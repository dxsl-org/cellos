# Raspberry Pi 3 VideoCore config — Cellos board-rpi3
#
# arm_64bit=1:    Load kernel8.img as AArch64 binary at 0x80000
# enable_uart=1:  Enable mini UART (UART1) on GPIO14/15 at 115200 baud
#                 PL011 is routed to Bluetooth; mini UART → serial console
#
# The mini UART baud divisor depends on the VPU core clock. Real Pi 3 firmware
# may select a different clock, so pin it to 250 MHz to keep 115200 baud stable.

arm_64bit=1
kernel=kernel8.img
device_tree=bcm2710-rpi-3-b.dtb
enable_uart=1
uart_2ndstage=1
core_freq=250
# Suppress HDMI EDID polling retries to eliminate 0.5s boot delay
hdmi_ignore_edid=0xa5000020
hdmi_force_hotplug=1
