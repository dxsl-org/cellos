2//! riscv64 CPU features: the H-extension, from the firmware device tree. The
3//! arch-neutral flags and accessors are `crate::cpu_features`.

5use core::sync::atomic::Ordering;

7/// The riscv64 part of `crate::cpu_features::detect`.
8pub(crate) fn detect(dtb: usize) {
9    detect_riscv(dtb);
10}
11
12/// The verdict of a board EL2 monitor: aarch64 only.
13pub(crate) fn el2_monitor_verified() -> Option<bool> {
14    None
15}

17fn detect_riscv(dtb: usize) {
18    if dtb == 0 {
19        return;
20    }
21    // SAFETY: dtb is the FDT pointer handed to the kernel by OpenSBI firmware.
22    // fdt::Fdt::from_ptr verifies the FDT magic number before reading any further.
23    let fdt = match unsafe { fdt::Fdt::from_ptr(dtb as *const u8) } {
24        Ok(f) => f,
25        Err(_) => return,
26    };
27    for cpu in fdt.cpus() {
28        // Prefer the newer property; fall back to the legacy ISA string.
29        //
30        // `riscv,isa-extensions` is a DT stringlist: NUL-separated tokens packed
31        // into one byte blob.  `as_str()` strips the trailing NUL; splitting on
32        // '\0' produces the individual extension names (e.g. "h", "smstateen").
33        let from_ext_list = cpu
34            .property("riscv,isa-extensions")
35            .and_then(|p| p.as_str())
36            .map(|s| s.split('\0').any(|ext| ext == "h"));
37
38        let from_isa_str = cpu
39            .property("riscv,isa")
40            .and_then(|p| p.as_str())
41            .map(isa_string_has_h);
42
43        if from_ext_list.or(from_isa_str).unwrap_or(false) {
44            crate::cpu_features::HAS_H_EXT.store(true, Ordering::Relaxed);
45            return;
46        }
47    }
48}

50/// Returns `true` if the `riscv,isa` string encodes the 'h' extension.
51///
52/// Scans past the `rv32`/`rv64` prefix, then iterates the single-letter extension
53/// zone.  Digits and `p` in version suffixes (e.g. `i2p1`) are skipped — NOT
54/// treated as terminators — so `"rv64i2p1mafdch"` correctly detects 'h'.
55/// Stops at `_` (start of the multi-char extension zone).
56fn isa_string_has_h(isa: &str) -> bool {
57    let after_prefix = if isa.len() >= 4 {
58        &isa[4..]
59    } else {
60        return false;
61    };
62    for c in after_prefix.chars() {
63        match c {
64            'h' => return true,
65            '_' => return false, // multi-char zone; no more single-letter exts
66            'a'..='z' | '0'..='9' => {} // extension letter or version component
67            _ => return false,
68        }
69    }
70    false
71}
