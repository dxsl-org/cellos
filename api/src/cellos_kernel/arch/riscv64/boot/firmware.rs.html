2//! riscv64 boot memory: the firmware DTB's memory map, else the selected
3//! board descriptor's fallback spans.
4
5use crate::boot::{
6    dtb_memory, fallback_memory_entry, MemoryMapEntry, SimpleBootInfo, DTB_BOOT_INFO,
7    DTB_MEMORY_MAP,
8};
9
10// RISC-V fallback RAM comes from the one selected board descriptor. Firmware
11// DTB/Limine data remains authoritative whenever it is available.
12const SELECTED_RISCV64_BOARD: &cellos_boards::BoardDescriptor = crate::board::compiled();
13
14static FALLBACK_MEMORY_MAP: [MemoryMapEntry; 3] = [
15    fallback_memory_entry(SELECTED_RISCV64_BOARD, 0),
16    fallback_memory_entry(SELECTED_RISCV64_BOARD, 1),
17    fallback_memory_entry(SELECTED_RISCV64_BOARD, 2),
18];
19
20pub static FALLBACK_BOOT_INFO: SimpleBootInfo = SimpleBootInfo {
21    memory_map: &FALLBACK_MEMORY_MAP,
22    kernel_phys_base: SELECTED_RISCV64_BOARD.fallback_memory[1].base,
23    hhdm_offset: 0x0,
24};
25
26/// Fallback boot info — static map on arches whose images stay small.
27///
28/// Only aarch64 QEMU-virt sizes its kernel region at runtime
29/// (`arch::aarch64::boot::firmware`);
30/// the other fallbacks keep their audited static spans.
31pub fn fallback_boot_info(dtb: usize) -> &'static SimpleBootInfo {
32    use core::ptr::{addr_of, addr_of_mut};
33    extern "C" {
34        static __kernel_end: u8;
35    }
36    #[cfg(not(feature = "board-vf2"))]
37    let _board = crate::board::active();
38
39    if dtb != 0 {
40        // SAFETY: firmware supplies a mapped FDT pointer; the parser validates its header.
41        if let Ok(tree) = unsafe { fdt::Fdt::from_ptr(dtb as *const u8) } {
42            let kernel_base = FALLBACK_BOOT_INFO.kernel_phys_base as usize;
43            let kernel_end = addr_of!(__kernel_end) as usize;
44            // SAFETY: early boot publishes these statics once before other tasks exist.
45            let result = unsafe {
46                dtb_memory::build(
47                    &tree,
48                    kernel_base,
49                    kernel_end,
50                    &mut *addr_of_mut!(DTB_MEMORY_MAP),
51                )
52            };
53            match result {
54                Ok(count) => unsafe {
55                    (*addr_of_mut!(DTB_BOOT_INFO)).memory_map = core::slice::from_raw_parts(
56                        addr_of!(DTB_MEMORY_MAP) as *const MemoryMapEntry,
57                        count,
58                    );
59                    (*addr_of_mut!(DTB_BOOT_INFO)).kernel_phys_base = kernel_base as u64;
60                    return &*addr_of!(DTB_BOOT_INFO);
61                },
62                Err(error) => log::warn!("[boot] DTB memory map rejected: {:?}", error),
63            }
64        } else {
65            log::warn!("[boot] invalid DTB memory map; using static fallback");
66        }
67    } else {
68        log::warn!("[boot] no DTB memory map; using static fallback");
69    }
70    &FALLBACK_BOOT_INFO
71}
