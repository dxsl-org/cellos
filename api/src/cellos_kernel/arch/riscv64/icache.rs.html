2//! `icache`: publishing loaded code to instruction fetch on every online hart.
3//!
4//! RV64 instruction fetch is not coherent with stores, and the hart that
5//! dequeues a freshly loaded task may not be the one that wrote its code: the
6//! loader fences locally, asks every other online hart through SBI RFENCE
7//! `remote_fence_i`, and falls back to an IPI epoch served with `fence.i`.
8
9use crate::memory::paging::Flags;
10use crate::sched::smp::{self, MAX_HARTS};
11use core::sync::atomic::{AtomicUsize, Ordering};
12use types::{VAddr, ViError, ViResult};
13
14/// Instruction-cache sync epochs: the IPI fallback a code publisher takes when
15/// SBI RFENCE `remote_fence_i` fails. Kept apart from the TLB epochs so a
16/// routine invalidation never pays for an instruction-cache flush. Served on
17/// the same trap path and in the same spin answers as the TLB request.
18static ICACHE_SYNC_REQUEST: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
19static ICACHE_SYNC_COMPLETE: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
20
21/// How long the IPI fallback waits for every asked hart's `fence.i` before
22/// refusing the spawn: the same 200 ms budget a TLB probe gets.
23const ICACHE_SYNC_PROBE_TICKS: u64 = 20 * hal::common::timer::TICKS_PER_10MS;
24
25/// Test-hooks: whether this boot already drove the IPI fallback once to emit
26/// the lane's evidence marker.
27#[cfg(feature = "test-hooks")]
28static ICACHE_SYNC_WITNESSED: core::sync::atomic::AtomicBool =
29    core::sync::atomic::AtomicBool::new(false);
30
31/// Make this hart's later instruction fetches observe every store it has
32/// observed (RV64 does not keep instruction fetch coherent with stores).
33#[inline]
34fn local_fence_i() {
35    // SAFETY: `fence.i` only orders this hart's later instruction fetches after
36    // the stores it has observed; it writes no register or memory.
37    unsafe { core::arch::asm!("fence.i", options(nostack, preserves_flags)) };
38}
39
40/// Ask `hart_id` to execute `fence.i` and return the epoch it must publish.
41///
42/// The Release half of the request orders the requester's code stores before
43/// the target's Acquire of the epoch, so the target's `fence.i` covers them.
44fn request_icache_sync(hart_id: usize) -> usize {
45    if hart_id >= MAX_HARTS {
46        return 0;
47    }
48    let epoch = ICACHE_SYNC_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
49    smp::send_ipi(hart_id);
50    epoch
51}
52
53/// Is `epoch` known complete on `hart_id`? Epoch 0 means "nothing requested".
54fn icache_sync_completed(hart_id: usize, epoch: usize) -> bool {
55    epoch == 0
56        || ICACHE_SYNC_COMPLETE
57            .get(hart_id)
58            .is_some_and(|complete| complete.load(Ordering::Acquire) >= epoch)
59}
60
61/// Serve one captured instruction-cache sync request on this hart. Called from
62/// the TLB service path, after its `hart_id < MAX_HARTS` check.
63pub(crate) fn service_sync_request(hart_id: usize) {
64    let epoch = ICACHE_SYNC_REQUEST[hart_id].load(Ordering::Acquire);
65    if ICACHE_SYNC_COMPLETE[hart_id].load(Ordering::Acquire) < epoch {
66        local_fence_i();
67        smp::publish_completion(&ICACHE_SYNC_COMPLETE[hart_id], epoch);
68    }
69}

71/// Publish the loader's final stores to `pages` to instruction fetch on every
72/// online hart, before the task can be registered: a no-op when none of them
73/// is executable.
74///
75/// # Errors
76/// `ViError::IO` when a hart asked through the fallback does not confirm.
77pub(crate) fn sync_loaded_code(pages: &[(VAddr, Flags)], cell: &str) -> ViResult<()> {
78    if pages
79        .iter()
80        .any(|(_, flags)| flags.bits() & Flags::EXECUTE != 0)
81    {
82        sync_instruction_caches(cell)?;
83    }
84    Ok(())
85}
86
87/// Make the loader's code stores fetchable on every online hart before the
88/// task can be registered, so no hart can dequeue it and fetch stale bytes a
89/// previous owner of the frame left in its instruction cache.
90///
91/// Local `fence.i`, then SBI RFENCE `remote_fence_i` to each other online hart.
92/// A hart whose RFENCE fails is asked through the IPI epoch path instead and
93/// must publish its own `fence.i` within the probe; a silent one refuses the
94/// spawn. The set is the harts online now: a later hart never fetched these
95/// bytes.
96///
97/// # Errors
98/// `ViError::IO` when a hart asked through the fallback does not confirm.
99fn sync_instruction_caches(cell: &str) -> ViResult<()> {
100    #[cfg(feature = "test-hooks")]
101    let mut remote = [false; MAX_HARTS];
102    let mut fallback = [false; MAX_HARTS];
103    let mut rfence_failed = false;
104    // Masked so the local fence and "every other hart" name the same hart.
105    let sstatus = hal::arch::save_and_disable_interrupts();
106    let me = crate::sched::hart_local::current_hart_id();
107    // Order the loader's stores before any remote hart is asked to fence.
108    core::sync::atomic::fence(Ordering::SeqCst);
109    local_fence_i();
110    for hart in smp::online_harts().filter(|hart| *hart != me) {
111        #[cfg(feature = "test-hooks")]
112        {
113            remote[hart] = true;
114        }
115        let sent = smp::logical_sbi_target(hart)
116            .map(|(mask, base)| hal::sbi::sbi_remote_fence_i(mask, base));
117        if !matches!(sent, Some(Ok(()))) {
118            fallback[hart] = true;
119            rfence_failed = true;
120        }
121    }
122    // SAFETY: restores exactly the interrupt state captured above.
123    unsafe { hal::arch::restore_sstatus(sstatus) };
124
125    if rfence_failed {
126        log::warn!(
127            "[wx] '{}': RFENCE remote_fence_i failed; using the IPI fallback",
128            cell
129        );
130    }
131    // Once per boot the fallback also runs after a successful RFENCE, so the
132    // lane witnesses both paths on every remote hart.
133    #[cfg(feature = "test-hooks")]
134    let harts = 1 + remote.iter().filter(|asked| **asked).count();
135    #[cfg(feature = "test-hooks")]
136    let witness = harts > 1 && !ICACHE_SYNC_WITNESSED.swap(true, Ordering::AcqRel);
137    #[cfg(feature = "test-hooks")]
138    if witness {
139        fallback = remote;
140    }
141
142    let asked = smp::RemoteEpochs::broadcast(
143        (0..MAX_HARTS).filter(|hart| fallback[*hart]),
144        request_icache_sync,
145    );
146    let deadline = hal::common::timer::read_mtime() + ICACHE_SYNC_PROBE_TICKS;
147    while let Some((hart, epoch)) = asked.first_outstanding(icache_sync_completed) {
148        if hal::common::timer::read_mtime() > deadline {
149            log::error!(
150                "[wx] '{}': hart {} did not confirm fence.i epoch {} — refusing spawn",
151                cell,
152                hart,
153                epoch
154            );
155            return Err(ViError::IO);
156        }
157        core::hint::spin_loop();
158        // Answer peers' requests so two harts waiting on each other cannot
159        // deadlock. The hart is re-read under the mask: with interrupts open
160        // between passes this context may have moved, and answering for a
161        // stale id would forge that hart's completion.
162        let sstatus = hal::arch::save_and_disable_interrupts();
163        smp::answer_tlb_flush_in_spin(crate::sched::hart_local::current_hart_id());
164        // SAFETY: restores exactly the interrupt state captured above.
165        unsafe { hal::arch::restore_sstatus(sstatus) };
166    }
167
168    #[cfg(feature = "test-hooks")]
169    if witness {
170        log::info!(
171            "[wx] icache sync: local+remote PASS harts={} rfence={} ipi-fallback=PASS",
172            harts,
173            if rfence_failed { "failed" } else { "ok" }
174        );
175    }
176    Ok(())
177}
