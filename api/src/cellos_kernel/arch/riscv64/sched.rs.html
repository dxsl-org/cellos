2//! `sched`: the scheduler's context-switch seam.
3//!
4//! The switch itself is the HAL's: `switch(SwitchPlan { out, inc, root, irq })`.
5//! `root: None` writes no `satp` (see `hal::arch::SwitchPlan`).
6
7pub use crate::hal::arch::{set_kernel_stack_for, switch, IrqSaved, Root, SwitchPlan};
8
9/// `yield_cpu` masks SIE on entry, before any scheduler lock or publication:
10/// cooperative yields can enter with SIE set, and masking only at `__switch`
11/// would leave the current/selected scheduler state interruptible.
12pub const MASK_AT_ENTRY: bool = true;
13
14/// RV64 reports its harts through the SMP and self-test lines instead.
15pub const REPORTS_DISPATCH: bool = false;
16
17/// The root for a top-level table at physical address `table` with tag `asid`
18/// (`satp` takes the table's page number).
19#[inline(always)]
20pub fn root_for_table(table: usize, asid: usize) -> Option<Root> {
21    Root::new(table >> 12, asid)
22}
