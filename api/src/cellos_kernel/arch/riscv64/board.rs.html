2//! `board`: the SoC policy of the active riscv64 board.
3
4/// Returns the SoC policy paired with the active RISC-V board descriptor.
5pub(crate) fn soc() -> &'static hal_soc_riscv::RiscvSocProfile {
6    use cellos_boards::SocId;
7
8    match crate::board::active().soc {
9        SocId::GenericRiscvVirt => &hal_soc_riscv::GENERIC_VIRT,
10        SocId::Jh7110 => &hal_soc_riscv::JH7110,
11        SocId::Sg2042 => &hal_soc_riscv::SG2042,
12        _ => panic!("[board] RISC-V descriptor has incompatible SoC identity"),
13    }
14}
