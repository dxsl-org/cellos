2//! `clock`: counters, uptime and wall-clock time.
3
4/// `GetTime(0)`: raw monotonic ticks (mtime).
5pub(crate) fn uptime_ticks() -> usize {
6    hal::common::timer::read_mtime() as usize
7}
8
9/// `GetTime(1)`: milliseconds since boot. 10 MHz mtime on QEMU RV64 → 10_000 ticks/ms.
10pub(crate) fn uptime_ms() -> usize {
11    (hal::common::timer::read_mtime() / 10_000) as usize
12}
13
14/// `GetTime(5)`: frequency of [`uptime_ticks`] in Hz.
15pub(crate) fn tick_hz() -> usize {
16    10_000_000usize
17}
18
19/// Nanoseconds since the Unix epoch (wall clock).
20pub(crate) fn epoch_ns() -> u64 {
21    hal::common::rtc::now_epoch_ns()
22}
23
24/// Monotonic nanoseconds on the same clock `GetTime(0)` exposes to cells.
25pub(crate) fn monotonic_ns() -> Option<u64> {
26    hal::common::timer::read_mtime().checked_mul(100)
27}
28
29/// Enable the S-mode timer interrupt and arm this hart's first 10 ms tick.
30pub(crate) fn start_tick() {
31    // SAFETY: sets STIE (bit 5 = mask 0x20) in sie from S-mode. Must use the
32    // register form of csrs — csrsi's immediate is only 5 bits (0..=31), so a
33    // 0x20 mask cannot be encoded as an immediate.
34    unsafe {
35        core::arch::asm!("csrs sie, {stie}", stie = in(reg) 0x20usize);
36    }
37    let next = hal::common::timer::read_mtime() + hal::common::timer::TICKS_PER_10MS;
38    hal::common::sbi::set_timer(next);
39}
