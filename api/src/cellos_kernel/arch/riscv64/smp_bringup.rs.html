7/// Test-hooks: how many preemption requests were pended for each logical hart.
8///
9/// The RT-wake fixture needs the *decision* observable, not the interrupt: it
10/// asserts that consuming a message pends exactly one for the sender's target
11/// hart, that a stale delivery token pends none, and that the decision follows the
12/// target hart's running priority rather than the waking hart's.
13#[cfg(feature = "test-hooks")]
14pub(super) static PREEMPT_PENDS: [AtomicUsize; MAX_HARTS] =
15    [const { AtomicUsize::new(0) }; MAX_HARTS];
16
17#[cfg(feature = "test-hooks")]
18#[inline]
19pub(crate) fn note_preempt_pend(hart_id: usize) {
20    if hart_id < MAX_HARTS {
21        PREEMPT_PENDS[hart_id].fetch_add(1, Ordering::AcqRel);
22    }
23}
24
25#[cfg(feature = "test-hooks")]
26#[inline]
27pub(crate) fn preempt_pends_for(hart_id: usize) -> usize {
28    PREEMPT_PENDS
29        .get(hart_id)
30        .map(|count| count.load(Ordering::Acquire))
31        .unwrap_or(0)
32}
33
34#[cfg(feature = "test-hooks")]
35pub(crate) fn reset_preempt_pends() {
36    for count in PREEMPT_PENDS.iter() {
37        count.store(0, Ordering::Release);
38    }
39}

41/// Publish the RV64 boot hart's numbering into [`CPU_TOPOLOGY`]: the boot hart
42/// is logical 0 and the hart it displaced takes its number
43/// ([`topology::boot_swapped_physical`]). Called once by `kmain` on the boot
44/// hart, before any secondary is started.
45pub fn set_boot_physical_hart(physical_hart: usize) {
46    assert!(
47        physical_hart < MAX_HARTS,
48        "unsupported RV64 boot hart {physical_hart}"
49    );
50    for logical in 0..MAX_HARTS {
51        assert!(
52            CPU_TOPOLOGY.publish(
53                logical,
54                topology::boot_swapped_physical(physical_hart, logical)
55            ),
56            "RV64 boot hart {physical_hart} conflicts with the published topology"
57        );
58    }
59}
60
61pub fn boot_physical_hart() -> Option<usize> {
62    logical_to_physical(0)
63}

65pub fn logical_sbi_target(logical_hart: usize) -> Option<(usize, usize)> {
66    logical_to_physical(logical_hart).map(|physical| (1, physical))
67}

69/// Return the one remote RV64 hart this boot can have, if it is online.
70///
71/// Every hart, the boot hart included, publishes readiness in `HART_ONLINE`
72/// with Release. RV64 runs at most two harts (`MAX_HARTS = 2`), so the remote
73/// hart is the RT hart for hart 0 and hart 0 for everyone else.
74pub fn remote_online_sbi_target() -> Option<(usize, usize)> {
75    const _: () = assert!(MAX_HARTS == 2);
76    let current = crate::sched::hart_local::current_hart_id();
77    let remote = if current == 0 { HART_RT } else { 0 };
78    let online = HART_ONLINE[remote].load(Ordering::Acquire);
79    online.then(|| logical_sbi_target(remote)).flatten()
80}

82/// How many 10 ms ticks hart 0 waits for each secondary to come online before
83/// logging a warning and continuing single-hart.  500 ms is generous for QEMU
84/// and for an SBI `hart_start` that has to power a core up. (AArch64 waits
85/// on the architected counter instead, which runs with IRQs masked.)
86pub(super) const SECONDARY_BOOT_TIMEOUT_TICKS: usize = 50;
87
88/// Called by hart 0 **after** `task::init()` to bring secondary harts online.
89///
90/// Each secondary is started via SBI HSM `hart_start`.  Hart 0 then spins
91/// (bounded) waiting for each secondary to set `HART_ONLINE[hart_id]`.
92/// If a secondary fails to start or times out, a warning is logged and the
93/// system continues single-hart — graceful degradation, never a panic.
94pub fn start_secondaries() {
95    use crate::task::stack::Stack;
96    use crate::task::STACK_PAGES;
97    use hal::common::sbi::{sbi_hart_get_status, sbi_hart_start, sbi_rfence_available};
98
99    let Some(boot_physical) = boot_physical_hart() else {
100        log::warn!("[smp] boot physical hart was not published");
101        return;
102    };
103    log::info!("[smp] physical {} -> logical 0 boot", boot_physical);
104
105    match sbi_rfence_available() {
106        Ok(true) => {}
107        Ok(false) => {
108            log::warn!("[smp] SBI RFENCE unavailable — keeping Cellos single-hart");
109            return;
110        }
111        Err(error) => {
112            log::warn!(
113                "[smp] SBI RFENCE probe failed (err={}) — keeping Cellos single-hart",
114                error
115            );
116            return;
117        }
118    }
119
120    extern "C" {
121        // Physical asm label defined in hal/arch/riscv/src/rv64/boot.rs.
122        // Runs bare (SATP=0); no relocation or BSS clear.
123        fn _secondary_entry();
124    }
125
126    for (hart_id, online) in HART_ONLINE.iter().enumerate().skip(1) {
127        let Some(physical_hart) = logical_to_physical(hart_id) else {
128            log::warn!("[smp] logical hart {} has no physical mapping", hart_id);
129            continue;
130        };
131        // Allocate a dedicated kernel stack for this hart.  Leak it — it lives
132        // for the entire lifetime of the hart.
133        let stack = match Stack::new_kernel(STACK_PAGES) {
134            Ok(s) => s,
135            Err(e) => {
136                log::warn!("[smp] hart {} stack alloc failed: {:?}", hart_id, e);
137                continue;
138            }
139        };
140        let stack_top = stack.top;
141        core::mem::forget(stack);
142
143        let Ok(state) = sbi_hart_get_status(physical_hart) else {
144            log::warn!(
145                "[smp] physical hart {} HSM status unavailable",
146                physical_hart
147            );
148            continue;
149        };
150        log::info!(
151            "[smp] physical {} -> logical {} HSM state = {}",
152            physical_hart,
153            hart_id,
154            state
155        );
156        if state != 1 {
157            log::warn!("[smp] physical hart {} is not HSM STOPPED", physical_hart);
158            continue;
159        }
160
161        // SAFETY: _secondary_entry is a physical-address asm label; the kernel
162        // is loaded at 0x80200000 with slide=0 so physical == virtual.
163        // stack_top is the usable top of a freshly-allocated kernel stack.
164        // SAFETY: casting function pointer to integer — use double-cast through
165        // *const () to avoid the "direct cast of function item" lint.
166        let entry_paddr = _secondary_entry as *const () as usize;
167        match sbi_hart_start(physical_hart, entry_paddr, stack_top) {
168            Ok(()) => log::info!(
169                "[smp] hart {} start requested (entry={:#x})",
170                physical_hart,
171                entry_paddr
172            ),
173            Err(e) => {
174                log::warn!("[smp] hart {} SBI hart_start failed: err={}", hart_id, e);
175                continue;
176            }
177        }
178
179        // Bounded spin: wait for the secondary to signal it is online.
180        let deadline = crate::task::system_ticks() + SECONDARY_BOOT_TIMEOUT_TICKS;
181        loop {
182            if online.load(Ordering::Acquire) {
183                log::info!("[smp] hart {} online, parked", hart_id);
184                break;
185            }
186            if crate::task::system_ticks() >= deadline {
187                log::warn!(
188                    "[smp] hart {} did not come online in time — continuing single-hart",
189                    hart_id
190                );
191                break;
192            }
193            core::hint::spin_loop();
194        }
195    }
196}

198/// `smp_hart_entry`'s riscv64 bring-up, up to interrupts enabled: map the
199/// physical hart to its logical id, enter the shared kernel root, install the
200/// hart-local state and trap vector, arm the tick, enable interrupts. Returns
201/// the logical hart.
202pub(super) fn secondary_start(physical_hart: usize) -> usize {
203    let hart_id = physical_to_logical(physical_hart)
204        .unwrap_or_else(|| panic!("unmapped RV64 physical hart {}", physical_hart));
205
206    {
207        let root = crate::memory::paging::KERNEL_ROOT
208            .lock()
209            .expect("RV64 secondary started before kernel paging root");
210        // SAFETY: the boot hart published a complete shared root before HSM
211        // startup; it maps this entry code, stack, and all kernel globals.
212        unsafe {
213            crate::memory::paging::activate_paging(root);
214        }
215        crate::arch::user_access::secondary_open();
216    }
217    // Install the trap vector (each hart has its own stvec CSR).
218    // `hal::ARCH.init()` sets stvec + enables SSIE.
219    {
220        crate::sched::hart_local::install(hart_id);
221        use hal::Arch;
222        hal::ARCH.init();
223        // ARCH.init installs the bootstrap-safe default vector. Restore this
224        // secondary's logical vector before any interrupt is enabled.
225        hal::trap::init_for_hart(hart_id);
226    }
227
228    // Enable S-mode timer interrupt and arm the first tick on this hart.
229    // Each hart has its own mtimecmp register via SBI; arming here starts
230    // the 10ms preemption slice for this hart.
231    {
232        crate::arch::clock::start_tick();
233    }
234
235    // `ARCH.init()` enables SSIE in `sie`, but it deliberately leaves the
236    // per-hart global SIE bit clear. A secondary otherwise wakes from WFI with
237    // a pending IPI but never takes the SSIP trap that enters `yield_cpu()`;
238    // explicitly enable delivery before advertising this hart as schedulable.
239    {
240        use hal::Arch;
241        hal::ARCH.enable_interrupts();
242        if !hal::ARCH.interrupts_enabled() {
243            panic!(
244                "[smp] hart {} could not enable supervisor interrupts",
245                hart_id
246            );
247        }
248    }
249    hart_id
