2//! `power`: system shutdown.
3
4/// Power the machine off (`Syscall::Shutdown`). Test images leave QEMU with
5/// success instead.
6pub(crate) fn shutdown() -> ! {
7    #[cfg(feature = "test-hooks")]
8    crate::qemu_exit(true);
9
10    #[cfg(not(feature = "test-hooks"))]
11    unsafe {
12        // SAFETY: ecall traps to OpenSBI which powers off QEMU; no return.
13        core::arch::asm!(
14            "li a7, 0x53525354", // SBI_EXT_SRST
15            "li a6, 0",          // fid = SYSTEM_RESET
16            "li a0, 0",          // reset_type = Shutdown
17            "li a1, 0",          // reset_reason = NoReason
18            "ecall",
19            options(noreturn)
20        );
21    }
22}
23
24/// Leave QEMU (test images): the SiFive test device at 0x100000.
25#[cfg(feature = "test-hooks")]
26pub(crate) fn qemu_exit(success: bool) -> ! {
27    use qemu_exit::QEMUExit;
28    // SAFETY: 0x100000 is the SiFive test device address on the QEMU `virt`
29    // machine used by this test-only exit path.
30    unsafe {
31        qemu_exit::RISCV64::new(0x100000).exit(if success { 0 } else { 1 });
32    }
33}

35/// After a kernel panic is printed: reboot through SBI SRST.
36#[cfg(not(all(test, not(target_os = "none"))))]
37pub(crate) fn reset_after_panic() {
38    crate::hal::sbi::system_reset(crate::hal::sbi::SBI_RESET_COLD_REBOOT, 0);
