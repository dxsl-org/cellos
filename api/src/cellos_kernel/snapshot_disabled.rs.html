2//! The snapshot API on kernels built without `snapshot-qualified`
3//! (architecture remediation Phase 09): the 5.7k-line engine in
4//! `snapshot_engine.rs` is not compiled at all, and this module is the only
5//! closed-gate behaviour.
6//!
7//! - no scratch region is reserved at boot;
8//! - a capture is refused with [`SnapshotError::GateClosed`];
9//! - a boot never restores, and clears a header an older image may have left
10//!   behind, so that image cannot replay it after a downgrade.
11
12use crate::snapshot_format::{SECTOR_SIZE, SNAPSHOT_BASE_LBA, SNAPSHOT_MAGIC};
13
14/// Why a capture was refused. Only the gate exists on this kernel.
15#[derive(Clone, Copy, PartialEq, Eq, Debug)]
16pub enum SnapshotError {
17    /// `snapshot-qualified` is off — the phase-01 gate.
18    GateClosed,
19}
20
21impl core::fmt::Display for SnapshotError {
22    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
23        f.write_str("capture unqualified (phase-01 gate: feature `snapshot-qualified`)")
24    }
25}
26
27/// No capture scratch region exists on this kernel.
28pub fn reserve_scratch_region_at_boot() {}
29
30/// Capture is not built into this kernel.
31pub fn serialize_snapshot() -> Result<u32, SnapshotError> {
32    Err(SnapshotError::GateClosed)
33}
34
35/// Cold boot only: clears a stale snapshot header and never restores.
36pub fn try_restore() -> bool {
37    if clear_stale_header(&mut KernelBlock) {
38        log::info!("[snapshot] snapshot invalidated");
39        log::warn!(
40            "[snapshot] stale header invalidated: capture/restore disabled by the phase-01 \
41             qualification gate (feature `snapshot-qualified`)"
42        );
43    }
44    false
45}
46
47/// The sector operations `clear_stale_header` needs.
48trait Sectors {
49    fn read(&mut self, lba: u64, buf: &mut [u8]) -> bool;
50    fn write(&mut self, lba: u64, buf: &[u8]) -> bool;
51    fn flush(&mut self) -> bool;
52}
53
54/// The kernel's block route.
55struct KernelBlock;
56
57impl Sectors for KernelBlock {
58    fn read(&mut self, lba: u64, buf: &mut [u8]) -> bool {
59        crate::dev::block::read_sector(lba, buf).is_ok()
60    }
61    fn write(&mut self, lba: u64, buf: &[u8]) -> bool {
62        crate::dev::block::write_sector(lba, buf).is_ok()
63    }
64    fn flush(&mut self) -> bool {
65        crate::dev::block::flush().is_ok()
66    }
67}
68
69/// Reads the header sector and zeroes it (then flushes) only when it carries
70/// the snapshot magic, so a board without a snapshot region sees no stray
71/// write. Returns whether a header was found and cleared; write and flush
72/// failures are non-fatal because nothing on this kernel reads the region.
73fn clear_stale_header(sectors: &mut impl Sectors) -> bool {
74    let mut sector = [0u8; SECTOR_SIZE];
75    if !sectors.read(SNAPSHOT_BASE_LBA, &mut sector) {
76        return false;
77    }
78    let magic = u32::from_le_bytes([sector[0], sector[1], sector[2], sector[3]]);
79    if magic != SNAPSHOT_MAGIC {
80        return false;
81    }
82    let _ = sectors.write(SNAPSHOT_BASE_LBA, &[0u8; SECTOR_SIZE]);
83    let _ = sectors.flush();
84    true
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90    use alloc::vec::Vec;
91
92    #[derive(Default)]
93    struct Fake {
94        header: Option<[u8; SECTOR_SIZE]>,
95        writes: Vec<(u64, [u8; SECTOR_SIZE])>,
96        flushes: usize,
97    }
98
99    impl Sectors for Fake {
100        fn read(&mut self, lba: u64, buf: &mut [u8]) -> bool {
101            assert_eq!(lba, SNAPSHOT_BASE_LBA);
102            match self.header {
103                Some(header) => {
104                    buf.copy_from_slice(&header);
105                    true
106                }
107                None => false,
108            }
109        }
110        fn write(&mut self, lba: u64, buf: &[u8]) -> bool {
111            self.writes.push((lba, buf.try_into().unwrap()));
112            true
113        }
114        fn flush(&mut self) -> bool {
115            self.flushes += 1;
116            true
117        }
118    }
119
120    fn header_with_magic(magic: u32) -> [u8; SECTOR_SIZE] {
121        let mut sector = [0xA5u8; SECTOR_SIZE];
122        sector[..4].copy_from_slice(&magic.to_le_bytes());
123        sector
124    }
125
126    #[test]
127    fn gate_closed_capture_is_refused() {
128        assert_eq!(serialize_snapshot(), Err(SnapshotError::GateClosed));
129    }
130
131    #[test]
132    fn unreadable_region_is_left_alone() {
133        let mut fake = Fake::default();
134        assert!(!clear_stale_header(&mut fake));
135        assert!(fake.writes.is_empty());
136        assert_eq!(fake.flushes, 0);
137    }
138
139    #[test]
140    fn foreign_sector_is_not_written() {
141        let mut fake = Fake {
142            header: Some(header_with_magic(SNAPSHOT_MAGIC ^ 1)),
143            ..Fake::default()
144        };
145        assert!(!clear_stale_header(&mut fake));
146        assert!(fake.writes.is_empty());
147        assert_eq!(fake.flushes, 0);
148    }
149
150    #[test]
151    fn stale_header_is_zeroed_and_flushed_once() {
152        let mut fake = Fake {
153            header: Some(header_with_magic(SNAPSHOT_MAGIC)),
154            ..Fake::default()
155        };
156        assert!(clear_stale_header(&mut fake));
157        assert_eq!(fake.writes.len(), 1);
158        assert_eq!(fake.writes[0].0, SNAPSHOT_BASE_LBA);
159        assert!(fake.writes[0].1.iter().all(|byte| *byte == 0));
160        assert_eq!(fake.flushes, 1);
161    }
