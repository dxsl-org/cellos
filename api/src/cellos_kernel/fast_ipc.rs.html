1//! Kernel-owned fast-IPC dispatch table — the single canonical instance.
2//!
3//! In a Single Address Space there is no privilege wall between Cells: a trusted
4//! A future direct path would reduce a service call to an indirect branch, but
5//! that path has no runtime measurement or loader bridge today. To work, ONE handler
6//! pointer must be shared by the VFS cell (which registers it), client cells
7//! (which call it), and the kernel (which nulls it if VFS faults).
8//!
9//! Because Cells are separately-loaded ELFs (each with its own copy of any
10//! `static`), the shared instance cannot live in a per-cell library — it lives
11//! HERE, in the kernel. No loader import-resolution bridge exists today.
12//! Separately linked Cells use their private `ostd` table and therefore take
13//! the message fallback. This table is retained only as design scaffolding for
14//! the ruled Tier-1 rewrite; the kernel uses
15//! `set_vfs_handler_cell`/`clear_vfs_if_cell` directly.
16//!
17//! ## Safety invariant
18//! The handler pointer is published once at VFS startup (before any client call)
19//! with `Release` ordering, read with `Acquire`, and only ever nulled on VFS
20//! fault. Single-hart QEMU: no concurrent modification.
21
22use api::caller_identity::CallerIdentity;
23use api::fast_ipc::{TrustedHandle, VfsCell};
24use api::ipc::{VfsRequest, IPC_BUF_SIZE};
25use core::sync::atomic::{AtomicPtr, AtomicUsize, Ordering};
26
27/// Signature of a registered VFS fast-IPC handler: read `req` on behalf of
28/// `caller`, write the response into `out`, return the number of bytes written.
29///
30/// `caller` is `None` when the kernel could not attribute the call to a live
31/// cell. The handler must then refuse the request: this path serves `GetFile`,
32/// which hands back a raw `DataPtr` — permanent, unrevocable read authority in a
33/// single address space — so an unauthorized answer here cannot be taken back.
34pub type VfsFastHandler = unsafe fn(
35    caller: Option<CallerIdentity>,
36    req: &VfsRequest<'_>,
37    out: &mut [u8; IPC_BUF_SIZE],
38) -> usize;
39
40static VFS_HANDLER_PTR: AtomicPtr<()> = AtomicPtr::new(core::ptr::null_mut());
41/// Raw CellId that registered the handler; 0 = unregistered. Lets the kernel
42/// null the pointer when (and only when) that specific cell faults.
43static VFS_HANDLER_CELL: AtomicUsize = AtomicUsize::new(0);
44
45/// Register the kernel-side VFS fast-IPC handler for the future Tier-1 bridge.
46/// No Cell import resolves here today.
47///
48/// The stable symbol name is retained for the reviewed bridge design.
49#[no_mangle]
50pub extern "Rust" fn register_vfs(handler: VfsFastHandler) {
51    // SAFETY: fn-ptr → *mut () for atomic storage; recovered with the same type
52    // in `call_vfs`. Published Release so the handler body is visible to Acquire readers.
53    VFS_HANDLER_PTR.store(handler as *mut (), Ordering::Release);
54}
55
56/// Record which cell owns the registered handler (kernel-internal; called from
57/// the VFS spawn path so a later fault of that cell can null the pointer).
58pub fn set_vfs_handler_cell(cell_id_raw: usize) {
59    VFS_HANDLER_CELL.store(cell_id_raw, Ordering::Relaxed);
60}
61
62/// Null the handler pointer iff `cell_id_raw` is the registered owner. Called by
63/// the kernel fault path so a future `call_vfs` does not jump into dead VFS code.
64pub fn clear_vfs_if_cell(cell_id_raw: usize) {
65    if VFS_HANDLER_CELL.load(Ordering::Relaxed) == cell_id_raw && cell_id_raw != 0 {
66        VFS_HANDLER_PTR.store(core::ptr::null_mut(), Ordering::Release);
67        VFS_HANDLER_CELL.store(0, Ordering::Relaxed);
68    }
69}
70
71pub(crate) fn is_registered_vfs_cell(cell_id_raw: usize) -> bool {
72    cell_id_raw != 0 && VFS_HANDLER_CELL.load(Ordering::Relaxed) == cell_id_raw
73}

75#[cfg(feature = "test-hooks")]
76pub(crate) fn vfs_handler_cell_snapshot() -> usize {
77    VFS_HANDLER_CELL.load(Ordering::Acquire)
78}

80/// Snapshot the registered handler pointer separately from its Cell owner.
81/// Fixture code may exercise owner routing but must never clear another
82/// component's registered handler.
83#[cfg(feature = "test-hooks")]
84pub(crate) fn vfs_handler_pointer_snapshot() -> usize {
85    VFS_HANDLER_PTR.load(Ordering::Acquire) as usize
86}

88/// Restore a pointer obtained from `vfs_handler_pointer_snapshot` after a
89/// test-only fixture temporarily registers its own handler.
90#[cfg(feature = "test-hooks")]
91pub(crate) fn restore_vfs_handler_pointer_for_test(handler: usize) {
92    VFS_HANDLER_PTR.store(handler as *mut (), Ordering::Release);
93}

95/// RAII guard that restores the S-mode interrupt-enable bit (SIE) on drop.
96///
97/// Constructed by disabling SIE and recording its prior state.  `Drop` restores
98/// it, so SIE is always restored even if the handler panics (Rust drop glue runs
99/// before the panic handler, giving the guard a chance to clean up).
100struct SieGuard(
101    /// `true` if SIE was set before we disabled it; `false` = was already clear.
102    bool,
103);
104
105impl SieGuard {
106    /// Disable SIE and return a guard that will restore it.
107    ///
108    /// # Safety
109    /// Must be called from S-mode.
110    #[inline]
111    unsafe fn disable() -> Self {
112        #[cfg(target_arch = "riscv64")]
113        {
114            let v: usize;
115            // SAFETY: csrrci reads-and-clears sstatus.SIE (bit 1) atomically.
116            core::arch::asm!("csrrci {}, sstatus, 0x2", out(reg) v);
117            Self(v & 0x2 != 0)
118        }
119        #[cfg(not(target_arch = "riscv64"))]
120        Self(false)
121    }
122}
123
124impl Drop for SieGuard {
125    fn drop(&mut self) {
126        if self.0 {
127            // SAFETY: restoring SIE to the value saved in disable(); S-mode only.
128            #[cfg(target_arch = "riscv64")]
129            unsafe {
130                core::arch::asm!("csrsi sstatus, 0x2");
131            }
132        }
133    }
134}

136/// Call the registered VFS handler directly, bypassing the `ecall` trap. Returns
137/// bytes written into `out`, or 0 if no handler is registered (caller falls back
138/// to the `sys_send`/`sys_recv` path).
139///
140/// The stable symbol name is retained for the reviewed bridge design; no client
141/// Cell import resolves here today.
142///
143/// # Note (PIE limitation)
144/// For non-PIE cells (current default), each cell ELF links `libs/ostd` statically
145/// and gets its own copy of `VFS_HANDLER_PTR` — so `call_vfs` in the shell reads
146/// null and always takes the ecall fallback.  The fast path becomes effective once
147/// cells are compiled as PIE and a reviewed loader import bridge resolves this
148/// kernel function. The fallback is always safe; that bridge is absent today.
149///
150/// # Caller identity
151/// This path does not go through `ecall`, so the request carries no sender tid
152/// and the handler has nothing of its own to authorize against. The identity is
153/// therefore derived HERE, from live scheduler state for the task currently
154/// running on this hart — never from an argument, because every argument on this
155/// path is chosen by the cell being authorized. That makes the fast path exactly
156/// as attested as the message path; without it, gating `GetFile` on the message
157/// path would only move the hole rather than close it.
158///
159/// `TrustedHandle` is not the control: its own contract says it is advisory.
160///
161/// # Safety
162/// The caller must own `out` exclusively for the call. `_handle` documents that
163/// the caller was granted fast-path access; it is not enforced at runtime.
164#[no_mangle]
165pub unsafe extern "Rust" fn call_vfs(
166    _handle: TrustedHandle<VfsCell>,
167    req: &VfsRequest<'_>,
168    out: &mut [u8; IPC_BUF_SIZE],
169) -> usize {
170    let ptr = VFS_HANDLER_PTR.load(Ordering::Acquire);
171    if ptr.is_null() {
172        return 0; // VFS not yet registered — caller falls back to ecall path.
173    }
174    // Resolve identity BEFORE disabling interrupts: it takes the scheduler lock,
175    // and holding that lock across the handler would deadlock the VFS backends.
176    let caller = crate::task::syscall::attested_identity_of(crate::task::current_task_id());
177
178    // SAFETY: ptr was stored by register_vfs from a valid VfsFastHandler.
179    let handler: VfsFastHandler = core::mem::transmute(ptr);
180
181    // Disable S-mode interrupts for the handler's duration. The VFS FAT16 driver
182    // holds a spinlock; timer preemption mid-handler to another VFS caller would
183    // deadlock on it. SieGuard restores SIE on drop — safe even on handler panic.
184    // SAFETY: called from S-mode trap handler context.
185    let _sie = SieGuard::disable();
186
187    handler(caller, req, out)
188}
