1//! CPU feature detection — parsed from the firmware-provided device tree.
2//!
3//! Call `detect(dtb)` once at kernel boot before any Cell is spawned.
4//! All other callers use the read-only `has_*()` accessors.
5
6#[cfg(target_arch = "x86_64")]
7use core::sync::atomic::AtomicU8;
8use core::sync::atomic::{AtomicBool, Ordering};
9
10static HAS_H_EXT: AtomicBool = AtomicBool::new(false);
11
12/// QEMU virt EL2-host is latched by detect(); Pi's EL1-host monitor becomes
13/// capable only after its HVC init AND boot-time guest smoke succeed.
14static HAS_EL2: AtomicBool = AtomicBool::new(false);
15
16/// x86 hardware-virtualization vendor detected via CPUID (Tier 3 x86 VMM).
17#[cfg(target_arch = "x86_64")]
18#[derive(Copy, Clone, Debug, PartialEq, Eq)]
19pub(crate) enum X86Virt {
20    /// AMD SVM (`CPUID.8000_0001:ECX[2]`) — first backend (QEMU TCG emulates it).
21    Svm,
22    /// Intel VT-x (`CPUID.1:ECX[5]`) — second backend (KVM/real-HW lane only).
23    Vmx,
24}
25
26/// 0 = none, 1 = SVM, 2 = VMX — written once by `detect()`.
27#[cfg(target_arch = "x86_64")]
28static X86_VIRT_KIND: AtomicU8 = AtomicU8::new(0);
29
30/// Latched by kmain AFTER root operation is actually entered (EFER.SVME set /
31/// VMXON succeeded).  The HypervisorCap gate keys off THIS, not raw CPUID —
32/// firmware may advertise the feature but lock it off.
33static HAS_X86_VIRT: AtomicBool = AtomicBool::new(false);
34
35/// Probe the device tree for CPU feature flags.
36///
37/// Must be called once at kernel boot before any Cell is spawned.
38/// No-op (and safe) on non-riscv64 targets.
39pub(crate) fn detect(dtb: usize) {
40    #[cfg(target_arch = "riscv64")]
41    detect_riscv(dtb);
42    #[cfg(not(target_arch = "riscv64"))]
43    let _ = dtb;
44    // Latch EL2 boot status (ARM64 only); no-op on other arches.
45    #[cfg(target_arch = "aarch64")]
46    if hal::aarch64::el2::is_el2() {
47        HAS_EL2.store(true, Ordering::Relaxed);
48    }
49    // Latch the x86 hardware-virt vendor (SVM preferred — the TCG-testable
50    // backend; VMX only when SVM is absent, i.e. genuine Intel).
51    #[cfg(target_arch = "x86_64")]
52    {
53        if hal::svm::supported() {
54            X86_VIRT_KIND.store(1, Ordering::Relaxed);
55        } else if hal::vmx::supported() {
56            X86_VIRT_KIND.store(2, Ordering::Relaxed);
57        }
58    }
59}

61/// Is this kernel running under a hypervisor? (`CPUID.1:ECX[31]`)
62///
63/// Nested SVM/VMX is not guaranteed even when the vendor bits are advertised:
64/// `VMXON` under a hypervisor that does not grant nested operation **faults**
65/// (`#GP`) rather than reporting failure, and root operation is an optional
66/// capability. The boot path consults this before attempting it.
67#[cfg(target_arch = "x86_64")]
68pub(crate) fn hypervisor_present() -> bool {
69    core::arch::x86_64::__cpuid(1).ecx & (1 << 31) != 0
70}

72/// The x86 virt vendor CPUID advertised, if any.  `None` before `detect()`.
73#[cfg(target_arch = "x86_64")]
74pub(crate) fn x86_virt_kind() -> Option<X86Virt> {
75    match X86_VIRT_KIND.load(Ordering::Relaxed) {
76        1 => Some(X86Virt::Svm),
77        2 => Some(X86Virt::Vmx),
78        _ => None,
79    }
80}

82/// Enter the CPUID-selected backend's x86 root operation on the CURRENT CPU.
83///
84/// Root operation is per-CPU state (`EFER.SVME` + `VM_HSAVE_PA`, or
85/// `CR4.VMXE` + `VMXON`), so every CPU gets its own freshly allocated HSAVE /
86/// VMXON frame. The frame is never freed: root operation lasts until reset, and
87/// it is never mapped into any guest NPT (P02 invariant). The BSP calls this
88/// before [`latch_x86_root_active`]; while [`has_x86_virt`] is set, each AP
89/// calls it before going online, and an AP that is refused stays offline.
90///
91/// `Err` names the refusal for the caller's log line.
92#[cfg(target_arch = "x86_64")]
93pub(crate) fn enter_x86_root_on_current_cpu() -> Result<X86Virt, &'static str> {
94    let kind = x86_virt_kind().ok_or("not supported by CPU")?;
95    let frame = {
96        let mut guard = crate::memory::frame::FRAME_ALLOCATOR.lock();
97        guard
98            .as_mut()
99            .expect("Frame allocator not initialized")
100            .allocate_frame()
101    };
102    match kind {
103        X86Virt::Svm => {
104            let pa = frame.ok_or("OOM allocating HSAVE frame")?;
105            // SAFETY: the HSAVE frame is freshly allocated, exclusively owned
106            // by this CPU and never freed (see above).
107            unsafe { hal::svm::enable(pa as u64) }
108                .map_err(|_| "SVM present but firmware-locked")?;
109        }
110        X86Virt::Vmx => {
111            let pa = frame.ok_or("OOM allocating VMXON frame")?;
112            let va = crate::memory::frame::phys_to_virt(pa) as *mut u32;
113            // SAFETY: pa/va name the same freshly allocated, exclusively owned
114            // frame (HHDM mapping), never freed (see above).
115            unsafe { hal::vmx::enter_root(pa as u64, va) }
116                .map_err(|_| "VMX present but unavailable")?;
117        }
118    }
119    Ok(kind)
120}

122/// Record that the BSP successfully entered x86 root operation.
123///
124/// Called by kmain after [`enter_x86_root_on_current_cpu`] returned `Ok` —
125/// the HypervisorCap gate reads [`has_x86_virt`]. APs that cannot enter the
126/// same root operation never go online, so the flag holds on every online CPU.
127#[cfg(target_arch = "x86_64")]
128pub(crate) fn latch_x86_root_active() {
129    HAS_X86_VIRT.store(true, Ordering::Relaxed);
130}

132/// Returns `true` if this kernel entered x86 root operation (SVM/VMX) at boot.
133///
134/// Always `false` on non-x86_64 targets and on CPUs where the feature is
135/// absent or firmware-locked.
136pub(crate) fn has_x86_virt() -> bool {
137    HAS_X86_VIRT.load(Ordering::Relaxed)
138}

140/// True only after entering actual EL2 host mode or validating the Pi monitor
141/// with an HVC/WFI/MMIO/virtual IRQ guest. Firmware that starts Pi at EL1 fails
142/// closed even if the CPU advertises EL2 in its ID registers.
143pub(crate) fn has_el2() -> bool {
144    #[cfg(target_arch = "aarch64")]
145    {
146        use hal::aarch64::backend::{ArmEl2Monitor, El2Monitor};
147        if El2Monitor::PRESENT {
148            return El2Monitor::is_verified();
149        }
150    }
151    HAS_EL2.load(Ordering::Relaxed)
152}

154/// Returns `true` if the RISC-V H-extension (hypervisor) is present.
155///
156/// Always `false` on non-riscv64 targets.
157pub(crate) fn has_h_ext() -> bool {
158    HAS_H_EXT.load(Ordering::Relaxed)

161#[cfg(target_arch = "riscv64")]
162fn detect_riscv(dtb: usize) {
163    if dtb == 0 {
164        return;
165    }
166    // SAFETY: dtb is the FDT pointer handed to the kernel by OpenSBI firmware.
167    // fdt::Fdt::from_ptr verifies the FDT magic number before reading any further.
168    let fdt = match unsafe { fdt::Fdt::from_ptr(dtb as *const u8) } {
169        Ok(f) => f,
170        Err(_) => return,
171    };
172    for cpu in fdt.cpus() {
173        // Prefer the newer property; fall back to the legacy ISA string.
174        //
175        // `riscv,isa-extensions` is a DT stringlist: NUL-separated tokens packed
176        // into one byte blob.  `as_str()` strips the trailing NUL; splitting on
177        // '\0' produces the individual extension names (e.g. "h", "smstateen").
178        let from_ext_list = cpu
179            .property("riscv,isa-extensions")
180            .and_then(|p| p.as_str())
181            .map(|s| s.split('\0').any(|ext| ext == "h"));
182
183        let from_isa_str = cpu
184            .property("riscv,isa")
185            .and_then(|p| p.as_str())
186            .map(isa_string_has_h);
187
188        if from_ext_list.or(from_isa_str).unwrap_or(false) {
189            HAS_H_EXT.store(true, Ordering::Relaxed);
190            return;
191        }
192    }
193}

195/// Returns `true` if the `riscv,isa` string encodes the 'h' extension.
196///
197/// Scans past the `rv32`/`rv64` prefix, then iterates the single-letter extension
198/// zone.  Digits and `p` in version suffixes (e.g. `i2p1`) are skipped — NOT
199/// treated as terminators — so `"rv64i2p1mafdch"` correctly detects 'h'.
200/// Stops at `_` (start of the multi-char extension zone).
201#[cfg(target_arch = "riscv64")]
202fn isa_string_has_h(isa: &str) -> bool {
203    let after_prefix = if isa.len() >= 4 {
204        &isa[4..]
205    } else {
206        return false;
207    };
208    for c in after_prefix.chars() {
209        match c {
210            'h' => return true,
211            '_' => return false, // multi-char zone; no more single-letter exts
212            'a'..='z' | '0'..='9' => {} // extension letter or version component
213            _ => return false,
214        }
215    }
216    false
217}
