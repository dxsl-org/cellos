1//! SMP: secondary hart startup and controlled park loop.
2//!
3//! Phase 01: brings each secondary hart online, installs its trap vector,
4//! then parks it in WFI.  Phase 03 replaces the park loop with a per-hart
5//! scheduler round.
6//!
7//! Invariant: hart 0 calls `start_secondaries()` only AFTER `task::init()`
8//! completes — the SCHEDULER and heap are live before any secondary runs.
9

24/// Maximum number of logical harts this kernel tracks. Every per-hart static
25/// is sized by it, so no heap allocation happens on the boot critical path.
26/// This is capacity, not the number of CPUs brought online.
27///
28/// - x86_64: the HAL's per-CPU table bound, `hal::x86_64::syscall::MAX_CPUS`
29///   (64). 64 keeps every `u64` hart mask exact, including the published
30///   `ViRetirementV1.remote_hart_mask` (plan 261103-2218 H-S2-5).
31/// - x86_64 G1 private lane (`feature = "g1-cpu-accounting"`): 2, which
32///   `task::cpu_accounting` asserts equal to its own hart count.
33/// - aarch64: 32 (spin-table and PSCI boards with more than two cores).
34/// - riscv64 / 32-bit targets: 2 (QEMU virt `-smp 2`, G2 entry).
35#[cfg(all(target_arch = "x86_64", not(feature = "g1-cpu-accounting")))]
36pub const MAX_HARTS: usize = hal::x86_64::syscall::MAX_CPUS;
37#[cfg(all(target_arch = "aarch64", not(feature = "g1-cpu-accounting")))]
38pub const MAX_HARTS: usize = 32;
39#[cfg(not(any(
40    all(target_arch = "x86_64", not(feature = "g1-cpu-accounting")),
41    all(target_arch = "aarch64", not(feature = "g1-cpu-accounting"))
42)))]
43pub const MAX_HARTS: usize = 2;

45/// The first RealTime hart: the remote hart that two-hart fixtures and RV64's
46/// SBI targets pair with hart 0. RT placement itself uses [`RT_HARTS`].
47pub const HART_RT: usize = 1;
48
49/// Harts that take RealTime-priority tasks (bit n: hart n). Every secondary
50/// hart by default, so RT work spreads instead of piling onto hart 1. RT tasks
51/// are never stolen (Phase 03 steal filter), so this mask and each task's last
52/// hart decide where they run (Phase 10 step 7).
53static RT_HARTS: AtomicUsize = AtomicUsize::new(!1);
54
55/// Whether `hart` is in the RealTime set.
56pub(crate) fn is_rt_hart(hart: usize) -> bool {
57    hart < usize::BITS as usize && RT_HARTS.load(Ordering::Relaxed) & (1 << hart) != 0
58}
59
60/// Where a RealTime task becomes runnable: its `last` hart when that hart is an
61/// online RT hart, otherwise the online RT hart with the fewest queued RT tasks
62/// (lowest hart on ties). `None` when no RT hart is online.
63pub(crate) fn rt_destination(last: Option<usize>) -> Option<usize> {
64    choose_rt_hart(
65        last,
66        |hart| is_rt_hart(hart) && HART_ONLINE[hart].load(Ordering::Acquire),
67        crate::sched::hart_local::ready::rt_queued,
68    )
69}
70
71/// [`rt_destination`]'s policy over explicit inputs.
72fn choose_rt_hart(
73    last: Option<usize>,
74    eligible: impl Fn(usize) -> bool,
75    queued: impl Fn(usize) -> usize,
76) -> Option<usize> {
77    if let Some(hart) = last.filter(|&hart| hart < MAX_HARTS && eligible(hart)) {
78        return Some(hart);
79    }
80    (0..MAX_HARTS)
81        .filter(|&hart| eligible(hart))
82        .min_by_key(|&hart| (queued(hart), hart))
83}

85/// Set to `true` by each secondary hart once its trap vector and timer are ready.
86/// Hart 0's bounded wait reads this via `Acquire` to observe all preceding stores.
87pub static HART_ONLINE: [AtomicBool; MAX_HARTS] = [const { AtomicBool::new(false) }; MAX_HARTS];
88
89/// Monotonic switch-completion epochs for root-retirement quiescence. A retiring
90/// generation cannot release its CellId slot until every requested hart has
91/// switched to a different context and published that completion.
92/// Per-hart invalidation epochs: a requester publishes a target, the target
93/// performs the local flush on its way through the trap path and publishes
94/// completion. Only then may the requester recycle a tag or release frames — a
95/// firmware call returning is not evidence that another hart stopped using the
96/// translation (phase 02 slice 3).
97static TLB_FLUSH_REQUEST: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
98static TLB_FLUSH_COMPLETE: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
99
100/// Deliver the kernel's cross-hart IPI to `hart_id`.
101///
102/// The requester only records an epoch; what makes the target *act* is this
103/// interrupt. Every path that needs a remote hart to do work (invalidate its
104/// TLB, switch out of a retiring context) goes through here so the delivery
105/// mechanism has one definition per architecture.
106#[inline]
107fn send_ipi(hart_id: usize) {
108    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
109    {
110        // Boards with no software-interrupt transport retain the request epoch.
111        #[cfg(target_arch = "aarch64")]
112        if !<hal::aarch64::backend::Board as hal::aarch64::backend::ArmIrqController>::HAS_IPI {
113            return;
114        }
115        if hart_id >= MAX_HARTS || hart_id == crate::sched::hart_local::current_hart_id() {
116            return;
117        }
118        #[cfg(target_arch = "riscv64")]
119        if let Some((mask, base)) = logical_sbi_target(hart_id) {
120            let _ = hal::common::sbi::sbi_send_ipi(mask, base);
121        }
122        // AArch64: the board's interrupt controller raises the IPI (a GIC SGI),
123        // addressed by CPU index or by affinity, whichever it uses.
124        #[cfg(target_arch = "aarch64")]
125        if let Some(cpu) = crate::arch::hart_local::physical_cpu_for(hart_id) {
126            if let Some(mpidr) = CPU_MPIDRS.mpidr(cpu) {
127                <hal::aarch64::backend::Board as hal::aarch64::backend::ArmIrqController>::send_ipi(
128                    cpu as u32, mpidr,
129                );
130            }
131        }
132    }
133    #[cfg(all(target_arch = "x86_64", not(all(test, not(target_os = "none")))))]
134    if hart_id < MAX_HARTS {
135        // No completion is forged on delivery failure. Keep this path free of
136        // locks/logging because a deferred-release requester may own a lock.
137        let _ = hal::x86_64::apic::send_ipi(hart_id);
138    }
139    // Boards with no software-interrupt transport retain the request epoch.
140    #[cfg(not(any(
141        target_arch = "riscv64",
142        target_arch = "aarch64",
143        all(target_arch = "x86_64", not(all(test, not(target_os = "none"))))
144    )))]
145    let _ = hart_id;
146}

148/// Ask `hart_id` to invalidate its local TLB and return the epoch it must publish.
149pub fn request_tlb_flush(hart_id: usize) -> usize {
150    if hart_id >= MAX_HARTS {
151        return 0;
152    }
153    let epoch = TLB_FLUSH_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
154    send_ipi(hart_id);
155    epoch
156}

158/// Is `epoch` known complete on `hart_id`? Epoch 0 means "nothing requested".
159pub fn tlb_flush_completed(hart_id: usize, epoch: usize) -> bool {
160    epoch == 0
161        || TLB_FLUSH_COMPLETE
162            .get(hart_id)
163            .is_some_and(|complete| complete.load(Ordering::Acquire) >= epoch)
164}

166/// Service one captured request. A newer request arriving during the local
167/// invalidate must NOT be acknowledged by that invalidate.
168pub fn service_tlb_flush(hart_id: usize) -> usize {
169    let (epoch, _advanced) = service_tlb_flush_captured(hart_id);
170    #[cfg(all(feature = "test-hooks", not(target_arch = "x86_64")))]
171    if _advanced {
172        log::info!(
173            "[selftest] TLB-ACK: stage=remote-flush-completed hart={} epoch={}",
174            hart_id,
175            epoch
176        );
177    }
178    epoch
179}

181/// [`service_tlb_flush`] for spin loops that may already own other locks
182/// (lock acquisition, cross-hart fixture waits): lock-free and log-free.
183///
184/// Every hart, hart 0 included, is a shootdown target, so a hart spinning with
185/// interrupts masked on progress that a peer makes only after this hart's
186/// acknowledgement must answer here, or both stall until the peer's bounded
187/// probe fails.
188#[inline]
189pub fn answer_tlb_flush_in_spin(hart_id: usize) {
190    let _ = service_tlb_flush_captured(hart_id);
191}

193/// Returns the captured request epoch and whether this call published it.
194fn service_tlb_flush_captured(hart_id: usize) -> (usize, bool) {
195    if hart_id >= MAX_HARTS {
196        return (0, false);
197    }
198    #[cfg(target_arch = "riscv64")]
199    service_icache_sync(hart_id);
200    let epoch = TLB_FLUSH_REQUEST[hart_id].load(Ordering::Acquire);
201    if TLB_FLUSH_COMPLETE[hart_id].load(Ordering::Acquire) < epoch {
202        crate::memory::paging::tlb_flush_all();
203        return (
204            epoch,
205            publish_completion(&TLB_FLUSH_COMPLETE[hart_id], epoch),
206        );
207    }
208    (epoch, false)
209}

211/// Publish `epoch` as complete without ever lowering `complete`; returns
212/// whether this call advanced it.
213///
214/// The service path can be re-entered by an IPI between its request snapshot
215/// and this publication (an x86 waiter spins with interrupts enabled), so the
216/// nested call may already have published a newer epoch that a plain store of
217/// the older snapshot would retract. Release orders the local invalidate
218/// before the requester's Acquire observation.
219fn publish_completion(complete: &AtomicUsize, epoch: usize) -> bool {
220    complete.fetch_max(epoch, Ordering::Release) < epoch
221}

223/// RV64 instruction-cache sync epochs: the IPI fallback a code publisher takes
224/// when SBI RFENCE `remote_fence_i` fails. Kept apart from the TLB epochs so a
225/// routine invalidation never pays for an instruction-cache flush. Served on
226/// the same trap path and in the same spin answers as the TLB request.
227#[cfg(target_arch = "riscv64")]
228static ICACHE_SYNC_REQUEST: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
229#[cfg(target_arch = "riscv64")]
230static ICACHE_SYNC_COMPLETE: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
231
232/// Make this hart's later instruction fetches observe every store it has
233/// observed (RV64 does not keep instruction fetch coherent with stores).
234#[cfg(target_arch = "riscv64")]
235#[inline]
236pub fn local_fence_i() {
237    // SAFETY: `fence.i` only orders this hart's later instruction fetches after
238    // the stores it has observed; it writes no register or memory.
239    unsafe { core::arch::asm!("fence.i", options(nostack, preserves_flags)) };
240}
241
242/// Ask `hart_id` to execute `fence.i` and return the epoch it must publish.
243///
244/// The Release half of the request orders the requester's code stores before
245/// the target's Acquire of the epoch, so the target's `fence.i` covers them.
246#[cfg(target_arch = "riscv64")]
247pub fn request_icache_sync(hart_id: usize) -> usize {
248    if hart_id >= MAX_HARTS {
249        return 0;
250    }
251    let epoch = ICACHE_SYNC_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
252    send_ipi(hart_id);
253    epoch
254}
255
256/// Is `epoch` known complete on `hart_id`? Epoch 0 means "nothing requested".
257#[cfg(target_arch = "riscv64")]
258pub fn icache_sync_completed(hart_id: usize, epoch: usize) -> bool {
259    epoch == 0
260        || ICACHE_SYNC_COMPLETE
261            .get(hart_id)
262            .is_some_and(|complete| complete.load(Ordering::Acquire) >= epoch)
263}
264
265/// Serve one captured instruction-cache sync request on this hart.
266#[cfg(target_arch = "riscv64")]
267fn service_icache_sync(hart_id: usize) {
268    let epoch = ICACHE_SYNC_REQUEST[hart_id].load(Ordering::Acquire);
269    if ICACHE_SYNC_COMPLETE[hart_id].load(Ordering::Acquire) < epoch {
270        local_fence_i();
271        publish_completion(&ICACHE_SYNC_COMPLETE[hart_id], epoch);
272    }
273}

283/// Does this hart owe the requester an invalidation completion?
284pub fn tlb_flush_pending(hart_id: usize) -> bool {
285    hart_id < MAX_HARTS
286        && TLB_FLUSH_REQUEST[hart_id].load(Ordering::Acquire)
287            > TLB_FLUSH_COMPLETE[hart_id].load(Ordering::Acquire)

290/// Epochs one requester asked a set of remote harts for: one slot per logical
291/// hart, 0 = not asked (epoch 0 is never a real request).
292///
293/// Broadcast-then-wait contract (plan 261103-2218 phase 07b): every request of
294/// the set is published and its IPI sent before the requester waits on any
295/// answer, so the targets work concurrently and the set completes in about one
296/// round trip, not one per hart. The set is fixed when it is built: a hart that
297/// comes online later was not asked and is not waited for (it joins only after
298/// its own bring-up flush handshake). Completion is per hart and monotone
299/// (`publish_completion` never lowers an epoch), so a hart found complete
300/// stays complete; the set is complete when no slot is outstanding.
301#[derive(Clone, Copy)]
302pub struct RemoteEpochs {
303    epochs: [usize; MAX_HARTS],
304}
305
306impl RemoteEpochs {
307    /// Ask each of `targets` once through `request` (which publishes the epoch
308    /// and delivers the IPI) and record the epoch it must confirm. Ids at or
309    /// above `MAX_HARTS` are ignored.
310    pub fn broadcast(
311        targets: impl IntoIterator<Item = usize>,
312        mut request: impl FnMut(usize) -> usize,
313    ) -> Self {
314        let mut epochs = [0; MAX_HARTS];
315        for hart in targets {
316            if let Some(slot) = epochs.get_mut(hart) {
317                *slot = request(hart);
318            }
319        }
320        Self { epochs }
321    }
322
323    /// The lowest-numbered asked hart whose epoch `completed` does not confirm
324    /// yet, with that epoch; `None` once the whole set has confirmed.
325    pub fn first_outstanding(
326        &self,
327        mut completed: impl FnMut(usize, usize) -> bool,
328    ) -> Option<(usize, usize)> {
329        self.epochs
330            .iter()
331            .enumerate()
332            .find(|&(hart, &epoch)| epoch != 0 && !completed(hart, epoch))
333            .map(|(hart, &epoch)| (hart, epoch))
334    }
335
336    /// How many harts were asked.
337    pub fn asked(&self) -> usize {
338        self.epochs.iter().filter(|epoch| **epoch != 0).count()
339    }
340}
341
342/// Broadcast a TLB invalidation request to `targets` (see [`RemoteEpochs`]).
343pub fn request_tlb_flush_set(targets: impl IntoIterator<Item = usize>) -> RemoteEpochs {
344    RemoteEpochs::broadcast(targets, request_tlb_flush)
345}
346
347/// The first hart of `set` that has not confirmed its invalidation epoch.
348pub fn tlb_flush_set_outstanding(set: &RemoteEpochs) -> Option<(usize, usize)> {
349    set.first_outstanding(tlb_flush_completed)
350}

352/// The logical harts that completed kernel bring-up, the boot hart included.
353///
354/// Every remote-invalidation and flush-epoch target set is built from this

356pub fn online_harts() -> impl Iterator<Item = usize> {

379/// Request that `hart_id` schedules through a retirement boundary and return
380/// the epoch that its incoming context must complete.
381pub fn request_retirement_switch(hart_id: usize) -> usize {
382    if hart_id >= MAX_HARTS {
383        return 0;
384    }
385    let epoch = RETIRE_SWITCH_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
386    #[cfg(feature = "test-hooks")]
387    log::info!(
388        "[selftest] SMP-RETIREMENT: stage=remote-switch-requested hart={} epoch={}",
389        hart_id,
390        epoch
391    );
392    send_ipi(hart_id);
393    epoch
394}

396/// Publish the requested epoch from the incoming side of `Context::switch`.
397///
398/// The release store is deliberately after the raw context switch has changed
399/// stacks: an IPI/trap entry only proves that the outgoing task entered the
400/// kernel, while this proves that its saved context no longer executes.
401pub fn complete_retirement_switch(hart_id: usize) {
402    if hart_id < MAX_HARTS {
403        let epoch = RETIRE_SWITCH_REQUEST[hart_id].load(Ordering::Acquire);
404        let completed = RETIRE_SWITCH_COMPLETE[hart_id].load(Ordering::Acquire);
405        if completed < epoch {
406            RETIRE_SWITCH_COMPLETE[hart_id].store(epoch, Ordering::Release);
407            #[cfg(feature = "test-hooks")]
408            log::info!(
409                "[selftest] SMP-RETIREMENT: stage=remote-switch-completed hart={} epoch={}",
410                hart_id,
411                epoch
412            );
413        }
414    }
415}

474/// Entry point for secondary harts, called from `_secondary_entry` asm.
475///
476/// a0 = hart_id (set by OpenSBI per SBI HSM §9.1.1).
477///
478/// Installs the trap vector, enables the timer, runs the per-hart scheduler loop.
479#[no_mangle]
480pub extern "C" fn smp_hart_entry(physical_hart: usize) -> ! {
481    #[cfg(target_arch = "riscv64")]
482    let hart_id = rv::secondary_start(physical_hart);
483    #[cfg(not(target_arch = "riscv64"))]
484    let hart_id = physical_hart;
485
486    // Signal hart 0's bounded wait only after this hart can actually take the
487    // dispatch IPI and run its local scheduler.
488    if hart_id < MAX_HARTS {
489        log::info!(
490            "[smp] physical {} -> logical {} trap-ready, interrupts-enabled",
491            physical_hart,
492            hart_id
493        );
494        #[cfg(feature = "test-hooks")]
495        log::info!(
496            "[selftest] SMP-RETIREMENT: stage=hart{}-interrupts-enabled",
497            hart_id
498        );
499        HART_ONLINE[hart_id].store(true, Ordering::Release);
500    }
501
502    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
503    crate::arch::memory::tlb_shootdown_selftest::run_secondary(hart_id);
504    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
505    crate::arch::cell::provenance_selftest::run_secondary(hart_id);
506
507    // Per-hart scheduler loop.  The timer ISR (vi_timer_tick) calls yield_cpu()
508    // which runs pick_next for THIS hart (work-stealing from hart 0 if idle).
509    // Between ticks we sit in WFI to save power.  Interrupts are enabled on
510    // entry (ARCH.init() sets sstatus.SIE=1), so WFI fires on the timer ISR.
511    loop {
512        #[cfg(feature = "test-hooks")]
513        crate::loader::atomic_publication_tests::observe_schedule_attempt();
514        // wfi suspends until the next interrupt; state is unchanged.
515        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
516        crate::arch::cpu::halt();
517        core::hint::spin_loop();
518    }
519}

525    /// An IPI-nested service publishes a newer epoch before the interrupted
526    /// outer service publishes its older snapshot; completion must not regress.
527    #[test]
528    fn nested_service_completion_is_not_retracted_by_stale_snapshot() {
529        let complete = AtomicUsize::new(4);
530        let outer_snapshot = 5;
531        assert!(publish_completion(&complete, 6));
532        assert!(!publish_completion(&complete, outer_snapshot));

553    /// request is issued before any completion is consulted, the set waits on
554    /// the lowest silent hart, and it completes only when all three confirmed.
555    #[test]
556    fn epoch_set_broadcasts_before_waiting_and_completes_over_four_harts() {
557        const HARTS: usize = 4;
558        let me = 0;
559        let request = [const { AtomicUsize::new(0) }; HARTS];
560        let complete = [const { AtomicUsize::new(0) }; HARTS];
561        let mut issued = alloc::vec::Vec::new();
562        let set = RemoteEpochs::broadcast((0..HARTS).filter(|hart| *hart != me), |hart| {
563            issued.push(hart);
564            request[hart].fetch_add(1, Ordering::AcqRel) + 1
565        });
566        assert_eq!(issued, [1, 2, 3], "one request per remote, in id order");
567        assert_eq!(set.asked(), 3);

569        assert_eq!(set.first_outstanding(completed), Some((1, 1)));
570        // Out of order: 3 and 1 answer, 2 stays silent.
571        assert!(publish_completion(&complete[3], 1));
572        assert!(publish_completion(&complete[1], 1));
573        assert_eq!(set.first_outstanding(completed), Some((2, 1)));
574        assert!(publish_completion(&complete[2], 1));
575        assert_eq!(set.first_outstanding(completed), None);
576        // The requester never asked itself.
577        assert_eq!(request[me].load(Ordering::Acquire), 0);

605    /// retires its own slot.

613        }
