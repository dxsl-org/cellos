1//! SMP: secondary hart startup and controlled park loop.
2//!
3//! Phase 01: brings each secondary hart online, installs its trap vector,
4//! then parks it in WFI.  Phase 03 replaces the park loop with a per-hart
5//! scheduler round.
6//!
7//! Invariant: hart 0 calls `start_secondaries()` only AFTER `task::init()`
8//! completes — the SCHEDULER and heap are live before any secondary runs.

23/// Maximum number of logical harts this kernel tracks. Every per-hart static
24/// is sized by it, so no heap allocation happens on the boot critical path.
25/// This is capacity, not the number of CPUs brought online.
26///
27/// - x86_64: the HAL's per-CPU table bound, `hal::x86_64::syscall::MAX_CPUS`
28///   (64). 64 keeps every `u64` hart mask exact, including the published
29///   `ViRetirementV1.remote_hart_mask` (plan 261103-2218 H-S2-5).
30/// - x86_64 G1 private lane (`feature = "g1-cpu-accounting"`): 2, which
31///   `task::cpu_accounting` asserts equal to its own hart count.
32/// - aarch64: 32 (spin-table and PSCI boards with more than two cores).
33/// - riscv64 / 32-bit targets: 2 (QEMU virt `-smp 2`, G2 entry).
34#[cfg(all(target_arch = "x86_64", not(feature = "g1-cpu-accounting")))]
35pub const MAX_HARTS: usize = hal::x86_64::syscall::MAX_CPUS;
36#[cfg(all(target_arch = "aarch64", not(feature = "g1-cpu-accounting")))]
37pub const MAX_HARTS: usize = 32;
38#[cfg(not(any(
39    all(target_arch = "x86_64", not(feature = "g1-cpu-accounting")),
40    all(target_arch = "aarch64", not(feature = "g1-cpu-accounting"))
41)))]
42pub const MAX_HARTS: usize = 2;
43// The HAL owns one GS record, GDT/TSS and IST set per logical CPU.
44#[cfg(target_arch = "x86_64")]
45const _: () = assert!(MAX_HARTS <= hal::x86_64::syscall::MAX_CPUS);

47/// The first RealTime hart: the remote hart that two-hart fixtures and RV64's
48/// SBI targets pair with hart 0. RT placement itself uses [`RT_HARTS`].
49pub const HART_RT: usize = 1;

51/// Harts that take RealTime-priority tasks (bit n: hart n). Every secondary
52/// hart by default, so RT work spreads instead of piling onto hart 1. RT tasks
53/// are never stolen (Phase 03 steal filter), so this mask and each task's last
54/// hart decide where they run (Phase 10 step 7).
55static RT_HARTS: AtomicUsize = AtomicUsize::new(!1);

57/// Whether `hart` is in the RealTime set.
58pub(crate) fn is_rt_hart(hart: usize) -> bool {
59    hart < usize::BITS as usize && RT_HARTS.load(Ordering::Relaxed) & (1 << hart) != 0
60}

62/// Where a RealTime task becomes runnable: its `last` hart when that hart is an
63/// online RT hart, otherwise the online RT hart with the fewest queued RT tasks
64/// (lowest hart on ties). `None` when no RT hart is online.
65pub(crate) fn rt_destination(last: Option<usize>) -> Option<usize> {
66    choose_rt_hart(
67        last,
68        |hart| is_rt_hart(hart) && HART_ONLINE[hart].load(Ordering::Acquire),
69        crate::sched::hart_local::ready::rt_queued,
70    )
71}

73/// [`rt_destination`]'s policy over explicit inputs.
74fn choose_rt_hart(
75    last: Option<usize>,
76    eligible: impl Fn(usize) -> bool,
77    queued: impl Fn(usize) -> usize,
78) -> Option<usize> {
79    if let Some(hart) = last.filter(|&hart| hart < MAX_HARTS && eligible(hart)) {
80        return Some(hart);
81    }
82    (0..MAX_HARTS)
83        .filter(|&hart| eligible(hart))
84        .min_by_key(|&hart| (queued(hart), hart))
85}

87/// Set to `true` by each secondary hart once its trap vector and timer are ready.
88/// Hart 0's bounded wait reads this via `Acquire` to observe all preceding stores.
89pub static HART_ONLINE: [AtomicBool; MAX_HARTS] = [const { AtomicBool::new(false) }; MAX_HARTS];

91/// Monotonic switch-completion epochs for root-retirement quiescence. A retiring
92/// generation cannot release its CellId slot until every requested hart has
93/// switched to a different context and published that completion.
94/// Per-hart invalidation epochs: a requester publishes a target, the target
95/// performs the local flush on its way through the trap path and publishes
96/// completion. Only then may the requester recycle a tag or release frames — a
97/// firmware call returning is not evidence that another hart stopped using the
98/// translation (phase 02 slice 3).
99static TLB_FLUSH_REQUEST: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
100static TLB_FLUSH_COMPLETE: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
101
102/// IPIs each hart has taken through its interrupt controller. The timer tick
103/// also services flush requests, so only this count tells [`run_ipi_selftest`]
104/// that the SGI itself — not the next tick — reached the hart.
105#[cfg(target_arch = "aarch64")]
106static IPI_TAKEN: [AtomicUsize; MAX_HARTS] = [const { AtomicUsize::new(0) }; MAX_HARTS];
107
108/// Record that `hart_id` took the kernel's IPI. Called by the IPI handler
109/// (`vi_ipi_service`) on the hart itself, before it services the request, so
110/// the requester's Acquire of that completion also makes this count visible.
111#[cfg(target_arch = "aarch64")]
112#[inline]
113pub(crate) fn note_ipi_taken(hart_id: usize) {
114    if let Some(taken) = IPI_TAKEN.get(hart_id) {
115        taken.fetch_add(1, Ordering::Relaxed);
116    }
117}

119/// Deliver the kernel's cross-hart IPI to `hart_id`.
120///
121/// The requester only records an epoch; what makes the target *act* is this
122/// interrupt. Every path that needs a remote hart to do work (invalidate its
123/// TLB, switch out of a retiring context) goes through here so the delivery
124/// mechanism has one definition per architecture.
125#[inline]
126fn send_ipi(hart_id: usize) {
127    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
128    {
129        // Boards with no software-interrupt transport retain the request epoch.
130        #[cfg(target_arch = "aarch64")]
131        if !<hal::aarch64::backend::Board as hal::aarch64::backend::ArmIrqController>::HAS_IPI {
132            return;
133        }
134        if hart_id >= MAX_HARTS || hart_id == crate::sched::hart_local::current_hart_id() {
135            return;
136        }
137        #[cfg(target_arch = "riscv64")]
138        if let Some((mask, base)) = logical_sbi_target(hart_id) {
139            let _ = hal::common::sbi::sbi_send_ipi(mask, base);
140        }
141        // AArch64: the board's interrupt controller raises the IPI (a GIC SGI),
142        // addressed by CPU index or by affinity, whichever it uses.
143        #[cfg(target_arch = "aarch64")]
144        if let Some(cpu) = crate::sched::hart_local::physical_cpu_for(hart_id) {
145            if let Some(mpidr) = CPU_MPIDRS.mpidr(cpu) {
146                <hal::aarch64::backend::Board as hal::aarch64::backend::ArmIrqController>::send_ipi(
147                    cpu as u32, mpidr,
148                );
149            }
150        }
151    }
152    #[cfg(all(target_arch = "x86_64", not(all(test, not(target_os = "none")))))]
153    if hart_id < MAX_HARTS {
154        // No completion is forged on delivery failure. Keep this path free of
155        // locks/logging because a deferred-release requester may own a lock.
156        let _ = hal::x86_64::apic::send_ipi(hart_id);
157    }
158    // Boards with no software-interrupt transport retain the request epoch.
159    #[cfg(not(any(
160        target_arch = "riscv64",
161        target_arch = "aarch64",
162        all(target_arch = "x86_64", not(all(test, not(target_os = "none"))))
163    )))]
164    let _ = hart_id;
165}

167/// Ask `hart_id` to invalidate its local TLB and return the epoch it must publish.
168pub fn request_tlb_flush(hart_id: usize) -> usize {
169    if hart_id >= MAX_HARTS {
170        return 0;
171    }
172    let epoch = TLB_FLUSH_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
173    send_ipi(hart_id);
174    epoch
175}

177/// Is `epoch` known complete on `hart_id`? Epoch 0 means "nothing requested".
178pub fn tlb_flush_completed(hart_id: usize, epoch: usize) -> bool {
179    epoch == 0
180        || TLB_FLUSH_COMPLETE
181            .get(hart_id)
182            .is_some_and(|complete| complete.load(Ordering::Acquire) >= epoch)
183}

185/// Service one captured request. A newer request arriving during the local
186/// invalidate must NOT be acknowledged by that invalidate.
187pub fn service_tlb_flush(hart_id: usize) -> usize {
188    let (epoch, _advanced) = service_tlb_flush_captured(hart_id);
189    #[cfg(all(feature = "test-hooks", not(target_arch = "x86_64")))]
190    if _advanced {
191        log::info!(
192            "[selftest] TLB-ACK: stage=remote-flush-completed hart={} epoch={}",
193            hart_id,
194            epoch
195        );
196    }
197    epoch
198}

200/// [`service_tlb_flush`] for spin loops that may already own other locks
201/// (lock acquisition, cross-hart fixture waits): lock-free and log-free.
202///
203/// Every hart, hart 0 included, is a shootdown target, so a hart spinning with
204/// interrupts masked on progress that a peer makes only after this hart's
205/// acknowledgement must answer here, or both stall until the peer's bounded
206/// probe fails.
207#[inline]
208pub fn answer_tlb_flush_in_spin(hart_id: usize) {
209    let _ = service_tlb_flush_captured(hart_id);
210}

212/// Returns the captured request epoch and whether this call published it.
213fn service_tlb_flush_captured(hart_id: usize) -> (usize, bool) {
214    if hart_id >= MAX_HARTS {
215        return (0, false);
216    }
217    let epoch = TLB_FLUSH_REQUEST[hart_id].load(Ordering::Acquire);
218    if TLB_FLUSH_COMPLETE[hart_id].load(Ordering::Acquire) < epoch {
219        crate::memory::paging::tlb_flush_all();
220        return (
221            epoch,
222            publish_completion(&TLB_FLUSH_COMPLETE[hart_id], epoch),
223        );
224    }
225    (epoch, false)
226}

228/// Publish `epoch` as complete without ever lowering `complete`; returns
229/// whether this call advanced it.
230///
231/// The service path can be re-entered by an IPI between its request snapshot
232/// and this publication (an x86 waiter spins with interrupts enabled), so the
233/// nested call may already have published a newer epoch that a plain store of
234/// the older snapshot would retract. Release orders the local invalidate
235/// before the requester's Acquire observation.
236fn publish_completion(complete: &AtomicUsize, epoch: usize) -> bool {
237    complete.fetch_max(epoch, Ordering::Release) < epoch
238}

255/// Epochs one requester asked a set of remote harts for: one slot per logical
256/// hart, 0 = not asked (epoch 0 is never a real request).
257///
258/// Broadcast-then-wait contract (plan 261103-2218 phase 07b): every request of
259/// the set is published and its IPI sent before the requester waits on any
260/// answer, so the targets work concurrently and the set completes in about one
261/// round trip, not one per hart. The set is fixed when it is built: a hart that
262/// comes online later was not asked and is not waited for (it joins only after
263/// its own bring-up flush handshake). Completion is per hart and monotone
264/// (`publish_completion` never lowers an epoch), so a hart found complete
265/// stays complete; the set is complete when no slot is outstanding.
266#[derive(Clone, Copy)]
267pub struct RemoteEpochs {
268    epochs: [usize; MAX_HARTS],
269}
270
271impl RemoteEpochs {
272    /// Ask each of `targets` once through `request` (which publishes the epoch
273    /// and delivers the IPI) and record the epoch it must confirm. Ids at or
274    /// above `MAX_HARTS` are ignored.
275    pub fn broadcast(
276        targets: impl IntoIterator<Item = usize>,
277        mut request: impl FnMut(usize) -> usize,
278    ) -> Self {
279        let mut epochs = [0; MAX_HARTS];
280        for hart in targets {
281            if let Some(slot) = epochs.get_mut(hart) {
282                *slot = request(hart);
283            }
284        }
285        Self { epochs }
286    }
287
288    /// The lowest-numbered asked hart whose epoch `completed` does not confirm
289    /// yet, with that epoch; `None` once the whole set has confirmed.
290    pub fn first_outstanding(
291        &self,
292        mut completed: impl FnMut(usize, usize) -> bool,
293    ) -> Option<(usize, usize)> {
294        self.epochs
295            .iter()
296            .enumerate()
297            .find(|&(hart, &epoch)| epoch != 0 && !completed(hart, epoch))
298            .map(|(hart, &epoch)| (hart, epoch))
299    }
300
301    /// How many harts were asked.
302    pub fn asked(&self) -> usize {
303        self.epochs.iter().filter(|epoch| **epoch != 0).count()
304    }
305}

391/// Request that `hart_id` schedules through a retirement boundary and return
392/// the epoch that its incoming context must complete.
393pub fn request_retirement_switch(hart_id: usize) -> usize {
394    if hart_id >= MAX_HARTS {
395        return 0;
396    }
397    let epoch = RETIRE_SWITCH_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
398    #[cfg(feature = "test-hooks")]
399    log::info!(
400        "[selftest] SMP-RETIREMENT: stage=remote-switch-requested hart={} epoch={}",
401        hart_id,
402        epoch
403    );
404    send_ipi(hart_id);
405    epoch
406}

408/// Publish the requested epoch from the incoming side of `Context::switch`.
409///
410/// The release store is deliberately after the raw context switch has changed
411/// stacks: an IPI/trap entry only proves that the outgoing task entered the
412/// kernel, while this proves that its saved context no longer executes.
413pub fn complete_retirement_switch(hart_id: usize) {
414    if hart_id < MAX_HARTS {
415        let epoch = RETIRE_SWITCH_REQUEST[hart_id].load(Ordering::Acquire);
416        let completed = RETIRE_SWITCH_COMPLETE[hart_id].load(Ordering::Acquire);
417        if completed < epoch {
418            RETIRE_SWITCH_COMPLETE[hart_id].store(epoch, Ordering::Release);
419            #[cfg(feature = "test-hooks")]
420            log::info!(
421                "[selftest] SMP-RETIREMENT: stage=remote-switch-completed hart={} epoch={}",
422                hart_id,
423                epoch
424            );
425        }
426    }
427}

485/// Return the one remote RV64 hart this boot can have, if it is online.
486///
487/// Every hart, the boot hart included, publishes readiness in `HART_ONLINE`
488/// with Release. RV64 runs at most two harts (`MAX_HARTS = 2`), so the remote
489/// hart is the RT hart for hart 0 and hart 0 for everyone else.

494    let remote = if current == 0 { HART_RT } else { 0 };

497}

521/// Called by hart 0 **after** `task::init()` to bring secondary harts online.
522///
523/// Each secondary is started via SBI HSM `hart_start`.  Hart 0 then spins
524/// (bounded) waiting for each secondary to set `HART_ONLINE[hart_id]`.

526/// system continues single-hart — graceful degradation, never a panic.
527#[cfg(target_arch = "riscv64")]
528pub fn start_secondaries() {
529    use crate::task::stack::Stack;
530    use crate::task::STACK_PAGES;
531    use hal::common::sbi::{sbi_hart_get_status, sbi_hart_start, sbi_rfence_available};
532
533    let Some(boot_physical) = boot_physical_hart() else {
534        log::warn!("[smp] boot physical hart was not published");
535        return;
536    };
537    log::info!("[smp] physical {} -> logical 0 boot", boot_physical);
538
539    match sbi_rfence_available() {
540        Ok(true) => {}
541        Ok(false) => {
542            log::warn!("[smp] SBI RFENCE unavailable — keeping Cellos single-hart");
543            return;
544        }
545        Err(error) => {
546            log::warn!(
547                "[smp] SBI RFENCE probe failed (err={}) — keeping Cellos single-hart",
548                error
549            );
550            return;
551        }
552    }
553
554    extern "C" {
555        // Physical asm label defined in hal/arch/riscv/src/rv64/boot.rs.
556        // Runs bare (SATP=0); no relocation or BSS clear.
557        fn _secondary_entry();
558    }
559
560    for (hart_id, online) in HART_ONLINE.iter().enumerate().skip(1) {
561        let Some(physical_hart) = logical_to_physical(hart_id) else {
562            log::warn!("[smp] logical hart {} has no physical mapping", hart_id);
563            continue;
564        };
565        // Allocate a dedicated kernel stack for this hart.  Leak it — it lives
566        // for the entire lifetime of the hart.
567        let stack = match Stack::new_kernel(STACK_PAGES) {
568            Ok(s) => s,
569            Err(e) => {
570                log::warn!("[smp] hart {} stack alloc failed: {:?}", hart_id, e);
571                continue;
572            }
573        };
574        let stack_top = stack.top;
575        core::mem::forget(stack);
576
577        let Ok(state) = sbi_hart_get_status(physical_hart) else {
578            log::warn!(
579                "[smp] physical hart {} HSM status unavailable",
580                physical_hart
581            );
582            continue;
583        };
584        log::info!(
585            "[smp] physical {} -> logical {} HSM state = {}",
586            physical_hart,
587            hart_id,
588            state
589        );
590        if state != 1 {
591            log::warn!("[smp] physical hart {} is not HSM STOPPED", physical_hart);
592            continue;
593        }
594
595        // SAFETY: _secondary_entry is a physical-address asm label; the kernel
596        // is loaded at 0x80200000 with slide=0 so physical == virtual.
597        // stack_top is the usable top of a freshly-allocated kernel stack.
598        // SAFETY: casting function pointer to integer — use double-cast through
599        // *const () to avoid the "direct cast of function item" lint.
600        let entry_paddr = _secondary_entry as *const () as usize;
601        match sbi_hart_start(physical_hart, entry_paddr, stack_top) {
602            Ok(()) => log::info!(
603                "[smp] hart {} start requested (entry={:#x})",
604                physical_hart,
605                entry_paddr
606            ),
607            Err(e) => {
608                log::warn!("[smp] hart {} SBI hart_start failed: err={}", hart_id, e);
609                continue;
610            }
611        }
612
613        // Bounded spin: wait for the secondary to signal it is online.
614        let deadline = crate::task::system_ticks() + SECONDARY_BOOT_TIMEOUT_TICKS;
615        loop {
616            if online.load(Ordering::Acquire) {
617                log::info!("[smp] hart {} online, parked", hart_id);
618                break;
619            }
620            if crate::task::system_ticks() >= deadline {
621                log::warn!(
622                    "[smp] hart {} did not come online in time — continuing single-hart",
623                    hart_id
624                );
625                break;
626            }
627            core::hint::spin_loop();
628        }
629    }
630}

665/// AArch64 starts a core through firmware (`PSCI_CPU_ON` over the conduit the
666/// device tree declares). Everything the new core needs — its stack, the boot
667/// core's live translation regime, the hardening bits in `SCTLR_EL1` — travels
668/// in one context block it reads with the MMU still off; `_secondary_entry`
669/// installs the regime and calls [`smp_aarch64_secondary_main`].
670///
671/// Bounded wait, graceful degradation, never a panic: a core that does not come
672/// online within the deadline leaves the system single-hart with a warning.
673#[cfg(target_arch = "aarch64")]
674fn start_psci_secondaries() {
675    use crate::boot::firmware_smp;
676    use crate::task::stack::Stack;
677    use crate::task::STACK_PAGES;
678
679    let at_el2 = hal::aarch64::el2::is_el2();
680    // Which instruction carries a PSCI call: the firmware tree's answer when
681    // there is one, otherwise what the architecture allows — with EL3 the
682    // conduit is EL3 firmware's (`smc`); without EL3 an `smc` is undefined and
683    // the only conduit that can exist is the hypervisor's (`hvc`).
684    let conduit = match firmware_smp::psci_conduit() {
685        Some(declared) => declared,
686        None => {
687            let inferred = hal::aarch64::psci::inferred_conduit();
688            log::info!(
689                "[smp] no firmware tree declares PSCI; EL3 {} — inferring conduit {:?}",
690                if hal::aarch64::psci::el3_implemented() {
691                    "implemented"
692                } else {
693                    "absent"
694                },
695                inferred
696            );
697            inferred
698        }
699    };
700    // A conduit that cannot work is reported, never attempted: `smc` without EL3
701    // and `hvc` from EL2 both trap as undefined instructions.
702    let usable = match (conduit, at_el2) {
703        // EL3 firmware answers an SMC from either exception level.
704        (hal::aarch64::psci::Conduit::Smc, _) => true,
705        // An HVC from EL1 traps to EL2, where the PSCI implementation lives; an
706        // HVC from EL2 would target EL3, which by construction has no PSCI here.
707        (hal::aarch64::psci::Conduit::Hvc, false) => true,
708        (hal::aarch64::psci::Conduit::Hvc, true) => false,
709    };
710    if !usable {
711        log::warn!(
712            "[smp] the kernel runs at EL2 and its conduit is HVC, which would target EL3 — \
713             keeping Cellos single-hart"
714        );
715        return;
716    }
717    log::info!(
718        "[smp] PSCI conduit {:?}, kernel at {}",
719        conduit,
720        if at_el2 { "EL2" } else { "EL1" }
721    );
722    // Logical hart N runs CPU index N (the N-th `/cpus` node), so the boot core
723    // has to *be* CPU 0 for the mapping to hold; anything else is reported
724    // rather than half-applied.
725    let boot_cpu = crate::sched::hart_local::physical_cpu_index();
726    if boot_cpu != Some(0) {
727        log::warn!(
728            "[smp] boot core reports CPU index {:?} (expected 0) — keeping Cellos single-hart",
729            boot_cpu
730        );
731        return;
732    }
733    // How many cores to try. The tree is authoritative when it says anything;
734    // with no tree the firmware's own answer to `CPU_ON` is what tells us
735    // whether a core exists, so every hart this kernel models is attempted and
736    // a refusal ends the search.
737    let described = firmware_smp::cpu_count();
738    if described == 1 {
739        log::info!("[smp] firmware tree describes 1 CPU — single-hart");
740        return;
741    }
742    let attempt = if described == 0 {
743        MAX_HARTS
744    } else {
745        MAX_HARTS.min(described)
746    };
747    match hal::aarch64::psci::version(conduit) {
748        Some(version) => log::info!(
749            "[smp] PSCI {}.{} over {:?}, {} CPU(s) described",
750            version >> 16,
751            version & 0xFFFF,
752            conduit,
753            described
754        ),
755        None => {
756            log::warn!("[smp] firmware does not answer PSCI_VERSION — keeping Cellos single-hart");
757            return;
758        }
759    }
760
761    extern "C" {
762        /// Physical entry point defined in hal/arch/arm/src/aarch64/boot.rs.
763        /// Runs with the MMU off; installs the boot core's translation regime.
764        fn _secondary_entry();
765    }
766    let entry = _secondary_entry as *const () as usize;
767    let kernel_root = match *crate::memory::paging::KERNEL_ROOT.lock() {
768        Some(root) => root,
769        None => {
770            log::warn!("[smp] kernel page tables are not published — keeping Cellos single-hart");
771            return;
772        }
773    };
774
775    for hart_id in 1..attempt {
776        // PSCI names the core by its full affinity, which only the firmware
777        // tree knows once a machine has more than one cluster.
778        let Some(mpidr) = CPU_MPIDRS.mpidr(hart_id) else {
779            log::info!(
780                "[smp] hart {}: no CPU {} described — done",
781                hart_id,
782                hart_id
783            );
784            break;
785        };
786        // The core is still off: this mapping has to exist before it reads it.
787        if !crate::sched::hart_local::publish_physical_cpu(hart_id, hart_id) {
788            log::warn!(
789                "[smp] hart {} conflicts with the published topology — not started",
790                hart_id
791            );
792            continue;
793        }
794        let stack = match Stack::new_kernel(STACK_PAGES) {
795            Ok(stack) => stack,
796            Err(error) => {
797                log::warn!("[smp] hart {} stack alloc failed: {:?}", hart_id, error);
798                continue;
799            }
800        };
801        let stack_top = stack.top;
802        // Leaked on purpose: a hart's stack lives as long as the hart does.
803        core::mem::forget(stack);
804
805        let context = hal::aarch64::boot::SecondaryContext::for_hart(
806            hart_id as u64,
807            stack_top as u64,
808            kernel_root as u64,
809        );
810        // SAFETY: single writer (hart 0, before the core exists), and no reader
811        // until `PSCI_CPU_ON` succeeds.
812        let context_addr = unsafe {
813            let slot = core::ptr::addr_of_mut!(SECONDARY_CONTEXT[hart_id]);
814            slot.write(context);
815            slot as usize
816        };
817        // The core reads this with caches off: publish it to the point of
818        // coherency, and order that before the firmware call.
819        hal::aarch64::cache::clean_data_cache_range(
820            context_addr,
821            core::mem::size_of::<hal::aarch64::boot::SecondaryContext>(),
822        );
823        core::sync::atomic::fence(Ordering::SeqCst);
824
825        match hal::aarch64::psci::cpu_on(conduit, mpidr, entry, context_addr as u64) {
826            Ok(()) => log::info!(
827                "[smp] hart {} start requested (cpu={} mpidr={:#x} entry={:#x})",
828                hart_id,
829                hart_id,
830                mpidr,
831                entry
832            ),
833            Err(status) => {
834                // With nothing describing the CPU population, "no such core" is
835                // the expected answer on a smaller machine — and it also ends the
836                // search, because cores are numbered from zero.
837                // PSCI 1.0: -2 INVALID_PARAMETERS, -7 NOT_PRESENT.
838                let missing = described == 0 && (status == -2 || status == -7);
839                if missing {
840                    log::info!(
841                        "[smp] hart {}: firmware reports no such core — single-hart",
842                        hart_id
843                    );
844                    break;
845                }
846                log::warn!(
847                    "[smp] hart {} PSCI CPU_ON refused: {} ({})",
848                    hart_id,
849                    hal::aarch64::psci::status_name(status),
850                    status
851                );
852                continue;
853            }
854        }
855
856        // Bounded wait using the hardware monotonic counter (CNTPCT_EL0) which
857        // advances even with interrupts masked on the boot hart.
858        let timeout_ticks = (hal::aarch64::timer::counter_frequency_hz() / 2).max(1);
859        let start = hal::aarch64::timer::counter_now();
860        loop {
861            if HART_ONLINE[hart_id].load(Ordering::Acquire) {
862                log::info!("[smp] hart {} online, parked", hart_id);
863                break;
864            }
865            if hal::aarch64::timer::counter_now().wrapping_sub(start) >= timeout_ticks {
866                log::warn!(
867                    "[smp] hart {} did not come online in time — continuing single-hart",
868                    hart_id
869                );
870                break;
871            }
872            core::hint::spin_loop();
873        }
874    }
875}

877/// AArch64 starts a core through the device tree's spin table: the core spins
878/// on its `cpu-release-addr` with the MMU off until that word holds an entry
879/// address. A release passes no argument, so `_spin_table_entry` loads the
880/// context from the single slot [`hal::aarch64::boot::publish_spin_table_context`]
881/// fills — which is why cores are released one at a time and the first one
882/// that does not come online ends the bring-up.
883///
884/// The release word is written by physical address ([`write_release_word`]),
885/// as the kernel image itself is reached by physical address in the entry path.
886#[cfg(target_arch = "aarch64")]
887fn start_spin_table_secondaries() {
888    use crate::boot::firmware_smp;
889    use crate::task::stack::Stack;
890    use crate::task::STACK_PAGES;
891
892    let described = firmware_smp::cpu_count();
893    if described <= 1 {
894        log::info!(
895            "[smp] spin-table: {} CPU(s) described — single-hart",
896            described
897        );
898        return;
899    }
900    // Release addresses are indexed by `/cpus` node order, with the boot core
901    // as node 0, exactly as the logical hart equals the CPU index.
902    let boot_cpu = crate::sched::hart_local::physical_cpu_index();
903    if boot_cpu != Some(0) {
904        log::warn!(
905            "[smp] boot core reports CPU index {:?} (expected 0) — keeping Cellos single-hart",
906            boot_cpu
907        );
908        return;
909    }
910    let attempt = MAX_HARTS.min(described);
911
912    extern "C" {
913        /// Spin-table trampoline in hal/arch/arm/src/aarch64/boot.rs: loads
914        /// the published context into x0, then runs `_secondary_entry`.
915        fn _spin_table_entry();
916    }
917    let entry = _spin_table_entry as *const () as usize;
918    let kernel_root = match *crate::memory::paging::KERNEL_ROOT.lock() {
919        Some(root) => root,
920        None => {
921            log::warn!("[smp] spin-table: kernel page tables not published");
922            return;
923        }
924    };
925
926    for hart_id in 1..attempt {
927        let release_addr = match firmware_smp::cpu_release_addr(hart_id) {
928            Some(addr) => addr as usize,
929            None => {
930                log::info!("[smp] hart {hart_id}: no cpu-release-addr in FDT");
931                continue;
932            }
933        };
934
935        if !crate::sched::hart_local::publish_physical_cpu(hart_id, hart_id) {
936            log::warn!("[smp] hart {hart_id} conflicts with the published topology — not started");
937            continue;
938        }
939
940        let stack = match Stack::new_kernel(STACK_PAGES) {
941            Ok(s) => s,
942            Err(e) => {
943                log::warn!("[smp] hart {hart_id} stack alloc failed: {e:?}");
944                continue;
945            }
946        };
947        let stack_top = stack.top;
948        // Leaked on purpose: a hart's stack lives as long as the hart does.
949        core::mem::forget(stack);
950
951        let context = hal::aarch64::boot::SecondaryContext::for_hart(
952            hart_id as u64,
953            stack_top as u64,
954            kernel_root as u64,
955        );
956        // SAFETY: single writer (hart 0, before the core is released), and no
957        // reader until the release word below is written.
958        let context_addr = unsafe {
959            let slot = core::ptr::addr_of_mut!(SECONDARY_CONTEXT[hart_id]);
960            slot.write(context);
961            slot as usize
962        };
963        // The core reads both with caches off: clean them to the point of
964        // coherency before the release (each clean ends in `dsb sy`).
965        hal::aarch64::cache::clean_data_cache_range(
966            context_addr,
967            core::mem::size_of::<hal::aarch64::boot::SecondaryContext>(),
968        );
969        hal::aarch64::boot::publish_spin_table_context(context_addr as u64);
970
971        if !write_release_word(release_addr, entry as u64) {
972            log::warn!(
973                "[smp] hart {hart_id}: release word {release_addr:#x} is unaligned or cannot be \
974                 mapped — stopping bring-up"
975            );
976            return;
977        }
978        // SAFETY: SEV only wakes cores parked in WFE; the preceding `dsb sy`
979        // orders the release word before it.
980        unsafe {
981            core::arch::asm!("sev", options(nomem, nostack));
982        }
983        log::info!("[smp] hart {hart_id} released via spin table at {release_addr:#x}");
984
985        let timeout_ticks = (hal::aarch64::timer::counter_frequency_hz() / 2).max(1);
986        let start = hal::aarch64::timer::counter_now();
987        while !HART_ONLINE[hart_id].load(Ordering::Acquire) {
988            if hal::aarch64::timer::counter_now().wrapping_sub(start) >= timeout_ticks {
989                // A late core would read the next core's context from the
990                // shared slot, so no further core is released.
991                log::warn!("[smp] hart {hart_id} did not come online in time — stopping bring-up");
992                return;
993            }
994            core::hint::spin_loop();
995        }
996        log::info!("[smp] hart {hart_id} online, parked");
997    }
998}

1000/// Write a spin-table release word and clean it to the point of coherency.
1001///
1002/// The word is firmware memory the kernel's identity map need not cover: on
1003/// the Raspberry Pi it sits in page 0 (0xd8–0xf0, beside the armstub), below
1004/// the kernel image and unmapped as the null guard. Such a page is mapped
1005/// kernel-only for the write and unmapped again, so the guard is back before
1006/// the next core is released. `false` when the word is not 8-byte aligned (the
1007/// binding requires it) or its page cannot be mapped.
1008#[cfg(target_arch = "aarch64")]
1009fn write_release_word(release_addr: usize, entry: u64) -> bool {
1010    use crate::memory::paging::{self, Flags, PAGE_SIZE};
1011
1012    if !release_addr.is_multiple_of(8) {
1013        return false;
1014    }
1015    let page = release_addr & !(PAGE_SIZE - 1);
1016    let transient = paging::virt_to_phys(page).is_none();
1017    if transient {
1018        let mut frames = crate::memory::frame::FRAME_ALLOCATOR.lock();
1019        let Some(allocator) = frames.as_mut() else {
1020            return false;
1021        };
1022        let flags = Flags::from_bits(
1023            Flags::VALID | Flags::READ | Flags::WRITE | Flags::ACCESSED | Flags::DIRTY,
1024        );
1025        if paging::map_page(allocator, page, page, flags).is_err() {
1026            return false;
1027        }
1028    }
1029    // SAFETY: `release_addr` is the firmware's aligned spin-table word for
1030    // this core, identity-mapped (permanently or just above) for the write.
1031    unsafe {
1032        core::ptr::write_volatile(release_addr as *mut u64, entry);
1033    }
1034    hal::aarch64::cache::clean_data_cache_range(release_addr, 8);
1035    if transient {
1036        // Never fails for a mapped page; it ends in a broadcast TLB invalidate.
1037        let _ = paging::unmap_page(page);
1038    }
1039    true
1040}

1042/// Prove every hart that came online actually takes the kernel's IPI.
1043///
1044/// The one cross-hart path a boot log cannot witness on its own: the request is
1045/// delivered by a GIC SGI, the target answers from *its* trap path, and this
1046/// hart's bounded wait ends only when that epoch lands. Without it, a hart that
1047/// came online but is deaf to the IPI looks exactly like a healthy one — and
1048/// every later remote invalidation would silently fall back to the retained-frame
1049/// path. Every remote hart is probed because the SGI target is per core: a
1050/// wrong affinity (another cluster, Aff0 ≥ 16) fails only for the cores it
1051/// names. Bounded and non-fatal: a failed probe is reported, it does not stop
1052/// the boot.
1053#[cfg(all(target_arch = "aarch64", not(cellos_posture = "release")))]
1054pub fn run_ipi_selftest() {
1055    let mut probed = 0;
1056    for remote in online_harts().filter(|hart| *hart != 0) {
1057        probed += 1;
1058        let taken_before = IPI_TAKEN[remote].load(Ordering::Relaxed);
1059        let epoch = request_tlb_flush(remote);
1060        // The architected counter, not the tick: the boot hart runs this with
1061        // IRQs masked, so `system_ticks` would never reach a deadline.
1062        let timeout_ticks = (hal::aarch64::timer::counter_frequency_hz() / 2).max(1);
1063        let start = hal::aarch64::timer::counter_now();
1064        let (mut acknowledged, mut sgi_taken) = (false, false);
1065        while hal::aarch64::timer::counter_now().wrapping_sub(start) < timeout_ticks {
1066            acknowledged = tlb_flush_completed(remote, epoch);
1067            sgi_taken = IPI_TAKEN[remote].load(Ordering::Relaxed) != taken_before;
1068            if acknowledged && sgi_taken {
1069                break;
1070            }
1071            core::hint::spin_loop();
1072        }
1073        match (acknowledged, sgi_taken) {
1074            (true, true) => {
1075                log::info!("[selftest] SMP-IPI: PASS hart={} epoch={}", remote, epoch)
1076            }
1077            // Its timer tick serviced the request; the IPI never arrived.
1078            (true, false) => log::error!(
1079                "[selftest] SMP-IPI: FAIL hart={} epoch={} (acknowledged by tick, no IPI taken)",
1080                remote,
1081                epoch
1082            ),
1083            _ => log::error!(
1084                "[selftest] SMP-IPI: FAIL hart={} epoch={} (no acknowledgement)",
1085                remote,
1086                epoch
1087            ),
1088        }
1089    }
1090    if probed == 0 {
1091        log::info!("[selftest] SMP-IPI: skipped, no remote hart online");
1092    }
1093}

1103// x86 AP bring-up handshake (phase 07b: every MP CPU up to `MAX_HARTS`).
1104//
1105// The BSP starts APs one at a time, in Limine MP order, each through the same
1106// four-step exchange; the next AP is released only after the previous one is
1107// online, so each handshake word below names at most one AP in flight and is
1108// a logical id (0 = none yet), never a flag that a later AP could confuse with
1109// an earlier one's:
1110//   1. AP: own flush proof on the BSP (request_tlb_flush(0) acknowledged), then
1111//      `X86_READY = logical` (or `X86_AP_ROOT_REFUSED = logical` and park).
1112//   2. BSP: flush proof on the AP (request_tlb_flush(logical) acknowledged),
1113//      then `X86_QUALIFIED = logical`.
1114//   3. AP: `HART_ONLINE[logical] = true`.
1115//   4. BSP: one catch-up flush acknowledged by the AP (it was outside every
1116//      shootdown target set until step 3), then the next AP.
1117// A refusal (root operation) or a start failure (stack, topology) ends
1118// bring-up, so the online set is always the contiguous range 0..online and
1119// the RT hart (1) is online whenever any AP is. An AP that was released but
1120// never answers a step within its bounded budget panics the boot
1121// (`x86_bounded_wait`): fail-closed, never a silently missing CPU.
1122#[cfg(target_arch = "x86_64")]
1123static X86_READY: AtomicUsize = AtomicUsize::new(0);
1124/// Highest logical AP the BSP has qualified (step 2); monotone, because APs
1125/// are qualified in increasing logical order.
1126#[cfg(target_arch = "x86_64")]
1127static X86_QUALIFIED: AtomicUsize = AtomicUsize::new(0);
1128#[cfg(target_arch = "x86_64")]
1129static X86_BOOT_COMPLETE: AtomicBool = AtomicBool::new(false);
1130/// Logical id of the AP that answered step 1 with a refusal instead of
1131/// `X86_READY`: it cannot enter the x86 root operation the BSP latched, so it
1132/// parks and never goes online.
1133#[cfg(target_arch = "x86_64")]
1134static X86_AP_ROOT_REFUSED: AtomicUsize = AtomicUsize::new(0);
1135#[cfg(target_arch = "x86_64")]
1136static X86_PREEMPT: [AtomicBool; MAX_HARTS] = [const { AtomicBool::new(false) }; MAX_HARTS];
1137
1138/// Budget of one bounded handshake wait (`x86_bounded_wait`).
1139#[cfg(target_arch = "x86_64")]
1140const X86_WAIT_STEPS: usize = 100_000_000;

1223#[cfg(target_arch = "x86_64")]

1225    // A bounded instruction budget does not depend on timer interrupts being
1226    // delivered to the waiting CPU. Fail closed once a callback was released.

1229            return;
1230        }
1231        core::hint::spin_loop();
1232    }
1233    panic!("[x86-smp] active AP safety handshake timed out");

1266/// Bring every Limine MP CPU online, up to `MAX_HARTS` logical harts.
1267///
1268/// Logical 0 is the BSP; APs take logical 1, 2, … in Limine MP order (1 is
1269/// the RT hart, the rest are normal harts). APs are started one at a time
1270/// through the handshake documented at `X86_READY`; a CPU beyond `MAX_HARTS`
1271/// is logged `cpu N ignored: MAX_HARTS` and left in Limine's park loop, a
1272/// LAPIC ID above 255 stays offline (no interrupt remapping, phase 12), and
1273/// the first AP that refuses root operation or cannot be started ends
1274/// bring-up with the harts already online.
1275///
1276/// `task::quiesce` fails closed above 8 online harts; on x86 it already refuses
1277/// every multi-hart set (no park hook), so bring-up does not cap below
1278/// `MAX_HARTS` for it (phase 07b decision D2).
1279#[cfg(target_arch = "x86_64")]
1280pub fn start_secondaries() {
1281    use crate::boot::limine;
1282    let bsp_id = hal::x86_64::apic::current_id();
1283    #[cfg(feature = "test-hooks")]
1284    x86_handoff_fixture();
1285    let finish_up = || {
1286        X86_BOOT_COMPLETE.store(true, Ordering::Release);
1287        log::info!(
1288            "[x86-smp] topology online=1 processor_ids={} qualified=yes",
1289            bsp_id
1290        );
1291    };
1292    // Logical 0 is the BSP. `CPU_TOPOLOGY` is the kernel's logical -> LAPIC map;
1293    // the HAL keeps its own write-once copy (`apic::register_cpu`) because its
1294    // IPI path cannot read kernel state. Both refuse a LAPIC ID above 255 and a
1295    // conflicting pair, so either refusal keeps the system uniprocessor.
1296    if !CPU_TOPOLOGY.publish(0, bsp_id as usize) || !hal::x86_64::apic::register_cpu(0, bsp_id) {
1297        // UP does not use APIC-directed remote invalidations.
1298        finish_up();
1299        return;
1300    }
1301    let Some(response) = limine::get_mp_response() else {
1302        finish_up();
1303        return;
1304    };
1305    let count = response.cpu_count as usize;
1306    // Response flag bit 0 = Limine enabled x2APIC; it must match this CPU.
1307    let x2apic_flag = response.flags & 1 != 0;
1308    if response.flags & !1 != 0
1309        || x2apic_flag != hal::x86_64::apic::x2apic_active()
1310        || response.bsp_lapic_id != bsp_id
1311        || count == 0
1312        || count > 256
1313        || !limine::mp_record_valid(response.cpus, count)
1314    {
1315        log::warn!("[x86-smp] unsupported MP topology; no AP activated");
1316        finish_up();
1317        return;
1318    }
1319    // Validate every record before any AP is released: one bad record keeps
1320    // the whole machine uniprocessor, as before.
1321    let mut ids = [false; 256];
1322    let mut processors = [0u32; 256];
1323    let mut bsp_seen = false;
1324    let mut secondaries: [*mut limine::LimineMpInfo; 256] = [core::ptr::null_mut(); 256];
1325    let mut secondary_count = 0;
1326    for index in 0..count {
1327        let info = unsafe { *response.cpus.add(index) };
1328        if !limine::mp_record_valid(info, 1) {
1329            finish_up();
1330            return;
1331        }
1332        let cpu = unsafe { &*info };
1333        let id = cpu.lapic_id as usize;
1334        if id > 255 {
1335            // Needs interrupt remapping for IOAPIC/MSI destinations (phase 12).
1336            log::warn!(
1337                "[x86-smp] cpu apic_id={} offline: needs interrupt remapping",
1338                cpu.lapic_id
1339            );
1340            continue;
1341        }
1342        if ids[id]
1343            || processors[..index].contains(&cpu.processor_id)
1344            || (cpu.lapic_id != bsp_id && cpu.goto_address.load(Ordering::Acquire) != 0)
1345        {
1346            log::warn!("[x86-smp] invalid MP CPU record; no AP activated");
1347            finish_up();
1348            return;
1349        }
1350        ids[id] = true;
1351        processors[index] = cpu.processor_id;
1352        if cpu.lapic_id == bsp_id {
1353            bsp_seen = true;
1354        } else {
1355            secondaries[secondary_count] = info;
1356            secondary_count += 1;
1357        }
1358    }
1359    if !bsp_seen || secondary_count == 0 {
1360        finish_up();
1361        return;
1362    }
1363    let kernel_cr3 = hal::domain::kernel_cr3();
1364    assert!(kernel_cr3 != 0, "AP requires permanent shared kernel root");
1365    // The BSP must receive each AP's reciprocal proof even if boot entered this
1366    // routine with interrupts masked. Restore its original state afterwards.
1367    let flags: usize;
1368    unsafe {
1369        core::arch::asm!("pushfq", "pop {}", "sti", out(reg) flags);
1370    }
1371    for &info in &secondaries[..secondary_count] {
1372        let (ap_id, processor_id) = unsafe { ((*info).lapic_id, (*info).processor_id) };
1373        // The next free logical index; the pair is published before the AP is
1374        // released, so the AP's own check in `x86_secondary_main` reads a
1375        // complete mapping.
1376        let logical = CPU_TOPOLOGY.count();
1377        if logical >= MAX_HARTS {
1378            log::warn!(
1379                "[x86-smp] cpu {} ignored: MAX_HARTS (apic_id={})",
1380                processor_id,
1381                ap_id
1382            );
1383            continue;
1384        }
1385        if !x86_start_ap(info, logical, ap_id, kernel_cr3) {
1386            break;
1387        }
1388    }
1389    let online = online_hart_count();
1390    let broadcast_ns = (online > 1).then(x86_broadcast_flush_proof);
1391    #[cfg(feature = "test-hooks")]
1392    if online > 1 {
1393        x86_shootdown_fixture_primary();
1394    }
1395    X86_BOOT_COMPLETE.store(true, Ordering::Release);
1396    if let Some(ack_ns) = broadcast_ns {
1397        log::info!("[x86-smp] transport IPI-invalidate bidirectional=pass");
1398        log::info!(
1399            "[x86-smp] transport IPI-invalidate broadcast targets={} ack_ns={}",
1400            online - 1,
1401            ack_ns
1402        );
1403    }
1404    log::info!(
1405        "[x86-smp] topology online={} processor_ids={} qualified=yes",
1406        online,
1407        X86OnlineApicIds { bsp_id }
1408    );
1409    if flags & (1 << 9) == 0 {
1410        unsafe {
1411            core::arch::asm!("cli", options(nomem, nostack));
1412        }
1413    }
1414}

1416/// Start one AP as `logical` and run the four-step handshake (see
1417/// `X86_READY`). Returns `false`, leaving that AP offline, when it cannot be
1418/// started or refuses root operation; the caller then stops bring-up.
1419#[cfg(target_arch = "x86_64")]
1420fn x86_start_ap(
1421    info: *mut crate::boot::limine::LimineMpInfo,
1422    logical: usize,
1423    ap_id: u32,
1424    kernel_cr3: usize,
1425) -> bool {
1426    let stack = match crate::task::stack::Stack::new_kernel(crate::task::STACK_PAGES) {
1427        Ok(stack) => stack,
1428        Err(_) => {
1429            log::warn!(
1430                "[x86-smp] AP {} stack allocation failed; bring-up stops",
1431                ap_id
1432            );
1433            return false;
1434        }
1435    };
1436    if !CPU_TOPOLOGY.publish(logical, ap_id as usize)
1437        || !hal::x86_64::apic::register_cpu(logical, ap_id)
1438    {
1439        log::warn!(
1440            "[x86-smp] AP {} refused as logical {}; bring-up stops",
1441            ap_id,
1442            logical
1443        );
1444        return false;
1445    }
1446    let stack_top = stack.top;
1447    core::mem::forget(stack);
1448    extern "C" {
1449        fn vi_x86_secondary_entry();
1450    }
1451    let context = &X86_SECONDARY[logical];
1452    // Relaxed: the Release store to `goto_address` below publishes these.
1453    context.stack_top.store(stack_top, Ordering::Relaxed);
1454    context.kernel_cr3.store(kernel_cr3, Ordering::Relaxed);
1455    context.logical.store(logical, Ordering::Relaxed);
1456    context.lapic_id.store(ap_id as usize, Ordering::Relaxed);
1457    unsafe {
1458        (*info).extra_argument = core::ptr::from_ref(context) as u64;
1459        (*info).goto_address.store(
1460            vi_x86_secondary_entry as *const () as usize,
1461            Ordering::Release,
1462        );
1463    }
1464    // Step 1 answered by this AP (no earlier AP can still write these words:
1465    // it is online or parked before this one is released).
1466    x86_bounded_wait(|| {
1467        X86_READY.load(Ordering::Acquire) == logical
1468            || X86_AP_ROOT_REFUSED.load(Ordering::Acquire) == logical
1469    });
1470    if X86_AP_ROOT_REFUSED.load(Ordering::Acquire) == logical {
1471        log::warn!(
1472            "[x86-smp] AP {} refused x86 root operation; it stays offline",
1473            ap_id
1474        );
1475        return false;
1476    }
1477    // Step 2: this AP answers the BSP's IPI-driven invalidation.
1478    let epoch = request_tlb_flush(logical);
1479    x86_bounded_wait(|| tlb_flush_completed(logical, epoch));
1480    X86_QUALIFIED.store(logical, Ordering::Release);
1481    // Step 3.
1482    x86_bounded_wait(|| HART_ONLINE[logical].load(Ordering::Acquire));
1483    // Step 4: catch-up invalidation. Until `HART_ONLINE[logical]` was visible,
1484    // every shootdown (another AP's, or the BSP's own) excluded this AP from its
1485    // target set, so a translation changed between step 2 and step 3 may still
1486    // be cached here. One more acknowledged flush, requested only after the
1487    // BSP observed the AP online, closes that window: every later shootdown
1488    // targets it.
1489    let epoch = request_tlb_flush(logical);
1490    x86_bounded_wait(|| tlb_flush_completed(logical, epoch));
1491    true
1492}

1494/// One broadcast invalidation to every online AP at once, the shape every
1495/// later shootdown takes (`tlb_shootdown::await_remote_invalidation`).
1496/// Returns its acknowledgement latency in clock nanoseconds (0 without a
1497/// clocksource). The APs still spin in their boot-complete wait with IF set.
1498#[cfg(target_arch = "x86_64")]
1499fn x86_broadcast_flush_proof() -> usize {
1500    let started = hal::x86_64::clock::now_ns();
1501    let requested = request_tlb_flush_set(online_harts().filter(|hart| *hart != 0));
1502    x86_bounded_wait(|| tlb_flush_set_outstanding(&requested).is_none());
1503    hal::x86_64::clock::now_ns().saturating_sub(started) as usize
1504}

1506#[cfg(target_arch = "x86_64")]
1507extern "C" fn x86_secondary_main(logical: usize, lapic_id: usize, kernel_cr3: usize) -> ! {
1508    // Any AP slot below MAX_HARTS is admissible, but only on the CPU whose LAPIC
1509    // ID the BSP published for it before releasing this one.
1510    assert!(
1511        logical != 0 && logical < MAX_HARTS && logical_to_physical(logical) == Some(lapic_id),
1512        "[x86-smp] AP logical {} lapic {} is not the published topology",
1513        logical,
1514        lapic_id
1515    );
1516    if !unsafe { hal::x86_64::init_secondary_cpu(logical, lapic_id as u32, kernel_cr3) } {
1517        panic!("[x86-smp] active AP feature/isolation initialization failed");
1518    }
1519    crate::sched::hart_local::install(logical);
1520    // HypervisorCap is system-wide once the BSP latched root operation, so this
1521    // CPU must enter the same root operation (own HSAVE/VMXON frame) before it
1522    // may go online. A refused AP parks with IF clear and is never published.
1523    if crate::cpu_features::has_x86_virt() {
1524        if let Err(reason) = crate::cpu_features::enter_x86_root_on_current_cpu() {
1525            log::warn!(
1526                "[x86-smp] AP {} x86 root operation refused: {}",
1527                lapic_id,
1528                reason
1529            );
1530            X86_AP_ROOT_REFUSED.store(logical, Ordering::Release);
1531            loop {
1532                unsafe {
1533                    core::arch::asm!("cli", "hlt", options(nomem, nostack));
1534                }
1535            }
1536        }
1537    }
1538    // Establish this CPU's own idle activation before enabling delivery.
1539    unsafe {
1540        core::arch::asm!("sti", options(nomem, nostack));
1541    }
1542    // Per-AP clock state: `init_secondary_cpu` ran the TSC skew check, armed
1543    // this CPU's LAPIC/TSC-deadline tick and, when the BSP uses kvmclock,
1544    // registered this vCPU's own pvclock slot (the MSR is per vCPU).
1545    log::info!(
1546        "[x86-smp] AP logical={} apic_id={} pvclock={}",
1547        logical,
1548        lapic_id,
1549        if hal::x86_64::clock::kvmclock::registered_area().is_some() {
1550            "own"
1551        } else {
1552            "none"
1553        }
1554    );
1555    // Step 1 (see `X86_READY`).
1556    let epoch = request_tlb_flush(0);
1557    x86_bounded_wait(|| tlb_flush_completed(0, epoch));
1558    X86_READY.store(logical, Ordering::Release);
1559    x86_bounded_wait(|| X86_QUALIFIED.load(Ordering::Acquire) >= logical);
1560    // Step 3.
1561    HART_ONLINE[logical].store(true, Ordering::Release);
1562    // Later APs are brought up while this one waits here. Each later AP costs
1563    // the BSP at most four bounded waits, plus the closing broadcast and
1564    // shootdown fixture, so five budgets per hart slot bound this wait.
1565    x86_bounded_wait_steps(X86_WAIT_STEPS.saturating_mul(5 * MAX_HARTS), || {
1566        // The shootdown fixture pairs the BSP with the RT hart only; any other
1567        // AP just answers its IPIs from here.
1568        #[cfg(feature = "test-hooks")]
1569        if logical == HART_RT {
1570            x86_shootdown_fixture_secondary();
1571        }
1572        X86_BOOT_COMPLETE.load(Ordering::Acquire)
1573    });
1574    loop {
1575        crate::task::yield_cpu();
1576        unsafe {
1577            core::arch::asm!("sti", "hlt", options(nomem, nostack));
1578        }
1579    }
1580}

1582// This behavioral fixture caches a genuinely GLOBAL shared translation on the
1583// AP, changes its binding on the BSP, then forces the AP to contend on the same
1584// page-table lock with IF clear. Success requires lock-spin local invalidation;
1585// an ISR-only acknowledgement deadlocks or retains the old binding. A second
1586// round rebinds the page while the AP, IF still clear, only takes that lock
1587// *uncontended* — a long masked kernel stretch such as a spawn reading its
1588// image. The BSP's shootdown must still be acknowledged before its probe
1589// expires: every acquisition, not only a contended one, is a service point.
1590#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
1591const X86_PROBE_VA: usize = 0x0600_0000;
1592#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
1593static X86_PROBE_STAGE: AtomicUsize = AtomicUsize::new(0);
1594#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
1595static X86_PROBE_BEFORE: AtomicUsize = AtomicUsize::new(0);
1596#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
1597static X86_PROBE_AFTER: AtomicUsize = AtomicUsize::new(0);
1598
1599#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
1600fn x86_shootdown_fixture_secondary() {
1601    if X86_PROBE_STAGE.load(Ordering::Acquire) != 1 {
1602        return;
1603    }
1604    X86_PROBE_BEFORE.store(
1605        unsafe { core::ptr::read_volatile(X86_PROBE_VA as *const usize) },
1606        Ordering::Release,
1607    );
1608    X86_PROBE_STAGE.store(2, Ordering::Release);
1609    x86_bounded_wait(|| X86_PROBE_STAGE.load(Ordering::Acquire) == 3);
1610    unsafe {
1611        core::arch::asm!("cli", options(nomem, nostack));
1612    }

1854/// Entry point for secondary harts, called from `_secondary_entry` asm.
1855///
1856/// a0 = hart_id (set by OpenSBI per SBI HSM §9.1.1).
1857///
1858/// Installs the trap vector, enables the timer, runs the per-hart scheduler loop.
1859#[no_mangle]
1860pub extern "C" fn smp_hart_entry(physical_hart: usize) -> ! {
1861    #[cfg(target_arch = "riscv64")]
1862    let hart_id = physical_to_logical(physical_hart)
1863        .unwrap_or_else(|| panic!("unmapped RV64 physical hart {}", physical_hart));
1864    #[cfg(not(target_arch = "riscv64"))]
1865    let hart_id = physical_hart;
1866
1867    #[cfg(target_arch = "riscv64")]
1868    {
1869        let root = crate::memory::paging::KERNEL_ROOT
1870            .lock()
1871            .expect("RV64 secondary started before kernel paging root");
1872        // SAFETY: the boot hart published a complete shared root before HSM
1873        // startup; it maps this entry code, stack, and all kernel globals.
1874        unsafe {
1875            crate::memory::paging::activate_paging(root);
1876            core::arch::asm!(
1877                "csrs sstatus, {sum}",
1878                sum = in(reg) 0x40000usize,
1879                options(nostack)
1880            );
1881        }
1882    }
1883    // Install the trap vector (each hart has its own stvec CSR).
1884    // `hal::ARCH.init()` sets stvec + enables SSIE.
1885    #[cfg(target_arch = "riscv64")]
1886    {
1887        crate::sched::hart_local::install(hart_id);
1888        use hal::Arch;
1889        hal::ARCH.init();
1890        // ARCH.init installs the bootstrap-safe default vector. Restore this
1891        // secondary's logical vector before any interrupt is enabled.
1892        hal::trap::init_for_hart(hart_id);
1893    }
1894
1895    // Enable S-mode timer interrupt and arm the first tick on this hart.
1896    // Each hart has its own mtimecmp register via SBI; arming here starts
1897    // the 10ms preemption slice for this hart.
1898    #[cfg(target_arch = "riscv64")]
1899    {
1900        // SAFETY: csrs on sie is always legal from S-mode (RISC-V priv spec §4.1.3).
1901        unsafe {
1902            core::arch::asm!("csrs sie, {stie}", stie = in(reg) 0x20usize);
1903        }
1904        let next = hal::common::timer::read_mtime() + hal::common::timer::TICKS_PER_10MS;
1905        hal::common::sbi::set_timer(next);
1906    }
1907
1908    // `ARCH.init()` enables SSIE in `sie`, but it deliberately leaves the
1909    // per-hart global SIE bit clear. A secondary otherwise wakes from WFI with
1910    // a pending IPI but never takes the SSIP trap that enters `yield_cpu()`;
1911    // explicitly enable delivery before advertising this hart as schedulable.
1912    #[cfg(target_arch = "riscv64")]
1913    {
1914        use hal::Arch;
1915        hal::ARCH.enable_interrupts();
1916        if !hal::ARCH.interrupts_enabled() {
1917            panic!(
1918                "[smp] hart {} could not enable supervisor interrupts",
1919                hart_id
1920            );
1921        }
1922    }
1923
1924    // Signal hart 0's bounded wait only after this hart can actually take the
1925    // dispatch IPI and run its local scheduler.
1926    if hart_id < MAX_HARTS {
1927        log::info!(
1928            "[smp] physical {} -> logical {} trap-ready, interrupts-enabled",
1929            physical_hart,
1930            hart_id
1931        );
1932        #[cfg(feature = "test-hooks")]
1933        log::info!(
1934            "[selftest] SMP-RETIREMENT: stage=hart{}-interrupts-enabled",
1935            hart_id
1936        );
1937        HART_ONLINE[hart_id].store(true, Ordering::Release);
1938    }
1939
1940    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
1941    crate::memory::tlb_shootdown_selftest::run_secondary(hart_id);
1942    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
1943    crate::cell::provenance_selftest::run_secondary(hart_id);
1944
1945    // Per-hart scheduler loop.  The timer ISR (vi_timer_tick) calls yield_cpu()
1946    // which runs pick_next for THIS hart (work-stealing from hart 0 if idle).
1947    // Between ticks we sit in WFI to save power.  Interrupts are enabled on
1948    // entry (ARCH.init() sets sstatus.SIE=1), so WFI fires on the timer ISR.
1949    loop {
1950        #[cfg(feature = "test-hooks")]
1951        crate::loader::atomic_publication_tests::observe_schedule_attempt();
1952        // SAFETY: wfi suspends until the next interrupt; state is unchanged.
1953        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1954        unsafe {
1955            core::arch::asm!("wfi", options(nomem, nostack))
1956        };
1957        core::hint::spin_loop();
1958    }
1959}
