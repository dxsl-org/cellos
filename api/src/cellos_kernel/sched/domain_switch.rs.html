1//! Scheduler-owned root decision. Compiled for every architecture that can
2//! carry a private root; the root encoding is the architecture's
3//! (`crate::arch::sched::root_for_table`).
4use crate::arch::sched::{IrqSaved, Root, SwitchPlan};
5use crate::hal::arch::Context;
6use crate::memory::address_space::AddressSpace;
7use crate::sched::hart_local;
8use crate::task::tcb::{Task, TaskAddressSpace};
9use alloc::sync::Arc;
10
11#[derive(Clone)]
12pub(crate) struct DomainRef(Arc<AddressSpace>);
13
14impl DomainRef {
15    /// No liveness re-check here: the execution pin acquired in pick_next_local
16    /// makes a later retire() legal and drain-safe, so a pinned task's plan
17    /// MUST still derive Activate instead of diverting to the safe root under
18    /// KERNEL_ROOT with none of its mappings.
19    fn from_task(task: &Task) -> Option<Self> {
20        match &task.address_space {
21            TaskAddressSpace::Sas => None,
22            TaskAddressSpace::Domain(space) => Some(Self(Arc::clone(space))),
23        }
24    }
25
26    pub(crate) fn tuple(&self) -> (u64, u64) {
27        (self.0.identity().raw(), self.0.generation())
28    }
29
30    #[allow(dead_code)]
31    pub(crate) fn from_address_space(space: &Arc<AddressSpace>) -> Self {
32        Self(Arc::clone(space))
33    }
34
35    #[allow(dead_code)]
36    pub(crate) fn address_space(&self) -> &Arc<AddressSpace> {
37        &self.0
38    }
39}

41/// A root decision made under `SCHEDULER`, carried to the raw-switch boundary.
42/// No assembly code is allowed to rediscover task or root ownership.
43///
44/// The decision's root side effects (publishing the domain identity, staging
45/// the safe-root acknowledge) run only in [`RootDecision::switch`], on the way
46/// into the switch that programs the root: nothing can record a domain it did
47/// not program. Fixtures inspect a decision through [`RootDecision::preview`].
48pub(crate) struct RootDecision {
49    pub outgoing: *mut Context,
50    pub incoming: *const Context,
51    transition: DomainTransition,
52}
53
54/// The root the switch programs for `space`.
55pub(crate) fn root_of(space: &AddressSpace) -> Option<Root> {
56    crate::arch::sched::root_for_table(space.root_ppn() << 12, space.asid())
57}
58
59/// The kernel root, ASID 0 (untagged).
60fn kernel_root() -> Option<Root> {
61    let root = crate::memory::paging::KERNEL_ROOT
62        .lock()
63        .expect("native domain requires SAS root");
64    crate::arch::sched::root_for_table(root, 0)
65}

67#[derive(Clone)]
68enum DomainTransition {
69    SasToSas,
70    Activate(DomainRef),
71    SameDomain(DomainRef),
72    ToSafeRoot,
73}
74
75impl DomainTransition {
76    /// The transition from the hart's published domain `current` to `domain`.
77    fn classify(current: (u64, u64), domain: Option<DomainRef>) -> Self {
78        match domain {
79            Some(domain) if current == domain.tuple() => Self::SameDomain(domain),
80            Some(domain) => Self::Activate(domain),
81            None if current.0 == 0 => Self::SasToSas,
82            None => Self::ToSafeRoot,
83        }
84    }
85}
86
87impl RootDecision {
88    pub(crate) fn new(
89        outgoing: *mut Context,
90        incoming: *const Context,
91        task: Option<&Task>,
92    ) -> Option<Self> {
93        // Test posture: a Context the scheduler never primed (no kernel stack,
94        // sp 0) must not reach the switch, which would load sp = 0 and surface
95        // two faults later as a misleading kernel-stack overflow.
96        #[cfg(feature = "test-hooks")]
97        if !incoming.is_null() {
98            // SAFETY: a non-null incoming pointer names the selected task's
99            // Context, stable under `SCHEDULER`.
100            let sp = unsafe { (*incoming).sp };
101            assert!(
102                sp != 0 && task.is_none_or(|task| task.kernel_stack.is_some()),
103                "[sched] dispatch of an unprimed Context: tid={} hart={} sp={:#x} kernel_stack={}",
104                task.map_or(0, |task| task.id),
105                hart_local::current_hart_id(),
106                sp,
107                task.is_some_and(|task| task.kernel_stack.is_some()),
108            );
109        }
110        // No liveness gate here: the execution pin was already acquired in
111        // pick_next_local's filter, before the task was dequeued or attributed,
112        // so a plan that reaches this point MUST proceed — returning `None`
113        // would strand a selected task. A retirement flipping the pinned root
114        // Dying after acquisition is legal and drain-safe.
115        let domain = task.and_then(DomainRef::from_task);
116        let transition = DomainTransition::classify(hart_local::current_domain(), domain);
117        match &transition {
118            DomainTransition::Activate(domain) => {
119                hart_local::advance_execution_pin(Some(Arc::clone(&domain.0)));
120            }
121            // Leaving the pinned private root: hand its release to the
122            // safe-root completion hook alongside outgoing attribution.
123            DomainTransition::ToSafeRoot => hart_local::advance_execution_pin(None),
124            DomainTransition::SasToSas | DomainTransition::SameDomain(_) => {}
125        }
126        Some(Self {
127            outgoing,
128            incoming,
129            transition,
130        })
131    }
132
133    /// Run the root side effects and return the root the switch must program
134    /// after saving `outgoing`. `None` is the explicit no-write path, and it
135    /// means exactly one thing: SAS to SAS, where the kernel root is already
136    /// live.
137    ///
138    /// Consumes the decision so its root reference is released before the raw
139    /// switch. The outgoing task's stack frame never unwinds when that task has
140    /// exited, so a decision still alive across the switch would leak one
141    /// reference to the incoming root and keep that domain (its tables and
142    /// ledger) alive forever. The hart's execution pin, not the decision, keeps
143    /// the programmed root alive past this point.
144    fn commit(self) -> Option<Root> {
145        match self.transition {
146            DomainTransition::Activate(domain) => {
147                let (id, generation) = domain.tuple();
148                let root = root_of(&domain.0);
149                // The execution pin was acquired at selection time under
150                // scheduler-stable state; programming the dead root's SATP here
151                // is exactly what begin_execution's recheck prevents.
152                hart_local::set_current_domain(id, generation);
153                crate::hal::domain::observe_switch_activation();
154                #[cfg(feature = "bench-counters")]
155                crate::bench_counters::root_programming();
156                #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
157                log::info!(
158                    "S22-RV64-SWITCH: PASS harts={}",
159                    crate::sched::smp::online_hart_count()
160                );
161                root
162            }
163            DomainTransition::SameDomain(domain) => {
164                // Same-domain resume must still program the root. Trap entry
165                // installs the kernel root, so "same domain" no longer implies
166                // "the root is already live"; skipping the write would resume a
167                // Cell under the kernel root with the kernel's mappings visible
168                // to its S-mode code — a silent isolation failure.
169                crate::hal::domain::observe_switch_activation();
170                #[cfg(feature = "bench-counters")]
171                crate::bench_counters::root_programming();
172                root_of(&domain.0)
173            }
174            DomainTransition::ToSafeRoot => {
175                hart_local::mark_safe_root_pending();
176                crate::hal::domain::observe_switch_activation();
177                #[cfg(feature = "bench-counters")]
178                crate::bench_counters::root_programming();
179                kernel_root()
180            }
181            DomainTransition::SasToSas => None,
182        }
183    }
184
185    /// Switch from `out` to `inc` (the selection's contexts, with the hart's
186    /// boot context in place of a null one), programming this decision's root
187    /// between the outgoing save and the incoming load.
188    ///
189    /// Returns this activation's interrupt token once it resumes.
190    ///
191    /// # Safety
192    /// `out` and `inc` must be valid context slots for this hart, and `irq`
193    /// must be the token captured before selection on this hart.
194    pub(crate) unsafe fn switch(
195        self,
196        out: *mut Context,
197        inc: *const Context,
198        irq: IrqSaved,
199    ) -> IrqSaved {
200        let root = self.commit();
201        #[cfg(feature = "test-hooks")]
202        rootless::note(root.is_none());
203        // The root is programmed *inside* the switch, after the outgoing
204        // context is saved and before the incoming stack is adopted.
205        // Programming it here instead would leave the incoming root live while
206        // the outgoing context is still being saved on a stack the new root
207        // does not map.
208        // SAFETY: the caller's contract; the root comes from a decision made
209        // under `SCHEDULER` with the execution pin held.
210        unsafe {
211            crate::arch::sched::switch(SwitchPlan {
212                out,
213                inc,
214                root,
215                irq,
216            })
217        }
218    }
219
220    /// What a switch for `task` would program on a hart whose published domain
221    /// is `current`, with none of the decision's side effects: no execution
222    /// pin, no published identity.
223    #[cfg(feature = "test-hooks")]
224    pub(crate) fn preview(current: (u64, u64), task: Option<&Task>) -> Preview {
225        let transition = DomainTransition::classify(current, task.and_then(DomainRef::from_task));
226        match transition {
227            DomainTransition::Activate(domain) => Preview {
228                root: root_of(&domain.0),
229                publishes: Some(domain.tuple()),
230            },
231            DomainTransition::SameDomain(domain) => Preview {
232                root: root_of(&domain.0),
233                publishes: None,
234            },
235            DomainTransition::ToSafeRoot => Preview {
236                root: kernel_root(),
237                publishes: None,
238            },
239            DomainTransition::SasToSas => Preview {
240                root: None,
241                publishes: None,
242            },
243        }
244    }
245}

247/// A side-effect-free view of a root decision ([`RootDecision::preview`]).
248#[cfg(feature = "test-hooks")]
249pub(crate) struct Preview {
250    /// The root the switch would program; `None` is the no-write path.
251    pub root: Option<Root>,
252    /// The domain identity the switch would publish (Activate only).
253    pub publishes: Option<(u64, u64)>,
254}

256/// Test posture: a switch that programmed no root must leave no domain
257/// recorded on its hart. [`RootDecision::switch`] notes a rootless switch; the
258/// incoming-side completion hook checks it.
259#[cfg(feature = "test-hooks")]
260mod rootless {
261    use crate::sched::hart_local;
262    use core::sync::atomic::{AtomicBool, Ordering};
263
264    static PENDING: [AtomicBool; crate::sched::smp::MAX_HARTS] =
265        [const { AtomicBool::new(false) }; crate::sched::smp::MAX_HARTS];
266
267    pub(super) fn note(rootless: bool) {
268        if let Some(slot) = PENDING.get(hart_local::current_hart_id()) {
269            slot.store(rootless, Ordering::Release);
270        }
271    }
272
273    /// Called by `vi_context_switch_complete` before the safe-root acknowledge.
274    pub(crate) fn check_completion(hart: usize) {
275        if PENDING
276            .get(hart)
277            .is_some_and(|slot| slot.swap(false, Ordering::AcqRel))
278        {
279            let (id, generation) = hart_local::current_domain();
280            assert!(
281                id == 0,
282                "[sched] hart {} completed a switch that programmed no root with domain ({}, {}) recorded",
283                hart,
284                id,
285                generation,
286            );
287        }
288    }
289}
