10/// Parse HPET table for event timer block MMIO address.
11///
12/// HPET table layout (after 36-byte SDT header):
13///   offset 36: event_timer_block_id (u32) — hardware rev, comparators, etc.
14///   offset 40: base_address (GAS, 12 bytes)
15///       GAS[0]:  address_space_id (u8) — 0 = MMIO
16///       GAS[4..12]: address (u64)
17///   offset 52: hpet_number (u8)
18///   offset 53: minimum_tick (u16)
19///   offset 55: page_protection (u8)
20pub(super) fn parse_hpet(virt: usize, length: usize, info: &mut AcpiInfo) {
21    // GAS starts at offset 40; address field is at GAS offset 4 → table offset 44.
22    if length < 56 {
23        log::warn!("[acpi] HPET table too short ({} bytes)", length);
24        return;
25    }
26
27    // GAS address_space_id at offset 40.
28    // SAFETY: offset 40 is within the validated HPET body.
29    let addr_space = unsafe { core::ptr::read_volatile((virt + 40) as *const u8) };
30    if addr_space != 0 {
31        // Non-zero address space means I/O ports or PCI config — not simple MMIO.
32        log::warn!(
33            "[acpi] HPET GAS address_space_id={} (not MMIO) — timer gate closed",
34            addr_space
35        );
36        return;
37    }
38
39    // GAS address at GAS offset +4 → table offset 44.
40    // SAFETY: offset 44 is within the validated HPET body.
41    let hpet_addr = unsafe { core::ptr::read_unaligned((virt + 44) as *const u64) };
42    if hpet_addr != 0 {
43        info.hpet_base = hpet_addr;
44        log::info!("[acpi] HPET: event timer block = {:#x}", hpet_addr);
45    } else {
46        log::warn!("[acpi] HPET: GAS address is 0 — timer gate closed");
47    }
48}
