12/// Parse MCFG (Memory Mapped Configuration Space) table.
13///
14/// Selects the first valid allocation for segment 0. MCFG stores a segment's
15/// bus-0-relative ECAM base; the admitted window starts at `base + bus_start×1 MiB`.
16/// MCFG body layout (after 36-byte SDT header):
17///   - 8 bytes reserved
18///   - then N × 16-byte allocation entries:
19///     base_address (u64), segment (u16), bus_start (u8), bus_end (u8), _reserved (u32)
20pub(super) fn parse_mcfg(virt: usize, length: usize, info: &mut AcpiInfo, topo: &mut AcpiTopology) {
21    // First allocation entry starts at offset 44 (36-byte header + 8-byte reserved).
22    if length < 44 + 16 {
23        log::warn!(
24            "[acpi] MCFG too short ({} bytes) for any allocation entry",
25            length
26        );
27        return;
28    }
29
30    let entry_count = (length - 44) / 16;
31    // SAFETY: the caller validated and mapped `[virt, virt + length)`.
32    let table = unsafe { TableBytes::new(virt, length) };
33    for index in 0..entry_count {
34        let at = 44 + index * 16;
35        topo.mcfg.push(McfgAlloc {
36            base: table.u64(at).unwrap_or(0),
37            segment: table.u16(at + 8).unwrap_or(0),
38            bus_start: table.u8(at + 10).unwrap_or(0),
39            bus_end: table.u8(at + 11).unwrap_or(0),
40        });
41    }
42    for index in 0..entry_count {
43        let entry = virt + 44 + index * 16;
44        // SAFETY: every field is within this validated 16-byte allocation entry.
45        let base_addr = unsafe { core::ptr::read_unaligned(entry as *const u64) };
46        let segment = unsafe { core::ptr::read_unaligned((entry + 8) as *const u16) };
47        let bus_start = unsafe { core::ptr::read_volatile((entry + 10) as *const u8) };
48        let bus_end = unsafe { core::ptr::read_volatile((entry + 11) as *const u8) };
49        let window_len = crate::dev::pcie_ecam::ecam_window_size(bus_start, bus_end);
50        let window_base = base_addr.checked_add((bus_start as u64) << 20);
51        if segment == 0
52            && base_addr != 0
53            && base_addr & 0xF_FFFF == 0
54            && window_base
55                .zip(window_len)
56                .and_then(|(base, len)| base.checked_add(len as u64))
57                .is_some()
58        {
59            let window_base = window_base.expect("validated MCFG window base");
60            info.ecam_base = window_base;
61            info.ecam_bus_start = bus_start;
62            info.ecam_bus_end = bus_end;
63            log::info!(
64                "[acpi] MCFG: segment 0 bus {}-{} ECAM window base = {:#x}",
65                bus_start,
66                bus_end,
67                window_base
68            );
69            return;
70        }
71    }
72    log::warn!("[acpi] MCFG has no valid segment 0 allocation — PCIe gate closed");
73}
