1//! Resource Registry — exclusive MMIO region grants for Driver Cells.
2//!
3//! A Driver Cell calls `sys_request_mmio(base, len)` (added in Phase 03).
4//! The kernel checks here before handing an `MmioRegion` to the Cell:
5//!
6//! 1. **Allowlist**: the requested range must fall within a known-safe
7//!    device window for the active target. Unknown ranges are rejected
8//!    so a misbehaving Cell cannot map arbitrary kernel memory as MMIO.
9//!
10//! 2. **Exclusive ownership**: at most one Cell may hold a given MMIO range.
11//!    A second `request_mmio` for an overlapping range returns `AlreadyExists`.
12//!
13//! 3. **Release-on-exit**: `release_for(cell_id)` frees all ranges owned by
14//!    a Cell.  Call this from every Cell-exit path alongside
15//!    `cell_quota::deregister`.
16//!
17//! # Current scope
18//! Allowlist is hardcoded per target (DTB discovery remains deferred).
19//!
20//! | Target | Device | Base | Size |
21//! |--------|--------|------|------|
22//! | QEMU ARM virt (aarch64) | PL011 UART0 | 0x0900_0000 | 0x1000 |
23//! | QEMU ARM virt (aarch64) | PL061 GPIO  | 0x0903_0000 | 0x1000 |
24//! | Raspberry Pi 3 | BCM2837 GPIO/BSC1/SPI0/AUX | exact 4-KiB windows | 0x1000 |
25//! | QEMU RISC-V virt (riscv64) | (none yet — kernel serial owns UART) | — | — |
26

35/// UART serial device window. Set in a cell's `mmio_devices` when its manifest
36/// declares `uart = true`.
37pub const DEV_UART: u8 = 1 << 0;
38/// GPIO controller window. Set when the manifest declares `gpio = true`.
39pub const DEV_GPIO: u8 = 1 << 1;
40/// PCIe device BAR window. Set on tasks with `PcieDriverCap`; PCIe authority is
41/// path-gated and intentionally remains separate from hardware manifest flags.
42pub const DEV_PCIE: u8 = 1 << 2;
43/// CAN bus controller window (v2 manifest — freed by the u16 flags widening).
44/// Set when the manifest declares `can = true`.
45pub const DEV_CAN: u8 = 1 << 3;
46/// ADC controller window (v2 manifest). Set when the manifest declares `adc = true`.
47pub const DEV_ADC: u8 = 1 << 4;
48/// I2C controller window (v2 manifest). Set when the manifest declares `i2c = true`.
49pub const DEV_I2C: u8 = 1 << 5;
50pub const DEV_SPI: u8 = 1 << 6;
51/// Firmware / display controller window.
52pub const DEV_DISPLAY: u8 = 1 << 7;
53// DEV_USB is intentionally absent: the u8 manifest byte is full (bits 0–7 used).
54// USB host controller authority requires policy v3 with an explicit signed byte.
55// Gate with a test matrix in policy::self_test before implementing.

150/// Maps MMIO base address → (len, owner CellId).
151static REGISTRY: Spinlock<BTreeMap<usize, (usize, CellId)>> = Spinlock::new(BTreeMap::new());
152
153/// Dynamically discovered PCIe BAR windows (base → len).
154/// Populated by `pcie_ecam::init()` after the ECAM scan; consumed by
155/// `request_mmio` when the caller holds `PcieDriverCap` (DEV_PCIE).
156static PCIE_BARS: Spinlock<BTreeMap<usize, usize>> = Spinlock::new(BTreeMap::new());
157
158fn static_range_allowed(
159    allowed: &[(usize, usize, u8)],
160    base: usize,
161    end: usize,
162    allowed_devices: u8,
163) -> bool {
164    allowed.iter().any(|&(window_base, window_len, class)| {
165        window_base
166            .checked_add(window_len)
167            .is_some_and(|window_end| {
168                base >= window_base && end <= window_end && class & allowed_devices != 0
169            })
170    })
171}

177/// Force-release this module's lock during fault teardown.
178///
179/// # Safety
180/// Single-hart kernel; called only from the fault/panic path with interrupts
181/// disabled.  Force-unlocking an already-free Spinlock is a no-op.
182pub unsafe fn force_unlock_locks() {
183    REGISTRY.force_unlock();
184    // SAFETY: same contract as REGISTRY above.
185    unsafe {
186        PCIE_BARS.force_unlock();
187    }
188}

190/// Register a PCIe BAR window discovered during ECAM scan.
191///
192/// Called by `pcie_ecam::init()` for every non-zero BAR on every device.
193/// Driver Cells may subsequently call `sys_request_mmio` on these ranges
194/// if they hold `PcieDriverCap`.
195pub fn register_pcie_bar(base: usize, len: usize) {
196    if !valid_pcie_bar_window(base, len) {
197        log::warn!(
198            "[pcie] rejected invalid BAR window base={:#x} len={:#x}",
199            base,
200            len
201        );
202        return;
203    }
204    PCIE_BARS.lock().insert(base, len);
205}

207/// Current Driver Cells support bounded conventional BARs only.
208///
209/// ReBAR and larger accelerator apertures need a separate policy because they
210/// materially widen a Cell's MMIO authority.
211fn valid_pcie_bar_window(base: usize, len: usize) -> bool {
212    const MAX_SUPPORTED_BAR_LEN: usize = 1 << 30;
213
214    base != 0
215        && len != 0
216        && len <= MAX_SUPPORTED_BAR_LEN
217        && len.is_power_of_two()
218        && base & (len - 1) == 0
219        && base.checked_add(len).is_some()
220}
221
222/// Return `true` if `[base, base+len)` is a known PCIe BAR window.
223///
224/// Used by the `RequestMmio` handler to decide whether to take the PCIe path.
225pub fn is_pcie_bar(base: usize, len: usize) -> bool {
226    if len == 0 {
227        return false;
228    }
229    let guard = PCIE_BARS.lock();
230    guard.get(&base).is_some_and(|&bar_len| len <= bar_len)
231}
232
233fn checked_mmio_end(base: usize, len: usize) -> ViResult<usize> {
234    if len == 0 {
235        return Err(ViError::InvalidInput);
236    }
237    base.checked_add(len).ok_or(ViError::InvalidInput)
238}

240/// Request exclusive MMIO ownership without allowlist validation (Platform Cell only).
241///
242/// Bypasses the per-arch ALLOWED list and the PCIE_BARS table. The overlap check
243/// still runs — two cells cannot share a byte. Used by the PlatformCap bypass path
244/// in `sys_request_mmio` so the Platform Cell can claim the ECAM config-space window
245/// (which is not a device BAR and therefore not in either allowlist).
246pub fn request_mmio_unchecked(cell_id: CellId, base: usize, len: usize) -> ViResult<()> {
247    let end = checked_mmio_end(base, len)?;
248    let mut reg = REGISTRY.lock();
249    for (&eb, &(el, _)) in reg.iter() {
250        let ee = eb + el;
251        if !(end <= eb || base >= ee) {
252            return Err(ViError::AlreadyExists);
253        }
254    }
255    reg.insert(base, (len, cell_id));
256    Ok(())
257}

259/// Request exclusive ownership of `[base, base+len)` for `cell_id`.
260///
261/// Returns:
262/// - `Ok(())` — range is now owned by the caller; construct `MmioRegion` and
263///   hand it to the Cell.
264/// - `Err(PermissionDenied)` — range not in allowlist, or its device class is
265///   not among `allowed_devices` (the cell's declared `mmio_devices`).
266/// - `Err(AlreadyExists)` — range overlaps an already-granted region.
267/// - `Err(InvalidInput)` — zero length or arithmetic overflow in `base + len`.
268pub fn request_mmio(cell_id: CellId, base: usize, len: usize, allowed_devices: u8) -> ViResult<()> {
269    // 1. Allowlist check — the range must fall inside a known device window
270    //    AND that window's device class must be one the cell declared.
271    let end = checked_mmio_end(base, len)?;
272
273    // PCIe path: validate against the dynamic BAR table populated by pcie_ecam.
274    let in_allowlist = if allowed_devices & DEV_PCIE != 0 {
275        let bars = PCIE_BARS.lock();
276        bars.get(&base).is_some_and(|&bar_len| len <= bar_len)
277    } else {
278        // SoC peripheral path: static per-arch allowlist.
279        static_range_allowed(ALLOWED, base, end, allowed_devices)
280    };
281    if !in_allowlist {
282        return Err(ViError::PermissionDenied);
283    }
284
285    // 2. Overlap check — no two cells may share a byte
286    let mut reg = REGISTRY.lock();
287    for (&eb, &(el, _owner)) in reg.iter() {
288        let ee = eb + el;
289        // Ranges overlap when NOT (end ≤ eb OR base ≥ ee)
290        if !(end <= eb || base >= ee) {
291            return Err(ViError::AlreadyExists);
292        }
293    }
294
295    reg.insert(base, (len, cell_id));
296    Ok(())
297}

303/// Power-on self-test of the MMIO allowlist. Non-RPi3 targets have nothing
304/// pinned yet; the test exists so the boot chain can call it unconditionally.
305#[cfg(not(all(target_arch = "aarch64", feature = "board-rpi3")))]
306pub fn self_test() -> bool {
307    true
308}
309
310/// Power-on self-test of the RPi3 production MMIO allowlist.
311///
312/// Pins two properties on the REAL table (`ALLOWED`), not a synthetic copy:
313///
314/// 1. Positive control — a known window (mailbox/DEV_DISPLAY) still authorizes.
315///    Without this, an accidentally emptied table would make every denial
316///    below vacuous.
317/// 2. DWC2 denial — the DWC2 USB window must be rejected for every device
318///    class, full masks included, over the whole aperture AND at its edges.
319///    The entry was previously present mis-tagged `DEV_PCIE`; it may return
320///    only via policy v3 with a signed USB-host authority byte.
321#[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
322pub fn self_test() -> bool {
323    let mmio = hal_soc_bcm27xx::BCM2837.mmio;
324
325    // 1. Positive control: the table is live.
326    let mbox_base = mmio.mailbox_base;
327    let mbox_end = mbox_base + mmio.mailbox_grant_size;
328    if !static_range_allowed(ALLOWED, mbox_base, mbox_end, DEV_DISPLAY) {
329        log::error!("[selftest] mmio-allowlist: FAIL — mailbox window no longer authorized");
330        return false;
331    }
332
333    // 2. DWC2 denial across the full aperture, per class and combined masks.
334    let dwc2_base = mmio.dwc2_base;
335    let dwc2_end = dwc2_base + mmio.dwc2_grant_size;
336    let classes = [
337        DEV_UART,
338        DEV_GPIO,
339        DEV_PCIE,
340        DEV_CAN,
341        DEV_ADC,
342        DEV_I2C,
343        DEV_SPI,
344        DEV_DISPLAY,
345        0xFF,
346    ];
347    for class in classes {
348        if static_range_allowed(ALLOWED, dwc2_base, dwc2_end, class) {
349            log::error!(
350                "[selftest] mmio-allowlist: FAIL — DWC2 window authorized for class {class:#04x}"
351            );
352            return false;
353        }
354    }
355
356    // Edge words: same full class sweep as the whole aperture — a future
357    // sub-range entry tagged with ANY class must fail the boot self-test.
358    let last_word = dwc2_end - 4;
359    for addr in [dwc2_base, last_word] {
360        for class in classes {
361            if static_range_allowed(ALLOWED, addr, addr + 4, class) {
362                log::error!(
363                    "[selftest] mmio-allowlist: FAIL — DWC2 edge word {addr:#x} authorized \
364                     for class {class:#04x}"
365                );
366                return false;
367            }
368        }
369    }
370    true
371}

373/// Release all MMIO regions owned by `cell_id`.
374///
375/// Call this from every Cell-exit path (Exit syscall, ForceExit, fault, watchdog).
376pub fn release_for(cell_id: CellId) {
377    REGISTRY
378        .lock()
379        .retain(|_base, &mut (_len, owner)| owner != cell_id);
380}
381
382/// Return the task ID (TID) of the cell that currently owns the MMIO region
383/// whose base address exactly matches `base`.
384///
385/// Returns `None` if no cell has requested that exact base address.
386/// Used by the GPIO IRQ handler to route interrupts to the current MMIO owner.
387pub fn lookup_mmio_owner(base: usize) -> Option<usize> {
388    REGISTRY
389        .lock()
390        .get(&base)
391        .map(|&(_len, cell_id)| cell_id.0 as usize)
392}
393
394/// Current number of registered regions (diagnostics).
395pub fn region_count() -> usize {
396    REGISTRY.lock().len()
397}

400// PCIe BDF ownership (for sys_grant_dma authorization)
401// ---------------------------------------------------------------------------

403/// Maps PCIe BDF → owning task ID.
404///
405/// Kernel drivers (NIC, NVMe) are not registered here — they bypass `sys_grant_dma`
406/// and call `iommu::map_dma_for_cell(0, bdf, ...)` directly during init.
407/// Only userspace Driver Cells that receive PCIe device ownership via capability
408/// delegation need to register here.
409static BDF_OWNERS: Spinlock<alloc::collections::BTreeMap<u32, usize>> =
410    Spinlock::new(alloc::collections::BTreeMap::new());
411
412/// Register a PCIe BDF as owned by task `tid`.
413///
414/// Called when a Driver Cell is granted ownership of a PCIe device.
415pub fn register_bdf_owner(bdf: u32, tid: usize) {
416    BDF_OWNERS.lock().insert(bdf, tid);
417}
418
419/// Return the task ID that currently owns `bdf`, or `None` if unowned.
420pub fn owner_of_bdf(bdf: u32) -> Option<usize> {
421    BDF_OWNERS.lock().get(&bdf).copied()
422}
423
424/// Release all BDF ownerships held by task `tid` (called on Cell exit).
425pub fn release_bdfs_for(tid: usize) {
426    BDF_OWNERS.lock().retain(|_bdf, &mut owner| owner != tid);
427}

429/// Force-unlock BDF_OWNERS during fault teardown.
430///
431/// # Safety
432/// See `force_unlock_locks` above — same contract.
433pub unsafe fn force_unlock_bdf_locks() {
434    BDF_OWNERS.force_unlock();
435}
