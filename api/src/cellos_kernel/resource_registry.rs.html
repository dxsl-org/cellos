1//! Resource Registry — exclusive MMIO region grants for Driver Cells.
2//!
3//! A Driver Cell calls `sys_request_mmio(base, len)` (added in Phase 03).
4//! The kernel checks here before handing an `MmioRegion` to the Cell:
5//!
6//! 1. **Allowlist**: the requested range must fall within a known-safe
7//!    device window for the current QEMU target.  Unknown ranges are rejected
8//!    so a misbehaving Cell cannot map arbitrary kernel memory as MMIO.
9//!
10//! 2. **Exclusive ownership**: at most one Cell may hold a given MMIO range.
11//!    A second `request_mmio` for an overlapping range returns `AlreadyExists`.
12//!
13//! 3. **Release-on-exit**: `release_for(cell_id)` frees all ranges owned by
14//!    a Cell.  Call this from every Cell-exit path alongside
15//!    `cell_quota::deregister`.
16//!
17//! # v1 scope
18//! Allowlist is hardcoded per QEMU target (DTB discovery deferred to v2).
19//!
20//! | Target | Device | Base | Size |
21//! |--------|--------|------|------|
22//! | QEMU ARM virt (aarch64) | PL011 UART0 | 0x0900_0000 | 0x1000 |
23//! | QEMU ARM virt (aarch64) | PL061 GPIO  | 0x0903_0000 | 0x1000 |
24//! | QEMU RISC-V virt (riscv64) | (none yet — kernel serial owns UART) | — | — |
25

34/// UART serial device window. Set in a cell's `mmio_devices` when its manifest
35/// declares `uart = true`.
36pub const DEV_UART: u8 = 1 << 0;
37/// GPIO controller window. Set when the manifest declares `gpio = true`.
38pub const DEV_GPIO: u8 = 1 << 1;
39/// PCIe device BAR window. Set on tasks with `PcieDriverCap` — not a manifest flag
40/// (all 8 manifest bits are occupied); gated by the ZST cap instead.
41pub const DEV_PCIE: u8 = 1 << 2;
42/// CAN bus controller window (v2 manifest — freed by the u16 flags widening).
43/// Set when the manifest declares `can = true`.
44pub const DEV_CAN: u8 = 1 << 3;
45/// ADC controller window (v2 manifest). Set when the manifest declares `adc = true`.
46pub const DEV_ADC: u8 = 1 << 4;

124// ---------------------------------------------------------------------------
125// Registry state
126// ---------------------------------------------------------------------------
127
128/// Maps MMIO base address → (len, owner CellId).
129static REGISTRY: Spinlock<BTreeMap<usize, (usize, CellId)>> = Spinlock::new(BTreeMap::new());
130
131/// Dynamically discovered PCIe BAR windows (base → len).
132/// Populated by `pcie_ecam::init()` after the ECAM scan; consumed by
133/// `request_mmio` when the caller holds `PcieDriverCap` (DEV_PCIE).
134static PCIE_BARS: Spinlock<BTreeMap<usize, usize>> = Spinlock::new(BTreeMap::new());
135
136// ---------------------------------------------------------------------------
137// Public API
138// ---------------------------------------------------------------------------
139
140/// Force-release this module's lock during fault teardown.
141///
142/// # Safety
143/// Single-hart kernel; called only from the fault/panic path with interrupts
144/// disabled.  Force-unlocking an already-free Spinlock is a no-op.
145pub unsafe fn force_unlock_locks() {
146    REGISTRY.force_unlock();
147    // SAFETY: same contract as REGISTRY above.
148    unsafe {
149        PCIE_BARS.force_unlock();
150    }
151}
152
153/// Register a PCIe BAR window discovered during ECAM scan.
154///
155/// Called by `pcie_ecam::init()` for every non-zero BAR on every device.
156/// Driver Cells may subsequently call `sys_request_mmio` on these ranges
157/// if they hold `PcieDriverCap`.
158pub fn register_pcie_bar(base: usize, len: usize) {
159    if base != 0 && len != 0 {
160        PCIE_BARS.lock().insert(base, len);
161    }
162}
163
164/// Return `true` if `[base, base+len)` is a known PCIe BAR window.
165///
166/// Used by the `RequestMmio` handler to decide whether to take the PCIe path.
167pub fn is_pcie_bar(base: usize, len: usize) -> bool {
168    let guard = PCIE_BARS.lock();
169    guard.get(&base).is_some_and(|&bar_len| len <= bar_len)
170}

172/// Request exclusive MMIO ownership without allowlist validation (Platform Cell only).
173///
174/// Bypasses the per-arch ALLOWED list and the PCIE_BARS table. The overlap check
175/// still runs — two cells cannot share a byte. Used by the PlatformCap bypass path
176/// in `sys_request_mmio` so the Platform Cell can claim the ECAM config-space window
177/// (which is not a device BAR and therefore not in either allowlist).
178pub fn request_mmio_unchecked(cell_id: CellId, base: usize, len: usize) -> ViResult<()> {
179    let end = base.checked_add(len).ok_or(ViError::InvalidInput)?;
180    let mut reg = REGISTRY.lock();
181    for (&eb, &(el, _)) in reg.iter() {
182        let ee = eb + el;
183        if !(end <= eb || base >= ee) {
184            return Err(ViError::AlreadyExists);
185        }
186    }
187    reg.insert(base, (len, cell_id));
188    Ok(())
189}

191/// Request exclusive ownership of `[base, base+len)` for `cell_id`.
192///
193/// Returns:
194/// - `Ok(())` — range is now owned by the caller; construct `MmioRegion` and
195///   hand it to the Cell.
196/// - `Err(PermissionDenied)` — range not in allowlist, or its device class is
197///   not among `allowed_devices` (the cell's declared `mmio_devices`).
198/// - `Err(AlreadyExists)` — range overlaps an already-granted region.
199/// - `Err(InvalidInput)` — arithmetic overflow in `base + len`.
200pub fn request_mmio(cell_id: CellId, base: usize, len: usize, allowed_devices: u8) -> ViResult<()> {
201    // 1. Allowlist check — the range must fall inside a known device window
202    //    AND that window's device class must be one the cell declared.
203    let end = base.checked_add(len).ok_or(ViError::InvalidInput)?;
204
205    // PCIe path: validate against the dynamic BAR table populated by pcie_ecam.
206    let in_allowlist = if allowed_devices & DEV_PCIE != 0 {
207        let bars = PCIE_BARS.lock();
208        bars.get(&base).is_some_and(|&bar_len| len <= bar_len)
209    } else {
210        // GPIO/UART path: static per-arch allowlist.
211        ALLOWED.iter().any(|&(ab, al, class)| {
212            let ae = ab + al;
213            base >= ab && end <= ae && (class & allowed_devices != 0)
214        })
215    };
216    if !in_allowlist {
217        return Err(ViError::PermissionDenied);
218    }
219
220    // 2. Overlap check — no two cells may share a byte
221    let mut reg = REGISTRY.lock();
222    for (&eb, &(el, _owner)) in reg.iter() {
223        let ee = eb + el;
224        // Ranges overlap when NOT (end ≤ eb OR base ≥ ee)
225        if !(end <= eb || base >= ee) {
226            return Err(ViError::AlreadyExists);
227        }
228    }
229
230    reg.insert(base, (len, cell_id));
231    Ok(())
232}

234/// Release all MMIO regions owned by `cell_id`.
235///
236/// Call this from every Cell-exit path (Exit syscall, ForceExit, fault, watchdog).
237pub fn release_for(cell_id: CellId) {
238    REGISTRY
239        .lock()
240        .retain(|_base, &mut (_len, owner)| owner != cell_id);
241}
242
243/// Return the task ID (TID) of the cell that currently owns the MMIO region
244/// whose base address exactly matches `base`.
245///
246/// Returns `None` if no cell has requested that exact base address.
247/// Used by the GPIO IRQ handler to route interrupts to the current MMIO owner.
248pub fn lookup_mmio_owner(base: usize) -> Option<usize> {
249    REGISTRY
250        .lock()
251        .get(&base)
252        .map(|&(_len, cell_id)| cell_id.0 as usize)
253}
254
255/// Current number of registered regions (diagnostics).
256pub fn region_count() -> usize {
257    REGISTRY.lock().len()
258}

260// ---------------------------------------------------------------------------
261// PCIe BDF ownership (for sys_grant_dma authorization)
262// ---------------------------------------------------------------------------
263
264/// Maps PCIe BDF → owning task ID.
265///
266/// Kernel drivers (NIC, NVMe) are not registered here — they bypass `sys_grant_dma`
267/// and call `iommu::map_dma_for_cell(0, bdf, ...)` directly during init.
268/// Only userspace Driver Cells that receive PCIe device ownership via capability
269/// delegation need to register here.
270static BDF_OWNERS: Spinlock<alloc::collections::BTreeMap<u32, usize>> =
271    Spinlock::new(alloc::collections::BTreeMap::new());
272
273/// Register a PCIe BDF as owned by task `tid`.
274///
275/// Called when a Driver Cell is granted ownership of a PCIe device.
276pub fn register_bdf_owner(bdf: u32, tid: usize) {
277    BDF_OWNERS.lock().insert(bdf, tid);
278}
279
280/// Return the task ID that currently owns `bdf`, or `None` if unowned.
281pub fn owner_of_bdf(bdf: u32) -> Option<usize> {
282    BDF_OWNERS.lock().get(&bdf).copied()
283}
284
285/// Release all BDF ownerships held by task `tid` (called on Cell exit).
286pub fn release_bdfs_for(tid: usize) {
287    BDF_OWNERS.lock().retain(|_bdf, &mut owner| owner != tid);
288}

290/// Force-unlock BDF_OWNERS during fault teardown.
291///
292/// # Safety
293/// See `force_unlock_locks` above — same contract.
294pub unsafe fn force_unlock_bdf_locks() {
295    BDF_OWNERS.force_unlock();
