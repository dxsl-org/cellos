1//! Resource Registry — exclusive MMIO region grants for Driver Cells.
2//!
3//! A Driver Cell calls `sys_request_mmio(base, len)` (added in Phase 03).
4//! The kernel checks here before handing an `MmioRegion` to the Cell:
5//!
6//! 1. **Allowlist**: the requested range must fall within a known-safe
7//!    device window for the active target. Unknown ranges are rejected
8//!    so a misbehaving Cell cannot map arbitrary kernel memory as MMIO.
9//!
10//! 2. **Exclusive ownership**: at most one task may hold a given MMIO range.
11//!    A second `request_mmio` for an overlapping range returns `AlreadyExists`.
12//!
13//! 3. **Release-on-exit**: `release_mmio_for(tid)` frees all ranges owned by
14//!    a task. Owners are task ids, the same key as the device records: the claiming
15//!    task holds the window, so retirement releases it per member task. The
16//!    released pages first return to their boot posture (kernel-only) on every
17//!    hart, lowered without the registry lock while the window refuses every
18//!    claim; a window whose lowering is unconfirmed stays owned by nobody until
19//!    a retry the timer poll arms confirms it.
20//!
21//! 4. **Grant is the mapping**: `RequestMmio` sets USER on the whole pages of
22//!    a granted window. A sub-page window (the BCM2837 mailbox, which shares
23//!    its page with the IRQ controller) is never USER-mapped; the kernel
24//!    performs the owner's accesses to its bytes (`mediation`).
25//!
26//! # Current scope
27//! Allowlist is hardcoded per target (DTB discovery remains deferred).
28//!
29//! | Target | Device | Base | Size |
30//! |--------|--------|------|------|
31//! | QEMU ARM virt (aarch64) | PL011 UART0 | 0x0900_0000 | 0x1000 |
32//! | QEMU ARM virt (aarch64) | PL061 GPIO  | 0x0903_0000 | 0x1000 |
33//! | Raspberry Pi 3 | BCM2837 GPIO/BSC1/SPI0/AUX | exact 4-KiB windows | 0x1000 |
34//! | Raspberry Pi 3 | VideoCore mailbox (mediated) | 0x3F00_B880 | 0x40 |
35//! | QEMU RISC-V virt (riscv64) | (none yet — kernel serial owns UART) | — | — |
36

46/// UART serial device window. Set in a cell's `mmio_devices` when its manifest
47/// declares `uart = true`.
48pub const DEV_UART: u8 = 1 << 0;
49/// GPIO controller window. Set when the manifest declares `gpio = true`.
50pub const DEV_GPIO: u8 = 1 << 1;
51/// PCIe device BAR window. Set on tasks with `PcieDriverCap`; PCIe authority is
52/// path-gated and intentionally remains separate from hardware manifest flags.
53pub const DEV_PCIE: u8 = 1 << 2;
54/// CAN bus controller window (v2 manifest — freed by the u16 flags widening).
55/// Set when the manifest declares `can = true`.
56pub const DEV_CAN: u8 = 1 << 3;
57/// ADC controller window (v2 manifest). Set when the manifest declares `adc = true`.
58pub const DEV_ADC: u8 = 1 << 4;
59/// I2C controller window (v2 manifest). Set when the manifest declares `i2c = true`.
60pub const DEV_I2C: u8 = 1 << 5;
61pub const DEV_SPI: u8 = 1 << 6;
62/// Firmware / display controller window.
63pub const DEV_DISPLAY: u8 = 1 << 7;

65/// Class of an owned MMIO window: the `DEV_*` bits, plus window kinds that have
66/// no `DEV_*` bit because they are not manifest-declared devices.
67///
68/// The class is stored with the region when it is granted. Re-deriving it at
69/// revoke time would have to guess: a PCIe BAR is in neither allowlist, and a
70/// granted sub-window is contained in — not equal to — the window it came from.
71pub type MmioClass = u16;
72
73/// The Platform Cell's ECAM config-space window (`request_ecam_mmio`).
74pub const CLASS_ECAM: MmioClass = 1 << 8;
75/// The DWC2 USB host aperture (`request_dwc2_mmio`).
76pub const CLASS_DWC2: MmioClass = 1 << 9;

146/// Test-hooks only: register a window with an explicit class, so a boot
147/// self-test can prove class-selective revocation on boards whose allowlist
148/// has no window of the class it needs.
149#[cfg(feature = "test-hooks")]
150pub(crate) fn test_register_region(tid: usize, base: usize, len: usize, class: MmioClass) -> bool {
151    if checked_mmio_end(base, len).is_err() {
152        return false;
153    }
154    let mut registry = REGISTRY.lock();
155    #[cfg(feature = "p01-trace")]
156    let _semantic = crate::memory::heap::p01_owner::creation(
157        None,
158        crate::memory::heap::p01_owner::DRIVER_SHARED,
159    );
160    registry
161        .insert(
162            base,
163            MmioRegion {
164                len,
165                owner_tid: tid,
166                class,
167            },
168        )
169        .is_none()
170}

180/// Device class of the allowlist window containing `[base, end)`, when the
181/// caller declared that class.
182///
183/// Returns the class rather than a bool because the grant records it: a revoke
184/// must be able to tell which device class a window belongs to without
185/// re-deriving it from a table that may have changed since.
186fn static_range_class(
187    allowed: &[(usize, usize, u8)],
188    base: usize,
189    end: usize,
190    allowed_devices: u8,
191) -> Option<u8> {
192    allowed
193        .iter()
194        .find(|&&(window_base, window_len, class)| {
195            window_base
196                .checked_add(window_len)
197                .is_some_and(|window_end| {
198                    base >= window_base && end <= window_end && class & allowed_devices != 0
199                })
200        })
201        .map(|&(_, _, class)| class)
202}

208/// Current Driver Cells support bounded conventional BARs only.
209///
210/// ReBAR and larger accelerator apertures need a separate policy because they
211/// materially widen a Cell's MMIO authority.
212pub(crate) fn valid_pcie_bar_window(base: usize, len: usize) -> bool {
213    const MAX_SUPPORTED_BAR_LEN: usize = 1 << 30;
214
215    base != 0
216        && len != 0
217        && len <= MAX_SUPPORTED_BAR_LEN
218        && len.is_power_of_two()
219        && base & (len - 1) == 0
220        && base.checked_add(len).is_some()
221}
222
223fn checked_mmio_end(base: usize, len: usize) -> ViResult<usize> {
224    if len == 0 {
225        return Err(ViError::InvalidInput);
226    }
227    base.checked_add(len).ok_or(ViError::InvalidInput)
228}

230/// Record an ECAM claim with only the overlap check (no window policy).
231///
232/// The `RequestMmio` syscall must go through [`request_ecam_mmio`]; this raw
233/// form exists for kernel-internal fixtures that register a window directly.
234pub fn request_mmio_unchecked(tid: usize, base: usize, len: usize) -> ViResult<()> {
235    let end = checked_mmio_end(base, len)?;
236    let mut reg = REGISTRY.lock();
237    if reg
238        .iter()
239        .any(|(&window_base, region)| claim_collides(window_base, region, base, end))
240    {
241        return Err(ViError::AlreadyExists);
242    }
243    reg.insert(
244        base,
245        MmioRegion {
246            len,
247            owner_tid: tid,
248            class: CLASS_ECAM,
249        },
250    );
251    Ok(())
252}

254/// Platform Cell (`PlatformCap`) claim: only inside the kernel-admitted ECAM
255/// window, and never over a managed RAM frame.
256///
257/// The claim is user-mapped by the caller, so a range reaching kernel RAM, page
258/// tables or the kernel image would hand the cell the kernel. The RAM test is
259/// kept even inside the window: the window comes from firmware (ACPI MCFG).
260pub fn request_ecam_mmio(tid: usize, base: usize, len: usize) -> ViResult<()> {
261    let end = checked_mmio_end(base, len)?;
262    let window = crate::dev::pcie_ecam::admitted_ecam_range();
263    let admitted = {
264        // No allocator means managed RAM is unknown: refuse rather than guess.
265        crate::memory::frame::with(|allocator| {
266            ecam_claim_admitted(
267                base,
268                end,
269                window,
270                allocator
271                    .managed_ranges()
272                    .map(|range| (range.start, range.start + range.frames * 4096)),
273            )
274        })
275        .unwrap_or(false)
276    };
277    if !admitted {
278        log::warn!(
279            "[mmio] platform claim {:#x}+{:#x} outside the ECAM window or over RAM",
280            base,
281            len
282        );
283        return Err(ViError::PermissionDenied);
284    }
285    request_mmio_unchecked(tid, base, len)
286}

288/// `[base, end)` lies inside `window` (`(base, len)`) and overlaps none of
289/// `ram` (`(start, end)` ranges).
290fn ecam_claim_admitted(
291    base: usize,
292    end: usize,
293    window: Option<(usize, usize)>,
294    mut ram: impl Iterator<Item = (usize, usize)>,
295) -> bool {
296    let Some((window_base, window_len)) = window else {
297        return false;
298    };
299    let inside = base >= window_base
300        && window_base
301            .checked_add(window_len)
302            .is_some_and(|window_end| end <= window_end);
303    inside && !ram.any(|(start, ram_end)| base < ram_end && start < end)
304}

317/// Check if `[base, base + len)` falls inside the platform's DWC2 host controller MMIO window.
318pub fn is_dwc2_mmio(base: usize, len: usize) -> bool {
319    dwc2_aperture().is_some_and(|(dwc2_base, dwc2_size)| {
320        len > 0
321            && base >= dwc2_base
322            && base
323                .checked_add(len)
324                .is_some_and(|end| end <= dwc2_base + dwc2_size)
325    })
326}

328/// Request exclusive ownership of a DWC2 MMIO region for a driver cell holding
329/// `UsbDriverCap`: the aperture's device record is claimed first (its owner,
330/// or its first claimant while free), and a first claim is undone when the
331/// window is refused.
332pub fn request_dwc2_mmio(tid: usize, base: usize, len: usize) -> ViResult<()> {
333    if !is_dwc2_mmio(base, len) {
334        return Err(ViError::PermissionDenied);
335    }
336    let end = checked_mmio_end(base, len)?;
337    claim_device_then(tid, base, len, true, || {
338        let mut reg = REGISTRY.lock();
339        if reg
340            .iter()
341            .any(|(&window_base, region)| claim_collides(window_base, region, base, end))
342        {
343            return Err(ViError::AlreadyExists);
344        }
345        #[cfg(feature = "p01-trace")]
346        let _semantic = crate::memory::heap::p01_owner::creation(
347            None,
348            crate::memory::heap::p01_owner::DRIVER_SHARED,
349        );
350        reg.insert(
351            base,
352            MmioRegion {
353                len,
354                owner_tid: tid,
355                class: CLASS_DWC2,
356            },
357        );
358        Ok(())
359    })
360}

362/// Claim the device record holding window `[base, base + len)` for `tid`, then
363/// run `grant`; a first claim of an MMIO device is undone when `grant` fails.
364///
365/// A PCI function's BAR is granted only to the function's owner (the 418/424
366/// claimant), so no driver can map, reset or reprogram a device it was not
367/// handed. A VirtIO slot or the DWC2 aperture is claimed by its first
368/// requester and refused to everyone else until its owner is reaped. Lock
369/// order: `DEVICES` is released before `grant` takes `REGISTRY`.
370fn claim_device_then(
371    tid: usize,
372    base: usize,
373    len: usize,
374    dwc2: bool,
375    grant: impl FnOnce() -> ViResult<()>,
376) -> ViResult<()> {
377    use crate::dev::ownership::{claim_window, unclaim, WindowClaim};
378    let claim = claim_window(tid, base, len, dwc2);
379    match claim {
380        WindowClaim::NoDevice | WindowClaim::NotOwner => return Err(ViError::PermissionDenied),
381        WindowClaim::Held => return Err(ViError::AlreadyExists),
382        WindowClaim::Owned | WindowClaim::Claimed(_) => {}
383    }
384    let granted = grant();
385    if let (Err(_), WindowClaim::Claimed(key)) = (&granted, claim) {
386        unclaim(tid, key);
387    }
388    granted
389}

391/// Request exclusive ownership of `[base, base+len)` for task `tid`.
392///
393/// Returns:
394/// - `Ok(())` — range is now owned by the caller; construct `MmioRegion` and
395///   hand it to the Cell.
396/// - `Err(PermissionDenied)` — range not in allowlist, or its device class is
397///   not among `allowed_devices` (the cell's declared `mmio_devices`), or it is
398///   the BAR of a PCI function the caller does not own.
399/// - `Err(AlreadyExists)` — range overlaps an already-granted region, or is a
400///   VirtIO slot another task owns.
401/// - `Err(InvalidInput)` — zero length or arithmetic overflow in `base + len`.
402pub fn request_mmio(tid: usize, base: usize, len: usize, allowed_devices: u8) -> ViResult<()> {
403    // 1. Allowlist check — the range must fall inside a known device window
404    //    AND that window's device class must be one the cell declared.
405    let end = checked_mmio_end(base, len)?;
406
407    // PCIe path: the window must be one a device record holds (a scanned or
408    // registered BAR, or a VirtIO slot), and the caller must own that device.
409    if allowed_devices & DEV_PCIE != 0 {
410        return claim_device_then(tid, base, len, false, || {
411            grant_window(tid, base, len, end, DEV_PCIE as MmioClass)
412        });
413    }
414    // SoC peripheral path: static per-arch allowlist.
415    match static_range_class(ALLOWED, base, end, allowed_devices) {
416        Some(class) => grant_window(tid, base, len, end, class as MmioClass),
417        None => Err(ViError::PermissionDenied),
418    }
419}

421/// Record `[base, end)` as `tid`'s window of `class` unless it overlaps a
422/// window another live task holds, or one being released or quarantined.
423///
424/// Lock order: SCHEDULER ranks above RESOURCE_REGISTRY, so owner liveness is
425/// read with the registry lock dropped. That is sound because tids are never
426/// re-issued (`task::task_id_selftest`) and a dead task never revives: an owner
427/// seen dead stays dead. A colliding owner not in that set, including one
428/// that claimed in between, is treated as live and refuses the grant.
429fn grant_window(tid: usize, base: usize, len: usize, end: usize, class: MmioClass) -> ViResult<()> {
430    let colliders: alloc::vec::Vec<usize> = REGISTRY
431        .lock()
432        .iter()
433        .filter(|&(&eb, region)| region.owner_tid != tid && claim_collides(eb, region, base, end))
434        .map(|(_, region)| region.owner_tid)
435        .collect();
436    let dead = dead_owners(&colliders);
437
438    // 2. Overlap check — no two live cells may share a byte
439    let mut reg = REGISTRY.lock();
440    reg.retain(|&eb, region| {
441        let ee = eb + region.len;
442        if !(end <= eb || base >= ee) && region.owner_tid == tid {
443            return false;
444        }
445        true
446    });
447    for (&eb, region) in reg.iter() {
448        if claim_collides(eb, region, base, end) {
449            // A quarantined or releasing window is never handed over
450            // (`release_mmio_for`).
451            if region.owner_tid == QUARANTINED_OWNER || region.owner_tid == RELEASING_OWNER {
452                return Err(ViError::AlreadyExists);
453            }
454            if dead.contains(&region.owner_tid) {
455                continue;
456            }
457            return Err(ViError::AlreadyExists);
458        }
459    }
460    #[cfg(feature = "p01-trace")]
461    let _semantic = crate::memory::heap::p01_owner::creation(
462        None,
463        crate::memory::heap::p01_owner::DRIVER_SHARED,
464    );
465    reg.insert(
466        base,
467        MmioRegion {
468            len,
469            owner_tid: tid,
470            class,
471        },
472    );
473    Ok(())
474}

476/// Owners among `tids` that are terminated or gone. Before the scheduler
477/// exists, nobody counts as dead, which keeps every overlap refused as the
478/// previous code did.
479fn dead_owners(tids: &[usize]) -> alloc::vec::Vec<usize> {
480    if tids.is_empty() {
481        return alloc::vec::Vec::new();
482    }
483    let guard = crate::task::SCHEDULER.lock();
484    let Some(sched) = guard.as_ref() else {
485        return alloc::vec::Vec::new();
486    };
487    tids.iter()
488        .copied()
489        .filter(|t| match sched.tasks.get(t) {
490            Some(task) => matches!(task.state, crate::task::tcb::TaskState::Terminated),
491            None => true,
492        })
493        .collect()
494}

513/// Release all MMIO regions owned by task `tid`.
514///
515/// Retirement calls this for every member task of the retiring incarnation.
516/// The windows' USER pages first go back to the boot posture (kernel-only) on
517/// every architecture: `RequestMmio` installed them, so without this a dead
518/// Cell's window would stay U-mode/EL0/ring-3-reachable for every other Cell.
519/// The windows pass to [`RELEASING_OWNER`] first, under `REGISTRY`, so every
520/// claim on them is refused; the lowering then runs without `REGISTRY`,
521/// which ranks after `KERNEL_ROOT` and the console that the lowering takes.
522/// A window is released only once its lowering is in force on every hart. A
523/// window whose lowering is not (a remote hart did not confirm the
524/// invalidation, or the leaf could not be rewritten) stays registered to
525/// [`QUARANTINED_OWNER`] — counted, logged, and handed to no Cell — because a
526/// stale USER translation may survive on that hart, until
527/// [`retry_quarantined_windows`] lowers it again and that lowering is
528/// confirmed.
529pub fn release_mmio_for(tid: usize) {
530    let release = release_windows(tid, lower_run);
531    for &(base, len, page, error) in &release.unlowered {
532        log::error!(
533            "[mmio] release tid={tid}: window {base:#x}+{len:#x} quarantined, page {page:#x} \
534             may stay USER ({error:?}); handed out again only once a retry lowers it \
535             ({} quarantined)",
536            release.quarantined
537        );
538    }
539    // A VideoCore allocation and its USER mapping remain valid until reboot.
540    // Do not clear Active metadata or permit a restarted cell to replace it:
541    // restart is fail-stop for display after registration.
542}

544/// Owner of a released window whose USER pages are not provably gone on every
545/// hart. No task id is ever `usize::MAX`, so no owner query matches it, and
546/// every claim overlapping it is refused until [`retry_quarantined_windows`]
547/// confirms its lowering.
548const QUARANTINED_OWNER: usize = usize::MAX;
549
550/// Owner of a window whose release lowering is in progress without
551/// `REGISTRY` ([`release_windows`]). No task id reaches it either: no owner
552/// query matches it, every claim overlapping the window or sharing one of its
553/// released pages is refused, and only the release that set it changes it.
554const RELEASING_OWNER: usize = usize::MAX - 1;
555
556/// Windows registered to [`QUARANTINED_OWNER`], so the timer poll and the
557/// housekeeping retry cost one load while there are none. Written only under
558/// `REGISTRY`.
559static QUARANTINED_WINDOWS: AtomicUsize = AtomicUsize::new(0);
560
561/// Set by the hart-0 timer poll ([`poll_quarantined_windows`]) and consumed
562/// only by [`retry_quarantined_windows`]: a housekeeping pass armed by any
563/// other producer (a zombie reap, a grant reap, a tag wait ending) does not
564/// lower the quarantined windows again.
565static RETRY_ARMED: AtomicBool = AtomicBool::new(false);

567/// Hart-0 timer ticks between two retries of the quarantined windows' lowering
568/// (about 1.3 s at 10 ms): the backoff that keeps a never-confirmed lowering
569/// from re-arming the housekeeping pass on every tick.
570const RETRY_EVERY_TICKS: usize = 128;

584/// What one [`release_windows`] did.
585struct Release {
586    /// `(base, len)` of every window released.
587    released: alloc::vec::Vec<(usize, usize)>,
588    unlowered: alloc::vec::Vec<Unlowered>,
589    /// Windows quarantined after it, all owners.
590    quarantined: usize,
591}
592
593/// Release `owner`'s windows: mark them releasing under `REGISTRY`, lower
594/// their runs through `lower` without it, then re-take it to release each
595/// window whose lowering is in force and quarantine the others.
596fn release_windows(
597    owner: usize,
598    lower: impl FnMut(usize, usize) -> Result<(), (usize, crate::memory::paging::PageTableError)>,
599) -> Release {
600    let windows = begin_release(&mut REGISTRY.lock(), owner);
601    let unlowered = lower_windows(&windows, lower);
602    let quarantined = finish_release(&mut REGISTRY.lock(), &windows, &unlowered);
603    let released = windows
604        .iter()
605        .filter(|window| !unlowered.iter().any(|failed| failed.0 == window.base))
606        .map(|window| (window.base, window.len))
607        .collect();
608    Release {
609        released,
610        unlowered,
611        quarantined,
612    }
613}
614
615/// Pass `owner`'s windows to [`RELEASING_OWNER`] and return them with the
616/// page runs to lower, computed while `REGISTRY` still shows which pages
617/// another live window covers.
618fn begin_release(
619    registry: &mut BTreeMap<usize, MmioRegion>,
620    owner: usize,
621) -> alloc::vec::Vec<Releasing> {
622    let windows: alloc::vec::Vec<Releasing> = registry
623        .iter()
624        .filter(|(_, region)| region.owner_tid == owner)
625        .map(|(&base, region)| Releasing {
626            base,
627            len: region.len,
628            runs: lowering_runs(registry, owner, base, region.len),
629        })
630        .collect();
631    for window in &windows {
632        if let Some(region) = registry.get_mut(&window.base) {
633            region.owner_tid = RELEASING_OWNER;
634        }
635    }
636    windows
637}

639/// The pages `RequestMmio` made USER for `[base, base + len)`: the whole pages
640/// on RISC-V and AArch64 (`user_pages`), every page the window touches on
641/// x86_64 (`map_mmio_user_x86` maps the page-rounded window).
642#[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
643fn released_pages(base: usize, len: usize) -> core::ops::Range<usize> {
644    user_pages(base, len)
645}
646
647/// [`released_pages`] where `RequestMmio` maps the page-rounded window.
648#[cfg(not(any(target_arch = "riscv64", target_arch = "aarch64")))]
649fn released_pages(base: usize, len: usize) -> core::ops::Range<usize> {
650    const PAGE: usize = 4096;
651    match base
652        .checked_add(len)
653        .and_then(|end| end.checked_add(PAGE - 1))
654    {
655        Some(end) if len != 0 => (base & !(PAGE - 1))..(end & !(PAGE - 1)),
656        _ => 0..0,
657    }
658}
659
660/// The released pages of window `[base, base + len)` held by `owner` that its
661/// release lowers, as contiguous runs `(start, pages)`. A page another
662/// owner's live window still covers (one handed over from a Terminated owner
663/// before this reap) keeps USER; a quarantined or releasing window has no
664/// owner to keep it for.
665fn lowering_runs(
666    registry: &BTreeMap<usize, MmioRegion>,
667    owner: usize,
668    base: usize,
669    len: usize,
670) -> alloc::vec::Vec<(usize, usize)> {
671    const PAGE: usize = 4096;
672    let mut runs: alloc::vec::Vec<(usize, usize)> = alloc::vec::Vec::new();
673    for page in released_pages(base, len).step_by(PAGE) {
674        let still_owned = registry.iter().any(|(&live, other)| {
675            other.owner_tid != owner
676                && other.owner_tid != QUARANTINED_OWNER
677                && other.owner_tid != RELEASING_OWNER
678                && live < page + PAGE
679                && page < live + other.len
680        });
681        if still_owned {
682            continue;
683        }
684        match runs.last_mut() {
685            Some((start, pages)) if *start + *pages * PAGE == page => *pages += 1,
686            _ => runs.push((page, 1)),
687        }
688    }
689    runs
690}

692/// Lower every run of `windows` through `lower`, with no `REGISTRY` held.
693/// Every run is attempted; a window is reported with its first page that
694/// failed.
695fn lower_windows(
696    windows: &[Releasing],
697    mut lower: impl FnMut(usize, usize) -> Result<(), (usize, crate::memory::paging::PageTableError)>,
698) -> alloc::vec::Vec<Unlowered> {
699    let mut unlowered = alloc::vec::Vec::new();
700    for window in windows {
701        let mut failed = None;
702        for &(start, pages) in &window.runs {
703            if let Err(failure) = lower(start, pages) {
704                failed.get_or_insert(failure);
705            }
706        }
707        if let Some((page, error)) = failed {
708            unlowered.push((window.base, window.len, page, error));
709        }
710    }
711    unlowered
712}

714/// Return a run to the boot posture on every hart: one shared flush per
715/// 2 MiB step, absent leaves included (`paging::restore_mmio_run`).
716#[cfg(all(cellos_wide, not(test)))]
717fn lower_run(
718    start: usize,
719    pages: usize,
720) -> Result<(), (usize, crate::memory::paging::PageTableError)> {
721    crate::memory::paging::restore_mmio_run(start, pages).map(drop)
722}

734/// Release each of `windows` whose lowering is in force on every hart and
735/// pass the others (`unlowered`) to [`QUARANTINED_OWNER`]. A releasing window
736/// is changed by no other path, so each is still registered as it was left.
737/// Returns the windows quarantined after it.
738fn finish_release(
739    registry: &mut BTreeMap<usize, MmioRegion>,
740    windows: &[Releasing],
741    unlowered: &[Unlowered],
742) -> usize {
743    for window in windows {
744        let Some(region) = registry
745            .get_mut(&window.base)
746            .filter(|region| region.owner_tid == RELEASING_OWNER && region.len == window.len)
747        else {
748            continue;
749        };
750        if unlowered.iter().any(|failed| failed.0 == window.base) {
751            region.owner_tid = QUARANTINED_OWNER;
752        } else {
753            registry.remove(&window.base);
754        }
755    }
756    count_quarantined(registry)
757}
758
759/// Lower every quarantined window again and release those whose lowering is
760/// now in force on every hart. The housekeeping pass calls it; it acts only
761/// when the timer poll armed it ([`RETRY_ARMED`]) and no remote hart owes an
762/// invalidation, and it lowers without `REGISTRY`, as [`release_mmio_for`]
763/// does. Returns the windows still quarantined.
764pub fn retry_quarantined_windows() -> usize {
765    if !RETRY_ARMED.swap(false, Ordering::AcqRel)
766        || QUARANTINED_WINDOWS.load(Ordering::Acquire) == 0
767    {
768        return QUARANTINED_WINDOWS.load(Ordering::Acquire);
769    }
770    // A hart that still owes an invalidation would leave this lowering
771    // unconfirmed too: the next poll arms the retry again.
772    #[cfg(cellos_wide)]
773    if crate::memory::tlb_shootdown::tag_invalidation_outstanding().is_some() {
774        return QUARANTINED_WINDOWS.load(Ordering::Acquire);
775    }
776    let release = release_windows(QUARANTINED_OWNER, lower_run);
777    for (base, len) in release.released {
778        log::warn!(
779            "[mmio] quarantined window {base:#x}+{len:#x} lowered on every hart: released \
780             ({} still quarantined)",
781            release.quarantined
782        );
783    }
784    release.quarantined
785}
786
787/// Hart-0 timer hook: while a window is quarantined, arm the retry and the
788/// housekeeping pass every [`RETRY_EVERY_TICKS`] once no remote hart owes an
789/// invalidation, so [`retry_quarantined_windows`] runs when its flush can
790/// confirm. One load while nothing is quarantined; never waits.
791#[cfg(cellos_wide)]
792pub fn poll_quarantined_windows() {
793    if QUARANTINED_WINDOWS.load(Ordering::Acquire) == 0
794        || !crate::task::system_ticks().is_multiple_of(RETRY_EVERY_TICKS)
795    {
796        return;
797    }
798    if crate::memory::tlb_shootdown::tag_invalidation_outstanding().is_none() {
799        RETRY_ARMED.store(true, Ordering::Release);
800        crate::task::note_housekeeping();
801    }
802}

804/// Windows registered to [`QUARANTINED_OWNER`]; refreshes
805/// [`QUARANTINED_WINDOWS`]. The caller holds `REGISTRY`.
806fn count_quarantined(registry: &BTreeMap<usize, MmioRegion>) -> usize {
807    let quarantined = registry
808        .values()
809        .filter(|region| region.owner_tid == QUARANTINED_OWNER)
810        .count();
811    QUARANTINED_WINDOWS.store(quarantined, Ordering::Release);
812    quarantined
813}

815/// Does a claim of `[base, end)` collide with the registered window at
816/// `window_base`? A window being released also refuses a claim that only
817/// shares one of its released pages: its lowering runs without `REGISTRY` and
818/// would take USER from the new owner's page.
819fn claim_collides(window_base: usize, region: &MmioRegion, base: usize, end: usize) -> bool {
820    let window_end = window_base + region.len;
821    let (start, stop) = if region.owner_tid == RELEASING_OWNER {
822        let pages = released_pages(window_base, region.len);
823        (window_base.min(pages.start), window_end.max(pages.end))
824    } else {
825        (window_base, window_end)
826    };
827    base < stop && start < end
828}

830/// The windows [`revoke_mmio_for`] would release, without releasing them: a
831/// revoke lowers their PTEs first and releases exactly these only once the
832/// lowering is in force on every hart ([`release_mmio_windows`]).
833pub fn mmio_windows_for(tid: usize, device_mask: MmioClass) -> alloc::vec::Vec<(usize, usize)> {
834    if device_mask == 0 {
835        return alloc::vec::Vec::new();
836    }
837    #[cfg(feature = "p01-trace")]
838    let _semantic = crate::memory::heap::p01_owner::creation(
839        None,
840        crate::memory::heap::p01_owner::DRIVER_SHARED,
841    );
842    REGISTRY
843        .lock()
844        .iter()
845        .filter(|(_, region)| region.owner_tid == tid && region.class & device_mask != 0)
846        .map(|(&base, region)| (base, region.len))
847        .collect()
848}

850/// Release exactly `windows` of `tid` (bases from [`mmio_windows_for`]); a
851/// window claimed since is untouched.
852pub fn release_mmio_windows(tid: usize, windows: &[(usize, usize)]) {
853    REGISTRY.lock().retain(|base, region| {
854        !(region.owner_tid == tid && windows.contains(&(*base, region.len)))
855    });
856    crate::dev::ownership::release_mmio_devices_in(tid, windows);
857}

859/// Release the regions owned by task `tid` whose class is in `device_mask`, and
860/// return their `(base, len)` so the caller can remove user accessibility.
861///
862/// Selective by construction: a class the caller did not name stays owned and
863/// mapped, so revoking one device class never disarms another. Used by the
864/// MMIO revoke self-test; `sys_cap_revoke` lowers first ([`mmio_windows_for`]),
865/// and task retirement uses [`release_mmio_for`], which drops every region
866/// regardless of class.
867pub fn revoke_mmio_for(tid: usize, device_mask: MmioClass) -> alloc::vec::Vec<(usize, usize)> {
868    let mut revoked = alloc::vec::Vec::new();
869    if device_mask == 0 {
870        return revoked;
871    }
872    REGISTRY.lock().retain(|&base, region| {
873        let hit = region.owner_tid == tid && region.class & device_mask != 0;
874        if hit {
875            #[cfg(feature = "p01-trace")]
876            let _semantic = crate::memory::heap::p01_owner::creation(
877                None,
878                crate::memory::heap::p01_owner::DRIVER_SHARED,
879            );
880            revoked.push((base, region.len));
881        }
882        !hit
883    });
884    revoked
885}

887/// Return the task ID (TID) of the cell that currently owns the MMIO region
888/// whose base address exactly matches `base`.
889///
890/// Returns `None` if no cell has requested that exact base address, or the
891/// window is being released or quarantined (no cell owns it any more).
892/// Used by the GPIO IRQ handler to route interrupts to the current MMIO owner.
893pub fn lookup_mmio_owner(base: usize) -> Option<usize> {
894    REGISTRY
895        .lock()
896        .get(&base)
897        .map(|region| region.owner_tid)
898        .filter(|&tid| tid != QUARANTINED_OWNER && tid != RELEASING_OWNER)
899}

901/// Return whether task `tid` owns exactly `[base, base + len)`.
902///
903/// Display registration uses this rather than an overlap query: the VideoCore
904/// mailbox authority is one fixed capability, so a subrange cannot be confused
905/// with an independently granted peripheral window.
906pub fn owns_exact_mmio(tid: usize, base: usize, len: usize) -> bool {
907    REGISTRY
908        .lock()
909        .get(&base)
910        .is_some_and(|region| region.len == len && region.owner_tid == tid)
911}

913/// Outcome of atomically reserving one framebuffer mapping transaction.
914#[derive(Debug, Clone, Copy, PartialEq, Eq)]
915pub enum DisplayFramebufferReservation {
916    Reserved,
917    ActiveReplay,
918}
919
920/// Why a framebuffer reservation could not be established.
921#[derive(Debug, Clone, Copy, PartialEq, Eq)]
922pub enum DisplayFramebufferReservationError {
923    /// Another pending or active framebuffer transaction owns the singleton.
924    Conflict,
925}
926
927fn display_framebuffer_candidate(
928    owner: CellId,
929    base: usize,
930    size: usize,
931    width: u16,
932    height: u16,
933    pitch: usize,
934) -> DisplayFramebuffer {
935    DisplayFramebuffer {
936        owner,
937        base,
938        size,
939        width,
940        height,
941        pitch,
942    }
943}
944
945/// Reserve one framebuffer mapping before page-table mutation.
946///
947/// A reservation serializes preflight/map/activation. Only an identical Active
948/// replay is accepted; a pending or different active request fails closed.
949pub fn reserve_display_framebuffer(
950    owner: CellId,
951    base: usize,
952    size: usize,
953    width: u16,
954    height: u16,
955    pitch: usize,
956) -> Result<DisplayFramebufferReservation, DisplayFramebufferReservationError> {
957    let candidate = display_framebuffer_candidate(owner, base, size, width, height, pitch);
958    let mut framebuffer = DISPLAY_FRAMEBUFFER.lock();
959    match *framebuffer {
960        DisplayFramebufferState::Empty => {
961            *framebuffer = DisplayFramebufferState::Pending(candidate);
962            Ok(DisplayFramebufferReservation::Reserved)
963        }
964        DisplayFramebufferState::Active(existing) if existing == candidate => {
965            Ok(DisplayFramebufferReservation::ActiveReplay)
966        }
967        DisplayFramebufferState::Pending(_) | DisplayFramebufferState::Active(_) => {
968            Err(DisplayFramebufferReservationError::Conflict)
969        }
970    }
971}
972
973/// Commit a previously reserved mapping after every page was installed.
974pub fn activate_display_framebuffer(
975    owner: CellId,
976    base: usize,
977    size: usize,
978    width: u16,
979    height: u16,
980    pitch: usize,
981) -> bool {
982    let candidate = display_framebuffer_candidate(owner, base, size, width, height, pitch);
983    let mut framebuffer = DISPLAY_FRAMEBUFFER.lock();
984    if *framebuffer == DisplayFramebufferState::Pending(candidate) {
985        *framebuffer = DisplayFramebufferState::Active(candidate);
986        true
987    } else {
988        false
989    }
990}
991
992/// Cancel a reservation when preflight or mapping fails before activation.
993pub fn cancel_display_framebuffer(
994    owner: CellId,
995    base: usize,
996    size: usize,
997    width: u16,
998    height: u16,
999    pitch: usize,
1000) {
1001    let candidate = display_framebuffer_candidate(owner, base, size, width, height, pitch);
1002    let mut framebuffer = DISPLAY_FRAMEBUFFER.lock();
1003    if *framebuffer == DisplayFramebufferState::Pending(candidate) {
1004        *framebuffer = DisplayFramebufferState::Empty;
1005    }
1006}

1008/// Return the active firmware scanout geometry, if registered.
1009pub fn display_framebuffer_resolution() -> Option<(u16, u16)> {
1010    match *DISPLAY_FRAMEBUFFER.lock() {
1011        DisplayFramebufferState::Active(framebuffer) => {
1012            Some((framebuffer.width, framebuffer.height))
1013        }
1014        DisplayFramebufferState::Empty | DisplayFramebufferState::Pending(_) => None,
1015    }
1016}

1018/// Return the active firmware scanout geometry and base, if registered.
1019pub fn display_framebuffer_info() -> Option<(usize, usize, u16, u16, usize)> {
1020    match *DISPLAY_FRAMEBUFFER.lock() {
1021        DisplayFramebufferState::Active(framebuffer) => Some((
1022            framebuffer.base,
1023            framebuffer.size,
1024            framebuffer.width,
1025            framebuffer.height,
1026            framebuffer.pitch,
1027        )),
1028        DisplayFramebufferState::Empty | DisplayFramebufferState::Pending(_) => None,
1029    }
1030}
