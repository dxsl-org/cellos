1//! Resource Registry — exclusive MMIO region grants for Driver Cells.
2//!
3//! A Driver Cell calls `sys_request_mmio(base, len)` (added in Phase 03).
4//! The kernel checks here before handing an `MmioRegion` to the Cell:
5//!
6//! 1. **Allowlist**: the requested range must fall within a known-safe
7//!    device window for the active target. Unknown ranges are rejected
8//!    so a misbehaving Cell cannot map arbitrary kernel memory as MMIO.
9//!
10//! 2. **Exclusive ownership**: at most one Cell may hold a given MMIO range.
11//!    A second `request_mmio` for an overlapping range returns `AlreadyExists`.
12//!
13//! 3. **Release-on-exit**: `release_for(cell_id)` frees all ranges owned by
14//!    a Cell.  Call this from every Cell-exit path alongside
15//!    `cell_quota::deregister`.
16//!
17//! # Current scope
18//! Allowlist is hardcoded per target (DTB discovery remains deferred).
19//!
20//! | Target | Device | Base | Size |
21//! |--------|--------|------|------|
22//! | QEMU ARM virt (aarch64) | PL011 UART0 | 0x0900_0000 | 0x1000 |
23//! | QEMU ARM virt (aarch64) | PL061 GPIO  | 0x0903_0000 | 0x1000 |
24//! | Raspberry Pi 3 | BCM2837 GPIO/BSC1/SPI0/AUX | exact 4-KiB windows | 0x1000 |
25//! | QEMU RISC-V virt (riscv64) | (none yet — kernel serial owns UART) | — | — |
26

35/// UART serial device window. Set in a cell's `mmio_devices` when its manifest
36/// declares `uart = true`.
37pub const DEV_UART: u8 = 1 << 0;
38/// GPIO controller window. Set when the manifest declares `gpio = true`.
39pub const DEV_GPIO: u8 = 1 << 1;
40/// PCIe device BAR window. Set on tasks with `PcieDriverCap`; PCIe authority is
41/// path-gated and intentionally remains separate from hardware manifest flags.
42pub const DEV_PCIE: u8 = 1 << 2;
43/// CAN bus controller window (v2 manifest — freed by the u16 flags widening).
44/// Set when the manifest declares `can = true`.
45pub const DEV_CAN: u8 = 1 << 3;
46/// ADC controller window (v2 manifest). Set when the manifest declares `adc = true`.
47pub const DEV_ADC: u8 = 1 << 4;
48/// I2C controller window (v2 manifest). Set when the manifest declares `i2c = true`.
49pub const DEV_I2C: u8 = 1 << 5;
50pub const DEV_SPI: u8 = 1 << 6;
51/// Firmware / display controller window.
52pub const DEV_DISPLAY: u8 = 1 << 7;

54/// Class of an owned MMIO window: the `DEV_*` bits, plus window kinds that have
55/// no `DEV_*` bit because they are not manifest-declared devices.
56///
57/// The class is stored with the region when it is granted. Re-deriving it at
58/// revoke time would have to guess: a PCIe BAR is in neither allowlist, and a
59/// granted sub-window is contained in — not equal to — the window it came from.
60pub type MmioClass = u16;

62/// The Platform Cell's ECAM config-space window (`request_mmio_unchecked`).
63pub const CLASS_ECAM: MmioClass = 1 << 8;
64/// The DWC2 USB host aperture (`request_dwc2_mmio`).
65pub const CLASS_DWC2: MmioClass = 1 << 9;

203/// Test-hooks only: register a window with an explicit class, so a boot
204/// self-test can prove class-selective revocation on boards whose allowlist
205/// has no window of the class it needs.
206#[cfg(feature = "test-hooks")]
207pub(crate) fn test_register_region(
208    cell_id: CellId,
209    base: usize,
210    len: usize,
211    class: MmioClass,
212) -> bool {
213    if checked_mmio_end(base, len).is_err() {
214        return false;
215    }
216    REGISTRY
217        .lock()
218        .insert(
219            base,
220            MmioRegion {
221                len,
222                owner: cell_id,
223                class,
224            },
225        )
226        .is_none()
227}

229/// The per-arch static allowlist, for callers that must reason about which
230/// windows this board can grant at all (the boot self-tests, which have to stay
231/// board-agnostic, and `request_mmio`'s class derivation below).
232pub(crate) fn allowed_windows() -> &'static [(usize, usize, u8)] {
233    ALLOWED
234}
235
236/// Device class of the allowlist window containing `[base, end)`, when the
237/// caller declared that class.
238///
239/// Returns the class rather than a bool because the grant records it: a revoke
240/// must be able to tell which device class a window belongs to without
241/// re-deriving it from a table that may have changed since.
242fn static_range_class(
243    allowed: &[(usize, usize, u8)],
244    base: usize,
245    end: usize,
246    allowed_devices: u8,
247) -> Option<u8> {
248    allowed
249        .iter()
250        .find(|&&(window_base, window_len, class)| {
251            window_base
252                .checked_add(window_len)
253                .is_some_and(|window_end| {
254                    base >= window_base && end <= window_end && class & allowed_devices != 0
255                })
256        })
257        .map(|&(_, _, class)| class)
258}

278/// Register a PCIe BAR window discovered during ECAM scan.
279///
280/// Called by `pcie_ecam::init()` for every non-zero BAR on every device.
281/// Driver Cells may subsequently call `sys_request_mmio` on these ranges
282/// if they hold `PcieDriverCap`.
283pub fn register_pcie_bar(base: usize, len: usize) {
284    if !valid_pcie_bar_window(base, len) {
285        log::warn!(
286            "[pcie] rejected invalid BAR window base={:#x} len={:#x}",
287            base,
288            len
289        );
290        return;
291    }
292    PCIE_BARS.lock().insert(base, len);
293}

295/// Current Driver Cells support bounded conventional BARs only.
296///
297/// ReBAR and larger accelerator apertures need a separate policy because they
298/// materially widen a Cell's MMIO authority.
299fn valid_pcie_bar_window(base: usize, len: usize) -> bool {
300    const MAX_SUPPORTED_BAR_LEN: usize = 1 << 30;
301
302    base != 0
303        && len != 0
304        && len <= MAX_SUPPORTED_BAR_LEN
305        && len.is_power_of_two()
306        && base & (len - 1) == 0
307        && base.checked_add(len).is_some()
308}

310/// Return `true` if `[base, base+len)` is a known PCIe BAR window.
311///
312/// Used by the `RequestMmio` handler to decide whether to take the PCIe path.
313pub fn is_pcie_bar(base: usize, len: usize) -> bool {
314    if len == 0 {
315        return false;
316    }
317    let guard = PCIE_BARS.lock();
318    guard.get(&base).is_some_and(|&bar_len| len <= bar_len)
319}

321fn checked_mmio_end(base: usize, len: usize) -> ViResult<usize> {
322    if len == 0 {
323        return Err(ViError::InvalidInput);
324    }
325    base.checked_add(len).ok_or(ViError::InvalidInput)
326}

328/// Request exclusive MMIO ownership without allowlist validation (Platform Cell only).
329///
330/// Bypasses the per-arch ALLOWED list and the PCIE_BARS table. The overlap check
331/// still runs — two cells cannot share a byte. Used by the PlatformCap bypass path
332/// in `sys_request_mmio` so the Platform Cell can claim the ECAM config-space window
333/// (which is not a device BAR and therefore not in either allowlist).
334pub fn request_mmio_unchecked(cell_id: CellId, base: usize, len: usize) -> ViResult<()> {
335    let end = checked_mmio_end(base, len)?;
336    let mut reg = REGISTRY.lock();
337    for (&eb, region) in reg.iter() {
338        let ee = eb + region.len;
339        if !(end <= eb || base >= ee) {
340            return Err(ViError::AlreadyExists);
341        }
342    }
343    reg.insert(
344        base,
345        MmioRegion {
346            len,
347            owner: cell_id,
348            class: CLASS_ECAM,
349        },
350    );
351    Ok(())
352}

354/// Check if `[base, base + len)` falls inside the platform's DWC2 host controller MMIO window.
355pub fn is_dwc2_mmio(base: usize, len: usize) -> bool {
356    #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
357    {
358        let dwc2_base = hal_soc_bcm27xx::BCM2837.mmio.dwc2_base;
359        let dwc2_len = hal_soc_bcm27xx::BCM2837.mmio.dwc2_grant_size;
360        len > 0
361            && base >= dwc2_base
362            && base
363                .checked_add(len)
364                .is_some_and(|end| end <= dwc2_base + dwc2_len)
365    }
366    #[cfg(not(all(target_arch = "aarch64", feature = "board-rpi3")))]
367    {
368        let _ = (base, len);
369        false
370    }
371}

373/// Request exclusive ownership of a DWC2 MMIO region for a driver cell holding `UsbDriverCap`.
374pub fn request_dwc2_mmio(cell_id: CellId, base: usize, len: usize) -> ViResult<()> {
375    if !is_dwc2_mmio(base, len) {
376        return Err(ViError::PermissionDenied);
377    }
378    let end = checked_mmio_end(base, len)?;
379    let mut reg = REGISTRY.lock();
380    for (&eb, region) in reg.iter() {
381        let ee = eb + region.len;
382        if !(end <= eb || base >= ee) {
383            return Err(ViError::AlreadyExists);
384        }
385    }
386    reg.insert(
387        base,
388        MmioRegion {
389            len,
390            owner: cell_id,
391            class: CLASS_DWC2,
392        },
393    );
394    Ok(())
395}

397/// Request exclusive ownership of `[base, base+len)` for `cell_id`.
398///
399/// Returns:
400/// - `Ok(())` — range is now owned by the caller; construct `MmioRegion` and
401///   hand it to the Cell.
402/// - `Err(PermissionDenied)` — range not in allowlist, or its device class is
403///   not among `allowed_devices` (the cell's declared `mmio_devices`).
404/// - `Err(AlreadyExists)` — range overlaps an already-granted region.
405/// - `Err(InvalidInput)` — zero length or arithmetic overflow in `base + len`.
406pub fn request_mmio(cell_id: CellId, base: usize, len: usize, allowed_devices: u8) -> ViResult<()> {
407    // 1. Allowlist check — the range must fall inside a known device window
408    //    AND that window's device class must be one the cell declared.
409    let end = checked_mmio_end(base, len)?;
410
411    // PCIe path: validate against the dynamic BAR table populated by pcie_ecam.
412    let class: MmioClass = if allowed_devices & DEV_PCIE != 0 {
413        let bars = PCIE_BARS.lock();
414        if !bars.get(&base).is_some_and(|&bar_len| len <= bar_len) {
415            return Err(ViError::PermissionDenied);
416        }
417        DEV_PCIE as MmioClass
418    } else {
419        // SoC peripheral path: static per-arch allowlist.
420        match static_range_class(ALLOWED, base, end, allowed_devices) {
421            Some(class) => class as MmioClass,
422            None => return Err(ViError::PermissionDenied),
423        }
424    };
425
426    // 2. Overlap check — no two live cells may share a byte
427    let mut reg = REGISTRY.lock();
428    reg.retain(|&eb, region| {
429        let ee = eb + region.len;
430        if !(end <= eb || base >= ee) && region.owner == cell_id {
431            return false;
432        }
433        true
434    });
435    for (&eb, region) in reg.iter() {
436        let ee = eb + region.len;
437        if !(end <= eb || base >= ee) {
438            if let Some(sched) = crate::task::SCHEDULER.lock().as_ref() {
439                if let Some(task) = sched.tasks.get(&(region.owner.0 as usize)) {
440                    if matches!(task.state, crate::task::tcb::TaskState::Terminated) {
441                        continue;
442                    }
443                } else {
444                    continue;
445                }
446            }
447            return Err(ViError::AlreadyExists);
448        }
449    }
450    reg.insert(
451        base,
452        MmioRegion {
453            len,
454            owner: cell_id,
455            class,
456        },
457    );
458    Ok(())
459}

465/// Power-on self-test of the MMIO allowlist. Non-RPi3 targets have nothing
466/// pinned yet; the test exists so the boot chain can call it unconditionally.
467#[cfg(not(all(target_arch = "aarch64", feature = "board-rpi3")))]
468pub fn self_test() -> bool {
469    true
470}
471
472/// Power-on self-test of the RPi3 production MMIO allowlist.
473///
474/// Pins two properties on the REAL table (`ALLOWED`), not a synthetic copy:
475///
476/// 1. Positive control — a known window (mailbox/DEV_DISPLAY) still authorizes.
477///    Without this, an accidentally emptied table would make every denial
478///    below vacuous.
479/// 2. DWC2 denial — the DWC2 USB window must be rejected for every device
480///    class, full masks included, over the whole aperture AND at its edges.
481///    The entry was previously present mis-tagged `DEV_PCIE`; it may return
482///    only via policy v3 with a signed USB-host authority byte.
483#[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
484pub fn self_test() -> bool {
485    let mmio = hal_soc_bcm27xx::BCM2837.mmio;
486
487    // 1. Positive control: the table is live.
488    let mbox_base = mmio.mailbox_base;
489    let mbox_end = mbox_base + mmio.mailbox_grant_size;
490    if static_range_class(ALLOWED, mbox_base, mbox_end, DEV_DISPLAY).is_none() {
491        log::error!("[selftest] mmio-allowlist: FAIL — mailbox window no longer authorized");
492        return false;
493    }
494
495    // 2. DWC2 denial across the full aperture, per class and combined masks.
496    let dwc2_base = mmio.dwc2_base;
497    let dwc2_end = dwc2_base + mmio.dwc2_grant_size;
498    let classes = [
499        DEV_UART,
500        DEV_GPIO,
501        DEV_PCIE,
502        DEV_CAN,
503        DEV_ADC,
504        DEV_I2C,
505        DEV_SPI,
506        DEV_DISPLAY,
507        0xFF,
508    ];
509    for class in classes {
510        if static_range_class(ALLOWED, dwc2_base, dwc2_end, class).is_some() {
511            log::error!(
512                "[selftest] mmio-allowlist: FAIL — DWC2 window authorized for class {class:#04x}"
513            );
514            return false;
515        }
516    }
517
518    // Edge words: same full class sweep as the whole aperture — a future
519    // sub-range entry tagged with ANY class must fail the boot self-test.
520    let last_word = dwc2_end - 4;
521    for addr in [dwc2_base, last_word] {
522        for class in classes {
523            if static_range_class(ALLOWED, addr, addr + 4, class).is_some() {
524                log::error!(
525                    "[selftest] mmio-allowlist: FAIL — DWC2 edge word {addr:#x} authorized \
526                     for class {class:#04x}"
527                );
528                return false;
529            }
530        }
531    }
532    true
533}

535/// Release all MMIO regions owned by `cell_id`.
536///
537/// Call this from every Cell-exit path (Exit syscall, ForceExit, fault, watchdog).
538pub fn release_for(cell_id: CellId) {
539    REGISTRY
540        .lock()
541        .retain(|_base, region| region.owner != cell_id);
542    // A VideoCore allocation and its USER mapping remain valid until reboot.
543    // Do not clear Active metadata or permit a restarted cell to replace it:
544    // restart is fail-stop for display after registration.
545}

547/// Release the regions owned by `cell_id` whose class is in `device_mask`, and
548/// return their `(base, len)` so the caller can remove user accessibility.
549///
550/// Selective by construction: a class the caller did not name stays owned and
551/// mapped, so revoking one device class never disarms another. Used by runtime
552/// revocation (`sys_cap_revoke`); Cell death uses [`release_for`], which drops
553/// every region regardless of class.
554pub fn revoke_mmio_for(cell_id: CellId, device_mask: MmioClass) -> alloc::vec::Vec<(usize, usize)> {
555    let mut revoked = alloc::vec::Vec::new();
556    if device_mask == 0 {
557        return revoked;
558    }
559    REGISTRY.lock().retain(|&base, region| {
560        let hit = region.owner == cell_id && region.class & device_mask != 0;
561        if hit {
562            revoked.push((base, region.len));
563        }
564        !hit
565    });
566    revoked
567}

581/// Return whether `cell_id` owns exactly `[base, base + len)`.

584/// mailbox authority is one fixed capability, so a subrange cannot be confused

629pub fn reserve_display_framebuffer(
630    owner: CellId,
631    base: usize,
632    size: usize,
633    width: u16,
634    height: u16,
635    pitch: usize,
636) -> Result<DisplayFramebufferReservation, DisplayFramebufferReservationError> {
637    let candidate = display_framebuffer_candidate(owner, base, size, width, height, pitch);
638    let mut framebuffer = DISPLAY_FRAMEBUFFER.lock();
639    match *framebuffer {
640        DisplayFramebufferState::Empty => {
641            *framebuffer = DisplayFramebufferState::Pending(candidate);
642            Ok(DisplayFramebufferReservation::Reserved)
643        }
644        DisplayFramebufferState::Active(existing) if existing == candidate => {
645            Ok(DisplayFramebufferReservation::ActiveReplay)
646        }
647        DisplayFramebufferState::Pending(_) | DisplayFramebufferState::Active(_) => {
648            Err(DisplayFramebufferReservationError::Conflict)
649        }
650    }
651}
652
653/// Commit a previously reserved mapping after every page was installed.
654pub fn activate_display_framebuffer(
655    owner: CellId,
656    base: usize,
657    size: usize,
658    width: u16,
659    height: u16,
660    pitch: usize,
661) -> bool {
662    let candidate = display_framebuffer_candidate(owner, base, size, width, height, pitch);
663    let mut framebuffer = DISPLAY_FRAMEBUFFER.lock();
664    if *framebuffer == DisplayFramebufferState::Pending(candidate) {
665        *framebuffer = DisplayFramebufferState::Active(candidate);
666        true
667    } else {
668        false
669    }
670}
671
672/// Cancel a reservation when preflight or mapping fails before activation.
673pub fn cancel_display_framebuffer(
674    owner: CellId,
675    base: usize,
676    size: usize,
677    width: u16,
678    height: u16,
679    pitch: usize,
680) {
681    let candidate = display_framebuffer_candidate(owner, base, size, width, height, pitch);
682    let mut framebuffer = DISPLAY_FRAMEBUFFER.lock();
683    if *framebuffer == DisplayFramebufferState::Pending(candidate) {
684        *framebuffer = DisplayFramebufferState::Empty;
685    }
686}
687
688/// Return the active firmware scanout geometry, if registered.
689pub fn display_framebuffer_resolution() -> Option<(u16, u16)> {
690    match *DISPLAY_FRAMEBUFFER.lock() {
691        DisplayFramebufferState::Active(framebuffer) => {
692            Some((framebuffer.width, framebuffer.height))
693        }
694        DisplayFramebufferState::Empty | DisplayFramebufferState::Pending(_) => None,
695    }
696}

698/// Current number of registered regions (diagnostics).
699pub fn region_count() -> usize {
700    REGISTRY.lock().len()
701}

742// ---------------------------------------------------------------------------
743// PCIe BDF ownership (for sys_grant_dma authorization)
744// ---------------------------------------------------------------------------
745
746/// Maps PCIe BDF → owning task ID.
747///
748/// Kernel drivers (NIC, NVMe) are not registered here — they bypass `sys_grant_dma`
749/// and call `iommu::map_dma_for_cell(0, bdf, ...)` directly during init.
750/// Only userspace Driver Cells that receive PCIe device ownership via capability
751/// delegation need to register here.
752static BDF_OWNERS: Spinlock<alloc::collections::BTreeMap<u32, usize>> =
753    Spinlock::new(alloc::collections::BTreeMap::new());
754
755/// Atomically claim an unowned PCIe BDF for task `tid`.
756///
757/// Repeated claims by the same task are idempotent. A live owner cannot be
758/// displaced; task reaping releases its claims through [`release_bdfs_for`].
759pub fn claim_bdf_owner(bdf: u32, tid: usize) -> bool {
760    let mut owners = BDF_OWNERS.lock();
761    match owners.get(&bdf).copied() {
762        Some(owner) => owner == tid,
763        None => {
764            owners.insert(bdf, tid);
765            true
766        }
767    }
768}
769
770/// Return the task ID that currently owns `bdf`, or `None` if unowned.
771pub fn owner_of_bdf(bdf: u32) -> Option<usize> {
772    BDF_OWNERS.lock().get(&bdf).copied()
773}
774
775/// Release all BDF ownerships held by task `tid` (called on Cell exit).
776pub fn release_bdfs_for(tid: usize) {
777    BDF_OWNERS.lock().retain(|_bdf, &mut owner| owner != tid);
778}
779
780/// Force-unlock BDF_OWNERS during fault teardown.
781///
782/// # Safety
783/// See `force_unlock_locks` above — same contract.
784pub unsafe fn force_unlock_bdf_locks() {
785    BDF_OWNERS.force_unlock();
786}
