40// ── VM entry ──────────────────────────────────────────────────────────────────
41
42#[cfg(target_arch = "aarch64")]
43struct Vm {
44    stage2: Stage2Table,
45    guest_pa: u64,
46    guest_pages: usize,
47    vcpus: Vec<AArch64Vcpu>,
48    /// Per-vCPU pending virtual IRQ set; intids set by inject_irq, drained into
49    /// GICH LRs just before each run_vcpu_impl call (Phase 09). Fixed-size
50    /// coalescing bitset — see `pending_irqs::PendingIrqs` for why this isn't
51    /// a queue.
52    vcpu_irqs: Vec<PendingIrqs>,
53    // reason: the Pi monitor's guest entry passes this to `monitor::run`; on the
54    // other AArch64 hosts the local `vmid` still carries the Stage-2 activation,
55    // so the field stays reserved for VM introspection until they read it too.
56    #[allow(dead_code)]
57    vmid: u16,
58    /// Set once the guest RAM window has been cleaned to PoC and the I-cache
59    /// invalidated, before the first vCPU entry.
60    entry_flushed: bool,
61}
62
63// VM_REGISTRY is keyed by (owner_tid, vm_id).
64// vm_id is assigned sequentially per owner; starts at 1.
65#[cfg(target_arch = "aarch64")]
66static VM_REGISTRY: Spinlock<Option<BTreeMap<(usize, usize), Vm>>> = Spinlock::new(None);
67
68#[cfg(target_arch = "aarch64")]
69static NEXT_VMID: core::sync::atomic::AtomicU16 = core::sync::atomic::AtomicU16::new(1);
70
71#[cfg(target_arch = "aarch64")]
72fn registry_lock() -> &'static Spinlock<Option<BTreeMap<(usize, usize), Vm>>> {
73    &VM_REGISTRY
74}

76// ── Sequential vm_id counter per owner ───────────────────────────────────────

78/// Per-owner sequential VM-id counter, stored alongside each owner's first VM.
79/// Simple: we just use the total registered VM count + 1 as the next id.
80// reason: kept for near-future VM lifecycle refactor (currently `create_vm`
81// inlines equivalent logic); not yet wired up as a callable helper.
82#[allow(dead_code)]
83#[cfg(target_arch = "aarch64")]
84fn next_vm_id_for(owner: usize) -> usize {
85    let guard = registry_lock().lock();
86    let count = guard
87        .as_ref()
88        .map_or(0, |m| m.keys().filter(|(o, _)| *o == owner).count());
89    count + 1
90}

94/// Allocate guest RAM + Stage-2 table; return opaque `vm_id`.
95pub fn create_vm(owner: usize, guest_pages: usize) -> ViResult<usize> {
96    #[cfg(target_arch = "aarch64")]
97    {
98        #[cfg(feature = "board-rpi3")]
99        if !hal::aarch64::monitor::is_verified() {
100            // The cell can only print one sentence for every failure, so the
101            // reason has to be here: a refused boot monitor and a failed carve
102            // are different bugs with the same symptom on the console.
103            log::warn!(
104                "[hv] create_vm refused: EL2 monitor not verified (ready={})",
105                hal::aarch64::monitor::is_ready()
106            );
107            return Err(ViError::NotSupported);
108        }
109        use crate::memory::paging::PAGE_SIZE;
110
111        let mut table = Stage2Table::new().ok_or_else(|| {
112            log::error!("[hv] create_vm: stage-2 root allocation failed");
113            ViError::OutOfMemory
114        })?;
115        let guest_pa = table
116            .carve_guest_ram(guest_pages)
117            .ok_or_else(|| {
118                let (total_mib, used_mib, largest_mib) = {
119                    let frames = crate::memory::frame::FRAME_ALLOCATOR.lock();
120                    frames
121                        .as_ref()
122                        .map(|allocator| {
123                            (
124                                allocator.total_memory() / (1024 * 1024),
125                                allocator.used_memory() / (1024 * 1024),
126                                allocator.largest_free_run() / 256,
127                            )
128                        })
129                        .unwrap_or((0, 0, 0))
130                };
131                log::error!(
132                    "[hv] create_vm: no contiguous guest run ({} pages, {} MiB; allocator {} MiB total, {} MiB used, largest free run {} MiB)",
133                    guest_pages,
134                    guest_pages / 256,
135                    total_mib,
136                    used_mib,
137                    largest_mib
138                );
139                ViError::OutOfMemory
140            })?;
141        // Map all guest RAM at IPA 0x40000000.
142        table
143            .map(0x4000_0000, guest_pa, guest_pages, true)
144            .map_err(|error| {
145                log::error!("[hv] create_vm: guest RAM stage-2 map failed: {:?}", error);
146                ViError::OutOfMemory
147            })?;
148        // Only QEMU virt supplies GICV/GICH. Pi GICC/GICD are software-MMIO
149        // devices and must remain unmapped Stage-2 holes.
150        #[cfg(not(feature = "board-rpi3"))]
151        table
152            .map_mmio_passthrough(0x0801_0000, 0x0804_0000, 16, false)
153            .map_err(|error| {
154                log::error!("[hv] create_vm: GICV stage-2 map failed: {:?}", error);
155                ViError::OutOfMemory
156            })?;
157
158        let vmid = NEXT_VMID.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
159        // SAFETY: table built and flushed; vmid ≥ 1; not yet active (enable later).
160        unsafe {
161            enable_stage2(vmid, table.root_pa());
162        }
163        #[cfg(not(feature = "board-rpi3"))]
164        unsafe {
165            vgic::enable();
166        }
167
168        let vm_id = {
169            let mut guard = registry_lock().lock();
170            if guard.is_none() {
171                *guard = Some(BTreeMap::new());
172            }
173            let map = guard.as_mut().unwrap();
174            let id = map.keys().filter(|(o, _)| *o == owner).count() + 1;
175            map.insert(
176                (owner, id),
177                Vm {
178                    stage2: table,
179                    guest_pa,
180                    guest_pages,
181                    vcpus: Vec::new(),
182                    vcpu_irqs: Vec::new(),
183                    vmid,
184                    entry_flushed: false,
185                },
186            );
187            let _ = PAGE_SIZE; // suppress unused warning
188            id
189        };
190        Ok(vm_id)
191    }
192    #[cfg(target_arch = "x86_64")]
193    {
194        super::svm_registry::create_vm(owner, guest_pages)
195    }
196    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
197    {
198        let _ = (owner, guest_pages);
199        Err(ViError::NotSupported)
200    }
201}

203/// Create a vCPU in `vm_id` with initial PC `entry_pc`; return `vcpu_id` (1-based).
204///
205/// Under `test-hooks`, writes a P04 HVC smoke blob (`MOVZ X0,#42; HVC #0; B .`)
206/// to the page containing `entry_pc` so the test cell does not need userspace
207/// memory access to guest RAM.
208pub fn create_vcpu(owner: usize, vm_id: usize, entry_pc: u64) -> ViResult<usize> {
209    #[cfg(target_arch = "aarch64")]
210    {
211        let mut guard = registry_lock().lock();
212        let map = guard.as_mut().ok_or(ViError::NotFound)?;
213        let vm = map.get_mut(&(owner, vm_id)).ok_or(ViError::NotFound)?;
214        let vcpu_id = vm.vcpus.len() + 1;
215
216        // test-hooks: write P04 HVC smoke blob so the test cell can verify Hvc exit.
217        #[cfg(feature = "test-hooks")]
218        {
219            const MOVZ_X0_42: u32 = 0xD280_0540; // MOVZ X0, #42
220            const HVC_0: u32 = 0xD400_0002; // HVC #0
221            const B_DOT: u32 = 0x1400_0000; // B .
222            const GUEST_IPA_BASE: u64 = 0x4000_0000;
223            let offset = (entry_pc - GUEST_IPA_BASE) as usize;
224            let blob_pa = vm.guest_pa as usize + offset;
225            // SAFETY: guest RAM is kernel-allocated identity-mapped memory; no active vCPU yet.
226            unsafe {
227                let ptr = blob_pa as *mut u32;
228                ptr.write(MOVZ_X0_42);
229                ptr.add(1).write(HVC_0);
230                ptr.add(2).write(B_DOT);
231            }
232        }
233
234        vm.vcpus.push(AArch64Vcpu::new(entry_pc));
235        vm.vcpu_irqs.push(PendingIrqs::new());
236        Ok(vcpu_id)
237    }
238    #[cfg(target_arch = "x86_64")]
239    {
240        super::svm_registry::create_vcpu(owner, vm_id, entry_pc)
241    }
242    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
243    {
244        let _ = (owner, vm_id, entry_pc);
245        Err(ViError::NotSupported)
246    }
247}

249/// Map guest IPA range in `vm_id`'s Stage-2.
250pub fn map_guest_memory(
251    owner: usize,
252    vm_id: usize,
253    ipa: u64,
254    size: usize,
255    writable: bool,
256) -> ViResult<()> {
257    #[cfg(target_arch = "aarch64")]
258    {
259        use crate::memory::paging::PAGE_SIZE;
260        let pages = size.div_ceil(PAGE_SIZE);
261        let mut guard = registry_lock().lock();
262        let map = guard.as_mut().ok_or(ViError::NotFound)?;
263        let vm = map.get_mut(&(owner, vm_id)).ok_or(ViError::NotFound)?;
264        // Extend guest RAM mapping to cover the requested IPA range.
265        vm.stage2
266            .map(ipa, vm.guest_pa, pages, writable)
267            .map_err(|_| ViError::OutOfMemory)?;
268        Ok(())
269    }
270    #[cfg(target_arch = "x86_64")]
271    {
272        super::svm_registry::map_guest_memory(owner, vm_id, ipa, size, writable)
273    }
274    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
275    {
276        let _ = (owner, vm_id, ipa, size, writable);
277        Err(ViError::NotSupported)
278    }
279}

281/// World-switch into vCPU; write `ViVmExit` to `exit_out`.
282///
283/// # Safety
284/// `exit_out` must point to a valid, writable `ViVmExit`-sized buffer in the
285/// caller's address space.  Validated by the syscall layer before this call.
286pub unsafe fn run_vcpu(
287    owner: usize,
288    vm_id: usize,
289    vcpu_id: usize,
290    _budget_ns: u64,
291    exit_out: *mut api::hypervisor::ViVmExit,
292) -> ViResult<usize> {
293    #[cfg(target_arch = "aarch64")]
294    {
295        let hal_exit = {
296            let mut guard = registry_lock().lock();
297            let map = guard.as_mut().ok_or(ViError::NotFound)?;
298            let vm = map.get_mut(&(owner, vm_id)).ok_or(ViError::NotFound)?;
299            let vcpu_idx = vcpu_id.saturating_sub(1);
300
301            // Pi's software GICC requests at most one edge per entry. Consume
302            // one request now, assert HCR_EL2.VI only during this guest run and
303            // clear VI in the EL2 exit trampoline before the host resumes.
304            #[cfg(feature = "board-rpi3")]
305            let mut virtual_irq = vm
306                .vcpu_irqs
307                .get_mut(vcpu_idx)
308                .and_then(PendingIrqs::take_lowest)
309                .is_some();
310            #[cfg(not(feature = "board-rpi3"))]
311            let mut num_loaded = 0usize;
312            #[cfg(not(feature = "board-rpi3"))]
313            if let Some(q) = vm.vcpu_irqs.get_mut(vcpu_idx) {
314                while num_loaded < vgic::MAX_LRS {
315                    let Some(intid) = q.take_lowest() else {
316                        break;
317                    };
318                    // SAFETY: EL2; GICH MMIO at 0x0803_0000; num_loaded < MAX_LRS.
319                    unsafe {
320                        vgic::load_lr(num_loaded, intid);
321                    }
322                    num_loaded += 1;
323                }
324            }
325
326            // ── Guest image: make it visible to the guest's own view ────────────
327            //
328            // The cell streams the guest kernel/initrd/DTB into guest RAM through
329            // `write_guest_memory`, which copies through the kernel's mapping; the
330            // guest fetches those pages through Stage-2, a different VA and ASID.
331            // Cache maintenance by VA does not cover that alias, so the first
332            // entry cleans the whole window to the point of coherency and drops
333            // the instruction cache. Once per VM: the streaming happens before
334            // the first run, and the guest owns the window afterwards.
335            if !vm.entry_flushed {
336                #[cfg(target_arch = "aarch64")]
337                {
338                    let start = vm.guest_pa as usize;
339                    let len = vm.guest_pages * crate::memory::paging::PAGE_SIZE;
340                    hal::aarch64::cache::clean_data_cache_range(start, len);
341                    hal::aarch64::cache::invalidate_instruction_cache_all();
342                }
343                vm.entry_flushed = true;
344            }
345
346            // ── World-switch into guest ──────────────────────────────────────────
347            let exit = {
348                let vcpu = vm.vcpus.get_mut(vcpu_idx).ok_or(ViError::NotFound)?;
349
350                // Resolve guest ID_AA64* reads (trapped by HCR_EL2.TID3) here:
351                // `ViVmExit::SysReg` cannot carry a return value. When the
352                // bounded batch ends, retry the faulting MRS on the next run
353                // instead of fabricating a CPU feature value.
354                const MAX_ID_REG_RESOLVES: u32 = 64;
355                let mut resolved = 0u32;
356                let exit = loop {
357                    #[cfg(feature = "board-rpi3")]
358                    let exit = {
359                        let vi = core::mem::replace(&mut virtual_irq, false);
360                        unsafe {
361                            hal::aarch64::monitor::run(vcpu, vm.vmid, vm.stage2.root_pa(), vi);
362                        }
363                        vcpu.decode_exit()
364                    };
365                    #[cfg(not(feature = "board-rpi3"))]
366                    let exit = unsafe { run_vcpu_impl(vcpu) };
367                    if let HalVmExit::SysReg {
368                        op0,
369                        op1,
370                        crn,
371                        crm,
372                        op2,
373                        rt,
374                        is_write,
375                    } = exit
376                    {
377                        if !is_write {
378                            if let Some(val) = read_trapped_id_reg(op0, op1, crn, crm, op2) {
379                                if resolved == MAX_ID_REG_RESOLVES {
380                                    break HalVmExit::Preempted;
381                                }
382                                if (rt as usize) < 31 {
383                                    vcpu.gp[rt as usize] = val;
384                                }
385                                vcpu.g_elr_el2 = vcpu.exit_elr.wrapping_add(4);
386                                resolved += 1;
387                                continue;
388                            }
389                        }
390                    }
391                    break exit;
392                };
393                // Unhandled guest trap: dump the guest's own EL1 exception bank.
394                // After a guest-internal exception these carry the ORIGINAL
395                // syndrome (the EL2 exit only sees the follow-on vector-fetch
396                // fault), which is the difference between a diagnosable failure
397                // and "unknown vmexit".
398                if let HalVmExit::Unknown { ec, iss } = exit {
399                    log::warn!(
400                        "[hv] unhandled guest trap ec={:#x} iss={:#x} | guest ELR_EL1={:#x} ESR_EL1={:#x} FAR_EL1={:#x} VBAR_EL1={:#x} SCTLR_EL1={:#x} SPSR_EL1={:#x}",
401                        ec, iss,
402                        vcpu.g_elr_el1, vcpu.g_esr_el1, vcpu.g_far_el1,
403                        vcpu.g_vbar_el1, vcpu.g_sctlr_el1, vcpu.g_spsr_el1,
404                    );
405                    log::warn!(
406                        "[hv]   guest TCR_EL1={:#x} TTBR0_EL1={:#x} TTBR1_EL1={:#x} MAIR_EL1={:#x}",
407                        vcpu.g_tcr_el1,
408                        vcpu.g_ttbr0_el1,
409                        vcpu.g_ttbr1_el1,
410                        vcpu.g_mair_el1,
411                    );
412                }
413                exit
414                // vcpu borrow ends here (NLL + nested block)
415            };
416
417            // ── Phase 09: drain GICH LRs after exit ─────────────────────────────
418            // Re-mark pending any LRs still in Active state (guest was preempted
419            // mid-handling). SAFETY: no vCPU running; EL2; GICH MMIO accessible.
420            #[cfg(not(feature = "board-rpi3"))]
421            if num_loaded > 0 {
422                let elrsr = unsafe { vgic::read_elrsr() };
423                for n in 0..num_loaded {
424                    if (elrsr >> n) & 1 == 0 {
425                        // LR occupied — re-mark pending if Active or Pending+Active.
426                        let lr_val = unsafe { vgic::read_lr(n) };
427                        if (lr_val >> 28) & 3 != 0 {
428                            if let Some(q) = vm.vcpu_irqs.get_mut(vcpu_idx) {
429                                q.set(lr_val & 0x3FF);
430                            }
431                        }
432                    }
433                    unsafe {
434                        vgic::clear_lr(n);
435                    }
436                }
437            }
438
439            exit
440        };
441
442        // Convert HAL ViVmExit → API ViVmExit (same fields, different crate paths).
443        let api_exit = match hal_exit {
444            HalVmExit::MmioRead { ipa, size, reg } => ApiVmExit::MmioRead { ipa, size, reg },
445            HalVmExit::MmioWrite { ipa, size, val } => ApiVmExit::MmioWrite { ipa, size, val },
446            HalVmExit::Hvc { imm, regs } => ApiVmExit::Hvc { imm, regs },
447            HalVmExit::Wfi => ApiVmExit::Wfi,
448            HalVmExit::SysReg {
449                op0,
450                op1,
451                crn,
452                crm,
453                op2,
454                rt,
455                is_write,
456            } => ApiVmExit::SysReg {
457                op0,
458                op1,
459                crn,
460                crm,
461                op2,
462                rt,
463                is_write,
464            },
465            HalVmExit::Preempted => ApiVmExit::Preempted,
466            HalVmExit::Shutdown => ApiVmExit::Shutdown,
467            HalVmExit::Unknown { ec, iss } => ApiVmExit::Unknown { ec, iss },
468            // x86-only exits (SVM/VT-x) never arise on the aarch64 world-switch.
469            HalVmExit::PortIn { .. }
470            | HalVmExit::PortOut { .. }
471            | HalVmExit::Hlt
472            | HalVmExit::Msr { .. } => ApiVmExit::Unknown { ec: 0, iss: 0 },
473        };
474        // SAFETY: exit_out validated by syscall layer.
475        unsafe {
476            core::ptr::write(exit_out, api_exit);
477        }
478        Ok(0)
479    }
480    #[cfg(target_arch = "x86_64")]
481    {
482        use api::hypervisor::ViVmExit as ApiVmExit;
483        use hal::ViVmExit as HalVmExit;
484        let hal_exit = super::svm_registry::run_vcpu_hal(owner, vm_id, vcpu_id)?;
485        // HAL → API conversion (VERSION 2 ABI — the x86 variants are frozen at
486        // discriminants 8-11). `reg` is not decoded for PortIn (guest `IN`
487        // always targets (E)AX) → 0.
488        let api_exit = match hal_exit {
489            HalVmExit::MmioRead { ipa, size, reg } => ApiVmExit::MmioRead { ipa, size, reg },
490            HalVmExit::MmioWrite { ipa, size, val } => ApiVmExit::MmioWrite { ipa, size, val },
491            HalVmExit::Preempted => ApiVmExit::Preempted,
492            HalVmExit::Shutdown => ApiVmExit::Shutdown,
493            HalVmExit::Unknown { ec, iss } => ApiVmExit::Unknown { ec, iss },
494            HalVmExit::PortIn { port, size } => ApiVmExit::PortIn { port, size, reg: 0 },
495            HalVmExit::PortOut { port, size, val } => ApiVmExit::PortOut { port, size, val },
496            HalVmExit::Hlt => ApiVmExit::Hlt,
497            HalVmExit::Msr {
498                index,
499                is_write,
500                value,
501            } => ApiVmExit::Msr {
502                index,
503                is_write,
504                val: value,
505            },
506            // ARM-only HAL variants — unreachable on x86 (no aarch64 exits here).
507            HalVmExit::Hvc { .. } | HalVmExit::Wfi | HalVmExit::SysReg { .. } => {
508                ApiVmExit::Unknown { ec: 0, iss: 0 }
509            }
510        };
511        // SAFETY: exit_out validated by the syscall layer.
512        unsafe {
513            core::ptr::write(exit_out, api_exit);
514        }
515        Ok(0)
516    }
517    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
518    {
519        let _ = (owner, vm_id, vcpu_id, _budget_ns, exit_out);
520        Err(ViError::NotSupported)
521    }
522}

524/// Modes 0/1 read/write 32 guest GP-register words. On Pi mode 2 reads a
525/// 32-word timer snapshot (CNTV_CTL_EL0, CNTV_CVAL_EL0, then thirty zeros).
526pub fn vcpu_regs(
527    owner: usize,
528    vm_id: usize,
529    vcpu_id: usize,
530    buf_ptr: usize,
531    mode: usize,
532) -> ViResult<usize> {
533    #[cfg(target_arch = "aarch64")]
534    {
535        let mut guard = registry_lock().lock();
536        let map = guard.as_mut().ok_or(ViError::NotFound)?;
537        let vm = map.get_mut(&(owner, vm_id)).ok_or(ViError::NotFound)?;
538        let vcpu = vm
539            .vcpus
540            .get_mut(vcpu_id.saturating_sub(1))
541            .ok_or(ViError::NotFound)?;
542        // buf_ptr points to 32×u64 (256 bytes), validated by syscall layer.
543        // SAFETY: buf_ptr validated; SAS — same VA in kernel and cell.
544        let buf = unsafe { core::slice::from_raw_parts_mut(buf_ptr as *mut u64, 32) };
545        match mode {
546            1 => {
547                for (i, v) in buf[..31].iter().enumerate() {
548                    vcpu.gp[i] = *v;
549                }
550                vcpu.g_elr_el2 = buf[31];
551            }
552            0 => {
553                for (i, v) in vcpu.gp.iter().enumerate() {
554                    buf[i] = *v;
555                }
556                buf[31] = vcpu.g_elr_el2;
557            }
558            #[cfg(feature = "board-rpi3")]
559            2 => {
560                buf.fill(0);
561                buf[0] = vcpu.g_cntv_ctl;
562                buf[1] = vcpu.g_cntv_cval;
563            }
564            _ => return Err(ViError::InvalidArgument),
565        }
566        Ok(0)
567    }
568    #[cfg(target_arch = "x86_64")]
569    {
570        if mode > 1 {
571            return Err(ViError::InvalidArgument);
572        }
573        super::svm_registry::vcpu_regs(owner, vm_id, vcpu_id, buf_ptr, mode == 1)
574    }
575    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
576    {
577        let _ = (owner, vm_id, vcpu_id, buf_ptr, mode);
578        Err(ViError::NotSupported)
579    }
580}

582/// Copy `len` bytes from caller's `src_ptr` into guest physical RAM at `gpa`.
583///
584/// # Preconditions (enforced by caller / syscall layer)
585/// - `src_ptr + len` is within the caller cell's valid address range (via `validate_user_buf`).
586/// - `gpa + len` does not wrap (overflow guard in syscall layer).
587///
588/// # Safety (kernel-internal)
589/// `src_ptr` is a valid cell VA; in SAS, VA == PA for kernel-managed regions, but
590/// the copy uses `copy_nonoverlapping` which only reads the source — no guest access.
591pub fn write_guest_memory(
592    owner: usize,
593    vm_id: usize,
594    gpa: u64,
595    src_ptr: usize,
596    len: usize,
597) -> ViResult<usize> {
598    #[cfg(target_arch = "aarch64")]
599    {
600        use crate::memory::paging::PAGE_SIZE;
601        const GUEST_IPA_BASE: u64 = 0x4000_0000;
602
603        let guard = registry_lock().lock();
604        let map = guard.as_ref().ok_or(ViError::NotFound)?;
605        let vm = map.get(&(owner, vm_id)).ok_or(ViError::NotFound)?;
606
607        // Validate gpa is within the mapped guest-RAM window.
608        let offset = gpa
609            .checked_sub(GUEST_IPA_BASE)
610            .ok_or(ViError::InvalidInput)? as usize;
611        let end = offset.checked_add(len).ok_or(ViError::InvalidInput)?;
612        if end > vm.guest_pages * PAGE_SIZE {
613            return Err(ViError::InvalidInput);
614        }
615
616        // SAFETY: guest RAM is kernel-allocated identity-mapped memory.
617        // src_ptr validated by syscall layer (validate_user_buf); SAS means it's
618        // also accessible here. No active vCPU reads this region while we copy
619        // (the caller holds no vcpu run in progress — that would require RunVcpu,
620        // which cannot be concurrent in a single-task cell).
621        unsafe {
622            let dst = (vm.guest_pa as usize + offset) as *mut u8;
623            let src = src_ptr as *const u8;
624            core::ptr::copy_nonoverlapping(src, dst, len);
625        }
626        Ok(len)
627    }
628    #[cfg(target_arch = "x86_64")]
629    {
630        super::svm_registry::write_guest_memory(owner, vm_id, gpa, src_ptr, len)
631    }
632    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
633    {
634        let _ = (owner, vm_id, gpa, src_ptr, len);
635        Err(ViError::NotSupported)
636    }
637}

697/// Queue one virtual IRQ request for the next vCPU run. On Pi the monitor
698/// asserts HCR_EL2.VI for that run only; its exit path clears VI before returning
699/// to the EL1 host. The software GICC owns enabled/pending/active state.
700pub fn inject_irq(owner: usize, vm_id: usize, vcpu_id: usize, intid: u32) -> ViResult<usize> {
701    #[cfg(target_arch = "aarch64")]
702    {
703        let mut guard = registry_lock().lock();
704        let map = guard.as_mut().ok_or(ViError::NotFound)?;
705        let vm = map.get_mut(&(owner, vm_id)).ok_or(ViError::NotFound)?;
706        let idx = vcpu_id.saturating_sub(1);
707        if let Some(q) = vm.vcpu_irqs.get_mut(idx) {
708            q.set(intid);
709        }
710        Ok(0)
711    }
712    #[cfg(target_arch = "x86_64")]
713    {
714        // `intid` is reinterpreted as an x86 interrupt vector (8259 line remap).
715        super::svm_registry::inject_irq(owner, vm_id, vcpu_id, intid)?;
716        Ok(0)
717    }
718    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
719    {
720        let _ = (owner, vm_id, vcpu_id, intid);
721        Ok(0)
722    }
723}

725// ── Teardown — called on every task-exit path ─────────────────────────────────
726
727/// Reclaim all VMs and guest RAM owned by `dead_tid`.
728///
729/// Called alongside `reap_grants_for_task` on task exit, fault, and watchdog kill.
730/// Lock order: VM_REGISTRY → FRAME_ALLOCATOR (same as grant reaper).
731pub fn reap_vms_for_task(dead_tid: usize) {
732    #[cfg(target_arch = "aarch64")]
733    {
734        // Collect entries to drop outside the lock (Stage2Table::drop frees frames).
735        let dead_vms: alloc::vec::Vec<Vm> = {
736            let mut guard = registry_lock().lock();
737            let Some(map) = guard.as_mut() else { return };
738            let dead_keys: alloc::vec::Vec<(usize, usize)> = map
739                .keys()
740                .filter(|(o, _)| *o == dead_tid)
741                .copied()
742                .collect();
743            dead_keys.iter().filter_map(|k| map.remove(k)).collect()
744        };
745        // Disable Stage-2 for each dying VM before dropping the table.
746        for vm in dead_vms {
747            // SAFETY: no vCPU is running (task is dead); safe to disable Stage-2.
748            unsafe {
749                disable_stage2();
750            }
751            drop(vm); // Stage2Table::drop frees all frames
752        }
753    }
754    #[cfg(target_arch = "x86_64")]
755    {
756        super::svm_registry::reap_vms_for_task(dead_tid);
757    }
758    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
759    {
760        let _ = dead_tid;
761    }
762}

764/// Pi's EL1-host capability gate: an actual monitor round-trip must run guest
765/// HVC, WFI, MMIO and software-injected virtual IRQ before any VM is admitted.
766#[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
767pub fn pi_monitor_smoke() {
768    use crate::memory::frame::phys_to_virt;
769    use hal::aarch64::{
770        cache::{clean_data_cache_range, invalidate_instruction_cache_all, sync_instruction_cache},
771        monitor,
772    };
773
774    if !monitor::is_ready() {
775        log::warn!(
776            "[pi-monitor] entry EL={} HVC init unavailable; HypervisorCap closed",
777            monitor::entry_el()
778        );
779        return;
780    }
781    const IPA: u64 = 0x4000_0000;
782    const VMID: u16 = 0xfffe;
783    let Some(mut table) = Stage2Table::new() else {
784        log::warn!("[pi-monitor] smoke: no Stage-2 root; HypervisorCap closed");
785        return;
786    };
787    let Some(guest_pa) = table.carve_guest_ram(1) else {
788        log::warn!("[pi-monitor] smoke: no guest page; HypervisorCap closed");
789        return;
790    };
791    if table.map(IPA, guest_pa, 1, true).is_err() {
792        log::warn!("[pi-monitor] smoke: guest map failed; HypervisorCap closed");
793        return;
794    }
795    let page = phys_to_virt(guest_pa as usize) as *mut u32;
796    // MOVZ X0,#42; HVC #0; WFI; MOVZ X0,#0x900,LSL#16;
797    // STR X1,[X0]; HVC #0. The final HVC bounds a failure if MMIO did not
798    // trap instead of leaving an unbounded guest loop.
799    let blob = [
800        0xD280_0540,
801        0xD400_0002,
802        0xD503_201F,
803        0xD2A1_2000,
804        0xF900_0001,
805        0xD400_0002,
806    ];
807    unsafe {
808        core::ptr::copy_nonoverlapping(blob.as_ptr(), page, blob.len());
809        // EL1h IRQ vector = VBAR_EL1+0x280; HVC reports that vector was reached.
810        page.add(0x280 / 4).write(0xD400_0002);
811        // +0x300: `B .` — a guest that never yields, for the preemption step.
812        page.add(0x300 / 4).write(0x1400_0000);
813        sync_instruction_cache(page as usize, page as usize, 4096);
814        // The guest fetches this page through Stage-2, i.e. a different VA and
815        // ASID, so the per-VA invalidate above does not cover the alias. Clean
816        // the page to the point of coherency and drop the whole instruction
817        // cache: on real hardware (not in TCG, which has no caches) a guest that
818        // fetched a stale alias faulted at its first instruction while EL2's own
819        // read of the same page showed the correct blob.
820        clean_data_cache_range(page as usize, 4096);
821        invalidate_instruction_cache_all();
822    }
823    let mut vcpu = AArch64Vcpu::new(IPA);
824    // These markers distinguish an unobserved EL2 write from a genuine trap
825    // whose syndrome and PC happen to be zero.
826    vcpu.exit_esr = 0x534d_4f4b_4553_52;
827    vcpu.exit_elr = 0x534d_4f4b_4550_43;
828    // Boot has DAIF.I set: defer the existing, still-routed physical timers
829    // for a bounded 100 ms guest smoke window. Otherwise their pre-boot
830    // pending IRQs can turn a conditional WFI into a NOP. Production vCPU
831    // runs retain the regular 10 ms host-tick preemption.
832    let prepare = hal::aarch64::timer::prepare_monitor_smoke_window;
833    // A physical tick can still race guest entry. Retry only genuine
834    // preemptions, bounded: a permanently asserted unrelated IRQ fails closed.
835    let run_step = |cpu: &mut AArch64Vcpu, vi: bool| {
836        for _ in 0..16 {
837            prepare();
838            unsafe {
839                monitor::run(cpu, VMID, table.root_pa(), vi);
840            }
841            if !matches!(cpu.decode_exit(), HalVmExit::Preempted) {
842                return;
843            }
844        }
845    };
846    let passed = {
847        log::info!(
848            "[pi-monitor] first-run EL1 entry={:#x} insn={:#x} esr-marker={:#x} elr-marker={:#x}",
849            vcpu.g_elr_el2,
850            unsafe { core::ptr::read_volatile(page) },
851            vcpu.exit_esr,
852            vcpu.exit_elr
853        );
854        prepare();
855        let first =
856            unsafe { monitor::run_probed(&mut vcpu, VMID, table.root_pa(), false, IPA, guest_pa) };
857        log::info!(
858            "[pi-monitor] first-run root={:#x} page={:#x} EL2 entry={:#x} s2=[{:#x},{:#x},{:#x}] insn={:#x} exit=[{:#x},{:#x},irq={}] EL1 exit=[{:#x},{:#x},irq={}]",
859            table.root_pa(), guest_pa, first.entry_pc, first.stage2_l1,
860            first.stage2_l2, first.stage2_l3, first.instruction,
861            first.exit_esr, first.exit_elr, first.exit_is_irq,
862            vcpu.exit_esr, vcpu.exit_elr, vcpu.exit_is_irq
863        );
864        if matches!(vcpu.decode_exit(), HalVmExit::Preempted) {
865            run_step(&mut vcpu, false);
866        }
867        let hvc = matches!(vcpu.decode_exit(), HalVmExit::Hvc { imm: 0, regs } if regs[0] == 42);
868        if !hvc {
869            // EL1's own view of the same words EL2 reported: if they differ the
870            // two regimes disagree about memory; if they agree, the divergence is
871            // in what the *hardware* walk/fetch saw, not in what software reads.
872            let el1_root_leaf = unsafe { core::ptr::read_volatile(phys_to_virt(table.root_pa() as usize) as *const u64) };
873            let el1_insn = unsafe { core::ptr::read_volatile(page) };
874            log::warn!(
875                "[pi-monitor] HVC exit={:?} pc={:#x} esr={:#x} EL1 leaf0={:#x} EL1 insn={:#x}",
876                vcpu.decode_exit(),
877                vcpu.exit_elr,
878                vcpu.exit_esr,
879                el1_root_leaf,
880                el1_insn
881            );
882            false
883        } else {
884            // Step 2 runs the guest's WFI. Whether it produces a `Wfi` trap is
885            // emulator/hardware dependent: QEMU's AArch64 WFI helper returns
886            // before the HCR_EL2.TWI trap whenever `cpu_has_work()` is true
887            // (target/arm/tcg/op_helper.c), so under QEMU-TCG the WFI can be an
888            // in-block NOP. Measured on raspi3b: the WFI is a NOP and the guest
889            // runs straight into the MMIO store. The gate therefore requires
890            // forward progress past the WFI — either exit — and logs which one.
891            run_step(&mut vcpu, false);
892            let (mmio_ok, wfi_trapped) = match vcpu.decode_exit() {
893                HalVmExit::Wfi => {
894                    run_step(&mut vcpu, false);
895                    (
896                        matches!(
897                            vcpu.decode_exit(),
898                            HalVmExit::MmioWrite {
899                                ipa: 0x0900_0000,
900                                size: 8,
901                                ..
902                            }
903                        ),
904                        true,
905                    )
906                }
907                HalVmExit::MmioWrite {
908                    ipa: 0x0900_0000,
909                    size: 8,
910                    ..
911                } => (true, false),
912                _ => (false, false),
913            };
914            if !mmio_ok {
915                log::warn!(
916                    "[pi-monitor] WFI/MMIO exit={:?} pc={:#x} esr={:#x} far={:#x} hpfar={:#x}",
917                    vcpu.decode_exit(),
918                    vcpu.exit_elr,
919                    vcpu.exit_esr,
920                    vcpu.exit_far,
921                    vcpu.exit_hpfar
922                );
923                let soc = hal_soc_bcm27xx::BCM2837;
924                // SAFETY: both controller apertures are identity-mapped before
925                // IRQs are enabled and the smoke runs after paging activation.
926                let (
927                    basic,
928                    pending1,
929                    pending2,
930                    enable1,
931                    enable2,
932                    core_src,
933                    core_fiq,
934                    core_timers,
935                    cntp_ctl,
936                ) = unsafe {
937                    (
938                        core::ptr::read_volatile(soc.mmio.legacy_irq_base as *const u32),
939                        core::ptr::read_volatile((soc.mmio.legacy_irq_base + 0x04) as *const u32),
940                        core::ptr::read_volatile((soc.mmio.legacy_irq_base + 0x08) as *const u32),
941                        core::ptr::read_volatile((soc.mmio.legacy_irq_base + 0x10) as *const u32),
942                        core::ptr::read_volatile((soc.mmio.legacy_irq_base + 0x14) as *const u32),
943                        core::ptr::read_volatile(
944                            (soc.mmio.local_controller_base + 0x60) as *const u32,
945                        ),
946                        core::ptr::read_volatile(
947                            (soc.mmio.local_controller_base + 0x70) as *const u32,
948                        ),
949                        core::ptr::read_volatile(
950                            (soc.mmio.local_controller_base + 0x40) as *const u32,
951                        ),
952                        {
953                            let v: u64;
954                            core::arch::asm!("mrs {v}, cntp_ctl_el0",
955                                             v = out(reg) v, options(nomem, nostack));
956                            v
957                        },
958                    )
959                };
960                log::warn!(
961                    "[pi-monitor]   irq: basic={:#x} pend1={:#x} pend2={:#x} en1={:#x} en2={:#x}",
962                    basic,
963                    pending1,
964                    pending2,
965                    enable1,
966                    enable2
967                );
968                log::warn!(
969                    "[pi-monitor]   core: irq_src={:#x} fiq_src={:#x} timers={:#x} cntp_ctl={:#x}",
970                    core_src,
971                    core_fiq,
972                    core_timers,
973                    cntp_ctl
974                );
975                false
976            } else {
977                log::info!(
978                    "[pi-monitor] WFI {}",
979                    if wfi_trapped {
980                        "trapped (TWI honoured)"
981                    } else {
982                        "NOP (emulator helper)"
983                    }
984                );
985                let mut irq_cpu = AArch64Vcpu::new(IPA);
986                irq_cpu.g_vbar_el1 = IPA;
987                irq_cpu.g_spsr_el2 &= !(1 << 7); // unmask guest IRQ
988                run_step(&mut irq_cpu, true);
989                // The vIRQ must be taken before the guest's first instruction:
990                // PC becomes VBAR_EL1 + 0x280 and that vector's `HVC #0` runs.
991                // ELR_EL2 for a natively executed HVC is HVC+4, so the exit PC
992                // is IPA + 0x284 — proof the guest entered its own IRQ vector.
993                let irq = matches!(irq_cpu.decode_exit(), HalVmExit::Hvc { imm: 0, .. })
994                    && irq_cpu.exit_elr == IPA + 0x284;
995                if !irq {
996                    log::warn!(
997                        "[pi-monitor] VI exit={:?} pc={:#x} esr={:#x} spsr={:#x}",
998                        irq_cpu.decode_exit(),
999                        irq_cpu.exit_elr,
1000                        irq_cpu.exit_esr,
1001                        irq_cpu.g_spsr_el2
1002                    );
1003                    false
1004                } else {
1005                    // A guest that never yields must still be preempted by the
1006                    // host tick; otherwise a runaway guest owns the CPU.
1007                    let mut spin_cpu = AArch64Vcpu::new(IPA);
1008                    spin_cpu.g_elr_el2 = IPA + 0x300;
1009                    hal::aarch64::timer::arm_monitor_smoke_preemption();
1010                    unsafe {
1011                        monitor::run(&mut spin_cpu, VMID, table.root_pa(), false);
1012                    }
1013                    let preempted = matches!(spin_cpu.decode_exit(), HalVmExit::Preempted);
1014                    if !preempted {
1015                        log::warn!(
1016                            "[pi-monitor] PREEMPT exit={:?} pc={:#x} esr={:#x}",
1017                            spin_cpu.decode_exit(),
1018                            spin_cpu.exit_elr,
1019                            spin_cpu.exit_esr
1020                        );
1021                    }
1022                    preempted
1023                }
1024            }
1025        }
1026    };
1027    // Flush before freeing the Stage-2 frames even on a failed smoke.
1028    unsafe {
1029        disable_stage2();
1030    }
1031    // Restore the ordinary host tick cadence before Cells can run.
1032    hal::aarch64::bcm2835_systimer::init();
1033    hal::aarch64::timer::reset();
1034    if passed {
1035        monitor::mark_verified();
1036        log::info!("[pi-monitor] HVC/MMIO/VI/PREEMPT smoke PASS; HypervisorCap open");
1037    } else {
1038        log::warn!("[pi-monitor] HVC/MMIO/VI/PREEMPT smoke FAILED; HypervisorCap closed");
1039    }
1040}
