1//! VM registry — per-owner RAII store for Stage-2 tables, guest RAM, and vCPUs.
2//!
3//! A VM belongs to the Cell incarnation that created it ([`VmKey`]), never to
4//! a tid: every operation names that incarnation, and the incarnation's VMs
5//! are reaped together when its root retires ([`reap_vms_for`]).
6//!
7//! # Lock order
8//! `VM_REGISTRY` → `FRAME_ALLOCATOR` (same order as grant reaper; never reverse).
9//!
10//! # Non-aarch64
11//! All public functions return `Err(ViError::NotSupported)` so the compiler
12//! produces a complete match for the hypervisor Syscall arms on every target.
13
14use crate::memory::cell_quota::{Charge, QuotaExceeded};

22/// A VM's registry key: its owner incarnation and an id never handed out
23/// twice in a boot, so a key can never name another incarnation's VM or a
24/// VM that was reaped.
25#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
26pub(crate) struct VmKey {
27    owner: (u64, u64),
28    vm_id: usize,
29}
30
31// reason: the targets without a hypervisor (riscv64, legacy) register no VM.
32#[allow(dead_code)]
33impl VmKey {
34    pub(crate) fn new(owner: Charge, vm_id: usize) -> Self {
35        Self {
36            owner: (owner.cell().0, owner.generation()),
37            vm_id,
38        }
39    }
40
41    pub(crate) fn owned_by(self, owner: Charge) -> bool {
42        self.owner == (owner.cell().0, owner.generation())
43    }
44
45    // reason: only the p01-trace root walk reads the raw parts.
46    #[allow(dead_code)]
47    pub(crate) fn parts(self) -> (u64, usize) {
48        (self.owner.0, self.vm_id)
49    }
50}
51
52/// The next VM id. Monotonic for the boot: an id is never reused, so a stale
53/// id held by a cell names nothing.
54static NEXT_VM_ID: AtomicUsize = AtomicUsize::new(1);
55
56/// Mint the id of a VM about to be registered.
57// reason: the targets without a hypervisor (riscv64, legacy) register no VM.
58#[allow(dead_code)]
59pub(crate) fn next_vm_id() -> usize {
60    NEXT_VM_ID.fetch_add(1, Ordering::Relaxed)
61}

63/// Why `create_vm` refused. The `CreateVm` syscall returns `OutOfMemory` for
64/// [`VmCreateError::Budget`] only; every other refusal stays `NotSupported`.
65#[derive(Clone, Copy, Debug, Eq, PartialEq)]
66pub(crate) enum VmCreateError {
67    /// The owner's guest budget cannot cover the VM. Nothing was allocated
68    /// and nothing stays charged.
69    Budget,
70    /// Any other refusal: no hypervisor, a failed root, carve or map, no
71    /// VMID. What it allocated is freed and its charge refunded.
72    Refused(ViError),
73}
74
75impl From<ViError> for VmCreateError {
76    fn from(error: ViError) -> Self {
77        Self::Refused(error)
78    }
79}
80
81impl From<QuotaExceeded> for VmCreateError {
82    fn from(_: QuotaExceeded) -> Self {
83        Self::Budget
84    }
85}

89/// Allocate guest RAM + Stage-2 table for `owner`; return opaque `vm_id`. The
90/// owner's guest budget is charged before anything else, for the most frames
91/// the VM can take (`arch::vm::guest_bytes`); a refusal is
