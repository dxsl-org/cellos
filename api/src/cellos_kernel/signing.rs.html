1//! Cell binary signing — Ed25519 signature verification for spawned cell ELFs.
2//!
3//! The kernel holds only a public key (fleet trust anchor). Cell binaries are
4//! signed offline with the corresponding private key and carry the 64-byte
5//! signature in an `__ViCell_sig` ELF section.
6//!
7//! Canonical signed payload: every byte of the final ELF container except the
8//! 64 signature bytes themselves. The signature section header remains covered,
9//! so section names, offsets, types, flags, links, relocation metadata, ELF
10//! headers, program headers, and all load-affecting payload bytes are immutable.
11//! The signer first adds a zero-filled signature section, signs that stable
12//! container, then replaces only those excluded bytes.
13//!
14//! Verify-only: the kernel never signs (private key lives offline).
15
16use alloc::vec::Vec;
17use types::{ViError, ViResult};

19/// Reused buffer for the signed payload (the ELF minus its 64 signature bytes).
20///
21/// The message a verification hashes is a *transient*: it exists for one
22/// `verify_cell` call. Allocating it per spawn put two 78 KiB buffers in flight at
23/// once (the ELF read plus this copy), and at ~194 parked cells the payload's
24/// 78,760-byte request was the one that could no longer find a contiguous run
25/// (measured 2026-10-03: 289 KiB free, no 78 KiB hole — the refusal that ended the
26/// capacity sweep). One shared buffer, grown once and never shrunk, removes that
27/// per-spawn churn; verification itself is unchanged (same bytes, same order).
28static PAYLOAD_SCRATCH: crate::sync::Spinlock<Vec<u8>> =
29    crate::sync::Spinlock::ranked(Vec::new(), crate::sync::rank::PAYLOAD_SCRATCH);

97/// Posture name and the first 8 bytes of the trust anchor, for the boot log.
98pub fn trust_anchor() -> (&'static str, [u8; 8]) {
99    let posture = if cfg!(feature = "release-signing-key") {
100        "release"
101    } else if cfg!(feature = "phase02-terminal-profile") {
102        "phase02-private"
103    } else if cfg!(feature = "dev-signing-key") {
104        "dev"
105    } else {
106        "none"
107    };
108    let mut prefix = [0u8; 8];
109    prefix.copy_from_slice(&CELL_SIGNER_PUBKEY[..8]);
110    (posture, prefix)
111}

113/// The `api::cell_image` anchor class reported for an image this kernel's
114/// signature check verified. Follows the key `CELL_SIGNER_PUBKEY` actually
115/// selects (the private profile key overrides the others), not the feature
116/// list, so the class always names the key that did the verifying.
117pub const fn anchor_class() -> u8 {
118    anchor_class_for(
119        cfg!(feature = "phase02-terminal-profile"),
120        cfg!(feature = "release-signing-key"),
121        cfg!(feature = "dev-signing-key"),
122    )
123}
124
125const fn anchor_class_for(private: bool, release: bool, dev: bool) -> u8 {
126    if private {
127        api::cell_image::ANCHOR_PRIVATE
128    } else if release {
129        api::cell_image::ANCHOR_RELEASE
130    } else if dev {
131        api::cell_image::ANCHOR_DEV
132    } else {
133        // The zero placeholder key verifies nothing, so no image ever reaches
134        // the caller of this function with this value.
135        api::cell_image::ANCHOR_UNSIGNED
136    }
137}

139// VFS admits a dev-anchor `sshd` because only a dev-anchor kernel can report
140// one. A release kernel reporting `ANCHOR_DEV` would turn the public dev seed
141// into release authority, so such a build must not exist.

143    !cfg!(feature = "release-signing-key") || anchor_class() != api::cell_image::ANCHOR_DEV

163/// Extract the 64-byte Ed25519 signature from the `__ViCell_sig` ELF section.
164///
165/// Returns `None` unless exactly one in-bounds signature section has 64 bytes.
166pub fn extract_sig(elf_bytes: &[u8]) -> Option<[u8; 64]> {
167    let (offset, size) = signature_range(elf_bytes)?;
168    if size != 64 {
169        return None;
170    }
171    let mut sig = [0u8; 64];
172    sig.copy_from_slice(&elf_bytes[offset..offset + size]);
173    Some(sig)
174}

176/// Verify the Ed25519 signature of a cell ELF binary.
177///
178/// Returns `Ok(false)` on any malformed input, missing section, or signature
179/// mismatch — never panics. The signature covers every final ELF byte except
180/// its own 64-byte payload, including all relocation section headers and data.
181///
182/// Returns `Err(ViError::OutOfMemory)` when the signed payload cannot be
183/// allocated: the caller must fail closed *and* report capacity, because a
184/// verification that never ran is not a signature mismatch.
185pub fn verify_cell(elf_bytes: &[u8], sig: &[u8; 64]) -> ViResult<bool> {
186    verify_cell_with_key(elf_bytes, sig, &CELL_SIGNER_PUBKEY)
187}
188
189/// Locate the single file-backed fixed-size signature payload with the bounded
190/// ELF parser. Returning its byte range lets verification cover the table
191/// itself without creating a signature recursion.
192fn signature_range(elf_bytes: &[u8]) -> Option<(usize, usize)> {
193    const SHT_PROGBITS: u64 = 1;
194
195    let elf = crate::loader::elf_section::ElfSections::parse(elf_bytes)?;
196    let names = elf.names()?;
197    let mut signature = None;
198    for index in 0..elf.count() {
199        let section = elf.section(index)?;
200        if elf.name(names, section)? != b"__ViCell_sig" {
201            continue;
202        }
203        // Only a file-backed, ordinary data section can carry the excluded
204        // bytes. Every other section's metadata stays inside the signed input.
205        if signature.is_some() || section.kind != SHT_PROGBITS || section.size != 64 {
206            return None;
207        }
208        elf.bytes(section)?;
209        signature = Some((section.offset, section.size));
210    }
211    signature
212}

214/// Inner implementation; accepts an explicit key so `self_test` can use
215/// the precomputed test key without touching `CELL_SIGNER_PUBKEY`.
216fn verify_cell_with_key(elf_bytes: &[u8], sig: &[u8; 64], pubkey: &[u8; 32]) -> ViResult<bool> {
217    let (signature_offset, signature_size) = match signature_range(elf_bytes) {
218        Some(range @ (_, 64)) => range,
219        _ => return Ok(false),
220    };
221    if extract_sig(elf_bytes).as_ref() != Some(sig) {
222        return Ok(false);
223    }
224
225    // The signature section itself is the only excluded interval. Its header
226    // and placement stay in the signed byte stream, so no parser-selected
227    // relocation data can move outside authenticated bytes.
228    //
229    // The payload is a full copy of the ELF minus its 64 signature bytes, so it
230    // must stay fallible: `Vec::with_capacity` reaches the infallible alloc error
231    // handler, which halts the kernel instead of refusing the spawn. It is built
232    // in `PAYLOAD_SCRATCH` rather than a fresh buffer because a per-spawn 78 KiB
233    // allocation interleaved with the retained per-cell state is what fragmented
234    // the heap: at 194 parked cells this exact request (78 760 bytes for
235    // `/bin/bench-probe`) was the one that no longer found a contiguous run
236    // (289 KiB free), and it ended the capacity sweep.
237    let mut scratch = PAYLOAD_SCRATCH.lock();
238    // Retained verification workspace is shared signing state, not an
239    // exclusive object of the unpublished task being verified.
240    #[cfg(feature = "p01-trace")]
241    let _semantic = crate::memory::heap::p01_owner::creation(
242        None,
243        crate::memory::heap::p01_owner::SIGNING_SHARED,
244    );
245    let payload = &mut *scratch;
246    payload.clear();
247    payload
248        .try_reserve_exact(elf_bytes.len() - signature_size)
249        .map_err(|_| {
250            log::error!(
251                "[signing] OOM: signed payload of {} bytes",
252                elf_bytes.len() - signature_size
253            );
254            ViError::OutOfMemory
255        })?;
256    payload.extend_from_slice(&elf_bytes[..signature_offset]);
257    payload.extend_from_slice(&elf_bytes[signature_offset + signature_size..]);
258    #[cfg(feature = "p01-trace")]
259    drop(_semantic);
260    Ok(crate::ed25519::verify(pubkey, payload, sig))
261}

263/// Boot-time self-test using a precomputed RFC-style test vector.
264///
265/// Uses a separate precomputed `(pubkey, payload, sig)` triple — **no private
266/// key in the kernel**. Returns `true` iff the known-good vector verifies AND
267/// a flipped-byte payload is rejected.
268pub fn self_test() -> bool {
269    // Precomputed vector — seed [0x43]*32, payload b"CellosSigningTest".
270    // Regenerate: `python scripts/sign-cell.py --emit-test-vector`
271    const TEST_PUBKEY: [u8; 32] = [
272        0x22, 0xfc, 0x29, 0x77, 0x92, 0xf0, 0xb6, 0xff, 0xc0, 0xbf, 0xcf, 0xdb, 0x7e, 0xdb, 0x0c,
273        0x0a, 0xa1, 0x4e, 0x02, 0x5a, 0x36, 0x5e, 0xc0, 0xe3, 0x42, 0xe8, 0x6e, 0x38, 0x29, 0xcb,
274        0x74, 0xb6,
275    ];
276    const TEST_SIG: [u8; 64] = [
277        0x22, 0xf6, 0x2e, 0xba, 0x53, 0x9c, 0x66, 0xa0, 0xc1, 0xed, 0x39, 0xc8, 0x90, 0x04, 0xf8,
278        0xfc, 0x46, 0xb0, 0xe5, 0x42, 0xc9, 0x97, 0x22, 0x2d, 0x3f, 0x10, 0x17, 0xf3, 0xa4, 0x56,
279        0x67, 0x58, 0x9b, 0x49, 0x98, 0x2b, 0x4a, 0x48, 0x23, 0x11, 0x90, 0x09, 0x25, 0xe3, 0x9f,
280        0x02, 0x0b, 0x0e, 0x34, 0x70, 0x25, 0xfa, 0x10, 0xe3, 0x7e, 0xac, 0xd4, 0xb1, 0x6c, 0x66,
281        0xcf, 0x7b, 0x1e, 0x0a,
282    ];
283    const TEST_PAYLOAD: &[u8] = b"CellosSigningTest";
284
285    // Positive: known-good vector must verify.
286    if !crate::ed25519::verify(&TEST_PUBKEY, TEST_PAYLOAD, &TEST_SIG) {
287        return false;
288    }
289    // Negative: flipped byte in payload must be rejected.
290    let mut bad_payload = alloc::vec![0u8; TEST_PAYLOAD.len()];
291    bad_payload.copy_from_slice(TEST_PAYLOAD);
292    bad_payload[0] ^= 0x01;
293    if crate::ed25519::verify(&TEST_PUBKEY, &bad_payload, &TEST_SIG) {
294        return false;
295    }
296    true
