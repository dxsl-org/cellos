1//! Cell binary signing — Ed25519 signature verification for spawned cell ELFs.
2//!
3//! The kernel holds only a public key (fleet trust anchor). Cell binaries are
4//! signed offline with the corresponding private key and carry the 64-byte
5//! signature in an `__ViCell_sig` ELF section.
6//!
7//! Canonical signed payload: every byte of the final ELF container except the
8//! 64 signature bytes themselves. The signature section header remains covered,
9//! so section names, offsets, types, flags, links, relocation metadata, ELF
10//! headers, program headers, and all load-affecting payload bytes are immutable.
11//! The signer first adds a zero-filled signature section, signs that stable
12//! container, then replaces only those excluded bytes.
13//!
14//! Verify-only: the kernel never signs (private key lives offline).
15
16use alloc::vec::Vec;
17
18/// Dev Ed25519 **public** key — derived from the fixed dev seed in
19/// `scripts/sign-cell.py` (seed `[0x43]*32`, reproducible; never shipped in release).
20const DEV_CELL_SIGNER_PUBKEY: [u8; 32] = [
21    0x22, 0xfc, 0x29, 0x77, 0x92, 0xf0, 0xb6, 0xff, 0xc0, 0xbf, 0xcf, 0xdb, 0x7e, 0xdb, 0x0c, 0x0a,
22    0xa1, 0x4e, 0x02, 0x5a, 0x36, 0x5e, 0xc0, 0xe3, 0x42, 0xe8, 0x6e, 0x38, 0x29, 0xcb, 0x74, 0xb6,
23];
24
25/// Fleet cell-signing trust anchor.
26/// `dev-signing-key` → the reproducible dev key above (matching `scripts/sign-cell.py --seed 0x43*32`).
27/// Otherwise → a zero placeholder that fails every verify (fail-closed until prod key is provisioned).
28#[cfg(feature = "dev-signing-key")]
29const CELL_SIGNER_PUBKEY: [u8; 32] = DEV_CELL_SIGNER_PUBKEY;
30
31#[cfg(not(feature = "dev-signing-key"))]
32const CELL_SIGNER_PUBKEY: [u8; 32] = [0u8; 32]; // TODO(prod): provisioned fleet key
33
34/// Returns `true` when the `signing-required` build feature is set (CI/prod posture).
35/// In dev mode (default) an absent signature is permitted; with `signing-required`
36/// an unsigned cell is denied the same as a tampered one.
37pub const fn signing_required() -> bool {
38    cfg!(feature = "signing-required")
39}

41/// Extract the 64-byte Ed25519 signature from the `__ViCell_sig` ELF section.
42///
43/// Returns `None` unless exactly one in-bounds signature section has 64 bytes.
44pub fn extract_sig(elf_bytes: &[u8]) -> Option<[u8; 64]> {
45    let (offset, size) = signature_range(elf_bytes)?;
46    if size != 64 {
47        return None;
48    }
49    let mut sig = [0u8; 64];
50    sig.copy_from_slice(&elf_bytes[offset..offset + size]);
51    Some(sig)
52}

54/// Verify the Ed25519 signature of a cell ELF binary.
55///
56/// Returns `false` on any malformed input, missing section, or signature
57/// mismatch — never panics. The signature covers every final ELF byte except
58/// its own 64-byte payload, including all relocation section headers and data.
59pub fn verify_cell(elf_bytes: &[u8], sig: &[u8; 64]) -> bool {
60    verify_cell_with_key(elf_bytes, sig, &CELL_SIGNER_PUBKEY)
61}
62
63/// Locate the single file-backed fixed-size signature payload with the bounded
64/// ELF parser. Returning its byte range lets verification cover the table
65/// itself without creating a signature recursion.
66fn signature_range(elf_bytes: &[u8]) -> Option<(usize, usize)> {
67    const SHT_PROGBITS: u64 = 1;
68
69    let elf = crate::loader::elf_section::ElfSections::parse(elf_bytes)?;
70    let names = elf.names()?;
71    let mut signature = None;
72    for index in 0..elf.count() {
73        let section = elf.section(index)?;
74        if elf.name(names, section)? != b"__ViCell_sig" {
75            continue;
76        }
77        // Only a file-backed, ordinary data section can carry the excluded
78        // bytes. Every other section's metadata stays inside the signed input.
79        if signature.is_some() || section.kind != SHT_PROGBITS || section.size != 64 {
80            return None;
81        }
82        elf.bytes(section)?;
83        signature = Some((section.offset, section.size));
84    }
85    signature
86}

88/// Inner implementation; accepts an explicit key so `self_test` can use
89/// the precomputed test key without touching `CELL_SIGNER_PUBKEY`.
90fn verify_cell_with_key(elf_bytes: &[u8], sig: &[u8; 64], pubkey: &[u8; 32]) -> bool {
91    let (signature_offset, signature_size) = match signature_range(elf_bytes) {
92        Some(range @ (_, 64)) => range,
93        _ => return false,
94    };
95    if extract_sig(elf_bytes).as_ref() != Some(sig) {
96        return false;
97    }
98
99    // The signature section itself is the only excluded interval. Its header
100    // and placement stay in the signed byte stream, so no parser-selected
101    // relocation data can move outside authenticated bytes.
102    let mut payload = Vec::with_capacity(elf_bytes.len() - signature_size);
103    payload.extend_from_slice(&elf_bytes[..signature_offset]);
104    payload.extend_from_slice(&elf_bytes[signature_offset + signature_size..]);
105    crate::ed25519::verify(pubkey, &payload, sig)
106}

108/// Boot-time self-test using a precomputed RFC-style test vector.
109///
110/// Uses a separate precomputed `(pubkey, payload, sig)` triple — **no private
111/// key in the kernel**. Returns `true` iff the known-good vector verifies AND
112/// a flipped-byte payload is rejected.
113pub fn self_test() -> bool {
114    // Precomputed vector — seed [0x43]*32, payload b"CellosSigningTest".
115    // Regenerate: `python scripts/sign-cell.py --emit-test-vector`
116    const TEST_PUBKEY: [u8; 32] = [
117        0x22, 0xfc, 0x29, 0x77, 0x92, 0xf0, 0xb6, 0xff, 0xc0, 0xbf, 0xcf, 0xdb, 0x7e, 0xdb, 0x0c,
118        0x0a, 0xa1, 0x4e, 0x02, 0x5a, 0x36, 0x5e, 0xc0, 0xe3, 0x42, 0xe8, 0x6e, 0x38, 0x29, 0xcb,
119        0x74, 0xb6,
120    ];
121    const TEST_SIG: [u8; 64] = [
122        0x22, 0xf6, 0x2e, 0xba, 0x53, 0x9c, 0x66, 0xa0, 0xc1, 0xed, 0x39, 0xc8, 0x90, 0x04, 0xf8,
123        0xfc, 0x46, 0xb0, 0xe5, 0x42, 0xc9, 0x97, 0x22, 0x2d, 0x3f, 0x10, 0x17, 0xf3, 0xa4, 0x56,
124        0x67, 0x58, 0x9b, 0x49, 0x98, 0x2b, 0x4a, 0x48, 0x23, 0x11, 0x90, 0x09, 0x25, 0xe3, 0x9f,
125        0x02, 0x0b, 0x0e, 0x34, 0x70, 0x25, 0xfa, 0x10, 0xe3, 0x7e, 0xac, 0xd4, 0xb1, 0x6c, 0x66,
126        0xcf, 0x7b, 0x1e, 0x0a,
127    ];
128    const TEST_PAYLOAD: &[u8] = b"CellosSigningTest";
129
130    // Positive: known-good vector must verify.
131    if !crate::ed25519::verify(&TEST_PUBKEY, TEST_PAYLOAD, &TEST_SIG) {
132        return false;
133    }
134    // Negative: flipped byte in payload must be rejected.
135    let mut bad_payload = alloc::vec![0u8; TEST_PAYLOAD.len()];
136    bad_payload.copy_from_slice(TEST_PAYLOAD);
137    bad_payload[0] ^= 0x01;
138    if crate::ed25519::verify(&TEST_PUBKEY, &bad_payload, &TEST_SIG) {
139        return false;
140    }
141    true
142}
