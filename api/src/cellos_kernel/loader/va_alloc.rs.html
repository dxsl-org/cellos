1//! Virtual address allocator for cell images.
2//!
3//! Assigns each cell image one aligned VA slot within the cell VA region
4//! [CELL_VA_START, CELL_VA_START + MAX_SLOTS * CELL_VA_STRIDE): a PIE takes
5//! any free slot (`alloc_cell_va`); a fixed-VA image claims the one slot it is
6//! linked into (`reserve_cell_va`). Every PT_LOAD must stay inside the slot
7//! its spawn owns (see `loader::elf::image_va`).
8//!
9//! This region lives entirely below the SV39 user-half boundary (256 GiB), so
10//! the `USER_VADDR_MAX` guard is never triggered by any allocated slot.
11//!
12//! # Design
13//! Simple two-level allocator: a bump pointer for never-allocated slots, and a
14//! free list (atomic bitset) for slots returned by `free_cell_va`.  Allocations
15//! are O(n_freed) in the worst case, O(1) when the free list is empty.
16
17// RV32 lacks native 64-bit atomics; portable-atomic polyfills AtomicU64 there
18// via the critical-section impl hal/arch/riscv registers.
19#[cfg(not(target_arch = "riscv32"))]
20use core::sync::atomic::AtomicU64;
21use core::sync::atomic::Ordering;
22#[cfg(target_arch = "riscv32")]
23use portable_atomic::AtomicU64;

25#[cfg(all(feature = "large-cell-va", target_pointer_width = "32"))]
26compile_error!("large-cell-va requires a 64-bit paged target");
27#[cfg(all(feature = "large-cell-va", feature = "cell-scale-experiment"))]
28compile_error!(
29    "large-cell-va cannot be combined with cell-scale-experiment: exceeds SV39 user half"
30);

32/// Cell VA region start — 64 GiB (0x10_0000_0000).
33///
34/// Must lie above ALL identity-mapped regions in the SAS, avoiding collision
35/// when physical RAM extends to >= 4 GiB (e.g. 0x1_0000_0000..0x1_8000_0000):
36///   - QEMU virt RISC-V: UART/VirtIO MMIO at 0x1000_0000–0x1001_0000
37///   - PCIe ECAM at 0x3000_0000 (1 MiB); VirtIO BAR below 0x8000_0000
38///   - RAM identity map: supports physical configurations up to 48 GiB
39///   - QEMU virt AArch64: GIC/peripheral MMIO below 0x1000_0000
40///     64 GiB clears all of the above with room to spare and remains well inside
41///     the SV39 user-half (256 GiB = 0x40_0000_0000).
42///
43/// RISC-V medany: even the 256 MiB desktop slot stays within ±2 GiB.
44///
45/// RV32 Nano boots with SATP=0 (bare physical, no paging — see specs/04), so
46/// PIE cells never occur there and `alloc_cell_va` is never actually called;
47/// this constant only needs to type-check as a 32-bit `usize` on that target.
48#[cfg(not(target_arch = "riscv32"))]
49pub(crate) const CELL_VA_START: usize = 0x10_0000_0000;
50#[cfg(target_arch = "riscv32")]
51pub(crate) const CELL_VA_START: usize = 0x2000_0000;
52
53/// Embedded keeps the historical 32 MiB spacing. Desktop images opt in to a
54/// 256 MiB slot, sufficient for a 128 MiB static arena plus code and data.
55#[cfg(not(feature = "large-cell-va"))]
56pub(crate) const CELL_VA_STRIDE: usize = 32 * 1024 * 1024;
57#[cfg(feature = "large-cell-va")]
58pub(crate) const CELL_VA_STRIDE: usize = 256 * 1024 * 1024;
59
60/// Maximum simultaneous PIE cell slots: normally 512 (16 GiB embedded or
61/// 128 GiB desktop). Both regions, starting at 64 GiB, fit SV39's user half.
62///
63/// `cell-scale-experiment` raises this to 4096 slots (4096 × 32 MiB = 128 GiB of
64/// VA, still inside the user half) so the D5 scale measurement measures memory
65/// rather than this constant. See `memory::cell_quota::MAX_CELLS`.
66#[cfg(not(feature = "cell-scale-experiment"))]
67const MAX_SLOTS: usize = 512;
68#[cfg(feature = "cell-scale-experiment")]
69const MAX_SLOTS: usize = 4096;
70
71#[cfg(target_pointer_width = "64")]
72const _: () = assert!(CELL_VA_START + MAX_SLOTS * CELL_VA_STRIDE <= 0x40_0000_0000);
73
74/// Exclusive end of a valid reservation, shared by ELF admission and release.
75pub(crate) fn slot_end(base: usize) -> Option<usize> {
76    let offset = base.checked_sub(CELL_VA_START)?;
77    if !offset.is_multiple_of(CELL_VA_STRIDE) || offset / CELL_VA_STRIDE >= MAX_SLOTS {
78        return None;
79    }
80    base.checked_add(CELL_VA_STRIDE)
81}
82
83/// Base of the slot containing `va`, or `None` outside the cell VA region.
84pub(crate) fn slot_containing(va: usize) -> Option<usize> {
85    let index = va.checked_sub(CELL_VA_START)? / CELL_VA_STRIDE;
86    (index < MAX_SLOTS).then(|| CELL_VA_START + index * CELL_VA_STRIDE)
87}
88
89/// Number of AtomicU64 words needed to cover MAX_SLOTS bits.
90const BITMAP_WORDS: usize = MAX_SLOTS.div_ceil(64);
91
92/// Slot ownership. Invariant: a slot is owned by exactly one spawn iff its
93/// index is below `bump` and its `free` bit is clear; every `free` bit lies
94/// below `bump`.
95pub(super) struct SlotAllocator {
96    /// The first slot that has NEVER been handed out.
97    bump: core::sync::atomic::AtomicUsize,
98    /// Bit N = 1 means slot N is available for reuse. Each repeat of the inline
99    /// `const` block is a distinct AtomicU64 (an inline const sidesteps
100    /// `clippy::declare_interior_mutable_const` without changing semantics).
101    free: [AtomicU64; BITMAP_WORDS],
102}
103
104static SLOTS: SlotAllocator = SlotAllocator::new();
105
106impl SlotAllocator {
107    pub(super) const fn new() -> Self {
108        Self {
109            bump: core::sync::atomic::AtomicUsize::new(0),
110            free: [const { AtomicU64::new(0) }; BITMAP_WORDS],
111        }
112    }
113
114    pub(super) fn alloc(&self) -> Option<usize> {
115        // 1. Try the free list first.
116        for (word_idx, word) in self.free.iter().enumerate() {
117            let mut val = word.load(Ordering::Relaxed);
118            while val != 0 {
119                let bit = val.trailing_zeros() as usize;
120                let slot = word_idx * 64 + bit;
121                if slot >= MAX_SLOTS {
122                    break;
123                }
124                let mask = 1u64 << bit;
125                match word.compare_exchange_weak(
126                    val,
127                    val & !mask,
128                    Ordering::AcqRel,
129                    Ordering::Relaxed,
130                ) {
131                    Ok(_) => return Some(CELL_VA_START + slot * CELL_VA_STRIDE),
132                    Err(cur) => val = cur, // retry with updated value
133                }
134            }
135        }
136
137        // 2. Bump-allocate a fresh slot.
138        let idx = self.bump.fetch_add(1, Ordering::SeqCst);
139        if idx >= MAX_SLOTS {
140            self.bump.fetch_sub(1, Ordering::SeqCst);
141            return None;
142        }
143        Some(CELL_VA_START + idx * CELL_VA_STRIDE)
144    }
145
146    /// Claim the exact slot at `base`, or refuse when another spawn owns it.
147    pub(super) fn reserve(&self, base: usize) -> bool {
148        if slot_end(base).is_none() {
149            return false;
150        }
151        let slot = (base - CELL_VA_START) / CELL_VA_STRIDE;
152        loop {
153            let bump = self.bump.load(Ordering::SeqCst);
154            if slot < bump {
155                // Handed out before: claimable only while it is on the free list.
156                let mask = 1u64 << (slot % 64);
157                return self.free[slot / 64].fetch_and(!mask, Ordering::AcqRel) & mask != 0;
158            }
159            if self
160                .bump
161                .compare_exchange(bump, slot + 1, Ordering::SeqCst, Ordering::SeqCst)
162                .is_ok()
163            {
164                // The slots jumped over were never handed out: they go on the
165                // free list so the ownership invariant keeps holding.
166                for skipped in bump..slot {
167                    self.free[skipped / 64].fetch_or(1u64 << (skipped % 64), Ordering::Release);
168                }
169                return true;
170            }
171        }
172    }
173
174    pub(super) fn free(&self, base: usize) {
175        if slot_end(base).is_none() {
176            return;
177        }
178        let slot = (base - CELL_VA_START) / CELL_VA_STRIDE;
179        // Restore the bump cursor exactly when this is the most recently minted
180        // slot. If another allocation raced ahead, publish it to the free bitmap.
181        if self
182            .bump
183            .compare_exchange(slot + 1, slot, Ordering::AcqRel, Ordering::Acquire)
184            .is_err()
185        {
186            let word = &self.free[slot / 64];
187            let mask = 1u64 << (slot % 64);
188            word.fetch_or(mask, Ordering::Release);
189        }
190    }
191}

193/// Allocate one profile-sized VA slot for a new PIE cell.
194///
195/// Returns `None` when all slots are exhausted (unlikely in practice: 512 cells
196/// and the free list recycles dead cells' slots).
197pub fn alloc_cell_va() -> Option<usize> {
198    SLOTS.alloc()
199}
200
201/// Claim the slot at `base` for a fixed-VA (ET_EXEC) image that is linked
202/// inside it. `false` when `base` is not a slot base or another cell owns it:
203/// a fixed-VA image may never land in a slot a PIE (or another image) holds.
204pub fn reserve_cell_va(base: usize) -> bool {
205    SLOTS.reserve(base)
206}
207
208/// Return a VA slot to the free list so it can be reused by a future cell.
209///
210/// `base` must be a value previously returned by `alloc_cell_va` or claimed
211/// by `reserve_cell_va`. Silently ignores invalid values.
212pub fn free_cell_va(base: usize) {
213    SLOTS.free(base)
214}

216#[cfg(feature = "test-hooks")]
217pub(crate) fn snapshot() -> (usize, alloc::vec::Vec<u64>) {
218    (
219        SLOTS.bump.load(Ordering::Acquire),
220        SLOTS
221            .free
222            .iter()
223            .map(|word| word.load(Ordering::Acquire))
224            .collect(),
225    )
226}
