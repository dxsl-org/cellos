1//! Virtual address allocators: cell image slots and the user-stack arena.
2//!
3//! Assigns each cell image one aligned VA slot within the cell VA region
4//! [CELL_VA_START, CELL_VA_START + MAX_SLOTS * CELL_VA_STRIDE): a PIE takes
5//! any free slot (`alloc_cell_va`); a fixed-VA image claims the one slot it is
6//! linked into (`reserve_cell_va`). Every PT_LOAD must stay inside the slot
7//! its spawn owns (see `loader::elf::image_va`). A fixed-VA image links into
8//! one of the lowest [`FIXED_VA_SLOTS`] slots: randomised order hands those to
9//! a PIE only once every other slot is held.
10//!
11//! Every user stack takes one `STACK_VA_STRIDE` slot of the stack arena right
12//! above the cell region (`alloc_stack_va`), so it is never mapped at its
13//! physical address; the slot returns (`free_stack_va`) only once its frames'
14//! translations are gone on every hart.
15//!
16//! Both regions live entirely below the SV39 user-half boundary (256 GiB), so
17//! the `USER_VADDR_MAX` guard is never triggered by any allocated slot.
18//!
19//! # Design
20//! Each region is one atomic ownership bitmap: bit N is set iff slot N has a
21//! holder. `alloc` probes from a start index ([`SlotOrder`]) for the first clear
22//! bit, wrapping once through the region, and sets it with one compare-exchange;
23//! `reserve` sets one named bit; `free` clears it. Slots are all one size, so
24//! any free slot fits any request: the region is exhausted only when every slot
25//! is held, whatever the start index. A probe reads at most `slots / 64 + 1`
26//! bitmap words, one more for a region with a fixed-VA band (10 for the 512
27//! cell slots, 513 for the 32 768 stack slots).
28//!
29//! The start index is random per allocation when the kernel has a hardware
30//! entropy source at boot (x86 RDSEED/RDRAND), so a cell's image and stack
31//! addresses differ between boots; it is then drawn above the region's
32//! fixed-VA band ([`Region::fixed`]), and the band is probed last. Without one
33//! (RV64, AArch64, an x86 CPU without RDRAND) every probe starts at slot 0:
34//! lowest free slot first, band included, the historical order.
35//!
36//! # Identity-mapped RAM
37//! Grants and SHM pages keep their physical address as their VA, and RV64 and
38//! AArch64 also map all RAM at VA == PA, so RAM must end below the cell slots:
39//! the frame allocator leaves RAM at or above [`IDENTITY_RAM_END`] unused
40//! (logged once at boot), and the boot maps assert it
41//! ([`assert_identity_ram_below_cells`]). MMIO is refused window by window
42//! where it would reach the slots or the arena ([`overlaps_allocated_va`]).
43

59/// Cell VA region start — 64 GiB (0x10_0000_0000).
60///
61/// Must lie above ALL identity-mapped regions in the SAS, avoiding collision
62/// when physical RAM extends to >= 4 GiB (e.g. 0x1_0000_0000..0x1_8000_0000):
63///   - QEMU virt RISC-V: UART/VirtIO MMIO at 0x1000_0000–0x1001_0000
64///   - PCIe ECAM at 0x3000_0000 (1 MiB); VirtIO BAR below 0x8000_0000
65///   - RAM identity map: supports physical configurations up to 48 GiB
66///   - QEMU virt AArch64: GIC/peripheral MMIO below 0x1000_0000
67///     64 GiB clears all of the above with room to spare and remains well inside
68///     the user window (`hal::paging::USER_LIMIT`, 256 GiB).
69///
70/// RISC-V medany: even the 256 MiB desktop slot stays within ±2 GiB.
71///
72/// RV32 Nano boots with SATP=0 (bare physical, no paging — see specs/04), so
73/// PIE cells never occur there and `alloc_cell_va` is never actually called;
74/// this constant only needs to type-check as a 32-bit `usize` on that target.
75#[cfg(not(target_arch = "riscv32"))]
76pub(crate) const CELL_VA_START: usize = 0x10_0000_0000;
77#[cfg(target_arch = "riscv32")]
78pub(crate) const CELL_VA_START: usize = 0x2000_0000;

80/// Embedded keeps the historical 32 MiB spacing. Desktop images opt in to a
81/// 256 MiB slot, sufficient for a 128 MiB static arena plus code and data.
82#[cfg(not(feature = "large-cell-va"))]
83pub(crate) const CELL_VA_STRIDE: usize = 32 * 1024 * 1024;
84#[cfg(feature = "large-cell-va")]
85pub(crate) const CELL_VA_STRIDE: usize = 256 * 1024 * 1024;

87/// Maximum simultaneous PIE cell slots: normally 512 (16 GiB embedded or
88/// 128 GiB desktop). Both regions, starting at 64 GiB, fit SV39's user half.
89///
90/// `cell-scale-experiment` raises this to 4096 slots (4096 × 32 MiB = 128 GiB of
91/// VA, still inside the user half) so the D5 scale measurement measures memory
92/// rather than this constant. See `memory::cell_quota::MAX_CELLS`.
93#[cfg(not(feature = "cell-scale-experiment"))]
94const MAX_SLOTS: usize = 512;
95#[cfg(feature = "cell-scale-experiment")]
96const MAX_SLOTS: usize = 4096;

98#[cfg(cellos_wide)]
99const _: () = assert!(CELL_VA_START + MAX_SLOTS * CELL_VA_STRIDE <= crate::hal::paging::USER_LIMIT);

101/// Exclusive end of the RAM the kernel manages and maps at VA == PA: the
102/// first cell slot. The frame allocator leaves usable RAM at or above it
103/// unused (`memory::frame::plan_managed_ranges`), so no grant, SHM page or
104/// RV64/AArch64 RAM alias can land on a cell slot or the stack arena. Bare
105/// physical targets map no cell slots and keep all RAM.
106#[cfg(cellos_wide)]
107pub(crate) const IDENTITY_RAM_END: usize = CELL_VA_START;
108#[cfg(not(cellos_wide))]
109pub(crate) const IDENTITY_RAM_END: usize = usize::MAX;

111/// The lowest cell slots, kept for fixed-VA (ET_EXEC) images: randomised
112/// order hands one to a PIE only once every other slot is held, so a fixed
113/// image linked into this band finds its slot free while any slot is. The
114/// deterministic order (no entropy) still takes slots lowest-first, band
115/// included, exactly as before randomisation. No shipped cell is fixed-VA.
116pub(crate) const FIXED_VA_SLOTS: usize = 8;
117
118/// A run of equal, aligned VA slots: `slots` slots of `stride` bytes from
119/// `start`; the lowest `fixed` of them are the fixed-VA band (see
120/// [`FIXED_VA_SLOTS`]; `0` for a region no fixed image links into).
121#[derive(Clone, Copy)]
122pub(super) struct Region {
123    start: usize,
124    stride: usize,
125    slots: usize,
126    fixed: usize,
127}
128
129impl Region {
130    /// Index of the slot whose base is exactly `base`.
131    fn index_of_base(&self, base: usize) -> Option<usize> {
132        let offset = base.checked_sub(self.start)?;
133        (offset.is_multiple_of(self.stride) && offset / self.stride < self.slots)
134            .then_some(offset / self.stride)
135    }
136
137    fn base(&self, index: usize) -> usize {
138        self.start + index * self.stride
139    }
140}

142/// The cell image slots.
143const CELLS: Region = Region {
144    start: CELL_VA_START,
145    stride: CELL_VA_STRIDE,
146    slots: MAX_SLOTS,
147    fixed: FIXED_VA_SLOTS,
148};
149
150/// Exclusive end of a valid reservation, shared by ELF admission and release.
151pub(crate) fn slot_end(base: usize) -> Option<usize> {
152    CELLS.index_of_base(base)?;
153    base.checked_add(CELL_VA_STRIDE)
154}
155
156/// Base of the slot containing `va`, or `None` outside the cell VA region.
157pub(crate) fn slot_containing(va: usize) -> Option<usize> {
158    let index = va.checked_sub(CELL_VA_START)? / CELL_VA_STRIDE;
159    (index < MAX_SLOTS).then(|| CELL_VA_START + index * CELL_VA_STRIDE)
160}

162/// Number of AtomicU64 words needed to cover MAX_SLOTS bits.
163const BITMAP_WORDS: usize = MAX_SLOTS.div_ceil(64);
164
165/// Where [`SlotAllocator::alloc`] starts probing.
166///
167/// Randomised iff [`seed`](Self::seed) found its source able to fill one
168/// 64-bit word; it is then read again for every allocation. Otherwise every
169/// start is 0 (lowest free slot first) and the source is never called again.
170pub(super) struct SlotOrder {
171    /// Fills a prefix of the buffer with hardware entropy and returns its length
172    /// (`dev::virtio_rng::get_random`'s contract); never allocates or blocks.
173    source: fn(&mut [u8]) -> usize,
174    randomised: AtomicBool,
175}
176
177impl SlotOrder {
178    pub(super) const fn new(source: fn(&mut [u8]) -> usize) -> Self {
179        Self {
180            source,
181            randomised: AtomicBool::new(false),
182        }
183    }
184
185    /// Probe the source once and fix the order for this boot: `true` (random
186    /// starts) iff it filled a whole word.
187    #[cfg(any(cellos_wide, test))]
188    pub(super) fn seed(&self) -> bool {
189        let randomised = self.word().is_some();
190        // Only this flag is published: the source holds its own state.
191        self.randomised.store(randomised, Ordering::Relaxed);
192        randomised
193    }
194
195    /// The random start offset for one allocation over a span of `span`
196    /// slots, below `span`; `None` in deterministic order. In randomised order
197    /// a source that fails this once (RDSEED and RDRAND both exhausted after
198    /// their bounded retries) starts that one allocation at offset 0; which
199    /// slot it gets never affects ownership.
200    fn random_start(&self, span: usize) -> Option<usize> {
201        if !self.randomised.load(Ordering::Relaxed) {
202            return None;
203        }
204        Some(self.word().map_or(0, |word| (word % span as u64) as usize))
205    }
206
207    fn word(&self) -> Option<u64> {
208        let mut bytes = [0u8; 8];
209        ((self.source)(&mut bytes) == bytes.len()).then(|| u64::from_le_bytes(bytes))
210    }
211}

213/// The start order of both regions, fixed once at boot by [`seed_slot_order`].
214static SLOT_ORDER: SlotOrder = SlotOrder::new(crate::dev::virtio_rng::get_random);
215
216/// Boot, once, before any cell image or user stack takes a slot: random slot
217/// starts if the kernel entropy source works, else lowest-first. Logs which.
218#[cfg(cellos_wide)]
219pub(crate) fn seed_slot_order() {
220    if SLOT_ORDER.seed() {
221        log::info!("[va] slot order: randomised");
222    } else {
223        log::info!("[va] slot order: deterministic (no entropy)");
224    }
225}

227/// Slot ownership over one [`Region`] (`WORDS` bitmap words cover its slots).
228/// Invariant: bit N of `owned` is set iff slot N has exactly one holder, and no
229/// bit at or past `region.slots` is ever set. Only an atomic read-modify-write
230/// that finds a bit clear sets it (`alloc`, `reserve`), so no slot is handed
231/// out twice; only `free` clears one.
232pub(super) struct SlotAllocator<const WORDS: usize = BITMAP_WORDS> {
233    region: Region,
234    /// Each repeat of the inline `const` block is a distinct AtomicU64 (an
235    /// inline const sidesteps `clippy::declare_interior_mutable_const`).
236    owned: [AtomicU64; WORDS],
237    /// `free` calls that found their slot already clear (see [`Self::free`]).
238    stale_frees: AtomicUsize,
239}
240
241static SLOTS: SlotAllocator = SlotAllocator::new();
242
243impl SlotAllocator {
244    /// The cell image slots.
245    pub(super) const fn new() -> Self {
246        Self::over(CELLS)
247    }
248}
249
250impl<const WORDS: usize> SlotAllocator<WORDS> {
251    pub(super) const fn over(region: Region) -> Self {
252        assert!(region.slots > 0 && region.slots <= WORDS * 64);
253        assert!(region.fixed < region.slots);
254        Self {
255            region,
256            owned: [const { AtomicU64::new(0) }; WORDS],
257            stale_frees: AtomicUsize::new(0),
258        }
259    }
260
261    /// Take a free slot in `order`; `None` only when every slot is held.
262    ///
263    /// Deterministic order: the lowest free slot. Randomised order: the first
264    /// free slot at or after a random start above the fixed-VA band, wrapping
265    /// once to the band's end; the band only when every other slot is held.
266    pub(super) fn alloc(&self, order: &SlotOrder) -> Option<usize> {
267        let (fixed, slots) = (self.region.fixed, self.region.slots);
268        match order.random_start(slots - fixed) {
269            None => self.claim_span(0, slots),
270            Some(offset) => {
271                let start = fixed + offset;
272                self.claim_span(start, slots)
273                    .or_else(|| self.claim_span(fixed, start))
274                    .or_else(|| self.claim_span(0, fixed))
275            }
276        }
277    }
278
279    /// Claim the first free slot at or after `start`, wrapping past the last
280    /// slot to slot 0: every slot is probed once. `start == 0` is lowest-first.
281    #[cfg(test)]
282    fn claim_from(&self, start: usize) -> Option<usize> {
283        self.claim_span(start, self.region.slots)
284            .or_else(|| self.claim_span(0, start))
285    }
286
287    /// Claim the lowest free slot in `[from, to)` (`to <= region.slots`),
288    /// reading each bitmap word the span touches once.
289    fn claim_span(&self, from: usize, to: usize) -> Option<usize> {
290        let mut next = from;
291        while next < to {
292            let index = next / 64;
293            let word_end = ((index + 1) * 64).min(to);
294            // Bits `next % 64 ..= (word_end - 1) % 64` of word `index`.
295            let probed = (u64::MAX << (next % 64)) & (u64::MAX >> (64 - (word_end - index * 64)));
296            let word = &self.owned[index];
297            let mut seen = word.load(Ordering::Relaxed);
298            loop {
299                let open = probed & !seen;
300                if open == 0 {
301                    break;
302                }
303                let bit = open.trailing_zeros() as usize;
304                // Acquire: the previous holder's `free` happens before this
305                // holder's use of the slot.
306                match word.compare_exchange_weak(
307                    seen,
308                    seen | 1u64 << bit,
309                    Ordering::AcqRel,
310                    Ordering::Relaxed,
311                ) {
312                    Ok(_) => return Some(self.region.base(index * 64 + bit)),
313                    Err(now) => seen = now,
314                }
315            }
316            next = word_end;
317        }
318        None
319    }
320
321    /// Claim the exact slot at `base`, or refuse when another holder owns it.
322    pub(super) fn reserve(&self, base: usize) -> bool {
323        let Some(slot) = self.region.index_of_base(base) else {
324            return false;
325        };
326        let mask = 1u64 << (slot % 64);
327        self.owned[slot / 64].fetch_or(mask, Ordering::AcqRel) & mask == 0
328    }
329
330    /// Release the slot at `base`. Not a slot base: ignored. A slot nobody
331    /// holds stays clear, so the ownership invariant holds, but a second free
332    /// means some holder's bookkeeping is wrong (a later one could clear a new
333    /// holder's bit): it is logged and counted, and test builds fail on it.
334    pub(super) fn free(&self, base: usize) {
335        let Some(slot) = self.region.index_of_base(base) else {
336            return;
337        };
338        let mask = 1u64 << (slot % 64);
339        if self.owned[slot / 64].fetch_and(!mask, Ordering::Release) & mask == 0 {
340            let stale = self.stale_frees.fetch_add(1, Ordering::Relaxed) + 1;
341            log::error!("[va] free of unheld slot {base:#x} ignored ({stale} so far)");
342            #[cfg(test)]
343            debug_assert!(false, "free of unheld slot {base:#x}");
344        }
345    }
346
347    /// `free` calls that found their slot already clear.
348    #[cfg(test)]
349    pub(super) fn stale_frees(&self) -> usize {
350        self.stale_frees.load(Ordering::Relaxed)
351    }
352
353    /// Slots owned right now.
354    #[cfg(test)]
355    pub(super) fn owned(&self) -> usize {
356        self.owned
357            .iter()
358            .map(|word| word.load(Ordering::Acquire).count_ones() as usize)
359            .sum()
360    }
361
362    /// The ownership bitmap, word by word.
363    #[cfg(feature = "test-hooks")]
364    fn snapshot(&self) -> alloc::vec::Vec<u64> {
365        self.owned
366            .iter()
367            .map(|word| word.load(Ordering::Acquire))
368            .collect()
369    }
370
371    /// Does the slot at `base` have a holder? `false` for a non-slot address.
372    #[cfg(all(cellos_wide, feature = "test-hooks"))]
373    fn held(&self, base: usize) -> bool {
374        self.region.index_of_base(base).is_some_and(|slot| {
375            self.owned[slot / 64].load(Ordering::Acquire) & (1u64 << (slot % 64)) != 0
376        })
377    }
378}

380/// First byte of the user-stack arena: right above the cell slots, so it lies
381/// inside the user window and above all identity-mapped RAM (which ends below
382/// the cell slots, [`IDENTITY_RAM_END`]). 80 GiB by default, 192 GiB with
383/// `large-cell-va` or `cell-scale-experiment`.
384#[cfg(cellos_wide)]
385pub(crate) const STACK_VA_START: usize = CELL_VA_START + MAX_SLOTS * CELL_VA_STRIDE;
386
387/// One user stack per slot: its bottom guards, then its usable pages, then
388/// unmapped VA up to the next slot. The largest stack the kernel allocates
389/// (`task::STACK_PAGES` plus `task::stack::STACK_GUARD_PAGES`) rounded up to a
390/// power of two: 512 KiB.
391pub(crate) const STACK_VA_STRIDE: usize =
392    (crate::task::STACK_PAGES + crate::task::stack::STACK_GUARD_PAGES).next_power_of_two()
393        * crate::memory::paging::PAGE_SIZE;
394
395/// Live user stacks at once (16 GiB of VA at the 512 KiB stride).
396#[cfg(cellos_wide)]
397const STACK_SLOTS: usize = 32 * 1024;
398
399#[cfg(cellos_wide)]
400const _: () =
401    assert!(STACK_VA_START + STACK_SLOTS * STACK_VA_STRIDE <= crate::hal::paging::USER_LIMIT);
402
403#[cfg(cellos_wide)]
404const STACKS: Region = Region {
405    start: STACK_VA_START,
406    stride: STACK_VA_STRIDE,
407    slots: STACK_SLOTS,
408    fixed: 0,
409};
410
411#[cfg(cellos_wide)]
412static STACK_ARENA: SlotAllocator<{ STACK_SLOTS / 64 }> = SlotAllocator::over(STACKS);
413
414/// Take one stack slot (`STACK_VA_STRIDE` bytes of VA, nothing mapped) in
415/// this boot's [`SlotOrder`], or `None` when every slot is held.
416#[cfg(cellos_wide)]
417pub(crate) fn alloc_stack_va() -> Option<usize> {
418    STACK_ARENA.alloc(&SLOT_ORDER)
419}

421/// Paging targets only: the bare-physical ones never map a stack.
422#[cfg(not(cellos_wide))]
423pub(crate) fn alloc_stack_va() -> Option<usize> {
424    None

427/// Return a stack slot. Only once nothing translates any page of it on any
428/// hart: the stack's `VaHold` calls this after its run's flush is confirmed.
429pub(crate) fn free_stack_va(base: usize) {
430    #[cfg(cellos_wide)]
431    STACK_ARENA.free(base);
432    #[cfg(not(cellos_wide))]
433    let _ = base;
434}

436/// Boot check: RAM mapped at VA == PA (grants and SHM pages keep their
437/// physical address as VA; RV64 and AArch64 also map all RAM there) ends at or
438/// below `ram_end`, which must not reach the cell slots. The frame allocator
439/// already leaves RAM at or above [`IDENTITY_RAM_END`] unused, so this holds
440/// unless firmware placed the kernel or bootloader memory there, which cannot
441/// be dropped. MMIO is not RAM: each boot or later MMIO window is refused on
442/// its own when it reaches the slots or the arena ([`overlaps_allocated_va`]).
443#[cfg(cellos_wide)]
444pub(crate) fn assert_identity_ram_below_cells(ram_end: usize) {
445    assert!(
446        ram_end <= IDENTITY_RAM_END,
447        "identity-mapped RAM ends at {ram_end:#x}, inside the cell VA slots at {CELL_VA_START:#x}"
448    );
449}

451/// End of the VA the kernel hands out from its allocators: the cell slots
452/// `[CELL_VA_START, STACK_VA_START)`, then the stack arena right above them.
453#[cfg(cellos_wide)]
454const ALLOCATED_VA_END: usize = STACK_VA_START + STACK_SLOTS * STACK_VA_STRIDE;
455
456/// Does `[start, start + len)` reach the VA the kernel hands out from its
457/// allocators, `[CELL_VA_START, ALLOCATED_VA_END)` (cell slots and stack
458/// arena)? A physical range that does can never be mapped at VA == PA: its
459/// leaves would alias a cell image or a user stack. A range whose end
460/// overflows counts as overlapping; an empty one never does.
461#[cfg(cellos_wide)]
462pub(crate) fn overlaps_allocated_va(start: usize, len: usize) -> bool {
463    match start.checked_add(len) {
464        Some(end) => len != 0 && start < ALLOCATED_VA_END && CELL_VA_START < end,
465        None => true,
466    }
467}

469/// Paging targets only: the bare-physical ones allocate no VA.
470#[cfg(not(cellos_wide))]
471pub(crate) fn overlaps_allocated_va(_start: usize, _len: usize) -> bool {
472    false
473}
