1//! Forward-edge CFI opt-in: the GNU property note of an admitted cell image.
2//!
3//! A cell asks for hardware forward-edge CFI exactly the way a Linux ELF does:
4//! the linker ANDs `GNU_PROPERTY_*_FEATURE_1_AND` across every input object and
5//! emits the result as a `NT_GNU_PROPERTY_TYPE_0` note in its own
6//! `PT_GNU_PROPERTY` segment. The bit survives only when *every* object was
7//! compiled with landing pads (`-Z branch-protection=bti` / `-mbranch-protection=bti`
8//! on AArch64, `-Z cf-protection=branch` / `-fcf-protection=branch` on x86_64),
9//! so it is the toolchain's statement that enforcing CFI cannot break the image.
10//!
11//! The note is read from the same bytes the loader maps — the image admission
12//! has already verified (signature, policy) — so the opt-in is covered by the
13//! signature like the rest of the image. Only `PT_GNU_PROPERTY` is consulted,
14//! as on Linux (`fs/binfmt_elf.c`): lld always emits it when a property note
15//! exists, unless a linker script declares `PHDRS` (the cell script does not).
16//!
17//! An image without the segment, or whose property lacks the bit, is not opted
18//! in and runs exactly as before. A *malformed* property segment refuses the
19//! spawn (Linux returns `ENOEXEC`): the note is signed toolchain output, so a
20//! malformed one means a broken or hostile image, not a choice to make silently.
21//!
22//! What the loader does with the result is [`arm`]: AArch64 maps the image's
23//! executable pages guarded (stage-1 `GP`), x86_64 gives the task a non-zero
24//! `IA32_U_CET`. Both only when the CPU implements the mechanism.
25
26use types::{ViError, ViResult};
27
28/// `p_type` of the program header that holds the property note.
29const PT_GNU_PROPERTY: u32 = 0x6474_e553;
30/// `n_type` of the property note.
31const NT_GNU_PROPERTY_TYPE_0: u32 = 5;
32/// `GNU_PROPERTY_AARCH64_FEATURE_1_AND` and its BTI bit.
33const GNU_PROPERTY_AARCH64_FEATURE_1_AND: u32 = 0xc000_0000;
34const GNU_PROPERTY_AARCH64_FEATURE_1_BTI: u32 = 1 << 0;
35/// `GNU_PROPERTY_X86_FEATURE_1_AND` and its IBT bit.
36const GNU_PROPERTY_X86_FEATURE_1_AND: u32 = 0xc000_0002;
37const GNU_PROPERTY_X86_FEATURE_1_IBT: u32 = 1 << 0;
38const EM_X86_64: u16 = 62;
39const EM_AARCH64: u16 = 183;
40/// Upper bound for the property segment. Real notes are 32 bytes; the bound
41/// keeps a hostile `p_filesz` from turning the walk into a long scan.
42const MAX_PROPERTY_BYTES: usize = 1024;
43/// ELF64 program-header entry size.
44const PHENT64: usize = 56;
45
46/// Forward-edge CFI mechanisms an image declares it is built for.
47#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
48pub struct ForwardEdgeCfi {
49    /// AArch64 BTI: every indirect-branch target carries a `BTI` landing pad.
50    pub bti: bool,
51    /// x86_64 CET-IBT: every indirect-branch target carries `ENDBR64`.
52    pub ibt: bool,
53}
54
55/// What the loader arms for one spawn, given the image's opt-in and the CPU.
56#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
57pub struct Armed {
58    /// AArch64: map the image's executable pages with the BTI `GP` bit.
59    pub guarded_code: bool,
60    /// x86_64: initial `IA32_U_CET` image for the task (0 = not armed).
61    pub user_cet: u64,
62}

64fn read_u16(data: &[u8], at: usize) -> ViResult<u16> {
65    let bytes = data
66        .get(at..at.checked_add(2).ok_or(ViError::InvalidInput)?)
67        .ok_or(ViError::InvalidInput)?;
68    Ok(u16::from_le_bytes([bytes[0], bytes[1]]))
69}
70
71fn read_u32(data: &[u8], at: usize) -> ViResult<u32> {
72    let bytes = data
73        .get(at..at.checked_add(4).ok_or(ViError::InvalidInput)?)
74        .ok_or(ViError::InvalidInput)?;
75    Ok(u32::from_le_bytes([bytes[0], bytes[1], bytes[2], bytes[3]]))
76}
77
78fn read_u64(data: &[u8], at: usize) -> ViResult<u64> {
79    let bytes = data
80        .get(at..at.checked_add(8).ok_or(ViError::InvalidInput)?)
81        .ok_or(ViError::InvalidInput)?;
82    let mut word = [0u8; 8];
83    word.copy_from_slice(bytes);
84    Ok(u64::from_le_bytes(word))
85}

87/// The forward-edge CFI opt-in of an ELF image.
88///
89/// Returns the default (nothing declared) for an image without
90/// `PT_GNU_PROPERTY`, for an ELF32 image (no 32-bit target has a mechanism to
91/// arm), and for a machine with no property this kernel arms.
92///
93/// # Errors
94/// `ViError::InvalidInput` for a truncated header or program-header table, more
95/// than one `PT_GNU_PROPERTY`, a segment outside the file or larger than
96/// [`MAX_PROPERTY_BYTES`], or a malformed note (see [`parse_property_note`]).
97pub fn forward_edge_cfi(elf: &[u8]) -> ViResult<ForwardEdgeCfi> {
98    // EI_CLASS 2 = ELF64, EI_DATA 1 = little-endian: the only layout any cell
99    // target with a CFI mechanism uses.
100    if elf.len() < 64 || elf[4] != 2 || elf[5] != 1 {
101        return Ok(ForwardEdgeCfi::default());
102    }
103    let machine = read_u16(elf, 18)?;
104    let phoff = usize::try_from(read_u64(elf, 32)?).map_err(|_| ViError::InvalidInput)?;
105    let phentsize = usize::from(read_u16(elf, 54)?);
106    let phnum = usize::from(read_u16(elf, 56)?);
107    if phnum == 0 {
108        return Ok(ForwardEdgeCfi::default());
109    }
110    if phentsize != PHENT64 {
111        return Err(ViError::InvalidInput);
112    }
113
114    let mut found: Option<&[u8]> = None;
115    for index in 0..phnum {
116        let ph = index
117            .checked_mul(PHENT64)
118            .and_then(|off| phoff.checked_add(off))
119            .ok_or(ViError::InvalidInput)?;
120        if read_u32(elf, ph)? != PT_GNU_PROPERTY {
121            continue;
122        }
123        if found.is_some() {
124            return Err(ViError::InvalidInput);
125        }
126        let offset = usize::try_from(read_u64(elf, ph + 8)?).map_err(|_| ViError::InvalidInput)?;
127        let size = usize::try_from(read_u64(elf, ph + 32)?).map_err(|_| ViError::InvalidInput)?;
128        if size > MAX_PROPERTY_BYTES {
129            return Err(ViError::InvalidInput);
130        }
131        let end = offset.checked_add(size).ok_or(ViError::InvalidInput)?;
132        found = Some(elf.get(offset..end).ok_or(ViError::InvalidInput)?);
133    }
134    match found {
135        Some(note) => parse_property_note(note, machine),
136        None => Ok(ForwardEdgeCfi::default()),
137    }
138}

140/// Parse the contents of a `PT_GNU_PROPERTY` segment for `machine` (ELF
141/// `e_machine`).
142///
143/// Layout (ELF64, every field little-endian, 8-byte alignment):
144/// `n_namesz = 4`, `n_descsz`, `n_type = NT_GNU_PROPERTY_TYPE_0`, `"GNU\0"`,
145/// then `n_descsz` bytes of properties, each `pr_type: u32`, `pr_datasz: u32`,
146/// `pr_data` padded to 8 bytes, in strictly ascending `pr_type` order.
147///
148/// # Errors
149/// `ViError::InvalidInput` when any of the above does not hold, or when the
150/// machine's `FEATURE_1_AND` property has a `pr_datasz` other than 4.
151pub fn parse_property_note(note: &[u8], machine: u16) -> ViResult<ForwardEdgeCfi> {
152    let namesz = read_u32(note, 0)?;
153    let descsz = usize::try_from(read_u32(note, 4)?).map_err(|_| ViError::InvalidInput)?;
154    let kind = read_u32(note, 8)?;
155    if namesz != 4 || kind != NT_GNU_PROPERTY_TYPE_0 || note.get(12..16) != Some(b"GNU\0") {
156        return Err(ViError::InvalidInput);
157    }
158    if !descsz.is_multiple_of(8) {
159        return Err(ViError::InvalidInput);
160    }
161    let desc = note
162        .get(16..16usize.checked_add(descsz).ok_or(ViError::InvalidInput)?)
163        .ok_or(ViError::InvalidInput)?;
164
165    let wanted = match machine {
166        EM_AARCH64 => Some(GNU_PROPERTY_AARCH64_FEATURE_1_AND),
167        EM_X86_64 => Some(GNU_PROPERTY_X86_FEATURE_1_AND),
168        _ => None,
169    };
170    let mut cfi = ForwardEdgeCfi::default();
171    let mut previous: Option<u32> = None;
172    let mut at = 0;
173    while at < desc.len() {
174        let pr_type = read_u32(desc, at)?;
175        let datasz = usize::try_from(read_u32(desc, at + 4)?).map_err(|_| ViError::InvalidInput)?;
176        let data_start = at + 8;
177        let data_end = data_start
178            .checked_add(datasz)
179            .filter(|&end| end <= desc.len())
180            .ok_or(ViError::InvalidInput)?;
181        if previous.is_some_and(|prev| pr_type <= prev) {
182            return Err(ViError::InvalidInput);
183        }
184        previous = Some(pr_type);
185        if Some(pr_type) == wanted {
186            if datasz != 4 {
187                return Err(ViError::InvalidInput);
188            }
189            let bits = read_u32(desc, data_start)?;
190            match machine {
191                EM_AARCH64 => cfi.bti = bits & GNU_PROPERTY_AARCH64_FEATURE_1_BTI != 0,
192                _ => cfi.ibt = bits & GNU_PROPERTY_X86_FEATURE_1_IBT != 0,
193            }
194        }
195        // `pr_data` is padded to the 8-byte ELF64 note alignment.
196        at = data_end
197            .checked_add(7)
198            .map(|end| end & !7)
199            .ok_or(ViError::InvalidInput)?;
200    }
201    Ok(cfi)
202}

204/// Decide what to arm for `cell` from its opt-in and this CPU, and log it.
205///
206/// The CPU half is what keeps an opted-in image running on hardware without
207/// the mechanism: BTI and ENDBR64 are NOPs there, and the `GP` bit is RES0
208/// before FEAT_BTI, so neither is set and the image runs unenforced — the same
209/// as an image that never opted in.
210pub fn arm(cfi: ForwardEdgeCfi, cell: &str) -> Armed {
211    Armed {
212        guarded_code: cfi.bti && arm_bti(cell),
213        user_cet: if cfi.ibt { arm_ibt(cell) } else { 0 },
214    }
215}
216
217/// AArch64: whether an image that opts into BTI gets guarded code pages.
218#[cfg(target_arch = "aarch64")]
219fn arm_bti(cell: &str) -> bool {
220    let available = crate::hal::cfi::detect().bti;
221    if available {
222        log::info!(
223            "[cfi] '{}': BTI armed (GNU property) — executable pages mapped guarded",
224            cell
225        );
226    } else {
227        log::info!(
228            "[cfi] '{}': image opts into BTI but the CPU lacks FEAT_BTI — not enforced",
229            cell
230        );
231    }
232    available
233}
234
235/// Other targets have no BTI; an AArch64 property never reaches them anyway
236/// (the parser only reads the property of the image's own `e_machine`).
237#[cfg(not(target_arch = "aarch64"))]
238fn arm_bti(_cell: &str) -> bool {
239    false
240}
241
242/// x86_64: the initial `IA32_U_CET` for a task whose image opts into IBT.
243#[cfg(all(target_arch = "x86_64", not(all(test, not(target_os = "none")))))]
244fn arm_ibt(cell: &str) -> u64 {
245    if crate::hal::cet::user_ibt_available() {
246        log::info!(
247            "[cfi] '{}': CET-IBT armed (GNU property) — IA32_U_CET.ENDBR_EN for this task",
248            cell
249        );
250        crate::hal::cet::USER_CET_IBT
251    } else {
252        log::info!(
253            "[cfi] '{}': image opts into CET-IBT but the CPU does not enumerate IBT — not enforced",
254            cell
255        );
256        0
257    }
258}

260/// Targets without CET (and the hosted test build, which cannot touch CR4).
261#[cfg(not(all(target_arch = "x86_64", not(all(test, not(target_os = "none"))))))]
262fn arm_ibt(_cell: &str) -> u64 {
263    0
264}

271    /// A property note carrying `props` (`(pr_type, pr_data)`), 8-byte padded.
272    fn note(props: &[(u32, &[u8])]) -> Vec<u8> {
273        let mut desc = Vec::new();
274        for (pr_type, data) in props {
275            desc.extend_from_slice(&pr_type.to_le_bytes());
276            desc.extend_from_slice(&(data.len() as u32).to_le_bytes());
277            desc.extend_from_slice(data);
278            while desc.len() % 8 != 0 {
279                desc.push(0);
280            }
281        }
282        let mut out = Vec::new();
283        out.extend_from_slice(&4u32.to_le_bytes());
284        out.extend_from_slice(&(desc.len() as u32).to_le_bytes());
285        out.extend_from_slice(&NT_GNU_PROPERTY_TYPE_0.to_le_bytes());
286        out.extend_from_slice(b"GNU\0");
287        out.extend_from_slice(&desc);
288        out
289    }

291    /// A minimal ELF64 LE image: header, program headers of `types`, then
292    /// `payload`; every `PT_GNU_PROPERTY` header points at `payload`.
293    fn elf(machine: u16, types: &[u32], payload: &[u8]) -> Vec<u8> {
294        let phoff = 64usize;
295        let data_off = phoff + types.len() * PHENT64;
296        let mut out = alloc::vec![0u8; data_off];
297        out[..4].copy_from_slice(b"\x7fELF");
298        out[4] = 2;
299        out[5] = 1;
300        out[18..20].copy_from_slice(&machine.to_le_bytes());
301        out[32..40].copy_from_slice(&(phoff as u64).to_le_bytes());
302        out[54..56].copy_from_slice(&(PHENT64 as u16).to_le_bytes());
303        out[56..58].copy_from_slice(&(types.len() as u16).to_le_bytes());
304        for (index, p_type) in types.iter().enumerate() {
305            let ph = phoff + index * PHENT64;
306            out[ph..ph + 4].copy_from_slice(&p_type.to_le_bytes());
307            out[ph + 8..ph + 16].copy_from_slice(&(data_off as u64).to_le_bytes());
308            out[ph + 32..ph + 40].copy_from_slice(&(payload.len() as u64).to_le_bytes());
309        }
310        out.extend_from_slice(payload);
311        out
312    }
