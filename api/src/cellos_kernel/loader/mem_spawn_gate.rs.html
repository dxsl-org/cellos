1//! Admission gate for ELF images handed to the kernel as caller-supplied bytes
2//! (`Syscall::SpawnFromMem`).
3//!
4//! Such a spawn arrives with a *name*, never a kernel-resolved path, and that name
5//! is fully caller-controlled. Every path-keyed decision inside
6//! [`super::spawn_gated`] — the `/bin/` manifest-privilege gate,
7//! [`super::legacy_path_caps`], `CapSet::with_path_caps`, the operator-policy
8//! lookup, the trusted-core recovery list, and the `/bin/vfs` block-region grant —
9//! would become caller-selectable if that name were passed through as the path: a
10//! `SpawnCap` holder naming `"/bin/vfs"` would inherit that install path's
11//! authority. This module therefore reduces the name to a label that provably
12//! matches none of those patterns, so the byte-derived trust checks (Ed25519
13//! signature, manifest) run while no path-derived authority can be forged.
14
15use alloc::string::String;
16use types::ViResult;
17
18/// Prefix carried by every label this module produces. The label invariant — never
19/// `/bin/`-prefixed, never equal to nor suffixed by a privileged install path — is
20/// what keeps a caller-supplied name from selecting path-based capabilities.
21const MEM_LABEL_PREFIX: &str = "/mem/";
22
23/// Component used when a name carries no character that survives filtering.
24const FALLBACK_NAME: &str = "cell";
25
26/// Longest label component retained; keeps the label far below `MAX_CELL_PATH`
27/// however long the caller's name is.
28const MAX_LABEL_NAME: usize = 64;
29
30/// Verify and spawn a cell from an ELF image the caller supplied as bytes.
31///
32/// Runs the one admission gate shared by every spawn path
33/// ([`super::spawn_gated`]): Ed25519 signature over the bytes (fail-closed under
34/// `signing-required`), manifest-privilege check, capability intersection with the
35/// spawner's ceiling, operator policy, syscall allowlist, cluster membership,
36/// memory quota, and integrity measurement. `caller_name` is advisory: it is
37/// reduced to a `/mem/` label before the gate sees it and can only ever cost the
38/// child privilege, never gain it.
39///
40/// The child's identity is derived by `task::spawn_from_mem` from the sentinel
41/// `CellId(0)` that `spawn_gated` passes — a cell spawned here must never keep
42/// `CellId(0)`, since both fault handlers treat a user-mode fault attributed to
43/// cell 0 as unattributable and panic.
44///
45/// # Errors
46/// - `ViError::PermissionDenied` — signature invalid, or absent under
47///   `signing-required`; or the image's manifest declares privilege that a cell
48///   outside `/bin/` may not hold.
49/// - `ViError::InvalidInput` — malformed ELF or unsupported relocation.
50/// - `ViError::OutOfMemory` — no frames available for the cell's segments.
51pub fn spawn_from_mem_gated(
52    elf_bytes: &[u8],
53    caller_name: &str,
54    spawner: crate::task::cap::Spawner,
55) -> ViResult<usize> {
56    let label = mem_label(caller_name);
57    log::info!(
58        "[loader] SpawnFromMem: {} ({} bytes, requested name {:?})",
59        label,
60        elf_bytes.len(),
61        caller_name
62    );
63    super::spawn_gated(elf_bytes, &label, spawner)
64}
65
66/// Reduce an untrusted name to a `/mem/`-prefixed advisory label.
67///
68/// Invariants upheld for every input, hostile ones included:
69/// - the component after the prefix holds no `/`, so no `ends_with` match against a
70///   `/bin/...` install path can succeed. Keeping only the final component is what
71///   enforces this; `is_label_char` rejecting `/` is redundant today and exists so
72///   the invariant survives a future change to the component extraction;
73/// - the result is never `/bin/`-prefixed, so the manifest-privilege gate treats
74///   the image as a user cell and `legacy_path_caps` grants nothing;
75/// - the result is never a member of the operator-policy or trusted-core path
76///   sets, both of which hold only exactly-matched `/bin/` paths.
77pub(super) fn mem_label(caller_name: &str) -> String {
78    let base = caller_name.rsplit('/').next().unwrap_or("");
79    let mut component = String::new();
80    for ch in base.chars().filter(is_label_char).take(MAX_LABEL_NAME) {
81        component.push(ch);
82    }
83    let mut label = String::from(MEM_LABEL_PREFIX);
84    label.push_str(match component.as_str() {
85        // "." and ".." are filtered out rather than kept: they are legal here (the
86        // consumers compare whole strings, never canonicalise) but would make the
87        // measurement-log label unreadable.
88        "" | "." | ".." => FALLBACK_NAME,
89        other => other,
90    });
91    label
92}
93
94/// Characters allowed in a label component; `/` is among those excluded, as the
95/// invariant documented on [`mem_label`] requires.
96fn is_label_char(c: &char) -> bool {
97    c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-')
98}
