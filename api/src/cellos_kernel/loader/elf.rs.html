6/// Maximum user-space virtual address for RISC-V SV39.
7///
8/// SV39 splits the 39-bit VA space into:
9///   0x0000_0000_0000 – 0x003F_FFFF_FFFF  (user / lower half, 256 GB)
10///   0xFFC0_0000_0000 – 0xFFFF_FFFF_FFFF  (kernel / upper half, 256 GB)
11///
12/// 0x8000_0000 was wrong — it only allowed 2 GB and coincidentally matched
13/// the RISC-V physical RAM base (0x8000_0000 on QEMU virt), causing every
14/// cell ELF compiled at 0x8800_0000+ to be rejected.
15///
16/// The real boundary is half the SV39 address space: 2^38 = 0x40_0000_0000.
17/// Cells compiled at 0x0040_0000 (4 MB) are safely within this range.
18// SV39 user-half upper bound (256 GB). On 32-bit targets, checked arithmetic
19// below is the address-space bound because usize::MAX cannot represent the
20// exclusive 4 GiB limit.
21#[cfg(not(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm")))]
22const USER_VADDR_MAX: usize = 0x40_0000_0000; // 256 GB — SV39 user half
23

26/// One 4 KiB page of a loaded cell image.
27///
28/// `final_flags` is the permission set the page must carry once the loader is
29/// done — the ELF's `p_flags`, OR-ed across every PT_LOAD that touches the page.
30/// It is deliberately NOT the set the page is mapped with during loading: see
31/// [`crate::loader::wx`] for why the two differ and when they converge.
32#[derive(Debug, Clone, Copy)]
33pub struct LoadedPage {
34    /// Page-aligned virtual address in the cell's address space.
35    pub va: VAddr,
36    /// Backing physical frame, reclaimed by `CellSegments` when the cell dies.
37    pub frame: PhysAddr,
38    /// Post-relocation permissions, applied by `wx::enforce`.
39    pub final_flags: crate::memory::paging::Flags,
40}
41
42impl ElfLoader {
43    /// Load loadable segments into memory.
44    ///
45    /// `load_base` is added to every segment's `p_vaddr` before mapping:
46    /// - For fixed-VA cells (ET_EXEC, non-PIE): pass `0` — uses p_vaddr directly.
47    /// - For PIE cells (ET_DYN): pass the VA base allocated by `va_alloc::alloc_cell_va`.
48    ///
49    /// Returns one [`LoadedPage`] per mapped page so the caller can (a) hand the
50    /// frames to `CellSegments` for reclamation and (b) run the W^X lowering pass
51    /// once relocations are applied.
52    ///
53    /// # Errors
54    /// - `ViError::InvalidInput` — malformed headers (file_size > mem_size, ranges
55    ///   outside the ELF buffer, arithmetic overflow).
56    /// - `ViError::PermissionDenied` — segment VA outside user space, a VA already
57    ///   owned by another live cell or kernel MMIO, or a PT_LOAD declaring W+X.
58    /// - `ViError::OutOfMemory` — no frame available, or the mapping failed.
59    pub fn load_segments(
60        &self,
61        data: &[u8],
62        frame_allocator: &mut crate::memory::frame::FrameAllocator,
63        load_base: usize,
64    ) -> ViResult<alloc::vec::Vec<LoadedPage>> {
65        let elf = ElfFile::new(data).map_err(|_| ViError::InvalidInput)?;
66        // Record each mapped page so the cell's segment frames can be reclaimed
67        // when it dies (see task::stack::CellSegments) — otherwise they leak — and
68        // so the post-relocation W^X pass knows each page's target permissions.
69        let mut mapped: alloc::vec::Vec<LoadedPage> = alloc::vec::Vec::new();
70
71        for (seg_index, ph) in elf.program_iter().enumerate() {
72            if let Ok(xmas_elf::program::Type::Load) = ph.get_type() {
73                let file_offset = usize::try_from(ph.offset())
74                    .map_err(|_| ViError::InvalidInput)
75                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
76                // For PIE cells load_base relocates every segment; for fixed-VA
77                // cells load_base == 0 so p_vaddr is used verbatim.
78                let segment_vaddr = usize::try_from(ph.virtual_addr())
79                    .map_err(|_| ViError::InvalidInput)
80                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
81                let vaddr = segment_vaddr
82                    .checked_add(load_base)
83                    .ok_or(ViError::InvalidInput)
84                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
85                let mem_size = usize::try_from(ph.mem_size())
86                    .map_err(|_| ViError::InvalidInput)
87                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
88                let file_size = usize::try_from(ph.file_size())
89                    .map_err(|_| ViError::InvalidInput)
90                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
91                let ph_flags = ph.flags();
92
93                // --- Header sanity checks ---
94                // W+X is refused before a single frame is allocated: a cell must
95                // not be able to opt out of W^X by declaring writable code.
96                super::wx::reject_wx_segment(
97                    seg_index,
98                    vaddr,
99                    ph_flags.is_write(),
100                    ph_flags.is_execute(),
101                )
102                .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
103
104                // file_size MUST NOT exceed mem_size (the rest is BSS).
105                if file_size > mem_size {
106                    log::error!(
107                        "ELF: rejecting segment with file_size={} > mem_size={}",
108                        file_size,
109                        mem_size
110                    );
111                    Self::unwind(&mapped, frame_allocator);
112                    return Err(ViError::InvalidInput);
113                }
114                // file_offset + file_size must fit inside the ELF buffer.
115                let file_end = file_offset
116                    .checked_add(file_size)
117                    .ok_or(ViError::InvalidInput)
118                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
119                if file_end > data.len() {
120                    log::error!(
121                        "ELF: segment file range {}..{} exceeds buffer len {}",
122                        file_offset,
123                        file_end,
124                        data.len()
125                    );
126                    Self::unwind(&mapped, frame_allocator);
127                    return Err(ViError::InvalidInput);
128                }
129                // vaddr + mem_size must not overflow and must lie below the
130                // kernel VA window — prevents user ELF clobbering kernel maps.
131                let end_addr = vaddr
132                    .checked_add(mem_size)
133                    .ok_or(ViError::InvalidInput)
134                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
135                #[cfg(not(any(
136                    target_arch = "riscv32",
137                    target_arch = "x86",
138                    target_arch = "arm"
139                )))]
140                if vaddr >= USER_VADDR_MAX || end_addr > USER_VADDR_MAX {
141                    log::error!(
142                        "ELF: segment VA range 0x{:X}-0x{:X} outside user space",
143                        vaddr,
144                        end_addr
145                    );
146                    Self::unwind(&mapped, frame_allocator);
147                    return Err(ViError::PermissionDenied);
148                }
149
150                let start_addr = vaddr;
151
152                // Align start/end to page boundaries
153                let start_page = start_addr & !(4096 - 1);
154                let end_page = end_addr
155                    .checked_add(4095)
156                    .ok_or(ViError::InvalidInput)
157                    .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?
158                    & !(4096 - 1);
159
160                // --- Translate ELF p_flags to page-table flags ---
161                // p_flags bits: 0x1=X, 0x2=W, 0x4=R.
162                //
163                // TWO flag sets per segment, and the difference is the whole point:
164                //
165                //   final_flags — what the ELF actually asks for. Recorded per page,
166                //                 OR-ed across PT_LOADs that share a boundary page,
167                //                 and applied by `wx::enforce` after relocation.
168                //   load_flags  — final_flags plus WRITE. Used only while the loader
169                //                 runs, because `.rela.dyn` patching writes through
170                //                 the cell's own VA on riscv64/x86_64.
171                //
172                // The WRITE bit therefore lives for exactly the load window. After
173                // `wx::enforce` a cell's `.text`/`.rodata` are hardware read-only,
174                // which is what stops one cell rewriting another's code in the SAS.
175                use crate::memory::paging::Flags;
176                let final_flags = super::wx::page_flags(ph_flags.is_write(), ph_flags.is_execute());
177                let flags = Flags::from_bits(final_flags.bits() | Flags::WRITE);
178
179                // Map pages
180                let mut current_page = start_page;
181                while current_page < end_page {
182                    // Overwrite guard: reject VA collision with kernel MMIO or a
183                    // *different* live cell. Allow shared pages from *this* cell's own
184                    // adjacent PT_LOAD segments (already_ours == true).
185                    //
186                    // PIE linker scripts may place adjacent sections (e.g. .text R-X
187                    // and .rodata R--) at non-page-aligned offsets so the two segments
188                    // share the same physical page.  In that case already_ours==true and
189                    // we reuse the frame, merging the new segment's flags into the page
190                    // rather than allocating a fresh one or blindly overwriting flags.
191                    let already_ours = mapped.iter().any(|p| p.va == current_page);
192                    if !already_ours && crate::memory::paging::virt_to_phys(current_page).is_some()
193                    {
194                        log::error!(
195                            "ELF: load VA 0x{:X} already mapped — rejecting spawn (VA collision with a live cell or kernel MMIO; fix the cell's linker script)",
196                            current_page
197                        );
198                        Self::unwind(&mapped, frame_allocator);
199                        return Err(ViError::PermissionDenied);
200                    }
201
202                    // Get the backing frame's kernel-accessible VA for copying.
203                    let frame_virt = if already_ours {
204                        // Shared page from an earlier LOAD segment (e.g. .rodata R-- and
205                        // .data/RELRO RW- sharing a boundary page).  Reuse the existing
206                        // frame but OR the new segment's permission bits in so the page
207                        // satisfies both segments' access requirements.
208                        let phys = crate::memory::paging::virt_to_phys(current_page)
209                            .expect("already_ours but virt_to_phys returned None");
210                        if let Some(entry) = mapped.iter_mut().find(|p| p.va == current_page) {
211                            // Merge the TARGET flags — this is the value `wx::enforce`
212                            // will apply, so the OR must happen here, on the page, and
213                            // never per-segment after the fact.
214                            let merged =
215                                Flags::from_bits(entry.final_flags.bits() | final_flags.bits());
216                            if merged != entry.final_flags {
217                                entry.final_flags = merged;
218                                let _ = crate::memory::paging::map_page(
219                                    frame_allocator,
220                                    current_page,
221                                    phys,
222                                    Flags::from_bits(merged.bits() | Flags::WRITE),
223                                );
224                            }
225                        }
226                        crate::memory::frame::phys_to_virt(phys)
227                    } else {
228                        let buf_frame = match frame_allocator.allocate_frame() {
229                            Some(f) => f,
230                            None => {
231                                log::warn!(
232                                    "[loader] segment frame allocation failed: va={:#x} mapped_pages={}",
233                                    current_page,
234                                    mapped.len()
235                                );
236                                Self::unwind(&mapped, frame_allocator);
237                                return Err(ViError::OutOfMemory);
238                            }
239                        };
240
241                        if crate::memory::paging::map_page(
242                            frame_allocator,
243                            current_page,
244                            buf_frame,
245                            flags,
246                        )
247                        .is_err()
248                        {
249                            log::warn!(
250                                "[loader] segment page-table allocation failed: va={:#x} mapped_pages={}",
251                                current_page,
252                                mapped.len()
253                            );
254                            // buf_frame is not in `mapped` yet — free it separately.
255                            frame_allocator.deallocate_frame(buf_frame);
256                            Self::unwind(&mapped, frame_allocator);
257                            return Err(ViError::OutOfMemory);
258                        }
259
260                        // Track for reclamation on cell death.
261                        mapped.push(LoadedPage {
262                            va: current_page,
263                            frame: buf_frame,
264                            final_flags,
265                        });
266
267                        // Zero the frame first (simplifies BSS and padding, and
268                        // prevents info-leak from previous frame owner).
269                        // Use phys_to_virt: on RISC-V it's a no-op (identity map);
270                        // on x86_64 physical RAM is only accessible via HHDM_BASE+phys.
271                        let fv = crate::memory::frame::phys_to_virt(buf_frame);
272                        unsafe {
273                            core::ptr::write_bytes(fv as *mut u8, 0, 4096);
274                        }
275                        fv
276                    };
277
278                    // Intersection of [page, page+4096) AND [vaddr, vaddr+file_size)
279                    let page_start_vaz = current_page;
280                    let page_end_vaz = current_page + 4096;
281                    let copy_start_v = core::cmp::max(page_start_vaz, vaddr);
282                    let copy_end_v = core::cmp::min(page_end_vaz, vaddr + file_size);
283
284                    if copy_start_v < copy_end_v {
285                        let len = copy_end_v - copy_start_v;
286                        let dst_offset = copy_start_v - page_start_vaz;
287                        let src_offset_in_file = file_offset + (copy_start_v - vaddr);
288                        // file_end was already validated above; this guards
289                        // arithmetic on `len` from any rounding surprise.
290                        let src_end = src_offset_in_file
291                            .checked_add(len)
292                            .ok_or(ViError::InvalidInput)
293                            .inspect_err(|_| Self::unwind(&mapped, frame_allocator))?;
294                        if src_end <= data.len() {
295                            let src = &data[src_offset_in_file..src_end];
296                            unsafe {
297                                let dst = (frame_virt as *mut u8).add(dst_offset);
298                                core::ptr::copy_nonoverlapping(src.as_ptr(), dst, len);
299                            }
300                        }
301                    }
302
303                    #[cfg(feature = "test-hooks")]
304                    crate::loader::atomic_publication_tests::observe_unpublished_pages(&mapped);
305
306                    if let Err(error) = crate::loader::atomic_checkpoint("AP-02") {
307                        Self::unwind(&mapped, frame_allocator);
308                        return Err(error);
309                    }
310
311                    current_page += 4096;
312                }
313
314                log::info!(
315                    "ELF LOAD: 0x{:X}-0x{:X} flags={}{}{}",
316                    start_addr,
317                    end_addr,
318                    if ph_flags.is_read() { 'R' } else { '-' },
319                    if ph_flags.is_write() { 'W' } else { '-' },
320                    if ph_flags.is_execute() { 'X' } else { '-' },
321                );
322            }
323        }
324        Ok(mapped)
325    }
326
327    /// Undo a partially loaded image: unmap every page recorded so far and
328    /// return its frame to the allocator.
329    ///
330    /// Every early return inside `load_segments` must call this. Without it a
331    /// rejected ELF leaves its already-mapped pages resident, which both leaks
332    /// the frames and poisons the VA-collision guard for the next spawn at the
333    /// same base.
334    fn unwind(mapped: &[LoadedPage], frame_allocator: &mut crate::memory::frame::FrameAllocator) {
335        for page in mapped {
336            let _ = crate::memory::paging::unmap_page(page.va);
337            crate::memory::tlb_shootdown::flush_page(page.va);
338        }
339        for page in mapped {
340            let reclaimed = crate::memory::paging::prune_empty_tables(page.va);
341            if !reclaimed.is_empty() {
342                crate::memory::paging::tlb_flush_all();
343            }
344            reclaimed.release(frame_allocator);
345            frame_allocator.deallocate_frame(page.frame);
346        }
347    }
348}

350impl ElfParser for ElfLoader {
351    fn parse_header(&self, data: &[u8]) -> ViResult<ElfHeader> {
352        let elf = ElfFile::new(data).map_err(|_| ViError::InvalidInput)?;
353
354        // Verify architecture (RISC-V 64)
355        // Header check is implicit in successful new(), but specific machine check?
356        // elf.header.pt2.machine() == xmas_elf::header::Machine::RISC_V
357
358        Ok(ElfHeader {
359            entry: elf.header.pt2.entry_point() as usize,
360            shoff: elf.header.pt2.sh_offset() as usize,
361        })
362    }
363
364    fn get_section<'a>(&self, data: &'a [u8], name: &str) -> ViResult<&'a [u8]> {
365        let elf = ElfFile::new(data).map_err(|_| ViError::InvalidInput)?;
366        match elf.find_section_by_name(name) {
367            Some(section) => Ok(section.raw_data(&elf)),
368            None => Err(ViError::NotFound),
369        }
370    }
