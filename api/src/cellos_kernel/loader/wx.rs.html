1//! W^X for cell pages: derive the final per-page permissions from the ELF's
2//! `p_flags` and apply them once relocation has finished.
3//!
4//! # Why a second pass exists
5//! `.rela.dyn` patching writes into `.text` and `.data.rel.ro`, so every cell
6//! page must be mapped WRITE while the loader runs. Leaving WRITE set afterwards
7//! means any cell holding one `unsafe` block can rewrite the code of every other
8//! cell in the single address space. This module closes the window.
9//!
10//! # Ordering contract (do not reorder)
11//! 1. `elf::load_segments` maps all pages WRITE and records the *target* flags,
12//!    OR-ing them across PT_LOADs that share a boundary page.
13//! 2. `reloc::apply_relocations` patches each owned frame through its kernel
14//!    alias (every arch), never through the cell's USER mapping.
15//! 3. `enforce` publishes the code bytes to instruction fetch on every online
16//!    hart (`crate::arch::icache`: AArch64 cache maintenance; RV64 local +
17//!    remote `fence.i`), then lowers each page to its recorded target flags.
18//! 4. Only then may the spawn path register the task with the scheduler.
19//!
20//! Applying step 3 per-segment instead of per-page would break a page shared by
21//! an R-X and an R-- segment: whichever segment was processed last would win
22//! and strip the other's rights.
23//!
24//! Step 4 is what turns "the cell never runs with a writable `.text`" from
25//! likely into guaranteed. Registration pushes the task onto a ready queue, from
26//! where another hart's work-stealing can start it on its next tick. RV64 pairs
27//! a changed PTE with ordered local `sfence.vma` and synchronous SBI RFENCE;
28//! x86_64 invalidates every PCID and global entry locally, then waits until
29//! every online CPU acknowledges its own local invalidation
30//! (`tlb_shootdown::flush_range`, one per contiguous run of lowered pages), and
31//! a CPU that does not acknowledge within the probe fails the lowering closed:
32//! the spawn is refused and its frames are quarantined. The loader keeps the
33//! stricter load -> relocate -> lower -> register ordering on every arch, even
34//! though AArch64 emits broadcast stage-1 TLBI today.
35//!
36//! # What is NOT guaranteed
37//! Remote stale-translation closure is arch-specific. RV64 requires a firmware
38//! RFENCE probe before secondary startup and still lacks its two-hart runtime
39//! witness; x86_64 closes it with the acknowledged shootdown above, witnessed
40//! on QEMU/KVM only; AArch64 already emits
41//! `tlbi vaae1is` (plus `vae2is` when EL2 is active) with the required
42//! shareable barriers but still lacks a checked two-PE runtime witness. See
43//! [`crate::memory::page_protect`] for the current contract.
44//!
45//! # What stays writable
46//! Cell stacks, heaps, grant pages, and MMIO windows are mapped by other paths
47//! and are untouched here. Kernel writes into cell memory (segment load, warm
48//! snapshot restore, relocation on every arch) go through the physical/HHDM
49//! alias, which carries kernel RW independently of the USER mapping's W bit.
50
51use crate::memory::paging::Flags;
52use types::{VAddr, ViError, ViResult};

54/// Base permission bits every cell page carries regardless of `p_flags`.
55///
56/// `READ` is unconditional: a PT_LOAD without `PF_R` is legal ELF but yields a
57/// page the cell cannot even fetch its own constants from, and no toolchain in
58/// this workspace emits one. `ACCESSED`/`DIRTY` are pre-set because the RISC-V
59/// spec permits an implementation to fault instead of updating them in hardware.
60const BASE: usize = Flags::VALID | Flags::USER | Flags::READ | Flags::ACCESSED | Flags::DIRTY;
61
62/// Final page-table flags for a page covered by a PT_LOAD with these `p_flags`.
63///
64/// `WRITE` and `EXECUTE` are the only bits the ELF's `p_flags` influence. OR
65/// the results together when several PT_LOADs share one page.
66///
67/// `guarded` is the loader's BTI decision for the whole image
68/// ([`super::gnu_property::arm`]): it adds `GUARDED` (AArch64 stage-1 `GP`) to
69/// executable pages only. A data page never carries it, and an image that did
70/// not opt in (or a CPU without FEAT_BTI) gets exactly the flags it always had.
71pub fn page_flags(write: bool, execute: bool, guarded: bool) -> Flags {
72    let mut bits = BASE;
73    if write {
74        bits |= Flags::WRITE;
75    }
76    if execute {
77        bits |= Flags::EXECUTE;
78        if guarded {
79            bits |= Flags::GUARDED;
80        }
81    }
82    Flags::from_bits(bits)
83}

85/// Reject a PT_LOAD that declares both `PF_W` and `PF_X`.
86///
87/// A self-declared W+X segment is the obvious way around this whole pass: the
88/// cell asks for writable code and the loader obliges. Refusing it means a cell
89/// that genuinely needs a JIT has to change its linker script or manifest in a
90/// diff a reviewer sees, rather than winning the permission silently at load
91/// time. No cell in this workspace declares W+X.
92///
93/// # Errors
94/// `ViError::PermissionDenied` when both bits are set.
95pub fn reject_wx_segment(index: usize, vaddr: VAddr, write: bool, execute: bool) -> ViResult<()> {
96    if write && execute {
97        log::error!(
98            "ELF: rejecting spawn — PT_LOAD #{} at 0x{:X} declares W+X; \
99             writable code is refused (split the segment or drop PF_X in the linker script)",
100            index,
101            vaddr
102        );
103        return Err(ViError::PermissionDenied);
104    }
105    Ok(())
106}

108/// Reject a page whose permissions come out both writable and executable after
109/// the boundary-page merge.
110///
111/// [`reject_wx_segment`] cannot catch this: two PT_LOADs, one R-X and one R-W,
112/// are each individually legal yet OR together into W+X when they share a 4 KiB
113/// page, and that merge does not exist yet when the per-segment check runs. A
114/// linker pads to a page boundary across a permission change, so a merged W+X
115/// page means a hand-authored or hostile section layout rather than toolchain
116/// output — and in a single address space one such page is a writable,
117/// executable region reachable from every cell in the system, which is exactly
118/// the primitive this pass exists to remove.
119///
120/// Dropping WRITE instead of refusing would silently break a legitimate `.data`
121/// boundary page; refusal is the only choice that cannot corrupt a valid cell.
122///
123/// # Errors
124/// `ViError::PermissionDenied` when both bits survive the merge.
125pub fn reject_wx_page(cell: &str, vaddr: VAddr, flags: Flags) -> ViResult<()> {
126    if flags.bits() & Flags::WRITE != 0 && flags.bits() & Flags::EXECUTE != 0 {
127        log::error!(
128            "[wx] refusing to spawn '{}': page 0x{:X} is W+X after the boundary merge — \
129             two PT_LOADs with different permissions share it; align the segments to 4 KiB",
130            cell,
131            vaddr
132        );
133        return Err(ViError::PermissionDenied);
134    }
135    Ok(())
136}

138/// Lower every loaded cell page from the loader's writable mapping to its final
139/// ELF-derived flags.
140///
141/// Call AFTER `.rela.dyn` has been applied — see the ordering contract above.
142///
143/// Fail-closed: a page that cannot be lowered aborts the whole pass so the
144/// caller kills the half-spawned cell, rather than letting it run with a
145/// writable `.text`.
146///
147/// # Errors
148/// - `ViError::PermissionDenied` if any page's merged flags are W+X. The ELF's
149///   section layout is malformed or hostile; no page has been touched yet.
150/// - `ViError::InvalidInput` if a page recorded at load time is no longer mapped,
151///   or the page table rejects the new flags. Both indicate a loader bug, not a
152///   malformed ELF, so the message names the page.
153#[cfg(not(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm")))]
154pub fn enforce(pages: &[(VAddr, Flags)], cell: &str) -> ViResult<()> {
155    // Validate the whole set before lowering any of it, so a refused ELF leaves
156    // the mapping uniformly writable for the caller to tear down, rather than
157    // half-lowered in a state no later code is written to expect.
158    for &(va, flags) in pages {
159        reject_wx_page(cell, va, flags)?;
160    }
161
162    // AArch64 and RV64 instruction fetch is not coherent with stores, and the
163    // hart that dequeues the task may not be this one: publish the relocated
164    // code bytes before removing WRITE and before the task can be registered.
165    crate::arch::icache::sync_loaded_code(pages, cell)?;
166
167    // One batch (phase 07b R1): `protect_pages` writes every leaf, then flushes
168    // each contiguous run once — on the error path too — and returns `Ok` only
169    // when every online CPU acknowledged. An unconfirmed CPU is the
170    // `InvalidationUnconfirmed` error, which refuses the spawn below, and the
171    // failed spawn's teardown quarantines its frames until that CPU answers.
172    // So the lowering is in force on every CPU before step 4 registers the
173    // task; riscv64/aarch64 keep the per-page pass.
174    let lowered = pages
175        .iter()
176        .filter(|(_, flags)| flags.bits() & Flags::WRITE == 0)
177        .count();
178    crate::memory::paging::protect_pages(pages).map_err(|(va, e)| {
179        log::error!(
180            "[wx] '{}' failed to protect page 0x{:X}: {:?} — killing cell",
181            cell,
182            va,
183            e
184        );
185        ViError::InvalidInput
186    })?;
187    log::info!(
188        "[wx] '{}': {}/{} cell pages lowered to read-only/execute-only",
189        cell,
190        lowered,
191        pages.len()
192    );
193    Ok(())
194}

196/// Bare-physical arches (riscv32 Nano, x86_32, arm32) run with no page tables,
197/// so there is no permission to lower. Report the gap rather than returning a
198/// silent success that reads like enforcement happened.
199#[cfg(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm"))]
200pub fn enforce(pages: &[(VAddr, Flags)], cell: &str) -> ViResult<()> {
201    log::warn!(
202        "[wx] '{}': {} pages left writable — no MMU on this target, W^X unavailable",
203        cell,
204        pages.len()
205    );
206    Ok(())
207}

209// ─── Boot-time self-tests ────────────────────────────────────────────────────
210//
211// Written as plain `pub fn` + `assert!` rather than `#[cfg(test)]` to match
212// `loader::elf_tests`: the kernel is built for bare-metal targets where
213// `cargo test` never runs, so a `#[cfg(test)]` module would be neither executed
214// nor even type-checked. These compile on every build and are invoked from
215// `elf_tests::run_all`.
216
217/// Assert the flag-derivation rules that the whole pass rests on.
218///
219/// # Panics
220/// Panics (halting boot) if any invariant is violated — a wrong flag table is
221/// worse than a failed boot because it silently disables the isolation.
222pub fn run_self_tests() {
223    // .text — R-X: must lose WRITE, keep EXECUTE and USER.
224    let text = page_flags(false, true, false);
225    assert_eq!(text.bits() & Flags::WRITE, 0, ".text must not be writable");
226    assert_ne!(
227        text.bits() & Flags::EXECUTE,
228        0,
229        ".text must stay executable"
230    );
231    assert_ne!(text.bits() & Flags::USER, 0, "cell pages stay user pages");
232
233    // .rodata — R--: neither writable nor executable.
234    let rodata = page_flags(false, false, false);
235    assert_eq!(rodata.bits() & Flags::WRITE, 0, ".rodata must be read-only");
236    assert_eq!(
237        rodata.bits() & Flags::EXECUTE,
238        0,
239        ".rodata must not be executable"
240    );
241    assert_ne!(rodata.bits() & Flags::READ, 0, ".rodata must be readable");
242
243    // .data — RW-: keeps WRITE.
244    let data = page_flags(true, false, false);
245    assert_ne!(data.bits() & Flags::WRITE, 0, ".data must stay writable");
246    assert_eq!(
247        data.bits() & Flags::EXECUTE,
248        0,
249        ".data must not be executable"
250    );
251
252    // Boundary page shared by .rodata (R--) and .data (RW-) ORs to RW-.
253    let merged = Flags::from_bits(rodata.bits() | data.bits());
254    assert_ne!(
255        merged.bits() & Flags::WRITE,
256        0,
257        "merged boundary page must satisfy the writable segment"
258    );
259    assert_eq!(merged.bits() & Flags::EXECUTE, 0);
260    assert!(
261        reject_wx_page("selftest", 0x4000, merged).is_ok(),
262        "an R-- / RW- boundary page is legitimate and must still load"
263    );
264
265    // Boundary page shared by .text (R-X) and .data (RW-) ORs to RWX. Neither
266    // segment declares W+X, so only the post-merge check can refuse it.
267    let hostile = Flags::from_bits(text.bits() | data.bits());
268    assert_ne!(hostile.bits() & Flags::WRITE, 0);
269    assert_ne!(hostile.bits() & Flags::EXECUTE, 0);
270    assert!(
271        reject_wx_segment(0, 0x5000, false, true).is_ok()
272            && reject_wx_segment(1, 0x5000, true, false).is_ok(),
273        "the two contributing segments are individually legal"
274    );
275    assert!(
276        reject_wx_page("selftest", 0x5000, hostile).is_err(),
277        "a merged W+X page must fail the spawn, not merely warn"
278    );
279
280    // R-X, R-- and RW- pages are the common case and must never be refused.
281    assert!(reject_wx_page("selftest", 0x6000, text).is_ok());
282    assert!(reject_wx_page("selftest", 0x7000, rodata).is_ok());
283    assert!(reject_wx_page("selftest", 0x8000, data).is_ok());
284
285    // W+X is refused; every single-permission combination is accepted.
286    assert!(
287        reject_wx_segment(0, 0x1000, true, true).is_err(),
288        "W+X segment must be rejected"
289    );
290    assert!(reject_wx_segment(0, 0x1000, true, false).is_ok());
291    assert!(reject_wx_segment(1, 0x2000, false, true).is_ok());
292    assert!(reject_wx_segment(2, 0x3000, false, false).is_ok());
293
294    assert_guard_selection();
295
296    log::info!("[wx] self-tests PASSED");
297}

299/// BTI `GP` selection: only executable pages of an armed image are guarded,
300/// and an image that is not armed gets exactly the unguarded flags.
301///
302/// # Panics
303/// Panics if a data page could be guarded, an armed code page is not, or the
304/// unarmed flags differ from what every cell received before the opt-in.
305pub fn assert_guard_selection() {
306    for write in [false, true] {
307        for execute in [false, true] {
308            let plain = page_flags(write, execute, false);
309            let armed = page_flags(write, execute, true);
310            assert_eq!(
311                plain.bits() & Flags::GUARDED,
312                0,
313                "an unarmed image is never guarded"
314            );
315            assert_eq!(
316                armed.bits() & Flags::GUARDED != 0,
317                execute,
318                "an armed image guards exactly its executable pages"
319            );
320            assert_eq!(
321                armed.bits() & !Flags::GUARDED,
322                plain.bits(),
323                "arming changes nothing but GUARDED"
324            );
325        }
326    }
327    // A boundary page merged from guarded R-X and unguarded R-- keeps the guard:
328    // its instructions still belong to the opted-in image.
329    let merged = page_flags(false, true, true).bits() | page_flags(false, false, true).bits();
330    assert_ne!(merged & Flags::GUARDED, 0);
331}
