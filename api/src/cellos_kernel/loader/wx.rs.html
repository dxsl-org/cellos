1//! W^X for cell pages: derive the final per-page permissions from the ELF's
2//! `p_flags` and apply them once relocation has finished.
3//!
4//! # Why a second pass exists
5//! `.rela.dyn` patching writes into `.text` and `.data.rel.ro`, so every cell
6//! page must be mapped WRITE while the loader runs. Leaving WRITE set afterwards
7//! means any cell holding one `unsafe` block can rewrite the code of every other
8//! cell in the single address space. This module closes the window.
9//!
10//! # Ordering contract (do not reorder)
11//! 1. `elf::load_segments` maps all pages WRITE and records the *target* flags,
12//!    OR-ing them across PT_LOADs that share a boundary page.
13//! 2. `reloc::apply_relocations` patches through those writable mappings.
14//! 3. `enforce` lowers each page to its recorded target flags.
15//! 4. Only then may the spawn path register the task with the scheduler.
16//!
17//! Applying step 3 per-segment instead of per-page would break a page shared by
18//! an R-X and an R-- segment: whichever segment was processed last would win
19//! and strip the other's rights.
20//!
21//! Step 4 is what turns "the cell never runs with a writable `.text`" from
22//! likely into guaranteed. Registration pushes the task onto a ready queue, from
23//! where another hart's work-stealing can start it on its next tick. RV64 pairs
24//! a changed PTE with ordered local `sfence.vma` and synchronous SBI RFENCE;
25//! x86_64 remains local `invlpg` only. The loader keeps the stricter load ->
26//! relocate -> lower -> register ordering on every arch, even though AArch64
27//! emits broadcast stage-1 TLBI today.
28//!
29//! # What is NOT guaranteed
30//! Remote stale-translation closure is arch-specific. RV64 requires a firmware
31//! RFENCE probe before secondary startup and still lacks its two-hart runtime
32//! witness; x86_64 lacks SMP IPI shootdown; AArch64 already emits
33//! `tlbi vaae1is` (plus `vae2is` when EL2 is active) with the required
34//! shareable barriers but still lacks a checked two-PE runtime witness. See
35//! [`crate::memory::page_protect`] for the current contract.
36//!
37//! # What stays writable
38//! Cell stacks, heaps, grant pages, and MMIO windows are mapped by other paths
39//! and are untouched here. Kernel writes into cell memory (segment load, warm
40//! snapshot restore, AArch64 relocation) go through the physical/HHDM alias,
41//! which carries kernel RW independently of the USER mapping's W bit.
42
43use crate::memory::paging::Flags;
44use types::{VAddr, ViError, ViResult};

46/// Base permission bits every cell page carries regardless of `p_flags`.
47///
48/// `READ` is unconditional: a PT_LOAD without `PF_R` is legal ELF but yields a
49/// page the cell cannot even fetch its own constants from, and no toolchain in
50/// this workspace emits one. `ACCESSED`/`DIRTY` are pre-set because the RISC-V
51/// spec permits an implementation to fault instead of updating them in hardware.
52const BASE: usize = Flags::VALID | Flags::USER | Flags::READ | Flags::ACCESSED | Flags::DIRTY;
53
54/// Final page-table flags for a page covered by a PT_LOAD with these `p_flags`.
55///
56/// `WRITE` and `EXECUTE` are the only bits the ELF gets to influence. OR the
57/// results together when several PT_LOADs share one page.
58pub fn page_flags(write: bool, execute: bool) -> Flags {
59    let mut bits = BASE;
60    if write {
61        bits |= Flags::WRITE;
62    }
63    if execute {
64        bits |= Flags::EXECUTE;
65    }
66    Flags::from_bits(bits)
67}

69/// Reject a PT_LOAD that declares both `PF_W` and `PF_X`.
70///
71/// A self-declared W+X segment is the obvious way around this whole pass: the
72/// cell asks for writable code and the loader obliges. Refusing it means a cell
73/// that genuinely needs a JIT has to change its linker script or manifest in a
74/// diff a reviewer sees, rather than winning the permission silently at load
75/// time. No cell in this workspace declares W+X.
76///
77/// # Errors
78/// `ViError::PermissionDenied` when both bits are set.
79pub fn reject_wx_segment(index: usize, vaddr: VAddr, write: bool, execute: bool) -> ViResult<()> {
80    if write && execute {
81        log::error!(
82            "ELF: rejecting spawn — PT_LOAD #{} at 0x{:X} declares W+X; \
83             writable code is refused (split the segment or drop PF_X in the linker script)",
84            index,
85            vaddr
86        );
87        return Err(ViError::PermissionDenied);
88    }
89    Ok(())
90}

92/// Reject a page whose permissions come out both writable and executable after
93/// the boundary-page merge.
94///
95/// [`reject_wx_segment`] cannot catch this: two PT_LOADs, one R-X and one R-W,
96/// are each individually legal yet OR together into W+X when they share a 4 KiB
97/// page, and that merge does not exist yet when the per-segment check runs. A
98/// linker pads to a page boundary across a permission change, so a merged W+X
99/// page means a hand-authored or hostile section layout rather than toolchain
100/// output — and in a single address space one such page is a writable,
101/// executable region reachable from every cell in the system, which is exactly
102/// the primitive this pass exists to remove.
103///
104/// Dropping WRITE instead of refusing would silently break a legitimate `.data`
105/// boundary page; refusal is the only choice that cannot corrupt a valid cell.
106///
107/// # Errors
108/// `ViError::PermissionDenied` when both bits survive the merge.
109pub fn reject_wx_page(cell: &str, vaddr: VAddr, flags: Flags) -> ViResult<()> {
110    if flags.bits() & Flags::WRITE != 0 && flags.bits() & Flags::EXECUTE != 0 {
111        log::error!(
112            "[wx] refusing to spawn '{}': page 0x{:X} is W+X after the boundary merge — \
113             two PT_LOADs with different permissions share it; align the segments to 4 KiB",
114            cell,
115            vaddr
116        );
117        return Err(ViError::PermissionDenied);
118    }
119    Ok(())
120}

122/// Lower every loaded cell page from the loader's writable mapping to its final
123/// ELF-derived flags.
124///
125/// Call AFTER `.rela.dyn` has been applied — see the ordering contract above.
126///
127/// Fail-closed: a page that cannot be lowered aborts the whole pass so the
128/// caller kills the half-spawned cell, rather than letting it run with a
129/// writable `.text`.
130///
131/// # Errors
132/// - `ViError::PermissionDenied` if any page's merged flags are W+X. The ELF's
133///   section layout is malformed or hostile; no page has been touched yet.
134/// - `ViError::InvalidInput` if a page recorded at load time is no longer mapped,
135///   or the page table rejects the new flags. Both indicate a loader bug, not a
136///   malformed ELF, so the message names the page.
137#[cfg(not(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm")))]
138pub fn enforce(pages: &[(VAddr, Flags)], cell: &str) -> ViResult<()> {
139    // Validate the whole set before lowering any of it, so a refused ELF leaves
140    // the mapping uniformly writable for the caller to tear down, rather than
141    // half-lowered in a state no later code is written to expect.
142    for &(va, flags) in pages {
143        reject_wx_page(cell, va, flags)?;
144    }
145
146    // AArch64 has non-coherent data and instruction caches. Relocations are the
147    // final writes to executable pages, so publish those bytes before removing
148    // WRITE and before another hart can dequeue the task.
149    #[cfg(target_arch = "aarch64")]
150    for &(va, flags) in pages {
151        if flags.bits() & Flags::EXECUTE != 0 {
152            let phys = crate::memory::paging::virt_to_phys(va).ok_or_else(|| {
153                log::error!(
154                    "[wx] '{}' executable page 0x{:X} disappeared before cache sync",
155                    cell,
156                    va
157                );
158                ViError::InvalidInput
159            })?;
160            let data_va = crate::memory::frame::phys_to_virt(phys);
161            // SAFETY: `data_va` is the physical alias used by segment copy and
162            // relocation; `va` maps that same page for the cell's instruction
163            // fetch. Both page-sized ranges are mapped and cannot overflow.
164            unsafe {
165                hal::sync_instruction_cache(data_va, va, crate::memory::paging::PAGE_SIZE);
166            }
167        }
168    }
169
170    let mut lowered = 0usize;
171    for &(va, flags) in pages {
172        if flags.bits() & Flags::WRITE == 0 {
173            lowered += 1;
174        }
175        crate::memory::paging::protect_page(va, flags).map_err(|e| {
176            log::error!(
177                "[wx] '{}' failed to protect page 0x{:X}: {:?} — killing cell",
178                cell,
179                va,
180                e
181            );
182            ViError::InvalidInput
183        })?;
184    }
185    log::info!(
186        "[wx] '{}': {}/{} cell pages lowered to read-only/execute-only",
187        cell,
188        lowered,
189        pages.len()
190    );
191    Ok(())
192}

194/// Bare-physical arches (riscv32 Nano, x86_32, arm32) run with no page tables,
195/// so there is no permission to lower. Report the gap rather than returning a
196/// silent success that reads like enforcement happened.
197#[cfg(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm"))]
198pub fn enforce(pages: &[(VAddr, Flags)], cell: &str) -> ViResult<()> {
199    log::warn!(
200        "[wx] '{}': {} pages left writable — no MMU on this target, W^X unavailable",
201        cell,
202        pages.len()
203    );
204    Ok(())
205}

207// ─── Boot-time self-tests ────────────────────────────────────────────────────
208//
209// Written as plain `pub fn` + `assert!` rather than `#[cfg(test)]` to match
210// `loader::elf_tests`: the kernel is built for bare-metal targets where
211// `cargo test` never runs, so a `#[cfg(test)]` module would be neither executed
212// nor even type-checked. These compile on every build and are invoked from
213// `elf_tests::run_all`.
214
215/// Assert the flag-derivation rules that the whole pass rests on.
216///
217/// # Panics
218/// Panics (halting boot) if any invariant is violated — a wrong flag table is
219/// worse than a failed boot because it silently disables the isolation.
220pub fn run_self_tests() {
221    // .text — R-X: must lose WRITE, keep EXECUTE and USER.
222    let text = page_flags(false, true);
223    assert_eq!(text.bits() & Flags::WRITE, 0, ".text must not be writable");
224    assert_ne!(
225        text.bits() & Flags::EXECUTE,
226        0,
227        ".text must stay executable"
228    );
229    assert_ne!(text.bits() & Flags::USER, 0, "cell pages stay user pages");
230
231    // .rodata — R--: neither writable nor executable.
232    let rodata = page_flags(false, false);
233    assert_eq!(rodata.bits() & Flags::WRITE, 0, ".rodata must be read-only");
234    assert_eq!(
235        rodata.bits() & Flags::EXECUTE,
236        0,
237        ".rodata must not be executable"
238    );
239    assert_ne!(rodata.bits() & Flags::READ, 0, ".rodata must be readable");
240
241    // .data — RW-: keeps WRITE.
242    let data = page_flags(true, false);
243    assert_ne!(data.bits() & Flags::WRITE, 0, ".data must stay writable");
244    assert_eq!(
245        data.bits() & Flags::EXECUTE,
246        0,
247        ".data must not be executable"
248    );
249
250    // Boundary page shared by .rodata (R--) and .data (RW-) ORs to RW-.
251    let merged = Flags::from_bits(rodata.bits() | data.bits());
252    assert_ne!(
253        merged.bits() & Flags::WRITE,
254        0,
255        "merged boundary page must satisfy the writable segment"
256    );
257    assert_eq!(merged.bits() & Flags::EXECUTE, 0);
258    assert!(
259        reject_wx_page("selftest", 0x4000, merged).is_ok(),
260        "an R-- / RW- boundary page is legitimate and must still load"
261    );
262
263    // Boundary page shared by .text (R-X) and .data (RW-) ORs to RWX. Neither
264    // segment declares W+X, so only the post-merge check can refuse it.
265    let hostile = Flags::from_bits(text.bits() | data.bits());
266    assert_ne!(hostile.bits() & Flags::WRITE, 0);
267    assert_ne!(hostile.bits() & Flags::EXECUTE, 0);
268    assert!(
269        reject_wx_segment(0, 0x5000, false, true).is_ok()
270            && reject_wx_segment(1, 0x5000, true, false).is_ok(),
271        "the two contributing segments are individually legal"
272    );
273    assert!(
274        reject_wx_page("selftest", 0x5000, hostile).is_err(),
275        "a merged W+X page must fail the spawn, not merely warn"
276    );
277
278    // R-X, R-- and RW- pages are the common case and must never be refused.
279    assert!(reject_wx_page("selftest", 0x6000, text).is_ok());
280    assert!(reject_wx_page("selftest", 0x7000, rodata).is_ok());
281    assert!(reject_wx_page("selftest", 0x8000, data).is_ok());
282
283    // W+X is refused; every single-permission combination is accepted.
284    assert!(
285        reject_wx_segment(0, 0x1000, true, true).is_err(),
286        "W+X segment must be rejected"
287    );
288    assert!(reject_wx_segment(0, 0x1000, true, false).is_ok());
289    assert!(reject_wx_segment(1, 0x2000, false, true).is_ok());
290    assert!(reject_wx_segment(2, 0x3000, false, false).is_ok());
291
292    log::info!("[wx] self-tests PASSED");
293}
