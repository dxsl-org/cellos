1//! Default-off, internal-only gate for future native-domain admission.
2//!
3//! A successful result is an immutable policy-generation lease, not permission
4//! to expose a Tier-2 route. No public loader or manifest path constructs a
5//! request; test hooks use this module solely to prove denial and drain rules.
6
7#![allow(dead_code)]
8use crate::task::cap::CapSet;
9use core::sync::atomic::{AtomicU64, Ordering};
10
11const DISABLED: u64 = 0;
12const ENABLED: u64 = 1;
13const DRAINING: u64 = 2;
14static POLICY: AtomicU64 = AtomicU64::new(DISABLED);
15static GENERATION: AtomicU64 = AtomicU64::new(1);
16
17#[derive(Clone, Copy, Debug, Eq, PartialEq)]
18pub(crate) enum DomainAdmissionDenial {
19    FeatureDisabled,
20    PolicyDisabled,
21    PolicyDraining,
22    UnsupportedArchitecture,
23    ResourceQuota,
24    ArtifactIneligible,
25    CopiedIpcUnavailable,
26    UnenforceableCapability,
27}
28
29/// A held policy generation. Publication must recheck it after every fallible
30/// operation; a drain invalidates outstanding leases before it can commit.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub(crate) struct DomainAdmissionLease(u64);
33
34impl DomainAdmissionLease {
35    pub(crate) fn remains_enabled(self) -> bool {
36        POLICY.load(Ordering::Acquire) == ENABLED && GENERATION.load(Ordering::Acquire) == self.0
37    }
38}
39
40/// Kernel-only request shape. It intentionally has no manifest/parser form.
41#[derive(Clone, Copy)]
42pub(crate) struct DomainAdmissionRequest {
43    pub(crate) resource_quota_available: bool,
44    pub(crate) artifact_eligible: bool,
45    pub(crate) copied_ipc_ready: bool,
46    pub(crate) requested_caps: CapSet,
47    pub(crate) requests_dma: bool,
48}
49
50impl DomainAdmissionRequest {
51    pub(crate) const fn fixture() -> Self {
52        Self {
53            resource_quota_available: true,
54            artifact_eligible: true,
55            copied_ipc_ready: true,
56            requested_caps: CapSet::EMPTY,
57            requests_dma: false,
58        }
59    }
60}
61
62/// Evaluate all enforceable predicates before a builder or task can exist.
63pub(crate) fn evaluate_domain_admission(
64    request: DomainAdmissionRequest,
65) -> Result<DomainAdmissionLease, DomainAdmissionDenial> {
66    if !cfg!(feature = "native-domains") {
67        return Err(DomainAdmissionDenial::FeatureDisabled);
68    }
69    if !cfg!(target_arch = "riscv64") {
70        return Err(DomainAdmissionDenial::UnsupportedArchitecture);
71    }
72    match POLICY.load(Ordering::Acquire) {
73        ENABLED => {}
74        DRAINING => return Err(DomainAdmissionDenial::PolicyDraining),
75        _ => return Err(DomainAdmissionDenial::PolicyDisabled),
76    }
77    if !request.resource_quota_available {
78        return Err(DomainAdmissionDenial::ResourceQuota);
79    }
80    if !request.artifact_eligible {
81        return Err(DomainAdmissionDenial::ArtifactIneligible);
82    }
83    if !request.copied_ipc_ready {
84        return Err(DomainAdmissionDenial::CopiedIpcUnavailable);
85    }
86    if request.requested_caps != CapSet::EMPTY || request.requests_dma {
87        return Err(DomainAdmissionDenial::UnenforceableCapability);
88    }
89    Ok(DomainAdmissionLease(GENERATION.load(Ordering::Acquire)))
90}

92/// Begin the boot-local one-way rollback. Existing domain teardown owns the
93/// transition from draining to disabled; no caller may re-enable this boot.
94pub(crate) fn begin_domain_drain() -> bool {
95    POLICY
96        .compare_exchange(ENABLED, DRAINING, Ordering::AcqRel, Ordering::Acquire)
97        .is_ok()
98}

100#[cfg(feature = "test-hooks")]
101pub(crate) fn run_selftest() {
102    let disabled = evaluate_domain_admission(DomainAdmissionRequest::fixture())
103        == Err(DomainAdmissionDenial::PolicyDisabled);
104    if disabled {
105        log::info!("S22-RV64-ADMISSION-DENY: PASS");
106    } else {
107        log::error!("S22-RV64-ADMISSION-DENY: FAIL");
108    }
109
110    POLICY.store(ENABLED, Ordering::Release);
111    let lease = evaluate_domain_admission(DomainAdmissionRequest::fixture());
112    let drained = begin_domain_drain()
113        && lease.is_ok_and(|lease| !lease.remains_enabled())
114        && evaluate_domain_admission(DomainAdmissionRequest::fixture())
115            == Err(DomainAdmissionDenial::PolicyDraining);
116    POLICY.store(DISABLED, Ordering::Release);
117    if drained {
118        log::info!("S22-RV64-ADMISSION-DRAIN: PASS");
119    } else {
120        log::error!("S22-RV64-ADMISSION-DRAIN: FAIL");
121    }
122}
