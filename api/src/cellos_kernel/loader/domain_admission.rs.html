1//! Tier 2 admission policy — the single on-path control for domain launches.
2//!
3//! ADR-0019: a domain launch is admitted only while this policy is `ENABLED`, the
4//! build has a qualified backend, the architecture is covered by that backend,
5//! the artifact class is domain-eligible, the copied-IPC boundary exists, the
6//! resource quota is available, and the requested authority is one a domain root
7//! can actually enforce. The loader evaluates the policy before it creates
8//! anything, holds the returned lease across every fallible admission step, and
9//! re-checks it immediately before publication — so a concurrent drain either
10//! linearizes before the admission or is observed by it, never in between.
11//!
12//! The cargo feature `native-domains` only selects whether a backend is compiled;
13//! it is never an admission decision (ADR-0019 §2.2). A build with the backend
14//! and no enabled policy denies domain-class artifacts; it never falls back to
15//! the shared address space.
16
17use crate::task::cap::CapSet;
18use core::sync::atomic::{AtomicU64, Ordering};
19use types::ViError;
20
21const DISABLED: u64 = 0;
22const ENABLED: u64 = 1;
23const DRAINING: u64 = 2;
24static POLICY: AtomicU64 = AtomicU64::new(DISABLED);
25static GENERATION: AtomicU64 = AtomicU64::new(1);
26
27#[derive(Clone, Copy, Debug, Eq, PartialEq)]
28pub(crate) enum DomainAdmissionDenial {
29    FeatureDisabled,
30    UnsupportedArchitecture,
31    PolicyDisabled,
32    PolicyDraining,
33    ResourceQuota,
34    ArtifactIneligible,
35    CopiedIpcUnavailable,
36    UnenforceableCapability,
37    /// This architecture's raw context switch activates the incoming root before
38    /// the outgoing context is saved (phase-02 gate).
39    SwitchOrderingUnqualified,
40    /// A grant record owned by, or shared to, a private-root task is live, so the
41    /// containment gate cannot guarantee its receiver PTEs are revoked.
42    LiveDomainGrant,
43}
44
45impl DomainAdmissionDenial {
46    /// Stable code for the audit ring, so a denial is greppable after the fact.
47    pub(crate) fn audit_code(self) -> u32 {
48        match self {
49            Self::FeatureDisabled => 1,
50            Self::UnsupportedArchitecture => 2,
51            Self::PolicyDisabled => 3,
52            Self::PolicyDraining => 4,
53            Self::ResourceQuota => 5,
54            Self::ArtifactIneligible => 6,
55            Self::CopiedIpcUnavailable => 7,
56            Self::UnenforceableCapability => 8,
57            Self::SwitchOrderingUnqualified => 9,
58            Self::LiveDomainGrant => 10,
59        }
60    }
61
62    /// The error the loader returns. A denial is final: it never downgrades to
63    /// SAS, and it never publishes a partial task or domain.
64    pub(crate) fn error(self) -> ViError {
65        match self {
66            Self::FeatureDisabled
67            | Self::UnsupportedArchitecture
68            | Self::SwitchOrderingUnqualified => ViError::NotSupported,
69            Self::ResourceQuota => ViError::OutOfMemory,
70            Self::PolicyDisabled
71            | Self::PolicyDraining
72            | Self::ArtifactIneligible
73            | Self::CopiedIpcUnavailable
74            | Self::UnenforceableCapability
75            | Self::LiveDomainGrant => ViError::PermissionDenied,
76        }
77    }
78}

80/// A held policy generation. Publication must recheck it after every fallible
81/// operation; a drain invalidates outstanding leases before it can commit.
82#[derive(Clone, Copy, Debug, Eq, PartialEq)]
83pub(crate) struct DomainAdmissionLease(u64);

85impl DomainAdmissionLease {
86    pub(crate) fn remains_enabled(self) -> bool {
87        POLICY.load(Ordering::Acquire) == ENABLED && GENERATION.load(Ordering::Acquire) == self.0
88    }
89}

92/// the on-disk artifact states a class, never an admission decision.

96    pub(crate) artifact_eligible: bool,

123/// Is this architecture's domain switch ordering qualified?
124///
125/// Phase-01 held every architecture but RV64 closed: `task.rs` activated the
126/// incoming private root before `Context::switch` saved the outgoing context, so
127/// a domain could go live with unsaved outgoing state. Phase 02 moved the root
128/// write *inside* the switch — `Context::switch_with_root` programs `TTBR0_EL1`
129/// after the outgoing context is stored and before the incoming stack is
130/// adopted — and proved on one AArch64 CPU that the root-*writing* path executes
131/// for SAS→domain and domain→domain, that the live `TTBR0_EL1` carries the
132/// expected ASID at every stop, that the kernel root is restored afterwards, and
133/// that a real domain cell runs at EL0 under its own root (`S22-AARCH64-ROOT-SWITCH`,
134/// `S22-AARCH64-DOMAIN-LIVE`).
135///
136/// The ordering change is structural hardening, not a demonstrated fix: the A/B
137/// in `a77545341` inverted the write order and the fixture still passed, because
138/// the AArch64 save writes only to the context struct and reads no stack memory.
139/// What the reopen rests on is therefore the executed root-writing path plus the
140/// per-root isolation witnesses, not a discriminating ordering test.
141///
142/// x86_64 grew the same in-switch root write — `switch_with_root` composes CR3
143/// with `domain::cr3_for` and programs it between the outgoing save and the
144/// incoming stack adopt — and the phase-02 x86 test image now enters a real
145/// domain cell through it, reading the live CR3 from inside the domain's own
146/// kernel context (`S22-X86-DOMAIN-LIVE`) and tearing the root down with every
147/// frame returned and nothing quarantined.
148///
149/// x86_64 single-CPU switch ordering is production-qualified across QEMU TCG,
150/// KVM and VMware (`docs/evidence/vmware-domain-test.txt`). Actual SMP admission
151/// separately requires this boot's bidirectional IPI/local-invalidation proof;
152/// configuration capacity alone is never that proof.
153/// AArch64 remains test-images only.
154pub(crate) const fn switch_ordering_qualified() -> bool {
155    cfg!(target_arch = "riscv64")
156        || cfg!(target_arch = "x86_64")
157        || cfg!(all(target_arch = "aarch64", feature = "test-hooks"))
158}
159
160/// AArch64 reopen is confined to test images.
161///
162/// `enable_for_boot` returns `false` (and the boot stays `PolicyDisabled`) on
163/// production AArch64 builds, so an unhardened image cannot admit a domain-class
164/// cell by accident. Const-evaluated, so it fails the build rather than a boot.
165#[cfg(all(
166    feature = "native-domains",
167    target_arch = "aarch64",
168    not(feature = "test-hooks")
169))]
170const _: () = assert!(!switch_ordering_qualified());

172/// Does this build contain a domain backend for this architecture?
173///
174/// This is the same cfg set as the route that creates a domain in
175/// `task::launch`; the policy and the route must not disagree about coverage
176/// (ADR-0019 §2.4).
177pub(crate) const fn architecture_covered() -> bool {
178    cfg!(feature = "native-domains")
179        && cfg!(any(
180            target_arch = "riscv64",
181            target_arch = "aarch64",
182            target_arch = "x86_64"
183        ))
184}

186/// Authority a Tier 2 domain root cannot enforce, and therefore must not hold.
187///
188/// A domain root maps only the cell's own image, stack, heap and explicit
189/// grants: it has no user MMIO window, and `RequestMmio` has no domain-PTE route,
190/// so device and DMA authority in a domain cell is dead authority whose request
191/// path would still mutate the shared root. Device-class cells (drivers,
192/// `/bin/vfs`, `/bin/platform`) are Tier 1 for that reason. Authority a domain
193/// *can* enforce — network client capability, spawn authority bounded by this
194/// same policy for every child, and service-registration authority — stays
195/// admissible.
196pub(crate) fn unenforceable_authority(caps: CapSet, requests_dma: bool) -> bool {
197    requests_dma
198        || caps.mmio_devices != 0
199        || caps.pcie_driver
200        || caps.usb_driver
201        || caps.platform
202        || caps.block_io
203        || caps.block_regions != 0
204        || caps.hypervisor
205        || caps.supervisor
206}

208/// Build the request for a domain-class launch from the state the loader holds.
209///
210/// The only caller is `task::launch::publish_prepared`, the single publication
211/// point for ELF tasks; a second admission path would defeat ADR-0019 §2.1.
212pub(crate) fn evaluate_for_launch(
213    granted: CapSet,
214) -> Result<DomainAdmissionLease, DomainAdmissionDenial> {
215    evaluate_domain_admission(DomainAdmissionRequest {
216        // The quota reservation is taken before this call, so an unavailable
217        // quota surfaces as `OutOfMemory` from the reservation itself.
218        resource_quota_available: true,
219        // `is_domain` is the class decision (unsigned, FFI, UNTRUSTED) that the
220        // signature gate already made; authority is bounded below instead.
221        artifact_eligible: true,
222        copied_ipc_ready: architecture_covered(),
223        requested_caps: granted,
224        requests_dma: granted.pcie_driver || granted.platform || granted.usb_driver,
225    })
226}

228/// Evaluate the launch and audit a denial before it becomes an error.
229///
230/// The audit record is what makes a refusal observable after the fact: a cell
231/// that never appears because admission refused it leaves no task to inspect.
232pub(crate) fn admit_for_launch(granted: CapSet) -> Result<DomainAdmissionLease, ViError> {
233    evaluate_for_launch(granted).map_err(|denial| {
234        crate::audit::log_event(
235            crate::audit::AuditEvent::CellSpawnDenied,
236            &crate::audit::encode_u32x2(0, denial.audit_code()),
237        );
238        denial.error()
239    })
240}

242/// Refuse a domain-class publication while a domain-backed grant record is live.
243///
244/// Deliberately **not** part of [`evaluate_domain_admission`]: the caller
245/// (`task::launch::publish_prepared`) holds the scheduler lock across that
246/// evaluation, and the grant tables must only ever be taken in the documented
247/// `*_GRANT_TABLE → SCHEDULER` order — never underneath `SCHEDULER`. The
248/// publication path calls this first, before taking that lock, so the preflight
249/// keeps its on-path position without the inversion.
250///
251/// A denial is final and audited: no task, no domain, and no SAS fallback.
252pub(crate) fn refuse_while_domain_grant_live() -> Result<(), ViError> {
253    if !crate::task::syscall::domain_grant_records_live() {
254        return Ok(());
255    }
256    let denial = DomainAdmissionDenial::LiveDomainGrant;
257    crate::audit::log_event(
258        crate::audit::AuditEvent::CellSpawnDenied,
259        &crate::audit::encode_u32x2(0, denial.audit_code()),
260    );
261    Err(denial.error())
262}
263
264/// Refuse a launch whose lease died between creation and publication.
265pub(crate) fn drain_refusal() -> ViError {
266    let denial = DomainAdmissionDenial::PolicyDraining;
267    crate::audit::log_event(
268        crate::audit::AuditEvent::CellSpawnDenied,
269        &crate::audit::encode_u32x2(0, denial.audit_code()),
270    );
271    denial.error()
272}

274/// Evaluate all enforceable predicates before a builder or task can exist.
275pub(crate) fn evaluate_domain_admission(
276    request: DomainAdmissionRequest,
277) -> Result<DomainAdmissionLease, DomainAdmissionDenial> {
278    if !cfg!(feature = "native-domains") {
279        return Err(DomainAdmissionDenial::FeatureDisabled);
280    }
281    if !architecture_covered() {
282        return Err(DomainAdmissionDenial::UnsupportedArchitecture);
283    }
284    if !switch_ordering_qualified() {
285        return Err(DomainAdmissionDenial::SwitchOrderingUnqualified);
286    }
287    match POLICY.load(Ordering::Acquire) {
288        ENABLED => {}
289        DRAINING => return Err(DomainAdmissionDenial::PolicyDraining),
290        _ => return Err(DomainAdmissionDenial::PolicyDisabled),
291    }
292    if !request.resource_quota_available {
293        return Err(DomainAdmissionDenial::ResourceQuota);
294    }
295    if !request.artifact_eligible {
296        return Err(DomainAdmissionDenial::ArtifactIneligible);
297    }
298    if !request.copied_ipc_ready {
299        return Err(DomainAdmissionDenial::CopiedIpcUnavailable);
300    }
301    if unenforceable_authority(request.requested_caps, request.requests_dma) {
302        return Err(DomainAdmissionDenial::UnenforceableCapability);
303    }
304    Ok(DomainAdmissionLease(GENERATION.load(Ordering::Acquire)))
305}

307/// Enter the boot's admission posture. Called once during boot for a profile
308/// that runs domain-class cells; a build that never calls it denies them
309/// (fail-closed), which is the fleet posture.
310pub(crate) fn enable_for_boot() -> bool {
311    // Phase-01 posture: a build whose raw switch ordering is unqualified stays
312    // fail-closed instead of entering the enabled posture, because the admission
313    // control must never be stronger than the mechanism it guards. On AArch64
314    // that is every production build; the test-hooks image qualifies and enables.
315    if !switch_ordering_qualified() {
316        return false;
317    }
318    // A PE that cannot program a private root cannot run one. At EL2 the AArch64
319    // switch has no root argument (`__switch_el2`, and `switch_with_root` asserts
320    // its absence), so a "qualified" ordering there would be a claim about a
321    // mechanism the machine does not have. Refuse the posture rather than admit a
322    // domain cell that would enter without its own mappings.
323    #[cfg(target_arch = "aarch64")]
324    if crate::hal::aarch64::el2::is_el2() {
325        return false;
326    }
327    // The x86_64 analogue: a PE that never recorded its own kernel CR3 cannot
328    // run a private root either. Admission runs after `init_kernel_paging_x86`
329    // published the kernel PML4 in this boot, but the switch composes the
330    // domain CR3 against a *known* kernel root, and trap entry only knows to
331    // install it from `VI_KERNEL_CR3`. With that unset a "qualified" ordering
332    // would be a claim about a mechanism this boot does not have — the same
333    // refusal, for the same reason, as the EL2 branch above.
334    #[cfg(target_arch = "x86_64")]
335    if crate::hal::domain::kernel_cr3() == 0
336        || !crate::task::smp::x86_substrate_qualified() {
337        return false;
338    }
339    POLICY
340        .compare_exchange(DISABLED, ENABLED, Ordering::AcqRel, Ordering::Acquire)
341        .is_ok()
342}

344/// Begin the boot-local one-way rollback. Existing domain teardown owns the
345/// transition from draining to disabled; no caller may re-enable this boot.
346#[allow(dead_code)] // reason: emergency admission rollback; no production caller until an operator channel lands (ADR-0019 §2.1), exercised by the boot selftest today
347pub(crate) fn begin_domain_drain() -> bool {
348    POLICY
349        .compare_exchange(ENABLED, DRAINING, Ordering::AcqRel, Ordering::Acquire)
350        .is_ok()
351}

366/// Marker prefix for the boot selftest: the RV64 lane greps `S22-RV64-ADMISSION-*`,
367/// the AArch64 lane `S22-AARCH64-ADMISSION-*`, the x86_64 lane `S22-X86-ADMISSION-*`.
368/// The rendered marker text on RV64 is byte-identical to the pre-phase-02 literals,

379    target_arch = "aarch64"

384    feature = "test-hooks",
385    target_arch = "x86_64"

389/// Boot-time assertions for the admission control. The posture cases are the
390/// evidence for ADR-0019: the policy denies when disabled or draining, an
391/// outstanding lease dies with a drain, and the *publication path* refuses a
392/// domain-class launch with nothing published and no SAS fallback.
393///
394/// The denial cases are the reason this runs on every architecture that can
395/// carry a private root: `DENY`/`DRAIN`/`PUBLICATION-DENY` are the fleet
396/// posture, asserted on the same code path the fleet profile leaves closed.
397#[cfg(all(
398    feature = "native-domains",
399    feature = "test-hooks",
400    any(
401        target_arch = "riscv64",
402        target_arch = "aarch64",
403        target_arch = "x86_64"
404    )
405))]
406pub(crate) fn run_selftest() -> bool {
407    let mut all_ok = true;
408
409    // The boot posture is part of the contract: a domain-class cell only runs
410    // because boot enabled admission, not because a default feature admitted it.
411    let enabled = policy_is_enabled();
412    all_ok &= enabled;
413    if enabled {
414        log::info!("S22-{}-ADMISSION-ENABLED: PASS", ADMISSION_TAG);
415    } else {
416        log::error!(
417            "S22-{}-ADMISSION-ENABLED: FAIL — boot posture did not enable admission",
418            ADMISSION_TAG
419        );
420    }
421
422    // Disabled posture denies, and a denial maps to a final error, never SAS.
423    POLICY.store(DISABLED, Ordering::Release);
424    let disabled = evaluate_domain_admission(DomainAdmissionRequest::fixture())
425        == Err(DomainAdmissionDenial::PolicyDisabled)
426        && DomainAdmissionDenial::PolicyDisabled.error() == ViError::PermissionDenied;
427    all_ok &= disabled;
428    if disabled {
429        log::info!("S22-{}-ADMISSION-DENY: PASS", ADMISSION_TAG);
430    } else {
431        log::error!("S22-{}-ADMISSION-DENY: FAIL", ADMISSION_TAG);
432    }
433
434    // Enabled admits, and a drain invalidates the lease an admission is holding.
435    POLICY.store(ENABLED, Ordering::Release);
436    let lease = evaluate_domain_admission(DomainAdmissionRequest::fixture());
437    let drained = begin_domain_drain()
438        && lease.is_ok_and(|lease| !lease.remains_enabled())
439        && evaluate_domain_admission(DomainAdmissionRequest::fixture())
440            == Err(DomainAdmissionDenial::PolicyDraining);
441    all_ok &= drained;
442    if drained {
443        log::info!("S22-{}-ADMISSION-DRAIN: PASS", ADMISSION_TAG);
444    } else {
445        log::error!("S22-{}-ADMISSION-DRAIN: FAIL", ADMISSION_TAG);
446    }
447
448    // The route itself: while draining, the single publication point must refuse
449    // a domain-class launch with no task, no domain, and no SAS fallback.
450    let publication_denied = publication_is_refused_while_draining();
451    all_ok &= publication_denied;
452    if publication_denied {
453        log::info!("S22-{}-ADMISSION-PUBLICATION-DENY: PASS", ADMISSION_TAG);
454    } else {
455        log::error!("S22-{}-ADMISSION-PUBLICATION-DENY: FAIL", ADMISSION_TAG);
456    }
457
458    // Unenforceable authority is refused by name, so a device-class artifact
459    // cannot be admitted as a domain by a future manifest edit.
460    let mut device_caps = CapSet::EMPTY;
461    device_caps.mmio_devices = 0b1;
462    let ceiling = unenforceable_authority(device_caps, false)
463        && unenforceable_authority(CapSet::EMPTY, true)
464        && !unenforceable_authority(CapSet::EMPTY, false);
465    all_ok &= ceiling;
466    if ceiling {
467        log::info!("S22-{}-ADMISSION-CEILING: PASS", ADMISSION_TAG);
468    } else {
469        log::error!("S22-{}-ADMISSION-CEILING: FAIL", ADMISSION_TAG);
470    }
471
472    // Leave the boot in the posture the boot policy chose: the cases above moved
473    // the policy for their own observation, and the rest of this boot runs cells.
474    POLICY.store(ENABLED, Ordering::Release);
475    all_ok
476}

478/// Drive the real publication path with a domain-class launch while the policy
479/// is draining and observe that nothing is published.
480#[cfg(all(
481    feature = "native-domains",
482    feature = "test-hooks",
483    any(
484        target_arch = "riscv64",
485        target_arch = "aarch64",
486        target_arch = "x86_64"
487    )
488))]
489fn publication_is_refused_while_draining() -> bool {
490    use crate::task::{LaunchRoutes, TaskLaunchState};
491
492    // AArch64 and x86_64 have no `domain_identity_counter`, so the "no domain was
493    // created" half of this proof is carried by the ledger a private root is
494    // made of: the
495    // snapshot is taken *before* the ELF is prepared, so a refusal that publishes
496    // nothing must return the ledger to exactly this value — the task's own
497    // stacks and segments are allocated and released inside the call, and a
498    // private root and its tables are frames too. (Taken after preparation the
499    // comparison is one-sided and meaningless: the refusal's release of the
500    // task's own frames reads as a decrease.)
501    #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
502    let frames_before = frames_in_use();
503    let Ok(prepared) = crate::task::prepare_elf_task(
504        crate::INIT_ELF,
505        "admission-publication-probe",
506        types::CellId(0),
507        alloc::vec::Vec::new(),
508    ) else {
509        return false;
510    };
511    let state = TaskLaunchState::complete(
512        None,
513        CapSet::EMPTY,
514        None,
515        None,
516        crate::memory::cell_quota::DEFAULT_QUOTA_BYTES,
517        u64::MAX,
518        0,
519        0,
520        api::TaskPriority::Normal as u8,
521        0,
522        0,
523        false,
524        0,
525        None,
526        LaunchRoutes {
527            block_io: false,
528            input: false,
529            development_silo: false,
530        },
531        None,
532        true,
533    );
534
535    let tasks_before = crate::task::scheduler_stats().0;
536    #[cfg(target_arch = "riscv64")]
537    let domains_before = crate::memory::address_space::domain_identity_counter();
538    let outcome = crate::task::publish_prepared(prepared, state);
539    let tasks_after = crate::task::scheduler_stats().0;
540    #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
541    let frames_after = frames_in_use();
542    // `domain_identity_counter` is RV64-only, but the claim it carries — a refused
543    // launch creates no domain — is arch-neutral and is the one that matters: a
544    // private root and its tables *are* frames, so a refusal that still moved the
545    // frame ledger would mean a domain was built and retained for a launch that
546    // never published. `create_cell_domain` is the only allocator on this path,
547    // and the drain refusal precedes it.
548    #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
549    let domains_unchanged = frames_before.is_some() && frames_after == frames_before;
550    #[cfg(target_arch = "riscv64")]
551    let domains_unchanged =
552        crate::memory::address_space::domain_identity_counter() == domains_before;
553
554    let refused = outcome == Err(ViError::PermissionDenied);
555    let unpublished = tasks_after == tasks_before;
556    if !(refused && unpublished && domains_unchanged) {
557        log::error!(
558            "S22-{}-ADMISSION-PUBLICATION-DENY detail: refused={} unpublished={} no_domain={} tasks {}->{} outcome={:?}",
559            ADMISSION_TAG,
560            refused,
561            unpublished,
562            domains_unchanged,
563            tasks_before,
564            tasks_after,
565            outcome
566        );
567        #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
568        log::error!(
569            "S22-{}-ADMISSION-PUBLICATION-DENY detail: frames {:?} -> {:?}",
570            ADMISSION_TAG,
571            frames_before,
572            frames_after
573        );
574    }
575
576    refused && unpublished && domains_unchanged
577}

579/// Frames the frame allocator has handed out, or `None` before it is published.
580///
581/// AArch64 and x86_64 have no domain-identity counter, so the refusal proof above
582/// is carried by the ledger a private root is actually made of. Its only caller is
583/// the test-hooks publication probe, so it carries that caller's cfg — a
584/// production AArch64 or x86_64 build has no use for it and `-D warnings` refuses
585/// dead code.
586#[cfg(all(
587    feature = "native-domains",
588    feature = "test-hooks",
589    any(target_arch = "aarch64", target_arch = "x86_64")
590))]
591fn frames_in_use() -> Option<usize> {
592    crate::memory::frame::FRAME_ALLOCATOR
593        .lock()
594        .as_ref()
595        .map(|allocator| allocator.used_frames())
596}
