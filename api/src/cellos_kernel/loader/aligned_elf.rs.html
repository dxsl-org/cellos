1//! Kernel-owned, 8-aligned ELF bytes for admission and loading.
2//!
3//! Admission reads an image several times — SHA-256, Ed25519 `verify_cell`,
4//! manifest classification, segment load — and every pass must see the bytes
5//! the previous one judged. Caller memory (a grant or user buffer) stays
6//! writable by the caller's other threads and by grant peers, so an image from
7//! there must be captured into an [`ElfSnapshot`] before the first pass.
8
9use alloc::vec::Vec;
10use types::{ViError, ViResult};
11
12/// An owned copy of an ELF image. The `u64` backing keeps it 8-aligned so
13/// [`bytes`] borrows it instead of copying a second time.
14pub(crate) struct ElfSnapshot {
15    words: Vec<u64>,
16    len: usize,
17}
18
19impl ElfSnapshot {
20    /// A zero-filled snapshot of `len` bytes for the caller to fill once.
21    ///
22    /// `len` is the image's own (caller-chosen) size, so the reservation is
23    /// fallible: `vec![0u64; ..]` would reach the infallible alloc error handler
24    /// and halt the kernel instead of refusing the spawn.
25    pub(crate) fn zeroed(len: usize) -> ViResult<Self> {
26        let words_len = len.div_ceil(8);
27        let mut words = Vec::new();
28        words
29            .try_reserve_exact(words_len)
30            .map_err(|_| ViError::OutOfMemory)?;
31        words.resize(words_len, 0);
32        Ok(Self { words, len })
33    }
34
35    pub(crate) fn bytes(&self) -> &[u8] {
36        // SAFETY: the `u64` backing holds at least `len` initialized bytes.
37        unsafe { core::slice::from_raw_parts(self.words.as_ptr().cast::<u8>(), self.len) }
38    }
39
40    pub(crate) fn bytes_mut(&mut self) -> &mut [u8] {
41        // SAFETY: the `u64` backing holds at least `len` initialized bytes and
42        // `&mut self` makes this the only view of them.
43        unsafe { core::slice::from_raw_parts_mut(self.words.as_mut_ptr().cast::<u8>(), self.len) }
44    }
45}
46
47pub(crate) enum AlignedElf<'a> {
48    Borrowed(&'a [u8]),
49    Owned(ElfSnapshot),
50}
51
52impl AsRef<[u8]> for AlignedElf<'_> {
53    fn as_ref(&self) -> &[u8] {
54        match self {
55            Self::Borrowed(data) => data,
56            Self::Owned(snapshot) => snapshot.bytes(),
57        }
58    }
59}
60
61/// View `data` 8-aligned, borrowing it when it already is.
62///
63/// Precondition: `data` is kernel-owned and immutable while the result lives
64/// (a disk/VIFS read, an embedded image, or an [`ElfSnapshot`]). The aligned
65/// fast path reads `data` itself on every pass, so caller memory passed here
66/// reopens the verify-then-load TOCTOU.
67pub(crate) fn bytes(data: &[u8]) -> ViResult<AlignedElf<'_>> {
68    if (data.as_ptr() as usize).is_multiple_of(8) {
69        return Ok(AlignedElf::Borrowed(data));
70    }
71    let mut copy = ElfSnapshot::zeroed(data.len())?;
72    copy.bytes_mut().copy_from_slice(data);
73    Ok(AlignedElf::Owned(copy))
74}

76#[cfg(all(
77    test,
78    feature = "dev-signing-key",
79    not(feature = "phase02-terminal-profile")
80))]
81mod tests {
82    use super::{bytes, AlignedElf, ElfSnapshot};
83    use crate::loader::elf_tests::{build_minimal_signed_elf, MINIMAL_ELF_DEV_SIG};
84
85    /// A writer that still holds the source after capture (another caller
86    /// thread, a grant peer) cannot change what admission verifies or loads.
87    #[test]
88    fn snapshot_keeps_signed_bytes_after_source_rewrite() {
89        let mut source = build_minimal_signed_elf(MINIMAL_ELF_DEV_SIG);
90        let original_digest = crate::sha256::sha256(&source);
91        let sig = crate::signing::extract_sig(&source).expect("fixture carries a signature");
92
93        let mut snapshot = ElfSnapshot::zeroed(source.len()).expect("fixture fits the heap");
94        snapshot.bytes_mut().copy_from_slice(&source);
95        source[120] ^= 0xff; // Rewrite the PT_LOAD code bytes after capture.
96
97        assert!(!crate::signing::verify_cell(&source, &sig).expect("verifier has memory"));
98        let admitted = bytes(snapshot.bytes()).expect("aligned view");
99        assert!(matches!(admitted, AlignedElf::Borrowed(_)));
100        assert!(crate::signing::verify_cell(admitted.as_ref(), &sig).expect("verifier has memory"));
101        assert_eq!(crate::sha256::sha256(admitted.as_ref()), original_digest);
102        assert_eq!(admitted.as_ref()[120], 0);
103    }
