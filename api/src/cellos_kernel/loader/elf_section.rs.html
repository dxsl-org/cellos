1//! Bounded, byte-oriented ELF section-table access for untrusted cell images.
2
3const ELF_MAGIC: &[u8; 4] = b"\x7fELF";
4const ELFCLASS32: u8 = 1;
5const ELFCLASS64: u8 = 2;
6const ELFDATA2LSB: u8 = 1;
7const ELFDATA2MSB: u8 = 2;
8const SHT_STRTAB: u64 = 3;
9pub(crate) const SHT_NOBITS: u64 = 8;
10const SHN_XINDEX: u16 = 0xffff;
11
12#[derive(Clone, Copy)]
13pub(crate) struct Section {
14    pub(crate) name: usize,
15    pub(crate) kind: u64,
16    pub(crate) offset: usize,
17    pub(crate) size: usize,
18}
19
20#[derive(Clone, Copy)]
21struct Header {
22    class: u8,
23    little: bool,
24    shoff: usize,
25    shentsize: usize,
26    shnum: usize,
27    shstrndx: usize,
28}
29
30/// Validated section-table view. File-backed ranges remain checked by callers:
31/// `SHT_NOBITS` intentionally has no backing bytes in an ELF container.
32pub(crate) struct ElfSections<'a> {
33    data: &'a [u8],
34    header: Header,
35}
36
37impl<'a> ElfSections<'a> {
38    /// Parse an ELF with structurally bounded program and section tables.
39    pub(crate) fn parse(data: &'a [u8]) -> Option<Self> {
40        if data.get(..4)? != ELF_MAGIC || *data.get(6)? != 1 {
41            return None;
42        }
43        let class = *data.get(4)?;
44        let little = match *data.get(5)? {
45            ELFDATA2LSB => true,
46            ELFDATA2MSB => false,
47            _ => return None,
48        };
49        let (header_size, phoff_at, phentsize_at, phnum_at, shoff_at, ehsize_at, shentsize_at, shnum_at, shstrndx_at, expected_phent, expected_shent) = match class {
50            ELFCLASS32 => (52, 28, 42, 44, 32, 40, 46, 48, 50, 32, 40),
51            ELFCLASS64 => (64, 32, 54, 56, 40, 52, 58, 60, 62, 56, 64),
52            _ => return None,
53        };
54        if data.len() < header_size
55            || read(data, 20, 4, little)? != 1
56            || read(data, ehsize_at, 2, little)? as usize != header_size
57        {
58            return None;
59        }
60        let width = if class == ELFCLASS32 { 4 } else { 8 };
61        let phoff = to_usize(read(data, phoff_at, width, little)?)?;
62        let phentsize = read(data, phentsize_at, 2, little)? as usize;
63        let phnum = read(data, phnum_at, 2, little)? as usize;
64        if phnum == 0 {
65            if phoff != 0 {
66                return None;
67            }
68        } else {
69            if phnum == 0xffff || phoff == 0 || phentsize != expected_phent {
70                return None;
71            }
72            checked_range(data, phoff, phentsize.checked_mul(phnum)?)?;
73            for index in 0..phnum {
74                let base = phoff.checked_add(phentsize.checked_mul(index)?)?;
75                let (offset_at, size_at) = if class == ELFCLASS32 { (4, 16) } else { (8, 32) };
76                let kind = read(data, base, 4, little)?;
77                let offset = to_usize(read(data, base.checked_add(offset_at)?, width, little)?)?;
78                let size = to_usize(read(data, base.checked_add(size_at)?, width, little)?)?;
79                if kind != 0 && size != 0 {
80                    checked_range(data, offset, size)?;
81                }
82            }
83        }
84        let header = Header {
85            class,
86            little,
87            shoff: to_usize(read(data, shoff_at, width, little)?)?,
88            shentsize: read(data, shentsize_at, 2, little)? as usize,
89            shnum: read(data, shnum_at, 2, little)? as usize,
90            shstrndx: read(data, shstrndx_at, 2, little)? as usize,
91        };
92        if header.shnum == 0 {
93            return (header.shoff == 0 && header.shstrndx == 0).then_some(Self { data, header });
94        }
95        if header.shstrndx == SHN_XINDEX as usize
96            || header.shoff == 0
97            || header.shentsize != expected_shent
98            || header.shstrndx == 0
99            || header.shstrndx >= header.shnum
100        {
101            return None;
102        }
103        checked_range(data, header.shoff, header.shentsize.checked_mul(header.shnum)?)?;
104        Some(Self { data, header })
105    }
106
107    pub(crate) fn count(&self) -> usize {
108        self.header.shnum
109    }
110
111    pub(crate) fn section(&self, index: usize) -> Option<Section> {
112        if index >= self.header.shnum {
113            return None;
114        }
115        let base = self
116            .header
117            .shoff
118            .checked_add(self.header.shentsize.checked_mul(index)?)?;
119        let (offset_at, size_at, width) = if self.header.class == ELFCLASS32 {
120            (16, 20, 4)
121        } else {
122            (24, 32, 8)
123        };
124        Some(Section {
125            name: to_usize(read(self.data, base, 4, self.header.little)?)?,
126            kind: read(self.data, base.checked_add(4)?, 4, self.header.little)?,
127            offset: to_usize(read(self.data, base.checked_add(offset_at)?, width, self.header.little)?)?,
128            size: to_usize(read(self.data, base.checked_add(size_at)?, width, self.header.little)?)?,
129        })
130    }
131
132    pub(crate) fn names(&self) -> Option<&'a [u8]> {
133        let header = self.section(self.header.shstrndx)?;
134        if header.kind != SHT_STRTAB {
135            return None;
136        }
137        self.bytes(header)
138    }
139
140    pub(crate) fn name<'b>(&self, names: &'b [u8], section: Section) -> Option<&'b [u8]> {
141        let tail = names.get(section.name..)?;
142        let end = tail.iter().position(|byte| *byte == 0)?;
143        Some(&tail[..end])
144    }
145
146    pub(crate) fn bytes(&self, section: Section) -> Option<&'a [u8]> {
147        checked_range(self.data, section.offset, section.size)
148    }
149}
150
151fn checked_range(data: &[u8], offset: usize, size: usize) -> Option<&[u8]> {
152    data.get(offset..offset.checked_add(size)?)
153}
154
155fn read(data: &[u8], offset: usize, width: usize, little: bool) -> Option<u64> {
156    let bytes = data.get(offset..offset.checked_add(width)?)?;
157    let mut value = 0u64;
158    if little {
159        for (index, byte) in bytes.iter().enumerate() {
160            value |= (*byte as u64) << (index * 8);
161        }
162    } else {
163        for byte in bytes {
164            value = (value << 8) | *byte as u64;
165        }
166    }
167    Some(value)
168}
169
170fn to_usize(value: u64) -> Option<usize> {
171    usize::try_from(value).ok()
