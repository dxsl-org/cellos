1//! Boot self-test for the per-path boot-ceiling table.
2//!
3//! Checks the two properties the table exists for: it is **per-path** (no row is
4//! the union of the others, so the ceiling can actually bind), and no row is
5//! *over*-tightened out of a cap its cell legitimately needs.
6//!
7//! Deliberately returns a bool and logs, rather than using `assert!`: it runs on
8//! every boot, so a wrong expectation here must print the offending row, not
9//! panic the kernel before anything has started.
10
11use super::{boot_ceiling, lookup, VFS_REGIONS};
12use crate::loader::launch_profile::{authorize, CallerLaunchState, LaunchRoute};
13use crate::task::cap::CapSet;
14
15/// One (path, cap-accessor) case: the cap that path must still hold.
16type PrivCapCase = (&'static str, fn(&CapSet) -> bool);
17
18const SHELL: CallerLaunchState<'static> = CallerLaunchState {
19    name: "shell",
20    has_spawn: false,
21    has_supervisor: false,
22};
23
24/// Returns `true` when every table property holds; logs each violation.
25pub fn run() -> bool {
26    let mut ok = true;
27
28    // An unknown path must yield no authority at all, and must be reported as
29    // unknown (not as a row that happens to grant nothing).
30    if boot_ceiling("/bin/no-such-cell") != CapSet::EMPTY || lookup("/bin/no-such-cell").is_some() {
31        ok = false;
32        log::error!("[selftest] boot-ceiling: unknown path did not fail closed");
33    }
34
35    // Not a union: for every cap, some row must LACK it. A union-shaped table
36    // fails all of these, which is the whole point of pinning them.
37    let platform_only = CapSet {
38        platform: true,
39        ..CapSet::EMPTY
40    };
41    let per_path = [
42        (
43            "/bin/vfs holds pcie_driver",
44            !boot_ceiling("/bin/vfs").pcie_driver,
45        ),
46        (
47            "/bin/nvme holds block_io",
48            !boot_ceiling("/bin/nvme").block_io,
49        ),
50        ("/bin/net holds spawn", !boot_ceiling("/bin/net").spawn),
51        (
52            "/bin/shell holds network",
53            !boot_ceiling("/bin/shell").network,
54        ),
55        ("/bin/shell holds spawn", !boot_ceiling("/bin/shell").spawn),
56        (
57            "/bin/shell holds block_regions",
58            boot_ceiling("/bin/shell").block_regions == 0,
59        ),
60        (
61            "/bin/supervisor holds mmio",
62            boot_ceiling("/bin/supervisor").mmio_devices == 0,
63        ),
64        (
65            "/bin/platform is wider than platform",
66            boot_ceiling("/bin/platform") == platform_only,
67        ),
68        (
69            "/bin/init holds platform",
70            !boot_ceiling("/bin/init").platform,
71        ),
72    ];
73    for (violation, holds) in per_path {
74        if !holds {
75            ok = false;
76            log::error!("[selftest] boot-ceiling: union collapse — {}", violation);
77        }
78    }
79
80    // The `/bin/vfs` cell-store region bit must survive the ceiling, which runs
81    // BEFORE policy: a `0b111` row silently zeroes it whatever the policy says.
82    if boot_ceiling("/bin/vfs").block_regions != VFS_REGIONS
83        || boot_ceiling("/bin/init").block_regions != VFS_REGIONS
84    {
85        ok = false;
86        log::error!("[selftest] boot-ceiling: /bin/vfs cell-store region would be zeroed");
87    }
88
89    // Positive direction: each privileged boot cell must still receive the cap
90    // its install path requests once the ceiling is intersected.
91    let privileged: [PrivCapCase; 9] = [
92        ("/bin/platform", |c| c.platform),
93        ("/bin/supervisor", |c| c.supervisor),
94        ("/bin/block", |c| c.pcie_driver),
95        ("/bin/nvme", |c| c.pcie_driver),
96        ("/bin/e1000", |c| c.pcie_driver),
97        ("/bin/virtio-net", |c| c.pcie_driver),
98        ("/bin/virtio-gpu", |c| c.pcie_driver),
99        ("/bin/bcm-display", |c| c.pcie_driver),
100        ("/bin/input", |c| c.pcie_driver),
101    ];
102    for (path, held) in privileged {
103        let requested = CapSet::EMPTY.with_path_caps(path);
104        if !held(&requested.intersect(boot_ceiling(path))) {
105            ok = false;
106            log::error!(
107                "[selftest] boot-ceiling: {} would LOSE its privileged cap — row over-tightened",
108                path
109            );
110        }
111    }
112
113    // Negative direction: the USB driver cells hold NO authority until policy v3
114    // adds a signed USB host byte. `with_path_caps` mints nothing for them and
115    // their ceiling rows are EMPTY; if either side starts granting `pcie_driver`
116    // or `DEV_DISPLAY`, this fails the power-on self-test.
117    for path in ["/bin/dwc2-usb", "/bin/lan9514"] {
118        let requested = CapSet::EMPTY.with_path_caps(path);
119        let granted = requested.intersect(boot_ceiling(path));
120        if granted != CapSet::EMPTY {
121            ok = false;
122            log::error!(
123                "[selftest] boot-ceiling: {} granted {:?} without USB policy v3 — must stay EMPTY",
124                path,
125                granted
126            );
127        }
128    }
129
130    if authorize(SHELL, LaunchRoute::Mem, "/mem/demo").is_some() {
131        ok = false;
132        log::error!("[selftest] launch-profile: shell mem launch must fail closed");
133    }
134    if authorize(SHELL, LaunchRoute::Path, "/bin/httpd").is_none() {
135        ok = false;
136        log::error!("[selftest] launch-profile: shell lost exact /bin/httpd launch edge");
137    }
138    if authorize(SHELL, LaunchRoute::Elf, "/bin/httpd").is_none() {
139        ok = false;
140        log::error!("[selftest] launch-profile: shell lost exact /bin/httpd ELF edge");
141    }
142    if authorize(SHELL, LaunchRoute::Elf, "/bin/vfs-test").is_none() {
143        ok = false;
144        log::error!("[selftest] launch-profile: shell lost capability-free ELF edge");
145    }
146    for route in [LaunchRoute::Path, LaunchRoute::Elf] {
147        match authorize(SHELL, route, "/bin/hotswap") {
148            Some(profile) if profile.child_ceiling == CapSet::EMPTY => {}
149            Some(_) => {
150                ok = false;
151                log::error!(
152                    "[selftest] launch-profile: /bin/hotswap gained ambient authority on {:?}",
153                    route
154                );
155            }
156            None => {
157                ok = false;
158                log::error!(
159                    "[selftest] launch-profile: shell lost exact /bin/hotswap edge on {:?}",
160                    route
161                );
162            }
163        }
164    }
165    if authorize(SHELL, LaunchRoute::Elf, "/bin/vfs").is_some() {
166        ok = false;
167        log::error!("[selftest] launch-profile: shell gained privileged /bin/vfs launch edge");
168    }
169
170    if ok {
171        log::info!("[selftest] boot-ceiling: PASS (per-path table, no union collapse)");
172    } else {
173        log::error!("[selftest] boot-ceiling: FAIL");
174    }
175    ok
176}
