7pub(super) fn reviewed_user_target_ceiling(target: &str) -> Option<CapSet> {
8    let caps = match target {
9        "/bin/adc-demo"
10        | "/bin/ai-test"
11        | "/bin/audio-demo"
12        | "/bin/bench-probe"
13        | "/bin/can-demo"
14        | "/bin/cat"
15        | "/bin/cfi-test"
16        | "/bin/curl"
17        | "/bin/c-pthread"
18        | "/bin/cpp-smoke"
19        | "/bin/tls-test"
20        | "/bin/futex-test"
21        | "/bin/pipe-test"
22        | "/bin/pipe-peer"
23        | "/bin/c-spawn"
24        | "/bin/backend-worker"
25        | "/bin/doom"
26        | "/bin/echo"
27        | "/bin/free"
28        | "/bin/gpio-test-rv"
29        | "/bin/http-smoke"
30        | "/bin/input-test"
31        | "/bin/window-policy-probe"
32        | "/bin/ls"
33        | "/bin/posix-shim-test"
34        | "/bin/ps"
35        | "/bin/robot-dashboard"
36        | "/bin/viui-demo"
37        | "/bin/tetris"
38        | "/bin/tetris-c"
39        | "/bin/tetris-lua"
40        | "/bin/vfs-test"
41        | "/bin/wx-test"
42        | "/bin/tier2-exploit"
43        | "/bin/tier2-smoke"
44        | "/bin/std-smoke"
45        | "/bin/desktop"

47        | "/bin/ocel-js" => CapSet::EMPTY,
48        // These clients and servers use typed IPC to the net service; they do
49        // not hold NetworkCap themselves. Keeping their launch ceiling empty

51        "/bin/httpd" | "/bin/https-demo" | "/bin/llm-gateway" | "/bin/mqtt" | "/bin/nc"

53        "/bin/net-broker" => CapSet {
54            network: true,

57        "/bin/dwc2-usb" => CapSet {
58            usb_driver: true,
59            ..CapSet::EMPTY
60        },
61        "/bin/periph-demo" | "/bin/periph-test" => console_mmio_capset(),
62        "/bin/pwm-demo" => gpio_mmio_capset(),
63        "/bin/sensor-demo" => sensor_mmio_capset(),
64        "/bin/spi-demo" => spi_demo_mmio_capset(),
65        "/bin/bench"
66        | "/bin/capacity-probe"
67        | "/bin/hypha"
68        | "/bin/tool-spawn"
69        // The B0 supervisor witnesses the actor/supervisor library: it must reach
70        // the shell edge with its SpawnCap intact, or it cannot spawn or watch
71        // children at all (ADR-0021 §2.4).
72        | "/bin/backend-supervisor"
73        | "/bin/hotswap-demo-v1"
74        | "/bin/hotswap-demo-v2" => CapSet {
75            spawn: true,
76            ..CapSet::EMPTY
77        },
78        "/bin/python" | "/bin/lua" | "/bin/tool-fs" | "/bin/tool-sys" => CapSet::EMPTY,
79        "/bin/robot-demo" => CapSet {
80            network: true,
81            mmio_devices: gpio_mmio_capset().mmio_devices,
82            ..CapSet::EMPTY
83        },
84        _ => return None,
85    };
86    Some(caps)
87}
