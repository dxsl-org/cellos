16pub(super) fn init_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
17    #[cfg(feature = "phase04-driver-profile")]
18    if target == "/bin/phase04-terminal-driver" {
19        return phase04_driver_init_profile(route, target);
20    }
21    if matches!(route, LaunchRoute::Mem | LaunchRoute::Pinned) {
22        return None;
23    }
24    match target {
25        "/bin/ai" | "/bin/ahci" | "/bin/xhci" | "/bin/bcm-display" | "/bin/block" | "/bin/compositor" | "/bin/config"
26        | "/bin/dwc2-usb" | "/bin/e1000" | "/bin/igb" | "/bin/fb-console" | "/bin/hypervisor" | "/bin/input"
27        | "/bin/kms" | "/bin/net" | "/bin/net-broker" | "/bin/nvme" | "/bin/serial" | "/bin/cellbox"
28        | "/bin/silo" | "/bin/ai-test" | "/bin/silo-test" | "/bin/srv-test" | "/bin/supervisor"
29        | "/bin/vfs" | "/bin/vfs-test" | "/bin/virtio-gpu" | "/bin/virtio-net"
30        | "/bin/std-smoke" | "/bin/desktop" | "/bin/sshd" | "/bin/chrome" | "/bin/vm-runner" | "/bin/files" | "/bin/terminal"
31        // Ocel's demand-only Tier 2 JavaScript engines. Init activates the exact
32        // fixed target on a document lease, not through the boot service plan.
33        // The empty boot ceiling grants no engine authority; domain admission
34        // independently decides whether the artifact may run in a private root.
35        | "/bin/ocel-js"
36        | "/bin/ocel-quickjs"
37        // Isolated native PDF rasterizer; no additional capabilities.
38        | "/bin/ocel-pdf"
39        // Phase-02 Tier-2 entry fixtures. Both carry `PROTECTION_CLASS_UNTRUSTED`
40        // manifests and `boot_ceiling` already gives them `CapSet::EMPTY`, so this
41        // row grants init nothing beyond the ability to launch them: the
42        // admission policy is still the only thing that decides whether they may
43        // run as domains, and a production image (which has no such cells and no
44        // qualified switch ordering) refuses them there.
45        | "/bin/tier2-smoke" | "/bin/tier2-exploit" | "/bin/cpp-smoke"
46        // Phase-02 slice B cross-tier exchange: the Tier-2 provider and the Tier-1
47        // driver. Both ceilings are `CapSet::EMPTY` (neither spawns anything), so
48        // this edge grants init nothing but the ability to launch them, and the
49        // admission policy still decides whether the provider may run as a domain.
50        | "/bin/tier2-rpc-provider" | "/bin/tier2-rpc-driver" => Some(LaunchProfile::new(
51            boot_ceiling::boot_ceiling(target),
52            "init-launch-edge",
53            true,
54        )),
55        // Autoload may select applications already reviewed for interactive
56        // launch. Reuse their exact ceiling, never init's union of authority;
57        // unknown paths still fail closed and lifecycle authority is required.
58        _ => reviewed_user_target_ceiling(target)
59            .map(|ceiling| LaunchProfile::new(ceiling, "init-autoload-edge", true)),
60    }
61}

63pub(super) fn cellbox_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
64    let ceiling = match route {
65        LaunchRoute::Path | LaunchRoute::Elf if target == "/bin/hotswap" => LEGACY,
66        // Retained power-off worker: the only caller of raw opcode 502 (see
67        // `RAW_SHUTDOWN_CALLERS` in the syscall layer). Path route only; the
68        // child holds no capability, the syscall gate is name-keyed.
69        LaunchRoute::Path if target == "/bin/shutdown" => LEGACY,
70        // Phase 09 D79-9: the cellbox ELF installed a second time for `cellbox ssh`.
71        // Only the shell may start it, only by the kernel-resolved path, and only it
72        // carries the outbound client rights plus, since KSSH K3, the KMS SSH sign
73        // right ([`SSH_CLIENT_EDGE`]); `/bin/cellbox` itself holds none. It is not
74        // in the generic reviewed table, so init autoload and the desktop cannot
75        // launch it, and its caller-owned ELF route stays denied below.
76        LaunchRoute::Path if target == "/bin/cellbox-ssh" => SSH_CLIENT_EDGE,
77        // KSSH K7: the admin cell that imports and deletes the single KMS SSH identity.
78        // Same shape as the ssh edge: shell only, kernel-resolved path route only, not
79        // in the generic reviewed table (so init autoload, the desktop, the terminal and
80        // the caller-owned ELF route cannot start it). Its ceiling carries the admin
81        // right and never the sign right ([`KMS_SSH_ADMIN_EDGE`]).
82        LaunchRoute::Path if target == "/bin/kms-ssh-admin" => KMS_SSH_ADMIN_EDGE,
83        LaunchRoute::Path | LaunchRoute::Elf => reviewed_user_target_ceiling(target)?,
84        LaunchRoute::Mem | LaunchRoute::Pinned => return None,
85    };
86    Some(LaunchProfile::new(ceiling, "cellbox-launch-edge", false))
87}

96pub(super) fn hypha_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
97    if !matches!(route, LaunchRoute::Path | LaunchRoute::Elf) {
98        return None;
99    }
100    let ceiling = match target {
101        // The gateway's reviewed ceiling carries its network-client rights; the
102        // edge reuses that row rather than restating it.
103        "/bin/llm-gateway" => reviewed_user_target_ceiling(target)?,
104        "/bin/tool-fs" | "/bin/tool-sys" => LEGACY,
105        "/bin/tool-spawn" => CapSet {
106            spawn: true,
107            ..LEGACY
108        },
109        _ => return None,
110    };
111    Some(LaunchProfile::new(ceiling, "hypha-launch-edge", false))
112}

155/// GUI Terminal (`terminal`) may run exactly one program: the multicall shell

167/// The child ceiling of every reviewed edge that hosts the interactive shell
168/// `/bin/cellbox`: `LEGACY` plus `config_write`, which service-config bounds to
169/// the shell key namespace (Phase 01 CP1, D-Q5). Must equal the init boot row
170/// `boot_ceiling("/bin/cellbox")`.
171pub(super) const SHELL_EDGE: CapSet = CapSet {
172    config_write: true,
173    ..LEGACY
174};

176/// The child ceiling of the one reviewed edge that starts `/bin/cellbox-ssh`
177/// (shell, kernel-resolved path route only): the outbound client rights of
178/// [`NET_CLIENT`] plus `kms_ssh_sign` (KSSH K3). A separate constant on purpose:
179/// `NET_CLIENT` is the ceiling of every other network client (curl, wget, mqtt,
180/// ocel, llm-gateway, https-demo, robot-demo) and must not carry the right. Only
181/// this edge carries it, so `request(/bin/cellbox-ssh) ∩ ceiling` is the sole
182/// way a Cell obtains `CALLER_FLAG_KMS_SSH_SIGN`.
183pub(super) const SSH_CLIENT_EDGE: CapSet = CapSet {
184    kms_ssh_sign: true,
185    ..NET_CLIENT
186};
187
188/// The child ceiling of the one reviewed edge that starts `/bin/kms-ssh-admin` (shell,
189/// kernel-resolved path route only; KSSH K7): the pre-annex X/M/W rights plus
190/// `kms_ssh_admin`. No network, no sign right: the admin and sign rights are never
191/// held by one Cell, so `request(/bin/kms-ssh-admin) ∩ ceiling` is the sole way a Cell
192/// obtains `CALLER_FLAG_KMS_SSH_ADMIN`. The constant lives here, not in `targets.rs`,
193/// for the same reason as [`SSH_CLIENT_EDGE`].
194pub(super) const KMS_SSH_ADMIN_EDGE: CapSet = CapSet {
195    kms_ssh_admin: true,
196    ..LEGACY
197};

199/// Private Phase04 fixture driver: shell hosting (the same ceiling as the GUI
200/// Terminal's shell edge, so the lane proves the production Q grant) plus the
201/// stock echo negative-output-grant probe on a capability-free edge. Neither
202/// edge exists in ordinary images.
203#[cfg(feature = "phase04-driver-profile")]
204pub(super) fn phase04_driver_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
205    if route != LaunchRoute::Path {
206        return None;
207    }
208    let (ceiling, label) = match target {
209        "/bin/cellbox" => (SHELL_EDGE, "phase04-driver-shell-edge"),
210        "/bin/echo" => (LEGACY, "phase04-driver-output-probe-edge"),
211        _ => return None,
212    };
213    Some(LaunchProfile::new(ceiling, label, false))
214}

333/// Compatibility edge: `periph-demo` uses `SpawnPinned("/bin/periph-demo", ..)`
334/// as a self-relaunch path despite manifest `spawn = false`. Preserve only that
335/// exact pinned edge, bounded to the reviewed console MMIO ceiling, so the demo
336/// keeps working without reviving ambient lifecycle authority.
337pub(super) fn pinned_profile(
338    caller_name: &str,
339    route: LaunchRoute,
340    target: &str,
341) -> Option<LaunchProfile> {
342    if !matches!(route, LaunchRoute::Pinned) {
343        return None;
344    }
345    let ceiling = match (caller_name, target) {
346        ("bench", "/bin/bench-probe") | ("capacity-probe", "/bin/bench-probe") => LEGACY,
347        ("capacity-probe", "/bin/heavy-probe") => LEGACY,
348        ("periph-demo", "/bin/periph-demo") => CapSet {
349            mmio_devices: CONSOLE_MMIO,
350            ..LEGACY
351        },
352        _ => return None,
353    };
354    Some(LaunchProfile::new(ceiling, "pinned-launch-edge", false))
355}
