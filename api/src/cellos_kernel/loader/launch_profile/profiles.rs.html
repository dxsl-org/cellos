13pub(super) fn init_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
14    #[cfg(feature = "phase04-driver-profile")]
15    if target == "/bin/phase04-terminal-driver" {
16        return phase04_driver_init_profile(route, target);
17    }
18    if matches!(route, LaunchRoute::Mem | LaunchRoute::Pinned) {
19        return None;
20    }
21    match target {
22        "/bin/ai" | "/bin/ahci" | "/bin/xhci" | "/bin/bcm-display" | "/bin/block" | "/bin/compositor" | "/bin/config"
23        | "/bin/dwc2-usb" | "/bin/e1000" | "/bin/e1000e" | "/bin/igb" | "/bin/fb-console" | "/bin/hypervisor" | "/bin/input"
24        | "/bin/kms" | "/bin/net" | "/bin/net-broker" | "/bin/nvme" | "/bin/serial" | "/bin/cellbox"
25        | "/bin/silo" | "/bin/supervisor"
26        | "/bin/vfs" | "/bin/virtio-gpu" | "/bin/virtio-net" | "/bin/virtio-scsi"
27        | "/bin/vmxnet3" | "/bin/pvscsi"
28        | "/bin/desktop" | "/bin/sshd" | "/bin/chrome" | "/bin/vm-runner" | "/bin/files" | "/bin/terminal"
29        // Phase 13: init's fixed bootstrap rows after the VFS. The machine-pin
30        // cell (x86) and the Ocel activator each keep their own boot ceiling;
31        // the activator alone reaches the demand engines (`activator_profile`).
32        | "/bin/machine-pin" | "/bin/activator"
33        // Test images only: the plan names the fixture launcher when the image
34        // packages it; its own reviewed edges are `fixture_launcher_profile`.
35        | "/bin/fixture-launcher" => Some(LaunchProfile::new(
36            boot_ceiling::boot_ceiling(target),
37            "init-launch-edge",
38            true,
39        )),
40        // The demand engines are the activator's alone (`activator_profile`);
41        // init reaches them neither on a fixed edge nor through autoload.
42        "/bin/ocel-js" | "/bin/ocel-quickjs" | "/bin/ocel-pdf" => None,
43        // Autoload may select applications already reviewed for interactive
44        // launch. Reuse their exact ceiling, never init's union of authority;
45        // unknown paths still fail closed and lifecycle authority is required.
46        _ => reviewed_user_target_ceiling(target)
47            .map(|ceiling| LaunchProfile::new(ceiling, "init-autoload-edge", true)),
48    }
49}

67        true,
68    ))

71/// The test fixture launcher (Phase 13): the fixture edges init held until the
72/// fixtures left it, each with the same child ceiling (`boot_ceiling`) init's
73/// edge granted. The five boot tests are present in ordinary test disks; the
74/// Tier-2, cross-tier, DMA-export and grant-pair fixtures run only on
75/// `test-hooks` kernels, so their edges exist only there.
76pub(super) fn fixture_launcher_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
77    if !matches!(route, LaunchRoute::Path | LaunchRoute::Elf) {
78        return None;
79    }
80    let reviewed = match target {
81        "/bin/silo-test" | "/bin/ai-test" | "/bin/vfs-test" | "/bin/srv-test"
82        | "/bin/std-smoke" => true,
83        // Phase-02 Tier-2 entry and cross-tier fixtures, the ADR-0033 CP1 §7
84        // `edu` export lane, and the phase-03 grant pair (owner on `/bin/silo`,
85        // receiver on `/bin/net-broker`, whose real services those images omit).
86        #[cfg(feature = "test-hooks")]
87        "/bin/tier2-smoke"
88        | "/bin/tier2-exploit"
89        | "/bin/cpp-smoke"
90        | "/bin/tier2-rpc-provider"
91        | "/bin/tier2-rpc-driver"
92        | "/bin/edu-holder"
93        | "/bin/export-client"
94        | "/bin/silo"
95        | "/bin/net-broker" => true,
96        _ => false,
97    };
98    reviewed.then(|| {
99        LaunchProfile::new(
100            boot_ceiling::boot_ceiling(target),
101            "fixture-launcher-edge",
102            true,
103        )
104    })
105}

107pub(super) fn cellbox_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
108    let ceiling = match route {
109        LaunchRoute::Path | LaunchRoute::Elf if target == "/bin/hotswap" => LEGACY,
110        // Retained power-off worker: the only caller of raw opcode 502 (see
111        // `RAW_SHUTDOWN_CALLERS` in the syscall layer). Path route only; the
112        // child holds no capability, the syscall gate is name-keyed.
113        LaunchRoute::Path if target == "/bin/shutdown" => LEGACY,
114        // Phase 09 D79-9: the cellbox ELF installed a second time for `cellbox ssh`.
115        // Only the shell may start it, only by the kernel-resolved path, and only it
116        // carries the outbound client rights plus, since KSSH K3, the KMS SSH sign
117        // right ([`SSH_CLIENT_EDGE`]); `/bin/cellbox` itself holds none. It is not
118        // in the generic reviewed table, so init autoload and the desktop cannot
119        // launch it, and its caller-owned ELF route stays denied below.
120        LaunchRoute::Path if target == "/bin/cellbox-ssh" => SSH_CLIENT_EDGE,
121        // KSSH K7: the admin cell that imports and deletes the single KMS SSH identity.
122        // Same shape as the ssh edge: shell only, kernel-resolved path route only, not
123        // in the generic reviewed table (so init autoload, the desktop, the terminal and
124        // the caller-owned ELF route cannot start it). Its ceiling carries the admin
125        // right and never the sign right ([`KMS_SSH_ADMIN_EDGE`]).
126        LaunchRoute::Path if target == "/bin/kms-ssh-admin" => KMS_SSH_ADMIN_EDGE,
127        // Tier 3 on demand (`hv`): the kernel-resolved path route only, exactly the
128        // hypervisor's own ceiling. Not in the generic reviewed table.
129        LaunchRoute::Path if target == HYPERVISOR => hypervisor_edge()?,
130        LaunchRoute::Path | LaunchRoute::Elf => reviewed_user_target_ceiling(target)?,
131        LaunchRoute::Mem | LaunchRoute::Pinned => return None,
132    };
133    Some(LaunchProfile::new(ceiling, "cellbox-launch-edge", false))
134}

235/// The child ceiling of every reviewed edge that hosts the interactive shell
236/// `/bin/cellbox`: `LEGACY` plus `config_write`, which service-config bounds to
237/// the shell key namespace (Phase 01 CP1, D-Q5). Must equal the init boot row
238/// `boot_ceiling("/bin/cellbox")`.
239pub(super) const SHELL_EDGE: CapSet = CapSet {
240    config_write: true,
241    ..LEGACY
242};
243
244/// The child ceiling of the one reviewed edge that starts `/bin/cellbox-ssh`
245/// (shell, kernel-resolved path route only): the outbound client rights of
246/// [`NET_CLIENT`] plus `kms_ssh_sign` (KSSH K3). A separate constant on purpose:
247/// `NET_CLIENT` is the ceiling of every other network client (curl, wget, mqtt,
248/// ocel, llm-gateway, https-demo, robot-demo) and must not carry the right. Only
249/// this edge carries it, so `request(/bin/cellbox-ssh) ∩ ceiling` is the sole
250/// way a Cell obtains `CALLER_FLAG_KMS_SSH_SIGN`.
251pub(super) const SSH_CLIENT_EDGE: CapSet = CapSet {
252    kms_ssh_sign: true,
253    ..NET_CLIENT
254};
255
256/// The child ceiling of the one reviewed edge that starts `/bin/kms-ssh-admin` (shell,
257/// kernel-resolved path route only; KSSH K7): the pre-annex X/M/W rights plus
258/// `kms_ssh_admin`. No network, no sign right: the admin and sign rights are never
259/// held by one Cell, so `request(/bin/kms-ssh-admin) ∩ ceiling` is the sole way a Cell
260/// obtains `CALLER_FLAG_KMS_SSH_ADMIN`. The constant lives here, not in `targets.rs`,
261/// for the same reason as [`SSH_CLIENT_EDGE`].
262pub(super) const KMS_SSH_ADMIN_EDGE: CapSet = CapSet {
263    kms_ssh_admin: true,
264    ..LEGACY
265};

267/// Private Phase04 fixture driver: shell hosting (the same ceiling as the GUI
268/// Terminal's shell edge, so the lane proves the production Q grant) plus the
269/// stock echo negative-output-grant probe on a capability-free edge. Neither
270/// edge exists in ordinary images.
271#[cfg(feature = "phase04-driver-profile")]
272pub(super) fn phase04_driver_profile(route: LaunchRoute, target: &str) -> Option<LaunchProfile> {
273    if route != LaunchRoute::Path {
274        return None;
275    }
276    let (ceiling, label) = match target {
277        "/bin/cellbox" => (SHELL_EDGE, "phase04-driver-shell-edge"),
278        "/bin/echo" => (LEGACY, "phase04-driver-output-probe-edge"),
279        _ => return None,
280    };
281    Some(LaunchProfile::new(ceiling, label, false))
282}

416/// Compatibility edge: `periph-demo` uses `SpawnPinned("/bin/periph-demo", ..)`
417/// as a self-relaunch path despite manifest `spawn = false`. Preserve only that
418/// exact pinned edge, bounded to the reviewed console MMIO ceiling, so the demo
419/// keeps working without reviving ambient lifecycle authority.
420pub(super) fn pinned_profile(
421    caller_name: &str,
422    route: LaunchRoute,
423    target: &str,
424) -> Option<LaunchProfile> {
425    if !matches!(route, LaunchRoute::Pinned) {
426        return None;
427    }
428    let ceiling = match (caller_name, target) {
429        ("bench", "/bin/bench-probe") | ("capacity-probe", "/bin/bench-probe") => LEGACY,
430        // Tier-2 echo peer for `ipc_send_recv_domain`; domain admission accepts
431        // LEGACY (observe/mutate are not domain-unenforceable).
432        ("bench", "/bin/bench-peer-t2") => LEGACY,
433        ("capacity-probe", "/bin/heavy-probe") => LEGACY,
434        ("periph-demo", "/bin/periph-demo") => CapSet {
435            mmio_devices: CONSOLE_MMIO,
436            ..LEGACY
437        },
438        _ => return None,
439    };
440    Some(LaunchProfile::new(ceiling, "pinned-launch-edge", false))
