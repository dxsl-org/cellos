1//! Per-path capability ceiling for the cells named in the boot manifest.
2//!
3//! This is the ceiling applied to a **kernel/boot-initiated** spawn
4//! (`Spawner::Root`) and the source of the root authority's own capability
5//! write. It replaces a single "full authority" constant.
6//!
7//! **Invariant — the table is per-path, never a union.** A ceiling only ever
8//! constrains when it is *smaller* than the request; the union of every row is
9//! by definition a superset of every row, so a union-shaped ceiling would admit
10//! everything while looking like a restriction. Each row therefore states the
11//! authority that ONE path is expected to hold, and nothing else. `self_test`
12//! pins that property so a later edit cannot quietly collapse the table into a
13//! union.
14//!
15//! A path with no row is **unknown**, and an unknown path gets `CapSet::EMPTY`
16//! (fail-closed). `lookup` returns `None` for it so the diagnostics can say
17//! "missing entry" rather than "entry grants nothing" — the two look identical
18//! in the resulting CapSet and need very different fixes.
19//!
20//! Rows are derived from what each cell's `__ViCell_manifest` requests plus the
21//! path-triggered caps `CapSet::with_path_caps` layers on. A row is therefore
22//! the *expected* authority of that binary, not a guess.
23
24use crate::resource_registry::{DEV_DISPLAY, DEV_GPIO, DEV_I2C, DEV_SPI, DEV_UART};
25use crate::task::cap::CapSet;

27mod selftest;

29/// Boot self-test of the table's load-bearing properties — that it is per-path
30/// rather than a union, and that no row is over-tightened. See `selftest`.
31pub use selftest::run as self_test;

33/// Block regions VFS must be able to hold after request ∩ ceiling ∩ policy:
34/// P1 | P4 | SRV | cell-store.
35///
36/// `/bin/vfs` now requests bit 3 in `CapSet::with_path_caps`, the boot ceiling
37/// must preserve it, and the signed `/POLICY.BIN` row must also preserve it.
38/// A `0b111` ceiling here would still zero the cell-store bit before policy
39/// runs, so this row remains `0b1111`.
40const VFS_REGIONS: u8 = 0b1111;
41
42/// The two console peripherals that path-triggered demo rows may request.
43const CONSOLE_MMIO: u8 = DEV_GPIO | DEV_UART;
44const ROOT_MMIO: u8 = CONSOLE_MMIO | DEV_I2C | DEV_SPI | DEV_DISPLAY;
45/// The expected ceiling for `path`, or `None` when `path` is not a boot cell.
46///
47/// Every row carries the reason that path holds those caps. Prefer a row that is
48/// marginally too generous over a missing one: one cap too many is a follow-up,
49/// a cell that cannot start is a dead system.
50pub fn lookup(path: &str) -> Option<CapSet> {
51    let caps = match path {
52        // Root authority. Not spawned from a path at all — the kernel writes
53        // these caps directly — but it is listed here so there is exactly one
54        // table describing boot authority. It is the widest row because a
55        // spawner's caps bound its children, and this cell brings up every
56        // service below. `platform` is deliberately absent: the Platform Cell is
57        // kernel-spawned, and `CapSet::apply_to` never writes `platform_cap`
58        // (the singleton latch owns it), so the root authority never actually
59        // held it and cannot delegate it.
60        "/bin/init" => CapSet {
61            block_io: true,             // delegated to /bin/vfs
62            network: true,              // delegated to /bin/net, /bin/net-broker
63            spawn: true,                // delegated to /bin/shell, /bin/supervisor
64            hypervisor: true,           // delegated to /bin/silo, /bin/hypervisor
65            mmio_devices: ROOT_MMIO,    // delegated through reviewed launch profiles
66            block_regions: VFS_REGIONS, // delegated to /bin/vfs
67            pcie_driver: true,          // delegated to the block/NIC/GPU/input drivers
68            platform: false,
69            supervisor: true, // delegated to /bin/supervisor
70        },
71        // PCIe ECAM enumeration + BAR registration. Kernel-spawned before init.
72        // Its manifest declares nothing; `with_path_caps` is the request signal,
73        // and the owned reservation still enforces one holder ever.
74        "/bin/platform" => CapSet {
75            platform: true,
76            ..CapSet::EMPTY
77        },
78        // Filesystem service: raw block syscalls plus its partitions.
79        "/bin/vfs" => CapSet {
80            block_io: true,
81            block_regions: VFS_REGIONS,
82            ..CapSet::EMPTY
83        },
84        // Network stack and the cluster broker; both declare `network` only.
85        "/bin/net" | "/bin/net-broker" => CapSet {
86            network: true,
87            ..CapSet::EMPTY
88        },
89        // Interactive shell: no ambient lifecycle authority. Exact launch edges
90        // are reviewed in `loader::launch_profile`; the shell itself holds none.
91        "/bin/shell" => CapSet::EMPTY,
92        // Hotswap orchestration: re-spawns cells (spawn) and freezes/resumes
93        // them (supervisor, from `with_path_caps`).
94        "/bin/supervisor" => CapSet {
95            spawn: true,
96            supervisor: true,
97            ..CapSet::EMPTY
98        },
99        // Both declare `hypervisor`. `CapSet::from_manifest` additionally gates
100        // that bit on the CPU actually reporting H-ext/EL2/VMX, so this row
101        // cannot conjure it on hardware that lacks virtualisation.
102        "/bin/silo" | "/bin/hypervisor" => CapSet {
103            hypervisor: true,
104            ..CapSet::EMPTY
105        },
106        // PCIe driver cells: each claims a BAR/MMIO range and authorises DMA.
107        // `pcie_driver` + `DEV_DISPLAY` are set here; `with_path_caps` adds them
108        // to the request before the ceiling intersection.
109        "/bin/block" | "/bin/nvme" | "/bin/e1000" | "/bin/virtio-net" | "/bin/virtio-gpu"
110        | "/bin/input" => CapSet {
111            pcie_driver: true,
112            mmio_devices: DEV_DISPLAY,
113            ..CapSet::EMPTY
114        },
115        // BCM uses the statically allowlisted VideoCore mailbox, never PCIe.
116        "/bin/bcm-display" => CapSet {
117            mmio_devices: DEV_DISPLAY,
118            ..CapSet::EMPTY
119        },
120        // USB driver cells: intentionally EMPTY until policy v3 introduces a
121        // signed USB host authority byte. `with_path_caps` grants them nothing;
122        // the ceiling here is the cross-check. Listed explicitly (not `_ => None`)
123        // so a boot-log denial can report "row intentionally empty" vs "row missing".
124        "/bin/dwc2-usb" | "/bin/lan9514" => CapSet::EMPTY,
125        // Known boot cells that need no authority at all: pure IPC clients. They
126        // are listed rather than left to the unknown-path fallback so a denial
127        // report can distinguish "needs nothing" from "row missing".
128        "/bin/config" | "/bin/compositor" | "/bin/fb-console" | "/bin/kms" | "/bin/silo-test"
129        | "/bin/vfs-test" | "/bin/srv-test" => CapSet::EMPTY,
130        _ => return None,
131    };
132    Some(caps)
133}
134
135/// The ceiling for `path`; an unknown path yields `CapSet::EMPTY` (fail-closed).
136pub fn boot_ceiling(path: &str) -> CapSet {
137    lookup(path).unwrap_or(CapSet::EMPTY)
138}

140/// Report, at `error` level, every cap the boot ceiling refused for `path`.
141///
142/// Written for a reader who has only a boot log: it prints whether the path is
143/// missing from the table at all, the three sets involved, and one line per
144/// refused cap named exactly as the table field to add. Nothing here needs to be
145/// re-derived from the source to act on it.
146pub fn log_refusal(path: &str, requested: CapSet, ceiling: CapSet, granted: CapSet) {
147    if lookup(path).is_none() {
148        log::error!(
149            "[loader] boot-ceiling: {:?} has NO ROW in boot_ceiling::lookup — ceiling is EMPTY. Add a row.",
150            path
151        );
152    } else {
153        log::error!(
154            "[loader] boot-ceiling: {:?} narrowed by its row — widen the row if the cap is legitimate.",
155            path
156        );
157    }
158    log::error!("[loader]   requested = {:?}", requested);
159    log::error!("[loader]   ceiling   = {:?}", ceiling);
160    log::error!("[loader]   granted   = {:?}", granted);
161    if requested.block_io && !granted.block_io {
162        log::error!("[loader]   refused: block_io");
163    }
164    if requested.network && !granted.network {
165        log::error!("[loader]   refused: network");
166    }
167    if requested.spawn && !granted.spawn {
168        log::error!("[loader]   refused: spawn");
169    }
170    if requested.hypervisor && !granted.hypervisor {
171        log::error!("[loader]   refused: hypervisor");
172    }
173    if requested.pcie_driver && !granted.pcie_driver {
174        log::error!("[loader]   refused: pcie_driver");
175    }
176    if requested.platform && !granted.platform {
177        log::error!("[loader]   refused: platform");
178    }
179    if requested.supervisor && !granted.supervisor {
180        log::error!("[loader]   refused: supervisor");
181    }
182    if requested.mmio_devices != granted.mmio_devices {
183        log::error!(
184            "[loader]   refused: mmio_devices {:#07b}",
185            requested.mmio_devices & !granted.mmio_devices
186        );
187    }
188    if requested.block_regions != granted.block_regions {
189        log::error!(
190            "[loader]   refused: block_regions {:#06b}",
191            requested.block_regions & !granted.block_regions
192        );
193    }
194}
