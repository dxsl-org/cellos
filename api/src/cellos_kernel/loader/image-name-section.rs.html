1//! Bounded read of the `__ViCell_image` program-name section, and the
2//! measured identity an admitted image is published with.
3//!
4//! The signing tool emits the section into the bytes it signs. The name feeds
5//! `ImageIdentity`, which VFS matches path rules against, so it is read with
6//! the duplicate-refusing `ElfSections` parser rather than
7//! `ElfLoader::get_section` (first match wins, no duplicate check).
8use super::elf_section::{ElfSections, SHT_NOBITS};
9use api::cell_image::{is_valid_image_name, ImageIdentity, IMAGE_NAME_SECTION};
10
11/// The identity recorded for an admitted image. `verified` is true only when
12/// its signature verified under this kernel's anchor: the name is believed
13/// only then, because only then are the section bytes covered by a signature
14/// the kernel checked. An unsigned image states its digest and nothing else,
15/// whatever name its section claims.
16pub(super) fn measured_identity(data: &[u8], digest: [u8; 32], verified: bool) -> ImageIdentity {
17    verified
18        .then(|| {
19            let name = declared_name(data).unwrap_or(&[]);
20            ImageIdentity::signed(digest, crate::signing::anchor_class(), name)
21        })
22        .flatten()
23        .unwrap_or(ImageIdentity::unsigned(digest))
24}
25
26/// The declared program name, or `None` when the image declares none or the
27/// declaration is unusable (duplicated, `NOBITS`, out of bounds, or not a valid
28/// name). Unusable is treated as absent: the image still spawns, it just
29/// carries no name, so no name-bound rule can ever match it.
30fn declared_name(data: &[u8]) -> Option<&[u8]> {
31    let elf = ElfSections::parse(data)?;
32    let names = elf.names()?;
33    let mut found = None;
34    for index in 0..elf.count() {
35        let section = elf.section(index)?;
36        if elf.name(names, section)? != IMAGE_NAME_SECTION.as_bytes() {
37            continue;
38        }
39        if section.kind == SHT_NOBITS || found.is_some() {
40            return None;
41        }
42        found = Some(elf.bytes(section)?);
43    }
44    found.filter(|name| is_valid_image_name(name))
45}

47#[cfg(test)]
48mod tests {
49    use super::{declared_name, measured_identity};
50
51    const NAMES: &[u8] = b"\0.shstrtab\0__ViCell_image\0";
52    const DATA: usize = 96;
53    const SHOFF: usize = 128;
54
55    fn put(elf: &mut [u8], at: usize, value: u64, width: usize) {
56        elf[at..at + width].copy_from_slice(&value.to_le_bytes()[..width]);
57    }
58
59    fn section(elf: &mut [u8], index: usize, name: u32, kind: u32, offset: usize, size: usize) {
60        let base = SHOFF + index * 64;
61        put(elf, base, u64::from(name), 4);
62        put(elf, base + 4, u64::from(kind), 4);
63        put(elf, base + 24, offset as u64, 8);
64        put(elf, base + 32, size as u64, 8);
65    }
66
67    /// Minimal ELF64: null section, `.shstrtab`, then `copies` image sections.
68    fn elf64(name: &[u8], copies: usize, kind: u32) -> alloc::vec::Vec<u8> {
69        let count = 2 + copies;
70        let mut elf = alloc::vec![0u8; SHOFF + count * 64];
71        elf[..7].copy_from_slice(b"\x7fELF\x02\x01\x01");
72        put(&mut elf, 16, 2, 2);
73        put(&mut elf, 18, 243, 2);
74        put(&mut elf, 20, 1, 4);
75        put(&mut elf, 40, SHOFF as u64, 8);
76        put(&mut elf, 52, 64, 2);
77        put(&mut elf, 58, 64, 2);
78        put(&mut elf, 60, count as u64, 2);
79        put(&mut elf, 62, 1, 2);
80        elf[64..64 + NAMES.len()].copy_from_slice(NAMES);
81        section(&mut elf, 1, 1, 3, 64, NAMES.len());
82        elf[DATA..DATA + name.len()].copy_from_slice(name);
83        for copy in 0..copies {
84            section(&mut elf, 2 + copy, 11, kind, DATA, name.len());
85        }
86        elf
87    }

89    #[test]
90    fn single_valid_section_is_the_name() {
91        assert_eq!(declared_name(&elf64(b"sshd", 1, 1)), Some(&b"sshd"[..]));
92    }
93
94    #[test]
95    fn absent_duplicate_nobits_or_invalid_declares_nothing() {
96        assert_eq!(declared_name(&elf64(b"sshd", 0, 1)), None);
97        assert_eq!(declared_name(&elf64(b"sshd", 2, 1)), None);
98        assert_eq!(declared_name(&elf64(b"sshd", 1, 8)), None);
99        assert_eq!(declared_name(&elf64(b"SSHD", 1, 1)), None);
100        assert_eq!(declared_name(&elf64(b"ssh\0", 1, 1)), None);
101        assert_eq!(declared_name(b"not an elf"), None);
102    }
103
104    #[test]
105    fn unsigned_image_named_sshd_carries_no_name() {
106        let elf = elf64(b"sshd", 1, 1);
107        let unsigned = measured_identity(&elf, [4; 32], false);
108        assert!(!unsigned.is_signed());
109        assert!(unsigned.name().is_empty());
110        assert!(!unsigned.is_signed_program("sshd"));
111        assert_eq!(unsigned.digest(), &[4; 32]);
112    }
113
114    #[test]
115    fn verified_image_carries_its_name_under_this_kernels_anchor() {
116        let signed = measured_identity(&elf64(b"sshd", 1, 1), [5; 32], true);
117        if crate::signing::anchor_class() == api::cell_image::ANCHOR_UNSIGNED {
118            // No anchor compiled in: nothing can verify, so nothing is named.
119            assert!(!signed.is_signed());
120        } else {
121            assert_eq!(signed.anchor(), crate::signing::anchor_class());
122            assert!(signed.is_signed_program("sshd"));
123        }
124        // A verified image without the section loads with no name.
125        let anonymous = measured_identity(&elf64(b"sshd", 0, 1), [5; 32], true);
126        assert!(!anonymous.is_signed_program("sshd"));
127    }
