1//! Kernel-mediated EL0 access to sub-page MMIO grants (H-18).
2//!
3//! A grant reaches EL0 through the MMU only as the whole pages that lie inside
4//! the granted bytes ([`user_pages`]): the kernel never sets USER on a page
5//! that also holds registers the Cell was not granted. The BCM2837 VideoCore
6//! mailbox is that case — its 0x40 bytes share a page with the legacy IRQ
7//! controller and ARM timer the kernel tick runs on.
8//!
9//! An EL0 load/store to such a page takes a permission fault. A stage-1
10//! abort taken to EL1 carries no instruction syndrome (ISV = 0 on Armv8.0),
11//! so the kernel decodes the faulting instruction itself; when it is one
12//! plain register load/store and the access lies entirely inside a window the
13//! current task owns, the kernel performs exactly that access and the trap
14//! resumes past the instruction. Anything else stays a Cell fault.
15
16use core::ops::Range;
17
18const PAGE: usize = 4096;
19
20/// Whole pages inside `[base, base + len)`: the only pages a grant may make
21/// EL0-reachable. Empty for a sub-page window or an overflowing range.
22pub(crate) fn user_pages(base: usize, len: usize) -> Range<usize> {
23    let (Some(end), Some(start)) = (base.checked_add(len), base.checked_add(PAGE - 1)) else {
24        return 0..0;
25    };
26    let (start, end) = (start & !(PAGE - 1), end & !(PAGE - 1));
27    if start < end {
28        start..end
29    } else {
30        0..0
31    }
32}

34#[cfg(any(target_arch = "aarch64", test))]
35/// One EL0 data access decoded from the faulting instruction.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub(crate) struct Access {
38    pub(crate) write: bool,
39    /// Bytes: 1, 2, 4 or 8.
40    pub(crate) size: usize,
41    /// Transfer register; 31 is the zero register for loads and stores.
42    pub(crate) reg: usize,
43    pub(crate) sign_extend: bool,
44    /// Load destination is an X register, else a W register.
45    pub(crate) wide: bool,
46}
47
48#[cfg(any(target_arch = "aarch64", test))]
49impl Access {
50    /// The register value a load of `raw` produces.
51    pub(crate) fn extend(self, raw: u64) -> u64 {
52        let bits = self.size * 8;
53        let value = if self.sign_extend && bits < 64 {
54            (((raw << (64 - bits)) as i64) >> (64 - bits)) as u64
55        } else {
56            raw
57        };
58        if self.wide {
59            value
60        } else {
61            value & 0xFFFF_FFFF
62        }
63    }
64}

66#[cfg(any(target_arch = "aarch64", test))]
67/// Decode a lower-EL data abort (`esr`) raised by the A64 instruction `insn`.
68///
69/// The abort must be a permission fault (the page is mapped kernel-only) with
70/// a valid FAR, from a 32-bit instruction, not a cache-maintenance or
71/// stage-1-walk fault. The instruction must be a single general-purpose
72/// LDR/STR (unsigned-offset, register-offset or unscaled form) whose direction
73/// matches the syndrome's WnR: writeback, pair, exclusive, unprivileged,
74/// SIMD/FP and prefetch forms are refused.
75pub(crate) fn decode(esr: u64, insn: u32) -> Option<Access> {
76    const FNV_CM_S1PTW: u64 = (1 << 10) | (1 << 8) | (1 << 7);
77    let permission_fault = (esr >> 26) & 0x3F == 0x24
78        && esr & (1 << 25) != 0
79        && esr & FNV_CM_S1PTW == 0
80        && matches!(esr & 0x3F, 0x0D..=0x0F);
81    // bits[29:24]: 0b111001 unsigned offset; 0b111000 with bit 21 and
82    // bits[11:10] selecting unscaled (0, 00) or register offset (1, 10).
83    let (form, bit21, mode) = ((insn >> 24) & 0x3F, (insn >> 21) & 1, (insn >> 10) & 3);
84    let plain =
85        (form == 0b11_1000 && matches!((bit21, mode), (0, 0) | (1, 2))) || form == 0b11_1001;
86    if !permission_fault || !plain {
87        return None;
88    }
89    let size = 1usize << (insn >> 30);
90    let (write, sign_extend, wide) = match ((insn >> 22) & 3, size) {
91        (0, _) => (true, false, false),
92        (1, _) => (false, false, size == 8),
93        (2, 1..=4) => (false, true, true),
94        (3, 1..=2) => (false, true, false),
95        _ => return None,
96    };
97    if write != (esr & (1 << 6) != 0) {
98        return None;
99    }
100    Some(Access {
101        write,
102        size,
103        reg: (insn & 0x1F) as usize,
104        sign_extend,
105        wide,
106    })
107}

109#[cfg(any(target_arch = "aarch64", test))]
110/// `[far, far + size)` is a naturally aligned access inside the window
111/// `[base, base + len)` on a page the grant left kernel-only.
112pub(crate) fn mediates(base: usize, len: usize, far: usize, size: usize) -> bool {
113    let (Some(window_end), Some(access_end)) = (base.checked_add(len), far.checked_add(size))
114    else {
115        return false;
116    };
117    far.is_multiple_of(size)
118        && far >= base
119        && access_end <= window_end
120        && !user_pages(base, len).contains(&(far & !(PAGE - 1)))
121}

123/// Complete a Cell's faulting access to a sub-page window it owns: `true`
124/// once performed (the trap skips the instruction), else a Cell fault.
125#[cfg(target_arch = "aarch64")]
126#[no_mangle]
127pub extern "Rust" fn vi_emulate_el0_mmio_aarch64(
128    esr: usize,
129    far: usize,
130    pc: usize,
131    regs: &mut [u64; 31],
132) -> bool {
133    use core::ptr::{read_volatile, write_volatile};
134    use core::sync::atomic::{AtomicBool, Ordering};
135    static ANNOUNCED: AtomicBool = AtomicBool::new(false);
136
137    // Read through the kernel's RAM alias of the frame the interrupted EL0
138    // context just executed, as the Cell-fault reporter does.
139    let Some(insn_phys) = crate::memory::paging::virt_to_phys(pc & !3) else {
140        return false;
141    };
142    // SAFETY: the frame holding an instruction that just executed is mapped
143    // RAM, and an aligned word inside it is in bounds of its kernel alias.
144    let insn =
145        unsafe { read_volatile(crate::memory::frame::phys_to_virt(insn_phys) as *const u32) };
146    let Some(access) = decode(esr as u64, insn) else {
147        return false;
148    };
149    let tid = crate::task::current_task_id();
150    // Held across the device access so a concurrent release cannot hand the
151    // window to another task between the ownership check and the access.
152    let registry = super::REGISTRY.lock();
153    let Some((&base, _)) = registry
154        .range(..=far)
155        .next_back()
156        .filter(|(&base, region)| {
157            region.owner_tid == tid && mediates(base, region.len, far, access.size)
158        })
159    else {
160        return false;
161    };
162    // SAFETY: `far` lies inside a device window the current task owns, on a
163    // page the kernel's identity map keeps as kernel-only device memory (the
164    // fault was a permission fault, so the page is mapped). The access has the
165    // Cell's own width and natural alignment and touches only granted bytes.
166    unsafe {
167        let value = regs.get(access.reg).copied().unwrap_or(0);
168        match (access.write, access.size) {
169            (true, 1) => write_volatile(far as *mut u8, value as u8),
170            (true, 2) => write_volatile(far as *mut u16, value as u16),
171            (true, 4) => write_volatile(far as *mut u32, value as u32),
172            (true, _) => write_volatile(far as *mut u64, value),
173            (false, size) => {
174                let raw = match size {
175                    1 => read_volatile(far as *const u8) as u64,
176                    2 => read_volatile(far as *const u16) as u64,
177                    4 => read_volatile(far as *const u32) as u64,
178                    _ => read_volatile(far as *const u64),
179                };
180                if let Some(slot) = regs.get_mut(access.reg) {
181                    *slot = access.extend(raw);
182                }
183            }
184        }
185    }
186    drop(registry);
187    if !ANNOUNCED.swap(true, Ordering::Relaxed) {
188        log::info!("[mmio] kernel-mediated EL0 access: tid={tid} window={base:#x} far={far:#x}");
189    }
190    true
191}
