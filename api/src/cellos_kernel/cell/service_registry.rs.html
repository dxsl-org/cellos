1//! Kernel Service Registry — stable `service_id → current provider tid` mapping.
2//!
3//! The supervisor (init) respawns a dead service under a NEW tid. Clients that
4//! addressed the service by its old tid would break. This registry adds an
5//! indirection: the supervisor registers each service's tid under a well-known
6//! `service_id` ([`api::syscall::service`]) and re-registers the new tid on
7//! respawn; a client resolves `service_id → tid` right before sending, so it
8//! reconnects transparently. Keeping the map in the kernel (the never-die core)
9//! means it survives any service's death, and a dead provider is auto-cleared
10//! ([`clear_tid`]) so a lookup in the death→respawn window returns "none" (the
11//! client retries) instead of a stale tid.
12//!
13//! Only `SpawnCap` holders may `register` (enforced at the syscall dispatch),
14//! so a cell cannot hijack, e.g., the VFS endpoint — the trusted supervisor owns
15//! the namespace. `lookup` is open to all cells.
16
17use crate::sync::Spinlock;
18use alloc::collections::BTreeMap;
19
20/// Upper bound on distinct registered services. Bounds kernel memory and matches
21/// the small, fixed set of well-known service IDs — a runaway registrar cannot
22/// grow the map without bound.
23pub const MAX_SERVICES: usize = 32;
24
25/// `service_id` → current provider task id. `0` is never stored (it is the ABI
26/// "no provider" sentinel returned by `lookup`).
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28enum ServiceEntry {
29    Active(usize),
30    Paused(usize),
31}
32
33static REGISTRY: Spinlock<BTreeMap<u16, ServiceEntry>> = Spinlock::new(BTreeMap::new());
34
35/// Force-release this module's lock during fault teardown.
36///
37/// # Safety
38/// Single-hart; called only from the fault/panic path with interrupts disabled.
39pub unsafe fn force_unlock_locks() {
40    REGISTRY.force_unlock();
41}
42
43/// Register `tid` as the current provider of `service_id`, replacing any prior
44/// entry. Returns `false` (rejected) if the registry is full and `service_id` is
45/// new, or if `tid` is 0 (the reserved "none" sentinel). The SpawnCap authority
46/// check is performed by the caller (syscall dispatch), not here.
47pub fn register(service_id: u16, tid: usize) -> bool {
48    if tid == 0 {
49        return false;
50    }
51    let mut map = REGISTRY.lock();
52    if map.len() >= MAX_SERVICES && !map.contains_key(&service_id) {
53        log::warn!(
54            "[service-registry] full ({} entries); rejecting id {}",
55            MAX_SERVICES,
56            service_id
57        );
58        return false;
59    }
60    map.insert(service_id, ServiceEntry::Active(tid));
61    log::info!("[service-registry] {} -> tid {}", service_id, tid);
62    true
63}
64
65/// Resolve `service_id` to its current provider tid, or `None` if no live
66/// provider is registered. The syscall layer maps `None` to the ABI value 0.
67pub fn lookup(service_id: u16) -> Option<usize> {
68    match REGISTRY.lock().get(&service_id).copied() {
69        Some(ServiceEntry::Active(tid)) => Some(tid),
70        Some(ServiceEntry::Paused(_)) | None => None,
71    }
72}
73
74/// Hide a service from new lookups while its current provider remains runnable.
75///
76/// The compare-and-pause contract prevents a stale supervisor request from
77/// pausing a replacement that another recovery path already registered.
78pub fn pause(service_id: u16, expected_tid: usize) -> bool {
79    let mut map = REGISTRY.lock();
80    match map.get(&service_id).copied() {
81        Some(ServiceEntry::Active(tid)) if tid == expected_tid => {
82            map.insert(service_id, ServiceEntry::Paused(expected_tid));
83            true
84        }
85        Some(ServiceEntry::Paused(tid)) if tid == expected_tid => true,
86        _ => false,
87    }
88}
89
90/// Publish `new_tid` only when `service_id` is still paused at `old_tid`.
91///
92/// The hot-swap barrier calls this while holding `SCHEDULER`, preserving the
93/// global `SCHEDULER -> service registry` lock order.
94pub(crate) fn commit_paused(service_id: u16, old_tid: usize, new_tid: usize) -> bool {
95    if new_tid == 0 {
96        return false;
97    }
98    let mut map = REGISTRY.lock();
99    match map.get(&service_id).copied() {
100        Some(ServiceEntry::Paused(tid)) if tid == old_tid => {
101            map.insert(service_id, ServiceEntry::Active(new_tid));
102            true
103        }
104        _ => false,
105    }
106}
107
108/// Check the exact paused provider without exposing registry representation.
109pub(crate) fn paused_matches(service_id: u16, expected_tid: usize) -> bool {
110    matches!(
111        REGISTRY.lock().get(&service_id).copied(),
112        Some(ServiceEntry::Paused(tid)) if tid == expected_tid
113    )
114}
115
116/// Return whether `tid` is hidden behind any paused service mapping.
117///
118/// IPC admission uses this as the quiesce barrier for callers that cached the
119/// provider tid before the mapping was paused.
120pub fn is_paused_tid(tid: usize) -> bool {
121    REGISTRY
122        .lock()
123        .values()
124        .any(|entry| matches!(entry, ServiceEntry::Paused(provider) if *provider == tid))
125}
126
127/// Return whether `tid` is the active or paused provider of any trusted
128/// supervisor-owned service registration.
129pub fn is_registered_tid(tid: usize) -> bool {
130    REGISTRY.lock().values().any(|entry| {
131        matches!(
132            entry,
133            ServiceEntry::Active(provider) | ServiceEntry::Paused(provider) if *provider == tid
134        )
135    })
136}
137
138/// Remove every registration that points at `tid`. Called from `exit_task` when a
139/// task dies so a client never resolves a service to a dead provider; the
140/// supervisor re-registers the replacement's tid on respawn.
141pub fn clear_tid(tid: usize) {
142    let mut map = REGISTRY.lock();
143    let before = map.len();
144    map.retain(|_, entry| match entry {
145        ServiceEntry::Active(provider) | ServiceEntry::Paused(provider) => *provider != tid,
146    });
147    if map.len() != before {
148        log::info!(
149            "[service-registry] cleared stale entries for dead tid {}",
150            tid
151        );
152    }
153}

155#[cfg(feature = "test-hooks")]
156pub(crate) fn snapshot() -> alloc::vec::Vec<(u16, usize, bool)> {
157    REGISTRY
158        .lock()
159        .iter()
160        .map(|(service_id, entry)| match entry {
161            ServiceEntry::Active(tid) => (*service_id, *tid, false),
162            ServiceEntry::Paused(tid) => (*service_id, *tid, true),
163        })
164        .collect()
165}

167#[cfg(test)]
168mod tests {
169    use super::*;
170
171    #[test]
172    fn register_then_lookup() {
173        assert!(register(api::syscall::service::VFS, 4));
174        assert_eq!(lookup(api::syscall::service::VFS), Some(4));
175    }
176
177    #[test]
178    fn reject_zero_tid() {
179        assert!(!register(api::syscall::service::NET, 0));
180        assert_eq!(lookup(api::syscall::service::NET), None);
181    }
182
183    #[test]
184    fn reregister_updates_tid() {
185        register(api::syscall::service::INPUT, 7);
186        register(api::syscall::service::INPUT, 9);
187        assert_eq!(lookup(api::syscall::service::INPUT), Some(9));
188    }
189
190    #[test]
191    fn clear_tid_removes_dead_provider() {
192        register(api::syscall::service::CONFIG, 12);
193        clear_tid(12);
194        assert_eq!(lookup(api::syscall::service::CONFIG), None);
195    }
196
197    #[test]
198    fn pause_hides_only_the_expected_provider() {
199        const TEST_SERVICE: u16 = 60_000;
200        register(TEST_SERVICE, 21);
201        assert!(!pause(TEST_SERVICE, 20));
202        assert_eq!(lookup(TEST_SERVICE), Some(21));
203
204        assert!(pause(TEST_SERVICE, 21));
205        assert_eq!(lookup(TEST_SERVICE), None);
206        assert!(pause(TEST_SERVICE, 21));
207        assert!(is_paused_tid(21));
208    }
209
210    #[test]
211    fn register_reactivates_a_paused_service() {
212        const TEST_SERVICE: u16 = 60_001;
213        register(TEST_SERVICE, 31);
214        assert!(pause(TEST_SERVICE, 31));
215        assert!(register(TEST_SERVICE, 31));
216        assert_eq!(lookup(TEST_SERVICE), Some(31));
217    }
218
219    #[test]
220    fn commit_requires_exact_paused_provider() {
221        const TEST_SERVICE: u16 = 60_002;
222        register(TEST_SERVICE, 41);
223        assert!(!commit_paused(TEST_SERVICE, 40, 42));
224        assert_eq!(lookup(TEST_SERVICE), Some(41));
225        assert!(pause(TEST_SERVICE, 41));
226        assert!(!commit_paused(TEST_SERVICE, 40, 42));
227        assert!(commit_paused(TEST_SERVICE, 41, 42));
228        assert_eq!(lookup(TEST_SERVICE), Some(42));
229    }
