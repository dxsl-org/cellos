1//! Kernel Service Registry — stable `service_id → current provider` mapping.
2//!
3//! The supervisor (init) respawns a dead service under a NEW tid. Clients that
4//! addressed the service by its old tid would break. This registry adds an
5//! indirection: the supervisor registers each service's tid under a well-known
6//! `service_id` ([`api::syscall::service`]) and re-registers the new tid on
7//! respawn; a client resolves `service_id → tid` right before sending, so it
8//! reconnects transparently. Keeping the map in the kernel (the never-die core)
9//! means it survives any service's death, and a dead provider is auto-cleared
10//! ([`clear_tid`]) so a lookup in the death→respawn window returns "none" (the
11//! client retries) instead of a stale tid.
12//!
13//! Each entry also records the provider's **Cell identity** — `(cell_id,
14//! generation)`. A tid says where to send; it is not an identity, because a
15//! `CellId` slot may be reused later while the generation cannot. Recording the
16//! identity here is what lets `LookupServiceBound` answer with a binding instead
17//! of a bare tid, and it is the same axis [`api::caller_identity::CallerIdentity`]
18//! and [`api::cell_owner::CellOwner`] already use — one identity concept, not two.
19//! A provider whose recorded identity is not live (`cell_id == 0` or
20//! `generation == 0`) is **not bindable** and is reported as "no live provider",
21//! so the kernel never states a binding it cannot stand behind.
22//!
23//! Only `SpawnCap` holders may `register` (enforced at the syscall dispatch),
24//! so a cell cannot hijack, e.g., the VFS endpoint — the trusted supervisor owns
25//! the namespace. `lookup` is open to all cells.
26
27use crate::sync::Spinlock;
28use alloc::collections::BTreeMap;
29
30/// Upper bound on distinct registered services. Bounds kernel memory and matches
31/// the small, fixed set of well-known service IDs — a runaway registrar cannot
32/// grow the map without bound.
33pub const MAX_SERVICES: usize = 32;
34
35/// `service_id` → current provider. `0` is never stored as a tid (it is the ABI
36/// "no provider" sentinel returned by `lookup`).
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38enum ServiceEntry {
39    Active {
40        tid: usize,
41        cell_id: u64,
42        generation: u64,
43    },
44    Paused {
45        tid: usize,
46        cell_id: u64,
47        generation: u64,
48    },
49}
50
51impl ServiceEntry {
52    fn tid(self) -> usize {
53        match self {
54            ServiceEntry::Active { tid, .. } | ServiceEntry::Paused { tid, .. } => tid,
55        }
56    }
57
58    /// The provider's Cell identity, or `None` when it names no live Cell.
59    ///
60    /// `cell_id == 0` is the kernel's own allocation rather than any Cell, and a
61    /// zero generation is not an epoch. Either one means the entry cannot support
62    /// a binding, so `LookupServiceBound` treats it exactly like an absent provider.
63    fn identity(self) -> Option<(u64, u64)> {
64        let (ServiceEntry::Active {
65            cell_id, generation, ..
66        }
67        | ServiceEntry::Paused {
68            cell_id, generation, ..
69        }) = self;
70        (cell_id != 0 && generation != 0).then_some((cell_id, generation))
71    }
72}
73
74static REGISTRY: Spinlock<BTreeMap<u16, ServiceEntry>> = Spinlock::new(BTreeMap::new());
75
76/// Force-release this module's lock during fault teardown.
77///
78/// # Safety
79/// Single-hart; called only from the fault/panic path with interrupts disabled.
80pub unsafe fn force_unlock_locks() {
81    REGISTRY.force_unlock();
82}

84/// Register `tid` as the current provider of `service_id`, replacing any prior
85/// entry. Returns `false` (rejected) if the registry is full and `service_id` is
86/// new, or if `tid` is 0 (the reserved "none" sentinel). The SpawnCap authority
87/// check is performed by the caller (syscall dispatch), not here.
88///
89/// `cell_id`/`generation` are the provider's live identity, read from the
90/// scheduler by the caller. They are recorded, never invented here.
91pub fn register(service_id: u16, tid: usize, cell_id: u64, generation: u64) -> bool {
92    if tid == 0 {
93        return false;
94    }
95    let mut map = REGISTRY.lock();
96    if map.len() >= MAX_SERVICES && !map.contains_key(&service_id) {
97        log::warn!(
98            "[service-registry] full ({} entries); rejecting id {}",
99            MAX_SERVICES,
100            service_id
101        );
102        return false;
103    }
104    map.insert(
105        service_id,
106        ServiceEntry::Active {
107            tid,
108            cell_id,
109            generation,
110        },
111    );
112    log::info!(
113        "[service-registry] {} -> tid {} (cell {} gen {})",
114        service_id,
115        tid,
116        cell_id,
117        generation
118    );
119    true
120}

122/// Resolve `service_id` to its current provider tid, or `None` if no live
123/// provider is registered. The syscall layer maps `None` to the ABI value 0.
124pub fn lookup(service_id: u16) -> Option<usize> {
125    match REGISTRY.lock().get(&service_id).copied() {
126        Some(ServiceEntry::Active { tid, .. }) => Some(tid),
127        Some(ServiceEntry::Paused { .. }) | None => None,
128    }
129}
130
131/// Resolve `service_id` to its current provider's full binding
132/// `(tid, cell_id, generation)`.
133///
134/// Returns `None` for an absent provider, a paused provider (the hot-swap quiesce
135/// barrier must stay in force for new work) and a provider whose recorded identity
136/// is not live. All three are "no live binding", and the syscall layer maps them to
137/// the ABI value 0 exactly like [`lookup`].
138pub fn lookup_bound(service_id: u16) -> Option<(usize, u64, u64)> {
139    let entry = REGISTRY.lock().get(&service_id).copied()?;
140    if !matches!(entry, ServiceEntry::Active { .. }) {
141        return None;
142    }
143    let (cell_id, generation) = entry.identity()?;
144    Some((entry.tid(), cell_id, generation))
145}

147/// Hide a service from new lookups while its current provider remains runnable.
148///
149/// The compare-and-pause contract prevents a stale supervisor request from
150/// pausing a replacement that another recovery path already registered.
151pub fn pause(service_id: u16, expected_tid: usize) -> bool {
152    let mut map = REGISTRY.lock();
153    match map.get(&service_id).copied() {
154        // Pausing preserves the recorded identity verbatim: the pause is a
155        // visibility change, not a re-registration, and a provider's identity does
156        // not change while it is being quiesced.
157        Some(ServiceEntry::Active {
158            tid,
159            cell_id,
160            generation,
161        }) if tid == expected_tid => {
162            map.insert(
163                service_id,
164                ServiceEntry::Paused {
165                    tid: expected_tid,
166                    cell_id,
167                    generation,
168                },
169            );
170            true
171        }
172        Some(ServiceEntry::Paused { tid, .. }) if tid == expected_tid => true,
173        _ => false,
174    }
175}

177/// Publish `new_tid` only when `service_id` is still paused at `old_tid`.
178///
179/// The hot-swap barrier calls this while holding `SCHEDULER`, preserving the
180/// global `SCHEDULER -> service registry` lock order. `cell_id`/`generation` are
181/// the replacement's live identity, read from the scheduler by that caller.
182pub(crate) fn commit_paused(
183    service_id: u16,
184    old_tid: usize,
185    new_tid: usize,
186    cell_id: u64,
187    generation: u64,
188) -> bool {
189    if new_tid == 0 {
190        return false;
191    }
192    let mut map = REGISTRY.lock();
193    match map.get(&service_id).copied() {
194        Some(ServiceEntry::Paused { tid, .. }) if tid == old_tid => {
195            map.insert(
196                service_id,
197                ServiceEntry::Active {
198                    tid: new_tid,
199                    cell_id,
200                    generation,
201                },
202            );
203            true
204        }
205        _ => false,
206    }
207}

209/// Check the exact paused provider without exposing registry representation.
210pub(crate) fn paused_matches(service_id: u16, expected_tid: usize) -> bool {
211    matches!(
212        REGISTRY.lock().get(&service_id).copied(),
213        Some(ServiceEntry::Paused { tid, .. }) if tid == expected_tid
214    )
215}

217/// Return whether `tid` is hidden behind any paused service mapping.
218///
219/// IPC admission uses this as the quiesce barrier for callers that cached the
220/// provider tid before the mapping was paused.
221pub fn is_paused_tid(tid: usize) -> bool {
222    REGISTRY
223        .lock()
224        .values()
225        .any(|entry| matches!(entry, ServiceEntry::Paused { tid: provider, .. } if *provider == tid))
226}
227
228/// Return whether `tid` is the active or paused provider of any trusted
229/// supervisor-owned service registration.
230pub fn is_registered_tid(tid: usize) -> bool {
231    REGISTRY
232        .lock()
233        .values()
234        .any(|entry| matches!(entry, ServiceEntry::Active { .. } | ServiceEntry::Paused { .. } if entry.tid() == tid))
235}
236
237/// Remove every registration that points at `tid`. Called from `exit_task` when a
238/// task dies so a client never resolves a service to a dead provider; the
239/// supervisor re-registers the replacement's tid on respawn.
240pub fn clear_tid(tid: usize) {
241    let mut map = REGISTRY.lock();
242    let before = map.len();
243    map.retain(|_, entry| entry.tid() != tid);
244    if map.len() != before {
245        log::info!(
246            "[service-registry] cleared stale entries for dead tid {}",
247            tid
248        );
249    }
250}

252#[cfg(feature = "test-hooks")]
253pub(crate) fn snapshot() -> alloc::vec::Vec<(u16, usize, bool)> {
254    REGISTRY
255        .lock()
256        .iter()
257        .map(|(service_id, entry)| match entry {
258            ServiceEntry::Active { tid, .. } => (*service_id, *tid, false),
259            ServiceEntry::Paused { tid, .. } => (*service_id, *tid, true),
260        })
261        .collect()
262}

271    #[test]
272    fn register_then_lookup() {
273        assert!(register(api::syscall::service::VFS, 4, CELL, GEN));

280        assert_eq!(lookup(api::syscall::service::NET), None);
281    }

283    #[test]
284    fn reregister_updates_tid() {
285        register(api::syscall::service::INPUT, 7, CELL, GEN);
286        register(api::syscall::service::INPUT, 9, CELL + 1, GEN + 1);
287        assert_eq!(lookup(api::syscall::service::INPUT), Some(9));
288    }

290    #[test]
291    fn clear_tid_removes_dead_provider() {
292        register(api::syscall::service::CONFIG, 12, CELL, GEN);
293        clear_tid(12);
294        assert_eq!(lookup(api::syscall::service::CONFIG), None);
295    }
296
297    #[test]
298    fn pause_hides_only_the_expected_provider() {
299        const TEST_SERVICE: u16 = 60_000;
300        register(TEST_SERVICE, 21, CELL, GEN);
301        assert!(!pause(TEST_SERVICE, 20));
302        assert_eq!(lookup(TEST_SERVICE), Some(21));
303
304        assert!(pause(TEST_SERVICE, 21));
305        assert_eq!(lookup(TEST_SERVICE), None);
306        assert!(pause(TEST_SERVICE, 21));
307        assert!(is_paused_tid(21));
308    }
309
310    #[test]
311    fn register_reactivates_a_paused_service() {
312        const TEST_SERVICE: u16 = 60_001;
313        register(TEST_SERVICE, 31, CELL, GEN);
314        assert!(pause(TEST_SERVICE, 31));
315        assert!(register(TEST_SERVICE, 31, CELL, GEN));
316        assert_eq!(lookup(TEST_SERVICE), Some(31));
317    }
318
319    #[test]
320    fn commit_requires_exact_paused_provider() {
321        const TEST_SERVICE: u16 = 60_002;
322        register(TEST_SERVICE, 41, CELL, GEN);
323        assert!(!commit_paused(TEST_SERVICE, 40, 42, CELL, GEN));
324        assert_eq!(lookup(TEST_SERVICE), Some(41));
325        assert!(pause(TEST_SERVICE, 41));
326        assert!(!commit_paused(TEST_SERVICE, 40, 42, CELL, GEN));
327        assert!(commit_paused(TEST_SERVICE, 41, 42, CELL, GEN + 1));
328        assert_eq!(lookup(TEST_SERVICE), Some(42));
329        // The replacement's identity is the one that was committed, not the source's.
330        assert_eq!(lookup_bound(TEST_SERVICE), Some((42, CELL, GEN + 1)));
331    }
332
333    #[test]
334    fn lookup_bound_reports_the_recorded_identity() {
335        const TEST_SERVICE: u16 = 60_003;
336        register(TEST_SERVICE, 51, CELL, GEN);
337        assert_eq!(lookup_bound(TEST_SERVICE), Some((51, CELL, GEN)));
338
339        // Re-registering a new incarnation replaces both tid and identity, so a
340        // holder of the old binding can tell the provider changed.
341        register(TEST_SERVICE, 52, CELL, GEN + 1);
342        assert_eq!(lookup_bound(TEST_SERVICE), Some((52, CELL, GEN + 1)));
343    }
344
345    #[test]
346    fn lookup_bound_denies_absent_paused_and_identityless_providers() {
347        const ABSENT: u16 = 60_004;
348        const PAUSED: u16 = 60_005;
349        const IDENTITYLESS: u16 = 60_006;
350
351        assert_eq!(lookup_bound(ABSENT), None);
352
353        register(PAUSED, 61, CELL, GEN);
354        assert!(pause(PAUSED, 61));
355        // A paused provider must not hand out a binding: the quiesce barrier holds.
356        assert_eq!(lookup_bound(PAUSED), None);
357
358        // A provider with no Cell identity (kernel-owned task) is not bindable, and
359        // the kernel must not state a binding it cannot stand behind.
360        register(IDENTITYLESS, 71, 0, GEN);
361        assert_eq!(lookup(IDENTITYLESS), Some(71));
362        assert_eq!(lookup_bound(IDENTITYLESS), None);
363        register(IDENTITYLESS, 72, CELL, 0);
364        assert_eq!(lookup_bound(IDENTITYLESS), None);
365    }
366
367    #[test]
368    fn clear_tid_removes_paused_and_active_entries() {
369        const ACTIVE: u16 = 60_007;
370        const PAUSED: u16 = 60_008;
371        register(ACTIVE, 81, CELL, GEN);
372        register(PAUSED, 82, CELL, GEN);
373        assert!(pause(PAUSED, 82));
374
375        clear_tid(81);
376        clear_tid(82);
377        assert_eq!(lookup_bound(ACTIVE), None);
378        assert_eq!(lookup_bound(PAUSED), None);
379        assert!(!is_paused_tid(82));
380    }
