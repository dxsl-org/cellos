1//! Kernel Service Registry — stable `service_id → current provider` mapping.
2//!
3//! The supervisor (init) respawns a dead service under a NEW tid. Clients that
4//! addressed the service by its old tid would break. This registry adds an
5//! indirection: the supervisor registers each service's tid under a well-known
6//! `service_id` ([`abi::syscall::service`]) and re-registers the new tid on
7//! respawn; a client resolves `service_id → tid` right before sending, so it
8//! reconnects transparently. Keeping the map in the kernel (the never-die core)
9//! means it survives any service's death, and a dead provider is auto-cleared
10//! ([`clear_tid`]) so a lookup in the death→respawn window returns "none" (the
11//! client retries) instead of a stale tid.
12//!
13//! Each entry also records the provider's **Cell identity** — `(cell_id,
14//! generation)`. A tid says where to send; it is not an identity, because a
15//! `CellId` slot may be reused later while the generation cannot. Recording the
16//! identity here is what lets `LookupServiceBound` answer with a binding instead
17//! of a bare tid, and it is the same axis [`abi::caller_identity::CallerIdentity`]
18//! and [`abi::cell_owner::CellOwner`] already use — one identity concept, not two.
19//! A provider whose recorded identity is not live (`cell_id == 0` or
20//! `generation == 0`) is **not bindable** and is reported as "no live provider",
21//! so the kernel never states a binding it cannot stand behind.
22//!
23//! Publishing is owner-bound ([`register_as`]): the well-known ids below
24//! [`PRIVATE_ID_BASE`] belong to the namespace owners (trusted init and
25//! `SupervisorCap` holders), so a cell that merely holds `SpawnCap` — e.g. a USB
26//! host driver that spawns its protocol child — cannot rebind VFS, NET, KMS or
27//! SUPERVISOR. A spawn-only cell may publish a private id, and the first such
28//! registrar owns it until it exits. `lookup` is open to all cells.
29
30use crate::sync::Spinlock;
31use alloc::collections::BTreeMap;
32
33/// Upper bound on distinct registered services. Bounds kernel memory and matches
34/// the small, fixed set of well-known service IDs — a runaway registrar cannot
35/// grow the map without bound.
36pub const MAX_SERVICES: usize = 32;
37
38/// `service_id` → current provider. `0` is never stored as a tid (it is the ABI
39/// "no provider" sentinel returned by `lookup`).
40#[derive(Clone, Copy, Debug, PartialEq, Eq)]
41struct ServiceEntry {
42    tid: usize,
43    cell_id: u64,
44    generation: u64,
45    /// [`abi::syscall::service_flags`] granted to this incarnation.
46    flags: usize,
47    /// Hidden from lookups by a hot-swap quiesce; the provider still runs.
48    paused: bool,
49}
50
51impl ServiceEntry {
52    /// The provider's Cell identity, or `None` when it names no live Cell.
53    ///
54    /// `cell_id == 0` is the kernel's own allocation rather than any Cell, and a
55    /// zero generation is not an epoch. Either one means the entry cannot support
56    /// a binding, so `LookupServiceBound` treats it exactly like an absent provider.
57    fn identity(self) -> Option<(u64, u64)> {
58        (self.cell_id != 0 && self.generation != 0).then_some((self.cell_id, self.generation))
59    }
60
61    fn names(self, cell_id: u64, generation: u64) -> bool {
62        self.identity() == Some((cell_id, generation))
63    }
64}
65
66static REGISTRY: Spinlock<BTreeMap<u16, ServiceEntry>> =
67    Spinlock::ranked(BTreeMap::new(), crate::sync::rank::SERVICE_REGISTRY);
68
69/// Private id → tid of the spawn-only registrar that owns it. Lock order:
70/// `OWNERS → REGISTRY`; nothing takes `OWNERS` while holding `REGISTRY`.
71static OWNERS: Spinlock<BTreeMap<u16, usize>> =
72    Spinlock::ranked(BTreeMap::new(), crate::sync::rank::OWNERS);
73
74/// Ids below this are the well-known namespace of [`abi::syscall::service`];
75/// ids at or above it are private ids a lifecycle parent publishes for its own
76/// children (the demo and test fixtures use `0x7Axx`).
77pub const PRIVATE_ID_BASE: u16 = 0x100;
78
79/// Who asks to publish a service id, as established by the syscall dispatch.
80#[derive(Clone, Copy, Debug, PartialEq, Eq)]
81pub enum Registrar {
82    /// Trusted init or a `SupervisorCap` holder: owns the whole namespace and
83    /// rebinds any id (respawn, hot-swap rollback).
84    Namespace,
85    /// A `SpawnCap`-only task, by tid: private ids only, first registrar owns.
86    Spawner(usize),
87}
88
89/// Why [`register_as`] refused.
90#[derive(Clone, Copy, Debug, PartialEq, Eq)]
91pub enum Refusal {
92    /// The id is well-known or owned by another live registrar, or flags were
93    /// asked by a registrar that is not the namespace owner.
94    NotOwner,
95    /// Zero tid, an undefined flag bit, or the registry/owner table is full.
96    Rejected,
97}
98
99/// Publish `tid` as the provider of `service_id` on behalf of `registrar`.
100///
101/// A [`Registrar::Spawner`] may publish only a private id, without flags, and
102/// only one that no other live spawner owns; its first successful publish makes
103/// it the owner until it exits ([`clear_tid`]), so it can rebind its own
104/// restarted child.
105pub fn register_as(
106    registrar: Registrar,
107    service_id: u16,
108    tid: usize,
109    cell_id: u64,
110    generation: u64,
111    flags: usize,
112) -> Result<(), Refusal> {
113    if flags & !abi::syscall::service_flags::ALL != 0 {
114        return Err(Refusal::Rejected);
115    }
116    let Registrar::Spawner(spawner) = registrar else {
117        return insert(service_id, tid, cell_id, generation, flags)
118            .then_some(())
119            .ok_or(Refusal::Rejected);
120    };
121    if service_id < PRIVATE_ID_BASE || flags != 0 {
122        return Err(Refusal::NotOwner);
123    }
124    let mut owners = OWNERS.lock();
125    match owners.get(&service_id) {
126        Some(&owner) if owner != spawner => return Err(Refusal::NotOwner),
127        None if owners.len() >= MAX_SERVICES => return Err(Refusal::Rejected),
128        _ => {}
129    }
130    if !register(service_id, tid, cell_id, generation) {
131        return Err(Refusal::Rejected);
132    }
133    #[cfg(feature = "p01-trace")]
134    let _semantic = crate::memory::heap::p01_owner::creation(
135        None,
136        crate::memory::heap::p01_owner::SERVICE_SHARED,
137    );
138    owners.insert(service_id, spawner);
139    Ok(())
140}

142/// Register `tid` as the current provider of `service_id`, without flags,
143/// replacing any prior entry. Returns `false` (rejected) if the registry is
144/// full and `service_id` is new, or if `tid` is 0 (the reserved "none"
145/// sentinel). No authority check: the generic `RegisterService` path uses
146/// [`register_as`]; the kernel-defined self-registration paths (GPU driver,
147/// Silo, demo) check their own authority.
148///
149/// `cell_id`/`generation` are the provider's live identity, read from the
150/// scheduler by the caller. They are recorded, never invented here.
151pub fn register(service_id: u16, tid: usize, cell_id: u64, generation: u64) -> bool {
152    insert(service_id, tid, cell_id, generation, 0)
153}
154
155fn insert(service_id: u16, tid: usize, cell_id: u64, generation: u64, flags: usize) -> bool {
156    if tid == 0 {
157        return false;
158    }
159    let mut map = REGISTRY.lock();
160    if map.len() >= MAX_SERVICES && !map.contains_key(&service_id) {
161        log::warn!(
162            "[service-registry] full ({} entries); rejecting id {}",
163            MAX_SERVICES,
164            service_id
165        );
166        return false;
167    }
168    #[cfg(feature = "p01-trace")]
169    let _semantic = crate::memory::heap::p01_owner::creation(
170        None,
171        crate::memory::heap::p01_owner::SERVICE_SHARED,
172    );
173    map.insert(
174        service_id,
175        ServiceEntry {
176            tid,
177            cell_id,
178            generation,
179            flags,
180            paused: false,
181        },
182    );
183    #[cfg(feature = "p01-trace")]
184    drop(_semantic);
185    log::info!(
186        "[service-registry] {} -> tid {} (cell {} gen {} flags {:#x})",
187        service_id,
188        tid,
189        cell_id,
190        generation,
191        flags
192    );
193    true
194}

196/// Resolve `service_id` to its current provider tid, or `None` if no live
197/// provider is registered. The syscall layer maps `None` to the ABI value 0.
198pub fn lookup(service_id: u16) -> Option<usize> {
199    REGISTRY
200        .lock()
201        .get(&service_id)
202        .filter(|entry| !entry.paused)
203        .map(|entry| entry.tid)
204}
205
206/// Resolve `service_id` to its current provider's full binding
207/// `(tid, cell_id, generation)`.
208///
209/// Returns `None` for an absent provider, a paused provider (the hot-swap quiesce
210/// barrier must stay in force for new work) and a provider whose recorded identity
211/// is not live. All three are "no live binding", and the syscall layer maps them to
212/// the ABI value 0 exactly like [`lookup`].
213pub fn lookup_bound(service_id: u16) -> Option<(usize, u64, u64)> {
214    let entry = REGISTRY.lock().get(&service_id).copied()?;
215    if entry.paused {
216        return None;
217    }
218    let (cell_id, generation) = entry.identity()?;
219    Some((entry.tid, cell_id, generation))
220}

222/// Whether the Cell incarnation `(cell_id, generation)` provides a service
223/// whose row carries `RETAINS_OUTER_CALLER`, Active or Paused.
224///
225/// Lock order SCHEDULER -> REGISTRY: callers hold the scheduler lock.
226pub fn retains_outer_caller(cell_id: u64, generation: u64) -> bool {
227    REGISTRY.lock().values().any(|entry| {
228        entry.flags & abi::syscall::service_flags::RETAINS_OUTER_CALLER != 0
229            && entry.names(cell_id, generation)
230    })
231}
232
233/// Whether the Cell incarnation `(cell_id, generation)` is the Active provider
234/// of a service whose row carries `OWNER_WATCH`. A Paused row grants nothing,
235/// and a provider bound to one row has no right through another's binding.
236pub fn grants_owner_watch(cell_id: u64, generation: u64) -> bool {
237    REGISTRY.lock().values().any(|entry| {
238        entry.flags & abi::syscall::service_flags::OWNER_WATCH != 0
239            && !entry.paused
240            && entry.names(cell_id, generation)
241    })
242}

244/// Whether the hot-swap path may never pause, freeze or republish
245/// `service_id`'s provider.
246///
247/// `MACHINE_PIN` is registered only by trusted init for its own `machine-pin`
248/// child (`RegisterService`, ADR-0029 §3); the VFS admits the live provider to
249/// the pin store and 433 answers only it. A swap would publish a
250/// supervisor-spawned successor that init then adopts, so a `SupervisorCap`
251/// holder could hand itself the machine facts and pin write access.
252pub fn swap_refused(service_id: u16) -> bool {
253    service_id == abi::syscall::service::MACHINE_PIN
254}
255
256/// Whether `tid` is the active or paused provider of a service the hot-swap
257/// path refuses ([`swap_refused`]).
258pub fn is_swap_refused_tid(tid: usize) -> bool {
259    REGISTRY
260        .lock()
261        .iter()
262        .any(|(&service_id, entry)| swap_refused(service_id) && entry.tid == tid)
263}
264
265/// Hide a service from new lookups while its current provider remains runnable.
266///
267/// The compare-and-pause contract prevents a stale supervisor request from
268/// pausing a replacement that another recovery path already registered.
269/// Pausing preserves the recorded identity and flags verbatim: it is a
270/// visibility change, not a re-registration. A [`swap_refused`] id is never
271/// paused.
272pub fn pause(service_id: u16, expected_tid: usize) -> bool {
273    if swap_refused(service_id) {
274        return false;
275    }
276    match REGISTRY.lock().get_mut(&service_id) {
277        Some(entry) if entry.tid == expected_tid => {
278            entry.paused = true;
279            true
280        }
281        _ => false,
282    }
283}

285/// Publish `new_tid` only when `service_id` is still paused at `old_tid`.
286///
287/// The hot-swap barrier calls this while holding `SCHEDULER`, preserving the
288/// global `SCHEDULER -> service registry` lock order. `cell_id`/`generation` are
289/// the replacement's live identity, read from the scheduler by that caller. The
290/// replacement inherits the row's flags. A [`swap_refused`] id is never
291/// republished.
292pub(crate) fn commit_paused(
293    service_id: u16,
294    old_tid: usize,
295    new_tid: usize,
296    cell_id: u64,
297    generation: u64,
298) -> bool {
299    if new_tid == 0 || swap_refused(service_id) {
300        return false;
301    }
302    match REGISTRY.lock().get_mut(&service_id) {
303        Some(entry) if entry.paused && entry.tid == old_tid => {
304            *entry = ServiceEntry {
305                tid: new_tid,
306                cell_id,
307                generation,
308                flags: entry.flags,
309                paused: false,
310            };
311            true
312        }
313        _ => false,
314    }
315}

317/// Check the exact paused provider without exposing registry representation.
318pub(crate) fn paused_matches(service_id: u16, expected_tid: usize) -> bool {
319    REGISTRY
320        .lock()
321        .get(&service_id)
322        .is_some_and(|entry| entry.paused && entry.tid == expected_tid)
323}

325/// Register `tid` as the provider of `service_id`, without flags, only while
326/// `service_id` has no row, active or paused: an existing row is never
327/// replaced. The check and the insert share one `REGISTRY` hold. `false`, with
328/// nothing changed, when a row exists, the registry is full, or `tid` is 0.
329pub(crate) fn register_if_absent(
330    service_id: u16,
331    tid: usize,
332    cell_id: u64,
333    generation: u64,
334) -> bool {
335    if tid == 0 {
336        return false;
337    }
338    let inserted = {
339        let mut map = REGISTRY.lock();
340        if map.contains_key(&service_id) || map.len() >= MAX_SERVICES {
341            false
342        } else {
343            #[cfg(feature = "p01-trace")]
344            let _semantic = crate::memory::heap::p01_owner::creation(
345                None,
346                crate::memory::heap::p01_owner::SERVICE_SHARED,
347            );
348            map.insert(
349                service_id,
350                ServiceEntry {
351                    tid,
352                    cell_id,
353                    generation,
354                    flags: 0,
355                    paused: false,
356                },
357            );
358            true
359        }
360    };
361    if inserted {
362        log::info!(
363            "[service-registry] {} -> tid {} (cell {} gen {}), registered while vacant",
364            service_id,
365            tid,
366            cell_id,
367            generation
368        );
369    }
370    inserted
371}

373/// Return whether `tid` is hidden behind any paused service mapping.
374///
375/// IPC admission uses this as the quiesce barrier for callers that cached the
376/// provider tid before the mapping was paused.
377pub fn is_paused_tid(tid: usize) -> bool {
378    REGISTRY
379        .lock()
380        .values()
381        .any(|entry| entry.paused && entry.tid == tid)
382}

384/// Return whether `tid` is the active or paused provider of any trusted
385/// supervisor-owned service registration.
386pub fn is_registered_tid(tid: usize) -> bool {
387    REGISTRY.lock().values().any(|entry| entry.tid == tid)
388}

390/// Remove every registration that points at `tid`. Called from `exit_task` when a
391/// task dies so a client never resolves a service to a dead provider; the
392/// supervisor re-registers the replacement's tid on respawn. A dead registrar
393/// also gives up its private ids, so its restarted successor can claim them.
394pub fn clear_tid(tid: usize) {
395    {
396        let mut map = REGISTRY.lock();
397        let before = map.len();
398        map.retain(|_, entry| entry.tid != tid);
399        if map.len() != before {
400            log::info!(
401                "[service-registry] cleared stale entries for dead tid {}",
402                tid
403            );
404        }
405    }
406    OWNERS.lock().retain(|_, owner| *owner != tid);
407}

409/// Re-point `service_id` from provider `from_tid` to `to = (tid, cell_id,
410/// generation)` in place: only an active row that still names `from_tid`
411/// changes, so a racing re-registration is never overwritten. The row keeps
412/// its map slot, so nothing is allocated (exit teardown, which may run in an
413/// ISR, calls it); it takes `to`'s identity with no flags and no pause, as
414/// [`register`] would write it. `false`, with nothing changed, when no row of
415/// `service_id` names `from_tid`, the row is paused, or `to` names tid 0.
416///
417/// A paused row belongs to a hot swap in flight: only [`commit_paused`] moves
418/// it, and [`clear_tid`] removes it when its provider dies, so the swap's
419/// commit then fails instead of publishing over a re-pointed row.
420pub(crate) fn repoint(service_id: u16, from_tid: usize, to: (usize, u64, u64)) -> bool {
421    let (tid, cell_id, generation) = to;
422    if tid == 0 {
423        return false;
424    }
425    {
426        let mut map = REGISTRY.lock();
427        let Some(entry) = map
428            .get_mut(&service_id)
429            .filter(|entry| entry.tid == from_tid && !entry.paused)
430        else {
431            return false;
432        };
433        *entry = ServiceEntry {
434            tid,
435            cell_id,
436            generation,
437            flags: 0,
438            paused: false,
439        };
440    }
441    log::info!(
442        "[service-registry] {} -> tid {} (cell {} gen {}), re-pointed from tid {}",
443        service_id,
444        tid,
445        cell_id,
446        generation,
447        from_tid
448    );
449    true
450}

452/// Remove `service_id`'s row while it is active and names `tid`: a live
453/// provider that lost the authority behind the row (a runtime capability
454/// revoke). A paused row is left for its hot swap, as in [`repoint`]. Frees
455/// at most one map node, never allocates. `false` when nothing was removed.
456pub(crate) fn remove_active(service_id: u16, tid: usize) -> bool {
457    let removed = {
458        let mut map = REGISTRY.lock();
459        let named = map
460            .get(&service_id)
461            .is_some_and(|entry| entry.tid == tid && !entry.paused);
462        named && map.remove(&service_id).is_some()
463    };
464    if removed {
465        log::info!(
466            "[service-registry] {} removed: tid {} lost its authority",
467            service_id,
468            tid
469        );
470    }
471    removed
472}

476pub(crate) fn snapshot() -> alloc::vec::Vec<(u16, usize, bool, usize)> {
477    REGISTRY
478        .lock()
479        .iter()
480        .map(|(service_id, entry)| (*service_id, entry.tid, entry.paused, entry.flags))

762    /// D1: flags come only from the namespace owner, an undefined bit is refused,
763    /// RETAINS_OUTER_CALLER holds through a quiesce while OWNER_WATCH does not,
764    /// and a hot-swap commit hands both to the replacement.
765    #[test]
766    fn registration_flags_follow_the_row() {
767        use abi::syscall::service_flags::{ALL, OWNER_WATCH, RETAINS_OUTER_CALLER};
768        const ID: u16 = 0x7A0C;
769        const OLD: usize = 0x5_11D0;
770        const NEW: usize = 0x5_11D1;
771        const NEW_CELL: u64 = CELL + 1;
772
773        assert_eq!(
774            register_as(Registrar::Namespace, ID, OLD, CELL, GEN, ALL + 1),
775            Err(Refusal::Rejected)
776        );
777        assert_eq!(
778            register_as(Registrar::Spawner(OLD), ID, OLD, CELL, GEN, OWNER_WATCH),
779            Err(Refusal::NotOwner)
780        );
781        assert_eq!(lookup(ID), None);
782
783        assert_eq!(
784            register_as(Registrar::Namespace, ID, OLD, CELL, GEN, ALL),
785            Ok(())
786        );
787        assert!(retains_outer_caller(CELL, GEN) && grants_owner_watch(CELL, GEN));
788        assert!(!retains_outer_caller(CELL, GEN + 1) && !grants_owner_watch(CELL, GEN + 1));
789
790        assert!(pause(ID, OLD));
791        assert!(retains_outer_caller(CELL, GEN));
792        assert!(!grants_owner_watch(CELL, GEN));
793
794        assert!(commit_paused(ID, OLD, NEW, NEW_CELL, GEN));
795        assert!(!retains_outer_caller(CELL, GEN));
796        assert!(retains_outer_caller(NEW_CELL, GEN) && grants_owner_watch(NEW_CELL, GEN));
797
798        // A plain re-registration states its own flags.
799        assert!(register(ID, NEW, NEW_CELL, GEN));
800        assert!(!retains_outer_caller(NEW_CELL, GEN) && !grants_owner_watch(NEW_CELL, GEN));
801        assert_eq!(
802            register_as(
803                Registrar::Namespace,
804                ID,
805                NEW,
806                NEW_CELL,
807                GEN,
808                RETAINS_OUTER_CALLER
809            ),
810            Ok(())
811        );
812        clear_tid(NEW);
813        assert!(!retains_outer_caller(NEW_CELL, GEN));
814    }
