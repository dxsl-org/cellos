1//! Kernel capability registry.
2//!
3//! Every open file (or other resource) in ViCell is represented by a `CapId`
4//! assigned by this registry.  The registry enforces single-ownership:
5//! only the cell that allocated or received a cap may use or revoke it.
6//!
7//! Current implementation is a simple global `BTreeMap` protected by a
8//! Spinlock.  A sharded design (one lock per 256 cap IDs) is deferred
9//! to Phase 13 when concurrent pressure is expected to be higher.
10
11use crate::sync::Spinlock;
12use alloc::collections::BTreeMap;
13use types::{CellId, ViError, ViResult};
14
15/// Capability identifier — re-exported from the API crate so kernel and
16/// user-space share exactly one definition.
17pub use api::cap::CapId;
18
19/// What resource a capability refers to.
20pub enum CapResource {
21    /// An open file backed by the kernel-internal FS (`VIFS1` / ramFS).
22    ///
23    /// The `file` field is `None` while a `ReadCap` syscall is in progress —
24    /// the kernel takes the `Box` out, releases the cap-table lock, performs I/O,
25    /// then re-acquires the lock and puts the `Box` back.  This prevents the
26    /// global cap-table lock from being held across potentially-slow I/O.
27    /// A concurrent `ReadCap` on the same cap while it is parked returns
28    /// `Err(ViError::TryAgain)`.
29    File {
30        file: Option<alloc::boxed::Box<dyn api::fs::ViFile + Send + Sync>>,
31    },
32}
33
34/// One capability entry in the table.
35pub struct CapEntry {
36    /// Cell that currently owns this capability.
37    pub owner: CellId,
38    /// The resource this capability refers to.
39    pub resource: CapResource,
40    /// Permissions this cap grants (`api::cap::CapPerms` bits).
41    pub perms: u32,
42    /// Optional lease expiry in kernel monotonic ticks.
43    ///
44    /// `None` = permanent (revoked only by explicit `close` or cell exit).
45    /// `Some(t)` = auto-revoked on the first `verify` call after tick `t`.
46    pub expires_at: Option<u64>,
47}
48
49/// The global capability table.
50pub struct CapTable {
51    entries: BTreeMap<CapId, CapEntry>,
52    next_id: u64,
53}
54
55impl Default for CapTable {
56    fn default() -> Self {
57        Self::new()
58    }
59}
60
61impl CapTable {
62    pub const fn new() -> Self {
63        Self {
64            entries: BTreeMap::new(),
65            next_id: 1, // 0 is the null sentinel — never assigned
66        }
67    }
68
69    /// Allocate a new capability for `owner` over `resource`.
70    ///
71    /// Returns the new `CapId` (always > 0 for the first 2^64-1 calls).
72    /// Panics in debug builds if `next_id` wraps to 0 (effectively unreachable).
73    pub fn alloc(&mut self, owner: CellId, resource: CapResource, perms: u32) -> CapId {
74        let id = CapId(self.next_id);
75        let next = self.next_id.wrapping_add(1);
76        debug_assert!(
77            next != 0,
78            "CapId counter wrapped to 0 — unreachable in practice"
79        );
80        self.next_id = if next == 0 { 1 } else { next };
81        self.entries.insert(
82            id,
83            CapEntry {
84                owner,
85                resource,
86                perms,
87                expires_at: None,
88            },
89        );
90        id
91    }
92
93    /// Allocate a capability with an automatic lease expiry.
94    ///
95    /// `expires_at` is the absolute kernel tick at which the cap becomes invalid.
96    /// Use `crate::task::system_ticks() + duration` to compute this.
97    pub fn alloc_with_lease(
98        &mut self,
99        owner: CellId,
100        resource: CapResource,
101        perms: u32,
102        expires_at: u64,
103    ) -> CapId {
104        let id = self.alloc(owner, resource, perms);
105        if let Some(e) = self.entries.get_mut(&id) {
106            e.expires_at = Some(expires_at);
107        }
108        id
109    }
110
111    /// Verify that `caller` owns `cap_id` and the lease has not expired.
112    ///
113    /// Lease expiry is checked lazily — the entry is revoked and
114    /// `ViError::PermissionDenied` returned if `expires_at` has passed.
115    ///
116    /// # Errors
117    /// - `ViError::PermissionDenied` — cap owned by a different cell or lease expired.
118    /// - `ViError::NotFound` — cap does not exist.
119    pub fn verify(&mut self, cap_id: CapId, caller: CellId) -> ViResult<()> {
120        let now = crate::task::system_ticks() as u64;
121        match self.entries.get(&cap_id) {
122            Some(e) if e.owner == caller => {
123                // Lazy lease check.
124                if let Some(exp) = e.expires_at {
125                    if now >= exp {
126                        // Expired — revoke and report.
127                        self.entries.remove(&cap_id);
128                        return Err(ViError::PermissionDenied);
129                    }
130                }
131                Ok(())
132            }
133            Some(_) => Err(ViError::PermissionDenied),
134            None => Err(ViError::NotFound),
135        }
136    }
137
138    /// Revoke a capability.  Idempotent — no-op if the cap is not in the table.
139    pub fn revoke(&mut self, cap_id: CapId) {
140        self.entries.remove(&cap_id);
141    }
142
143    /// Revoke all capabilities owned by `owner`.
144    ///
145    /// Called when a cell exits to prevent orphaned caps.
146    pub fn revoke_all_for(&mut self, owner: CellId) {
147        self.entries.retain(|_, e| e.owner != owner);
148    }
149
150    /// Return an immutable reference to an entry if owned by `caller`.
151    pub fn get_if_owner(&self, cap_id: CapId, caller: CellId) -> Option<&CapEntry> {
152        self.entries.get(&cap_id).filter(|e| e.owner == caller)
153    }
154
155    /// Return a mutable reference to an entry (does NOT verify ownership —
156    /// callers must call `verify` first).
157    pub fn get_mut_unchecked(&mut self, cap_id: CapId) -> Option<&mut CapEntry> {
158        self.entries.get_mut(&cap_id)
159    }
160
161    /// Take the `Box<dyn ViFile>` out of a `CapResource::File` entry so that
162    /// I/O can be performed with the cap-table lock released.
163    ///
164    /// Returns `Err` if the cap is not found, already parked (concurrent read),
165    /// or is not a `File` cap.
166    pub fn park_file(
167        &mut self,
168        cap_id: CapId,
169        owner: types::CellId,
170    ) -> ViResult<alloc::boxed::Box<dyn api::fs::ViFile + Send + Sync>> {
171        self.verify(cap_id, owner)?;
172        let entry = self.entries.get_mut(&cap_id).ok_or(ViError::NotFound)?;
173        // Single variant today; match for future exhaustiveness when more cap types land.
174        let CapResource::File { ref mut file } = entry.resource;
175        // None means a concurrent read is in progress (single-core: shouldn't happen).
176        file.take().ok_or(ViError::InvalidInput)
177    }
178
179    /// Return a previously parked `Box<dyn ViFile>` back into the cap entry.
180    ///
181    /// No-ops if the cap was revoked while the file was parked.
182    pub fn unpark_file(
183        &mut self,
184        cap_id: CapId,
185        boxed_file: alloc::boxed::Box<dyn api::fs::ViFile + Send + Sync>,
186    ) {
187        if let Some(CapEntry {
188            resource: CapResource::File { ref mut file },
189            ..
190        }) = self.entries.get_mut(&cap_id)
191        {
192            *file = Some(boxed_file);
193        }
194        // If the cap was revoked while parked, just drop `boxed_file` here.
195    }
196}

198/// Global kernel capability table.
199pub static CAP_TABLE: Spinlock<CapTable> = Spinlock::new(CapTable::new());
