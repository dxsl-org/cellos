1//! Futex wait queues — wait-on-address for cell threads (ADR-0018 §2.1).
2//!
3//! A futex wait parks the calling task until another task wakes the same *key*.
4//! The key is `(address-space identity, generation, word address)`, which is what
5//! makes the primitive correct in both execution tiers:
6//!
7//! * Tier 1 cells share one address space, so the identity is the shared root and
8//!   the address alone names the word — two cells that can both write a word can
9//!   also wake each other on it, which is the standard shared-memory contract.
10//! * Tier 2 cells have private page tables, so the same virtual address means
11//!   different words in different domains. The identity and generation in the key
12//!   keep a peer domain from waking (or being woken by) a wait it does not share.
13//!
14//! Correctness of the compare-and-park is the whole point of this module:
15//!
16//! 1. the caller's word is read through the validated, domain-aware copy path —
17//!    never by dereferencing a raw user pointer;
18//! 2. the *deciding* read happens while `SCHEDULER` is held, and the enqueue
19//!    happens under the same lock;
20//! 3. a waker updates the word before it calls wake and takes `SCHEDULER` to
21//!    select waiters.
22//!
23//! Together those give the standard result: a wake that raced the park is either
24//! observed by the deciding read (the waiter returns `VALUE_MISMATCH` and never
25//! sleeps) or arrives after the waiter is enqueued (and wakes it). There is no
26//! window in which a woken waiter sleeps forever.
27//!
28//! Out of scope by design: `FUTEX_REQUEUE`, priority inheritance, and robust-list
29//! semantics. A waiter that dies leaves no entry behind — the scheduler sweep and
30//! the wake path both drop entries whose task is no longer waiting.
31
32use crate::sync::Spinlock;
33use alloc::collections::{BTreeMap, VecDeque};
34use types::VAddr;
35
36/// Wait outcome: the task was woken by a `FutexWake`.
37pub(crate) const OUTCOME_WOKEN: usize = 0;
38/// Wait outcome: the word did not hold the expected value, so the caller never
39/// parked (or parked and was woken without the value having changed).
40pub(crate) const OUTCOME_VALUE_MISMATCH: usize = 1;
41/// Wait outcome: the deadline elapsed before any wake.
42pub(crate) const OUTCOME_TIMED_OUT: usize = 2;
43
44/// A wait key. `space` is the address-space identity (0 = the shared SAS root) and
45/// `generation` pins the specific domain incarnation, so a recycled domain identity
46/// cannot inherit an old domain's waiters.
47#[derive(Clone, Copy, Debug, Eq, PartialEq, Ord, PartialOrd)]
48pub struct FutexKey {
49    pub space: u64,
50    pub generation: u64,
51    pub addr: VAddr,
52}
53
54/// Key for a word in the calling task's address space.
55pub(crate) fn key_for(addr: VAddr) -> FutexKey {
56    let (space, generation) = super::hart_local::current_domain();
57    FutexKey {
58        space,
59        generation,
60        addr,
61    }
62}
63
64/// Waiter queues, one entry per key. A leaf lock: taken while `SCHEDULER` is held
65/// (the same order as the ready queues), never the reverse.
66///
67/// The queue is bounded by the per-cell thread cap (`MAX_THREADS_PER_CELL`): only a
68/// task that is parked can be enqueued, and a cell cannot have more parked tasks
69/// than it has threads. It never grows without bound.
70static WAITERS: Spinlock<BTreeMap<FutexKey, VecDeque<usize>>> = Spinlock::new(BTreeMap::new());
71
72fn enqueue(key: FutexKey, tid: usize) {
73    let mut waiters = WAITERS.lock();
74    let queue = waiters.entry(key).or_default();
75    if !queue.contains(&tid) {
76        queue.push_back(tid);
77    }
78}
79
80/// Drop one waiter. Called on timeout and on any exit from the wait state.
81pub(crate) fn remove_waiter(key: FutexKey, tid: usize) {
82    let mut waiters = WAITERS.lock();
83    if let Some(queue) = waiters.get_mut(&key) {
84        queue.retain(|queued| *queued != tid);
85        if queue.is_empty() {
86            waiters.remove(&key);
87        }
88    }
89}
90
91/// Take up to `count` waiters off a key's queue (`count == 0` takes every waiter).
92fn take_waiters(key: FutexKey, count: usize, out: &mut VecDeque<usize>) {
93    let mut waiters = WAITERS.lock();
94    if let Some(queue) = waiters.get_mut(&key) {
95        while count == 0 || out.len() < count {
96            match queue.pop_front() {
97                Some(tid) => out.push_back(tid),
98                None => break,
99            }
100        }
101        if queue.is_empty() {
102            waiters.remove(&key);
103        }
104    }
105}

107/// Read the caller's futex word through the validated, domain-aware copy path.
108///
109/// Allocation-free: a four-byte stack buffer and one copy through the caller's
110/// view. Null, kernel, unmapped, and peer-domain addresses fail here and become the
111/// ABI's recoverable error — the kernel never dereferences the raw pointer.
112pub(crate) fn read_word(
113    view: &super::copy_glue::TaskCopyView,
114    addr: VAddr,
115) -> Result<u32, super::syscall::SyscallError> {
116    if addr == 0 || !addr.is_multiple_of(core::mem::align_of::<u32>()) {
117        return Err(super::syscall::SyscallError::InvalidInput);
118    }
119    let mut bytes = [0u8; 4];
120    view.read_into(addr, &mut bytes)
121        .map_err(|_| super::syscall::SyscallError::InvalidInput)?;
122    Ok(u32::from_ne_bytes(bytes))
123}

125/// Park the caller until woken, the word changes, or the deadline elapses.
126///
127/// `timeout_ticks == 0` blocks indefinitely (the `WaitEvent` convention). Returns
128/// the wait outcome, or `Err` when the caller has no live task record.
129pub(crate) fn wait(
130    caller_id: usize,
131    addr: VAddr,
132    expected: u32,
133    timeout_ticks: u64,
134) -> Result<usize, super::syscall::SyscallError> {
135    let key = key_for(addr);
136
137    // The copy view is taken *before* `SCHEDULER`: building it locks the scheduler
138    // (it snapshots the caller's task), so taking it under the lock would
139    // self-deadlock. The view is a plain descriptor; the copy itself is lock-free
140    // and validated per call, so using it under the lock is safe.
141    let view = super::syscall::caller_copy_view_for(caller_id)?;
142
143    // First check, outside the lock: the common case is "the value already moved",
144    // and it needs no queue traffic at all.
145    if read_word(&view, addr)? != expected {
146        return Ok(OUTCOME_VALUE_MISMATCH);
147    }
148
149    let deadline = if timeout_ticks == 0 {
150        None
151    } else {
152        Some(super::system_ticks() as u64 + timeout_ticks)
153    };
154
155    // Decide and park under one lock: the word is re-read here, so a wake that
156    // already ran (it must take this same lock to select waiters) is observed
157    // instead of being lost.
158    let mut scheduler = super::SCHEDULER.lock();
159    let sched = scheduler
160        .as_mut()
161        .ok_or(super::syscall::SyscallError::Unknown)?;
162    if read_word(&view, addr)? != expected {
163        return Ok(OUTCOME_VALUE_MISMATCH);
164    }
165    match sched.tasks.get_mut(&caller_id) {
166        Some(task) => {
167            enqueue(key, caller_id);
168            task.state = super::tcb::TaskState::FutexWait { key, deadline };
169            // A waker writes the outcome into this slot before pushing the task to
170            // a run queue; the handler reads it back after the park.
171            task.trap_frame.regs[10] = OUTCOME_WOKEN as _;
172        }
173        None => return Err(super::syscall::SyscallError::PermissionDenied),
174    }
175    drop(scheduler);
176
177    super::yield_cpu();
178
179    // Resumed: the waker or the deadline sweep published the outcome.
180    let outcome = super::SCHEDULER
181        .lock()
182        .as_ref()
183        .and_then(|sched| sched.tasks.get(&caller_id))
184        .map(|task| task.trap_frame.regs[10])
185        .unwrap_or(OUTCOME_WOKEN);
186    // The task left the wait state; make sure no stale queue entry survives.
187    remove_waiter(key, caller_id);
188    Ok(outcome)
189}

191/// Wake up to `count` waiters on the caller's key. Returns how many were woken.
192pub(crate) fn wake(
193    caller_id: usize,
194    addr: VAddr,
195    count: usize,
196) -> Result<usize, super::syscall::SyscallError> {
197    let key = key_for(addr);
198    let mut scheduler = super::SCHEDULER.lock();
199    let sched = scheduler
200        .as_mut()
201        .ok_or(super::syscall::SyscallError::Unknown)?;
202    if !sched.tasks.contains_key(&caller_id) {
203        return Err(super::syscall::SyscallError::PermissionDenied);
204    }
205
206    let mut candidates = VecDeque::new();
207    take_waiters(key, count, &mut candidates);
208
209    let mut woken = 0;
210    while let Some(tid) = candidates.pop_front() {
211        if let Some(task) = sched.tasks.get_mut(&tid) {
212            // Only a task still parked on this key may be woken: a timeout or an
213            // exit that raced this wake already removed it from the wait state.
214            if matches!(task.state, super::tcb::TaskState::FutexWait { key: k, .. } if k == key) {
215                task.state = super::tcb::TaskState::Ready;
216                task.trap_frame.regs[10] = OUTCOME_WOKEN as _;
217                sched.push_ready(tid);
218                woken += 1;
219            }
220        }
221    }
222    Ok(woken)
223}

225/// Boot-time assertion for the wait key's discriminating power.
226///
227/// The runtime cross-domain case (two Tier 2 cells waiting on the same virtual
228/// address) needs two cooperating cells; this proves the same property at the
229/// queue: a waiter enqueued under one address space is invisible to a wake from
230/// another, and to a later generation of the same space.
231#[cfg(feature = "test-hooks")]
232pub(crate) fn run_selftest() {
233    let addr = 0x1000usize;
234    let owner = FutexKey {
235        space: 1,
236        generation: 1,
237        addr,
238    };
239    let peer_space = FutexKey {
240        space: 2,
241        generation: 1,
242        addr,
243    };
244    let later_generation = FutexKey {
245        space: 1,
246        generation: 2,
247        addr,
248    };
249
250    const TID: usize = 7;
251    enqueue(owner, TID);
252    enqueue(owner, TID + 1);
253
254    let mut out = VecDeque::new();
255    take_waiters(peer_space, 8, &mut out);
256    let cross_space_invisible = out.is_empty();
257
258    out.clear();
259    take_waiters(later_generation, 8, &mut out);
260    let cross_generation_invisible = out.is_empty();
261
262    out.clear();
263    take_waiters(owner, 0, &mut out);
264    let owner_sees_broadcast = out.len() == 2 && out[0] == TID && out[1] == TID + 1;
265
266    // Leave no residue for the rest of the boot.
267    out.clear();
268    remove_waiter(owner, TID);
269    remove_waiter(owner, TID + 1);
270
271    if cross_space_invisible && cross_generation_invisible && owner_sees_broadcast {
272        log::info!("S22-RV64-FUTEX-KEY: PASS");
273    } else {
274        log::error!(
275            "S22-RV64-FUTEX-KEY: FAIL space={} generation={} broadcast={}",
276            cross_space_invisible,
277            cross_generation_invisible,
278            owner_sees_broadcast
279        );
280    }
281}

283/// Deadline sweep arm: a parked waiter whose deadline elapsed.
284///
285/// Called from the scheduler's global sweep, which already holds `SCHEDULER`.
286pub(crate) fn on_deadline(task: &mut super::tcb::Task, tid: usize, key: FutexKey) {
287    remove_waiter(key, tid);
288    task.trap_frame.regs[10] = OUTCOME_TIMED_OUT as _;
289}

291/// Drop a waiter whose task is leaving the wait state for any other reason
292/// (exit, forced exit, fault).
293pub(crate) fn on_task_leaves_wait(tid: usize) {
294    let mut waiters = WAITERS.lock();
295    let keys: alloc::vec::Vec<FutexKey> = waiters
296        .iter()
297        .filter(|(_, queue)| queue.contains(&tid))
298        .map(|(key, _)| *key)
299        .collect();
300    for key in keys {
301        if let Some(queue) = waiters.get_mut(&key) {
302            queue.retain(|queued| *queued != tid);
303            if queue.is_empty() {
304                waiters.remove(&key);
305            }
306        }
307    }
