1//! Kernel side of `ResolveCellImage` (syscall 441, `abi::cell_image`).
2//!
3//! The identity lives on the Cell's root task (`Task::bind_image`), bound from
4//! the admission measurement before the task is published and never rewritten.
5//! It is reached only through the live owner slot, so the answer is always for
6//! the exact `(cell_id, generation)` asked about: a respawn under the same
7//! `CellId` is a different generation with its own root and its own image, and
8//! a dead or stale generation has no answer at all.
9
10use abi::cell_image::CellImage;
11use abi::cell_owner::CellOwnerRequest;
12use types::CellId;
13
14impl crate::sched::scheduler::Scheduler {
15    /// Image identity of the live `(cell_id, generation)`. `None` when that
16    /// generation is not live (stale, retired, never existed) or its root was
17    /// not admitted through a measured image.
18    pub(crate) fn resolve_cell_image(&self, cell_id: CellId, generation: u64) -> Option<CellImage> {
19        let owner = self.resolve_live_cell_owner(cell_id, generation)?;
20        let root = self.tasks.get(&(owner.root_tid as usize))?;
21        Some(CellImage {
22            cell_id: owner.cell_id,
23            generation: owner.generation,
24            identity: root.image()?,
25        })
26    }
27}
28
29/// Dispatch rule for 441: answered for a task of the registered VFS Cell, or
30/// for a `SupervisorCap` holder (architecture remediation Phase 06, CP1 option
31/// A: `/bin/supervisor` authorises `OP_POWER`/`OP_HOTSWAP` by the sender's signed
32/// program).
33///
34/// Not tied to the principal VFS is currently receiving from (unlike 247):
35/// VFS re-authorizes deferred work (handle reads, prepared directories) after
36/// its receive context has moved on, and an image identity grants nothing by
37/// itself. Every other Cell is refused, so it cannot learn which program
38/// another Cell runs.
39///
40/// The supervisor gets an answer only for a Cell whose name is its launch
41/// identity (`Task::launch_named`): a signed `shutdown` copied under `/home`, or
42/// bytes another caller supplied, is the same program but not the reviewed
43/// install, and gets `None` (refused) — the rule the kernel's own 502 gate uses.
44/// VFS keeps the unfiltered record (`/srv/sshd` is not a `/bin` install).
45pub(super) fn resolve_for_caller(
46    sched: &crate::sched::scheduler::Scheduler,
47    caller_id: usize,
48    request: CellOwnerRequest,
49    retains_outer_caller: impl Fn(u64, u64) -> bool,
50) -> Option<CellImage> {
51    let caller = sched.tasks.get(&caller_id)?;
52    if retains_outer_caller(caller.cell_id.0, caller.cell_generation) {
53        return sched.resolve_cell_image(CellId(request.cell_id), request.generation);
54    }
55    if !sched.cell_rights(caller_id).supervisor {
56        return None;
57    }
58    let owner = sched.resolve_live_cell_owner(CellId(request.cell_id), request.generation)?;
59    let root = sched.tasks.get(&(owner.root_tid as usize))?;
60    if !root.launch_named {
61        return None;
62    }
63    sched.resolve_cell_image(CellId(request.cell_id), request.generation)
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use crate::sched::scheduler::Scheduler;
70    use crate::task::Task;
71    use abi::cell_image::{ImageIdentity, ANCHOR_RELEASE};
72    use abi::cell_owner::CellOwner;
73    use alloc::boxed::Box;
74
75    const ROOT: usize = 60_441;
76    const CELL: u64 = 62;
77    const VFS_TID: usize = 60_442;
78    const VFS_CELL: u64 = 63;
79
80    fn sshd() -> ImageIdentity {
81        ImageIdentity::signed([7; 32], ANCHOR_RELEASE, b"sshd").expect("valid identity")
82    }
83
84    /// One published root with `image` bound, plus a VFS task. Returns the
85    /// root's generation.
86    fn fixture(sched: &mut Scheduler, image: Option<ImageIdentity>) -> u64 {
87        let mut root = Box::new(Task::new(ROOT, CellId(CELL), "sshd", alloc::vec![]));
88        if let Some(image) = image {
89            assert!(root.bind_image(image));
90        }
91        let generation = root.cell_generation;
92        sched.tasks.insert(ROOT, root);
93        assert!(sched.publish_live_cell_owner(CellOwner::new(CELL, generation, ROOT as u64)));
94        let vfs = Box::new(Task::new(VFS_TID, CellId(VFS_CELL), "vfs", alloc::vec![]));
95        sched.tasks.insert(VFS_TID, vfs);
96        generation
97    }
98
99    #[test]
100    fn bound_identity_is_immutable() {
101        let _guard = crate::TEST_STATE_LOCK.lock();
102        let mut task = Task::new(ROOT, CellId(CELL), "sshd", alloc::vec![]);
103        assert_eq!(task.image(), None);
104        assert!(task.bind_image(sshd()));
105        assert!(!task.bind_image(ImageIdentity::unsigned([9; 32])));
106        assert_eq!(task.image(), Some(sshd()));
107    }
108
109    #[test]
110    fn exact_live_generation_resolves_and_stale_ones_fail() {
111        let _guard = crate::TEST_STATE_LOCK.lock();
112        let mut sched = Scheduler::new();
113        let generation = fixture(&mut sched, Some(sshd()));
114        let image = sched
115            .resolve_cell_image(CellId(CELL), generation)
116            .expect("live generation resolves");
117        assert_eq!((image.cell_id, image.generation), (CELL, generation));
118        assert_eq!(image.identity, sshd());
119        for stale in [0, generation - 1, generation + 1] {
120            assert!(sched.resolve_cell_image(CellId(CELL), stale).is_none());
121        }
122        assert!(sched
123            .resolve_cell_image(CellId(CELL + 1), generation)
124            .is_none());
125        // Once the root is gone the incarnation is over: no answer survives it.
126        sched.tasks.remove(&ROOT);
127        assert!(sched.resolve_cell_image(CellId(CELL), generation).is_none());
128    }
129
130    #[test]
131    fn unmeasured_root_has_no_identity() {
132        let _guard = crate::TEST_STATE_LOCK.lock();
133        let mut sched = Scheduler::new();
134        let generation = fixture(&mut sched, None);
135        assert!(sched.resolve_cell_image(CellId(CELL), generation).is_none());
136    }
137
138    #[test]
139    fn only_the_registered_vfs_cell_is_answered() {
140        let _guard = crate::TEST_STATE_LOCK.lock();
141        let mut sched = Scheduler::new();
142        let generation = fixture(&mut sched, Some(sshd()));
143        let request = CellOwnerRequest::new(CELL, generation);
144        let is_vfs = |cell: u64, _generation: u64| cell == VFS_CELL;
145        assert!(resolve_for_caller(&sched, VFS_TID, request, is_vfs).is_some());
146        // The subject itself, an unknown task, and VFS asking about a stale
147        // generation all get nothing.
148        assert!(resolve_for_caller(&sched, ROOT, request, is_vfs).is_none());
149        assert!(resolve_for_caller(&sched, 1, request, is_vfs).is_none());
150        let stale = CellOwnerRequest::new(CELL, generation + 1);
151        assert!(resolve_for_caller(&sched, VFS_TID, stale, is_vfs).is_none());
152    }
153
154    /// CP1 option A: a `SupervisorCap` holder is answered too, but only for a
155    /// Cell whose signed name is its launch identity (`Task::launch_named`).
156    #[test]
157    fn supervisor_is_answered_only_for_named_launches() {
158        const SUPERVISOR_TID: usize = 60_443;
159        let _guard = crate::TEST_STATE_LOCK.lock();
160        let mut sched = Scheduler::new();
161        let generation = fixture(&mut sched, Some(sshd()));
162        let mut supervisor = Box::new(Task::new(
163            SUPERVISOR_TID,
164            CellId(64),
165            "supervisor",
166            alloc::vec![],
167        ));
168        crate::task::cap::CapSet {
169            supervisor: true,
170            ..crate::task::cap::CapSet::EMPTY
171        }
172        .apply_to(&mut supervisor);
173        sched.tasks.insert(SUPERVISOR_TID, supervisor);
174        let request = CellOwnerRequest::new(CELL, generation);
175        let no_vfs = |_: u64, _: u64| false;
176
177        // Same bytes, but not a reviewed install (sandbox copy, caller ELF).
178        assert!(resolve_for_caller(&sched, SUPERVISOR_TID, request, no_vfs).is_none());
179        sched.tasks.get_mut(&ROOT).expect("root").launch_named = true;
180        let image = resolve_for_caller(&sched, SUPERVISOR_TID, request, no_vfs)
181            .expect("supervisor answered for a named launch");
182        assert_eq!(image.identity, sshd());
183        // A task without SupervisorCap still learns nothing.
184        assert!(resolve_for_caller(&sched, VFS_TID, request, no_vfs).is_none());
185        let stale = CellOwnerRequest::new(CELL, generation + 1);
186        assert!(resolve_for_caller(&sched, SUPERVISOR_TID, stale, no_vfs).is_none());
187    }
