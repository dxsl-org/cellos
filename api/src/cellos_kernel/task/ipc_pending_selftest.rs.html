1//! Boot self-test for receiver-owned IPC delivery.
2//!
3//! Runs after scheduler initialization and before secondary harts start. Each
4//! producer receives a synthetic `Recv` task whose buffer pointer is deliberately
5//! invalid: success proves the producer queued owned bytes without touching the
6//! foreign buffer. Teardown removes every synthetic task and ready-queue entry.
7

10use super::tcb::{Task, TaskState, HOTSWAP_MSG_QUEUE_DEPTH, INPUT_EVENT_QUEUE_DEPTH};
11use alloc::vec::Vec;
12use core::sync::atomic::Ordering;
13use types::CellId;

15const SENDER: usize = 9321;
16const RECEIVER: usize = 9322;
17const TEST_CELL: u64 = (crate::memory::cell_quota::MAX_CELLS - 3) as u64;
18const QUOTA_CELL: u64 = (crate::memory::cell_quota::MAX_CELLS - 1) as u64;
19const INVALID_RECV_PTR: usize = usize::MAX - 64;

21fn fail(reason: &str) -> bool {
22    log::error!("[selftest] IPC-PENDING: FAIL — {}", reason);
23    false
24}

26fn prepare_receiver(mask: usize) {
27    insert(SENDER, TEST_CELL);
28    insert(RECEIVER, TEST_CELL + 1);
29    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
30        if let Some(task) = sched.tasks.get_mut(&RECEIVER) {
31            task.state = TaskState::Blocked {
32                wait: WaitKind::Recv {
33                    mask,
34                    buf_ptr: INVALID_RECV_PTR,
35                    buf_len: 16,
36                },
37                deadline: None,
38            };
39        }
40    }
41}

43fn prepare_completion_wait_receiver() {
44    insert(SENDER, TEST_CELL);
45    insert(RECEIVER, TEST_CELL + 1);
46    if let Some(task) = super::SCHEDULER
47        .lock()
48        .as_mut()
49        .and_then(|sched| sched.tasks.get_mut(&RECEIVER))

52            wait: WaitKind::Completion {
53                source: api::completion::source::NET_RX,
54                delivery: Delivery::Queue,
55            },
56            deadline: None,
57        };
58    }

61fn verify_delivery(expected: &[u8]) -> bool {
62    let guard = super::SCHEDULER.lock();
63    let Some(task) = guard.as_ref().and_then(|sched| sched.tasks.get(&RECEIVER)) else {
64        return fail("synthetic receiver disappeared");
65    };
66    if task.state != TaskState::Ready {
67        return fail("delivery did not wake the receiver");
68    }
69    match task.pending_msgs.as_slice() {
70        [msg] if msg.sender_tid == SENDER && msg.payload() == expected => true,
71        _ => fail("delivery did not leave exactly one owned mailbox message"),
72    }
73}

75fn reset() {
76    remove(SENDER);
77    remove(RECEIVER);

80fn all_producers_defer_foreign_writes() -> bool {
81    // RecvScatter leaves the same WaitKind::Recv shape behind, but its temporary
82    // buffer is already gone. INVALID_RECV_PTR models that stale pointer: exercising
83    // every producer here proves none of them dereferences either Recv variant's
84    // retained destination outside the receiver's own syscall context.
85    let payload = b"owned-ipc";
86    let mut ok = true;
87
88    prepare_receiver(SENDER);
89    if super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()) != Ok(0)
90        || !verify_delivery(payload)
91    {
92        ok = fail("ipc_send did not defer matched Recv delivery");
93    }
94    reset();
95
96    prepare_receiver(usize::MAX);
97    if super::ipc_post_nonblock(SENDER, RECEIVER, payload).is_err() || !verify_delivery(payload) {
98        ok = fail("ipc_post_nonblock did not defer matched Recv delivery");
99    }
100    reset();
101
102    prepare_receiver(SENDER);
103    if super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_err()
104        || !verify_delivery(payload)
105    {
106        ok = fail("ipc_try_send did not defer matched Recv delivery");
107    }
108    reset();
109    ok
110}

112/// A target already blocked sending to this caller must not park the caller too.
113///
114/// Two single-threaded cells that send to each other deadlock if the second send
115/// blocks: neither can receive the message that would complete the other's send.
116/// The board hit exactly this (Net Cell in `Sending { target: 4 }`, NIC driver in
117/// `Sending { target: 5 }`) and the liveness heartbeat then killed the Net Cell
118/// every second, so the kernel queues instead and lets the caller return to
119/// receive.
120fn send_to_a_target_blocked_sending_to_us_queues() -> bool {
121    let payload = b"cycle-break";
122    insert(SENDER, TEST_CELL);
123    insert(RECEIVER, TEST_CELL + 1);
124    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
125        if let Some(task) = sched.tasks.get_mut(&RECEIVER) {
126            task.state = TaskState::Blocked {
127                wait: WaitKind::Send {
128                    target: SENDER,
129                    delivery_id: 1,
130                },
131                deadline: None,
132            };
133        }
134    }
135    let result = super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len());
136    let ok = if result != Ok(0) {
137        fail("ipc_send parked behind a target blocked sending to the caller")
138    } else {
139        let guard = super::SCHEDULER.lock();
140        match guard.as_ref().and_then(|sched| sched.tasks.get(&RECEIVER)) {
141            Some(task)
142                if matches!(task.state, TaskState::Blocked { wait: WaitKind::Send { target, .. }, .. } if target == SENDER)
143                    && matches!(
144                        task.pending_msgs.as_slice(),
145                        [msg] if msg.sender_tid == SENDER && msg.payload() == payload
146                    ) =>
147            {
148                true
149            }
150            _ => fail("cycle break did not leave the message queued for the target"),
151        }
152    };
153    reset();
154    ok
155}

157/// `ipc_send_kernel` backs every blocking send. Unlike `ipc_post_nonblock`, it
158/// must not wake a VFS task parked in a nested receive for a different sender.
159fn kernel_send_respects_receiver_mask() -> bool {
160    let payload = b"kernel-mask";
161    prepare_receiver(usize::MAX);
162    let sent = super::ipc_send_kernel(SENDER, RECEIVER, payload) == Ok(1);
163    let preserved = {
164        let guard = super::SCHEDULER.lock();
165        let Some(sched) = guard.as_ref() else {
166            reset();
167            return fail("scheduler missing after kernel send");
168        };
169        let receiver = sched.tasks.get(&RECEIVER);
170        let sender = sched.tasks.get(&SENDER);
171        matches!(
172            receiver.map(|task| &task.state),
173            Some(TaskState::Blocked { wait: WaitKind::Recv { mask, .. }, .. }) if *mask == usize::MAX
174        ) && receiver.is_some_and(|task| {
175            matches!(task.pending_msgs.as_slice(), [msg] if msg.sender_tid == SENDER && msg.payload() == payload)
176        }) && matches!(
177            sender.map(|task| &task.state),
178            Some(TaskState::Blocked { wait: WaitKind::Send { target, .. }, .. }) if *target == RECEIVER
179        )
180    };
181    reset();
182    sent && preserved || fail("kernel send ignored a receiver mask")
183}

185fn verify_completion_wait_wake(expected: &[u8], ready_before: usize, sender_blocks: bool) -> bool {
186    let guard = super::SCHEDULER.lock();
187    let Some(sched) = guard.as_ref() else {
188        return fail("scheduler missing after completion-wait IPC wake");
189    };
190    let Some(receiver) = sched.tasks.get(&RECEIVER) else {
191        return fail("completion-wait receiver disappeared");
192    };
193    let sender_generation = sched
194        .tasks
195        .get(&SENDER)
196        .map(|task| task.cell_generation)
197        .unwrap_or(0);
198    let receiver_ok = receiver.state == TaskState::Ready
199        && matches!(
200            receiver.pending_msgs.as_slice(),
201            [msg]
202                if msg.sender_tid == SENDER
203                    && msg.payload() == expected
204                    && msg.wire_header().is_some_and(|header| {
205                        header.sender_tid == SENDER
206                            && header.sender_cell_id == TEST_CELL
207                            && header.sender_generation == sender_generation
208                            && header.delivery_id != 0
209                    })
210        );
211    let sender_ok = sched.tasks.get(&SENDER).is_some_and(|sender| {
212        if sender_blocks {
213            matches!(
214                sender.state,
215                TaskState::Blocked { wait: WaitKind::Send { target, .. }, .. } if target == RECEIVER
216            )
217        } else {
218            sender.state == TaskState::Ready
219        }
220    });
221    receiver_ok
222        && sender_ok
223        && crate::sched::hart_local::ready::total_ready_count() == ready_before + 1
224}

226/// The state publication is the handoff boundary: a producer arriving before
227/// `yield_cpu` or after the switch observes the same state and must perform the
228/// same single ready transition.
229fn all_producers_wake_net_rx_completion_wait() -> bool {
230    let payload = b"completion-ipc";
231    let mut ok = true;
232
233    prepare_completion_wait_receiver();
234    let ready_before = crate::sched::hart_local::ready::total_ready_count();
235    if super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()) != Ok(1)
236        || !verify_completion_wait_wake(payload, ready_before, true)
237    {
238        ok = fail("ipc_send did not wake NET_RX wait while preserving blocking send");
239    }
240    reset();
241
242    prepare_completion_wait_receiver();
243    let ready_before = crate::sched::hart_local::ready::total_ready_count();
244    if super::ipc_post_nonblock(SENDER, RECEIVER, payload).is_err()
245        || !verify_completion_wait_wake(payload, ready_before, false)
246    {
247        ok = fail("ipc_post_nonblock did not wake NET_RX completion wait");
248    }
249    reset();
250
251    prepare_completion_wait_receiver();
252    let ready_before = crate::sched::hart_local::ready::total_ready_count();
253    let saved_input = crate::dev::driver_cell::INPUT_CELL_TID.load(Ordering::Acquire);
254    crate::dev::driver_cell::INPUT_CELL_TID.store(SENDER, Ordering::Release);
255    let sent =
256        super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_ok();
257    crate::dev::driver_cell::INPUT_CELL_TID.store(saved_input, Ordering::Release);
258    if !sent || !verify_completion_wait_wake(payload, ready_before, false) {
259        ok = fail("trusted ipc_try_send did not wake NET_RX completion wait");
260    }
261    reset();
262
263    prepare_completion_wait_receiver();
264    let ready_before = crate::sched::hart_local::ready::total_ready_count();
265    let saved_input = crate::dev::driver_cell::INPUT_CELL_TID.load(Ordering::Acquire);
266    crate::dev::driver_cell::INPUT_CELL_TID.store(0, Ordering::Release);
267    let rejected =
268        super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_err();
269    crate::dev::driver_cell::INPUT_CELL_TID.store(saved_input, Ordering::Release);
270    let unchanged = {
271        let guard = super::SCHEDULER.lock();
272        guard
273            .as_ref()
274            .and_then(|sched| sched.tasks.get(&RECEIVER))
275            .is_some_and(|task| {
276                matches!(
277                    task.state,
278                    TaskState::Blocked {
279                        wait: WaitKind::Completion {
280                            source: api::completion::source::NET_RX,
281                            delivery: Delivery::Queue,
282                            ..
283                        },
284                        ..
285                    }
286                ) && task.pending_msgs.is_empty()
287            })
288    };
289    if !rejected
290        || !unchanged
291        || crate::sched::hart_local::ready::total_ready_count() != ready_before
292    {
293        ok = fail("untrusted ipc_try_send broadened completion-wait admission");
294    }
295    reset();
296    ok
297}

299fn ready_receiver_is_not_enqueued_twice() -> bool {
300    prepare_completion_wait_receiver();
301    let ready_before = crate::sched::hart_local::ready::total_ready_count();
302    let first = super::ipc_post_nonblock(SENDER, RECEIVER, b"first");
303    let second = super::ipc_post_nonblock(SENDER, RECEIVER, b"second");
304    let unchanged_ready_depth =
305        crate::sched::hart_local::ready::total_ready_count() == ready_before + 1;
306    let messages_preserved = {
307        let guard = super::SCHEDULER.lock();
308        guard
309            .as_ref()
310            .and_then(|sched| sched.tasks.get(&RECEIVER))
311            .is_some_and(|task| {
312                matches!(
313                    task.pending_msgs.as_slice(),
314                    [first, second]
315                        if first.payload() == b"first" && second.payload() == b"second"
316                )
317            })
318    };
319    reset();
320    first.is_ok() && second.is_ok() && unchanged_ready_depth && messages_preserved
321        || fail("IPC queued a receiver that was already Ready more than once")
322}

324fn timer_completion_wait_remains_deadline_only() -> bool {
325    insert(SENDER, TEST_CELL);
326    insert(RECEIVER, TEST_CELL + 1);
327    let ready_before = crate::sched::hart_local::ready::total_ready_count();
328    let queued_before_park = super::ipc_post_nonblock(SENDER, RECEIVER, b"before-timer-park");
329    let decision = {
330        let mut guard = super::SCHEDULER.lock();
331        guard
332            .as_mut()
333            .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
334            .map(|task| {
335                super::completion_wait::publish_wait_state_locked(
336                    task,
337                    RECEIVER,
338                    api::completion::source::TIMER,
339                    Some(u64::MAX),
340                )
341            })
342    };
343    let posted_while_waiting = super::ipc_post_nonblock(SENDER, RECEIVER, b"timer-ipc");
344    let unchanged = {
345        let guard = super::SCHEDULER.lock();
346        guard
347            .as_ref()
348            .and_then(|sched| sched.tasks.get(&RECEIVER))
349            .is_some_and(|task| {
350                matches!(
351                    task.state,
352                    TaskState::Blocked {
353                        wait: WaitKind::Completion {
354                            source: api::completion::source::TIMER,
355                            delivery: Delivery::Queue,
356                            ..
357                        },
358                        ..
359                    }
360                ) && matches!(
361                    task.pending_msgs.as_slice(),
362                    [before, during]
363                        if before.payload() == b"before-timer-park"
364                            && during.payload() == b"timer-ipc"
365                )
366            })
367    };
368    let ready_unchanged = crate::sched::hart_local::ready::total_ready_count() == ready_before;
369    reset();
370    queued_before_park.is_ok()
371        && decision == Some(super::completion_wait::CompletionParkDecision::Parked)
372        && posted_while_waiting.is_ok()
373        && unchanged
374        && ready_unchanged
375        || fail("IPC interrupted a TIMER completion wait")
376}

378fn fill_completion_wait_mailbox(depth: usize) -> bool {
379    prepare_completion_wait_receiver();
380    let mut guard = super::SCHEDULER.lock();
381    let Some(task) = guard
382        .as_mut()
383        .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
384    else {
385        return false;
386    };
387    for _ in 0..depth {

389            return false;
390        }
391    }
392    true
393}

419fn full_completion_wait_mailbox_never_wakes() -> bool {
420    let payload = b"overflow";
421    let mut ok = true;
422
423    if !fill_completion_wait_mailbox(HOTSWAP_MSG_QUEUE_DEPTH) {
424        reset();
425        return fail("could not fill Send completion-wait mailbox");
426    }
427    let ready_before = crate::sched::hart_local::ready::total_ready_count();
428    let refused = super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len())
429        == Err(super::IpcSendError::Backpressure);
430    if !refused || !completion_wait_mailbox_is_unchanged(HOTSWAP_MSG_QUEUE_DEPTH, ready_before) {
431        ok = fail("full mailbox Send mutated or woke completion waiter");
432    }
433    reset();
434
435    if !fill_completion_wait_mailbox(HOTSWAP_MSG_QUEUE_DEPTH) {
436        reset();
437        return fail("could not fill post completion-wait mailbox");
438    }
439    let ready_before = crate::sched::hart_local::ready::total_ready_count();
440    let refused = super::ipc_post_nonblock(SENDER, RECEIVER, payload).is_err();
441    if !refused || !completion_wait_mailbox_is_unchanged(HOTSWAP_MSG_QUEUE_DEPTH, ready_before) {
442        ok = fail("full mailbox post mutated or woke completion waiter");
443    }
444    reset();
445
446    if !fill_completion_wait_mailbox(INPUT_EVENT_QUEUE_DEPTH) {
447        reset();
448        return fail("could not fill TrySend completion-wait mailbox");
449    }
450    let ready_before = crate::sched::hart_local::ready::total_ready_count();
451    let saved_input = crate::dev::driver_cell::INPUT_CELL_TID.load(Ordering::Acquire);
452    crate::dev::driver_cell::INPUT_CELL_TID.store(SENDER, Ordering::Release);
453    let refused =
454        super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_err();
455    crate::dev::driver_cell::INPUT_CELL_TID.store(saved_input, Ordering::Release);
456    if !refused || !completion_wait_mailbox_is_unchanged(INPUT_EVENT_QUEUE_DEPTH, ready_before) {
457        ok = fail("full mailbox TrySend mutated or woke completion waiter");
458    }
459    reset();
460    ok
461}

485    crate::sched::hart_local::ready::set_current_task_id(hart, saved_current);
486    crate::sched::hart_local::ready::begin_outgoing_context_save(hart, saved_outgoing);
487    reset();
488    decision == Some(super::completion_wait::CompletionParkDecision::Parked) && armed
489        || fail("completion-wait publication did not arm the existing SMP handoff")
490}

497fn ipc_before_park_returns_raw_zero() -> bool {
498    insert(SENDER, TEST_CELL);
499    insert(RECEIVER, TEST_CELL + 1);
500    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
501        super::completion::deliver_pending_wakes(sched);
502    }
503    let Some(completions) = queue(RECEIVER) else {
504        reset();
505        return fail("pre-park case could not reach completion queue");
506    };
507    let Some(slot) = completions.reserve() else {
508        reset();
509        return fail("pre-park case could not reserve completion slot");
510    };
511    let Some(waiter) = super::completion_wait::WaiterRegistration::new(
512        &completions,
513        RECEIVER,
514        slot,
515        api::completion::source::NET_RX,
516    ) else {
517        let _ = completions.release(slot);
518        reset();
519        return fail("pre-park case could not register waiter");
520    };
521    super::waker::arm_net_rx(completions.clone(), slot);
522
523    let ready_before = crate::sched::hart_local::ready::total_ready_count();
524    let posted = super::ipc_post_nonblock(SENDER, RECEIVER, b"pre-park");
525    let decision = {
526        let mut guard = super::SCHEDULER.lock();
527        guard
528            .as_mut()
529            .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
530            .map(|task| {
531                super::completion_wait::publish_wait_state_locked(
532                    task,
533                    RECEIVER,
534                    api::completion::source::NET_RX,
535                    None,
536                )
537            })
538    };
539    let mut untouched = [0xa5u8; api::completion::COMPLETION_LEN];
540    let outcome = super::completion_wait::finish_net_rx_wait(
541        RECEIVER,
542        untouched.as_mut_ptr() as usize,
543        &completions,
544    );
545    drop(waiter);
546
547    let task_ok = {
548        let guard = super::SCHEDULER.lock();
549        guard
550            .as_ref()
551            .and_then(|sched| sched.tasks.get(&RECEIVER))
552            .is_some_and(|task| {
553                task.state == TaskState::Ready
554                    && task.completion_wait.is_none()
555                    && matches!(
556                        task.pending_msgs.as_slice(),
557                        [msg] if msg.payload() == b"pre-park"
558                    )
559            })
560    };
561    let ok = posted.is_ok()
562        && decision == Some(super::completion_wait::CompletionParkDecision::MailboxPending)
563        && outcome == Ok(0)
564        && untouched == [0xa5u8; api::completion::COMPLETION_LEN]
565        && completions.reserved() == 0
566        && completions.drainable() == 0
567        && !super::completion::wakes_pending()
568        && crate::sched::hart_local::ready::total_ready_count() == ready_before
569        && task_ok;
570    reset();
571    ok || fail("IPC-before-park did not abort NET_RX wait cleanly with raw zero")
572}

574fn full_mailbox_refuses_without_wake() -> bool {
575    prepare_receiver(SENDER);
576    let fill_ok = {
577        let mut guard = super::SCHEDULER.lock();
578        let task = guard
579            .as_mut()
580            .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
581            .expect("synthetic receiver exists");
582        let mut ok = true;
583        for _ in 0..HOTSWAP_MSG_QUEUE_DEPTH {
584            if super::queue_pending_msg(task, SENDER, b"x", HOTSWAP_MSG_QUEUE_DEPTH).is_err() {
585                ok = false;
586                break;
587            }
588        }
589        ok
590    };
591    if !fill_ok {
592        reset();
593        return fail("mailbox refused an entry below its depth");
594    }
595
596    let refused = super::ipc_post_nonblock(SENDER, RECEIVER, b"overflow").is_err();
597    let unchanged = {
598        let guard = super::SCHEDULER.lock();
599        guard
600            .as_ref()
601            .and_then(|sched| sched.tasks.get(&RECEIVER))
602            .is_some_and(|task| {
603                matches!(
604                    task.state,
605                    TaskState::Blocked {
606                        wait: WaitKind::Recv { .. },
607                        ..
608                    }
609                ) && task.caller.current.is_none()
610                    && task.pending_msgs.len() == HOTSWAP_MSG_QUEUE_DEPTH
611            })
612    };
613    reset();
614    refused && unchanged || fail("full mailbox changed receiver state or accepted overflow")
615}

899                    wire: None,
900                    enqueued_tick: 0,

907        RECEIVER,
908        super::syscall::Syscall::TryRecv {
909            mask: SENDER,
910            buf_ptr: recv_buf.as_mut_ptr() as usize,

930/// Multi-source selection: one parked waiter sees NET_RX, an IPC terminal and
931/// its deadline exactly once each within three selections.
932///
933/// This runs on production boots, so it never touches the global NET_RX
934/// reservation or level flag: NET_RX readiness is a local value handed to the
935/// selector, exactly where the wait loop passes its own observation. The IPC
936/// terminal is a real peer-death transition on a seeded dispatched operation.
937/// The syscall validator must accept the mixed mask only with a finite
938/// deadline, and reject unknown bits next to IPC. The real reservation and
939/// level-flag settlement is covered by the host tests in `completion_wait`.
940fn multi_source_wait_reports_each_source_once() -> bool {
941    use super::completion_wait as cw;
942    const NET_RX: u32 = api::completion::source::NET_RX;
943    const TIMER: u32 = api::completion::source::TIMER;
944    const IPC: u32 = api::completion::source::IPC;
945    let mask = NET_RX | IPC | TIMER;
946
947    let syscall_rule = cw::source_is_valid(IPC, None)
948        && cw::source_is_valid(mask, Some(1))
949        && !cw::source_is_valid(mask, None)
950        && !cw::source_is_valid(IPC | (1 << 4), Some(1));
951
952    insert(SENDER, TEST_CELL);
953    insert(RECEIVER, TEST_CELL + 1);
954    // A NET_RX observation, held locally: the selector consumes it once.
955    let mut net_rx_level = true;
956
957    let mut token = None;
958    let mut woke = false;
959    let mut gate = None;
960    let mut records = [(0u32, 0isize); 4];
961    let mut releveled = false;
962    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
963        token = super::async_ipc::selftest_accept_dispatched(sched, RECEIVER, SENDER);
964        if let Some(task) = sched.tasks.get_mut(&RECEIVER) {
965            task.state = TaskState::Blocked {
966                wait: WaitKind::Completion {
967                    source: mask,
968                    delivery: Delivery::Queue,
969                },
970                deadline: None,
971            };
972        }
973        super::async_ipc::peer_died(sched, SENDER, TEST_CELL, 1);
974        if let Some(task) = sched.tasks.get_mut(&RECEIVER) {
975            woke = task.state == TaskState::Ready;
976            gate = Some(cw::publish_wait_state_locked(task, RECEIVER, mask, Some(1)));
977            for record in &mut records {
978                if let Some(chosen) = cw::select_locked(task, mask, net_rx_level, true) {
979                    *record = (chosen.source, chosen.result);
980                    if chosen.source == NET_RX {
981                        net_rx_level = false;
982                    }
983                    releveled |= chosen.relevel_net_rx;
984                }
985            }
986        }
987    }
988    // The NET_RX observation was reported first, so nothing had to be re-levelled.
989    let not_relevelled = !releveled;
990    reset();
991
992    let ipc_record = token.map(|token| (IPC, token as isize));
993    let ok = syscall_rule
994        && not_relevelled
995        && token.is_some()
996        && woke
997        && gate == Some(cw::CompletionParkDecision::IpcPending)
998        && records[0] == (NET_RX, NET_RX as isize)
999        && Some(records[1]) == ipc_record
1000        && records[2] == (TIMER, 0)
1001        && records[3] == (TIMER, 0);
1002    ok || fail("multi-source wait did not deliver NET_RX, IPC and TIMER once each")
1003}
