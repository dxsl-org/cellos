1//! Boot self-test for receiver-owned IPC delivery.
2//!
3//! Runs after scheduler initialization and before secondary harts start. Each
4//! producer receives a synthetic `Recv` task whose buffer pointer is deliberately
5//! invalid: success proves the producer queued owned bytes without touching the
6//! foreign buffer. Teardown removes every synthetic task and ready-queue entry.
7
8use super::completion_selftest::{insert, queue, remove};
9use super::tcb::{Task, TaskState, HOTSWAP_MSG_QUEUE_DEPTH, INPUT_EVENT_QUEUE_DEPTH};
10use alloc::vec::Vec;
11use core::sync::atomic::Ordering;
12use types::CellId;

14const SENDER: usize = 9321;
15const RECEIVER: usize = 9322;
16const TEST_CELL: u64 = (crate::memory::cell_quota::MAX_CELLS - 3) as u64;
17const QUOTA_CELL: u64 = (crate::memory::cell_quota::MAX_CELLS - 1) as u64;
18const INVALID_RECV_PTR: usize = usize::MAX - 64;

20fn fail(reason: &str) -> bool {
21    log::error!("[selftest] IPC-PENDING: FAIL — {}", reason);
22    false
23}

25fn prepare_receiver(mask: usize) {
26    insert(SENDER, TEST_CELL);
27    insert(RECEIVER, TEST_CELL + 1);
28    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
29        if let Some(task) = sched.tasks.get_mut(&RECEIVER) {
30            task.state = TaskState::Recv {
31                mask,
32                buf_ptr: INVALID_RECV_PTR,
33                buf_len: 16,
34                deadline: None,
35            };
36        }
37    }
38}

40fn prepare_completion_wait_receiver() {
41    insert(SENDER, TEST_CELL);
42    insert(RECEIVER, TEST_CELL + 1);
43    if let Some(task) = super::SCHEDULER
44        .lock()
45        .as_mut()
46        .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
47    {
48        task.state = TaskState::WaitCompletion {
49            source: api::completion::source::NET_RX,
50            deadline: None,
51        };
52    }
53}

55fn verify_delivery(expected: &[u8]) -> bool {
56    let guard = super::SCHEDULER.lock();
57    let Some(task) = guard.as_ref().and_then(|sched| sched.tasks.get(&RECEIVER)) else {
58        return fail("synthetic receiver disappeared");
59    };
60    if task.state != TaskState::Ready {
61        return fail("delivery did not wake the receiver");
62    }
63    match task.pending_msgs.as_slice() {
64        [msg] if msg.sender_tid == SENDER && msg.payload() == expected => true,
65        _ => fail("delivery did not leave exactly one owned mailbox message"),
66    }
67}

69fn reset() {
70    remove(SENDER);
71    remove(RECEIVER);
72}

74fn all_producers_defer_foreign_writes() -> bool {
75    // RecvScatter leaves the same TaskState::Recv shape behind, but its temporary
76    // buffer is already gone. INVALID_RECV_PTR models that stale pointer: exercising
77    // every producer here proves none of them dereferences either Recv variant's
78    // retained destination outside the receiver's own syscall context.
79    let payload = b"owned-ipc";
80    let mut ok = true;
81
82    prepare_receiver(SENDER);
83    if super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()) != Ok(0)
84        || !verify_delivery(payload)
85    {
86        ok = fail("ipc_send did not defer matched Recv delivery");
87    }
88    reset();
89
90    prepare_receiver(usize::MAX);
91    if super::ipc_post_nonblock(SENDER, RECEIVER, payload).is_err() || !verify_delivery(payload) {
92        ok = fail("ipc_post_nonblock did not defer matched Recv delivery");
93    }
94    reset();
95
96    prepare_receiver(SENDER);
97    if super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_err()
98        || !verify_delivery(payload)
99    {
100        ok = fail("ipc_try_send did not defer matched Recv delivery");
101    }
102    reset();
103    ok
104}

106fn verify_completion_wait_wake(expected: &[u8], ready_before: usize, sender_blocks: bool) -> bool {
107    let guard = super::SCHEDULER.lock();
108    let Some(sched) = guard.as_ref() else {
109        return fail("scheduler missing after completion-wait IPC wake");
110    };
111    let Some(receiver) = sched.tasks.get(&RECEIVER) else {
112        return fail("completion-wait receiver disappeared");
113    };
114    let sender_generation = sched
115        .tasks
116        .get(&SENDER)
117        .map(|task| task.cell_generation)
118        .unwrap_or(0);
119    let receiver_ok = receiver.state == TaskState::Ready
120        && matches!(
121            receiver.pending_msgs.as_slice(),
122            [msg]
123                if msg.sender_tid == SENDER
124                    && msg.payload() == expected
125                    && msg.wire_header().is_some_and(|header| {
126                        header.sender_tid == SENDER
127                            && header.sender_cell_id == TEST_CELL
128                            && header.sender_generation == sender_generation
129                            && header.delivery_id != 0
130                    })
131        );
132    let sender_ok = sched.tasks.get(&SENDER).is_some_and(|sender| {
133        if sender_blocks {
134            matches!(
135                sender.state,
136                TaskState::Sending { target, .. } if target == RECEIVER
137            )
138        } else {
139            sender.state == TaskState::Ready
140        }
141    });
142    receiver_ok && sender_ok && super::hart_local::ready::total_ready_count() == ready_before + 1
143}

145/// The state publication is the handoff boundary: a producer arriving before
146/// `yield_cpu` or after the switch observes the same state and must perform the
147/// same single ready transition.
148fn all_producers_wake_net_rx_completion_wait() -> bool {
149    let payload = b"completion-ipc";
150    let mut ok = true;
151
152    prepare_completion_wait_receiver();
153    let ready_before = super::hart_local::ready::total_ready_count();
154    if super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()) != Ok(1)
155        || !verify_completion_wait_wake(payload, ready_before, true)
156    {
157        ok = fail("ipc_send did not wake NET_RX wait while preserving blocking send");
158    }
159    reset();
160
161    prepare_completion_wait_receiver();
162    let ready_before = super::hart_local::ready::total_ready_count();
163    if super::ipc_post_nonblock(SENDER, RECEIVER, payload).is_err()
164        || !verify_completion_wait_wake(payload, ready_before, false)
165    {
166        ok = fail("ipc_post_nonblock did not wake NET_RX completion wait");
167    }
168    reset();
169
170    prepare_completion_wait_receiver();
171    let ready_before = super::hart_local::ready::total_ready_count();
172    let saved_input = super::drivers::driver_cell::INPUT_CELL_TID.load(Ordering::Acquire);
173    super::drivers::driver_cell::INPUT_CELL_TID.store(SENDER, Ordering::Release);
174    let sent =
175        super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_ok();
176    super::drivers::driver_cell::INPUT_CELL_TID.store(saved_input, Ordering::Release);
177    if !sent || !verify_completion_wait_wake(payload, ready_before, false) {
178        ok = fail("trusted ipc_try_send did not wake NET_RX completion wait");
179    }
180    reset();
181
182    prepare_completion_wait_receiver();
183    let ready_before = super::hart_local::ready::total_ready_count();
184    let saved_input = super::drivers::driver_cell::INPUT_CELL_TID.load(Ordering::Acquire);
185    super::drivers::driver_cell::INPUT_CELL_TID.store(0, Ordering::Release);
186    let rejected =
187        super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_err();
188    super::drivers::driver_cell::INPUT_CELL_TID.store(saved_input, Ordering::Release);
189    let unchanged = {
190        let guard = super::SCHEDULER.lock();
191        guard
192            .as_ref()
193            .and_then(|sched| sched.tasks.get(&RECEIVER))
194            .is_some_and(|task| {
195                matches!(
196                    task.state,
197                    TaskState::WaitCompletion {
198                        source: api::completion::source::NET_RX,
199                        ..
200                    }
201                ) && task.pending_msgs.is_empty()
202            })
203    };
204    if !rejected || !unchanged || super::hart_local::ready::total_ready_count() != ready_before {
205        ok = fail("untrusted ipc_try_send broadened completion-wait admission");
206    }
207    reset();
208    ok
209}

211fn ready_receiver_is_not_enqueued_twice() -> bool {
212    prepare_completion_wait_receiver();
213    let ready_before = super::hart_local::ready::total_ready_count();
214    let first = super::ipc_post_nonblock(SENDER, RECEIVER, b"first");
215    let second = super::ipc_post_nonblock(SENDER, RECEIVER, b"second");
216    let unchanged_ready_depth = super::hart_local::ready::total_ready_count() == ready_before + 1;
217    let messages_preserved = {
218        let guard = super::SCHEDULER.lock();
219        guard
220            .as_ref()
221            .and_then(|sched| sched.tasks.get(&RECEIVER))
222            .is_some_and(|task| {
223                matches!(
224                    task.pending_msgs.as_slice(),
225                    [first, second]
226                        if first.payload() == b"first" && second.payload() == b"second"
227                )
228            })
229    };
230    reset();
231    first.is_ok() && second.is_ok() && unchanged_ready_depth && messages_preserved
232        || fail("IPC queued a receiver that was already Ready more than once")
233}

235fn timer_completion_wait_remains_deadline_only() -> bool {
236    insert(SENDER, TEST_CELL);
237    insert(RECEIVER, TEST_CELL + 1);
238    let ready_before = super::hart_local::ready::total_ready_count();
239    let queued_before_park = super::ipc_post_nonblock(SENDER, RECEIVER, b"before-timer-park");
240    let decision = {
241        let mut guard = super::SCHEDULER.lock();
242        guard
243            .as_mut()
244            .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
245            .map(|task| {
246                super::completion_wait::publish_wait_state_locked(
247                    task,
248                    RECEIVER,
249                    api::completion::source::TIMER,
250                    Some(u64::MAX),
251                )
252            })
253    };
254    let posted_while_waiting = super::ipc_post_nonblock(SENDER, RECEIVER, b"timer-ipc");
255    let unchanged = {
256        let guard = super::SCHEDULER.lock();
257        guard
258            .as_ref()
259            .and_then(|sched| sched.tasks.get(&RECEIVER))
260            .is_some_and(|task| {
261                matches!(
262                    task.state,
263                    TaskState::WaitCompletion {
264                        source: api::completion::source::TIMER,
265                        ..
266                    }
267                ) && matches!(
268                    task.pending_msgs.as_slice(),
269                    [before, during]
270                        if before.payload() == b"before-timer-park"
271                            && during.payload() == b"timer-ipc"
272                )
273            })
274    };
275    let ready_unchanged = super::hart_local::ready::total_ready_count() == ready_before;
276    reset();
277    queued_before_park.is_ok()
278        && decision == Some(super::completion_wait::CompletionParkDecision::Parked)
279        && posted_while_waiting.is_ok()
280        && unchanged
281        && ready_unchanged
282        || fail("IPC interrupted a TIMER completion wait")
283}

285fn fill_completion_wait_mailbox(depth: usize) -> bool {
286    prepare_completion_wait_receiver();
287    let mut guard = super::SCHEDULER.lock();
288    let Some(task) = guard
289        .as_mut()
290        .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
291    else {
292        return false;
293    };
294    for _ in 0..depth {
295        if super::queue_pending_msg(task, SENDER, b"x", depth).is_err() {
296            return false;
297        }
298    }
299    true

302fn completion_wait_mailbox_is_unchanged(depth: usize, ready_before: usize) -> bool {
303    let guard = super::SCHEDULER.lock();
304    let Some(sched) = guard.as_ref() else {
305        return false;
306    };

308        matches!(
309            task.state,
310            TaskState::WaitCompletion {
311                source: api::completion::source::NET_RX,
312                ..

315    }) && sched
316        .tasks

319        && super::hart_local::ready::total_ready_count() == ready_before

322fn full_completion_wait_mailbox_never_wakes() -> bool {
323    let payload = b"overflow";
324    let mut ok = true;
325
326    if !fill_completion_wait_mailbox(HOTSWAP_MSG_QUEUE_DEPTH) {
327        reset();
328        return fail("could not fill Send completion-wait mailbox");
329    }
330    let ready_before = super::hart_local::ready::total_ready_count();
331    let refused = super::ipc_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len())
332        == Err(super::IpcSendError::Backpressure);
333    if !refused || !completion_wait_mailbox_is_unchanged(HOTSWAP_MSG_QUEUE_DEPTH, ready_before) {
334        ok = fail("full mailbox Send mutated or woke completion waiter");
335    }
336    reset();
337
338    if !fill_completion_wait_mailbox(HOTSWAP_MSG_QUEUE_DEPTH) {
339        reset();
340        return fail("could not fill post completion-wait mailbox");
341    }
342    let ready_before = super::hart_local::ready::total_ready_count();
343    let refused = super::ipc_post_nonblock(SENDER, RECEIVER, payload).is_err();
344    if !refused || !completion_wait_mailbox_is_unchanged(HOTSWAP_MSG_QUEUE_DEPTH, ready_before) {
345        ok = fail("full mailbox post mutated or woke completion waiter");
346    }
347    reset();
348
349    if !fill_completion_wait_mailbox(INPUT_EVENT_QUEUE_DEPTH) {
350        reset();
351        return fail("could not fill TrySend completion-wait mailbox");
352    }
353    let ready_before = super::hart_local::ready::total_ready_count();
354    let saved_input = super::drivers::driver_cell::INPUT_CELL_TID.load(Ordering::Acquire);
355    super::drivers::driver_cell::INPUT_CELL_TID.store(SENDER, Ordering::Release);
356    let refused =
357        super::ipc_try_send(SENDER, RECEIVER, payload.as_ptr() as usize, payload.len()).is_err();
358    super::drivers::driver_cell::INPUT_CELL_TID.store(saved_input, Ordering::Release);
359    if !refused || !completion_wait_mailbox_is_unchanged(INPUT_EVENT_QUEUE_DEPTH, ready_before) {
360        ok = fail("full mailbox TrySend mutated or woke completion waiter");
361    }
362    reset();
363    ok
364}

399fn ipc_before_park_returns_raw_zero() -> bool {
400    insert(SENDER, TEST_CELL);
401    insert(RECEIVER, TEST_CELL + 1);
402    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
403        super::completion::deliver_pending_wakes(sched);
404    }
405    let Some(completions) = queue(RECEIVER) else {
406        reset();
407        return fail("pre-park case could not reach completion queue");
408    };
409    let Some(slot) = completions.reserve() else {
410        reset();
411        return fail("pre-park case could not reserve completion slot");
412    };
413    let Some(waiter) = super::completion_wait::WaiterRegistration::new(
414        &completions,
415        RECEIVER,
416        slot,
417        api::completion::source::NET_RX,
418    ) else {
419        let _ = completions.release(slot);
420        reset();
421        return fail("pre-park case could not register waiter");
422    };
423    super::waker::arm_net_rx(completions.clone(), slot);
424
425    let ready_before = super::hart_local::ready::total_ready_count();
426    let posted = super::ipc_post_nonblock(SENDER, RECEIVER, b"pre-park");
427    let decision = {
428        let mut guard = super::SCHEDULER.lock();
429        guard
430            .as_mut()
431            .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
432            .map(|task| {
433                super::completion_wait::publish_wait_state_locked(
434                    task,
435                    RECEIVER,
436                    api::completion::source::NET_RX,
437                    None,
438                )
439            })
440    };
441    let mut untouched = [0xa5u8; api::completion::COMPLETION_LEN];
442    let outcome = super::completion_wait::finish_net_rx_wait(
443        RECEIVER,
444        untouched.as_mut_ptr() as usize,
445        &completions,
446    );
447    drop(waiter);
448
449    let task_ok = {
450        let guard = super::SCHEDULER.lock();
451        guard
452            .as_ref()
453            .and_then(|sched| sched.tasks.get(&RECEIVER))
454            .is_some_and(|task| {
455                task.state == TaskState::Ready
456                    && task.completion_wait.is_none()
457                    && matches!(
458                        task.pending_msgs.as_slice(),
459                        [msg] if msg.payload() == b"pre-park"
460                    )
461            })
462    };
463    let ok = posted.is_ok()
464        && decision == Some(super::completion_wait::CompletionParkDecision::MailboxPending)
465        && outcome == Ok(0)
466        && untouched == [0xa5u8; api::completion::COMPLETION_LEN]
467        && completions.reserved() == 0
468        && completions.drainable() == 0
469        && !super::completion::wakes_pending()
470        && super::hart_local::ready::total_ready_count() == ready_before
471        && task_ok;
472    reset();
473    ok || fail("IPC-before-park did not abort NET_RX wait cleanly with raw zero")
474}

476fn full_mailbox_refuses_without_wake() -> bool {
477    prepare_receiver(SENDER);
478    let fill_ok = {
479        let mut guard = super::SCHEDULER.lock();
480        let task = guard
481            .as_mut()
482            .and_then(|sched| sched.tasks.get_mut(&RECEIVER))
483            .expect("synthetic receiver exists");
484        let mut ok = true;
485        for _ in 0..HOTSWAP_MSG_QUEUE_DEPTH {
486            if super::queue_pending_msg(task, SENDER, b"x", HOTSWAP_MSG_QUEUE_DEPTH).is_err() {
487                ok = false;
488                break;
489            }
490        }
491        ok
492    };
493    if !fill_ok {
494        reset();
495        return fail("mailbox refused an entry below its depth");
496    }
497
498    let refused = super::ipc_post_nonblock(SENDER, RECEIVER, b"overflow").is_err();
499    let unchanged = {
500        let guard = super::SCHEDULER.lock();
501        guard
502            .as_ref()
503            .and_then(|sched| sched.tasks.get(&RECEIVER))
504            .is_some_and(|task| {
505                matches!(task.state, TaskState::Recv { .. })
506                    && task.current_caller.is_none()
507                    && task.pending_msgs.len() == HOTSWAP_MSG_QUEUE_DEPTH
508            })
509    };
510    reset();
511    refused && unchanged || fail("full mailbox changed receiver state or accepted overflow")
512}

514fn quota_failure_is_fallible() -> bool {
515    let mut task = Task::new(9330, CellId(QUOTA_CELL), "ipc-quota-selftest", Vec::new());
516    crate::memory::cell_quota::register(CellId(QUOTA_CELL), 0);
517    let inline_ok =
518        super::queue_pending_msg(&mut task, SENDER, b"x", HOTSWAP_MSG_QUEUE_DEPTH).is_ok();
519    if inline_ok {
520        drop(task.pending_msgs.remove(0));
521    }
522    let heap_backed_payload = [0x5a; 33];
523    let result = super::queue_pending_msg(
524        &mut task,
525        SENDER,
526        &heap_backed_payload,
527        HOTSWAP_MSG_QUEUE_DEPTH,
528    );
529    crate::memory::cell_quota::deregister(CellId(QUOTA_CELL));
530
531    if inline_ok && result.is_err() && task.pending_msgs.is_empty() {
532        true
533    } else {
534        fail("inline delivery allocated or heap quota failure mutated the mailbox")
535    }
536}

548    crate::memory::cell_quota::deregister(CellId(QUOTA_CELL));
549    queued.is_ok() && charged >= payload.len() && refunded == 0
550        || fail("heap payload was not charged and refunded to the receiver")
551}

570fn pending_drain_keeps_sender_context_without_relocking() -> bool {
571    insert(SENDER, TEST_CELL);
572    insert(RECEIVER, TEST_CELL + 1);
573    let payload = b"context";
574    let mut recv_buf = [0u8; 16];
575    let result = {
576        let mut guard = super::SCHEDULER.lock();
577        let Some(sched) = guard.as_mut() else {
578            return fail("scheduler missing during IPC pending drain selftest");
579        };
580        let receiver = sched
581            .tasks
582            .get_mut(&RECEIVER)
583            .expect("synthetic receiver exists");
584        receiver.pending_msgs.try_push(super::tcb::PendingMsg {
585            sender_tid: SENDER,
586            data: super::pending_mailbox::PendingMsgData::try_copy(payload, TEST_CELL as usize)
587                .expect("inline payload copy must fit"),
588            wire: None,
589            enqueued_tick: 0,
590        })
591    };
592    if result.is_err() {
593        reset();
594        return fail("could not enqueue the synthetic pending IPC message");
595    }
596
597    let sender_generation = {
598        let guard = super::SCHEDULER.lock();
599        guard
600            .as_ref()
601            .and_then(|sched| sched.tasks.get(&SENDER))
602            .map(|task| task.cell_generation)
603            .unwrap_or(0)
604    };
605    let delivered = super::syscall::handle_syscall(
606        RECEIVER,
607        super::syscall::Syscall::TryRecv {
608            mask: SENDER,
609            buf_ptr: recv_buf.as_mut_ptr() as usize,
610            buf_len: payload.len(),
611            attest_caller: false,
612        },
613    );
614    let preserved = {
615        let guard = super::SCHEDULER.lock();
616        guard
617            .as_ref()
618            .and_then(|sched| sched.tasks.get(&RECEIVER))
619            .is_some_and(|task| {
620                task.current_caller == Some(SENDER)
621                    && task.current_caller_cell_id == TEST_CELL
622                    && task.current_caller_cell_generation == sender_generation
623                    && task.pending_msgs.is_empty()
624            })
625    };
626    reset();
627    matches!(delivered, Ok(id) if id == SENDER)
628        && recv_buf[..payload.len()] == *payload
629        && preserved
630        || fail("pending-message drain did not preserve sender context under one scheduler lock")
631}

633fn try_recv_attestation_writes_identity_trailer() -> bool {
634    prepare_receiver(0);
635    let payload = b"attested-try-recv";
636    let mut recv_buf = [0u8; 64];
637    let result = {
638        let mut guard = super::SCHEDULER.lock();
639        let Some(sched) = guard.as_mut() else {
640            return fail("scheduler missing during try_recv attestation selftest");
641        };
642        let receiver = sched
643            .tasks
644            .get_mut(&RECEIVER)
645            .expect("synthetic receiver exists");
646        receiver.pending_msgs.try_push(super::tcb::PendingMsg {
647            sender_tid: SENDER,
648            data: super::pending_mailbox::PendingMsgData::try_copy(payload, TEST_CELL as usize)
649                .expect("inline payload copy must fit"),
650            wire: None,
651            enqueued_tick: 0,
652        })
653    };
654    if result.is_err() {
655        reset();
656        return fail("could not enqueue pending message for try_recv attestation test");
657    }
658
659    let sender_generation = {
660        let guard = super::SCHEDULER.lock();
661        guard
662            .as_ref()
663            .and_then(|sched| sched.tasks.get(&SENDER))
664            .map(|task| task.cell_generation)
665            .unwrap_or(0)
666    };
667
668    // Case 1: attest_caller = true
669    let delivered = super::syscall::handle_syscall(
670        RECEIVER,
671        super::syscall::Syscall::TryRecv {
672            mask: SENDER,
673            buf_ptr: recv_buf.as_mut_ptr() as usize,
674            buf_len: recv_buf.len(),
675            attest_caller: true,
676        },
677    );
678    let identity = api::caller_identity::CallerIdentity::from_recv_buf(&recv_buf);
679    let attested_ok = matches!(delivered, Ok(id) if id == SENDER)
680        && recv_buf[..payload.len()] == *payload
681        && identity
682            == Some(api::caller_identity::CallerIdentity {
683                flags: 0,
684                cell_id: TEST_CELL,
685                generation: sender_generation,
686                sender_tid: SENDER as u64,
687            });
688
689    // Case 2: SENDER has bit 63 set and allowlist != u64::MAX -> flags = CALLER_FLAG_VFS_MUTATE
690    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
691        if let Some(task) = sched.tasks.get_mut(&SENDER) {
692            task.syscall_allowlist = 1u64 << 63;
693        }
694        if let Some(receiver) = sched.tasks.get_mut(&RECEIVER) {
695            receiver.pending_msgs.try_push(super::tcb::PendingMsg {
696                sender_tid: SENDER,
697                data: super::pending_mailbox::PendingMsgData::try_copy(payload, TEST_CELL as usize)
698                    .expect("inline payload copy must fit"),
699                wire: None,
700                enqueued_tick: 0,
701            }).ok();
702        }
703    }
704    recv_buf.fill(0);
705    let mutate_delivered = super::syscall::handle_syscall(
706        RECEIVER,
707        super::syscall::Syscall::TryRecv {
708            mask: SENDER,
709            buf_ptr: recv_buf.as_mut_ptr() as usize,
710            buf_len: recv_buf.len(),
711            attest_caller: true,
712        },
713    );
714    let mutate_identity = api::caller_identity::CallerIdentity::from_recv_buf(&recv_buf);
715    let mutate_ok = matches!(mutate_delivered, Ok(id) if id == SENDER)
716        && mutate_identity
717            == Some(api::caller_identity::CallerIdentity {
718                flags: api::caller_identity::CALLER_FLAG_VFS_MUTATE,
719                cell_id: TEST_CELL,
720                generation: sender_generation,
721                sender_tid: SENDER as u64,
722            });
723
724    reset();
725    (attested_ok && mutate_ok)
726        || fail("try_recv with attest_caller=true did not write valid CallerIdentity trailer or mutate flags")
727}
