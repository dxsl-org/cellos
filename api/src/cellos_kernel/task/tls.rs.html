1//! Per-task user thread pointer (the TLS base primitive, ADR-0018 §2.1).
2//!
3//! The kernel owns exactly one word per task and installs it as the *user* thread
4//! pointer. The value is opaque: a cell allocates its own TLS blocks and tells the
5//! kernel where they are. Nothing here parses, dereferences, or bounds-checks the
6//! base — a bad value is the cell's own fault and is contained by its tier (LBI on
7//! Tier 1, an MMU fault on Tier 2), which is why this syscall carries no authority
8//! and needs no allowlist bit.
9//!
10//! The carrier differs per architecture, and each choice is deliberate:
11//!
12//! * **riscv64** — the user `tp` (x4) is part of the trap frame. `HART_TRAP_ENTRY`
13//!   parks the user's `tp` in the frame and `__trap_exit` restores it, so writing
14//!   the frame is enough; no switch work is needed. The kernel's own `tp` is a
15//!   different value (the HartLocal pointer, reloaded on every U→S transition), so
16//!   the two never collide.
17//! * **aarch64** — `TPIDR_EL0` is not banked by the exception model and is not part
18//!   of the trap frame, so the kernel writes it on every resume and when the cell
19//!   sets a base. `TPIDR_EL1` (kernel stack) is untouched.
20//! * **x86_64** — `FS_BASE` (`IA32_FS_BASE`) is written on resume and on set; the
21//!   `GS_BASE`/`KERNEL_GS_BASE` pair stays kernel context state.
22//!
23//! Contract: the kernel owns this register. A cell must use the `SetTlsBase`
24//! syscall rather than writing the register directly, because a direct user write
25//! is overwritten on the next resume. `SetTlsBase` returns the previous base, which
26//! is also the only supported way for a cell to read it back.
27
28use super::tcb::Task;

30/// Store the caller's base and make it effective for the current user return.
31pub(crate) fn set_base(task: &mut Task, base: usize) {
32    task.tls_base = base;
33    #[cfg(target_arch = "riscv64")]
34    write_frame_tp(task, base);
35    #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
36    write_user_thread_pointer(base);
37}
38
39/// Install the selected task's base before it returns to user mode.
40///
41/// The scheduler publishes the base while holding `SCHEDULER`. This runs both
42/// before a raw switch (so a fresh task's direct jump to `__trap_exit` has the
43/// right carrier) and after a switched-back context resumes. RISC-V carries the
44/// value in the task trap frame instead.
45pub(crate) fn install_published_base() {
46    #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
47    write_user_thread_pointer(super::hart_local::current_tls_base());
48}
49
50/// Reinstall a switched-back task's base after its raw context switch returns.
51#[inline]
52pub(crate) fn apply_on_resume() {
53    install_published_base();
54}
55
56/// Give a newly spawned thread the creator's base, unless the creator had none.
57///
58/// This is deliberately distinct from [`set_base`]: the new task is not the
59/// running task, so installing an AArch64/x86 user pointer here would corrupt
60/// the creator's hardware register. On RISC-V the fully primed trap frame owns
61/// the register; other architectures install the stored base before the raw
62/// switch that enters the child.
63pub(crate) fn inherit_from(task: &mut Task, parent_base: usize) {
64    if parent_base == 0 {
65        return;
66    }
67    task.tls_base = parent_base;
68    #[cfg(target_arch = "riscv64")]
69    write_frame_tp(task, parent_base);
70}

72/// Write the user `tp` slot of a task's trap frame.
73///
74/// The frame sits at the top of the task's kernel stack in every case: the spawn
75/// path primes it there (`task.rs`: `kernel_stack.top - TRAP_FRAME_SIZE`), and a
76/// U→S trap rebuilds it at the same address. For a task currently in a syscall
77/// this is the live frame, so the write takes effect on the trap return.
78#[cfg(target_arch = "riscv64")]
79fn write_frame_tp(task: &Task, base: usize) {
80    let Some(stack) = task.kernel_stack.as_ref() else {
81        return;
82    };
83    let frame = (stack.top - super::TRAP_FRAME_SIZE) as *mut crate::hal::arch::ViTrapFrame;
84    // SAFETY: the address is the task's own trap frame inside its kernel stack,
85    // which is allocated and owned by this task for its whole lifetime. x4 is the
86    // thread pointer slot; `__trap_exit` loads it with `ld x4, 4*8(sp)`.
87    unsafe {
88        (*frame).regs[4] = base;
89    }
90}

92/// Write the architecture's user thread pointer register (aarch64/x86_64).
93#[cfg(target_arch = "aarch64")]
94fn write_user_thread_pointer(base: usize) {
95    // SAFETY: TPIDR_EL0 is the EL0-visible thread pointer; writing it from EL1 is
96    // always permitted and touches no kernel state (the kernel uses TPIDR_EL1).
97    unsafe {
98        core::arch::asm!("msr tpidr_el0, {0}", in(reg) base, options(nomem, nostack));
99    }
100}
101
102#[cfg(target_arch = "x86_64")]
103fn write_user_thread_pointer(base: usize) {
104    const IA32_FS_BASE: u32 = 0xC000_0100;
105    // SAFETY: WRMSR of IA32_FS_BASE from ring 0 is always permitted; it sets the
106    // user-mode FS segment base and leaves the kernel's GS-based per-CPU state
107    // (GS_BASE/KERNEL_GS_BASE) untouched.
108    unsafe {
109        core::arch::asm!(
110            "wrmsr",
111            in("ecx") IA32_FS_BASE,
112            in("eax") base as u32,
113            in("edx") (base >> 32) as u32,
114            options(nomem, nostack)
115        );
116    }
117}
