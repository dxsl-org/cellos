1//! Per-hart local state, accessed in O(1) via the `tp` (thread-pointer) CSR.
2//!
3//! Each hart keeps `tp = &HART_LOCALS[hart_id]` at all times while running
4//! kernel code.  The trap entry restores the kernel `tp` on U→S transitions
5//! so all kernel code (syscall handler, allocator, scheduler) always sees the
6//! correct HartLocal.  Cells run with the value stored in
7//! `ViHartLocal::kernel_tp_for_cells`, which is independent of HartLocal.
8//!
9//! Phase 02 replaces the single global `CURRENT_CELL_ID` with a per-hart
10//! `current_cell_id` field inside `ViHartLocal`.  Phase 03 adds per-hart
11//! ready queues and the work-stealing scheduler.
12

23/// Per-hart local state.
24///
25/// LAYOUT IS ABI — the trap.S reads `kernel_tp_for_cells` by hardcoded offset
26/// and the field order is FIXED.  Add new fields AFTER existing ones.
27/// `#[repr(C)]` ensures Rust does not reorder or pad unexpectedly.
28#[repr(C)]
29pub struct ViHartLocal {
30    /// This hart's id (0 = boot hart).
31    pub hart_id: usize, // offset 0
32    /// Cell ID currently running on this hart.  0 = kernel (no quota limit).
33    pub current_cell_id: AtomicUsize, // offset 8  (AtomicUsize is transparent over usize)
34    /// Value of `gp` captured at `install()` time — handed to new cells.
35    pub kernel_gp: usize, // offset 16
36    /// Value of `tp` that cells inherit on context switch.  Currently 0 (cells
37    /// have no TLS); Phase 05 may give each cell a private tp.
38    pub kernel_tp_for_cells: usize, // offset 24
39    /// Per-hart ready queues keyed by priority (`u8`; higher = higher priority).
40    /// Leaf lock: may be locked while holding SCHEDULER, never the reverse.
41    pub ready: crate::sync::Spinlock<BTreeMap<u8, VecDeque<usize>>>,
42    /// Task selected by the scheduler for this hart.  This can lead the actual
43    /// CPU context while `Context::switch` is in flight.
44    pub current_task_id: AtomicUsize,
45    /// Task whose saved context is actually executing on this hart.  Published
46    /// by the incoming context only after the raw switch has changed stacks.
47    pub executing_task_id: AtomicUsize,
48    /// Incoming task whose raw Context has been selected but has not yet
49    /// completed the stack/register switch.  This is a transient ownership pin:
50    /// retirement must treat it as live until the incoming-side completion hook
51    /// publishes `executing_task_id` and clears this field.
52    pub selected_task_id: AtomicUsize,
53    /// Runnable outgoing task whose Context save is in flight.  It may be on
54    /// this hart's ready queue for round-robin fairness, but another hart must
55    /// not steal it until the raw switch has saved its Context.
56    pub outgoing_context_save_task_id: AtomicUsize,
57    /// Generation paired with `current_cell_id` for the task currently selected
58    /// on this hart.  Trap code snapshots it with the CellId before surrendering
59    /// allocation attribution to the kernel.
60    current_cell_generation: AtomicU64,
61    /// Private-root identity selected for this hart. Zero is the shared SAS root.
62    current_domain_id: AtomicU64,
63    /// Immutable generation paired with `current_domain_id`.
64    current_domain_generation: AtomicU64,
65    /// Last safe-root completion acknowledgement for remote retirement.
66    domain_ack_generation: AtomicU64,
67    /// A safe root is active but its incoming Context has not yet acknowledged it.
68    safe_root_pending: AtomicUsize,
69    /// Fixed retirement handoff slot.  A trap fault or clean `Exit` writes
70    /// scalar state here before scheduler-owned collections are touched; the
71    /// scheduler consumes it later with kernel allocation attribution.
72    deferred_retirement_pending: AtomicUsize,
73    deferred_retirement_kind: AtomicUsize,
74    deferred_retirement_tid: AtomicUsize,
75    deferred_retirement_cell_id: AtomicUsize,
76    deferred_retirement_generation: AtomicU64,
77    deferred_retirement_exit_code: AtomicUsize,
78    deferred_retirement_fault_cause: AtomicUsize,
79    deferred_retirement_fault_pc: AtomicUsize,
80    deferred_retirement_fault_addr: AtomicUsize,
81    /// Private root whose execution pin this hart currently holds. Mirrors
82    /// `current_domain_id` in lifetime: set on successful activation, moved out
83    /// when the hart plans a transition away.
84    #[cfg(all(feature = "native-domains", target_arch = "riscv64"))]
85    pinned_domain: execution_pin::HartOwnedSlot,
86    /// Outgoing root awaiting release at the switch-completion hook. Staged
87    /// under `SCHEDULER`, consumed exactly once per completed transition away.
88    #[cfg(all(feature = "native-domains", target_arch = "riscv64"))]
89    staged_domain_release: execution_pin::HartOwnedSlot,
90    /// Recoverable user-copy fault guard (RV64). Armed by `task::user_copy`
91    /// around exactly one guarded byte-copy window and read by the trap
92    /// handler through `vi_user_copy_guard_fault`. Hart-owned: only the
93    /// executing hart touches its own slot, so plain atomics need no locks.
94    /// Zero resume PC means "no landing pad"; the active flag is the gate.
95    #[cfg(target_arch = "riscv64")]
96    pub user_copy_guard_active: AtomicUsize,
97    #[cfg(target_arch = "riscv64")]
98    pub user_copy_guard_resume_pc: AtomicUsize,
99    /// Inclusive start of the user range covered by the armed guard. The
100    /// trap hook rejects faults whose `stval` falls outside [start, end).
101    #[cfg(target_arch = "riscv64")]
102    pub user_copy_guard_start: AtomicUsize,
103    /// Exclusive end of the guarded user range.
104    #[cfg(target_arch = "riscv64")]
105    pub user_copy_guard_end: AtomicUsize,
106}

108/// Static array of per-hart local state, one entry per supported hart.
109/// Accessed without any lock: hart N only writes HART_LOCALS[N] from N.
110/// SAFETY: interior mutability via AtomicUsize; the `usize` fields are only
111/// written during `install()` before the hart handles any interrupt.
112pub static HART_LOCALS: [ViHartLocal; MAX_HARTS] = {
113    // `ZERO` is consumed exactly once below, by the `[ZERO; MAX_HARTS]` array-repeat
114    // expression — rustc evaluates a `const` operand fresh per array slot, so each
115    // hart gets its OWN independent `AtomicUsize`/`Spinlock`, not a shared instance.
116    // That is the desired behaviour here (see module doc: hart N only touches slot N),
117    // so this is not the aliasing footgun `declare_interior_mutable_const` warns about;
118    // switching to `static` would break the repeat expression (requires a `const` for
119    // non-`Copy` element types).
120    #[allow(clippy::declare_interior_mutable_const)]
121    const ZERO: ViHartLocal = ViHartLocal {
122        hart_id: 0,
123        current_cell_id: AtomicUsize::new(0),
124        kernel_gp: 0,
125        kernel_tp_for_cells: 0,
126        ready: crate::sync::Spinlock::new(BTreeMap::new()),
127        current_task_id: AtomicUsize::new(0),
128        executing_task_id: AtomicUsize::new(0),
129        selected_task_id: AtomicUsize::new(0),
130        outgoing_context_save_task_id: AtomicUsize::new(0),
131        current_cell_generation: AtomicU64::new(0),
132        current_domain_id: AtomicU64::new(0),
133        current_domain_generation: AtomicU64::new(0),
134        domain_ack_generation: AtomicU64::new(0),
135        safe_root_pending: AtomicUsize::new(0),
136        deferred_retirement_pending: AtomicUsize::new(0),
137        deferred_retirement_kind: AtomicUsize::new(0),
138        deferred_retirement_tid: AtomicUsize::new(0),
139        deferred_retirement_cell_id: AtomicUsize::new(0),
140        deferred_retirement_generation: AtomicU64::new(0),
141        deferred_retirement_exit_code: AtomicUsize::new(0),
142        deferred_retirement_fault_cause: AtomicUsize::new(0),
143        deferred_retirement_fault_pc: AtomicUsize::new(0),
144        deferred_retirement_fault_addr: AtomicUsize::new(0),
145        #[cfg(all(feature = "native-domains", target_arch = "riscv64"))]
146        pinned_domain: execution_pin::HartOwnedSlot(core::cell::UnsafeCell::new(None)),
147        #[cfg(all(feature = "native-domains", target_arch = "riscv64"))]
148        staged_domain_release: execution_pin::HartOwnedSlot(core::cell::UnsafeCell::new(None)),
149        #[cfg(target_arch = "riscv64")]
150        user_copy_guard_active: AtomicUsize::new(0),
151        #[cfg(target_arch = "riscv64")]
152        user_copy_guard_resume_pc: AtomicUsize::new(0),
153        #[cfg(target_arch = "riscv64")]
154        user_copy_guard_start: AtomicUsize::new(0),
155        #[cfg(target_arch = "riscv64")]
156        user_copy_guard_end: AtomicUsize::new(0),
157    };
158    [ZERO; MAX_HARTS]
159};

161/// Pointer to the calling hart's `ViHartLocal`, stored as a plain `usize`.
162///
163/// `trap.S` loads `tp` from this address on every U→S trap so kernel code
164/// always runs with a valid hart-local pointer.  Exposed `#[no_mangle]` so
165/// the assembler can reference it by name without mangling.
166///
167/// Phase 03 upgrades this to the `sscratch = &HartLocal` protocol for full
168/// SMP correctness.  For Phase 02 (single active hart), the single-entry
169/// array gives the correct result.
170#[no_mangle]
171pub static HART_LOCAL_TP_ADDRS: [AtomicUsize; MAX_HARTS] =
172    [AtomicUsize::new(0), AtomicUsize::new(0)];
173
174/// Initialize the calling hart's `ViHartLocal` and write `tp` to point at it.
175///
176/// Call BEFORE enabling the scheduler or handling any interrupt on this hart.
177/// Hart 0 calls this as the first action of `task::init()`.
178/// Secondary harts call this from `smp_hart_entry`, after installing stvec.
179pub fn install(hart_id: usize) {
180    assert!(
181        hart_id < MAX_HARTS,
182        "hart_id {} >= MAX_HARTS {}",
183        hart_id,
184        MAX_HARTS
185    );
186
187    // Capture the current gp and tp so we can hand them to cells unchanged.
188    let (gp, tp) = crate::hal::arch::get_gp_tp();
189
190    // SAFETY: hart_id < MAX_HARTS; we only write HART_LOCALS[hart_id] from
191    // the hart with that id — no concurrent writers for this index.
192    let hl = &HART_LOCALS[hart_id];
193
194    // Write hart_id once (not atomic — no other hart touches this slot yet).
195    // SAFETY: single writer; written before any reader could observe this hart.
196    unsafe {
197        let ptr = hl as *const ViHartLocal as *mut ViHartLocal;
198        core::ptr::addr_of_mut!((*ptr).hart_id).write(hart_id);
199        core::ptr::addr_of_mut!((*ptr).kernel_gp).write(gp);
200        core::ptr::addr_of_mut!((*ptr).kernel_tp_for_cells).write(tp);
201    }
202    hl.current_cell_id.store(0, Ordering::Relaxed);
203    hl.current_cell_generation.store(0, Ordering::Relaxed);
204    hl.current_domain_id.store(0, Ordering::Relaxed);
205    hl.current_domain_generation.store(0, Ordering::Relaxed);
206    hl.domain_ack_generation.store(0, Ordering::Relaxed);
207    hl.safe_root_pending.store(0, Ordering::Relaxed);
208    hl.deferred_retirement_pending.store(0, Ordering::Relaxed);
209
210    // Publish this logical hart's restore pointer before installing its stvec.
211    // The hart-specific trap stub reads only its own array slot.
212    let hl_addr = hl as *const ViHartLocal as usize;
213    HART_LOCAL_TP_ADDRS[hart_id].store(hl_addr, Ordering::Release);
214
215    // Write tp CSR to point at this hart's HartLocal.
216    // SAFETY: tp is a callee-save GPR used here as a kernel-internal pointer;
217    // cells receive `kernel_tp_for_cells` (not this pointer) on context switch.
218    unsafe { write_tp(hl_addr) };
219
220    #[cfg(target_arch = "riscv64")]
221    crate::hal::trap::init_for_hart(hart_id);
222}

224/// Return a reference to the calling hart's `ViHartLocal`.
225///
226/// On RISC-V reads the `tp` CSR. On other architectures (x86_64 single-hart
227/// bring-up) returns HART_LOCALS[0] directly.
228///
229/// # Safety
230/// On RISC-V: `tp` must point to a valid `ViHartLocal` (guaranteed after `install()`).
231#[inline(always)]
232pub unsafe fn current_hart() -> &'static ViHartLocal {
233    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
234    {
235        let tp: usize;
236        core::arch::asm!("mv {}, tp", out(reg) tp, options(nomem, nostack, preserves_flags));
237        &*(tp as *const ViHartLocal)
238    }
239    #[cfg(not(any(target_arch = "riscv64", target_arch = "riscv32")))]
240    {
241        &HART_LOCALS[0]
242    }
243}

245/// Return the calling hart's id.
246///
247/// Returns 0 if called before `install()` (safe — hart 0 is hart 0).
248#[inline(always)]
249pub fn current_hart_id() -> usize {
250    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
251    {
252        let tp: usize;
253        unsafe {
254            core::arch::asm!("mv {}, tp", out(reg) tp, options(nomem, nostack, preserves_flags));
255        }
256        if tp == 0 {
257            return 0;
258        }
259        unsafe { (*(tp as *const ViHartLocal)).hart_id }
260    }
261    #[cfg(not(any(target_arch = "riscv64", target_arch = "riscv32")))]
262    {
263        0
264    }
265}

267/// Cell ID currently running on this hart (0 = kernel, no quota).
268///
269/// On RISC-V reads `tp` CSR. On other architectures returns HART_LOCALS[0].current_cell_id.
270#[inline(always)]
271pub fn current_cell_id() -> usize {
272    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
273    {
274        let tp: usize;
275        unsafe {
276            core::arch::asm!("mv {}, tp", out(reg) tp, options(nomem, nostack, preserves_flags));
277        }
278        if tp == 0 {
279            return 0;
280        }
281        unsafe {
282            (*(tp as *const ViHartLocal))
283                .current_cell_id
284                .load(Ordering::Relaxed)
285        }
286    }
287    #[cfg(not(any(target_arch = "riscv64", target_arch = "riscv32")))]
288    {
289        HART_LOCALS[0].current_cell_id.load(Ordering::Relaxed)
290    }
291}

293/// Provenance required to place a fault in the recoverable scheduler funnel.
294///
295/// A Cell attribution is accounting state, not evidence that the current CPU
296/// context was executing Cell code.  In particular, kernel code servicing a
297/// Cell syscall retains that attribution while it holds kernel locks.  Only a
298/// trap handler that has established a U-mode origin can mint this capability.
299#[derive(Clone, Copy)]
300pub struct TrapProvenUserFault(());
301
302impl TrapProvenUserFault {
303    #[inline(always)]
304    pub(super) const fn new() -> Self {
305        Self(())
306    }
307}
308
309/// Origin of a fault considered for deferred Cell retirement.
310#[derive(Clone, Copy, Debug, Eq, PartialEq)]
311enum FaultOrigin {
312    TrapProvenUser,
313    #[cfg(test)]
314    KernelPanic,
315}
316
317impl FaultOrigin {
318    #[inline(always)]
319    const fn permits_deferred_retirement(self) -> bool {
320        matches!(self, Self::TrapProvenUser)
321    }
322}

324/// Scalar fault state captured by a trap-proven U-mode fault and consumed by
325/// the scheduler.
326///
327/// This deliberately contains no task name or heap-backed diagnostic payload:
328/// quota-exhausted Cell faults must be able to hand off without allocating.
329#[derive(Clone, Copy)]
330pub struct DeferredFault {
331    pub tid: usize,
332    pub cell_id: usize,
333    pub generation: u64,
334    pub cause: usize,
335    pub pc: usize,
336    pub fault_addr: usize,
337    origin: FaultOrigin,
338}
339
340/// Scalar clean-exit state captured before the exiting Cell's allocation
341/// attribution is surrendered.  Root versus worker classification belongs to
342/// the scheduler, where it can use the authoritative task table.
343#[derive(Clone, Copy)]
344pub struct DeferredExit {
345    pub tid: usize,
346    pub cell_id: usize,
347    pub generation: u64,
348    pub code: usize,
349}

351/// One allocation-free handoff record per hart.  A task can reach only one of
352/// these paths before yielding, so a single fixed slot preserves the ordering
353/// between the attribution handoff and scheduler retirement.
354#[derive(Clone, Copy)]
355pub enum DeferredRetirement {
356    Fault(DeferredFault),
357    Exit(DeferredExit),
358}
359
360impl DeferredFault {
361    /// Construct the sole recoverable deferred-fault record.
362    ///
363    /// `TrapProvenUserFault` cannot be minted outside `task`; this prevents
364    /// Cell accounting attribution or a kernel panic from entering the
365    /// scheduler retirement path.
366    #[inline(always)]
367    pub(super) fn from_user_trap(
368        _provenance: TrapProvenUserFault,
369        tid: usize,
370        cell_id: usize,
371        generation: u64,
372        cause: usize,
373        pc: usize,
374        fault_addr: usize,
375    ) -> Self {
376        Self::from_origin(
377            FaultOrigin::TrapProvenUser,
378            tid,
379            cell_id,
380            generation,
381            cause,
382            pc,
383            fault_addr,
384        )
385        .expect("trap-proven U-mode origin must permit deferred retirement")
386    }
387
388    #[inline(always)]
389    fn from_origin(
390        origin: FaultOrigin,
391        tid: usize,
392        cell_id: usize,
393        generation: u64,
394        cause: usize,
395        pc: usize,
396        fault_addr: usize,
397    ) -> Option<Self> {
398        origin.permits_deferred_retirement().then_some(Self {
399            tid,
400            cell_id,
401            generation,
402            cause,
403            pc,
404            fault_addr,
405            origin,
406        })
407    }
408
409    #[inline(always)]
410    pub fn is_trap_proven_user_fault(self) -> bool {
411        self.origin.permits_deferred_retirement()
412    }
413}

516/// Execution-pin bookkeeping for native domains. The slots are touched only by
517/// the owning hart, and only inside the interrupt-masked window from plan
518/// construction to the incoming-side completion hook — no other hart can
519/// observe a partially updated slot.
520#[cfg(all(feature = "native-domains", target_arch = "riscv64"))]
521mod execution_pin {
522    use super::current_hart;
523    use alloc::sync::Arc;
524    use core::cell::UnsafeCell;
525    use crate::memory::address_space::AddressSpace;
526
527    /// Hart-owned slot. `HART_LOCALS` is shared across harts as a static, so
528    /// the wrapper carries `Sync`; soundness rests on the single-owner rule
529    /// documented on [`super::ViHartLocal`], not on this impl.
530    pub(crate) struct HartOwnedSlot(pub(crate) UnsafeCell<Option<Arc<AddressSpace>>>);
531    unsafe impl Sync for HartOwnedSlot {}
532
533    /// Advance this hart's execution-pin slot to `next`, staging any displaced
534    /// root for release at the switch-completion hook. Called under `SCHEDULER`
535    /// while planning a transition, so the displaced root cannot be selected
536    /// again on this hart before its pin is released.
537    pub(crate) fn advance(next: Option<Arc<AddressSpace>>) {
538        let hart = unsafe { current_hart() };
539        // SAFETY: sole access is this hart within the masked plan→completion
540        // window documented on the slot fields.
541        unsafe {
542            let pinned = &mut *hart.pinned_domain.0.get();
543            let displaced = pinned.take();
544            *pinned = next;
545            if displaced.is_some() {
546                *hart.staged_domain_release.0.get() = displaced;
547            }
548        }
549    }
550
551    /// Consume the staged outgoing-root release. Returns `None` on every switch
552    /// that did not leave a pinned root.
553    pub(crate) fn take_staged() -> Option<Arc<AddressSpace>> {
554        // SAFETY: sole access is this hart, within the masked window above.
555        unsafe { (*current_hart().staged_domain_release.0.get()).take() }
556    }
557}

586const DEFERRED_RETIREMENT_EXIT: usize = 1;
587const DEFERRED_RETIREMENT_FAULT: usize = 2;
588
589/// Publish a fixed, per-hart fault record without allocating or acquiring a
590/// lock. Trap entry has interrupts masked, so only this hart can produce it.
591#[inline(always)]
592pub fn defer_fault(fault: DeferredFault) {
593    let hart = unsafe { current_hart() };
594    hart.deferred_retirement_tid
595        .store(fault.tid, Ordering::Relaxed);
596    hart.deferred_retirement_cell_id
597        .store(fault.cell_id, Ordering::Relaxed);
598    hart.deferred_retirement_generation
599        .store(fault.generation, Ordering::Relaxed);
600    hart.deferred_retirement_fault_cause
601        .store(fault.cause, Ordering::Relaxed);
602    hart.deferred_retirement_fault_pc
603        .store(fault.pc, Ordering::Relaxed);
604    hart.deferred_retirement_fault_addr
605        .store(fault.fault_addr, Ordering::Relaxed);
606    hart.deferred_retirement_kind
607        .store(DEFERRED_RETIREMENT_FAULT, Ordering::Relaxed);
608    hart.deferred_retirement_pending.store(1, Ordering::Release);
609}

611/// Publish a clean task exit before surrendering the exiting Cell's allocation
612/// attribution.  Root classification and every heap-backed lifecycle action
613/// remain deferred to the scheduler.
614#[inline(always)]
615pub fn defer_exit(exit: DeferredExit) {
616    let hart = unsafe { current_hart() };
617    hart.deferred_retirement_tid
618        .store(exit.tid, Ordering::Relaxed);
619    hart.deferred_retirement_cell_id
620        .store(exit.cell_id, Ordering::Relaxed);
621    hart.deferred_retirement_generation
622        .store(exit.generation, Ordering::Relaxed);
623    hart.deferred_retirement_exit_code
624        .store(exit.code, Ordering::Relaxed);
625    hart.deferred_retirement_kind
626        .store(DEFERRED_RETIREMENT_EXIT, Ordering::Relaxed);
627    hart.deferred_retirement_pending.store(1, Ordering::Release);
628}

630/// Consume this hart's pending retirement record after allocation attribution
631/// has switched to Cell 0 and normal scheduler locking is permitted.
632#[inline(always)]
633pub fn take_deferred_retirement() -> Option<DeferredRetirement> {
634    let hart = unsafe { current_hart() };
635    if hart.deferred_retirement_pending.swap(0, Ordering::AcqRel) == 0 {
636        return None;
637    }
638
639    let tid = hart.deferred_retirement_tid.load(Ordering::Relaxed);
640    let cell_id = hart.deferred_retirement_cell_id.load(Ordering::Relaxed);
641    let generation = hart.deferred_retirement_generation.load(Ordering::Relaxed);
642    match hart.deferred_retirement_kind.load(Ordering::Relaxed) {
643        DEFERRED_RETIREMENT_EXIT => Some(DeferredRetirement::Exit(DeferredExit {
644            tid,
645            cell_id,
646            generation,
647            code: hart.deferred_retirement_exit_code.load(Ordering::Relaxed),
648        })),
649        DEFERRED_RETIREMENT_FAULT => {
650            Some(DeferredRetirement::Fault(DeferredFault::from_user_trap(
651                TrapProvenUserFault::new(),
652                tid,
653                cell_id,
654                generation,
655                hart.deferred_retirement_fault_cause.load(Ordering::Relaxed),
656                hart.deferred_retirement_fault_pc.load(Ordering::Relaxed),
657                hart.deferred_retirement_fault_addr.load(Ordering::Relaxed),
658            )))
659        }
660        kind => panic!("[task] invalid deferred retirement kind {kind}"),
661    }
662}

664/// Update the cell-id attribution for the calling hart.
665#[inline(always)]
666pub fn set_current_cell_id(id: usize) {
667    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
668    {
669        let tp: usize;
670        unsafe {
671            core::arch::asm!("mv {}, tp", out(reg) tp, options(nomem, nostack, preserves_flags));
672        }
673        if tp == 0 {
674            return;
675        }
676        unsafe {
677            (*(tp as *const ViHartLocal))
678                .current_cell_id
679                .store(id, Ordering::Relaxed)
680        };
681    }
682    #[cfg(not(any(target_arch = "riscv64", target_arch = "riscv32")))]
683    {
684        HART_LOCALS[0].current_cell_id.store(id, Ordering::Relaxed);
685    }
686}

688/// Write the `tp` register.
689///
690/// # Safety
691/// Caller is responsible for ensuring `val` is a valid `ViHartLocal` pointer
692/// (or 0 for the pre-install sentinel).  Must run with interrupts disabled or
693/// from boot context where no concurrent trap can misread a partial write.
694#[inline(always)]
695#[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
696pub unsafe fn write_tp(val: usize) {
697    // SAFETY: writing tp CSR is always safe from S-mode; the value is either
698    // a valid HART_LOCALS pointer or 0 (pre-install). Caller ensures context.
699    core::arch::asm!("mv tp, {}", in(reg) val, options(nomem, nostack, preserves_flags));
700}
