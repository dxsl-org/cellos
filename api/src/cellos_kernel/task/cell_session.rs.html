1//! CellSessionV1: scheduler-serialized operation, holder and launch authority.
2//! Rings belong to pipe.rs; Cell identity belongs exclusively to Scheduler.
3//!
4//! Every transition below runs under the one SCHEDULER lock, so death, expiry,
5//! abort, cancel and commit race through a single canonical owner funnel. The
6//! only bookkeeping that is not allocation-free is *reservation*: the final
7//! `commit_child` consumes capacity promised during `PrepareChild`.

42/// Explicit per-principal accounting. Rings charge their creator in `pipe.rs`;
43/// every other record is charged here and refunded exactly once on drop, unless
44/// the principal's Cell has already retired (`forfeit`): its quota row is then
45/// deregistered and its CellId may name a different generation.
46struct Charge { cell: usize, bytes: usize }
47impl Charge {
48    fn reserve(actor: Principal, bytes: usize) -> Result<Self> {
49        if !crate::memory::cell_quota::charge(actor.cell_id as usize, bytes) { return Err(Status::QuotaExceeded); }
50        Ok(Self { cell: actor.cell_id as usize, bytes })
51    }
52    fn forfeit(&mut self, cell: usize) { if self.cell == cell { self.bytes = 0; } }
53}
54impl Drop for Charge { fn drop(&mut self) { crate::memory::cell_quota::refund(self.cell, self.bytes); } }

110/// Kind-6 (InputFrontend) receipt seal: exactly the frontend Ref and the
111/// prepared child. The context's slot-8 grant must name that same frontend.
112fn check_frontend_seal(seal: &[u8], object: Ref, grant: Option<Ref>, child: Child) -> Result<()> {
113    let mut sr = Reader::new(seal);
114    let frontend = sr.reference()?; let sealed = sr.child()?; sr.finish()?;
115    if sealed != child { return Err(Status::InvalidChild); }
116    if frontend == ZERO || frontend != object || grant != Some(frontend) { return Err(Status::Denied); }
117    Ok(())
118}

122impl State {
123    pub(crate) const fn new() -> Self { Self { next: 1, terminal_next: 1,
124        operations: Vec::new(), endpoints: Vec::new(), prepared: Vec::new(), receipts: Vec::new(),
125        bootstraps: Vec::new(), ledgers: Vec::new(), pass: 0, pending_cancels: 0, cascading: false,
126        dead_rings: Vec::new(), dead_overflow: false, drain_operation_cursor: 0, drain_prepared_cursor: 0 } }
127    fn mint(&mut self) -> Result<Ref> {
128        let id = self.next;
129        self.next = id.checked_add(1).ok_or(Status::QuotaExceeded)?;
130        Ok(Ref { id, generation: 1 })
131    }
132    /// Operations, holders, receipts and bootstraps are appended in mint order
133    /// and only ever removed, so each table stays sorted by reference id.
134    fn op_index(&self, reference: Ref) -> Result<usize> {
135        let i = self.operations.binary_search_by_key(&reference.id, |o| o.reference.id).map_err(|_| Status::Stale)?;
136        if self.operations[i].reference == reference { Ok(i) } else { Err(Status::Stale) }
137    }
138    fn op_member(&self, actor: Principal, operation: Ref) -> Result<usize> {
139        let i = self.op_index(operation)?;
140        if self.operations[i].owner != actor && !self.endpoints.iter().any(|h| h.endpoint.holder == actor
141            && h.endpoint.operation == operation && h.endpoint.prepared.is_none() && h.endpoint.state == 0) {
142            return Err(Status::WrongOwner);
143        }
144        Ok(i)
145    }
146    fn op_active(&self, actor: Principal, operation: Ref) -> Result<usize> {
147        let i = self.op_member(actor, operation)?;
148        if self.operations[i].state != 0 { return Err(Status::Cancelled); }
149        Ok(i)
150    }
151    fn endpoint_index(&self, reference: Ref) -> Result<usize> {
152        let i = self.endpoints.binary_search_by_key(&reference.id, |h| h.endpoint.reference.id).map_err(|_| Status::Stale)?;
153        if self.endpoints[i].endpoint.reference == reference { Ok(i) } else { Err(Status::Stale) }
154    }
155    /// Resolve an open, enabled, non-staged endpoint held by `actor` that
156    /// carries every right in `rights` within a live operation.
157    fn endpoint(&self, actor: Principal, reference: Ref, rights: u32) -> Result<Endpoint> {
158        let e = self.endpoints[self.endpoint_index(reference)?].endpoint;
159        if e.holder != actor { return Err(Status::WrongOwner); }
160        if e.prepared.is_some() || !e.enabled { return Err(Status::Denied); }
161        match e.state { 0 => {}, 1 => return Err(Status::Stale), 2 => return Err(Status::Revoked),
162            _ => return Err(Status::Cancelled) }
163        if self.operations[self.op_index(e.operation)?].state != 0 { return Err(Status::Cancelled); }
164        if e.rights & rights != rights {
165            if rights & 3 != 0 && e.rights & 3 & rights == 0 { return Err(Status::WrongDirection); }
166            return Err(Status::Denied);
167        }
168        Ok(e)
169    }
170    fn preparation(&self, reference: Ref) -> Result<usize> {
171        self.prepared.iter().position(|p| p.reference == reference).ok_or(Status::Stale)
172    }
173    fn preparing(&self, actor: Principal, reference: Ref) -> Result<usize> {
174        let i = self.preparation(reference)?;
175        if self.prepared[i].preparer != actor { return Err(Status::WrongOwner); }
176        if self.prepared[i].state != 0 { return Err(Status::InvalidChild); }
177        Ok(i)
178    }
179    fn receipt_index(&self, reference: Ref) -> Result<usize> {
180        let i = self.receipts.binary_search_by_key(&reference.id, |r| r.reference.id).map_err(|_| Status::Stale)?;
181        if self.receipts[i].reference == reference { Ok(i) } else { Err(Status::Stale) }
182    }
183    fn bootstrap_index(&self, reference: Ref) -> Result<usize> {
184        let i = self.bootstraps.binary_search_by_key(&reference.id, |r| r.reference.id).map_err(|_| Status::Stale)?;
185        if self.bootstraps[i].reference == reference { Ok(i) } else { Err(Status::Stale) }
186    }
187    /// At most 64 live holder nodes per operation (retained closed records do
188    /// not count), with the vector capacity reserved before any mutation.
189    fn holder_capacity(&mut self, operation: Ref, count: usize) -> Result<()> {
190        let live = self.endpoints.iter().filter(|h| h.endpoint.operation == operation
191            && (h.endpoint.state == 0 || h.endpoint.state == 4)).count();
192        if live + count > 64 { return Err(Status::QuotaExceeded); }
193        self.endpoints.try_reserve(count).map_err(|_| Status::QuotaExceeded)
194    }
195    #[allow(clippy::too_many_arguments)]
196    fn add_holder(&mut self, charged: Principal, holder: Principal, operation: Ref, handle: PipeHandle,
197        rights: u32, parent: Ref, prepared: Option<Child>) -> Result<Ref> {
198        self.holder_capacity(operation, 1)?;
199        let charge = Charge::reserve(charged, core::mem::size_of::<Holder>())?;
200        let reference = self.mint()?;
201        self.endpoints.push(Holder { endpoint: Endpoint { reference, holder, operation, handle,
202            rights, parent, prepared, enabled: true, state: 0, issuer_held: false }, charge });
203        Ok(reference)
204    }
205    fn create_stream(&mut self, actor: Principal, operation: Ref, capacity: u32) -> Result<[Ref; 2]> {
206        self.op_active(actor, operation)?;
207        if capacity != 0 && !(64..=65536).contains(&capacity) { return Err(Status::Malformed); }
208        self.holder_capacity(operation, 2)?;
209        let first = Charge::reserve(actor, core::mem::size_of::<Holder>())?;
210        let second = Charge::reserve(actor, core::mem::size_of::<Holder>())?;
211        let reader = self.mint()?;
212        let writer = self.mint()?;
213        let (read, write) = pipe::create(capacity as usize, actor.cell_id as usize).map_err(|_| Status::QuotaExceeded)?;
214        for (reference, handle, rights, charge) in [(reader, read, READ|DUPLICATE|DELEGATE|CLOSE, first),
215            (writer, write, WRITE|DUPLICATE|DELEGATE|CLOSE, second)] {
216            self.endpoints.push(Holder { endpoint: Endpoint { reference, holder: actor, operation,
217                parent: ZERO, handle, rights, state: 0, prepared: None, enabled: true, issuer_held: false }, charge });
218        }
219        Ok([reader, writer])
220    }
221    fn is_subtree(&self, child: Ref, root: Ref) -> bool {
222        if child == root { return true; }
223        let (Ok(mut at), Ok(top)) = (self.op_index(child), self.op_index(root)) else { return false; };
224        let target = self.operations[top].depth;
225        while self.operations[at].depth > target {
226            let Ok(parent) = self.op_index(self.operations[at].parent) else { return false; };
227            at = parent;
228        }
229        at == top
230    }
231    /// Revoke one holder node and run any cancellation it causes (see `close_one`).
232    fn close_holder(&mut self, sched: &mut Scheduler, index: usize, state: u32) {
233        self.close_one(sched, index, state);
234        self.cascade(sched);
235    }
236    /// Revoke one holder node. A holder with a ring reference releases it; a
237    /// staged holder (prepared) or reserved peer (state 4) owns none yet. Any
238    /// waiter parked on the ring wakes to recheck its typed state. Losing the
239    /// final consumer of a ring only *records* the ring: the producers on it are
240    /// gathered in one pass by `close_producers`.
241    fn close_one(&mut self, sched: &mut Scheduler, index: usize, state: u32) {
242        let e = self.endpoints[index].endpoint;
243        if e.state == 4 { self.endpoints[index].endpoint.state = state; return; }
244        if e.state != 0 { return; }
245        self.endpoints[index].endpoint.state = state;
246        if e.prepared.is_some() { return; }
247        let final_reader = pipe::session_close(sched, e.handle);
248        pipe::session_wake_all(sched, e.handle);
249        if e.handle.end == PipeEnd::Read && final_reader && !self.dead_overflow {
250            if self.dead_rings.try_reserve(1).is_ok() { self.dead_rings.push(e.handle.pipe); }
251            else { self.dead_overflow = true; }
252        }
253    }
254    /// A holder placed inside a bootstrap's own operation by that bootstrap is
255    /// shared session state (the controller's retained writers): the operation
256    /// holds other sessions and the controller's own holders, so losing the
257    /// consumer closes this holder but never cancels the operation.
258    fn bootstrap_holder(&self, holder: &Endpoint) -> bool {
259        self.bootstraps.iter().any(|b| b.operation == holder.operation && b.endpoints.contains(&holder.reference))
260    }
261    /// One pass over every holder for the rings whose final reader just closed:
262    /// each upstream writer on such a ring is closed, and the operation that
263    /// owns a producer's stage is queued for cancellation. Only the holders on
264    /// those rings and the stages that produce into them are touched; a bootstrap
265    /// root operation and its other sessions are not.
266    fn close_producers(&mut self, sched: &mut Scheduler) {
267        let overflow = core::mem::take(&mut self.dead_overflow);
268        if self.dead_rings.is_empty() && !overflow { return; }
269        let mut rings = core::mem::take(&mut self.dead_rings);
270        rings.sort_unstable();
271        for j in 0..self.endpoints.len() {
272            let producer = self.endpoints[j].endpoint;
273            if producer.state != 0 || producer.handle.end != PipeEnd::Write || producer.issuer_held { continue; }
274            let dead = if overflow {
275                pipe::session_counts(producer.handle).is_none_or(|(readers, _)| readers == 0)
276            } else { rings.binary_search(&producer.handle.pipe).is_ok() };
277            if !dead { continue; }
278            self.close_one(sched, j, 2);
279            if !self.bootstrap_holder(&producer) { self.request_cancel(producer.operation, 3); }
280        }
281        rings.clear();
282        if self.dead_rings.is_empty() { self.dead_rings = rings; }
283    }
284    /// Close without cascading to the producer's operation (bootstrap teardown).
285    fn close_reservation_holder(&mut self, sched: &mut Scheduler, index: usize) {
286        let e = self.endpoints[index].endpoint;
287        if e.state == 4 { self.endpoints[index].endpoint.state = 2; return; }
288        if e.state == 0 {
289            self.endpoints[index].endpoint.state = 2;
290            if e.prepared.is_none() {
291                pipe::session_close(sched, e.handle);
292                pipe::session_wake_all(sched, e.handle);
293            }
294        }
295    }
296    fn terminal_sequence(&mut self) -> u64 {
297        let value = self.terminal_next;
298        if value == 0 { return 0; }
299        self.terminal_next = value.checked_add(1).unwrap_or(0);
300        value
301    }
302    /// Abort or expire an unpublished child. The reservation guards drop here,
303    /// which refunds every creator claim exactly once and releases the owner
304    /// slot, task row, log promise and ready promise without publishing.
305    fn abort_prepared(&mut self, sched: &mut Scheduler, index: usize, expired: bool) {
306        if self.prepared[index].state != 0 { return; }
307        let reference = self.prepared[index].reference;
308        let child = self.prepared[index].child;
309        if let Some(publication) = self.prepared[index].publication.take() {
310            sched.abort_session_owner(publication.owner);
311            drop(publication);
312        }
313        self.prepared[index].state = if expired { 5 } else { 4 };
314        self.prepared[index].reason = if expired { 6 } else { 5 };
315        let sequence = self.terminal_sequence();
316        self.prepared[index].terminal_sequence = sequence;
317        self.prepared[index].context = Vec::new();
318        self.prepared[index].parsed = None;
319        for r in &mut self.receipts {
320            if r.prepared == reference && r.state == 0 { r.state = if expired { 3 } else { 2 }; }
321        }
322        // Reserved issuer peers hold no ring count; releasing them is a state change.
323        for i in 0..self.receipts.len() {
324            if self.receipts[i].prepared == reference {
325                for producer in self.receipts[i].producers {
326                    if let Ok(ei) = self.endpoint_index(producer) { self.close_holder(sched, ei, 2); }
327                }
328            }
329        }
330        for i in 0..self.endpoints.len() {
331            if self.endpoints[i].endpoint.prepared == Some(child) { self.close_holder(sched, i, 2); }
332        }
333    }
334    /// Cancel an operation subtree: mark, revoke every holder, abort staged
335    /// children and flag live committed children for the canonical kill funnel.
336    fn cancel(&mut self, sched: &mut Scheduler, operation: Ref, reason: u32) {
337        self.request_cancel(operation, reason);
338        self.cascade(sched);
339    }
340    /// Queue a subtree cancellation. An operation that is already cancelled has a
341    /// fully cancelled subtree (no child or holder can be added to a cancelled
342    /// operation), so there is nothing left to do for it.
343    fn request_cancel(&mut self, operation: Ref, reason: u32) {
344        let Ok(i) = self.op_index(operation) else { return; };
345        if self.operations[i].state != 0 || self.operations[i].pending != 0 { return; }
346        self.operations[i].pending = reason.max(1);
347        self.pending_cancels += 1;
348    }
349    /// Flat cancellation worklist: consumer loss cascades upstream through
350    /// `close_producers` without recursion, one pass per round.
351    fn cascade(&mut self, sched: &mut Scheduler) {
352        if self.cascading { return; }
353        self.cascading = true;
354        while self.pending_cancels != 0 || !self.dead_rings.is_empty() || self.dead_overflow {
355            for i in 0..self.operations.len() {
356                let reason = self.operations[i].pending;
357                if reason == 0 { continue; }
358                self.operations[i].pending = 0;
359                self.pending_cancels -= 1;
360                if self.operations[i].state == 0 { self.cancel_subtree(sched, i, reason); }
361            }
362            self.close_producers(sched);
363        }
364        self.cascading = false;
365    }
366    /// Cancel the still-live subtree below operation `root`. Operations are kept
367    /// in creation order and a parent always precedes its children, so a single
368    /// forward pass finds every descendant; holders and staged children are then
369    /// matched by their operation's pass mark.
370    fn cancel_subtree(&mut self, sched: &mut Scheduler, root: usize, reason: u32) {
371        self.pass += 1;
372        let pass = self.pass;
373        self.operations[root].state = 1;
374        self.operations[root].cancelled_by = pass;
375        for i in root + 1..self.operations.len() {
376            if self.operations[i].state != 0 || self.operations[i].parent == ZERO { continue; }
377            let parent = self.operations[i].parent;
378            let marked = {
379                let known = &self.operations[root..i];
380                known.binary_search_by_key(&parent.id, |o| o.reference.id).ok()
381                    .is_some_and(|at| known[at].reference == parent && known[at].cancelled_by == pass)
382            };
383            if marked { self.operations[i].state = 1; self.operations[i].cancelled_by = pass; }
384        }
385        for i in 0..self.endpoints.len() {
386            let e = &self.endpoints[i].endpoint;
387            if e.state != 0 && e.state != 4 { continue; }
388            let Ok(oi) = self.op_index(e.operation) else { continue; };
389            if self.operations[oi].cancelled_by == pass { self.close_one(sched, i, 3); }
390        }
391        for i in 0..self.prepared.len() {
392            if self.prepared[i].state > 1 { continue; }
393            let Ok(oi) = self.op_index(self.prepared[i].operation) else { continue; };
394            if self.operations[oi].cancelled_by != pass { continue; }
395            if self.prepared[i].state == 0 { self.abort_prepared(sched, i, false); }
396            else {
397                if self.prepared[i].reason == 0 { self.prepared[i].reason = reason; }
398                self.prepared[i].kill = true;
399            }
400        }
401    }
402    fn forfeit_charges(&mut self, cell: usize) {
403        for o in &mut self.operations { o.charge.forfeit(cell); }
404        for h in &mut self.endpoints { h.charge.forfeit(cell); }
405        for p in &mut self.prepared { p.charge.forfeit(cell); }
406        for r in &mut self.receipts { r.charge.forfeit(cell); }
407        for b in &mut self.bootstraps { b.charge.forfeit(cell); }
408        for l in &mut self.ledgers { for o in &mut l.outcomes { o.charge.forfeit(cell); } }
409    }
410    /// Close a bootstrap: abort the session's committed foreground child, then
411    /// revoke every bootstrap endpoint. `new_state`: 3 aborted, 4 expired, 5 closed.
412    fn close_bootstrap(&mut self, sched: &mut Scheduler, bi: usize, new_state: u32) {
413        let session = self.bootstraps[bi].session;
414        let endpoints = self.bootstraps[bi].endpoints;
415        self.bootstraps[bi].state = new_state;
416        if session != ZERO {
417            for i in 0..self.receipts.len() {
418                let rr = &self.receipts[i];
419                if rr.kind == 1 && rr.state == 1 && !rr.restored && read_ref(&rr.seal, 0).ok() == Some(session) {
420                    let operation = rr.operation;
421                    self.cancel(sched, operation, 2);
422                }
423            }
424        }
425        for reference in endpoints {
426            if let Ok(ei) = self.endpoint_index(reference) { self.close_reservation_holder(sched, ei); }
427        }
428    }

435fn binding(sched: &Scheduler, service: u16) -> Result<Binding> {
436    let (tid, cell, generation) = crate::cell::service_registry::lookup_bound(service).ok_or(Status::Revoked)?;
437    let owner = sched.resolve_live_cell_owner(CellId(cell), generation).ok_or(Status::Revoked)?;
438    let task = sched.tasks.get(&tid).ok_or(Status::Revoked)?;
439    if task.cell_id.0 != owner.cell_id || task.cell_generation != generation { return Err(Status::Revoked); }
440    Ok(Binding::new(tid as u64, cell, generation))
441}
442/// The caller must be the exact currently registered provider of `service`.
443fn issuer(sched: &Scheduler, actor: Principal, service: u16) -> Result<Binding> {
444    let current = binding(sched, service)?;
445    if current.cell_id != actor.cell_id || current.generation != actor.generation { return Err(Status::Denied); }
446    Ok(current)
447}

450    ViError::PermissionDenied => Status::Denied, ViError::NotSupported => Status::Unsupported,
451    ViError::InvalidInput | ViError::InvalidArgument => Status::Malformed, _ => Status::Io } }

453impl State {
454    /// Expiry, owner/issuer death and registry drain, all under the owner lock.
455    /// Every pass is bounded: the common scans are linear in the record tables,
456    /// and the costly per-record work (aborting an unpublished child, probing a
457    /// dead owner's records for drainability) is capped per pass, with the rest
458    /// left for the next dispatch or tick, so no request or timer tick can be made
459    /// to hold SCHEDULER for super-linear time.
460    fn maintenance(&mut self, sched: &mut Scheduler) {
461        let now = super::system_ticks() as u64;
462        let mut aborts = ABORT_BUDGET;
463        for i in 0..self.prepared.len() {
464            if aborts == 0 { break; }
465            if self.prepared[i].state == 0 {
466                let expired = now >= self.prepared[i].deadline;
467                if expired || !owner_live(sched, self.prepared[i].preparer) {
468                    self.abort_prepared(sched, i, expired);
469                    aborts -= 1;
470                }
471            }
472        }
473        let input = binding(sched, api::syscall::service::INPUT).ok();
474        let vfs = binding(sched, api::syscall::service::VFS).ok();
475        for i in 0..self.receipts.len() {
476            let rr = &self.receipts[i];
477            if rr.state > 1 { continue; }
478            let current = if receipt_service(rr.kind) == api::syscall::service::INPUT { input } else { vfs };
479            if current == Some(rr.issuer) { continue; }
480            let (operation, prepared) = (rr.operation, rr.prepared);
481            if let Ok(p) = self.preparation(prepared) {
482                let current_state = self.prepared[p].state;
483                match current_state {
484                    0 if aborts != 0 => { self.abort_prepared(sched, p, false); aborts -= 1; }
485                    1 => self.cancel(sched, operation, 4),
486                    _ => {}
487                }
488            }
489        }
490        for i in 0..self.bootstraps.len() {
491            let b = &self.bootstraps[i];
492            if b.state > 2 { continue; }
493            let invalid = input != Some(b.issuer) || !owner_live(sched, b.controller);
494            let expired = b.state < 2 && now >= b.deadline;
495            if invalid || expired { self.close_bootstrap(sched, i, if invalid { 5 } else { 4 }); }
496        }
497        // Receipts whose issuer is no longer the registered provider are
498        // unreadable by anyone once their child is terminal: drain them so a
499        // restarted issuer cannot leave records pinning the shared caps.
500        let prepared = &self.prepared;
501        self.receipts.retain(|rr| {
502            let current = if receipt_service(rr.kind) == api::syscall::service::INPUT { input } else { vfs };
503            current == Some(rr.issuer)
504                || prepared.iter().find(|p| p.reference == rr.prepared).is_some_and(|p| p.state < 3)
505        });
506        self.drain_dead(sched, input);
507    }
508    /// Creator death drains owned records once nothing depends on them. At most
509    /// `DRAIN_BUDGET` dead-owner records are probed per pass (each probe scans the
510    /// tables), resuming from where the previous pass stopped so none starves.
511    fn drain_dead(&mut self, sched: &Scheduler, input: Option<Binding>) {
512        let mut budget = DRAIN_BUDGET;
513        let total = self.prepared.len();
514        let mut i = self.drain_prepared_cursor;
515        let mut visited = 0;
516        while visited < total && budget != 0 && !self.prepared.is_empty() {
517            if i >= self.prepared.len() { i = 0; }
518            visited += 1;
519            let p = &self.prepared[i];
520            if p.state >= 3 && !owner_live(sched, p.preparer) {
521                budget -= 1;
522                let reference = p.reference;
523                if !self.receipts.iter().any(|r| r.prepared == reference) {
524                    self.prepared.remove(i);
525                    continue;
526                }
527            }
528            i += 1;
529        }
530        self.drain_prepared_cursor = i;
531        // A terminal bootstrap nobody can still read (controller dead, or the
532        // issuer that recorded it replaced) holds no further authority.
533        self.bootstraps.retain(|b| b.state <= 2 || (owner_live(sched, b.controller) && input == Some(b.issuer)));
534        let total = self.operations.len();
535        let mut at = self.operations.partition_point(|o| o.reference.id <= self.drain_operation_cursor);
536        let mut last = self.drain_operation_cursor;
537        let (mut visited, mut budget) = (0, DRAIN_BUDGET);
538        while visited < total && budget != 0 && !self.operations.is_empty() {
539            if at >= self.operations.len() { at = 0; }
540            visited += 1;
541            let reference = self.operations[at].reference;
542            last = reference.id;
543            if !owner_live(sched, self.operations[at].owner) {
544                budget -= 1;
545                if self.drained(reference) {
546                    self.remove_operation(at);
547                    continue;
548                }
549            }
550            at += 1;
551        }
552        self.drain_operation_cursor = last;
553    }
554    /// Remove an operation, its holder records and every request ledger scoped
555    /// to it. Operation ids are never reused, so no fence can reopen.
556    fn remove_operation(&mut self, index: usize) {
557        let reference = self.operations.remove(index).reference;
558        self.endpoints.retain(|h| h.endpoint.operation != reference);
559        self.ledgers.retain(|l| l.operation != reference);
560        // Receipt-less terminal child records are retained for their preparer
561        // (`QueryChild`) until here; nothing else can refer to them.
562        self.prepared.retain(|p| p.operation != reference);
563    }
564    /// A finished child record (terminal 3, aborted 4, expired 5) that no receipt names
565    /// any more: only its preparer can read it (`QueryChild`), and it is released with its
566    /// operation, so it never blocks closing it. Neither a VFS (cwd) receipt nor an
567    /// input foreground receipt pins the record to its issuer once resolved: the issuer's
568    /// resolution removes the receipt, never the preparer's view of the child's exit.
569    fn retained_terminal(&self, p: &Prepared) -> bool {
570        p.state >= 3 && !self.receipts.iter().any(|rr| rr.prepared == p.reference)
571    }
572    fn drained(&self, operation: Ref) -> bool {
573        !self.operations.iter().any(|o| o.parent == operation)
574            && !self.prepared.iter().any(|p| p.operation == operation && !self.retained_terminal(p))
575            && !self.receipts.iter().any(|rr| rr.operation == operation)
576            && !self.bootstraps.iter().any(|b| b.operation == operation && b.state <= 2)
577            && !self.endpoints.iter().any(|h| h.endpoint.operation == operation
578                && (h.endpoint.state == 0 || h.endpoint.state == 4))
579    }
580    /// Correlation scope of a request. A ledgered request must come from a
581    /// principal already authorised in that scope (an existing ledger proves
582    /// it was), checked before any ledger or outcome slot is created, so a
583    /// guessed reference can neither mint a ledger nor consume an outcome.
584    fn request_operation(&self, sched: &Scheduler, actor: Principal, selector: u16, body: &[u8],
585        ledgered: bool) -> Result<Ref> {
586        if matches!(selector, 17 | 26) { return Ok(ZERO); }
587        let reference = Reader::new(body).reference()?;
588        let is_actor = |b: Binding| b.cell_id == actor.cell_id && b.generation == actor.generation;
589        let (operation, authorised) = match selector {
590            // Zero is the correlation scope only for own-principal root
591            // operation creation (and its acknowledgement/query).
592            1 | 2 | 8 | 9 | 10 | 20 | 21 | 25 | 28 => {
593                if reference == ZERO && !matches!(selector, 1 | 25 | 28) { return Err(Status::Malformed); }
594                if !ledgered || reference == ZERO { return Ok(reference); }
595                let i = self.op_index(reference)?;
596                let member = if selector == 8 {
597                    // The owner, or the exact input issuer that holds a receipt on it.
598                    self.operations[i].owner == actor
599                        || self.receipts.iter().any(|rr| rr.operation == reference && is_actor(rr.issuer))
600                } else { self.op_member(actor, reference).is_ok() };
601                (reference, member)
602            }
603            3..=7 | 19 => {
604                let e = self.endpoints[self.endpoint_index(reference)?].endpoint;
605                (e.operation, e.holder == actor)
606            }
607            11..=15 | 18 | 23 | 27 => {
608                let p = &self.prepared[self.preparation(reference)?];
609                let allowed = if selector == 15 {
610                    issuer(sched, actor, api::syscall::service::INPUT).is_ok()
611                        || issuer(sched, actor, api::syscall::service::VFS).is_ok()
612                } else { p.preparer == actor };
613                (p.operation, allowed)
614            }
615            16 | 24 | 29 => {
616                let rr = &self.receipts[self.receipt_index(reference)?];
617                (rr.operation, is_actor(rr.issuer))
618            }
619            30 => {
620                let child = Reader::new(body).child()?;
621                let p = &self.prepared[self.preparation(child.prepared_ref())?];
622                (p.operation, p.preparer == actor)
623            }
624            22 => {
625                let b = &self.bootstraps[self.bootstrap_index(reference)?];
626                (b.operation, b.controller == actor || is_actor(b.issuer))
627            }
628            _ => return Err(Status::UnknownOperation),
629        };
630        if ledgered && !authorised && self.ledger_index(actor, operation).is_none() {
631            return Err(Status::WrongOwner);
632        }
633        Ok(operation)
634    }
635    fn ledger_index(&self, actor: Principal, operation: Ref) -> Option<usize> {
636        self.ledgers.iter().position(|l| l.actor == actor && l.operation == operation)
637    }
638    fn ledger(&mut self, actor: Principal, operation: Ref) -> Result<usize> {
639        if let Some(i) = self.ledger_index(actor, operation) { return Ok(i); }
640        if self.ledgers.len() >= MAX_LEDGERS
641            || self.ledgers.iter().filter(|l| l.actor == actor).count() >= MAX_LEDGERS_PER_ACTOR {
642            return Err(Status::QuotaExceeded);
643        }
644        self.ledgers.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
645        self.ledgers.push(Ledger { actor, operation, next: 1, acknowledged: 0, outcomes: Vec::new() });
646        Ok(self.ledgers.len()-1)
647    }
648    /// Scope: authenticated sender generation x operation. A repeated ID replays
649    /// its exact retained outcome; only the next monotonic ID is new, and a
650    /// charged outcome record is reserved before any effect.
651    fn begin_request(&mut self, actor: Principal, operation: Ref, header: RequestHeaderV1,
652        request: &[u8], result: &mut [u8]) -> Result<Begin> {
653        let li = self.ledger(actor, operation)?;
654        let ledger = &mut self.ledgers[li];
655        if let Some(outcome) = ledger.outcomes.iter_mut().find(|o| o.request_id == header.request_id) {
656            if outcome.request != request { return Err(Status::Malformed); }
657            if !outcome.complete { return Err(Status::WouldBlock); }
658            result[..outcome.result.len()].copy_from_slice(&outcome.result);
659            outcome.observed = true;
660            return Ok(Begin::Replay(outcome.result.len()));
661        }
662        if header.request_id != ledger.next { return Err(Status::UnknownRequest); }
663        let next = ledger.next.checked_add(1).ok_or(Status::QuotaExceeded)?;
664        if ledger.outcomes.len() >= MAX_OUTCOMES { return Err(Status::QuotaExceeded); }
665        ledger.outcomes.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
666        let charge = Charge::reserve(actor, request.len() + RESULT_MAX + core::mem::size_of::<Outcome>())?;
667        let mut saved = Vec::new();
668        saved.try_reserve_exact(request.len()).map_err(|_| Status::QuotaExceeded)?;
669        saved.extend_from_slice(request);
670        let mut response = Vec::new();
671        response.try_reserve_exact(RESULT_MAX).map_err(|_| Status::QuotaExceeded)?;
672        ledger.outcomes.push(Outcome { request_id: header.request_id, request: saved, result: response,
673            complete: false, observed: false, charge });
674        ledger.next = next;
675        Ok(Begin::New)
676    }
677    /// Completion records the result; it is not yet *observed*: the caller has
678    /// only seen it once the copy-out succeeded (`delivered`), a replay, or a
679    /// QueryRequest.
680    fn finish_request(&mut self, actor: Principal, operation: Ref, id: u64, result: &[u8]) {
681        let Some(li) = self.ledger_index(actor, operation) else { return; };
682        if let Some(o) = self.ledgers[li].outcomes.iter_mut().find(|o| o.request_id == id) {
683            if result.len() <= o.result.capacity() { o.result.extend_from_slice(result); }
684            o.complete = true;
685        }
686    }
687    /// The reply bytes of exactly this completed request reached the caller.
688    fn mark_observed(&mut self, actor: Principal, request: &[u8], id: u64) {
689        for l in self.ledgers.iter_mut().filter(|l| l.actor == actor) {
690            for o in l.outcomes.iter_mut() {
691                if o.request_id == id && o.complete && o.request == request { o.observed = true; }
692            }
693        }
694    }
695    /// True while `principal` is a published child still (or just) alive; its
696    /// legacy stdio is session-routed and never falls back to the console.
697    fn session_child(&self, principal: Principal) -> bool {
698        self.prepared.iter().any(|p| p.child.principal == principal && (1..=3).contains(&p.state))
699    }

705impl State {
706    /// QueryRequest: the original result bytes, never a re-execution.
707    fn query_request(&mut self, actor: Principal, body: &[u8], out: &mut [u8]) -> Result<usize> {
708        let mut r = Reader::new(body);
709        let operation = r.reference()?; let id = r.u64()?; r.finish()?;
710        let li = self.ledger_index(actor, operation).ok_or(Status::UnknownRequest)?;
711        let o = self.ledgers[li].outcomes.iter_mut().find(|o| o.request_id == id).ok_or(Status::UnknownRequest)?;
712        if !o.complete { return Err(Status::WouldBlock); }
713        out[..o.result.len()].copy_from_slice(&o.result);
714        o.observed = true;
715        Ok(o.result.len() | RAW)
716    }
717    fn execute(&mut self, sched: &mut Scheduler, actor: Principal, selector: u16,
718        body: &[u8], out: &mut [u8]) -> Result<usize> {
719        if selector == 28 { return self.query_request(actor, body, out); }
720        let mut r = Reader::new(body);
721        let mut w = Writer::new(out);
722        match selector {
723            1 => {
724                let parent = r.reference()?; r.finish()?;
725                let mut depth = 0;
726                if parent != ZERO {
727                    let pi = self.op_active(actor, parent)?;
728                    depth = self.operations[pi].depth.checked_add(1).ok_or(Status::QuotaExceeded)?;
729                    if self.operations[pi].owner != actor && !self.endpoints.iter().any(|h|
730                        h.endpoint.holder == actor && h.endpoint.operation == parent && h.endpoint.state == 0
731                        && h.endpoint.rights & DELEGATE != 0 && h.endpoint.prepared.is_none()) {
732                        return Err(Status::Denied);
733                    }
734                    if self.operations.iter().filter(|o| o.parent == parent).count() >= 32 {
735                        return Err(Status::QuotaExceeded);
736                    }
737                }
738                if self.operations.len() >= 512 || self.operations.iter().filter(|o| o.owner == actor).count() >= 64 {
739                    return Err(Status::QuotaExceeded);
740                }
741                let charge = Charge::reserve(actor, core::mem::size_of::<Operation>())?;
742                self.operations.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
743                let reference = self.mint()?;
744                self.operations.push(Operation { reference, owner: actor, parent, state: 0, charge,
745                    depth, cancelled_by: 0, pending: 0 });
746                w.reference(reference)?;
747            }
748            2 => {
749                let operation = r.reference()?; let capacity = r.u32()?; zero(r.u32()?)?; r.finish()?;
750                let [reader, writer] = self.create_stream(actor, operation, capacity)?;
751                w.reference(reader)?; w.reference(writer)?;
752            }
753            3 | 4 => {
754                let source = r.reference()?;
755                let target = if selector == 4 { Some(r.child()?) } else { None };
756                let rights = r.u32()?; zero(r.u32()?)?; r.finish()?;
757                if rights == 0 || rights & !31 != 0 { return Err(Status::Malformed); }
758                let e = self.endpoint(actor, source, if selector == 3 { DUPLICATE } else { DELEGATE })?;
759                if rights & !e.rights != 0 || rights & 3 != e.rights & 3 { return Err(Status::Denied); }
760                let (holder, operation) = if let Some(child) = target {
761                    let pi = self.preparing(actor, child.prepared_ref())?;
762                    if self.prepared[pi].child != child || !self.is_subtree(self.prepared[pi].operation, e.operation) {
763                        return Err(Status::InvalidChild);
764                    }
765                    (child.principal, self.prepared[pi].operation)
766                } else { (actor, e.operation) };
767                let reference = self.add_holder(actor, holder, operation, e.handle, rights, source, target)?;
768                if target.is_none() && !pipe::duplicate_end(e.handle) {
769                    self.endpoints.pop();
770                    return Err(Status::Stale);
771                }
772                w.reference(reference)?;
773            }
774            7 => {
775                let reference = r.reference()?; r.finish()?;
776                let i = self.endpoint_index(reference)?;
777                let e = self.endpoints[i].endpoint;
778                if e.holder != actor { return Err(Status::WrongOwner); }
779                if e.rights & CLOSE == 0 || e.prepared.is_some() { return Err(Status::Denied); }
780                self.close_holder(sched, i, 1);
781            }
782            8 => {
783                let operation = r.reference()?; let reason = r.u32()?; zero(r.u32()?)?; r.finish()?;
784                let i = self.op_index(operation)?;
785                if self.operations[i].owner != actor {
786                    // The exact registered input issuer may cut the committed
787                    // foreground child's own operation, and nothing else.
788                    let input = issuer(sched, actor, api::syscall::service::INPUT)?;
789                    if !self.receipts.iter().any(|rr| rr.kind == 1 && rr.state == 1 && !rr.restored
790                        && rr.operation == operation && rr.issuer == input
791                        && self.preparation(rr.prepared).ok().is_some_and(|p| self.prepared[p].state == 1)) {
792                        return Err(Status::WrongOwner);
793                    }
794                }
795                self.cancel(sched, operation, if reason == 3 { 3 } else { 2 });
796            }
797            9 => {
798                let operation = r.reference()?; r.finish()?;
799                // Members, and the exact input issuer probing a frontend
800                // holder's operation lifetime (read-only).
801                let i = match self.op_member(actor, operation) {
802                    Err(Status::WrongOwner) if issuer(sched, actor, api::syscall::service::INPUT).is_ok() => self.op_index(operation)?,
803                    other => other?,
804                };
805                w.u32(self.operations[i].state)?;
806                w.u32(self.endpoints.iter().filter(|h| h.endpoint.operation == operation && h.endpoint.state == 0).count() as u32)?;
807                w.u32(self.prepared.iter().filter(|p| p.operation == operation && p.state <= 2).count() as u32)?;
808                w.u32(0)?;
809            }
810            11 => self.write_context(sched, actor, r, &mut w)?,
811            12 => {
812                let reference = r.reference()?; let kind = r.u16()?; let slot = r.u16()?; zero(r.u32()?)?;
813                let owner = r.binding()?; let receipt = r.reference()?; r.finish()?;
814                let pi = self.preparing(actor, reference)?;
815                let ri = self.receipt_index(receipt)?;
816                let rr = &self.receipts[ri];
817                if rr.prepared != reference || rr.kind != kind || rr.slot != slot || rr.issuer != owner || rr.state != 0 {
818                    return Err(Status::Denied);
819                }
820                let index = match kind { 1 => 0, 2 => 1, GRANT_KIND_INPUT_FRONTEND => 2, _ => return Err(Status::Unsupported) };
821                // The receipt must still be the exact live issuer's.
822                if binding(sched, receipt_service(kind))? != owner { return Err(Status::Revoked); }
823                if self.prepared[pi].receipts[index] != ZERO && self.prepared[pi].receipts[index] != receipt { return Err(Status::Denied); }
824                self.prepared[pi].receipts[index] = receipt;
825                self.receipts[ri].bound = true;
826            }
827            13 => {
828                let reference = r.reference()?; r.finish()?;
829                let pi = self.preparing(actor, reference)?;
830                self.commit_child(sched, pi)?;
831                w.child(self.prepared[pi].child)?;
832            }
833            14 => {
834                let reference = r.reference()?; r.finish()?;
835                let pi = self.preparation(reference)?;
836                if self.prepared[pi].preparer != actor { return Err(Status::WrongOwner); }
837                if (1..=3).contains(&self.prepared[pi].state) { return Err(Status::InvalidChild); }
838                self.abort_prepared(sched, pi, false);
839            }
840            15 => self.register_receipt(sched, actor, r, &mut w)?,
841            16 => {
842                let reference = r.reference()?; let action = r.u32()?; zero(r.u32()?)?; r.finish()?;
843                let ri = self.receipt_index(reference)?;
844                let kind = self.receipts[ri].kind;
845                if issuer(sched, actor, receipt_service(kind))? != self.receipts[ri].issuer { return Err(Status::WrongOwner); }
846                if action > 2 { return Err(Status::Malformed); }
847                if action == 1 && self.receipts[ri].state == 0 {
848                    let pi = self.preparation(self.receipts[ri].prepared)?;
849                    self.abort_prepared(sched, pi, false);
850                }
851                let (state, child, producers) = {
852                    let rr = &self.receipts[ri];
853                    (rr.state, rr.child, if rr.state == 1 { rr.producers } else { [ZERO; 2] })
854                };
855                if action == 2 {
856                    self.receipts[ri].materialized = true;
857                    let operation = self.receipts[ri].operation;
858                    self.reclaim(operation);
859                }
860                w.u32(state)?; w.u32(0)?; w.child(child)?;
861                for producer in producers { w.reference(producer)?; }
862            }
863            17 => {
864                let offset = r.u32()? as usize; let max = r.u32()? as usize; r.finish()?;
865                if !(1..=512).contains(&max) { return Err(Status::Malformed); }
866                let owner = sched.resolve_live_cell_owner(CellId(actor.cell_id), actor.generation).ok_or(Status::Stale)?;
867                let bytes = sched.tasks.get(&(owner.root_tid as usize)).and_then(|t| t.child_context.as_ref()).ok_or(Status::Denied)?;
868                if offset > bytes.len() { return Err(Status::Malformed); }
869                let count = max.min(bytes.len()-offset);
870                w.u32(count as u32)?; w.u32(bytes.len() as u32)?; w.bytes(&bytes[offset..offset+count])?;
871            }
872            18 => self.inspect_grant(sched, actor, r, &mut w)?,
873            19 => {
874                let reference = r.reference()?; r.finish()?;
875                let e = self.endpoints[self.endpoint_index(reference)?].endpoint;
876                if e.holder != actor && issuer(sched, actor, api::syscall::service::INPUT).is_err() { return Err(Status::WrongOwner); }
877                w.principal(e.holder)?; w.reference(e.operation)?; w.u32(e.rights)?;
878                // A reserved issuer peer is not yet a revoked node.
879                w.u32(if e.state == 4 { 0 } else { e.state })?;
880                w.u32(if e.handle.end == PipeEnd::Read { 0 } else { 1 })?; w.u32(0)?;
881            }
882            20 => {
883                let operation = r.reference()?; r.finish()?;
884                let i = self.op_index(operation)?;
885                if self.operations[i].owner != actor { return Err(Status::WrongOwner); }
886                if self.endpoints.iter().any(|h| h.endpoint.operation == operation
887                        && (h.endpoint.state == 0 || h.endpoint.state == 4))
888                    || self.operations.iter().any(|o| o.parent == operation)
889                    || self.prepared.iter().any(|p| p.operation == operation && !self.retained_terminal(p))
890                    || self.receipts.iter().any(|rr| rr.operation == operation)
891                    || self.bootstraps.iter().any(|b| b.operation == operation && b.state <= 2) { return Err(Status::WouldBlock); }
892                self.remove_operation(i);
893            }
894            21 => self.bootstrap(sched, actor, r, &mut w)?,
895            22 => self.resolve_bootstrap(sched, actor, r, &mut w)?,
896            23 => {
897                let reference = r.reference()?; r.finish()?;
898                issuer(sched, actor, api::syscall::service::VFS)?;
899                let p = &self.prepared[self.preparation(reference)?];
900                if p.state != 0 { return Err(Status::InvalidChild); }
901                w.child(p.child)?; w.reference(p.operation)?; w.principal(p.preparer)?;
902            }
903            24 => {
904                let reference = r.reference()?; r.finish()?;
905                let rr = &self.receipts[self.receipt_index(reference)?];
906                if issuer(sched, actor, receipt_service(rr.kind))? != rr.issuer { return Err(Status::WrongOwner); }
907                let p = &self.prepared[self.preparation(rr.prepared)?];
908                w.child(p.child)?; w.u32(p.state)?; w.u32(p.reason)?; w.i32(p.exit_code)?; w.u32(0)?;
909                w.u64(p.terminal_sequence)?;
910            }
911            30 => {
912                let child = r.child()?; r.finish()?;
913                let p = &self.prepared[self.preparation(child.prepared_ref())?];
914                if p.preparer != actor { return Err(Status::WrongOwner); }
915                if p.child != child { return Err(Status::Stale); }
916                if p.state == 0 { return Err(Status::InvalidChild); }
917                w.u32(p.state)?; w.u32(p.reason)?; w.i32(p.exit_code)?; w.u32(0)?;
918                w.u64(p.terminal_sequence)?;
919            }
920            25 => {
921                let operation = r.reference()?; let through = r.u64()?; r.finish()?;
922                let li = self.ledger_index(actor, operation).ok_or(Status::UnknownRequest)?;
923                let l = &mut self.ledgers[li];
924                if through <= l.acknowledged { return Ok(0); }
925                if through >= l.next || l.outcomes.iter().any(|o| o.request_id <= through && (!o.complete || !o.observed)) {
926                    return Err(Status::UnknownRequest);
927                }
928                l.outcomes.retain(|o| o.request_id > through);
929                l.acknowledged = through;
930                self.reclaim(operation);
931            }
932            26 => {
933                r.finish()?;
934                let rr = self.receipts.iter().find(|rr| rr.kind == 1 && rr.state == 1 && !rr.restored
935                    && rr.child.principal == actor).ok_or(Status::Denied)?;
936                w.reference(read_ref(&rr.seal, 0)?)?; w.reference(read_ref(&rr.seal, 16)?)?;
937                w.u64(rr.next_epoch)?; w.reference(read_ref(&rr.seal, 144)?)?; w.binding(rr.issuer)?;
938            }
939            27 => {
940                let reference = r.reference()?; let input = r.binding()?;
941                let writers = [r.reference()?, r.reference()?]; r.finish()?;
942                let pi = self.preparing(actor, reference)?;
943                if binding(sched, api::syscall::service::INPUT)? != input { return Err(Status::Revoked); }
944                if self.prepared[pi].parsed.is_none() || self.prepared[pi].intent.is_some() || writers[0] == writers[1] {
945                    return Err(Status::Denied);
946                }
947                let mut rings = [0usize; 2];
948                for (n, writer) in writers.iter().enumerate() {
949                    let e = self.endpoint(actor, *writer, WRITE|DELEGATE)?;
950                    if e.operation != self.prepared[pi].operation || pipe::session_counts(e.handle).map(|p| p.1) != Some(1) {
951                        return Err(Status::Denied);
952                    }
953                    rings[n] = e.handle.pipe;
954                }
955                if rings[0] == rings[1] { return Err(Status::Denied); }
956                let intent = self.mint()?;
957                self.prepared[pi].intent = Some(Intent { reference: intent, issuer: input, writers });
958                w.reference(intent)?;
959            }
960            29 => self.restore(sched, actor, r, &mut w)?,
961            _ => return Err(Status::UnknownOperation),
962        }
963        Ok(w.position())
964    }
965    /// WriteChildContext: strictly sequential, no holes, no overwrite. The
966    /// completed context is validated once; any invalid final context aborts the
967    /// preparation (no byte of it can be rewritten).
968    fn write_context(&mut self, sched: &mut Scheduler, actor: Principal, mut r: Reader<'_>, w: &mut Writer<'_>) -> Result<()> {
969        let reference = r.reference()?; let offset = r.u32()? as usize; let count = r.u32()? as usize;
970        let bytes = r.take(count)?; r.finish()?;
971        if !(1..=512).contains(&count) { return Err(Status::Malformed); }
972        let p = self.preparing(actor, reference)?;
973        let end = offset.checked_add(count).ok_or(Status::Malformed)?;
974        if offset != self.prepared[p].written || end > self.prepared[p].context.len() { return Err(Status::Malformed); }
975        self.prepared[p].context[offset..end].copy_from_slice(bytes);
976        self.prepared[p].written = end;
977        if end == self.prepared[p].context.len() {
978            if let Err(status) = self.install_context(p) {
979                self.abort_prepared(sched, p, false);
980                return Err(status);
981            }
982        }
983        w.u32(end as u32)?; w.u32(0)
984    }
985    fn install_context(&mut self, p: usize) -> Result<()> {
986        let parsed = child_context::validate(&self.prepared[p].context, self.prepared[p].operation)?;
987        let argv = &self.prepared[p].context[parsed.argv.clone()];
988        let argv = if argv.is_empty() { None } else {
989            let mut copy = Vec::new();
990            copy.try_reserve_exact(argv.len()).map_err(|_| Status::QuotaExceeded)?;
991            copy.extend_from_slice(argv);
992            Some(copy)
993        };
994        let cwd = core::str::from_utf8(&self.prepared[p].context[parsed.cwd.clone()]).map_err(|_| Status::UnsupportedInput)?;
995        let mut cwd_copy = alloc::string::String::new();
996        cwd_copy.try_reserve_exact(cwd.len()).map_err(|_| Status::QuotaExceeded)?;
997        cwd_copy.push_str(cwd);
998        self.prepared[p].parsed = Some(parsed);
999        // Staged kernel grants must already exist, match and be delegated by
1000        // this preparer; an invalid grant is never silently dropped.
1001        self.staged_grants(p)?;
1002        let mut stdio = [ZERO; 4];
1003        {
1004            let parsed = self.prepared[p].parsed.as_ref().ok_or(Status::InvalidChild)?;
1005            for (slot, reference) in stdio.iter_mut().enumerate() {
1006                *reference = parsed.grants[slot].map_or(ZERO, |g| g.object);
1007            }
1008        }
1009        let publication = self.prepared[p].publication.as_mut().ok_or(Status::InvalidChild)?;
1010        publication.task.inherited_argv = argv;
1011        publication.task.cwd = cwd_copy;
1012        publication.task.child_stdio = stdio;
1013        Ok(())
1014    }
1015    /// Every kernel-endpoint grant (kind 1: stdio slots 0..=3 and the slot-5
1016    /// completion writer) must name a holder staged for this child by its
1017    /// preparer's delegation; every staged holder must be named.
1018    fn staged_grants(&self, pi: usize) -> Result<[Option<usize>; child_context::MAX_GRANTS]> {
1019        let p = &self.prepared[pi];
1020        let parsed = p.parsed.as_ref().ok_or(Status::WouldBlock)?;
1021        let mut staged = [None; child_context::MAX_GRANTS];
1022        for slot in 0..child_context::MAX_GRANTS {
1023            let Some(grant) = parsed.grants[slot] else { continue };
1024            if grant.kind != 1 { continue; }
1025            let ei = self.endpoint_index(grant.object)?;
1026            let e = self.endpoints[ei].endpoint;
1027            if e.holder != p.child.principal || e.prepared != Some(p.child) || e.state != 0
1028                || e.operation != p.operation || e.rights != grant.rights as u32 { return Err(Status::Denied); }
1029            let source = self.endpoint(p.preparer, e.parent, DELEGATE)?;
1030            if source.handle != e.handle || e.rights & !source.rights != 0 { return Err(Status::Denied); }
1031            staged[slot] = Some(ei);
1032        }
1033        if self.endpoints.iter().any(|h| h.endpoint.prepared == Some(p.child) && h.endpoint.state == 0
1034            && !staged.iter().flatten().any(|i| self.endpoints[*i].endpoint.reference == h.endpoint.reference)) {
1035            return Err(Status::Denied);
1036        }
1037        Ok(staged)
1038    }

1042    fn inspect_grant(&self, sched: &Scheduler, actor: Principal, mut r: Reader<'_>, w: &mut Writer<'_>) -> Result<()> {
1043        let reference = r.reference()?; let kind = r.u16()?; let slot = r.u16()?;
1044        let offset = r.u32()? as usize; let max = r.u32()? as usize; zero(r.u32()?)?; r.finish()?;
1045        if !(1..=512).contains(&max) { return Err(Status::Malformed); }
1046        let p = &self.prepared[self.preparation(reference)?];
1047        if p.state != 0 { return Err(Status::InvalidChild); }
1048        let parsed = p.parsed.as_ref().ok_or(Status::WouldBlock)?;
1049        let stdio;
1050        let mut identity = [0u8; 32];
1051        let (grant, projection): (&child_context::Grant, &[u8]) = match (kind, slot) {
1052            (1, 0) => {
1053                let intent = p.intent.ok_or(Status::Denied)?;
1054                if issuer(sched, actor, api::syscall::service::INPUT)? != intent.issuer { return Err(Status::Denied); }
1055                stdio = parsed.stdio_projection()?;
1056                (parsed.grants[0].as_ref().ok_or(Status::Denied)?, &stdio)
1057            }
1058            (2, 4) => {
1059                issuer(sched, actor, api::syscall::service::VFS)?;
1060                (parsed.grants[4].as_ref().ok_or(Status::Denied)?, &p.context[parsed.cwd.clone()])
1061            }
1062            (GRANT_KIND_INPUT_FRONTEND, SLOT_FRONTEND) => {
1063                // Frontend binding: the input issuer proves the child is prepared
1064                // by the frontend holder under an operation the holder owns.
1065                issuer(sched, actor, api::syscall::service::INPUT)?;
1066                let grant = parsed.grants[SLOT_FRONTEND as usize].as_ref().ok_or(Status::Denied)?;
1067                let owner = self.operations[self.op_index(p.operation)?].owner;
1068                identity[..16].copy_from_slice(&p.preparer.to_bytes());
1069                identity[16..].copy_from_slice(&owner.to_bytes());
1070                (grant, &identity)
1071            }
1072            _ => return Err(Status::Denied),
1073        };
1074        if offset > projection.len() { return Err(Status::Malformed); }
1075        let count = max.min(projection.len()-offset);
1076        w.child(p.child)?; w.reference(p.operation)?; w.bytes(&parsed.digest)?; w.bytes(&grant.wire)?;
1077        w.u32(projection.len() as u32)?; w.u32(offset as u32)?; w.u32(count as u32)?; w.u32(0)?;
1078        w.bytes(&projection[offset..offset+count])
1079    }
1080    fn register_receipt(&mut self, sched: &Scheduler, actor: Principal, mut r: Reader<'_>, w: &mut Writer<'_>) -> Result<()> {
1081        let reference = r.reference()?; let kind = r.u16()?; let slot = r.u16()?; zero(r.u32()?)?;
1082        let object = r.reference()?; let expected_epoch = r.u64()?; let next_epoch = r.u64()?;
1083        let digest: [u8; 32] = r.take(32)?.try_into().map_err(|_| Status::Malformed)?;
1084        let seal_len = r.u32()? as usize; zero(r.u32()?)?;
1085        let seal = r.take(seal_len)?; r.finish()?;
1086        let pi = self.preparation(reference)?;
1087        let p = &self.prepared[pi];
1088        if p.state != 0 { return Err(Status::InvalidChild); }
1089        let parsed = p.parsed.as_ref().ok_or(Status::WouldBlock)?;
1090        if digest != parsed.digest { return Err(Status::Denied); }
1091        if self.receipts.iter().any(|rr| rr.prepared == reference && rr.kind == kind && rr.state <= 1) {
1092            return Err(Status::Denied);
1093        }
1094        let service = match (kind, slot, seal_len) {
1095            (1, 0, 192) => api::syscall::service::INPUT,
1096            (2, 4, 128) => api::syscall::service::VFS,
1097            (GRANT_KIND_INPUT_FRONTEND, SLOT_FRONTEND, FRONTEND_SEAL_LEN) => api::syscall::service::INPUT,
1098            _ => return Err(Status::Malformed),
1099        };
1100        let issuer = issuer(sched, actor, service)?;
1101        if kind == 1 {
1102            let intent = p.intent.ok_or(Status::Denied)?;
1103            if intent.issuer != issuer { return Err(Status::Denied); }
1104            let mut sr = Reader::new(seal);
1105            let session = sr.reference()?; let lease = sr.reference()?; let operation = sr.reference()?;
1106            let child = sr.child()?; let expected = sr.u64()?; let next = sr.u64()?;
1107            let streams = [sr.reference()?, sr.reference()?, sr.reference()?, sr.reference()?];
1108            let writers = [sr.reference()?, sr.reference()?]; sr.finish()?;
1109            if session == ZERO || lease != object || child != p.child || operation != p.operation
1110                || expected != expected_epoch || next != next_epoch || next != expected.checked_add(1).ok_or(Status::WrongEpoch)?
1111                || writers != intent.writers { return Err(Status::Denied); }
1112            // The seal's session must be the bootstrap-bound SessionRef, and the
1113            // child's operation must descend from the bootstrap's root through
1114            // kernel-owned operation links, never a payload claim.
1115            let b = self.bootstraps.iter().find(|b| b.state == 2 && b.session == session && b.issuer == issuer
1116                && b.controller == p.preparer).ok_or(Status::WrongSession)?;
1117            if !self.is_subtree(p.operation, b.operation) || p.operation == b.operation { return Err(Status::Denied); }
1118            // Foreground exclusivity and monotonic epochs per session.
1119            for rr in self.receipts.iter().filter(|rr| rr.kind == 1 && read_ref(&rr.seal, 0).ok() == Some(session)) {
1120                if (rr.state == 0 || (rr.state == 1 && !rr.restored)) && rr.prepared != reference {
1121                    return Err(Status::WrongEpoch);
1122                }
1123                if rr.state == 1 && expected < rr.next_epoch { return Err(Status::WrongEpoch); }
1124            }
1125            for (slot, stream) in streams.iter().enumerate() {
1126                let grant = parsed.grants[slot].ok_or(Status::Denied)?;
1127                let endpoint = self.endpoints[self.endpoint_index(*stream)?].endpoint;
1128                if grant.object != *stream || endpoint.holder != p.child.principal || endpoint.prepared != Some(p.child)
1129                    || endpoint.operation != p.operation || endpoint.rights != grant.rights as u32 || endpoint.state != 0 {
1130                    return Err(Status::Denied);
1131                }
1132            }
1133            let ring = |reference| -> Result<usize> {
1134                Ok(self.endpoints[self.endpoint_index(reference)?].endpoint.handle.pipe)
1135            };
1136            // stdout/stderr continue on the session's rings; stdin/control are
1137            // new exclusive rings distinct from the baseline and every earlier epoch.
1138            if ring(streams[1])? != ring(b.endpoints[1])? || ring(streams[2])? != ring(b.endpoints[2])?
1139                || ring(streams[0])? == ring(streams[3])?
1140                || [streams[0], streams[3]].iter().any(|new| [b.endpoints[0], b.endpoints[3]]
1141                    .iter().any(|old| ring(*new).ok() == ring(*old).ok())) {
1142                return Err(Status::WrongSession);
1143            }
1144            for previous in self.receipts.iter().filter(|rr| rr.kind == 1 && read_ref(&rr.seal, 0).ok() == Some(session)) {
1145                for offset in [96, 144] {
1146                    if let Ok(old) = read_ref(&previous.seal, offset).and_then(ring) {
1147                        if old == ring(streams[0])? || old == ring(streams[3])? { return Err(Status::WrongEpoch); }
1148                    }
1149                }
1150            }
1151            for (n, writer) in writers.iter().enumerate() {
1152                let e = self.endpoint(p.preparer, *writer, WRITE|DELEGATE)?;
1153                let target = self.endpoints[self.endpoint_index(streams[if n == 0 { 0 } else { 3 }])?].endpoint;
1154                if e.handle.pipe != target.handle.pipe || e.operation != p.operation
1155                    || pipe::session_counts(e.handle) != Some((1, 1)) { return Err(Status::Denied); }
1156            }
1157        } else if kind == GRANT_KIND_INPUT_FRONTEND {
1158            // No producer peers and no session side effect: the receipt only
1159            // admits the slot-8 grant of this exact child at CommitChild.
1160            if expected_epoch != 0 || next_epoch != 0 { return Err(Status::WrongEpoch); }
1161            check_frontend_seal(seal, object, parsed.grants[SLOT_FRONTEND as usize].map(|g| g.object), p.child)?;
1162            // One live receipt per frontend: a frontend binds at most one child.
1163            if self.receipts.iter().any(|rr| rr.kind == GRANT_KIND_INPUT_FRONTEND && rr.object == object
1164                && rr.issuer == issuer && rr.state <= 1) { return Err(Status::Denied); }
1165        } else {
1166            if expected_epoch != 0 || next_epoch != 0 { return Err(Status::WrongEpoch); }
1167            let mut sr = Reader::new(seal);
1168            let source = sr.principal()?; let source_grant = sr.reference()?; let child = sr.child()?;
1169            let operation = sr.reference()?; let cwd_digest = sr.take(32)?; let child_grant = sr.reference()?; sr.finish()?;
1170            let grant = parsed.grants[4].ok_or(Status::Denied)?;
1171            if child_grant.generation != child.principal.generation { return Err(Status::InvalidChild); }
1172            if source != p.preparer || source_grant == ZERO || child != p.child || operation != p.operation
1173                || child_grant != grant.object || object != child_grant
1174                || cwd_digest != crate::sha256::sha256(&p.context[parsed.cwd.clone()]) {
1175                return Err(Status::Denied);
1176            }
1177        }
1178        if self.receipts.len() >= 1024 { return Err(Status::QuotaExceeded); }
1179        let (child, operation, preparer, intent) = (p.child, p.operation, p.preparer, p.intent);
1180        self.receipts.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
1181        let charge = Charge::reserve(preparer, core::mem::size_of::<Receipt>())?;
1182        let mut seal_bytes = [0u8; 256];
1183        seal_bytes[..seal_len].copy_from_slice(seal);
1184        let receipt = self.mint()?;
1185        let mut producers = [ZERO; 2];
1186        if kind == 1 {
1187            // The issuer's producer peers are reserved now, so commit can move
1188            // the sole writer count to them without allocating or failing.
1189            self.holder_capacity(operation, 2)?;
1190            let charges = [Charge::reserve(preparer, core::mem::size_of::<Holder>())?,
1191                Charge::reserve(preparer, core::mem::size_of::<Holder>())?];
1192            let writers = intent.ok_or(Status::Denied)?.writers;
1193            let sources = [self.endpoint(preparer, writers[0], WRITE|DELEGATE)?,
1194                self.endpoint(preparer, writers[1], WRITE|DELEGATE)?];
1195            producers = [self.mint()?, self.mint()?];
1196            for (n, holder_charge) in charges.into_iter().enumerate() {
1197                self.endpoints.push(Holder { endpoint: Endpoint { reference: producers[n],
1198                    holder: Principal { cell_id: issuer.cell_id, generation: issuer.generation },
1199                    operation, parent: writers[n], handle: sources[n].handle, rights: WRITE|CLOSE,
1200                    state: 4, prepared: None, enabled: false, issuer_held: true }, charge: holder_charge });
1201            }
1202        }
1203        self.receipts.push(Receipt { reference: receipt, prepared: reference, child, operation, issuer,
1204            kind, slot, object, expected_epoch, next_epoch, digest, seal: seal_bytes, seal_len,
1205            state: 0, bound: false, producers, restored: false, materialized: false, charge });
1206        w.reference(receipt)
1207    }
1208}

1210impl State {
1211    /// Final child publication. Every check precedes the first mutation, and
1212    /// the mutation sequence below allocates nothing and cannot fail: holder
1213    /// storage, ring counts, owner slot, task row, CSpace row, measurement
1214    /// entry and ready slot were all reserved during preparation.
1215    fn commit_child(&mut self, sched: &mut Scheduler, pi: usize) -> Result<()> {
1216        let (parent, operation, reference, deadline, intent) = {
1217            let p = &self.prepared[pi];
1218            (p.preparer, p.operation, p.reference, p.deadline, p.intent)
1219        };
1220        self.op_active(parent, operation)?;
1221        if super::system_ticks() as u64 >= deadline {
1222            self.abort_prepared(sched, pi, true);
1223            return Err(Status::TimedOut);
1224        }
1225        // A receipt registered for this child but never bound by the preparer
1226        // would outlive the commit as a stale "Prepared" record and keep the
1227        // issuer reading a running child as pending: refuse the commit.
1228        if self.receipts.iter().any(|rr| rr.prepared == reference && rr.state == 0 && !rr.bound) {
1229            return Err(Status::Denied);
1230        }
1231        let (digest, has_cwd, has_stdio, frontend) = {
1232            let parsed = self.prepared[pi].parsed.as_ref().ok_or(Status::WouldBlock)?;
1233            (parsed.digest, parsed.grants[4].is_some(), (0..4).all(|slot| parsed.grants[slot].is_some()),
1234                parsed.grants[SLOT_FRONTEND as usize].map(|g| g.object))
1235        };
1236        // A frontend grant is admitted only with its own bound, unresolved
1237        // kind-6 receipt for this exact child and frontend object.
1238        if let Some(object) = frontend {
1239            let child = self.prepared[pi].child;
1240            let named = self.prepared[pi].receipts[2];
1241            if !self.receipt_index(named).ok().is_some_and(|ri| {
1242                let rr = &self.receipts[ri];
1243                rr.kind == GRANT_KIND_INPUT_FRONTEND && rr.prepared == reference && rr.child == child
1244                    && rr.object == object && rr.bound && rr.state == 0
1245            }) { return Err(Status::InvalidChild); }
1246        }
1247        if !self.prepared[pi].publication.as_ref().is_some_and(|p| p.current(sched)) { return Err(Status::Revoked); }
1248        let mut receipt_indices: [Option<usize>; 3] = [None; 3];
1249        for n in 0..3 {
1250            let named = self.prepared[pi].receipts[n];
1251            if named == ZERO { continue; }
1252            let ri = self.receipt_index(named)?;
1253            let rr = &self.receipts[ri];
1254            let service = if n == 1 { api::syscall::service::VFS } else { api::syscall::service::INPUT };
1255            if rr.prepared != reference || !rr.bound || rr.state != 0 || rr.digest != digest
1256                || binding(sched, service)? != rr.issuer { return Err(Status::Revoked); }
1257            receipt_indices[n] = Some(ri);
1258        }
1259        if has_cwd && receipt_indices[1].is_none() { return Err(Status::Denied); }
1260        let input = receipt_indices[0];
1261        if input.is_some() && !has_stdio { return Err(Status::Denied); }
1262        let staged = self.staged_grants(pi)?;
1263        // Reader grants are moves: the preparer's source reader is consumed so
1264        // the projected post-commit reader set is exactly the child.
1265        let mut consumed: [Option<usize>; 2] = [None; 2];
1266        let mut reader_rings = [0usize; 2];
1267        for (slot, staged_index) in staged.iter().enumerate() {
1268            let Some(staged_index) = staged_index else { continue };
1269            if slot != 0 && slot != 3 { continue; }
1270            let endpoint = self.endpoints[*staged_index].endpoint;
1271            if pipe::session_counts(endpoint.handle).map(|c| c.0) != Some(1) { return Err(Status::Denied); }
1272            let n = if slot == 0 { 0 } else { 1 };
1273            consumed[n] = Some(self.endpoint_index(endpoint.parent)?);
1274            reader_rings[n] = endpoint.handle.pipe;
1275        }
1276        if consumed[0].is_some() && consumed[1].is_some() && reader_rings[0] == reader_rings[1] {
1277            return Err(Status::Denied);
1278        }
1279        let mut producers = [ZERO; 2];
1280        let mut writer_indices: [Option<usize>; 2] = [None; 2];
1281        if let Some(ri) = input {
1282            let intent = intent.ok_or(Status::Denied)?;
1283            if intent.issuer != self.receipts[ri].issuer { return Err(Status::Revoked); }
1284            for n in 0..2 {
1285                let source = self.endpoint(parent, intent.writers[n], WRITE|DELEGATE)?;
1286                let target_index = staged[if n == 0 { 0 } else { 3 }].ok_or(Status::Denied)?;
1287                let target = self.endpoints[target_index].endpoint;
1288                if source.operation != operation || source.handle.pipe != target.handle.pipe
1289                    || pipe::session_counts(source.handle) != Some((1, 1)) { return Err(Status::Denied); }
1290                writer_indices[n] = Some(self.endpoint_index(intent.writers[n])?);
1291            }
1292            producers = self.receipts[ri].producers;
1293            for producer in producers {
1294                let ei = self.endpoint_index(producer)?;
1295                if self.endpoints[ei].endpoint.state != 4 { return Err(Status::Revoked); }
1296            }
1297        }
1298        // ---- No fallible step below this line. ----
1299        if input.is_some() {
1300            for n in 0..2 {
1301                let ei = self.endpoint_index(producers[n]).expect("validated reserved peer");
1302                self.endpoints[ei].endpoint.state = 0;
1303                self.endpoints[ei].endpoint.enabled = true;
1304                // The sole writer count moves to the issuer; it is not duplicated.
1305                self.endpoints[writer_indices[n].expect("validated source writer")].endpoint.state = 1;
1306            }
1307        }
1308        for (slot, staged_index) in staged.into_iter().enumerate() {
1309            let Some(index) = staged_index else { continue };
1310            if slot == 0 || slot == 3 {
1311                self.endpoints[consumed[if slot == 0 { 0 } else { 1 }].expect("validated source reader")].endpoint.state = 1;
1312            } else {
1313                assert!(pipe::duplicate_end(self.endpoints[index].endpoint.handle),
1314                    "validated child writer ring vanished");
1315            }
1316            self.endpoints[index].endpoint.prepared = None;
1317        }
1318        for ri in receipt_indices.into_iter().flatten() {
1319            self.receipts[ri].state = 1;
1320            if self.receipts[ri].kind == 1 { self.receipts[ri].producers = producers; }
1321        }
1322        let mut publication = self.prepared[pi].publication.take().expect("reserved publication");
1323        publication.task.child_context = Some(core::mem::take(&mut self.prepared[pi].context));
1324        self.prepared[pi].state = 1;
1325        publication.publish(sched);
1326        Ok(())
1327    }
1328}

1331    fn bootstrap(&mut self, sched: &mut Scheduler, actor: Principal, mut r: Reader<'_>, w: &mut Writer<'_>) -> Result<()> {
1332        let operation = r.reference()?; let owner = r.binding()?; let capacity = r.u32()?; zero(r.u32()?)?; r.finish()?;
1333        let oi = self.op_active(actor, operation)?;
1334        if self.operations[oi].owner != actor || binding(sched, api::syscall::service::INPUT)? != owner {
1335            return Err(Status::WrongOwner);
1336        }
1337        if self.bootstraps.len() >= 256 || self.bootstraps.iter().filter(|b| b.state <= 2).count() >= 64
1338            || self.bootstraps.iter().filter(|b| b.controller == actor && b.state <= 2).count() >= 8 {
1339            return Err(Status::QuotaExceeded);
1340        }
1341        let (reference, endpoints) = self.install_bootstrap(sched, actor, operation, owner, capacity)?;
1342        w.reference(reference)?;
1343        for endpoint in &endpoints[..6] { w.reference(*endpoint)?; }
1344        Ok(())
1345    }
1346    /// Create the bootstrap record and its four rings, every precondition having
1347    /// been checked by the caller.
1348    fn install_bootstrap(&mut self, sched: &mut Scheduler, actor: Principal, operation: Ref, owner: Binding,
1349        capacity: u32) -> Result<(Ref, [Ref; 8])> {
1350        self.holder_capacity(operation, 8)?;
1351        self.bootstraps.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
1352        let charge = Charge::reserve(actor, core::mem::size_of::<Bootstrap>())?;
1353        let reference = self.mint()?;
1354        let deadline = (super::system_ticks() as u64).checked_add(PREPARE_TICKS).ok_or(Status::QuotaExceeded)?;
1355        let mut streams = [[ZERO; 2]; 4];
1356        for n in 0..4 {
1357            match self.create_stream(actor, operation, capacity) {
1358                Ok(pair) => streams[n] = pair,
1359                Err(error) => {
1360                    for pair in streams {
1361                        for created in pair {
1362                            if let Ok(ei) = self.endpoint_index(created) { self.close_reservation_holder(sched, ei); }
1363                        }
1364                    }
1365                    return Err(error);
1366                }
1367            }
1368        }
1369        // stdin reader, stdout writer, stderr writer, control reader, frontend
1370        // output/diagnostic readers, and issuer-only producer peers.
1371        let endpoints = [streams[0][0], streams[1][1], streams[2][1], streams[3][0],
1372            streams[1][0], streams[2][0], streams[0][1], streams[3][1]];
1373        for (n, endpoint) in endpoints.iter().enumerate() {
1374            let ei = self.endpoint_index(*endpoint)?;
1375            let e = &mut self.endpoints[ei].endpoint;
1376            e.enabled = false;
1377            e.rights = match n { 0 | 3 => READ|CLOSE, 1 | 2 => WRITE|DUPLICATE|DELEGATE|CLOSE,
1378                4 | 5 => READ|DUPLICATE|CLOSE, _ => WRITE|CLOSE };
1379            if n >= 6 {
1380                e.holder = Principal { cell_id: owner.cell_id, generation: owner.generation };
1381                e.issuer_held = true;
1382            }
1383        }
1384        self.bootstraps.push(Bootstrap { reference, controller: actor, operation, issuer: owner,
1385            endpoints, session: ZERO, state: 0, deadline, charge });
1386        Ok((reference, endpoints))
1387    }
1388    fn resolve_bootstrap(&mut self, sched: &mut Scheduler, actor: Principal, mut r: Reader<'_>, w: &mut Writer<'_>) -> Result<()> {
1389        let reference = r.reference()?; let action = r.u32()?; zero(r.u32()?)?; let session = r.reference()?; r.finish()?;
1390        let bi = self.bootstrap_index(reference)?;
1391        let (controller, recorded, state) = {
1392            let b = &self.bootstraps[bi];
1393            (b.controller, b.issuer, b.state)
1394        };
1395        let is_issuer = issuer(sched, actor, api::syscall::service::INPUT).ok() == Some(recorded);
1396        if actor != controller && !is_issuer { return Err(Status::WrongOwner); }
1397        if action > 3 || (action != 3 && session != ZERO) { return Err(Status::Malformed); }
1398        match action {
1399            0 => {},
1400            1 => {
1401                if !is_issuer { return Err(Status::Denied); }
1402                if state == 0 { self.bootstraps[bi].state = 1; }
1403                else if state != 1 { return Err(Status::Stale); }
1404            }
1405            2 => {
1406                // Controller or issuer: abort a reserved bootstrap, or close an
1407                // active one (cancelling its committed foreground child).
1408                if state <= 2 { self.close_bootstrap(sched, bi, if state == 2 { 5 } else { 3 }); }
1409            }
1410            3 => {
1411                if !is_issuer || session == ZERO || state != 1 { return Err(Status::Denied); }
1412                if self.bootstraps.iter().any(|b| b.state == 2 && b.session == session && b.issuer == recorded) {
1413                    return Err(Status::WrongSession);
1414                }
1415                // Resolve every endpoint before the first mutation: nothing may
1416                // fail once the bootstrap is active.
1417                let mut indices = [0usize; 8];
1418                for (n, endpoint) in self.bootstraps[bi].endpoints.iter().enumerate() {
1419                    indices[n] = self.endpoint_index(*endpoint)?;
1420                }
1421                self.bootstraps[bi].session = session;
1422                self.bootstraps[bi].state = 2;
1423                for ei in indices { self.endpoints[ei].endpoint.enabled = true; }
1424            }
1425            _ => return Err(Status::Malformed),
1426        }
1427        let b = &self.bootstraps[bi];
1428        w.u32(b.state)?; w.u32(0)?; w.principal(b.controller)?; w.reference(b.operation)?;
1429        for (n, endpoint) in b.endpoints.iter().enumerate() {
1430            w.reference(if n >= 6 && !is_issuer { ZERO } else { *endpoint })?;
1431        }
1432        Ok(())
1433    }
1434    /// RestoreSessionInput: exact input issuer, exact canonical terminal child.
1435    /// Validates both baseline rings before any close or discard, then closes the
1436    /// child's producer peers and discards stale baseline bytes in one transition.
1437    fn restore(&mut self, sched: &mut Scheduler, actor: Principal, mut r: Reader<'_>, w: &mut Writer<'_>) -> Result<()> {
1438        let reference = r.reference()?; let terminal = r.u64()?; let expected_epoch = r.u64()?; r.finish()?;
1439        let ri = self.receipt_index(reference)?;
1440        let rr = &self.receipts[ri];
1441        if rr.kind != 1 || rr.state != 1 || issuer(sched, actor, api::syscall::service::INPUT)? != rr.issuer {
1442            return Err(Status::WrongOwner);
1443        }
1444        let (prepared, next_epoch, rr_issuer, producers, restored) = (rr.prepared, rr.next_epoch, rr.issuer, rr.producers, rr.restored);
1445        let session = read_ref(&rr.seal, 0)?;
1446        let p = &self.prepared[self.preparation(prepared)?];
1447        if p.state != 3 || terminal == 0 || p.terminal_sequence != terminal { return Err(Status::InvalidChild); }
1448        let preparer = p.preparer;
1449        if next_epoch != expected_epoch { return Err(Status::WrongEpoch); }
1450        let bi = self.bootstraps.iter().position(|b| b.state == 2 && b.session == session && b.issuer == rr_issuer
1451            && b.controller == preparer).ok_or(Status::WrongSession)?;
1452        if self.receipts.iter().any(|other| other.kind == 1 && other.state == 1 && !other.restored
1453            && other.reference != reference && read_ref(&other.seal, 0).ok() == Some(session)
1454            && other.next_epoch > expected_epoch) { return Err(Status::WrongEpoch); }
1455        let baseline = [self.bootstraps[bi].endpoints[0], self.bootstraps[bi].endpoints[3]];
1456        if !restored {
1457            let controller = self.bootstraps[bi].controller;
1458            let handles = [self.endpoint(controller, baseline[0], READ)?.handle,
1459                self.endpoint(controller, baseline[1], READ)?.handle];
1460            for handle in handles { if pipe::session_counts(handle).is_none() { return Err(Status::Stale); } }
1461            for producer in producers {
1462                if let Ok(ei) = self.endpoint_index(producer) { self.close_holder(sched, ei, 1); }
1463            }
1464            for handle in handles { pipe::session_flush(sched, handle); }
1465            self.receipts[ri].restored = true;
1466        }
1467        w.reference(baseline[0])?; w.reference(baseline[1])
1468    }
1469    /// Release terminal records of `operation` that nothing can still refer to.
1470    /// A retained outcome keeps exactly the record named by its request: a replay
1471    /// resolves that record to find its correlation scope, so removing it would
1472    /// turn the retained result into `Stale`. Every other terminal record is
1473    /// reclaimed regardless of unrelated outstanding or in-flight outcomes.
1474    fn reclaim(&mut self, operation: Ref) {
1475        let ledgers = &self.ledgers;
1476        let outcomes = ledgers.iter().any(|l| l.operation == operation && !l.outcomes.is_empty());
1477        let pinned = |reference: Ref| outcomes && ledgers.iter().any(|l| l.operation == operation
1478            && l.outcomes.iter().any(|o| o.request.get(HEADER_LEN..HEADER_LEN + 16)
1479                .is_some_and(|named| Ref::from_bytes(named).ok() == Some(reference))));
1480        self.receipts.retain(|rr| rr.operation != operation || !rr.materialized
1481            || (rr.state == 1 && !rr.restored && rr.kind == 1)
1482            || self.prepared.iter().any(|p| p.reference == rr.prepared && p.state <= 2)
1483            || pinned(rr.reference));
1484        // A terminal (exited) record stays readable by its preparer (`QueryChild`) even
1485        // after the receipts bound to it (VFS cwd, input foreground) were resolved and
1486        // removed: a controller reads its foreground child's exit status there, and the
1487        // input owner may restore and acknowledge before the controller asks.
1488        // `remove_operation` releases it. An aborted or expired record is dropped here as
1489        // soon as no receipt names it.
1490        self.prepared.retain(|p| p.operation != operation || p.state <= 2
1491            || self.receipts.iter().any(|rr| rr.prepared == p.reference) || pinned(p.reference)
1492            || p.state == 3);
1493        self.endpoints.retain(|h| h.endpoint.operation != operation || h.endpoint.state == 0
1494            || h.endpoint.state == 4 || pinned(h.endpoint.reference));
1495        self.bootstraps.retain(|b| b.operation != operation || b.state <= 2 || pinned(b.reference));
1496    }
1497}

1499/// Temporarily borrow canonical session bookkeeping so its transitions can
1500/// update the scheduler without any second owner lock or recursive acquisition.
1501/// Bookkeeping heap is kernel memory; per-principal charges are explicit.
1502fn with_state<T>(sched: &mut Scheduler, f: impl FnOnce(&mut State, &mut Scheduler) -> T) -> T {
1503    let _kernel = KernelAllocationContext::enter();
1504    let mut state = core::mem::replace(&mut sched.cell_session, State::new());
1505    let result = f(&mut state, sched);
1506    sched.cell_session = state;
1507    result
1508}
1509/// Root retirement began: the lifetime record is no longer live. Terminal
1510/// sequence is minted later, by `retired`, after resource teardown.
1511pub(crate) fn retiring(sched: &mut Scheduler, owner: api::cell_owner::CellOwner, exit_reason: usize) {
1512    with_state(sched, |state, sched| {
1513        let principal = Principal { cell_id: owner.cell_id, generation: owner.generation };
1514        for i in 0..state.prepared.len() {
1515            if state.prepared[i].child.principal == principal && state.prepared[i].state == 1 {
1516                state.prepared[i].state = 2;
1517                if state.prepared[i].reason == 0 {
1518                    state.prepared[i].reason = if exit_reason == usize::MAX { 1 } else { 0 };
1519                    state.prepared[i].exit_code = if exit_reason == usize::MAX { 0 } else { exit_reason as i32 };
1520                }
1521            }
1522            if state.prepared[i].preparer == principal && state.prepared[i].state == 0 {
1523                state.abort_prepared(sched, i, false);
1524            }
1525        }
1526        for i in 0..state.operations.len() {
1527            if state.operations[i].owner == principal {
1528                let operation = state.operations[i].reference;
1529                state.cancel(sched, operation, 2);
1530            }
1531        }
1532        for i in 0..state.endpoints.len() {
1533            if state.endpoints[i].endpoint.holder == principal { state.close_holder(sched, i, 1); }
1534        }
1535        state.maintenance(sched);
1536    });
1537}
1538/// Root retirement completed (owner slot empty, quota still registered): mint
1539/// the canonical terminal sequence and retire everything charged to the dead
1540/// principal so a reused CellId can never be refunded or charged for it.
1541pub(crate) fn retired(sched: &mut Scheduler, owner: api::cell_owner::CellOwner) {
1542    with_state(sched, |state, _| {
1543        for i in 0..state.prepared.len() {
1544            let p = &state.prepared[i];
1545            if p.child.principal.cell_id == owner.cell_id && p.child.principal.generation == owner.generation && p.state == 2 {
1546                let terminal = state.terminal_sequence();
1547                state.prepared[i].state = 3;
1548                state.prepared[i].terminal_sequence = terminal;
1549            }
1550        }
1551        state.forfeit_charges(owner.cell_id as usize);
1552        state.ledgers.retain(|l| l.actor.cell_id != owner.cell_id || l.actor.generation != owner.generation);
1553    });
1554    pipe::disown_cell(owner.cell_id as usize);
1555}
1556pub(crate) fn sweep(sched: &mut Scheduler) {
1557    with_state(sched, |state, sched| state.maintenance(sched));
1558    loop {
1559        let tid = with_state(sched, |state, _| {
1560            for p in state.prepared.iter_mut() {
1561                if p.kill {
1562                    p.kill = false;
1563                    if p.state == 1 { return Some(p.child.root_tid as usize); }
1564                }
1565            }
1566            None
1567        });
1568        match tid { Some(tid) => sched.exit_task(tid, usize::MAX), None => break }
1569    }
1570}

1572fn mutating(selector: u16, body: &[u8]) -> bool {
1573    if matches!(selector, 9 | 17 | 18 | 19 | 23 | 24 | 25 | 26 | 28 | 30) { return false; }
1574    if matches!(selector, 16 | 22) {
1575        return body.get(16..20).and_then(|b| b.try_into().ok()).map(u32::from_le_bytes) != Some(0);
1576    }
1577    true
1578}
1579/// Largest success record for `selector`; reserved before any mutation so a
1580/// short result buffer can never leave an outcome ambiguous.
1581pub(crate) fn required_result(selector: u16, body: &[u8]) -> usize {
1582    HEADER_LEN + match selector {
1583        1 | 3 | 4 | 15 | 27 => 16, 2 | 13 | 29 => 32, 10 => 48,
1584        5 => 8 + body.get(16..20).and_then(|b| b.try_into().ok()).map(u32::from_le_bytes).unwrap_or(512).min(512) as usize,
1585        6 | 11 => 8, 9 => 16, 16 => 72, 17 => 520, 18 => 648, 19 => 48,
1586        21 => 112, 22 => 168, 23 => 64, 24 => 56, 26 => 80, 28 => RESULT_MAX-HEADER_LEN, 30 => 24,
1587        _ => 0,
1588    }
1589}
1590fn encode_result(header: RequestHeaderV1, status: Status, payload: usize, out: &mut [u8]) -> usize {
1591    let len = HEADER_LEN + payload;
1592    ResultHeaderV1 { selector: header.selector, status, total_len: len as u32,
1593        request_id: header.request_id }.encode(out).expect("bounded result header");
1594    len
1595}
1596fn render(header: RequestHeaderV1, outcome: Result<usize>, result: &mut [u8]) -> usize {
1597    match outcome {
1598        Ok(n) if n & RAW != 0 => {
1599            let len = n & !RAW;
1600            result.copy_within(HEADER_LEN..HEADER_LEN + len, 0);
1601            len
1602        }
1603        Ok(n) => encode_result(header, Status::Ok, n, result),
1604        Err(status) => encode_result(header, status, 0, result),
1605    }
1606}
1607/// One CellSessionV1 transaction. Returns the byte length written to `result`.
1608/// Selectors that block (5, 6) or load an artifact (10) run their slow part
1609/// outside the owner lock; every other selector is a single locked transition.
1610pub(crate) fn dispatch(caller: usize, request: &[u8], result: &mut [u8]) -> usize {
1611    let header = match RequestHeaderV1::decode(request) {
1612        Ok(header) => header,
1613        Err(status) => {
1614            // Header-only syntax rejections may echo a zero request ID.
1615            let selector = request.get(4..6).and_then(|b| b.try_into().ok()).map(u16::from_le_bytes).unwrap_or(0);
1616            let request_id = request.get(16..24).and_then(|b| b.try_into().ok()).map(u64::from_le_bytes).unwrap_or(0);
1617            ResultHeaderV1 { selector, request_id, status, total_len: HEADER_LEN as u32 }.encode(result).ok();
1618            return HEADER_LEN;
1619        }
1620    };
1621    let body = &request[HEADER_LEN..];
1622    let selector = header.selector;
1623    let ledgered = mutating(selector, body);
1624    let (principal, operation) = {
1625        let _kernel = KernelAllocationContext::enter();
1626        let mut guard = super::SCHEDULER.lock();
1627        let Some(sched) = guard.as_mut() else { return encode_result(header, Status::Denied, 0, result); };
1628        // Death, expiry and kill propagation run first: a caller retired by
1629        // this sweep must not complete a request after its retirement.
1630        sweep(sched);
1631        let principal = match resolve_actor(sched, caller) {
1632            Ok(principal) => principal,
1633            Err(status) => return encode_result(header, status, 0, result),
1634        };
1635        let operation = match sched.cell_session.request_operation(&*sched, principal, selector, body, ledgered) {
1636            Ok(operation) => operation,
1637            Err(status) => return encode_result(header, status, 0, result),
1638        };
1639        if ledgered {
1640            match sched.cell_session.begin_request(principal, operation, header, request, result) {
1641                Ok(Begin::New) => {}
1642                Ok(Begin::Replay(len)) => return len,
1643                Err(status) => return encode_result(header, status, 0, result),
1644            }
1645        }
1646        if !matches!(selector, 5 | 6 | 10) {
1647            let outcome = with_state(sched, |state, sched| {
1648                state.execute(sched, principal, selector, body, &mut result[HEADER_LEN..])
1649            });
1650            let len = render(header, outcome, result);
1651            if ledgered { sched.cell_session.finish_request(principal, operation, header.request_id, &result[..len]); }
1652            sweep(sched);
1653            return len;
1654        }
1655        (principal, operation)
1656    };
1657    let outcome = if selector == 10 {
1658        prepare_child(caller, principal, operation, body, &mut result[HEADER_LEN..])
1659    } else {
1660        stream_io(caller, principal, selector, body, &mut result[HEADER_LEN..])
1661    };
1662    let len = render(header, outcome, result);
1663    let _kernel = KernelAllocationContext::enter();
1664    let mut guard = super::SCHEDULER.lock();
1665    if let Some(sched) = guard.as_mut() {
1666        if ledgered { sched.cell_session.finish_request(principal, operation, header.request_id, &result[..len]); }
1667        sweep(sched);
1668    }
1669    len
1670}
1671/// Context staging bytes are kernel memory: allocated and released under the
1672/// kernel accounting identity, with the creator's explicit `Charge` as the quota.
1673struct KernelBytes(Vec<u8>);
1674impl Drop for KernelBytes {
1675    fn drop(&mut self) {
1676        let _kernel = KernelAllocationContext::enter();
1677        drop(core::mem::take(&mut self.0));
1678    }

1680fn prepare_child(caller: usize, principal: Principal, operation: Ref, body: &[u8], out: &mut [u8]) -> Result<usize> {
1681    let deadline = (super::system_ticks() as u64).checked_add(PREPARE_TICKS).ok_or(Status::QuotaExceeded)?;
1682    let mut r = Reader::new(body);
1683    if r.reference()? != operation { return Err(Status::Malformed); }
1684    let route = r.u32()?; let path_len = r.u32()? as usize; let context_len = r.u32()? as usize; zero(r.u32()?)?;
1685    let path = r.take(path_len)?; r.finish()?;
1686    if route != 1 || !(1..=4096).contains(&path_len) || path.contains(&0)
1687        || !(48..=child_context::MAX_LEN).contains(&context_len) { return Err(Status::Malformed); }
1688    let path = core::str::from_utf8(path).map_err(|_| Status::Malformed)?;
1689    {

1704    }

1706    let (prepared, state) = request.prepare_session_path(path).map_err(vi)?;
1707    if super::system_ticks() as u64 >= deadline { return Err(Status::TimedOut); }
1708    let _kernel = KernelAllocationContext::enter();
1709    let mut guard = super::SCHEDULER.lock(); let sched = guard.as_mut().ok_or(Status::Revoked)?;
1710    if resolve_actor(sched, caller)? != principal { return Err(Status::Revoked); }
1711    sched.cell_session.op_active(principal, operation)?;
1712    if sched.cell_session.prepared.iter().any(|p| p.operation == operation && p.state == 0)
1713        || sched.cell_session.prepared.iter().filter(|p| p.preparer == principal && p.state == 0).count()
1714            >= MAX_STAGING_PER_PRINCIPAL {
1715        return Err(Status::QuotaExceeded);
1716    }
1717    sched.cell_session.prepared.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
1718    let publication = super::launch::SessionPublication::reserve(sched, prepared, state).map_err(vi)?;
1719    let child = Child { principal: Principal { cell_id: publication.owner.cell_id, generation: publication.owner.generation },
1720        root_tid: publication.owner.root_tid, publication_generation: publication.owner.generation };
1721    let reference = child.prepared_ref();
1722    sched.cell_session.prepared.push(Prepared { reference, child, preparer: principal, operation,
1723        deadline, written: 0, context: core::mem::take(&mut context.0), parsed: None, publication: Some(publication),
1724        state: 0, reason: 0, exit_code: 0, terminal_sequence: 0, kill: false,
1725        intent: None, receipts: [ZERO; 3], charge });
1726    let mut w = Writer::new(out); w.reference(reference)?; w.child(child)?;
1727    Ok(w.position())
1728}

1730/// ReadStream / WriteStream. Wait registration, endpoint checks, cancellation and
1731/// readiness are one SCHEDULER-serialized check-and-park; a wake rechecks typed
1732/// state, so timeout is never confused with EOF.
1733fn stream_io(caller: usize, principal: Principal, selector: u16, body: &[u8], out: &mut [u8]) -> Result<usize> {
1734    let mut r = Reader::new(body); let reference = r.reference()?; let count = r.u32()? as usize;
1735    zero(r.u32()?)?; let deadline_ns = r.u64()?;
1736    if !(1..=512).contains(&count) { return Err(Status::Malformed); }
1737    let bytes = if selector == 6 { r.take(count)? } else { &[] }; r.finish()?;
1738    loop {
1739        let mut guard = super::SCHEDULER.lock(); let sched = guard.as_mut().ok_or(Status::Revoked)?;
1740        if resolve_actor(sched, caller)? != principal { return Err(Status::Revoked); }
1741        let e = sched.cell_session.endpoint(principal, reference, if selector == 5 { READ } else { WRITE })?;
1742        let mut scratch = [0u8; 512];
1743        let progress = if selector == 5 {
1744            match pipe::session_read(sched, e.handle, &mut scratch[..count]).ok_or(Status::Stale)? {
1745                pipe::ReadOutcome::Read(n) => Some(n), pipe::ReadOutcome::Eof => return Err(Status::Eof),
1746                pipe::ReadOutcome::Empty => None,
1747            }
1748        } else {
1749            match pipe::session_write(sched, e.handle, bytes).ok_or(Status::Stale)? {
1750                pipe::WriteOutcome::Wrote(n) => Some(n), pipe::WriteOutcome::NoReader => return Err(Status::BrokenPipe),
1751                pipe::WriteOutcome::Full => None,
1752            }
1753        };
1754        if let Some(n) = progress {
1755            let mut w = Writer::new(out); w.u32(n as u32)?; w.u32(0)?;
1756            if selector == 5 { w.bytes(&scratch[..n])?; }
1757            return Ok(w.position());
1758        }
1759        if deadline_ns == 0 { return Err(Status::WouldBlock); }
1760        // Absolute monotonic ns -> scheduler ticks, rounded up plus one tick so
1761        // the wake can never precede the requested deadline.
1762        let wait = if deadline_ns == u64::MAX { None } else {
1763            let now = super::telemetry::monotonic_ns().ok_or(Status::Unsupported)?;
1764            if now >= deadline_ns { return Err(Status::TimedOut); }
1765            let ticks = (deadline_ns - now).div_ceil(TICK_NS);
1766            Some((super::system_ticks() as u64).checked_add(ticks).and_then(|t| t.checked_add(1)).ok_or(Status::Malformed)?)
1767        };
1768        if selector == 5 { pipe::park_reader(e.handle, caller); } else { pipe::park_writer(e.handle, caller); }
1769        let task = sched.tasks.get_mut(&caller).ok_or(Status::Revoked)?;
1770        task.state = if selector == 5 { super::tcb::TaskState::PipeRead { handle: e.handle, deadline: wait } }
1771            else { super::tcb::TaskState::PipeWrite { handle: e.handle, deadline: wait } };
1772        drop(guard);
1773        super::yield_cpu();
1774    }
1775}

1777/// Legacy fd 0/1/2 of a published session child, as
1778/// `Some(Ok((principal, endpoint)))`; `None` only for a task that is not a
1779/// session child, which keeps the unchanged console path. A published child
1780/// never falls back to the console, even if its root row is already gone.
1781fn stdio_ref(caller: usize, fd: usize) -> Option<Result<(Principal, Ref)>> {
1782    let guard = super::SCHEDULER.lock();
1783    let sched = guard.as_ref()?;
1784    let task = sched.tasks.get(&caller)?;
1785    let principal = Principal { cell_id: task.cell_id.0, generation: task.cell_generation };
1786    let root = sched.tasks.get(&task.root_tid);
1787    let has_context = root.is_some_and(|root| root.child_context.is_some());
1788    if !has_context && !sched.cell_session.session_child(principal) { return None; }
1789    let Some(root) = root.filter(|_| has_context) else { return Some(Err(Status::Revoked)); };
1790    if sched.resolve_live_cell_owner(task.cell_id, task.cell_generation).is_none() {
1791        return Some(Err(Status::Revoked));
1792    }
1793    let reference = root.child_stdio.get(fd).copied().unwrap_or(ZERO);
1794    Some(if reference == ZERO { Err(Status::Denied) } else { Ok((principal, reference)) })
1795}
1796/// The reply of `request` was copied out to the caller: only now may an
1797/// acknowledgement cover its outcome. Called by the syscall layer after a
1798/// successful copy-out; a failed copy-out leaves the outcome unobserved so the
1799/// caller must recover it with QueryRequest before acknowledging.
1800pub(crate) fn delivered(caller: usize, request: &[u8]) {
1801    let Ok(header) = RequestHeaderV1::decode(request) else { return; };
1802    if !mutating(header.selector, request.get(HEADER_LEN..).unwrap_or(&[])) { return; }
1803    let mut guard = super::SCHEDULER.lock();
1804    let Some(sched) = guard.as_mut() else { return; };
1805    let Ok(actor) = resolve_actor(sched, caller) else { return; };
1806    sched.cell_session.mark_observed(actor, request, header.request_id);
1807}
1808pub(crate) fn stdio_read(caller: usize, dst: &mut [u8]) -> Option<usize> {
1809    let (principal, reference) = match stdio_ref(caller, 0)? { Ok(pair) => pair, Err(_) => return Some(usize::MAX) };
1810    if dst.is_empty() { return Some(0); }
1811    let mut body = [0u8; 32]; let mut w = Writer::new(&mut body);
1812    w.reference(reference).ok()?; w.u32(dst.len().min(512) as u32).ok()?; w.u32(0).ok()?; w.u64(u64::MAX).ok()?;
1813    let mut out = [0u8; 520];
1814    Some(match stream_io(caller, principal, 5, &body, &mut out) {
1815        Ok(_) => { let count = u32::from_le_bytes(out[..4].try_into().unwrap()) as usize;
1816            dst[..count].copy_from_slice(&out[8..8+count]); count },
1817        Err(Status::Eof) => 0,
1818        Err(_) => usize::MAX,
1819    })
1820}
1821pub(crate) fn stdio_write(caller: usize, fd: usize, src: &[u8]) -> Option<usize> {
1822    let (principal, reference) = match stdio_ref(caller, fd)? { Ok(pair) => pair, Err(_) => return Some(usize::MAX) };
1823    if src.is_empty() { return Some(0); }
1824    let count = src.len().min(512);
1825    let mut body = [0u8; 544]; let mut w = Writer::new(&mut body);
1826    w.reference(reference).ok()?; w.u32(count as u32).ok()?; w.u32(0).ok()?; w.u64(u64::MAX).ok()?;
1827    w.bytes(&src[..count]).ok()?;
1828    let mut out = [0u8; 8];
1829    Some(match stream_io(caller, principal, 6, &body[..32+count], &mut out) {
1830        Ok(_) => u32::from_le_bytes(out[..4].try_into().unwrap()) as usize, Err(_) => usize::MAX,
1831    })
1832}

2089    /// Record an unbound kind-6 receipt on `prepared` sealed for `child` and
2090    /// `frontend`, as `RegisterBindReceipt` would for the fixture issuer.
2091    pub(crate) fn stage_frontend_receipt(sched: &mut Scheduler, prepared: Ref, child: Child, frontend: Ref) -> Option<Ref> {
2092        let _kernel = KernelAllocationContext::enter();
2093        let state = &mut sched.cell_session;
2094        let pi = state.preparation(prepared).ok()?;
2095        let (operation, preparer, digest) = (state.prepared[pi].operation, state.prepared[pi].preparer,
2096            state.prepared[pi].parsed.as_ref()?.digest);
2097        state.receipts.try_reserve(1).ok()?;
2098        let receipt = state.mint().ok()?;
2099        let mut seal = [0u8; 256];
2100        seal[..16].copy_from_slice(&frontend.to_bytes());
2101        seal[16..48].copy_from_slice(&child.to_bytes());
2102        state.receipts.push(Receipt { reference: receipt, prepared, child, operation,
2103            issuer: Binding::new(u64::MAX, u64::MAX, u64::MAX), kind: GRANT_KIND_INPUT_FRONTEND, slot: SLOT_FRONTEND,

2105            bound: false, producers: [ZERO; 2], restored: false, materialized: false,
