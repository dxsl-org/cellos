1//! Kernel-internal capability tokens.
2//!
3//! Each token is a zero-sized type (ZST).  Constructors are `pub(crate)` so
4//! only kernel code can create them — Cell crates are separate Rust
5//! compilation units and cannot call `pub(crate)` items from this crate.
6//!
7//! `Option<ZST>` uses Rust's niche optimization: exactly 1 byte on the wire.
8//! Three caps together are 3 bytes, smaller than the previous `KernelPerms(u32)`.
9
10/// Permits raw block-device syscalls (BlkRead, BlkWrite, BlkFlush).
11/// Granted to `/bin/vfs` at spawn.

14

152/// Singleton latch: held by either the one live reservation or the permanently
153/// committed Platform task. A reservation owns the only transition back to
154/// `false`; once committed, the latch intentionally remains set forever.
155static PLATFORM_CAP_GRANTED: AtomicBool = AtomicBool::new(false);

164/// Rollback owner for the Platform singleton while a task is unpublished.
165pub(crate) struct PlatformCapReservation {
166    committed: bool,
167}
168
169impl PlatformCapReservation {
170    /// Permanently transfer the singleton grant into a complete unpublished TCB.
171    pub(crate) fn commit_into(mut self, task: &mut super::tcb::Task) {
172        task.platform_cap = Some(PlatformCap::new());
173        self.committed = true;
174    }
175}
176
177impl Drop for PlatformCapReservation {
178    fn drop(&mut self) {
179        if !self.committed {
180            let reset = PLATFORM_CAP_GRANTED.compare_exchange(
181                true,
182                false,
183                Ordering::SeqCst,
184                Ordering::SeqCst,
185            );
186            debug_assert!(
187                reset.is_ok(),
188                "Platform reservation lost singleton ownership"
189            );
190        }
191    }
192}
193
194/// Reserve the singleton grant until atomic task publication commits.
195pub(crate) fn reserve_platform() -> Result<PlatformCapReservation, types::ViError> {
196    PLATFORM_CAP_GRANTED
197        .compare_exchange(false, true, Ordering::SeqCst, Ordering::SeqCst)
198        .map(|_| PlatformCapReservation { committed: false })
199        .map_err(|_| types::ViError::PermissionDenied)
200}

202#[cfg(feature = "test-hooks")]
203pub(crate) fn platform_reserved_or_committed() -> bool {
204    PLATFORM_CAP_GRANTED.load(Ordering::SeqCst)
205}

209/// A plain-data snapshot of a Task's capabilities, used to enforce spawn-time
210/// **intersection**: a child is granted `manifest ∩ spawner`, so no cell can
211/// hand a child a capability it does not itself hold (Fuchsia/Genode monotonic
212/// downgrade). Single source of truth for "what caps does X hold".
213#[derive(Copy, Clone, Debug, PartialEq, Eq)]
214pub struct CapSet {
215    pub block_io: bool,
216    pub network: bool,
217    pub spawn: bool,
218    pub hypervisor: bool,
219    pub mmio_devices: u8,  // bitmask of resource_registry::DEV_*
220    pub block_regions: u8, // P03 partition bitmask
221    // P-TRUST: the privileged path-triggered caps now live in the CapSet so the
222    // SAME spawn-time intersection that bounds every other cap also bounds them.
223    // Before this, they were minted by a raw `path ==` match AFTER (and blind to)
224    // the ceiling intersection — reachable via sys_spawn_from_elf to hand any
225    // SpawnCap holder PcieDriverCap → DMA-anywhere (LBI bypass). They have no
226    // manifest flag bit (v1 manifest is full); the install path is the request
227    // signal, but the request is now `∩ ceiling` like everything else.
228    pub pcie_driver: bool,
229    pub platform: bool,
230    pub supervisor: bool,
231    pub usb_driver: bool,
232    /// Port-I/O serial authority: lets `/bin/serial` drive the 16550 ports the
233    /// kernel itself probed. No DMA and no MMIO — it reaches only the declared
234    /// ports, which is why it is its own cap rather than a reuse of `pcie_driver`.
235    pub serial_port: bool,
236    // Coarse per-Cell network rights (Phase 04 S2-N). Enforced by service-net,
237    // not by any kernel syscall: the kernel only carries them from the bounded
238    // spawn grant into the attested caller trailer (`CALLER_FLAG_NET_*`). Like
239    // the P-TRUST fields they have no manifest bit — the install path is the
240    // request (`with_path_caps`) and the reviewed ceiling bounds it — but they
241    // are NOT P-TRUST: `without_ptrust` keeps them.
242    /// `NetRequest::Resolve`.
243    pub net_resolve: bool,
244    /// Outbound sockets: `TcpConnect`, `UdpCreate`, `UdpSend`.
245    pub net_connect: bool,
246    /// Inbound sockets: `TcpListen`, `UdpBind`, `MulticastJoin`.
247    pub net_listen: bool,
248    /// Raw Ethernet frames: `L2Send`, `L2Recv`.
249    pub net_l2: bool,
250    // Edge-bounded telemetry and VFS-mutation rights (Phase 04 S2-T/S2-W annex).
251    // Kernel-only: no manifest bit; the install path is the request
252    // (`with_path_caps`: every path except the `/apps/`, `/data/bin/`, `/home/`
253    // sandbox) and the reviewed ceiling and operator policy bound it. Not P-TRUST.
254    /// X: `GetProcs` / `GetProcs2` are refused at dispatch without it.
255    pub observe_tasks: bool,
256    /// M: `MemInfo` is refused at dispatch without it.
257    pub observe_memory: bool,
258    /// W: without it the kernel never attests `CALLER_FLAG_VFS_MUTATE`, even when
259    /// allowlist bit 63 is declared, so the VFS refuses every classified mutation.
260    pub vfs_mutate: bool,
261    /// Q (Phase 01 CP1, 2026-10-25): write the shell key namespace of
262    /// service-config (`ConfigRequest::Set` / `Delete`). Enforced by
263    /// service-config, not by any kernel syscall: the kernel only carries it from
264    /// the bounded spawn grant into the attested caller trailer
265    /// (`CALLER_FLAG_CONFIG_WRITE`). No manifest bit; `/bin/cellbox` is the only
266    /// install path that requests it (`with_path_config_rights`). Not P-TRUST.
267    pub config_write: bool,
268}

367    /// Derive the caps a manifest *requests*. Mirrors the historical loader grant
368    /// logic exactly — in particular `block_regions` replicates the SRV-bit
369    /// co-grant `data | (lfs<<1) | (lfs<<2)` (NOT a 1:1 copy) so the VFS service
370    /// keeps its P5 range after intersection. The H-ext gate is baked in here so
371    /// `hypervisor` can never be held on a CPU lacking H-ext.
372    pub fn from_manifest(m: &api::manifest::CellManifest) -> CapSet {
373        let hv = m.has_hypervisor()
374            && (crate::cpu_features::has_h_ext()
375                || crate::cpu_features::has_el2()
376                || crate::cpu_features::has_x86_virt());
377        let mut mmio = 0u8;
378        if m.has_gpio() {
379            mmio |= crate::resource_registry::DEV_GPIO;
380        }
381        if m.has_uart() {
382            mmio |= crate::resource_registry::DEV_UART;
383        }
384        if m.has_can() {
385            mmio |= crate::resource_registry::DEV_CAN;
386        }
387        if m.has_adc() {
388            mmio |= crate::resource_registry::DEV_ADC;
389        }
390        if m.has_i2c() {
391            mmio |= crate::resource_registry::DEV_I2C;
392        }
393        if m.has_spi() {
394            mmio |= crate::resource_registry::DEV_SPI;
395        }
396        CapSet {
397            block_io: m.has_block_io(),
398            network: m.has_network(),
399            spawn: m.has_spawn(),
400            hypervisor: hv,
401            mmio_devices: mmio,
402            block_regions: (m.has_part_data() as u8)
403                | ((m.has_part_lfs() as u8) << 1)
404                | ((m.has_part_lfs() as u8) << 2),
405            // The manifest never requests the privileged path-caps (no flag bits);
406            // they are layered on by `with_path_caps` from the install path.
407            pcie_driver: false,
408            platform: false,
409            supervisor: false,
410            usb_driver: false,
411            serial_port: false,
412            // Network rights have no manifest bit either; `with_path_caps`
413            // requests them from the install path.
414            net_resolve: false,
415            net_connect: false,
416            net_listen: false,
417            net_l2: false,
418            // Nor do the S2-T/S2-W rights; `with_path_caps` requests them.
419            observe_tasks: false,
420            observe_memory: false,
421            vfs_mutate: false,
422            config_write: false,
423        }
424    }

426    /// Layer the path-triggered privileged authority onto a requested CapSet.
427    /// These caps have no manifest flag bit (v1 manifest is full), so the install
428    /// path is the request signal — but the resulting request is still run through
429    /// the same `∩ ceiling` intersection as every other cap. This is the P-TRUST
430    /// fix: the loader used to mint these by raw `path ==` AFTER the intersection,
431    /// so `sys_spawn_from_elf(bytes, "/bin/nvme")` handed any SpawnCap holder
432    /// `PcieDriverCap` regardless of its ceiling → DMA-anywhere.
433    ///
434    /// `/bin/vfs` requests the cell-store block region here so the same ceiling and
435    /// operator-policy intersection that bounds every other authority also bounds bit 3.
436    pub fn with_path_caps(mut self, path: &str) -> CapSet {
437        if path == "/bin/vfs" {
438            self.block_regions |= 0b1000;
439        }
440        if matches!(
441            path,
442            "/bin/nvme"
443                | "/bin/e1000"
444                | "/bin/igb"
445                | "/bin/virtio-net"
446                | "/bin/block"
447                | "/bin/input"
448                | "/bin/virtio-gpu"
449                | "/bin/ahci"
450        ) {
451            self.pcie_driver = true;
452            self.mmio_devices |= crate::resource_registry::DEV_DISPLAY;
453        }
454        if path == "/bin/bcm-display" {
455            self.mmio_devices |= crate::resource_registry::DEV_DISPLAY;
456        }
457        // The xHCI USB host controller cell: PCIe BAR claim (sys_find_pcie_device
458        // / sys_request_mmio) plus USB-host authority, which the input service
459        // verifies via `sys_lookup_service(service::NIC_DRIVER)` before accepting
460        // this cell's raw key events — the same producer mechanism /bin/dwc2-usb
461        // uses.
462        if path == "/bin/xhci" {
463            self.pcie_driver = true;
464            self.usb_driver = true;
465            self.mmio_devices |= crate::resource_registry::DEV_DISPLAY;
466        }
467        if path == "/bin/dwc2-usb" {
468            self.usb_driver = true;
469        }
470        // The x86 serial Driver Cell: port-I/O authority over the 16550 ports the
471        // kernel probed. It gets no PCIe/MMIO authority: the ports are legacy I/O
472        // space, and the kernel already proved each one answers.
473        if path == "/bin/serial" {
474            self.serial_port = true;
475        }
476        if path == "/bin/platform" {
477            self.platform = true;
478        }
479        if path == "/bin/supervisor" {
480            self.supervisor = true;
481        }
482        self.with_path_net_rights(path)
483            .with_path_observe_mutate_rights(path)
484            .with_path_config_rights(path)
485    }

487    /// Request the coarse network rights of the reviewed grant table (Phase 04
488    /// S2-N annex §5) for `path`. Any path not listed requests none — that
489    /// includes the shell, `/bin/lua`, `/bin/python` and the `/apps`, `/data/bin`,
490    /// `/home` sandbox. The request is still bounded by the launch/boot ceiling
491    /// and the operator policy like every other field.
492    ///
493    /// Kept out of `with_path_caps`' own body on purpose: `scripts/sign-policy.py`
494    /// reads that body as the list of P-TRUST-minting paths, and these rights are
495    /// not P-TRUST.
496    fn with_path_net_rights(mut self, path: &str) -> CapSet {
497        let (resolve, connect, listen, l2) = match path {
498            "/bin/curl" | "/bin/wget" | "/bin/mqtt" | "/bin/ocel" | "/bin/llm-gateway"
499            | "/bin/https-demo" | "/bin/robot-demo" | "/bin/cellbox-ssh" => (true, true, false, false),
500            "/bin/nc" | "/bin/net-broker" => (true, true, true, false),
501            "/bin/httpd" | "/bin/sshd" => (false, false, true, false),
502            "/bin/hypervisor" => (false, false, false, true),
503            _ => (false, false, false, false),
504        };
505        self.net_resolve |= resolve;
506        self.net_connect |= connect;
507        self.net_listen |= listen;
508        self.net_l2 |= l2;
509        self
510    }

512    /// Request the S2-T/S2-W rights (annex §3, decision D-TW1) for `path`: every
513    /// path requests all three **except** the user-app sandbox prefixes, which
514    /// request none. The request is bounded by the launch/boot ceiling and the
515    /// operator policy like every other field, so a restricted row can still
516    /// withhold them. Kept out of `with_path_caps`' body for the same reason as
517    /// the network rights: `scripts/sign-policy.py` reads that body as the
518    /// P-TRUST path list.
519    fn with_path_observe_mutate_rights(mut self, path: &str) -> CapSet {
520        let sandbox = path.starts_with("/apps/")
521            || path.starts_with("/data/bin/")
522            || path.starts_with("/home/");
523        if !sandbox {
524            self.observe_tasks = true;
525            self.observe_memory = true;
526            self.vfs_mutate = true;
527        }
528        self

545    /// Whether `with_path_caps` layers any privileged (P-TRUST) authority onto a
546    /// request for `path` — i.e. whether the install path alone is enough to ask
547    /// for `pcie_driver` / `platform` / `supervisor`.
548    ///
549    /// Derived from `with_path_caps` itself rather than a second path list, so the
550    /// match arms above stay the single source of truth. Callers use it to decide
551    /// how dangerous a *missing* authorisation for `path` is: for these paths a
552    /// permissive default hands out DMA-anywhere or cell-orchestration authority,
553    /// so they must fail closed where an ordinary path may not.
554    pub fn path_mints_ptrust(path: &str) -> bool {
555        let requested = CapSet::EMPTY.with_path_caps(path);
556        requested.pcie_driver
557            || requested.platform
558            || requested.supervisor
559            || requested.usb_driver
560            || requested.serial_port
561    }

563    /// Drop every privileged (P-TRUST) cap, keeping the ordinary ones.
564    ///
565    /// The privileged three are the caps whose holder can DMA anywhere or drive
566    /// other cells; removing only those keeps a cell runnable (and its failure
567    /// diagnosable) where zeroing the whole set would look like a crash.
568    pub fn without_ptrust(self) -> CapSet {
569        CapSet {
570            pcie_driver: false,
571            platform: false,
572            supervisor: false,
573            usb_driver: false,
574            serial_port: false,
575            ..self
576        }
577    }

579    /// Drop the three S2-T/S2-W rights, keeping everything else.

699/// Compute the granted x86 PKU protection class from the granted caps and the
700/// manifest's requested byte (Manifest v2). Pure logic, host/target-agnostic —
701/// extracted so it is unit-testable independent of the loader's spawn plumbing.
702///
703/// The requested byte is a FLOOR, not a ceiling — the inverse of a capability. A
704/// higher value means MORE isolation / LESS authority. A cell may always RAISE
705/// its own protection class (self-restrict further); it may NEVER lower it below
706/// the floor derived from its granted caps (that would be a privilege
707/// escalation). Hence `max(requested_protection_class, floor)`, not a plain
708/// assignment.
709///
710/// `floor`: cells holding real system authority (block_io/network/spawn/
711/// hypervisor — the pre-v2 "is_trusted" set) may run at
712/// `PROTECTION_CLASS_TRUSTED_CORE` (0, unfenced). Everything else has a floor of
713/// `PROTECTION_CLASS_STANDARD` (1) — it can never claim class 0 no matter what
714/// it asks for.
715///
716/// `PROTECTION_CLASS_LEGACY` (manifest absent, or a tier-less manifest) means
717/// "no explicit request" — the granted class is exactly the floor,
718/// reproducing the pre-v2 behaviour byte-for-byte (NOT `max(0xFF, floor)`,
719/// which would wrongly force every such cell to the most-isolated class).
720pub fn granted_protection_class(granted: &CapSet, requested_protection_class: u8) -> u8 {
721    use api::manifest::{
722        PROTECTION_CLASS_LEGACY, PROTECTION_CLASS_STANDARD, PROTECTION_CLASS_TRUSTED_CORE,
723    };
724    let is_trusted = granted.block_io || granted.network || granted.spawn || granted.hypervisor;
725    let floor: u8 = if is_trusted {
726        PROTECTION_CLASS_TRUSTED_CORE
727    } else {
728        PROTECTION_CLASS_STANDARD
729    };
730    if requested_protection_class == PROTECTION_CLASS_LEGACY {
731        floor
732    } else {
733        core::cmp::max(requested_protection_class, floor)
734    }
735}

737/// Compatibility wrapper for callers still using the legacy "tier" terminology.
738#[deprecated(
739    note = "use granted_protection_class(); the manifest byte is now named protection class"
740)]
741pub fn granted_tier(granted: &CapSet, requested_tier: u8) -> u8 {
742    granted_protection_class(granted, requested_tier)
743}
