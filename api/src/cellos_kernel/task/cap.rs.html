1//! Kernel-internal capability tokens.
2//!
3//! Each token is a zero-sized type (ZST).  Constructors are `pub(crate)` so
4//! only kernel code can create them — Cell crates are separate Rust
5//! compilation units and cannot call `pub(crate)` items from this crate.
6//!
7//! `Option<ZST>` uses Rust's niche optimization: exactly 1 byte on the wire.
8//! Three caps together are 3 bytes, smaller than the previous `KernelPerms(u32)`.
9
10/// Permits raw block-device syscalls (BlkRead, BlkWrite, BlkFlush).
11/// Granted to `/bin/vfs` at spawn.

15/// Permits network transmit and receive syscalls (NetTx, NetRx).

62/// Permits `sys_freeze_cell`, `sys_resume_cell`, `sys_kill_cell`.
63///
64/// Carried in `CapSet` (P-TRUST) and gated by the spawn-time ceiling: the
65/// `/bin/supervisor` install path *requests* it (`with_path_caps`), and the
66/// request is intersected against the spawner's ceiling like every other cap, so

96/// Permits PCIe ECAM enumeration and BAR registration via `sys_register_pcie_bar`.
97///
98/// Granted by exact path match in `loader.rs` to `/bin/platform` ONLY, and is a
99/// singleton — the kernel refuses to grant it a second time (second `/bin/platform`

107/// Singleton latch: set when PlatformCap has been granted to any task.
108/// Once true, `try_grant_platform()` returns `None` for all future callers.
109static PLATFORM_CAP_GRANTED: AtomicBool = AtomicBool::new(false);

113    pub(crate) fn new() -> Self {
114        Self(())
115    }

118/// Attempt to grant `PlatformCap`.
119///
120/// Returns `Some(PlatformCap)` on the first call, `None` on all subsequent calls
121/// (singleton enforcement). The compare_exchange is sequentially consistent to
122/// avoid races on SMP (even though Cellos is currently UP, the invariant must hold
123/// under future SMP enablement).
124pub(crate) fn try_grant_platform() -> Option<PlatformCap> {
125    PLATFORM_CAP_GRANTED
126        .compare_exchange(false, true, Ordering::SeqCst, Ordering::SeqCst)
127        .ok()
128        .map(|_| PlatformCap::new())
129}

133/// A plain-data snapshot of a Task's capabilities, used to enforce spawn-time
134/// **intersection**: a child is granted `manifest ∩ spawner`, so no cell can
135/// hand a child a capability it does not itself hold (Fuchsia/Genode monotonic
136/// downgrade). Single source of truth for "what caps does X hold".
137#[derive(Copy, Clone, Debug, PartialEq, Eq)]
138pub struct CapSet {
139    pub block_io: bool,
140    pub network: bool,
141    pub spawn: bool,
142    pub hypervisor: bool,
143    pub mmio_devices: u8,  // bitmask of resource_registry::DEV_*
144    pub block_regions: u8, // P03 partition bitmask
145    // P-TRUST: the privileged path-triggered caps now live in the CapSet so the
146    // SAME spawn-time intersection that bounds every other cap also bounds them.
147    // Before this, they were minted by a raw `path ==` match AFTER (and blind to)
148    // the ceiling intersection — reachable via sys_spawn_from_elf to hand any
149    // SpawnCap holder PcieDriverCap → DMA-anywhere (LBI bypass). They have no
150    // manifest flag bit (v1 manifest is full); the install path is the request
151    // signal, but the request is now `∩ ceiling` like everything else.
152    pub pcie_driver: bool,
153    pub platform: bool,
154    pub supervisor: bool,
155}
156
157impl CapSet {
158    /// No capabilities (used for an unknown spawner — fail-safe).
159    pub const EMPTY: CapSet = CapSet {
160        block_io: false,
161        network: false,
162        spawn: false,
163        hypervisor: false,
164        mmio_devices: 0,
165        block_regions: 0,
166        pcie_driver: false,
167        platform: false,
168        supervisor: false,
169    };

171    /// Every cap this kernel can express — a **reference upper bound for
172    /// ceiling self-tests only**. It is NOT granted to any task.
173    ///
174    /// Boot authority comes from the per-path table in
175    /// `crate::loader::boot_ceiling`, which states what ONE path may hold. This
176    /// constant is the union of everything and therefore never constrains
177    /// anything; using it as a ceiling would admit every request while reading
178    /// like a restriction. The self-tests use it in the opposite direction — as
179    /// the widest possible ceiling, to prove a request is not *over*-tightened.
180    pub const ALL: CapSet = CapSet {
181        block_io: true,
182        network: true,
183        spawn: true,
184        hypervisor: true,
185        mmio_devices: crate::resource_registry::DEV_GPIO | crate::resource_registry::DEV_UART,
186        block_regions: 0b1111,
187        pcie_driver: true,
188        platform: true,
189        supervisor: true,
190    };

207    /// Derive the caps a manifest *requests*. Mirrors the historical loader grant
208    /// logic exactly — in particular `block_regions` replicates the SRV-bit
209    /// co-grant `data | (lfs<<1) | (lfs<<2)` (NOT a 1:1 copy) so the VFS service
210    /// keeps its P5 range after intersection. The H-ext gate is baked in here so
211    /// `hypervisor` can never be held on a CPU lacking H-ext.
212    pub fn from_manifest(m: &api::manifest::CellManifest) -> CapSet {
213        let hv = m.has_hypervisor()
214            && (crate::cpu_features::has_h_ext()
215                || crate::cpu_features::has_el2()
216                || crate::cpu_features::has_x86_virt());
217        let mut mmio = 0u8;
218        if m.has_gpio() {
219            mmio |= crate::resource_registry::DEV_GPIO;
220        }
221        if m.has_uart() {
222            mmio |= crate::resource_registry::DEV_UART;
223        }
224        if m.has_can() {
225            mmio |= crate::resource_registry::DEV_CAN;
226        }
227        if m.has_adc() {
228            mmio |= crate::resource_registry::DEV_ADC;
229        }
230        CapSet {
231            block_io: m.has_block_io(),
232            network: m.has_network(),
233            spawn: m.has_spawn(),
234            hypervisor: hv,
235            mmio_devices: mmio,
236            block_regions: (m.has_part_data() as u8)
237                | ((m.has_part_lfs() as u8) << 1)
238                | ((m.has_part_lfs() as u8) << 2),
239            // The manifest never requests the privileged path-caps (no flag bits);
240            // they are layered on by `with_path_caps` from the install path.
241            pcie_driver: false,
242            platform: false,
243            supervisor: false,
244        }
245    }
246
247    /// Layer the path-triggered privileged authority onto a requested CapSet.
248    /// These caps have no manifest flag bit (v1 manifest is full), so the install
249    /// path is the request signal — but the resulting request is still run through
250    /// the same `∩ ceiling` intersection as every other cap. This is the P-TRUST
251    /// fix: the loader used to mint these by raw `path ==` AFTER the intersection,
252    /// so `sys_spawn_from_elf(bytes, "/bin/nvme")` handed any SpawnCap holder
253    /// `PcieDriverCap` regardless of its ceiling → DMA-anywhere.
254    ///
255    /// `/bin/vfs` requests the cell-store block region here so the same ceiling and
256    /// operator-policy intersection that bounds every other authority also bounds bit 3.
257    pub fn with_path_caps(mut self, path: &str) -> CapSet {
258        if path == "/bin/vfs" {
259            self.block_regions |= 0b1000;
260        }
261        if matches!(
262            path,
263            "/bin/nvme"
264                | "/bin/e1000"
265                | "/bin/virtio-net"
266                | "/bin/block"
267                | "/bin/input"
268                | "/bin/virtio-gpu"
269        ) {
270            self.pcie_driver = true;
271        }
272        if path == "/bin/platform" {
273            self.platform = true;
274        }
275        if path == "/bin/supervisor" {
276            self.supervisor = true;
277        }
278        self
279    }
280
281    /// Whether `with_path_caps` layers any privileged (P-TRUST) authority onto a
282    /// request for `path` — i.e. whether the install path alone is enough to ask
283    /// for `pcie_driver` / `platform` / `supervisor`.
284    ///
285    /// Derived from `with_path_caps` itself rather than a second path list, so the
286    /// match arms above stay the single source of truth. Callers use it to decide
287    /// how dangerous a *missing* authorisation for `path` is: for these paths a
288    /// permissive default hands out DMA-anywhere or cell-orchestration authority,
289    /// so they must fail closed where an ordinary path may not.
290    pub fn path_mints_ptrust(path: &str) -> bool {
291        let requested = CapSet::EMPTY.with_path_caps(path);
292        requested.pcie_driver || requested.platform || requested.supervisor
293    }
294
295    /// Drop every privileged (P-TRUST) cap, keeping the ordinary ones.
296    ///
297    /// The privileged three are the caps whose holder can DMA anywhere or drive
298    /// other cells; removing only those keeps a cell runnable (and its failure
299    /// diagnosable) where zeroing the whole set would look like a crash.
300    pub fn without_ptrust(self) -> CapSet {
301        CapSet {
302            pcie_driver: false,
303            platform: false,
304            supervisor: false,
305            ..self
306        }
307    }

309    /// Field-wise minimum (bool AND, bitmask AND). The monotonic-downgrade core.
310    pub fn intersect(self, o: CapSet) -> CapSet {
311        CapSet {
312            block_io: self.block_io && o.block_io,
313            network: self.network && o.network,
314            spawn: self.spawn && o.spawn,
315            hypervisor: self.hypervisor && o.hypervisor,
316            mmio_devices: self.mmio_devices & o.mmio_devices,
317            block_regions: self.block_regions & o.block_regions,
318            pcie_driver: self.pcie_driver && o.pcie_driver,
319            platform: self.platform && o.platform,
320            supervisor: self.supervisor && o.supervisor,
321        }
322    }

324    /// Write the granted caps into a child Task's TCB fields. Pure data — block-IO
325    /// VFS-handler registration and any other side effects stay in the loader,
326    /// keyed off the *granted* (not requested) bits.
327    pub fn apply_to(self, t: &mut super::tcb::Task) {
328        t.block_io_cap = self.block_io.then(BlockIoCap::new);
329        t.network_cap = self.network.then(NetworkCap::new);
330        t.spawn_cap = self.spawn.then(SpawnCap::new);
331        t.hypervisor_cap = self.hypervisor.then(HypervisorCap::new);
332        t.mmio_devices = self.mmio_devices;
333        t.block_regions = self.block_regions;
334        t.pcie_driver_cap = self.pcie_driver.then(PcieDriverCap::new);
335        t.supervisor_cap = self.supervisor.then(SupervisorCap::new);
336        // NOTE: `platform_cap` is deliberately NOT written here. It is a singleton
337        // (`try_grant_platform` enforces one-holder-ever); the loader consults the
338        // latch when `granted.platform` is set. Writing it from a plain bool would
339        // bypass the latch and allow two holders.
340    }
341}

343/// Compute the granted x86 PKU isolation tier from the granted caps and the
344/// manifest's requested tier (Manifest v2). Pure logic, host/target-agnostic —
345/// extracted so it is unit-testable independent of the loader's spawn plumbing.
346///
347/// `tier` is a FLOOR, not a ceiling — the inverse of a capability. A higher tier
348/// number means MORE isolation / LESS authority. A cell may always RAISE its own
349/// tier (self-restrict further); it may NEVER lower it below the floor derived
350/// from its granted caps (that would be a privilege escalation). Hence
351/// `max(requested_tier, floor)`, not a plain assignment.
352///
353/// `floor`: cells holding real system authority (block_io/network/spawn/
354/// hypervisor — the pre-v2 "is_trusted" set) may run at `TIER_TRUSTED_CORE` (0,
355/// unfenced). Everything else has a floor of `TIER_STANDARD` (1) — it can never
356/// claim tier 0 no matter what it asks for.
357///
358/// `TIER_LEGACY` (manifest absent, or a tier-less manifest) means "no explicit
359/// request" — the granted tier is exactly the floor, reproducing the pre-v2
360/// behaviour byte-for-byte (NOT `max(0xFF, floor)`, which would wrongly force
361/// every such cell to the most-isolated tier).
362pub fn granted_tier(granted: &CapSet, requested_tier: u8) -> u8 {
363    use api::manifest::{TIER_LEGACY, TIER_STANDARD, TIER_TRUSTED_CORE};
364    let is_trusted = granted.block_io || granted.network || granted.spawn || granted.hypervisor;
365    let floor: u8 = if is_trusted {
366        TIER_TRUSTED_CORE
367    } else {
368        TIER_STANDARD
369    };
370    if requested_tier == TIER_LEGACY {
371        floor
372    } else {
373        core::cmp::max(requested_tier, floor)
374    }
375}

377/// Who initiated a spawn — determines the capability ceiling for the new cell.
378#[derive(Copy, Clone, Debug)]
379pub enum Spawner {
380    /// Kernel/boot-initiated (only `init`). No intersection — grant full manifest.
381    Root,
382    /// User-cell-initiated via syscall. Child caps = `manifest ∩ caps_of(tid)`.
383    User(usize),
384    /// Kernel-internal re-spawn (HotSwap) bounded by an explicit ceiling
385    /// (the replaced cell's caps) — NOT the `Root` exemption.
386    Ceiling(CapSet),
387}

394    fn intersect_is_monotonic_downgrade() {
395        let spawner = CapSet {
396            block_io: false,
397            network: true,
398            spawn: true,
399            hypervisor: false,
400            mmio_devices: 0b01,
401            block_regions: 0b010,

404        let child = CapSet {

406            network: true,
407            spawn: false,
408            hypervisor: true,

410            block_regions: 0b111,
411            ..CapSet::EMPTY
412        };
413        let g = child.intersect(spawner);
414        assert!(!g.block_io, "child cannot gain block_io its spawner lacks");
415        assert!(g.network);
416        assert!(!g.spawn);
417        assert!(!g.hypervisor);
