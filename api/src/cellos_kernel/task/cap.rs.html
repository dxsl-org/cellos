1//! Kernel-internal capability tokens.
2//!
3//! Each token is a zero-sized type (ZST).  Constructors are `pub(crate)` so
4//! only kernel code can create them — Cell crates are separate Rust
5//! compilation units and cannot call `pub(crate)` items from this crate.
6//!
7//! `Option<ZST>` uses Rust's niche optimization: exactly 1 byte on the wire.
8//! Three caps together are 3 bytes, smaller than the previous `KernelPerms(u32)`.
9

133/// Rollback owner for the Platform singleton while a task is unpublished.
134pub(crate) struct PlatformCapReservation {
135    committed: bool,
136}
137
138impl PlatformCapReservation {
139    /// Permanently transfer the singleton grant into a complete unpublished TCB.
140    pub(crate) fn commit_into(mut self, task: &mut super::tcb::Task) {
141        task.platform_cap = Some(PlatformCap::new());
142        self.committed = true;
143    }
144}
145
146impl Drop for PlatformCapReservation {
147    fn drop(&mut self) {
148        if !self.committed {
149            let reset = PLATFORM_CAP_GRANTED.compare_exchange(
150                true,
151                false,
152                Ordering::SeqCst,
153                Ordering::SeqCst,
154            );
155            debug_assert!(
156                reset.is_ok(),
157                "Platform reservation lost singleton ownership"
158            );
159        }
160    }
161}
162
163/// Reserve the singleton grant until atomic task publication commits.
164pub(crate) fn reserve_platform() -> Result<PlatformCapReservation, types::ViError> {
165    PLATFORM_CAP_GRANTED
166        .compare_exchange(false, true, Ordering::SeqCst, Ordering::SeqCst)
167        .map(|_| PlatformCapReservation { committed: false })
168        .map_err(|_| types::ViError::PermissionDenied)
169}
170
171#[cfg(feature = "test-hooks")]
172pub(crate) fn platform_reserved_or_committed() -> bool {
173    PLATFORM_CAP_GRANTED.load(Ordering::SeqCst)
174}

178/// A plain-data snapshot of a Task's capabilities, used to enforce spawn-time
179/// **intersection**: a child is granted `manifest ∩ spawner`, so no cell can
180/// hand a child a capability it does not itself hold (Fuchsia/Genode monotonic
181/// downgrade). Single source of truth for "what caps does X hold".
182#[derive(Copy, Clone, Debug, PartialEq, Eq)]
183pub struct CapSet {
184    pub block_io: bool,
185    pub network: bool,
186    pub spawn: bool,
187    pub hypervisor: bool,
188    pub mmio_devices: u8,  // bitmask of resource_registry::DEV_*
189    pub block_regions: u8, // P03 partition bitmask
190    // P-TRUST: the privileged path-triggered caps now live in the CapSet so the
191    // SAME spawn-time intersection that bounds every other cap also bounds them.
192    // Before this, they were minted by a raw `path ==` match AFTER (and blind to)
193    // the ceiling intersection — reachable via sys_spawn_from_elf to hand any
194    // SpawnCap holder PcieDriverCap → DMA-anywhere (LBI bypass). They have no
195    // manifest flag bit (v1 manifest is full); the install path is the request
196    // signal, but the request is now `∩ ceiling` like everything else.
197    pub pcie_driver: bool,
198    pub platform: bool,
199    pub supervisor: bool,
200}

202impl CapSet {
203    /// No capabilities (used for an unknown spawner — fail-safe).
204    pub const EMPTY: CapSet = CapSet {
205        block_io: false,
206        network: false,
207        spawn: false,
208        hypervisor: false,
209        mmio_devices: 0,
210        block_regions: 0,
211        pcie_driver: false,
212        platform: false,
213        supervisor: false,
214    };

216    /// Every cap this kernel can express — a **reference upper bound for
217    /// ceiling self-tests only**. It is NOT granted to any task.
218    ///
219    /// Boot authority comes from the per-path table in
220    /// `crate::loader::boot_ceiling`, which states what ONE path may hold. This
221    /// constant is the union of everything and therefore never constrains
222    /// anything; using it as a ceiling would admit every request while reading
223    /// like a restriction. The self-tests use it in the opposite direction — as
224    /// the widest possible ceiling, to prove a request is not *over*-tightened.
225    pub const ALL: CapSet = CapSet {
226        block_io: true,
227        network: true,
228        spawn: true,
229        hypervisor: true,
230        mmio_devices: crate::resource_registry::DEV_GPIO
231            | crate::resource_registry::DEV_UART
232            | crate::resource_registry::DEV_CAN
233            | crate::resource_registry::DEV_ADC
234            | crate::resource_registry::DEV_I2C
235            | crate::resource_registry::DEV_SPI
236            | crate::resource_registry::DEV_DISPLAY,
237        block_regions: 0b1111,
238        pcie_driver: true,
239        platform: true,
240        supervisor: true,
241    };

258    /// Derive the caps a manifest *requests*. Mirrors the historical loader grant
259    /// logic exactly — in particular `block_regions` replicates the SRV-bit
260    /// co-grant `data | (lfs<<1) | (lfs<<2)` (NOT a 1:1 copy) so the VFS service
261    /// keeps its P5 range after intersection. The H-ext gate is baked in here so
262    /// `hypervisor` can never be held on a CPU lacking H-ext.
263    pub fn from_manifest(m: &api::manifest::CellManifest) -> CapSet {
264        let hv = m.has_hypervisor()
265            && (crate::cpu_features::has_h_ext()
266                || crate::cpu_features::has_el2()
267                || crate::cpu_features::has_x86_virt());
268        let mut mmio = 0u8;
269        if m.has_gpio() {
270            mmio |= crate::resource_registry::DEV_GPIO;
271        }
272        if m.has_uart() {
273            mmio |= crate::resource_registry::DEV_UART;
274        }
275        if m.has_can() {
276            mmio |= crate::resource_registry::DEV_CAN;
277        }
278        if m.has_adc() {
279            mmio |= crate::resource_registry::DEV_ADC;
280        }
281        if m.has_i2c() {
282            mmio |= crate::resource_registry::DEV_I2C;
283        }
284        if m.has_spi() {
285            mmio |= crate::resource_registry::DEV_SPI;
286        }
287        CapSet {
288            block_io: m.has_block_io(),
289            network: m.has_network(),
290            spawn: m.has_spawn(),
291            hypervisor: hv,
292            mmio_devices: mmio,
293            block_regions: (m.has_part_data() as u8)
294                | ((m.has_part_lfs() as u8) << 1)
295                | ((m.has_part_lfs() as u8) << 2),
296            // The manifest never requests the privileged path-caps (no flag bits);
297            // they are layered on by `with_path_caps` from the install path.
298            pcie_driver: false,
299            platform: false,
300            supervisor: false,
301        }
302    }

304    /// Layer the path-triggered privileged authority onto a requested CapSet.
305    /// These caps have no manifest flag bit (v1 manifest is full), so the install
306    /// path is the request signal — but the resulting request is still run through
307    /// the same `∩ ceiling` intersection as every other cap. This is the P-TRUST
308    /// fix: the loader used to mint these by raw `path ==` AFTER the intersection,
309    /// so `sys_spawn_from_elf(bytes, "/bin/nvme")` handed any SpawnCap holder
310    /// `PcieDriverCap` regardless of its ceiling → DMA-anywhere.
311    ///
312    /// `/bin/vfs` requests the cell-store block region here so the same ceiling and
313    /// operator-policy intersection that bounds every other authority also bounds bit 3.
314    pub fn with_path_caps(mut self, path: &str) -> CapSet {
315        if path == "/bin/vfs" {
316            self.block_regions |= 0b1000;
317        }
318        if matches!(
319            path,
320            "/bin/nvme"
321                | "/bin/e1000"
322                | "/bin/virtio-net"
323                | "/bin/block"
324                | "/bin/input"
325                | "/bin/virtio-gpu"
326        ) {
327            self.pcie_driver = true;
328            self.mmio_devices |= crate::resource_registry::DEV_DISPLAY;
329        }
330        if path == "/bin/bcm-display" {
331            self.mmio_devices |= crate::resource_registry::DEV_DISPLAY;
332        }
333        // NOTE: /bin/dwc2-usb and /bin/lan9514 intentionally receive NO path-triggered
334        // caps here. USB host controller authority (DWC2 MMIO + IRQ) requires policy v3
335        // with a signed USB byte before it can be expressed through the capability system.
336        // Gate with a test matrix in policy::self_test first. See resource_registry.rs.
337        if path == "/bin/platform" {
338            self.platform = true;
339        }
340        if path == "/bin/supervisor" {
341            self.supervisor = true;
342        }
343        self
344    }

346    /// Whether `with_path_caps` layers any privileged (P-TRUST) authority onto a
347    /// request for `path` — i.e. whether the install path alone is enough to ask
348    /// for `pcie_driver` / `platform` / `supervisor`.
349    ///
350    /// Derived from `with_path_caps` itself rather than a second path list, so the
351    /// match arms above stay the single source of truth. Callers use it to decide
352    /// how dangerous a *missing* authorisation for `path` is: for these paths a
353    /// permissive default hands out DMA-anywhere or cell-orchestration authority,
354    /// so they must fail closed where an ordinary path may not.
355    pub fn path_mints_ptrust(path: &str) -> bool {
356        let requested = CapSet::EMPTY.with_path_caps(path);
357        requested.pcie_driver || requested.platform || requested.supervisor
358    }

360    /// Drop every privileged (P-TRUST) cap, keeping the ordinary ones.
361    ///
362    /// The privileged three are the caps whose holder can DMA anywhere or drive
363    /// other cells; removing only those keeps a cell runnable (and its failure
364    /// diagnosable) where zeroing the whole set would look like a crash.
365    pub fn without_ptrust(self) -> CapSet {
366        CapSet {
367            pcie_driver: false,
368            platform: false,
369            supervisor: false,
370            ..self
371        }
372    }

374    /// Field-wise minimum (bool AND, bitmask AND). The monotonic-downgrade core.
375    pub fn intersect(self, o: CapSet) -> CapSet {
376        CapSet {
377            block_io: self.block_io && o.block_io,
378            network: self.network && o.network,
379            spawn: self.spawn && o.spawn,
380            hypervisor: self.hypervisor && o.hypervisor,
381            mmio_devices: self.mmio_devices & o.mmio_devices,
382            block_regions: self.block_regions & o.block_regions,
383            pcie_driver: self.pcie_driver && o.pcie_driver,
384            platform: self.platform && o.platform,
385            supervisor: self.supervisor && o.supervisor,
386        }
387    }

389    /// Write the granted caps into a child Task's TCB fields. Pure data — block-IO
390    /// VFS-handler registration and any other side effects stay in the loader,
391    /// keyed off the *granted* (not requested) bits.
392    pub fn apply_to(self, t: &mut super::tcb::Task) {
393        t.block_io_cap = self.block_io.then(BlockIoCap::new);
394        t.network_cap = self.network.then(NetworkCap::new);
395        t.spawn_cap = self.spawn.then(SpawnCap::new);
396        t.hypervisor_cap = self.hypervisor.then(HypervisorCap::new);
397        t.mmio_devices = self.mmio_devices;
398        t.block_regions = self.block_regions;
399        t.pcie_driver_cap = self.pcie_driver.then(PcieDriverCap::new);
400        t.supervisor_cap = self.supervisor.then(SupervisorCap::new);
401        // `platform_cap` is transferred only from `PlatformCapReservation`.
402        // Applying a plain capability snapshot must never bypass the singleton.
403    }
404}

406/// Compute the granted x86 PKU protection class from the granted caps and the
407/// manifest's requested byte (Manifest v2). Pure logic, host/target-agnostic —
408/// extracted so it is unit-testable independent of the loader's spawn plumbing.
409///
410/// The requested byte is a FLOOR, not a ceiling — the inverse of a capability. A
411/// higher value means MORE isolation / LESS authority. A cell may always RAISE
412/// its own protection class (self-restrict further); it may NEVER lower it below
413/// the floor derived from its granted caps (that would be a privilege
414/// escalation). Hence `max(requested_protection_class, floor)`, not a plain
415/// assignment.
416///
417/// `floor`: cells holding real system authority (block_io/network/spawn/
418/// hypervisor — the pre-v2 "is_trusted" set) may run at
419/// `PROTECTION_CLASS_TRUSTED_CORE` (0, unfenced). Everything else has a floor of
420/// `PROTECTION_CLASS_STANDARD` (1) — it can never claim class 0 no matter what
421/// it asks for.
422///
423/// `PROTECTION_CLASS_LEGACY` (manifest absent, or a tier-less manifest) means
424/// "no explicit request" — the granted class is exactly the floor,
425/// reproducing the pre-v2 behaviour byte-for-byte (NOT `max(0xFF, floor)`,
426/// which would wrongly force every such cell to the most-isolated class).
427pub fn granted_protection_class(granted: &CapSet, requested_protection_class: u8) -> u8 {
428    use api::manifest::{
429        PROTECTION_CLASS_LEGACY, PROTECTION_CLASS_STANDARD, PROTECTION_CLASS_TRUSTED_CORE,
430    };
431    let is_trusted = granted.block_io || granted.network || granted.spawn || granted.hypervisor;
432    let floor: u8 = if is_trusted {
433        PROTECTION_CLASS_TRUSTED_CORE
434    } else {
435        PROTECTION_CLASS_STANDARD
436    };
437    if requested_protection_class == PROTECTION_CLASS_LEGACY {
438        floor
439    } else {
440        core::cmp::max(requested_protection_class, floor)
441    }
442}

448pub fn granted_tier(granted: &CapSet, requested_tier: u8) -> u8 {

452/// Who initiated a spawn — determines the capability ceiling for the new cell.
453#[derive(Copy, Clone, Debug)]
454pub enum Spawner {
455    /// Kernel/boot-initiated (only `init`). No intersection — grant full manifest.
456    Root,
457    /// User-cell-initiated via syscall. Child caps = `manifest ∩ caps_of(tid)`.
458    User(usize),
459    /// Kernel-internal re-spawn (HotSwap) bounded by an explicit ceiling
460    /// (the replaced cell's caps) — NOT the `Root` exemption.
461    Ceiling(CapSet),
462}
