13/// Checked copy from user memory into `dst`. On ANY failure `dst` is left
14/// untouched: the probe pass moves no bytes and the commit pass only starts
15/// after every page of `src` was proven present and readable.
16///
17/// Requires `src.len() <= dst.len()`; a caller passing a short destination has
18/// misdescribed the transaction and gets the same recoverable error.
19pub(crate) fn copy_from_user(
20    view: &CopyView,
21    src: UserReadSlice,
22    dst: &mut [u8],
23) -> Result<(), CopyError> {
24    if src.len() > dst.len() {
25        return Err(CopyError::InvalidAddress);
26    }
27    execute(
28        view,
29        Direction::FromUser,
30        src.ptr(),
31        dst.as_mut_ptr(),
32        src.len(),
33    )
34}

36/// Checked copy from kernel `src` into user memory. On ANY failure the user
37/// destination is untouched: the commit pass only starts after every page of
38/// `dst` was proven present and writable.
39pub(crate) fn copy_to_user(
40    view: &CopyView,
41    dst: UserWriteSlice,
42    src: &[u8],
43) -> Result<(), CopyError> {
44    if dst.len() > src.len() {
45        return Err(CopyError::InvalidAddress);
46    }
47    execute(
48        view,
49        Direction::ToUser,
50        dst.ptr(),
51        src.as_ptr().cast_mut(),
52        dst.len(),
53    )
54}
55/// Probe a destination without moving any bytes.
56///
57/// The caller may use this before an irreversible operation and then repeat the
58/// full copy under its own short-lived ownership lease.
59pub(crate) fn probe_writable(view: &CopyView, dst: UserWriteSlice) -> Result<(), CopyError> {
60    if dst.len() == 0 {
61        return Ok(());
62    }
63    match view {
64        CopyView::Domain(arc) => {
65            if !arc.is_live() {
66                return Err(CopyError::InvalidAddress);
67            }
68            drop(stage_domain(arc, Direction::ToUser, dst.ptr(), dst.len())?);
69            Ok(())
70        }
71        CopyView::Sas => probe_sas(dst.ptr(), dst.len(), Direction::ToUser),
72    }
73}

75/// Shared transaction body. `other` is the kernel-side buffer pointer; the
76/// user-side pointer comes from the validated slice.
77fn execute(
78    view: &CopyView,
79    direction: Direction,
80    user_ptr: usize,
81    other: *mut u8,
82    len: usize,
83) -> Result<(), CopyError> {
84    if len == 0 {
85        return Ok(());
86    }
87    match view {
88        CopyView::Domain(arc) => {
89            // Reject Dying before the pin (the double-check inside
90            // acquire_copy_reader covers the post-pin edge).
91            if !arc.is_live() {
92                return Err(CopyError::InvalidAddress);
93            }
94            let staged = stage_domain(arc, direction, user_ptr, len)?;
95            staged.commit(other)
96        }
97        CopyView::Sas => {
98            probe_sas(user_ptr, len, direction)?;
99            commit_sas(direction, user_ptr, other, len)
100        }
101    }
102}

104/// A probed, pinned user range awaiting its byte-copy commit.
105pub(crate) struct PinnedCopy<'a> {
106    pub(super) arc: &'a AddressSpace,
107    /// Held purely for its drain side effect: while this lease exists no
108    /// revocation path may tear down a PTE of the pinned root.
109    #[allow(dead_code)]
110    pub(super) reader: CopyReader<'a>,
111    pub(super) direction: Direction,
112    pub(super) user_ptr: usize,
113    pub(super) len: usize,
114}
115
116impl PinnedCopy<'_> {
117    /// Run the commit pass. `other` is the kernel-side buffer.
118    ///
119    /// FAULT-FREEDOM PROOF AT THE COPY SITE: the reader pin blocks every
120    /// revocation path (`unmap_private_page`, `unmap_grant_page`) until it
121    /// drains. Every page is re-walked directly from `(user_ptr, len)` under
122    /// that pin with zero heap allocation.
123    pub(crate) fn commit(self, other: *mut u8) -> Result<(), CopyError> {
124        let root_pa = self.arc.root_ppn() << 12;
125        if super::sv39_probe::current_satp_root() == Some(root_pa) {
126            let (src, dst) = if self.direction == Direction::FromUser {
127                (self.user_ptr as *const u8, other)
128            } else {
129                (other as *const u8, self.user_ptr as *mut u8)
130            };
131            return commit_guarded(src, dst, self.len, self.user_ptr, self.user_ptr + self.len);
132        }
133        let need_pte = if self.direction == Direction::ToUser {
134            SV39_WRITE
135        } else {
136            SV39_READ
137        };
138        let end = self.user_ptr + self.len;
139        let mut page = self.user_ptr & !(PAGE_SIZE - 1);
140        let mut result = Ok(());
141        while page < end {
142            let lo = page.max(self.user_ptr);
143            let hi = (page + PAGE_SIZE).min(end);
144            let chunk = hi - lo;
145            let Some((bits, pa)) = sv39_leaf(root_pa, page) else {
146                return Err(CopyError::InvalidAddress);
147            };
148            if bits & (SV39_VALID | SV39_USER | need_pte) != SV39_VALID | SV39_USER | need_pte {
149                return Err(CopyError::InvalidAddress);
150            }
151            let offset = lo - page;
152            let alias = phys_to_virt(pa + offset);
153            let (src, dst): (*const u8, *mut u8) = if self.direction == Direction::FromUser {
154                (alias as *const u8, other.wrapping_add(lo - self.user_ptr))
155            } else {
156                (other.wrapping_add(lo - self.user_ptr), alias as *mut u8)
157            };
158            if commit_guarded(src, dst, chunk, alias, alias + chunk).is_err() {
159                result = Err(CopyError::InvalidAddress);
160                break;
161            }
162            page += PAGE_SIZE;
163        }
164        result
165    }
166}

168/// Sas commit: direct VA byte copy under the active shared root, inside the
169/// recoverable guard window. A racing revocation of a validated page surfaces
170/// as `CopyError::InvalidAddress` — retained legacy Sas mapping exposure,
171/// documented in the phase plan.
172pub(super) fn commit_sas(
173    direction: Direction,
174    user_ptr: usize,
175    other: *mut u8,
176    len: usize,
177) -> Result<(), CopyError> {
178    let (src, dst) = if direction == Direction::FromUser {
179        (user_ptr as *const u8, other)
180    } else {
181        (other as *const u8, user_ptr as *mut u8)
182    };
183    commit_guarded(src, dst, len, user_ptr, user_ptr + len)
184}
