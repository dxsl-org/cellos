14/// Checked copy from user memory into `dst`. On ANY failure `dst` is left
15/// untouched: the probe pass moves no bytes and the commit pass only starts
16/// after every page of `src` was proven present and readable.
17///
18/// Requires `src.len() <= dst.len()`; a caller passing a short destination has
19/// misdescribed the transaction and gets the same recoverable error.
20pub(crate) fn copy_from_user(
21    view: &CopyView,
22    src: UserReadSlice,
23    dst: &mut [u8],
24) -> Result<(), CopyError> {
25    if src.len() > dst.len() {
26        return Err(CopyError::InvalidAddress);
27    }
28    execute(
29        view,
30        Direction::FromUser,
31        src.ptr(),
32        dst.as_mut_ptr(),
33        src.len(),
34    )
35}

37/// Checked copy from kernel `src` into user memory. On ANY failure the user
38/// destination is untouched: the commit pass only starts after every page of
39/// `dst` was proven present and writable.
40pub(crate) fn copy_to_user(
41    view: &CopyView,
42    dst: UserWriteSlice,
43    src: &[u8],
44) -> Result<(), CopyError> {
45    if dst.len() > src.len() {
46        return Err(CopyError::InvalidAddress);
47    }
48    execute(
49        view,
50        Direction::ToUser,
51        dst.ptr(),
52        src.as_ptr().cast_mut(),
53        dst.len(),
54    )
55}
56/// Probe a destination without moving any bytes.
57///
58/// The caller may use this before an irreversible operation and then repeat the
59/// full copy under its own short-lived ownership lease.
60pub(crate) fn probe_writable(view: &CopyView, dst: UserWriteSlice) -> Result<(), CopyError> {
61    if dst.len() == 0 {
62        return Ok(());
63    }
64    match view {
65        CopyView::Domain(arc) => {
66            if !arc.is_live() {
67                return Err(CopyError::InvalidAddress);
68            }
69            drop(stage_domain(arc, Direction::ToUser, dst.ptr(), dst.len())?);
70            Ok(())
71        }
72        CopyView::Sas => probe_sas(dst.ptr(), dst.len(), Direction::ToUser),
73    }
74}

76/// Shared transaction body. `other` is the kernel-side buffer pointer; the
77/// user-side pointer comes from the validated slice.
78fn execute(
79    view: &CopyView,
80    direction: Direction,
81    user_ptr: usize,
82    other: *mut u8,
83    len: usize,
84) -> Result<(), CopyError> {
85    if len == 0 {
86        return Ok(());
87    }
88    match view {
89        CopyView::Domain(arc) => {
90            // Reject Dying before the pin (the double-check inside
91            // acquire_copy_reader covers the post-pin edge).
92            if !arc.is_live() {
93                return Err(CopyError::InvalidAddress);
94            }
95            let staged = stage_domain(arc, direction, user_ptr, len)?;
96            staged.commit(other)
97        }
98        CopyView::Sas => {
99            probe_sas(user_ptr, len, direction)?;
100            commit_sas(direction, user_ptr, other, len)
101        }
102    }
103}

105/// A probed, pinned user range awaiting its byte-copy commit.
106pub(crate) struct PinnedCopy<'a> {
107    pub(super) arc: &'a AddressSpace,
108    /// Held purely for its drain side effect: while this lease exists no
109    /// revocation path may tear down a PTE of the pinned root.
110    #[allow(dead_code)]
111    pub(super) reader: CopyReader<'a>,
112    pub(super) direction: Direction,
113    pub(super) user_ptr: usize,
114    pub(super) len: usize,
115}
116
117impl PinnedCopy<'_> {
118    /// Run the commit pass. `other` is the kernel-side buffer.
119    ///
120    /// FAULT-FREEDOM PROOF AT THE COPY SITE: the reader pin blocks every
121    /// revocation path (`unmap_private_page`, `unmap_grant_page`) until it
122    /// drains. Every page is re-walked directly from `(user_ptr, len)` under
123    /// that pin with zero heap allocation.
124    pub(crate) fn commit(self, other: *mut u8) -> Result<(), CopyError> {
125        let root_pa = self.arc.root_ppn() << 12;
126        if super::sv39_probe::current_satp_root() == Some(root_pa) {
127            let (src, dst) = if self.direction == Direction::FromUser {
128                (self.user_ptr as *const u8, other)
129            } else {
130                (other as *const u8, self.user_ptr as *mut u8)
131            };
132            return commit_guarded(src, dst, self.len, self.user_ptr, self.user_ptr + self.len);
133        }
134        let need_pte = if self.direction == Direction::ToUser {
135            SV39_WRITE
136        } else {
137            SV39_READ
138        };
139        let end = self.user_ptr + self.len;
140        let mut page = self.user_ptr & !(PAGE_SIZE - 1);
141        let mut result = Ok(());
142        while page < end {
143            let lo = page.max(self.user_ptr);
144            let hi = (page + PAGE_SIZE).min(end);
145            let chunk = hi - lo;
146            let Some((bits, pa)) = sv39_leaf(root_pa, page) else {
147                return Err(CopyError::InvalidAddress);
148            };
149            if bits & (SV39_VALID | SV39_USER | need_pte) != SV39_VALID | SV39_USER | need_pte {
150                return Err(CopyError::InvalidAddress);
151            }
152            let offset = lo - page;
153            let alias = phys_to_virt(pa + offset);
154            let (src, dst): (*const u8, *mut u8) = if self.direction == Direction::FromUser {
155                (alias as *const u8, other.wrapping_add(lo - self.user_ptr))
156            } else {
157                (other.wrapping_add(lo - self.user_ptr), alias as *mut u8)
158            };
159            if commit_guarded(src, dst, chunk, alias, alias + chunk).is_err() {
160                result = Err(CopyError::InvalidAddress);
161                break;
162            }
163            page += PAGE_SIZE;
164        }
165        result
166    }
167}
