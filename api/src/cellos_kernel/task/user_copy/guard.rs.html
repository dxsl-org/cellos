1//! Per-hart recoverable-fault guard: window arming, the guarded byte-copy
2//! core, and the trap-facing glue contract. The RV64 trap handler routes
3//! guard-owned page faults here by rewinding `sepc` to the landing pad
4//! published in `user_copy_guard_resume_pc`.
5
6use super::range::CopyError;
7use crate::task::hart_local;
8use core::sync::atomic::Ordering;

10/// Run one byte-copy chunk inside the per-hart recoverable fault window.
11///
12/// `guard_lo`/`guard_hi` describe the faulting-address window published to the
13/// trap hook: Sas commits publish the user range (faulting `stval` is the user
14/// VA); Domain commits publish the kernel-linear alias range actually touched.
15pub(super) fn commit_guarded(
16    src: *const u8,
17    dst: *mut u8,
18    len: usize,
19    guard_lo: usize,
20    guard_hi: usize,
21) -> Result<(), CopyError> {
22    if len == 0 {
23        return Ok(());
24    }
25    let window = GuardWindow::arm(guard_lo, guard_hi);
26    let hart = unsafe { hart_local::current_hart() };
27    // SAFETY: `hart` is the executing hart's static ViHartLocal; the atomic is
28    // USIZE-repr and only this hart touches its own slot, so publishing the
29    // landing-pad address through a plain store is race-free.
30    let slot = core::ptr::addr_of!(hart.user_copy_guard_resume_pc) as *mut usize;
31    // SAFETY: the guard window is armed around the call; see
32    // guarded_byte_copy for the register/trap contract.
33    let failed = unsafe { guarded_byte_copy(src, dst, len, slot) };
34    drop(window);
35    if failed == 0 {
36        Ok(())
37    } else {
38        Err(CopyError::InvalidAddress)
39    }
40}

42/// Arm the per-hart recoverable-fault window. Interrupts are masked FIRST so a
43/// timer preempting mid-window can never switch tasks with the guard set; the
44/// active flag publishes last (Release) so the trap hook never sees an armed
45/// guard with an unpublished range.
46struct GuardWindow {
47    saved_sstatus: usize,
48}
49
50impl GuardWindow {
51    fn arm(lo: usize, hi: usize) -> Self {
52        let hart = unsafe { hart_local::current_hart() };
53        hart.user_copy_guard_start.store(lo, Ordering::Relaxed);
54        hart.user_copy_guard_end.store(hi, Ordering::Relaxed);
55        hart.user_copy_guard_resume_pc.store(0, Ordering::Relaxed);
56        let saved_sstatus = crate::hal::arch::save_and_disable_interrupts();
57        let sum_bit: usize = 1 << 18;
58        unsafe {
59            core::arch::asm!(
60                "csrs sstatus, {sum}",
61                sum = in(reg) sum_bit,
62                options(nomem, nostack),
63            );
64        }
65        hart.user_copy_guard_active.store(1, Ordering::Release);
66        Self { saved_sstatus }
67    }
68}
69
70impl Drop for GuardWindow {
71    fn drop(&mut self) {
72        let hart = unsafe { hart_local::current_hart() };
73        hart.user_copy_guard_active.store(0, Ordering::Release);
74        // SAFETY: paired with save_and_disable_interrupts in arm(); no lock,
75        // allocation, or callback ran while interrupts were masked.
76        unsafe { crate::hal::arch::restore_sstatus(self.saved_sstatus) };
77    }
78}

80/// Byte-copy loop with a trap-recoverable error landing pad (local label `3`).
81/// Direction-neutral: reading `src` and writing `dst` byte-by-byte covers both
82/// copy directions through the boundary; the caller picks which side points
83/// at user memory and publishes the matching guard range.
84///
85/// Register contract: the block touches ONLY caller-saved registers and never
86/// changes `sp` or `ra`. When the trap hook recovers a guard fault it rewrites
87/// `sepc` to label `3`; the trap epilogue then restores the fault-time register
88/// image, which is exactly the state label `3` expects — it sets the return
89/// value, clears the resume slot, and returns through the untouched `ra`.
90///
91/// # Safety
92/// The guard window must be armed around the call, `resume_slot` must point at
93/// this hart's `user_copy_guard_resume_pc`, and both ranges must be resolved
94/// or recoverable per the module docs.
95#[inline(never)]
96unsafe fn guarded_byte_copy(
97    src: *const u8,
98    dst: *mut u8,
99    len: usize,
100    resume_slot: *mut usize,
101) -> usize {
102    let ret;
103    // SAFETY: see the function contract above.
104    unsafe {
105        core::arch::asm!(
106            "la   {tmp}, 3f",
107            "sd   {tmp}, 0({slot})",
108            "li   {idx}, 0",
109            "2:",
110            "bgeu {idx}, {cnt}, 1f",
111            "add  {sa}, {sptr}, {idx}",
112            "lbu  {byte}, 0({sa})",
113            "add  {da}, {dptr}, {idx}",
114            "sb   {byte}, 0({da})",
115            "addi {idx}, {idx}, 1",
116            "j    2b",
117            "1:",
118            "li   {out}, 0",
119            "j    4f",
120            "3:",
121            "li   {out}, 1",
122            "4:",
123            "sd   zero, 0({slot})",
124            sptr = in(reg) src,
125            dptr = in(reg) dst,
126            cnt = in(reg) len,
127            slot = in(reg) resume_slot,
128            tmp = out(reg) _,
129            idx = out(reg) _,
130            sa = out(reg) _,
131            da = out(reg) _,
132            byte = out(reg) _,
133            out = lateout(reg) ret,
134            options(nostack)
135        );
136    }
137    ret
138}

140/// Defense-in-depth: a hart switching tasks must never carry an armed guard
141/// into the next task. The guarded window masks interrupts, so in the current
142/// design this can only fire if a future change reintroduces preemption
143/// mid-window; clearing costs one store either way.
144pub(crate) fn clear_guard_for_context_switch() {
145    let hart = unsafe { hart_local::current_hart() };
146    hart.user_copy_guard_active.store(0, Ordering::Release);
147}

149/// TEST HOOK: fire one genuine guard-recovered page fault through the full
150/// trap path — asm landing pad, `vi_user_copy_guard_fault` claim, `sepc`
151/// rewind — by loading from a kernel-linear alias of an unmapped physical
152/// hole inside an armed window. Returns whether the landing pad reported the
153/// recovered fault and cleared its own resume slot.
154#[cfg(feature = "test-hooks")]
155pub(crate) fn forced_guard_fault_recovers_for_test(hole_pa: usize) -> bool {
156    use crate::memory::paging::PAGE_SIZE;
157    let hole_va = crate::memory::frame::phys_to_virt(hole_pa);
158    let mut sink = [0u8; 8];
159    let window = GuardWindow::arm(hole_va, hole_va + PAGE_SIZE);
160    let hart = unsafe { hart_local::current_hart() };
161    let slot = core::ptr::addr_of!(hart.user_copy_guard_resume_pc) as *mut usize;
162    // SAFETY: the window is armed around the call; the load faults on purpose
163    // and recovery resumes at the landing pad without writing `sink`.
164    let failed = unsafe { guarded_byte_copy(hole_va as *const u8, sink.as_mut_ptr(), 8, slot) };
165    drop(window);
166    failed == 1 && hart.user_copy_guard_resume_pc.load(Ordering::Acquire) == 0
167}
