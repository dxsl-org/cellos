10pub(super) const SV39_VALID: usize = 1 << 0;
11pub(super) const SV39_READ: usize = 1 << 1;
12pub(super) const SV39_WRITE: usize = 1 << 2;
13pub(super) const SV39_USER: usize = 1 << 4;
14
15/// Walk one Sv39 root and return `(PTE flag bits, resolved physical address)`
16/// for the leaf translating `va`. Reads page-table memory through the kernel
17/// linear mapping, so the walk itself can never take a user page fault.
18pub(crate) fn sv39_leaf(root_pa: usize, va: usize) -> Option<(usize, usize)> {
19    #[cfg(target_arch = "riscv64")]
20    {
21        let mut table_pa = root_pa;
22        for level in 0..3usize {
23            let index = (va >> (30 - 9 * level)) & 0x1FF;
24            let pte = unsafe {
25                core::ptr::read_volatile(
26                    crate::memory::frame::phys_to_virt(table_pa + index * 8) as *const u64
27                )
28            } as usize;
29            if pte & SV39_VALID == 0 {
30                return None;
31            }
32            if pte & (SV39_READ | SV39_WRITE | (1 << 3)) != 0 {
33                let shift = 12 + 9 * (2 - level);
34                let off_mask = (1usize << shift) - 1;
35                let pa = (((pte >> 10) & ((1usize << 44) - 1)) << 12) | (va & off_mask);
36                return Some((pte & 0xFF, pa));
37            }
38            table_pa = ((pte >> 10) & ((1usize << 44) - 1)) << 12;
39        }
40        None
41    }
42    #[cfg(target_arch = "aarch64")]
43    {
44        let table =
45            unsafe { &*(crate::memory::frame::phys_to_virt(root_pa) as *const hal::PageTable) };
46        let leaf = table.leaf_entry(va)?;
47        let pa = (leaf as usize) & 0x0000_ffff_ffff_f000;
48        let mut bits = 0usize;
49        if leaf & 1 != 0 {
50            bits |= SV39_VALID;
51        }
52        if leaf & (1 << 6) != 0 {
53            bits |= SV39_USER;
54        }
55        bits |= SV39_READ;
56        if leaf & (1 << 7) == 0 {
57            bits |= SV39_WRITE;
58        }
59        Some((bits, pa))
60    }
61    #[cfg(target_arch = "x86_64")]
62    {
63        let mut table_pa = root_pa;
64        for level in 0..4usize {
65            let shift = 39 - 9 * level;
66            let index = (va >> shift) & 0x1FF;
67            let pte = unsafe {
68                core::ptr::read_volatile(
69                    crate::memory::frame::phys_to_virt(table_pa + index * 8) as *const u64
70                )
71            } as usize;
72            if pte & 1 == 0 {
73                return None;
74            }
75            if level == 3 || pte & (1 << 7) != 0 {
76                let pa = pte & 0x000f_ffff_ffff_f000;
77                let mut bits = SV39_VALID | SV39_READ;
78                if pte & (1 << 1) != 0 {
79                    bits |= SV39_WRITE;
80                }
81                if pte & (1 << 2) != 0 {
82                    bits |= SV39_USER;
83                }
84                return Some((bits, pa));
85            }
86            table_pa = pte & 0x000f_ffff_ffff_f000;
87        }
88        None
89    }
90    #[cfg(not(any(
91        target_arch = "riscv64",
92        target_arch = "aarch64",
93        target_arch = "x86_64"
94    )))]
95    None
96}

98/// Physical root of the currently resident domain table, if paging is active.
99pub(crate) fn current_satp_root() -> Option<usize> {
100    #[cfg(target_arch = "riscv64")]
101    {
102        let satp: usize;
103        unsafe {
104            core::arch::asm!(
105                "csrr {satp}, satp",
106                satp = out(reg) satp,
107                options(nostack, nomem)
108            );
109        }
110        if (satp >> 60) != 8 {
111            return None;
112        }
113        Some((satp & ((1usize << 44) - 1)) << 12)
114    }
115    #[cfg(target_arch = "aarch64")]
116    {
117        let ttbr0: usize;
118        unsafe {
119            core::arch::asm!(
120                "mrs {ttbr0}, ttbr0_el1",
121                ttbr0 = out(reg) ttbr0,
122                options(nostack, nomem)
123            );
124        }
125        Some(ttbr0 & 0x0000_ffff_ffff_f000)
126    }
127    #[cfg(target_arch = "x86_64")]
128    {
129        let cr3: usize;
130        unsafe {
131            core::arch::asm!(
132                "mov {cr3}, cr3",
133                cr3 = out(reg) cr3,
134                options(nostack, nomem)
135            );
136        }
137        Some(cr3 & !0xFFF)
138    }
139    #[cfg(not(any(
140        target_arch = "riscv64",
141        target_arch = "aarch64",
142        target_arch = "x86_64"
143    )))]
144    None
145}

147/// Probe every page of `[ptr, ptr+len)` against BOTH the mapping ledger and
148/// the live Sv39 PTEs of the private root under an acquired reader pin.
149/// Zero heap allocations: pages are validated and committed directly from `(ptr, len)`.
150pub(super) fn stage_domain<'a>(
151    arc: &'a AddressSpace,
152    direction: Direction,
153    ptr: usize,
154    len: usize,
155) -> Result<PinnedCopy<'a>, CopyError> {
156    // 1. Acquire the reader pin FIRST: revoke paths must wait for us to drain
157    // before they may remove a single PTE or reuse frames.
158    let reader = arc
159        .acquire_copy_reader()
160        .map_err(|_| CopyError::InvalidAddress)?;
161
162    let end = ptr.checked_add(len).ok_or(CopyError::InvalidAddress)?;
163    let root_pa = arc.root_ppn() << 12;
164    let need_flag = if direction == Direction::ToUser {
165        Flags::WRITE
166    } else {
167        Flags::READ
168    };
169    let need_pte = if direction == Direction::ToUser {
170        SV39_WRITE
171    } else {
172        SV39_READ
173    };
174
175    // 2. Validate every page under the held pin.
176    let mut page = ptr & !(PAGE_SIZE - 1);
177    let satp_is_domain = current_satp_root() == Some(root_pa);
178    while page < end {
179        let (ledger_flags, ledger_pa) =
180            arc.page_proof_for(page).ok_or(CopyError::InvalidAddress)?;
181        // Strict U + R/W contract for private roots.
182        if ledger_flags.bits() & (Flags::USER | need_flag) != Flags::USER | need_flag {
183            return Err(CopyError::InvalidAddress);
184        }
185        if !satp_is_domain {
186            let (pte_bits, pte_pa) = sv39_leaf(root_pa, page).ok_or(CopyError::InvalidAddress)?;
187            if pte_bits & (SV39_VALID | SV39_USER | need_pte) != SV39_VALID | SV39_USER | need_pte {
188                return Err(CopyError::InvalidAddress);
189            }
190            if pte_pa != ledger_pa {
191                return Err(CopyError::InvalidAddress);
192            }
193        }
194        page += PAGE_SIZE;
195    }
196
197    Ok(PinnedCopy {
198        arc,
199        reader,
200        direction,
201        user_ptr: ptr,
202        len,
203    })
204}

206/// Probe `[ptr, ptr+len)` against the currently resident root.
207///
208/// Sas RETAINS today's shared-address-space semantics: cells and the kernel
209/// share one map and the kernel dereferences through SUM=1, so any live
210/// mapping with the required R/W right qualifies — the U bit is deliberately
211/// NOT required here. Kernel-image rejection for Sas therefore reduces to the
212/// canonical bound plus whatever the walk refuses; private-root views enforce
213/// the strict U + R/W ledger contract above.
214///
215/// Documented retained legacy exposure: Sas identity-maps RAM, so a wild
216/// pointer inside the identity-mapped RAM range passes this probe by design —
217/// only VAs with no live Sv39 leaf (true holes between mapped regions) are
218/// rejected before any byte moves. Concurrent-unmap residual is legacy Sas
219/// semantics, converted to a recoverable error by the guard.
220pub(super) fn probe_sas(ptr: usize, len: usize, direction: Direction) -> Result<(), CopyError> {
221    let end = ptr.checked_add(len).ok_or(CopyError::InvalidAddress)?;
222    let Some(root_pa) = current_satp_root() else {
223        return Err(CopyError::InvalidAddress);
224    };
225    let need = if direction == Direction::ToUser {
226        SV39_WRITE | SV39_USER
227    } else {
228        SV39_READ | SV39_USER
229    };
230    let mut page = ptr & !(PAGE_SIZE - 1);
231    while page < end {
232        let (bits, _) = sv39_leaf(root_pa, page).ok_or(CopyError::InvalidAddress)?;
233        if bits & (SV39_VALID | need) != SV39_VALID | need {
234            return Err(CopyError::InvalidAddress);
235        }
236        page += PAGE_SIZE;
237    }
238    Ok(())
239}

241/// TEST HOOK: stage (probe + pin) a domain copy without committing it, so a
242/// fixture can inject a protocol violation between the two passes.
243#[cfg(feature = "test-hooks")]
244pub(crate) fn stage_domain_for_test<'a>(
245    arc: &'a AddressSpace,
246    ptr: usize,
247    len: usize,
248    write: bool,
249) -> Result<PinnedCopy<'a>, CopyError> {
250    let direction = if write {
251        Direction::ToUser
252    } else {
253        Direction::FromUser
254    };
255    stage_domain(arc, direction, ptr, len)
