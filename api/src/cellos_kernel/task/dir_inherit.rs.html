1//! Carrying a directory-handle set from a spawner to the cell it spawns.
2//!
3//! The kernel's whole role is transport and provenance. It copies a set of
4//! opaque values from one task to another and can later state which task the
5//! values came from. It never resolves a handle, never checks one, and holds no
6//! table of its own — the filesystem service issues, validates and revokes, and
7//! remains the only place a handle means anything.
8//!
9//! ## Why the record cannot be a second source of truth
10//! A cell acquires handles from the filesystem service after it starts, so the
11//! set recorded here is not "what the cell holds" and must never be read as
12//! that. It is only "what its spawner named at spawn". Treating it as authority
13//! would create a copy that drifts from the service's table, and drift in this
14//! direction widens what a cell may reach without anything failing to compile.
15//!
16//! ## Ordering contract
17//! [`take_for_launch`] runs while the scheduler lock is held and produces the
18//! complete record installed in the unpublished child before insertion.
19
20use api::dir_handles::{DirHandleSet, InheritedDirHandles};
21
22/// Record `set` as the handles `tid`'s next spawn should pass on.
23///
24/// Replaces any previously staged set: the most recent statement wins, and a
25/// staged set that is never spawned against costs one bounded inline array.
26pub fn stage(tid: usize, set: DirHandleSet) {
27    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
28        if let Some(task) = sched.tasks.get_mut(&tid) {
29            task.staged_dirs = set;
30        }
31    }
32}
33
34/// Discard whatever `tid` has staged.
35///
36/// Called after every spawn `tid` attempts, successful or not. A failed spawn
37/// that left its set staged would hand it to whichever child `tid` created
38/// next — an over-broad grant nobody asked for and nobody would see.
39pub fn clear_staged(tid: usize) {
40    stage(tid, DirHandleSet::EMPTY);
41}
42
43/// Consume `spawner`'s staged set and build the child's immutable provenance.
44///
45/// Call while holding the scheduler lock immediately before atomic publication.
46pub(crate) fn take_for_launch(
47    sched: &mut super::scheduler::Scheduler,
48    spawner: usize,
49) -> InheritedDirHandles {
50    if spawner == 0 {
51        return InheritedDirHandles::NONE;
52    }
53    let Some(spawner_task) = sched.tasks.get_mut(&spawner) else {
54        return InheritedDirHandles::NONE;
55    };
56    let set = core::mem::replace(&mut spawner_task.staged_dirs, DirHandleSet::EMPTY);
57    if set.is_empty() || spawner_task.cell_id.0 == 0 {
58        return InheritedDirHandles::NONE;
59    }
60    let inherited = InheritedDirHandles {
61        spawner_cell_id: spawner_task.cell_id.0,
62        spawner_generation: spawner_task.cell_generation,
63        set,
64    };
65    log::info!(
66        "[dirs] next cell inherits {} dir handle(s) from cell {}",
67        set.len(),
68        inherited.spawner_cell_id
69    );
70    inherited
71}

73/// Returns `None` when `cell_id` names no live root-owned Cell generation.
74/// Cell IDs are bounded reusable quota slots, never task-table indexes.
75pub fn attestation_for(cell_id: u64) -> Option<api::dir_attestation::ViDirHandleAttestation> {
76    if cell_id == 0 {
77        return None;
78    }
79    let guard = super::SCHEDULER.lock();
80    let sched = guard.as_ref()?;
81    let owner = sched.live_cell_owner_for_id(types::CellId(cell_id))?;
82    let task = sched.tasks.get(&(owner.root_tid as usize))?;
83    Some(api::dir_attestation::ViDirHandleAttestation {
84        cell_id,
85        generation: owner.generation,
86        spawner_cell_id: task.inherited_dirs.spawner_cell_id,
87        spawner_generation: task.inherited_dirs.spawner_generation,
88        set: task.inherited_dirs.set,
89    })
