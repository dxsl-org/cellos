1//! Carrying a directory-handle set from a spawner to the cell it spawns.
2//!
3//! The kernel's whole role is transport and provenance. It copies a set of
4//! opaque values from one task to another and can later state which task the
5//! values came from. It never resolves a handle, never checks one, and holds no
6//! table of its own — the filesystem service issues, validates and revokes, and
7//! remains the only place a handle means anything.
8//!
9//! ## Why the record cannot be a second source of truth
10//! A cell acquires handles from the filesystem service after it starts, so the
11//! set recorded here is not "what the cell holds" and must never be read as
12//! that. It is only "what its spawner named at spawn". Treating it as authority
13//! would create a copy that drifts from the service's table, and drift in this
14//! direction widens what a cell may reach without anything failing to compile.
15//!
16//! ## Ordering contract
17//! [`install_on_child`] runs inside the scheduler-lock critical section that
18//! creates the child, before the child is reachable by any hart. A child is
19//! therefore never observable with an unset record, so the service can never
20//! read "no set" for a cell that was in fact given one.
21
22use api::dir_handles::{DirHandleSet, InheritedDirHandles};
23
24/// Record `set` as the handles `tid`'s next spawn should pass on.
25///
26/// Replaces any previously staged set: the most recent statement wins, and a
27/// staged set that is never spawned against costs one bounded inline array.
28pub fn stage(tid: usize, set: DirHandleSet) {
29    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
30        if let Some(task) = sched.tasks.get_mut(&tid) {
31            task.staged_dirs = set;
32        }
33    }
34}
35
36/// Discard whatever `tid` has staged.
37///
38/// Called after every spawn `tid` attempts, successful or not. A failed spawn
39/// that left its set staged would hand it to whichever child `tid` created
40/// next — an over-broad grant nobody asked for and nobody would see.
41pub fn clear_staged(tid: usize) {
42    stage(tid, DirHandleSet::EMPTY);
43}
44
45/// Move `spawner`'s staged set onto the freshly created `child`.
46///
47/// Call while holding the scheduler lock that created `child` and before
48/// releasing it; see the module ordering contract. `spawner` of `0` (kernel
49/// boot, no calling task) installs nothing.
50///
51/// The spawner's identity is captured here rather than trusted later: by the
52/// time the filesystem service asks, the spawner may be gone or replaced.
53pub fn install_on_child(sched: &mut super::scheduler::Scheduler, child: usize, spawner: usize) {
54    if spawner == 0 || spawner == child {
55        return;
56    }
57    let Some(spawner_task) = sched.tasks.get_mut(&spawner) else {
58        return;
59    };
60    let set = core::mem::replace(&mut spawner_task.staged_dirs, DirHandleSet::EMPTY);
61    if set.is_empty() {
62        return;
63    }
64    let spawner_cell_id = spawner_task.cell_id.0;
65    let spawner_generation = spawner_task.cell_generation;
66    // A set whose source the kernel cannot name is worthless to the service that
67    // must check it against that source, so drop it rather than pass an
68    // unattributable grant.
69    if spawner_cell_id == 0 {
70        log::warn!("[dirs] dropping staged set from unattributable spawner tid {spawner}");
71        return;
72    }
73    if let Some(child_task) = sched.tasks.get_mut(&child) {
74        child_task.inherited_dirs = InheritedDirHandles {
75            spawner_cell_id,
76            spawner_generation,
77            set,
78        };
79        log::info!(
80            "[dirs] cell {} inherits {} dir handle(s) from cell {}",
81            child,
82            set.len(),
83            spawner_cell_id
84        );
85    }
86}

88/// What the kernel is prepared to state about `cell_id`'s inherited handles.
89///
90/// Returns `None` when `cell_id` names no live cell. The record lives on the
91/// cell's own task — the one whose tid the `CellId` is derived from — so a
92/// thread of that cell resolves to the same answer rather than to an empty set
93/// of its own.
94pub fn attestation_for(cell_id: u64) -> Option<api::dir_attestation::ViDirHandleAttestation> {
95    if cell_id == 0 {
96        return None;
97    }
98    let guard = super::SCHEDULER.lock();
99    let task = guard.as_ref()?.tasks.get(&(cell_id as usize))?;
100    if task.cell_id.0 != cell_id {
101        return None;
102    }
103    Some(api::dir_attestation::ViDirHandleAttestation {
104        cell_id,
105        generation: task.cell_generation,
106        spawner_cell_id: task.inherited_dirs.spawner_cell_id,
107        spawner_generation: task.inherited_dirs.spawner_generation,
108        set: task.inherited_dirs.set,
109    })
