1//! Per-cell completion queue: the kernel-owned place an asynchronous result lands.
2//!
3//! The queue is heap memory the kernel allocates and holds through the task
4//! record. It is deliberately **not** a grant: a grant can be unregistered or
5//! freed by the cell that owns it, and a queue that can vanish under an
6//! in-flight operation turns every completion into a write to whatever was
7//! handed the frame next. Because the queue is kernel-owned, appending needs no
8//! address resolution and no allocator, which is what lets the append path hold
9//! one leaf lock and nothing else.
10//!
11//! Two rules make a completion infallible:
12//!
13//! - **A slot is reserved when the operation is submitted**, from the submitting
14//!   context, where a refusal is an ordinary error the caller can act on. By the
15//!   time anything is in flight its landing place already exists.
16//! - **Appending never allocates and never grows the queue.** The drainable ring
17//!   holds slot indexes and is exactly as long as the slot array, so it cannot
18//!   overflow: a slot contributes at most one entry between reservation and
19//!   release.
20//!
21//! Waking the task that is waiting is a *separate*, deferred step. An append may
22//! run in interrupt context, and making a task runnable needs the scheduler lock,
23//! which must never be taken from there. So an append only raises a flag; the
24//! flag is turned into a scheduler wake by [`deliver_pending_wakes`], called from
25//! `yield_cpu` after `SCHEDULER` is free — the same deferral the grant reap uses
26//! for work that cannot run under the sweep's locks.
27//!
28//! Lock order: this module's leaf lock is taken alone. It is never held across a
29//! call that takes `SCHEDULER`, `FRAME_ALLOCATOR` or `KERNEL_ROOT`, and none of
30//! those is held when it is taken.
31

40/// Operations one cell may have outstanding at once.
41///
42/// The cap is the backpressure: a cell that submits without draining is refused
43/// at submission rather than served, so the failure is visible, attributable and
44/// confined to the cell that caused it.
45pub const QUEUE_CAPACITY: usize = 32;
46/// Maximum number of concurrent registered waiters per cell completion queue.
47pub const MAX_WAITERS: usize = 32;
48
49/// Names one submission within the cell that reserved it.
50///
51/// A slot — not a task id — is the identifier a completion carries. Task ids are
52/// reused, do not survive a restart, and cannot tell two operations of one task
53/// apart. A slot is meaningful only inside the cell that owns the queue and dies
54/// with it.
55#[derive(Clone, Copy, PartialEq, Eq, Debug)]
56pub struct SlotId(u16);
57
58impl SlotId {
59    /// Position of this slot in its queue. Always `< QUEUE_CAPACITY`.
60    pub fn index(self) -> usize {
61        self.0 as usize
62    }
63}

65/// A finished operation, ready to be handed to whoever was waiting.
66#[derive(Clone, Copy, PartialEq, Eq, Debug)]
67pub struct Completion {
68    pub slot: SlotId,
69    /// One public source bit from `api::completion::source`, or zero for
70    /// kernel-only queue self-tests that do not model a public producer.
71    pub source: u32,
72    /// Operation result. Negative values are reserved for errors; the exact
73    /// encoding belongs with the first operation migrated onto the queue, not
74    /// here, so nothing is frozen before a real caller exists.
75    pub result: isize,
76}
77
78#[derive(Clone, Copy, PartialEq, Eq)]
79enum Slot {
80    /// No submission owns this index.
81    Free,
82    /// A submission owns this index; its result has not arrived.
83    Reserved,
84    /// The result arrived and has not been drained.
85    Done { source: u32, result: isize },
86}
87
88struct Ring {
89    slots: [Slot; QUEUE_CAPACITY],
90    /// Indexes of slots holding an undrained result, oldest first.
91    drainable: [u16; QUEUE_CAPACITY],
92    head: usize,
93    len: usize,
94    #[cfg(feature = "p01-trace")]
95    p01_obligations: [Option<crate::p01_resources::Obligation>; QUEUE_CAPACITY],
96}
97
98/// Raised by any append, cleared by [`deliver_pending_wakes`]. A single relaxed
99/// load keeps the common tick — no completions anywhere — at one atomic read.
100static WAKES_PENDING: AtomicBool = AtomicBool::new(false);
101
102/// The landing ground for one cell's completions.
103///
104/// Shared by every thread of the cell through `Arc`, so the last reference to
105/// disappear frees it. A completion source that still holds a reference keeps
106/// the queue alive past the cell's death: the append then lands in a queue
107/// nobody drains, which leaks a bounded allocation rather than corrupting
108/// whatever took the memory next.
109pub struct CompletionQueue {
110    cell: CellId,
111    ring: Spinlock<Ring>,
112    /// Task to make runnable when a completion arrives; 0 means nobody is
113    /// waiting. See [`CompletionQueue::register_waiter`] for the contract.
114    /// Registered waiters in this Cell.
115    waiters: Spinlock<Vec<usize>>,
116    /// Task to make runnable when a completion arrives; 0 means nobody is
117    /// waiting. Kept for atomic single-waiter fast path.
118    waiter: AtomicUsize,
119    wake_requested: AtomicBool,
120    #[cfg(feature = "p01-trace")]
121    p01_tid: AtomicUsize,
122}

124impl CompletionQueue {
125    /// Capacity of the pre-sized waiter table (test observation only).
126    #[cfg(test)]
127    pub(crate) fn waiter_capacity(&self) -> usize {
128        self.waiters.lock().capacity()
129    }

131    /// A queue whose waiter table already holds [`MAX_WAITERS`] entries, so
132    /// [`register_waiter`](Self::register_waiter) never allocates on the wait
133    /// path. `None` when the table cannot be reserved.
134    fn new(cell: CellId) -> Option<Self> {
135        let mut waiters = Vec::new();
136        waiters.try_reserve_exact(MAX_WAITERS).ok()?;
137        Some(Self {
138            cell,
139            ring: Spinlock::new(Ring {
140                slots: [Slot::Free; QUEUE_CAPACITY],
141                drainable: [0; QUEUE_CAPACITY],
142                head: 0,
143                len: 0,
144                #[cfg(feature = "p01-trace")]
145                p01_obligations: [None; QUEUE_CAPACITY],
146            }),
147            waiters: Spinlock::new(waiters),
148            waiter: AtomicUsize::new(0),
149            wake_requested: AtomicBool::new(false),
150            #[cfg(feature = "p01-trace")]
151            p01_tid: AtomicUsize::new(0),
152        })
153    }
154
155    #[cfg(feature = "p01-trace")]
156    pub(crate) fn p01_bind(&self, owner: crate::p01_measurement::Owner) {
157        let refused = {
158            let previous = self.p01_tid.load(Ordering::Acquire);
159            if previous == owner.tid as usize {
160                return;
161            }
162            self.p01_tid
163                .compare_exchange(0, owner.tid as usize, Ordering::Release, Ordering::Relaxed)
164                .is_err()
165        };
166        if refused {
167            crate::p01_resources::unknown(owner, 2);
168        }
169    }

171    /// Cell this queue belongs to.
172    pub fn cell(&self) -> CellId {
173        self.cell
174    }

176    /// Take a slot for an operation about to be submitted.
177    ///
178    /// # Errors
179    /// `None` when every slot is taken. Call this from the submitting cell's own
180    /// context and turn it into an ordinary error there: refusing an interrupt
181    /// handler that is trying to report completion would deadlock the system it
182    /// was serving, so the refusal has to happen here, before anything is in
183    /// flight.
184    pub fn reserve(&self) -> Option<SlotId> {
185        let mut ring = self.ring.lock();
186        let Some(index) = ring.slots.iter().position(|s| *s == Slot::Free) else {
187            #[cfg(feature = "p01-trace")]
188            if let Some(owner) =
189                crate::p01_resources::owner_for_tid(self.p01_tid.load(Ordering::Acquire))
190            {
191                crate::p01_resources::unknown(owner, 2);
192            }
193            return None;
194        };
195        ring.slots[index] = Slot::Reserved;
196        #[cfg(feature = "p01-trace")]
197        if let Some(owner) =
198            crate::p01_resources::owner_for_tid(self.p01_tid.load(Ordering::Acquire))
199        {
200            ring.p01_obligations[index] = crate::p01_resources::completion_obligation(owner);
201        }
202        Some(SlotId(index as u16))
203    }

205    /// Record the result of the operation holding `slot`.
206    ///
207    /// Takes this queue's leaf lock and nothing else, so it is safe from
208    /// interrupt context. Waking the waiter is deliberately not done here — see
209    /// [`deliver_pending_wakes`].
210    ///
211    /// # Returns
212    /// `true` once the result is stored. `false` only for a kernel-side protocol
213    /// violation — completing a slot that was never reserved, or completing one
214    /// twice — which is logged and leaves the queue untouched. A slot obtained
215    /// from [`reserve`](Self::reserve) and completed once always succeeds, which
216    /// is the property the whole design rests on.
217    #[must_use = "a refused completion means a slot was misused and the waiter will hang"]
218    pub fn complete(&self, slot: SlotId, result: isize) -> bool {
219        self.complete_from(slot, api::completion::source::UNSPECIFIED, result)
220    }
221
222    /// Record a public source and result for the operation holding `slot`.
223    ///
224    /// Production completion sources must use this method so the userspace
225    /// record identifies why the parked executor woke. The zero-source
226    /// [`complete`](Self::complete) wrapper remains for kernel-only queue tests.
227    #[must_use = "a refused completion means a slot was misused and the waiter will hang"]
228    pub fn complete_from(&self, slot: SlotId, source: u32, result: isize) -> bool {
229        let index = slot.index();
230        // The whole lock-holding region is this block, and it contains no call
231        // that could reach another lock — not the logger, which is why the
232        // refusal below reports after the guard is dropped.
233        let refused = {
234            let mut ring = self.ring.lock();
235            match ring.slots.get(index) {
236                Some(Slot::Reserved) => {
237                    ring.slots[index] = Slot::Done { source, result };
238                    #[cfg(feature = "p01-trace")]
239                    crate::p01_resources::reservation_terminal(&mut ring.p01_obligations[index]);
240                    // Cannot overflow: `drainable` is as long as `slots`, and a
241                    // slot is only pushed on the Reserved -> Done edge, which
242                    // happens once per reservation.
243                    let position = (ring.head + ring.len) % QUEUE_CAPACITY;
244                    ring.drainable[position] = index as u16;
245                    ring.len += 1;
246                    // Publish the wake while the ring transition is serialized.
247                    // A drain that empties the ring can then cancel this exact
248                    // request without erasing a later completion's wake.
249                    self.wake_requested.store(true, Ordering::Release);
250                    None
251                }
252                Some(Slot::Free) => Some("free"),
253                Some(Slot::Done { .. }) => Some("already done"),
254                None => Some("out of range"),
255            }
256        };
257
258        if let Some(why) = refused {
259            log::error!(
260                "[completion] cell {} slot {} completed while {}",
261                self.cell.0,
262                index,
263                why
264            );
265            return false;
266        }
267
268        // The queue-local request is already visible with the entry. Raising the
269        // global scan flag afterwards may cause a harmless empty scan if the
270        // submitter drains first, but can never hide a completion.
271        WAKES_PENDING.store(true, Ordering::Release);
272        true
273    }

275    /// Withdraw a reservation whose operation never started.
276    ///
277    /// This is not a completion and must never be expressed as one: completing
278    /// a slot raises a wake request, and a request raised by the submitter
279    /// itself — which is already running — is still outstanding when that same
280    /// task parks next, so the park is cancelled the instant it begins. A
281    /// submitter that withdraws and re-submits in a loop would then never
282    /// sleep at all.
283    ///
284    /// # Returns
285    /// `true` when the slot was reserved and is now free. `false` when it
286    /// already holds a result: the submitter lost the race with the source and
287    /// must [`drain`](Self::drain) that result rather than discard it.
288    #[must_use = "a refused withdrawal means a result is waiting and would be lost"]
289    pub fn release(&self, slot: SlotId) -> bool {
290        let mut ring = self.ring.lock();
291        match ring.slots.get(slot.index()) {
292            Some(Slot::Reserved) => {
293                ring.slots[slot.index()] = Slot::Free;
294                #[cfg(feature = "p01-trace")]
295                crate::p01_resources::finish(ring.p01_obligations[slot.index()].take(), 1);
296                true
297            }
298            _ => false,
299        }
300    }

302    /// Take the oldest undrained completion, releasing its slot for reuse.
303    ///
304    /// Holds the same single leaf lock as [`complete`](Self::complete); the
305    /// report of an inconsistency is deliberately made after the guard drops.
306    pub fn drain(&self) -> Option<Completion> {
307        let taken = {
308            let mut ring = self.ring.lock();
309            if ring.len == 0 {
310                return None;
311            }
312            let index = ring.drainable[ring.head] as usize;
313            ring.head = (ring.head + 1) % QUEUE_CAPACITY;
314            ring.len -= 1;
315            if ring.len == 0 {
316                self.wake_requested.store(false, Ordering::Release);
317            }
318            match ring.slots[index] {
319                Slot::Done { source, result } => {
320                    ring.slots[index] = Slot::Free;
321                    #[cfg(feature = "p01-trace")]
322                    crate::p01_resources::finish(ring.p01_obligations[index].take(), 1);
323                    Ok(Completion {
324                        slot: SlotId(index as u16),
325                        source,
326                        result,
327                    })
328                }
329                // Unreachable while the ring and the slot array agree; treat a
330                // disagreement as an empty drain rather than a stale read.
331                _ => Err(index),
332            }
333        };
334
335        match taken {
336            Ok(completion) => Some(completion),
337            Err(index) => {
338                log::error!(
339                    "[completion] cell {} drained slot {} that holds no result",
340                    self.cell.0,
341                    index
342                );
343                None
344            }
345        }
346    }

348    /// Slots taken by submissions that have not yet reported a result.
349    pub fn reserved(&self) -> usize {
350        let ring = self.ring.lock();
351        ring.slots.iter().filter(|s| **s == Slot::Reserved).count()
352    }

354    /// Completions recorded and not yet drained.
355    pub fn drainable(&self) -> usize {
356        self.ring.lock().len
357    }

359    /// Name `tid` as the task to make runnable when a completion arrives.
360    ///
361    /// # Contract
362    /// The registrant must publish its `WaitKind::Completion` park before yielding.
363    /// A NET_RX waiter is also intentionally interruptible by queued IPC; that
364    /// path readies the task but leaves this queue's reservation and any
365    /// concurrent completion under the waiter's normal cleanup protocol.
366    /// TIMER waiters remain deadline-only.
367    pub fn register_waiter(&self, tid: usize) {
368        self.waiter.store(tid, Ordering::Release);
369        let mut list = self.waiters.lock();
370        if !list.contains(&tid) && list.len() < MAX_WAITERS {
371            list.push(tid);
372        }
373    }
374
375    /// Stop waking `tid` for this queue if it still owns the registration.
376    ///
377    /// Returns `false` when another waiter replaced `tid`; the older wait must
378    /// not erase the newer waiter's registration while unwinding.
379    pub fn clear_waiter(&self, tid: usize) -> bool {
380        let _ = self
381            .waiter
382            .compare_exchange(tid, 0, Ordering::AcqRel, Ordering::Acquire);
383        let mut list = self.waiters.lock();
384        if let Some(pos) = list.iter().position(|&x| x == tid) {
385            list.swap_remove(pos);
386            true
387        } else {
388            false
389        }
390    }
391
392    /// Consume a pending wake request, yielding the task to make runnable.
393    pub fn take_wake_request(&self) -> Option<usize> {
394        if !self.wake_requested.swap(false, Ordering::AcqRel) {
395            return None;
396        }
397        match self.waiter.load(Ordering::Acquire) {
398            0 => None,
399            tid => Some(tid),
400        }
401    }
402
403    /// Consume pending wake requests, yielding all registered tasks to make runnable.
404    pub fn drain_wake_requests(&self, out: &mut Vec<usize>) {
405        if !self.wake_requested.swap(false, Ordering::AcqRel) {
406            return;
407        }
408        let list = self.waiters.lock();
409        for &tid in list.iter() {
410            if tid != 0 && !out.contains(&tid) {
411                out.push(tid);
412            }
413        }
414        let w = self.waiter.load(Ordering::Acquire);
415        if w != 0 && !out.contains(&w) {
416            out.push(w);
417        }
418    }
419}

421#[cfg(feature = "p01-trace")]
422impl Drop for CompletionQueue {
423    fn drop(&mut self) {
424        let mut ring = self.ring.lock();
425        for token in &mut ring.p01_obligations {
426            crate::p01_resources::finish(token.take(), 1);
427        }
428        if let Some(owner) =
429            crate::p01_resources::owner_for_tid(self.p01_tid.load(Ordering::Acquire))
430        {
431            crate::p01_resources::checked(owner, 2);
432        }
433    }
434}

436/// The queue serving `tid`'s cell, created on first use.
437///
438/// Threads of one cell share one queue, so the handle is looked up across the
439/// cell before a new one is allocated. Returns `None` when `tid` names no live task
440/// or the queue cannot be allocated.
441///
442/// The queue and its fixed waiter table are kernel bookkeeping, charged to the
443/// kernel (cell 0) as `async_ipc::Operations::push` charges its storage. They
444/// are allocated once per Cell here, so later waits allocate nothing.
445///
446/// Call from a submitting context with `SCHEDULER` held. The append path never
447/// comes here — it uses the handle it was given, which is the reason it needs no
448/// lock but its own.
449pub fn queue_for(sched: &mut Scheduler, tid: usize) -> Option<Arc<CompletionQueue>> {
450    let cell = sched.tasks.get(&tid)?.cell_id;
451    if let Some(existing) = sched.tasks.get(&tid).and_then(|t| t.completion.clone()) {
452        return Some(existing);
453    }
454    let queue = match sched
455        .tasks
456        .values()
457        .filter(|t| t.cell_id == cell)
458        .find_map(|t| t.completion.clone())
459    {
460        Some(shared) => shared,
461        None => {
462            #[cfg(feature = "p01-trace")]
463            let _semantic = crate::memory::heap::p01_owner::creation(
464                None,
465                crate::memory::heap::p01_owner::COMPLETION_SHARED,
466            );
467            let previous = crate::sched::hart_local::current_cell_id();
468            crate::sched::hart_local::set_current_cell_id(0);
469            let created = CompletionQueue::new(cell).map(Arc::new);
470            crate::sched::hart_local::set_current_cell_id(previous);
471            created?
472        }
473    };
474    #[cfg(feature = "p01-trace")]
475    if let Some(owner) = crate::p01_resources::owner_for_tid(tid) {
476        queue.p01_bind(owner);
477    }
478    sched.tasks.get_mut(&tid)?.completion = Some(queue.clone());
479    Some(queue)
480}

482/// Whether any append has asked for a wake since the last delivery.
483pub fn wakes_pending() -> bool {
484    WAKES_PENDING.load(Ordering::Acquire)
485}

487/// A state a completion wake may leave, versus one it must not disturb.
488///
489/// `Ready`/`Running` are already runnable, and terminal task records must not
490/// be revived. `Frozen` is held deliberately by hot-swap — unfreezing it here
491/// would run a cell mid-swap.
492fn is_parked(state: &TaskState) -> bool {
493    !matches!(
494        state,
495        TaskState::Ready
496            | TaskState::Running
497            | TaskState::Terminated
498            | TaskState::Retiring
499            | TaskState::Frozen { .. }
500    )
501}

503/// Turn flagged completions into scheduler wakes.
504///
505/// Runs where `SCHEDULER` may be taken and neither `FRAME_ALLOCATOR` nor
506/// `KERNEL_ROOT` is held — that is, from `yield_cpu` and not from an append.
507/// Requires `SCHEDULER` to be held by the caller.
508pub fn deliver_pending_wakes(sched: &mut Scheduler) {
509    // Cleared before the scan: an append landing during the scan re-raises the
510    // flag, so at worst its wake waits for the next tick. Clearing afterwards
511    // would swallow it entirely.
512    if !WAKES_PENDING.swap(false, Ordering::AcqRel) {
513        return;
514    }
515
516    // Reached through the task table, so a queue kept alive only by a completion
517    // source outlives its cell unreachable — which is the intended end state: its
518    // registered waiter is dead, and the alternative, a wake for a tid the table
519    // has since reissued, would make a stranger runnable.
520    let mut requested: Vec<usize> = Vec::new();
521    for task in sched.tasks.values() {
522        if let Some(queue) = task.completion.as_ref() {
523            queue.drain_wake_requests(&mut requested);
524        }
525    }
526
527    let mut runnable: Vec<usize> = Vec::new();
528    for tid in requested {
529        if let Some(task) = sched.tasks.get_mut(&tid) {
530            if is_parked(&task.state) {
531                task.state = TaskState::Ready;
532                runnable.push(tid);
533            }
534        }
535    }
536    for tid in runnable {
537        sched.push_ready(tid);
538    }
