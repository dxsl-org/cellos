99fn authority_is_current(
100    sched: &super::scheduler::Scheduler,
101    authority: &CallerLaunchAuthority,
102) -> bool {
103    sched.tasks.get(&authority.tid).is_some_and(|task| {
104        task.cell_generation == authority.generation
105            && super::cap::CapSet::of_task(task).intersect(authority.ceiling) == authority.ceiling
106    })
107}

109/// Publish a fully prepared ELF task. Every fallible check precedes mutation;
110/// the final ready-queue push is the only runnable publication point.
111pub fn publish_prepared(
112    prepared: super::PreparedElfTask,
113    mut state: TaskLaunchState,
114) -> Result<(usize, usize), ViError> {
115    // Phase-01 containment preflight (ADR-0019 §2.1): a live grant record that
116    // names a private-root owner or receiver cannot be drained by this kernel, so
117    // a domain-class publication is refused while one exists. This runs before the
118    // scheduler lock is taken because the grant tables are only ever acquired in
119    // the documented `*_GRANT_TABLE → SCHEDULER` order.
120    #[cfg(feature = "native-domains")]
121    if state.is_domain {
122        crate::loader::domain_admission::refuse_while_domain_grant_live()?;
123    }
124    let mut scheduler = super::SCHEDULER.lock();
125    let sched = scheduler.as_mut().ok_or(ViError::Unknown)?;
126    let tid = sched.next_task_id;
127    let requested_cell_id = prepared.requested_cell_id();
128    if requested_cell_id.0 != 0
129        && (requested_cell_id.0 as usize) >= crate::memory::cell_quota::MAX_CELLS
130    {
131        return Err(ViError::PermissionDenied);
132    }
133    if state
134        .caller
135        .as_ref()
136        .is_some_and(|a| !authority_is_current(sched, a))
137    {
138        return Err(ViError::PermissionDenied);
139    }
140    // AP-11 must be injectable at the binding boundary even when this boot
141    // corpus has no frozen replacement source to invalidate.
142    crate::loader::atomic_checkpoint("AP-11")?;
143    if state
144        .replacement
145        .as_ref()
146        .is_some_and(|r| !r.can_bind(sched))
147    {
148        return Err(ViError::PermissionDenied);
149    }
150
151    let quota = if requested_cell_id == CellId(0) {
152        crate::memory::cell_quota::QuotaReservation::reserve_next(state.quota_limit)?
153    } else {
154        crate::memory::cell_quota::QuotaReservation::reserve(requested_cell_id, state.quota_limit)?
155    };
156    let cell_id = quota.cell_id();
157    if !sched.cell_owner_slot_is_empty(cell_id) {
158        return Err(ViError::AlreadyExists);
159    }
160    crate::loader::atomic_checkpoint("AP-07")?;
161    crate::loader::atomic_checkpoint("AP-08")?;
162    let _kernel_alloc = super::scheduler::KernelAllocationContext::enter();
163
164    let inherited_dirs = super::dir_inherit::take_for_launch(sched, state.inherit_from);
165    // InheritedDirHandles is bounded inline state, not an independently
166    // allocated buffer: its storage is already part of the task allocation.
167    let (mut task, load_base) = prepared.into_task(tid, cell_id)?;
168    task.inherited_dirs = inherited_dirs;
169    // The command line lives on the task record itself, so it is reachable by
170    // this task's own `StateRestore` and by nothing else, and it is installed
171    // before the task can ever be selected.
172    task.inherited_argv = state.argv.take();
173    #[cfg(cellos_p01_trace)]
174    crate::memory::heap::p01_owner::adopt_task(&task);
175    task.syscall_allowlist = state.syscall_allowlist;
176    task.cluster_mode = state.cluster_mode;
177    task.cluster_id = state.cluster_id;
178    task.priority = state.priority;
179    task.pku_key = state.pku_key;
180    task.pku_value = state.pku_value;
181    task.is_critical = state.is_critical;
182    task.root_tid = tid;
183    let owner = api::cell_owner::CellOwner::new(cell_id.0, task.cell_generation, tid as u64);
184    state.granted.apply_to(&mut task);
185    #[cfg(feature = "test-hooks")]
186    {
187        task.development_silo_registration_cap = state
188            .routes
189            .development_silo
190            .then(super::cap::DevelopmentSiloRegistrationCap::new);
191    }
192    if let Some(platform) = state.platform.take() {
193        platform.commit_into(&mut task);
194    }
195    if let Some(replacement) = state.replacement.take() {
196        replacement.commit_into(&mut task);
197    }
198    #[cfg(all(
199        feature = "native-domains",
200        any(
201            target_arch = "riscv64",
202            target_arch = "aarch64",
203            target_arch = "x86_64"
204        )
205    ))]
206    if state.is_domain {
207        // ADR-0019 §2.1 — the single on-path admission control. The lease is
208        // taken before anything is created and re-checked immediately before the
209        // task becomes runnable, so a concurrent drain either linearizes before
210        // this admission or is observed by it. A denial is final: no task, no
211        // domain, and no SAS fallback.
212        let path = state.measurement.as_ref().map_or("", |measurement| measurement.path.as_str());
213        let lease = crate::loader::domain_admission::admit_for_launch(path, state.granted)?;
214        if let (Some(ks), Some(us), Some(seg)) =
215            (&task.kernel_stack, &task.user_stack, &task.segment_mem)
216        {
217            let domain = crate::memory::address_space::create_cell_domain(ks, us, seg,
218                #[cfg(cellos_p01_trace)] crate::p01_resources::owner(&task))
219                .map_err(|_| ViError::OutOfMemory)?;
220            if !lease.remains_enabled() {
221                return Err(crate::loader::domain_admission::drain_refusal());
222            }
223            #[cfg(target_arch = "riscv64")]
224            let arch_tag = "SATP";
225            #[cfg(target_arch = "aarch64")]
226            let arch_tag = "TTBR0";
227            #[cfg(target_arch = "x86_64")]
228            let arch_tag = "CR3";
229            log::warn!(
230                "[domain] admitted cell '{}' to Tier 2 Paged Domain ({} isolation)",
231                task.name,
232                arch_tag
233            );
234            task.bind_address_space(domain);
235        } else {
236            log::error!(
237                "[domain] cannot create Tier 2 domain for '{}': missing stacks or segments",
238                task.name
239            );
240            return Err(ViError::InvalidInput);
241        }
242    }
243    #[cfg(not(all(
244        feature = "native-domains",
245        any(
246            target_arch = "riscv64",
247            target_arch = "aarch64",
248            target_arch = "x86_64"
249        )
250    )))]
251    if state.is_domain {
252        log::error!(
253            "[domain] Tier 2 Native Domain requested for '{}' but unsupported on this architecture/configuration",
254            task.name
255        );
256        return Err(ViError::NotSupported);
257    }
258
259    #[cfg(cellos_p01_trace)]
260    crate::memory::heap::p01_owner::task_backing(&task);
261    // Private policy selection is tied to the measured artifact and the live
262    // kernel-bootstrap publisher, never a caller-selected budget or task name.
263    #[cfg(cellos_g1_cpu)]
264    if let Some(measurement) = state.measurement.as_ref() {
265        if measurement.path == "/bin/bench" {
266            let attempt = state.trace.ok_or(ViError::PermissionDenied)?;
267            let caller = state.caller.as_ref().ok_or(ViError::PermissionDenied)?;
268            let publisher = sched.tasks.get(&caller.tid).ok_or(ViError::PermissionDenied)?;
269            if !attempt.selected()
270                || attempt.parent != crate::p01_measurement::Owner::of(publisher)
271                || caller.generation != publisher.cell_generation
272                || attempt.parent.tid != attempt.parent.root
273                || !matches!(attempt.route, 1 | 2)
274                || measurement.digest.as_slice() != &crate::p01_measurement::MANIFEST[32..64]
275                || crate::p01_measurement::selected_owner(
276                    publisher.cell_id.0, publisher.cell_generation).is_some()
277            {
278                return Err(ViError::PermissionDenied);
279            }
280            super::cpu_accounting::bind(sched, owner).map_err(|_| ViError::PermissionDenied)?;
281        }
282    }
283
284    // Scheduler maps/route/audit storage are kernel bookkeeping, not child
285    // objects merely because they are allocated during a selected launch.
286    #[cfg(cellos_p01_trace)]
287    let _heap_bookkeeping = crate::memory::heap::p01_owner::bookkeeping();
288    {
289        #[cfg(cellos_p01_trace)]
290        let _semantic = crate::memory::heap::p01_owner::creation(None,
291            crate::memory::heap::p01_owner::SCHEDULER_SHARED);
292        sched.tasks.insert(tid, task);
293    }
294    assert!(
295        sched.publish_live_cell_owner(owner),
296        "reserved CellId owner slot must be empty"
297    );
298    sched.next_task_id = tid.checked_add(1).expect("task id space exhausted");
299    quota.commit();
300    crate::loader::commit_launch_routes(tid, cell_id, state.routes);
301    #[cfg(cellos_p01_trace)]
302    {
303        let task = sched.tasks.get_mut(&tid).expect("committed publication");
304        let diagnostic_owner = crate::p01_measurement::Owner::of(task);
305        if let Some(measurement) = state.measurement.as_ref() {
306            if state.bootstrap_init {
307                crate::p01_measurement::arm(diagnostic_owner, measurement.digest);
308            } else {
309                crate::p01_measurement::publication(diagnostic_owner, state.trace, measurement.digest);
310            }
311        }
312        crate::memory::heap::p01_owner::bind(state.trace, diagnostic_owner);
313        crate::p01_resources::published(task);
314    }
315    if let Some(measurement) = state.measurement.take() {
316        crate::measurement_log::commit_staged(tid, measurement.path, measurement.digest);
317    }
318    crate::audit::log_event(
319        crate::audit::AuditEvent::CellSpawn,
320        &crate::audit::encode_u32x2(tid as u32, 0),
321    );
322    crate::loader::observe_pre_ready(sched, tid);
323    sched.push_ready(tid);
324    Ok((tid, load_base))
325}

327/// Creator-quota claim for resources a prepared child will retain or briefly
328/// occupy. It is charged to the creator's exact live Cell before preparation
329/// succeeds and refunded exactly once when the preparation is dropped; a
330/// successful commit transfers the retained portion (the historical measurement
331/// entry and its path) to the retained log owner instead of refunding it.
332struct PublicationClaim {
333    cell: usize,
334    retained: usize,
335    transient: usize,
336    armed: bool,
337}
338impl PublicationClaim {
339    fn charge(cell: usize, retained: usize, transient: usize) -> Result<Self, ViError> {
340        let total = retained.checked_add(transient).ok_or(ViError::OutOfMemory)?;
341        if !crate::memory::cell_quota::charge(cell, total) {
342            return Err(ViError::OutOfMemory);
343        }
344        Ok(Self { cell, retained, transient, armed: true })
345    }
346    /// Commit: the queue slot is consumed by the runnable task, so only that
347    /// transient part is refunded; the retained log history stays attributed.
348    fn transfer(mut self) {
349        crate::memory::cell_quota::refund(self.cell, self.transient);
350        self.armed = false;
351    }
352}
353impl Drop for PublicationClaim {
354    fn drop(&mut self) {
355        if self.armed {
356            crate::memory::cell_quota::refund(self.cell, self.retained + self.transient);
357        }
358    }
359}

361/// The task, quota, CSpace, measurement-log and ready-queue slots remain
362/// detached promises until the single session owner transaction commits. This
363/// is not a second live-owner table: the Cell owner slot is `Reserved`, which
364/// no resolve/watch path can observe as live.
365pub(crate) struct SessionPublication {
366    pub(crate) task: alloc::boxed::Box<super::Task>,
367    pub(crate) owner: api::cell_owner::CellOwner,
368    state: TaskLaunchState,
369    quota: crate::memory::cell_quota::QuotaReservation,
370    authority: crate::cell::cspace::AuthorityReservation,
371    measurement: crate::measurement_log::PublicationReservation,
372    ready: super::hart_local::ready::PublicationReservation,
373    claim: PublicationClaim,
374}
375impl SessionPublication {
376    /// Every fallible allocation, quota, owner-slot, log and ready-queue claim
377    /// happens here; `publish` allocates nothing and cannot fail.
378    pub(crate) fn reserve(sched: &mut super::scheduler::Scheduler,
379        prepared: super::PreparedElfTask, state: TaskLaunchState) -> Result<Self, ViError> {
380        let Some(caller) = state.caller.as_ref() else { return Err(ViError::PermissionDenied); };
381        if !authority_is_current(sched, caller) {
382            return Err(ViError::PermissionDenied);
383        }
384        let (creator, creator_generation) = sched.tasks.get(&caller.tid)
385            .map(|t| (t.cell_id.0 as usize, t.cell_generation))
386            .ok_or(ViError::PermissionDenied)?;
387        // A session child is an ordinary governed path launch: never a boot
388        // singleton, domain, hotswap replacement, or service route publisher.
389        if prepared.requested_cell_id() != CellId(0) || state.is_domain || state.platform.is_some()
390            || state.replacement.is_some() || state.routes.block_io || state.routes.input {
391            return Err(ViError::PermissionDenied);
392        }
393        #[cfg(feature = "test-hooks")]
394        if state.routes.development_silo { return Err(ViError::PermissionDenied); }
395        // Heap bookkeeping for the unpublished task is kernel memory, exactly as
396        // in `publish_prepared`; per-principal accounting is the explicit claim.
397        let _kernel_alloc = super::scheduler::KernelAllocationContext::enter();
398        let tid = sched.next_task_id;
399        let next = tid.checked_add(1).ok_or(ViError::OutOfMemory)?;
400        let quota = crate::memory::cell_quota::QuotaReservation::reserve_next(state.quota_limit)?;
401        let cell = quota.cell_id();
402        let (mut task, _) = prepared.into_task(tid, cell)?;
403        task.root_tid = tid;
404        task.syscall_allowlist = state.syscall_allowlist;
405        task.cluster_mode = state.cluster_mode;
406        task.cluster_id = state.cluster_id;
407        task.priority = state.priority;
408        task.pku_key = state.pku_key;
409        task.pku_value = state.pku_value;
410        task.is_critical = state.is_critical;
411        state.granted.apply_to(&mut task);
412        #[cfg(feature = "test-hooks")]
413        { task.development_silo_registration_cap = state.routes.development_silo
414            .then(super::cap::DevelopmentSiloRegistrationCap::new); }
415        let owner = api::cell_owner::CellOwner::new(cell.0, task.cell_generation, tid as u64);
416        let authority = crate::cell::cspace::AuthorityReservation::reserve(cell, owner.generation, state.granted)?;
417        let path_len = state.measurement.as_ref().map_or(0, |m| m.path.len());
418        let claim = PublicationClaim::charge(creator,
419            crate::measurement_log::PublicationReservation::ENTRY_BYTES + path_len,
420            super::hart_local::ready::PublicationReservation::SLOT_BYTES)?;
421        let measurement = crate::measurement_log::PublicationReservation::reserve(
422            (creator as u64, creator_generation))?;
423        let ready = super::hart_local::ready::PublicationReservation::reserve(state.priority)?;
424        sched.reserve_session_owner(owner)?;
425        sched.next_task_id = next;
426        Ok(Self { task, owner, state, quota, authority, measurement, ready, claim })
427    }
428    pub(crate) fn current(&self, sched: &super::scheduler::Scheduler) -> bool {
429        if !sched.session_owner_reserved(self.owner)
430            || self.state.caller.as_ref().is_none_or(|a| !authority_is_current(sched, a)) {
431            return false;
432        }
433        let Some(caller) = self.state.caller.as_ref().and_then(|a| sched.tasks.get(&a.tid)) else { return false; };
434        let Some(measurement) = self.state.measurement.as_ref() else { return false; };
435        let state = crate::loader::launch_profile::CallerLaunchState { name: &caller.name,
436            has_spawn: caller.spawn_cap.is_some(), has_supervisor: caller.supervisor_cap.is_some() };
437        crate::loader::launch_profile::authorize(state, crate::loader::launch_profile::LaunchRoute::Path,
438            &measurement.path).is_some_and(|profile| self.state.granted.intersect(profile.child_ceiling)
439                == self.state.granted && (!profile.requires_lifecycle_authority || caller.spawn_cap.is_some()))
440    }
441    /// Final publication. Called with every owner check already passed, under
442    /// SCHEDULER, and performs no allocation: capacity for the task row, CSpace
443    /// row, log entry and ready entry was promised in `reserve`.
444    pub(crate) fn publish(mut self, sched: &mut super::scheduler::Scheduler) {
445        let tid = self.owner.root_tid as usize;
446        #[cfg(cellos_p01_trace)]
447        {
448            crate::memory::heap::p01_owner::adopt_task(&self.task);
449            crate::memory::heap::p01_owner::task_backing(&self.task);
450        }
451        self.authority.commit();
452        sched.tasks.publish_reserved(tid, self.task);
453        sched.publish_session_owner(self.owner);
454        self.quota.commit();
455        crate::loader::commit_launch_routes(tid, CellId(self.owner.cell_id), self.state.routes);
456        #[cfg(cellos_p01_trace)]
457        {
458            let task = sched.tasks.get_mut(&tid).expect("committed publication");
459            let diagnostic_owner = crate::p01_measurement::Owner::of(task);
460            if let Some(measurement) = self.state.measurement.as_ref() {
461                if self.state.bootstrap_init {
462                    crate::p01_measurement::arm(diagnostic_owner, measurement.digest);
463                } else {
464                    crate::p01_measurement::publication(diagnostic_owner, self.state.trace, measurement.digest);
465                }
466            }
467            crate::memory::heap::p01_owner::bind(self.state.trace, diagnostic_owner);
468            crate::p01_resources::published(task);
469        }
470        if let Some(measurement) = self.state.measurement.take() {
471            self.measurement.commit(tid, measurement.path, measurement.digest);
472        }
473        crate::audit::log_event(
474            crate::audit::AuditEvent::CellSpawn,
475            &crate::audit::encode_u32x2(tid as u32, 0),
476        );
477        crate::loader::observe_pre_ready(sched, tid);
478        // The notification follows the reservation's pinned destination, not
479        // whichever hart happens to commit or whatever RT hart is online now.
480        let (hart, priority) = self.ready.destination();
481        self.ready.commit(tid);
482        self.claim.transfer();
483        sched.pend_preempt_on_hart(hart, priority);
484    }
