109fn authority_is_current(
110    sched: &crate::sched::scheduler::Scheduler,
111    authority: &CallerLaunchAuthority,
112) -> bool {
113    sched.tasks.get(&authority.tid).is_some_and(|task| {
114        task.cell_generation == authority.generation
115            && sched
116                .cell_rights(authority.tid)
117                .intersect(authority.ceiling)
118                == authority.ceiling
119    })
120}

122/// Publish a fully prepared ELF task. Every fallible check precedes mutation;
123/// the final ready-queue push is the only runnable publication point.
124pub fn publish_prepared(
125    prepared: super::PreparedElfTask,
126    mut state: TaskLaunchState,
127) -> Result<(usize, usize), ViError> {
128    // Phase-01 containment preflight (ADR-0019 §2.1): a live grant record that
129    // names a private-root owner or receiver cannot be drained by this kernel, so
130    // a domain-class publication is refused while one exists. This runs before the
131    // scheduler lock is taken because the grant tables are only ever acquired in
132    // the documented `*_GRANT_TABLE → SCHEDULER` order.
133    #[cfg(feature = "native-domains")]
134    if state.is_domain {
135        crate::loader::domain_admission::refuse_while_domain_grant_live()?;
136    }
137    let mut scheduler = super::SCHEDULER.lock();
138    let sched = scheduler.as_mut().ok_or(ViError::Unknown)?;
139    let tid = sched.next_task_id;
140    let requested_cell_id = prepared.requested_cell_id();
141    if requested_cell_id.0 != 0
142        && (requested_cell_id.0 as usize) >= crate::memory::cell_quota::MAX_CELLS
143    {
144        return Err(ViError::PermissionDenied);
145    }
146    if state
147        .caller
148        .as_ref()
149        .is_some_and(|a| !authority_is_current(sched, a))
150    {
151        return Err(ViError::PermissionDenied);
152    }
153    // AP-11 must be injectable at the binding boundary even when this boot
154    // corpus has no frozen replacement source to invalidate.
155    crate::loader::atomic_checkpoint("AP-11")?;
156    if state
157        .replacement
158        .as_ref()
159        .is_some_and(|r| !r.can_bind(sched))
160    {
161        return Err(ViError::PermissionDenied);
162    }
163
164    // The quota row carries the task's generation, so it is minted first.
165    let generation = super::tcb::next_cell_generation();
166    let limits = crate::memory::cell_quota::Limits {
167        memory: state.quota_limit,
168        guest: state.guest_limit,
169    };
170    let quota = if requested_cell_id == CellId(0) {
171        crate::memory::cell_quota::QuotaReservation::reserve_next(generation, limits)?
172    } else {
173        crate::memory::cell_quota::QuotaReservation::reserve(requested_cell_id, generation, limits)?
174    };
175    let cell_id = quota.cell_id();
176    if !sched.cell_owner_slot_is_empty(cell_id) {
177        return Err(ViError::AlreadyExists);
178    }
179    crate::loader::atomic_checkpoint("AP-07")?;
180    crate::loader::atomic_checkpoint("AP-08")?;
181    // Heap bookkeeping for the unpublished task is kernel memory; the creator
182    // is charged explicitly through its `PublicationClaim`.
183    let _kernel =
184        crate::memory::heap::ChargeScope::enter(crate::memory::cell_quota::Charge::KERNEL);
185
186    let inherited_dirs = super::dir_inherit::take_for_launch(sched, state.inherit_from);
187    // InheritedDirHandles is bounded inline state, not an independently
188    // allocated buffer: its storage is already part of the task allocation.
189    let (mut task, load_base) = prepared.into_task(tid, cell_id, generation)?;
190    task.inherited_dirs = inherited_dirs;
191    // The command line lives on the task record itself, so it is reachable by
192    // this task's own `StateRestore` and by nothing else, and it is installed
193    // before the task can ever be selected.
194    task.inherited_argv = state.argv.take();
195    #[cfg(feature = "p01-trace")]
196    crate::memory::heap::p01_owner::adopt_task(&task);
197    task.syscall_allowlist = state.syscall_allowlist;
198    task.priority = state.priority;
199    task.pku_key = state.pku_key;
200    task.pku_value = state.pku_value;
201    task.is_critical = state.is_critical;
202    task.root_tid = tid;
203    task.spawner = state
204        .caller
205        .as_ref()
206        .map(|caller| (caller.cell_id, caller.generation));
207    if let Some(measurement) = state.measurement.as_ref() {
208        // A fresh task from `into_task`, so the first binding always lands.
209        task.bind_image(measurement.image);
210        task.launch_named = measurement.launch_named;
211    }
212    let owner = abi::cell_owner::CellOwner::new(cell_id.0, task.cell_generation, tid as u64);
213    state.granted.apply_to(&mut task);
214    #[cfg(feature = "test-hooks")]
215    {
216        task.development_silo_registration_cap = state
217            .routes
218            .development_silo
219            .then(super::cap::DevelopmentSiloRegistrationCap::new);
220    }
221    if let Some(platform) = state.platform.take() {
222        platform.commit_into(&mut task);
223    }
224    if let Some(replacement) = state.replacement.take() {
225        replacement.commit_into(&mut task);
226    }
227    if state.is_domain {
228        // ADR-0019 §2.1: the one admission policy (see `admit_and_bind_domain`).
229        let path = state
230            .measurement
231            .as_ref()
232            .map_or("", |measurement| measurement.path.as_str());
233        admit_and_bind_domain(&mut task, path, state.granted)?;
234        log_domain_admitted(&task.name);
235    }
236
237    #[cfg(feature = "p01-trace")]
238    crate::memory::heap::p01_owner::task_backing(&task);
239    // Private policy selection is tied to the measured artifact and the live
240    // kernel-bootstrap publisher, never a caller-selected budget or task name.
241    #[cfg(feature = "g1-cpu-accounting")]
242    if let Some(measurement) = state.measurement.as_ref() {
243        if measurement.path == "/bin/bench" {
244            let attempt = state.trace.ok_or(ViError::PermissionDenied)?;
245            let caller = state.caller.as_ref().ok_or(ViError::PermissionDenied)?;
246            let publisher = sched
247                .tasks
248                .get(&caller.tid)
249                .ok_or(ViError::PermissionDenied)?;
250            if !attempt.selected()
251                || attempt.parent != crate::p01_measurement::Owner::of(publisher)
252                || caller.generation != publisher.cell_generation
253                || attempt.parent.tid != attempt.parent.root
254                || !matches!(attempt.route, 1 | 2)
255                || measurement.digest.as_slice() != &crate::p01_measurement::MANIFEST[32..64]
256                || crate::p01_measurement::selected_owner(
257                    publisher.cell_id.0,
258                    publisher.cell_generation,
259                )
260                .is_some()
261            {
262                return Err(ViError::PermissionDenied);
263            }
264            crate::sched::cpu_accounting::bind(sched, owner)
265                .map_err(|_| ViError::PermissionDenied)?;
266        }
267    }
268
269    // Scheduler maps/route/audit storage are kernel bookkeeping, not child
270    // objects merely because they are allocated during a selected launch.
271    #[cfg(feature = "p01-trace")]
272    let _heap_bookkeeping = crate::memory::heap::p01_owner::bookkeeping();
273    {
274        #[cfg(feature = "p01-trace")]
275        let _semantic = crate::memory::heap::p01_owner::creation(
276            None,
277            crate::memory::heap::p01_owner::SCHEDULER_SHARED,
278        );
279        sched.tasks.insert(tid, task);
280    }
281    assert!(
282        sched.publish_live_cell_owner(owner),
283        "reserved CellId owner slot must be empty"
284    );
285    sched.next_task_id = tid.checked_add(1).expect("task id space exhausted");
286    quota.commit();
287    crate::loader::commit_launch_routes(tid, state.routes);
288    #[cfg(feature = "p01-trace")]
289    {
290        let task = sched.tasks.get_mut(&tid).expect("committed publication");
291        let diagnostic_owner = crate::p01_measurement::Owner::of(task);
292        if let Some(measurement) = state.measurement.as_ref() {
293            if state.bootstrap_init {
294                crate::p01_measurement::arm(diagnostic_owner, measurement.digest);
295            } else {
296                crate::p01_measurement::publication(
297                    diagnostic_owner,
298                    state.trace,
299                    measurement.digest,
300                );
301            }
302        }
303        crate::memory::heap::p01_owner::bind(state.trace, diagnostic_owner);
304        crate::p01_resources::published(task);
305    }
306    if let Some(measurement) = state.measurement.take() {
307        crate::measurement_log::commit_staged(tid, measurement.path, measurement.digest);
308    }
309    crate::audit::log_event(
310        crate::audit::AuditEvent::CellSpawn,
311        &crate::audit::encode_u32x2(tid as u32, 0),
312    );
313    crate::loader::observe_pre_ready(sched, tid);
314    if state.trusted_init {
315        // Init's tid keys its launch identity and the `MACHINE_PIN`
316        // registration restriction (ADR-0029 §3); publish before it can run.
317        crate::task::syscall::publish_trusted_init(tid);
318    }
319    sched.push_ready(tid);
320    Ok((tid, load_base))
321}

323/// Creator-quota claim for resources a prepared child will retain or briefly
324/// occupy. It is charged to the creator's exact incarnation before preparation
325/// succeeds and refunded exactly once when the preparation is dropped; a
326/// successful commit keeps the retained portion (the historical measurement
327/// entry and its path) charged to the creator instead of refunding it.
328struct PublicationClaim {
329    retained: crate::memory::cell_quota::Reservation,
330    transient: crate::memory::cell_quota::Reservation,
331}
332impl PublicationClaim {
333    fn charge(
334        creator: crate::memory::cell_quota::Charge,
335        retained: usize,
336        transient: usize,
337    ) -> Result<Self, ViError> {
338        let total = retained
339            .checked_add(transient)
340            .ok_or(ViError::OutOfMemory)?;
341        let mut transient = crate::memory::cell_quota::Reservation::take(
342            creator,
343            crate::memory::cell_quota::Account::Memory,
344            total,
345        )
346        .map_err(|_| ViError::OutOfMemory)?;
347        let retained = transient
348            .split(retained)
349            .expect("the claim covers its retained part");
350        Ok(Self {
351            retained,
352            transient,
353        })
354    }
355    /// Commit: the queue slot is consumed by the runnable task, so only that
356    /// transient part is refunded; the retained log history stays attributed.
357    fn transfer(self) {
358        let Self {
359            retained,
360            transient,
361        } = self;
362        retained.keep();
363        drop(transient);
364    }
365}

367/// The ADR-0019 decision at a session commit.
368#[derive(Clone, Copy, Debug, PartialEq, Eq)]
369pub(crate) enum DomainCommit {
370    /// Not a domain child, or a domain child whose lease and grant state allow it.
371    Admit,
372    /// The admission lease no longer holds (a drain since prepare): `Revoked`.
373    Drained,
374    /// A domain-backed grant record is live: `Denied`.
375    LiveGrant,
376}
377
378/// `lease_enabled` is `None` for an ordinary child. A drain wins over a live
379/// grant: the policy generation the child was admitted under is gone, so the
380/// grant state no longer matters.
381pub(crate) fn domain_commit_gate(lease_enabled: Option<bool>, grant_live: bool) -> DomainCommit {
382    match lease_enabled {
383        None => DomainCommit::Admit,
384        Some(false) => DomainCommit::Drained,
385        Some(true) if grant_live => DomainCommit::LiveGrant,
386        Some(true) => DomainCommit::Admit,
387    }
388}

426/// ADR-0019 §2.1, the single admission policy applied at every publication
427/// point (`publish_prepared` and `SessionPublication`): take the lease before
428/// anything is created, build the private root, recheck the lease, bind. A
429/// denial is final: no domain and no SAS fallback. The caller still rechecks
430/// the returned lease immediately before the task becomes runnable.
431#[cfg(all(feature = "native-domains", cellos_wide))]
432fn admit_and_bind_domain(
433    task: &mut super::Task,
434    path: &str,
435    granted: super::cap::CapSet,
436) -> Result<DomainLease, ViError> {
437    let lease = crate::loader::domain_admission::admit_for_launch(path, granted)?;
438    let (Some(ks), Some(us), Some(seg)) = (&task.kernel_stack, &task.user_stack, &task.segment_mem)
439    else {
440        log::error!(
441            "[domain] cannot create Tier 2 domain for '{}': missing stacks or segments",
442            task.name
443        );
444        return Err(ViError::InvalidInput);
445    };
446    let domain = crate::memory::address_space::create_cell_domain(
447        crate::memory::cell_quota::Charge::of(task),
448        ks,
449        us,
450        seg,
451        #[cfg(feature = "p01-trace")]
452        crate::p01_resources::owner(task),
453    )
454    .map_err(|_| ViError::OutOfMemory)?;
455    if !lease.remains_enabled() {
456        return Err(crate::loader::domain_admission::drain_refusal());
457    }
458    task.bind_address_space(domain);
459    Ok(lease)
460}

491/// The task, quota, CSpace, measurement-log and ready-queue slots remain
492/// detached promises until the single session owner transaction commits. This
493/// is not a second live-owner table: the Cell owner slot is `Reserved`, which
494/// no resolve/watch path can observe as live.
495pub(crate) struct SessionPublication {
496    pub(crate) task: alloc::boxed::Box<super::Task>,
497    pub(crate) owner: abi::cell_owner::CellOwner,
498    state: TaskLaunchState,
499    quota: crate::memory::cell_quota::QuotaReservation,
500    authority: crate::cell::cspace::AuthorityReservation,
501    measurement: crate::measurement_log::PublicationReservation,
502    ready: crate::sched::hart_local::ready::PublicationReservation,
503    claim: PublicationClaim,
504    /// ADR-0019 lease of a domain child, rechecked in `current()` under the same
505    /// SCHEDULER section as `publish`; `None` for an ordinary child.
506    domain_lease: Option<DomainLease>,
507    /// The launch edge route this child was admitted under (`Path` for a kernel
508    /// read, `Elf` for caller-supplied bytes); `current()` re-authorizes it.
509    route: crate::loader::launch_profile::LaunchRoute,
510}
511impl SessionPublication {
512    /// Every fallible allocation, quota, owner-slot, log and ready-queue claim
513    /// happens here; `publish` allocates nothing and cannot fail.
514    pub(crate) fn reserve(
515        sched: &mut crate::sched::scheduler::Scheduler,
516        prepared: super::PreparedElfTask,
517        state: TaskLaunchState,
518        route: crate::loader::launch_profile::LaunchRoute,
519    ) -> Result<Self, ViError> {
520        let Some(caller) = state.caller.as_ref() else {
521            return Err(ViError::PermissionDenied);
522        };
523        if !authority_is_current(sched, caller) {
524            return Err(ViError::PermissionDenied);
525        }
526        let creator = sched
527            .tasks
528            .get(&caller.tid)
529            .map(|t| crate::memory::cell_quota::Charge::of(t))
530            .ok_or(ViError::PermissionDenied)?;
531        // A session child is an ordinary governed path launch: never a boot
532        // singleton, hotswap replacement, or service route publisher. A domain
533        // child goes through the same ADR-0019 policy as `publish_prepared`.
534        if prepared.requested_cell_id() != CellId(0)
535            || state.platform.is_some()
536            || state.replacement.is_some()
537            || state.routes.block_io
538            || state.routes.input
539        {
540            return Err(ViError::PermissionDenied);
541        }
542        #[cfg(feature = "test-hooks")]
543        if state.routes.development_silo {
544            return Err(ViError::PermissionDenied);
545        }
546        // Heap bookkeeping for the unpublished task is kernel memory, exactly as
547        // in `publish_prepared`; per-principal accounting is the explicit claim.
548        let _kernel =
549            crate::memory::heap::ChargeScope::enter(crate::memory::cell_quota::Charge::KERNEL);
550        let tid = sched.next_task_id;
551        let next = tid.checked_add(1).ok_or(ViError::OutOfMemory)?;
552        let generation = super::tcb::next_cell_generation();
553        let quota = crate::memory::cell_quota::QuotaReservation::reserve_next(
554            generation,
555            crate::memory::cell_quota::Limits {
556                memory: state.quota_limit,
557                guest: state.guest_limit,
558            },
559        )?;
560        let cell = quota.cell_id();
561        let (mut task, _) = prepared.into_task(tid, cell, generation)?;
562        task.root_tid = tid;
563        task.syscall_allowlist = state.syscall_allowlist;
564        task.priority = state.priority;
565        task.pku_key = state.pku_key;
566        task.pku_value = state.pku_value;
567        task.is_critical = state.is_critical;
568        state.granted.apply_to(&mut task);
569        if let Some(measurement) = state.measurement.as_ref() {
570            task.bind_image(measurement.image);
571            task.launch_named = measurement.launch_named;
572        }
573        let domain_lease = if state.is_domain {
574            let path = state.measurement.as_ref().map_or("", |m| m.path.as_str());
575            Some(admit_and_bind_domain(&mut task, path, state.granted)?)
576        } else {
577            None
578        };
579        #[cfg(feature = "test-hooks")]
580        {
581            task.development_silo_registration_cap = state
582                .routes
583                .development_silo
584                .then(super::cap::DevelopmentSiloRegistrationCap::new);
585        }
586        let owner = abi::cell_owner::CellOwner::new(cell.0, task.cell_generation, tid as u64);
587        let authority = crate::cell::cspace::AuthorityReservation::reserve(cell, owner.generation)?;
588        let path_len = state.measurement.as_ref().map_or(0, |m| m.path.len());
589        let claim = PublicationClaim::charge(
590            creator,
591            crate::measurement_log::PublicationReservation::ENTRY_BYTES + path_len,
592            crate::sched::hart_local::ready::PublicationReservation::SLOT_BYTES,
593        )?;
594        let measurement = crate::measurement_log::PublicationReservation::reserve((
595            creator.cell().0,
596            creator.generation(),
597        ))?;
598        let ready =
599            crate::sched::hart_local::ready::PublicationReservation::reserve(state.priority)?;
600        sched.reserve_session_owner(owner)?;
601        sched.next_task_id = next;
602        Ok(Self {
603            task,
604            owner,
605            state,
606            quota,
607            authority,
608            measurement,
609            ready,
610            claim,
611            domain_lease,
612            route,
613        })
614    }
615    /// ADR-0019 commit decision for this child, given the live-grant preflight
616    /// `dispatch` took before SCHEDULER. Call under SCHEDULER, right before
617    /// `publish`, after `current()`.
618    pub(crate) fn domain_commit(&self, grant_live: bool) -> DomainCommit {
619        domain_commit_gate(
620            self.domain_lease.map(|lease| lease.remains_enabled()),
621            grant_live,
622        )
623    }
624    pub(crate) fn current(&self, sched: &crate::sched::scheduler::Scheduler) -> bool {
625        if !sched.session_owner_reserved(self.owner)
626            || self
627                .state
628                .caller
629                .as_ref()
630                .is_none_or(|a| !authority_is_current(sched, a))
631        {
632            return false;
633        }
634        let Some(caller_tid) = self.state.caller.as_ref().map(|a| a.tid) else {
635            return false;
636        };
637        // The caller's measured launch identity, never its task name (C-1).
638        let Some(identity) = super::syscall::launch_identity_in(sched, caller_tid) else {
639            return false;
640        };
641        let rights = sched.cell_rights(caller_tid);
642        let Some(measurement) = self.state.measurement.as_ref() else {
643            return false;
644        };
645        let state = crate::loader::launch_profile::CallerLaunchState {
646            name: &identity,
647            has_spawn: rights.spawn,
648            has_supervisor: rights.supervisor,
649        };
650        crate::loader::launch_profile::authorize(state, self.route, &measurement.path).is_some_and(
651            |profile| {
652                self.state.granted.intersect(profile.child_ceiling) == self.state.granted
653                    && (!profile.requires_lifecycle_authority || rights.spawn)
654            },
655        )
656    }
657    /// Final publication. Called with every owner check already passed, under
658    /// SCHEDULER, and performs no allocation: capacity for the task row, CSpace
659    /// row, log entry and ready entry was promised in `reserve`.
660    pub(crate) fn publish(mut self, sched: &mut crate::sched::scheduler::Scheduler) {
661        let tid = self.owner.root_tid as usize;
662        if self.domain_lease.is_some() {
663            log_domain_admitted(&self.task.name);
664        }
665        #[cfg(feature = "p01-trace")]
666        {
667            crate::memory::heap::p01_owner::adopt_task(&self.task);
668            crate::memory::heap::p01_owner::task_backing(&self.task);
669        }
670        self.authority.commit();
671        sched.tasks.publish_reserved(tid, self.task);
672        sched.publish_session_owner(self.owner);
673        self.quota.commit();
674        crate::loader::commit_launch_routes(tid, self.state.routes);
675        #[cfg(feature = "p01-trace")]
676        {
677            let task = sched.tasks.get_mut(&tid).expect("committed publication");
678            let diagnostic_owner = crate::p01_measurement::Owner::of(task);
679            if let Some(measurement) = self.state.measurement.as_ref() {
680                if self.state.bootstrap_init {
681                    crate::p01_measurement::arm(diagnostic_owner, measurement.digest);
682                } else {
683                    crate::p01_measurement::publication(
684                        diagnostic_owner,
685                        self.state.trace,
686                        measurement.digest,
687                    );
688                }
689            }
690            crate::memory::heap::p01_owner::bind(self.state.trace, diagnostic_owner);
691            crate::p01_resources::published(task);
692        }
693        if let Some(measurement) = self.state.measurement.take() {
694            self.measurement
695                .commit(tid, measurement.path, measurement.digest);
696        }
697        crate::audit::log_event(
698            crate::audit::AuditEvent::CellSpawn,
699            &crate::audit::encode_u32x2(tid as u32, 0),
700        );
701        crate::loader::observe_pre_ready(sched, tid);
702        // The notification follows the reservation's pinned destination, not
703        // whichever hart happens to commit or whatever RT hart is online now.
704        let (hart, priority) = self.ready.destination();
705        self.ready.commit(tid);
706        self.claim.transfer();
707        sched.pend_preempt_on_hart(hart, priority);
708    }
709}
