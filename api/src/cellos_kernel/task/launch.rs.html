104fn authority_is_current(
105    sched: &crate::sched::scheduler::Scheduler,
106    authority: &CallerLaunchAuthority,
107) -> bool {
108    sched.tasks.get(&authority.tid).is_some_and(|task| {
109        task.cell_generation == authority.generation
110            && sched
111                .cell_rights(authority.tid)
112                .intersect(authority.ceiling)
113                == authority.ceiling
114    })
115}

117/// Publish a fully prepared ELF task. Every fallible check precedes mutation;
118/// the final ready-queue push is the only runnable publication point.
119pub fn publish_prepared(
120    prepared: super::PreparedElfTask,
121    mut state: TaskLaunchState,
122) -> Result<(usize, usize), ViError> {
123    // Phase-01 containment preflight (ADR-0019 §2.1): a live grant record that
124    // names a private-root owner or receiver cannot be drained by this kernel, so
125    // a domain-class publication is refused while one exists. This runs before the
126    // scheduler lock is taken because the grant tables are only ever acquired in
127    // the documented `*_GRANT_TABLE → SCHEDULER` order.
128    #[cfg(feature = "native-domains")]
129    if state.is_domain {
130        crate::loader::domain_admission::refuse_while_domain_grant_live()?;
131    }
132    let mut scheduler = super::SCHEDULER.lock();
133    let sched = scheduler.as_mut().ok_or(ViError::Unknown)?;
134    let tid = sched.next_task_id;
135    let requested_cell_id = prepared.requested_cell_id();
136    if requested_cell_id.0 != 0
137        && (requested_cell_id.0 as usize) >= crate::memory::cell_quota::MAX_CELLS
138    {
139        return Err(ViError::PermissionDenied);
140    }
141    if state
142        .caller
143        .as_ref()
144        .is_some_and(|a| !authority_is_current(sched, a))
145    {
146        return Err(ViError::PermissionDenied);
147    }
148    // AP-11 must be injectable at the binding boundary even when this boot
149    // corpus has no frozen replacement source to invalidate.
150    crate::loader::atomic_checkpoint("AP-11")?;
151    if state
152        .replacement
153        .as_ref()
154        .is_some_and(|r| !r.can_bind(sched))
155    {
156        return Err(ViError::PermissionDenied);
157    }
158
159    let quota = if requested_cell_id == CellId(0) {
160        crate::memory::cell_quota::QuotaReservation::reserve_next(state.quota_limit)?
161    } else {
162        crate::memory::cell_quota::QuotaReservation::reserve(requested_cell_id, state.quota_limit)?
163    };
164    let cell_id = quota.cell_id();
165    if !sched.cell_owner_slot_is_empty(cell_id) {
166        return Err(ViError::AlreadyExists);
167    }
168    crate::loader::atomic_checkpoint("AP-07")?;
169    crate::loader::atomic_checkpoint("AP-08")?;
170    let _kernel_alloc = crate::sched::scheduler::KernelAllocationContext::enter();
171
172    let inherited_dirs = super::dir_inherit::take_for_launch(sched, state.inherit_from);
173    // InheritedDirHandles is bounded inline state, not an independently
174    // allocated buffer: its storage is already part of the task allocation.
175    let (mut task, load_base) = prepared.into_task(tid, cell_id)?;
176    task.inherited_dirs = inherited_dirs;
177    // The command line lives on the task record itself, so it is reachable by
178    // this task's own `StateRestore` and by nothing else, and it is installed
179    // before the task can ever be selected.
180    task.inherited_argv = state.argv.take();
181    #[cfg(feature = "p01-trace")]
182    crate::memory::heap::p01_owner::adopt_task(&task);
183    task.syscall_allowlist = state.syscall_allowlist;
184    task.priority = state.priority;
185    task.pku_key = state.pku_key;
186    task.pku_value = state.pku_value;
187    task.is_critical = state.is_critical;
188    task.root_tid = tid;
189    if let Some(measurement) = state.measurement.as_ref() {
190        // A fresh task from `into_task`, so the first binding always lands.
191        task.bind_image(measurement.image);
192        task.launch_named = measurement.launch_named;
193    }
194    let owner = api::cell_owner::CellOwner::new(cell_id.0, task.cell_generation, tid as u64);
195    state.granted.apply_to(&mut task);
196    #[cfg(feature = "test-hooks")]
197    {
198        task.development_silo_registration_cap = state
199            .routes
200            .development_silo
201            .then(super::cap::DevelopmentSiloRegistrationCap::new);
202    }
203    if let Some(platform) = state.platform.take() {
204        platform.commit_into(&mut task);
205    }
206    if let Some(replacement) = state.replacement.take() {
207        replacement.commit_into(&mut task);
208    }
209    if state.is_domain {
210        // ADR-0019 §2.1: the one admission policy (see `admit_and_bind_domain`).
211        let path = state
212            .measurement
213            .as_ref()
214            .map_or("", |measurement| measurement.path.as_str());
215        admit_and_bind_domain(&mut task, path, state.granted)?;
216        log_domain_admitted(&task.name);
217    }
218
219    #[cfg(feature = "p01-trace")]
220    crate::memory::heap::p01_owner::task_backing(&task);
221    // Private policy selection is tied to the measured artifact and the live
222    // kernel-bootstrap publisher, never a caller-selected budget or task name.
223    #[cfg(feature = "g1-cpu-accounting")]
224    if let Some(measurement) = state.measurement.as_ref() {
225        if measurement.path == "/bin/bench" {
226            let attempt = state.trace.ok_or(ViError::PermissionDenied)?;
227            let caller = state.caller.as_ref().ok_or(ViError::PermissionDenied)?;
228            let publisher = sched
229                .tasks
230                .get(&caller.tid)
231                .ok_or(ViError::PermissionDenied)?;
232            if !attempt.selected()
233                || attempt.parent != crate::p01_measurement::Owner::of(publisher)
234                || caller.generation != publisher.cell_generation
235                || attempt.parent.tid != attempt.parent.root
236                || !matches!(attempt.route, 1 | 2)
237                || measurement.digest.as_slice() != &crate::p01_measurement::MANIFEST[32..64]
238                || crate::p01_measurement::selected_owner(
239                    publisher.cell_id.0,
240                    publisher.cell_generation,
241                )
242                .is_some()
243            {
244                return Err(ViError::PermissionDenied);
245            }
246            crate::sched::cpu_accounting::bind(sched, owner)
247                .map_err(|_| ViError::PermissionDenied)?;
248        }
249    }
250
251    // Scheduler maps/route/audit storage are kernel bookkeeping, not child
252    // objects merely because they are allocated during a selected launch.
253    #[cfg(feature = "p01-trace")]
254    let _heap_bookkeeping = crate::memory::heap::p01_owner::bookkeeping();
255    {
256        #[cfg(feature = "p01-trace")]
257        let _semantic = crate::memory::heap::p01_owner::creation(
258            None,
259            crate::memory::heap::p01_owner::SCHEDULER_SHARED,
260        );
261        sched.tasks.insert(tid, task);
262    }
263    assert!(
264        sched.publish_live_cell_owner(owner),
265        "reserved CellId owner slot must be empty"
266    );
267    sched.next_task_id = tid.checked_add(1).expect("task id space exhausted");
268    quota.commit();
269    crate::loader::commit_launch_routes(tid, state.routes);
270    #[cfg(feature = "p01-trace")]
271    {
272        let task = sched.tasks.get_mut(&tid).expect("committed publication");
273        let diagnostic_owner = crate::p01_measurement::Owner::of(task);
274        if let Some(measurement) = state.measurement.as_ref() {
275            if state.bootstrap_init {
276                crate::p01_measurement::arm(diagnostic_owner, measurement.digest);
277            } else {
278                crate::p01_measurement::publication(
279                    diagnostic_owner,
280                    state.trace,
281                    measurement.digest,
282                );
283            }
284        }
285        crate::memory::heap::p01_owner::bind(state.trace, diagnostic_owner);
286        crate::p01_resources::published(task);
287    }
288    if let Some(measurement) = state.measurement.take() {
289        crate::measurement_log::commit_staged(tid, measurement.path, measurement.digest);
290    }
291    crate::audit::log_event(
292        crate::audit::AuditEvent::CellSpawn,
293        &crate::audit::encode_u32x2(tid as u32, 0),
294    );
295    crate::loader::observe_pre_ready(sched, tid);
296    if state.trusted_init {
297        // Init's tid keys its launch identity and the `MACHINE_PIN`
298        // registration restriction (ADR-0029 §3); publish before it can run.
299        crate::task::syscall::publish_trusted_init(tid);
300    }
301    sched.push_ready(tid);
302    Ok((tid, load_base))
303}

305/// Creator-quota claim for resources a prepared child will retain or briefly
306/// occupy. It is charged to the creator's exact live Cell before preparation
307/// succeeds and refunded exactly once when the preparation is dropped; a
308/// successful commit transfers the retained portion (the historical measurement
309/// entry and its path) to the retained log owner instead of refunding it.
310struct PublicationClaim {
311    cell: usize,
312    retained: usize,
313    transient: usize,
314    armed: bool,
315}
316impl PublicationClaim {
317    fn charge(cell: usize, retained: usize, transient: usize) -> Result<Self, ViError> {
318        let total = retained
319            .checked_add(transient)
320            .ok_or(ViError::OutOfMemory)?;
321        if !crate::memory::cell_quota::charge(cell, total) {
322            return Err(ViError::OutOfMemory);
323        }
324        Ok(Self {
325            cell,
326            retained,
327            transient,
328            armed: true,
329        })
330    }
331    /// Commit: the queue slot is consumed by the runnable task, so only that
332    /// transient part is refunded; the retained log history stays attributed.
333    fn transfer(mut self) {
334        crate::memory::cell_quota::refund(self.cell, self.transient);
335        self.armed = false;
336    }
337}
338impl Drop for PublicationClaim {
339    fn drop(&mut self) {
340        if self.armed {
341            crate::memory::cell_quota::refund(self.cell, self.retained + self.transient);
342        }
343    }
344}

346/// The ADR-0019 decision at a session commit.
347#[derive(Clone, Copy, Debug, PartialEq, Eq)]
348pub(crate) enum DomainCommit {
349    /// Not a domain child, or a domain child whose lease and grant state allow it.
350    Admit,
351    /// The admission lease no longer holds (a drain since prepare): `Revoked`.
352    Drained,
353    /// A domain-backed grant record is live: `Denied`.
354    LiveGrant,
355}
356
357/// `lease_enabled` is `None` for an ordinary child. A drain wins over a live
358/// grant: the policy generation the child was admitted under is gone, so the
359/// grant state no longer matters.
360pub(crate) fn domain_commit_gate(lease_enabled: Option<bool>, grant_live: bool) -> DomainCommit {
361    match lease_enabled {
362        None => DomainCommit::Admit,
363        Some(false) => DomainCommit::Drained,
364        Some(true) if grant_live => DomainCommit::LiveGrant,
365        Some(true) => DomainCommit::Admit,
366    }
367}

405/// ADR-0019 §2.1, the single admission policy applied at every publication
406/// point (`publish_prepared` and `SessionPublication`): take the lease before
407/// anything is created, build the private root, recheck the lease, bind. A
408/// denial is final: no domain and no SAS fallback. The caller still rechecks
409/// the returned lease immediately before the task becomes runnable.
410#[cfg(all(
411    feature = "native-domains",
412    any(
413        target_arch = "riscv64",
414        target_arch = "aarch64",
415        target_arch = "x86_64"
416    )
417))]
418fn admit_and_bind_domain(
419    task: &mut super::Task,
420    path: &str,
421    granted: super::cap::CapSet,
422) -> Result<DomainLease, ViError> {
423    let lease = crate::loader::domain_admission::admit_for_launch(path, granted)?;
424    let (Some(ks), Some(us), Some(seg)) = (&task.kernel_stack, &task.user_stack, &task.segment_mem)
425    else {
426        log::error!(
427            "[domain] cannot create Tier 2 domain for '{}': missing stacks or segments",
428            task.name
429        );
430        return Err(ViError::InvalidInput);
431    };
432    let domain = crate::memory::address_space::create_cell_domain(
433        ks,
434        us,
435        seg,
436        #[cfg(feature = "p01-trace")]
437        crate::p01_resources::owner(task),
438    )
439    .map_err(|_| ViError::OutOfMemory)?;
440    if !lease.remains_enabled() {
441        return Err(crate::loader::domain_admission::drain_refusal());
442    }
443    task.bind_address_space(domain);
444    Ok(lease)
445}

487/// The task, quota, CSpace, measurement-log and ready-queue slots remain
488/// detached promises until the single session owner transaction commits. This
489/// is not a second live-owner table: the Cell owner slot is `Reserved`, which
490/// no resolve/watch path can observe as live.
491pub(crate) struct SessionPublication {
492    pub(crate) task: alloc::boxed::Box<super::Task>,
493    pub(crate) owner: api::cell_owner::CellOwner,
494    state: TaskLaunchState,
495    quota: crate::memory::cell_quota::QuotaReservation,
496    authority: crate::cell::cspace::AuthorityReservation,
497    measurement: crate::measurement_log::PublicationReservation,
498    ready: crate::sched::hart_local::ready::PublicationReservation,
499    claim: PublicationClaim,
500    /// ADR-0019 lease of a domain child, rechecked in `current()` under the same
501    /// SCHEDULER section as `publish`; `None` for an ordinary child.
502    domain_lease: Option<DomainLease>,
503    /// The launch edge route this child was admitted under (`Path` for a kernel
504    /// read, `Elf` for caller-supplied bytes); `current()` re-authorizes it.
505    route: crate::loader::launch_profile::LaunchRoute,
506}
507impl SessionPublication {
508    /// Every fallible allocation, quota, owner-slot, log and ready-queue claim
509    /// happens here; `publish` allocates nothing and cannot fail.
510    pub(crate) fn reserve(
511        sched: &mut crate::sched::scheduler::Scheduler,
512        prepared: super::PreparedElfTask,
513        state: TaskLaunchState,
514        route: crate::loader::launch_profile::LaunchRoute,
515    ) -> Result<Self, ViError> {
516        let Some(caller) = state.caller.as_ref() else {
517            return Err(ViError::PermissionDenied);
518        };
519        if !authority_is_current(sched, caller) {
520            return Err(ViError::PermissionDenied);
521        }
522        let (creator, creator_generation) = sched
523            .tasks
524            .get(&caller.tid)
525            .map(|t| (t.cell_id.0 as usize, t.cell_generation))
526            .ok_or(ViError::PermissionDenied)?;
527        // A session child is an ordinary governed path launch: never a boot
528        // singleton, hotswap replacement, or service route publisher. A domain
529        // child goes through the same ADR-0019 policy as `publish_prepared`.
530        if prepared.requested_cell_id() != CellId(0)
531            || state.platform.is_some()
532            || state.replacement.is_some()
533            || state.routes.block_io
534            || state.routes.input
535        {
536            return Err(ViError::PermissionDenied);
537        }
538        #[cfg(feature = "test-hooks")]
539        if state.routes.development_silo {
540            return Err(ViError::PermissionDenied);
541        }
542        // Heap bookkeeping for the unpublished task is kernel memory, exactly as
543        // in `publish_prepared`; per-principal accounting is the explicit claim.
544        let _kernel_alloc = crate::sched::scheduler::KernelAllocationContext::enter();
545        let tid = sched.next_task_id;
546        let next = tid.checked_add(1).ok_or(ViError::OutOfMemory)?;
547        let quota = crate::memory::cell_quota::QuotaReservation::reserve_next(state.quota_limit)?;
548        let cell = quota.cell_id();
549        let (mut task, _) = prepared.into_task(tid, cell)?;
550        task.root_tid = tid;
551        task.syscall_allowlist = state.syscall_allowlist;
552        task.priority = state.priority;
553        task.pku_key = state.pku_key;
554        task.pku_value = state.pku_value;
555        task.is_critical = state.is_critical;
556        state.granted.apply_to(&mut task);
557        if let Some(measurement) = state.measurement.as_ref() {
558            task.bind_image(measurement.image);
559            task.launch_named = measurement.launch_named;
560        }
561        let domain_lease = if state.is_domain {
562            let path = state.measurement.as_ref().map_or("", |m| m.path.as_str());
563            Some(admit_and_bind_domain(&mut task, path, state.granted)?)
564        } else {
565            None
566        };
567        #[cfg(feature = "test-hooks")]
568        {
569            task.development_silo_registration_cap = state
570                .routes
571                .development_silo
572                .then(super::cap::DevelopmentSiloRegistrationCap::new);
573        }
574        let owner = api::cell_owner::CellOwner::new(cell.0, task.cell_generation, tid as u64);
575        let authority = crate::cell::cspace::AuthorityReservation::reserve(cell, owner.generation)?;
576        let path_len = state.measurement.as_ref().map_or(0, |m| m.path.len());
577        let claim = PublicationClaim::charge(
578            creator,
579            crate::measurement_log::PublicationReservation::ENTRY_BYTES + path_len,
580            crate::sched::hart_local::ready::PublicationReservation::SLOT_BYTES,
581        )?;
582        let measurement = crate::measurement_log::PublicationReservation::reserve((
583            creator as u64,
584            creator_generation,
585        ))?;
586        let ready =
587            crate::sched::hart_local::ready::PublicationReservation::reserve(state.priority)?;
588        sched.reserve_session_owner(owner)?;
589        sched.next_task_id = next;
590        Ok(Self {
591            task,
592            owner,
593            state,
594            quota,
595            authority,
596            measurement,
597            ready,
598            claim,
599            domain_lease,
600            route,
601        })
602    }
603    /// ADR-0019 commit decision for this child, given the live-grant preflight
604    /// `dispatch` took before SCHEDULER. Call under SCHEDULER, right before
605    /// `publish`, after `current()`.
606    pub(crate) fn domain_commit(&self, grant_live: bool) -> DomainCommit {
607        domain_commit_gate(
608            self.domain_lease.map(|lease| lease.remains_enabled()),
609            grant_live,
610        )
611    }
612    pub(crate) fn current(&self, sched: &crate::sched::scheduler::Scheduler) -> bool {
613        if !sched.session_owner_reserved(self.owner)
614            || self
615                .state
616                .caller
617                .as_ref()
618                .is_none_or(|a| !authority_is_current(sched, a))
619        {
620            return false;
621        }
622        let Some(caller_tid) = self.state.caller.as_ref().map(|a| a.tid) else {
623            return false;
624        };
625        // The caller's measured launch identity, never its task name (C-1).
626        let Some(identity) = super::syscall::launch_identity_in(sched, caller_tid) else {
627            return false;
628        };
629        let rights = sched.cell_rights(caller_tid);
630        let Some(measurement) = self.state.measurement.as_ref() else {
631            return false;
632        };
633        let state = crate::loader::launch_profile::CallerLaunchState {
634            name: &identity,
635            has_spawn: rights.spawn,
636            has_supervisor: rights.supervisor,
637        };
638        crate::loader::launch_profile::authorize(state, self.route, &measurement.path).is_some_and(
639            |profile| {
640                self.state.granted.intersect(profile.child_ceiling) == self.state.granted
641                    && (!profile.requires_lifecycle_authority || rights.spawn)
642            },
643        )
644    }
645    /// Final publication. Called with every owner check already passed, under
646    /// SCHEDULER, and performs no allocation: capacity for the task row, CSpace
647    /// row, log entry and ready entry was promised in `reserve`.
648    pub(crate) fn publish(mut self, sched: &mut crate::sched::scheduler::Scheduler) {
649        let tid = self.owner.root_tid as usize;
650        if self.domain_lease.is_some() {
651            log_domain_admitted(&self.task.name);
652        }
653        #[cfg(feature = "p01-trace")]
654        {
655            crate::memory::heap::p01_owner::adopt_task(&self.task);
656            crate::memory::heap::p01_owner::task_backing(&self.task);
657        }
658        self.authority.commit();
659        sched.tasks.publish_reserved(tid, self.task);
660        sched.publish_session_owner(self.owner);
661        self.quota.commit();
662        crate::loader::commit_launch_routes(tid, self.state.routes);
663        #[cfg(feature = "p01-trace")]
664        {
665            let task = sched.tasks.get_mut(&tid).expect("committed publication");
666            let diagnostic_owner = crate::p01_measurement::Owner::of(task);
667            if let Some(measurement) = self.state.measurement.as_ref() {
668                if self.state.bootstrap_init {
669                    crate::p01_measurement::arm(diagnostic_owner, measurement.digest);
670                } else {
671                    crate::p01_measurement::publication(
672                        diagnostic_owner,
673                        self.state.trace,
674                        measurement.digest,
675                    );
676                }
677            }
678            crate::memory::heap::p01_owner::bind(self.state.trace, diagnostic_owner);
679            crate::p01_resources::published(task);
680        }
681        if let Some(measurement) = self.state.measurement.take() {
682            self.measurement
683                .commit(tid, measurement.path, measurement.digest);
684        }
685        crate::audit::log_event(
686            crate::audit::AuditEvent::CellSpawn,
687            &crate::audit::encode_u32x2(tid as u32, 0),
688        );
689        crate::loader::observe_pre_ready(sched, tid);
690        // The notification follows the reservation's pinned destination, not
691        // whichever hart happens to commit or whatever RT hart is online now.
692        let (hart, priority) = self.ready.destination();
693        self.ready.commit(tid);
694        self.claim.transfer();
695        sched.pend_preempt_on_hart(hart, priority);
696    }
697}
