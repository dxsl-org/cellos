12/// Maximum number of IPC messages buffered for a Frozen cell during hot-swap.
13///
14/// When a cell is mid-swap (state `Frozen`), incoming `sys_send` calls enqueue
15/// here instead of blocking the caller or returning an error.  If the buffer
16/// fills, additional senders receive `SyscallError::TryAgain` so they can
17/// back off.  Messages are drained to the new cell in Step 5 (UNFREEZE).
18pub const HOTSWAP_MSG_QUEUE_DEPTH: usize = 64;
19
20/// Deeper bound for nonblocking input-service events queued to a client.
21///
22/// Keyboard events use blocking `sys_send`, so they do not rely on this mailbox
23/// as a paste reservoir. Nonblocking input-service traffic such as mouse events
24/// may use this scheduling cushion while the target is briefly outside `Recv`.
25/// Keeping the bound below the 4096-byte UART RX ring limits scheduler critical-
26/// section work. All other `sys_try_send` callers keep strict drop-if-not-ready
27/// semantics.
28pub const INPUT_EVENT_QUEUE_DEPTH: usize = 512;

31pub enum TaskState {
32    Ready,
33    Running,
34    /// Sleeping until a specific monotonic time (ticks/ms)
35    Sleeping {
36        until: usize,
37    },
38    /// Blocked waiting to send a message to `target_id`.
39    /// Stores the message pointer and length temporarily.
40    Sending {
41        target: usize,
42        msg_ptr: VAddr,
43        msg_len: usize,
44    },
45    /// Blocked waiting to receive a message.
46    ///
47    /// `mask`: sender filter (0 = any sender).
48    /// `deadline`: optional monotonic tick count after which the kernel wakes
49    ///   this task with `ViError::Timeout`.  `None` = wait indefinitely.
50    Recv {
51        mask: usize,
52        buf_ptr: VAddr,
53        buf_len: usize,
54        deadline: Option<u64>,
55    },
56    /// This task has finished running.
57    Terminated,
58    /// Blocked on a Futex wait.
59    /// `addr`: The address being waited on.
60    FutexWait {
61        addr: VAddr,
62    },
63    /// Waiting for another task to exit (Join).
64    Waiting {
65        target: usize,
66    },
67    /// Polling an async future (e.g. syscall)
68    Polling,
69    /// Blocked in `WaitForEvent(mask, timeout)`.  Woken by `waker::signal_net_rx()`
70    /// (or equivalent) when any bit in `mask` fires, or when `deadline` ticks pass.
71    /// `deadline = None` means block indefinitely.
72    WaitEvent {
73        mask: u32,
74        deadline: Option<u64>,
75    },
76    /// Blocked in `WaitCompletion`. Source completions wake through the queue;
77    /// a finite TIMER wakes when `deadline` elapses.
78    WaitCompletion {
79        source: u32,
80        deadline: Option<u64>,
81    },
82    /// Cell is suspended during a hot-swap sequence identified by `swap_id`.
83    ///
84    /// Invariants while Frozen:
85    /// - NOT in any scheduler ready queue.
86    /// - Cannot be killed by external signal (returns `ViError::PermissionDenied`).
87    /// - Can only be terminated by the hotswap orchestrator via `exit_task` after a
88    ///   successful swap, or rolled back to `Ready` on failure.
89    /// - Incoming IPC is queued in `crate::cell::hotswap::FROZEN` registry instead
90    ///   of being delivered (Phase 02 will drain the queue; P01 stubs this).
91    Frozen {
92        swap_id: u64,
93    },
94    /// Blocked in `sys_wait_irq(irq, mmio_base)` until hardware IRQ `irq` fires.
95    ///
96    /// Woken by the scheduler sweep (NOT from ISR — ISR only sets `IRQ_PENDING[irq]`).
97    /// No deadline: IRQ-only wake; Driver Cells must implement their own poll-fallback
98    /// if the device can hang without raising an IRQ.
99    WaitIrq {
100        irq: u8,
101    },
102}

356    ///

358    /// read, a quota ledger row — must not hand that state to a *different* cell
359    /// that later shows up under the same id. `CellId` is `CellId(tid)` and the
360    /// scheduler's `next_task_id` only ever increments, so ids are not recycled

362    /// because it is minted fresh per cell and never reused.

365    /// `Scheduler::spawn_thread`) so a thread is indistinguishable from its cell

375    /// must never treat this as a second record of filesystem authority — the
376    /// filesystem service owns that, and a second copy would drift in the
377    /// direction of silently widening what a cell may reach. The only claim the

398    ///
399    /// Kernel-owned heap memory, never a grant: a cell cannot unregister or free

402    /// dies with the last reference to it — see [`crate::task::completion`].

404    /// Queue slot held by an in-progress `WaitCompletion` syscall.
405    ///
406    /// Kept outside `TaskState` because a deferred source wake changes the state
407    /// to `Ready` before the syscall has released its reservation. Task exit
408    /// takes this field and defers TIMER release outside the scheduler lock.
409    pub completion_wait: Option<CompletionWait>,

553    ) {
554        // VFS may perform nested IPC while serving a request. Its outer caller
555        // remains the authority for grants and owner-death watches until VFS
556        // sends that caller's response; a storage reply must not replace it.
557        if crate::fast_ipc::is_registered_vfs_cell(self.cell_id.0 as usize)
558            && self.current_caller.is_some()
559        {
560            return;
561        }
562        self.set_current_caller_context(sender_tid, sender_cell_id, sender_generation);
563    }
564
565    pub fn begin_receive_context(&mut self, mask: usize) {
566        // VFS uses a wildcard receive only at its public request loop. Masked
567        // receives are nested dependency replies and keep the outer authority.
568        if mask == 0 && crate::fast_ipc::is_registered_vfs_cell(self.cell_id.0 as usize) {
569            self.clear_current_caller_context();
570        }
571    }
572
573    pub fn allows_current_caller_owner_watch(&self, watched: usize) -> bool {
574        self.current_caller_cell_id != 0 && watched == self.current_caller_cell_id as usize
575    }
576
577    pub fn clear_current_caller_context(&mut self) {
578        self.current_caller = None;
579        self.current_caller_cell_id = 0;
580        self.current_caller_cell_generation = 0;
581        self.current_caller_request_generation = 0;
582    }
583
584    pub fn clear_current_caller_context_if(
585        &mut self,
586        sender_tid: usize,
587        request_generation: u64,
588    ) -> bool {
589        if self.current_caller == Some(sender_tid)
590            && self.current_caller_request_generation == request_generation
591        {
592            self.clear_current_caller_context();
593            return true;
594        }
595        false
596    }
