1//! SMP: secondary hart startup and controlled park loop.
2//!
3//! Phase 01: brings each secondary hart online, installs its trap vector,
4//! then parks it in WFI.  Phase 03 replaces the park loop with a per-hart
5//! scheduler round.
6//!
7//! Invariant: hart 0 calls `start_secondaries()` only AFTER `task::init()`
8//! completes — the SCHEDULER and heap are live before any secondary runs.
9
10use core::sync::atomic::AtomicUsize;
11use core::sync::atomic::{AtomicBool, Ordering};
12
13/// Maximum number of harts this kernel tracks.  2 covers QEMU virt `-smp 2`
14/// (G2 entry target).  Constant so secondary stacks and HART_ONLINE are
15/// statically sized — no heap allocation during the boot critical path.
16pub const MAX_HARTS: usize = 2;
17
18/// Hart dedicated to RealTime-priority cells.  RT tasks are enqueued here by
19/// `push_ready` and never stolen (Phase 03 steal filter excludes RT).
20pub const HART_RT: usize = 1;
21
22/// Set to `true` by each secondary hart once its trap vector and timer are ready.
23/// Hart 0's bounded wait reads this via `Acquire` to observe all preceding stores.
24pub static HART_ONLINE: [AtomicBool; MAX_HARTS] = [AtomicBool::new(false), AtomicBool::new(false)];
25
26/// Monotonic switch-completion epochs for root-retirement quiescence. A retiring
27/// generation cannot release its CellId slot until every requested hart has
28/// switched to a different context and published that completion.
29/// Per-hart invalidation epochs: a requester publishes a target, the target
30/// performs the local flush on its way through the trap path and publishes
31/// completion. Only then may the requester recycle a tag or release frames — a
32/// firmware call returning is not evidence that another hart stopped using the
33/// translation (phase 02 slice 3).
34static TLB_FLUSH_REQUEST: [AtomicUsize; MAX_HARTS] = [AtomicUsize::new(0), AtomicUsize::new(0)];
35static TLB_FLUSH_COMPLETE: [AtomicUsize; MAX_HARTS] = [AtomicUsize::new(0), AtomicUsize::new(0)];
36
37/// Deliver the kernel's cross-hart IPI to `hart_id`.
38///
39/// The requester only records an epoch; what makes the target *act* is this
40/// interrupt. Every path that needs a remote hart to do work (invalidate its
41/// TLB, switch out of a retiring context) goes through here so the delivery
42/// mechanism has one definition per architecture.
43#[inline]
44fn send_ipi(hart_id: usize) {
45    #[cfg(any(
46        target_arch = "riscv64",
47        all(target_arch = "aarch64", not(feature = "board-rpi3"))
48    ))]
49    {
50        if hart_id >= MAX_HARTS || hart_id == crate::task::hart_local::current_hart_id() {
51            return;
52        }
53        #[cfg(target_arch = "riscv64")]
54        if let Some((mask, base)) = logical_sbi_target(hart_id) {
55            let _ = hal::common::sbi::sbi_send_ipi(mask, base);
56        }
57        // AArch64: the BCM2836 (RPi3) local controller has no software-interrupt
58        // path in this kernel, and that board starts no secondary, so the GIC SGI
59        // is the only delivery mechanism that exists.
60        #[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
61        if let Some(cpu) = crate::task::hart_local::physical_cpu_for(hart_id) {
62            hal::aarch64::gic::send_sgi(cpu as u32, hal::aarch64::gic::SGI_IPI);
63        }
64    }
65    #[cfg(all(target_arch = "x86_64", not(all(test, not(target_os = "none")))))]
66    if hart_id < MAX_HARTS {
67        // No completion is forged on delivery failure. Keep this path free of
68        // locks/logging because a deferred-release requester may own a lock.
69        let _ = hal::x86_64::apic::send_ipi(hart_id);
70    }
71    // Boards with no software-interrupt transport retain the request epoch.
72    #[cfg(not(any(
73        target_arch = "riscv64",
74        all(target_arch = "x86_64", not(all(test, not(target_os = "none")))),
75        all(target_arch = "aarch64", not(feature = "board-rpi3"))
76    )))]
77    let _ = hart_id;
78}

80/// Ask `hart_id` to invalidate its local TLB and return the epoch it must publish.
81pub fn request_tlb_flush(hart_id: usize) -> usize {
82    if hart_id >= MAX_HARTS {
83        return 0;
84    }
85    let epoch = TLB_FLUSH_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
86    send_ipi(hart_id);
87    epoch
88}

90/// Is `epoch` known complete on `hart_id`? Epoch 0 means "nothing requested".
91pub fn tlb_flush_completed(hart_id: usize, epoch: usize) -> bool {
92    epoch == 0
93        || TLB_FLUSH_COMPLETE
94            .get(hart_id)
95            .is_some_and(|complete| complete.load(Ordering::Acquire) >= epoch)
96}
97
98/// Service one captured request. A newer request arriving during the local
99/// invalidate must NOT be acknowledged by that invalidate.
100pub fn service_tlb_flush(hart_id: usize) -> usize {
101    if hart_id >= MAX_HARTS {
102        return 0;
103    }
104    let epoch = TLB_FLUSH_REQUEST[hart_id].load(Ordering::Acquire);
105    if TLB_FLUSH_COMPLETE[hart_id].load(Ordering::Acquire) < epoch {
106        crate::memory::paging::tlb_flush_all();
107        TLB_FLUSH_COMPLETE[hart_id].store(epoch, Ordering::Release);
108        #[cfg(all(feature = "test-hooks", not(target_arch = "x86_64")))]
109        log::info!(
110            "[selftest] TLB-ACK: stage=remote-flush-completed hart={} epoch={}",
111            hart_id,
112            epoch
113        );
114    }
115    epoch
116}

127/// Does this hart owe the requester an invalidation completion?
128pub fn tlb_flush_pending(hart_id: usize) -> bool {
129    hart_id < MAX_HARTS
130        && TLB_FLUSH_REQUEST[hart_id].load(Ordering::Acquire)
131            > TLB_FLUSH_COMPLETE[hart_id].load(Ordering::Acquire)
132}

135pub fn online_harts() -> impl Iterator<Item = usize> {

139static RETIRE_SWITCH_REQUEST: [AtomicUsize; MAX_HARTS] = [AtomicUsize::new(0), AtomicUsize::new(0)];
140static RETIRE_SWITCH_COMPLETE: [AtomicUsize; MAX_HARTS] =
141    [AtomicUsize::new(0), AtomicUsize::new(0)];
142
143#[cfg(target_arch = "x86_64")]
144static RETIRE_SWITCH_INFLIGHT: [AtomicUsize; MAX_HARTS] =
145    [const { AtomicUsize::new(0) }; MAX_HARTS];
146
147/// Capture before scheduler selection: a later request may concern the incoming
148/// task and must wait for that task's next switch-away.
149#[cfg(target_arch = "x86_64")]
150pub fn capture_retirement_boundary(hart: usize) {
151    RETIRE_SWITCH_INFLIGHT[hart].store(
152        RETIRE_SWITCH_REQUEST[hart].load(Ordering::Acquire), Ordering::Release,
153    );
154}
155
156#[cfg(target_arch = "x86_64")]
157pub fn complete_captured_retirement_boundary(hart: usize) {
158    let epoch = RETIRE_SWITCH_INFLIGHT[hart].swap(0, Ordering::AcqRel);
159    RETIRE_SWITCH_COMPLETE[hart].fetch_max(epoch, Ordering::Release);
160}

162/// Test-hooks: how many preemption requests were pended for each logical hart.
163///
164/// The RT-wake fixture needs the *decision* observable, not the interrupt: it
165/// asserts that consuming a message pends exactly one for the sender's target
166/// hart, that a stale delivery token pends none, and that the decision follows the
167/// target hart's running priority rather than the waking hart's.
168#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
169static PREEMPT_PENDS: [AtomicUsize; MAX_HARTS] = [AtomicUsize::new(0), AtomicUsize::new(0)];
170
171#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
172#[inline]
173pub(crate) fn note_preempt_pend(hart_id: usize) {
174    if hart_id < MAX_HARTS {
175        PREEMPT_PENDS[hart_id].fetch_add(1, Ordering::AcqRel);
176    }
177}
178
179#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
180#[inline]
181pub(crate) fn preempt_pends_for(hart_id: usize) -> usize {
182    PREEMPT_PENDS
183        .get(hart_id)
184        .map(|count| count.load(Ordering::Acquire))
185        .unwrap_or(0)
186}
187
188#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
189pub(crate) fn reset_preempt_pends() {
190    for count in PREEMPT_PENDS.iter() {
191        count.store(0, Ordering::Release);
192    }
193}

195/// Request that `hart_id` schedules through a retirement boundary and return
196/// the epoch that its incoming context must complete.
197pub fn request_retirement_switch(hart_id: usize) -> usize {
198    if hart_id >= MAX_HARTS {
199        return 0;
200    }
201    let epoch = RETIRE_SWITCH_REQUEST[hart_id].fetch_add(1, Ordering::AcqRel) + 1;
202    #[cfg(feature = "test-hooks")]
203    log::info!(
204        "[selftest] SMP-RETIREMENT: stage=remote-switch-requested hart={} epoch={}",
205        hart_id,
206        epoch
207    );
208    send_ipi(hart_id);
209    epoch
210}

212/// Publish the requested epoch from the incoming side of `Context::switch`.
213///
214/// The release store is deliberately after the raw context switch has changed
215/// stacks: an IPI/trap entry only proves that the outgoing task entered the
216/// kernel, while this proves that its saved context no longer executes.
217pub fn complete_retirement_switch(hart_id: usize) {
218    if hart_id < MAX_HARTS {
219        let epoch = RETIRE_SWITCH_REQUEST[hart_id].load(Ordering::Acquire);
220        let completed = RETIRE_SWITCH_COMPLETE[hart_id].load(Ordering::Acquire);
221        if completed < epoch {
222            RETIRE_SWITCH_COMPLETE[hart_id].store(epoch, Ordering::Release);
223            #[cfg(feature = "test-hooks")]
224            log::info!(
225                "[selftest] SMP-RETIREMENT: stage=remote-switch-completed hart={} epoch={}",
226                hart_id,
227                epoch
228            );
229        }
230    }
231}

233/// Does `hart_id` still owe a completion for a requested retirement switch?
234pub fn retirement_switch_pending(hart_id: usize) -> bool {
235    hart_id < MAX_HARTS
236        && RETIRE_SWITCH_REQUEST[hart_id].load(Ordering::Acquire)
237            > RETIRE_SWITCH_COMPLETE[hart_id].load(Ordering::Acquire)
238}

240pub fn retirement_switch_completed(hart_id: usize, epoch: usize) -> bool {
241    epoch == 0
242        || RETIRE_SWITCH_COMPLETE
243            .get(hart_id)
244            .is_some_and(|complete| complete.load(Ordering::Acquire) >= epoch)
245}

250#[cfg(target_arch = "riscv64")]
251pub fn set_boot_physical_hart(physical_hart: usize) {
252    assert!(
253        physical_hart < MAX_HARTS,
254        "unsupported RV64 boot hart {physical_hart}"
255    );
256    BOOT_PHYSICAL_HART.store(physical_hart, Ordering::Release);
257}

259#[cfg(target_arch = "riscv64")]
260pub fn boot_physical_hart() -> Option<usize> {
261    let physical = BOOT_PHYSICAL_HART.load(Ordering::Acquire);
262    (physical < MAX_HARTS).then_some(physical)
263}
264
265#[cfg(target_arch = "riscv64")]
266pub fn logical_to_physical(logical_hart: usize) -> Option<usize> {
267    let boot = BOOT_PHYSICAL_HART.load(Ordering::Acquire);
268    match logical_hart {
269        0 if boot < MAX_HARTS => Some(boot),
270        HART_RT if boot < MAX_HARTS => Some(boot ^ 1),
271        _ => None,
272    }
273}
274
275#[cfg(target_arch = "riscv64")]
276pub fn physical_to_logical(physical_hart: usize) -> Option<usize> {
277    let boot = BOOT_PHYSICAL_HART.load(Ordering::Acquire);
278    if physical_hart == boot {
279        Some(0)
280    } else if physical_hart < MAX_HARTS && physical_hart == (boot ^ 1) {
281        Some(HART_RT)
282    } else {
283        None
284    }
285}

292/// Return every online RV64 hart except the one executing this call.
293///
294/// Hart 0 is running whenever this kernel reaches normal execution but is not
295/// represented by `HART_ONLINE`; secondary harts publish readiness with Release.
296#[cfg(target_arch = "riscv64")]
297pub fn remote_online_sbi_target() -> Option<(usize, usize)> {
298    let current = crate::task::hart_local::current_hart_id();
299    let remote = if current == 0 { HART_RT } else { 0 };
300    let online = remote == 0 || HART_ONLINE[remote].load(Ordering::Acquire);
301    online.then(|| logical_sbi_target(remote)).flatten()
302}

304/// Return the harts that completed kernel bring-up for the current boot.
305///
306/// Hart zero is the active boot hart; each secondary contributes only after
307/// publishing `HART_ONLINE`, so test evidence cannot confuse configured SMP
308/// capacity with observed runtime availability.
309#[cfg(all(
310    feature = "native-domains",
311    feature = "test-hooks",
312    any(target_arch = "riscv64", target_arch = "x86_64")
313))]
314pub(crate) fn online_hart_count() -> usize {
315    1 + HART_ONLINE
316        .iter()
317        .skip(1)
318        .filter(|online| online.load(Ordering::Acquire))
319        .count()
320}

322/// How many 10 ms ticks hart 0 waits for each secondary to come online before
323/// logging a warning and continuing single-hart.  500 ms is generous for QEMU
324/// and for a firmware PSCI call that has to power a core up.
325#[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
326const SECONDARY_BOOT_TIMEOUT_TICKS: usize = 50;
327
328/// Called by hart 0 **after** `task::init()` to bring secondary harts online.
329///
330/// Each secondary is started via SBI HSM `hart_start`.  Hart 0 then spins
331/// (bounded) waiting for each secondary to set `HART_ONLINE[hart_id]`.
332/// If a secondary fails to start or times out, a warning is logged and the
333/// system continues single-hart — graceful degradation, never a panic.
334#[cfg(target_arch = "riscv64")]
335pub fn start_secondaries() {
336    use crate::task::stack::Stack;
337    use crate::task::STACK_PAGES;
338    use hal::common::sbi::{sbi_hart_get_status, sbi_hart_start, sbi_rfence_available};
339
340    let Some(boot_physical) = boot_physical_hart() else {
341        log::warn!("[smp] boot physical hart was not published");
342        return;
343    };
344    log::info!("[smp] physical {} -> logical 0 boot", boot_physical);
345
346    match sbi_rfence_available() {
347        Ok(true) => {}
348        Ok(false) => {
349            log::warn!("[smp] SBI RFENCE unavailable — keeping Cellos single-hart");
350            return;
351        }
352        Err(error) => {
353            log::warn!(
354                "[smp] SBI RFENCE probe failed (err={}) — keeping Cellos single-hart",
355                error
356            );
357            return;
358        }
359    }
360
361    extern "C" {
362        // Physical asm label defined in hal/arch/riscv/src/rv64/boot.rs.
363        // Runs bare (SATP=0); no relocation or BSS clear.
364        fn _secondary_entry();
365    }
366
367    for (hart_id, online) in HART_ONLINE.iter().enumerate().skip(1) {
368        let Some(physical_hart) = logical_to_physical(hart_id) else {
369            log::warn!("[smp] logical hart {} has no physical mapping", hart_id);
370            continue;
371        };
372        // Allocate a dedicated kernel stack for this hart.  Leak it — it lives
373        // for the entire lifetime of the hart.
374        let stack = match Stack::new_kernel(STACK_PAGES) {
375            Ok(s) => s,
376            Err(e) => {
377                log::warn!("[smp] hart {} stack alloc failed: {:?}", hart_id, e);
378                continue;
379            }
380        };
381        let stack_top = stack.top;
382        core::mem::forget(stack);
383
384        let Ok(state) = sbi_hart_get_status(physical_hart) else {
385            log::warn!(
386                "[smp] physical hart {} HSM status unavailable",
387                physical_hart
388            );
389            continue;
390        };
391        log::info!(
392            "[smp] physical {} -> logical {} HSM state = {}",
393            physical_hart,
394            hart_id,
395            state
396        );
397        if state != 1 {
398            log::warn!("[smp] physical hart {} is not HSM STOPPED", physical_hart);
399            continue;
400        }
401
402        // SAFETY: _secondary_entry is a physical-address asm label; the kernel
403        // is loaded at 0x80200000 with slide=0 so physical == virtual.
404        // stack_top is the usable top of a freshly-allocated kernel stack.
405        // SAFETY: casting function pointer to integer — use double-cast through
406        // *const () to avoid the "direct cast of function item" lint.
407        let entry_paddr = _secondary_entry as *const () as usize;
408        match sbi_hart_start(physical_hart, entry_paddr, stack_top) {
409            Ok(()) => log::info!(
410                "[smp] hart {} start requested (entry={:#x})",
411                physical_hart,
412                entry_paddr
413            ),
414            Err(e) => {
415                log::warn!("[smp] hart {} SBI hart_start failed: err={}", hart_id, e);
416                continue;
417            }
418        }
419
420        // Bounded spin: wait for the secondary to signal it is online.
421        let deadline = crate::task::system_ticks() + SECONDARY_BOOT_TIMEOUT_TICKS;
422        loop {
423            if online.load(Ordering::Acquire) {
424                log::info!("[smp] hart {} online, parked", hart_id);
425                break;
426            }
427            if crate::task::system_ticks() >= deadline {
428                log::warn!(
429                    "[smp] hart {} did not come online in time — continuing single-hart",
430                    hart_id
431                );
432                break;
433            }
434            core::hint::spin_loop();
435        }
436    }
437}

439/// Boot state for one secondary core, published before `PSCI_CPU_ON`.
440///
441/// Written once by the boot hart and read by a core whose caches are off, so
442/// the write is followed by a clean to the point of coherency. `static mut`
443/// with a single writer and no reader until the core is started is the whole
444/// synchronisation argument.
445#[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
446static mut SECONDARY_CONTEXT: [hal::aarch64::boot::SecondaryContext; MAX_HARTS] =
447    [hal::aarch64::boot::SecondaryContext {
448        hart_id: 0,
449        stack_top: 0,
450        mair: 0,
451        tcr: 0,
452        ttbr0: 0,
453        sctlr: 0,
454    }; MAX_HARTS];
455
456/// Called by hart 0 **after** `task::init()` to bring secondary cores online.
457///
458/// AArch64 starts a core through firmware (`PSCI_CPU_ON` over the conduit the
459/// device tree declares). Everything the new core needs — its stack, the boot
460/// core's live translation regime, the hardening bits in `SCTLR_EL1` — travels
461/// in one context block it reads with the MMU still off; `_secondary_entry`
462/// installs the regime and calls [`smp_aarch64_secondary_main`].
463///
464/// Bounded wait, graceful degradation, never a panic: a core that does not come
465/// online within the deadline leaves the system single-hart with a warning.
466#[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
467pub fn start_secondaries() {
468    use crate::boot::firmware_smp;
469    use crate::task::stack::Stack;
470    use crate::task::STACK_PAGES;
471
472    let at_el2 = hal::aarch64::el2::is_el2();
473    // Which instruction carries a PSCI call: the firmware tree's answer when
474    // there is one, otherwise what the architecture allows — with EL3 the
475    // conduit is EL3 firmware's (`smc`); without EL3 an `smc` is undefined and
476    // the only conduit that can exist is the hypervisor's (`hvc`).
477    let conduit = match firmware_smp::psci_conduit() {
478        Some(declared) => declared,
479        None => {
480            let inferred = hal::aarch64::psci::inferred_conduit();
481            log::info!(
482                "[smp] no firmware tree declares PSCI; EL3 {} — inferring conduit {:?}",
483                if hal::aarch64::psci::el3_implemented() {
484                    "implemented"
485                } else {
486                    "absent"
487                },
488                inferred
489            );
490            inferred
491        }
492    };
493    // A conduit that cannot work is reported, never attempted: `smc` without EL3
494    // and `hvc` from EL2 both trap as undefined instructions.
495    let usable = match (conduit, at_el2) {
496        // EL3 firmware answers an SMC from either exception level.
497        (hal::aarch64::psci::Conduit::Smc, _) => true,
498        // An HVC from EL1 traps to EL2, where the PSCI implementation lives; an
499        // HVC from EL2 would target EL3, which by construction has no PSCI here.
500        (hal::aarch64::psci::Conduit::Hvc, false) => true,
501        (hal::aarch64::psci::Conduit::Hvc, true) => false,
502    };
503    if !usable {
504        log::warn!(
505            "[smp] the kernel runs at EL2 and its conduit is HVC, which would target EL3 — \
506             keeping Cellos single-hart"
507        );
508        return;
509    }
510    log::info!(
511        "[smp] PSCI conduit {:?}, kernel at {}",
512        conduit,
513        if at_el2 { "EL2" } else { "EL1" }
514    );
515    // The logical hart equals the physical CPU index on every AArch64 platform
516    // this kernel supports, so the boot core has to *be* CPU 0 for the mapping
517    // to hold; anything else is reported rather than half-applied.
518    let boot_cpu = crate::task::hart_local::physical_cpu_index();
519    if boot_cpu != 0 {
520        log::warn!(
521            "[smp] boot core reports CPU index {} (expected 0) — keeping Cellos single-hart",
522            boot_cpu
523        );
524        return;
525    }
526    // How many cores to try. The tree is authoritative when it says anything;
527    // with no tree the firmware's own answer to `CPU_ON` is what tells us
528    // whether a core exists, so every hart this kernel models is attempted and
529    // a refusal ends the search.
530    let described = firmware_smp::cpu_count();
531    if described == 1 {
532        log::info!("[smp] firmware tree describes 1 CPU — single-hart");
533        return;
534    }
535    let attempt = if described == 0 {
536        MAX_HARTS
537    } else {
538        MAX_HARTS.min(described)
539    };
540    match hal::aarch64::psci::version(conduit) {
541        Some(version) => log::info!(
542            "[smp] PSCI {}.{} over {:?}, {} CPU(s) described",
543            version >> 16,
544            version & 0xFFFF,
545            conduit,
546            described
547        ),
548        None => {
549            log::warn!("[smp] firmware does not answer PSCI_VERSION — keeping Cellos single-hart");
550            return;
551        }
552    }
553
554    extern "C" {
555        /// Physical entry point defined in hal/arch/arm/src/aarch64/boot.rs.
556        /// Runs with the MMU off; installs the boot core's translation regime.
557        fn _secondary_entry();
558    }
559    let entry = _secondary_entry as *const () as usize;
560    let kernel_root = match *crate::memory::paging::KERNEL_ROOT.lock() {
561        Some(root) => root,
562        None => {
563            log::warn!("[smp] kernel page tables are not published — keeping Cellos single-hart");
564            return;
565        }
566    };
567
568    for hart_id in 1..attempt {
569        // The core is still off: this mapping has to exist before it reads it.
570        crate::task::hart_local::publish_physical_cpu(hart_id, hart_id);
571        let stack = match Stack::new_kernel(STACK_PAGES) {
572            Ok(stack) => stack,
573            Err(error) => {
574                log::warn!("[smp] hart {} stack alloc failed: {:?}", hart_id, error);
575                continue;
576            }
577        };
578        let stack_top = stack.top;
579        // Leaked on purpose: a hart's stack lives as long as the hart does.
580        core::mem::forget(stack);
581
582        let context = hal::aarch64::boot::SecondaryContext::for_hart(
583            hart_id as u64,
584            stack_top as u64,
585            kernel_root as u64,
586        );
587        // SAFETY: single writer (hart 0, before the core exists), and no reader
588        // until `PSCI_CPU_ON` succeeds.
589        let context_addr = unsafe {
590            let slot = core::ptr::addr_of_mut!(SECONDARY_CONTEXT[hart_id]);
591            slot.write(context);
592            slot as usize
593        };
594        // The core reads this with caches off: publish it to the point of
595        // coherency, and order that before the firmware call.
596        hal::aarch64::cache::clean_data_cache_range(
597            context_addr,
598            core::mem::size_of::<hal::aarch64::boot::SecondaryContext>(),
599        );
600        core::sync::atomic::fence(Ordering::SeqCst);
601
602        // Aff0 is the CPU index on every supported platform; the firmware call
603        // takes the full MPIDR, which those platforms build from it alone.
604        let mpidr = hart_id as u64;
605        match hal::aarch64::psci::cpu_on(conduit, mpidr, entry, context_addr as u64) {
606            Ok(()) => log::info!(
607                "[smp] hart {} start requested (cpu={} entry={:#x})",
608                hart_id,
609                hart_id,
610                entry
611            ),
612            Err(status) => {
613                // With nothing describing the CPU population, "no such core" is
614                // the expected answer on a smaller machine — and it also ends the
615                // search, because cores are numbered from zero.
616                // PSCI 1.0: -2 INVALID_PARAMETERS, -7 NOT_PRESENT.
617                let missing = described == 0 && (status == -2 || status == -7);
618                if missing {
619                    log::info!(
620                        "[smp] hart {}: firmware reports no such core — single-hart",
621                        hart_id
622                    );
623                    break;
624                }
625                log::warn!(
626                    "[smp] hart {} PSCI CPU_ON refused: {} ({})",
627                    hart_id,
628                    hal::aarch64::psci::status_name(status),
629                    status
630                );
631                continue;
632            }
633        }
634
635        let deadline = crate::task::system_ticks() + SECONDARY_BOOT_TIMEOUT_TICKS;
636        loop {
637            if HART_ONLINE[hart_id].load(Ordering::Acquire) {
638                log::info!("[smp] hart {} online, parked", hart_id);
639                break;
640            }
641            if crate::task::system_ticks() >= deadline {
642                log::warn!(
643                    "[smp] hart {} did not come online in time — continuing single-hart",
644                    hart_id
645                );
646                break;
647            }
648            core::hint::spin_loop();
649        }
650    }
651}

653/// Prove the hart that just came online actually takes the kernel's IPI.
654///
655/// The one cross-hart path a boot log cannot witness on its own: the request is
656/// delivered by a GIC SGI, the target answers from *its* trap path, and this
657/// hart's bounded wait ends only when that epoch lands. Without it, a hart that
658/// came online but is deaf to the IPI looks exactly like a healthy one — and
659/// every later remote invalidation would silently fall back to the retained-frame
660/// path. Bounded and non-fatal: a failed probe is reported, it does not stop the
661/// boot.
662#[cfg(target_arch = "aarch64")]
663pub fn run_ipi_selftest() {
664    let Some(remote) = online_harts().find(|hart| *hart != 0) else {
665        log::info!("[selftest] SMP-IPI: skipped, no remote hart online");
666        return;
667    };
668    let epoch = request_tlb_flush(remote);
669    let deadline = crate::task::system_ticks() + SECONDARY_BOOT_TIMEOUT_TICKS;
670    let mut acknowledged = false;
671    while crate::task::system_ticks() < deadline {
672        if tlb_flush_completed(remote, epoch) {
673            acknowledged = true;
674            break;
675        }
676        core::hint::spin_loop();
677    }
678    if acknowledged {
679        log::info!("[selftest] SMP-IPI: PASS hart={} epoch={}", remote, epoch);
680    } else {
681        log::error!(
682            "[selftest] SMP-IPI: FAIL hart={} epoch={} (no acknowledgement)",
683            remote,
684            epoch
685        );
686    }
687}

773#[cfg(target_arch = "x86_64")]
774fn x86_bounded_wait(mut done: impl FnMut() -> bool) {
775    // A bounded instruction budget does not depend on timer interrupts being
776    // delivered to the waiting CPU. Fail closed once a callback was released.
777    for _ in 0..100_000_000usize {
778        if done() { return; }
779        core::hint::spin_loop();
780    }
781    panic!("[x86-smp] active AP safety handshake timed out");
782}

784#[cfg(target_arch = "x86_64")]
785pub fn start_secondaries() {
786    use crate::boot::limine;
787    let bsp_id = core::arch::x86_64::__cpuid(1).ebx >> 24;
788    HART_ONLINE[0].store(true, Ordering::Release);
789    #[cfg(feature = "test-hooks")]
790    x86_handoff_fixture();
791    let finish_up = || {
792        X86_BOOT_COMPLETE.store(true, Ordering::Release);
793        log::info!("[x86-smp] topology online=1 processor_ids={} qualified=yes", bsp_id);
794    };
795    if !hal::x86_64::apic::register_cpu(0, bsp_id) {
796        // UP does not use APIC-directed remote invalidations.
797        finish_up();
798        return;
799    }
800    let Some(response) = limine::get_mp_response() else { finish_up(); return };
801    let count = response.cpu_count as usize;
802    if response.flags != 0 || response.bsp_lapic_id != bsp_id
803        || count == 0 || count > 256 || !limine::mp_record_valid(response.cpus, count) {
804        log::warn!("[x86-smp] unsupported MP topology; no AP activated");
805        finish_up();
806        return;
807    }
808    let mut ids = [false; 256];
809    let mut processors = [0u32; 256];
810    let mut bsp_seen = false;
811    let mut secondary: *mut limine::LimineMpInfo = core::ptr::null_mut();
812    for index in 0..count {
813        let info = unsafe { *response.cpus.add(index) };
814        if !limine::mp_record_valid(info, 1) { finish_up(); return; }
815        let cpu = unsafe { &*info };
816        let id = cpu.lapic_id as usize;
817        if id > 255 || ids[id]
818            || processors[..index].contains(&cpu.processor_id)
819            || (cpu.lapic_id != bsp_id && cpu.goto_address.load(Ordering::Acquire) != 0) {
820            log::warn!("[x86-smp] invalid MP CPU record; no AP activated");
821            finish_up(); return;
822        }
823        ids[id] = true;
824        processors[index] = cpu.processor_id;
825        if cpu.lapic_id == bsp_id { bsp_seen = true; }
826        else if secondary.is_null() { secondary = info; }
827    }
828    if !bsp_seen || secondary.is_null() { finish_up(); return; }
829    let ap_id = unsafe { (*secondary).lapic_id };
830    if !hal::x86_64::apic::register_cpu(1, ap_id) { finish_up(); return; }
831    let stack = match crate::task::stack::Stack::new_kernel(crate::task::STACK_PAGES) {
832        Ok(stack) => stack,
833        Err(_) => { finish_up(); return; }
834    };
835    let kernel_cr3 = hal::domain::kernel_cr3();
836    assert!(kernel_cr3 != 0, "AP requires permanent shared kernel root");
837    let stack_top = stack.top;
838    core::mem::forget(stack);
839    extern "C" { fn vi_x86_secondary_entry(); }
840    unsafe {
841        X86_SECONDARY = X86SecondaryContext { stack_top, kernel_cr3, logical: 1, lapic_id: ap_id as usize };
842        (*secondary).extra_argument = (&raw const X86_SECONDARY) as u64;
843        (*secondary).goto_address.store(vi_x86_secondary_entry as *const () as usize, Ordering::Release);
844    }
845    // The BSP must receive the AP's reciprocal proof even if boot entered this
846    // routine with interrupts masked. Restore its original state afterwards.
847    let flags: usize;
848    unsafe { core::arch::asm!("pushfq", "pop {}", "sti", out(reg) flags); }
849    x86_bounded_wait(|| X86_READY.load(Ordering::Acquire));
850    let epoch = request_tlb_flush(1);
851    x86_bounded_wait(|| tlb_flush_completed(1, epoch));
852    X86_QUALIFIED.store(true, Ordering::Release);
853    x86_bounded_wait(|| HART_ONLINE[1].load(Ordering::Acquire));
854    #[cfg(feature = "test-hooks")]
855    x86_shootdown_fixture_primary();
856    X86_BOOT_COMPLETE.store(true, Ordering::Release);
857    log::info!("[x86-smp] transport IPI-invalidate bidirectional=pass");
858    log::info!("[x86-smp] topology online=2 processor_ids={};{} qualified=yes", bsp_id, ap_id);
859    if flags & (1 << 9) == 0 { unsafe { core::arch::asm!("cli", options(nomem, nostack)); } }
860}

862#[cfg(target_arch = "x86_64")]
863extern "C" fn x86_secondary_main(logical: usize, lapic_id: usize, kernel_cr3: usize) -> ! {
864    assert!(logical == 1);
865    if !unsafe { hal::x86_64::init_secondary_cpu(logical, lapic_id as u32, kernel_cr3) } {
866        panic!("[x86-smp] active AP feature/isolation initialization failed");
867    }
868    crate::task::hart_local::install(logical);
869    // Establish this CPU's own idle activation before enabling delivery.
870    unsafe { core::arch::asm!("sti", options(nomem, nostack)); }
871    let epoch = request_tlb_flush(0);
872    x86_bounded_wait(|| tlb_flush_completed(0, epoch));
873    X86_READY.store(true, Ordering::Release);
874    x86_bounded_wait(|| X86_QUALIFIED.load(Ordering::Acquire));
875    HART_ONLINE[logical].store(true, Ordering::Release);
876    x86_bounded_wait(|| {
877        #[cfg(feature = "test-hooks")]
878        x86_shootdown_fixture_secondary();
879        X86_BOOT_COMPLETE.load(Ordering::Acquire)
880    });
881    loop {
882        crate::task::yield_cpu();
883        unsafe { core::arch::asm!("sti", "hlt", options(nomem, nostack)); }
884    }
885}

887// This behavioral fixture caches a genuinely GLOBAL shared translation on the
888// AP, changes its binding on the BSP, then forces the AP to contend on the same
889// page-table lock with IF clear. Success requires lock-spin local invalidation;
890// an ISR-only acknowledgement deadlocks or retains the old binding.
891#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
892const X86_PROBE_VA: usize = 0x0600_0000;
893#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
894static X86_PROBE_STAGE: AtomicUsize = AtomicUsize::new(0);
895#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
896static X86_PROBE_BEFORE: AtomicUsize = AtomicUsize::new(0);
897#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
898static X86_PROBE_AFTER: AtomicUsize = AtomicUsize::new(0);
899
900#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
901fn x86_shootdown_fixture_secondary() {
902    if X86_PROBE_STAGE.load(Ordering::Acquire) != 1 { return; }
903    X86_PROBE_BEFORE.store(
904        unsafe { core::ptr::read_volatile(X86_PROBE_VA as *const usize) }, Ordering::Release,
905    );
906    X86_PROBE_STAGE.store(2, Ordering::Release);
907    x86_bounded_wait(|| X86_PROBE_STAGE.load(Ordering::Acquire) == 3);
908    unsafe { core::arch::asm!("cli", options(nomem, nostack)); }
909    X86_PROBE_STAGE.store(4, Ordering::Release);
910    let root = crate::memory::paging::KERNEL_ROOT.lock();
911    X86_PROBE_AFTER.store(
912        unsafe { core::ptr::read_volatile(X86_PROBE_VA as *const usize) }, Ordering::Release,
913    );
914    drop(root);
915    X86_PROBE_STAGE.store(5, Ordering::Release);
916    unsafe { core::arch::asm!("sti", options(nomem, nostack)); }
917}

919#[cfg(all(target_arch = "x86_64", feature = "test-hooks"))]
920fn x86_shootdown_fixture_primary() {
921    use crate::memory::frame::{phys_to_virt, FRAME_ALLOCATOR};
922    use crate::memory::paging::{KERNEL_ROOT, PAGE_SIZE};
923    use hal::paging::{PTE_PRESENT, PTE_WRITABLE, PTE_NX, walk_read, walk_create};
924    let flags = PTE_PRESENT | PTE_WRITABLE | PTE_NX | (1 << 8);
925    let (old, new) = {
926        let mut allocator = FRAME_ALLOCATOR.lock();
927        let allocator = allocator.as_mut().expect("fixture frame allocator");
928        let old = allocator.allocate_frame().expect("fixture old frame");
929        let new = allocator.allocate_frame().expect("fixture replacement frame");
930        unsafe {
931            core::ptr::write_volatile(phys_to_virt(old) as *mut usize, 0x1357);
932            core::ptr::write_volatile(phys_to_virt(new) as *mut usize, 0x2468);
933        }
934        let root = KERNEL_ROOT.lock();
935        let root_pointer = phys_to_virt(root.expect("fixture root")) as *const u64;
936        assert!(unsafe { walk_read(root_pointer, X86_PROBE_VA) }.is_none_or(|pte| pte & PTE_PRESENT == 0),
937            "shootdown fixture VA already mapped");
938        drop(root);
939        crate::memory::paging::map_page_x86(allocator, X86_PROBE_VA, old, flags).expect("fixture map");
940        (old, new)
941    };
942    X86_PROBE_STAGE.store(1, Ordering::Release);
943    x86_bounded_wait(|| X86_PROBE_STAGE.load(Ordering::Acquire) == 2);
944    {
945        let root = KERNEL_ROOT.lock();
946        let root_pointer = phys_to_virt(root.expect("fixture root")) as *mut u64;
947        let mut no_allocate = || None;
948        let leaf = unsafe { walk_create(root_pointer, X86_PROBE_VA, &mut no_allocate) }.expect("fixture leaf");
949        unsafe { core::ptr::write_volatile(leaf, new as u64 | flags); }
950        X86_PROBE_STAGE.store(3, Ordering::Release);
951        x86_bounded_wait(|| X86_PROBE_STAGE.load(Ordering::Acquire) == 4);
952        crate::memory::tlb_shootdown::flush_range(X86_PROBE_VA, PAGE_SIZE);
953    }
954    x86_bounded_wait(|| X86_PROBE_STAGE.load(Ordering::Acquire) == 5);
955    assert_eq!(X86_PROBE_BEFORE.load(Ordering::Acquire), 0x1357);
956    assert_eq!(X86_PROBE_AFTER.load(Ordering::Acquire), 0x2468);
957    crate::memory::paging::unmap_page_x86(X86_PROBE_VA).expect("fixture unmap");
958    let mut allocator = FRAME_ALLOCATOR.lock();
959    let allocator = allocator.as_mut().expect("fixture frame allocator");
960    allocator.deallocate_frame(old);
961    allocator.deallocate_frame(new);
962    log::info!("S22-X86-REMOTE-SHOOTDOWN: PASS harts=2 global=true irq_masked_lock=true before=4951 after=9320");
963}

1040/// Rust entry point of a secondary core started by `PSCI_CPU_ON`.
1041///
1042/// Reached from `_secondary_entry` with the boot core's translation regime
1043/// installed, this hart's stack in place, interrupts masked, and `hart_id` in
1044/// x0. Returns only by never returning: the hart parks and lives out its life
1045/// in the timer/IPI trap path, which is where its scheduler round runs.
1046#[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
1047#[no_mangle]
1048pub extern "C" fn smp_aarch64_secondary_main(hart_id: usize) -> ! {
1049    crate::task::hart_local::install(hart_id);
1050    hal::aarch64::init_secondary_hart();
1051    {
1052        use hal::Arch;
1053        hal::ARCH.enable_interrupts();
1054        if !hal::ARCH.interrupts_enabled() {
1055            panic!("[smp] hart {} could not enable interrupts", hart_id);
1056        }
1057    }
1058    if hart_id < MAX_HARTS {
1059        log::info!("[smp] hart {} trap-ready, interrupts-enabled", hart_id);
1060        HART_ONLINE[hart_id].store(true, Ordering::Release);
1061    }
1062    loop {
1063        // SAFETY: WFI suspends until the next interrupt; no state changes.
1064        unsafe { core::arch::asm!("wfi", options(nomem, nostack)) };
1065        core::hint::spin_loop();
1066    }
1067}

1070///
1071/// Used by the scheduler to fall back to hart 0 on single-hart systems
1072/// (e.g. QEMU without `-smp 2`) so RT-priority tasks still get scheduled.
1073#[inline]

1078/// Entry point for secondary harts, called from `_secondary_entry` asm.
1079///
1080/// a0 = hart_id (set by OpenSBI per SBI HSM §9.1.1).
1081///
1082/// Installs the trap vector, enables the timer, runs the per-hart scheduler loop.
1083#[no_mangle]
1084pub extern "C" fn smp_hart_entry(physical_hart: usize) -> ! {
1085    #[cfg(target_arch = "riscv64")]
1086    let hart_id = physical_to_logical(physical_hart)
1087        .unwrap_or_else(|| panic!("unmapped RV64 physical hart {}", physical_hart));
1088    #[cfg(not(target_arch = "riscv64"))]
1089    let hart_id = physical_hart;
1090
1091    #[cfg(target_arch = "riscv64")]
1092    {
1093        let root = crate::memory::paging::KERNEL_ROOT
1094            .lock()
1095            .expect("RV64 secondary started before kernel paging root");
1096        // SAFETY: the boot hart published a complete shared root before HSM
1097        // startup; it maps this entry code, stack, and all kernel globals.
1098        unsafe {
1099            crate::memory::paging::activate_paging(root);
1100            core::arch::asm!(
1101                "csrs sstatus, {sum}",
1102                sum = in(reg) 0x40000usize,
1103                options(nostack)
1104            );
1105        }
1106    }
1107    // Install the trap vector (each hart has its own stvec CSR).
1108    // `hal::ARCH.init()` sets stvec + enables SSIE.
1109    #[cfg(target_arch = "riscv64")]
1110    {
1111        crate::task::hart_local::install(hart_id);
1112        use hal::Arch;
1113        hal::ARCH.init();
1114        // ARCH.init installs the bootstrap-safe default vector. Restore this
1115        // secondary's logical vector before any interrupt is enabled.
1116        hal::trap::init_for_hart(hart_id);
1117    }
1118
1119    // Enable S-mode timer interrupt and arm the first tick on this hart.
1120    // Each hart has its own mtimecmp register via SBI; arming here starts
1121    // the 10ms preemption slice for this hart.
1122    #[cfg(target_arch = "riscv64")]
1123    {
1124        // SAFETY: csrs on sie is always legal from S-mode (RISC-V priv spec §4.1.3).
1125        unsafe {
1126            core::arch::asm!("csrs sie, {stie}", stie = in(reg) 0x20usize);
1127        }
1128        let next = hal::common::timer::read_mtime() + hal::common::timer::TICKS_PER_10MS;
1129        hal::common::sbi::set_timer(next);
1130    }
1131
1132    // `ARCH.init()` enables SSIE in `sie`, but it deliberately leaves the
1133    // per-hart global SIE bit clear. A secondary otherwise wakes from WFI with
1134    // a pending IPI but never takes the SSIP trap that enters `yield_cpu()`;
1135    // explicitly enable delivery before advertising this hart as schedulable.
1136    #[cfg(target_arch = "riscv64")]
1137    {
1138        use hal::Arch;
1139        hal::ARCH.enable_interrupts();
1140        if !hal::ARCH.interrupts_enabled() {
1141            panic!(
1142                "[smp] hart {} could not enable supervisor interrupts",
1143                hart_id
1144            );
1145        }
1146    }
1147
1148    // Signal hart 0's bounded wait only after this hart can actually take the
1149    // dispatch IPI and run its local scheduler.
1150    if hart_id < MAX_HARTS {
1151        log::info!(
1152            "[smp] physical {} -> logical {} trap-ready, interrupts-enabled",
1153            physical_hart,
1154            hart_id
1155        );
1156        #[cfg(feature = "test-hooks")]
1157        log::info!(
1158            "[selftest] SMP-RETIREMENT: stage=hart{}-interrupts-enabled",
1159            hart_id
1160        );
1161        HART_ONLINE[hart_id].store(true, Ordering::Release);
1162    }
1163
1164    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
1165    crate::memory::tlb_shootdown_selftest::run_secondary(hart_id);
1166
1167    // Per-hart scheduler loop.  The timer ISR (vi_timer_tick) calls yield_cpu()
1168    // which runs pick_next for THIS hart (work-stealing from hart 0 if idle).
1169    // Between ticks we sit in WFI to save power.  Interrupts are enabled on
1170    // entry (ARCH.init() sets sstatus.SIE=1), so WFI fires on the timer ISR.
1171    loop {
1172        #[cfg(feature = "test-hooks")]
1173        crate::loader::atomic_publication_tests::observe_schedule_attempt();
1174        // SAFETY: wfi suspends until the next interrupt; state is unchanged.
1175        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1176        unsafe {
1177            core::arch::asm!("wfi", options(nomem, nostack))
1178        };
1179        core::hint::spin_loop();
1180    }
1181}
