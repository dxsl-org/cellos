3use alloc::string::{String, ToString};
4use alloc::vec::Vec;

7/// Every allocation and mapping needed by an unpublished ELF task.
8pub struct PreparedElfTask {
9    name: String,
10    requested_cell_id: CellId,
11    allowed_drivers: Vec<usize>,
12    kstack: super::stack::Stack,
13    ustack: super::stack::Stack,
14    segments: super::stack::CellSegments,
15    entry: usize,
16    load_base: usize,
17}
18
19impl PreparedElfTask {
20    pub(super) fn into_task(
21        self,
22        id: usize,
23        cell_id: CellId,
24    ) -> (alloc::boxed::Box<super::Task>, usize) {
25        let mut task = alloc::boxed::Box::new(super::Task::new(
26            id,
27            cell_id,
28            &self.name,
29            self.allowed_drivers,
30        ));
31        task.kernel_stack = Some(self.kstack);
32        task.user_stack = Some(self.ustack);
33        task.segment_mem = Some(self.segments);
34        super::prime_user_mode_entry(&mut task, self.entry, 0);
35        (task, self.load_base)
36    }
37
38    pub(super) fn requested_cell_id(&self) -> CellId {
39        self.requested_cell_id
40    }
41}

43/// Parse, map, relocate, protect, and stack an ELF without touching scheduler state.
44pub fn prepare_elf_task(
45    data: &[u8],
46    name: &str,
47    requested_cell_id: CellId,
48    allowed_drivers: Vec<usize>,
49) -> Result<PreparedElfTask, ViError> {
50    use crate::loader::{ElfLoader, ElfParser};
51
52    if data.len() < 4 || &data[..4] != b"\x7fELF" {
53        return Err(ViError::InvalidInput);
54    }
55    let aligned = crate::loader::aligned_elf::bytes(data);
56    let elf_data = aligned.as_ref();
57
58    let loader = ElfLoader;
59    let header = loader.parse_header(elf_data)?;
60    let load_base = if super::elf_is_pie(elf_data) {
61        crate::loader::va_alloc::alloc_cell_va().ok_or(ViError::OutOfMemory)?
62    } else {
63        0
64    };
65    if let Err(error) = crate::loader::atomic_checkpoint("AP-01") {
66        if load_base != 0 {
67            crate::loader::va_alloc::free_cell_va(load_base);
68        }
69        return Err(error);
70    }
71
72    let seg_pages = {
73        let mut frames = crate::memory::frame::FRAME_ALLOCATOR.lock();
74        let allocator = frames.as_mut().ok_or(ViError::OutOfMemory)?;
75        match loader.load_segments(elf_data, allocator, load_base) {
76            Ok(pages) => pages,
77            Err(error) => {
78                if load_base != 0 {
79                    crate::loader::va_alloc::free_cell_va(load_base);
80                }
81                return Err(error);
82            }
83        }
84    };
85    let final_flags = seg_pages
86        .iter()
87        .map(|p| (p.va, p.final_flags))
88        .collect::<Vec<_>>();
89    let segments = super::stack::CellSegments::with_writable_pages(
90        seg_pages.iter().map(|p| (p.va, p.frame)).collect(),
91        seg_pages
92            .iter()
93            .filter(|p| p.final_flags.bits() & crate::memory::paging::Flags::WRITE != 0)
94            .map(|p| p.va)
95            .collect(),
96        load_base,
97    );
98    #[cfg(feature = "test-hooks")]
99    crate::loader::atomic_publication_tests::observe_unpublished_segments(&segments);
100    crate::loader::atomic_checkpoint("AP-02")?;
101
102    if load_base != 0 {
103        if let Ok(rela) = loader.get_section(elf_data, ".rela.dyn") {
104            crate::loader::reloc::apply_relocations(load_base, &seg_pages, rela)?;
105        }
106    }
107    crate::loader::atomic_checkpoint("AP-03")?;
108    crate::loader::wx::enforce(&final_flags, name)?;
109
110    let pages = super::stack_pages_for(name);
111    let kstack = super::stack::Stack::new_kernel(pages).map_err(|_| ViError::OutOfMemory)?;
112    // SAFETY: the freshly allocated stack's usable range is exclusively owned.
113    unsafe {
114        core::ptr::write_bytes(kstack.usable_start() as *mut u8, 0, kstack.usable_bytes());
115    }
116    #[cfg(feature = "test-hooks")]
117    kstack.test_hook_prime_watermark();
118    crate::loader::atomic_checkpoint("AP-04")?;
119    let ustack = super::stack::Stack::new_user(pages).map_err(|_| ViError::OutOfMemory)?;
120    #[cfg(feature = "test-hooks")]
121    ustack.test_hook_prime_watermark();
122
123    Ok(PreparedElfTask {
124        name: name.to_string(),
125        requested_cell_id,
126        allowed_drivers,
127        kstack,
128        ustack,
129        segments,
130        entry: header.entry.wrapping_add(load_base),
131        load_base,
132    })
133}
