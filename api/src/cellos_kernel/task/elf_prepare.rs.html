7/// Every allocation and mapping needed by an unpublished ELF task.
8pub struct PreparedElfTask {
9    name: String,
10    requested_cell_id: CellId,
11    allowed_drivers: Vec<usize>,
12    kstack: super::stack::Stack,
13    ustack: super::stack::Stack,
14    segments: crate::memory::cell_segments::CellSegments,
15    entry: usize,
16    load_base: usize,
17    /// Entry stack pointer: the Cell initial stack block (CP1 AX-2) on 64-bit
18    /// targets, the bare user stack top elsewhere.
19    user_sp: usize,
20    /// Initial `IA32_U_CET` (x86_64 CET-IBT opt-in); 0 everywhere else.
21    user_cet: u64,
22}
23
24impl PreparedElfTask {
25    /// The task of incarnation `generation`, minted before its quota row.
26    pub(super) fn into_task(
27        self,
28        id: usize,
29        cell_id: CellId,
30        generation: u64,
31    ) -> Result<(alloc::boxed::Box<super::Task>, usize), ViError> {
32        let mut task = crate::memory::heap::try_box(super::Task::of_incarnation(
33            id,
34            cell_id,
35            generation,
36            &self.name,
37            self.allowed_drivers,
38        ))?;
39        task.kernel_stack = Some(self.kstack);
40        task.user_stack = Some(self.ustack);
41        task.segment_mem = Some(self.segments);
42        task.user_cet = self.user_cet;
43        super::prime_user_mode_entry_at(&mut task, self.entry, 0, self.user_sp);
44        Ok((task, self.load_base))
45    }
46
47    /// Where the image and the user stack landed: (PIE load base, `0` for a
48    /// fixed-VA image; user-stack slot base).
49    pub(crate) fn placement(&self) -> (usize, usize) {
50        (self.load_base, self.ustack.base)
51    }
52
53    pub(super) fn requested_cell_id(&self) -> CellId {
54        self.requested_cell_id
55    }
56}

58/// `Iterator::collect` allocates through the infallible allocator: a full heap
59/// halts the kernel instead of refusing the spawn. Every vector here is sized by
60/// the ELF's page count (4 KiB for `/bin/bench-probe`), so reserve fallibly and
61/// fill — the spawn then returns `ViError::OutOfMemory` like any other capacity
62/// refusal. Measured 2026-10-03: these collects were what ended the capacity
63/// sweep once the spawn path's larger transients stopped binding.
64fn collect_reserved<T>(iter: impl ExactSizeIterator<Item = T>) -> Result<Vec<T>, ViError> {
65    let mut out = Vec::new();
66    out.try_reserve_exact(iter.len())
67        .map_err(|_| ViError::OutOfMemory)?;
68    for item in iter {
69        out.push(item);
70    }
71    Ok(out)
72}

74/// Own the image's VA slot before any subsequent fallible preparation step.
75/// An unwind hands the slot to its last retired run instead
76/// ([`Self::hold_in`]), so a VA a remote hart may still translate is not
77/// handed out again before the unwind's flush confirms (census risk 2).
78struct PreparationVa(core::cell::Cell<usize>);
79impl PreparationVa {
80    /// Keep the slot reserved until the runs `batch` retired are free; the
81    /// guard no longer releases it then.
82    fn hold_in(&self, batch: &mut crate::memory::frame_run::Batch) {
83        let slot = self.0.get();
84        if slot != 0
85            && batch.hold_va(crate::memory::frame_run::VaHold {
86                release: crate::memory::cell_segments::release_cell_va,
87                start: slot,
88                end: slot,
89            })
90        {
91            self.0.set(0);
92        }
93    }
94}
95impl Drop for PreparationVa {
96    fn drop(&mut self) {
97        let slot = self.0.get();
98        if slot != 0 {
99            crate::loader::va_alloc::free_cell_va(slot);
100        }
101    }
102}

104/// LoadedPage is metadata, so bridge its mappings into CellSegments ownership
105/// without another allocation. This guard must be declared after the page Vec
106/// and the VA guard.
107struct PreparationMappings<'a>(
108    Option<&'a [crate::loader::elf::LoadedPage]>,
109    &'a PreparationVa,
110);
111impl Drop for PreparationMappings<'_> {
112    fn drop(&mut self) {
113        if let Some(pages) = self.0.take() {
114            let mut batch = crate::memory::frame_run::Batch::new();
115            crate::loader::ElfLoader::unwind(pages, &mut batch);
116            self.1.hold_in(&mut batch);
117            let _ = batch.complete();
118        }
119    }
120}

122/// Parse, map, relocate, protect, and stack an ELF without touching scheduler state.
123pub fn prepare_elf_task(
124    data: &[u8],
125    name: &str,
126    requested_cell_id: CellId,
127    allowed_drivers: Vec<usize>,
128) -> Result<PreparedElfTask, ViError> {
129    use crate::loader::{ElfLoader, ElfParser};
130
131    if data.len() < 4 || &data[..4] != b"\x7fELF" {
132        return Err(ViError::InvalidInput);
133    }
134    let aligned = crate::loader::aligned_elf::bytes(data)?;
135    let elf_data = aligned.as_ref();
136
137    let loader = ElfLoader;
138    let header = loader.parse_header(elf_data)?;
139    // Forward-edge CFI opt-in, read from the admitted bytes before any mapping
140    // so a malformed property refuses the spawn with nothing to unwind.
141    let cfi = crate::loader::gnu_property::arm(
142        crate::loader::gnu_property::forward_edge_cfi(elf_data)?,
143        name,
144    );
145    // Every image owns exactly one VA slot and may map only inside it: a PIE
146    // takes any free slot; a fixed-VA image claims the slot it is linked into
147    // and is refused when another spawn holds it, so no image page can land
148    // on the null page, a stack guard or a foreign slot.
149    let (load_base, slot_base) = if super::elf_is_pie(elf_data) {
150        let base = crate::loader::va_alloc::alloc_cell_va().ok_or(ViError::OutOfMemory)?;
151        (base, base)
152    } else {
153        let slot = loader.fixed_image_slot(elf_data)?;
154        if !crate::loader::va_alloc::reserve_cell_va(slot) {
155            log::error!("ELF: fixed-VA image slot {slot:#x} is owned by another cell");
156            return Err(ViError::PermissionDenied);
157        }
158        (0, slot)
159    };
160    let va_rollback = PreparationVa(core::cell::Cell::new(slot_base));
161    crate::loader::atomic_checkpoint("AP-01")?;
162
163    let mut load_batch = crate::memory::frame_run::Batch::new();
164    let loaded = crate::memory::frame::with(|allocator| {
165        loader.load_segments(
166            elf_data,
167            allocator,
168            &mut load_batch,
169            load_base,
170            slot_base,
171            cfi.guarded_code,
172        )
173    });
174    // A failed load unwound its pages into the batch: they hold the slot.
175    if !matches!(loaded, Some(Ok(_))) {
176        va_rollback.hold_in(&mut load_batch);
177    }
178    // One shared flush for the whole image (or for its unwind), after the frame
179    // allocator is released and before anything relocates through it.
180    let flushed = load_batch.complete();
181    let seg_pages = loaded.ok_or(ViError::OutOfMemory)??;
182    let mut mapping_rollback = PreparationMappings(Some(&seg_pages), &va_rollback);
183    if !flushed.is_confirmed() {
184        log::error!("ELF: image mapping invalidation unconfirmed — refusing the spawn");
185        return Err(ViError::OutOfMemory);
186    }
187    #[cfg(feature = "test-hooks")]
188    crate::loader::atomic_publication_tests::observe_unpublished_pages(&seg_pages);
189    let final_flags = crate::loader::atomic_checkpoint("AP-RF")
190        .map_err(|_| ViError::OutOfMemory)
191        .and_then(|_| collect_reserved(seg_pages.iter().map(|p| (p.va, p.final_flags))))
192        .inspect_err(|_| {
193            #[cfg(feature = "p01-trace")]
194            p01_post_load_failure(&seg_pages, load_base, 2);
195        })?;
196    let mapped_pages = crate::loader::atomic_checkpoint("AP-RM")
197        .map_err(|_| ViError::OutOfMemory)
198        .and_then(|_| collect_reserved(seg_pages.iter().map(|p| (p.va, p.frame))))
199        .inspect_err(|_| {
200            #[cfg(feature = "p01-trace")]
201            p01_post_load_failure(&seg_pages, load_base, 3);
202        })?;
203    let writable_count = seg_pages
204        .iter()
205        .filter(|p| p.final_flags.bits() & crate::memory::paging::Flags::WRITE != 0)
206        .count();
207    let mut writable_pages = Vec::new();
208    crate::loader::atomic_checkpoint("AP-RW")
209        .map_err(|_| ViError::OutOfMemory)
210        .and_then(|_| {
211            writable_pages
212                .try_reserve_exact(writable_count)
213                .map_err(|_| ViError::OutOfMemory)
214        })
215        .map_err(|_| {
216            #[cfg(feature = "p01-trace")]
217            p01_post_load_failure(&seg_pages, load_base, 4);
218            ViError::OutOfMemory
219        })?;
220    for page in seg_pages.iter() {
221        if page.final_flags.bits() & crate::memory::paging::Flags::WRITE != 0 {
222            writable_pages.push(page.va);
223        }
224    }
225    #[cfg(feature = "p01-trace")]
226    {
227        // These exclusive backing arrays survive in CellSegments. Other loader
228        // scratch/shared-helper allocations retain origin-only classification.
229        crate::memory::heap::p01_owner::adopt_vec(&mapped_pages);
230        crate::memory::heap::p01_owner::adopt_vec(&writable_pages);
231    }
232    let segments = crate::memory::cell_segments::CellSegments::with_writable_pages(
233        mapped_pages,
234        writable_pages,
235        slot_base,
236    )
237    .with_guarded_code(cfi.guarded_code);
238    // Constructor is infallible: the segment owner now exclusively owns both
239    // frame reclamation and PIE release, including every later failure edge.
240    mapping_rollback.0 = None;
241    va_rollback.0.set(0);
242    #[cfg(feature = "test-hooks")]
243    crate::loader::atomic_publication_tests::observe_unpublished_segments(&segments);
244    crate::loader::atomic_checkpoint("AP-02")?;
245
246    if load_base != 0 {
247        if let Ok(rela) = loader.get_section(elf_data, ".rela.dyn") {
248            crate::loader::reloc::apply_relocations(load_base, &seg_pages, rela)?;
249        }
250    }
251    crate::loader::atomic_checkpoint("AP-03")?;
252    crate::loader::wx::enforce(&final_flags, name)?;
253
254    let pages = super::stack_pages_for(name);
255    let kstack = super::stack::Stack::new_kernel(pages).map_err(|_| ViError::OutOfMemory)?;
256    kstack.zero_usable();
257    #[cfg(feature = "test-hooks")]
258    kstack.test_hook_prime_watermark();
259    crate::loader::atomic_checkpoint("AP-04")?;
260    let ustack = super::stack::Stack::new_user(pages).map_err(|_| ViError::OutOfMemory)?;
261    // Frames come back from the allocator holding their last owner's bytes, and
262    // the initial-stack writer fills only the top of the range: zero all of it so
263    // a new cell can never read another cell's data off its own stack. Through
264    // the frames' kernel alias, never the user stack's USER VA (census G12).
265    ustack.zero_usable();
266    #[cfg(feature = "test-hooks")]
267    ustack.test_hook_prime_watermark();
268    let entry = header.entry.wrapping_add(load_base);
269    #[cfg(target_pointer_width = "64")]
270    let user_sp = write_initial_stack(elf_data, &ustack, load_base, entry, name)?;
271    #[cfg(not(target_pointer_width = "64"))]
272    let user_sp = ustack.top;
273
274    Ok(PreparedElfTask {
275        name: name.to_string(),
276        requested_cell_id,
277        allowed_drivers,
278        kstack,
279        ustack,
280        segments,
281        entry,
282        load_base,
283        user_sp,
284        user_cet: cfi.user_cet,
285    })
286}

288/// Write the Cell initial stack (argc, argv, envp, auxv; CP1 AX-2…AX-5) below
289/// the top of the root user stack and return the entry `sp`. Runs before the
290/// task exists, so the stack is reached through the shared address space where
291/// `Stack::new_user` mapped it; a Tier-2 domain maps the same frames later.
292#[cfg(target_pointer_width = "64")]
293fn write_initial_stack(
294    elf: &[u8],
295    ustack: &super::stack::Stack,
296    load_base: usize,
297    entry: usize,
298    name: &str,
299) -> Result<usize, ViError> {
300    use cell_initial_stack::{build, Image, MAX_PHDRS, PHENT};
301
302    let field = |at: usize, width: usize| -> Option<u64> {
303        let bytes = elf.get(at..at + width)?;
304        let mut word = [0u8; 8];
305        word[..width].copy_from_slice(bytes);
306        Some(u64::from_le_bytes(word))
307    };
308    // ELF64 header: e_phoff @32 (8), e_phentsize @54 (2), e_phnum @56 (2).
309    let (Some(phoff), Some(phentsize), Some(phnum)) = (field(32, 8), field(54, 2), field(56, 2))
310    else {
311        return Err(ViError::InvalidInput);
312    };
313    if phentsize as usize != PHENT || phnum as usize > MAX_PHDRS {
314        log::warn!(
315            "[loader] {}: {} program headers of {} bytes exceed the initial-stack bound ({} × {})",
316            name,
317            phnum,
318            phentsize,
319            MAX_PHDRS,
320            PHENT
321        );
322        return Err(ViError::InvalidInput);
323    }
324    let start = usize::try_from(phoff).map_err(|_| ViError::InvalidInput)?;
325    let phdrs = start
326        .checked_add(phnum as usize * PHENT)
327        .and_then(|end| elf.get(start..end))
328        .ok_or(ViError::InvalidInput)?;
329
330    let mut seed = [0u8; 16];
331    let random = (crate::dev::virtio_rng::get_random(&mut seed) == seed.len()).then_some(seed);
332    let image = Image {
333        phdrs,
334        load_base: load_base as u64,
335        entry: entry as u64,
336        random,
337    };
338    let block = build(ustack.top as u64, &image).map_err(|_| ViError::InvalidInput)?;
339    super::copy_glue::TaskCopyView::sas()
340        .write_bytes(block.sp() as usize, block.bytes())
341        .map_err(|_| ViError::OutOfMemory)?;
342    Ok(block.sp() as usize)
343}

345/// Observe each exact reserve failure while rollback still owns the live frames.
346/// This witness precedes real unwind; it is not a final retained-leak assertion.
347#[cfg(feature = "p01-trace")]
348fn p01_post_load_failure(pages: &[crate::loader::elf::LoadedPage], load_base: usize, edge: u64) {
349    // Source lock only, no scheduler/owner invention. Preserve one coherent
350    // physical witness batch while the mapped leaves remain allocator-resident.
351    let _ = crate::memory::frame::with(|_frames| {
352        let _physical_write = crate::p01_retained::physical_write();
353        for page in pages {
354            crate::p01_retained::physical_prepare_failed(page.frame, page.va, load_base, edge);
355        }
356    });
