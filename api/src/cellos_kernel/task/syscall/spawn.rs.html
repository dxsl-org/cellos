9/// Called outside source locks after exact root teardown. Never infer pin/DMA or
10/// historical heap ownership from grant-table absence.
11#[cfg(feature = "p01-trace")]
12pub(crate) fn p01_retirement_check(owner: crate::p01_measurement::Owner) {
13    let page_clear =
14        {
15            let table = grant_table_lock().lock();
16            table.as_ref().map_or(Some(true), |table| {
17                if table.values().nth(128).is_some() {
18                    return None;
19                }
20                Some(!table.values().any(|g| {
21                    g.owner.cell.0 == owner.cell && g.owner.generation == owner.generation
22                }))
23            })
24        };
25    let registered_clear =
26        {
27            let table = reg_grant_table_lock().lock();
28            table.as_ref().map_or(Some(true), |table| {
29                if table.values().nth(128).is_some() {
30                    return None;
31                }
32                Some(!table.values().any(|g| {
33                    g.owner.cell.0 == owner.cell && g.owner.generation == owner.generation
34                }))
35            })
36        };
37    if page_clear == Some(true) && registered_clear == Some(true) {
38        crate::p01_resources::checked(owner, 3);
39    } else {
40        crate::p01_resources::unknown(owner, 3);
41    }
42}

58/// Record `tid` as trusted init. Only the first publication counts.
59pub(crate) fn publish_trusted_init(tid: usize) {
60    use core::sync::atomic::Ordering;
61    let _first_wins =
62        TRUSTED_INIT_TID.compare_exchange(0, tid, Ordering::AcqRel, Ordering::Acquire);
63}

65pub(super) fn caller_is_trusted_init(caller_id: usize) -> bool {
66    caller_id != 0 && TRUSTED_INIT_TID.load(core::sync::atomic::Ordering::Acquire) == caller_id

110/// The caller's launch identity, spawn and supervisor flags.
111///
112/// The identity is **never the task name** (the basename of a spawner-chosen
113/// path, architecture review C-1). It is `"init"` for any task of the Cell
114/// whose root the kernel published as trusted init, otherwise the program name
115/// from the verified `__ViCell_image` section of the caller's Cell root
116/// (`Task::image`) when the kernel itself resolved that root's install path
117/// (`Task::launch_named`), and empty otherwise — unsigned or unnamed images,
118/// caller-supplied ELF/Mem bytes, sandbox paths and aliases match no launch row. Threads share their Cell root's identity.
119pub(super) fn caller_launch_state(caller_id: usize) -> Option<(String, bool, bool)> {
120    crate::task::SCHEDULER.lock().as_ref().and_then(|sched| {
121        let identity = launch_identity_in(sched, caller_id)?;
122        let rights = sched.cell_rights(caller_id);
123        Some((identity, rights.spawn, rights.supervisor))
124    })
125}

127/// Non-owning peek at the next delivery event. Payload bytes are copied into
128/// kernel memory; the original record is NOT removed. Call `commit_resume`
129/// after a successful copy-out to remove exactly the same event.
130pub(in crate::task) enum ResumeSnapshot {
131    /// A peer died or an exit reason was synthesised. `is_owner_death` flags
132    /// which record to commit: owner-deaths queue vs. `pending_exit_reason`.
133    Death {
134        sender_tid: usize,
135        reason: usize,
136        is_owner_death: bool,
137    },
138    /// A queued message. Payload already staged; commit key is `delivery_id`
139    /// (wire) or `sender_tid` (inline) — whichever is `Some`/applicable.
140    Message {
141        sender_tid: usize,
142        wire_header: Option<crate::ipc::ipc_wire::IpcWireHeader>,
143        delivery_id: Option<u64>,
144        sender_cell_id: u64,
145        sender_generation: u64,
146        payload: MsgPayload,
147    },
148    /// Woken with no pending message (timeout, manual wake, or 0-return).
149    Wake { sender_tid: usize },
150}
151
152/// Peek at the next delivery event matching `mask` without removing anything.
153/// Returns `Err(())` on allocation failure (OOM staging payload bytes).
154pub(in crate::task) fn snapshot_resume(
155    task: &crate::task::Task,
156    mask: usize,
157) -> Result<ResumeSnapshot, ()> {
158    // Owner-death events take priority (same order as take_resume_delivery).
159    if crate::task::Task::owner_death_matches_receive_mask(mask) {
160        if let Some((_, root_tid, reason)) = task.pending_owner_deaths.first().copied() {
161            return Ok(ResumeSnapshot::Death {
162                sender_tid: root_tid,
163                reason,
164                is_owner_death: true,
165            });
166        }
167    }
168    if let Some(reason) = task.pending_exit_reason {
169        return Ok(ResumeSnapshot::Death {
170            sender_tid: task.caller.current.unwrap_or(0),
171            reason,
172            is_owner_death: false,
173        });
174    }
175    if let Some(index) = task
176        .pending_msgs
177        .iter()
178        .position(|m| mask == 0 || m.sender_tid == mask)
179    {
180        let msg = &task.pending_msgs.as_slice()[index];
181        let wire_header = msg.wire_header();
182        let delivery_id = wire_header.as_ref().map(|h| h.delivery_id);
183        let (sender_cell_id, sender_generation) = match &wire_header {
184            Some(h) => (h.sender_cell_id, h.sender_generation),
185            None => (0, 0),
186        };
187        #[cfg(feature = "p01-trace")]
188        let _semantic = crate::memory::heap::p01_owner::creation(
189            Some(crate::p01_resources::owner(task)),
190            crate::memory::heap::p01_owner::RECEIVE_SCRATCH,
191        );
192        let payload = MsgPayload::from_slice(msg.payload())?;
193        return Ok(ResumeSnapshot::Message {
194            sender_tid: msg.sender_tid,
195            wire_header,
196            delivery_id,
197            sender_cell_id,
198            sender_generation,
199            payload,
200        });
201    }
202    Ok(ResumeSnapshot::Wake {
203        sender_tid: task.caller.current.unwrap_or(0),
204    })
205}

215/// Build the guard only for a spawn syscall.
216///
217/// `consumes_staged_spawn_argv(&syscall).then_some(StagedSpawnArgvCleanup { .. })`
218/// looks equivalent but is not: `then_some` takes its argument **by value**, so the
219/// guard was constructed for *every* syscall and dropped again immediately when the
220/// predicate was false — its `Drop` then discarded the caller's staged command
221/// line. Staging an argv and then issuing any other syscall before the spawn (the
222/// shell's VFS read is exactly that) silently lost the command line; the cell then
223/// started with no arguments at all (`httpd 9091 /tmp/resp.txt` listened on :8080).
224pub(super) fn staged_spawn_argv_cleanup(
225    syscall: &Syscall,
226    caller_id: usize,
227) -> Option<StagedSpawnArgvCleanup> {
228    if consumes_staged_spawn_argv(syscall) {
229        Some(StagedSpawnArgvCleanup { caller_id })
230    } else {
231        None
232    }
233}

235pub(super) fn governed_spawn_request(
236    caller_id: usize,
237    route: crate::loader::launch_profile::LaunchRoute,
238    child_ceiling: crate::task::cap::CapSet,
239    priority: u8,
240    #[cfg(feature = "p01-trace")] trace: Option<crate::p01_measurement::Attempt>,
241) -> Result<crate::loader::SpawnRequest, SyscallError> {
242    let (incarnation, caller_authority, caller_root) = crate::task::SCHEDULER
243        .lock()
244        .as_ref()
245        .and_then(|sched| {
246            let task = sched.tasks.get(&caller_id)?;
247            #[cfg(feature = "p01-trace")]
248            if trace.is_some_and(|a| a.parent != crate::p01_measurement::Owner::of(task)) {
249                crate::p01_measurement::invalidate(crate::p01_measurement::IDENTITY);
250            }
251            Some((
252                (task.cell_id.0, task.cell_generation),
253                sched.cell_rights(caller_id),
254                task.root_tid,
255            ))
256        })
257        .ok_or(SyscallError::PermissionDenied)?;
258    let argv = crate::cell::state_stash::take_spawn_argv(caller_id);
259    Ok(crate::loader::SpawnRequest::governed_caller(
260        caller_id,
261        incarnation,
262        caller_authority,
263        child_ceiling,
264        priority,
265        argv,
266        names_child(caller_root, route),
267    ))
268}

297/// The launch identity of `caller_id` (see [`caller_launch_state`]) read under
298/// a scheduler the caller already holds. `None` when the task is gone.
299pub(in crate::task) fn launch_identity_in(
300    sched: &crate::sched::scheduler::Scheduler,
301    caller_id: usize,
302) -> Option<String> {
303    let init_tid = TRUSTED_INIT_TID.load(core::sync::atomic::Ordering::Acquire);
304    let task = sched.tasks.get(&caller_id)?;
305    Some(if init_tid != 0 && task.root_tid == init_tid {
306        String::from("init")
307    } else {
308        sched
309            .tasks
310            .get(&task.root_tid)
311            .filter(|root| root.launch_named)
312            .and_then(|root| root.image())
313            .filter(|image| image.is_signed())
314            .and_then(|image| core::str::from_utf8(image.name()).ok().map(String::from))
315            .unwrap_or_default()
316    })
317}

319pub(super) fn authorize_launch_edge(
320    caller_id: usize,
321    route: crate::loader::launch_profile::LaunchRoute,
322    target: &str,
323) -> Result<crate::loader::launch_profile::LaunchProfile, SyscallError> {
324    let (caller_name, has_spawn, has_supervisor) =
325        caller_launch_state(caller_id).unwrap_or((String::from("<unknown>"), false, false));
326    let caller = crate::loader::launch_profile::CallerLaunchState {
327        name: &caller_name,
328        has_spawn,
329        has_supervisor,
330    };
331    let profile = crate::loader::launch_profile::authorize(caller, route, target).ok_or_else(|| {
332        log::warn!(
333            "[loader] DENY launch edge: caller={} name={} route={:?} target={} spawn_cap={} supervisor_cap={}",
334            caller_id,
335            caller.name,
336            route,
337            target,
338            caller.has_spawn,
339            caller.has_supervisor
340        );
341        SyscallError::PermissionDenied
342    })?;
343    if profile.requires_lifecycle_authority && !caller.has_spawn {
344        log::warn!(
345            "[loader] DENY launch edge: caller={} name={} route={:?} target={} lacks lifecycle authority",
346            caller_id,
347            caller.name,
348            route,
349            target
350        );
351        return Err(SyscallError::PermissionDenied);
352    }
353    Ok(profile)
354}

356pub(crate) fn session_launch_request(
357    caller: usize,
358    path: &str,
359    route: crate::loader::launch_profile::LaunchRoute,
360) -> Result<crate::loader::SpawnRequest, SyscallError> {
361    let profile = authorize_launch_edge(caller, route, path)?;
362    let guard = crate::task::SCHEDULER.lock();
363    let task = guard
364        .as_ref()
365        .and_then(|s| s.tasks.get(&caller))
366        .ok_or(SyscallError::PermissionDenied)?;
367    // The CellSessionV1 transport itself carries no allowlist bit, so the
368    // launch it performs must hold the same bit as the equivalent syscall:
369    // SpawnFromPath for the path route, SpawnFromElf for caller-owned bytes.
370    let syscall = if matches!(route, crate::loader::launch_profile::LaunchRoute::Elf) {
371        ViSyscall::SpawnFromElf
372    } else {
373        ViSyscall::SpawnFromPath
374    };
375    if let Some(bit) = syscall.allowlist_bit() {
376        if (task.syscall_allowlist >> bit) & 1 == 0 {
377            log::warn!(
378                "[kernel] session launch denied for tid {}: {:?} bit {} not allowed",
379                caller,
380                syscall,
381                bit
382            );
383            return Err(SyscallError::PermissionDenied);
384        }
385    }
386    let ceiling = guard
387        .as_ref()
388        .map_or(crate::task::cap::CapSet::EMPTY, |s| s.cell_rights(caller));
389    Ok(crate::loader::SpawnRequest::governed_caller(
390        caller,
391        (task.cell_id.0, task.cell_generation),
392        ceiling,
393        profile.child_ceiling,
394        abi::TaskPriority::Normal as u8,
395        None,
396        names_child(task.root_tid, route),
397    ))
398}

400/// Bind the dispatch boundary while the receive snapshot owns the scheduler
401/// lock, before an independent caller can cancel and remove the queued record.
402pub(super) fn snapshot_resume_dispatched(
403    sched: &mut crate::sched::scheduler::Scheduler,
404    caller_id: usize,
405    mask: usize,
406) -> Result<ResumeSnapshot, ()> {
407    let snap = match sched.tasks.get(&caller_id) {
408        Some(task) => snapshot_resume(task, mask)?,
409        None => ResumeSnapshot::Wake { sender_tid: 0 },
410    };
411    if let ResumeSnapshot::Message {
412        wire_header: Some(header),
413        ..
414    } = &snap
415    {
416        crate::task::async_ipc::dispatch(sched, caller_id, *header);
417    }
418    Ok(snap)
419}

421/// Take a blocked receive's resume snapshot, or park the receiver again when it
422/// woke with nothing to take.
423///
424/// Messages reach a receiver only through its mailbox, and deaths only through
425/// the death fields, so a `Wake` snapshot means the receiver was made runnable
426/// for a message that is gone: a submitted request that was cancelled before the
427/// receiver ran, or a post that `mask` does not admit. Completing the receive
428/// would hand userspace a sender tid over its previous buffer contents. Instead,
429/// under the same lock that found the mailbox empty, the receiver parks again
430/// with the same mask, buffer and deadline, so the next matching publish wakes
431/// it as usual.
432///
433/// A receive whose `deadline` has passed reports the timeout as sender 0.
434///
435/// Returns `Ok(None)` when the receiver parked again: the caller yields and
436/// retries. `Err(())` is the snapshot's allocation failure.
437pub(in crate::task) fn resume_or_repark(
438    sched: &mut crate::sched::scheduler::Scheduler,
439    caller_id: usize,
440    mask: usize,
441    buf_ptr: usize,
442    buf_len: usize,
443    deadline: Option<u64>,
444) -> Result<Option<ResumeSnapshot>, ()> {
445    let snap = snapshot_resume_dispatched(sched, caller_id, mask)?;
446    if !matches!(snap, ResumeSnapshot::Wake { .. }) {
447        return Ok(Some(snap));
448    }
449    if deadline.is_some_and(|deadline| crate::task::system_ticks() as u64 >= deadline) {
450        return Ok(Some(ResumeSnapshot::Wake { sender_tid: 0 }));
451    }
452    let Some(task) = sched.tasks.get_mut(&caller_id) else {
453        return Ok(Some(snap));
454    };
455    crate::task::arm_ipc_block_handoff(caller_id);
456    task.state = crate::task::tcb::TaskState::Blocked {
457        wait: crate::task::tcb::WaitKind::Recv {
458            mask,
459            buf_ptr,
460            buf_len,
461        },
462        deadline,
463    };
464    Ok(None)
465}

483/// Handler for `Syscall::CellSessionV1` (extracted from the dispatcher).
484pub(super) fn cell_session_v1(
485    caller_id: usize,
486    request_ptr: usize,
487    request_len: usize,
488    result_ptr: usize,
489    result_capacity: usize,
490) -> SyscallResult {
491    const MAX: usize = abi::cell_session::MAX_REQUEST_LEN;
492    if !(abi::cell_session::HEADER_LEN..=MAX).contains(&request_len)
493        || result_capacity < abi::cell_session::HEADER_LEN
494    {
495        return Err(SyscallError::InvalidInput);
496    }
497    let request_end = request_ptr
498        .checked_add(request_len)
499        .ok_or(SyscallError::InvalidInput)?;
500    let result_end = result_ptr
501        .checked_add(result_capacity)
502        .ok_or(SyscallError::InvalidInput)?;
503    if request_ptr < result_end && result_ptr < request_end {
504        return Err(SyscallError::InvalidInput);
505    }
506    validate_user_buf(request_ptr, request_len, MAX)?;
507    validate_user_buf(result_ptr, result_capacity, MAX_USER_BUF)?;
508    preflight_user_output(caller_id, result_ptr, result_capacity)?;
509    let view = caller_copy_view(caller_id)?;
510    let mut request = [0u8; MAX];
511    let _reply_pin = SessionOutputPin::reserve(caller_id, result_ptr, result_capacity, &view)?;
512    view.read_into(request_ptr, &mut request[..request_len])
513        .map_err(|_| SyscallError::InvalidInput)?;
514    let header = abi::cell_session::RequestHeaderV1::decode(&request[..request_len]);
515    let mut result = [0u8; 1536];
516    if let Ok(header) = header {
517        let required = crate::ipc::cell_session::required_result(
518            header.selector,
519            &request[abi::cell_session::HEADER_LEN..request_len],
520        );
521        if result_capacity < required {
522            abi::cell_session::ResultHeaderV1 {
523                selector: header.selector,
524                status: abi::cell_session::Status::BufferTooSmall,
525                total_len: required as u32,
526                request_id: header.request_id,
527            }
528            .encode(&mut result)
529            .map_err(|_| SyscallError::InvalidInput)?;
530            view.write_bytes(result_ptr, &result[..abi::cell_session::HEADER_LEN])
531                .map_err(|_| SyscallError::InvalidInput)?;
532            return Ok(0);
533        }
534    }
535    let len = crate::ipc::cell_session::dispatch(caller_id, &request[..request_len], &mut result);
536    view.write_bytes(result_ptr, &result[..len])
537        .map_err(|_| SyscallError::InvalidInput)?;
538    // Only a reply that reached the caller may be acknowledged.
539    crate::ipc::cell_session::delivered(caller_id, &request[..request_len]);
540    Ok(0)
541}

543/// Handler for `Syscall::Spawn` (extracted from the dispatcher).
544pub(super) fn spawn(caller_id: usize, entry: usize, arg: usize) -> SyscallResult {
545    // Build and fully inherit the child under one scheduler lock before
546    // publication. A runnable thread must never observe Task defaults
547    // (SAS, permissive allowlist, or zero PKU state) between construction
548    // and its creator's authority transfer.
549    match crate::task::spawn_with_arg(caller_id, "thread", entry, arg) {
550        Ok(tid) => Ok(tid),
551        Err(ViError::OutOfMemory) => Err(SyscallError::TryAgain),
552        Err(_) => Err(SyscallError::Unknown),
553    }
554}

556/// Handler for `Syscall::SetTlsBase` (extracted from the dispatcher).
557pub(super) fn set_tls_base(caller_id: usize, base: usize) -> SyscallResult {
558    // Self-only by construction: the ABI has no target task parameter, so
559    // this can only ever change the caller's own thread pointer, and the
560    // base carries no authority — the kernel never dereferences it. A bad
561    // value is contained by the cell's own tier (LBI on Tier 1, an MMU
562    // fault on Tier 2).
563    //
564    // The previous value is returned so a cell can read its base back
565    // without touching the register, which the kernel owns (a direct user
566    // write is overwritten on the next resume).
567    if let Some(task) = crate::task::SCHEDULER
568        .lock()
569        .as_mut()
570        .and_then(|sched| sched.tasks.get_mut(&caller_id))
571    {
572        let previous = task.tls_base;
573        crate::task::tls::set_base(task, base);
574        Ok(previous)
575    } else {
576        Err(SyscallError::PermissionDenied)
577    }
578}

580/// Handler for `Syscall::Exit` (extracted from the dispatcher).
581pub(super) fn exit(caller_id: usize, code: usize) -> SyscallResult {
582    // TID zero is the explicit kernel-context sentinel. It has no Cell
583    // generation to retire; a kernel scheduling point remains explicit
584    // rather than fabricating a user-exit record.
585    if caller_id == 0 {
586        crate::task::yield_cpu();
587        return Ok(0);
588    }
589
590    #[cfg(feature = "bench-counters")]
591    crate::bench_counters::note_exit(caller_id);
592
593    // Do not inspect the task table, log, audit, or call
594    // `Scheduler::exit_task` while the exiting Cell is still charged.
595    // A root exit can retire every member and grow scheduler vectors;
596    // the fixed record is therefore the entire victim-attributed path.
597    let exit = crate::sched::hart_local::DeferredExit {
598        tid: caller_id,
599        cell_id: crate::sched::hart_local::current_cell_id(),
600        generation: crate::sched::hart_local::current_cell_generation(),
601        code,
602    };
603    crate::sched::hart_local::defer_exit(exit);
604    crate::sched::hart_local::set_current_cell_id(0);
605    // The identity write above is the trap-attribution handoff; this scope is
606    // the charging one: everything after the record allocates as the kernel.
607    let _kernel =
608        crate::memory::heap::ChargeScope::enter(crate::memory::cell_quota::Charge::KERNEL);
609
610    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
611    crate::arch::task::retirement_selftest::observe_exit_deferred_record_commit(exit);
612
613    // `yield_cpu` consumes the fixed record under normal scheduler
614    // locking and routes roots through quiescent retirement. A worker
615    // still reaches the task-local branch of `exit_task`.
616    crate::task::yield_cpu();
617    Ok(0)
618}

620/// Handler for `Syscall::ForceExit` (extracted from the dispatcher).
621pub(super) fn force_exit(caller_id: usize, tid: usize) -> SyscallResult {
622    // Self-kill rejected before touching the scheduler (cheap early check).
623    if tid == caller_id {
624        return Err(SyscallError::InvalidCommand);
625    }
626
627    // Single SCHEDULER lock: SpawnCap gate + all cleanup in one scope.
628    // Two separate acquisitions would create a TOCTOU window where the target
629    // self-exits between them, causing a spurious InvalidCommand return.
630    if let Some(sched) = crate::task::SCHEDULER.lock().as_mut() {
631        // Gate 1: only SpawnCap holders (init/shell) may force-terminate tasks.
632        let has_spawn = sched.tasks.contains_key(&caller_id) && sched.cell_rights(caller_id).spawn;
633        if !has_spawn {
634            return Err(SyscallError::PermissionDenied);
635        }
636
637        // Gate 2: protect system service cells (VFS=block_io_cap, net=network_cap).
638        // Killing them mid-I/O leaves driver state inconsistent; use hot-swap instead.
639        let target_rights = sched.cell_rights(tid);
640        let target_is_system = target_rights.block_io || target_rights.network;
641        if target_is_system {
642            return Err(SyscallError::PermissionDenied);
643        }
644
645        // Gate 3: protect Frozen cells — they are mid-swap and cannot be killed
646        // by external actors; only the supervisor-owned cutover path may terminate
647        // them via the internal exit_task path after a successful swap.
648        let target_is_frozen = sched
649            .tasks
650            .get(&tid)
651            .map(|t| matches!(t.state, TaskState::Frozen { .. }))
652            .unwrap_or(false);
653        if target_is_frozen {
654            return Err(SyscallError::PermissionDenied);
655        }
656
657        let task = match sched.tasks.get_mut(&tid) {
658            Some(t) => t,
659            // Target self-exited between the lock boundary — already dead; mission done.
660            None => return Ok(0),
661        };
662        task.exit_code = Some(usize::MAX); // sentinel: force-killed
663
664        // exit_task: zombie move + ready-queue purge + stuck-sender unblock,
665        // and wakes sys_wait(tid) waiters with the force-kill sentinel.
666        sched.exit_task(tid, usize::MAX);
667    } else {
668        return Err(SyscallError::InvalidCommand);
669    }
670
671    crate::audit::log_event(
672        crate::audit::AuditEvent::CellExit,
673        &crate::audit::encode_u32x2(tid as u32, 0xFFFF_FFFFu32), // force-kill marker
674    );
675
676    log::info!(
677        "[kernel] ForceExit: task {} killed by task {}",
678        tid,
679        caller_id
680    );
681
682    Ok(0) // non-blocking — caller keeps running; do NOT yield_cpu
683}

685/// Handler for `Syscall::NotifyOnExit` (extracted from the dispatcher).
686pub(super) fn notify_on_exit(caller_id: usize, watched: usize) -> SyscallResult {
687    // Privileged: only SpawnCap holders (supervisors like init) may watch
688    // arbitrary tasks — same authority gate as ForceExit. The watcher's
689    // next Recv returns `watched` when it dies (see exit_task delivery).
690    //
691    // Race: the watched task may have already exited before this call.
692    // Subscribe while SCHEDULER still proves the target is live; exit_task
693    // takes DEATH_SUBSCRIBERS under the same outer lock, so publication and
694    // delivery are atomic with respect to task death.
695    {
696        let mut sched_opt = crate::task::SCHEDULER.lock();
697        let sched = match sched_opt.as_mut() {
698            Some(s) => s,
699            None => return Ok(0),
700        };
701        let has_spawn = sched.tasks.contains_key(&caller_id) && sched.cell_rights(caller_id).spawn;
702        if !has_spawn {
703            return Err(SyscallError::PermissionDenied);
704        }
705        if sched
706            .tasks
707            .get(&watched)
708            .is_some_and(|task| !matches!(task.state, TaskState::Retiring | TaskState::Terminated))
709        {
710            crate::sched::scheduler::subscribe_death(watched, caller_id);
711        } else {
712            // Task already dead — queue synthetic death so watcher never stalls.
713            if let Some(wt) = sched.tasks.get_mut(&caller_id) {
714                crate::sched::scheduler::queue_pending_death(wt, watched, 0);
715            }
716        }
717    }
718    Ok(0)
719}

721/// Handler for `Syscall::QueryRetirement` (extracted from the dispatcher).
722pub(super) fn query_retirement(
723    caller_id: usize,
724    root_tid: usize,
725    out_ptr: usize,
726    out_len: usize,
727) -> SyscallResult {
728    // Admitted past the allowlist; the verdict table is the gate: a
729    // death subscriber of that root at its retirement, or a supervisor.
730    if out_len != abi::retirement::RETIREMENT_RECORD_LEN {
731        return Err(SyscallError::InvalidInput);
732    }
733    let supervisor = caller_has_supervisor(caller_id);
734    let record = crate::cell::retirement_verdict::query(caller_id, supervisor, root_tid).map_err(
735        |error| match error {
736            crate::cell::retirement_verdict::QueryError::NotFound => SyscallError::FileNotFound,
737            crate::cell::retirement_verdict::QueryError::Denied => SyscallError::PermissionDenied,
738        },
739    )?;
740    write_user_slice(
741        caller_id,
742        out_ptr,
743        &record.encode(),
744        abi::retirement::RETIREMENT_RECORD_LEN,
745    )?;
746    Ok(abi::retirement::RETIREMENT_RECORD_LEN)
747}

757    caller_id: usize,
758    path_ptr: usize,
759    path_len: usize,

819                    "[loader] spawn OOM: op=SpawnFromPath caller={} path={}",
820                    caller_id,
821                    path_str
822                );
823                SyscallError::OutOfMemory
824            }
825            _ => SyscallError::InvalidInput,
826        }

1086/// Handler for `Syscall::TelemetrySnapshot` (extracted from the dispatcher).
1087pub(super) fn telemetry_snapshot(
1088    caller_id: usize,
1089    out_ptr: usize,
1090    out_len: usize,
1091    flags: usize,
1092    reserved: usize,
1093) -> SyscallResult {
1094    use abi::telemetry::request;
1095    // The dispatch gate already required `observe_tasks` (X). Arguments
1096    // are refused before anything is sampled.
1097    if !crate::task::telemetry::request_is_valid(out_ptr, out_len, flags, reserved) {
1098        return Err(SyscallError::InvalidInput);
1099    }
1100    validate_user_buf(out_ptr, out_len, MAX_USER_BUF)?;
1101    let memory_allowed = flags & request::SYSTEM_MEMORY != 0
1102        && caller_has_cap(caller_id, |rights| rights.observe_memory);
1103    let epoch = crate::task::telemetry::provider_epoch();
1104    // Identity, membership and CPU in one SCHEDULER hold. The transient
1105    // rows are allocated there, charged to the caller like GetProcs2.
1106    let capture = {
1107        let mut guard = crate::task::SCHEDULER.lock();
1108        let sched = guard.as_mut().ok_or(SyscallError::Unknown)?;
1109        crate::task::telemetry::capture(
1110            sched,
1111            epoch,
1112            crate::arch::clock::monotonic_ns(),
1113            out_len,
1114            flags & request::TASKS != 0,
1115        )
1116        .map_err(|_| SyscallError::OutOfMemory)?
1117    };
1118    // Frames are read under FRAME_ALLOCATOR only after SCHEDULER is released.
1119    let memory = if flags & request::SYSTEM_MEMORY == 0 {
1120        crate::task::telemetry::SystemMemory::NotRequested
1121    } else if !memory_allowed {
1122        crate::task::telemetry::SystemMemory::Denied
1123    } else {
1124        match crate::memory::frame::stats() {
1125            Some(frames) => {
1126                crate::task::telemetry::SystemMemory::Present(abi::syscall::ViMemInfoV1 {
1127                    total_frames: frames.total_frames as u64,
1128                    used_frames: frames.used_frames as u64,
1129                    free_frames: frames.free_frames as u64,
1130                    page_size: crate::memory::paging::PAGE_SIZE as u64,
1131                })
1132            }
1133            None => crate::task::telemetry::SystemMemory::Unavailable,
1134        }
1135    };
1136    let bytes = capture.finish(memory, crate::arch::clock::monotonic_ns());
1137    write_user_slice(caller_id, out_ptr, &bytes, MAX_USER_BUF)?;
1138    Ok(bytes.len())
1139}

1168    let bytes = read_user_slice(
1169        caller_id,
1170        buf_ptr,
1171        buf_len,
1172        crate::cell::state_stash::MAX_STASH_LEN,
1173    )?;
1174    let stored = crate::cell::state_stash::stash(owner, raw_key, &bytes);
1175    #[cfg(feature = "bench-counters")]
1176    if stored > 0 {
1177        crate::bench_counters::milestone("state-stashed", caller_id);
1178    }
1179    Ok(stored)
1180}

1182/// Handler for `Syscall::StateRestore` (extracted from the dispatcher).
1183pub(super) fn state_restore(
1184    caller_id: usize,
1185    key: usize,
1186    buf_ptr: usize,
1187    buf_len: usize,
1188) -> SyscallResult {
1189    if key as u64 == SPAWN_ARGV_KEY {
1190        validate_user_buf(
1191            buf_ptr,
1192            buf_len,
1193            crate::cell::state_stash::MAX_SPAWN_ARGV_LEN,
1194        )?;
1195        let Some(argv) = crate::cell::state_stash::take_inherited_argv(caller_id) else {
1196            return Ok(0);
1197        };
1198        let n = argv.len().min(buf_len);
1199        if n > 0 {
1200            write_user_slice(
1201                caller_id,
1202                buf_ptr,
1203                &argv[..n],
1204                crate::cell::state_stash::MAX_SPAWN_ARGV_LEN,
1205            )?;
1206        }
1207        return Ok(n);
1208    }
1209    let reader = stash_incarnation(caller_id)?;
1210    // Empty buffer: length probe only (owner, destination or supervisor).
1211    if buf_len == 0 {
1212        let supervisor = caller_has_supervisor(caller_id);
1213        return Ok(crate::cell::state_stash::status(
1214            reader, supervisor, key as u64,
1215        ));
1216    }
1217    validate_user_buf(buf_ptr, buf_len, crate::cell::state_stash::MAX_STASH_LEN)?;
1218    let mut kbuf = alloc::vec::Vec::new();
1219    kbuf.try_reserve_exact(buf_len)
1220        .map_err(|_| SyscallError::OutOfMemory)?;
1221    kbuf.resize(buf_len, 0);
1222    let n = crate::cell::state_stash::restore(reader, key as u64, &mut kbuf);
1223    if n > 0 {
1224        write_user_slice(
1225            caller_id,
1226            buf_ptr,
1227            &kbuf[..n],
1228            crate::cell::state_stash::MAX_STASH_LEN,
1229        )?;
1230        #[cfg(feature = "bench-counters")]
1231        crate::bench_counters::milestone("state-restored", caller_id);
1232    }
1233    Ok(n)
1234}

1236/// Handler for `Syscall::StateStashClear` (extracted from the dispatcher).
1237pub(super) fn state_stash_clear(caller_id: usize, key: usize) -> SyscallResult {
1238    if key as u64 == SPAWN_ARGV_KEY {
1239        crate::cell::state_stash::discard_spawn_argv(caller_id);
1240    } else {
1241        let caller = stash_incarnation(caller_id)?;
1242        let supervisor = caller_has_supervisor(caller_id);
1243        if !crate::cell::state_stash::clear(caller, supervisor, key as u64) {
1244            return Err(SyscallError::PermissionDenied);
1245        }
1246    }

1250/// Handler for `Syscall::QueryHotswapReady` (extracted from the dispatcher).
1251pub(super) fn query_hotswap_ready(caller_id: usize, target_tid: usize) -> SyscallResult {
1252    if !caller_has_supervisor(caller_id) {
1253        return Err(SyscallError::PermissionDenied);
1254    }
1255    let ready = crate::task::SCHEDULER
1256        .lock()
1257        .as_ref()
1258        .and_then(|s| s.tasks.get(&target_tid).map(|t| t.hotswap.ready));
1259    match ready {
1260        Some(true) => Ok(1),
1261        Some(false) => Ok(0),
1262        None => Err(SyscallError::FileNotFound), // tid does not exist
1263    }
1264}

1266/// Handler for `Syscall::SpawnReplacement` (extracted from the dispatcher).
1267pub(super) fn spawn_replacement(
1268    caller_id: usize,
1269    old_tid: usize,
1270    path_ptr: usize,
1271    path_len: usize,
1272) -> SyscallResult {
1273    if !caller_has_supervisor(caller_id) {
1274        return Err(SyscallError::PermissionDenied);
1275    }
1276    if old_tid == 0 || path_len == 0 || path_len > crate::loader::disk_layout::MAX_CELL_PATH {
1277        return Err(SyscallError::InvalidInput);
1278    }
1279    // The machine-pin provider is never swapped (`swap_refused`).
1280    if crate::cell::service_registry::is_swap_refused_tid(old_tid) {
1281        return Err(SyscallError::PermissionDenied);
1282    }
1283    let path_str = read_user_string(
1284        caller_id,
1285        path_ptr,
1286        path_len,
1287        crate::loader::disk_layout::MAX_CELL_PATH,
1288    )?;
1289    if !path_str.starts_with('/') {
1290        return Err(SyscallError::InvalidInput);
1291    }
1292    let profile = authorize_launch_edge(
1293        caller_id,
1294        crate::loader::launch_profile::LaunchRoute::Path,
1295        &path_str,
1296    )?;
1297    let replacement = crate::cell::hotswap::reserve_frozen_replacement(old_tid)
1298        .ok_or(SyscallError::PermissionDenied)?;
1299    let old_incarnation = stash_incarnation(old_tid)?;
1300    let request = match governed_replacement_request(caller_id, profile.child_ceiling, replacement)
1301    {
1302        Ok(request) => request,
1303        Err(error) => {
1304            crate::task::dir_inherit::clear_staged(caller_id);
1305            return Err(error);
1306        }
1307    };
1308    let spawned = crate::loader::spawn_from_path(&path_str, request);
1309    // Mirror the regular spawn contract: a successful replacement spawn
1310    // consumes any staged directory set, and a failed one must clear it
1311    // so it cannot attach to an unrelated later child.
1312    crate::task::dir_inherit::clear_staged(caller_id);
1313    let task_id = spawned.map_err(|e| match e {
1314        types::ViError::NotFound => SyscallError::FileNotFound,
1315        types::ViError::OutOfMemory => {
1316            log::warn!(
1317                "[loader] spawn OOM: op=SpawnReplacement caller={} old_tid={} path={}",
1318                caller_id,
1319                old_tid,
1320                path_str
1321            );
1322            SyscallError::OutOfMemory
1323        }
1324        types::ViError::PermissionDenied => SyscallError::PermissionDenied,
1325        _ => SyscallError::InvalidInput,
1326    })?;
1327    // Hand the frozen incarnation's stash entries to exactly this
1328    // replacement: the old writer is fenced, only the new one may read.
1329    if let Ok(new_incarnation) = stash_incarnation(task_id) {
1330        crate::cell::state_stash::seal_for_replacement(old_incarnation, new_incarnation);
1331        #[cfg(feature = "bench-counters")]
1332        crate::bench_counters::milestone("replacement-sealed", task_id);
1333    }
1334    Ok(task_id)
1335}

1338pub(super) fn hot_swap_ready(caller_id: usize) -> SyscallResult {
1339    // The new cell signals that it has finished deserializing state.

1349/// Handler for `Syscall::Snapshot` (extracted from the dispatcher).
1350pub(super) fn snapshot(caller_id: usize) -> SyscallResult {
1351    if !caller_has_supervisor(caller_id) {
1352        log::warn!("[snapshot] denied: caller {caller_id} has no SupervisorCap");
1353        return Err(SyscallError::PermissionDenied);
1354    }
1355    // Cells must be quiesced before calling this (all at yield points).
1356    // For MVP: the shell is the only active task while the snapshot runs.
1357    match crate::snapshot::serialize_snapshot() {
1358        Ok(frame_count) => Ok(frame_count as usize),
1359        Err(reason) => {
1360            log::warn!("[snapshot] unavailable: {reason}");
1361            Err(SyscallError::Unknown)
1362        }
1363    }
1364}
