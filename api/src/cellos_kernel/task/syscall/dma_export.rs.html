1//! Client-memory DMA export leases: `ExportDmaRange = 437`, `MapDmaExport =
2//! 438`, `RevokeDmaExport = 439` (ADR-0033, CP1 `cp1-gpu-kernel-abi.md` §5-§6).
3//!
4//! A client Cell (the *owner*) exports a page-aligned range of memory it owns to
5//! the live bound provider of a service in [`DMA_EXPORT_SERVICES`] (the
6//! *holder*). The holder maps the export into its own VT-d domain at a
7//! kernel-allocated IOVA window in `[2^38, 2^39)` and programs that IOVA into its
8//! device. The kernel alone revokes: it clears the window's leaves and drains the
9//! domain IOTLB, and only an acknowledged drain releases the owner's range. It
10//! never waits on the holder (D-G5).
11//!
12//! # State machine
13//!
14//! ```text
15//! 437 ──► Live{unmapped} ──438──► Live{mapped} ──439/owner death──► Revoking
16//!            │                       │                                  │
17//!            └──439/owner death──────┴──holder domain revoked──► dropped ◄─ ack
18//! ```
19//!
20//! `Revoking` keeps the owner range locked (and, after owner death, its frames in
21//! quarantine under `ReleaseKey::DmaExport`) until an invalidation is
22//! acknowledged; [`retry_pending`] repeats it on every scheduler pass.
23//!
24//! # Locks
25//!
26//! `EXPORTS` is taken first, never under another lock: `IOMMU_CLEANUP_SERIAL →
27//! EXPORTS` (holder hook) and `EXPORTS → grant table → SCHEDULER → pin
28//! REGISTRY` / `EXPORTS → VT-d UNITS`. Every lease transition happens under it,
29//! so 437/438/439, owner death and holder death are serialized.
30

42/// Lowest IOVA of the export window region (D-G3): `2^38`.
43pub(crate) const WINDOW_BASE: u64 = 1 << 38;

45pub(crate) const WINDOW_END: u64 = 1 << 39;

52/// Test-only holder service id, admitted only in `test-hooks` images: the

81/// Outcome of publishing a window in the holder's domain.
82#[derive(Clone, Copy, Debug, PartialEq, Eq)]
83pub(crate) enum MapOutcome {
84    /// Leaves written and the invalidation acknowledged.
85    Mapped,
86    /// Nothing was published (no unit, DIDs/tables exhausted, window occupied).
87    Rejected,
88    /// Leaves are hardware-visible but the invalidation timed out. Only the
89    /// VT-d backend produces it; other targets never publish a window.
90    #[cfg_attr(not(any(test, target_arch = "x86_64")), allow(dead_code))]

94/// Everything the lease state machine needs from the rest of the kernel.
95///
96/// The production implementation is [`KernelPlatform`]; host tests drive the
97/// same state machine with a fake whose remapper acknowledgement can fail.
98pub(crate) trait Platform {
99    /// `(cell, generation)` of a live (not retiring) task.
100    fn binding(&self, tid: usize) -> Option<Incarnation>;
101    /// Whether any live task belongs to `who`.
102    fn incarnation_live(&self, who: Incarnation) -> bool;
103    /// A translating remapper is active and every RAM frame lies below
104    /// [`WINDOW_BASE`], so no window can alias an identity leaf.
105    fn exports_enabled(&self) -> bool;
106    /// The live bound provider of `service`.
107    fn provider(&self, service: u16) -> Option<Provider>;
108    /// Atomically prove `[base, base + len)` is memory `tid` owns and may write
109    /// (an owned SAS Grant always is) and lock it under `export_id`. No effect
110    /// on refusal.
111    fn lock_owned_range(&mut self, req: &RangeLock) -> Result<(), SyscallError>;
112    /// Drop the range lock of `export_id` and free the frames nothing else
113    /// protects; returns pages freed.
114    fn unlock_range(&mut self, export_id: u64) -> usize;
115    /// `tid` holds `PcieDriverCap`.
116    fn has_pcie_driver(&self, tid: usize) -> bool;
117    /// `tid` owns `bdf` (418/424).
118    fn owns_bdf(&self, tid: usize, bdf: u32) -> bool;
119    /// The remapping unit covering `bdf` translates.
120    fn translating(&self, bdf: u32) -> bool;
121    /// Charge `bytes` to `cell`'s DMA quota.
122    fn reserve_quota(&mut self, cell: u64, bytes: usize) -> bool;
123    /// Return `bytes` to `cell`'s DMA quota.
124    fn release_quota(&mut self, cell: u64, bytes: usize);
125    /// Publish `[iova, iova + len) → [phys, phys + len)` in `holder_tid`'s domain for `bdf`.
126    fn map_window(&mut self, w: &WindowMap) -> MapOutcome;
127    /// Clear the window's leaves and drain the domain IOTLB; `true` once acknowledged.
128    fn unmap_window(&mut self, holder_tid: usize, iova: u64, len: usize) -> bool;
129    /// Enable bus mastering for `bdf`.
130    fn enable_bus_master(&mut self, bdf: u32);
131}

184/// Free IOVA ranges of one holder task's window region, `start → len`.
185#[derive(Debug)]
186struct Windows {
187    free: BTreeMap<u64, u64>,
188}
189
190impl Windows {
191    fn new() -> Self {
192        let mut free = BTreeMap::new();
193        free.insert(WINDOW_BASE, WINDOW_END - WINDOW_BASE);
194        Self { free }
195    }
196
197    /// First fit; 2 MiB aligned for windows of at least 2 MiB so the device
198    /// side may use large pages where frames allow.
199    fn allocate(&mut self, len: u64) -> Option<u64> {
200        let align = if len >= LARGE_PAGE {
201            LARGE_PAGE
202        } else {
203            PAGE as u64
204        };
205        let (start, span, base) = self.free.iter().find_map(|(&start, &span)| {
206            let base = start.checked_add(align - 1)? & !(align - 1);
207            (base.checked_add(len)? <= start + span).then_some((start, span, base))
208        })?;
209        self.free.remove(&start);
210        if base > start {
211            self.free.insert(start, base - start);
212        }
213        let end = base + len;
214        if end < start + span {
215            self.free.insert(end, start + span - end);
216        }
217        Some(base)
218    }
219
220    /// Return `[base, base + len)` and coalesce with its neighbours.
221    fn release(&mut self, base: u64, len: u64) {
222        let mut start = base;
223        let mut end = base + len;
224        if let Some((&prev, &span)) = self.free.range(..base).next_back() {
225            if prev + span == base {
226                self.free.remove(&prev);
227                start = prev;
228            }
229        }
230        if let Some(span) = self.free.remove(&end) {
231            end += span;
232        }
233        self.free.insert(start, end - start);
234    }
235}

237/// The lease table: every live or revoking export, plus per-holder windows.
238pub(crate) struct ExportTable {
239    leases: BTreeMap<u64, Lease>,
240    windows: BTreeMap<usize, Windows>,
241}
242
243static EXPORTS: Spinlock<ExportTable> =
244    Spinlock::ranked(ExportTable::new(), crate::sync::rank::DMA_EXPORTS);
245/// Next export id; boot-wide, never reused. An id taken by an export whose
246/// range proof then fails is skipped, never reissued.
247static NEXT_EXPORT_ID: AtomicU64 = AtomicU64::new(1);
248/// Leases in `Revoking`, so the scheduler pass skips `EXPORTS` when zero.
249static REVOKING: AtomicUsize = AtomicUsize::new(0);
250/// Live leases of any state, so death hooks skip `EXPORTS` when zero.
251static LEASES: AtomicUsize = AtomicUsize::new(0);
252
253/// Upper bound on the SLPT pages one window needs (leaf tables plus their
254/// parents), charged to the holder's DMA quota (CP1 §5.2).
255fn table_bytes(len: usize) -> usize {
256    let leaf_tables = len.div_ceil(1 << 21) + 1;
257    let parents = len.div_ceil(1 << 30) + 1;
258    (leaf_tables + parents) * PAGE
259}
260
261fn rights(flags: u16) -> &'static str {
262    if flags & EXPORT_DEVICE_WRITE != 0 {
263        "RW"
264    } else {
265        "R"
266    }
267}

269impl ExportTable {
270    pub(crate) const fn new() -> Self {
271        Self {
272            leases: BTreeMap::new(),
273            windows: BTreeMap::new(),
274        }
275    }
276
277    fn set_state(&mut self, id: u64, state: LeaseState) {
278        if let Some(lease) = self.leases.get_mut(&id) {
279            if lease.state != state {
280                if state == LeaseState::Revoking {
281                    REVOKING.fetch_add(1, Ordering::Relaxed);
282                } else {
283                    REVOKING.fetch_sub(1, Ordering::Relaxed);
284                }
285                lease.state = state;
286            }
287        }
288    }
289
290    /// Release a mapping's window and quota after its leaves are gone.
291    fn retire_mapping(
292        &mut self,
293        platform: &mut impl Platform,
294        holder: Incarnation,
295        m: Mapping,
296        len: usize,
297    ) {
298        if let Some(windows) = self.windows.get_mut(&m.holder_tid) {
299            windows.release(m.iova, len as u64);
300        }
301        platform.release_quota(holder.cell, m.quota_bytes);
302    }
303
304    /// Remove lease `id` entirely: its mapping is gone (or never existed).
305    fn drop_lease(&mut self, platform: &mut impl Platform, id: u64, why: &str) {
306        let Some(mut lease) = self.leases.remove(&id) else {
307            return;
308        };
309        LEASES.fetch_sub(1, Ordering::Relaxed);
310        if lease.state == LeaseState::Revoking {
311            REVOKING.fetch_sub(1, Ordering::Relaxed);
312        }
313        if let Some(m) = lease.mapping.take() {
314            self.retire_mapping(platform, lease.holder, m, lease.len);
315        }
316        let freed = platform.unlock_range(id);
317        log::warn!(
318            "[export] drop id={id} ({why}): owner range {:#x}+{} unlocked, {freed} quarantined page(s) freed",
319            lease.base,
320            lease.len
321        );
322    }
323
324    /// `Live → Revoking`, then clear and drain the window. Drops the lease on
325    /// acknowledgement; otherwise leaves it `Revoking` for [`retry_pending`].
326    fn revoke(
327        &mut self,
328        platform: &mut impl Platform,
329        id: u64,
330        why: &str,
331    ) -> Result<(), SyscallError> {
332        self.set_state(id, LeaseState::Revoking);
333        let Some(lease) = self.leases.get(&id) else {
334            return Err(SyscallError::FileNotFound);
335        };
336        if let Some(m) = &lease.mapping {
337            if !platform.unmap_window(m.holder_tid, m.iova, lease.len) {
338                log::warn!(
339                    "[export] revoke id={id} ({why}): iova={:#x} invalidation unacknowledged; lease Revoking, range locked",
340                    m.iova
341                );
342                return Err(SyscallError::TryAgain);
343            }
344            log::warn!(
345                "[export] revoke id={id} ({why}): iova={:#x} len={} cleared, IOTLB drain acknowledged",
346                m.iova,
347                lease.len
348            );
349        }
350        self.drop_lease(platform, id, why);
351        Ok(())
352    }

354    /// 437: export `[va, va + len)` of `caller` to the provider of `service`.
355    pub(crate) fn export(
356        &mut self,
357        platform: &mut impl Platform,
358        caller: usize,
359        va: usize,
360        len: usize,
361        service: usize,
362        flags: usize,
363    ) -> Result<u64, SyscallError> {
364        if !va.is_multiple_of(PAGE)
365            || len == 0
366            || !len.is_multiple_of(PAGE)
367            || len as u64 > EXPORT_MAX_BYTES
368            || va.checked_add(len).is_none()
369            || service > u16::MAX as usize
370            || flags & !(EXPORT_FLAGS_KNOWN as usize) != 0
371        {
372            return Err(SyscallError::InvalidInput);
373        }
374        let service = service as u16;
375        let flags = flags as u16;
376        if !holder_service_allowed(service) {
377            log::warn!("[export] task {caller} refused: service {service:#x} cannot hold exports");
378            return Err(SyscallError::PermissionDenied);
379        }
380        if !platform.exports_enabled() {
381            log::warn!("[export] task {caller} refused: no translating remapper (exports never use identity DMA)");
382            return Err(SyscallError::NotSupported);
383        }
384        let owner = platform
385            .binding(caller)
386            .ok_or(SyscallError::PermissionDenied)?;
387        let provider = platform
388            .provider(service)
389            .ok_or(SyscallError::FileNotFound)?;
390        if !provider.pcie_driver || provider.who == owner {
391            log::warn!(
392                "[export] task {caller} refused: holder of service {service:#x} is not eligible"
393            );
394            return Err(SyscallError::PermissionDenied);
395        }
396        if self.leases.len() >= MAX_LEASES
397            || self
398                .leases
399                .values()
400                .filter(|l| l.owner_tid == caller)
401                .count()
402                >= MAX_LEASES_PER_OWNER
403        {
404            return Err(SyscallError::OutOfMemory);
405        }
406        let id = NEXT_EXPORT_ID.fetch_add(1, Ordering::Relaxed);
407        platform.lock_owned_range(&RangeLock {
408            tid: caller,
409            owner,
410            base: va,
411            len,
412            holder_cell: provider.who.cell,
413            export_id: id,
414        })?;
415        self.leases.insert(
416            id,
417            Lease {
418                owner_tid: caller,
419                owner,
420                holder: provider.who,
421                base: va,
422                len,
423                flags,
424                state: LeaseState::Live,
425                mapping: None,
426            },
427        );
428        LEASES.fetch_add(1, Ordering::Relaxed);
429        log::warn!(
430            "[export] id={id} owner tid={caller} cell={}/{} holder cell={}/{} va={va:#x} len={len} rights={}",
431            owner.cell,
432            owner.generation,
433            provider.who.cell,
434            provider.who.generation,
435            rights(flags)
436        );
437        Ok(id)
438    }

440    /// 438: map export `id` for `bdf` into `caller`'s domain; the record to copy out.
441    pub(crate) fn map(
442        &mut self,
443        platform: &mut impl Platform,
444        caller: usize,
445        id: u64,
446        bdf: usize,
447    ) -> Result<ViDmaExportV1, SyscallError> {
448        if bdf > 0xFFFF {
449            return Err(SyscallError::InvalidInput);
450        }
451        let bdf = bdf as u32;
452        if !platform.has_pcie_driver(caller) || !platform.owns_bdf(caller, bdf) {
453            return Err(SyscallError::PermissionDenied);
454        }
455        let holder = platform
456            .binding(caller)
457            .ok_or(SyscallError::PermissionDenied)?;
458        let lease = self.leases.get(&id).ok_or(SyscallError::FileNotFound)?;
459        if lease.holder != holder {
460            log::warn!("[export] map id={id} refused: task {caller} is not the lease holder");
461            return Err(SyscallError::PermissionDenied);
462        }
463        if lease.state != LeaseState::Live {
464            return Err(SyscallError::FileNotFound);
465        }
466        if !platform.translating(bdf) {
467            return Err(SyscallError::NotSupported);
468        }
469        let writable = lease.flags & EXPORT_DEVICE_WRITE != 0;
470        let record = |iova| {
471            ViDmaExportV1::new(
472                lease.flags,
473                id,
474                lease.owner.cell,
475                lease.owner.generation,
476                iova,
477                lease.len as u64,
478            )
479        };
480        if let Some(m) = &lease.mapping {
481            if m.holder_tid != caller {
482                return Err(SyscallError::PermissionDenied);
483            }
484            if !m.confirmed {
485                return Err(SyscallError::Unknown);
486            }
487            let (iova, len, base, record) = (m.iova, lease.len, lease.base, record(m.iova));
488            if m.bdfs.contains(&bdf) {
489                return Ok(record);
490            }
491            // Another function of the same holder: attach it to the same window.
492            let outcome = platform.map_window(&WindowMap {
493                holder_tid: caller,
494                bdf,
495                iova,
496                phys: base as u64,
497                len,
498                writable,
499            });
500            let lease = self.leases.get_mut(&id).ok_or(SyscallError::FileNotFound)?;
501            let m = lease.mapping.as_mut().ok_or(SyscallError::FileNotFound)?;
502            return match outcome {
503                MapOutcome::Mapped => {
504                    m.bdfs.push(bdf);
505                    platform.enable_bus_master(bdf);
506                    Ok(record)
507                }
508                MapOutcome::Rejected => Err(SyscallError::OutOfMemory),
509                MapOutcome::Unconfirmed => {
510                    m.confirmed = false;
511                    Err(SyscallError::Unknown)
512                }
513            };
514        }
515
516        let (len, base, flags) = (lease.len, lease.base, lease.flags);
517        let quota_bytes = table_bytes(len);
518        let iova = self
519            .windows
520            .entry(caller)
521            .or_insert_with(Windows::new)
522            .allocate(len as u64)
523            .ok_or(SyscallError::OutOfMemory)?;
524        if !platform.reserve_quota(holder.cell, quota_bytes) {
525            if let Some(w) = self.windows.get_mut(&caller) {
526                w.release(iova, len as u64);
527            }
528            return Err(SyscallError::OutOfMemory);
529        }
530        let outcome = platform.map_window(&WindowMap {
531            holder_tid: caller,
532            bdf,
533            iova,
534            phys: base as u64,
535            len,
536            writable,
537        });
538        if outcome == MapOutcome::Rejected {
539            platform.release_quota(holder.cell, quota_bytes);
540            if let Some(w) = self.windows.get_mut(&caller) {
541                w.release(iova, len as u64);
542            }
543            log::warn!("[export] map id={id} refused: remapper rejected the window (tables, DIDs or occupied)");
544            return Err(SyscallError::OutOfMemory);
545        }
546        let confirmed = outcome == MapOutcome::Mapped;
547        let lease = self.leases.get_mut(&id).ok_or(SyscallError::FileNotFound)?;
548        lease.mapping = Some(Mapping {
549            holder_tid: caller,
550            iova,
551            quota_bytes,
552            confirmed,
553            bdfs: alloc::vec![bdf],
554        });
555        if !confirmed {
556            // Hardware-visible but unacknowledged: like GrantDma, keep the
557            // mapping (only revoke/death cleans it) and do not enable BME.
558            log::warn!("[export] map id={id} iova={iova:#x} unconfirmed: invalidation timed out");
559            return Err(SyscallError::Unknown);
560        }
561        platform.enable_bus_master(bdf);
562        log::warn!(
563            "[export] map id={id} holder tid={caller} bdf={:02x}:{:02x}.{} iova={iova:#x} len={len} rights={}",
564            (bdf >> 8) & 0xFF,
565            (bdf >> 3) & 0x1F,
566            bdf & 7,
567            rights(flags)
568        );
569        Ok(ViDmaExportV1::new(
570            flags,
571            id,
572            lease.owner.cell,
573            lease.owner.generation,
574            iova,
575            len as u64,
576        ))
577    }

579    /// 439: revoke export `id`; only its owner or holder may ask.
580    pub(crate) fn revoke_by(
581        &mut self,
582        platform: &mut impl Platform,
583        caller: usize,
584        id: u64,
585    ) -> Result<(), SyscallError> {
586        let who = platform.binding(caller).ok_or(SyscallError::FileNotFound)?;
587        let lease = self.leases.get(&id).ok_or(SyscallError::FileNotFound)?;
588        let why = if lease.owner == who {
589            "owner"
590        } else if lease.holder == who {
591            "holder"
592        } else {
593            // No existence oracle for third parties.
594            return Err(SyscallError::FileNotFound);
595        };
596        self.revoke(platform, id, why)
597    }

599    /// Owner death (first step of retirement): revoke every lease `tid` owns.
600    pub(crate) fn owner_died(&mut self, platform: &mut impl Platform, tid: usize) {
601        let ids: Vec<u64> = self
602            .leases
603            .iter()
604            .filter(|(_, l)| l.owner_tid == tid)
605            .map(|(&id, _)| id)
606            .collect();
607        for id in ids {
608            let _ = self.revoke(platform, id, "owner death");
609        }
610    }

612    /// The holder task's whole domain was revoked and acknowledged
613    /// (`teardown_dma_authority`): its windows no longer exist.
614    pub(crate) fn holder_domain_revoked(&mut self, platform: &mut impl Platform, tid: usize) {
615        let ids: Vec<u64> = self
616            .leases
617            .iter()
618            .filter(|(_, l)| l.mapping.as_ref().is_some_and(|m| m.holder_tid == tid))
619            .map(|(&id, _)| id)
620            .collect();
621        for id in ids {
622            let Some(lease) = self.leases.get_mut(&id) else {
623                continue;
624            };
625            let (holder, len) = (lease.holder, lease.len);
626            if let Some(m) = lease.mapping.take() {
627                platform.release_quota(holder.cell, m.quota_bytes);
628            }
629            let revoking = lease.state == LeaseState::Revoking;
630            if revoking || !platform.incarnation_live(holder) {
631                self.drop_lease(platform, id, "holder domain revoked");
632            } else {
633                log::warn!("[export] id={id}: holder tid {tid} lost its domain; lease stays Live unmapped ({len} bytes)");
634            }
635        }
636        // The domain and every window in it are gone; a new domain starts empty.
637        self.windows.remove(&tid);
638        self.drop_orphans(platform);
639    }
640
641    /// Drop unmapped leases whose holder incarnation no longer has a live task.
642    fn drop_orphans(&mut self, platform: &mut impl Platform) {
643        let ids: Vec<u64> = self
644            .leases
645            .iter()
646            .filter(|(_, l)| l.mapping.is_none() && !platform.incarnation_live(l.holder))
647            .map(|(&id, _)| id)
648            .collect();
649        for id in ids {
650            self.drop_lease(platform, id, "holder incarnation gone");
651        }
652    }
653
654    /// Scheduler pass: retry every `Revoking` lease's invalidation.
655    pub(crate) fn retry(&mut self, platform: &mut impl Platform) {
656        let ids: Vec<u64> = self
657            .leases
658            .iter()
659            .filter(|(_, l)| l.state == LeaseState::Revoking)
660            .map(|(&id, _)| id)
661            .collect();
662        for id in ids {
663            let _ = self.revoke(platform, id, "deferred retry");
664        }
665    }
666
667    #[cfg(test)]
668    fn state(&self, id: u64) -> Option<(LeaseState, Option<u64>)> {
669        self.leases
670            .get(&id)
671            .map(|l| (l.state, l.mapping.as_ref().map(|m| m.iova)))
672    }
673}

676pub(crate) fn export(
677    caller: usize,
678    va: usize,
679    len: usize,
680    service: usize,
681    flags: usize,
682) -> Result<usize, SyscallError> {
683    let mut platform = KernelPlatform;

685        .lock()
686        .export(&mut platform, caller, va, len, service, flags)
687        .map(|id| id as usize)
688}

691pub(crate) fn map(caller: usize, id: usize, bdf: usize) -> Result<ViDmaExportV1, SyscallError> {
692    let mut platform = KernelPlatform;
693    EXPORTS.lock().map(&mut platform, caller, id as u64, bdf)
694}

696/// 439 entry.
697pub(crate) fn revoke(caller: usize, id: usize) -> Result<usize, SyscallError> {
698    let mut platform = KernelPlatform;
699    EXPORTS
700        .lock()
701        .revoke_by(&mut platform, caller, id as u64)
702        .map(|()| 0)
703}

705/// First retirement step for `tid` (CP1 §6): revoke every lease it owns before
706/// any of its frames can be reclaimed.
707pub(crate) fn reap_owner(tid: usize) {
708    if LEASES.load(Ordering::Relaxed) == 0 {
709        return;
710    }
711    EXPORTS.lock().owner_died(&mut KernelPlatform, tid);
712}

714/// Holder hook: `tid`'s VT-d domain was torn down and acknowledged.
715///
716/// Always takes `EXPORTS`: the leases a dying holder affects were created by
717/// other tasks, so no program order makes a lock-free zero check exact here.
718pub(crate) fn holder_domain_revoked(tid: usize) {
719    EXPORTS
720        .lock()
721        .holder_domain_revoked(&mut KernelPlatform, tid);
722}

724/// Scheduler pass: retry unacknowledged revokes (CP1 §5.3 deferred retry).
725pub(crate) fn retry_pending() {
726    if REVOKING.load(Ordering::Relaxed) == 0 {
727        return;
728    }
729    EXPORTS.lock().retry(&mut KernelPlatform);
730}

732/// Prove `req.base..+len` lies inside one live Grant `req.tid` owns, under the
733/// owner's live binding, and lock it — all inside the grant-table transaction,
734/// so neither `GrantFree`/`GrantUnregister` nor the reaper can remove the Grant
735/// between the proof and the lock (same shape as `GrantDma`'s reservation).
736///
737/// SAS Grants are identity-mapped (VA == PA), physically contiguous, mapped
738/// USER RW for their owner, and never MMIO, framebuffer, ELF or received
739/// memory, so "inside a caller-owned Grant" is CP1 §5.1's whole ownership
740/// gate. A range under a live VFS lease is refused. A private-root caller is
741/// refused until Tier-2 anonymous memory exists (D-G0).
742fn lock_exported_range(req: &RangeLock) -> Result<(), SyscallError> {
743    use super::super::tcb::TaskState;
744    if super::domain_grant_task(req.tid) {
745        log::warn!(
746            "[export] task {} refused: private-root exports need Tier-2 anonymous memory (D-G0)",
747            req.tid
748        );
749        return Err(SyscallError::PermissionDenied);
750    }
751    let binding = (types::CellId(req.owner.cell), req.owner.generation);
752    let live_binding = || {
753        SCHEDULER
754            .lock()
755            .as_ref()
756            .and_then(|sched| sched.tasks.get(&req.tid))
757            .filter(|task| !matches!(task.state, TaskState::Retiring | TaskState::Terminated))
758            .map(|task| (task.cell_id, task.cell_generation))
759    };
760    let lock = || {
761        if crate::memory::pin::vfs_holder_of_owner(req.base, req.len, req.tid).is_some() {
762            log::warn!(
763                "[export] task {} refused: range is under a live VFS lease",
764                req.tid
765            );
766            return Err(SyscallError::PermissionDenied);
767        }
768        crate::memory::pin::lock_dma_export(
769            req.base,
770            req.len,
771            req.tid,
772            req.holder_cell,
773            req.export_id,
774        )
775        .map_err(|_| SyscallError::OutOfMemory)
776    };
777    {
778        let grants = super::grant_table_lock().lock();
779        if live_binding() != Some(binding) {
780            return Err(SyscallError::PermissionDenied);
781        }
782        let owned = grants.as_ref().is_some_and(|grants| {
783            grants
784                .values()
785                .any(|g| super::page_grant_authorizes_dma(g, req.tid, binding, req.base, req.len))
786        });
787        if owned {
788            return lock();
789        }
790    }
791    {
792        let grants = super::reg_grant_table_lock().lock();
793        if live_binding() != Some(binding) {
794            return Err(SyscallError::PermissionDenied);
795        }
796        let owned = grants.as_ref().is_some_and(|grants| {
797            grants
798                .values()
799                .any(|g| super::reg_grant_authorizes_dma(g, req.tid, binding, req.base, req.len))
800        });
801        if owned {
802            return lock();
803        }
804    }
805    log::warn!(
806        "[export] task {} refused: {:#x}+{} is not inside a live Grant it owns",
807        req.tid,
808        req.base,
809        req.len
810    );
811    Err(SyscallError::PermissionDenied)
812}

814/// The production [`Platform`]: scheduler, grant tables, pin registry, VT-d.
815pub(crate) struct KernelPlatform;
816
817impl Platform for KernelPlatform {
818    fn binding(&self, tid: usize) -> Option<Incarnation> {
819        super::live_task_binding(tid).map(|(cell, generation)| Incarnation {
820            cell: cell.0,
821            generation,
822        })
823    }
824
825    fn incarnation_live(&self, who: Incarnation) -> bool {
826        use super::super::tcb::TaskState;
827        SCHEDULER.lock().as_ref().is_some_and(|sched| {
828            sched.tasks.values().any(|t| {
829                t.cell_id.0 == who.cell
830                    && t.cell_generation == who.generation
831                    && !matches!(t.state, TaskState::Retiring | TaskState::Terminated)
832            })
833        })
834    }
835
836    fn exports_enabled(&self) -> bool {
837        #[cfg(target_arch = "x86_64")]
838        {
839            if !crate::dev::iommu::is_active() {
840                return false;
841            }
842            let top = crate::memory::frame::FRAME_ALLOCATOR
843                .lock()
844                .as_ref()
845                .map_or(usize::MAX, |a| a.memory_end());
846            if top as u64 > WINDOW_BASE {
847                log::warn!("[export] RAM reaches {top:#x} >= 2^38: export windows could alias identity leaves");
848                return false;
849            }
850            true
851        }
852        #[cfg(not(target_arch = "x86_64"))]
853        false
854    }
855
856    fn provider(&self, service: u16) -> Option<Provider> {
857        let (tid, cell, generation) = crate::cell::service_registry::lookup_bound(service)?;
858        Some(Provider {
859            tid,
860            who: Incarnation { cell, generation },
861            pcie_driver: super::caller_has_pcie_driver(tid),
862        })
863    }
864
865    fn lock_owned_range(&mut self, req: &RangeLock) -> Result<(), SyscallError> {
866        lock_exported_range(req)
867    }
868
869    fn unlock_range(&mut self, export_id: u64) -> usize {
870        let mut freed = 0;
871        for range in crate::memory::pin::unlock_dma_export(export_id) {
872            freed += range.pages;
873            super::free_released_range(range);
874        }
875        freed
876    }
877
878    fn has_pcie_driver(&self, tid: usize) -> bool {
879        super::caller_has_pcie_driver(tid)
880    }
881
882    fn owns_bdf(&self, tid: usize, bdf: u32) -> bool {
883        crate::resource_registry::owner_of_bdf(bdf) == Some(tid)
884    }
885
886    fn translating(&self, bdf: u32) -> bool {
887        #[cfg(target_arch = "x86_64")]
888        {
889            crate::dev::iommu_x86::bdf_translating(bdf)
890        }
891        #[cfg(not(target_arch = "x86_64"))]
892        {
893            let _ = bdf;
894            false
895        }
896    }
897
898    fn reserve_quota(&mut self, cell: u64, bytes: usize) -> bool {
899        match crate::memory::cell_quota::try_reserve_dma(cell as usize, bytes) {
900            Some(reservation) => {
901                reservation.commit();
902                true
903            }
904            None => false,
905        }
906    }
907
908    fn release_quota(&mut self, cell: u64, bytes: usize) {
909        crate::memory::cell_quota::record_dma_unmapped(cell as usize, bytes);
910    }
911
912    fn map_window(&mut self, w: &WindowMap) -> MapOutcome {
913        #[cfg(target_arch = "x86_64")]
914        {
915            use crate::dev::iommu::DmaMapResult;
916            match crate::dev::iommu_x86::map_window_for_cell(
917                w.holder_tid as u64,
918                w.bdf,
919                w.iova,
920                w.phys,
921                w.len,
922                w.writable,
923            ) {
924                DmaMapResult::Mapped(_) => MapOutcome::Mapped,
925                DmaMapResult::PublishedUnconfirmed => MapOutcome::Unconfirmed,
926                DmaMapResult::Rejected => MapOutcome::Rejected,
927            }
928        }
929        #[cfg(not(target_arch = "x86_64"))]
930        {
931            let _ = w;
932            MapOutcome::Rejected
933        }
934    }
935
936    fn unmap_window(&mut self, holder_tid: usize, iova: u64, len: usize) -> bool {
937        #[cfg(target_arch = "x86_64")]
938        {
939            crate::dev::iommu_x86::unmap_window_for_cell(holder_tid as u64, iova, len)
940        }
941        #[cfg(not(target_arch = "x86_64"))]
942        {
943            let _ = (holder_tid, iova, len);
944            true
945        }
946    }
947
948    fn enable_bus_master(&mut self, bdf: u32) {
949        crate::dev::pcie_ecam::enable_bus_master(bdf);
950    }
951}
