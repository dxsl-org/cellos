9pub(super) fn caller_has_hypervisor(caller_id: usize) -> bool {
10    caller_has_cap(caller_id, |t| t.hypervisor)
11}

13/// Staging granule for guest-memory copies. Cell bytes cross the copy view
14/// into a kernel buffer of at most this size; the registry only ever sees that
15/// buffer, never a cell address.
16pub(super) const GUEST_COPY_CHUNK: usize = 256 * 1024;
17
18/// A zeroed kernel staging buffer for a `len`-byte guest-memory copy.
19pub(super) fn guest_copy_staging(len: usize) -> Result<Vec<u8>, SyscallError> {
20    let size = len.min(GUEST_COPY_CHUNK);
21    let mut staging = Vec::new();
22    staging
23        .try_reserve_exact(size)
24        .map_err(|_| SyscallError::OutOfMemory)?;
25    staging.resize(size, 0);
26    Ok(staging)
27}

29/// Refuse a guest-memory request unless `[gpa, gpa + len)` lies wholly inside
30/// the VM's guest-RAM window, so no chunk is copied for a request that would
31/// fail part-way on bounds.
32pub(super) fn check_guest_range(
33    caller_id: usize,
34    vm_id: usize,
35    gpa: u64,
36    len: usize,
37) -> Result<(), SyscallError> {
38    let (base, bytes) = crate::hypervisor::registry::guest_window(caller_id, vm_id)
39        .map_err(|_| SyscallError::InvalidInput)?;
40    let end = gpa
41        .checked_sub(base)
42        .and_then(|offset| offset.checked_add(len as u64))
43        .ok_or(SyscallError::InvalidInput)?;
44    if end > bytes as u64 {
45        return Err(SyscallError::InvalidInput);
46    }
47    Ok(())
48}

50/// Handler for `Syscall::CreateVm` (extracted from the dispatcher).
51pub(super) fn create_vm(caller_id: usize, guest_pages: usize) -> SyscallResult {
52    if !caller_has_hypervisor(caller_id) {
53        return Err(SyscallError::PermissionDenied);
54    }
55    // An empty carve disables the nested table's SAS check, and the
56    // window's byte size must be representable for every later bound.
57    if guest_pages == 0
58        || guest_pages
59            .checked_mul(crate::memory::paging::PAGE_SIZE)
60            .is_none()
61    {
62        return Err(SyscallError::InvalidInput);
63    }
64    crate::hypervisor::registry::create_vm(caller_id, guest_pages)
65        .map_err(|_| SyscallError::NotSupported)
66}

77/// Handler for `Syscall::MapGuestMemory` (extracted from the dispatcher).
78pub(super) fn map_guest_memory(
79    caller_id: usize,
80    vm_id: usize,
81    ipa: u64,
82    size: usize,
83    writable: bool,
84) -> SyscallResult {
85    if !caller_has_hypervisor(caller_id) {
86        return Err(SyscallError::PermissionDenied);
87    }
88    // C3: overflow guard on IPA + size.
89    ipa.checked_add(size as u64)
90        .ok_or(SyscallError::InvalidInput)?;
91    // C-2: the range is backed from the start of the guest carve, so
92    // it may never be larger than the carve (the table re-checks).
93    let (_, guest_bytes) = crate::hypervisor::registry::guest_window(caller_id, vm_id)
94        .map_err(|_| SyscallError::NotSupported)?;
95    if size == 0 || size > guest_bytes {
96        return Err(SyscallError::InvalidInput);
97    }
98    crate::hypervisor::registry::map_guest_memory(caller_id, vm_id, ipa, size, writable)
99        .map(|_| 0usize)
100        .map_err(|_| SyscallError::NotSupported)
101}

103/// Handler for `Syscall::RunVcpu` (extracted from the dispatcher).
104pub(super) fn run_vcpu(
105    caller_id: usize,
106    vm_id: usize,
107    vcpu_id: usize,
108    budget_ns: u64,
109    out_ptr: usize,
110) -> SyscallResult {
111    if !caller_has_hypervisor(caller_id) {
112        return Err(SyscallError::PermissionDenied);
113    }
114    use crate::hypervisor::registry::vmm_wire;
115    validate_user_buf(out_ptr, vmm_wire::EXIT_LEN, MAX_USER_BUF)?;
116    let view = caller_copy_view(caller_id)?;
117    // Prove the record is writable before the guest runs: an exit
118    // that cannot be delivered would be lost with the guest advanced.
119    view.write_bytes(out_ptr, &[0u8; vmm_wire::EXIT_LEN])
120        .map_err(|_| SyscallError::InvalidInput)?;
121    let exit = crate::hypervisor::registry::run_vcpu(caller_id, vm_id, vcpu_id, budget_ns)
122        .map_err(|_| SyscallError::NotSupported)?;
123    view.write_bytes(out_ptr, &vmm_wire::encode_exit(&exit))
124        .map_err(|_| SyscallError::InvalidInput)?;
125    Ok(0)
126}

128/// Handler for `Syscall::VcpuRegs` (extracted from the dispatcher).
129pub(super) fn vcpu_regs(
130    caller_id: usize,
131    vm_id: usize,
132    vcpu_id: usize,
133    buf_ptr: usize,
134    mode: usize,
135) -> SyscallResult {
136    if !caller_has_hypervisor(caller_id) {
137        return Err(SyscallError::PermissionDenied);
138    }
139    // All modes access the full 32-word (256-byte) user buffer.
140    // Mode 2 is Pi-only, read-only, and clears unused words.
141    #[cfg(target_arch = "aarch64")]
142    let max_mode =
143        <hal::aarch64::vcpu::VcpuExt as hal::aarch64::backend::ArmVcpuExt>::REGS_MAX_MODE;
144    #[cfg(not(target_arch = "aarch64"))]
145    let max_mode = 1;
146    if mode > max_mode {
147        return Err(SyscallError::InvalidInput);
148    }
149    use crate::hypervisor::registry::vmm_wire;
150    validate_user_buf(buf_ptr, vmm_wire::REGS_LEN, MAX_USER_BUF)?;
151    let view = caller_copy_view(caller_id)?;
152    // Read the buffer in every mode: mode 1 consumes it, and the read
153    // modes leave words the backend does not model as the cell had them.
154    let mut bytes = [0u8; vmm_wire::REGS_LEN];
155    view.read_into(buf_ptr, &mut bytes)
156        .map_err(|_| SyscallError::InvalidInput)?;
157    let mut regs = vmm_wire::regs_from_bytes(&bytes);
158    let result = crate::hypervisor::registry::vcpu_regs(caller_id, vm_id, vcpu_id, &mut regs, mode)
159        .map_err(|_| SyscallError::NotSupported)?;
160    if mode != 1 {
161        view.write_bytes(buf_ptr, &vmm_wire::regs_to_bytes(&regs))
162            .map_err(|_| SyscallError::InvalidInput)?;
163    }
164    Ok(result)
165}

167/// Handler for `Syscall::InjectIrq` (extracted from the dispatcher).
168pub(super) fn inject_irq(
169    caller_id: usize,
170    vm_id: usize,
171    vcpu_id: usize,
172    intid: u32,
173) -> SyscallResult {
174    if !caller_has_hypervisor(caller_id) {
175        return Err(SyscallError::PermissionDenied);
176    }
177    // m3: GICv2 has SPIs/PPIs/SGIs up to intid 1019.
178    if intid > 1019 {
179        return Err(SyscallError::InvalidInput);
180    }
181    crate::hypervisor::registry::inject_irq(caller_id, vm_id, vcpu_id, intid)
182        .map_err(|_| SyscallError::NotSupported)
183}

185/// Handler for `Syscall::WriteGuestMemory` (extracted from the dispatcher).
186pub(super) fn write_guest_memory(
187    caller_id: usize,
188    vm_id: usize,
189    gpa: u64,
190    src_ptr: usize,
191    len: usize,
192) -> SyscallResult {
193    if !caller_has_hypervisor(caller_id) {
194        return Err(SyscallError::PermissionDenied);
195    }
196    validate_user_buf(src_ptr, len, MAX_USER_BUF)?;
197    check_guest_range(caller_id, vm_id, gpa, len)?;
198    let view = caller_copy_view(caller_id)?;
199    let mut staging = guest_copy_staging(len)?;
200    let mut done = 0;
201    while done < len {
202        let chunk = &mut staging[..(len - done).min(GUEST_COPY_CHUNK)];
203        view.read_into(src_ptr + done, chunk)
204            .map_err(|_| SyscallError::InvalidInput)?;
205        crate::hypervisor::registry::write_guest_memory(caller_id, vm_id, gpa + done as u64, chunk)
206            .map_err(|_| SyscallError::InvalidInput)?;
207        done += chunk.len();
208    }
209    Ok(len)
210}

212/// Handler for `Syscall::ReadGuestMemory` (extracted from the dispatcher).
213pub(super) fn read_guest_memory(
214    caller_id: usize,
215    vm_id: usize,
216    gpa: u64,
217    dst_ptr: usize,
218    len: usize,
219) -> SyscallResult {
220    if !caller_has_hypervisor(caller_id) {
221        return Err(SyscallError::PermissionDenied);
222    }
223    validate_user_buf(dst_ptr, len, MAX_USER_BUF)?;
224    check_guest_range(caller_id, vm_id, gpa, len)?;
225    let view = caller_copy_view(caller_id)?;
226    let mut staging = guest_copy_staging(len)?;
227    let mut done = 0;
228    while done < len {
229        let chunk = &mut staging[..(len - done).min(GUEST_COPY_CHUNK)];
230        crate::hypervisor::registry::read_guest_memory(caller_id, vm_id, gpa + done as u64, chunk)
231            .map_err(|_| SyscallError::InvalidInput)?;
232        view.write_bytes(dst_ptr + done, chunk)
233            .map_err(|_| SyscallError::InvalidInput)?;
234        done += chunk.len();
235    }
236    Ok(len)
