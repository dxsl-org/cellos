9/// 434/435 results: the typed sentinels of [`encode_ipc_result`] plus
10/// `NotSupported` and `OutOfMemory` (vectors exhausted), which a driver must
11/// tell apart to fall back to polling (`api::pcie_function::ERR_*`).
12#[cfg(not(target_arch = "riscv32"))]
13pub(super) fn encode_pcie_function_result(result: SyscallResult) -> usize {
14    match result {
15        Err(SyscallError::NotSupported) => api::pcie_function::ERR_NOT_SUPPORTED,
16        Err(SyscallError::OutOfMemory) => api::pcie_function::ERR_NO_VECTORS,
17        other => encode_ipc_result(other),
18    }
19}

29/// 436 results: each refusal keeps its own `api::syscall::system_power`
30/// sentinel. Success never returns to the caller.
31#[cfg(not(target_arch = "riscv32"))]
32pub(super) fn encode_system_power_result(result: SyscallResult) -> usize {
33    use api::syscall::system_power as sp;
34    match result {
35        Ok(value) => value,
36        Err(SyscallError::InvalidInput) => sp::ERR_INVALID_INPUT,
37        Err(SyscallError::NotSupported) => sp::ERR_NOT_SUPPORTED,
38        Err(_) => sp::ERR_PERMISSION_DENIED,
39    }
40}

42/// `WaitIrq` admission for its `mmio_base` argument.
43///
44/// Zero means "no InterruptACK write". A nonzero base is stored and later
45/// dereferenced by the riscv64/aarch64 ISR, so it must be the exact base of an
46/// MMIO window the resource registry granted to the caller.
47pub(super) fn wait_irq_base_permitted(caller_id: usize, mmio_base: usize) -> bool {
48    mmio_base == 0 || crate::resource_registry::lookup_mmio_owner(mmio_base) == Some(caller_id)
49}

59/// Gate for raw opcode 502. Runs before any platform shutdown code.
60pub(super) fn authorize_raw_shutdown(caller_id: usize) -> Result<(), SyscallError> {
61    let (name, _, _) =
62        caller_launch_state(caller_id).unwrap_or((String::from("<unknown>"), false, false));
63    if raw_shutdown_permitted(&name) {
64        return Ok(());
65    }
66    log::warn!("[kernel] raw opcode 502 denied for {}", name);
67    Err(SyscallError::PermissionDenied)
68}

99/// 423: authorize and publish the USB HID event producer role.
100///
101/// Deliberately separate from [`register_driver_service`]: that helper accepts
102/// either `PcieDriverCap` or `UsbDriverCap` and publishes the singleton NIC
103/// role. A USB host cell on a PC that already has e1000/virtio-net must hold
104/// only `UsbDriverCap` and must never overwrite `service::NIC_DRIVER`, so this
105/// path checks `usb_driver_cap` alone and publishes `service::USB_HID_PRODUCER`.
106pub(super) fn register_usb_hid_producer_role(caller_id: usize) -> Result<usize, SyscallError> {
107    let scheduler = crate::task::SCHEDULER.lock();
108    let authorized = scheduler
109        .as_ref()
110        .and_then(|scheduler| {
111            let rights = scheduler.cell_rights(caller_id);
112            Some((scheduler.tasks.get(&caller_id)?, rights))
113        })
114        .is_some_and(|(task, rights)| {
115            rights.usb_driver
116                && !matches!(
117                    &task.state,
118                    crate::task::tcb::TaskState::Retiring | crate::task::tcb::TaskState::Terminated
119                )
120        });
121    if !authorized {
122        return Err(SyscallError::PermissionDenied);
123    }
124
125    let Some((cell_id, generation)) = scheduler
126        .as_ref()
127        .and_then(|s| s.tasks.get(&caller_id))
128        .map(|task| (task.cell_id.0, task.cell_generation))
129    else {
130        return Err(SyscallError::InvalidInput);
131    };
132
133    let published = crate::dev::driver_cell::publish_role_or_rollback(
134        caller_id,
135        crate::dev::driver_cell::register_usb_hid_producer,
136        || {
137            crate::cell::service_registry::register(
138                api::syscall::service::USB_HID_PRODUCER,
139                caller_id,
140                cell_id,
141                generation,
142            )
143        },
144        crate::dev::driver_cell::deregister_usb_hid_producer,
145    );
146    drop(scheduler);
147    if published {
148        Ok(0)
149    } else {
150        Err(SyscallError::InvalidInput)
151    }
152}

199/// 436 after the `SupervisorCap` and reserved-argument gates. Reboot does not
200/// return; poweroff returns only when the machine is still running.
201#[cfg(target_arch = "x86_64")]
202pub(super) fn system_power(caller_id: usize, action: usize) -> SyscallResult {
203    use api::syscall::system_power as sp;
204    match action {
205        sp::ACTION_REBOOT => {
206            log::warn!(
207                "[kernel] SystemPower: reboot requested by tid {}",
208                caller_id
209            );
210            crate::acpi::power::reboot()
211        }
212        sp::ACTION_POWEROFF => {
213            log::warn!(
214                "[kernel] SystemPower: poweroff requested by tid {}",
215                caller_id
216            );
217            crate::acpi::power::poweroff();
218            Err(SyscallError::NotSupported)
219        }
220        _ => Err(SyscallError::InvalidInput),
221    }
222}

234pub(super) fn register_driver_service(
235    caller_id: usize,
236    service_id: u16,
237    publish_role: fn(usize),
238    rollback_role: fn(usize),
239) -> Result<usize, SyscallError> {
240    let scheduler = crate::task::SCHEDULER.lock();
241    let authorized = scheduler
242        .as_ref()
243        .and_then(|scheduler| {
244            let rights = scheduler.cell_rights(caller_id);
245            Some((scheduler.tasks.get(&caller_id)?, rights))
246        })
247        .is_some_and(|(task, rights)| {
248            (rights.pcie_driver || rights.usb_driver)
249                && !matches!(
250                    &task.state,
251                    crate::task::tcb::TaskState::Retiring | crate::task::tcb::TaskState::Terminated
252                )
253        });
254    if !authorized {
255        return Err(SyscallError::PermissionDenied);
256    }
257
258    // The identity is read under the same scheduler lock that authorized the
259    // caller, so the published binding names an incarnation the kernel can stand
260    // behind. A task with no live Cell identity is refused rather than recorded
261    // as an unbindable route.
262    let Some((cell_id, generation)) = scheduler
263        .as_ref()
264        .and_then(|s| s.tasks.get(&caller_id))
265        .map(|task| (task.cell_id.0, task.cell_generation))
266    else {
267        return Err(SyscallError::InvalidInput);
268    };
269
270    let published = crate::dev::driver_cell::publish_role_or_rollback(
271        caller_id,
272        publish_role,
273        || crate::cell::service_registry::register(service_id, caller_id, cell_id, generation),
274        rollback_role,
275    );
276    drop(scheduler);
277    if published {
278        Ok(0)
279    } else {
280        Err(SyscallError::InvalidInput)
281    }
282}

284/// 416: bind the route of the block device the caller owns.
285///
286/// Checks in order, each before any effect: decode (`Malformed` →
287/// `InvalidInput`), a live PcieDriverCap/UsbDriverCap caller (`PermissionDenied`),
288/// ownership of the named device (`NotOwner` → `FileNotFound`), then the route
289/// slot under `ROLE_PUBLICATION` (a refusal is `PermissionDenied`).
290///
291/// Ownership is read without SCHEDULER held: `request_mmio` takes the MMIO
292/// registry before SCHEDULER, so the opposite order here could deadlock. Both
293/// owner tables key the claim by the claiming task id (`RequestMmio` records
294/// `CellId(tid)`), and a claim is released only at that task's retirement, so
295/// it still holds when the caller is found live again under SCHEDULER, where
296/// its `(cell, generation)` is read and the slot is published so exit teardown
297/// cannot interleave. No service-registry entry is written.
298pub(super) fn register_block_device_route(
299    caller_id: usize,
300    kind: usize,
301    id: usize,
302    reserved: (usize, usize),
303) -> Result<usize, SyscallError> {
304    use api::block_route::{DeviceKey, BLOCK_DEVICE_MMIO, BLOCK_DEVICE_PCI};
305    let key = u16::try_from(kind)
306        .ok()
307        .filter(|_| reserved == (0, 0))
308        .and_then(|kind| DeviceKey::new(kind, id as u64))
309        .ok_or(SyscallError::InvalidInput)?;
310
311    let live_driver = |scheduler: &Option<crate::task::Scheduler>| {
312        scheduler.as_ref().and_then(|scheduler| {
313            let rights = scheduler.cell_rights(caller_id);
314            let task = scheduler.tasks.get(&caller_id)?;
315            ((rights.pcie_driver || rights.usb_driver)
316                && !matches!(&task.state, TaskState::Retiring | TaskState::Terminated))
317            .then_some((task.cell_id.0, task.cell_generation))
318        })
319    };
320    if live_driver(&crate::task::SCHEDULER.lock()).is_none() {
321        return Err(SyscallError::PermissionDenied);
322    }
323
324    let owned = match key.kind {
325        BLOCK_DEVICE_PCI => {
326            crate::resource_registry::owner_of_bdf(key.id as u32) == Some(caller_id)
327        }
328        BLOCK_DEVICE_MMIO => {
329            crate::resource_registry::lookup_mmio_owner(key.id as usize) == Some(caller_id)
330        }
331        _ => false,
332    };
333    if !owned {
334        return Err(SyscallError::FileNotFound);
335    }
336
337    let scheduler = crate::task::SCHEDULER.lock();
338    let Some((cell_id, generation)) = live_driver(&scheduler) else {
339        return Err(SyscallError::PermissionDenied);
340    };
341    let published =
342        crate::dev::driver_cell::publish_block_route(key, (caller_id, cell_id, generation));
343    drop(scheduler);
344    published.map(|_| 0).ok_or(SyscallError::PermissionDenied)
345}

368/// Serialize a matched PCIe device into the 48-byte `PcieDeviceInfo` wire record
369/// (padded from the 44 meaningful bytes by `vendor_id`/`device_id` alignment) and
370/// write it to the caller's `out_ptr`.
371///
372/// Shared by the class-triple (`FindPcieDevice`) and exact-identity
373/// (`FindPcieDeviceByVendor`) queries so the two hand-out paths cannot drift.
374/// Claims BDF ownership first and returns 0 without writing when a live Driver
375/// Cell already owns the device.
376pub(super) fn offer_pcie_device(
377    caller_id: usize,
378    out_ptr: usize,
379    dev: crate::dev::pcie_ecam::PciDevice,
380) -> Result<usize, SyscallError> {
381    let bdf = dev.canonical_bdf();
382    let bar0_base = dev.bars[0].base_addr();
383    let bar0_len: u64 = match dev.bars[0] {
384        crate::dev::pcie_ecam::Bar::Memory32 { size, .. } => size as u64,
385        crate::dev::pcie_ecam::Bar::Memory64 { size, .. } => size,
386        _ => 0x4000, // fallback 16 KiB
387    };
388    // First *memory* BAR with a non-zero base, for devices whose BAR0 is I/O or
389    // empty (ICH9's and VMware's AHCI decode their ABAR at BAR5). Every recorded
390    // function carries its full BAR set: the boot scan decodes bus 0, and
391    // `register_device` decodes a Platform-Cell-registered function (bus ≥ 1,
392    // e.g. behind VMware's `pciBridge0`) the same way. `bar0_*` keep their
393    // existing first-BAR meaning; NVMe/e1000 are unaffected because their BAR0
394    // is memory.
395    let (bar_mem_base, bar_mem_len): (u64, u64) =
396        crate::dev::pcie_ecam::first_memory_bar(&dev.bars)
397            .map(|(base, size)| (base, if size == 0 { 0x4000 } else { size }))
398            .unwrap_or((bar0_base, bar0_len));
399    // Under VT-d, a virtio function without ACCESS_PLATFORM or a PVSCSI
400    // adapter would DMA around the translation; no cell may own it (ADR-0031
401    // phase 05 review HIGH; H-S3-13). Judged only before the first claim: the
402    // virtio probe remaps the common-config page kernel-only, which must not
403    // happen under a live owner's mapping.
404    #[cfg(target_arch = "x86_64")]
405    if crate::resource_registry::owner_of_bdf(bdf).is_none()
406        && crate::dev::pcie_msi::dma_bypasses_iommu(&dev)
407    {
408        return Ok(0);
409    }
410    // Claim without displacing a live Driver Cell. A competing instance observes
411    // "not found" and may retry after reap.
412    if !crate::resource_registry::claim_bdf_owner(bdf, caller_id) {
413        return Ok(0);
414    }
415    // The owner may now reach its BARs: memory decoding on, bus mastering
416    // still off until its first GrantDma, after the cell has reset the device.
417    // Only when firmware placed every memory BAR: an unassigned one would
418    // decode at address 0 (the cell then fails closed on base 0, as before).
419    if crate::dev::pcie_ecam::memory_bars_assigned(&dev.bars) {
420        crate::dev::pcie_ecam::enable_memory_decode(bdf);
421    }
422    if out_ptr != 0 {
423        let mut dev_bytes = [0u8; 48];
424        dev_bytes[0..4].copy_from_slice(&bdf.to_ne_bytes());
425        dev_bytes[4..8].copy_from_slice(&1u32.to_ne_bytes());
426        dev_bytes[8..16].copy_from_slice(&bar0_base.to_ne_bytes());
427        dev_bytes[16..24].copy_from_slice(&bar0_len.to_ne_bytes());
428        dev_bytes[24..32].copy_from_slice(&bar_mem_base.to_ne_bytes());
429        dev_bytes[32..40].copy_from_slice(&bar_mem_len.to_ne_bytes());
430        dev_bytes[40..42].copy_from_slice(&dev.vendor_id.to_ne_bytes());
431        dev_bytes[42..44].copy_from_slice(&dev.device_id.to_ne_bytes());
432        write_user_slice(caller_id, out_ptr, &dev_bytes, MAX_USER_BUF)?;
433    }
434    Ok(1)
435}

437/// Handler for `Syscall::LookupBlockRoute` (extracted from the dispatcher).
438pub(super) fn lookup_block_route(
439    caller_id: usize,
440    index: usize,
441    out_ptr: usize,
442    out_len: usize,
443    reserved: usize,
444) -> SyscallResult {
445    // Admitted past the allowlist; the VFS (BlockIoCap) and init
446    // (SpawnCap) are the route readers.
447    if !(caller_has_block_io(caller_id) || caller_has_spawn(caller_id)) {
448        return Err(SyscallError::PermissionDenied);
449    }
450    let len = api::block_route::BLOCK_ROUTE_RECORD_LEN;
451    if out_len != len || reserved != 0 {
452        return Err(SyscallError::InvalidInput);
453    }
454    let record =
455        crate::dev::driver_cell::block_route_record(index).ok_or(SyscallError::FileNotFound)?;
456    write_user_slice(caller_id, out_ptr, &record.encode(), len)?;
457    Ok(len)
458}

460/// Handler for `Syscall::QueryMachineFacts` (extracted from the dispatcher).
461pub(super) fn query_machine_facts(
462    caller_id: usize,
463    out_ptr: usize,
464    out_len: usize,
465    a2: usize,
466    a3: usize,
467) -> SyscallResult {
468    // Admitted past the allowlist; init (SpawnCap) is the reader.
469    if !caller_has_spawn(caller_id) {
470        return Err(SyscallError::PermissionDenied);
471    }
472    if out_len != api::machine_facts::MACHINE_FACTS_CAPACITY || a2 != 0 || a3 != 0 {
473        return Err(SyscallError::InvalidInput);
474    }
475    let record = crate::machine_facts::query().ok_or(SyscallError::FileNotFound)?;
476    write_user_slice(caller_id, out_ptr, &record, record.len())?;
477    Ok(record.len())
478}

507/// Handler for `Syscall::GpuGetResolution` (extracted from the dispatcher).
508pub(super) fn gpu_get_resolution() -> SyscallResult {
509    // Firmware scanout registration is authoritative when present;
510    // generic VirtIO retains the legacy fallback before its driver
511    // publishes a query IPC path.
512    // Packed as (width << 32) | height, which only fits a 64-bit
513    // register; RV32 Nano has no GPU cell so this path never fires there.
514    #[cfg(not(target_arch = "riscv32"))]
515    {
516        if let Some((width, height)) = crate::resource_registry::display_framebuffer_resolution() {
517            return Ok(((width as usize) << 32) | height as usize);
518        }
519        Ok(((1280usize) << 32) | 800usize)
520    }
521    #[cfg(target_arch = "riscv32")]
522    {
523        Ok(0)
524    }
525}

552/// Handler for `Syscall::FindPcieDevice` (extracted from the dispatcher).

665/// Handler for `Syscall::SerialWrite` (extracted from the dispatcher).
666pub(super) fn serial_write(caller_id: usize, index: u32, ptr: usize, len: usize) -> SyscallResult {
667    if !caller_has_serial_port(caller_id) {
668        return Err(SyscallError::PermissionDenied);
669    }
670    if !serial_port::present(index) {
671        return Ok(usize::MAX);
672    }
673    // Bound one call: a cell cannot hold the console path for an
674    // unbounded burst through a blocking port.
675    const SERIAL_BURST_MAX: usize = 4096;
676    let len = len.min(SERIAL_BURST_MAX);
677    if len == 0 {
678        return Ok(0);
679    }
680    let view = caller_copy_view(caller_id)?;
681    let mut buf = [0u8; SERIAL_BURST_MAX];
682    let buf = &mut buf[..len];
683    view.read_into(ptr, buf)
684        .map_err(|_| SyscallError::InvalidInput)?;
685    let mut written = 0usize;
686    for byte in buf.iter() {
687        if !serial_port::write_byte(index, *byte) {
688            break;
689        }
690        written += 1;
691    }
692    Ok(written)
693}

695/// Handler for `Syscall::SerialRead` (extracted from the dispatcher).
696pub(super) fn serial_read(caller_id: usize, index: u32, ptr: usize, len: usize) -> SyscallResult {
697    if !caller_has_serial_port(caller_id) {
698        return Err(SyscallError::PermissionDenied);
699    }
700    if !serial_port::present(index) {
701        return Ok(usize::MAX);
702    }
703    const SERIAL_BURST_MAX: usize = 4096;
704    let len = len.min(SERIAL_BURST_MAX);
705    if len == 0 {
706        return Ok(0);
707    }
708    let mut buf = [0u8; SERIAL_BURST_MAX];
709    let mut read = 0usize;
710    while read < len {
711        let Some(byte) = serial_port::read_byte(index) else {
712            break;
713        };
714        buf[read] = byte;
715        read += 1;
716    }
717    write_user_slice(caller_id, ptr, &buf[..read], len)?;
718    Ok(read)
719}

736/// Handler for `Syscall::WaitIrq` (extracted from the dispatcher).
737pub(super) fn wait_irq(caller_id: usize, irq: u8, mmio_base: usize) -> SyscallResult {
738    if !caller_has_pcie_driver(caller_id)
739        && !caller_has_platform(caller_id)
740        && !caller_has_usb_driver(caller_id)
741    {
742        return Err(SyscallError::PermissionDenied);
743    }
744    #[cfg(target_arch = "aarch64")]
745    {
746        use hal::aarch64::backend::{ArmIrqController, Board};
747        if let Some(usb_irq) = <Board as ArmIrqController>::USB_ONE_SHOT_IRQ {
748            if caller_has_usb_driver(caller_id) && irq == usb_irq as u8 {
749                // One-shot unmask: re-enable the USB legacy IRQ after the
750                // driver has cleared GINTSTS.
751                <Board as ArmIrqController>::unmask_usb_irq();
752            }
753        }
754    }
755    // The ISR dereferences a stored nonzero base (VirtIO InterruptACK),
756    // so only an MMIO window the caller was granted is accepted.
757    if !wait_irq_base_permitted(caller_id, mmio_base) {
758        return Err(SyscallError::PermissionDenied);
759    }
760    // x86 IRQ numbers are MSI vectors from 435; a cell waits only on its
761    // own, so it can neither steal nor drain another driver's interrupts.
762    #[cfg(target_arch = "x86_64")]
763    if !crate::dev::pcie_msi::irq_owned_by(irq, caller_id) {
764        return Err(SyscallError::PermissionDenied);
765    }
766    // Lost-wakeup guard: if IRQ already fired before this call, return immediately.
767    if crate::dev::irq_wait::take_pending(irq) {
768        return Ok(0);
769    }
770    // Single-waiter policy: second caller on same IRQ gets TryAgain.
771    if !crate::dev::irq_wait::register_waiter(irq, caller_id, mmio_base) {
772        return Err(SyscallError::TryAgain);
773    }
774    // Park the task — scheduler sweep will wake it when IRQ_PENDING[irq] is set.
775    if let Some(sched) = crate::task::SCHEDULER.lock().as_mut() {
776        if let Some(task) = sched.tasks.get_mut(&caller_id) {
777            task.state = crate::task::tcb::TaskState::Blocked {
778                wait: crate::task::tcb::WaitKind::Completion {
779                    source: api::completion::source::IRQ,
780                    delivery: crate::task::tcb::Delivery::IrqLine(irq),
781                },
782                deadline: None,
783            };
784        }
785    }
786    crate::task::yield_cpu();
787    Ok(0)
788}

840/// Handler for `Syscall::Shutdown` (extracted from the dispatcher).
841pub(super) fn shutdown(caller_id: usize) -> SyscallResult {
842    authorize_raw_shutdown(caller_id)?;
843    #[cfg(all(feature = "test-hooks", target_arch = "aarch64"))]
844    crate::qemu_exit(true);
845
846    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
847    crate::qemu_exit(true);
848
849    #[cfg(all(not(feature = "test-hooks"), target_arch = "riscv64"))]
850    unsafe {
851        // SAFETY: ecall traps to OpenSBI which powers off QEMU; no return.
852        core::arch::asm!(
853            "li a7, 0x53525354", // SBI_EXT_SRST
854            "li a6, 0",          // fid = SYSTEM_RESET
855            "li a0, 0",          // reset_type = Shutdown
856            "li a1, 0",          // reset_reason = NoReason
857            "ecall",
858            options(noreturn)
859        );
860    }
861    #[cfg(all(not(feature = "test-hooks"), target_arch = "aarch64"))]
862    unsafe {
863        // PSCI SYSTEM_OFF via HVC (QEMU virt machine default)
864        core::arch::asm!(
865            "mov x0, #0x0008",
866            "movk x0, #0x8400, lsl #16",
867            "hvc #0",
868            options(noreturn)
869        );
870    }
871    #[cfg(target_arch = "x86_64")]
872    loop {
873        unsafe {
874            core::arch::asm!("hlt", options(nomem, nostack));
875        }
876    }
877    #[cfg(not(any(
878        target_arch = "riscv64",
879        target_arch = "aarch64",
880        target_arch = "x86_64"
881    )))]
882    loop {
883        unsafe {
884            core::arch::asm!("wfi", options(nomem, nostack));
885        }
886    }
887}

889/// Handler for `Syscall::BlkRead` (extracted from the dispatcher).
890pub(super) fn blk_read(caller_id: usize, sector: u64, buf_ptr: usize) -> SyscallResult {
891    if !caller_has_block_io(caller_id) {
892        log::warn!(
893            "BlkRead denied: task {} lacks block-I/O capability",
894            caller_id
895        );
896        return Err(SyscallError::PermissionDenied);
897    }
898    // Per-cell partition range gate — a runaway FAT offset must never
899    // reach kernel-owned LBAs (P2 cell table, P3 snapshot). Returns 0 = failure.
900    if !check_block_access(caller_id, sector, 1) {
901        return Ok(0);
902    }
903    validate_user_buf(buf_ptr, 512, MAX_USER_BUF)?;
904
905    // Bounce buffer: VirtioHal::share() treats the buffer's virtual address
906    // as its physical address (identity-map assumption). Stack frames ARE
907    // identity-mapped; ELF BSS/heap pages are NOT — DMA would land at the
908    // wrong physical address without the bounce. Read into an on-stack buffer
909    // (always identity-mapped), then copy to the user buffer under SUM=1.
910    let mut bounce = [0u8; 512];
911    match crate::dev::block::read_sector(sector, &mut bounce) {
912        Ok(()) => {
913            write_user_slice(caller_id, buf_ptr, &bounce, MAX_USER_BUF)?;
914            Ok(1)
915        }
916        Err(_) => Ok(0),
917    }
918}

920/// Handler for `Syscall::BlkWrite` (extracted from the dispatcher).
921pub(super) fn blk_write(caller_id: usize, sector: u64, buf_ptr: usize) -> SyscallResult {
922    if !caller_has_block_io(caller_id) {
923        log::warn!(
924            "BlkWrite denied: task {} lacks block-I/O capability",
925            caller_id
926        );
927        return Err(SyscallError::PermissionDenied);
928    }
929    // Per-cell partition range gate — prevents a cell from corrupting
930    // the loader's table or the snapshot region. Returns 0 = failure.
931    if !check_block_access(caller_id, sector, 1) {
932        return Ok(0);
933    }
934    let user = read_user_slice(caller_id, buf_ptr, 512, MAX_USER_BUF)?;
935    let mut bounce = [0u8; 512];
936    bounce.copy_from_slice(&user);
937    match crate::dev::block::write_sector(sector, &bounce) {
938        Ok(()) => Ok(1),
939        Err(error) => {
940            log::warn!("[blk] write LBA={} failed: {:?}", sector, error);
941            Ok(0)
942        }
943    }
944}

946/// Handler for `Syscall::RegisterDisplayFramebuffer` (extracted from the dispatcher).
947pub(super) fn register_display_framebuffer(
948    caller_id: usize,
949    base: usize,
950    size: usize,
951    packed_dimensions: usize,
952    pitch: usize,
953) -> SyscallResult {
954    #[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
955    {
956        const MAX_DIMENSION: usize = 8192;
957        const PAGE_SIZE: usize = 4096;
958        // AArch64: only a board with a firmware display path registers a
959        // framebuffer; any other board refuses before every other check.
960        #[cfg(target_arch = "aarch64")]
961        let Some(display) =
962            <hal::aarch64::backend::Board as hal::aarch64::backend::ArmResources>::DISPLAY
963        else {
964            let _ = (base, size, packed_dimensions, pitch);
965            return Err(SyscallError::Unknown);
966        };
967        if !caller_has_mmio_device(caller_id, crate::resource_registry::DEV_DISPLAY) {
968            return Err(SyscallError::PermissionDenied);
969        }
970        #[cfg(target_arch = "aarch64")]
971        if !crate::resource_registry::owns_exact_mmio(
972            caller_id,
973            display.mailbox.base,
974            display.mailbox.size,
975        ) {
976            return Err(SyscallError::PermissionDenied);
977        }
978        if base == 0
979            || size == 0
980            || base & (PAGE_SIZE - 1) != 0
981            || packed_dimensions > u32::MAX as usize
982        {
983            return Err(SyscallError::InvalidInput);
984        }
985        let width = packed_dimensions >> 16;
986        let height = packed_dimensions & 0xffff;
987        if width == 0 || height == 0 || width > MAX_DIMENSION || height > MAX_DIMENSION {
988            return Err(SyscallError::InvalidInput);
989        }
990        let minimum_pitch = width.checked_mul(4).ok_or(SyscallError::InvalidInput)?;
991        if pitch < minimum_pitch || pitch & 3 != 0 {
992            return Err(SyscallError::InvalidInput);
993        }
994        let covered = pitch
995            .checked_mul(height)
996            .ok_or(SyscallError::InvalidInput)?;
997        let end = base.checked_add(size).ok_or(SyscallError::InvalidInput)?;
998        #[cfg(target_arch = "aarch64")]
999        if covered > size || end > display.framebuffer_limit {
1000            return Err(SyscallError::InvalidInput);
1001        }
1002        #[cfg(not(target_arch = "aarch64"))]
1003        {
1004            if covered > size {
1005                return Err(SyscallError::InvalidInput);
1006            }
1007        }
1008        // The framebuffer is MMIO: it must not overlap *any* managed frame,
1009        // not merely sit above one end of the allocator's memory.
1010        //
1011        // `match` rather than `?`: the early return has to happen outside the
1012        // `guard` borrow, or the RPi3 build refuses the guard's lifetime
1013        // (E0597) with the borrowed allocator still in scope.
1014        let overlaps_managed = {
1015            let guard = crate::memory::frame::FRAME_ALLOCATOR.lock();
1016            match guard.as_ref() {
1017                Some(allocator) => allocator.managed_ranges().any(|range| {
1018                    let range_end = range.start + range.frames * PAGE_SIZE;
1019                    base < range_end && range.start < end
1020                }),
1021                None => return Err(SyscallError::Unknown),
1022            }
1023        };
1024        if overlaps_managed {
1025            return Err(SyscallError::InvalidInput);
1026        }
1027        let rounded = size
1028            .checked_add(PAGE_SIZE - 1)
1029            .ok_or(SyscallError::InvalidInput)?
1030            & !(PAGE_SIZE - 1);
1031        let pages = rounded / PAGE_SIZE;
1032        let owner = types::CellId(caller_id as u64);
1033        match crate::resource_registry::reserve_display_framebuffer(
1034            owner,
1035            base,
1036            size,
1037            width as u16,
1038            height as u16,
1039            pitch,
1040        ) {
1041            Ok(crate::resource_registry::DisplayFramebufferReservation::ActiveReplay) => {
1042                return Ok(0);
1043            }
1044            Err(crate::resource_registry::DisplayFramebufferReservationError::Conflict) => {
1045                return Err(SyscallError::PermissionDenied);
1046            }
1047            Ok(crate::resource_registry::DisplayFramebufferReservation::Reserved) => {}
1048        }
1049        if crate::memory::paging::map_display_framebuffer_user(base, pages).is_err() {
1050            crate::resource_registry::cancel_display_framebuffer(
1051                owner,
1052                base,
1053                size,
1054                width as u16,
1055                height as u16,
1056                pitch,
1057            );
1058            return Err(SyscallError::Unknown);
1059        }
1060        if crate::resource_registry::activate_display_framebuffer(
1061            owner,
1062            base,
1063            size,
1064            width as u16,
1065            height as u16,
1066            pitch,
1067        ) {
1068            #[cfg(target_arch = "x86_64")]
1069            crate::dev::fb_console::handover();
1070            Ok(0)
1071        } else {
1072            // This cannot be recovered safely: pages are firmware-owned
1073            // and remain mapped until reboot, so display stays fail-stop.
1074            Err(SyscallError::Unknown)
1075        }
1076    }
1077    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
1078    {
1079        let _ = (caller_id, base, size, packed_dimensions, pitch);
1080        Err(SyscallError::Unknown)
1081    }
1082}

1084/// Handler for `Syscall::RequestMmio` (extracted from the dispatcher).
1085pub(super) fn request_mmio(caller_id: usize, base: usize, len: usize) -> SyscallResult {
1086    // PlatformCap bypass: the Platform Cell may claim only the kernel-admitted
1087    // ECAM config-space window (not in PCIE_BARS), never managed RAM — the
1088    // claim is user-mapped. Overlap check still applies.
1089    // x86 SAS: the boot identity map covers MMIO kernel-only; a granted
1090    // window must gain PTE_USER (+PCD, NX) before the ring-3 cell touches
1091    // it. BAR windows may not be mapped at all yet — this also creates
1092    // the identity mapping.
1093    #[cfg(target_arch = "x86_64")]
1094    fn user_map(base: usize, len: usize) {
1095        crate::memory::paging::map_mmio_user_x86(base, len);
1096    }
1097    // RISC-V/AArch64 make an allowlisted window reachable from EL0 with the
1098    // *boot* identity map, so a window whose access was revoked earlier —
1099    // by a capability revoke, or by the self-test that exercises one — stays
1100    // kernel-only unless the grant re-arms it. This is the non-x86 half of
1101    // the same rule: a granted window is a reachable window.
1102    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
1103    fn user_map(base: usize, len: usize) {
1104        let _ = crate::memory::paging::grant_mmio_user(base, len);
1105    }
1106    #[cfg(not(any(
1107        target_arch = "riscv64",
1108        target_arch = "aarch64",
1109        target_arch = "x86_64"
1110    )))]
1111    fn user_map(_base: usize, _len: usize) {}
1112    // x86: a PCIe BAR is user-mapped without its MSI-X table/PBA pages
1113    // (ADR-0031 CP1 §4.3), so a cell cannot retarget its device's
1114    // interrupts by rewriting table entries.
1115    #[cfg(target_arch = "x86_64")]
1116    fn user_map_pcie_bar(base: usize, len: usize) {
1117        for (start, size) in crate::dev::pcie_msi::user_ranges(base, len) {
1118            crate::memory::paging::map_mmio_user_x86(start, size);
1119        }
1120    }
1121    #[cfg(not(target_arch = "x86_64"))]
1122    fn user_map_pcie_bar(base: usize, len: usize) {
1123        user_map(base, len);
1124    }
1125
1126    if caller_has_platform(caller_id) {
1127        return match crate::resource_registry::request_ecam_mmio(caller_id, base, len) {
1128            Ok(()) => {
1129                user_map(base, len);
1130                Ok(0)
1131            }
1132            Err(types::ViError::PermissionDenied) => Ok(1),
1133            Err(types::ViError::AlreadyExists) => Ok(2),
1134            Err(_) => Ok(3),
1135        };
1136    }
1137    // PCIe BAR path: cells with PcieDriverCap may claim any BAR
1138    // discovered during ECAM scan (registered in resource_registry::PCIE_BARS).
1139    if caller_has_pcie_driver(caller_id) && crate::resource_registry::is_pcie_bar(base, len) {
1140        return match crate::resource_registry::request_mmio(
1141            caller_id,
1142            base,
1143            len,
1144            crate::resource_registry::DEV_PCIE,
1145        ) {
1146            Ok(()) => {
1147                user_map_pcie_bar(base, len);
1148                Ok(0)
1149            }
1150            Err(types::ViError::PermissionDenied) => Ok(1),
1151            Err(types::ViError::AlreadyExists) => Ok(2),
1152            Err(_) => Ok(3),
1153        };
1154    }
1155    // USB Host path: cells with UsbDriverCap may claim the DWC2 MMIO aperture.
1156    if caller_has_usb_driver(caller_id) && crate::resource_registry::is_dwc2_mmio(base, len) {
1157        return match crate::resource_registry::request_dwc2_mmio(caller_id, base, len) {
1158            Ok(()) => {
1159                user_map(base, len);
1160                Ok(0)
1161            }
1162            Err(types::ViError::PermissionDenied) => Ok(1),
1163            Err(types::ViError::AlreadyExists) => Ok(2),
1164            Err(_) => Ok(3),
1165        };
1166    }
1167    // GPIO/UART path: gate on manifest-declared device classes.
1168    let allowed_devices = {
1169        let sched = crate::task::SCHEDULER.lock();
1170        sched
1171            .as_ref()
1172            .map_or(0, |s| s.cell_rights(caller_id).mmio_devices)
1173    };
1174    if allowed_devices == 0 {
1175        return Err(SyscallError::PermissionDenied);
1176    }
1177    match crate::resource_registry::request_mmio(caller_id, base, len, allowed_devices) {
1178        Ok(()) => {
1179            user_map(base, len);
1180            Ok(0)
1181        }
1182        Err(types::ViError::PermissionDenied) => {
1183            log::warn!(
1184                "[mmio] DENY caller={} base={:#x} len={:#x} allowed_devices={:#04x}",
1185                caller_id,
1186                base,
1187                len,
1188                allowed_devices
1189            );
1190            Ok(1)
1191        }
1192        Err(types::ViError::AlreadyExists) => Ok(2),
1193        Err(_) => Ok(3),
1194    }
1195}

1197/// Handler for `Syscall::ProbeVirtioMmio` (extracted from the dispatcher).
1198pub(super) fn probe_virtio_mmio(
1199    caller_id: usize,
1200    base: usize,
1201    reserved: (usize, usize, usize),
1202) -> SyscallResult {
1203    if !caller_has_pcie_driver(caller_id) {
1204        return Err(SyscallError::PermissionDenied);
1205    }
1206    if reserved != (0, 0, 0) {
1207        return Err(SyscallError::InvalidInput);
1208    }
1209    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
1210    {
1211        use crate::dev::virtio_common::{probe_slot, SlotProbe};
1212        match probe_slot(base) {
1213            SlotProbe::Identity(identity) => Ok(identity),
1214            SlotProbe::NoTransport => Err(SyscallError::FileNotFound),
1215            SlotProbe::NotASlot => Err(SyscallError::InvalidInput),
1216        }
1217    }
1218    // x86_64 reaches VirtIO over PCI: there is no MMIO slot to probe.
1219    #[cfg(not(any(target_arch = "riscv64", target_arch = "aarch64")))]
1220    {
1221        let _ = base;
1222        Err(SyscallError::NotSupported)
1223    }
1224}

1226/// Handler for `Syscall::BlkReadAsync` (extracted from the dispatcher).
1227pub(super) fn blk_read_async(caller_id: usize, sector: u64, grant_id: usize) -> SyscallResult {
1228    if !caller_has_block_io(caller_id) {
1229        return Err(SyscallError::PermissionDenied);
1230    }
1231    if !check_block_access(caller_id, sector, 1) {
1232        return Ok(0);
1233    }
1234    // Validate ownership and minimum size (must hold ≥ 512 bytes).
1235    let buf_paddr = {
1236        let tbl = grant_table_lock().lock();
1237        tbl.as_ref()
1238            .and_then(|m| m.get(&grant_id))
1239            .filter(|g| g.owner == caller_id && g.size >= 512)
1240            .map(|g| g.base)
1241    };
1242    let buf_paddr = match buf_paddr {
1243        Some(p) => p,
1244        None => return Ok(0),
1245    };
1246    // Grant pages are identity-mapped (vaddr == paddr), so DMA addresses are correct.
1247    // SAFETY: buf_paddr is a physically contiguous, identity-mapped grant page; valid
1248    // for 512 bytes of VirtIO DMA read.
1249
1250    let buf = unsafe { core::slice::from_raw_parts_mut(buf_paddr as *mut u8, 512) };
1251    match crate::dev::block::read_sector(sector, buf) {
1252        Ok(()) => Ok(1), // async_id = 1 means immediately complete (Phase 04 for real async)
1253        Err(_) => Ok(0),
1254    }
1255}
