9/// Advance a row's domain lifecycle to `Revoked` outside every table lock.
10///
11/// Returns `false` when the receiver or owner invalidation is not acknowledged:
12/// the caller must keep the row (so a retry is idempotent) and must not release
13/// a single frame.
14#[cfg(all(
15    feature = "native-domains",
16    any(
17        target_arch = "riscv64",
18        all(target_arch = "aarch64", feature = "test-hooks")
19    )
20))]
21pub(super) fn revoke_domain_record(record: &DomainRecord, id: usize, kind: &str) -> bool {
22    match record.revoke() {
23        Ok(()) => true,
24        Err(error) => {
25            log::warn!(
26                "[grant] {kind} {id:#x}: domain revoke deferred ({error:?}); frames retained and \
27                 record kept Revoking for idempotent retry"
28            );
29            false
30        }
31    }
32}

45/// Returns `true` if the calling task satisfies the given capability check.
46///
47/// Lock-ordering: acquires SCHEDULER, drops before returning.
48/// Gate on the caller's **Cell** rights (`Scheduler::cell_rights`, the root's
49/// incarnation record), never on a per-thread copy: a revoke aimed at any
50/// thread of the Cell is seen by all of them. A missing task fails closed.
51pub(super) fn caller_has_cap<F: Fn(&crate::task::cap::CapSet) -> bool>(
52    caller_id: usize,
53    check: F,
54) -> bool {
55    crate::task::SCHEDULER.lock().as_ref().is_some_and(|sched| {
56        sched.tasks.contains_key(&caller_id) && check(&sched.cell_rights(caller_id))
57    })
58}

60/// Read a task's live Cell identity `(cell_id, generation)`.
61///
62/// Takes the scheduler lock itself, so a caller must not already hold it. The
63/// service registry records this identity so `LookupServiceBound` can answer with
64/// a binding instead of a bare tid; a task whose identity is missing yields
65/// `None` and the caller decides whether that is a refusal or an unbindable route.
66pub(super) fn live_cell_identity(tid: usize) -> Option<(u64, u64)> {
67    crate::task::SCHEDULER
68        .lock()
69        .as_ref()
70        .and_then(|sched| sched.tasks.get(&tid))
71        .map(|t| (t.cell_id.0, t.cell_generation))
72}

88/// ADR-0031 CP1 §6: 418/424 offer a function that is unowned or already the
89/// caller's, and skip one a live task owns, so a second instance of one driver
90/// reaches the next identical function.
91/// H-3: a PCI query (418/424) must fall inside the caller's catalog row,
92/// keyed by the caller Cell's measured program name (`catalog::PCI_DRIVERS`).
93/// Logs the refused query so a missing row is visible in the boot log.
94pub(super) fn caller_pci_query_allowed(caller_id: usize, allowed: impl Fn(&str) -> bool) -> bool {
95    let program = caller_launch_state(caller_id)
96        .map(|(name, _, _)| name)
97        .unwrap_or_default();
98    let ok = allowed(&program);
99    if !ok {
100        log::warn!(
101            "[pcie] DENY query from caller={} program={:?}: outside its catalog::PCI_DRIVERS row",
102            caller_id,
103            program
104        );
105    }
106    ok
107}

109/// Look up the attested identity of `sender_tid`.
110///
111/// Returns `None` when the sender is gone (a reply to a dead cell, a queued
112/// message whose sender already exited) or is not attributable to a cell. A
113/// receiver must read that as "unknown caller" and deny, never as "some caller
114/// that owns nothing" — owning nothing still reads unowned state.
115///
116/// The flag word carries the VFS mutation bit, the four network rights, the Q
117/// Config-write right and the two KMS SSH rights (`CALLER_FLAG_KMS_SSH_SIGN`,
118/// `CALLER_FLAG_KMS_SSH_ADMIN`, KSSH K3). Every right except the VFS bit is read
119/// from the Cell root's `CapSet`, so every thread of a Cell reports the same
120/// right; the kernel attests it and the receiving service enforces it.
121pub fn attested_identity_of(sender_tid: usize) -> Option<api::caller_identity::CallerIdentity> {
122    let sched_guard = crate::task::SCHEDULER.lock();
123    let sched = sched_guard.as_ref()?;
124    let task = sched.tasks.get(&sender_tid)?;
125    let owner = sched.resolve_live_cell_owner(task.cell_id, task.cell_generation)?;
126    // Every attested right is a property of the Cell: read from the
127    // incarnation record (`Scheduler::cell_rights`), the CapSet spawn bounded by
128    // request ∩ ceiling ∩ policy and revokes narrowed. `resolve_live_cell_owner`
129    // has already proven the root is live and is this exact Cell generation, so
130    // a thread can neither claim more nor less than its Cell holds.
131    let root_caps = sched.cell_rights(sender_tid);
132    let mut flags = vfs_mutate_caller_flag(task.syscall_allowlist, root_caps.vfs_mutate);
133    flags |= root_caps.net_caller_flags();
134    // Q (Phase 01 CP1) is a Cell right exactly like the network rights: read
135    // from the same root CapSet, so a thread cannot hold more or less of it.
136    flags |= root_caps.config_caller_flags();
137    // KSSH K3: the KMS SSH sign/admin rights are Cell rights too, read from the
138    // same root CapSet for the same reason. Attested, not enforced, here.
139    flags |= root_caps.kms_caller_flags();
140    (owner.cell_id == task.cell_id.0 && owner.generation == task.cell_generation).then_some(
141        api::caller_identity::CallerIdentity {
142            flags,
143            cell_id: task.cell_id.0,
144            generation: task.cell_generation,
145            sender_tid: sender_tid as u64,
146        },
147    )
148}

150/// Write the caller-identity trailer into the last bytes of a receiver's buffer.
151///
152/// MUST be called only after the payload has been copied into `buf_ptr`: the
153/// trailer's unforgeability rests on the kernel writing it last, so a sender that
154/// pads its message across the whole buffer cannot pre-place a forged one.
155///
156/// A sender the scheduler can no longer attribute leaves the tail zeroed, which
157/// parses back as "no identity" and therefore as deny.
158pub(super) fn write_caller_identity(
159    caller_id: usize,
160    buf_ptr: usize,
161    buf_len: usize,
162    sender_tid: usize,
163) {
164    let len = api::caller_identity::CALLER_IDENTITY_LEN;
165    let Some(offset) = buf_len.checked_sub(len) else {
166        return; // buffer cannot hold a trailer; receiver sees no identity → deny
167    };
168    let dst = buf_ptr.wrapping_add(offset);
169    if validate_user_buf(dst, len, MAX_USER_BUF).is_err() {
170        return;
171    }
172    let trailer = attested_identity_of(sender_tid)
173        .map(|id| id.to_trailer())
174        .unwrap_or([0u8; api::caller_identity::CALLER_IDENTITY_LEN]);
175    if let Ok(view) = caller_copy_view(caller_id) {
176        view.write_bytes(dst, &trailer).ok();
177    } else {
178        TaskCopyView::sas().write_bytes(dst, &trailer).ok();
179    }
180}

183/// succeeded; a record already gone (or another one now at the head) is left alone.

194/// Every task of `tid`'s Cell incarnation (root and threads), in a fixed
195/// buffer: no allocation, bounded by the per-Cell thread cap.
196pub(super) fn incarnation_members(
197    sched: &crate::sched::scheduler::Scheduler,
198    tid: usize,
199) -> incarnation_tids::Tids {
200    let mut out = incarnation_tids::Tids::default();
201    if let Some(target) = sched.tasks.get(&tid) {
202        let incarnation = (target.cell_id, target.cell_generation);
203        for (&id, task) in sched.tasks.iter() {
204            if (task.cell_id, task.cell_generation) == incarnation {
205                out.push(id);
206            }
207        }
208    }
209    out
210}

244/// Notify `target_tid` that `mask` was revoked: `[MAGIC, KIND_CAP_REVOKED,
245/// mask_le4]` ([`api::services::proto::app_envelope`], `spec/17` §3).
246///
247/// Best-effort by design: the teardown already happened, and a Cell that ignores
248/// the event still lost the hardware. Returns `false` when the target is gone or
249/// its mailbox is full.
250pub(crate) fn send_cap_revoked_event(target_tid: usize, mask: u32) -> bool {
251    use api::services::proto::app_envelope::{
252        CAP_REVOKED_LEN, HEADER_LEN, KIND_CAP_REVOKED, MAGIC,
253    };
254    let mut msg = [0u8; CAP_REVOKED_LEN];
255    msg[0] = MAGIC;
256    msg[1] = KIND_CAP_REVOKED;
257    msg[HEADER_LEN..].copy_from_slice(&mask.to_le_bytes());
258    crate::task::ipc_post_nonblock(0, target_tid, &msg).is_ok()
259}

261/// Annex S2-T/S2-W §3: `GetProcs` and `GetProcs2` additionally need the
262/// caller's `observe_tasks` right and `MemInfo` its `observe_memory` right, on
263/// top of the allowlist bit (and whatever a permit-all allowlist says). Every
264/// other opcode needs none, and neither does the kernel-context sentinel (tid 0),
265/// which has no task. A missing or terminal task is denied.
266pub(super) fn observe_right_held(sc: ViSyscall, caller_id: usize) -> bool {
267    let held: fn(&crate::task::cap::CapSet) -> bool = match sc {
268        ViSyscall::GetProcs | ViSyscall::GetProcs2 | ViSyscall::TelemetrySnapshot => {
269            |rights| rights.observe_tasks
270        }
271        ViSyscall::MemInfo => |rights| rights.observe_memory,
272        _ => return true,
273    };
274    if caller_id == 0 {
275        return true;
276    }
277    caller_has_cap(caller_id, held)
278}

280pub(super) fn read_cell_owner_request(
281    caller_id: usize,
282    request_ptr: usize,
283    request_len: usize,
284) -> Result<api::cell_owner::CellOwnerRequest, SyscallError> {
285    let len = api::cell_owner::CELL_OWNER_REQUEST_LEN;
286    if request_len < len {
287        return Err(SyscallError::BufferTooSmall);
288    }
289    validate_user_buf(request_ptr, len, MAX_USER_BUF)?;
290    let mut bytes = [0u8; api::cell_owner::CELL_OWNER_REQUEST_LEN];
291    let view = caller_copy_view(caller_id)?;
292    view.read_into(request_ptr, &mut bytes)
293        .map_err(|_| SyscallError::InvalidInput)?;
294    api::cell_owner::CellOwnerRequest::from_bytes(&bytes).ok_or(SyscallError::InvalidInput)
295}

297/// Withdraw `cap_mask` from the target's Cell incarnation after its Class-2
298/// teardown has finished: re-run the gate, then narrow the incarnation record
299/// (`root_tid`'s `Task::rights`, captured before the teardown) that every
300/// thread of the Cell is gated on. The record is narrowed even if the target
301/// thread exited meanwhile; only a retired incarnation (root gone) has nothing
302/// left to narrow. Returns whether the target itself is still present. Takes
303/// `SCHEDULER` only.
304pub(super) fn commit_cap_revoke(
305    caller_id: usize,
306    target_tid: usize,
307    root_tid: usize,
308    incarnation: (types::CellId, u64),
309    cap_mask: u32,
310) -> Result<bool, SyscallError> {
311    let mut sched_opt = crate::task::SCHEDULER.lock();
312    let sched = match sched_opt.as_mut() {
313        Some(s) => s,
314        None => return Err(SyscallError::InvalidCommand),
315    };
316    cap_revoke_gate(sched, caller_id, target_tid, cap_mask)?;
317    if let Some(root) = sched.tasks.get_mut(&root_tid) {
318        if (root.cell_id, root.cell_generation) == incarnation {
319            root.rights = root.rights.without_revoke_mask(cap_mask);
320            root.revoke_pending &= !cap_mask;
321        }
322    }
323    Ok(sched.tasks.contains_key(&target_tid))
324}

326/// Authority gates for `CapRevoke`.
327///
328/// Runs twice — before the teardown and before the fields are cleared — so a
329/// `SpawnCap` withdrawn while the teardown runs cannot authorize the clear.
330///
331/// Gate 1: the caller must hold `SpawnCap`, the same authority `ForceExit` needs.
332/// Gate 1b: refuse bits whose authority is AMBIENT with no teardown path —
333///   `HYPERVISOR` (H-extension CSR access is not syscall-gated). MMIO device
334///   windows and the privileged path caps are torn down by the revoke itself.
335/// Gate 2: a system service cell (block_io/network) is refused — revoking I/O
336///   authority from a running VFS/net service mid-flight corrupts driver state;
337///   the supervisor freeze/replace flow is the supported path.
338pub(super) fn cap_revoke_gate(
339    sched: &mut crate::sched::scheduler::Scheduler,
340    caller_id: usize,
341    target_tid: usize,
342    cap_mask: u32,
343) -> Result<(), SyscallError> {
344    use api::syscall::cap_mask as CM;
345
346    if !sched.tasks.contains_key(&caller_id) || !sched.cell_rights(caller_id).spawn {
347        return Err(SyscallError::PermissionDenied);
348    }
349
350    if cap_mask & CM::HYPERVISOR != 0 {
351        crate::audit::log_event(
352            crate::audit::AuditEvent::CapRevoked,
353            &crate::audit::encode_subject(
354                crate::audit::Subject::of(sched, caller_id),
355                target_tid as u32,
356                cap_mask,
357            ),
358        );
359        log::warn!(
360            "[kernel] CapRevoke: refused ambient-authority bits mask={:#010x} from task {} \
361             (no teardown for hypervisor authority)",
362            cap_mask,
363            caller_id
364        );
365        return Err(SyscallError::NotSupported);
366    }
367
368    let target_rights = sched.cell_rights(target_tid);
369    let target_is_system = target_rights.block_io || target_rights.network;
370    if target_is_system {
371        return Err(SyscallError::PermissionDenied);
372    }
373    Ok(())
374}

376/// Release task `tid`'s MMIO windows of `classes` and take their user
377/// accessibility away.
378///
379/// Fail-closed: an error means a window the task can still reach, so the caller
380/// returns without clearing the capability.
381pub(super) fn revoke_mmio_classes(
382    tid: usize,
383    classes: crate::resource_registry::MmioClass,
384) -> Result<(), SyscallError> {
385    for (base, len) in crate::resource_registry::revoke_mmio_for(tid, classes) {
386        if let Err(e) = crate::memory::paging::revoke_mmio_user(base, len) {
387            log::error!(
388                "[kernel] CapRevoke: MMIO window {base:#x}+{len:#x} is still user-reachable \
389                 ({e:?}); capability left intact"
390            );
391            return Err(SyscallError::NotSupported);
392        }
393    }
394    Ok(())
395}

397/// Handler for `Syscall::CapRevoke` (extracted from the dispatcher).
398pub(super) fn cap_revoke(caller_id: usize, target_tid: usize, cap_mask: u32) -> SyscallResult {
399    use api::syscall::cap_mask as CM;
400
401    if target_tid == caller_id {
402        return Err(SyscallError::InvalidCommand);
403    }
404
405    let _serial = RevokeSerial::acquire().ok_or(SyscallError::TryAgain)?;
406
407    // Gates, then the facts the teardown needs. The teardown itself runs
408    // with SCHEDULER released: the grant reclaim reaches
409    // `live_task_binding` and the notify reaches `ipc_post_nonblock`, and
410    // this spinlock is not reentrant. Both passes re-run the gates, so
411    // nothing is revoked on authority withdrawn while it ran. The mask is
412    // published as pending on the root first, so no task of the Cell —
413    // including one spawned during the teardown — can re-claim what is
414    // being torn down.
415    let (members, pending) = {
416        let mut sched_opt = crate::task::SCHEDULER.lock();
417        let sched = match sched_opt.as_mut() {
418            Some(s) => s,
419            None => return Err(SyscallError::InvalidCommand),
420        };
421        cap_revoke_gate(sched, caller_id, target_tid, cap_mask)?;
422        let target = sched
423            .tasks
424            .get(&target_tid)
425            .ok_or(SyscallError::InvalidCommand)?;
426        let incarnation = (target.cell_id, target.cell_generation);
427        // Rights are per Cell incarnation, so revoking a sibling thread
428        // is revoking oneself — refused like `target_tid == caller_id`.
429        if sched
430            .tasks
431            .get(&caller_id)
432            .is_some_and(|c| (c.cell_id, c.cell_generation) == incarnation)
433        {
434            return Err(SyscallError::InvalidCommand);
435        }
436        let root_tid = target.root_tid;
437        // Same incarnation guard as the commit and `PendingRevoke::drop`,
438        // so the bits set here are exactly the bits cleared there.
439        if let Some(root) = sched.tasks.get_mut(&root_tid) {
440            if (root.cell_id, root.cell_generation) == incarnation {
441                root.revoke_pending |= cap_mask;
442            }
443        }
444        (
445            incarnation_members(sched, target_tid),
446            PendingRevoke {
447                root_tid,
448                incarnation,
449                mask: cap_mask,
450            },
451        )
452    };
453
454    // ── Class-2 teardown, BEFORE the record changes ──────────────────
455    // Ambient authority — a mapped MMIO window, an already-shared grant —
456    // is exercised without a per-use syscall re-check, so clearing the
457    // record alone would leave it live. Resources are owned by the tid
458    // that claimed them, so every thread of the incarnation is torn down.
459    // Fail-closed: a teardown that cannot finish returns with the
460    // capability still in the record (the pending mask is withdrawn on
461    // return). Members torn down before the failure stay torn down; a
462    // retry of the same revoke is idempotent.
463    let mmio_revoke = ((cap_mask >> CM::MMIO_SHIFT) & 0xFF) as u8;
464    for &tid in members.iter() {
465        if mmio_revoke != 0 {
466            revoke_mmio_classes(tid, mmio_revoke as crate::resource_registry::MmioClass)?;
467            // A revoked device class can be the authority behind a grant
468            // the target owns (grants carry no source-cap tag), so its
469            // owned grants are reclaimed with it. Frames a live pin still
470            // protects are quarantined rather than freed.
471            reclaim_owned_grants(tid);
472        }
473
474        if cap_mask & CM::PCIE_DRIVER != 0 {
475            // The DMA-anywhere surface: the whole domain, its requester
476            // entries, and the BAR windows the Cell claimed. Exactly the
477            // teardown a dying driver Cell gets.
478            if !crate::task::teardown_dma_authority(tid) {
479                log::error!(
480                    "[kernel] CapRevoke: DMA teardown for task {tid} was not \
481                             acknowledged; capability left in the record, retry the revoke"
482                );
483                return Err(SyscallError::NotSupported);
484            }
485            revoke_mmio_classes(
486                tid,
487                crate::resource_registry::DEV_PCIE as crate::resource_registry::MmioClass,
488            )?;
489            reclaim_owned_grants(tid);
490        }
491
492        if cap_mask & CM::PLATFORM != 0 {
493            // The Platform Cell's ECAM config-space window.
494            revoke_mmio_classes(tid, crate::resource_registry::CLASS_ECAM)?;
495        }
496    }
497
498    let target_present = commit_cap_revoke(
499        caller_id,
500        target_tid,
501        pending.root_tid,
502        pending.incarnation,
503        cap_mask,
504    )?;
505    let root_tid = pending.root_tid;
506    drop(pending);
507
508    crate::audit::log_event(
509        crate::audit::AuditEvent::CapRevoked,
510        &crate::audit::encode_subject(audit_subject(caller_id), target_tid as u32, cap_mask),
511    );
512    log::info!(
513        "[kernel] CapRevoke: task {} revoked mask={:#010x} from task {}",
514        caller_id,
515        cap_mask,
516        target_tid
517    );
518
519    // Courtesy notification, last and best-effort: the teardown above is
520    // what enforces the revocation, so a target that is gone or whose
521    // mailbox is full does not fail the call. The Cell's root (its app
522    // loop) is told too, since the revoke applied to the whole Cell.
523    if target_present && !send_cap_revoked_event(target_tid, cap_mask) {
524        log::warn!("[kernel] CapRevoke: task {target_tid} not notified (exited or mailbox full)");
525    }
526    if root_tid != target_tid && !send_cap_revoked_event(root_tid, cap_mask) {
527        log::warn!(
528            "[kernel] CapRevoke: Cell root {root_tid} not notified (exited or mailbox full)"
529        );
530    }
531
532    Ok(0)
533}

535/// Handler for `Syscall::RegisterService` (extracted from the dispatcher).
536pub(super) fn register_service(
537    caller_id: usize,
538    service_id: u16,
539    tid: usize,
540    flags: usize,
541) -> SyscallResult {
542    if flags & !api::syscall::service_flags::ALL != 0 {
543        return Err(SyscallError::InvalidInput);
544    }
545    // Flags are the namespace owner's decision for a named provider; the narrow
546    // `tid=0` self-registration paths below publish rows without them.
547    if flags != 0 {
548        if tid == 0 {
549            return Err(SyscallError::InvalidInput);
550        }
551        if !(caller_is_trusted_init(caller_id) || caller_has_supervisor(caller_id)) {
552            return Err(SyscallError::PermissionDenied);
553        }
554    }
555    // `BLOCK_DRIVER` is retired: block routes are per device (416/432),
556    // so no caller may publish the singleton service.
557    if service_id == api::syscall::service::BLOCK_DRIVER {
558        return Err(SyscallError::PermissionDenied);
559    }
560    // The machine pin endpoint is init's own: only trusted init may
561    // register it, and only naming itself (ADR-0029 §3). The VFS admits
562    // the live provider to the pin store, so a hijacked registration
563    // would hand over pin write access.
564    if service_id == api::syscall::service::MACHINE_PIN
565        && !(tid == caller_id && caller_is_trusted_init(caller_id))
566    {
567        return Err(SyscallError::PermissionDenied);
568    }
569    // Development Silo readiness is self-published only after artifact,
570    // VM, entropy, guest READY, and public-key validation. The dedicated
571    // non-delegable cap is minted solely for the governed `/bin/silo`
572    // root task in test-hooks kernels; HypervisorCap alone is insufficient.
573    #[cfg(feature = "test-hooks")]
574    if service_id == api::syscall::service::SILO
575        && tid == 0
576        && caller_has_development_silo_registration(caller_id)
577    {
578        let Some((cell_id, generation)) = live_cell_identity(caller_id) else {
579            return Err(SyscallError::InvalidInput);
580        };
581        return if crate::cell::service_registry::register(
582            service_id, caller_id, cell_id, generation,
583        ) {
584            Ok(0)
585        } else {
586            Err(SyscallError::InvalidInput)
587        };
588    }
589    // The state-transfer demo owns one non-production service ID and may
590    // self-register (`tid=0`) so QEMU can exercise the real Supervisor path.
591    if service_id == api::syscall::service::HOTSWAP_DEMO && tid == 0 && caller_has_spawn(caller_id)
592    {
593        // The caller is executing, so its identity is present. If it were
594        // somehow absent, the entry records 0/0 and the legacy `lookup`
595        // path keeps working while `LookupServiceBound` correctly reports
596        // that no binding exists for it.
597        let (cell_id, generation) = live_cell_identity(caller_id).unwrap_or((0, 0));
598        crate::cell::service_registry::register(service_id, caller_id, cell_id, generation);
599        return Ok(0);
600    }
601    // Driver Cell self-registration path: a PCIe GPU driver or the
602    // display-only BCM mailbox driver may register itself with tid=0.
603    if (caller_has_pcie_driver(caller_id)
604        || caller_has_mmio_device(caller_id, crate::resource_registry::DEV_DISPLAY))
605        && tid == 0
606        && service_id == api::syscall::service::GPU_DRIVER
607    {
608        let (cell_id, generation) = live_cell_identity(caller_id).unwrap_or((0, 0));
609        // Role and route publish as one transaction against exact-TID
610        // teardown. This path has always answered 0 once authorized, so
611        // a rejected route only rolls the role back; it adds no refusal.
612        crate::dev::driver_cell::publish_role_or_rollback(
613            caller_id,
614            crate::dev::driver_cell::register_gpu_driver,
615            || crate::cell::service_registry::register(service_id, caller_id, cell_id, generation),
616            crate::dev::driver_cell::deregister_gpu_driver,
617        );
618        return Ok(0);
619    }
620    // Privileged: SpawnCap is necessary but not sufficient. Well-known ids
621    // belong to trusted init and SupervisorCap holders; a spawn-only cell
622    // (e.g. a USB host driver with a protocol child) may publish only a
623    // private id it owns, so it cannot hijack VFS/NET/KMS/SUPERVISOR.
624    if !caller_has_spawn(caller_id) {
625        return Err(SyscallError::PermissionDenied);
626    }
627    let registrar = if caller_is_trusted_init(caller_id) || caller_has_supervisor(caller_id) {
628        crate::cell::service_registry::Registrar::Namespace
629    } else {
630        crate::cell::service_registry::Registrar::Spawner(caller_id)
631    };
632    // The liveness check and the identity capture share one scheduler lock,
633    // so a provider cannot be recorded with a binding that was already stale
634    // when it was written.
635    let provider = crate::task::SCHEDULER
636        .lock()
637        .as_ref()
638        .and_then(|scheduler| {
639            scheduler.tasks.get(&tid).and_then(|task| {
640                (!matches!(task.state, TaskState::Retiring | TaskState::Terminated))
641                    .then_some((task.cell_id.0, task.cell_generation))
642            })
643        });
644    let Some((cell_id, generation)) = provider else {
645        return Err(SyscallError::InvalidInput);
646    };
647    match crate::cell::service_registry::register_as(
648        registrar, service_id, tid, cell_id, generation, flags,
649    ) {
650        Ok(()) => Ok(0),
651        Err(crate::cell::service_registry::Refusal::NotOwner) => {
652            Err(SyscallError::PermissionDenied)
653        }
654        Err(crate::cell::service_registry::Refusal::Rejected) => Err(SyscallError::InvalidInput),
655    }
656}

658/// Handler for `Syscall::LookupServiceBound` (extracted from the dispatcher).
659pub(super) fn lookup_service_bound(
660    caller_id: usize,
661    service_id: u16,
662    out_ptr: usize,
663    out_len: usize,
664) -> SyscallResult {
665    // Open to all cells (bit 37): where to send, and which provider
666    // incarnation is there.
667    //
668    // The record is written whole or not at all. A short buffer is refused
669    // with the same `BufferTooSmall` error the other fixed-record writers
670    // use (`QueryDirHandles`, `ResolveCellOwner`) rather than partially
671    // filled; an absent, paused or identity-less provider returns the `0`
672    // absence sentinel — never a binding the kernel cannot stand behind.
673    if out_len < api::service_binding::SERVICE_BINDING_LEN {
674        return Err(SyscallError::BufferTooSmall);
675    }
676    let Some((tid, cell_id, generation)) = crate::cell::service_registry::lookup_bound(service_id)
677    else {
678        return Ok(0);
679    };
680    let binding = api::service_binding::ServiceBinding::new(tid as u64, cell_id, generation);
681    write_user_slice(caller_id, out_ptr, &binding.to_bytes(), MAX_USER_BUF)?;
682    Ok(api::service_binding::SERVICE_BINDING_LEN)
683}

685/// Handler for `Syscall::ServiceLookup` (extracted from the dispatcher).
686pub(super) fn service_lookup(caller_id: usize, name_ptr: usize, name_len: usize) -> SyscallResult {
687    let name = read_user_string(caller_id, name_ptr, name_len, MAX_LOG_MSG)?;
688    let id: usize = match name.as_str() {
689        "vfs" => 3,
690        "config" => 4,
691        "input" => 5,
692        "net" => 6,
693        "compositor" => 7,
694        "shell" => 8,
695        _ => return Err(SyscallError::FileNotFound),
696    };
697    Ok(id)
698}

700/// Handler for `Syscall::ResolveCellOwner` (extracted from the dispatcher).
701pub(super) fn resolve_cell_owner(
702    caller_id: usize,
703    cell_id: u64,
704    generation: u64,
705    out_ptr: usize,
706    out_len: usize,
707) -> SyscallResult {
708    let len = api::cell_owner::CELL_OWNER_LEN;
709    if out_len < len {
710        return Err(SyscallError::BufferTooSmall);
711    }
712    let owner = {
713        let guard = crate::task::SCHEDULER.lock();
714        let sched = guard.as_ref().ok_or(SyscallError::PermissionDenied)?;
715        let owner = sched
716            .resolve_live_cell_owner(types::CellId(cell_id), generation)
717            .ok_or(SyscallError::PermissionDenied)?;
718        let vfs_principal = sched.tasks.get(&caller_id).is_some_and(|task| {
719            crate::cell::service_registry::retains_outer_caller(
720                task.cell_id.0,
721                task.cell_generation,
722            ) && task.caller.cell_id == cell_id
723                && task.caller.cell_generation == generation
724        });
725        let registered_service =
726            crate::cell::service_registry::is_registered_tid(owner.root_tid as usize);
727        if !vfs_principal && !registered_service {
728            return Err(SyscallError::PermissionDenied);
729        }
730        owner
731    };
732    let bytes = owner.to_bytes();
733    write_user_slice(caller_id, out_ptr, &bytes, MAX_USER_BUF)?;
734    Ok(len)
735}

737/// Handler for `Syscall::WatchCellOwner` (extracted from the dispatcher).
738pub(super) fn watch_cell_owner(
739    caller_id: usize,
740    cell_id: u64,
741    generation: u64,
742    out_ptr: usize,

747        return Err(SyscallError::BufferTooSmall);

758    usize::try_from(token).map_err(|_| SyscallError::PermissionDenied)
759}

861/// Handler for `Syscall::FreezeCell` (extracted from the dispatcher).
862pub(super) fn freeze_cell(caller_id: usize, target_tid: usize) -> SyscallResult {
863    if !caller_has_supervisor(caller_id) {
864        return Err(SyscallError::PermissionDenied);
865    }
866    // Refuse to freeze the kernel (tid 0) or the caller itself.
867    if target_tid == 0 || target_tid == caller_id {
868        return Err(SyscallError::InvalidInput);
869    }
870    // swap_id u64::MAX = admin freeze (not a hotswap sequence).
871    crate::cell::hotswap::freeze_task_with_ceiling(target_tid, u64::MAX)
872        .map(|_| 0)
873        .map_err(|error| {
874            if error == types::ViError::NotFound {
875                SyscallError::FileNotFound
876            } else {
877                SyscallError::PermissionDenied
878            }
879        })
880}

882/// Handler for `Syscall::PauseService` (extracted from the dispatcher).
883pub(super) fn pause_service(
884    caller_id: usize,
885    service_id: u16,
886    expected_tid: usize,
887) -> SyscallResult {
888    if !caller_has_supervisor(caller_id) {
889        return Err(SyscallError::PermissionDenied);
890    }
891    if expected_tid == 0 {
892        return Err(SyscallError::InvalidInput);
893    }
894    if !crate::cell::service_registry::pause(service_id, expected_tid) {
895        return Err(SyscallError::TryAgain);
896    }
897    if !crate::task::inbound_ipc_drained(expected_tid) {
898        return Err(SyscallError::TryAgain);
899    }
900    Ok(0)
901}

903/// Handler for `Syscall::ResumeCell` (extracted from the dispatcher).
904pub(super) fn resume_cell(
905    caller_id: usize,
906    target_tid: usize,
907    source_tid: usize,
908    service_id: usize,
909    reserved: usize,
910) -> SyscallResult {
911    if !caller_has_supervisor(caller_id) {
912        return Err(SyscallError::PermissionDenied);
913    }
914    if target_tid == 0 || reserved != 0 {
915        return Err(SyscallError::InvalidInput);
916    }
917    if source_tid == 0 {
918        if service_id != 0 {
919            return Err(SyscallError::InvalidInput);
920        }
921        crate::cell::hotswap::unfreeze_task(target_tid);
922        return Ok(0);
923    }
924    if source_tid == target_tid || service_id > u16::MAX as usize {
925        return Err(SyscallError::InvalidInput);
926    }
927    crate::cell::hotswap::commit_hotswap_barrier(source_tid, target_tid, service_id as u16)
928        .map(|()| 0)
929        .map_err(|error| match error {
930            ViError::NotFound => SyscallError::FileNotFound,
931            ViError::PermissionDenied => SyscallError::PermissionDenied,
932            ViError::WouldBlock | ViError::OutOfMemory => SyscallError::TryAgain,
933            ViError::InvalidArgument | ViError::InvalidInput => SyscallError::InvalidInput,
934            _ => SyscallError::Unknown,
935        })
936}

938/// Handler for `Syscall::KillCell` (extracted from the dispatcher).
939pub(super) fn kill_cell(caller_id: usize, target_tid: usize, exit_code: u32) -> SyscallResult {
940    if !caller_has_supervisor(caller_id) {
941        return Err(SyscallError::PermissionDenied);
942    }
943    // Never kill kernel (tid 0) or the caller itself.
944    if target_tid == 0 || target_tid == caller_id {
945        return Err(SyscallError::InvalidInput);
946    }
947    // Never kill critical cells (init) — would collapse the restart tree.
948    // Single lock acquisition to get both is_critical and cell_id.
949    let (is_crit, cell_id) = {
950        let guard = crate::task::SCHEDULER.lock();
951        match guard.as_ref().and_then(|s| s.tasks.get(&target_tid)) {
952            None => return Err(SyscallError::FileNotFound),
953            Some(t) => (t.is_critical, t.cell_id),
954        }
955    };
956    if is_crit {
957        return Err(SyscallError::PermissionDenied);
958    }
959    crate::cell::hotswap::exit_task_internal(target_tid, cell_id);
960    Ok(exit_code as usize)
961}
