18struct SumAccessGuard {
19    restore: bool,
20}

22#[cfg(all(
23    feature = "test-hooks",
24    any(target_arch = "riscv64", target_arch = "riscv32")
25))]
26impl SumAccessGuard {
27    fn enter() -> Self {
28        const SUM_MASK: usize = 1usize << 18;
29        let status: usize;
30        // SAFETY: test hooks run in supervisor mode. SUM is changed only for
31        // mapped stack sampling and restored by Drop before this scope exits.
32        unsafe {
33            core::arch::asm!("csrr {status}, sstatus", status = out(reg) status, options(nostack));
34            if status & SUM_MASK == 0 {
35                core::arch::asm!("csrs sstatus, {mask}", mask = in(reg) SUM_MASK, options(nostack));
36            }
37        }
38        Self {
39            restore: status & SUM_MASK == 0,
40        }
41    }
42}
43
44#[cfg(all(
45    feature = "test-hooks",
46    any(target_arch = "riscv64", target_arch = "riscv32")
47))]
48impl Drop for SumAccessGuard {
49    fn drop(&mut self) {
50        if self.restore {
51            const SUM_MASK: usize = 1usize << 18;
52            // SAFETY: this reverses the matching supervisor CSR change made by
53            // `enter`; the guard is not shared across harts or task switches.
54            unsafe {
55                core::arch::asm!("csrc sstatus, {mask}", mask = in(reg) SUM_MASK, options(nostack));
56            }
57        }
58    }
59}

61/// Bottom guard reservation for every kernel and user stack.
62pub const STACK_GUARD_PAGES: usize = 2;

64/// Hand `total_pages` frames starting at `base` back to `allocator`, restoring each
65/// to the boot identity mapping (kernel RWX) first.
66///
67/// Invariant this upholds: every frame on the allocator's free list is
68/// identity-mapped kernel-RWX. The cell loader zeroes a freshly allocated frame
69/// through its identity address, so a frame released while unmapped (as the guard
70/// frame is) store-faults for its next owner, and one released carrying USER flags
71/// hands that owner the wrong permissions. Unmap-then-map normalises the PTE
72/// whatever state the frame was left in.
73///
74/// The caller must already hold `FRAME_ALLOCATOR`; only `KERNEL_ROOT` (a leaf) is
75/// taken here, which is the documented order.
76fn release_frames(allocator: &mut FrameAllocator, base: VAddr, total_pages: usize) {
77    let kernel_rwx = Flags::from_bits(
78        Flags::VALID | Flags::READ | Flags::WRITE | Flags::EXECUTE | Flags::ACCESSED | Flags::DIRTY,
79    );
80    for i in 0..total_pages {
81        let frame = base + (i * PAGE_SIZE);
82        let _ = paging::unmap_page(frame);
83        let _ = paging::map_page(allocator, frame, frame, kernel_rwx);
84        allocator.deallocate_frame(frame);
85    }
86    paging::tlb_flush_all();
87}

89/// Represents an allocated Stack.
90/// Implements Drop to automatically free pages.
91#[derive(Debug)]
92pub struct Stack {
93    /// Base address (lowest address) of the allocated range.
94    /// This includes the guard pages at the bottom.
95    pub base: VAddr,
96    /// Number of usable pages (excluding guard pages).
97    pub pages: usize,
98    /// Number of verified-unmapped pages below the usable range.
99    pub guard_pages: usize,
100    /// Top of the stack (initial SP).
101    pub top: VAddr,
102    /// Test-only lifetime registration for a kernel stack selected by a
103    /// native-domain private root.
104    #[cfg(all(
105        feature = "native-domains",
106        feature = "test-hooks",
107        target_arch = "riscv64"
108    ))]
109    supervisor_registration: Option<crate::memory::domain_supervisor_registry::SupervisorRangeId>,
110}
111
112impl Stack {
113    /// Allocate a new Kernel Stack of `pages` usable pages plus the configured
114    /// bottom guard reservation.
115    ///
116    /// # Errors
117    /// - `OutOfMemory` — no contiguous run of usable plus guard frames exists, or a
118    ///   page-table mapping could not be installed.
119    /// - `NotSupported` — every guard page could not be established. No stack is
120    ///   returned in that case; see [`Self::allocate`].
121    pub fn new_kernel(pages: usize) -> Result<Self, ViError> {
122        Self::allocate(pages, STACK_GUARD_PAGES, false)
123    }
124
125    /// Allocate a new User Stack of `pages` usable pages plus the configured
126    /// bottom guards. Usable pages are mapped USER RW.
127    ///
128    /// # Errors
129    /// Same as [`Self::new_kernel`].
130    pub fn new_user(pages: usize) -> Result<Self, ViError> {
131        Self::allocate(pages, STACK_GUARD_PAGES, true)
132    }
133
134    /// Internal allocation logic.
135    ///
136    /// Contract: this either returns a stack whose guard page is *verified* absent
137    /// from the page tables, or it returns an error having released every frame it
138    /// took. There is deliberately no third outcome. An unguarded stack is not a
139    /// degraded stack — in a single address space the frame below it belongs to
140    /// another cell, so an overflow that should have trapped instead corrupts a
141    /// neighbour with no fault and no log, and the victim dies later somewhere
142    /// unrelated. A log line at allocation time is not a mitigation: nothing reads
143    /// it in the microseconds before the overflow.
144    fn allocate(pages: usize, guard_pages: usize, user_mode: bool) -> Result<Self, ViError> {
145        let total_pages = pages.checked_add(guard_pages).ok_or(ViError::OutOfMemory)?;
146
147        let mut frame_guard = FRAME_ALLOCATOR.lock();
148        let allocator = frame_guard.as_mut().ok_or_else(|| {
149            error!(
150                "Stack alloc failed: frame allocator unavailable (user={}, pages={})",
151                user_mode, total_pages
152            );
153            ViError::OutOfMemory
154        })?;
155
156        // A stack must be ONE contiguous run of frames, and that is a consequence of
157        // SAS rather than a shortcut: everything is identity-mapped (VA == PA) and
158        // there is no virtual-address allocator, so virtual contiguity can only come
159        // from physical contiguity.
160        //
161        // The cost: this can fail on a fragmented allocator while plenty of memory is
162        // free, and no amount of lazy commit helps — a guard page cannot make VA != PA.
163        // So callers must read `OutOfMemory` here as "no run this long exists right
164        // now", which is recoverable, and never as grounds to panic. Lifting the
165        // constraint means adding a VA allocator so stack pages can be scattered.
166        let base_frame = allocator.allocate_contiguous(total_pages).ok_or_else(|| {
167            error!(
168                "Stack alloc failed: no contiguous run (user={}, pages={}, bytes={})",
169                user_mode,
170                total_pages,
171                total_pages * PAGE_SIZE
172            );
173            ViError::OutOfMemory
174        })?;
175
176        let base_addr = base_frame; // Identity Map
177
178        // 2. Map Pages
179        // Bottom guard frames are left unmapped; usable frames begin after them.
180
181        let usable_start_idx = guard_pages;
182
183        // SAS identity map: all RAM is already identity-mapped RWX by
184        // init_kernel_paging. The usable pages are re-mapped below, then every
185        // guard frame is unmapped after the loop so an overflow traps.
186
187        // Usable Pages
188        let flags = if user_mode {
189            // User Read/Write (Exec?)
190            Flags::from_bits(
191                Flags::VALID
192                    | Flags::READ
193                    | Flags::WRITE
194                    | Flags::USER
195                    | Flags::ACCESSED
196                    | Flags::DIRTY,
197            )
198        } else {
199            // Kernel Read/Write
200            Flags::from_bits(
201                Flags::VALID | Flags::READ | Flags::WRITE | Flags::ACCESSED | Flags::DIRTY,
202            )
203        };
204
205        for i in usable_start_idx..total_pages {
206            let addr = base_addr + (i * PAGE_SIZE);
207            if paging::map_page(allocator, addr, addr, flags).is_err() {
208                // The run is not yet owned by a Stack, so nothing will drop it —
209                // release it here or the frames are lost until reboot.
210                error!("Stack alloc failed: cannot map page 0x{:X}", addr);
211                release_frames(allocator, base_addr, total_pages);
212                return Err(ViError::OutOfMemory);
213            }
214        }
215
216        // Guard pages: drop the bottom frames' pre-existing identity mappings so a
217        // stack overflow (a write below base_addr+PAGE_SIZE) faults instead of
218        // silently corrupting the neighbouring frame. The spawn paths zero only
219        // the usable pages (skipping base_addr), so nothing legitimately writes to
220        // the guard frames. They stay owned by this Stack (freed in Drop); only
221        // their PTEs are cleared. unmap_page locks KERNEL_ROOT (not FRAME_ALLOCATOR,
222        // which we still hold) — no deadlock.
223        //
224        // The guard is verified by translation, not by the unmap's return code: on
225        // some arches `unmap_page` reports success for a page it never touched
226        // (paging root absent, or the PTE was not a 4 KiB leaf). Asking the page
227        // tables whether the frame still resolves is the only answer that matches
228        // what the hardware will do on overflow.
229        for guard_index in 0..guard_pages {
230            let guard_addr = base_addr + (guard_index * PAGE_SIZE);
231            let unmap_ok = paging::unmap_page(guard_addr).is_ok();
232            paging::tlb_flush_all();
233            if !unmap_ok || paging::virt_to_phys(guard_addr).is_some() {
234                error!(
235                    "Stack alloc refused: guard frame 0x{:X} still mapped (unmap_ok={})",
236                    guard_addr, unmap_ok
237                );
238                release_frames(allocator, base_addr, total_pages);
239                return Err(ViError::NotSupported);
240            }
241        }
242
243        // Calculate Top (Stack grows down)
244        // Top is at the END of the allocated range.
245        let top = base_addr + (total_pages * PAGE_SIZE);
246
247        #[cfg(all(
248            feature = "native-domains",
249            feature = "test-hooks",
250            target_arch = "riscv64"
251        ))]
252        let supervisor_registration =
253            if user_mode || !crate::memory::domain_supervisor_registry::is_active() {
254                None
255            } else {
256                match crate::memory::domain_supervisor_registry::register(
257                    base_addr + (guard_pages * PAGE_SIZE),
258                    top,
259                    crate::memory::domain_supervisor_registry::SupervisorRangeKind::KernelStack,
260                    crate::memory::domain_supervisor_registry::SupervisorRangeOwner::TaskStack,
261                ) {
262                    Ok(id) => Some(id),
263                    Err(()) => {
264                        release_frames(allocator, base_addr, total_pages);
265                        return Err(ViError::NotSupported);
266                    }
267                }
268            };
269        trace!(
270            "Allocated Stack: Base=0x{:X}, Top=0x{:X}, Pages={}, Guards={}, User={}",
271            base_addr,
272            top,
273            pages,
274            guard_pages,
275            user_mode
276        );
277
278        Ok(Stack {
279            base: base_addr,
280            pages,
281            guard_pages,
282            top,
283            #[cfg(all(
284                feature = "native-domains",
285                feature = "test-hooks",
286                target_arch = "riscv64"
287            ))]
288            supervisor_registration,
289        })
290    }

292    /// Total physical bytes reserved for this stack, including guard pages.
293    pub fn allocated_bytes(&self) -> usize {
294        (self.pages + self.guard_pages) * PAGE_SIZE
295    }
296
297    /// Task-owned usable stack bytes, excluding any kernel-only guard reservation.
298    pub fn usable_bytes(&self) -> usize {
299        self.pages * PAGE_SIZE
300    }
301
302    /// Lowest mapped byte in the usable stack range.
303    pub fn usable_start(&self) -> usize {
304        self.base + (self.guard_pages * PAGE_SIZE)
305    }
306    /// Whether `[ptr, ptr + len)` stays within this task's mapped stack bytes.
307    pub fn contains_usable_range(&self, ptr: usize, len: usize) -> bool {
308        let Some(end) = ptr.checked_add(len) else {
309            return false;
310        };
311        ptr >= self.usable_start() && end <= self.top
312    }

314    #[cfg(feature = "test-hooks")]
315    /// Prime the usable stack range with a sentinel pattern so later scans can
316    /// estimate the deepest downward growth without adding a runtime ABI.
317    pub fn test_hook_prime_watermark(&self) {
318        assert!(paging::virt_to_phys(self.usable_start()).is_some());
319        assert!(paging::virt_to_phys(self.top - 1).is_some());
320        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
321        let _sum_guard = SumAccessGuard::enter();
322        // SAFETY: Stack allocation maps the asserted usable range contiguously;
323        // this test hook owns the stack before the task can run.
324        unsafe {
325            core::ptr::write_bytes(
326                self.usable_start() as *mut u8,
327                STACK_WATERMARK_PATTERN,
328                self.usable_bytes(),
329            );
330        }
331    }
332
333    #[cfg(feature = "test-hooks")]
334    /// Return the deepest observed stack footprint in bytes since the sentinel
335    /// pattern was primed.
336    pub fn test_hook_watermark_bytes(&self) -> usize {
337        assert!(paging::virt_to_phys(self.usable_start()).is_some());
338        assert!(paging::virt_to_phys(self.top - 1).is_some());
339        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
340        let _sum_guard = SumAccessGuard::enter();
341
342        let usable = self.usable_bytes();
343        let start = self.usable_start() as *const u8;
344        let mut untouched_prefix = 0usize;
345        while untouched_prefix < usable {
346            // SAFETY: Stack allocation maps the asserted usable range, and the
347            // scan stays within it. Volatile reads observe the live watermark.
348            let byte = unsafe { core::ptr::read_volatile(start.add(untouched_prefix)) };
349            if byte != STACK_WATERMARK_PATTERN {
350                break;
351            }
352            untouched_prefix += 1;
353        }
354        usable.saturating_sub(untouched_prefix)
355    }

358#[cfg(feature = "test-hooks")]
359pub fn stack_probe_self_test() -> bool {
360    let stack = match Stack::new_kernel(2) {
361        Ok(stack) => stack,
362        Err(_) => return false,
363    };
364    if stack.guard_pages != STACK_GUARD_PAGES {
365        return false;
366    }
367    for guard_index in 0..stack.guard_pages {
368        let guard_addr = stack.base + (guard_index * PAGE_SIZE);
369        if paging::virt_to_phys(guard_addr).is_some() {
370            return false;
371        }
372    }
373    let deliberate_overflow_target = stack.usable_start().saturating_sub(8);
374    if paging::virt_to_phys(deliberate_overflow_target).is_some() {
375        return false;
376    }
377    stack.test_hook_prime_watermark();
378    if stack.test_hook_watermark_bytes() != 0 {
379        return false;
380    }
381    let probe_bytes = 96usize;
382    unsafe {
383        core::ptr::write_bytes((stack.top - probe_bytes) as *mut u8, 0x11, probe_bytes);
384    }
385    stack.test_hook_watermark_bytes() == probe_bytes
386}

388impl Drop for Stack {
389    fn drop(&mut self) {
390        #[cfg(all(
391            feature = "native-domains",
392            feature = "test-hooks",
393            target_arch = "riscv64"
394        ))]
395        if let Some(id) = self.supervisor_registration.take() {
396            let unregistered = crate::memory::domain_supervisor_registry::unregister(id);
397            debug_assert!(unregistered);
398        }
399        trace!("Dropping Stack at 0x{:X}", self.base);
400
401        let total_pages = self.pages + self.guard_pages;
402
403        let mut frame_guard = FRAME_ALLOCATOR.lock();
404        if let Some(allocator) = frame_guard.as_mut() {
405            release_frames(allocator, self.base, total_pages);
406        }
407    }
408}

410/// Physical frames mapped for a cell's ELF segments, recorded as `(vaddr, frame)`
411/// at load time so they can be reclaimed when the cell dies.
412///
413/// `Stack::drop` only frees stacks; without this a cell's code/data frames leak
414/// on every death (a supervised service restarted repeatedly would grow to OOM).
415/// Segment frames are allocated exclusively for the cell by `load_segments`
416/// (IPC/shared buffers use separate frames), so freeing them on death is safe.
417#[derive(Debug)]
418pub struct CellSegments {
419    pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
420    /// ELF pages whose final mapping grants the Cell write permission.
421    writable_pages: alloc::vec::Vec<types::VAddr>,
422    /// VA base allocated by `va_alloc::alloc_cell_va` for PIE cells; `0` for
423    /// fixed-VA cells.  Returned to the allocator's free list on drop.
424    pie_va_base: usize,
425}
426
427impl CellSegments {
428    pub fn new(
429        pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
430        pie_va_base: usize,
431    ) -> Self {
432        Self {
433            pages,
434            writable_pages: alloc::vec::Vec::new(),
435            pie_va_base,
436        }
437    }
438
439    pub fn with_writable_pages(
440        pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
441        writable_pages: alloc::vec::Vec<types::VAddr>,
442        pie_va_base: usize,
443    ) -> Self {
444        Self {
445            pages,
446            writable_pages,
447            pie_va_base,
448        }
449    }
450    /// Return the exclusive end of the writable page containing `ptr`.
451    pub fn writable_page_end_containing(&self, ptr: usize) -> Option<usize> {
452        let page = ptr & !(PAGE_SIZE - 1);
453        self.writable_pages
454            .contains(&page)
455            .then(|| page.checked_add(PAGE_SIZE))
456            .flatten()
457    }
458
459    /// Whether every page touched by this range is writable in this Cell image.
460    pub fn contains_writable_range(&self, ptr: usize, len: usize) -> bool {
461        let Some(end) = ptr.checked_add(len) else {
462            return false;
463        };
464        if len == 0 {
465            return false;
466        }
467        let mut page = ptr & !(PAGE_SIZE - 1);
468        while page < end {
469            if !self.writable_pages.contains(&page) {
470                return false;
471            }
472            page = match page.checked_add(PAGE_SIZE) {
473                Some(next) => next,
474                None => return false,
475            };
476        }
477        true
478    }
479
480    #[cfg(feature = "test-hooks")]
481    pub(crate) fn unpublished_pages(&self) -> &[(types::VAddr, types::PhysAddr)] {
482        &self.pages
483    }
484
485    /// Total physical bytes reserved for this cell's ELF segment frames.
486    pub fn allocated_bytes(&self) -> usize {
487        self.pages.len() * PAGE_SIZE
488    }
489
490    /// Unmap this cell's segment VAs immediately at death — WITHOUT freeing the
491    /// frames (those are freed lazily when the zombie is reaped, in `drop`).
492    ///
493    /// Frees the address space right away so (a) a respawn can reuse the fixed VA
494    /// and (b) the load-time overwrite guard (`load_segments`) only ever observes
495    /// LIVE cells' (and kernel MMIO) mappings, never a dead-but-unreaped cell's.
496    /// Locks only `KERNEL_ROOT` (a leaf), so it is safe under the SCHEDULER lock.
497    pub fn eager_unmap(&self) {
498        for &(vaddr, frame) in &self.pages {
499            // Only unmap a VA that still resolves to OUR frame (it won't if a
500            // respawn already re-pointed it — leave the new mapping intact).
501            if paging::virt_to_phys(vaddr) == Some(frame) {
502                let _ = paging::unmap_page(vaddr);
503            }
504            // Flush even when a replacement mapping won the race: an old remote
505            // translation for this VA can still point at this segment's frame.
506            crate::memory::tlb_shootdown::flush_page(vaddr);
507        }
508    }
509}
510
511impl Drop for CellSegments {
512    fn drop(&mut self) {
513        // Leaf and parent invalidations must reach every hart before either
514        // backing or page-table frames return to the allocator.
515        self.eager_unmap();
516        for &(vaddr, _frame) in &self.pages {
517            let reclaimed = paging::prune_empty_tables(vaddr);
518            if !reclaimed.is_empty() {
519                crate::memory::tlb_shootdown::flush_page(vaddr);
520                let mut frames = FRAME_ALLOCATOR.lock();
521                if let Some(allocator) = frames.as_mut() {
522                    reclaimed.release(allocator);
523                }
524            }
525        }
526        let mut frame_guard = FRAME_ALLOCATOR.lock();
527        if let Some(allocator) = frame_guard.as_mut() {
528            for &(_vaddr, frame) in &self.pages {
529                allocator.deallocate_frame(frame);
530            }
531        }
532        if self.pie_va_base != 0 {
533            crate::loader::va_alloc::free_cell_va(self.pie_va_base);
534        }
535    }
