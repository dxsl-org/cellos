29/// Hand a stack run mapped at `va` to `batch`: unmap each of its pages
30/// (recording the shared-root invalidation), detach the tables only the run
31/// needed where no alias is restored (x86_64, and every user stack, whose
32/// frames keep their kernel alias), and retire the run, which the batch frees
33/// only once that invalidation is confirmed — restoring its kernel alias first
34/// where the run says so.
35///
36/// `hold`: the stack's VA, released only with the frames (or after them, once
37/// a quarantined run is freed): its `STACK_RESERVATIONS` entry, and for a user
38/// stack its arena slot. `None` before either was taken.
39/// Takes only `KERNEL_ROOT` (a leaf); the caller may hold `FRAME_ALLOCATOR`.
40fn retire_run(batch: &mut Batch, mut run: FrameRun, va: VAddr, hold: Option<VaHold>) {
41    let pages = run.pages();
42    for i in 0..pages {
43        let _ = paging::unmap_page(batch, va + i * PAGE_SIZE);
44    }
45    if run.alias() == KernelAlias::Intact {
46        for i in 0..pages {
47            paging::prune_empty_tables(va + i * PAGE_SIZE).retire_into(batch);
48        }
49    }
50    // Every leaf of the run is gone from the kernel root. A run in any other
51    // state than `Mapped` cannot reach here; the batch would leak it (counted),
52    // never free it.
53    let unpublished = run.unpublish();
54    debug_assert!(unpublished.is_ok(), "stack run retired twice");
55    batch.retire(match hold {
56        Some(hold) => run.with_va_hold(hold),
57        None => run,
58    });
59}

61/// [`VaHold`] release for a kernel stack's run: only now are its guard VAs
62/// identity-mapped again, so the loader's collision check takes over.
63fn release_stack_reservation(start: VAddr, end: VAddr) {
64    STACK_RESERVATIONS.remove(start, end);
65}

67/// [`VaHold`] release for a user stack: nothing translates any page of its

69fn release_stack_slot(start: VAddr, end: VAddr) {

74/// Retire `pages` user-published pages at VA `base` (a stack, or a grant at its
75/// physical address) from the shared root before their frames return to the
76/// allocator.
77///
78/// A Sas user copy that probed one of these pages earlier must not land after
79/// its frame is freed (and, where the page was identity-mapped, re-mapped
80/// kernel-RW, which the kernel writes without faulting): the copy would hit
81/// whatever the frame becomes next (page table, another cell). Every PTE is
82/// cleared here and recorded in `batch`,
83/// whose finish invalidates every hart and then waits out any Sas commit
84/// window still open (see `user_copy::sas_lease`) before it frees a run. The
85/// frames stay UNMAPPED; the caller retires them into the same batch.
86///
87/// Lock order: KERNEL_ROOT (leaf) per page. The batch flushes only when it is
88/// finished, after the caller released `FRAME_ALLOCATOR`.
89pub(crate) fn unpublish_user_frames(batch: &mut Batch, base: VAddr, pages: usize) {
90    // Before paging is live no translation (and no user copy) can exist.
91    if pages == 0 || paging::KERNEL_ROOT.lock().is_none() {
92        return;
93    }
94    for i in 0..pages {
95        let _ = paging::unmap_page(batch, base + i * PAGE_SIZE);
96    }
97    batch.invalidate_range(base, pages * PAGE_SIZE);
98    batch.drain_user_copies();
99}

101/// Represents an allocated Stack.
102/// Implements Drop to automatically free pages.
103///
104/// Placement: a kernel stack is one contiguous run identity-mapped at its
105/// physical address (VA == PA), guard frames included and unmapped. A user
106/// stack is a contiguous run of only its usable frames, mapped USER RW at
107/// `base + guard_pages * PAGE_SIZE` inside a stack-arena slot
108/// (`loader::va_alloc::alloc_stack_va`); its guards are VA with neither a leaf
109/// nor a frame behind them, and its frames keep their kernel alias. Either way
110/// the run backs the top `run.pages()` pages of `[base, top)`.
111#[derive(Debug)]
112pub struct Stack {
113    /// Base address (lowest address) of the allocated range.
114    /// This includes the guard pages at the bottom.
115    pub base: VAddr,
116    /// Number of usable pages (excluding guard pages).
117    pub pages: usize,
118    /// Number of verified-unmapped pages below the usable range.
119    pub guard_pages: usize,
120    /// Top of the stack (initial SP).
121    pub top: VAddr,
122    /// The frames (a kernel stack's guards included); `None` only once retired
123    /// (or for a host test's stack, which owns no frames).
124    run: Option<FrameRun>,
125    /// `base` is a stack-arena slot (a user stack), returned with the frames.
126    in_arena: bool,
127    #[cfg(feature = "p01-trace")]
128    pub(crate) p01_obligation: Option<crate::p01_resources::Obligation>,
129    /// Test-only lifetime registration for a kernel stack selected by a
130    /// native-domain private root.
131    #[cfg(all(
132        feature = "native-domains",
133        feature = "test-hooks",
134        target_arch = "riscv64"
135    ))]
136    supervisor_registration: Option<crate::memory::domain_supervisor_registry::SupervisorRangeId>,
137}

143    /// # Errors

189    /// Internal allocation logic.
190    ///
191    /// Contract: this either returns a stack whose guard page is *verified* absent
192    /// from the page tables, or it returns an error having released every frame it
193    /// took. There is deliberately no third outcome. An unguarded stack is not a
194    /// degraded stack — in a single address space the frame below it belongs to
195    /// another cell, so an overflow that should have trapped instead corrupts a
196    /// neighbour with no fault and no log, and the victim dies later somewhere
197    /// unrelated. A log line at allocation time is not a mitigation: nothing reads
198    /// it in the microseconds before the overflow.
199    fn allocate(
200        pages: usize,
201        guard_pages: usize,
202        user_mode: bool,
203        bill: RunBill,
204    ) -> Result<Self, ViError> {
205        let total_pages = pages.checked_add(guard_pages).ok_or(ViError::OutOfMemory)?;
206
207        // Remapped usable pages and unmapped guards are invalidated on every
208        // hart before the stack is handed out (or, on failure, before its run
209        // is freed): the batch finishes after the frame allocator is released.
210        let mut batch = Batch::new();
211        let allocated = crate::memory::frame::with(|allocator| {
212            if user_mode {
213                Self::allocate_user_in(allocator, &mut batch, pages, guard_pages, total_pages, bill)
214            } else {
215                Self::allocate_kernel_in(
216                    allocator,
217                    &mut batch,
218                    pages,
219                    guard_pages,
220                    total_pages,
221                    bill,
222                )
223            }
224        })
225        .unwrap_or_else(|| {
226            error!(
227                "Stack alloc failed: frame allocator unavailable (user={}, pages={})",
228                user_mode, total_pages
229            );
230            Err(ViError::OutOfMemory)
231        });
232        let flushed = batch.complete();
233        let stack = allocated?;
234        if !flushed.is_confirmed() {
235            // A remote hart may still translate a guard or the old rights of a
236            // usable page: no stack, and its own teardown quarantines the run.
237            error!(
238                "Stack alloc refused: invalidation of run 0x{:X} unconfirmed",
239                stack.base
240            );
241            drop(stack);
242            return Err(ViError::NotSupported);
243        }
244        Ok(stack)
245    }

247    /// A kernel stack, with the frame allocator held: one contiguous run of
248    /// `total_pages` frames identity-mapped kernel RW at its physical address,
249    /// its bottom `guard_pages` frames unmapped. Every changed translation, and
250    /// on failure the whole run, goes to `batch`.
251    fn allocate_kernel_in(
252        allocator: &mut FrameAllocator,
253        batch: &mut Batch,
254        pages: usize,
255        guard_pages: usize,
256        total_pages: usize,
257        bill: RunBill,
258    ) -> Result<Self, ViError> {
259        // A kernel stack stays ONE contiguous run at VA == PA: a secondary RV64
260        // hart runs `smp_hart_entry` on its kernel stack with translation off
261        // (`_secondary_entry` takes the stack top from the SBI opaque), so that
262        // address must be valid before and after paging is on. User stacks map
263        // from the stack arena instead (`allocate_user_in`).
264        //
265        // The cost: this can fail on a fragmented allocator while plenty of memory is
266        // free, and no amount of lazy commit helps — a guard page cannot make VA != PA.
267        // So callers must read `OutOfMemory` here as "no run this long exists right
268        // now", which is recoverable, and never as grounds to panic.
269        let base_frame = allocator.allocate_contiguous(total_pages).ok_or_else(|| {
270            error!(
271                "Stack alloc failed: no contiguous run (user=false, pages={}, bytes={})",
272                total_pages,
273                total_pages * PAGE_SIZE
274            );
275            ViError::OutOfMemory
276        })?;
277
278        let base_addr = base_frame; // Identity Map
279                                    // Owned from here: every failure below retires it into `batch`, and a
280                                    // run dropped un-retired is leaked (counted), never freed.
281        let run = FrameRun::mapped(
282            base_frame,
283            total_pages,
284            bill,
285            RunKind::Leaf,
286            KernelAlias::of_identity_remap(base_frame),
287        );
288
289        // 2. Map Pages
290        // Bottom guard frames are left unmapped; usable frames begin after them.
291
292        let usable_start_idx = guard_pages;
293
294        // SAS identity map: all RAM is already identity-mapped RW+NX by
295        // init_kernel_paging. The usable pages are re-mapped below, then every
296        // guard frame is unmapped after the loop so an overflow traps.
297        let flags = Flags::from_bits(
298            Flags::VALID | Flags::READ | Flags::WRITE | Flags::ACCESSED | Flags::DIRTY,
299        );
300
301        for i in usable_start_idx..total_pages {
302            let addr = base_addr + (i * PAGE_SIZE);
303            if paging::map_page(allocator, batch, addr, addr, flags).is_err() {
304                // The run is not yet owned by a Stack, so nothing will drop it —
305                // release it here or the frames are lost until reboot.
306                error!("Stack alloc failed: cannot map page 0x{:X}", addr);
307                retire_run(batch, run, base_addr, None);
308                return Err(ViError::OutOfMemory);
309            }
310        }
311
312        // Guard pages: drop the bottom frames' pre-existing identity mappings so a
313        // stack overflow (a write below base_addr+PAGE_SIZE) faults instead of
314        // silently corrupting the neighbouring frame. The spawn paths zero only
315        // the usable pages (skipping base_addr), so nothing legitimately writes to
316        // the guard frames. They stay owned by this Stack (freed in Drop); only
317        // their PTEs are cleared. unmap_page locks KERNEL_ROOT (not FRAME_ALLOCATOR,
318        // which we still hold) — no deadlock.
319        //
320        // The guard is verified by translation, not by the unmap's return code: on
321        // some arches `unmap_page` reports success for a page it never touched
322        // (paging root absent, or the PTE was not a 4 KiB leaf). Asking the page
323        // tables whether the frame still resolves is the only answer that matches
324        // what the hardware will do on overflow.
325        for guard_index in 0..guard_pages {
326            let guard_addr = base_addr + (guard_index * PAGE_SIZE);
327            let unmap_ok = paging::unmap_page(batch, guard_addr).is_ok();
328            paging::tlb_flush_all();
329            if !unmap_ok || paging::virt_to_phys(guard_addr).is_some() {
330                error!(
331                    "Stack alloc refused: guard frame 0x{:X} still mapped (unmap_ok={})",
332                    guard_addr, unmap_ok
333                );
334                retire_run(batch, run, base_addr, None);
335                return Err(ViError::NotSupported);
336            }
337        }
338
339        // Calculate Top (Stack grows down)
340        // Top is at the END of the allocated range.
341        let top = base_addr + (total_pages * PAGE_SIZE);
342
343        // The guard VAs are now unmapped but owned: record the run so no cell
344        // image can be placed over them (see `reservations`). FRAME_ALLOCATOR,
345        // held since before the unmap, also excludes the ELF loader until then.
346        if let Err(error) = STACK_RESERVATIONS.insert(base_addr, top) {
347            error!("Stack alloc refused: run 0x{:X} not recorded", base_addr);
348            retire_run(batch, run, base_addr, None);
349            return Err(error);
350        }
351
352        #[cfg(all(
353            feature = "native-domains",
354            feature = "test-hooks",
355            target_arch = "riscv64"
356        ))]
357        let supervisor_registration = if !crate::memory::domain_supervisor_registry::is_active() {
358            None
359        } else {
360            match crate::memory::domain_supervisor_registry::register(
361                base_addr + (guard_pages * PAGE_SIZE),
362                top,
363                crate::memory::domain_supervisor_registry::SupervisorRangeKind::KernelStack,
364                crate::memory::domain_supervisor_registry::SupervisorRangeOwner::TaskStack,
365            ) {
366                Ok(id) => Some(id),
367                Err(()) => {
368                    let hold = VaHold {
369                        release: release_stack_reservation,
370                        start: base_addr,
371                        end: top,
372                    };
373                    retire_run(batch, run, base_addr, Some(hold));
374                    return Err(ViError::NotSupported);
375                }
376            }
377        };
378        trace!(
379            "Allocated Stack: Base=0x{:X}, Top=0x{:X}, Pages={}, Guards={}, User=false",
380            base_addr,
381            top,
382            pages,
383            guard_pages
384        );
385
386        Ok(Stack {
387            base: base_addr,
388            pages,
389            guard_pages,
390            top,
391            run: Some(run),
392            in_arena: false,
393            #[cfg(feature = "p01-trace")]
394            p01_obligation: None,
395            #[cfg(all(
396                feature = "native-domains",
397                feature = "test-hooks",
398                target_arch = "riscv64"
399            ))]
400            supervisor_registration,
401        })
402    }

404    /// A user stack, with the frame allocator held: a stack-arena slot, and one
405    /// contiguous run of `pages` frames mapped USER RW right above the slot's
406    /// `guard_pages` guard pages. The guards are never mapped and have no
407    /// frame; the frames' kernel alias is never touched (`KernelAlias::Intact`).
408    /// Every new translation, and on failure the run and the slot, goes to
409    /// `batch`; the slot returns only after the batch's flush (`VaHold`).
410    fn allocate_user_in(
411        allocator: &mut FrameAllocator,
412        batch: &mut Batch,
413        pages: usize,
414        guard_pages: usize,
415        total_pages: usize,
416        bill: RunBill,
417    ) -> Result<Self, ViError> {
418        use crate::loader::va_alloc;
419
420        if total_pages > va_alloc::STACK_VA_STRIDE / PAGE_SIZE {
421            error!(
422                "Stack alloc refused: {} pages exceed a {}-byte stack slot",
423                total_pages,
424                va_alloc::STACK_VA_STRIDE
425            );
426            return Err(ViError::OutOfMemory);
427        }
428        let base = va_alloc::alloc_stack_va().ok_or_else(|| {
429            error!("Stack alloc failed: every stack arena slot is held");
430            ViError::OutOfMemory
431        })?;
432        let top = base + total_pages * PAGE_SIZE;
433        // Recorded before anything is mapped, so every failure below returns
434        // the slot through the run's `VaHold`, after the invalidation.
435        if let Err(error) = STACK_RESERVATIONS.insert(base, top) {
436            error!("Stack alloc refused: slot 0x{:X} not recorded", base);
437            va_alloc::free_stack_va(base);
438            return Err(error);
439        }
440        // Physically contiguous only to keep Phase 16's one-run release contract.
441        let Some(base_frame) = allocator.allocate_contiguous(pages) else {
442            error!(
443                "Stack alloc failed: no contiguous run (user=true, pages={}, bytes={})",
444                pages,
445                pages * PAGE_SIZE
446            );
447            // Nothing was mapped in the slot: no translation to wait out.
448            release_stack_slot(base, top);
449            return Err(ViError::OutOfMemory);
450        };
451        // Owned from here: every failure below retires it into `batch`.
452        let run = FrameRun::mapped(base_frame, pages, bill, RunKind::Leaf, KernelAlias::Intact);
453        let hold = VaHold {
454            release: release_stack_slot,
455            start: base,
456            end: top,
457        };
458        let usable = base + guard_pages * PAGE_SIZE;
459        let flags = Flags::from_bits(
460            Flags::VALID
461                | Flags::READ
462                | Flags::WRITE
463                | Flags::USER
464                | Flags::ACCESSED
465                | Flags::DIRTY,
466        );
467        for i in 0..pages {
468            let (va, frame) = (usable + i * PAGE_SIZE, base_frame + i * PAGE_SIZE);
469            if paging::map_page(allocator, batch, va, frame, flags).is_err() {
470                error!("Stack alloc failed: cannot map page 0x{:X}", va);
471                retire_run(batch, run, usable, Some(hold));
472                return Err(ViError::OutOfMemory);
473            }
474        }
475        // The guards were never mapped: verify by translation, as for a kernel
476        // stack, so a stale leaf in a reused slot can never pass as a guard.
477        for guard_index in 0..guard_pages {
478            let guard_addr = base + guard_index * PAGE_SIZE;
479            if paging::virt_to_phys(guard_addr).is_some() {
480                error!("Stack alloc refused: guard 0x{:X} is mapped", guard_addr);
481                retire_run(batch, run, usable, Some(hold));
482                return Err(ViError::NotSupported);
483            }
484        }
485        trace!(
486            "Allocated Stack: Base=0x{:X}, Top=0x{:X}, Frames=0x{:X}, Pages={}, Guards={}, User=true",
487            base,
488            top,
489            base_frame,
490            pages,
491            guard_pages
492        );
493
494        Ok(Stack {
495            base,
496            pages,
497            guard_pages,
498            top,
499            run: Some(run),
500            in_arena: true,
501            #[cfg(feature = "p01-trace")]
502            p01_obligation: None,
503            #[cfg(all(
504                feature = "native-domains",
505                feature = "test-hooks",
506                target_arch = "riscv64"
507            ))]
508            supervisor_registration: None,
509        })
510    }

512    /// Physical bytes this stack holds: its run, guard frames included for a
513    /// kernel stack (a user stack's guards have no frame).
514    pub fn allocated_bytes(&self) -> usize {
515        self.run
516            .as_ref()
517            .map_or((self.pages + self.guard_pages) * PAGE_SIZE, |run| {
518                run.pages() * PAGE_SIZE
519            })
520    }

522    /// The run's physical base and frame count, while the stack holds it.
523    pub fn frames(&self) -> Option<(usize, usize)> {
524        self.run.as_ref().map(|run| (run.base(), run.pages()))
525    }

527    /// Physical address of the lowest usable byte: the run backs the top
528    /// `run.pages()` pages of `[base, top)`.
529    pub fn usable_phys(&self) -> Option<usize> {
530        self.run
531            .as_ref()
532            .map(|run| run.base() + (run.pages() - self.pages) * PAGE_SIZE)
533    }

535    /// Zero the usable pages through the frames' kernel alias
536    /// (`frame::phys_to_virt`), never through the stack's own VA: a user
537    /// stack's VA is a USER mapping, which the kernel writes only inside a
538    /// user-access window. Frames come back from the allocator holding their
539    /// last owner's bytes, so every spawn path zeroes a stack before it runs.
540    pub fn zero_usable(&self) {
541        let Some(usable) = self.usable_phys() else {
542            return;
543        };
544        // SAFETY: the run is this stack's alone and physically contiguous, and
545        // its kernel alias is writable and contiguous: the identity map on
546        // RV64/AArch64 (a user stack's frames keep the free-frame alias, a
547        // kernel stack's usable pages are its own kernel RW leaves), the HHDM
548        // on x86_64.
549        unsafe {
550            core::ptr::write_bytes(
551                crate::memory::frame::phys_to_virt(usable) as *mut u8,
552                0,
553                self.usable_bytes(),
554            );
555        }
556    }

558    /// Task-owned usable stack bytes, excluding any kernel-only guard reservation.
559    pub fn usable_bytes(&self) -> usize {
560        self.pages * PAGE_SIZE
561    }

563    /// Are this stack's frames billed to a cell's quota (a thread's stacks)?
564    pub fn is_billed(&self) -> bool {
565        self.run
566            .as_ref()
567            .is_some_and(|run| run.charge().account.is_some())
568    }

570    /// Lowest mapped byte in the usable stack range.
571    pub fn usable_start(&self) -> usize {
572        self.base + (self.guard_pages * PAGE_SIZE)
573    }
574    /// Whether `[ptr, ptr + len)` stays within this task's mapped stack bytes.
575    pub fn contains_usable_range(&self, ptr: usize, len: usize) -> bool {
576        let Some(end) = ptr.checked_add(len) else {
577            return false;
578        };
579        ptr >= self.usable_start() && end <= self.top
580    }

582    #[cfg(feature = "test-hooks")]
583    /// Prime the usable stack range with a sentinel pattern so later scans can
584    /// estimate the deepest downward growth without adding a runtime ABI.
585    /// Writes through the frames' kernel alias, never the stack's own VA: a
586    /// user stack's VA is a USER mapping (SMAP on x86_64).
587    pub fn test_hook_prime_watermark(&self) {
588        assert!(paging::virt_to_phys(self.usable_start()).is_some());
589        assert!(paging::virt_to_phys(self.top - 1).is_some());
590        let Some(usable) = self.usable_phys() else {
591            return;
592        };
593        // SAFETY: the run is this stack's alone and its kernel alias is
594        // writable and contiguous (see `zero_usable`); this test hook owns the
595        // stack before the task can run.
596        unsafe {
597            core::ptr::write_bytes(
598                crate::memory::frame::phys_to_virt(usable) as *mut u8,
599                STACK_WATERMARK_PATTERN,
600                self.usable_bytes(),
601            );
602        }
603    }
604
605    #[cfg(feature = "test-hooks")]
606    /// Return the deepest observed stack footprint in bytes since the sentinel
607    /// pattern was primed, read through the frames' kernel alias.
608    pub fn test_hook_watermark_bytes(&self) -> usize {
609        assert!(paging::virt_to_phys(self.usable_start()).is_some());
610        assert!(paging::virt_to_phys(self.top - 1).is_some());
611        let Some(usable_pa) = self.usable_phys() else {
612            return 0;
613        };
614        let usable = self.usable_bytes();
615        let start = crate::memory::frame::phys_to_virt(usable_pa) as *const u8;
616        let mut untouched_prefix = 0usize;
617        while untouched_prefix < usable {
618            // SAFETY: the scan stays within the run's usable bytes, read
619            // through its kernel alias. Volatile reads observe the live
620            // watermark.
621            let byte = unsafe { core::ptr::read_volatile(start.add(untouched_prefix)) };
622            if byte != STACK_WATERMARK_PATTERN {
623                break;
624            }
625            untouched_prefix += 1;
626        }
627        usable.saturating_sub(untouched_prefix)
628    }
629}

631#[cfg(feature = "test-hooks")]
632pub fn stack_probe_self_test() -> bool {
633    let stack = match Stack::new_kernel(2) {
634        Ok(stack) => stack,
635        Err(_) => return false,
636    };
637    if stack.guard_pages != STACK_GUARD_PAGES {
638        return false;
639    }
640    for guard_index in 0..stack.guard_pages {
641        let guard_addr = stack.base + (guard_index * PAGE_SIZE);
642        if paging::virt_to_phys(guard_addr).is_some() {
643            return false;
644        }
645    }
646    let deliberate_overflow_target = stack.usable_start().saturating_sub(8);
647    if paging::virt_to_phys(deliberate_overflow_target).is_some() {
648        return false;
649    }
650    stack.test_hook_prime_watermark();
651    if stack.test_hook_watermark_bytes() != 0 {
652        return false;
653    }
654    let probe_bytes = 96usize;
655    unsafe {
656        core::ptr::write_bytes((stack.top - probe_bytes) as *mut u8, 0x11, probe_bytes);
657    }
658    stack.test_hook_watermark_bytes() == probe_bytes && user_stack_probe()
659}

661/// A user stack is mapped from the stack arena, not at its physical address:
662/// its usable VA translates to its run but not to itself, both guards and the
663/// overflow target below them are unmapped, and the stack's top word
664/// (`top - 8`) translates to the run's last usable byte; a word written there
665/// through the frames' kernel alias reads back through the same alias. The
666/// kernel never dereferences the user stack VA (SMAP on x86_64).
667#[cfg(feature = "test-hooks")]
668fn user_stack_probe() -> bool {
669    const MARK: u64 = 0x5354_4b41_5245_4e41;
670    let Ok(stack) = Stack::new_user(2) else {
671        return false;
672    };
673    let (Some(usable_pa), Some((_, frames))) = (stack.usable_phys(), stack.frames()) else {
674        return false;
675    };
676    let usable = stack.usable_start();
677    let off_identity = stack.in_arena
678        && frames == stack.pages
679        && paging::virt_to_phys(usable) == Some(usable_pa)
680        && usable_pa != usable;
681    let guarded = (0..stack.guard_pages)
682        .all(|guard| paging::virt_to_phys(stack.base + guard * PAGE_SIZE).is_none())
683        && paging::virt_to_phys(usable - 8).is_none();
684    let top_pa = usable_pa + stack.usable_bytes() - 8;
685    let alias = crate::memory::frame::phys_to_virt(top_pa);
686    // SAFETY: the top 8 bytes of this stack's run, through its kernel alias;
687    // the stack is owned here and runs no task.
688    unsafe { core::ptr::write_volatile(alias as *mut u64, MARK) };
689    // SAFETY: the same word through the same alias, still owned here.
690    let landed = paging::virt_to_phys(stack.top - 8) == Some(top_pa)
691        && unsafe { core::ptr::read_volatile(alias as *const u64) } == MARK;
692    if !(off_identity && guarded && landed) {
693        error!(
694            "stack-probe: user stack base=0x{:X} usable_pa=0x{:X} off_identity={} guarded={} landed={}",
695            stack.base, usable_pa, off_identity, guarded, landed
696        );
697    }
698    off_identity && guarded && landed
699}

701/// Test-hooks: a free frame's kernel alias is `paging::FREE_FRAME_ALIAS`
702/// (`valid && write && !exec && !user`, translating to itself) for a frame
703/// fresh from the boot map, for the frames of a `Removed` run once freed (a
704/// kernel stack, dropped: `KernelBackend::normalise` restored the alias), and
705/// throughout the life of a user stack's frames, whose alias is never taken.
706///
707/// `None` where RAM has no identity alias (x86_64 reaches it through the HHDM).
708#[cfg(feature = "test-hooks")]
709pub fn free_frame_nx_self_test() -> Option<bool> {
710    fn free_alias(frame: usize) -> bool {
711        let Some(root) = *paging::KERNEL_ROOT.lock() else {
712            return false;
713        };
714        // SAFETY: the live kernel root; its tables outlive this walk.
715        unsafe { crate::hal::paging::leaf(root, frame) }.is_some_and(|leaf| {
716            leaf.pa == frame && leaf.valid && leaf.write && !leaf.exec && !leaf.user
717        })
718    }
719    fn alias_of(stack: &Stack) -> Option<KernelAlias> {
720        stack.run.as_ref().map(FrameRun::alias)
721    }
722
723    let Some(fresh) = crate::memory::frame::allocate_frame() else {
724        return Some(false);
725    };
726    if crate::memory::frame::phys_to_virt(fresh) != fresh {
727        crate::memory::frame::deallocate_frame(fresh);
728        return None;
729    }
730    let fresh_ok = free_alias(fresh);
731    crate::memory::frame::deallocate_frame(fresh);
732
733    let Ok(kernel) = Stack::new_kernel(1) else {
734        return Some(false);
735    };
736    let removed = alias_of(&kernel) == Some(KernelAlias::Removed);
737    let (guard, usable) = (kernel.base, kernel.usable_start());
738    // While the stack lives its guard frame is unmapped.
739    let live = paging::virt_to_phys(guard).is_none();
740    drop(kernel);
741    let restored = free_alias(guard) && free_alias(usable);
742
743    let Ok(user) = Stack::new_user(1) else {
744        return Some(false);
745    };
746    let Some((frame, _)) = user.frames() else {
747        return Some(false);
748    };
749    let intact = alias_of(&user) == Some(KernelAlias::Intact) && free_alias(frame);
750    drop(user);
751    let kept = free_alias(frame);
752
753    let ok = fresh_ok && removed && live && restored && intact && kept;
754    if !ok {
755        error!(
756            "[selftest] FREE-FRAME-NX: fresh={} removed={} live={} restored={} intact={} kept={}",
757            fresh_ok, removed, live, restored, intact, kept
758        );
759    }
760    Some(ok)
761}

763impl Stack {
764    /// The teardown up to its flush: unregister, unpublish every page, and
765    /// retire the run (still holding its reservation) into `batch`.
766    fn retire_run_into(&mut self, batch: &mut Batch) {
767        #[cfg(all(
768            feature = "native-domains",
769            feature = "test-hooks",
770            target_arch = "riscv64"
771        ))]
772        if let Some(id) = self.supervisor_registration.take() {
773            let unregistered = crate::memory::domain_supervisor_registry::unregister(id);
774            debug_assert!(unregistered);
775        }
776        trace!("Dropping Stack at 0x{:X}", self.base);
777
778        let Some(run) = self.run.take() else {
779            return;
780        };
781        let va = self.top - run.pages() * PAGE_SIZE;
782        unpublish_user_frames(batch, va, run.pages());
783        let release = if self.in_arena {
784            release_stack_slot
785        } else {
786            release_stack_reservation
787        };
788        let hold = VaHold {
789            release,
790            start: self.base,
791            end: self.top,
792        };
793        retire_run(batch, run, va, Some(hold));
794    }
795
796    /// Tear this stack down as part of a larger operation: its run joins
797    /// `batch`, whose one shared flush covers every owner retired into it,
798    /// and is freed — or quarantined, reservation included — when the batch
799    /// finishes. A p01 obligation the stack still holds is dropped unfinished.
800    #[cfg(any(feature = "test-hooks", all(feature = "native-domains", cellos_wide)))]
801    fn retire_without_drop(self, batch: &mut Batch) {
802        let mut stack = core::mem::ManuallyDrop::new(self);
803        stack.retire_run_into(batch);
804    }
805
806    /// [`Self::retire_without_drop`] for the x86 silent-CPU fixture.
807    #[cfg(feature = "test-hooks")]
808    // reason: only the x86 silent-CPU fixture retires owners into a shared batch.
809    #[allow(dead_code)]
810    pub(crate) fn retire_into(self, batch: &mut Batch) {
811        #[cfg(feature = "p01-trace")]
812        debug_assert!(
813            self.p01_obligation.is_none(),
814            "a stack with a p01 obligation is retired only by its drop"
815        );
816        self.retire_without_drop(batch);
817    }
818
819    /// Tear a thread's stacks down after their unmap from a domain root could
820    /// not confirm private tag `asid`: a remote hart may still resolve the
821    /// stale private entry, so the runs are quarantined behind that tag (and a
822    /// fresh epoch from every remote) whatever the shared flush says, and the
823    /// reaper frees them — reservations with them — once it is confirmed.
824    /// Before this phase they were forgotten and never reused (census F5).
825    ///
826    /// The caller settles any p01 obligation first; one still held is dropped
827    /// unfinished, as the old `mem::forget` did.
828    #[cfg(all(feature = "native-domains", cellos_wide))]
829    pub(crate) fn retire_behind_tag(asid: usize, stacks: [Stack; 2]) {
830        let mut batch = Batch::new();
831        for stack in stacks {
832            stack.retire_without_drop(&mut batch);
833        }
834        batch.await_tag(asid);
835        let _ = batch.complete();
836    }

838    /// A stack descriptor that owns no frames, for host tests of range checks.
839    #[cfg(test)]
840    pub(crate) fn for_tests(base: VAddr, pages: usize, guard_pages: usize, top: VAddr) -> Self {
841        Self {
842            base,
843            pages,

845            top,
846            run: None,
847            in_arena: false,

850            #[cfg(all(
851                feature = "native-domains",
852                feature = "test-hooks",
853                target_arch = "riscv64"
854            ))]
855            supervisor_registration: None,
856        }
857    }

859    /// A descriptor over synthetic frames billed with `charge`, for host tests
860    /// that read [`Self::is_billed`]. The test must forget it: its frames are

876impl Drop for Stack {
877    fn drop(&mut self) {
878        // One batch for the whole run: one shared invalidation, not one per
879        // page. A sibling thread's Sas copy may still target this user stack,
880        // so the batch drains copy windows after the flush and before the free.
881        // An unconfirmed flush quarantines the run (and keeps its reservation)
882        // until every hart acknowledges.
883        let mut batch = Batch::new();
884        #[cfg(feature = "p01-trace")]
885        let frames = self.frames();
886        self.retire_run_into(&mut batch);
887        #[allow(unused_variables)] // reason: only the p01-trace ledger reads it
888        let flushed = batch.complete();
889        #[cfg(feature = "p01-trace")]
890        if flushed.is_confirmed() {
891            crate::p01_resources::finish(self.p01_obligation.take(), 1);
892        }
893        #[cfg(feature = "p01-trace")]
894        if self.p01_obligation.is_some() {
895            let owner = crate::p01_resources::obligation_owner(self.p01_obligation);
896            if let (Some(_), Some((base, pages))) = (owner, frames) {
897                crate::p01_retained::begin(
898                    owner,
899                    base,
900                    pages,
901                    crate::p01_retained::Reason::TaskStackForget,
902                );
903            }
904            crate::p01_resources::finish(self.p01_obligation.take(), 6);
905        }
906    }
