1//! IOMMU common API — three-phase DMA isolation.
2//!
3//! Phase 1 `init()`             — probe hardware, allocate page tables, stay passthrough.
4//! Phase 2 `map_dma()`          — drivers register each DMA buffer's physical range.
5//! Phase 3 `activate_isolation()` — switch from passthrough to enforced page-table mode.
6//!
7//! Call order in `main.rs`:
8//!   `iommu::init()` → driver DMA allocs (call `map_dma()`) → `iommu::activate_isolation()`
9
10use core::sync::atomic::{AtomicBool, Ordering};
11
12#[derive(Clone, Copy, Debug, Eq, PartialEq)]
13pub enum DmaMapResult {
14    Mapped(u64),
15    Rejected,
16    /// Hardware may observe the published mapping; its pin must be retained.
17    PublishedUnconfirmed,
18}
19
20/// Outcome of a per-range DMA teardown.
21///
22/// Mirrors [`DmaMapResult`]: a teardown that zeroed leaves the hardware may
23/// still translate is reported as unconfirmed, never as done — the caller must
24/// keep the frames quarantined until the invalidation is retried.
25#[derive(Clone, Copy, Debug, Eq, PartialEq)]
26pub enum DmaUnmapResult {
27    /// Nothing was mapped in the range: no teardown was needed.
28    NothingMapped,
29    /// Every cleared leaf's translation is gone from hardware.
30    Unmapped { pages: usize },
31    /// Leaves were zeroed but the IOMMU did not acknowledge the invalidation.
32    PublishedUnconfirmed { pages: usize },
33}
34
35/// Classify a mapping whose device context was written before invalidation.
36///
37/// Either a command-queue publication failure or a missing IOFENCE
38/// acknowledgement leaves hardware visibility uncertain and keeps the pin.
39#[cfg_attr(not(target_arch = "riscv64"), allow(dead_code))]
40pub(crate) const fn classify_dma_publication(
41    iova: u64,
42    invalidation_published: bool,
43    fence_acknowledged: bool,
44) -> DmaMapResult {
45    if invalidation_published && fence_acknowledged {
46        DmaMapResult::Mapped(iova)
47    } else {
48        DmaMapResult::PublishedUnconfirmed
49    }
50}

52static IOMMU_ISOLATED: AtomicBool = AtomicBool::new(false);
53/// Set when `pcie_ecam::init()` is removed from the boot path.
54/// `try_deferred_init()` (called from `RegisterPciDevice` handler) checks this
55/// and runs `init()` once PCI_DEVICES is populated with the IOMMU device entry.
56static IOMMU_DEFERRED: AtomicBool = AtomicBool::new(false);

58/// Phase 1: probe IOMMU hardware and allocate isolation data structures.
59///
60/// Must be called after `pcie_ecam::init()` and before any DMA allocation.
61/// Does NOT enable enforcement yet — hardware stays in passthrough mode.
62pub fn init() {
63    #[cfg(target_arch = "riscv64")]
64    super::iommu_riscv::init_hw();
65    #[cfg(target_arch = "x86_64")]
66    super::iommu_x86::init_hw();
67}

69/// Register a DMA physical range or authorize identity DMA when x86 has no
70/// remapping hardware. A present-but-inactive remapper rejects the request.
71#[inline]
72pub fn map_dma(phys: u64, size: usize) -> Option<u64> {
73    match map_dma_for_cell(0, 0, phys, size) {
74        DmaMapResult::Mapped(iova) => Some(iova),
75        DmaMapResult::Rejected | DmaMapResult::PublishedUnconfirmed => None,
76    }
77}
78
79/// Register `[phys, phys+size)` for Cell `tid` owning device `bdf`.
80///
81/// Distinguishes a clean rejection from a mapping that was published before an
82/// invalidation timeout; the latter requires the caller to retain the DMA pin.
83pub fn map_dma_for_cell(tid: u64, bdf: u32, phys: u64, size: usize) -> DmaMapResult {
84    if size == 0 {
85        return DmaMapResult::Rejected;
86    }
87    if !is_active() {
88        #[cfg(target_arch = "x86_64")]
89        return if super::iommu_x86::is_present() {
90            DmaMapResult::Rejected
91        } else {
92            // No remapping hardware exists; ownership/pinning still succeeded,
93            // and the machine's DMA contract is identity addressing.
94            DmaMapResult::Mapped(phys)
95        };
96        #[cfg(not(target_arch = "x86_64"))]
97        return DmaMapResult::Rejected;
98    }
99    #[cfg(target_arch = "riscv64")]
100    let mapped = super::iommu_riscv::map_range_for_cell(tid, bdf, phys, size);
101    #[cfg(target_arch = "x86_64")]
102    let mapped = super::iommu_x86::map_range_for_cell(tid, bdf, phys, size);
103    #[cfg(not(any(target_arch = "riscv64", target_arch = "x86_64")))]
104    let mapped = {
105        let _ = (tid, bdf, phys, size);
106        DmaMapResult::Rejected
107    };
108    mapped
109}

111/// Tear down `[iova, iova+size)` from Cell `tid`'s DMA domain.
112///
113/// Zeros the leaf entries and invalidates the translations covering them, so a
114/// device the Cell authorised faults on its next access to the range. Cell-death
115/// teardown uses [`revoke_dma_for_cell`] instead: same invalidation, but it also
116/// drops the requester entries and the domain itself.
117#[inline]
118pub fn unmap_dma(tid: u64, iova: u64, size: usize) -> DmaUnmapResult {
119    #[cfg(target_arch = "riscv64")]
120    {
121        super::iommu_riscv::unmap_range_for_cell(tid, iova, size)
122    }
123    #[cfg(target_arch = "x86_64")]
124    {
125        super::iommu_x86::unmap_range_for_cell(tid, iova, size)
126    }
127    #[cfg(not(any(target_arch = "riscv64", target_arch = "x86_64")))]
128    {
129        let _ = (tid, iova, size);
130        DmaUnmapResult::NothingMapped
131    }
132}
133
134/// Tear down `tid`'s whole DMA domain: clear its requester entries, zero its
135/// DDT/context, and invalidate its cached translations.
136///
137/// The single domain-teardown path, reached from two callers: Cell death (the
138/// task-exit sequence, before grant frames are released) and a runtime revoke
139/// of `pcie_driver` — a Cell that loses DMA authority mid-life gets exactly the
140/// teardown it would get by dying.
141///
142/// Returns `true` only when hardware acknowledged teardown. Callers must keep
143/// pinned frames quarantined when it returns `false`.
144pub fn revoke_dma_for_cell(tid: u64) -> bool {
145    #[cfg(target_arch = "riscv64")]
146    {
147        super::iommu_riscv::unmap_cell(tid)
148    }
149    #[cfg(target_arch = "x86_64")]
150    {
151        super::iommu_x86::unmap_cell_domain(tid)
152    }
153    #[cfg(not(any(target_arch = "riscv64", target_arch = "x86_64")))]
154    {
155        let _ = tid;
156        true
157    }
158}

160/// Phase 3: switch IOMMU from passthrough to page-table enforcement.
161///
162/// On RISC-V: writes DDTP with MODE=1LVL + pre-built Sv39 DDT → faults any
163///   IOVA not in a registered DMA range.
164/// On x86_64: fills VT-d context entries with TT=TRANSLATED+SLPT, enables TE.
165///
166/// Call after all driver DMA buffers are registered via `map_dma()`.
167pub fn activate_isolation() {
168    #[cfg(target_arch = "riscv64")]
169    super::iommu_riscv::activate();
170    #[cfg(target_arch = "x86_64")]
171    super::iommu_x86::activate();
172}
173
174/// Returns `true` once `activate_isolation()` has completed successfully.
175#[inline]
176pub fn is_active() -> bool {
177    IOMMU_ISOLATED.load(Ordering::Relaxed)
178}
179
180/// Mark DMA isolation as active. Called by arch backends on successful activation.
181#[cfg(any(target_arch = "riscv64", target_arch = "x86_64"))]
182pub(super) fn set_active() {
183    IOMMU_ISOLATED.store(true, Ordering::Relaxed);
184}

186/// Arm deferred IOMMU init.
187///
188/// Call from `main.rs` instead of `init()` when the Platform Cell owns PCIe
189/// enumeration. `try_deferred_init()` will call `init()` + `activate_isolation()`
190/// once the IOMMU device entry appears in `PCI_DEVICES` via `RegisterPciDevice`.
191pub fn set_deferred_init_pending() {
192    IOMMU_DEFERRED.store(true, Ordering::Relaxed);
193}
194
195/// Attempt IOMMU init if deferred and the IOMMU device has been registered.
196///
197/// Called from the `RegisterPciDevice` syscall handler after each device is added
198/// to `PCI_DEVICES`. Returns immediately if already initialized or not deferred.
199///
200/// Phase 3 (`activate_isolation`) runs immediately after `init_hw` here. Until
201/// activation succeeds, a present remapper causes DMA grants to fail closed;
202/// later mappings take effect through the active per-Cell backend.
203pub fn try_deferred_init() {
204    if !IOMMU_DEFERRED.load(Ordering::Relaxed) {
205        return;
206    }
207    if IOMMU_ISOLATED.load(Ordering::Relaxed) {
208        return;
209    }
210
211    // init() calls arch init_hw() which calls find_class() — succeeds only once
212    // the IOMMU device has been registered in PCI_DEVICES.
213    init();
214
215    // If init_hw() found the IOMMU hardware (BAR0 != 0), activate isolation.
216    // activate() is a no-op when init_hw() returned early (device not found yet).
217    activate_isolation();
218
219    if IOMMU_ISOLATED.load(Ordering::Relaxed) {
220        IOMMU_DEFERRED.store(false, Ordering::Relaxed);
221        log::info!("[iommu] deferred init complete — DMA isolation active");
222    }
223}
