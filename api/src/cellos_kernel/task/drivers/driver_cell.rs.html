1//! Driver Cell registration — tracks which cells own the block, NIC, and GPU roles.
2//!
3//! When a Tier-1 Driver Cell calls `sys_register_block_driver`,
4//! `sys_register_nic_driver`, or `sys_register_gpu_driver`, the kernel records
5//! its TID here. These registrations support service routing and interrupt
6//! ownership checks until the owning Cell exits.
7//!
8//! `0` means "no driver cell registered". Block can fall back to kernel-resident
9//! virtio-blk or MMC; NIC has no kernel fallback and is always a Driver Cell.
10//! GPU has no kernel fallback; compositor refuses to init until a GPU Cell registers.
11
12use crate::sync::Spinlock;
13use core::sync::atomic::{AtomicUsize, Ordering};
14
15/// TID of the registered block Driver Cell (0 = none; kernel virtio_blk/MMC is the fallback).
16pub static BLOCK_DRIVER_CELL: AtomicUsize = AtomicUsize::new(0);
17
18#[derive(Clone, Copy)]
19struct NicDriverState {
20    owner_tid: usize,
21    virtio_irq: u32,
22}
23
24impl NicDriverState {
25    const fn empty() -> Self {
26        Self {
27            owner_tid: 0,
28            virtio_irq: 0,
29        }
30    }
31}
32
33/// Registered NIC owner paired with the proven VirtIO IRQ, if any.
34static NIC_DRIVER_STATE: Spinlock<NicDriverState> = Spinlock::new(NicDriverState::empty());
35
36/// TID of the registered GPU Driver Cell (0 = none; no kernel GPU fallback).
37pub static GPU_DRIVER_CELL: AtomicUsize = AtomicUsize::new(0);
38
39const VIRTIO_DEVICE_ID_OFFSET: usize = 0x8;
40const VIRTIO_DEVICE_ID_NETWORK: u32 = 1;
41
42fn registered_virtio_irq_for_owner(tid: usize) -> Option<u32> {
43    crate::task::drivers::virtio_common::virtio_slots()
44        .find(|slot| {
45            if crate::resource_registry::lookup_mmio_owner(slot.base) != Some(tid) {
46                return false;
47            }
48            // SAFETY: this runs only in NIC registration context, never from ISR.
49            // `virtio_slots()` yields the kernel's enumerated, identity-mapped
50            // VirtIO MMIO bases, and the ownership check above proves this TID
51            // already claimed the slot via `RequestMmio` before we inspect the
52            // read-only `device_id` register at offset 0x8.
53            let device_id = unsafe {
54                core::ptr::read_volatile((slot.base + VIRTIO_DEVICE_ID_OFFSET) as *const u32)
55            };
56            device_id == VIRTIO_DEVICE_ID_NETWORK
57        })
58        .map(|slot| slot.irq)
59}

61/// Record `tid` as the active block driver.  Overwrites any previous registration.
62///
63/// Logged at `warn!` for the same reason as `set_input_cell`: Driver Cells are
64/// spawned by init AFTER the kernel drops its log level to Warn, this is a
65/// one-time boot-integrity event (the kernel now routes all sector I/O to this
66/// TID), and it is the marker the x86 nvme/nic integration tests assert on.
67pub fn register_block_driver(tid: usize) {
68    BLOCK_DRIVER_CELL.store(tid, Ordering::Release);
69    log::warn!("[driver_cell] block driver registered: tid={}", tid);
70}
71
72/// Record `tid` as the active NIC driver.  Overwrites any previous registration.
73/// `warn!` — see `register_block_driver`.
74pub fn register_nic_driver(tid: usize) {
75    let proven_irq = registered_virtio_irq_for_owner(tid).unwrap_or(0);
76
77    // Fail closed during replacement: an IRQ only becomes network-owned again
78    // after the replacement publishes both the owner and its proven IRQ together.
79    *NIC_DRIVER_STATE.lock() = NicDriverState {
80        owner_tid: tid,
81        virtio_irq: proven_irq,
82    };
83    log::warn!("[driver_cell] NIC driver registered: tid={}", tid);
84}
85
86/// Clear the block driver registration (called on cell exit/kill).
87pub fn deregister_block_driver(tid: usize) {
88    BLOCK_DRIVER_CELL
89        .compare_exchange(tid, 0, Ordering::AcqRel, Ordering::Relaxed)
90        .ok();
91}
92
93/// Clear the NIC driver registration (called on cell exit/kill).
94pub fn deregister_nic_driver(tid: usize) {
95    let mut state = NIC_DRIVER_STATE.lock();
96    if state.owner_tid == tid {
97        *state = NicDriverState::empty();
98    }
99}
100
101/// Returns true only for the cached VirtIO IRQ proven to belong to the active NIC.
102pub fn owns_registered_nic_irq(irq: u32) -> bool {
103    let state = NIC_DRIVER_STATE.lock();
104    state.owner_tid != 0 && irq != 0 && state.virtio_irq == irq
105}
106
107/// Record `tid` as the active GPU driver.  Overwrites any previous registration.
108pub fn register_gpu_driver(tid: usize) {
109    GPU_DRIVER_CELL.store(tid, Ordering::Release);
110    log::info!("[driver_cell] GPU driver registered: tid={}", tid);
111}
112
113/// Clear the GPU driver registration (called on cell exit/kill).
114pub fn deregister_gpu_driver(tid: usize) {
115    GPU_DRIVER_CELL
116        .compare_exchange(tid, 0, Ordering::AcqRel, Ordering::Relaxed)
117        .ok();
118}

120/// TID of the registered input service Cell (0 = unregistered).
121/// Set by the loader after spawning `/bin/input`; cleared on its death.
122pub static INPUT_CELL_TID: AtomicUsize = AtomicUsize::new(0);
123
124/// Register the input service cell.  Called by the loader after spawning `/bin/input`.
125///
126/// Logged at `warn!`, not `info!`: `/bin/input` spawns after the kernel drops
127/// its log level to Warn at end-of-early-boot, so an info line here is
128/// suppressed. This is a one-time boot-integrity event (the kernel now trusts
129/// this TID as the sole keyboard-event sink) — worth surfacing, and the marker
130/// the input-registration integration test asserts on.
131pub fn set_input_cell(tid: usize) {
132    INPUT_CELL_TID.store(tid, Ordering::Release);
133    log::warn!("[input] registered input service TID {}", tid);
134}
135
136/// Clear the input service registration if it matches `tid` (called on cell death).
137pub fn clear_input_cell_if(tid: usize) {
138    INPUT_CELL_TID
139        .compare_exchange(tid, 0, Ordering::AcqRel, Ordering::Relaxed)
140        .ok();
141}
142
143/// Clear every well-known Driver Cell role owned by `tid`.
144///
145/// This is the single lifecycle teardown entrypoint for task death, force-exit,
146/// and hotswap retirement. Each role clears only on an exact TID match so an
147/// unrelated replacement cannot lose its registration.
148pub fn deregister_all_for(tid: usize) {
149    clear_input_cell_if(tid);
150    deregister_block_driver(tid);
151    deregister_nic_driver(tid);
152    deregister_gpu_driver(tid);
153}

155#[cfg(test)]
156mod tests {
157    use super::{
158        deregister_all_for, NicDriverState, BLOCK_DRIVER_CELL, GPU_DRIVER_CELL, INPUT_CELL_TID,
159        NIC_DRIVER_STATE,
160    };
161    use core::sync::atomic::Ordering;
162
163    fn seed_roles(owner_tid: usize, nic_irq: u32) {
164        BLOCK_DRIVER_CELL.store(owner_tid, Ordering::Release);
165        *NIC_DRIVER_STATE.lock() = NicDriverState {
166            owner_tid,
167            virtio_irq: nic_irq,
168        };
169        GPU_DRIVER_CELL.store(owner_tid, Ordering::Release);
170        INPUT_CELL_TID.store(owner_tid, Ordering::Release);
171    }
172
173    fn reset_roles() {
174        seed_roles(0, 0);
175    }
176
177    #[test]
178    fn deregister_all_for_respects_tid_matches_and_clears_stale_nic_irq_cache() {
179        reset_roles();
180        seed_roles(41, 29);
181
182        deregister_all_for(7);
183
184        assert_eq!(BLOCK_DRIVER_CELL.load(Ordering::Acquire), 41);
185        let state = *NIC_DRIVER_STATE.lock();
186        assert_eq!(state.owner_tid, 41);
187        assert_eq!(state.virtio_irq, 29);
188        assert_eq!(GPU_DRIVER_CELL.load(Ordering::Acquire), 41);
189        assert_eq!(INPUT_CELL_TID.load(Ordering::Acquire), 41);
190
191        deregister_all_for(41);
192
193        assert_eq!(BLOCK_DRIVER_CELL.load(Ordering::Acquire), 0);
194        let state = *NIC_DRIVER_STATE.lock();
195        assert_eq!(state.owner_tid, 0);
196        assert_eq!(state.virtio_irq, 0);
197        assert_eq!(GPU_DRIVER_CELL.load(Ordering::Acquire), 0);
198        assert_eq!(INPUT_CELL_TID.load(Ordering::Acquire), 0);
199
200        seed_roles(22, 0);
201
202        let state = *NIC_DRIVER_STATE.lock();
203        assert_eq!(state.owner_tid, 22);
204        assert_eq!(
205            state.virtio_irq, 0,
206            "stale NIC IRQ cache must not survive owner teardown"
207        );
208
209        reset_roles();
210    }
