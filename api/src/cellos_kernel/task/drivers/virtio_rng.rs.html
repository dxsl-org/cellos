1//! Kernel entropy source: x86_64 uses CPUID-gated RDSEED/RDRAND.
2//!
3//! VirtIO MMIO probing remains disabled on other architectures until a safe
4//! slot-skip strategy exists (probing claimed block/net slots hangs on RISC-V).
5//! Those targets return zero bytes unless the isolated RV64 output-copy fixture
6//! explicitly enables its deterministic test source.
7pub fn init_driver() {
8    #[cfg(target_arch = "x86_64")]
9    let _ = x86_rng::features();
10}

12#[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
13static TEST_ENTROPY_ENABLED: core::sync::atomic::AtomicBool =
14    core::sync::atomic::AtomicBool::new(false);
15
16#[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
17static TEST_ENTROPY_REQUESTS: core::sync::atomic::AtomicUsize =
18    core::sync::atomic::AtomicUsize::new(0);
19
20#[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
21pub(crate) struct TestEntropyGuard;
22
23#[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
24impl Drop for TestEntropyGuard {
25    fn drop(&mut self) {
26        TEST_ENTROPY_ENABLED.store(false, core::sync::atomic::Ordering::Release);
27    }
28}

30#[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
31pub(crate) fn enable_test_entropy() -> TestEntropyGuard {
32    TEST_ENTROPY_ENABLED.store(true, core::sync::atomic::Ordering::Release);
33    TEST_ENTROPY_REQUESTS.store(0, core::sync::atomic::Ordering::Release);
34    TestEntropyGuard
35}

37/// Return test-only entropy requests since the latest guard activation.
38#[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
39pub(crate) fn test_entropy_requests() -> usize {
40    TEST_ENTROPY_REQUESTS.load(core::sync::atomic::Ordering::Acquire)
41}

43/// Fill a prefix with hardware entropy, returning its length.
44///
45/// Zero means no secure source is available or its bounded retries exhausted.
46/// A short read contains only successful hardware samples, never filler bytes.
47pub fn get_random(buf: &mut [u8]) -> usize {
48    #[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
49    TEST_ENTROPY_REQUESTS.fetch_add(1, core::sync::atomic::Ordering::AcqRel);
50    #[cfg(all(feature = "getrandom-sas-test", target_arch = "riscv64"))]
51    if TEST_ENTROPY_ENABLED.load(core::sync::atomic::Ordering::Acquire) {
52        for (index, byte) in buf.iter_mut().enumerate() {
53            *byte = 0xA5 ^ index as u8;
54        }
55        return buf.len();
56    }
57    #[cfg(target_arch = "x86_64")]
58    {
59        x86_rng::get_random(buf)
60    }
61    #[cfg(not(target_arch = "x86_64"))]
62    {
63        let _ = buf;
64        0
65    }
66}

68#[cfg(target_arch = "x86_64")]
69mod x86_rng {
70    use core::arch::x86_64::{__cpuid, __cpuid_count, _rdrand64_step, _rdseed64_step};
71    use core::sync::atomic::{AtomicU8, Ordering};
72
73    const DETECTED: u8 = 1;
74    const RDSEED: u8 = 2;
75    const RDRAND: u8 = 4;
76    const ATTEMPTS: usize = 10;
77    static FEATURES: AtomicU8 = AtomicU8::new(0);
78
79    pub(super) fn features() -> u8 {
80        let cached = FEATURES.load(Ordering::Relaxed);
81        if cached != 0 {
82            return cached;
83        }
84        let max_leaf = __cpuid(0).eax;
85        let mut detected = DETECTED;
86        if max_leaf >= 1 && __cpuid(1).ecx & (1 << 30) != 0 {
87            detected |= RDRAND;
88        }
89        if max_leaf >= 7 && __cpuid_count(7, 0).ebx & (1 << 18) != 0 {
90            detected |= RDSEED;
91        }
92        // The feature bits are the entire cached state; no dependent memory
93        // needs publication. Concurrent initial probes are harmless.
94        FEATURES.store(detected, Ordering::Relaxed);
95        detected
96    }
97
98    fn retry(mut sample: impl FnMut() -> Option<u64>) -> Option<u64> {
99        for _ in 0..ATTEMPTS {
100            if let Some(word) = sample() {
101                return Some(word);
102            }
103            core::hint::spin_loop();
104        }
105        None
106    }
107
108    fn next_word(features: u8) -> Option<u64> {
109        // Prefer RDSEED's conditioned seed entropy. Exhaustion may fall back
110        // only to the CPU's cryptographic RDRAND generator, never a software PRNG.
111        if features & RDSEED != 0 {
112            let word = retry(|| {
113                let mut word = 0;
114                // SAFETY: CPUID.7.0:EBX[18] gated entry to this instruction.
115                let ready = unsafe { _rdseed64_step(&mut word) };
116                (ready == 1).then_some(word)
117            });
118            if word.is_some() {
119                return word;
120            }
121        }
122        if features & RDRAND != 0 {
123            return retry(|| {
124                let mut word = 0;
125                // SAFETY: CPUID.1:ECX[30] gated entry to this instruction.
126                let ready = unsafe { _rdrand64_step(&mut word) };
127                (ready == 1).then_some(word)
128            });
129        }
130        None
131    }
132
133    fn fill(buf: &mut [u8], mut sample: impl FnMut() -> Option<u64>) -> usize {
134        let mut written = 0;
135        for chunk in buf.chunks_mut(core::mem::size_of::<u64>()) {
136            let Some(word) = sample() else {
137                break;
138            };
139            chunk.copy_from_slice(&word.to_le_bytes()[..chunk.len()]);
140            written += chunk.len();
141        }
142        written
143    }
144
145    pub(super) fn get_random(buf: &mut [u8]) -> usize {
146        let features = features();
147        fill(buf, || next_word(features))
148    }

151    mod tests {
152        use super::*;
153
154        #[test]
155        fn retry_exhaustion_is_bounded_and_zero_is_a_valid_sample() {
156            let mut calls = 0;
157            assert_eq!(retry(|| {
158                calls += 1;
159                None
160            }), None);
161            assert_eq!(calls, ATTEMPTS);
162
163            calls = 0;
164            assert_eq!(retry(|| {
165                calls += 1;
166                (calls == ATTEMPTS).then_some(0)
167            }), Some(0));
168            assert_eq!(calls, ATTEMPTS);
169        }
170
171        #[test]
172        fn exhausted_source_returns_only_the_successful_prefix() {
173            let mut buf = [0xA5; 17];
174            let mut calls = 0;
175            let written = fill(&mut buf, || {
176                calls += 1;
177                (calls == 1).then_some(0x0807_0605_0403_0201)
178            });
179            assert_eq!(written, 8);
180            assert_eq!(calls, 2);
181            assert_eq!(&buf[..8], &[1, 2, 3, 4, 5, 6, 7, 8]);
182            assert_eq!(&buf[8..], &[0xA5; 9]);
183        }
184
185        #[test]
186        fn final_partial_word_is_bounded_and_empty_output_consumes_nothing() {
187            let mut buf = [0xA5; 9];
188            let mut calls = 0;
189            assert_eq!(fill(&mut buf, || {
190                calls += 1;
191                Some(0)
192            }), 9);
193            assert_eq!(calls, 2);
194            assert_eq!(buf, [0; 9]);
195            assert_eq!(fill(&mut [], || panic!("empty output sampled entropy")), 0);
196        }
