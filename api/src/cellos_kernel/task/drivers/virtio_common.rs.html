1//! Shared VirtIO MMIO slot enumeration.
2//!
3//! `virtio_blk`, `input_irq_ack`, and Driver Cells use `virtio_slots()` to
4//! iterate all VirtIO MMIO slots for the current platform.
5//!
6//! AArch64 QEMU virt: scans all 32 slots at 0x0a000000, stride 0x200.
7//! Raspberry Pi 3 exposes no VirtIO MMIO window, so board-rpi3 yields no slots.
8//! QEMU assigns devices to slots in an implementation-defined order so we must
9//! probe all 32 there.  The identity map in paging.rs covers the full
10//! 0x0a004000 range only on non-RPi3 AArch64 builds.
11//!
12//! RISC-V: reads DTB-confirmed slots from `platform::PLATFORM`.
13//! PCIe-only arches expose no VirtIO-MMIO slots.
14
15extern crate alloc;
16use alloc::vec::Vec;
17
18/// A VirtIO MMIO slot with base address and IRQ.
19pub struct VirtioSlot {
20    pub base: usize,
21    pub irq: u32,
22}
23
24/// Iterator over all VirtIO MMIO slots for the current platform.
25pub fn virtio_slots() -> impl Iterator<Item = VirtioSlot> {
26    #[cfg(all(
27        target_arch = "aarch64",
28        any(feature = "board-rpi3", feature = "board-rpi4")
29    ))]
30    {
31        Vec::new().into_iter()
32    }
33    #[cfg(all(
34        target_arch = "aarch64",
35        not(feature = "board-rpi3"),
36        not(feature = "board-rpi4")
37    ))]
38    {
39        let layout = hal_soc_arm_virt::QEMU_ARM_VIRT.virtio;
40        let slots: Vec<VirtioSlot> = (0..layout.count)
41            .map(|i| VirtioSlot {
42                base: layout.slot_base(i).expect("bounded VirtIO slot"),
43                irq: layout.spi(i).expect("bounded VirtIO IRQ"),
44            })
45            .collect();
46        slots.into_iter()
47    }
48    #[cfg(target_arch = "riscv64")]
49    {
50        let slots: Vec<VirtioSlot> = crate::platform::with(|p| {
51            p.virtio_mmio
52                .iter()
53                .filter_map(|e| {
54                    e.as_ref().map(|e| VirtioSlot {
55                        base: e.base,
56                        irq: e.irq,
57                    })
58                })
59                .collect()
60        });
61        slots.into_iter()
62    }
63    #[cfg(not(any(target_arch = "aarch64", target_arch = "riscv64")))]
64    {
65        Vec::new().into_iter()
66    }
67}

69/// Execute one VirtIO MMIO ACK sequence with RV64 SUM enabled only for that scope.
70///
71/// On RV64, supervisor-mode MMIO ACK helpers may touch user-accessible identity-mapped
72/// device pages while trap handling runs with SUM clear. This helper snapshots whether
73/// SUM was already set, raises it only for `ack`, and restores the prior state
74/// immediately afterwards. Other architectures execute `ack` unchanged.
75pub(crate) fn with_riscv_sum_for_virtio_ack<T>(ack: impl FnOnce() -> T) -> T {
76    #[cfg(target_arch = "riscv64")]
77    {
78        const SUM_MASK: usize = 1usize << 18;
79        let sstatus: usize;
80        // SAFETY: this helper is used only for short-lived VirtIO MMIO ACK sequences.
81        // `csrr/csrs/csrc sstatus` are privileged RV64 supervisor CSR operations; the
82        // code records the prior SUM bit, sets only SUM for the duration of `ack`, and
83        // restores the exact prior SUM state before returning. It does not widen trap
84        // entry globally or alter SSIE/SEIE/STIE.
85        unsafe {
86            core::arch::asm!("csrr {value}, sstatus", value = out(reg) sstatus, options(nostack));
87            if sstatus & SUM_MASK == 0 {
88                core::arch::asm!("csrs sstatus, {mask}", mask = in(reg) SUM_MASK, options(nostack));
89            }
90        }
91        let result = ack();
92        unsafe {
93            if sstatus & SUM_MASK == 0 {
94                core::arch::asm!("csrc sstatus, {mask}", mask = in(reg) SUM_MASK, options(nostack));
95            }
96        }
97        result
98    }
99    #[cfg(not(target_arch = "riscv64"))]
100    {
101        ack()
102    }
103}

105/// Read VirtIO `InterruptStatus` and ACK the exact bits it reports.
106///
107/// Returns the observed status bits; `0` means there was nothing pending to ACK.
108pub(crate) fn ack_virtio_interrupt_status(mmio_base: usize) -> u32 {
109    let status = with_riscv_sum_for_virtio_ack(|| {
110        // SAFETY: caller provides a live, identity-mapped VirtIO MMIO base. The
111        // status register is read-only, and the scoped SUM helper contains the
112        // RV64 privilege widening to this one MMIO access sequence only.
113        unsafe { core::ptr::read_volatile((mmio_base + 0x60) as *const u32) }
114    });
115    if status != 0 {
116        with_riscv_sum_for_virtio_ack(|| {
117            // SAFETY: same MMIO contract as the status read above; the ACK register
118            // consumes exactly the pending status bits returned by the device.
119            unsafe { core::ptr::write_volatile((mmio_base + 0x64) as *mut u32, status) };
120        });
121    }
122    status
123}

125/// Acknowledge an interrupt from an as-yet unclaimed VirtIO MMIO slot.
126///
127/// Driver Cells may need several synchronous queue transactions to initialise
128/// before they can enter `sys_wait_irq`. Leaving those level-triggered
129/// interrupts asserted starves the Cell before registration completes.
130fn ack_unclaimed(irq: u32) -> bool {
131    #[cfg(all(
132        target_arch = "aarch64",
133        not(feature = "board-rpi3"),
134        not(feature = "board-rpi4")
135    ))]
136    let base = {
137        let layout = hal_soc_arm_virt::QEMU_ARM_VIRT.virtio;
138        irq.checked_sub(layout.first_spi)
139            .and_then(|index| layout.slot_base(index as usize))
140    };
141    #[cfg(all(
142        target_arch = "aarch64",
143        any(feature = "board-rpi3", feature = "board-rpi4")
144    ))]
145    let base: Option<usize> = {
146        let _ = irq;
147        None
148    };
149    #[cfg(target_arch = "riscv64")]
150    let base = crate::platform::virtio_mmio_base_for_irq(irq);
151    // x86_64 routes VirtIO through PCI MSI-X, not a fixed MMIO IRQ window, so
152    // there is no slot base to derive and nothing to acknowledge here.
153    #[cfg(not(any(target_arch = "aarch64", target_arch = "riscv64")))]
154    let base: Option<usize> = {
155        let _ = irq;
156        None
157    };
158
159    let Some(base) = base else {
160        return false;
161    };
162    ack_virtio_interrupt_status(base) != 0
163}

165/// VirtIO MMIO IRQ dispatcher — called from the arch trap handlers (riscv64/aarch64)
166/// when any VirtIO MMIO IRQ fires. Routes the IRQ to whichever Driver Cell registered
167/// for it via `sys_wait_irq` (the block / net / gpu cells all rely on this), ACKs the
168/// input slot when the input service Cell isn't up yet, and warns on an unclaimed slot.
169///
170/// Relocated here from the deleted kernel `virtio_blk` driver (G2 loader redesign
171/// phase 06): it is VirtIO-common, not block-specific. The former kernel-block ACK
172/// branch is gone — the virtio-blk Driver Cell now ACKs its own IRQ via the
173/// `sys_wait_irq` path above.
174#[no_mangle]
175pub extern "Rust" fn vi_handle_virtio_irq(irq: u32) {
176    let nic_irq = crate::task::drivers::driver_cell::owns_registered_nic_irq(irq);
177
178    // Driver Cell IRQ routing: a Cell registered for this IRQ via sys_wait_irq —
179    // signal it (sets IRQ_PENDING + writes VirtIO InterruptACK) and return.
180    if crate::task::drivers::irq_wait::has_waiter(irq as u8) {
181        crate::task::drivers::irq_wait::signal_irq(irq as u8);
182        if nic_irq {
183            crate::task::waker::signal_net_rx();
184        }
185        return;
186    }
187    // Input (keyboard) slot: ACK to prevent an interrupt storm before the input
188    // service Cell is up; event routing lives entirely in that Cell.
189    if crate::task::drivers::input_irq_ack::ack_if_input(irq) {
190        return;
191    }
192    if ack_unclaimed(irq) {
193        if nic_irq {
194            crate::task::waker::signal_net_rx();
195        }
196        return;
197    }
198    // Unknown VirtIO slot — no device registered. InterruptStatus is already cleared
199    // by plic_complete in the trap handler.
200    log::warn!(
201        "[virtio] unhandled IRQ {} — no registered device for this slot",
202        irq
203    );
