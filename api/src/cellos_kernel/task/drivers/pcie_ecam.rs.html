1//! PCIe ECAM (Enhanced Configuration Access Mechanism) walker.
2//!
3//! The kernel fallback scans one admitted bus, probing functions per the
4//! header-type multi-function bit. Exposes a kernel-internal device snapshot.
5//!
6//! ECAM bases per machine:
7//!   x86_64      : runtime ACPI MCFG (no q35 fallback)
8//!   RISC-V virt : 0x3000_0000
9//!   ARM64 virt  : 0x3F00_0000
10//!
11//! The Platform Cell may map and enumerate the complete firmware-admitted bus
12//! range. Kernel fallback enumeration remains bus-0-only on non-x86 targets.
13//!
14//! Call ordering: `platform::init()` → `init_kernel_paging*()` → `pcie_ecam::init()`.
15

76/// ECAM bytes per PCI bus (32 devices × 8 functions × 4 KiB).
77pub const ECAM_BYTES_PER_BUS: usize = 0x10_0000;

79/// Checked byte length for an inclusive MCFG bus range.
80pub fn ecam_window_size(bus_start: u8, bus_end: u8) -> Option<usize> {
81    let buses = (bus_end as usize)
82        .checked_sub(bus_start as usize)?
83        .checked_add(1)?;
84    buses.checked_mul(ECAM_BYTES_PER_BUS)
85}

113/// A Base Address Register decoded from PCI config space.
114#[derive(Clone, Copy, Debug)]
115pub enum Bar {
116    /// 32-bit MMIO BAR: base address and probed size.
117    Memory32 { addr: u32, size: u32 },
118    /// 64-bit MMIO BAR: base address (64-bit) and probed size. Consumes two slots.
119    Memory64 { addr: u64, size: u64 },
120    /// I/O port BAR (not mapped by this driver; skipped).
121    Io,
122    /// Unused / empty BAR slot.
123    None,
124}
125
126impl Bar {
127    /// Physical base address of this BAR, or 0 for I/O / empty.
128    pub fn base_addr(&self) -> u64 {
129        match self {
130            Bar::Memory32 { addr, .. } => *addr as u64,
131            Bar::Memory64 { addr, .. } => *addr,
132            _ => 0,
133        }
134    }
135}

200/// Compute an MMIO address relative to an admitted inclusive bus range.
201#[inline(always)]
202fn config_addr_in_range(
203    ecam_base: usize,
204    bus_start: u8,
205    bus_end: u8,
206    bus: u8,
207    dev: u8,
208    fun: u8,
209    off: usize,
210) -> Option<*mut u8> {
211    if bus < bus_start || bus > bus_end || dev >= 32 || fun >= 8 || off >= 0x1000 {
212        return None;
213    }
214    let bus_offset = (bus as usize)
215        .checked_sub(bus_start as usize)?
216        .checked_mul(ECAM_BYTES_PER_BUS)?;
217    let offset = bus_offset
218        .checked_add((dev as usize) << 15)?
219        .checked_add((fun as usize) << 12)?
220        .checked_add(off)?;
221    Some(ecam_base.checked_add(offset)? as *mut u8)
222}

224/// Read vendor/device identity from one checked ECAM function.
225///
226/// # Safety
227/// The admitted ECAM window must be identity-mapped and remain live.
228unsafe fn read_identity_in_range(
229    ecam_base: usize,
230    bus_start: u8,
231    bus_end: u8,
232    bus: u8,
233    dev: u8,
234    fun: u8,
235) -> Option<(u16, u16)> {
236    let ptr = config_addr_in_range(ecam_base, bus_start, bus_end, bus, dev, fun, CFG_VENDOR_ID)?;
237    // SAFETY: the checked function header lies in the caller-provided live ECAM window.
238    let vendor = unsafe { core::ptr::read_volatile(ptr.cast::<u16>()) };
239    // SAFETY: device ID immediately follows vendor ID in the same function header.
240    let device = unsafe { core::ptr::read_volatile(ptr.add(CFG_DEVICE_ID).cast::<u16>()) };
241    Some((vendor, device))
242}

244/// Compute MMIO address for the fixed bus-0 kernel fallback scanners.
245#[inline(always)]
246fn config_addr(ecam_base: usize, bus: u8, dev: u8, fun: u8, off: usize) -> *mut u8 {
247    config_addr_in_range(ecam_base, 0, u8::MAX, bus, dev, fun, off)

251/// Read a u32 from PCI config space via ECAM MMIO.
252///
253/// # Safety
254/// `config_addr(ecam_base, bus, dev, fun, off)` must point into the identity-
255/// mapped ECAM window established by `init_kernel_paging*()`.  `off` must be
256/// 4-byte-aligned.
257unsafe fn read32(ecam_base: usize, bus: u8, dev: u8, fun: u8, off: usize) -> u32 {
258    // SAFETY: caller guarantees the ECAM window is identity-mapped.
259    let ptr = config_addr(ecam_base, bus, dev, fun, off) as *const u32;
260    // SAFETY: volatile prevents the compiler from optimising away hardware reads.
261    unsafe { core::ptr::read_volatile(ptr) }
262}

315/// Decode all BARs for a type-0 function.
316///
317/// The write-all-ones / read-back size-probe modifies the command register to
318/// disable memory decode before each probe and restores it after, preventing
319/// the device from responding to MMIO accesses during the transient probe window.
320///
321/// # Safety
322/// `ecam_base` must be the identity-mapped ECAM window base.
323unsafe fn decode_bars(ecam_base: usize, bus: u8, dev: u8, fun: u8) -> [Bar; 6] {
324    let mut bars = [Bar::None; 6];
325    let mut i = 0usize;
326    while i < 6 {
327        let off = CFG_BAR0 + i * 4;
328        // SAFETY: ECAM window is identity-mapped; off is within type-0 header.
329        let raw = unsafe { read32(ecam_base, bus, dev, fun, off) };
330        if raw & 1 == 1 {
331            // I/O BAR — skip.
332            bars[i] = Bar::Io;
333            i += 1;
334            continue;
335        }
336        let bar_type = (raw >> 1) & 0x3; // bits [2:1]
337        let addr32 = raw & !0xF;
338
339        if bar_type == 0x2 {
340            // 64-bit MMIO: base spans this slot + next.
341            if i + 1 >= 6 {
342                bars[i] = Bar::None;
343                i += 1;
344                continue;
345            }
346            let off_hi = CFG_BAR0 + (i + 1) * 4;
347            // SAFETY: ECAM window is identity-mapped.
348            let raw_hi = unsafe { read32(ecam_base, bus, dev, fun, off_hi) };
349            let addr64 = (addr32 as u64) | ((raw_hi as u64) << 32);
350            let size64 = unsafe { probe_bar_size64(ecam_base, bus, dev, fun, i) };
351            bars[i] = Bar::Memory64 {
352                addr: addr64,
353                size: size64,
354            };
355            bars[i + 1] = Bar::None; // second slot consumed
356            i += 2;
357        } else {
358            // 32-bit MMIO.
359            let size32 = unsafe { probe_bar_size32(ecam_base, bus, dev, fun, i) };
360            bars[i] = Bar::Memory32 {
361                addr: addr32,
362                size: size32,
363            };
364            i += 1;
365        }
366    }
367    bars
368}

370/// Probe size of a 32-bit MMIO BAR via the write-all-ones / read-back method.
371///
372/// Saves/restores both the command register (disabling memory decode during the
373/// probe) and the BAR value so the device is left in its original state.
374///
375/// # Safety
376/// `ecam_base` must be the identity-mapped ECAM window base.
377unsafe fn probe_bar_size32(ecam_base: usize, bus: u8, dev: u8, fun: u8, bar_idx: usize) -> u32 {
378    let off = CFG_BAR0 + bar_idx * 4;
379    // Save original values.
380    // SAFETY: ECAM window is identity-mapped.
381    let orig_cmd = unsafe { read16(ecam_base, bus, dev, fun, CFG_COMMAND) };
382    // SAFETY: ECAM window is identity-mapped.
383    let orig_bar = unsafe { read32(ecam_base, bus, dev, fun, off) };
384
385    // Disable memory decode before writing all-ones to BAR.
386    // This prevents the device from claiming MMIO during the probe window.
387    // SAFETY: writing command register disables decode only for this function.
388    unsafe {
389        write16(
390            ecam_base,
391            bus,
392            dev,
393            fun,
394            CFG_COMMAND,
395            orig_cmd & !CMD_MEM_SPACE,
396        );
397    }
398
399    // Write all-ones to BAR.
400    // SAFETY: ECAM window is identity-mapped; BAR write is bounded to this function.
401    unsafe {
402        write32(ecam_base, bus, dev, fun, off, 0xFFFF_FFFF);
403    }
404    // SAFETY: ECAM window is identity-mapped.
405    let readback = unsafe { read32(ecam_base, bus, dev, fun, off) };
406
407    // Restore BAR and command register.
408    // SAFETY: ECAM window is identity-mapped.
409    unsafe {
410        write32(ecam_base, bus, dev, fun, off, orig_bar);
411    }
412    unsafe {
413        write16(ecam_base, bus, dev, fun, CFG_COMMAND, orig_cmd);
414    }
415
416    // Size = ~(readback & mask) + 1 where mask = 0xFFFFFFF0 (clear lower 4 bits).
417    let mask = readback & 0xFFFF_FFF0;
418    if mask == 0 {
419        return 0;
420    }
421    (!mask).wrapping_add(1)
422}

424/// Probe size of a 64-bit MMIO BAR (both low and high dwords).
425///
426/// # Safety
427/// `ecam_base` must be the identity-mapped ECAM window base.
428unsafe fn probe_bar_size64(ecam_base: usize, bus: u8, dev: u8, fun: u8, bar_idx: usize) -> u64 {
429    let off_lo = CFG_BAR0 + bar_idx * 4;
430    let off_hi = CFG_BAR0 + (bar_idx + 1) * 4;
431
432    // SAFETY: ECAM window is identity-mapped.
433    let orig_cmd = unsafe { read16(ecam_base, bus, dev, fun, CFG_COMMAND) };
434    // SAFETY: ECAM window is identity-mapped.
435    let orig_lo = unsafe { read32(ecam_base, bus, dev, fun, off_lo) };
436    // SAFETY: ECAM window is identity-mapped.
437    let orig_hi = unsafe { read32(ecam_base, bus, dev, fun, off_hi) };
438
439    // Disable memory decode during probe.
440    // SAFETY: ECAM window is identity-mapped.
441    unsafe {
442        write16(
443            ecam_base,
444            bus,
445            dev,
446            fun,
447            CFG_COMMAND,
448            orig_cmd & !CMD_MEM_SPACE,
449        );
450    }
451
452    // Write all-ones to both halves.
453    // SAFETY: ECAM window is identity-mapped.
454    unsafe {
455        write32(ecam_base, bus, dev, fun, off_lo, 0xFFFF_FFFF);
456    }
457    unsafe {
458        write32(ecam_base, bus, dev, fun, off_hi, 0xFFFF_FFFF);
459    }
460
461    // SAFETY: ECAM window is identity-mapped.
462    let rb_lo = unsafe { read32(ecam_base, bus, dev, fun, off_lo) };
463    // SAFETY: ECAM window is identity-mapped.
464    let rb_hi = unsafe { read32(ecam_base, bus, dev, fun, off_hi) };
465
466    // Restore.
467    // SAFETY: ECAM window is identity-mapped.
468    unsafe {
469        write32(ecam_base, bus, dev, fun, off_lo, orig_lo);
470    }
471    unsafe {
472        write32(ecam_base, bus, dev, fun, off_hi, orig_hi);
473    }
474    unsafe {
475        write16(ecam_base, bus, dev, fun, CFG_COMMAND, orig_cmd);
476    }
477
478    let mask64 = ((rb_hi as u64) << 32) | ((rb_lo & 0xFFFF_FFF0) as u64);
479    if mask64 == 0 {
480        return 0;
481    }
482    (!mask64).wrapping_add(1)
483}

487/// Walk the capability list for a type-0 function; return MSI-X and PM caps.
488///
489/// # Safety
490/// `ecam_base` must be the identity-mapped ECAM window base.
491unsafe fn walk_caps(
492    ecam_base: usize,
493    bus: u8,
494    dev: u8,
495    fun: u8,
496) -> (Option<MsixCap>, Option<PmCap>) {
497    let mut msix = None;
498    let mut pm = None;
499
500    // Capability list is present only when status bit 4 is set.
501    // SAFETY: ECAM window is identity-mapped.
502    let status = unsafe { read16(ecam_base, bus, dev, fun, CFG_STATUS) };
503    if status & (1 << 4) == 0 {
504        return (None, None);
505    }
506
507    // SAFETY: ECAM window is identity-mapped.
508    let mut cap_ptr = unsafe { read8(ecam_base, bus, dev, fun, CFG_CAP_PTR) } & 0xFC;
509    let mut budget = 64u8; // guard against malformed circular lists
510
511    while cap_ptr != 0 && budget > 0 {
512        budget -= 1;
513        // SAFETY: ECAM window is identity-mapped.
514        let cap_id = unsafe { read8(ecam_base, bus, dev, fun, cap_ptr as usize) };
515        // SAFETY: ECAM window is identity-mapped.
516        let next = unsafe { read8(ecam_base, bus, dev, fun, cap_ptr as usize + 1) } & 0xFC;
517
518        match cap_id {
519            CAP_ID_MSIX => {
520                // SAFETY: ECAM window is identity-mapped.
521                let msg_ctrl = unsafe { read16(ecam_base, bus, dev, fun, cap_ptr as usize + 2) };
522                msix = Some(MsixCap {
523                    cap_offset: cap_ptr,
524                    msg_ctrl,
525                });
526            }
527            CAP_ID_PM => {
528                pm = Some(PmCap {
529                    cap_offset: cap_ptr,
530                });
531            }
532            _ => {}
533        }
534        cap_ptr = next;
535    }
536
537    (msix, pm)
538}

542/// Scan one bus of the ECAM window and populate the global device list.
543///
544/// # Safety
545/// `ecam_base + (config_bus << 20)` must point to the identity-mapped ECAM
546/// window for `reported_bus`. Nonzero-start allocations use `config_bus = 0`
547/// because their allocation base already denotes the first admitted bus.
548unsafe fn scan(ecam_base: usize, config_bus: u8, reported_bus: u8) {
549    let mut devices = PCI_DEVICES.lock();
550
551    for dev in 0u8..32 {
552        // SAFETY: the selected ECAM bus window is identity-mapped.
553        let vendor = unsafe { read16(ecam_base, config_bus, dev, 0, CFG_VENDOR_ID) };
554        if vendor == 0xFFFF {
555            continue;
556        }
557
558        // SAFETY: the selected ECAM bus window is identity-mapped.
559        let hdr_type = unsafe { read8(ecam_base, config_bus, dev, 0, CFG_HEADER_TYPE) };
560        let max_fun: u8 = if hdr_type & 0x80 != 0 { 8 } else { 1 };
561
562        for fun in 0u8..max_fun {
563            // SAFETY: the selected ECAM bus window is identity-mapped.
564            let vid = unsafe { read16(ecam_base, config_bus, dev, fun, CFG_VENDOR_ID) };
565            if vid == 0xFFFF {
566                continue;
567            }
568
569            // SAFETY: the selected ECAM bus window is identity-mapped.
570            let did = unsafe { read16(ecam_base, config_bus, dev, fun, CFG_DEVICE_ID) };
571            let prog_if = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_CLASS_PROG) };
572            let subclass = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_SUBCLASS) };
573            let class = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_CLASS_CODE) };
574
575            // Decode BARs for type-0 headers (endpoints). Skip bridges (type-1).
576            let hdr =
577                unsafe { read8(ecam_base, config_bus, dev, fun, CFG_HEADER_TYPE) } & 0x7F;
578            let bars = if hdr == 0 {
579                // SAFETY: the selected ECAM bus window is identity-mapped.
580                unsafe { decode_bars(ecam_base, config_bus, dev, fun) }
581            } else {
582                [Bar::None; 6]
583            };
584
585            // SAFETY: the selected ECAM bus window is identity-mapped.
586            let (msix, pm) = unsafe { walk_caps(ecam_base, config_bus, dev, fun) };
587
588            let bar0_addr = bars[0].base_addr();
589            log::info!(
590                "[pcie] {:02x}:{:02x}.{} vendor={:04x} device={:04x} \
591                 class={:02x}:{:02x}:{:02x} bar0={:#x}",
592                reported_bus,
593                dev,
594                fun,
595                vid,
596                did,
597                class,
598                subclass,
599                prog_if,
600                bar0_addr
601            );
602            if class == 0x02
603                && subclass == 0x00
604                && prog_if == 0x00
605                && (vid != 0x8086 || did != 0x100E)
606            {
607                log::warn!(
608                    "[e1000] unsupported Ethernet {:04x}:{:04x}; driver gate closed",
609                    vid,
610                    did
611                );
612            }
613
614            devices.push(PciDevice {
615                bdf: (reported_bus, dev, fun),
616                vendor_id: vid,
617                device_id: did,
618                class,
619                subclass,
620                prog_if,
621                bars,
622                msix,
623                pm,
624            });
625        }
626    }
627}

629/// Enable MMIO decoding and bus mastering after the caller has established
630/// exclusive BDF ownership and an IOMMU mapping.
631pub fn enable_bus_master(bdf: u32) {
632    let bus = ((bdf >> 8) & 0xFF) as u8;
633    let dev = ((bdf >> 3) & 0x1F) as u8;
634    let fun = (bdf & 0x07) as u8;
635
636    #[cfg(target_arch = "x86_64")]
637    let (ecam_base, bus_start, bus_end) = {
638        let (start, end) = ecam_bus_range_x86();
639        (ecam_base_x86(), start, end)
640    };
641    #[cfg(target_arch = "riscv64")]
642    let (ecam_base, bus_start, bus_end) = (ECAM_BASE_RISCV, 0u8, 0u8);
643    #[cfg(all(
644        target_arch = "aarch64",
645        not(feature = "board-rpi3"),
646        not(feature = "board-rpi4")
647    ))]
648    let (ecam_base, bus_start, bus_end) = (ECAM_BASE_AARCH64, 0u8, 0u8);
649    #[cfg(any(
650        all(
651            target_arch = "aarch64",
652            any(feature = "board-rpi3", feature = "board-rpi4")
653        ),
654        not(any(
655            target_arch = "x86_64",
656            target_arch = "riscv64",
657            target_arch = "aarch64"
658        ))
659    ))]
660    let (ecam_base, bus_start, bus_end) = (0usize, 0u8, 0u8);
661    if ecam_base == 0 {
662        return;
663    }
664    let Some(command_ptr) =
665        config_addr_in_range(ecam_base, bus_start, bus_end, bus, dev, fun, CFG_COMMAND)
666    else {
667        log::warn!(
668            "[pcie] refusing bus-master enable outside ECAM range: {:02x}:{:02x}.{}",
669            bus,
670            dev,
671            fun
672        );
673        return;
674    };
675
676    // SAFETY: the complete admitted ECAM window is identity-mapped before
677    // Driver Cells run, and command_ptr was checked against that exact range.
678    unsafe {
679        let command = core::ptr::read_volatile(command_ptr as *const u16);
680        core::ptr::write_volatile(
681            command_ptr as *mut u16,
682            command | CMD_MEM_SPACE | CMD_BUS_MASTER,
683        );
684    }
685}

689/// Initialise the ECAM scanner.
690///
691/// Must be called after paging is active (ECAM window mapped) and before
692/// Driver Cells are spawned (they call sys_find_pcie_device to discover their
693/// endpoint). Idempotent; safe to call more than once (later calls are no-ops).
694pub fn init() {
695    // Determine ECAM base for the current architecture.
696    // x86_64: use only the runtime value from validated ACPI MCFG.
697    #[cfg(target_arch = "x86_64")]
698    let ecam_base = ecam_base_x86();
699    #[cfg(target_arch = "riscv64")]
700    let ecam_base = ECAM_BASE_RISCV;
701    #[cfg(all(
702        target_arch = "aarch64",
703        not(feature = "board-rpi3"),
704        not(feature = "board-rpi4")
705    ))]
706    let ecam_base = ECAM_BASE_AARCH64;
707    #[cfg(all(
708        target_arch = "aarch64",
709        any(feature = "board-rpi3", feature = "board-rpi4")
710    ))]
711    let ecam_base = 0usize;
712    #[cfg(not(any(
713        target_arch = "x86_64",
714        target_arch = "riscv64",
715        target_arch = "aarch64"
716    )))]
717    let ecam_base = 0usize; // bare-physical arches: no PCIe
718
719    #[cfg(target_arch = "x86_64")]
720    let scan_bus = ecam_bus_range_x86().0;
721    #[cfg(not(target_arch = "x86_64"))]
722    let scan_bus = 0u8;
723
724    if ecam_base == 0 {
725        log::info!("[pcie] ECAM: no PCIe on this architecture");
726        return;
727    }
728
729    // If already populated (e.g. called twice), skip re-scan.
730    if !PCI_DEVICES.lock().is_empty() {
731        log::info!("[pcie] ECAM: already scanned, skipping");
732        return;
733    }
734    log::info!(
735        "[pcie] ECAM fallback scan bus {} @ {:#x} (1 MiB window)",
736        scan_bus,
737        ecam_base
738    );
739
740    // SAFETY: the first admitted ECAM bus window is identity-mapped before this
741    // function is called. The MCFG allocation base denotes `scan_bus`, so
742    // config-space access is relative to local bus offset zero.
743    unsafe {
744        scan(ecam_base, 0, scan_bus);
745    }
746
747    let count = PCI_DEVICES.lock().len();
748    if count == 0 {
749        log::warn!(
750            "[pcie] ECAM scan found no devices on bus {} — check base and MMIO mapping",
751            scan_bus
752        );
753    } else {
754        log::info!("[pcie] ECAM scan complete: {} device(s) found", count);
755    }
756
757    // Register all discovered BAR windows in the Resource Registry so Driver
758    // Cells with PcieDriverCap can call sys_request_mmio on them.
759    for dev in PCI_DEVICES.lock().iter() {
760        for bar in &dev.bars {
761            let base = bar.base_addr() as usize;
762            if base == 0 {
763                continue;
764            }
765            // Probe size is stored in Bar::Memory32/Memory64; use a safe default
766            // of 16 KiB if the probed size is 0 (identity entry).
767            let len = match bar {
768                Bar::Memory32 { size, .. } => {
769                    if *size == 0 {
770                        0x4000
771                    } else {
772                        *size as usize
773                    }
774                }
775                Bar::Memory64 { size, .. } => {
776                    if *size == 0 {
777                        0x4000
778                    } else {
779                        *size as usize
780                    }
781                }
782                _ => continue,
783            };
784            crate::resource_registry::register_pcie_bar(base, len);
785        }
786    }
787}

789/// Register a device announced by the Platform Cell via `sys_register_pci_device`.
790///
791/// Decodes `bdf` and `cls` into a `PciDevice` stub with BAR0 populated.
792/// Skips duplicate BDFs (idempotent: kernel ECAM scan may already have found the device).
793/// Also registers BAR0 in the Resource Registry for `sys_request_mmio` validation.
794pub fn register_device(bdf: u32, cls: u32, bar0_base: usize, bar0_size: usize) {
795    // Canonical BDF wire layout everywhere (Platform Cell, FindPcieDevice,
796    // GrantDma, iommu_x86): bus[15:8] | dev[7:3] | fun[2:0]. An earlier decode
797    // here assumed bus<<16|dev<<8|fun, which garbled the stored (bus,dev,fun)
798    // tuple for every device (it only round-tripped through FindPcieDevice's
799    // re-encode by luck on bus 0).
800    let bus = ((bdf >> 8) & 0xFF) as u8;
801    let dev = ((bdf >> 3) & 0x1F) as u8;
802    let fun = (bdf & 0x07) as u8;
803    let class = ((cls >> 16) & 0xFF) as u8;
804    let subclass = ((cls >> 8) & 0xFF) as u8;
805    let prog_if = (cls & 0xFF) as u8;
806    #[cfg(target_arch = "x86_64")]
807    let (vendor_id, device_id) = {
808        let ecam_base = ecam_base_x86();
809        let (bus_start, bus_end) = ecam_bus_range_x86();
810        if ecam_base == 0 {
811            (0, 0)
812        } else {
813            // SAFETY: x86 paging maps the validated MCFG allocation before the
814            // Platform Cell can issue registration syscalls.
815            unsafe {
816                read_identity_in_range(ecam_base, bus_start, bus_end, bus, dev, fun)
817                    .unwrap_or((0, 0))
818            }
819        }
820    };
821    #[cfg(not(target_arch = "x86_64"))]
822    let (vendor_id, device_id) = (0, 0);
823
824    {
825        let devices = PCI_DEVICES.lock();
826        if devices.iter().any(|d| d.bdf == (bus, dev, fun)) {
827            return; // already present (kernel ECAM scan ran first)
828        }
829    }
830
831    let bars = if bar0_base != 0 {
832        let bar0 = if bar0_size <= u32::MAX as usize {
833            Bar::Memory32 {
834                addr: bar0_base as u32,
835                size: bar0_size as u32,
836            }
837        } else {
838            Bar::Memory64 {
839                addr: bar0_base as u64,
840                size: bar0_size as u64,
841            }
842        };
843        let mut arr = [Bar::None; 6];
844        arr[0] = bar0;
845        arr
846    } else {
847        [Bar::None; 6]
848    };
849
850    log::info!(
851        "[pcie] platform-reg {:02x}:{:02x}.{} class={:02x}:{:02x}:{:02x} bar0={:#x}",
852        bus,
853        dev,
854        fun,
855        class,
856        subclass,
857        prog_if,
858        bar0_base
859    );
860
861    PCI_DEVICES.lock().push(PciDevice {
862        bdf: (bus, dev, fun),
863        vendor_id,
864        device_id,
865        class,
866        subclass,
867        prog_if,
868        bars,
869        msix: None,
870        pm: None,
871    });
872
873    if bar0_base != 0 && bar0_size != 0 {
874        crate::resource_registry::register_pcie_bar(bar0_base, bar0_size);
875    }
876}

878/// Return a cloned snapshot of the discovered device list.
879pub fn devices() -> Vec<PciDevice> {
880    PCI_DEVICES.lock().clone()
881}

883/// Find the first supported device matching (class, subclass, prog_if).
884///
885/// Example: NVMe = `find_class(0x01, 0x08, 0x02)`.
886pub fn find_class(class: u8, subclass: u8, prog_if: u8) -> Option<PciDevice> {
887    PCI_DEVICES
888        .lock()
889        .iter()
890        .find(|d| {
891            if d.class != class || d.subclass != subclass || d.prog_if != prog_if {
892                return false;
893            }
894            // The current Driver Cell implements only Intel 82540EM. Binding
895            // an I219/e1000e-class endpoint by generic Ethernet class would
896            // program an incompatible register model.
897            #[cfg(target_arch = "x86_64")]
898            if class == 0x02 && subclass == 0x00 && prog_if == 0x00 {
899                return d.vendor_id == 0x8086 && d.device_id == 0x100E;
900            }
901            true
902        })
903        .cloned()
904}
