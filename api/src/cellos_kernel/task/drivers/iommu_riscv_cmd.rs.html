1//! RISC-V IOMMU v1.0.1 command-queue encoders.
2//!
3//! Field positions follow `iommu_in_memory_queues.adoc`. Each command is two
4//! 64-bit words; the commands used here have a zero second word.
5
6const OPCODE_IOTINVAL: u64 = 0x1;
7const OPCODE_IOFENCE: u64 = 0x2;
8const OPCODE_IODIR: u64 = 0x3;
9
10const PSCID_MASK: u64 = (1 << 20) - 1;
11const DID_MASK: u64 = (1 << 24) - 1;
12const PPN_MASK: u64 = (1 << 44) - 1;
13
14/// CQB with PPN in bits 53:10 and LOG2SZ-1 in bits 4:0.
15pub(crate) const fn encode_cqb(queue_phys: u64, log2_entries: u8) -> u64 {
16    let ppn = (queue_phys >> 12) & PPN_MASK;
17    (ppn << 10) | ((log2_entries.saturating_sub(1) as u64) & 0x1F)
18}
19
20/// IOTINVAL.VMA with PSCV=1 and AV=GV=0.
21pub(crate) const fn encode_iotinval_vma(pscid: u32) -> (u64, u64) {
22    let word0 = OPCODE_IOTINVAL | (((pscid as u64) & PSCID_MASK) << 12) | (1 << 32);
23    (word0, 0)
24}
25
26/// IOFENCE.C with PR=PW=1 and AV=WSI=0.
27///
28/// Ordering both prior device reads and writes is required before DMA-accessible
29/// frames can leave quarantine and be reclaimed.
30pub(crate) const fn encode_iofence_c() -> (u64, u64) {
31    (OPCODE_IOFENCE | (1 << 12) | (1 << 13), 0)
32}
33
34/// IODIR.INVAL_DDT with DV=1 and the exact 24-bit device ID.
35pub(crate) const fn encode_iodir_inval_ddt(device_id: u32) -> (u64, u64) {
36    let word0 = OPCODE_IODIR | (1 << 33) | (((device_id as u64) & DID_MASK) << 40);
37    (word0, 0)
38}
