1//! Minimal 16550 UART Driver for QEMU RISC-V Virt
2//!
3//! Used for kernel logging and early debug output.
4//! Base Address: 0x10000000
5
6use crate::sync::Spinlock;
7use core::fmt;
8use core::fmt::Write as _;

10/// UART Registers (offset from base)
11const _RHR: usize = 0; // Receive Holding Register (read)
12const _THR: usize = 0; // Transmit Holding Register (write)
13const IER: usize = 1; // Interrupt Enable Register
14const FCR: usize = 2; // FIFO Control Register
15const _ISR: usize = 2; // Interrupt Status Register
16const LCR: usize = 3; // Line Control Register
17const LSR: usize = 5; // Line Status Register
18
19/// Line Status Flags
20const _LSR_RX_READY: u8 = 1 << 0;
21const _LSR_TX_EMPTY: u8 = 1 << 5;
22
23#[allow(non_camel_case_types)]
24pub struct viUART {
25    base_addr: usize,
26}
27
28impl viUART {
29    /// Create a new viUART instance (unsafe because base_addr must be valid)
30    ///
31    /// # Safety
32    /// `base_addr` must be either 0 (sentinel meaning "no NS16550 MMIO on this
33    /// board" — all I/O methods on this instance check for it and fall back to
34    /// SBI DBCN/port-I/O) or a valid, identity-mapped 16550 UART MMIO base that
35    /// remains mapped for the lifetime of the returned `viUART`.
36    pub const unsafe fn new(base_addr: usize) -> Self {
37        Self { base_addr }
38    }
39
40    /// Update the MMIO base address. Called once by `uart::init` from DTB info.
41    pub fn set_base(&mut self, base: usize) {
42        self.base_addr = base;
43    }
44
45    /// Initialize the UART
46    pub fn init(&mut self) {
47        if self.base_addr == 0 {
48            // No NS16550 MMIO on this board (e.g. Pioneer SG2042 — UART at sv39-inaccessible
49            // address). All console I/O goes via SBI DBCN; skip MMIO init entirely.
50            return;
51        }
52        unsafe {
53            let ptr = self.base_addr as *mut u8;
54
55            // Disable interrupts
56            ptr.add(IER).write_volatile(0x00);
57
58            // Enable + clear FIFO (bit0=enable, bit1=clear RX, bit2=clear TX).
59            ptr.add(FCR).write_volatile(0x07);
60
61            // Set 8-bit mode (Word Length Select bits 0 and 1)
62            ptr.add(LCR).write_volatile(0x03);
63
64            // Keep UART RX interrupts DISABLED (IER=0): the console driver polls
65            // the RHR directly. If RX IRQs were enabled, OpenSBI's M-mode console
66            // handler could drain the RHR before the kernel's S-mode poll sees
67            // the byte, swallowing all keyboard input.
68            ptr.add(IER).write_volatile(0x00);
69        }
70    }
71
72    // /// Write a single byte (Unused - Output via SBI)
73    // pub fn write_byte(&mut self, byte: u8) { ... }
74}

81// Direct writer to avoid stack buffering issues
82struct DirectWriter;
83
84impl fmt::Write for DirectWriter {
85    fn write_str(&mut self, s: &str) -> fmt::Result {
86        for c in s.bytes() {
87            #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
88            {
89                let _ = crate::hal::sbi::console_putchar(c);
90            }
91            #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
92            {
93                write_rpi3_console_byte(c);
94            }
95            #[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
96            {
97                crate::hal::uart_pl011::putchar(c);
98            }
99            #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
100            {
101                crate::hal::uart_16550::putchar(c);
102            }
103        }
104        Ok(())
105    }
106}

108#[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
109fn write_rpi3_console_byte(byte: u8) {
110    if byte == b'\n' {
111        write_rpi3_console_byte(b'\r');
112    }
113    loop {
114        // Synchronous exceptions mask IRQs while `sys_log` prints. Keep draining
115        // the eight-symbol RX FIFO into the 4096-byte kernel ring so full-duplex
116        // terminal traffic cannot overrun while TX waits for space.
117        vi_handle_uart_irq();
118        if crate::hal::uart_bcm_mini::try_putchar(byte) {
119            break;
120        }
121        core::hint::spin_loop();
122    }
123}

125/// Which writer owns the console line that is currently open.
126///
127/// The console is one UART shared by the kernel logger, every Cell and the
128/// hypervisor's guest PL011 forwarding, and each of them writes records
129/// independently. With a single "cursor is at a line start" flag, a record from
130/// one writer landed *inside* the partial line of another: the guest shell
131/// echoes one `sys_log` per keypress, so a diagnostic printed between two keys
132/// shredded the command the operator was typing
133/// (`USER: ping -c 3 192.16[net-loop] turns=215 …`). The owner is whichever
134/// writer last started a line it has not terminated; a *different* writer that
135/// begins a record closes that line first, so no writer can ever land inside
136/// another writer's partial line. Consecutive records from the same writer (the
137/// per-key echo, `print()` followed by its `println()`) stay inline as before.
138#[derive(Clone, Copy)]
139struct ConsoleLine {
140    owner: usize,
141}
142
143/// No writer owns the line: the console is at the start of a fresh line.
144const LINE_START: usize = usize::MAX;
145
146/// Owner tag for kernel `log` records. No task id can hold it, so a kernel
147/// record and a Cell record always see each other as foreign writers.
148const KERNEL_LOG_OWNER: usize = usize::MAX - 1;
149
150impl ConsoleLine {
151    const fn new() -> Self {
152        Self { owner: LINE_START }
153    }
154
155    /// Whether the console is between lines (nothing written since the last `\n`).
156    fn at_line_start(&self) -> bool {
157        self.owner == LINE_START
158    }
159
160    /// Claim the line for `writer`.
161    ///
162    /// Returns `(emit_break, owes_prefix)`: `emit_break` when the line was left
163    /// open by a *different* writer and must be closed with a newline first, and
164    /// `owes_prefix` when the record starts at a line start (either the line was
165    /// already fresh, or the break just made it so) and therefore owes the
166    /// per-line prefix.
167    fn begin_record_state(&mut self, writer: usize) -> (bool, bool) {
168        let at_start = self.at_line_start();
169        let foreign = self.owner != LINE_START && self.owner != writer;
170        self.owner = writer;
171        (foreign, at_start || foreign)
172    }
173
174    /// Record the bytes just written: a trailing newline ends the line.
175    fn wrote(&mut self, chunk: &str) {
176        if chunk.ends_with('\n') {
177            self.owner = LINE_START;
178        }
179    }
180}
181
182/// Console owner state, guarded together with the UART write so the owner a
183/// record sees cannot change between its line-break decision and its first byte.
184static CONSOLE: Spinlock<ConsoleLine> = Spinlock::new(ConsoleLine::new());
185
186/// Emit one `print_user_log` record: the `USER: ` prefix at each line start.
187///
188/// The body is a free function over an `emit` sink so the host test drives the
189/// exact loop an image drives (with a `String` in place of the UART), instead of
190/// re-implementing the discipline beside it. `line` carries the owner across
191/// records; `emit` receives the bytes in order.
192fn emit_user_record(
193    line: &mut ConsoleLine,
194    writer: usize,
195    msg: &str,
196    mut emit: impl FnMut(&str),
197) {
198    let (broke, fresh) = line.begin_record_state(writer);
199    if broke {
200        emit("\n");
201    }
202    let mut at_start = fresh;
203    let mut rest = msg;
204    while !rest.is_empty() {
205        if at_start {
206            emit("USER: ");
207            at_start = false;
208        }
209        match rest.find('\n') {
210            Some(i) => {
211                emit(&rest[..=i]);
212                line.wrote(&rest[..=i]);
213                at_start = true;
214                rest = &rest[i + 1..];
215            }
216            None => {
217                emit(rest);
218                line.wrote(rest);
219                rest = "";
220            }
221        }
222    }
223}

225/// Write one Cell console record owned by `writer`, with the `USER: ` prefix.
226///
227/// USER stdout (cell `println`/`sys_log`) MUST always appear regardless of the
228/// kernel's `log::max_level` — it is application output, not kernel debug chatter.
229/// Routing it through `log::info!` (as `print_user_log` once did) meant lowering
230/// the kernel log level to silence boot spam also silenced the shell prompt.
231///
232/// `writer` is the calling task's id, or [`KERNEL_LOG_OWNER`] for the kernel
233/// logger. A record that would land inside another writer's open line first emits
234/// the newline that closes it — one line, one writer.
235pub fn write_user_record(writer: usize, msg: &str) {
236    let mut line = CONSOLE.lock();
237    emit_user_record(&mut line, writer, msg, |s| {
238        let _ = DirectWriter.write_str(s);
239    });
240}

242/// One kernel `log` record: holds the console for the whole record so its bytes
243/// are emitted contiguously, and closes a foreign open line before it starts.
244struct ConsoleRecord<'a> {
245    state: crate::sync::SpinlockGuard<'a, ConsoleLine>,
246}
247
248impl ConsoleRecord<'_> {
249    /// Append `s` to this record.
250    pub fn write(&mut self, s: &str) {
251        let _ = DirectWriter.write_str(s);
252        self.state.wrote(s);
253    }
254}
255
256impl fmt::Write for ConsoleRecord<'_> {
257    fn write_str(&mut self, s: &str) -> fmt::Result {
258        self.write(s);
259        Ok(())
260    }
261}
262
263/// Begin a console record owned by `writer` (the kernel logger's entry point).
264fn begin_record(writer: usize) -> ConsoleRecord<'static> {
265    let mut state = CONSOLE.lock();
266    let (broke, _) = state.begin_record_state(writer);
267    if broke {
268        let _ = DirectWriter.write_str("\n");
269    }
270    ConsoleRecord { state }
271}

276impl log::Log for SimpleLogger {
277    fn enabled(&self, _metadata: &log::Metadata) -> bool {
278        true
279    }
280
281    fn log(&self, record: &log::Record) {
282        if self.enabled(record.metadata()) {
283            use fmt::Write;
284            let mut out = begin_record(KERNEL_LOG_OWNER);
285            let _ = writeln!(out, "[{:>5}] {}", record.level(), record.args());
286        }
287    }
288
289    fn flush(&self) {}
290}

293pub fn init() {
294    #[cfg(target_arch = "riscv64")]
295    {
296        // Read UART base from DTB (platform::init must have run first).
297        let base = crate::platform::with(|p| p.uart_base);
298        SERIAL.lock().set_base(base);
299        SERIAL.lock().init();
300    }
301    // Register the log backend (works on all architectures; DirectWriter routes
302    // to the correct UART per target_arch inside write_str).
303    let _ = log::set_logger(&LOGGER).map(|()| log::set_max_level(log::LevelFilter::Info));
304    // Logged here (not in platform::init) because the logger only just came up —
305    // platform::init's own log line is emitted before set_logger and is lost.
306    #[cfg(target_arch = "riscv64")]
307    log::info!("[uart] RX/TX base = {:#x}", SERIAL.lock().base_addr);
308}

312use alloc::collections::VecDeque;
313
314// Global RX Buffer (Initialized late)
315pub static RX_BUFFER: Spinlock<Option<VecDeque<u8>>> = Spinlock::new(None);
316const MAX_RX_BUFFERED: usize = 4096;
317
318/// Initialize Input Buffer (Must be called after Heap Init)
319pub fn init_input() {
320    *RX_BUFFER.lock() = Some(VecDeque::with_capacity(MAX_RX_BUFFERED));
321    log::info!("UART Input Buffer Initialized");
322}
323
324/// Poll for a character from the IRQ-filled buffer.
325pub fn getchar() -> Option<u8> {
326    if let Some(buf) = RX_BUFFER.lock().as_mut() {
327        return buf.pop_front();
328    }
329    None
330}

332/// Directly poll the 16550 Receive Holding Register.
333///
334/// This is the most robust input path on QEMU virt: it does not depend on
335/// PLIC interrupt delegation to S-mode (which OpenSBI may keep in M-mode) nor
336/// on the SBI DBCN console-read extension being implemented. Returns the byte
337/// if LSR.DR (Data Ready, bit 0) is set, else `None`.
338pub fn poll_rhr() -> Option<u8> {
339    let base = SERIAL.lock().base_addr;
340    if base == 0 {
341        // No NS16550 MMIO (e.g. Pioneer SG2042). Fall back to SBI DBCN console read
342        // so the interactive shell still receives keystrokes via the firmware console.
343        #[cfg(target_arch = "riscv64")]
344        {
345            let c = crate::hal::sbi::console_getchar();
346            return if c >= 0 { Some(c as u8) } else { None };
347        }
348        #[cfg(not(target_arch = "riscv64"))]
349        return None;
350    }
351    // SAFETY: base_addr is the identity-mapped UART MMIO region, mapped in
352    // init_kernel_paging. RHR (offset 0) is read-only; we gate on LSR.DR anyway.
353    unsafe {
354        let ptr = base as *mut u8;
355        if (ptr.add(LSR).read_volatile() & _LSR_RX_READY) != 0 {
356            Some(ptr.add(_RHR).read_volatile())
357        } else {
358            None
359        }
360    }
361}

363/// Called from the UART RX IRQ handler.
364///
365/// On RISC-V / AArch64: reads from the MMIO base stored in SERIAL.
366/// On x86_64: delegates to the configured HAL port-I/O UART mechanism.
367/// Handles CR→LF normalisation and pushes bytes into RX_BUFFER for the shell.
368#[no_mangle]
369pub extern "Rust" fn vi_handle_uart_irq() {
370    if let Some(buf) = RX_BUFFER.lock().as_mut() {
371        // Drain the UART receive FIFO; stop when no more data is ready.
372        loop {
373            // Read LSR (offset 5) to check Data Ready (bit 0).
374            let (lsr, rhr_byte): (u8, Option<u8>) = {
375                #[cfg(target_arch = "x86_64")]
376                {
377                    let byte = crate::hal::uart_16550::poll_input();
378                    (byte.map_or(0, |_| _LSR_RX_READY), byte)
379                }
380                #[cfg(target_arch = "x86")]
381                {
382                    let lsr = unsafe {
383                        let value: u8;
384                        core::arch::asm!(
385                            "in al, dx",
386                            in("dx") (0x3F8u16 + LSR as u16),
387                            out("al") value,
388                            options(nomem, nostack)
389                        );
390                        value
391                    };
392                    let byte = if lsr & _LSR_RX_READY != 0 {
393                        let value: u8;
394                        unsafe {
395                            core::arch::asm!(
396                                "in al, dx",
397                                in("dx") 0x3F8u16,
398                                out("al") value,
399                                options(nomem, nostack)
400                            );
401                        }
402                        Some(value)
403                    } else {
404                        None
405                    };
406                    (lsr, byte)
407                }
408                #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
409                {
410                    let byte = crate::hal::uart_bcm_mini::poll_rx();
411                    (byte.map_or(0, |_| _LSR_RX_READY), byte)
412                }
413                #[cfg(not(any(
414                    target_arch = "x86_64",
415                    target_arch = "x86",
416                    all(target_arch = "aarch64", feature = "board-rpi3")
417                )))]
418                {
419                    // MMIO path for RISC-V / AArch64.
420                    let serial = SERIAL.lock();
421                    let ptr = serial.base_addr as *mut u8;
422                    // No MMIO UART on this board (e.g. Pioneer SG2042): bail out.
423                    if ptr.is_null() {
424                        break;
425                    }
426                    // SAFETY: MMIO region is identity-mapped and valid.
427                    let lsr_val = unsafe { ptr.add(LSR).read_volatile() };
428                    let byte = if lsr_val & _LSR_RX_READY != 0 {
429                        Some(unsafe { ptr.add(_RHR).read_volatile() })
430                    } else {
431                        None
432                    };
433                    (lsr_val, byte)
434                }
435            };
436            let _ = lsr;
437            match rhr_byte {
438                None => break,
439                Some(c) => {
440                    let c = if c == b'\r' { b'\n' } else { c };
441                    if buf.len() < MAX_RX_BUFFERED {
442                        buf.push_back(c);
443                    }
444                }
445            }
446        }
447    }
448}

453#[cfg(test)]
454mod tests {
455    use super::{emit_user_record, ConsoleLine, KERNEL_LOG_OWNER};
456
457    /// The bytes a real console would receive for one record, through the same
458    /// loop an image runs — only the UART is replaced by a `String`.
459    fn record(line: &mut ConsoleLine, writer: usize, msg: &str) -> alloc::string::String {
460        let mut out = alloc::string::String::new();
461        emit_user_record(line, writer, msg, |s| out.push_str(s));
462        out
463    }
464
465    /// `print()` concatenation and the `USER: ` prefix: consecutive records from
466    /// one writer stay on one line, and the prefix is paid once per line — a
467    /// per-record prefix would render `help` as four `USER: h`/`e`/`l`/`p` lines.
468    #[test]
469    fn same_writer_stays_inline() {
470        let mut line = ConsoleLine::new();
471        assert_eq!(record(&mut line, 7, "h"), "USER: h");
472        assert_eq!(record(&mut line, 7, "elp"), "elp");
473        assert_eq!(record(&mut line, 7, "\n"), "\n");
474        assert_eq!(record(&mut line, 7, "next"), "USER: next");
475    }
476
477    /// The board defect this discipline exists for: the guest echoes its command
478    /// one byte per `sys_log` from the hypervisor, so a diagnostic from another
479    /// cell used to be spliced into the middle of the line the operator was
480    /// typing (`USER: ping -c 3 192.16[net-loop] turns=215 …`). A foreign writer
481    /// now closes that line first instead of landing inside it.
482    #[test]
483    fn guest_echo_is_not_spliced_by_another_writer() {
484        const GUEST: usize = 6;
485        const NET: usize = 5;
486        let mut line = ConsoleLine::new();
487        assert_eq!(record(&mut line, GUEST, "ping -c 3 192.16"), "USER: ping -c 3 192.16");
488        assert_eq!(
489            record(&mut line, NET, "[net-loop] turns=215"),
490            "\nUSER: [net-loop] turns=215"
491        );
492        assert_eq!(record(&mut line, NET, "\n"), "\n");
493        // The guest's next byte starts a fresh line and re-owes the prefix: the
494        // command is split across two lines but never absorbed into the
495        // diagnostic's line.
496        assert_eq!(record(&mut line, GUEST, "8.42.1"), "USER: 8.42.1");
497    }
498
499    /// A kernel `log` record is its own writer: it closes a Cell's open line (the
500    /// break) and starts at a line start, and it never borrows the `USER: `
501    /// prefix — the kernel logger formats through `ConsoleRecord`, which carries
502    /// its own `[ INFO]` tag instead.
503    #[test]
504    fn kernel_record_closes_a_cell_line_first() {
505        let mut line = ConsoleLine::new();
506        assert_eq!(record(&mut line, 5, "Cellos > "), "USER: Cellos > ");
507        assert_eq!(line.begin_record_state(KERNEL_LOG_OWNER), (true, true));
508        line.wrote("[ INFO] x\n");
509        // The kernel record ended its own line, so the Cell starts a fresh one.
510        assert_eq!(record(&mut line, 5, "vfs"), "USER: vfs");
511    }
