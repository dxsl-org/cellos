1//! Minimal 16550 UART Driver for QEMU RISC-V Virt
2//!
3//! Used for kernel logging and early debug output.
4//! Base Address: 0x10000000
5

10/// UART Registers (offset from base)
11const _RHR: usize = 0; // Receive Holding Register (read)
12const _THR: usize = 0; // Transmit Holding Register (write)
13const IER: usize = 1; // Interrupt Enable Register
14const FCR: usize = 2; // FIFO Control Register
15const _ISR: usize = 2; // Interrupt Status Register
16const LCR: usize = 3; // Line Control Register
17const LSR: usize = 5; // Line Status Register
18
19/// Line Status Flags
20const _LSR_RX_READY: u8 = 1 << 0;
21const _LSR_TX_EMPTY: u8 = 1 << 5;
22
23#[allow(non_camel_case_types)]
24pub struct viUART {
25    base_addr: usize,
26}
27
28impl viUART {
29    /// Create a new viUART instance (unsafe because base_addr must be valid)
30    ///
31    /// # Safety
32    /// `base_addr` must be either 0 (sentinel meaning "no NS16550 MMIO on this
33    /// board" — all I/O methods on this instance check for it and fall back to
34    /// SBI DBCN/port-I/O) or a valid, identity-mapped 16550 UART MMIO base that
35    /// remains mapped for the lifetime of the returned `viUART`.
36    pub const unsafe fn new(base_addr: usize) -> Self {
37        Self { base_addr }
38    }
39
40    /// Update the MMIO base address. Called once by `uart::init` from DTB info.
41    pub fn set_base(&mut self, base: usize) {
42        self.base_addr = base;
43    }
44
45    /// Initialize the UART
46    pub fn init(&mut self) {
47        if self.base_addr == 0 {
48            // No NS16550 MMIO on this board (e.g. Pioneer SG2042 — UART at sv39-inaccessible
49            // address). All console I/O goes via SBI DBCN; skip MMIO init entirely.
50            return;
51        }
52        unsafe {
53            let ptr = self.base_addr as *mut u8;
54
55            // Disable interrupts
56            ptr.add(IER).write_volatile(0x00);
57
58            // Enable + clear FIFO (bit0=enable, bit1=clear RX, bit2=clear TX).
59            ptr.add(FCR).write_volatile(0x07);
60
61            // Set 8-bit mode (Word Length Select bits 0 and 1)
62            ptr.add(LCR).write_volatile(0x03);
63
64            // Keep UART RX interrupts DISABLED (IER=0): the console driver polls
65            // the RHR directly. If RX IRQs were enabled, OpenSBI's M-mode console
66            // handler could drain the RHR before the kernel's S-mode poll sees
67            // the byte, swallowing all keyboard input.
68            ptr.add(IER).write_volatile(0x00);
69        }
70    }
71
72    // /// Write a single byte (Unused - Output via SBI)
73    // pub fn write_byte(&mut self, byte: u8) { ... }
74}

84impl fmt::Write for DirectWriter {
85    fn write_str(&mut self, s: &str) -> fmt::Result {
86        for c in s.bytes() {
87            #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
88            {
89                let _ = crate::hal::sbi::console_putchar(c);
90            }
91            #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
92            {
93                write_rpi3_console_byte(c);
94            }
95            #[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
96            {
97                crate::hal::uart_pl011::putchar(c);
98            }
99            #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
100            {
101                crate::hal::uart_16550::putchar(c);
102            }
103        }
104        Ok(())
105    }
106}

108#[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
109fn write_rpi3_console_byte(byte: u8) {
110    if byte == b'\n' {
111        write_rpi3_console_byte(b'\r');
112    }
113    loop {
114        // Synchronous exceptions mask IRQs while `sys_log` prints. Keep draining
115        // the eight-symbol RX FIFO into the 4096-byte kernel ring so full-duplex
116        // terminal traffic cannot overrun while TX waits for space.
117        vi_handle_uart_irq();
118        if crate::hal::uart_bcm_mini::try_putchar(byte) {
119            break;
120        }
121        core::hint::spin_loop();
122    }
123}

125/// Which writer owns the console line that is currently open.
126///
127/// The console is one UART shared by the kernel logger, every Cell and the
128/// hypervisor's guest PL011 forwarding, and each of them writes records
129/// independently. With a single "cursor is at a line start" flag, a record from
130/// one writer landed *inside* the partial line of another: the guest shell
131/// echoes one `sys_log` per keypress, so a diagnostic printed between two keys
132/// shredded the command the operator was typing
133/// (`USER: ping -c 3 192.16[net-loop] turns=215 …`). The owner is whichever
134/// writer emitted the last bytes of a line it has not terminated; a *different*
135/// writer that writes closes that line first, so no writer can ever land inside
136/// another writer's partial line. Consecutive records from the same writer (the
137/// per-key echo, `print()` followed by its `println()`, a long write split into
138/// several records) stay inline.
139///
140/// The owner is derived from the bytes alone, per fragment: a fragment ending in
141/// `\n` leaves the console at a line start, any other non-empty fragment leaves
142/// its writer owning an open line. A record's unterminated tail after an inner
143/// newline therefore owns the line it opened, and a record that writes nothing
144/// changes nothing.
145///
146/// Every physical line carries exactly one tag at its start: `USER: ` for a Cell
147/// line, `[LEVEL]` for a kernel line. When a kernel record interrupts an open
148/// Cell line, that Cell line is *suspended*: the kernel line starts on its own
149/// line, and the suspended writer's next bytes start a line tagged `USER+ `
150/// ([`USER_CONTINUATION`]) instead of `USER: `. A `USER+ ` line is the exact
151/// continuation of the last `USER: `/`USER+ ` line before the kernel lines
152/// between them — the forced line break is not part of the Cell's output — so a
153/// parser rebuilds the Cell's bytes by dropping kernel lines and joining each
154/// `USER+ ` line onto the preceding Cell line. Any other Cell writer starting a
155/// line ends the suspension (its line would sit between the two halves), and the
156/// suspended writer then resumes with a plain `USER: `: a split by another Cell
157/// is a real, unmarked line break.
158#[derive(Clone, Copy)]
159struct ConsoleLine {
160    owner: usize,
161    /// The Cell writer whose open line a kernel record closed, until it resumes.
162    suspended: Option<usize>,
163}
164
165/// No writer owns the line: the console is at the start of a fresh line.
166const LINE_START: usize = usize::MAX;
167
168/// Owner tag for kernel `log` records. No task id can hold it, so a kernel
169/// record and a Cell record always see each other as foreign writers.
170const KERNEL_LOG_OWNER: usize = usize::MAX - 1;
171
172/// Per-line tag of Cell console output; kernel records carry their own tag.
173const USER_PREFIX: &str = "USER: ";
174
175/// Tag of a Cell line that continues the Cell line a kernel record interrupted.
176const USER_CONTINUATION: &str = "USER+ ";
177
178impl ConsoleLine {
179    const fn new() -> Self {
180        Self { owner: LINE_START, suspended: None }
181    }
182}
183
184/// Console owner state, guarded together with the UART write so the owner a
185/// record sees cannot change between its line-break decision and its first byte.
186static CONSOLE: Spinlock<ConsoleLine> = Spinlock::new(ConsoleLine::new());
187
188/// Line tags of one writer: `prefix` starts each line it begins, `continuation`
189/// starts the line resuming its own line that a kernel record interrupted.
190struct Tags {
191    prefix: &'static str,
192    continuation: &'static str,
193}
194
195const USER_TAGS: Tags = Tags { prefix: USER_PREFIX, continuation: USER_CONTINUATION };
196/// Kernel records format their own `[LEVEL]` tag.
197const KERNEL_TAGS: Tags = Tags { prefix: "", continuation: "" };
198
199/// Emit `msg` as console bytes owned by `writer`, tagging every physical line it
200/// begins with `tags`.
201///
202/// Per line fragment: when the open line belongs to another writer it is closed
203/// with `\n` first (a kernel writer closing a Cell line suspends that Cell, see
204/// [`ConsoleLine`]); when the fragment starts a line (fresh, or just closed) it
205/// gets the prefix, or the continuation tag when it resumes `writer`'s suspended
206/// line; a fragment continuing `writer`'s own open line gets neither. Cell
207/// records and kernel `log` records both run through here.
208///
209/// The body is a free function over an `emit` sink so the host test drives the
210/// exact loop an image drives (with a `String` in place of the UART), instead of
211/// re-implementing the discipline beside it. `line` carries the state across
212/// records; `emit` receives the bytes in order.
213fn emit_record(
214    line: &mut ConsoleLine,
215    writer: usize,
216    tags: &Tags,
217    msg: &str,
218    mut emit: impl FnMut(&str),
219) {
220    let kernel = writer == KERNEL_LOG_OWNER;
221    let mut rest = msg;
222    while !rest.is_empty() {
223        let end = rest.find('\n').map_or(rest.len(), |i| i + 1);
224        let (fragment, tail) = rest.split_at(end);
225        if line.owner != writer {
226            if line.owner != LINE_START {
227                emit("\n");
228                if kernel {
229                    line.suspended = Some(line.owner);
230                }
231            }
232            let tag = if line.suspended == Some(writer) { tags.continuation } else { tags.prefix };
233            if !kernel {
234                line.suspended = None;
235            }
236            if !tag.is_empty() {
237                emit(tag);
238            }
239        }
240        emit(fragment);
241        line.owner = if fragment.ends_with('\n') { LINE_START } else { writer };
242        rest = tail;
243    }
244}

246/// Write one Cell console record owned by `writer`, tagged `USER: ` per line
247/// (`USER+ ` on the line resuming a line a kernel record interrupted).
248///
249/// USER stdout (cell `println`/`sys_log`) MUST always appear regardless of the
250/// kernel's `log::max_level` — it is application output, not kernel debug chatter.
251/// Routing it through `log::info!` (as `print_user_log` once did) meant lowering
252/// the kernel log level to silence boot spam also silenced the shell prompt.
253///
254/// `writer` is the calling task's id. A record that would land inside another
255/// writer's open line first emits the newline that closes it — one line, one
256/// writer, one tag.
257pub fn write_user_record(writer: usize, msg: &str) {
258    let mut line = CONSOLE.lock();
259    emit_record(&mut line, writer, &USER_TAGS, msg, |s| {
260        let _ = DirectWriter.write_str(s);
261    });
262}
263
264/// One kernel `log` record: holds the console for the whole record so its bytes
265/// are emitted contiguously, and closes a foreign open line before its first byte.
266struct ConsoleRecord<'a> {
267    state: crate::sync::SpinlockGuard<'a, ConsoleLine>,
268}
269
270impl fmt::Write for ConsoleRecord<'_> {
271    fn write_str(&mut self, s: &str) -> fmt::Result {
272        emit_record(&mut self.state, KERNEL_LOG_OWNER, &KERNEL_TAGS, s, |s| {
273            let _ = DirectWriter.write_str(s);
274        });
275        Ok(())
276    }
277}

279/// Begin a kernel `log` record (the kernel logger's entry point).
280fn begin_record() -> ConsoleRecord<'static> {
281    ConsoleRecord { state: CONSOLE.lock() }
282}

304pub fn init() {
305    #[cfg(target_arch = "riscv64")]
306    {
307        // Read UART base from DTB (platform::init must have run first).
308        let base = crate::platform::with(|p| p.uart_base);
309        SERIAL.lock().set_base(base);
310        SERIAL.lock().init();
311    }
312    // Register the log backend (works on all architectures; DirectWriter routes
313    // to the correct UART per target_arch inside write_str).
314    let _ = log::set_logger(&LOGGER).map(|()| log::set_max_level(log::LevelFilter::Info));
315    // Logged here (not in platform::init) because the logger only just came up —
316    // platform::init's own log line is emitted before set_logger and is lost.
317    #[cfg(target_arch = "riscv64")]
318    log::info!("[uart] RX/TX base = {:#x}", SERIAL.lock().base_addr);
319}

326pub static RX_BUFFER: Spinlock<Option<VecDeque<u8>>> = Spinlock::new(None);
327const MAX_RX_BUFFERED: usize = 4096;

329/// Initialize Input Buffer (Must be called after Heap Init)
330pub fn init_input() {
331    let mut input = RX_BUFFER.lock();
332    #[cfg(cellos_p01_trace)]
333    let _semantic = crate::memory::heap::p01_owner::creation(None,
334        crate::memory::heap::p01_owner::DRIVER_SHARED);
335    *input = Some(VecDeque::with_capacity(MAX_RX_BUFFERED));
336    #[cfg(cellos_p01_trace)]
337    drop(_semantic);
338    drop(input);
339    log::info!("UART Input Buffer Initialized");
340}

342/// Poll for a character from the IRQ-filled buffer.
343pub fn getchar() -> Option<u8> {
344    if let Some(buf) = RX_BUFFER.lock().as_mut() {
345        return buf.pop_front();
346    }
347    None

350/// Directly poll the 16550 Receive Holding Register.
351///
352/// This is the most robust input path on QEMU virt: it does not depend on
353/// PLIC interrupt delegation to S-mode (which OpenSBI may keep in M-mode) nor
354/// on the SBI DBCN console-read extension being implemented. Returns the byte
355/// if LSR.DR (Data Ready, bit 0) is set, else `None`.
356pub fn poll_rhr() -> Option<u8> {
357    let base = SERIAL.lock().base_addr;
358    if base == 0 {
359        // No NS16550 MMIO (e.g. Pioneer SG2042). Fall back to SBI DBCN console read
360        // so the interactive shell still receives keystrokes via the firmware console.
361        #[cfg(target_arch = "riscv64")]
362        {
363            let c = crate::hal::sbi::console_getchar();
364            return if c >= 0 { Some(c as u8) } else { None };
365        }
366        #[cfg(not(target_arch = "riscv64"))]
367        return None;
368    }
369    // SAFETY: base_addr is the identity-mapped UART MMIO region, mapped in
370    // init_kernel_paging. RHR (offset 0) is read-only; we gate on LSR.DR anyway.
371    unsafe {
372        let ptr = base as *mut u8;
373        if (ptr.add(LSR).read_volatile() & _LSR_RX_READY) != 0 {
374            Some(ptr.add(_RHR).read_volatile())
375        } else {
376            None
377        }
378    }
379}

381/// Called from the UART RX IRQ handler.
382///
383/// On RISC-V / AArch64: reads from the MMIO base stored in SERIAL.
384/// On x86_64: delegates to the configured HAL port-I/O UART mechanism.
385/// Pushes received bytes, unchanged, into RX_BUFFER. CR is not rewritten: the only
386/// reader (`console_drv`) relays raw bytes to the input service and converts CR→LF
387/// itself in the legacy no-input-service fallback.
388#[no_mangle]
389pub extern "Rust" fn vi_handle_uart_irq() {
390    if let Some(buf) = RX_BUFFER.lock().as_mut() {
391        // Drain the UART receive FIFO; stop when no more data is ready.
392        loop {
393            // Read LSR (offset 5) to check Data Ready (bit 0).
394            let (lsr, rhr_byte): (u8, Option<u8>) = {
395                #[cfg(target_arch = "x86_64")]
396                {
397                    let byte = crate::hal::uart_16550::poll_input();
398                    (byte.map_or(0, |_| _LSR_RX_READY), byte)
399                }
400                #[cfg(target_arch = "x86")]
401                {
402                    let lsr = unsafe {
403                        let value: u8;
404                        core::arch::asm!(
405                            "in al, dx",
406                            in("dx") (0x3F8u16 + LSR as u16),
407                            out("al") value,
408                            options(nomem, nostack)
409                        );
410                        value
411                    };
412                    let byte = if lsr & _LSR_RX_READY != 0 {
413                        let value: u8;
414                        unsafe {
415                            core::arch::asm!(
416                                "in al, dx",
417                                in("dx") 0x3F8u16,
418                                out("al") value,
419                                options(nomem, nostack)
420                            );
421                        }
422                        Some(value)
423                    } else {
424                        None
425                    };
426                    (lsr, byte)
427                }
428                #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
429                {
430                    let byte = crate::hal::uart_bcm_mini::poll_rx();
431                    (byte.map_or(0, |_| _LSR_RX_READY), byte)
432                }
433                #[cfg(not(any(
434                    target_arch = "x86_64",
435                    target_arch = "x86",
436                    all(target_arch = "aarch64", feature = "board-rpi3")
437                )))]
438                {
439                    // MMIO path for RISC-V / AArch64.
440                    let serial = SERIAL.lock();
441                    let ptr = serial.base_addr as *mut u8;
442                    // No MMIO UART on this board (e.g. Pioneer SG2042): bail out.
443                    if ptr.is_null() {
444                        break;
445                    }
446                    // SAFETY: MMIO region is identity-mapped and valid.
447                    let lsr_val = unsafe { ptr.add(LSR).read_volatile() };
448                    let byte = if lsr_val & _LSR_RX_READY != 0 {
449                        Some(unsafe { ptr.add(_RHR).read_volatile() })
450                    } else {
451                        None
452                    };
453                    (lsr_val, byte)
454                }
455            };
456            let _ = lsr;
457            match rhr_byte {
458                None => break,
459                Some(c) => {
460                    if buf.len() < MAX_RX_BUFFERED {
461                        buf.push_back(c);
462                    }
463                }
464            }
465        }
466    }
467}

478    /// The bytes a real console would receive for one Cell record, through the
479    /// same loop an image runs — only the UART is replaced by a `String`.
480    fn record(line: &mut ConsoleLine, writer: usize, msg: &str) -> String {
481        let mut out = String::new();
482        emit_record(line, writer, &USER_TAGS, msg, |s| out.push_str(s));
483        out
484    }
485
486    /// One kernel `log` record as the logger emits it: `writeln!` hands the
487    /// formatted line and its newline to `ConsoleRecord` as separate fragments.
488    fn kernel(line: &mut ConsoleLine, text: &str) -> String {
489        let mut out = String::new();
490        for fragment in [text, "\n"] {
491            emit_record(line, KERNEL_LOG_OWNER, &KERNEL_TAGS, fragment, |s| out.push_str(s));
492        }
493        out
494    }
495
496    /// Rebuild one Cell's bytes from a console capture by the documented rule:
497    /// drop kernel lines, strip the line-start `USER: ` tag, and join every
498    /// `USER+ ` line onto the Cell line before it. Panics on any physical line
499    /// that does not start with exactly one known tag.
500    fn rebuild(console: &str) -> String {
501        let mut out = String::new();
502        for physical in console.split_inclusive('\n') {
503            if physical.starts_with('[') {
504                continue;
505            }
506            if let Some(body) = physical.strip_prefix("USER: ") {
507                out.push_str(body);
508            } else if let Some(body) = physical.strip_prefix("USER+ ") {
509                // The forced break that preceded the kernel line is not Cell output.
510                assert_eq!(out.pop(), Some('\n'), "continuation without an interrupted line");
511                out.push_str(body);
512            } else {
513                panic!("untagged physical line {physical:?}");
514            }
515        }
516        out
517    }

522    #[test]

586        assert_eq!(record(&mut line, 5, "\n"), "USER+ \n");
587        assert_eq!(record(&mut line, 5, "done\n"), "USER: done\n");
588    }

602    /// End to end over the shape of the phase03 capture: a long multi-line
603    /// stream relayed in 512-byte records, with kernel lines landing between
604    /// records (mid-line and at line starts). Every physical line carries
605    /// exactly one tag at its start, and the documented rebuild rule recovers
606    /// the Cell's bytes exactly.
607    #[test]
608    fn chunked_stream_with_kernel_lines_rebuilds_exactly() {
609        let mut stream = String::new();
610        for i in 0..40 {
611            stream.push_str(&alloc::format!(
612                "{{\"frame\":\"row\",\"value\":{{\"cell_id\":{i},\"name\":\"cell-{i}\",\"pad\":\"{}\"}}}}\n",
613                "x".repeat(i * 7 % 61)
614            ));
615        }
616        stream.push_str("unterminated tail");
617        let bytes = stream.as_bytes();
618        let chunks: Vec<&str> = bytes
619            .chunks(512)
620            .map(|c| core::str::from_utf8(c).unwrap())
621            .collect();
622        assert!(chunks.len() > 4);
623        let mut line = ConsoleLine::new();
624        let mut console = String::new();
625        for (n, chunk) in chunks.iter().enumerate() {
626            console.push_str(&record(&mut line, 7, chunk));
627            if n % 2 == 1 {
628                console.push_str(&kernel(&mut line, &alloc::format!("[ INFO] Syscall::Exit: task {n}")));
629            }
630        }
631        console.push_str(&kernel(&mut line, "[ INFO] end"));
632        for physical in console.lines() {
633            let body = physical
634                .strip_prefix("USER: ")
635                .or_else(|| physical.strip_prefix("USER+ "))
636                .or_else(|| physical.strip_prefix("[ INFO] "))
637                .unwrap_or_else(|| panic!("untagged physical line {physical:?}"));
638            assert!(!body.contains("USER"), "tag inside a line: {physical:?}");
639            assert!(!body.contains("[ INFO]"), "kernel record spliced: {physical:?}");
640        }
641        assert!(console.contains("\nUSER+ "), "fixture must exercise a mid-line kernel interrupt");
642        assert_eq!(rebuild(&console), stream + "\n");
643    }
