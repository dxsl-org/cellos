1//! Kernel-owned, bounded request/reply operations. All transitions are serialized
2//! by SCHEDULER; neither a reply nor a wake depends on caller-owned memory.
3
4use super::{copy_glue::TaskCopyView, ipc_wire, scheduler::Scheduler, tcb::{Task, TaskState}, SCHEDULER};
5use alloc::vec::Vec;
6use core::sync::atomic::{AtomicUsize, Ordering};

8pub const MAX_OPERATIONS: usize = 64;
9/// Accepted operations have a kernel-enforced terminal deadline (30 seconds at
10/// the scheduler's 10 ms tick). Service replies that never arrive cannot hold
11/// completion reservations forever.
12pub const OPERATION_TIMEOUT_TICKS: u64 = 3_000;
13static NEXT_OPERATION: AtomicUsize = AtomicUsize::new(1);
14
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum Failure { Busy, PeerGone, Invalid, TooSmall }
17
18#[derive(Debug)]
19enum Phase { Queued, Dispatched, Terminal { kind: u32, len: usize } }
20
21#[derive(Debug)]
22pub struct Operation {
23    id: usize,
24    owner_cell: u64,
25    owner_generation: u64,
26    peer_tid: usize,
27    peer_cell: u64,
28    peer_generation: u64,
29    deadline: u64,
30    phase: Phase,
31    reply: ipc_wire::IpcWireMessage,
32}
33
34impl Operation {
35    fn owned_by(&self, task: &Task) -> bool {
36        self.owner_cell == task.cell_id.0 && self.owner_generation == task.cell_generation
37    }
38}

40/// Task-local, kernel-charged storage. Tokens are global monotonic identifiers,
41/// never slot indexes; reuse after take cannot alias a late service reply.
42#[derive(Default)]
43pub struct Operations(Vec<Operation>);
44
45impl Operations {
46    pub fn new() -> Self { Self(Vec::new()) }
47    pub fn len(&self) -> usize { self.0.len() }
48    pub fn find(&self, id: usize) -> Option<&Operation> { self.0.iter().find(|op| op.id == id) }
49    fn find_mut(&mut self, id: usize) -> Option<&mut Operation> { self.0.iter_mut().find(|op| op.id == id) }
50    fn remove(&mut self, id: usize) -> Option<Operation> {
51        let pos = self.0.iter().position(|op| op.id == id)?;
52        Some(self.0.remove(pos))
53    }
54    fn push(&mut self, op: Operation) -> Result<(), Failure> {
55        if self.0.len() == MAX_OPERATIONS { return Err(Failure::Busy); }
56        if self.0.len() == self.0.capacity() {
57            let previous = super::hart_local::current_cell_id();
58            super::hart_local::set_current_cell_id(0);
59            let reserved = self.0.try_reserve(1);
60            super::hart_local::set_current_cell_id(previous);
61            reserved.map_err(|_| Failure::Busy)?;
62        }
63        self.0.push(op);
64        Ok(())
65    }
66    pub fn peer_died(&mut self, tid: usize, cell: u64, generation: u64) -> bool {
67        let mut changed = false;
68        for op in &mut self.0 {
69            if op.peer_tid == tid && op.peer_cell == cell && op.peer_generation == generation
70                && !matches!(op.phase, Phase::Terminal { .. }) {
71                op.phase = Phase::Terminal { kind: api::syscall::ipc_status::PEER_GONE, len: 0 };
72                changed = true;
73            }
74        }
75        changed
76    }
77    fn has_terminal(&self) -> bool { self.0.iter().any(|op| matches!(op.phase, Phase::Terminal { .. })) }
78}
79
80impl Drop for Operations {
81    fn drop(&mut self) {
82        let rows = core::mem::take(&mut self.0);
83        let previous = super::hart_local::current_cell_id();
84        super::hart_local::set_current_cell_id(0);
85        drop(rows);
86        super::hart_local::set_current_cell_id(previous);
87    }
88}

90fn live(task: &Task) -> bool { !matches!(task.state, TaskState::Retiring | TaskState::Terminated) }

92fn wake(sched: &mut Scheduler, owner: usize) {
93    if let Some(task) = sched.tasks.get_mut(&owner) {
94        if matches!(task.state, TaskState::WaitIpc { .. }) {
95            task.state = TaskState::Ready;
96            let priority = sched.push_ready(owner);
97            sched.pend_preempt_if_needed(priority);
98        }
99    }
100}

102fn new_token() -> Result<usize, Failure> {
103    NEXT_OPERATION.fetch_update(Ordering::Relaxed, Ordering::Relaxed,
104        |next| (next <= isize::MAX as usize).then_some(next + 1))
105        .map_err(|_| Failure::Busy)
106}

108pub fn submit(owner: usize, peer: usize, ptr: usize, len: usize) -> Result<usize, Failure> {
109    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD { return Err(Failure::Invalid); }
110    let (view, binding, peer_binding) = {
111        let guard = SCHEDULER.lock();
112        let sched = guard.as_ref().ok_or(Failure::PeerGone)?;
113        let task = sched.tasks.get(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
114        let target = sched.tasks.get(&peer).filter(|task| live(task)).ok_or(Failure::PeerGone)?;
115        (TaskCopyView::of(task), (task.cell_id.0, task.cell_generation),
116         (target.cell_id.0, target.cell_generation))
117    };
118    let token = new_token()?;
119    let header = ipc_wire::IpcWireHeader {
120        sender_tid: owner, sender_cell_id: binding.0,
121        sender_generation: binding.1,
122        delivery_id: super::next_delivery_id(), async_op: token,
123    };
124    // Copy outside SCHEDULER; no peer or user pointer persists in a task slot.
125    let wire = ipc_wire::IpcWireMessage::try_from_user(header, &view, ptr, len)
126        .map_err(|_| Failure::Invalid)?;
127    let reply_header = ipc_wire::IpcWireHeader {
128        sender_tid: peer, sender_cell_id: peer_binding.0,
129        sender_generation: peer_binding.1, delivery_id: 0, async_op: token,
130    };
131    let reply = ipc_wire::IpcWireMessage::try_new(
132        reply_header, &[0u8; ipc_wire::MAX_IPC_WIRE_PAYLOAD]).map_err(|_| Failure::Busy)?;
133    let mut guard = SCHEDULER.lock();
134    let sched = guard.as_mut().ok_or(Failure::PeerGone)?;
135    let caller = sched.tasks.get(&owner).filter(|task| live(task)
136        && (task.cell_id.0, task.cell_generation) == binding).ok_or(Failure::Invalid)?;
137    if caller.async_operations.len() >= MAX_OPERATIONS { return Err(Failure::Busy); }
138    let receiver = sched.tasks.get(&peer).filter(|task| live(task)
139        && (task.cell_id.0, task.cell_generation) == peer_binding).ok_or(Failure::PeerGone)?;
140    if super::paused_target_rejects(sched, owner, peer) { return Err(Failure::Busy); }
141    if receiver.pending_msgs.len() >= super::tcb::HOTSWAP_MSG_QUEUE_DEPTH { return Err(Failure::Busy); }
142    sched.tasks.get_mut(&owner).ok_or(Failure::Invalid)?.async_operations.push(Operation {
143        id: token, owner_cell: binding.0, owner_generation: binding.1,
144        peer_tid: peer, peer_cell: peer_binding.0, peer_generation: peer_binding.1,
145        deadline: (super::system_ticks() as u64).saturating_add(OPERATION_TIMEOUT_TICKS),
146        phase: Phase::Queued, reply,
147    })?;
148    let target = sched.tasks.get_mut(&peer).ok_or(Failure::PeerGone)?;
149    let recv_eligible = matches!(target.state, TaskState::Recv { mask, .. } if mask == 0 || mask == owner);
150    if super::queue_wire_msg(target, wire, super::tcb::HOTSWAP_MSG_QUEUE_DEPTH).is_err() {
151        sched.tasks.get_mut(&owner).unwrap().async_operations.remove(token);
152        return Err(Failure::Busy);
153    }
154    let wake_cause = super::wake_after_ipc_publish(target, recv_eligible);
155    if wake_cause.made_runnable() {
156        let priority = sched.push_ready(peer);
157        sched.pend_preempt_if_needed(priority);
158    }
159    Ok(token)
160}

162/// Snapshotting the request is the dispatch boundary. If a copy-out later
163/// faults, cancellation stays conservatively indeterminate, not falsely safe.
164pub fn dispatch(sched: &mut Scheduler, peer: usize, header: ipc_wire::IpcWireHeader) {
165    if header.async_op == 0 { return; }
166    let Some(provider) = sched.tasks.get(&peer).filter(|task| live(task)) else { return; };
167    let peer_binding = (provider.cell_id.0, provider.cell_generation);
168    let Some(owner) = sched.tasks.get_mut(&header.sender_tid).filter(|task| live(task)) else { return; };
169    if owner.cell_id.0 != header.sender_cell_id || owner.cell_generation != header.sender_generation { return; }
170    let owner_binding = (owner.cell_id.0, owner.cell_generation);
171    let Some(op) = owner.async_operations.find_mut(header.async_op) else { return; };
172    if (op.owner_cell, op.owner_generation) == owner_binding
173        && (op.peer_tid, op.peer_cell, op.peer_generation) == (peer, peer_binding.0, peer_binding.1)
174        && matches!(op.phase, Phase::Queued) {
175        op.phase = Phase::Dispatched;
176    }
177}

179fn terminal(sched: &mut Scheduler, owner_tid: usize, token: usize, provider: usize,
180            exact_peer: bool, ptr: usize, len: usize) -> Result<(), Failure> {
181    let Some(owner) = sched.tasks.get(&owner_tid).filter(|task| live(task)) else {
182        return Err(Failure::PeerGone);
183    };
184    let Some(op) = owner.async_operations.find(token) else { return Err(Failure::Invalid); };
185    let Some(peer) = sched.tasks.get(&provider).filter(|task| live(task)) else {
186        return Err(Failure::PeerGone);
187    };
188    // Explicit replies may be delegated to a worker of the accepted provider's
189    // cell incarnation. Only the implicit bridge requires the served TID too.
190    // Authority always comes from scheduler identity, never caller input.
191    if !op.owned_by(owner) || (exact_peer && op.peer_tid != provider)
192        || op.peer_cell != peer.cell_id.0 || op.peer_generation != peer.cell_generation
193        || !matches!(op.phase, Phase::Dispatched) { return Err(Failure::Invalid); }
194    let view = TaskCopyView::of(peer);
195    let op = sched.tasks.get_mut(&owner_tid).unwrap().async_operations.find_mut(token).unwrap();
196    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD {
197        op.phase = Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, len: 0 };
198        wake(sched, owner_tid);
199        return Err(Failure::Invalid);
200    }
201    let copied = view.read_into(ptr, &mut op.reply.as_mut_slice()[..len]);
202    op.phase = Phase::Terminal {
203        kind: if copied.is_ok() { api::syscall::ipc_status::REPLY }
204            else { api::syscall::ipc_status::INDETERMINATE },
205        len: if copied.is_ok() { len } else { 0 },
206    };
207    wake(sched, owner_tid);
208    copied.map_err(|_| Failure::Invalid)
209}

211/// Only a Send to the exact currently served caller can be a legacy reply.
212/// Its token remains as a tombstone until another receive replaces the context:
213/// duplicate/cancelled/late replies must not fall through to the raw mailbox.
214/// Services sending same-destination events or deferring across receives must
215/// capture the token and use explicit replies instead of this bridge.
216pub fn reply_current(peer: usize, destination: usize, ptr: usize, len: usize)
217    -> Option<Result<(), Failure>> {
218    let mut guard = SCHEDULER.lock();
219    reply_current_in_sched(guard.as_mut()?, peer, destination, ptr, len)
220}
221
222fn reply_current_in_sched(sched: &mut Scheduler, peer: usize, destination: usize,
223                          ptr: usize, len: usize) -> Option<Result<(), Failure>> {
224    let provider = sched.tasks.get(&peer).filter(|task| live(task))?;
225    let token = provider.current_async_op;
226    if token == 0 || provider.current_caller != Some(destination) { return None; }
227    let owner_binding = (provider.current_caller_cell_id, provider.current_caller_cell_generation);
228    let owner_matches = sched.tasks.get(&destination)
229        .is_some_and(|task| (task.cell_id.0, task.cell_generation) == owner_binding);
230    if !owner_matches { return Some(Ok(())); }
231    Some(match terminal(sched, destination, token, peer, true, ptr, len) {
232        // Invalid tokens and terminal results are quarantined, never raw Send.
233        Err(Failure::Invalid) => Ok(()),
234        other => other,
235    })
236}

249fn reply_in_sched(sched: &mut Scheduler, peer: usize, token: usize,
250                  ptr: usize, len: usize) -> Result<(), Failure> {
251    if token == 0 { return Err(Failure::Invalid); }
252    let owner = sched.tasks.iter().find_map(|(&id, task)| task.async_operations.find(token)
253        .map(|_| id));
254    let result = owner.ok_or(Failure::Invalid)
255        .and_then(|owner| terminal(sched, owner, token, peer, false, ptr, len));
256    // An explicit response ends this request even if cancellation/take already
257    // retired its token. Otherwise later one-way events would be mistaken for
258    // stale implicit replies. VFS's exact lease cleanup is handled by syscall.
259    if let Some(binding) = sched.tasks.get(&peer).filter(|task| live(task))
260        .map(|task| (task.cell_id, task.cell_generation)) {
261        for task in sched.tasks.values_mut() {
262            if (task.cell_id, task.cell_generation) == binding
263                && task.current_async_op == token
264                && !crate::fast_ipc::is_registered_vfs_cell(task.cell_id.0 as usize) {
265                task.clear_current_caller_context();
266            }
267        }
268    }
269    result
270}

278fn cancel_in_sched(sched: &mut Scheduler, owner: usize, token: usize) -> Result<(), Failure> {
279    let task = sched.tasks.get(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
280    let op = task.async_operations.find(token).ok_or(Failure::Invalid)?;
281    if !op.owned_by(task) { return Err(Failure::Invalid); }
282    let queued = matches!(op.phase, Phase::Queued);
283    let peer = op.peer_tid;
284    if queued {
285        // No receiver snapshot has begun; remove only the matching token.
286        if let Some(receiver) = sched.tasks.get_mut(&peer) {
287            if let Some(pos) = receiver.pending_msgs.iter().position(|msg|
288                msg.wire_header().is_some_and(|header| header.async_op == token)) {
289                receiver.pending_msgs.remove(pos);
290            }
291        }
292    }
293    let op = sched.tasks.get_mut(&owner).unwrap().async_operations.find_mut(token).unwrap();
294    if !matches!(op.phase, Phase::Terminal { .. }) {
295        op.phase = Phase::Terminal { kind: if queued { api::syscall::ipc_status::CANCELLED }
296            else { api::syscall::ipc_status::INDETERMINATE }, len: 0 };
297        wake(sched, owner);
298    }
299    Ok(())
300}

302pub fn take(owner: usize, token: usize, reply_ptr: usize, reply_len: usize,
303            status_ptr: usize) -> Result<bool, Failure> {
304    let mut guard = SCHEDULER.lock();
305    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
306    let task = sched.tasks.get(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
307    let op = task.async_operations.find(token).ok_or(Failure::Invalid)?;
308    if !op.owned_by(task) { return Err(Failure::Invalid); }
309    let Phase::Terminal { kind, len } = &op.phase else { return Ok(false); };
310    let data = &op.reply.as_slice()[..*len];
311    if data.len() > reply_len { return Err(Failure::TooSmall); }
312    let view = TaskCopyView::of(task);
313    view.validate_writable(status_ptr, api::syscall::IPC_STATUS_LEN).map_err(|_| Failure::Invalid)?;
314    if !data.is_empty() {
315        view.validate_writable(reply_ptr, data.len()).map_err(|_| Failure::Invalid)?;
316        view.write_bytes(reply_ptr, data).map_err(|_| Failure::Invalid)?;
317    }
318    let mut status = [0u8; api::syscall::IPC_STATUS_LEN];
319    status[..4].copy_from_slice(&api::syscall::IPC_STATUS_VERSION.to_le_bytes());
320    status[4..8].copy_from_slice(&kind.to_le_bytes());
321    status[8..12].copy_from_slice(&(data.len() as u32).to_le_bytes());
322    view.write_bytes(status_ptr, &status).map_err(|_| Failure::Invalid)?;
323    sched.tasks.get_mut(&owner).unwrap().async_operations.remove(token);
324    Ok(true)
325}

327pub fn wait(owner: usize, ticks: u64) -> Result<bool, Failure> {
328    let deadline = (ticks != 0).then(|| (super::system_ticks() as u64).saturating_add(ticks));
329    {
330        let mut guard = SCHEDULER.lock();
331        let sched = guard.as_mut().ok_or(Failure::Invalid)?;
332        let task = sched.tasks.get_mut(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
333        if task.async_operations.has_terminal() { return Ok(true); }
334        super::arm_ipc_block_handoff(owner);
335        task.state = TaskState::WaitIpc { deadline };
336    }
337    super::yield_cpu();
338    let guard = SCHEDULER.lock();
339    Ok(guard.as_ref().and_then(|s| s.tasks.get(&owner))
340        .is_some_and(|task| task.async_operations.has_terminal()))
341}

343/// Peer death is an infallible terminal transition; storage was reserved on submit.
344pub fn peer_died(sched: &mut Scheduler, peer: usize, cell: u64, generation: u64) {
345    let mut after = 0usize;
346    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
347        after = match owner.checked_add(1) { Some(next) => next, None => break };
348        if sched.tasks.get_mut(&owner).is_some_and(|task|
349            task.async_operations.peer_died(peer, cell, generation)) {
350            wake(sched, owner);
351        }
352    }
353}

355/// Drop queued requests from an owner whose task generation has exited.
356/// Dispatched requests retain their service-side grant lease until the
357/// service's existing reply/next-public-receive cleanup boundary.
358pub fn owner_died(sched: &mut Scheduler, owner: usize, cell: u64, generation: u64) {
359    for task in sched.tasks.values_mut() {
360        let mut index = 0;
361        while index < task.pending_msgs.len() {
362            let queued = task.pending_msgs.as_slice()[index].wire_header()
363                .is_some_and(|header| header.async_op != 0
364                    && header.sender_tid == owner
365                    && header.sender_cell_id == cell
366                    && header.sender_generation == generation);
367            if queued { task.pending_msgs.remove(index); } else { index += 1; }
368        }
369    }
370}

372/// Complete queued requests tied to a provider being replaced. Called only
373/// after the successful hot-swap commit, under SCHEDULER. The replacement
374/// must not receive these wires because it is a different generation.
375pub(crate) fn cutover_peer(sched: &mut Scheduler, peer: usize, cell: u64, generation: u64) {
376    let mut after = 0usize;
377    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
378        after = match owner.checked_add(1) { Some(next) => next, None => break };
379        let changed = if let Some(task) = sched.tasks.get_mut(&owner) {
380            let mut changed = false;
381            for op in &mut task.async_operations.0 {
382                if op.peer_tid == peer && op.peer_cell == cell && op.peer_generation == generation
383                    && matches!(op.phase, Phase::Queued) {
384                    op.phase = Phase::Terminal {
385                        kind: api::syscall::ipc_status::PEER_GONE, len: 0,
386                    };
387                    changed = true;
388                }
389            }
390            changed
391        } else { false };
392        if changed { wake(sched, owner); }
393    }
394}

396/// Expire accepted work without allocating in the scheduler's timer sweep.
397/// An undispatched request is removed from its peer mailbox before reporting
398/// PreDispatchTimeout. A dispatched request is conservatively indeterminate.
399pub fn expire(sched: &mut Scheduler, now: u64) {
400    let mut after = 0usize;
401    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
402        after = match owner.checked_add(1) { Some(next) => next, None => break };
403        for index in 0..MAX_OPERATIONS {
404            let expired = {
405                let Some(task) = sched.tasks.get_mut(&owner) else { break };
406                let Some(op) = task.async_operations.0.get_mut(index) else { break };
407                if now < op.deadline || matches!(op.phase, Phase::Terminal { .. }) {
408                    None
409                } else {
410                    let queued = matches!(op.phase, Phase::Queued);
411                    op.phase = Phase::Terminal {
412                        kind: if queued { api::syscall::ipc_status::PRE_DISPATCH_TIMEOUT }
413                            else { api::syscall::ipc_status::INDETERMINATE },
414                        len: 0,
415                    };
416                    Some((op.id, op.peer_tid, queued))
417                }
418            };
419            if let Some((token, peer, queued)) = expired {
420                if queued {
421                    if let Some(target) = sched.tasks.get_mut(&peer) {
422                        if let Some(pos) = target.pending_msgs.iter().position(|msg|
423                            msg.wire_header().is_some_and(|header| header.async_op == token)) {
424                            target.pending_msgs.remove(pos);
425                        }
426                    }
427                }
428                wake(sched, owner);
429            }
430        }
431    }

790    #[test]
791    fn deterministic_two_hart_publication_and_wake_races() {
792        let _guard = crate::TEST_STATE_LOCK.lock();
793        let mut sched = fixture();
794
795        // 1. Race 1: Pre-park publication race
796        // Provider publishes reply before caller on another hart arms WaitIpc.
797        add_operation(&mut sched, OWNER, PROVIDER, 101, Phase::Dispatched);
798        serve(&mut sched, PROVIDER, OWNER, 101);
799        sched.tasks.get_mut(&OWNER).unwrap().state = TaskState::Ready;
800        assert_eq!(reply_current_in_sched(&mut sched, PROVIDER, OWNER, 0, 0), Some(Ok(())));
801        // When caller subsequently executes wait, has_terminal is true immediately,
802        // so caller does not block or lose the wakeup.
803        assert!(sched.tasks[&OWNER].async_operations.has_terminal());
804        assert!(matches!(sched.tasks[&OWNER].async_operations.find(101).unwrap().phase,
805            Phase::Terminal { kind: api::syscall::ipc_status::REPLY, .. }));
806
807        // 2. Race 2: Cross-hart parked wake race
808        // Caller is parked in WaitIpc when Provider settles the call.
809        add_operation(&mut sched, OWNER, PROVIDER, 102, Phase::Dispatched);
810        serve(&mut sched, PROVIDER, OWNER, 102);
811        sched.tasks.get_mut(&OWNER).unwrap().state = TaskState::WaitIpc { deadline: None };
812        assert_eq!(reply_current_in_sched(&mut sched, PROVIDER, OWNER, 0, 0), Some(Ok(())));
813        // Caller must be transitioned to Ready and pushed to the ready queue.
814        assert!(matches!(sched.tasks[&OWNER].state, TaskState::Ready));
815        assert!(sched.tasks[&OWNER].async_operations.has_terminal());
816        super::super::hart_local::ready::remove_from_all(OWNER);
817
818        // 3. Race 3: Remote peer-death vs parked caller race
819        // Caller is parked in WaitIpc when Provider crashes on another hart.
820        add_operation(&mut sched, OWNER, PROVIDER, 103, Phase::Dispatched);
821        sched.tasks.get_mut(&OWNER).unwrap().state = TaskState::WaitIpc { deadline: None };
822        peer_died(&mut sched, PROVIDER, 1_003, 7);
823        // Caller must be transitioned to Ready with PeerGone terminal.
824        assert!(matches!(sched.tasks[&OWNER].state, TaskState::Ready));
825        assert!(matches!(sched.tasks[&OWNER].async_operations.find(103).unwrap().phase,
826            Phase::Terminal { kind: api::syscall::ipc_status::PEER_GONE, .. }));
827        super::super::hart_local::ready::remove_from_all(OWNER);
828
829        // 4. Race 4: Remote deadline expiration vs reply race
830        // Case A: Reply wins lock first, deadline cannot overwrite it.
831        add_operation(&mut sched, OWNER, WORKER, 104, Phase::Dispatched);
832        serve(&mut sched, WORKER, OWNER, 104);
833        assert_eq!(reply_current_in_sched(&mut sched, WORKER, OWNER, 0, 0), Some(Ok(())));
834        expire(&mut sched, u64::MAX);
835        assert!(matches!(sched.tasks[&OWNER].async_operations.find(104).unwrap().phase,
836            Phase::Terminal { kind: api::syscall::ipc_status::REPLY, .. }));
837
838        // Case B: Expire wins lock first, subsequent reply is rejected.
839        add_operation(&mut sched, OWNER, WORKER, 105, Phase::Dispatched);
840        serve(&mut sched, WORKER, OWNER, 105);
841        expire(&mut sched, 100);
842        assert!(matches!(sched.tasks[&OWNER].async_operations.find(105).unwrap().phase,
843            Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, .. }));
844        assert_eq!(reply_current_in_sched(&mut sched, WORKER, OWNER, 0, 0), Some(Ok(())));
845        // Terminal remains Indeterminate, not overwritten by late reply.
846        assert!(matches!(sched.tasks[&OWNER].async_operations.find(105).unwrap().phase,
847            Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, .. }));
848    }
