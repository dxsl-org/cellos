22/// Phase 03e admission headroom: the last `PEER_QUEUE_HEADROOM` slots of a
23/// receiver's mailbox admit an operation only from a Cell incarnation that
24/// holds no queued operation wire there (all its threads count as one holder).
25/// One Cell can then hold at most `HOTSWAP_MSG_QUEUE_DEPTH - PEER_QUEUE_HEADROOM`
26/// queued wires and, while no raw or kernel-inserted records are queued at the
27/// receiver, a Cell holding none is refused for capacity only when at least
28/// `PEER_QUEUE_HEADROOM + 1` other Cells each hold one (Spec 20 §2.6).
29const PEER_QUEUE_HEADROOM: usize = 8;
30
31/// Whether `receiver`'s mailbox admits one more operation wire from the Cell
32/// incarnation `sender`. Identity is the kernel-written wire header; the scan
33/// is bounded by the mailbox depth and allocates nothing.
34fn peer_queue_admits(receiver: &Task, sender: (u64, u64), headroom: usize) -> bool {
35    let depth = super::tcb::HOTSWAP_MSG_QUEUE_DEPTH;
36    let len = receiver.pending_msgs.len();
37    if len >= depth {
38        return false;
39    }
40    len < depth - headroom
41        || !receiver.pending_msgs.iter().any(|message| {
42            message.wire_header().is_some_and(|header| {
43                header.async_op != 0 && (header.sender_cell_id, header.sender_generation) == sender
44            })
45        })
46}

107/// Task-local, kernel-charged storage. Tokens are global monotonic identifiers,
108/// never slot indexes; reuse after take cannot alias a late service reply.

110pub struct Operations(Vec<Operation>);

112impl Operations {
113    pub fn new() -> Self {
114        Self(Vec::new())
115    }
116    pub(crate) fn len(&self) -> usize {
117        self.0.len()
118    }
119    #[cfg(feature = "p01-trace")]
120    fn observe_owner(&self, owner: crate::p01_measurement::Owner) {
121        crate::memory::heap::p01_owner::runtime_vec(
122            &self.0,
123            owner,
124            crate::memory::heap::p01_owner::OPERATIONS,
125        );
126    }
127    pub fn find(&self, id: usize) -> Option<&Operation> {
128        self.0.iter().find(|op| op.id == id)
129    }
130    fn find_mut(&mut self, id: usize) -> Option<&mut Operation> {
131        self.0.iter_mut().find(|op| op.id == id)
132    }
133    fn remove(&mut self, id: usize) -> Option<Operation> {
134        let pos = self.0.iter().position(|op| op.id == id)?;
135        Some(self.0.swap_remove(pos))
136    }
137    fn push(&mut self, op: Operation) -> Result<(), Failure> {
138        if self.0.len() == MAX_OPERATIONS {
139            return Err(Failure::Busy);
140        }
141        if self.0.len() == self.0.capacity() {
142            crate::memory::heap::alloc_for(crate::memory::cell_quota::Charge::KERNEL, || {
143                self.0.try_reserve(1)
144            })
145            .map_err(|_| Failure::Busy)?;
146        }
147        self.0.push(op);
148        Ok(())
149    }
150    pub fn peer_died(&mut self, tid: usize, cell: u64, generation: u64) -> bool {
151        let mut changed = false;
152        for op in &mut self.0 {
153            if op.peer_tid == tid
154                && op.peer_cell == cell
155                && op.peer_generation == generation
156                && !matches!(op.phase, Phase::Terminal { .. })
157            {
158                op.phase = Phase::Terminal {
159                    kind: abi::syscall::ipc_status::PEER_GONE,
160                    len: 0,
161                };
162                #[cfg(feature = "p01-trace")]
163                crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
164                changed = true;
165            }
166        }
167        changed
168    }
169    fn has_terminal(&self) -> bool {
170        self.0
171            .iter()
172            .any(|op| matches!(op.phase, Phase::Terminal { .. }))
173    }
174    fn notifiable(op: &Operation, waiter: Waiter) -> bool {
175        !op.notified
176            && matches!(op.phase, Phase::Terminal { .. })
177            && (op.owner_cell, op.owner_generation) == (waiter.cell, waiter.generation)
178            && (op.consumer_tid == waiter.tid || op.owner_tid == waiter.tid)
179    }
180}

201/// One consumed IPC completion notification.
202#[derive(Clone, Copy, Debug, PartialEq, Eq)]
203pub(crate) struct IpcNotification {
204    /// Exact operation token; the terminal kind/bytes stay behind `IpcTake`.
205    pub token: usize,
206    /// Operation records examined to choose it (bounded by `MAX_OPERATIONS`).
207    pub scanned: usize,
208}

210/// Whether `task` owns a terminal operation that still owes its one-shot
211/// notification. Caller holds `SCHEDULER`, the lock every terminal
212/// transition also holds, so this check and a park decision are atomic.
213pub(crate) fn has_pending_notification(task: &Task) -> bool {
214    let waiter = Waiter::of(task);
215    task.async_operations
216        .0
217        .iter()
218        .any(|op| Operations::notifiable(op, waiter))
219}
220
221/// Consume the notification of the oldest submission (lowest token) among
222/// `task`'s un-notified terminal operations. The operation itself stays
223/// retained and charged until `take`. No allocation; one pass over at most
224/// `MAX_OPERATIONS` records. Caller holds `SCHEDULER`.
225pub(crate) fn take_notification(task: &mut Task) -> Option<IpcNotification> {
226    let waiter = Waiter::of(task);
227    let ops = &mut task.async_operations.0;
228    let scanned = ops.len();
229    let op = ops
230        .iter_mut()
231        .filter(|op| Operations::notifiable(op, waiter))
232        .min_by_key(|op| op.id)?;
233    op.notified = true;
234    Some(IpcNotification {
235        token: op.id,
236        scanned,
237    })
238}

240/// Re-arm a consumed notification whose record could not be copied out, so the
241/// next wait reports it again. Only the waiter's own retained terminal can be
242/// re-armed; a token taken meanwhile, or another incarnation's, is left alone.
243/// Caller holds `SCHEDULER`.
244pub(crate) fn restore_notification(task: &mut Task, token: usize) -> bool {
245    let waiter = Waiter::of(task);
246    let Some(op) = task.async_operations.find_mut(token) else {
247        return false;
248    };
249    let restorable = op.notified
250        && matches!(op.phase, Phase::Terminal { .. })
251        && (op.owner_cell, op.owner_generation) == (waiter.cell, waiter.generation)
252        && (op.consumer_tid == waiter.tid || op.owner_tid == waiter.tid);
253    if restorable {
254        op.notified = false;
255    }
256    restorable
257}

263/// Owner of every live operation token, and every task whose request context
264/// was set to a token (architecture remediation Phase 10).
265///
266/// These two tables answer what the reply path used to find by scanning every
267/// task. Owned by the [`Scheduler`] and only touched under `SCHEDULER`. A task
268/// holds at most one token at a time, so `holders` keeps at most one row per
269/// task; a row whose task has since moved on is stale and is checked against
270/// `current_async_op` before use. Capacity for both is reserved when an
271/// operation is accepted, so delivery and removal do not allocate.
272pub struct AsyncIndex {
273    /// `(token, owner tid)`, sorted by token.
274    owners: Vec<(usize, usize)>,
275    /// `(token, tid)`: the last token delivered as `tid`'s request context.
276    holders: Vec<(usize, usize)>,
277}
278
279impl AsyncIndex {
280    pub const fn new() -> Self {
281        Self {
282            owners: Vec::new(),
283            holders: Vec::new(),
284        }
285    }
286
287    /// Reserve room for one more operation, charged to the kernel like the
288    /// operation record itself.
289    fn reserve(&mut self) -> Result<(), Failure> {
290        crate::memory::heap::alloc_for(crate::memory::cell_quota::Charge::KERNEL, || {
291            self.owners
292                .try_reserve(1)
293                .and_then(|()| self.holders.try_reserve(self.owners.len() + 1))
294        })
295        .map_err(|_| Failure::Busy)
296    }
297
298    fn insert(&mut self, token: usize, owner: usize) {
299        let at = self.owners.partition_point(|&(t, _)| t < token);
300        self.owners.insert(at, (token, owner));
301    }
302
303    fn owner(&self, token: usize) -> Option<usize> {
304        let at = self.owners.binary_search_by_key(&token, |&(t, _)| t).ok()?;
305        Some(self.owners[at].1)
306    }
307
308    fn remove(&mut self, token: usize) {
309        if let Ok(at) = self.owners.binary_search_by_key(&token, |&(t, _)| t) {
310            self.owners.remove(at);
311        }
312    }
313
314    /// Record that `tid`'s request context now names `token` (0: none);
315    /// refused when the table cannot grow, with `tid` then holding no row.
316    ///
317    /// Submit reserved a row per accepted operation, so this normally does not
318    /// allocate; a token delivered twice (a wire that outlived its operation)
319    /// can still need a row, which is then charged to the kernel, never to the
320    /// receiving Cell.
321    fn try_note_holder(&mut self, tid: usize, token: usize) -> Result<(), Failure> {
322        self.forget_holder(tid);
323        if token == 0 {
324            return Ok(());
325        }
326        if self.holders.len() == self.holders.capacity() {
327            crate::memory::heap::alloc_for(crate::memory::cell_quota::Charge::KERNEL, || {
328                self.holders.try_reserve(1)
329            })
330            .map_err(|_| Failure::Busy)?;
331        }
332        self.holders.push((token, tid));
333        Ok(())
334    }
335
336    /// `tid` no longer holds a request context.
337    fn forget_holder(&mut self, tid: usize) {
338        if let Some(at) = self.holders.iter().position(|&(_, t)| t == tid) {
339            self.holders.swap_remove(at);
340        }
341    }
342}

344/// Record that `tid`'s request context now names `token`. Call after
345/// `Task::set_received_request_context` returned that token. If the index
346/// cannot grow, `tid` keeps the caller context but no async token: its reply
347/// settles no operation, which then ends by its deadline, cancel or owner
348/// death like any unanswered one.
349pub(crate) fn note_holder(sched: &mut Scheduler, tid: usize, token: usize) {
350    if sched.async_index.try_note_holder(tid, token).is_err() {
351        if let Some(task) = sched.tasks.get_mut(&tid) {
352            task.current_async_op = 0;
353        }
354    }
355}

363/// Accept `op` into its owner's list and the index; fails without side
364/// effect when either is full.
365fn push_operation(sched: &mut Scheduler, owner: usize, op: Operation) -> Result<(), Failure> {
366    sched.async_index.reserve()?;
367    let token = op.id;
368    sched
369        .tasks
370        .get_mut(&owner)
371        .ok_or(Failure::Invalid)?
372        .async_operations
373        .push(op)?;
374    sched.async_index.insert(token, owner);
375    Ok(())
376}

379fn remove_operation(sched: &mut Scheduler, owner: usize, token: usize) -> Option<Operation> {
380    let op = sched

383        .async_operations

389/// Index upkeep when `task` leaves the task table: its operations die with it,
390/// and it no longer holds a request context.
391pub(crate) fn forget_task(sched: &mut Scheduler, task: &Task) {
392    for op in &task.async_operations.0 {
393        sched.async_index.remove(op.id);
394    }
395    sched.async_index.forget_holder(task.id);

419/// Make the owner runnable after one of its operations turned terminal: any
420/// completion-source wait that names IPC (`WaitCompletion(IPC)` or `IpcWait`).
421fn wake(sched: &mut Scheduler, owner: usize) {
422    if let Some(task) = sched.tasks.get_mut(&owner) {
423        let waiting = matches!(
424            task.state,
425            TaskState::Blocked {
426                wait: WaitKind::Completion { source, .. },
427                ..
428            } if source & abi::completion::source::IPC != 0
429        );
430        if waiting {
431            task.state = TaskState::Ready;
432            let priority = sched.push_ready(owner);
433            sched.pend_preempt_if_needed(priority);
434        }
435    }
436}

438fn new_token() -> Result<usize, Failure> {
439    NEXT_OPERATION
440        .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |next| {
441            (next <= isize::MAX as usize).then_some(next + 1)
442        })
443        .map_err(|_| Failure::Busy)
444}

468/// Submit one exact operation from `owner` to `peer`. The operation belongs to
469/// `owner` exactly as an `IpcSubmit` would; the caller must take or cancel it.
470pub fn submit_request(owner: usize, peer: usize, request: Request<'_>) -> Result<usize, Failure> {
471    let len = request.len();
472    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD {
473        return Err(Failure::Invalid);
474    }
475    #[cfg(feature = "p01-trace")]
476    let heap_owner;
477    let (view, binding, peer_binding) = {
478        let guard = SCHEDULER.lock();
479        let sched = guard.as_ref().ok_or(Failure::PeerGone)?;
480        let task = sched
481            .tasks
482            .get(&owner)
483            .filter(|task| live(task))
484            .ok_or(Failure::Invalid)?;
485        let target = sched
486            .tasks
487            .get(&peer)
488            .filter(|task| live(task))
489            .ok_or(Failure::PeerGone)?;
490        // Refuse a full or headroom-reserved queue before the request copy and
491        // the 4 KiB reply buffer are allocated. The authoritative check repeats
492        // in the commit hold below, where the push happens.
493        if !peer_queue_admits(
494            target,
495            (task.cell_id.0, task.cell_generation),
496            PEER_QUEUE_HEADROOM,
497        ) {
498            return Err(Failure::Busy);
499        }
500        #[cfg(feature = "p01-trace")]
501        {
502            heap_owner = crate::p01_resources::owner(task);
503        }
504        (
505            TaskCopyView::of(task),
506            (task.cell_id.0, task.cell_generation),
507            (target.cell_id.0, target.cell_generation),
508        )
509    };
510    let token = new_token()?;
511    let header = ipc_wire::IpcWireHeader {
512        sender_tid: owner,
513        sender_cell_id: binding.0,
514        sender_generation: binding.1,
515        delivery_id: super::next_delivery_id(),
516        async_op: token,
517    };
518    // Copy outside SCHEDULER; no peer or user pointer persists in a task slot.
519    let wire = match request {
520        Request::User { ptr, len } => ipc_wire::IpcWireMessage::try_from_user(
521            header,
522            &view,
523            ptr,
524            len,
525            #[cfg(feature = "p01-trace")]
526            heap_owner,
527        ),
528        Request::Kernel(bytes) => {
529            let wire = ipc_wire::IpcWireMessage::try_new(header, bytes);
530            #[cfg(feature = "p01-trace")]
531            if let Ok(wire) = &wire {
532                wire.observe_owner(heap_owner, crate::memory::heap::p01_owner::SENDER_WIRE);
533            }
534            wire
535        }
536    }
537    .map_err(|_| Failure::Invalid)?;
538    #[cfg(feature = "bench-counters")]
539    crate::bench_counters::ipc_bytes(len);
540    let reply_header = ipc_wire::IpcWireHeader {
541        sender_tid: peer,
542        sender_cell_id: peer_binding.0,
543        sender_generation: peer_binding.1,
544        delivery_id: 0,
545        async_op: token,
546    };
547    // The reply storage is the submitter's: it is charged to `binding`, the
548    // incarnation that asked, never to the server the reply header names, so
549    // no client can spend a peer's quota by submitting.
550    let reply = ipc_wire::IpcWireMessage::try_new_charged(
551        reply_header,
552        &[0u8; ipc_wire::MAX_IPC_WIRE_PAYLOAD],
553        crate::memory::cell_quota::Charge::new(types::CellId(binding.0), binding.1),
554    )
555    .map_err(|_| Failure::Busy)?;
556    #[cfg(feature = "p01-trace")]
557    reply.observe_owner(heap_owner, crate::memory::heap::p01_owner::REPLY);
558    let mut guard = SCHEDULER.lock();
559    let sched = guard.as_mut().ok_or(Failure::PeerGone)?;
560    let caller = sched
561        .tasks
562        .get(&owner)
563        .filter(|task| live(task) && (task.cell_id.0, task.cell_generation) == binding)
564        .ok_or(Failure::Invalid)?;
565    if caller.async_operations.len() >= MAX_OPERATIONS {
566        return Err(Failure::Busy);
567    }
568    let receiver = sched
569        .tasks
570        .get(&peer)
571        .filter(|task| live(task) && (task.cell_id.0, task.cell_generation) == peer_binding)
572        .ok_or(Failure::PeerGone)?;
573    if super::paused_target_rejects(sched, owner, peer) {
574        return Err(Failure::Busy);
575    }
576    if !peer_queue_admits(receiver, binding, PEER_QUEUE_HEADROOM) {
577        return Err(Failure::Busy);
578    }
579    #[cfg(feature = "p01-trace")]
580    let p01_obligation = ObservedOperation(crate::p01_resources::operation_obligation(
581        crate::p01_resources::owner(caller),
582    ));
583    #[cfg(feature = "p01-trace")]
584    let _operations = crate::memory::heap::p01_owner::creation(
585        Some(heap_owner),
586        crate::memory::heap::p01_owner::OPERATIONS,
587    );
588    let pushed = push_operation(
589        sched,
590        owner,
591        Operation {
592            id: token,
593            owner_cell: binding.0,
594            owner_generation: binding.1,
595            owner_tid: owner,
596            consumer_tid: owner,
597            peer_tid: peer,
598            peer_cell: peer_binding.0,
599            peer_generation: peer_binding.1,
600            deadline: (super::system_ticks() as u64).saturating_add(OPERATION_TIMEOUT_TICKS),
601            phase: Phase::Queued,
602            notified: false,
603            reply,
604            #[cfg(feature = "p01-trace")]
605            _p01_obligation: p01_obligation,
606        },
607    );
608    #[cfg(feature = "p01-trace")]
609    drop(_operations);
610    #[cfg(feature = "p01-trace")]
611    sched
612        .tasks
613        .get(&owner)
614        .unwrap()
615        .async_operations
616        .observe_owner(heap_owner);
617    pushed?;
618    let target = sched.tasks.get_mut(&peer).ok_or(Failure::PeerGone)?;
619    let recv_eligible = matches!(target.state, TaskState::Blocked { wait: WaitKind::Recv { mask, .. }, .. } if mask == 0 || mask == owner);
620    if super::queue_wire_msg(target, wire, super::tcb::HOTSWAP_MSG_QUEUE_DEPTH).is_err() {
621        remove_operation(sched, owner, token);
622        return Err(Failure::Busy);
623    }
624    let wake_cause = super::wake_after_ipc_publish(target, recv_eligible);
625    if wake_cause.made_runnable() {
626        let priority = sched.push_ready(peer);
627        sched.pend_preempt_if_needed(priority);
628    }
629    Ok(token)
630}

632/// Snapshotting the request is the dispatch boundary. If a copy-out later
633/// faults, cancellation stays conservatively indeterminate, not falsely safe.
634pub fn dispatch(sched: &mut Scheduler, peer: usize, header: ipc_wire::IpcWireHeader) {
635    if header.async_op == 0 {
636        return;
637    }
638    let Some(provider) = sched.tasks.get(&peer).filter(|task| live(task)) else {
639        return;
640    };
641    let peer_binding = (provider.cell_id.0, provider.cell_generation);
642    let Some(owner) = sched
643        .tasks
644        .get_mut(&header.sender_tid)
645        .filter(|task| live(task))
646    else {
647        return;
648    };
649    if owner.cell_id.0 != header.sender_cell_id || owner.cell_generation != header.sender_generation
650    {
651        return;
652    }
653    let owner_binding = (owner.cell_id.0, owner.cell_generation);
654    let Some(op) = owner.async_operations.find_mut(header.async_op) else {
655        return;
656    };
657    if (op.owner_cell, op.owner_generation) == owner_binding
658        && (op.peer_tid, op.peer_cell, op.peer_generation) == (peer, peer_binding.0, peer_binding.1)
659        && matches!(op.phase, Phase::Queued)
660    {
661        op.phase = Phase::Dispatched;
662    }
663}

665fn terminal(
666    sched: &mut Scheduler,
667    owner_tid: usize,
668    token: usize,
669    provider: usize,
670    exact_peer: bool,
671    ptr: usize,
672    len: usize,
673) -> Result<(), Failure> {
674    let Some(owner) = sched.tasks.get(&owner_tid).filter(|task| live(task)) else {
675        return Err(Failure::PeerGone);
676    };
677    let Some(op) = owner.async_operations.find(token) else {
678        return Err(Failure::Invalid);
679    };
680    let Some(peer) = sched.tasks.get(&provider).filter(|task| live(task)) else {
681        return Err(Failure::PeerGone);
682    };
683    // Explicit replies may be delegated to a worker of the accepted provider's
684    // cell incarnation. Only the implicit bridge requires the served TID too.
685    // Authority always comes from scheduler identity, never caller input.
686    if !op.owned_by(owner)
687        || (exact_peer && op.peer_tid != provider)
688        || op.peer_cell != peer.cell_id.0
689        || op.peer_generation != peer.cell_generation
690        || !matches!(op.phase, Phase::Dispatched)
691    {
692        return Err(Failure::Invalid);
693    }
694    let view = TaskCopyView::of(peer);
695    let op = sched
696        .tasks
697        .get_mut(&owner_tid)
698        .unwrap()
699        .async_operations
700        .find_mut(token)
701        .unwrap();
702    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD {
703        op.phase = Phase::Terminal {
704            kind: abi::syscall::ipc_status::INDETERMINATE,
705            len: 0,
706        };
707        #[cfg(feature = "p01-trace")]
708        crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
709        wake(sched, owner_tid);
710        return Err(Failure::Invalid);
711    }
712    let copied = view.read_into(ptr, &mut op.reply.as_mut_slice()[..len]);
713    #[cfg(feature = "bench-counters")]
714    if copied.is_ok() {
715        crate::bench_counters::ipc_bytes(len);
716    }
717    op.phase = Phase::Terminal {
718        kind: if copied.is_ok() {
719            abi::syscall::ipc_status::REPLY
720        } else {
721            abi::syscall::ipc_status::INDETERMINATE
722        },
723        len: if copied.is_ok() { len } else { 0 },
724    };
725    #[cfg(feature = "p01-trace")]
726    crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
727    wake(sched, owner_tid);
728    copied.map_err(|_| Failure::Invalid)
729}

737    peer: usize,
738    destination: usize,
739    ptr: usize,
740    len: usize,

743    reply_current_in_sched(guard.as_mut()?, peer, destination, ptr, len)

746fn reply_current_in_sched(
747    sched: &mut Scheduler,
748    peer: usize,
749    destination: usize,
750    ptr: usize,
751    len: usize,
752) -> Option<Result<(), Failure>> {
753    let provider = sched.tasks.get(&peer).filter(|task| live(task))?;
754    let token = provider.current_async_op;
755    if token == 0 || provider.caller.current != Some(destination) {
756        return None;
757    }
758    let owner_binding = (provider.caller.cell_id, provider.caller.cell_generation);
759    let owner_matches = sched
760        .tasks

764        return Some(Ok(()));
765    }

768            // Invalid tokens and terminal results are quarantined, never raw Send.
769            Err(Failure::Invalid) => Ok(()),
770            other => other,
771        },
772    )

790fn reply_in_sched(
791    sched: &mut Scheduler,
792    peer: usize,
793    token: usize,
794    ptr: usize,
795    len: usize,
796) -> Result<(), Failure> {
797    if token == 0 {
798        return Err(Failure::Invalid);
799    }
800    let owner = owner_of(sched, token);
801    let result = owner
802        .ok_or(Failure::Invalid)
803        .and_then(|owner| terminal(sched, owner, token, peer, false, ptr, len));
804    // An explicit response ends this request even if cancellation/take already
805    // retired its token. Otherwise later one-way events would be mistaken for
806    // stale implicit replies. VFS's exact lease cleanup is handled by syscall.
807    if let Some(binding) = sched
808        .tasks
809        .get(&peer)
810        .filter(|task| live(task))
811        .map(|task| (task.cell_id, task.cell_generation))
812    {
813        #[cfg(any(feature = "test-hooks", test))]
814        for (&id, task) in sched.tasks.iter() {
815            assert!(
816                task.current_async_op != token || sched.async_index.holders.contains(&(token, id)),
817                "task {id} holds async token {token} unrecorded by the index"
818            );
819        }
820        // Visit the recorded holders of `token`; drop rows that are cleared
821        // here or no longer hold it, keep the rest for a later reply.
822        let holders = &mut sched.async_index.holders;
823        let mut at = 0;
824        while at < holders.len() {
825            let (held, tid) = holders[at];
826            if held != token {
827                at += 1;
828                continue;
829            }
830            let keep = match sched.tasks.get_mut(&tid) {
831                Some(task) if task.current_async_op == token => {
832                    let clear = (task.cell_id, task.cell_generation) == binding
833                        && !crate::cell::service_registry::retains_outer_caller(
834                            task.cell_id.0,
835                            task.cell_generation,
836                        );
837                    if clear {
838                        task.clear_current_caller_context();
839                    }
840                    !clear
841                }
842                _ => false,
843            };
844            if keep {
845                at += 1;
846            } else {
847                holders.swap_remove(at);
848            }
849        }
850    }
851    result
852}

860fn cancel_in_sched(sched: &mut Scheduler, owner: usize, token: usize) -> Result<(), Failure> {
861    let task = sched
862        .tasks
863        .get(&owner)
864        .filter(|task| live(task))
865        .ok_or(Failure::Invalid)?;
866    let op = task.async_operations.find(token).ok_or(Failure::Invalid)?;
867    if !op.owned_by(task) {
868        return Err(Failure::Invalid);
869    }
870    let queued = matches!(op.phase, Phase::Queued);
871    let peer = op.peer_tid;
872    if queued {
873        // No receiver snapshot has begun; remove only the matching token.
874        if let Some(receiver) = sched.tasks.get_mut(&peer) {
875            if let Some(pos) = receiver.pending_msgs.iter().position(|msg| {
876                msg.wire_header()
877                    .is_some_and(|header| header.async_op == token)
878            }) {
879                receiver.pending_msgs.remove(pos);
880            }
881        }
882    }
883    let op = sched
884        .tasks
885        .get_mut(&owner)
886        .unwrap()
887        .async_operations
888        .find_mut(token)
889        .unwrap();
890    if !matches!(op.phase, Phase::Terminal { .. }) {
891        op.phase = Phase::Terminal {
892            kind: if queued {
893                abi::syscall::ipc_status::CANCELLED
894            } else {
895                abi::syscall::ipc_status::INDETERMINATE
896            },
897            len: 0,
898        };
899        #[cfg(feature = "p01-trace")]
900        crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
901        wake(sched, owner);
902    }
903    Ok(())
904}

906pub fn take(
907    consumer: usize,
908    token: usize,
909    reply_ptr: usize,
910    reply_len: usize,
911    status_ptr: usize,
912) -> Result<bool, Failure> {
913    let mut guard = SCHEDULER.lock();
914    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
915    let caller = sched
916        .tasks
917        .get(&consumer)
918        .filter(|task| live(task))
919        .ok_or(Failure::Invalid)?;
920    let caller_cell = (caller.cell_id.0, caller.cell_generation);
921
922    let owner_tid = owner_of(sched, token)
923        .filter(|owner| {
924            let task = &sched.tasks[owner];
925            (task.cell_id.0, task.cell_generation) == caller_cell
926        })
927        .ok_or(Failure::Invalid)?;
928    let op_ref = sched.tasks[&owner_tid]
929        .async_operations
930        .find(token)
931        .unwrap();
932    if !op_ref.can_consume(caller) {
933        return Err(Failure::Invalid);
934    }
935    let Phase::Terminal { kind, len } = &op_ref.phase else {
936        return Ok(false);
937    };
938    let data = &op_ref.reply.as_slice()[..*len];
939    if data.len() > reply_len {
940        return Err(Failure::TooSmall);
941    }
942    let view = TaskCopyView::of(caller);
943    view.validate_writable(status_ptr, abi::syscall::IPC_STATUS_LEN)
944        .map_err(|_| Failure::Invalid)?;
945    if !data.is_empty() {
946        view.validate_writable(reply_ptr, data.len())
947            .map_err(|_| Failure::Invalid)?;
948        view.write_bytes(reply_ptr, data)
949            .map_err(|_| Failure::Invalid)?;
950        #[cfg(feature = "bench-counters")]
951        crate::bench_counters::ipc_bytes(data.len());
952    }
953    let mut status = [0u8; abi::syscall::IPC_STATUS_LEN];
954    status[..4].copy_from_slice(&abi::syscall::IPC_STATUS_VERSION.to_le_bytes());
955    status[4..8].copy_from_slice(&kind.to_le_bytes());
956    status[8..12].copy_from_slice(&(data.len() as u32).to_le_bytes());
957    view.write_bytes(status_ptr, &status)
958        .map_err(|_| Failure::Invalid)?;
959    remove_operation(sched, owner_tid, token);
960    Ok(true)
961}

963/// Take `owner`'s operation `token` into a kernel buffer if it is terminal;
964/// otherwise park `owner` until one of its operations changes, and return
965/// `None` so the caller yields and asks again.
966///
967/// A terminal returns its kind ([`abi::syscall::ipc_status`]) and reply length,
968/// and the operation is gone. Parking checks this operation only: another
969/// operation of `owner` that is already terminal must not turn the wait into a
970/// spin. For a syscall the kernel serves with [`submit_request`].
971pub fn take_or_park_kernel(
972    owner: usize,
973    token: usize,
974    out: &mut [u8; ipc_wire::MAX_IPC_WIRE_PAYLOAD],
975) -> Result<Option<(u32, usize)>, Failure> {
976    let mut guard = SCHEDULER.lock();
977    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
978    let task = sched
979        .tasks
980        .get_mut(&owner)
981        .filter(|task| live(task))
982        .ok_or(Failure::Invalid)?;
983    let op = task.async_operations.find(token).ok_or(Failure::Invalid)?;
984    let Phase::Terminal { kind, len } = op.phase else {
985        super::arm_ipc_block_handoff(owner);
986        task.state = TaskState::Blocked {
987            wait: WaitKind::Completion {
988                source: abi::completion::source::IPC,
989                delivery: Delivery::IpcTerminal,
990            },
991            deadline: None,
992        };
993        return Ok(None);
994    };
995    out[..len].copy_from_slice(&op.reply.as_slice()[..len]);
996    remove_operation(sched, owner, token);
997    Ok(Some((kind, len)))
998}

1041/// Drop or indeterminate all operations owned by an exiting thread.
1042/// If take won before exit, the operation was already removed, ensuring
1043/// single-winner serialization against take.
1044pub fn on_owner_exits(sched: &mut Scheduler, owner_tid: usize) {
1045    let to_cancel: alloc::vec::Vec<(usize, usize, bool)> = match sched.tasks.get(&owner_tid) {
1046        Some(task) => task
1047            .async_operations
1048            .0
1049            .iter()
1050            .filter_map(|op| {
1051                if op.owner_tid == owner_tid {
1052                    match op.phase {
1053                        Phase::Queued => Some((op.id, op.peer_tid, true)),
1054                        Phase::Dispatched => Some((op.id, op.peer_tid, false)),
1055                        Phase::Terminal { .. } => None,
1056                    }
1057                } else {
1058                    None
1059                }
1060            })
1061            .collect(),
1062        None => return,
1063    };
1064
1065    for &(token, peer, queued) in &to_cancel {
1066        if queued {
1067            if let Some(receiver) = sched.tasks.get_mut(&peer) {
1068                if let Some(pos) = receiver.pending_msgs.iter().position(|msg| {
1069                    msg.wire_header()
1070                        .is_some_and(|header| header.async_op == token)
1071                }) {
1072                    receiver.pending_msgs.remove(pos);
1073                }
1074            }
1075        }
1076    }
1077
1078    if let Some(task) = sched.tasks.get_mut(&owner_tid) {
1079        for (token, _, queued) in to_cancel {
1080            if let Some(op) = task.async_operations.find_mut(token) {
1081                op.phase = Phase::Terminal {
1082                    kind: if queued {
1083                        abi::syscall::ipc_status::CANCELLED
1084                    } else {
1085                        abi::syscall::ipc_status::INDETERMINATE
1086                    },
1087                    len: 0,
1088                };
1089            }
1090        }
1091    }
1092}

1094pub fn wait(owner: usize, ticks: u64) -> Result<bool, Failure> {
1095    let deadline = (ticks != 0).then(|| (super::system_ticks() as u64).saturating_add(ticks));
1096    {
1097        let mut guard = SCHEDULER.lock();
1098        let sched = guard.as_mut().ok_or(Failure::Invalid)?;
1099        let task = sched
1100            .tasks
1101            .get_mut(&owner)
1102            .filter(|task| live(task))
1103            .ok_or(Failure::Invalid)?;
1104        if task.async_operations.has_terminal() {
1105            return Ok(true);
1106        }
1107        super::arm_ipc_block_handoff(owner);
1108        task.state = TaskState::Blocked {
1109            wait: WaitKind::Completion {
1110                source: abi::completion::source::IPC,
1111                delivery: Delivery::IpcTerminal,
1112            },
1113            deadline,
1114        };
1115    }
1116    super::yield_cpu();
1117    let guard = SCHEDULER.lock();
1118    Ok(guard
1119        .as_ref()
1120        .and_then(|s| s.tasks.get(&owner))
1121        .is_some_and(|task| task.async_operations.has_terminal()))
1122}

1143/// Dispatched requests retain their service-side grant lease until the
1144/// service's existing reply/next-public-receive cleanup boundary.
1145pub fn owner_died(sched: &mut Scheduler, owner: usize, cell: u64, generation: u64) {

1155                        && header.sender_generation == generation
1156                });
1157            if queued {
1158                task.pending_msgs.remove(index);
1159            } else {
1160                index += 1;
1161            }
1162        }
1163    }

1203/// Expire accepted work without allocating in the scheduler's timer sweep.
1204/// An undispatched request is removed from its peer mailbox before reporting
1205/// PreDispatchTimeout. A dispatched request is conservatively indeterminate.
1206pub fn expire(sched: &mut Scheduler, now: u64) {
1207    let mut after = 0usize;
1208    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
1209        after = match owner.checked_add(1) {
1210            Some(next) => next,
1211            None => break,
1212        };
1213        for index in 0..MAX_OPERATIONS {
1214            let expired = {
1215                let Some(task) = sched.tasks.get_mut(&owner) else {
1216                    break;
1217                };
1218                let Some(op) = task.async_operations.0.get_mut(index) else {
1219                    break;
1220                };
1221                if now < op.deadline || matches!(op.phase, Phase::Terminal { .. }) {
1222                    None
1223                } else {
1224                    let queued = matches!(op.phase, Phase::Queued);
1225                    op.phase = Phase::Terminal {
1226                        kind: if queued {
1227                            abi::syscall::ipc_status::PRE_DISPATCH_TIMEOUT
1228                        } else {
1229                            abi::syscall::ipc_status::INDETERMINATE
1230                        },
1231                        len: 0,
1232                    };
1233                    #[cfg(feature = "p01-trace")]
1234                    crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
1235                    Some((op.id, op.peer_tid, queued))
1236                }
1237            };
1238            if let Some((token, peer, queued)) = expired {
1239                if queued {
1240                    if let Some(target) = sched.tasks.get_mut(&peer) {
1241                        if let Some(pos) = target.pending_msgs.iter().position(|msg| {
1242                            msg.wire_header()
1243                                .is_some_and(|header| header.async_op == token)
1244                        }) {
1245                            target.pending_msgs.remove(pos);
1246                        }
1247                    }
1248                }
1249                wake(sched, owner);
1250            }
1251        }
1252    }
1253}

1256/// Boot self-test seam for the multi-source wait: record one accepted,
1257/// already-dispatched operation from `owner` to `peer` without a user copy.
1258///
1259/// Synthetic self-test tasks have no address space, so they cannot go through
1260/// [`submit`]. Identity is still read from the scheduler records exactly as
1261/// `submit` reads it, and the token comes from the same global counter, so the
1262/// operation is indistinguishable from a real one to every terminal path.
1263/// Caller holds `SCHEDULER`.
1264pub(super) fn selftest_accept_dispatched(
1265    sched: &mut Scheduler,
1266    owner: usize,
1267    peer: usize,
1268) -> Option<usize> {
1269    let (peer_cell, peer_generation) = sched
1270        .tasks
1271        .get(&peer)
1272        .map(|task| (task.cell_id.0, task.cell_generation))?;
1273    let token = new_token().ok()?;
1274    let reply = ipc_wire::IpcWireMessage::try_new(
1275        ipc_wire::IpcWireHeader {
1276            sender_tid: peer,
1277            sender_cell_id: peer_cell,
1278            sender_generation: peer_generation,
1279            delivery_id: 0,
1280            async_op: token,
1281        },
1282        &[],
1283    )
1284    .ok()?;
1285    let task = sched.tasks.get(&owner)?;
1286    let (owner_cell, owner_generation) = (task.cell_id.0, task.cell_generation);
1287    push_operation(
1288        sched,
1289        owner,
1290        Operation {
1291            id: token,
1292            owner_cell,
1293            owner_generation,
1294            owner_tid: owner,
1295            consumer_tid: owner,
1296            peer_tid: peer,
1297            peer_cell,
1298            peer_generation,
1299            deadline: u64::MAX,
1300            phase: Phase::Dispatched,
1301            notified: false,
1302            reply,
1303            #[cfg(feature = "p01-trace")]
1304            _p01_obligation: ObservedOperation(None),
1305        },
1306    )
1307    .ok()?;
1308    Some(token)
1309}

2921    /// Inside [`admitting_by_quota`](alloc_probe::admitting_by_quota) it also
2922    /// refuses (returns null for) an allocation the quota of this thread's
2923    /// open charge scope would refuse, as the kernel allocator does.
2924    mod alloc_probe {
2925        use std::alloc::{GlobalAlloc, Layout, System};
2926        use std::cell::Cell;
2927
2928        std::thread_local! {
2929            static COUNT: Cell<usize> = const { Cell::new(0) };
2930            static FREED: Cell<usize> = const { Cell::new(0) };
2931            static ADMIT_BY_QUOTA: Cell<bool> = const { Cell::new(false) };
2932        }
2933
2934        pub struct Counting;
2935
2936        fn bump() {
2937            let _ = COUNT.try_with(|count| count.set(count.get() + 1));
2938        }
2939
2940        /// Whether the quota of this thread's open charge scope admits `size`
2941        /// bytes. The charge is the scope's per-thread record
2942        /// (`heap::thread_charge`), not the hart slot every test thread
2943        /// shares. Only a check: an admitted charge is refunded at once, so no
2944        /// free has to find it.
2945        fn admitted(size: usize) -> bool {
2946            if !ADMIT_BY_QUOTA.try_with(Cell::get).unwrap_or(false) {
2947                return true;
2948            }
2949            let charge = crate::memory::heap::thread_charge();
2950            match crate::memory::cell_quota::charge_owner(charge, size) {
2951                Some(owner) => {
2952                    crate::memory::cell_quota::refund_owner(owner, size);
2953                    true
2954                }
2955                None => false,
2956            }
2957        }
2958
2959        // SAFETY: every admitted call forwards unchanged to `System`; a refused
2960        // one returns null, which `GlobalAlloc` permits. The thread-locals are
2961        // const-initialised `Cell`s that never allocate.
2962        unsafe impl GlobalAlloc for Counting {
2963            unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
2964                bump();
2965                if !admitted(layout.size()) {
2966                    return core::ptr::null_mut();
2967                }
2968                unsafe { System.alloc(layout) }
2969            }
2970            unsafe fn alloc_zeroed(&self, layout: Layout) -> *mut u8 {
2971                bump();
2972                if !admitted(layout.size()) {
2973                    return core::ptr::null_mut();
2974                }
2975                unsafe { System.alloc_zeroed(layout) }
2976            }
2977            unsafe fn realloc(&self, ptr: *mut u8, layout: Layout, new_size: usize) -> *mut u8 {
2978                bump();
2979                if !admitted(new_size) {
2980                    return core::ptr::null_mut();
2981                }
2982                unsafe { System.realloc(ptr, layout, new_size) }
2983            }
2984            unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
2985                let _ = FREED.try_with(|freed| freed.set(freed.get() + layout.size()));
2986                unsafe { System.dealloc(ptr, layout) }
2987            }
2988        }

3681        use types::CellId;
3682        const CELL: CellId = CellId((cell_quota::MAX_CELLS - 1) as u64);
3683        const GENERATION: u64 = 4;

3685        let queue_for = |limit: usize| {

3699                super::super::completion::queue_for(&mut sched, OWNER)
