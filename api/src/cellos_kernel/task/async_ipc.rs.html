1//! Kernel-owned, bounded request/reply operations. All transitions are serialized
2//! by SCHEDULER; neither a reply nor a wake depends on caller-owned memory.
3
4use super::{copy_glue::TaskCopyView, ipc_wire, scheduler::Scheduler, tcb::{Task, TaskState}, SCHEDULER};
5use alloc::vec::Vec;
6use core::sync::atomic::{AtomicUsize, Ordering};

8pub const MAX_OPERATIONS: usize = 64;
9/// Accepted operations have a kernel-enforced terminal deadline (30 seconds at
10/// the scheduler's 10 ms tick). Service replies that never arrive cannot hold
11/// completion reservations forever.
12pub const OPERATION_TIMEOUT_TICKS: u64 = 3_000;
13static NEXT_OPERATION: AtomicUsize = AtomicUsize::new(1);
14
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum Failure { Busy, PeerGone, Invalid, TooSmall }
17
18#[derive(Debug)]
19enum Phase { Queued, Dispatched, Terminal { kind: u32, len: usize } }
20
21#[derive(Debug)]
22pub struct Operation {
23    id: usize,
24    owner_cell: u64,
25    owner_generation: u64,
26    pub(crate) owner_tid: usize,
27    pub(crate) consumer_tid: usize,
28    peer_tid: usize,
29    peer_cell: u64,
30    peer_generation: u64,
31    deadline: u64,
32    phase: Phase,
33    reply: ipc_wire::IpcWireMessage,
34    #[cfg(cellos_p01_trace)]
35    _p01_obligation: ObservedOperation,
36}
37
38impl Operation {
39    fn owned_by(&self, task: &Task) -> bool {
40        self.owner_cell == task.cell_id.0 && self.owner_generation == task.cell_generation
41    }
42
43    pub(crate) fn can_consume(&self, task: &Task) -> bool {
44        self.owned_by(task) && (self.consumer_tid == task.id || self.owner_tid == task.id)
45    }
46}

48#[cfg(cellos_p01_trace)]
49#[derive(Debug)]
50struct ObservedOperation(Option<crate::p01_resources::Obligation>);
51
52#[cfg(cellos_p01_trace)]
53impl Drop for ObservedOperation {
54    fn drop(&mut self) {
55        // This last field drops after reply storage, including a push rollback.
56        // Terminal phase alone never releases the operation's retained storage.
57        crate::p01_resources::finish(self.0.take(), 1);
58    }
59}

61/// Task-local, kernel-charged storage. Tokens are global monotonic identifiers,
62/// never slot indexes; reuse after take cannot alias a late service reply.
63#[derive(Default)]
64pub struct Operations(Vec<Operation>);
65
66impl Operations {
67    pub fn new() -> Self { Self(Vec::new()) }
68    pub fn len(&self) -> usize { self.0.len() }
69    #[cfg(cellos_p01_trace)]
70    fn observe_owner(&self, owner: crate::p01_measurement::Owner) {
71        crate::memory::heap::p01_owner::runtime_vec(&self.0, owner, crate::memory::heap::p01_owner::OPERATIONS);
72    }
73    pub fn find(&self, id: usize) -> Option<&Operation> { self.0.iter().find(|op| op.id == id) }
74    fn find_mut(&mut self, id: usize) -> Option<&mut Operation> { self.0.iter_mut().find(|op| op.id == id) }
75    fn remove(&mut self, id: usize) -> Option<Operation> {
76        let pos = self.0.iter().position(|op| op.id == id)?;
77        Some(self.0.remove(pos))
78    }
79    fn push(&mut self, op: Operation) -> Result<(), Failure> {
80        if self.0.len() == MAX_OPERATIONS { return Err(Failure::Busy); }
81        if self.0.len() == self.0.capacity() {
82            let previous = super::hart_local::current_cell_id();
83            super::hart_local::set_current_cell_id(0);
84            let reserved = self.0.try_reserve(1);
85            super::hart_local::set_current_cell_id(previous);
86            reserved.map_err(|_| Failure::Busy)?;
87        }
88        self.0.push(op);
89        Ok(())
90    }
91    pub fn peer_died(&mut self, tid: usize, cell: u64, generation: u64) -> bool {
92        let mut changed = false;
93        for op in &mut self.0 {
94            if op.peer_tid == tid && op.peer_cell == cell && op.peer_generation == generation
95                && !matches!(op.phase, Phase::Terminal { .. }) {
96                op.phase = Phase::Terminal { kind: api::syscall::ipc_status::PEER_GONE, len: 0 };
97                #[cfg(cellos_p01_trace)]
98                crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
99                changed = true;
100            }
101        }
102        changed
103    }
104    fn has_terminal(&self) -> bool { self.0.iter().any(|op| matches!(op.phase, Phase::Terminal { .. })) }
105}

107impl Drop for Operations {
108    fn drop(&mut self) {
109        let rows = core::mem::take(&mut self.0);
110        let previous = super::hart_local::current_cell_id();
111        super::hart_local::set_current_cell_id(0);
112        drop(rows);
113        super::hart_local::set_current_cell_id(previous);
114    }
115}

117fn live(task: &Task) -> bool { !matches!(task.state, TaskState::Retiring | TaskState::Terminated) }

119fn wake(sched: &mut Scheduler, owner: usize) {
120    if let Some(task) = sched.tasks.get_mut(&owner) {
121        if matches!(task.state, TaskState::WaitIpc { .. }) {
122            task.state = TaskState::Ready;
123            let priority = sched.push_ready(owner);
124            sched.pend_preempt_if_needed(priority);
125        }
126    }
127}

129fn new_token() -> Result<usize, Failure> {
130    NEXT_OPERATION.fetch_update(Ordering::Relaxed, Ordering::Relaxed,
131        |next| (next <= isize::MAX as usize).then_some(next + 1))
132        .map_err(|_| Failure::Busy)
133}

135pub fn submit(owner: usize, peer: usize, ptr: usize, len: usize) -> Result<usize, Failure> {
136    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD { return Err(Failure::Invalid); }
137    #[cfg(cellos_p01_trace)]
138    let heap_owner;
139    let (view, binding, peer_binding) = {
140        let guard = SCHEDULER.lock();
141        let sched = guard.as_ref().ok_or(Failure::PeerGone)?;
142        let task = sched.tasks.get(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
143        let target = sched.tasks.get(&peer).filter(|task| live(task)).ok_or(Failure::PeerGone)?;
144        #[cfg(cellos_p01_trace)]
145        { heap_owner = crate::p01_resources::owner(task); }
146        (TaskCopyView::of(task), (task.cell_id.0, task.cell_generation),
147         (target.cell_id.0, target.cell_generation))
148    };
149    let token = new_token()?;
150    let header = ipc_wire::IpcWireHeader {
151        sender_tid: owner, sender_cell_id: binding.0,
152        sender_generation: binding.1,
153        delivery_id: super::next_delivery_id(), async_op: token,
154    };
155    // Copy outside SCHEDULER; no peer or user pointer persists in a task slot.
156    let wire = ipc_wire::IpcWireMessage::try_from_user(header, &view, ptr, len,
157        #[cfg(cellos_p01_trace)] heap_owner)
158        .map_err(|_| Failure::Invalid)?;
159    let reply_header = ipc_wire::IpcWireHeader {
160        sender_tid: peer, sender_cell_id: peer_binding.0,
161        sender_generation: peer_binding.1, delivery_id: 0, async_op: token,
162    };
163    let reply = ipc_wire::IpcWireMessage::try_new(
164        reply_header, &[0u8; ipc_wire::MAX_IPC_WIRE_PAYLOAD]).map_err(|_| Failure::Busy)?;
165    #[cfg(cellos_p01_trace)]
166    reply.observe_owner(heap_owner, crate::memory::heap::p01_owner::REPLY);
167    let mut guard = SCHEDULER.lock();
168    let sched = guard.as_mut().ok_or(Failure::PeerGone)?;
169    let caller = sched.tasks.get(&owner).filter(|task| live(task)
170        && (task.cell_id.0, task.cell_generation) == binding).ok_or(Failure::Invalid)?;
171    if caller.async_operations.len() >= MAX_OPERATIONS { return Err(Failure::Busy); }
172    let receiver = sched.tasks.get(&peer).filter(|task| live(task)
173        && (task.cell_id.0, task.cell_generation) == peer_binding).ok_or(Failure::PeerGone)?;
174    if super::paused_target_rejects(sched, owner, peer) { return Err(Failure::Busy); }
175    if receiver.pending_msgs.len() >= super::tcb::HOTSWAP_MSG_QUEUE_DEPTH { return Err(Failure::Busy); }
176    #[cfg(cellos_p01_trace)]
177    let p01_obligation = ObservedOperation(crate::p01_resources::operation_obligation(
178        crate::p01_resources::owner(caller)));
179    #[cfg(cellos_p01_trace)]
180    let _operations = crate::memory::heap::p01_owner::creation(Some(heap_owner),
181        crate::memory::heap::p01_owner::OPERATIONS);
182    let pushed = sched.tasks.get_mut(&owner).ok_or(Failure::Invalid)?.async_operations.push(Operation {
183        id: token, owner_cell: binding.0, owner_generation: binding.1,
184        owner_tid: owner, consumer_tid: owner,
185        peer_tid: peer, peer_cell: peer_binding.0, peer_generation: peer_binding.1,
186        deadline: (super::system_ticks() as u64).saturating_add(OPERATION_TIMEOUT_TICKS),
187        phase: Phase::Queued, reply,
188        #[cfg(cellos_p01_trace)]
189        _p01_obligation: p01_obligation,
190    });
191    #[cfg(cellos_p01_trace)]
192    drop(_operations);
193    #[cfg(cellos_p01_trace)]
194    sched.tasks.get(&owner).unwrap().async_operations.observe_owner(heap_owner);
195    pushed?;
196    let target = sched.tasks.get_mut(&peer).ok_or(Failure::PeerGone)?;
197    let recv_eligible = matches!(target.state, TaskState::Recv { mask, .. } if mask == 0 || mask == owner);
198    if super::queue_wire_msg(target, wire, super::tcb::HOTSWAP_MSG_QUEUE_DEPTH).is_err() {
199        sched.tasks.get_mut(&owner).unwrap().async_operations.remove(token);
200        return Err(Failure::Busy);
201    }
202    let wake_cause = super::wake_after_ipc_publish(target, recv_eligible);
203    if wake_cause.made_runnable() {
204        let priority = sched.push_ready(peer);
205        sched.pend_preempt_if_needed(priority);
206    }
207    Ok(token)
208}

210/// Snapshotting the request is the dispatch boundary. If a copy-out later
211/// faults, cancellation stays conservatively indeterminate, not falsely safe.
212pub fn dispatch(sched: &mut Scheduler, peer: usize, header: ipc_wire::IpcWireHeader) {
213    if header.async_op == 0 { return; }
214    let Some(provider) = sched.tasks.get(&peer).filter(|task| live(task)) else { return; };
215    let peer_binding = (provider.cell_id.0, provider.cell_generation);
216    let Some(owner) = sched.tasks.get_mut(&header.sender_tid).filter(|task| live(task)) else { return; };
217    if owner.cell_id.0 != header.sender_cell_id || owner.cell_generation != header.sender_generation { return; }
218    let owner_binding = (owner.cell_id.0, owner.cell_generation);
219    let Some(op) = owner.async_operations.find_mut(header.async_op) else { return; };
220    if (op.owner_cell, op.owner_generation) == owner_binding
221        && (op.peer_tid, op.peer_cell, op.peer_generation) == (peer, peer_binding.0, peer_binding.1)
222        && matches!(op.phase, Phase::Queued) {
223        op.phase = Phase::Dispatched;
224    }
225}

227fn terminal(sched: &mut Scheduler, owner_tid: usize, token: usize, provider: usize,
228            exact_peer: bool, ptr: usize, len: usize) -> Result<(), Failure> {
229    let Some(owner) = sched.tasks.get(&owner_tid).filter(|task| live(task)) else {
230        return Err(Failure::PeerGone);
231    };
232    let Some(op) = owner.async_operations.find(token) else { return Err(Failure::Invalid); };
233    let Some(peer) = sched.tasks.get(&provider).filter(|task| live(task)) else {
234        return Err(Failure::PeerGone);
235    };
236    // Explicit replies may be delegated to a worker of the accepted provider's
237    // cell incarnation. Only the implicit bridge requires the served TID too.
238    // Authority always comes from scheduler identity, never caller input.
239    if !op.owned_by(owner) || (exact_peer && op.peer_tid != provider)
240        || op.peer_cell != peer.cell_id.0 || op.peer_generation != peer.cell_generation
241        || !matches!(op.phase, Phase::Dispatched) { return Err(Failure::Invalid); }
242    let view = TaskCopyView::of(peer);
243    let op = sched.tasks.get_mut(&owner_tid).unwrap().async_operations.find_mut(token).unwrap();
244    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD {
245        op.phase = Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, len: 0 };
246        #[cfg(cellos_p01_trace)]
247        crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
248        wake(sched, owner_tid);
249        return Err(Failure::Invalid);
250    }
251    let copied = view.read_into(ptr, &mut op.reply.as_mut_slice()[..len]);
252    op.phase = Phase::Terminal {
253        kind: if copied.is_ok() { api::syscall::ipc_status::REPLY }
254            else { api::syscall::ipc_status::INDETERMINATE },
255        len: if copied.is_ok() { len } else { 0 },
256    };
257    #[cfg(cellos_p01_trace)]
258    crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
259    wake(sched, owner_tid);
260    copied.map_err(|_| Failure::Invalid)
261}

263/// Only a Send to the exact currently served caller can be a legacy reply.
264/// Its token remains as a tombstone until another receive replaces the context:
265/// duplicate/cancelled/late replies must not fall through to the raw mailbox.
266/// Services sending same-destination events or deferring across receives must
267/// capture the token and use explicit replies instead of this bridge.
268pub fn reply_current(peer: usize, destination: usize, ptr: usize, len: usize)
269    -> Option<Result<(), Failure>> {
270    let mut guard = SCHEDULER.lock();
271    reply_current_in_sched(guard.as_mut()?, peer, destination, ptr, len)
272}
273
274fn reply_current_in_sched(sched: &mut Scheduler, peer: usize, destination: usize,
275                          ptr: usize, len: usize) -> Option<Result<(), Failure>> {
276    let provider = sched.tasks.get(&peer).filter(|task| live(task))?;
277    let token = provider.current_async_op;
278    if token == 0 || provider.current_caller != Some(destination) { return None; }
279    let owner_binding = (provider.current_caller_cell_id, provider.current_caller_cell_generation);
280    let owner_matches = sched.tasks.get(&destination)
281        .is_some_and(|task| (task.cell_id.0, task.cell_generation) == owner_binding);
282    if !owner_matches { return Some(Ok(())); }
283    Some(match terminal(sched, destination, token, peer, true, ptr, len) {
284        // Invalid tokens and terminal results are quarantined, never raw Send.
285        Err(Failure::Invalid) => Ok(()),
286        other => other,
287    })
288}

301fn reply_in_sched(sched: &mut Scheduler, peer: usize, token: usize,
302                  ptr: usize, len: usize) -> Result<(), Failure> {
303    if token == 0 { return Err(Failure::Invalid); }
304    let owner = sched.tasks.iter().find_map(|(&id, task)| task.async_operations.find(token)
305        .map(|_| id));
306    let result = owner.ok_or(Failure::Invalid)
307        .and_then(|owner| terminal(sched, owner, token, peer, false, ptr, len));
308    // An explicit response ends this request even if cancellation/take already
309    // retired its token. Otherwise later one-way events would be mistaken for
310    // stale implicit replies. VFS's exact lease cleanup is handled by syscall.
311    if let Some(binding) = sched.tasks.get(&peer).filter(|task| live(task))
312        .map(|task| (task.cell_id, task.cell_generation)) {
313        for task in sched.tasks.values_mut() {
314            if (task.cell_id, task.cell_generation) == binding
315                && task.current_async_op == token
316                && !crate::fast_ipc::is_registered_vfs_cell(task.cell_id.0 as usize) {
317                task.clear_current_caller_context();
318            }
319        }
320    }
321    result
322}

330fn cancel_in_sched(sched: &mut Scheduler, owner: usize, token: usize) -> Result<(), Failure> {
331    let task = sched.tasks.get(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
332    let op = task.async_operations.find(token).ok_or(Failure::Invalid)?;
333    if !op.owned_by(task) { return Err(Failure::Invalid); }
334    let queued = matches!(op.phase, Phase::Queued);
335    let peer = op.peer_tid;
336    if queued {
337        // No receiver snapshot has begun; remove only the matching token.
338        if let Some(receiver) = sched.tasks.get_mut(&peer) {
339            if let Some(pos) = receiver.pending_msgs.iter().position(|msg|
340                msg.wire_header().is_some_and(|header| header.async_op == token)) {
341                receiver.pending_msgs.remove(pos);
342            }
343        }
344    }
345    let op = sched.tasks.get_mut(&owner).unwrap().async_operations.find_mut(token).unwrap();
346    if !matches!(op.phase, Phase::Terminal { .. }) {
347        op.phase = Phase::Terminal { kind: if queued { api::syscall::ipc_status::CANCELLED }
348            else { api::syscall::ipc_status::INDETERMINATE }, len: 0 };
349        #[cfg(cellos_p01_trace)]
350        crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
351        wake(sched, owner);
352    }
353    Ok(())
354}

356pub fn take(consumer: usize, token: usize, reply_ptr: usize, reply_len: usize,
357            status_ptr: usize) -> Result<bool, Failure> {
358    let mut guard = SCHEDULER.lock();
359    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
360    let caller = sched.tasks.get(&consumer).filter(|task| live(task)).ok_or(Failure::Invalid)?;
361    let caller_cell = (caller.cell_id.0, caller.cell_generation);
362
363    let (owner_tid, op_ref) = sched
364        .tasks
365        .iter()
366        .filter(|(_, t)| (t.cell_id.0, t.cell_generation) == caller_cell)
367        .find_map(|(&id, t)| t.async_operations.find(token).map(|op| (id, op)))
368        .ok_or(Failure::Invalid)?;
369    if !op_ref.can_consume(caller) { return Err(Failure::Invalid); }
370    let Phase::Terminal { kind, len } = &op_ref.phase else { return Ok(false); };
371    let data = &op_ref.reply.as_slice()[..*len];
372    if data.len() > reply_len { return Err(Failure::TooSmall); }
373    let view = TaskCopyView::of(caller);
374    view.validate_writable(status_ptr, api::syscall::IPC_STATUS_LEN).map_err(|_| Failure::Invalid)?;
375    if !data.is_empty() {
376        view.validate_writable(reply_ptr, data.len()).map_err(|_| Failure::Invalid)?;
377        view.write_bytes(reply_ptr, data).map_err(|_| Failure::Invalid)?;
378    }
379    let mut status = [0u8; api::syscall::IPC_STATUS_LEN];
380    status[..4].copy_from_slice(&api::syscall::IPC_STATUS_VERSION.to_le_bytes());
381    status[4..8].copy_from_slice(&kind.to_le_bytes());
382    status[8..12].copy_from_slice(&(data.len() as u32).to_le_bytes());
383    view.write_bytes(status_ptr, &status).map_err(|_| Failure::Invalid)?;
384    sched.tasks.get_mut(&owner_tid).unwrap().async_operations.remove(token);
385    Ok(true)
386}

388pub fn transfer_consumption(caller_tid: usize, token: usize, target_tid: usize) -> Result<(), Failure> {
389    let mut guard = SCHEDULER.lock();
390    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
391    let caller = sched.tasks.get(&caller_tid).filter(|t| live(t)).ok_or(Failure::Invalid)?;
392    let target = sched.tasks.get(&target_tid).filter(|t| live(t)).ok_or(Failure::Invalid)?;
393    if (caller.cell_id, caller.cell_generation) != (target.cell_id, target.cell_generation) {
394        return Err(Failure::Invalid);
395    }
396    let caller_cell = (caller.cell_id.0, caller.cell_generation);
397    let op = sched
398        .tasks
399        .values_mut()
400        .filter(|t| (t.cell_id.0, t.cell_generation) == caller_cell)
401        .find_map(|t| t.async_operations.find_mut(token))
402        .ok_or(Failure::Invalid)?;
403    if op.consumer_tid != caller_tid && op.owner_tid != caller_tid {
404        return Err(Failure::Invalid);
405    }
406    op.consumer_tid = target_tid;
407    Ok(())
408}

410/// Drop or indeterminate all operations owned by an exiting thread.
411/// If take won before exit, the operation was already removed, ensuring
412/// single-winner serialization against take.
413pub fn on_owner_exits(sched: &mut Scheduler, owner_tid: usize) {
414    let to_cancel: alloc::vec::Vec<(usize, usize, bool)> = match sched.tasks.get(&owner_tid) {
415        Some(task) => task.async_operations.0.iter().filter_map(|op| {
416            if op.owner_tid == owner_tid {
417                match op.phase {
418                    Phase::Queued => Some((op.id, op.peer_tid, true)),
419                    Phase::Dispatched => Some((op.id, op.peer_tid, false)),
420                    Phase::Terminal { .. } => None,
421                }
422            } else {
423                None
424            }
425        }).collect(),
426        None => return,
427    };
428
429    for &(token, peer, queued) in &to_cancel {
430        if queued {
431            if let Some(receiver) = sched.tasks.get_mut(&peer) {
432                if let Some(pos) = receiver.pending_msgs.iter().position(|msg|
433                    msg.wire_header().is_some_and(|header| header.async_op == token)) {
434                    receiver.pending_msgs.remove(pos);
435                }
436            }
437        }
438    }
439
440    if let Some(task) = sched.tasks.get_mut(&owner_tid) {
441        for (token, _, queued) in to_cancel {
442            if let Some(op) = task.async_operations.find_mut(token) {
443                op.phase = Phase::Terminal {
444                    kind: if queued { api::syscall::ipc_status::CANCELLED } else { api::syscall::ipc_status::INDETERMINATE },
445                    len: 0,
446                };
447            }
448        }
449    }
450}

452pub fn wait(owner: usize, ticks: u64) -> Result<bool, Failure> {
453    let deadline = (ticks != 0).then(|| (super::system_ticks() as u64).saturating_add(ticks));
454    {
455        let mut guard = SCHEDULER.lock();
456        let sched = guard.as_mut().ok_or(Failure::Invalid)?;
457        let task = sched.tasks.get_mut(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
458        if task.async_operations.has_terminal() { return Ok(true); }
459        super::arm_ipc_block_handoff(owner);
460        task.state = TaskState::WaitIpc { deadline };
461    }
462    super::yield_cpu();
463    let guard = SCHEDULER.lock();
464    Ok(guard.as_ref().and_then(|s| s.tasks.get(&owner))
465        .is_some_and(|task| task.async_operations.has_terminal()))
466}

468/// Peer death is an infallible terminal transition; storage was reserved on submit.
469pub fn peer_died(sched: &mut Scheduler, peer: usize, cell: u64, generation: u64) {
470    let mut after = 0usize;
471    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
472        after = match owner.checked_add(1) { Some(next) => next, None => break };
473        if sched.tasks.get_mut(&owner).is_some_and(|task|
474            task.async_operations.peer_died(peer, cell, generation)) {
475            wake(sched, owner);
476        }
477    }
478}

480/// Drop queued requests from an owner whose task generation has exited.
481/// Dispatched requests retain their service-side grant lease until the
482/// service's existing reply/next-public-receive cleanup boundary.
483pub fn owner_died(sched: &mut Scheduler, owner: usize, cell: u64, generation: u64) {
484    for task in sched.tasks.values_mut() {
485        let mut index = 0;
486        while index < task.pending_msgs.len() {
487            let queued = task.pending_msgs.as_slice()[index].wire_header()
488                .is_some_and(|header| header.async_op != 0
489                    && header.sender_tid == owner
490                    && header.sender_cell_id == cell
491                    && header.sender_generation == generation);
492            if queued { task.pending_msgs.remove(index); } else { index += 1; }
493        }
494    }
495}

497/// Complete queued requests tied to a provider being replaced. Called only
498/// after the successful hot-swap commit, under SCHEDULER. The replacement
499/// must not receive these wires because it is a different generation.
500pub(crate) fn cutover_peer(sched: &mut Scheduler, peer: usize, cell: u64, generation: u64) {
501    let mut after = 0usize;
502    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
503        after = match owner.checked_add(1) { Some(next) => next, None => break };
504        let changed = if let Some(task) = sched.tasks.get_mut(&owner) {
505            let mut changed = false;
506            for op in &mut task.async_operations.0 {
507                if op.peer_tid == peer && op.peer_cell == cell && op.peer_generation == generation
508                    && matches!(op.phase, Phase::Queued) {
509                    op.phase = Phase::Terminal {
510                        kind: api::syscall::ipc_status::PEER_GONE, len: 0,
511                    };
512                    #[cfg(cellos_p01_trace)]
513                    crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
514                    changed = true;
515                }
516            }
517            changed
518        } else { false };
519        if changed { wake(sched, owner); }
520    }
521}

523/// Expire accepted work without allocating in the scheduler's timer sweep.
524/// An undispatched request is removed from its peer mailbox before reporting
525/// PreDispatchTimeout. A dispatched request is conservatively indeterminate.
526pub fn expire(sched: &mut Scheduler, now: u64) {
527    let mut after = 0usize;
528    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
529        after = match owner.checked_add(1) { Some(next) => next, None => break };
530        for index in 0..MAX_OPERATIONS {
531            let expired = {
532                let Some(task) = sched.tasks.get_mut(&owner) else { break };
533                let Some(op) = task.async_operations.0.get_mut(index) else { break };
534                if now < op.deadline || matches!(op.phase, Phase::Terminal { .. }) {
535                    None
536                } else {
537                    let queued = matches!(op.phase, Phase::Queued);
538                    op.phase = Phase::Terminal {
539                        kind: if queued { api::syscall::ipc_status::PRE_DISPATCH_TIMEOUT }
540                            else { api::syscall::ipc_status::INDETERMINATE },
541                        len: 0,
542                    };
543                    #[cfg(cellos_p01_trace)]
544                    crate::p01_resources::reservation_terminal(&mut op._p01_obligation.0);
545                    Some((op.id, op.peer_tid, queued))
546                }
547            };
548            if let Some((token, peer, queued)) = expired {
549                if queued {
550                    if let Some(target) = sched.tasks.get_mut(&peer) {
551                        if let Some(pos) = target.pending_msgs.iter().position(|msg|
552                            msg.wire_header().is_some_and(|header| header.async_op == token)) {
553                            target.pending_msgs.remove(pos);
554                        }
555                    }
556                }
557                wake(sched, owner);
558            }
559        }
560    }

925    #[test]
926    fn deterministic_two_hart_publication_and_wake_races() {
927        let _guard = crate::TEST_STATE_LOCK.lock();
928        let mut sched = fixture();
929
930        // 1. Race 1: Pre-park publication race
931        // Provider publishes reply before caller on another hart arms WaitIpc.
932        add_operation(&mut sched, OWNER, PROVIDER, 101, Phase::Dispatched);
933        serve(&mut sched, PROVIDER, OWNER, 101);
934        sched.tasks.get_mut(&OWNER).unwrap().state = TaskState::Ready;
935        assert_eq!(reply_current_in_sched(&mut sched, PROVIDER, OWNER, 0, 0), Some(Ok(())));
936        // When caller subsequently executes wait, has_terminal is true immediately,
937        // so caller does not block or lose the wakeup.
938        assert!(sched.tasks[&OWNER].async_operations.has_terminal());
939        assert!(matches!(sched.tasks[&OWNER].async_operations.find(101).unwrap().phase,
940            Phase::Terminal { kind: api::syscall::ipc_status::REPLY, .. }));
941
942        // 2. Race 2: Cross-hart parked wake race
943        // Caller is parked in WaitIpc when Provider settles the call.
944        add_operation(&mut sched, OWNER, PROVIDER, 102, Phase::Dispatched);
945        serve(&mut sched, PROVIDER, OWNER, 102);
946        sched.tasks.get_mut(&OWNER).unwrap().state = TaskState::WaitIpc { deadline: None };
947        assert_eq!(reply_current_in_sched(&mut sched, PROVIDER, OWNER, 0, 0), Some(Ok(())));
948        // Caller must be transitioned to Ready and pushed to the ready queue.
949        assert!(matches!(sched.tasks[&OWNER].state, TaskState::Ready));
950        assert!(sched.tasks[&OWNER].async_operations.has_terminal());
951        super::super::hart_local::ready::remove_from_all(OWNER);
952
953        // 3. Race 3: Remote peer-death vs parked caller race
954        // Caller is parked in WaitIpc when Provider crashes on another hart.
955        add_operation(&mut sched, OWNER, PROVIDER, 103, Phase::Dispatched);
956        sched.tasks.get_mut(&OWNER).unwrap().state = TaskState::WaitIpc { deadline: None };
957        peer_died(&mut sched, PROVIDER, 1_003, 7);
958        // Caller must be transitioned to Ready with PeerGone terminal.
959        assert!(matches!(sched.tasks[&OWNER].state, TaskState::Ready));
960        assert!(matches!(sched.tasks[&OWNER].async_operations.find(103).unwrap().phase,
961            Phase::Terminal { kind: api::syscall::ipc_status::PEER_GONE, .. }));
962        super::super::hart_local::ready::remove_from_all(OWNER);
963
964        // 4. Race 4: Remote deadline expiration vs reply race
965        // Case A: Reply wins lock first, deadline cannot overwrite it.
966        add_operation(&mut sched, OWNER, WORKER, 104, Phase::Dispatched);
967        serve(&mut sched, WORKER, OWNER, 104);
968        assert_eq!(reply_current_in_sched(&mut sched, WORKER, OWNER, 0, 0), Some(Ok(())));
969        expire(&mut sched, u64::MAX);
970        assert!(matches!(sched.tasks[&OWNER].async_operations.find(104).unwrap().phase,
971            Phase::Terminal { kind: api::syscall::ipc_status::REPLY, .. }));
972
973        // Case B: Expire wins lock first, subsequent reply is rejected.
974        add_operation(&mut sched, OWNER, WORKER, 105, Phase::Dispatched);
975        serve(&mut sched, WORKER, OWNER, 105);
976        expire(&mut sched, 100);
977        assert!(matches!(sched.tasks[&OWNER].async_operations.find(105).unwrap().phase,
978            Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, .. }));
979        assert_eq!(reply_current_in_sched(&mut sched, WORKER, OWNER, 0, 0), Some(Ok(())));
980        // Terminal remains Indeterminate, not overwritten by late reply.
981        assert!(matches!(sched.tasks[&OWNER].async_operations.find(105).unwrap().phase,
982            Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, .. }));
983    }
