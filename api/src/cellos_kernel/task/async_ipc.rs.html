1//! Kernel-owned, bounded request/reply operations. All transitions are serialized
2//! by SCHEDULER; neither a reply nor a wake depends on caller-owned memory.
3
4use super::{copy_glue::TaskCopyView, ipc_wire, scheduler::Scheduler, tcb::{Task, TaskState}, SCHEDULER};
5use alloc::vec::Vec;
6use core::sync::atomic::{AtomicUsize, Ordering};

8pub const MAX_OPERATIONS: usize = 64;
9/// Accepted operations have a kernel-enforced terminal deadline (30 seconds at
10/// the scheduler's 10 ms tick). Service replies that never arrive cannot hold
11/// completion reservations forever.
12pub const OPERATION_TIMEOUT_TICKS: u64 = 3_000;
13static NEXT_OPERATION: AtomicUsize = AtomicUsize::new(1);
14
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum Failure { Busy, PeerGone, Invalid, TooSmall }
17
18#[derive(Debug)]
19enum Phase { Queued, Dispatched, Terminal { kind: u32, len: usize } }
20
21#[derive(Debug)]
22pub struct Operation {
23    id: usize,
24    peer_tid: usize,
25    peer_cell: u64,
26    peer_generation: u64,
27    deadline: u64,
28    phase: Phase,
29    reply: ipc_wire::IpcWireMessage,
30}
31
32/// Task-local, kernel-charged storage. Tokens are global monotonic identifiers,
33/// never slot indexes; reuse after take cannot alias a late service reply.
34#[derive(Default)]
35pub struct Operations(Vec<Operation>);
36
37impl Operations {
38    pub fn new() -> Self { Self(Vec::new()) }
39    pub fn len(&self) -> usize { self.0.len() }
40    pub fn find(&self, id: usize) -> Option<&Operation> { self.0.iter().find(|op| op.id == id) }
41    fn find_mut(&mut self, id: usize) -> Option<&mut Operation> { self.0.iter_mut().find(|op| op.id == id) }
42    fn remove(&mut self, id: usize) -> Option<Operation> {
43        let pos = self.0.iter().position(|op| op.id == id)?;
44        Some(self.0.remove(pos))
45    }
46    fn push(&mut self, op: Operation) -> Result<(), Failure> {
47        if self.0.len() == MAX_OPERATIONS { return Err(Failure::Busy); }
48        if self.0.len() == self.0.capacity() {
49            let previous = super::hart_local::current_cell_id();
50            super::hart_local::set_current_cell_id(0);
51            let reserved = self.0.try_reserve(1);
52            super::hart_local::set_current_cell_id(previous);
53            reserved.map_err(|_| Failure::Busy)?;
54        }
55        self.0.push(op);
56        Ok(())
57    }
58    pub fn peer_died(&mut self, tid: usize, cell: u64, generation: u64) -> bool {
59        let mut changed = false;
60        for op in &mut self.0 {
61            if op.peer_tid == tid && op.peer_cell == cell && op.peer_generation == generation
62                && !matches!(op.phase, Phase::Terminal { .. }) {
63                op.phase = Phase::Terminal { kind: api::syscall::ipc_status::PEER_GONE, len: 0 };
64                changed = true;
65            }
66        }
67        changed
68    }
69    fn has_terminal(&self) -> bool { self.0.iter().any(|op| matches!(op.phase, Phase::Terminal { .. })) }
70}
71
72impl Drop for Operations {
73    fn drop(&mut self) {
74        let rows = core::mem::take(&mut self.0);
75        let previous = super::hart_local::current_cell_id();
76        super::hart_local::set_current_cell_id(0);
77        drop(rows);
78        super::hart_local::set_current_cell_id(previous);
79    }
80}

82fn live(task: &Task) -> bool { !matches!(task.state, TaskState::Retiring | TaskState::Terminated) }

84fn wake(sched: &mut Scheduler, owner: usize) {
85    if let Some(task) = sched.tasks.get_mut(&owner) {
86        if matches!(task.state, TaskState::WaitIpc { .. }) {
87            task.state = TaskState::Ready;
88            let priority = sched.push_ready(owner);
89            sched.pend_preempt_if_needed(priority);
90        }
91    }
92}

94fn new_token() -> Result<usize, Failure> {
95    NEXT_OPERATION.fetch_update(Ordering::Relaxed, Ordering::Relaxed,
96        |next| (next <= isize::MAX as usize).then_some(next + 1))
97        .map_err(|_| Failure::Busy)
98}

100pub fn submit(owner: usize, peer: usize, ptr: usize, len: usize) -> Result<usize, Failure> {
101    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD { return Err(Failure::Invalid); }
102    let (view, binding, peer_binding) = {
103        let guard = SCHEDULER.lock();
104        let sched = guard.as_ref().ok_or(Failure::PeerGone)?;
105        let task = sched.tasks.get(&owner).filter(|task| live(task)).ok_or(Failure::Invalid)?;
106        let target = sched.tasks.get(&peer).filter(|task| live(task)).ok_or(Failure::PeerGone)?;
107        (TaskCopyView::of(task), (task.cell_id.0, task.cell_generation),
108         (target.cell_id.0, target.cell_generation))
109    };
110    let token = new_token()?;
111    let header = ipc_wire::IpcWireHeader {
112        sender_tid: owner, sender_cell_id: binding.0,
113        sender_generation: binding.1,
114        delivery_id: super::next_delivery_id(), async_op: token,
115    };
116    // Copy outside SCHEDULER; no peer or user pointer persists in a task slot.
117    let wire = ipc_wire::IpcWireMessage::try_from_user(header, &view, ptr, len)
118        .map_err(|_| Failure::Invalid)?;
119    let reply_header = ipc_wire::IpcWireHeader {
120        sender_tid: peer, sender_cell_id: peer_binding.0,
121        sender_generation: peer_binding.1, delivery_id: 0, async_op: token,
122    };
123    let reply = ipc_wire::IpcWireMessage::try_new(
124        reply_header, &[0u8; ipc_wire::MAX_IPC_WIRE_PAYLOAD]).map_err(|_| Failure::Busy)?;
125    let mut guard = SCHEDULER.lock();
126    let sched = guard.as_mut().ok_or(Failure::PeerGone)?;
127    let caller = sched.tasks.get(&owner).filter(|task| live(task)
128        && (task.cell_id.0, task.cell_generation) == binding).ok_or(Failure::Invalid)?;
129    if caller.async_operations.len() >= MAX_OPERATIONS { return Err(Failure::Busy); }
130    let receiver = sched.tasks.get(&peer).filter(|task| live(task)
131        && (task.cell_id.0, task.cell_generation) == peer_binding).ok_or(Failure::PeerGone)?;
132    if super::paused_target_rejects(sched, owner, peer) { return Err(Failure::Busy); }
133    if receiver.pending_msgs.len() >= super::tcb::HOTSWAP_MSG_QUEUE_DEPTH { return Err(Failure::Busy); }
134    sched.tasks.get_mut(&owner).ok_or(Failure::Invalid)?.async_operations.push(Operation {
135        id: token, peer_tid: peer, peer_cell: peer_binding.0, peer_generation: peer_binding.1,
136        deadline: (super::system_ticks() as u64).saturating_add(OPERATION_TIMEOUT_TICKS),
137        phase: Phase::Queued, reply,
138    })?;
139    let target = sched.tasks.get_mut(&peer).ok_or(Failure::PeerGone)?;
140    let recv_eligible = matches!(target.state, TaskState::Recv { mask, .. } if mask == 0 || mask == owner);
141    if super::queue_wire_msg(target, wire, super::tcb::HOTSWAP_MSG_QUEUE_DEPTH).is_err() {
142        sched.tasks.get_mut(&owner).unwrap().async_operations.remove(token);
143        return Err(Failure::Busy);
144    }
145    let wake_cause = super::wake_after_ipc_publish(target, recv_eligible);
146    if wake_cause.made_runnable() {
147        let priority = sched.push_ready(peer);
148        sched.pend_preempt_if_needed(priority);
149    }
150    Ok(token)
151}

153/// Snapshotting the request is the dispatch boundary. If a copy-out later
154/// faults, cancellation stays conservatively indeterminate, not falsely safe.
155pub fn dispatch(sched: &mut Scheduler, peer: usize, header: ipc_wire::IpcWireHeader) {
156    if header.async_op == 0 { return; }
157    let Some(owner) = sched.tasks.get_mut(&header.sender_tid) else { return; };
158    if owner.cell_id.0 != header.sender_cell_id || owner.cell_generation != header.sender_generation { return; }
159    let Some(op) = owner.async_operations.find_mut(header.async_op) else { return; };
160    if op.peer_tid == peer && matches!(op.phase, Phase::Queued) {
161        op.phase = Phase::Dispatched;
162    }
163}

165fn terminal(sched: &mut Scheduler, owner_tid: usize, token: usize, peer: usize,
166            peer_cell: u64, peer_generation: u64, ptr: usize, len: usize) -> Result<(), Failure> {
167    let Some(owner) = sched.tasks.get(&owner_tid) else { return Err(Failure::PeerGone); };
168    if !live(owner) { return Err(Failure::PeerGone); }
169    let Some(op) = owner.async_operations.find(token) else { return Err(Failure::Invalid); };
170    if op.peer_tid != peer || op.peer_cell != peer_cell || op.peer_generation != peer_generation
171        || !matches!(op.phase, Phase::Dispatched) { return Err(Failure::Invalid); }
172    let view = sched.tasks.get(&peer).map(|task| TaskCopyView::of(task)).ok_or(Failure::PeerGone)?;
173    let op = sched.tasks.get_mut(&owner_tid).unwrap().async_operations.find_mut(token).unwrap();
174    if len > ipc_wire::MAX_IPC_WIRE_PAYLOAD {
175        op.phase = Phase::Terminal { kind: api::syscall::ipc_status::INDETERMINATE, len: 0 };
176        wake(sched, owner_tid);
177        return Err(Failure::Invalid);
178    }
179    let copied = view.read_into(ptr, &mut op.reply.as_mut_slice()[..len]);
180    op.phase = Phase::Terminal {
181        kind: if copied.is_ok() { api::syscall::ipc_status::REPLY }
182            else { api::syscall::ipc_status::INDETERMINATE },
183        len: if copied.is_ok() { len } else { 0 },
184    };
185    wake(sched, owner_tid);
186    copied.map_err(|_| Failure::Invalid)
187}

189/// Even a late legacy service reply is intercepted rather than misdelivered
190/// through the synchronous client mailbox.
191pub fn reply_current(peer: usize, destination: usize, ptr: usize, len: usize)
192    -> Option<Result<(), Failure>> {
193    let mut guard = SCHEDULER.lock();
194    let sched = guard.as_mut()?;
195    let provider = sched.tasks.get(&peer)?;
196    let token = provider.current_async_op;
197    if token == 0 || provider.current_caller != Some(destination) { return None; }
198    let binding = (provider.cell_id.0, provider.cell_generation);
199    Some(match terminal(sched, destination, token, peer, binding.0, binding.1, ptr, len) {
200        // Cancellation/reuse is terminal; quarantine the late response.
201        Err(Failure::Invalid) if !sched.tasks.get(&destination)
202            .and_then(|task| task.async_operations.find(token))
203            .is_some_and(|op| matches!(op.phase, Phase::Dispatched)) => Ok(()),
204        other => other,
205    })
206}

208pub fn current(peer: usize) -> usize {
209    SCHEDULER.lock().as_ref().and_then(|sched| sched.tasks.get(&peer))
210        .filter(|task| live(task)).map_or(0, |task| task.current_async_op)
211}

213pub fn reply(peer: usize, token: usize, ptr: usize, len: usize) -> Result<(), Failure> {
214    if token == 0 { return Err(Failure::Invalid); }
215    let mut guard = SCHEDULER.lock();
216    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
217    let provider = sched.tasks.get(&peer).filter(|task| live(task)).ok_or(Failure::PeerGone)?;
218    let (cell, generation) = (provider.cell_id.0, provider.cell_generation);
219    let owner = sched.tasks.iter().find_map(|(&id, task)| task.async_operations.find(token)
220        .map(|_| id)).ok_or(Failure::Invalid)?;
221    let result = terminal(sched, owner, token, peer, cell, generation, ptr, len);
222    if result.is_ok() {
223        if let Some(task) = sched.tasks.get_mut(&peer) {
224            if task.current_async_op == token
225                && !crate::fast_ipc::is_registered_vfs_cell(task.cell_id.0 as usize) {
226                task.clear_current_caller_context();
227            }
228        }
229    }
230    result
231}

233pub fn cancel(owner: usize, token: usize) -> Result<(), Failure> {
234    let mut guard = SCHEDULER.lock();
235    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
236    let op = sched.tasks.get_mut(&owner).ok_or(Failure::Invalid)?
237        .async_operations.find_mut(token).ok_or(Failure::Invalid)?;
238    let queued = matches!(op.phase, Phase::Queued);
239    let peer = op.peer_tid;
240    if queued {
241        // No receiver snapshot has begun; remove only the matching token.
242        if let Some(receiver) = sched.tasks.get_mut(&peer) {
243            if let Some(pos) = receiver.pending_msgs.iter().position(|msg|
244                msg.wire_header().is_some_and(|header| header.async_op == token)) {
245                receiver.pending_msgs.remove(pos);
246            }
247        }
248    }
249    let op = sched.tasks.get_mut(&owner).unwrap().async_operations.find_mut(token).unwrap();
250    if !matches!(op.phase, Phase::Terminal { .. }) {
251        op.phase = Phase::Terminal { kind: if queued { api::syscall::ipc_status::CANCELLED }
252            else { api::syscall::ipc_status::INDETERMINATE }, len: 0 };
253        wake(sched, owner);
254    }
255    Ok(())
256}

258pub fn take(owner: usize, token: usize, reply_ptr: usize, reply_len: usize,
259            status_ptr: usize) -> Result<bool, Failure> {
260    let mut guard = SCHEDULER.lock();
261    let sched = guard.as_mut().ok_or(Failure::Invalid)?;
262    let task = sched.tasks.get(&owner).ok_or(Failure::Invalid)?;
263    let op = task.async_operations.find(token).ok_or(Failure::Invalid)?;
264    let Phase::Terminal { kind, len } = &op.phase else { return Ok(false); };
265    let data = &op.reply.as_slice()[..*len];
266    if data.len() > reply_len { return Err(Failure::TooSmall); }
267    let view = TaskCopyView::of(task);
268    view.validate_writable(status_ptr, api::syscall::IPC_STATUS_LEN).map_err(|_| Failure::Invalid)?;
269    if !data.is_empty() {
270        view.validate_writable(reply_ptr, data.len()).map_err(|_| Failure::Invalid)?;
271        view.write_bytes(reply_ptr, data).map_err(|_| Failure::Invalid)?;
272    }
273    let mut status = [0u8; api::syscall::IPC_STATUS_LEN];
274    status[..4].copy_from_slice(&api::syscall::IPC_STATUS_VERSION.to_le_bytes());
275    status[4..8].copy_from_slice(&kind.to_le_bytes());
276    status[8..12].copy_from_slice(&(data.len() as u32).to_le_bytes());
277    view.write_bytes(status_ptr, &status).map_err(|_| Failure::Invalid)?;
278    sched.tasks.get_mut(&owner).unwrap().async_operations.remove(token);
279    Ok(true)
280}

282pub fn wait(owner: usize, ticks: u64) -> Result<bool, Failure> {
283    let deadline = (ticks != 0).then(|| (super::system_ticks() as u64).saturating_add(ticks));
284    {
285        let mut guard = SCHEDULER.lock();
286        let sched = guard.as_mut().ok_or(Failure::Invalid)?;
287        let task = sched.tasks.get_mut(&owner).ok_or(Failure::Invalid)?;
288        if task.async_operations.has_terminal() { return Ok(true); }
289        super::arm_ipc_block_handoff(owner);
290        task.state = TaskState::WaitIpc { deadline };
291    }
292    super::yield_cpu();
293    let guard = SCHEDULER.lock();
294    Ok(guard.as_ref().and_then(|s| s.tasks.get(&owner))
295        .is_some_and(|task| task.async_operations.has_terminal()))
296}

298/// Peer death is an infallible terminal transition; storage was reserved on submit.
299pub fn peer_died(sched: &mut Scheduler, peer: usize, cell: u64, generation: u64) {
300    let mut after = 0usize;
301    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
302        after = match owner.checked_add(1) { Some(next) => next, None => break };
303        if sched.tasks.get_mut(&owner).is_some_and(|task|
304            task.async_operations.peer_died(peer, cell, generation)) {
305            wake(sched, owner);
306        }
307    }
308}

310/// Drop queued requests from an owner whose task generation has exited.
311/// Dispatched requests retain their service-side grant lease until the
312/// service's existing reply/next-public-receive cleanup boundary.
313pub fn owner_died(sched: &mut Scheduler, owner: usize, cell: u64, generation: u64) {
314    for task in sched.tasks.values_mut() {
315        let mut index = 0;
316        while index < task.pending_msgs.len() {
317            let queued = task.pending_msgs.as_slice()[index].wire_header()
318                .is_some_and(|header| header.async_op != 0
319                    && header.sender_tid == owner
320                    && header.sender_cell_id == cell
321                    && header.sender_generation == generation);
322            if queued { task.pending_msgs.remove(index); } else { index += 1; }
323        }
324    }
325}

327/// Complete queued requests tied to a provider being replaced. Called only
328/// after the successful hot-swap commit, under SCHEDULER. The replacement
329/// must not receive these wires because it is a different generation.
330pub(crate) fn cutover_peer(sched: &mut Scheduler, peer: usize, cell: u64, generation: u64) {
331    let mut after = 0usize;
332    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
333        after = match owner.checked_add(1) { Some(next) => next, None => break };
334        let changed = if let Some(task) = sched.tasks.get_mut(&owner) {
335            let mut changed = false;
336            for op in &mut task.async_operations.0 {
337                if op.peer_tid == peer && op.peer_cell == cell && op.peer_generation == generation
338                    && matches!(op.phase, Phase::Queued) {
339                    op.phase = Phase::Terminal {
340                        kind: api::syscall::ipc_status::PEER_GONE, len: 0,
341                    };
342                    changed = true;
343                }
344            }
345            changed
346        } else { false };
347        if changed { wake(sched, owner); }
348    }
349}

351/// Expire accepted work without allocating in the scheduler's timer sweep.
352/// An undispatched request is removed from its peer mailbox before reporting
353/// PreDispatchTimeout. A dispatched request is conservatively indeterminate.
354pub fn expire(sched: &mut Scheduler, now: u64) {
355    let mut after = 0usize;
356    while let Some((&owner, _)) = sched.tasks.range(after..).next() {
357        after = match owner.checked_add(1) { Some(next) => next, None => break };
358        for index in 0..MAX_OPERATIONS {
359            let expired = {
360                let Some(task) = sched.tasks.get_mut(&owner) else { break };
361                let Some(op) = task.async_operations.0.get_mut(index) else { break };
362                if now < op.deadline || matches!(op.phase, Phase::Terminal { .. }) {
363                    None
364                } else {
365                    let queued = matches!(op.phase, Phase::Queued);
366                    op.phase = Phase::Terminal {
367                        kind: if queued { api::syscall::ipc_status::PRE_DISPATCH_TIMEOUT }
368                            else { api::syscall::ipc_status::INDETERMINATE },
369                        len: 0,
370                    };
371                    Some((op.id, op.peer_tid, queued))
372                }
373            };
374            if let Some((token, peer, queued)) = expired {
375                if queued {
376                    if let Some(target) = sched.tasks.get_mut(&peer) {
377                        if let Some(pos) = target.pending_msgs.iter().position(|msg|
378                            msg.wire_header().is_some_and(|header| header.async_op == token)) {
379                            target.pending_msgs.remove(pos);
380                        }
381                    }
382                }
383                wake(sched, owner);
384            }
385        }
386    }
