1//! Boot self-test for real user-thread first entry.
2//!
3//! This proves more than "the thread became Ready": the real `Syscall::Spawn`
4//! path must switch into U-mode, hand the thread its `arg`, and let it complete
5//! a syscall/exit round-trip before control returns to the boot context.
6

14use types::CellId;

19#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
20const PARENT_TID: usize = 9202;
21#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
22const TEST_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 5) as u64;
23#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
24const ENTRY: usize = 0x0001_4000;
25#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
26const MSG_OFFSET: usize = 0x40;
27#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
28const MSG: &[u8] = b"[selftest] THREAD-ENTRY arg=ok\n";
29
30#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
31fn insert_parent() {
32    let mut parent = Box::new(Task::new(
33        PARENT_TID,
34        CellId(TEST_CELL_ID),
35        "selftest",
36        Vec::new(),
37    ));
38    parent.cell_generation = 1;
39    parent.root_tid = PARENT_TID;
40    parent.spawn_cap = Some(super::cap::SpawnCap::new());
41    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
42        let owner = api::cell_owner::CellOwner::new(TEST_CELL_ID, 1, PARENT_TID as u64);
43        sched.publish_live_cell_owner(owner);
44        sched.tasks.insert(PARENT_TID, parent);
45    }
46}

48#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
49fn remove(tid: usize) {
50    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
51        if let Some(task) = sched.tasks.remove(&tid) {
52            if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
53                let owner = api::cell_owner::CellOwner::new(
54                    task.cell_id.0,
55                    task.cell_generation,
56                    task.root_tid as u64,
57                );
58                sched.clear_live_cell_owner_for_test(owner);
59            }
60        }
61    }
62    super::hart_local::ready::remove_from_all(tid);
63}
64#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
65fn reap() {
66    let dead = super::SCHEDULER
67        .lock()
68        .as_mut()
69        .map(|sched| sched.take_reapable_zombies())
70        .unwrap_or_default();
71    drop(dead);
72}

74#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
75fn drain_user_log() -> Vec<u8> {
76    let mut out = Vec::new();
77    let mut buf = [0u8; 128];
78    loop {
79        let read = crate::task::read_log_ring(&mut buf);
80        if read == 0 {
81            return out;
82        }
83        out.extend_from_slice(&buf[..read]);
84    }
85}

87#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
88fn rv_addi(rd: u32, imm: u32) -> u32 {
89    (imm << 20) | (rd << 7) | 0x13
90}

92#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
93fn map_test_page() -> Result<usize, &'static str> {
94    let mut frame_guard = FRAME_ALLOCATOR.lock();
95    let allocator = frame_guard.as_mut().ok_or("allocator unavailable")?;
96    let frame = allocator.allocate_frame().ok_or("frame unavailable")?;
97    let flags = paging::Flags::from_bits(
98        paging::Flags::VALID
99            | paging::Flags::READ
100            | paging::Flags::EXECUTE
101            | paging::Flags::USER
102            | paging::Flags::ACCESSED
103            | paging::Flags::DIRTY,
104    );
105    paging::map_page(allocator, ENTRY, frame, flags).map_err(|_| "map failed")?;
106    unsafe {
107        let base = frame as *mut u32;
108        *base.add(0) = rv_addi(11, MSG.len() as u32);
109        *base.add(1) = 0x00B0_0893;
110        *base.add(2) = 0x0000_0073;
111        *base.add(3) = 0x0000_0513;
112        *base.add(4) = 0x03C0_0893;
113        *base.add(5) = 0x0000_0073;
114        core::ptr::copy_nonoverlapping(MSG.as_ptr(), (frame + MSG_OFFSET) as *mut u8, MSG.len());
115    }
116    Ok(frame)
117}

119#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
120fn unmap_test_page(frame: usize) {
121    let _ = paging::unmap_page(ENTRY);
122    paging::tlb_flush_all();
123    if let Some(allocator) = FRAME_ALLOCATOR.lock().as_mut() {
124        allocator.deallocate_frame(frame);
125    }
126}

128#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
129pub fn self_test() -> Option<bool> {
130    let saved_next_tid = super::SCHEDULER
131        .lock()
132        .as_ref()
133        .map(|sched| sched.next_task_id);
134    let _ = drain_user_log();
135    let frame = match map_test_page() {
136        Ok(frame) => frame,
137        Err(err) => {
138            log::error!("[selftest] THREAD-ENTRY: FAIL — {}", err);
139            return Some(false);
140        }
141    };
142
143    cell_quota::register(CellId(TEST_CELL_ID), cell_quota::DEFAULT_QUOTA_BYTES);
144    insert_parent();
145    let spawned = handle_syscall(
146        PARENT_TID,
147        Syscall::Spawn {
148            entry: ENTRY,
149            arg: ENTRY + MSG_OFFSET,
150        },
151    );
152
153    let mut ok = true;
154    let mut spawned_tid = None;
155    match spawned {
156        Ok(tid) if tid != 0 => {
157            spawned_tid = Some(tid);
158            crate::task::yield_cpu();
159        }
160        other => {
161            ok = false;
162            log::error!("[selftest] THREAD-ENTRY: FAIL — spawn returned {:?}", other);
163        }
164    }
165
166    reap();
167    let log_bytes = drain_user_log();
168    if !log_bytes.windows(MSG.len()).any(|window| window == MSG) {
169        ok = false;
170        log::error!("[selftest] THREAD-ENTRY: FAIL — marker missing after yield");
171    }
172
173    if let Some(tid) = spawned_tid {
174        let still_live = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
175            sched.tasks.contains_key(&tid) || sched.zombies.iter().any(|task| task.id == tid)
176        });
177        if still_live {
178            ok = false;
179            log::error!("[selftest] THREAD-ENTRY: FAIL — tid {} never exited", tid);
180            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
181                sched.exit_task(tid, usize::MAX);
182            }
183            reap();
184        }
185    }
186
187    remove(PARENT_TID);
188    cell_quota::deregister(CellId(TEST_CELL_ID));
189    unmap_test_page(frame);
190    if let (Some(sched), Some(next_tid)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
191        sched.next_task_id = next_tid;
192    }
193
194    if ok {
195        log::info!("[selftest] THREAD-ENTRY: PASS (user entry reached arg)");
196    } else {
197        log::error!("[selftest] THREAD-ENTRY: FAIL");
198    }
199    Some(ok)
200}
