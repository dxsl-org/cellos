1//! Boot self-test for thread-stack quota accounting.
2//!
3//! A thread's kernel and user stacks are real memory held for the thread's whole
4//! life. If they are not charged to the cell that asked for the thread, a cell
5//! can grow its true footprint while every number the kernel reports about it
6//! stays still — and the only remaining bound is the thread count, which says
7//! nothing about memory.
8//!
9//! Two directions are proven here, because getting one right and the other wrong is
10//! worse than neither: a charge that is never released turns the quota into a slow
11//! leak that eventually refuses legitimate work, and a release without a charge
12//! makes the whole ledger decorative.
13//!
14//! Runs in the same single-hart window as the other task self-tests — after
15//! `task::init()` and before `smp::start_secondaries()` — so the real thread it
16//! spawns cannot be picked up by another hart between creation and teardown. The
17//! scheduler and the quota tables are left exactly as they were found.
18
19use super::syscall::{handle_syscall, Syscall, SyscallError};
20use super::tcb::Task;
21use crate::memory::cell_quota;
22use types::CellId;

24/// Synthetic tid outside any range the boot sequence has assigned yet.
25const PARENT_TID: usize = 9201;
26
27/// Quota slot to exercise. Must be `< MAX_CELLS` for the charge to be tracked at
28/// all (higher ids are uncapped by design), and is the last slot precisely because
29/// cell ids are handed out from `next_task_id` upward — nothing has reached it at
30/// this point in boot, and the test deregisters it before anything can.
31const QUOTA_CELL: u64 = (cell_quota::MAX_CELLS - 1) as u64;
32
33/// The parent incarnation's generation; its quota row is registered with it.
34const PARENT_GENERATION: u64 = 1;
35
36fn register_quota(memory: usize) {
37    cell_quota::register(
38        CellId(QUOTA_CELL),
39        PARENT_GENERATION,
40        cell_quota::Limits { memory, guest: 0 },
41    );
42}
43
44/// Bytes one thread occupies: a kernel stack with its guard frames, plus a user
45/// stack, whose guards are arena VA with no frame.
46fn stack_charge_bytes() -> usize {
47    let pages = crate::task::stack_pages_for("thread");
48    super::stack::Stack::kernel_reserved_bytes(pages)
49        + super::stack::Stack::user_reserved_bytes(pages)
50}
51
52fn insert_parent() {
53    let mut parent = alloc::boxed::Box::new(Task::new(
54        PARENT_TID,
55        CellId(QUOTA_CELL),
56        "selftest",
57        alloc::vec::Vec::new(),
58    ));
59    parent.cell_generation = PARENT_GENERATION;
60    parent.root_tid = PARENT_TID;
61    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
62        if (QUOTA_CELL as usize) < crate::memory::cell_quota::MAX_CELLS {
63            let owner =
64                abi::cell_owner::CellOwner::new(QUOTA_CELL, PARENT_GENERATION, PARENT_TID as u64);
65            sched.publish_live_cell_owner(owner);
66        }
67        #[cfg(feature = "p01-trace")]
68        crate::memory::heap::p01_owner::task_backing(&parent);
69        #[cfg(feature = "p01-trace")]
70        let _semantic = crate::memory::heap::p01_owner::creation(
71            None,
72            crate::memory::heap::p01_owner::SCHEDULER_SHARED,
73        );
74        sched.tasks.insert(PARENT_TID, parent);
75    }
76}

78fn remove(tid: usize) {
79    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
80        if let Some(task) = sched.tasks.remove(&tid) {
81            if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
82                let owner = abi::cell_owner::CellOwner::new(
83                    task.cell_id.0,
84                    task.cell_generation,
85                    task.root_tid as u64,
86                );
87                sched.clear_live_cell_owner_for_test(owner);
88            }
89        }
90    }
91    crate::sched::hart_local::ready::remove_from_all(tid);
92}

94/// Drop every reapable zombie OUTSIDE the scheduler lock, which is what actually
95/// returns a dead thread's stack frames to the allocator.
96fn reap() {
97    let dead = super::SCHEDULER
98        .lock()
99        .as_mut()
100        .map(|s| s.take_reapable_zombies())
101        .unwrap_or_default();
102    drop(dead);
103}

105/// Charge appears at spawn and is gone once the dead thread's stacks are freed.
106fn charge_then_release(expected: usize) -> bool {
107    register_quota(cell_quota::DEFAULT_QUOTA_BYTES);
108    insert_parent();
109
110    let mut ok = true;
111    let spawned = handle_syscall(
112        PARENT_TID,
113        Syscall::Spawn {
114            entry: 0x1000,
115            arg: 0,
116        },
117    );
118
119    match spawned {
120        Ok(tid) if tid != 0 => {
121            let charged = cell_quota::in_use(CellId(QUOTA_CELL));
122            if charged != expected {
123                ok = false;
124                log::error!(
125                    "[selftest] THREAD-QUOTA: FAIL — charged {} bytes, expected {}",
126                    charged,
127                    expected
128                );
129            }
130            // Death by the funnel every real death path uses, then the reap that
131            // frees the stacks: their runs refund the charge with their frames,
132            // and not before.
133            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
134                sched.exit_task(tid, 0);
135            }
136            let held = cell_quota::in_use(CellId(QUOTA_CELL));
137            if held != expected {
138                ok = false;
139                log::error!(
140                    "[selftest] THREAD-QUOTA: FAIL — {} bytes charged between exit and reap, expected {}",
141                    held,
142                    expected
143                );
144            }
145            reap();
146            let after = cell_quota::in_use(CellId(QUOTA_CELL));
147            if after != 0 {
148                ok = false;
149                log::error!(
150                    "[selftest] THREAD-QUOTA: FAIL — {} bytes still charged after reap",
151                    after
152                );
153            }
154        }
155        other => {
156            ok = false;
157            log::error!("[selftest] THREAD-QUOTA: FAIL — spawn returned {:?}", other);
158        }
159    }
160
161    remove(PARENT_TID);
162    cell_quota::deregister(CellId(QUOTA_CELL));
163    ok
164}

166/// A cell whose quota cannot absorb another thread's stacks is refused, and the refusal
167/// leaves no charge behind.
168fn refused_when_unaffordable() -> bool {
169    // One page of quota — below a stack by three orders of magnitude, so the
170    // refusal cannot be an artefact of a borderline comparison.
171    register_quota(crate::memory::paging::PAGE_SIZE);
172    insert_parent();
173
174    let mut ok = true;
175    let refused = handle_syscall(
176        PARENT_TID,
177        Syscall::Spawn {
178            entry: 0x1000,
179            arg: 0,
180        },
181    );
182    if !matches!(refused, Err(SyscallError::TryAgain)) {
183        ok = false;
184        log::error!(
185            "[selftest] THREAD-QUOTA: FAIL — unaffordable spawn returned {:?}, expected TryAgain",
186            refused
187        );
188        if let Ok(stray) = refused {
189            remove(stray);
190        }
191    }
192    let leaked = cell_quota::in_use(CellId(QUOTA_CELL));
193    if leaked != 0 {
194        ok = false;
195        log::error!(
196            "[selftest] THREAD-QUOTA: FAIL — refused spawn left {} bytes charged",
197            leaked
198        );
199    }
200
201    remove(PARENT_TID);
202    cell_quota::deregister(CellId(QUOTA_CELL));
203    ok
204}

206/// Returns true iff a thread stack is charged, released, and enforced as specified.
207/// Logs a decisive serial line.
208///
209/// Transparent to the boot sequence: the successful spawn advances `next_task_id`,
210/// so the counter is snapshotted on entry and restored on exit — the first real cell
211/// still gets the tid it would have without this test, keeping cell-id assignment
212/// stable whether or not the test is compiled in.
213pub fn self_test() -> bool {
214    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
215
216    let expected = stack_charge_bytes();
217    let ok = charge_then_release(expected) & refused_when_unaffordable();
218
219    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
220        sched.next_task_id = n;
221    }
222
223    if ok {
224        log::info!("[selftest] THREAD-QUOTA: PASS (charged, released, enforced)");
225    } else {
226        log::error!("[selftest] THREAD-QUOTA: FAIL");
227    }
228    ok
229}

231/// Thread spawn/exit rounds in [`stack_arena_self_test`].
232#[cfg(all(feature = "test-hooks", cellos_wide))]
233const ARENA_ROUNDS: usize = 512;
234
235/// How long [`stack_arena_self_test`] lets the frame-run reaper return slots
236/// still quarantined behind an unconfirmed flush.
237#[cfg(all(feature = "test-hooks", cellos_wide))]
238const ARENA_DRAIN_NS: u64 = 2_000_000_000;
239
240/// Returns true iff every thread's user stack maps from the stack arena (never
241/// at its physical address) and none of the slots `ARENA_ROUNDS`
242/// spawn/exit/reap rounds took (random under randomised slot order) is still
243/// held once the frame-run reaper has drained: a slot whose run waited behind
244/// an unconfirmed flush returns only when the reaper frees it, and a slot held
245/// by a live task's user stack (another hart spawning meanwhile) is not a
246/// leak. Slots the test never took are not compared. Logs
247/// `[selftest] STACK-ARENA`. Leaves `next_task_id` as found, as [`self_test`].
248#[cfg(all(feature = "test-hooks", cellos_wide))]
249pub fn stack_arena_self_test() -> bool {
250    use crate::loader::va_alloc;
251
252    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
253    register_quota(cell_quota::DEFAULT_QUOTA_BYTES);
254    insert_parent();
255
256    let mut taken = alloc::collections::BTreeSet::new();
257    let mut failure = None;
258    for round in 0..ARENA_ROUNDS {
259        let tid = match handle_syscall(
260            PARENT_TID,
261            Syscall::Spawn {
262                entry: 0x1000,
263                arg: 0,
264            },
265        ) {
266            Ok(tid) if tid != 0 => tid,
267            other => {
268                failure = Some(alloc::format!("round {round}: spawn returned {other:?}"));
269                break;
270            }
271        };
272        let placed = super::SCHEDULER.lock().as_mut().and_then(|sched| {
273            let stack = sched.tasks.get(&tid)?.user_stack.as_ref()?;
274            let base = stack.base;
275            let in_arena = va_alloc::in_stack_arena(base);
276            let off_identity = stack.usable_phys() != Some(stack.usable_start());
277            sched.exit_task(tid, 0);
278            Some((base, in_arena && off_identity))
279        });
280        reap();
281        match placed {
282            Some((base, true)) => {
283                taken.insert(base);
284            }
285            _ => {
286                failure = Some(alloc::format!(
287                    "round {round}: user stack placed {placed:?}"
288                ));
289                break;
290            }
291        }
292    }
293
294    remove(PARENT_TID);
295    cell_quota::deregister(CellId(QUOTA_CELL));
296    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
297        sched.next_task_id = n;
298    }
299
300    // A taken slot still held with no live user stack on it, once drained.
301    let leaked = || {
302        let live: alloc::collections::BTreeSet<usize> = super::SCHEDULER
303            .lock()
304            .as_ref()
305            .map(|sched| {
306                sched
307                    .tasks
308                    .values()
309                    .filter_map(|task| task.user_stack.as_ref().map(|stack| stack.base))
310                    .collect()
311            })
312            .unwrap_or_default();
313        taken
314            .iter()
315            .filter(|base| va_alloc::stack_slot_held(**base) && !live.contains(*base))
316            .count()
317    };
318    if failure.is_none() {
319        let start = crate::arch::clock::monotonic_ns();
320        loop {
321            crate::memory::frame_run::reap();
322            let still = leaked();
323            if still == 0 {
324                break;
325            }
326            match (start, crate::arch::clock::monotonic_ns()) {
327                (Some(start), Some(now)) if now.saturating_sub(start) < ARENA_DRAIN_NS => {
328                    core::hint::spin_loop()
329                }
330                _ => {
331                    failure = Some(alloc::format!(
332                        "{still} of {} slots taken still held after the reaper drained",
333                        taken.len()
334                    ));
335                    break;
336                }
337            }
338        }
339    }
340    match failure {
341        None => {
342            log::info!(
343                "[selftest] STACK-ARENA: PASS ({ARENA_ROUNDS} thread spawn/exit rounds, no slot leaked)"
344            );
345            true
346        }
347        Some(why) => {
348            log::error!("[selftest] STACK-ARENA: FAIL — {why}");
349            false
350        }
351    }
352}

354/// Test hooks (x86_64): `[selftest] SMAP-GRANT-SLICE`. A real thread of a Sas
355/// cell (its user stack from the arena, mapped USER in the shared root)
356/// allocates a one-page grant and slices it with `size_out` on its own user
357/// stack, through the production `handle_syscall` path. The size must land in
358/// that slot, read back through the user-copy window; the slot is primed with
359/// a sentinel through the same window first, so a skipped write cannot pass.
360/// Under SMAP a raw kernel store to the slot is a kernel `#PF` (fatal), so a
361/// PASS with `smap=on` proves the write goes through the window. The domain
362/// lane requires the marker when its boot line says `smap=on`. Leaves the
363/// scheduler, the quota tables and `next_task_id` as found.
364#[cfg(all(
365    feature = "test-hooks",
366    feature = "native-domains",
367    cellos_wide,
368    target_arch = "x86_64"
369))]
370pub fn smap_grant_slice_self_test() -> bool {
371    let smap = if crate::arch::user_access::smap_active() {
372        "on"
373    } else {
374        "off"
375    };
376    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
377    register_quota(cell_quota::DEFAULT_QUOTA_BYTES);
378    insert_parent();
379
380    let outcome = match handle_syscall(
381        PARENT_TID,
382        Syscall::Spawn {
383            entry: 0x1000,
384            arg: 0,
385        },
386    ) {
387        Ok(tid) if tid != 0 => {
388            let outcome = grant_slice_size_out_lands(tid);
389            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
390                sched.exit_task(tid, 0);
391            }
392            reap();
393            outcome
394        }
395        other => Err(alloc::format!("spawn returned {other:?}")),
396    };
397
398    remove(PARENT_TID);
399    cell_quota::deregister(CellId(QUOTA_CELL));
400    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
401        sched.next_task_id = n;
402    }
403    match outcome {
404        Ok(()) => {
405            log::info!(
406                "[selftest] SMAP-GRANT-SLICE: PASS (size_out landed on the user stack, smap={smap})"
407            );
408            true
409        }
410        Err(why) => {
411            log::error!("[selftest] SMAP-GRANT-SLICE: FAIL (smap={smap}) — {why}");
412            false
413        }
414    }
415}

417/// One `GrantAlloc` / `GrantSlice(size_out)` / `GrantFree` round by `tid`,
418/// with `size_out` at the top word of its user stack.
419#[cfg(all(
420    feature = "test-hooks",
421    feature = "native-domains",
422    cellos_wide,
423    target_arch = "x86_64"
424))]
425fn grant_slice_size_out_lands(tid: usize) -> Result<(), alloc::string::String> {
426    use alloc::format;
427    use core::mem::size_of;
428    const SENTINEL: usize = 0x5a5a_5a5a_5a5a_5a5a;
429    let size = crate::memory::paging::PAGE_SIZE;
430
431    let (slot, view) = super::SCHEDULER
432        .lock()
433        .as_ref()
434        .and_then(|sched| {
435            let task = sched.tasks.get(&tid)?;
436            let stack = task.user_stack.as_ref()?;
437            Some((
438                stack.top - size_of::<usize>(),
439                super::copy_glue::TaskCopyView::of(task),
440            ))
441        })
442        .ok_or_else(|| format!("thread {tid} has no user stack"))?;
443    if !view.is_sas() {
444        return Err(format!("thread {tid} is not a Sas task"));
445    }
446    view.write_bytes(slot, &SENTINEL.to_ne_bytes())
447        .map_err(|()| format!("priming slot {slot:#x} through the window failed"))?;
448    let grant_id = match handle_syscall(tid, Syscall::GrantAlloc { size }) {
449        Ok(id) if id != 0 => id,
450        other => return Err(format!("GrantAlloc returned {other:?}")),
451    };
452    let sliced = handle_syscall(
453        tid,
454        Syscall::GrantSlice {
455            grant_id,
456            size_out_ptr: slot,
457            expected_owner_tid: 0,
458            flags: 0,
459        },
460    );
461    let mut landed = [0u8; size_of::<usize>()];
462    let read = view.read_into(slot, &mut landed);
463    let freed = handle_syscall(tid, Syscall::GrantFree { grant_id });
464    if !matches!(sliced, Ok(base) if base == grant_id) {
465        return Err(format!(
466            "GrantSlice returned {sliced:?}, expected {grant_id:#x}"
467        ));
468    }
469    read.map_err(|()| format!("reading slot {slot:#x} back through the window failed"))?;
470    let value = usize::from_ne_bytes(landed);
471    if value != size {
472        return Err(format!("size_out holds {value:#x}, expected {size:#x}"));
473    }
474    freed.map_err(|error| format!("GrantFree returned {error:?}"))?;
475    Ok(())
