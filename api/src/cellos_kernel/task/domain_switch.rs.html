1//! Scheduler-owned root transition plan.  This module is RV64-only by construction.
2use super::{
3    hart_local,
4    tcb::{Task, TaskAddressSpace},
5};
6use crate::hal::arch::Context;
7use crate::memory::address_space::AddressSpace;
8use alloc::sync::Arc;
9
10#[derive(Clone)]
11pub(crate) struct DomainRef(Arc<AddressSpace>);
12
13impl DomainRef {
14    /// No liveness re-check here: the execution pin acquired in pick_next_local
15    /// makes a later retire() legal and drain-safe, so a pinned task's plan
16    /// MUST still derive Activate instead of diverting to the safe root under
17    /// KERNEL_ROOT with none of its mappings.
18    fn from_task(task: &Task) -> Option<Self> {
19        match &task.address_space {
20            TaskAddressSpace::Sas => None,
21            TaskAddressSpace::Domain(space) => Some(Self(Arc::clone(space))),
22        }
23    }
24
25    pub(crate) fn tuple(&self) -> (u64, u64) {
26        (self.0.identity().raw(), self.0.generation())
27    }
28
29    #[allow(dead_code)]
30    pub(crate) fn from_address_space(space: &Arc<AddressSpace>) -> Self {
31        Self(Arc::clone(space))
32    }
33
34    #[allow(dead_code)]
35    pub(crate) fn address_space(&self) -> &Arc<AddressSpace> {
36        &self.0
37    }
38}

40/// A root decision made under `SCHEDULER`, carried to the raw-switch boundary.
41/// No assembly code is allowed to rediscover task or root ownership.
42pub(crate) struct SwitchPlan {
43    pub outgoing: *mut Context,
44    pub incoming: *const Context,
45    transition: DomainTransition,
46}
47
48#[derive(Clone)]
49enum DomainTransition {
50    SasToSas,
51    Activate(DomainRef),
52    SameDomain,
53    ToSafeRoot,
54}
55
56impl SwitchPlan {
57    pub(crate) fn new(
58        outgoing: *mut Context,
59        incoming: *const Context,
60        task: Option<&Task>,
61    ) -> Option<Self> {
62        // No liveness gate here: the execution pin was already acquired in
63        // pick_next_local's filter, before the task was dequeued or attributed,
64        // so a plan that reaches this point MUST proceed — returning `None`
65        // would strand a selected task. A retirement flipping the pinned root
66        // Dying after acquisition is legal and drain-safe.
67        let transition = match task.and_then(DomainRef::from_task) {
68            Some(domain) if hart_local::current_domain() == domain.tuple() => {
69                DomainTransition::SameDomain
70            }
71            Some(domain) => {
72                hart_local::advance_execution_pin(Some(Arc::clone(&domain.0)));
73                DomainTransition::Activate(domain)
74            }
75            None if hart_local::current_domain().0 == 0 => DomainTransition::SasToSas,
76            None => {
77                // Leaving the pinned private root: hand its release to the
78                // safe-root completion hook alongside outgoing attribution.
79                hart_local::advance_execution_pin(None);
80                DomainTransition::ToSafeRoot
81            }
82        };
83        Some(Self {
84            outgoing,
85            incoming,
86            transition,
87        })
88    }
89
90    /// Return the root switch that assembly must issue after saving `outgoing`.
91    /// A zero PPN is the explicit no-write SAS/same-domain fast path.
92    pub(crate) fn root_switch(&self) -> (usize, usize) {
93        match &self.transition {
94            DomainTransition::Activate(domain) => {
95                let (id, generation) = domain.tuple();
96                let root = (domain.0.root_ppn(), domain.0.asid());
97                // The execution pin was acquired at selection time under
98                // scheduler-stable state; programming the dead root's SATP here
99                // is exactly what begin_execution's recheck prevents.
100                hart_local::set_current_domain(id, generation);
101                crate::hal::domain::observe_switch_activation();
102                #[cfg(feature = "test-hooks")]
103                log::info!(
104                    "S22-RV64-SWITCH: PASS harts={}",
105                    super::smp::online_hart_count()
106                );
107                root
108            }
109            DomainTransition::ToSafeRoot => {
110                let root = crate::memory::paging::KERNEL_ROOT
111                    .lock()
112                    .expect("native domain requires SAS root");
113                hart_local::mark_safe_root_pending();
114                crate::hal::domain::observe_switch_activation();
115                (root >> 12, 0)
116            }
117            DomainTransition::SasToSas | DomainTransition::SameDomain => (0, 0),
118        }
119    }
120
121    #[cfg(feature = "test-hooks")]
122    pub(crate) fn is_sas_fast_path(&self) -> bool {
123        matches!(
124            self.transition,
125            DomainTransition::SasToSas | DomainTransition::SameDomain
126        )
127    }
128}
