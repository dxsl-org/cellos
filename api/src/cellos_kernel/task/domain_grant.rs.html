1//! CPU-only domain grants between two private roots.
2//!
3//! One kernel-owned record per grant carries the owner root, the receiver root
4//! (`Arc<AddressSpace>`, retained while the receiver PTE is published), the
5//! exact owner/receiver identity and virtual range, the rights copied into the
6//! receiver PTE, and a `Live → Revoking → Revoked` state. The grant row's
7//! `memory::region::Region` owns the backing as one `FrameRun`, and a revoker
8//! releases it only once the record is `Revoked`: revocation removes the
9//! receiver PTE, performs the synchronous CPU shootdown, removes the owner PTE,
10//! and only then may the frames leave the grant table.
11//!
12//! A grant that cannot observe safe-root quiescence (the receiver root still
13//! current on a hart, or the remote-fence transport unacknowledged) stays
14//! `Revoking`: its frames are retained and no entry point may publish a new
15//! mapping over the old one.
16//!
17//! The state machine is architecture-independent: the owner/receiver invalidation
18//! goes through `AddressSpace`, whose private-root leaves are non-global on
19//! RV64, AArch64 and x86_64, so no step here is target-specific. The module is
20//! compiled on the targets whose lifecycle is open (`cellos_domain_grants`:
21//! `native-domains` on RV64 and x86_64, and AArch64 test images — see
22//! `super::syscall::domain_grant_lifecycle_supported`).
23//!
24//! The record is owned by the `PAGE_GRANT_TABLE`/`REG_GRANT_TABLE` rows in
25//! `super::syscall`; this module is the state machine they embed. Publication
26//! and revocation are ordered by [`GrantLifecycle`] (lock order and the
27//! no-mapping-after-revoke invariant are documented there).
28

52/// Test-hooks: `(present, user)` for a domain owner's backing in the SAS/global
53/// root, observed where each party would reach it.
54///
55/// `present` is read at the kernel's own alias of the frame (`phys_to_virt`:
56/// the identity map on RV64/AArch64, the HHDM on x86_64, where RAM has no
57/// identity alias), and `user` at the grant address itself, the VA a SAS cell
58/// would dereference. The backing must stay reachable by the kernel and never
59/// become USER in the shared root. Shared with `super::grant_gate_selftest`.
60#[cfg(feature = "test-hooks")]
61pub(crate) fn sas_root_state(base: PhysAddr) -> (bool, bool) {
62    let alias = crate::memory::frame::phys_to_virt(base);
63    let (present, _) = crate::memory::paging::mapping_state(alias);
64    let (_, user) = crate::memory::paging::mapping_state(base);
65    (present, user)
66}

68#[derive(Clone, Copy, Debug, Eq, PartialEq)]
69pub(crate) enum DomainGrantError {
70    /// The requested mapping cannot be represented (unaligned, overlapping an
71    /// existing receiver mapping, or naming a non-live root).
72    InvalidMapping,
73    /// The record is not `Live`; a `Revoking` record refuses new slices/shares.
74    NotLive,
75    /// The receiver PTE could not be published or removed.
76    Mapping(AddressSpaceError),
77    /// Unmap succeeded but the root/tag invalidation is not acknowledged. The
78    /// record stays `Revoking` with its frames retained for an idempotent retry.
79    AwaitingSafeRoot,
80    /// The requested rights are not enforceable on an ordinary domain page.
81    UnsupportedRights,
82}

84/// The receiver half of a domain grant.
85pub(crate) struct DomainReceiver {
86    pub(crate) root: Arc<AddressSpace>,
87    pub(crate) cell: CellId,
88    pub(crate) generation: u64,
89    pub(crate) base: VAddr,
90    pub(crate) size: usize,
91    pub(crate) rights: Flags,

96/// Resolve the exact PTE rights for a domain pair, or `None` when the requested
97/// permission has no enforceable page representation.
98///
99/// `ReadOnly` → R+NX, `ReadWrite` → RW+NX. A genuine write-only page is not
100/// representable on any supported target, so it is refused rather than widened
101/// to RW.
102pub(crate) fn rights_for(perm: GrantPerm) -> Option<Flags> {
103    match perm {
104        GrantPerm::ReadOnly => Some(Flags::from_bits(Flags::READ)),
105        GrantPerm::ReadWrite => Some(Flags::from_bits(Flags::READ | Flags::WRITE)),
106        GrantPerm::WriteOnly => None,
107    }
108}

147    /// Is the receiver mapping already exactly `(cell, generation, root, rights)`?
148    /// A redundant re-share must not tear down a live mapping.
149    pub(crate) fn receiver_matches(
150        &self,
151        cell: CellId,
152        generation: u64,
153        root: &Arc<AddressSpace>,
154        rights: Flags,
155    ) -> bool {
156        self.receiver.lock().as_ref().is_some_and(|receiver| {
157            !receiver.drained
158                && receiver.cell == cell
159                && receiver.generation == generation
160                && Arc::ptr_eq(&receiver.root, root)
161                && receiver.rights.bits() == rights.bits()
162        })
163    }
164
165    /// Publish `rights` for `root` at the grant's physical base, replacing any
166    /// previous receiver mapping.
167    ///
168    /// The tuple `(cell, generation, root, rights)` is compared first: only a
169    /// change revokes the old PTE. Every page is mapped transactionally; on any
170    /// failure the pages already mapped are unmapped (with acknowledgement) and
171    /// nothing is published.
172    ///
173    /// The `Live` check, the map loop and the receiver record run inside one
174    /// hold of the lifecycle gate, so a concurrent [`Self::revoke`] either
175    /// drains this mapping or makes this call map nothing.
176    pub(crate) fn publish(
177        &self,
178        root: &Arc<AddressSpace>,
179        cell: CellId,
180        generation: u64,
181        rights: Flags,
182    ) -> Result<(), DomainGrantError> {
183        #[cfg(feature = "test-hooks")]
184        let held = gate_hold::Hold::default();
185        let result = self.lifecycle.while_live(DomainGrantError::NotLive, || {
186            #[cfg(feature = "test-hooks")]
187            let _timer = held.time();
188            self.publish_live(root, cell, generation, rights)
189        });
190        #[cfg(feature = "test-hooks")]
191        held.record(&gate_hold::PUBLISH_MAX_NS);
192        result
193    }
194
195    /// Body of [`Self::publish`]; the caller holds the gate and saw `Live`.
196    fn publish_live(
197        &self,
198        root: &Arc<AddressSpace>,
199        cell: CellId,
200        generation: u64,
201        rights: Flags,
202    ) -> Result<(), DomainGrantError> {
203        if rights.bits() & Flags::EXECUTE != 0 || rights.bits() & Flags::READ == 0 {
204            return Err(DomainGrantError::UnsupportedRights);
205        }
206        if !root.is_live() {
207            return Err(DomainGrantError::InvalidMapping);
208        }
209        if self.receiver_matches(cell, generation, root, rights) {
210            return Ok(());
211        }
212        // A different tuple (re-share, target change or same-recipient
213        // downgrade) revokes the old receiver PTE and shoots it down before the
214        // new one is published.
215        self.drain_receiver_gated()?;
216
217        let base = self.owner_base;
218        let n_pages = self.owner_size.div_ceil(PAGE_SIZE).max(1);
219        let mut mapped = 0usize;
220        for index in 0..n_pages {
221            let va = base + index * PAGE_SIZE;
222            match root.map_grant_page(va, va, rights) {
223                Ok(()) => {
224                    // The hart that writes a PTE must invalidate before any
225                    // translation of it is relied on; the receiver root is
226                    // inactive on this hart, so the local page flush is the
227                    // whole requirement here.
228                    crate::hal::paging::flush_tlb_page(va);
229                    mapped += 1;
230                }
231                Err(error) => {
232                    for undo in 0..mapped {
233                        let _ = root.unmap_grant_page(base + undo * PAGE_SIZE);
234                    }
235                    // The pages above already queued their tag invalidation if it
236                    // could not be confirmed; a second blocking wait here would buy
237                    // nothing, so the rollback only re-probes once.
238                    let _ = crate::memory::tlb_shootdown::confirm_tag_invalidation(root.asid());
239                    return Err(DomainGrantError::Mapping(error));
240                }
241            }
242        }
243        *self.receiver.lock() = Some(DomainReceiver {
244            root: Arc::clone(root),
245            cell,
246            generation,
247            base,
248            size: self.owner_size,
249            rights,
250            drained: false,
251        });
252        Ok(())
253    }

255    /// Remove the receiver PTE and confirm the invalidation. `Ok(())` means the
256    /// receiver half is drained and its retained root may be dropped.
257    pub(crate) fn drain_receiver(&self) -> Result<(), DomainGrantError> {
258        #[cfg(feature = "test-hooks")]
259        let held = gate_hold::Hold::default();
260        let result = self.lifecycle.serialized(|| {
261            #[cfg(feature = "test-hooks")]
262            let _timer = held.time();
263            self.drain_receiver_gated()
264        });
265        #[cfg(feature = "test-hooks")]
266        held.record(&gate_hold::DRAIN_MAX_NS);
267        result
268    }
269
270    /// [`Self::drain_receiver`] for a caller already holding the gate.
271    fn drain_receiver_gated(&self) -> Result<(), DomainGrantError> {
272        let Some((root, base, size)) = ({
273            let guard = self.receiver.lock();
274            guard
275                .as_ref()
276                .filter(|receiver| !receiver.drained)
277                .map(|receiver| (Arc::clone(&receiver.root), receiver.base, receiver.size))
278        }) else {
279            return Ok(());
280        };
281        let n_pages = size.div_ceil(PAGE_SIZE).max(1);
282        for index in 0..n_pages {
283            match root.unmap_grant_page(base + index * PAGE_SIZE) {
284                Ok(()) | Err(AddressSpaceError::NotFound) => {}
285                Err(_) => return Err(DomainGrantError::AwaitingSafeRoot),
286            }
287        }
288        // The memory layer recorded the tag invalidation it could not confirm
289        // synchronously; its reaper completes it without blocking anyone, so this
290        // path asks whether it is confirmed instead of waiting for it. `Err` keeps
291        // exactly its old meaning — the receiver half is not drained and the
292        // record stays retryable — it just no longer costs a retry budget.
293        if crate::memory::deferred_release::tag_invalidation_unconfirmed(root.asid()) {
294            return Err(DomainGrantError::AwaitingSafeRoot);
295        }
296        // Confirmed: the addresses the failed attempts reserved may be published
297        // again, and the frames behind them are already released.
298        for index in 0..n_pages {
299            root.release_reserved_address(base + index * PAGE_SIZE);
300        }
301        if let Some(receiver) = self.receiver.lock().as_mut() {
302            receiver.drained = true;
303        }
304        Ok(())
305    }
306
307    /// Is an undrained half waiting on its root's tag invalidation? While it
308    /// is, a revoke retry cannot complete that half, and would only scan its
309    /// pages again: a housekeeping retry skips the record until the reaper
310    /// confirms the tag.
311    pub(crate) fn awaits_tag(&self) -> bool {
312        let receiver_tag = self
313            .receiver
314            .lock()
315            .as_ref()
316            .filter(|receiver| !receiver.drained)
317            .map(|receiver| receiver.root.asid());
318        let owner_tag = (!self.owner_drained.load(Ordering::Acquire))
319            .then(|| self.owner.address_space().asid());
320        [receiver_tag, owner_tag]
321            .into_iter()
322            .flatten()
323            .any(crate::memory::deferred_release::tag_invalidation_unconfirmed)
324    }
325
326    /// Remove the owner PTE and confirm the invalidation.
327    fn drain_owner(&self) -> Result<(), DomainGrantError> {
328        if self.owner_drained.load(Ordering::Acquire) {
329            return Ok(());
330        }
331        let root = Arc::clone(self.owner.address_space());
332        let n_pages = self.owner_size.div_ceil(PAGE_SIZE).max(1);
333        for index in 0..n_pages {
334            match root.unmap_grant_page(self.owner_base + index * PAGE_SIZE) {
335                Ok(()) | Err(AddressSpaceError::NotFound) => {}
336                Err(_) => return Err(DomainGrantError::AwaitingSafeRoot),
337            }
338        }
339        if crate::memory::deferred_release::tag_invalidation_unconfirmed(root.asid()) {
340            return Err(DomainGrantError::AwaitingSafeRoot);
341        }
342        for index in 0..n_pages {
343            root.release_reserved_address(self.owner_base + index * PAGE_SIZE);
344        }
345        self.owner_drained.store(true, Ordering::Release);
346        Ok(())
347    }

349    /// Linearize revocation before PTE removal and drive both halves to
350    /// quiescence. Idempotent: a record left `Revoking` by an unacknowledged
351    /// invalidation is retried with the same effect.
352    pub(crate) fn revoke(&self) -> Result<(), DomainGrantError> {
353        // Any publication that saw `Live` has recorded its receiver by the time
354        // this returns, so the drain below sees it; later ones see `Revoking`.
355        if self.lifecycle.begin_revoke() == DomainGrantState::Revoked {
356            return Ok(());
357        }
358        // Test-hooks CONTROL (`GuardedCheck::RevokeDrainsReceiver`): the
359        // revoke-while-mapped fixture must catch a revoke that skips this.
360        #[cfg(feature = "test-hooks")]
361        let drain = !super::fixture_controls::disabled(
362            super::fixture_controls::GuardedCheck::RevokeDrainsReceiver,
363        );
364        #[cfg(not(feature = "test-hooks"))]
365        let drain = true;
366        if drain {
367            self.drain_receiver()?;
368        }
369        self.drain_owner()?;
370        self.lifecycle.finish_revoke();
371        Ok(())
372    }
373}

375/// Test-hooks measurement of how long the lifecycle gate is held (Phase 15,
376/// risk R4).
377///
378/// The gate is a `Spinlock`, so its holder runs with interrupts masked, and a
379/// drain inside it includes the bounded wait for every online CPU to
380/// acknowledge the tag invalidation. `PUBLISH_MAX_NS` is the longest publish
381/// hold — the one taken under a grant-table lock — and `DRAIN_MAX_NS` the
382/// longest drain hold, which runs under no table lock (revoke, receiver death,
383/// and a share's pre-drain, see `super::syscall`). The x86 lane reports both
384/// rather than assuming either is short.
385#[cfg(feature = "test-hooks")]
386pub(crate) mod gate_hold {
387    use core::cell::Cell;
388    use core::sync::atomic::{AtomicU64, Ordering};
389
390    /// Longest single publish hold of any record's gate this boot, in ns.
391    pub(crate) static PUBLISH_MAX_NS: AtomicU64 = AtomicU64::new(0);
392    /// Longest single drain hold of any record's gate this boot, in ns.
393    pub(crate) static DRAIN_MAX_NS: AtomicU64 = AtomicU64::new(0);
394
395    fn now() -> u64 {
396        crate::arch::clock::monotonic_ns().unwrap_or(0)
397    }
398
399    /// One gate hold, measured inside the gate and recorded after release.
400    #[derive(Default)]
401    pub(super) struct Hold(Cell<u64>);
402
403    impl Hold {
404        /// Start timing; the returned guard stops when the gate's body ends.
405        pub(super) fn time(&self) -> Timer<'_> {
406            Timer {
407                hold: self,
408                start: now(),
409            }
410        }
411
412        /// Fold this hold into `max`; called once the gate is released.
413        pub(super) fn record(&self, max: &AtomicU64) {
414            max.fetch_max(self.0.get(), Ordering::AcqRel);
415        }
416    }
417
418    pub(super) struct Timer<'a> {
419        hold: &'a Hold,
420        start: u64,
421    }
422
423    impl Drop for Timer<'_> {
424        fn drop(&mut self) {
425            self.hold.0.set(now().saturating_sub(self.start));
426        }
427    }
428}

430/// Boot self-test: the table path is the single implementation, so this drives
431/// the production `handle_syscall` entry points with two real
432/// `TaskAddressSpace::Domain` tasks and observes the ABI result of each step.
433///
434/// Emits the established `{ARCH_TAG}-GRANT-REVOKE` terminal plus a property
435/// marker per assertion. Every synthetic task and grant is torn down before it
436/// returns.
437///
438/// It witnesses rights, transactional publish, revoke and the deferred outcome
439/// on the arch whose lane runs it; it does **not** witness the two-cell public
440/// handoff or an address-classified receiver store fault — those belong to each
441/// lane's `grant-pair` cells (`cells/tests/tier2-grant-*`).
442#[cfg(feature = "test-hooks")]
443pub(crate) fn run_selftest() {
444    use super::syscall::{handle_syscall, Syscall};
445    use super::tcb::TaskAddressSpace;
446    use super::thread_cap_selftest::{insert, mk_task, remove};
447    use crate::memory::address_space::{AddressSpaceBuilder, MappingKind};
448
449    const OWNER_TID: usize = 9601;
450    const RECEIVER_TID: usize = 9602;
451    const DEAD_TID: usize = 9603;
452    const OWNER_CELL: u64 = (crate::memory::cell_quota::MAX_CELLS - 20) as u64;
453    const RECEIVER_CELL: u64 = (crate::memory::cell_quota::MAX_CELLS - 21) as u64;
454    const PAGE: usize = PAGE_SIZE;
455    const GRANT_SIZE: usize = PAGE;
456    /// A private-root VA for each fixture task, above the RAM a grant frame (VA =
457    /// PA) can come from on every lane's machine, x86_64 included.
458    const OWNER_VA: usize = 0x30_0400_0000;
459    const RECEIVER_VA: usize = 0x30_0800_0000;
460
461    fn space(va: usize) -> Option<Arc<AddressSpace>> {
462        let mut builder = AddressSpaceBuilder::new();
463        builder
464            .map_user_page(
465                va,
466                MappingKind::Private,
467                Flags::from_bits(Flags::READ | Flags::WRITE),
468            )
469            .ok()?;
470        builder.build().ok()
471    }
472
473    fn grant_alloc_size(tid: usize, size: usize) -> Option<usize> {
474        match handle_syscall(tid, Syscall::GrantAlloc { size }) {
475            Ok(base) if base != 0 => Some(base),
476            _ => None,
477        }
478    }
479
480    fn grant_alloc(tid: usize) -> Option<usize> {
481        grant_alloc_size(tid, GRANT_SIZE)
482    }
483
484    fn grant_slice(tid: usize, grant_id: usize) -> Option<usize> {
485        match handle_syscall(
486            tid,
487            Syscall::GrantSlice {
488                grant_id,
489                size_out_ptr: 0,
490                expected_owner_tid: 0,
491                flags: 0,
492            },
493        ) {
494            Ok(base) if base != usize::MAX => Some(base),
495            _ => None,
496        }
497    }
498
499    let (Some(owner_space), Some(receiver_space)) = (space(OWNER_VA), space(RECEIVER_VA)) else {
500        log::error!("{ARCH_TAG}-GRANT-REVOKE: FAIL fixture-setup");
501        return;
502    };
503
504    let mut owner_task = mk_task(OWNER_TID, OWNER_CELL);
505    owner_task.address_space = TaskAddressSpace::Domain(Arc::clone(&owner_space));
506    insert(owner_task);
507    let mut receiver_task = mk_task(RECEIVER_TID, RECEIVER_CELL);
508    receiver_task.address_space = TaskAddressSpace::Domain(Arc::clone(&receiver_space));
509    insert(receiver_task);
510
511    let mut ok = true;
512    let mut report = |marker: &str, property: bool| {
513        if property {
514            log::info!("{ARCH_TAG}-GRANT-REVOKE-{marker}: PASS");
515        } else {
516            log::error!("{ARCH_TAG}-GRANT-REVOKE-{marker}: FAIL");
517            ok = false;
518        }
519    };
520
521    // 1. Owner allocation: the pointer is published in the owner's private root
522    //    and the SAS/global root must stay supervisor-only.
523    let allocated = grant_alloc(OWNER_TID);
524    let owner_mapped = allocated.is_some_and(|base| {
525        let (present, user) = sas_root_state(base);
526        let ledger_owner = owner_space.ledger().into_iter().any(|entry| {
527            entry.virtual_address == base
528                && entry.kind == MappingKind::Grant
529                && entry.flags.bits() & Flags::WRITE != 0
530                && entry.flags.bits() & Flags::EXECUTE == 0
531        });
532        present && !user && ledger_owner
533    });
534    report("OWNER-MAPPED", owner_mapped);
535
536    // 2. The owner resolves its own grant without a second mapping.
537    let owner_slice = allocated.is_some_and(|base| grant_slice(OWNER_TID, base) == Some(base));
538    report("OWNER-SLICE", owner_slice);
539
540    // 3. Read-only slice: R+NX in the receiver root, no write bit.
541    let ro_readable = allocated.is_some_and(|base| {
542        let shared = handle_syscall(
543            OWNER_TID,
544            Syscall::GrantShare {
545                grant_id: base,
546                target_cell: RECEIVER_TID,
547                perm: GrantPerm::ReadOnly as usize,
548            },
549        )
550        .is_ok();
551        let sliced = grant_slice(RECEIVER_TID, base) == Some(base);
552        let ro_pte = receiver_space.ledger().into_iter().any(|entry| {
553            entry.virtual_address == base
554                && entry.kind == MappingKind::Grant
555                && entry.flags.bits() & Flags::READ != 0
556                && entry.flags.bits() & Flags::WRITE == 0
557                && entry.flags.bits() & Flags::EXECUTE == 0
558        });
559        shared && sliced && ro_pte
560    });
561    report("SLICE-RO", ro_readable);
562
563    // 4. Same-recipient upgrade RW→RO must republish: the new PTE is RW+NX and
564    //    the old read-only PTE is gone.
565    let rw_writable = allocated.is_some_and(|base| {
566        handle_syscall(
567            OWNER_TID,
568            Syscall::GrantShare {
569                grant_id: base,
570                target_cell: RECEIVER_TID,
571                perm: GrantPerm::ReadWrite as usize,
572            },
573        )
574        .is_ok()
575            && grant_slice(RECEIVER_TID, base) == Some(base)
576            && receiver_space.ledger().into_iter().any(|entry| {
577                entry.virtual_address == base
578                    && entry.kind == MappingKind::Grant
579                    && entry.flags.bits() & Flags::WRITE != 0
580                    && entry.flags.bits() & Flags::EXECUTE == 0
581            })
582    });
583    report("SLICE-RW", rw_writable);
584
585    // 5. Write-only has no ordinary-page representation: refused, never widened.
586    let write_only_refused = allocated.is_some_and(|base| {
587        handle_syscall(
588            OWNER_TID,
589            Syscall::GrantShare {
590                grant_id: base,
591                target_cell: RECEIVER_TID,
592                perm: GrantPerm::WriteOnly as usize,
593            },
594        )
595        .is_err()
596    });
597    report("WO-REFUSED", write_only_refused);
598
599    // 6. A non-domain peer cannot be named as a domain receiver.
600    let foreign_peer_refused = allocated.is_some_and(|base| {
601        handle_syscall(
602            OWNER_TID,
603            Syscall::GrantShare {
604                grant_id: base,
605                target_cell: 0,
606                perm: GrantPerm::ReadOnly as usize,
607            },
608        )
609        .is_err()
610    });
611    report("FOREIGN-PEER", foreign_peer_refused);
612
613    // 7. Revoke through GrantFree: both PTEs disappear, the receiver root no
614    //    longer resolves the address, and a re-slice is refused.
615    let revoked = allocated.is_some_and(|base| {
616        let freed = handle_syscall(OWNER_TID, Syscall::GrantFree { grant_id: base }).is_ok();
617        let receiver_gone = !receiver_space
618            .ledger()
619            .into_iter()
620            .any(|entry| entry.virtual_address == base);
621        let owner_gone = !owner_space
622            .ledger()
623            .into_iter()
624            .any(|entry| entry.virtual_address == base);
625        freed && receiver_gone && owner_gone && grant_slice(RECEIVER_TID, base).is_none()
626    });
627    report("REVOKE", revoked);
628
629    // 8. Frame reuse: an allocation handed the same frame must be zeroed before
630    //    it can be published, so no byte of the revoked grant survives.
631    let reuse_private = grant_alloc(OWNER_TID).is_some_and(|base| {
632        let alias = crate::memory::frame::phys_to_virt(base);
633        // SAFETY: `phys_to_virt` is the kernel's supervisor alias of the frame
634        // (identity on RV64/AArch64, HHDM on x86_64), present in this root.
635        let zeroed = (0..GRANT_SIZE).all(|offset| {
636            (unsafe { core::ptr::read_volatile((alias + offset) as *const u8) }) == 0
637        });
638        let _ = handle_syscall(OWNER_TID, Syscall::GrantFree { grant_id: base });
639        zeroed
640    });
641    report("FRAME-REUSE", reuse_private);
642
643    // 9. A multi-page receiver publish that fails on its second page must undo
644    //    the first: the owner's grant is untouched, no half-mapping is visible in
645    //    the receiver root, and the pre-existing private mapping survives.
646    let partial_map = grant_alloc_size(OWNER_TID, 2 * PAGE).is_some_and(|base| {
647        let occupied = receiver_space
648            .map_private_page(
649                base + PAGE,
650                MappingKind::Private,
651                Flags::from_bits(Flags::READ | Flags::WRITE),
652            )
653            .is_ok();
654        let shared = handle_syscall(
655            OWNER_TID,
656            Syscall::GrantShare {
657                grant_id: base,
658                target_cell: RECEIVER_TID,
659                perm: GrantPerm::ReadWrite as usize,
660            },
661        )
662        .is_err();
663        let no_partial = !receiver_space
664            .ledger()
665            .into_iter()
666            .any(|entry| entry.virtual_address == base && entry.kind == MappingKind::Grant);
667        let private_intact = receiver_space.ledger().into_iter().any(|entry| {
668            entry.virtual_address == base + PAGE && entry.kind == MappingKind::Private
669        });
670        let _ = handle_syscall(OWNER_TID, Syscall::GrantFree { grant_id: base });
671        occupied && shared && no_partial && private_intact
672    });
673    report("PARTIAL-MAP", partial_map);
674
675    // 10. A retired private root is not a capability: its owner is refused with
676    //     the alloc-safe sentinel, and a receiver naming it is refused.
677    let mut dead_task = mk_task(DEAD_TID, RECEIVER_CELL + 1);
678    dead_task.address_space = TaskAddressSpace::Domain(Arc::clone(&receiver_space));
679    insert(dead_task);
680    receiver_space.retire();
681    let dead_refused = grant_alloc(DEAD_TID).is_none()
682        && handle_syscall(
683            DEAD_TID,
684            Syscall::GrantSlice {
685                grant_id: 0,
686                size_out_ptr: 0,
687                expected_owner_tid: 0,
688                flags: 0,
689            },
690        )
691        .is_ok_and(|result| result == usize::MAX);
692    remove(DEAD_TID);
693    report("DEAD-ROOT", dead_refused);
694
695    // Teardown: the owner-side reaper drains and releases whatever the probe
696    // left behind, then the synthetic tasks go away.
697    super::syscall::reclaim_owned_grants(OWNER_TID);
698    super::syscall::reclaim_owned_grants(RECEIVER_TID);
699    remove(OWNER_TID);
700    remove(RECEIVER_TID);
701
702    if ok {
703        log::info!("{ARCH_TAG}-GRANT-REVOKE: PASS");
704        log::info!("{ARCH_TAG}-DMA-QUARANTINE: DENY");
705    } else {
706        log::error!("{ARCH_TAG}-GRANT-REVOKE: FAIL");
707    }
708}
