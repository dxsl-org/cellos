1//! Boot self-tests for two trust-model fixes (2026-07-13):
2//!   #7 — spawned threads inherit the parent cell's identity (CellId + CapSet +
3//!        syscall_allowlist + PKU domain), closing the `CellId(0)` quota escape.
4//!   #5 — `sys_cap_revoke` is honest about ambient authority: MMIO windows are
5//!        torn down (unmapped + released, `.agents/260712-1901` P01-P04) and the
6//!        victim is notified, while HYPERVISOR — which has no teardown path — is
7//!        still refused with `NotSupported`.
8//!
9//! Runs in the single-hart window AFTER `task::init()` but BEFORE
10//! `smp::start_secondaries()` (see `main.rs`), so a synthetic thread inserted into
11//! the boot hart's ready queue can never be picked up by another hart before this
12//! test removes it. All synthetic tasks are created here and torn down before the
13//! function returns — the scheduler is left exactly as it was found.
14//!
15//! Invariant proven, not assumed: after the fix the spawned thread's `cell_id`
16//! equals the parent's (not `CellId(0)`), which is what routes its allocations to
17//! the parent's quota (the charge path keys off `cell_id`; the cell-spawn path in
18//! `loader.rs` already relies on this identity).
19
20use super::cap::CapSet;
21use super::syscall::{handle_syscall, Syscall, SyscallError};
22use super::tcb::Task;
23use api::syscall::cap_mask as CM;
24use types::CellId;
25
26// Synthetic tids outside any range the boot sequence has assigned yet. Removed
27// before return, so they never collide with real cells.
28const PARENT_TID: usize = 9001;
29const TARGET_TID: usize = 9002;
30const CTRL_TID: usize = 9003;
31/// Target of the privileged-cap revoke (P05): holds pcie_driver/platform/supervisor.
32const DRIVER_TID: usize = 9004;
33/// A synthetic BAR window, registered only under `test-hooks` so the global
34/// BAR table keeps exactly what the ECAM scan put in it.
35#[cfg(feature = "test-hooks")]
36const DRIVER_BAR: usize = 0x5100_0000;
37/// A synthetic requester ID; nothing else claims it during the boot window.
38const DRIVER_BDF: u32 = 0x02_00_00;
39/// Separate cell for the thread-cap section, so its fillers cannot be mistaken
40/// for (or collide with) the identity-inheritance section's parent cell.
41const DOS_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 4) as u64;
42const DOS_FILLER_BASE: usize = 9100;
43
44// Distinctive sentinels so an inherited value is unambiguously the parent's.
45const TEST_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 2) as u64;
46// A restricted allowlist that still PERMITS the Spawn syscall (so the global
47// allowlist gate in handle_syscall lets the call through) yet differs from the
48// Task::new default of u64::MAX — bit 63 is unassigned (real syscalls use bits
49// ≤54), so clearing it is behaviourally inert but proves the field was inherited
50// rather than left at the permit-all default.
51const TEST_ALLOWLIST: u64 = !(1u64 << 63);
52const TEST_PKU_KEY: u8 = 2;
53const TEST_PKU_VALUE: u32 = 0xABCD_1234;
54
55/// Build a bare task with the given tid/cell and no caps. Only the fields a
56/// test reads/writes need to be meaningful; the task is never scheduled.
57///
58/// Shared with the sibling boot self-tests (`grant_reclaim_selftest`,
59/// `mmio_revoke_selftest`) so each keeps one fixture implementation.
60pub(crate) fn mk_task(tid: usize, cell: u64) -> alloc::boxed::Box<Task> {
61    let mut task = alloc::boxed::Box::new(Task::new(
62        tid,
63        CellId(cell),
64        "selftest",
65        alloc::vec::Vec::new(),
66    ));
67    task.cell_generation = 1;
68    task.root_tid = tid;
69    task
70}
71
72/// Insert a task into the scheduler map (no ready-queue push — never scheduled).
73pub(crate) fn insert(task: alloc::boxed::Box<Task>) {
74    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
75        if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
76            let owner = api::cell_owner::CellOwner::new(
77                task.cell_id.0,
78                task.cell_generation,
79                task.root_tid as u64,
80            );
81            sched.publish_live_cell_owner(owner);
82        }
83        sched.tasks.insert(task.id, task);
84    }
85}
86
87/// Remove a tid from the scheduler map AND every hart's ready queue.
88pub(crate) fn remove(tid: usize) {
89    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
90        if let Some(task) = sched.tasks.remove(&tid) {
91            if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
92                let owner = api::cell_owner::CellOwner::new(
93                    task.cell_id.0,
94                    task.cell_generation,
95                    task.root_tid as u64,
96                );
97                sched.clear_live_cell_owner_for_test(owner);
98            }
99        }
100    }
101    super::hart_local::ready::remove_from_all(tid);
102}

104/// Returns true iff both fixes behave as specified. Logs a decisive serial line.
105///
106/// Transparent to the boot sequence: the real thread spawn it performs advances
107/// `next_task_id`, so the counter is snapshotted on entry and restored on exit —
108/// the first real cell (Platform) still gets the tid it would have without this
109/// test, keeping CellId assignment stable whether or not the test is compiled in.
110pub fn self_test() -> bool {
111    let mut ok = true;
112
113    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
114
115    // ── #7: thread identity inheritance ────────────────────────────────────────
116    // Parent cell: a distinctive CellId, a restricted allowlist, a non-zero PKU
117    // domain, and one transferable cap (network) + SpawnCap (needed to spawn).
118    {
119        let mut parent = mk_task(PARENT_TID, TEST_CELL_ID);
120        parent.network_cap = Some(super::cap::NetworkCap::new());
121        parent.spawn_cap = Some(super::cap::SpawnCap::new());
122        parent.syscall_allowlist = TEST_ALLOWLIST;
123        parent.pku_key = TEST_PKU_KEY;
124        parent.pku_value = TEST_PKU_VALUE;
125        let parent_caps = CapSet::of_task(&parent);
126        insert(parent);
127
128        // Invoke the REAL Spawn path as if the parent called it. entry/arg are
129        // placeholders; the thread is removed before it can run.
130        let spawned = handle_syscall(
131            PARENT_TID,
132            Syscall::Spawn {
133                entry: 0x1000,
134                arg: 0,
135            },
136        );
137
138        match spawned {
139            Ok(thread_tid) if thread_tid != 0 => {
140                if let Some(sched) = super::SCHEDULER.lock().as_ref() {
141                    if let Some(t) = sched.tasks.get(&thread_tid) {
142                        let cell_ok = t.cell_id.0 == TEST_CELL_ID; // NOT CellId(0)
143                        let caps_ok = CapSet::of_task(t) == parent_caps;
144                        let allow_ok = t.syscall_allowlist == TEST_ALLOWLIST; // NOT u64::MAX
145                        let pku_ok = t.pku_key == TEST_PKU_KEY && t.pku_value == TEST_PKU_VALUE;
146                        if !(cell_ok && caps_ok && allow_ok && pku_ok) {
147                            ok = false;
148                            log::error!(
149                                "[selftest] THREAD-INHERIT: FAIL \
150                                cell={} caps={} allow={} pku={}",
151                                cell_ok,
152                                caps_ok,
153                                allow_ok,
154                                pku_ok
155                            );
156                        }
157                    } else {
158                        ok = false;
159                        log::error!("[selftest] THREAD-INHERIT: FAIL — thread tid absent");
160                    }
161                }
162                remove(thread_tid);
163            }
164            _ => {
165                ok = false;
166                log::error!(
167                    "[selftest] THREAD-INHERIT: FAIL — spawn returned {:?}",
168                    spawned
169                );
170            }
171        }
172        remove(PARENT_TID);
173    }
174
175    // ── #5: honest revoke (ambient bits torn down, hypervisor still refused) ────
176    {
177        // Caller holds SpawnCap (Gate 1). The target holds an MMIO device bit, an
178        // owned grant and a hypervisor cap. An MMIO window is ambient authority —
179        // a mapped window is reachable with no syscall on the access path — so
180        // revoking it must tear the window down, not just clear the field
181        // (`.agents/260712-1901` P01-P04 replaced P00's blanket refusal).
182        // HYPERVISOR has no teardown path and must still be refused.
183        let mut caller = mk_task(PARENT_TID, TEST_CELL_ID);
184        caller.spawn_cap = Some(super::cap::SpawnCap::new());
185        insert(caller);
186
187        let mut target = mk_task(TARGET_TID, 0x1111);
188        target.hypervisor_cap = Some(super::cap::HypervisorCap::new());
189        target.mmio_devices = crate::resource_registry::DEV_GPIO;
190        insert(target);
191
192        // (a) HYPERVISOR bit → NotSupported, cap untouched.
193        let r_hyp = handle_syscall(
194            PARENT_TID,
195            Syscall::CapRevoke {
196                target_tid: TARGET_TID,
197                cap_mask: CM::HYPERVISOR,
198            },
199        );
200        let hyp_refused = matches!(r_hyp, Err(SyscallError::NotSupported))
201            && if let Some(sched) = super::SCHEDULER.lock().as_ref() {
202                sched
203                    .tasks
204                    .get(&TARGET_TID)
205                    .map(|t| t.hypervisor_cap.is_some())
206                    .unwrap_or(false)
207            } else {
208                false
209            };
210        if !hyp_refused {
211            ok = false;
212            log::error!("[selftest] REVOKE-HYPERVISOR: FAIL r={:?}", r_hyp);
213        }
214
215        // The target owns a grant, and — where the board has an allowlisted
216        // window — an MMIO region of the class it is about to lose.
217        let owned_grant = match handle_syscall(TARGET_TID, Syscall::GrantAlloc { size: 4096 }) {
218            Ok(base) if base != 0 => Some(base),
219            _ => None,
220        };
221        let window = crate::resource_registry::allowed_windows()
222            .first()
223            .copied()
224            .and_then(|(base, len, class)| {
225                crate::resource_registry::request_mmio(CellId(0x1111), base, len, class)
226                    .is_ok()
227                    .then_some((base, len))
228            });
229
230        // (b) MMIO bits → Ok, window released, owned grant reclaimed, the MMIO
231        //     field cleared (the hypervisor cap untouched), and the victim
232        //     notified with `[0xAC, 0xF2, mask_le4]`.
233        let r_mmio = handle_syscall(
234            PARENT_TID,
235            Syscall::CapRevoke {
236                target_tid: TARGET_TID,
237                cap_mask: CM::MMIO_MASK,
238            },
239        );
240        let target_state = super::SCHEDULER.lock().as_ref().and_then(|sched| {
241            sched.tasks.get(&TARGET_TID).map(|t| {
242                let notified = t
243                    .pending_msgs
244                    .as_slice()
245                    .iter()
246                    .filter_map(|msg| msg.wire.as_ref())
247                    .any(|wire| {
248                        let payload = wire.as_slice();
249                        payload.len() == 6
250                            && payload[0] == 0xAC
251                            && payload[1] == 0xF2
252                            && u32::from_le_bytes([payload[2], payload[3], payload[4], payload[5]])
253                                == CM::MMIO_MASK
254                    });
255                (t.mmio_devices, t.hypervisor_cap.is_some(), notified)
256            })
257        });
258        let window_released = window
259            .map(|(base, _)| crate::resource_registry::lookup_mmio_owner(base).is_none())
260            .unwrap_or(true);
261        let grant_reclaimed = owned_grant
262            .map(|base| {
263                matches!(
264                    handle_syscall(TARGET_TID, Syscall::GrantFree { grant_id: base }),
265                    Err(SyscallError::PermissionDenied)
266                )
267            })
268            .unwrap_or(true);
269
270        let mmio_ok = r_mmio.is_ok()
271            && target_state.is_some_and(|(devices, hyp, notified)| devices == 0 && hyp && notified)
272            && window_released
273            && grant_reclaimed;
274        if !mmio_ok {
275            ok = false;
276            log::error!(
277                "[selftest] REVOKE-MMIO: FAIL r={:?} state={:?} window_released={} \
278                 grant_reclaimed={}",
279                r_mmio,
280                target_state,
281                window_released,
282                grant_reclaimed
283            );
284        }
285
286        // (c) Positive control: SPAWN is a lazy (syscall-gated) bit — revoke of a
287        // non-system target must still SUCCEED and clear the field.
288        let mut ctrl = mk_task(CTRL_TID, 0x2222);
289        ctrl.spawn_cap = Some(super::cap::SpawnCap::new());
290        insert(ctrl);
291        let r_spawn = handle_syscall(
292            PARENT_TID,
293            Syscall::CapRevoke {
294                target_tid: CTRL_TID,
295                cap_mask: CM::SPAWN,
296            },
297        );
298        let cleared = if let Some(sched) = super::SCHEDULER.lock().as_ref() {
299            sched
300                .tasks
301                .get(&CTRL_TID)
302                .map(|t| t.spawn_cap.is_none())
303                .unwrap_or(false)
304        } else {
305            false
306        };
307        if !(r_spawn.is_ok() && cleared) {
308            ok = false;
309            log::error!(
310                "[selftest] REVOKE-ALLOW: FAIL r={:?} cleared={}",
311                r_spawn,
312                cleared
313            );
314        }
315
316        // (d) Privileged caps are revocable end to end (`.agents/260712-1901`
317        //     P05): `pcie_driver` drains the DMA authority and releases the BDF,
318        //     `platform` releases the ECAM window, `supervisor` is a field clear.
319        let mut driver = mk_task(DRIVER_TID, 0x3333);
320        driver.pcie_driver_cap = Some(super::cap::PcieDriverCap::new());
321        driver.platform_cap = Some(super::cap::PlatformCap::new());
322        driver.supervisor_cap = Some(super::cap::SupervisorCap::new());
323        insert(driver);
324
325        let bdf_claimed = crate::resource_registry::claim_bdf_owner(DRIVER_BDF, DRIVER_TID);
326        // The BAR table is boot-discovered data with no removal path, so the
327        // synthetic window is registered only where the test hook exists.
328        #[cfg(feature = "test-hooks")]
329        let bar_claimed = crate::resource_registry::test_register_region(
330            CellId(0x3333),
331            DRIVER_BAR,
332            0x1000,
333            crate::resource_registry::DEV_PCIE as crate::resource_registry::MmioClass,
334        );
335
336        let r_priv = handle_syscall(
337            PARENT_TID,
338            Syscall::CapRevoke {
339                target_tid: DRIVER_TID,
340                cap_mask: CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR,
341            },
342        );
343        let priv_state = super::SCHEDULER.lock().as_ref().and_then(|sched| {
344            sched.tasks.get(&DRIVER_TID).map(|t| {
345                let notified = t
346                    .pending_msgs
347                    .as_slice()
348                    .iter()
349                    .filter_map(|msg| msg.wire.as_ref())
350                    .any(|wire| {
351                        let payload = wire.as_slice();
352                        payload.len() == 6
353                            && payload[0] == 0xAC
354                            && payload[1] == 0xF2
355                            && u32::from_le_bytes([payload[2], payload[3], payload[4], payload[5]])
356                                == CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR
357                    });
358                (
359                    t.pcie_driver_cap.is_none(),
360                    t.platform_cap.is_none(),
361                    t.supervisor_cap.is_none(),
362                    notified,
363                )
364            })
365        });
366        let bdf_released =
367            bdf_claimed && crate::resource_registry::owner_of_bdf(DRIVER_BDF).is_none();
368        #[cfg(feature = "test-hooks")]
369        let bar_released =
370            bar_claimed && crate::resource_registry::lookup_mmio_owner(DRIVER_BAR).is_none();
371        #[cfg(not(feature = "test-hooks"))]
372        let bar_released = true;
373
374        let priv_ok = r_priv.is_ok()
375            && priv_state.is_some_and(|(pcie, platform, supervisor, notified)| {
376                pcie && platform && supervisor && notified
377            })
378            && bdf_released
379            && bar_released;
380        if !priv_ok {
381            ok = false;
382            log::error!(
383                "[selftest] REVOKE-PRIVILEGED: FAIL r={:?} state={:?} bdf_released={} \
384                 bar_released={}",
385                r_priv,
386                priv_state,
387                bdf_released,
388                bar_released
389            );
390        }
391
392        remove(PARENT_TID);
393        remove(TARGET_TID);
394        remove(CTRL_TID);
395        remove(DRIVER_TID);
396    }
397
398    // ── thread-spawn DoS bound ─────────────────────────────────────────────────
399    // `Syscall::Spawn` is gated by the syscall allowlist but NOT by SpawnCap, and
400    // each thread costs a contiguous run of STACK_PAGES+1 frames. Before the cap an
401    // unprivileged cell could loop here until the allocator fragmented, at which
402    // point the scheduler's `.expect("OOM Stack")` panicked the kernel — never-die
403    // broken from userspace. The refusal must be an error the caller survives.
404    //
405    // The cell is filled with STACKLESS synthetic tasks: the cap counts live tasks
406    // sharing a CellId, so proving the bound costs 31 Task structs instead of 31
407    // real 65-frame stack allocations.
408    {
409        let mut parent = mk_task(PARENT_TID, DOS_CELL_ID);
410        parent.spawn_cap = Some(super::cap::SpawnCap::new());
411        insert(parent);
412
413        let filler: alloc::vec::Vec<usize> = (0..super::scheduler::MAX_THREADS_PER_CELL - 1)
414            .map(|i| DOS_FILLER_BASE + i)
415            .collect();
416        for &tid in &filler {
417            insert(mk_task(tid, DOS_CELL_ID));
418        }
419
420        // Parent + fillers occupy the whole budget, so the next thread must be
421        // refused — with TryAgain (a recoverable refusal), not a panic, and not a
422        // silent success.
423        let refused = handle_syscall(
424            PARENT_TID,
425            Syscall::Spawn {
426                entry: 0x1000,
427                arg: 0,
428            },
429        );
430        if !matches!(refused, Err(SyscallError::TryAgain)) {
431            ok = false;
432            log::error!(
433                "[selftest] THREAD-CAP-DOS: FAIL — spawn at cap returned {:?}, expected TryAgain",
434                refused
435            );
436            // A spawn that unexpectedly succeeded left a task behind; drop it so the
437            // scheduler is still returned exactly as it was found.
438            if let Ok(stray) = refused {
439                remove(stray);
440            }
441        }
442
443        for tid in filler {
444            remove(tid);
445        }
446        remove(PARENT_TID);
447    }
448
449    // Restore the spawn counter so the test leaves no trace on tid assignment.
450    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
451        sched.next_task_id = n;
452    }
453
454    if ok {
455        log::info!("[selftest] THREAD-CAP: PASS (thread-inherit + honest-revoke + spawn bound)");
456    } else {
457        log::error!("[selftest] THREAD-CAP: FAIL");
458    }
459    ok
460}
