1//! Boot self-tests for two trust-model fixes (2026-07-13):
2//!   #7 — spawned threads inherit the parent cell's identity (CellId + CapSet +
3//!        syscall_allowlist + PKU domain), closing the `CellId(0)` quota escape.
4//!   #5 — `sys_cap_revoke` is honest about ambient authority: MMIO windows are
5//!        torn down (unmapped + released, `.agents/260712-1901` P01-P04) and the
6//!        victim is notified, while HYPERVISOR — which has no teardown path — is
7//!        still refused with `NotSupported`.
8//!
9//! Runs in the single-hart window AFTER `task::init()` but BEFORE
10//! `smp::start_secondaries()` (see `main.rs`), so a synthetic thread inserted into
11//! the boot hart's ready queue can never be picked up by another hart before this
12//! test removes it. All synthetic tasks are created here and torn down before the
13//! function returns — the scheduler is left exactly as it was found.
14//!
15//! Invariant proven, not assumed: after the fix the spawned thread's `cell_id`
16//! equals the parent's (not `CellId(0)`), which is what routes its allocations to
17//! the parent's quota (the charge path keys off `cell_id`; the cell-spawn path in
18//! `loader.rs` already relies on this identity).
19
20use super::cap::CapSet;
21use super::syscall::{handle_syscall, Syscall, SyscallError};
22use super::tcb::Task;
23use api::syscall::cap_mask as CM;
24use types::CellId;

26// Synthetic tids outside any range the boot sequence has assigned yet. Removed
27// before return, so they never collide with real cells.
28const PARENT_TID: usize = 9001;
29const TARGET_TID: usize = 9002;
30const CTRL_TID: usize = 9003;
31/// Target of the privileged-cap revoke (P05): holds pcie_driver/platform/supervisor.
32const DRIVER_TID: usize = 9004;
33/// A synthetic BAR window, registered only under `test-hooks` so the global
34/// BAR table keeps exactly what the ECAM scan put in it.
35#[cfg(feature = "test-hooks")]
36const DRIVER_BAR: usize = 0x5100_0000;
37/// A synthetic requester ID: 00:1f.7, a well-formed BDF that no supported board
38/// populates, so retirement quiescence finds no function and touches nothing.
39/// Where the kernel has no config-space window (AArch64) the function cannot be
40/// shown quiescent at all, and the revoke must refuse instead.
41/// (The previous `0x02_00_00` overflowed the 16-bit encoding and aliased the
42/// real host bridge at 00:00.0.)
43const DRIVER_BDF: u32 = 0x00_FF;
44/// Separate cell for the thread-cap section, so its fillers cannot be mistaken
45/// for (or collide with) the identity-inheritance section's parent cell.
46const DOS_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 4) as u64;
47const DOS_FILLER_BASE: usize = 9100;
48
49// Distinctive sentinels so an inherited value is unambiguously the parent's.
50const TEST_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 2) as u64;
51// A restricted allowlist that still PERMITS the Spawn syscall (so the global
52// allowlist gate in handle_syscall lets the call through) yet differs from the
53// Task::new default of u64::MAX — bit 63 is unassigned (real syscalls use bits
54// ≤54), so clearing it is behaviourally inert but proves the field was inherited
55// rather than left at the permit-all default.
56const TEST_ALLOWLIST: u64 = !(1u64 << 63);
57const TEST_PKU_KEY: u8 = 2;
58const TEST_PKU_VALUE: u32 = 0xABCD_1234;
59
60/// Build a bare task with the given tid/cell and no caps. Only the fields a
61/// test reads/writes need to be meaningful; the task is never scheduled.
62///
63/// Shared with the sibling boot self-tests (`grant_reclaim_selftest`,
64/// `mmio_revoke_selftest`) so each keeps one fixture implementation.
65pub(crate) fn mk_task(tid: usize, cell: u64) -> alloc::boxed::Box<Task> {
66    let mut task = alloc::boxed::Box::new(Task::new(
67        tid,
68        CellId(cell),
69        "selftest",
70        alloc::vec::Vec::new(),
71    ));
72    task.cell_generation = 1;
73    task.root_tid = tid;
74    task
75}

77/// Insert a task into the scheduler map (no ready-queue push — never scheduled).
78pub(crate) fn insert(task: alloc::boxed::Box<Task>) {
79    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
80        if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
81            let owner = api::cell_owner::CellOwner::new(
82                task.cell_id.0,
83                task.cell_generation,
84                task.root_tid as u64,
85            );
86            sched.publish_live_cell_owner(owner);
87        }
88        #[cfg(feature = "p01-trace")]
89        crate::memory::heap::p01_owner::task_backing(&task);
90        #[cfg(feature = "p01-trace")]
91        let _semantic = crate::memory::heap::p01_owner::creation(
92            None,
93            crate::memory::heap::p01_owner::SCHEDULER_SHARED,
94        );
95        sched.tasks.insert(task.id, task);
96    }
97}

99/// Remove a tid from the scheduler map AND every hart's ready queue.
100pub(crate) fn remove(tid: usize) {
101    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
102        if let Some(task) = sched.tasks.remove(&tid) {
103            if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
104                let owner = api::cell_owner::CellOwner::new(
105                    task.cell_id.0,
106                    task.cell_generation,
107                    task.root_tid as u64,
108                );
109                sched.clear_live_cell_owner_for_test(owner);
110            }
111        }
112    }
113    crate::sched::hart_local::ready::remove_from_all(tid);
114}

116/// Returns true iff both fixes behave as specified. Logs a decisive serial line.
117///
118/// Transparent to the boot sequence: the real thread spawn it performs advances
119/// `next_task_id`, so the counter is snapshotted on entry and restored on exit —
120/// the first real cell (Platform) still gets the tid it would have without this
121/// test, keeping CellId assignment stable whether or not the test is compiled in.
122pub fn self_test() -> bool {
123    let mut ok = true;
124
125    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
126
127    // ── #7: thread identity inheritance ────────────────────────────────────────
128    // Parent cell: a distinctive CellId, a restricted allowlist, a non-zero PKU
129    // domain, and one transferable cap (supervisor) + SpawnCap (needed to spawn).
130    {
131        let mut parent = mk_task(PARENT_TID, TEST_CELL_ID);
132        // Two of the three S2-T/S2-W rights (memory withheld), so equality below
133        // proves each one is the Cell's rather than defaulted.
134        // No `network`/`block_io`: those mark a system Cell, which `CapRevoke`
135        // refuses, and the Cell-wide revoke below must be admissible.
136        let parent_caps = CapSet {
137            supervisor: true,
138            spawn: true,
139            observe_tasks: true,
140            vfs_mutate: true,
141            ..CapSet::EMPTY
142        };
143        parent_caps.apply_to(&mut parent);
144        parent.syscall_allowlist = TEST_ALLOWLIST;
145        parent.pku_key = TEST_PKU_KEY;
146        parent.pku_value = TEST_PKU_VALUE;
147        insert(parent);
148
149        // Invoke the REAL Spawn path as if the parent called it. entry/arg are
150        // placeholders; the thread is removed before it can run.
151        let spawned = handle_syscall(
152            PARENT_TID,
153            Syscall::Spawn {
154                entry: 0x1000,
155                arg: 0,
156            },
157        );
158
159        match spawned {
160            Ok(thread_tid) if thread_tid != 0 => {
161                if let Some(sched) = super::SCHEDULER.lock().as_ref() {
162                    if let Some(t) = sched.tasks.get(&thread_tid) {
163                        let cell_ok = t.cell_id.0 == TEST_CELL_ID; // NOT CellId(0)
164                                                                   // The thread is gated on its Cell's record, not a copy.
165                        let caps_ok = sched.cell_rights(thread_tid) == parent_caps;
166                        let allow_ok = t.syscall_allowlist == TEST_ALLOWLIST; // NOT u64::MAX
167                        let pku_ok = t.pku_key == TEST_PKU_KEY && t.pku_value == TEST_PKU_VALUE;
168                        if !(cell_ok && caps_ok && allow_ok && pku_ok) {
169                            ok = false;
170                            log::error!(
171                                "[selftest] THREAD-INHERIT: FAIL \
172                                cell={} caps={} allow={} pku={}",
173                                cell_ok,
174                                caps_ok,
175                                allow_ok,
176                                pku_ok
177                            );
178                        }
179                    } else {
180                        ok = false;
181                        log::error!("[selftest] THREAD-INHERIT: FAIL — thread tid absent");
182                    }
183                }
184                // H-1: a revoke aimed at the thread removes the right from the
185                // whole Cell — the root included — and no thread keeps it.
186                let mut ctrl = mk_task(CTRL_TID, 0x2222);
187                CapSet {
188                    spawn: true,
189                    ..CapSet::EMPTY
190                }
191                .apply_to(&mut ctrl);
192                insert(ctrl);
193                let revoked = handle_syscall(
194                    CTRL_TID,
195                    Syscall::CapRevoke {
196                        target_tid: thread_tid,
197                        cap_mask: CM::SPAWN,
198                    },
199                );
200                let cell_wide = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
201                    !sched.cell_rights(PARENT_TID).spawn && !sched.cell_rights(thread_tid).spawn
202                });
203                if !(revoked.is_ok() && cell_wide) {
204                    ok = false;
205                    log::error!(
206                        "[selftest] REVOKE-CELL-WIDE: FAIL r={:?} cell_wide={}",
207                        revoked,
208                        cell_wide
209                    );
210                }
211                remove(CTRL_TID);
212                remove(thread_tid);
213            }
214            _ => {
215                ok = false;
216                log::error!(
217                    "[selftest] THREAD-INHERIT: FAIL — spawn returned {:?}",
218                    spawned
219                );
220            }
221        }
222        remove(PARENT_TID);
223    }
224
225    // ── #5: honest revoke (ambient bits torn down, hypervisor still refused) ────
226    {
227        // Caller holds SpawnCap (Gate 1). The target holds an MMIO device bit, an
228        // owned grant and a hypervisor cap. An MMIO window is ambient authority —
229        // a mapped window is reachable with no syscall on the access path — so
230        // revoking it must tear the window down, not just clear the field
231        // (`.agents/260712-1901` P01-P04 replaced P00's blanket refusal).
232        // HYPERVISOR has no teardown path and must still be refused.
233        let mut caller = mk_task(PARENT_TID, TEST_CELL_ID);
234        CapSet {
235            spawn: true,
236            ..CapSet::EMPTY
237        }
238        .apply_to(&mut caller);
239        insert(caller);
240
241        let mut target = mk_task(TARGET_TID, 0x1111);
242        CapSet {
243            hypervisor: true,
244            mmio_devices: crate::resource_registry::DEV_GPIO,
245            ..CapSet::EMPTY
246        }
247        .apply_to(&mut target);
248        insert(target);
249
250        // (a) HYPERVISOR bit → NotSupported, cap untouched.
251        let r_hyp = handle_syscall(
252            PARENT_TID,
253            Syscall::CapRevoke {
254                target_tid: TARGET_TID,
255                cap_mask: CM::HYPERVISOR,
256            },
257        );
258        let hyp_refused = matches!(r_hyp, Err(SyscallError::NotSupported))
259            && super::SCHEDULER
260                .lock()
261                .as_ref()
262                .is_some_and(|sched| sched.cell_rights(TARGET_TID).hypervisor);
263        if !hyp_refused {
264            ok = false;
265            log::error!("[selftest] REVOKE-HYPERVISOR: FAIL r={:?}", r_hyp);
266        }
267
268        // The target owns a grant, and — where the board has an allowlisted
269        // window — an MMIO region of the class it is about to lose.
270        let owned_grant = match handle_syscall(TARGET_TID, Syscall::GrantAlloc { size: 4096 }) {
271            Ok(base) if base != 0 => Some(base),
272            _ => None,
273        };
274        let window = crate::resource_registry::allowed_windows()
275            .first()
276            .copied()
277            .and_then(|(base, len, class)| {
278                crate::resource_registry::request_mmio(TARGET_TID, base, len, class)
279                    .is_ok()
280                    .then_some((base, len))
281            });
282
283        // (b) MMIO bits → Ok, window released, owned grant reclaimed, the MMIO
284        //     field cleared (the hypervisor cap untouched), and the victim
285        //     notified with `[0xAC, 0xF2, mask_le4]`.
286        let r_mmio = handle_syscall(
287            PARENT_TID,
288            Syscall::CapRevoke {
289                target_tid: TARGET_TID,
290                cap_mask: CM::MMIO_MASK,
291            },
292        );
293        let target_state = super::SCHEDULER.lock().as_ref().and_then(|sched| {
294            sched.tasks.get(&TARGET_TID).map(|t| {
295                let notified = t
296                    .pending_msgs
297                    .as_slice()
298                    .iter()
299                    .filter_map(|msg| msg.wire.as_ref())
300                    .any(|wire| {
301                        let payload = wire.as_slice();
302                        payload.len() == 6
303                            && payload[0] == 0xAC
304                            && payload[1] == 0xF2
305                            && u32::from_le_bytes([payload[2], payload[3], payload[4], payload[5]])
306                                == CM::MMIO_MASK
307                    });
308                let rights = sched.cell_rights(TARGET_TID);
309                (rights.mmio_devices, rights.hypervisor, notified)
310            })
311        });
312        let window_released = window
313            .map(|(base, _)| crate::resource_registry::lookup_mmio_owner(base).is_none())
314            .unwrap_or(true);
315        let grant_reclaimed = owned_grant
316            .map(|base| {
317                matches!(
318                    handle_syscall(TARGET_TID, Syscall::GrantFree { grant_id: base }),
319                    Err(SyscallError::PermissionDenied)
320                )
321            })
322            .unwrap_or(true);
323
324        let mmio_ok = r_mmio.is_ok()
325            && target_state.is_some_and(|(devices, hyp, notified)| devices == 0 && hyp && notified)
326            && window_released
327            && grant_reclaimed;
328        if !mmio_ok {
329            ok = false;
330            log::error!(
331                "[selftest] REVOKE-MMIO: FAIL r={:?} state={:?} window_released={} \
332                 grant_reclaimed={}",
333                r_mmio,
334                target_state,
335                window_released,
336                grant_reclaimed
337            );
338        }
339
340        // The window above came from the *board's* allowlist, so revoking it
341        // changed its mapping for the whole system. A self-test must not leave
342        // the machine in a state the boot did not produce, so the window goes
343        // back to its boot posture whatever the assertions said: USER where the
344        // board boot-maps it for Cells (RISC-V, QEMU virt — `/bin/periph-demo`
345        // lost the PL011 window at boot before this existed), kernel-only on
346        // the Pi boards, where only a live `RequestMmio` grant sets USER.
347        if let Some((base, len)) = window {
348            #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
349            match crate::memory::paging::restore_mmio_boot_posture(base, len) {
350                Ok(pages) => log::info!(
351                    "[selftest] REVOKE-MMIO: window {:#x} restored to boot posture ({} page(s))",
352                    base,
353                    pages
354                ),
355                Err(e) => log::error!(
356                    "[selftest] REVOKE-MMIO: window {:#x} could not be restored: {:?}",
357                    base,
358                    e
359                ),
360            }
361            #[cfg(target_arch = "x86_64")]
362            crate::memory::paging::map_mmio_user_x86(base, len);
363        }
364
365        // (c) Positive control: SPAWN is a lazy (syscall-gated) bit — revoke of a
366        // non-system target must still SUCCEED and clear the field.
367        let mut ctrl = mk_task(CTRL_TID, 0x2222);
368        CapSet {
369            spawn: true,
370            ..CapSet::EMPTY
371        }
372        .apply_to(&mut ctrl);
373        insert(ctrl);
374        let r_spawn = handle_syscall(
375            PARENT_TID,
376            Syscall::CapRevoke {
377                target_tid: CTRL_TID,
378                cap_mask: CM::SPAWN,
379            },
380        );
381        let cleared = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
382            sched.tasks.contains_key(&CTRL_TID) && !sched.cell_rights(CTRL_TID).spawn
383        });
384        if !(r_spawn.is_ok() && cleared) {
385            ok = false;
386            log::error!(
387                "[selftest] REVOKE-ALLOW: FAIL r={:?} cleared={}",
388                r_spawn,
389                cleared
390            );
391        }
392
393        // (d) Privileged caps are revocable end to end (`.agents/260712-1901`
394        //     P05): `pcie_driver` drains the DMA authority and releases the BDF,
395        //     `platform` releases the ECAM window, `supervisor` is a field clear.
396        let mut driver = mk_task(DRIVER_TID, 0x3333);
397        CapSet {
398            pcie_driver: true,
399            supervisor: true,
400            ..CapSet::EMPTY
401        }
402        .apply_to(&mut driver);
403        // `apply_to` never grants the Platform singleton; the fixture sets the
404        // record bit directly (no reservation is taken).
405        driver.rights.platform = true;
406        insert(driver);
407
408        let bdf_claimed = crate::resource_registry::claim_bdf_owner(DRIVER_BDF, DRIVER_TID);
409        // The BAR table is boot-discovered data with no removal path, so the
410        // synthetic window is registered only where the test hook exists.
411        #[cfg(feature = "test-hooks")]
412        let bar_claimed = crate::resource_registry::test_register_region(
413            DRIVER_TID,
414            DRIVER_BAR,
415            0x1000,
416            crate::resource_registry::DEV_PCIE as crate::resource_registry::MmioClass,
417        );
418
419        // Without config space the owned function cannot be shown quiescent, so the
420        // revoke must fail closed: error, BDF still owned, every capability intact.
421        // The claim is then dropped so the rest of the teardown is still proven.
422        if bdf_claimed && !crate::dev::pcie_ecam::config_space_admitted() {
423            let refused = handle_syscall(
424                PARENT_TID,
425                Syscall::CapRevoke {
426                    target_tid: DRIVER_TID,
427                    cap_mask: CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR,
428                },
429            );
430            let caps_intact = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
431                let rights = sched.cell_rights(DRIVER_TID);
432                rights.pcie_driver && rights.platform && rights.supervisor
433            });
434            let bdf_retained =
435                crate::resource_registry::owner_of_bdf(DRIVER_BDF) == Some(DRIVER_TID);
436            let refused_closed = matches!(refused, Err(SyscallError::NotSupported));
437            if !(refused_closed && caps_intact && bdf_retained) {
438                ok = false;
439                log::error!(
440                    "[selftest] REVOKE-PRIVILEGED: no-config-space revoke did not fail closed \
441                     r={:?} caps_intact={} bdf_retained={}",
442                    refused,
443                    caps_intact,
444                    bdf_retained
445                );
446            }
447            crate::resource_registry::release_bdfs_for(DRIVER_TID);
448        }
449
450        let r_priv = handle_syscall(
451            PARENT_TID,
452            Syscall::CapRevoke {
453                target_tid: DRIVER_TID,
454                cap_mask: CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR,
455            },
456        );
457        let priv_state = super::SCHEDULER.lock().as_ref().and_then(|sched| {
458            sched.tasks.get(&DRIVER_TID).map(|t| {
459                let notified = t
460                    .pending_msgs
461                    .as_slice()
462                    .iter()
463                    .filter_map(|msg| msg.wire.as_ref())
464                    .any(|wire| {
465                        let payload = wire.as_slice();
466                        payload.len() == 6
467                            && payload[0] == 0xAC
468                            && payload[1] == 0xF2
469                            && u32::from_le_bytes([payload[2], payload[3], payload[4], payload[5]])
470                                == CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR
471                    });
472                let rights = sched.cell_rights(DRIVER_TID);
473                (
474                    !rights.pcie_driver,
475                    !rights.platform,
476                    !rights.supervisor,
477                    notified,
478                )
479            })
480        });
481        let bdf_released =
482            bdf_claimed && crate::resource_registry::owner_of_bdf(DRIVER_BDF).is_none();
483        #[cfg(feature = "test-hooks")]
484        let bar_released =
485            bar_claimed && crate::resource_registry::lookup_mmio_owner(DRIVER_BAR).is_none();
486        #[cfg(not(feature = "test-hooks"))]
487        let bar_released = true;
488
489        let priv_ok = r_priv.is_ok()
490            && priv_state.is_some_and(|(pcie, platform, supervisor, notified)| {
491                pcie && platform && supervisor && notified
492            })
493            && bdf_released
494            && bar_released;
495        if !priv_ok {
496            ok = false;
497            log::error!(
498                "[selftest] REVOKE-PRIVILEGED: FAIL r={:?} state={:?} bdf_released={} \
499                 bar_released={}",
500                r_priv,
501                priv_state,
502                bdf_released,
503                bar_released
504            );
505        }
506
507        remove(PARENT_TID);
508        remove(TARGET_TID);
509        remove(CTRL_TID);
510        remove(DRIVER_TID);
511    }
512
513    // ── thread-spawn DoS bound ─────────────────────────────────────────────────
514    // `Syscall::Spawn` is gated by the syscall allowlist but NOT by SpawnCap, and
515    // each thread costs a contiguous run of STACK_PAGES+1 frames. Before the cap an
516    // unprivileged cell could loop here until the allocator fragmented, at which
517    // point the scheduler's `.expect("OOM Stack")` panicked the kernel — never-die
518    // broken from userspace. The refusal must be an error the caller survives.
519    //
520    // The cell is filled with STACKLESS synthetic tasks: the cap counts live tasks
521    // sharing a CellId, so proving the bound costs 31 Task structs instead of 31
522    // real 65-frame stack allocations.
523    {
524        let mut parent = mk_task(PARENT_TID, DOS_CELL_ID);
525        CapSet {
526            spawn: true,
527            ..CapSet::EMPTY
528        }
529        .apply_to(&mut parent);
530        insert(parent);
531
532        let filler: alloc::vec::Vec<usize> = (0..crate::sched::scheduler::MAX_THREADS_PER_CELL - 1)
533            .map(|i| DOS_FILLER_BASE + i)
534            .collect();
535        for &tid in &filler {
536            insert(mk_task(tid, DOS_CELL_ID));
537        }
538
539        // Parent + fillers occupy the whole budget, so the next thread must be
540        // refused — with TryAgain (a recoverable refusal), not a panic, and not a
541        // silent success.
542        let refused = handle_syscall(
543            PARENT_TID,
544            Syscall::Spawn {
545                entry: 0x1000,
546                arg: 0,
547            },
548        );
549        if !matches!(refused, Err(SyscallError::TryAgain)) {
550            ok = false;
551            log::error!(
552                "[selftest] THREAD-CAP-DOS: FAIL — spawn at cap returned {:?}, expected TryAgain",
553                refused
554            );
555            // A spawn that unexpectedly succeeded left a task behind; drop it so the
556            // scheduler is still returned exactly as it was found.
557            if let Ok(stray) = refused {
558                remove(stray);
559            }
560        }
561
562        for tid in filler {
563            remove(tid);
564        }
565        remove(PARENT_TID);
566    }
567
568    // Restore the spawn counter so the test leaves no trace on tid assignment.
569    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
570        sched.next_task_id = n;
571    }
572
573    if ok {
574        log::info!("[selftest] THREAD-CAP: PASS (thread-inherit + honest-revoke + spawn bound)");
575    } else {
576        log::error!("[selftest] THREAD-CAP: FAIL");
577    }
578    ok
579}
