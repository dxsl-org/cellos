1//! Boot self-tests for two trust-model fixes (2026-07-13):
2//!   #7 — spawned threads inherit the parent cell's identity (CellId + CapSet +
3//!        syscall_allowlist + PKU domain), closing the `CellId(0)` quota escape.
4//!   #5 — `sys_cap_revoke` is honest about ambient authority: MMIO windows are
5//!        torn down (unmapped + released, `.agents/260712-1901` P01-P04) and the
6//!        victim is notified, while HYPERVISOR — which has no teardown path — is
7//!        still refused with `NotSupported`.
8//!
9//! Runs in the single-hart window AFTER `task::init()` but BEFORE
10//! `smp::start_secondaries()` (see `main.rs`), so a synthetic thread inserted into
11//! the boot hart's ready queue can never be picked up by another hart before this
12//! test removes it. All synthetic tasks are created here and torn down before the
13//! function returns — the scheduler is left exactly as it was found.
14//!
15//! Invariant proven, not assumed: after the fix the spawned thread's `cell_id`
16//! equals the parent's (not `CellId(0)`), which is what routes its allocations to
17//! the parent's quota (the charge path keys off `cell_id`; the cell-spawn path in
18//! `loader.rs` already relies on this identity).
19
20use super::cap::CapSet;
21use super::syscall::{handle_syscall, Syscall, SyscallError};
22use super::tcb::Task;
23use abi::app_envelope;
24use abi::syscall::cap_mask as CM;
25use types::CellId;

27// Synthetic tids outside any range the boot sequence has assigned yet. Removed
28// before return, so they never collide with real cells.
29const PARENT_TID: usize = 9001;
30const TARGET_TID: usize = 9002;
31const CTRL_TID: usize = 9003;
32/// Target of the privileged-cap revoke (P05): holds pcie_driver/platform/supervisor.
33const DRIVER_TID: usize = 9004;
34/// A synthetic BAR window, registered only under `test-hooks` so the global
35/// BAR table keeps exactly what the ECAM scan put in it.
36#[cfg(feature = "test-hooks")]
37const DRIVER_BAR: usize = 0x5100_0000;
38/// A synthetic requester ID: 00:1f.7, a well-formed BDF that no supported board
39/// populates, so retirement quiescence finds no function and touches nothing.
40/// Where the kernel has no config-space window (AArch64) the function cannot be
41/// shown quiescent at all, and the revoke must refuse instead.
42/// (The previous `0x02_00_00` overflowed the 16-bit encoding and aliased the
43/// real host bridge at 00:00.0.)
44const DRIVER_BDF: u32 = 0x00_FF;
45/// A later owner of `DRIVER_BDF`, which must find the revoked driver's NIC
46/// route vacant.
47#[cfg(feature = "test-hooks")]
48const REPLACEMENT_TID: usize = 9005;
49/// Separate cell for the thread-cap section, so its fillers cannot be mistaken
50/// for (or collide with) the identity-inheritance section's parent cell.
51const DOS_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 4) as u64;
52const DOS_FILLER_BASE: usize = 9100;
53
54// Distinctive sentinels so an inherited value is unambiguously the parent's.
55const TEST_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 2) as u64;
56// A restricted allowlist that still PERMITS the Spawn syscall (so the global
57// allowlist gate in handle_syscall lets the call through) yet differs from the
58// Task::new default of u64::MAX — bit 63 is unassigned (real syscalls use bits
59// ≤54), so clearing it is behaviourally inert but proves the field was inherited
60// rather than left at the permit-all default.
61const TEST_ALLOWLIST: u64 = !(1u64 << 63);
62const TEST_PKU_KEY: u8 = 2;
63const TEST_PKU_VALUE: u32 = 0xABCD_1234;
64
65/// Build a bare task with the given tid/cell and no caps. Only the fields a
66/// test reads/writes need to be meaningful; the task is never scheduled.
67///
68/// Shared with the sibling boot self-tests (`grant_reclaim_selftest`,
69/// `mmio_revoke_selftest`) so each keeps one fixture implementation.
70pub(crate) fn mk_task(tid: usize, cell: u64) -> alloc::boxed::Box<Task> {
71    let mut task = alloc::boxed::Box::new(Task::new(
72        tid,
73        CellId(cell),
74        "selftest",
75        alloc::vec::Vec::new(),
76    ));
77    task.cell_generation = 1;
78    task.root_tid = tid;
79    task
80}

82/// Insert a task into the scheduler map (no ready-queue push — never scheduled).
83pub(crate) fn insert(task: alloc::boxed::Box<Task>) {
84    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
85        if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
86            let owner = abi::cell_owner::CellOwner::new(
87                task.cell_id.0,
88                task.cell_generation,
89                task.root_tid as u64,
90            );
91            sched.publish_live_cell_owner(owner);
92        }
93        #[cfg(feature = "p01-trace")]
94        crate::memory::heap::p01_owner::task_backing(&task);
95        #[cfg(feature = "p01-trace")]
96        let _semantic = crate::memory::heap::p01_owner::creation(
97            None,
98            crate::memory::heap::p01_owner::SCHEDULER_SHARED,
99        );
100        sched.tasks.insert(task.id, task);
101    }
102}

104/// Remove a tid from the scheduler map AND every hart's ready queue.
105pub(crate) fn remove(tid: usize) {
106    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
107        if let Some(task) = sched.tasks.remove(&tid) {
108            if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
109                let owner = abi::cell_owner::CellOwner::new(
110                    task.cell_id.0,
111                    task.cell_generation,
112                    task.root_tid as u64,
113                );
114                sched.clear_live_cell_owner_for_test(owner);
115            }
116        }
117    }
118    crate::sched::hart_local::ready::remove_from_all(tid);
119}

121/// Returns true iff both fixes behave as specified. Logs a decisive serial line.
122///
123/// Transparent to the boot sequence: the real thread spawn it performs advances
124/// `next_task_id`, so the counter is snapshotted on entry and restored on exit —
125/// the first real cell (Platform) still gets the tid it would have without this
126/// test, keeping CellId assignment stable whether or not the test is compiled in.
127pub fn self_test() -> bool {
128    let mut ok = true;
129
130    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
131
132    // ── #7: thread identity inheritance ────────────────────────────────────────
133    // Parent cell: a distinctive CellId, a restricted allowlist, a non-zero PKU
134    // domain, and one transferable cap (supervisor) + SpawnCap (needed to spawn).
135    {
136        let mut parent = mk_task(PARENT_TID, TEST_CELL_ID);
137        // Two of the three S2-T/S2-W rights (memory withheld), so equality below
138        // proves each one is the Cell's rather than defaulted.
139        // No `network`/`block_io`: those mark a system Cell, which `CapRevoke`
140        // refuses, and the Cell-wide revoke below must be admissible.
141        let parent_caps = CapSet {
142            supervisor: true,
143            spawn: true,
144            observe_tasks: true,
145            vfs_mutate: true,
146            ..CapSet::EMPTY
147        };
148        parent_caps.apply_to(&mut parent);
149        parent.syscall_allowlist = TEST_ALLOWLIST;
150        parent.pku_key = TEST_PKU_KEY;
151        parent.pku_value = TEST_PKU_VALUE;
152        insert(parent);
153
154        // Invoke the REAL Spawn path as if the parent called it. entry/arg are
155        // placeholders; the thread is removed before it can run.
156        let spawned = handle_syscall(
157            PARENT_TID,
158            Syscall::Spawn {
159                entry: 0x1000,
160                arg: 0,
161            },
162        );
163
164        match spawned {
165            Ok(thread_tid) if thread_tid != 0 => {
166                if let Some(sched) = super::SCHEDULER.lock().as_ref() {
167                    if let Some(t) = sched.tasks.get(&thread_tid) {
168                        let cell_ok = t.cell_id.0 == TEST_CELL_ID; // NOT CellId(0)
169                                                                   // The thread is gated on its Cell's record, not a copy.
170                        let caps_ok = sched.cell_rights(thread_tid) == parent_caps;
171                        let allow_ok = t.syscall_allowlist == TEST_ALLOWLIST; // NOT u64::MAX
172                        let pku_ok = t.pku_key == TEST_PKU_KEY && t.pku_value == TEST_PKU_VALUE;
173                        if !(cell_ok && caps_ok && allow_ok && pku_ok) {
174                            ok = false;
175                            log::error!(
176                                "[selftest] THREAD-INHERIT: FAIL \
177                                cell={} caps={} allow={} pku={}",
178                                cell_ok,
179                                caps_ok,
180                                allow_ok,
181                                pku_ok
182                            );
183                        }
184                    } else {
185                        ok = false;
186                        log::error!("[selftest] THREAD-INHERIT: FAIL — thread tid absent");
187                    }
188                }
189                // H-1: a revoke aimed at the thread removes the right from the
190                // whole Cell — the root included — and no thread keeps it.
191                let mut ctrl = mk_task(CTRL_TID, 0x2222);
192                CapSet {
193                    spawn: true,
194                    ..CapSet::EMPTY
195                }
196                .apply_to(&mut ctrl);
197                insert(ctrl);
198                let revoked = handle_syscall(
199                    CTRL_TID,
200                    Syscall::CapRevoke {
201                        target_tid: thread_tid,
202                        cap_mask: CM::SPAWN,
203                    },
204                );
205                let cell_wide = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
206                    !sched.cell_rights(PARENT_TID).spawn && !sched.cell_rights(thread_tid).spawn
207                });
208                if !(revoked.is_ok() && cell_wide) {
209                    ok = false;
210                    log::error!(
211                        "[selftest] REVOKE-CELL-WIDE: FAIL r={:?} cell_wide={}",
212                        revoked,
213                        cell_wide
214                    );
215                }
216                remove(CTRL_TID);
217                remove(thread_tid);
218            }
219            _ => {
220                ok = false;
221                log::error!(
222                    "[selftest] THREAD-INHERIT: FAIL — spawn returned {:?}",
223                    spawned
224                );
225            }
226        }
227        remove(PARENT_TID);
228    }
229
230    // ── #5: honest revoke (ambient bits torn down, hypervisor still refused) ────
231    {
232        // Caller holds SpawnCap (Gate 1). The target holds an MMIO device bit, an
233        // owned grant and a hypervisor cap. An MMIO window is ambient authority —
234        // a mapped window is reachable with no syscall on the access path — so
235        // revoking it must tear the window down, not just clear the field
236        // (`.agents/260712-1901` P01-P04 replaced P00's blanket refusal).
237        // HYPERVISOR has no teardown path and must still be refused.
238        let mut caller = mk_task(PARENT_TID, TEST_CELL_ID);
239        CapSet {
240            spawn: true,
241            ..CapSet::EMPTY
242        }
243        .apply_to(&mut caller);
244        insert(caller);
245
246        let mut target = mk_task(TARGET_TID, 0x1111);
247        CapSet {
248            hypervisor: true,
249            mmio_devices: crate::resource_registry::DEV_GPIO,
250            ..CapSet::EMPTY
251        }
252        .apply_to(&mut target);
253        insert(target);
254
255        // (a) HYPERVISOR bit → NotSupported, cap untouched.
256        let r_hyp = handle_syscall(
257            PARENT_TID,
258            Syscall::CapRevoke {
259                target_tid: TARGET_TID,
260                cap_mask: CM::HYPERVISOR,
261            },
262        );
263        let hyp_refused = matches!(r_hyp, Err(SyscallError::NotSupported))
264            && super::SCHEDULER
265                .lock()
266                .as_ref()
267                .is_some_and(|sched| sched.cell_rights(TARGET_TID).hypervisor);
268        if !hyp_refused {
269            ok = false;
270            log::error!("[selftest] REVOKE-HYPERVISOR: FAIL r={:?}", r_hyp);
271        }
272
273        // The target owns a grant, and — where the board has an allowlisted
274        // window — an MMIO region of the class it is about to lose.
275        let owned_grant = match handle_syscall(TARGET_TID, Syscall::GrantAlloc { size: 4096 }) {
276            Ok(base) if base != 0 => Some(base),
277            _ => None,
278        };
279        let window = crate::resource_registry::allowed_windows()
280            .first()
281            .copied()
282            .and_then(|(base, len, class)| {
283                crate::resource_registry::request_mmio(TARGET_TID, base, len, class)
284                    .is_ok()
285                    .then_some((base, len))
286            });
287
288        // (b) MMIO bits → Ok, window released, owned grant reclaimed, the MMIO
289        //     field cleared (the hypervisor cap untouched), and the victim
290        //     notified with the `KIND_CAP_REVOKED` envelope.
291        let r_mmio = handle_syscall(
292            PARENT_TID,
293            Syscall::CapRevoke {
294                target_tid: TARGET_TID,
295                cap_mask: CM::MMIO_MASK,
296            },
297        );
298        let target_state = super::SCHEDULER.lock().as_ref().and_then(|sched| {
299            sched.tasks.get(&TARGET_TID).map(|t| {
300                let notified = t
301                    .pending_msgs
302                    .as_slice()
303                    .iter()
304                    .filter_map(|msg| msg.wire.as_ref())
305                    .any(|wire| {
306                        let payload = wire.as_slice();
307                        payload.len() == 6
308                            && payload[0] == app_envelope::MAGIC
309                            && payload[1] == app_envelope::KIND_CAP_REVOKED
310                            && u32::from_le_bytes([payload[2], payload[3], payload[4], payload[5]])
311                                == CM::MMIO_MASK
312                    });
313                let rights = sched.cell_rights(TARGET_TID);
314                (rights.mmio_devices, rights.hypervisor, notified)
315            })
316        });
317        let window_released = window
318            .map(|(base, _)| crate::resource_registry::lookup_mmio_owner(base).is_none())
319            .unwrap_or(true);
320        let grant_reclaimed = owned_grant
321            .map(|base| {
322                matches!(
323                    handle_syscall(TARGET_TID, Syscall::GrantFree { grant_id: base }),
324                    Err(SyscallError::PermissionDenied)
325                )
326            })
327            .unwrap_or(true);
328
329        let mmio_ok = r_mmio.is_ok()
330            && target_state.is_some_and(|(devices, hyp, notified)| devices == 0 && hyp && notified)
331            && window_released
332            && grant_reclaimed;
333        if !mmio_ok {
334            ok = false;
335            log::error!(
336                "[selftest] REVOKE-MMIO: FAIL r={:?} state={:?} window_released={} \
337                 grant_reclaimed={}",
338                r_mmio,
339                target_state,
340                window_released,
341                grant_reclaimed
342            );
343        }
344
345        // The window above came from the *board's* allowlist, so revoking it
346        // changed its mapping for the whole system. A self-test must not leave
347        // the machine in a state the boot did not produce, so the window goes
348        // back to its boot posture whatever the assertions said: USER where the
349        // board boot-maps it for Cells (RISC-V, QEMU virt — `/bin/periph-demo`
350        // lost the PL011 window at boot before this existed), kernel-only on
351        // the Pi boards, where only a live `RequestMmio` grant sets USER.
352        if let Some((base, len)) = window {
353            #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
354            match crate::memory::paging::restore_mmio_boot_posture(base, len) {
355                Ok(pages) => log::info!(
356                    "[selftest] REVOKE-MMIO: window {:#x} restored to boot posture ({} page(s))",
357                    base,
358                    pages
359                ),
360                Err(e) => log::error!(
361                    "[selftest] REVOKE-MMIO: window {:#x} could not be restored: {:?}",
362                    base,
363                    e
364                ),
365            }
366            #[cfg(target_arch = "x86_64")]
367            crate::memory::paging::map_mmio_user_x86(base, len);
368        }
369
370        // (c) Positive control: SPAWN is a lazy (syscall-gated) bit — revoke of a
371        // non-system target must still SUCCEED and clear the field.
372        let mut ctrl = mk_task(CTRL_TID, 0x2222);
373        CapSet {
374            spawn: true,
375            ..CapSet::EMPTY
376        }
377        .apply_to(&mut ctrl);
378        insert(ctrl);
379        let r_spawn = handle_syscall(
380            PARENT_TID,
381            Syscall::CapRevoke {
382                target_tid: CTRL_TID,
383                cap_mask: CM::SPAWN,
384            },
385        );
386        let cleared = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
387            sched.tasks.contains_key(&CTRL_TID) && !sched.cell_rights(CTRL_TID).spawn
388        });
389        if !(r_spawn.is_ok() && cleared) {
390            ok = false;
391            log::error!(
392                "[selftest] REVOKE-ALLOW: FAIL r={:?} cleared={}",
393                r_spawn,
394                cleared
395            );
396        }
397
398        // (d) Privileged caps are revocable end to end (`.agents/260712-1901`
399        //     P05): `pcie_driver` drains the DMA authority and releases the BDF,
400        //     `platform` releases the ECAM window, `supervisor` is a field clear.
401        let mut driver = mk_task(DRIVER_TID, 0x3333);
402        CapSet {
403            pcie_driver: true,
404            supervisor: true,
405            ..CapSet::EMPTY
406        }
407        .apply_to(&mut driver);
408        // `apply_to` never grants the Platform singleton; the fixture sets the
409        // record bit directly (no reservation is taken).
410        driver.rights.platform = true;
411        insert(driver);
412
413        let bdf_claimed = crate::dev::ownership::claim_pci(DRIVER_BDF, DRIVER_TID);
414        #[cfg(feature = "test-hooks")]
415        let nic_route = bind_driver_nic_route(bdf_claimed);
416        // The BAR table is boot-discovered data with no removal path, so the
417        // synthetic window is registered only where the test hook exists.
418        #[cfg(feature = "test-hooks")]
419        let bar_claimed = crate::resource_registry::test_register_region(
420            DRIVER_TID,
421            DRIVER_BAR,
422            0x1000,
423            crate::resource_registry::DEV_PCIE as crate::resource_registry::MmioClass,
424        );
425
426        // Without config space the owned function cannot be shown quiescent, so the
427        // revoke must fail closed: error, BDF still owned, every capability intact.
428        // The claim is then dropped so the rest of the teardown is still proven.
429        if bdf_claimed && !crate::dev::pcie_ecam::config_space_admitted() {
430            let refused = handle_syscall(
431                PARENT_TID,
432                Syscall::CapRevoke {
433                    target_tid: DRIVER_TID,
434                    cap_mask: CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR,
435                },
436            );
437            let caps_intact = super::SCHEDULER.lock().as_ref().is_some_and(|sched| {
438                let rights = sched.cell_rights(DRIVER_TID);
439                rights.pcie_driver && rights.platform && rights.supervisor
440            });
441            let bdf_retained = crate::dev::ownership::pci_owner(DRIVER_BDF) == Some(DRIVER_TID);
442            // A refused revoke freed nothing, so the route keeps its provider.
443            #[cfg(feature = "test-hooks")]
444            let route_kept = nic_route.is_some()
445                && crate::cell::service_registry::lookup(abi::syscall::service::NIC_DRIVER)
446                    == Some(DRIVER_TID);
447            #[cfg(not(feature = "test-hooks"))]
448            let route_kept = true;
449            let refused_closed = matches!(refused, Err(SyscallError::NotSupported));
450            if !(refused_closed && caps_intact && bdf_retained && route_kept) {
451                ok = false;
452                log::error!(
453                    "[selftest] REVOKE-PRIVILEGED: no-config-space revoke did not fail closed \
454                     r={:?} caps_intact={} bdf_retained={} route_kept={}",
455                    refused,
456                    caps_intact,
457                    bdf_retained,
458                    route_kept
459                );
460            }
461            crate::dev::ownership::release_for(DRIVER_TID, crate::dev::ownership::Scope::Retired);
462        }
463
464        let r_priv = handle_syscall(
465            PARENT_TID,
466            Syscall::CapRevoke {
467                target_tid: DRIVER_TID,
468                cap_mask: CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR,
469            },
470        );
471        let priv_state = super::SCHEDULER.lock().as_ref().and_then(|sched| {
472            sched.tasks.get(&DRIVER_TID).map(|t| {
473                let notified = t
474                    .pending_msgs
475                    .as_slice()
476                    .iter()
477                    .filter_map(|msg| msg.wire.as_ref())
478                    .any(|wire| {
479                        let payload = wire.as_slice();
480                        payload.len() == 6
481                            && payload[0] == app_envelope::MAGIC
482                            && payload[1] == app_envelope::KIND_CAP_REVOKED
483                            && u32::from_le_bytes([payload[2], payload[3], payload[4], payload[5]])
484                                == CM::PCIE_DRIVER | CM::PLATFORM | CM::SUPERVISOR
485                    });
486                let rights = sched.cell_rights(DRIVER_TID);
487                (
488                    !rights.pcie_driver,
489                    !rights.platform,
490                    !rights.supervisor,
491                    notified,
492                )
493            })
494        });
495        let bdf_released = bdf_claimed && crate::dev::ownership::pci_owner(DRIVER_BDF).is_none();
496        #[cfg(feature = "test-hooks")]
497        let bar_released =
498            bar_claimed && crate::resource_registry::lookup_mmio_owner(DRIVER_BAR).is_none();
499        #[cfg(not(feature = "test-hooks"))]
500        let bar_released = true;
501        #[cfg(feature = "test-hooks")]
502        let route_vacated = nic_route.is_some_and(driver_nic_route_vacated);
503        #[cfg(not(feature = "test-hooks"))]
504        let route_vacated = true;
505
506        let priv_ok = r_priv.is_ok()
507            && priv_state.is_some_and(|(pcie, platform, supervisor, notified)| {
508                pcie && platform && supervisor && notified
509            })
510            && bdf_released
511            && bar_released
512            && route_vacated;
513        if !priv_ok {
514            ok = false;
515            log::error!(
516                "[selftest] REVOKE-PRIVILEGED: FAIL r={:?} state={:?} bdf_released={} \
517                 bar_released={} route_vacated={}",
518                r_priv,
519                priv_state,
520                bdf_released,
521                bar_released,
522                route_vacated
523            );
524        }
525        // Leaves no route provider or `service::NIC_DRIVER` row behind, even
526        // when an assertion above failed.
527        #[cfg(feature = "test-hooks")]
528        crate::dev::driver_cell::deregister_all_for(DRIVER_TID);
529
530        remove(PARENT_TID);
531        remove(TARGET_TID);
532        remove(CTRL_TID);
533        remove(DRIVER_TID);
534    }
535
536    // ── thread-spawn DoS bound ─────────────────────────────────────────────────
537    // `Syscall::Spawn` is gated by the syscall allowlist but NOT by SpawnCap, and
538    // each thread costs a contiguous run of STACK_PAGES+1 frames. Before the cap an
539    // unprivileged cell could loop here until the allocator fragmented, at which
540    // point the scheduler's `.expect("OOM Stack")` panicked the kernel — never-die
541    // broken from userspace. The refusal must be an error the caller survives.
542    //
543    // The cell is filled with STACKLESS synthetic tasks: the cap counts live tasks
544    // sharing a CellId, so proving the bound costs 31 Task structs instead of 31
545    // real 65-frame stack allocations.
546    {
547        let mut parent = mk_task(PARENT_TID, DOS_CELL_ID);
548        CapSet {
549            spawn: true,
550            ..CapSet::EMPTY
551        }
552        .apply_to(&mut parent);
553        insert(parent);
554
555        let filler: alloc::vec::Vec<usize> = (0..crate::sched::scheduler::MAX_THREADS_PER_CELL - 1)
556            .map(|i| DOS_FILLER_BASE + i)
557            .collect();
558        for &tid in &filler {
559            insert(mk_task(tid, DOS_CELL_ID));
560        }
561
562        // Parent + fillers occupy the whole budget, so the next thread must be
563        // refused — with TryAgain (a recoverable refusal), not a panic, and not a
564        // silent success.
565        let refused = handle_syscall(
566            PARENT_TID,
567            Syscall::Spawn {
568                entry: 0x1000,
569                arg: 0,
570            },
571        );
572        if !matches!(refused, Err(SyscallError::TryAgain)) {
573            ok = false;
574            log::error!(
575                "[selftest] THREAD-CAP-DOS: FAIL — spawn at cap returned {:?}, expected TryAgain",
576                refused
577            );
578            // A spawn that unexpectedly succeeded left a task behind; drop it so the
579            // scheduler is still returned exactly as it was found.
580            if let Ok(stray) = refused {
581                remove(stray);
582            }
583        }
584
585        for tid in filler {
586            remove(tid);
587        }
588        remove(PARENT_TID);
589    }
590
591    // Restore the spawn counter so the test leaves no trace on tid assignment.
592    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
593        sched.next_task_id = n;
594    }
595
596    if ok {
597        log::info!("[selftest] THREAD-CAP: PASS (thread-inherit + honest-revoke + spawn bound)");
598    } else {
599        log::error!("[selftest] THREAD-CAP: FAIL");
600    }
601    ok
602}

604#[cfg(feature = "test-hooks")]
605fn driver_nic_key() -> Option<abi::block_route::DeviceKey> {
606    abi::block_route::DeviceKey::new(abi::block_route::BLOCK_DEVICE_PCI, u64::from(DRIVER_BDF))
607}

609/// Bind the driver's NIC route as `RegisterNicDriver` (417) does, before any
610/// real NIC registers, so it is the primary and publishes `service::NIC_DRIVER`.
611/// `Some(route index)` only then: otherwise the revoke's effect on the service
612/// row could not be observed.
613///
614/// The route table has no removal path, so the slot stays keyed (vacant) for
615/// the boot; hence test-hooks only, like the synthetic BAR window.
616#[cfg(feature = "test-hooks")]
617fn bind_driver_nic_route(bdf_claimed: bool) -> Option<usize> {
618    use crate::cell::service_registry;
619    use abi::syscall::service::NIC_DRIVER;
620    // A row this early would be replaced by the publication below.
621    if !bdf_claimed || service_registry::lookup(NIC_DRIVER).is_some() {
622        log::error!("[selftest] REVOKE-PRIVILEGED: NIC route fixture unavailable");
623        return None;
624    }
625    let provider = (DRIVER_TID, 0x3333, 1);
626    let index = crate::dev::driver_cell::publish_nic_route(driver_nic_key()?, provider, || {
627        service_registry::register(NIC_DRIVER, provider.0, provider.1, provider.2)
628    })?;
629    (service_registry::lookup(NIC_DRIVER) == Some(DRIVER_TID)).then_some(index)
630}
631
632/// After the revoke freed `DRIVER_BDF` from the still-live driver: its
633/// `service::NIC_DRIVER` row is gone (the primary had no live successor) and a
634/// new owner of the function binds the same route rather than being refused as
635/// held. The replacement is torn down again before returning.
636#[cfg(feature = "test-hooks")]
637fn driver_nic_route_vacated(index: usize) -> bool {
638    use crate::dev::ownership;
639    let row_gone = crate::cell::service_registry::lookup(abi::syscall::service::NIC_DRIVER)
640        != Some(DRIVER_TID);
641    let rebound = ownership::claim_pci(DRIVER_BDF, REPLACEMENT_TID)
642        && driver_nic_key().is_some_and(|key| {
643            crate::dev::driver_cell::publish_nic_route(key, (REPLACEMENT_TID, 0x3334, 1), || true)
644                == Some(index)
645        });
646    crate::dev::driver_cell::deregister_all_for(REPLACEMENT_TID);
647    ownership::release_for(REPLACEMENT_TID, ownership::Scope::Retired);
648    if !(row_gone && rebound) {
649        log::error!(
650            "[selftest] REVOKE-PRIVILEGED: NIC route {} not vacated (row_gone={} rebound={})",
651            index,
652            row_gone,
653            rebound
654        );
655    }
656    row_gone && rebound
657}
