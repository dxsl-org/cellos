1//! Boot self-tests for two trust-model fixes (2026-07-13):
2//!   #7 — spawned threads inherit the parent cell's identity (CellId + CapSet +
3//!        syscall_allowlist + PKU domain), closing the `CellId(0)` quota escape.
4//!   #5 — `sys_cap_revoke` refuses ambient-authority bits (HYPERVISOR, MMIO) with
5//!        `NotSupported` instead of silently clearing a field it cannot truly revoke.
6//!
7//! Runs in the single-hart window AFTER `task::init()` but BEFORE
8//! `smp::start_secondaries()` (see `main.rs`), so a synthetic thread inserted into
9//! the boot hart's ready queue can never be picked up by another hart before this
10//! test removes it. All synthetic tasks are created here and torn down before the
11//! function returns — the scheduler is left exactly as it was found.
12//!
13//! Invariant proven, not assumed: after the fix the spawned thread's `cell_id`
14//! equals the parent's (not `CellId(0)`), which is what routes its allocations to
15//! the parent's quota (the charge path keys off `cell_id`; the cell-spawn path in
16//! `loader.rs` already relies on this identity).
17
18use super::cap::CapSet;
19use super::syscall::{handle_syscall, Syscall, SyscallError};
20use super::tcb::Task;
21use api::syscall::cap_mask as CM;
22use types::CellId;
23
24// Synthetic tids outside any range the boot sequence has assigned yet. Removed
25// before return, so they never collide with real cells.
26const PARENT_TID: usize = 9001;
27const TARGET_TID: usize = 9002;
28const CTRL_TID: usize = 9003;
29/// Separate cell for the thread-cap section, so its fillers cannot be mistaken
30/// for (or collide with) the identity-inheritance section's parent cell.
31const DOS_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 4) as u64;
32const DOS_FILLER_BASE: usize = 9100;
33
34// Distinctive sentinels so an inherited value is unambiguously the parent's.
35const TEST_CELL_ID: u64 = (crate::memory::cell_quota::MAX_CELLS - 2) as u64;
36// A restricted allowlist that still PERMITS the Spawn syscall (so the global
37// allowlist gate in handle_syscall lets the call through) yet differs from the
38// Task::new default of u64::MAX — bit 63 is unassigned (real syscalls use bits
39// ≤54), so clearing it is behaviourally inert but proves the field was inherited
40// rather than left at the permit-all default.
41const TEST_ALLOWLIST: u64 = !(1u64 << 63);
42const TEST_PKU_KEY: u8 = 2;
43const TEST_PKU_VALUE: u32 = 0xABCD_1234;
44
45/// Build a bare task with the given tid/cell and no caps. Only the fields this
46/// test reads/writes need to be meaningful; the task is never scheduled.
47fn mk_task(tid: usize, cell: u64) -> alloc::boxed::Box<Task> {
48    let mut task = alloc::boxed::Box::new(Task::new(
49        tid,
50        CellId(cell),
51        "selftest",
52        alloc::vec::Vec::new(),
53    ));
54    task.cell_generation = 1;
55    task.root_tid = tid;
56    task
57}
58
59/// Insert a task into the scheduler map (no ready-queue push — never scheduled).
60fn insert(task: alloc::boxed::Box<Task>) {
61    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
62        if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
63            let owner = api::cell_owner::CellOwner::new(
64                task.cell_id.0,
65                task.cell_generation,
66                task.root_tid as u64,
67            );
68            sched.publish_live_cell_owner(owner);
69        }
70        sched.tasks.insert(task.id, task);
71    }
72}
73
74/// Remove a tid from the scheduler map AND every hart's ready queue.
75fn remove(tid: usize) {
76    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
77        if let Some(task) = sched.tasks.remove(&tid) {
78            if (task.cell_id.0 as usize) < crate::memory::cell_quota::MAX_CELLS {
79                let owner = api::cell_owner::CellOwner::new(
80                    task.cell_id.0,
81                    task.cell_generation,
82                    task.root_tid as u64,
83                );
84                sched.clear_live_cell_owner_for_test(owner);
85            }
86        }
87    }
88    super::hart_local::ready::remove_from_all(tid);
89}

91/// Returns true iff both fixes behave as specified. Logs a decisive serial line.
92///
93/// Transparent to the boot sequence: the real thread spawn it performs advances
94/// `next_task_id`, so the counter is snapshotted on entry and restored on exit —
95/// the first real cell (Platform) still gets the tid it would have without this
96/// test, keeping CellId assignment stable whether or not the test is compiled in.
97pub fn self_test() -> bool {
98    let mut ok = true;
99
100    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
101
102    // ── #7: thread identity inheritance ────────────────────────────────────────
103    // Parent cell: a distinctive CellId, a restricted allowlist, a non-zero PKU
104    // domain, and one transferable cap (network) + SpawnCap (needed to spawn).
105    {
106        let mut parent = mk_task(PARENT_TID, TEST_CELL_ID);
107        parent.network_cap = Some(super::cap::NetworkCap::new());
108        parent.spawn_cap = Some(super::cap::SpawnCap::new());
109        parent.syscall_allowlist = TEST_ALLOWLIST;
110        parent.pku_key = TEST_PKU_KEY;
111        parent.pku_value = TEST_PKU_VALUE;
112        let parent_caps = CapSet::of_task(&parent);
113        insert(parent);
114
115        // Invoke the REAL Spawn path as if the parent called it. entry/arg are
116        // placeholders; the thread is removed before it can run.
117        let spawned = handle_syscall(
118            PARENT_TID,
119            Syscall::Spawn {
120                entry: 0x1000,
121                arg: 0,
122            },
123        );
124
125        match spawned {
126            Ok(thread_tid) if thread_tid != 0 => {
127                if let Some(sched) = super::SCHEDULER.lock().as_ref() {
128                    if let Some(t) = sched.tasks.get(&thread_tid) {
129                        let cell_ok = t.cell_id.0 == TEST_CELL_ID; // NOT CellId(0)
130                        let caps_ok = CapSet::of_task(t) == parent_caps;
131                        let allow_ok = t.syscall_allowlist == TEST_ALLOWLIST; // NOT u64::MAX
132                        let pku_ok = t.pku_key == TEST_PKU_KEY && t.pku_value == TEST_PKU_VALUE;
133                        if !(cell_ok && caps_ok && allow_ok && pku_ok) {
134                            ok = false;
135                            log::error!(
136                                "[selftest] THREAD-INHERIT: FAIL \
137                                cell={} caps={} allow={} pku={}",
138                                cell_ok,
139                                caps_ok,
140                                allow_ok,
141                                pku_ok
142                            );
143                        }
144                    } else {
145                        ok = false;
146                        log::error!("[selftest] THREAD-INHERIT: FAIL — thread tid absent");
147                    }
148                }
149                remove(thread_tid);
150            }
151            _ => {
152                ok = false;
153                log::error!(
154                    "[selftest] THREAD-INHERIT: FAIL — spawn returned {:?}",
155                    spawned
156                );
157            }
158        }
159        remove(PARENT_TID);
160    }
161
162    // ── #5: honest revoke (ambient bits refused, lazy bits still work) ──────────
163    {
164        // Caller holds SpawnCap (Gate 1). Target holds an ambient cap (hypervisor)
165        // and an MMIO device bit — neither is truly revocable yet, so revoke must
166        // REFUSE with NotSupported and leave the fields intact.
167        let mut caller = mk_task(PARENT_TID, TEST_CELL_ID);
168        caller.spawn_cap = Some(super::cap::SpawnCap::new());
169        insert(caller);
170
171        let mut target = mk_task(TARGET_TID, 0x1111);
172        target.hypervisor_cap = Some(super::cap::HypervisorCap::new());
173        target.mmio_devices = crate::resource_registry::DEV_GPIO;
174        insert(target);
175
176        // (a) HYPERVISOR bit → NotSupported, cap untouched.
177        let r_hyp = handle_syscall(
178            PARENT_TID,
179            Syscall::CapRevoke {
180                target_tid: TARGET_TID,
181                cap_mask: CM::HYPERVISOR,
182            },
183        );
184        // (b) MMIO bits → NotSupported, device mask untouched.
185        let r_mmio = handle_syscall(
186            PARENT_TID,
187            Syscall::CapRevoke {
188                target_tid: TARGET_TID,
189                cap_mask: CM::MMIO_MASK,
190            },
191        );
192
193        let refused = matches!(r_hyp, Err(SyscallError::NotSupported))
194            && matches!(r_mmio, Err(SyscallError::NotSupported));
195        let intact = if let Some(sched) = super::SCHEDULER.lock().as_ref() {
196            sched
197                .tasks
198                .get(&TARGET_TID)
199                .map(|t| {
200                    t.hypervisor_cap.is_some()
201                        && t.mmio_devices == crate::resource_registry::DEV_GPIO
202                })
203                .unwrap_or(false)
204        } else {
205            false
206        };
207        if !(refused && intact) {
208            ok = false;
209            log::error!(
210                "[selftest] REVOKE-REFUSE: FAIL refused={} intact={} (hyp={:?} mmio={:?})",
211                refused,
212                intact,
213                r_hyp,
214                r_mmio
215            );
216        }
217
218        // (c) Positive control: SPAWN is a lazy (syscall-gated) bit — revoke of a
219        // non-system target must still SUCCEED and clear the field.
220        let mut ctrl = mk_task(CTRL_TID, 0x2222);
221        ctrl.spawn_cap = Some(super::cap::SpawnCap::new());
222        insert(ctrl);
223        let r_spawn = handle_syscall(
224            PARENT_TID,
225            Syscall::CapRevoke {
226                target_tid: CTRL_TID,
227                cap_mask: CM::SPAWN,
228            },
229        );
230        let cleared = if let Some(sched) = super::SCHEDULER.lock().as_ref() {
231            sched
232                .tasks
233                .get(&CTRL_TID)
234                .map(|t| t.spawn_cap.is_none())
235                .unwrap_or(false)
236        } else {
237            false
238        };
239        if !(r_spawn.is_ok() && cleared) {
240            ok = false;
241            log::error!(
242                "[selftest] REVOKE-ALLOW: FAIL r={:?} cleared={}",
243                r_spawn,
244                cleared
245            );
246        }
247
248        remove(PARENT_TID);
249        remove(TARGET_TID);
250        remove(CTRL_TID);
251    }
252
253    // ── thread-spawn DoS bound ─────────────────────────────────────────────────
254    // `Syscall::Spawn` is gated by the syscall allowlist but NOT by SpawnCap, and
255    // each thread costs a contiguous run of STACK_PAGES+1 frames. Before the cap an
256    // unprivileged cell could loop here until the allocator fragmented, at which
257    // point the scheduler's `.expect("OOM Stack")` panicked the kernel — never-die
258    // broken from userspace. The refusal must be an error the caller survives.
259    //
260    // The cell is filled with STACKLESS synthetic tasks: the cap counts live tasks
261    // sharing a CellId, so proving the bound costs 31 Task structs instead of 31
262    // real 65-frame stack allocations.
263    {
264        let mut parent = mk_task(PARENT_TID, DOS_CELL_ID);
265        parent.spawn_cap = Some(super::cap::SpawnCap::new());
266        insert(parent);
267
268        let filler: alloc::vec::Vec<usize> = (0..super::scheduler::MAX_THREADS_PER_CELL - 1)
269            .map(|i| DOS_FILLER_BASE + i)
270            .collect();
271        for &tid in &filler {
272            insert(mk_task(tid, DOS_CELL_ID));
273        }
274
275        // Parent + fillers occupy the whole budget, so the next thread must be
276        // refused — with TryAgain (a recoverable refusal), not a panic, and not a
277        // silent success.
278        let refused = handle_syscall(
279            PARENT_TID,
280            Syscall::Spawn {
281                entry: 0x1000,
282                arg: 0,
283            },
284        );
285        if !matches!(refused, Err(SyscallError::TryAgain)) {
286            ok = false;
287            log::error!(
288                "[selftest] THREAD-CAP-DOS: FAIL — spawn at cap returned {:?}, expected TryAgain",
289                refused
290            );
291            // A spawn that unexpectedly succeeded left a task behind; drop it so the
292            // scheduler is still returned exactly as it was found.
293            if let Ok(stray) = refused {
294                remove(stray);
295            }
296        }
297
298        for tid in filler {
299            remove(tid);
300        }
301        remove(PARENT_TID);
302    }
303
304    // Restore the spawn counter so the test leaves no trace on tid assignment.
305    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
306        sched.next_task_id = n;
307    }
308
309    if ok {
310        log::info!("[selftest] THREAD-CAP: PASS (thread-inherit + honest-revoke + spawn bound)");
311    } else {
312        log::error!("[selftest] THREAD-CAP: FAIL");
313    }
314    ok
315}
