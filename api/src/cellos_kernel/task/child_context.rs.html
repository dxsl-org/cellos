1//! Immutable, bounded launch context validation before scheduler publication.
2use api::cell_session::{ObjectRefV1, Reader, Status, CONTEXT_MAX, CONTEXT_MAX_GRANTS,
3    GRANT_KIND_INPUT_FRONTEND, SLOT_FRONTEND};
4use core::ops::Range;
5
6pub(crate) const MAX_LEN: usize = CONTEXT_MAX;
7pub(crate) const MAX_GRANTS: usize = CONTEXT_MAX_GRANTS;
8/// Kind-1 completion-record writer slot (write-only, same rules as stdout).
9pub(crate) const SLOT_COMPLETION: u16 = 5;
10/// Kind-1/slot-0 inspection projection: the four stdio grant records.
11pub(crate) const STDIO_PROJECTION_LEN: usize = 4 * 40;
12#[derive(Clone, Copy, Debug)]
13pub(crate) struct Grant {
14    pub kind: u16,
15    pub rights: u16,
16    pub slot: u16,
17    pub owner_service: u64,
18    pub object: ObjectRefV1,
19    pub wire: [u8; 40],
20}
21pub(crate) struct Context {
22    pub argv: Range<usize>,
23    pub cwd: Range<usize>,
24    /// Indexed by slot: 0..=3 kernel endpoints (stdio), 4 VFS cwd, 5 kernel
25    /// completion writer, 6..=7 reserved, 8 input frontend.
26    pub grants: [Option<Grant>; MAX_GRANTS],
27    pub digest: [u8; 32],
28}
29
30pub(crate) fn validate(bytes: &[u8], operation: ObjectRefV1) -> Result<Context, Status> {
31    if !(48..=MAX_LEN).contains(&bytes.len()) { return Err(Status::Malformed); }
32    let mut r = Reader::new(bytes);
33    if r.u32()? != 1 { return Err(Status::UnsupportedVersion); }
34    if r.u32()? as usize != bytes.len() || r.u32()? != 0 { return Err(Status::Malformed); }
35    let argv_len = r.u32()? as usize;
36    let env_count = r.u16()? as usize;
37    let grant_count = r.u16()? as usize;
38    let cwd_len = r.u32()? as usize;
39    let env_len = r.u32()? as usize;
40    if r.u32()? != 0 || r.reference()? != operation || argv_len > 512 || cwd_len > 4096
41        || env_count > 64 || env_len > 4096 || grant_count > MAX_GRANTS { return Err(Status::Malformed); }
42    let argv = r.take(argv_len)?;
43    // The prefix is the established structured argv carrier, with no argv[0].
44    if !argv.is_empty() {
45        if !argv.starts_with(b"\0argv1\0") || argv.last() != Some(&0) { return Err(Status::Malformed); }
46        for arg in argv[7..].split_inclusive(|b| *b == 0) {
47            if arg.last() != Some(&0) || core::str::from_utf8(&arg[..arg.len()-1]).is_err() {
48                return Err(Status::Malformed);
49            }
50        }
51    }
52    let cwd = r.take(cwd_len)?;
53    if cwd.contains(&0) { return Err(Status::Malformed); }
54    let mut env = Reader::new(r.take(env_len)?);
55    let mut previous: Option<&[u8]> = None;
56    for _ in 0..env_count {
57        let kl = env.u16()? as usize;
58        let vl = env.u16()? as usize;
59        let key = env.take(kl)?;
60        let value = env.take(vl)?;
61        if key.is_empty() || key.contains(&0) || key.contains(&b'=') || value.contains(&0)
62            || core::str::from_utf8(key).is_err() || core::str::from_utf8(value).is_err()
63            || previous.is_some_and(|p| p >= key) { return Err(Status::Malformed); }
64        previous = Some(key);
65    }
66    env.finish()?;
67    let mut grants = [None; MAX_GRANTS];
68    for _ in 0..grant_count {
69        let wire: [u8; 40] = r.take(40)?.try_into().map_err(|_| Status::Malformed)?;
70        let mut g = Reader::new(&wire);
71        let kind = g.u16()?;
72        let rights = g.u16()?;
73        let slot = g.u16()?;
74        if g.u16()? != 0 { return Err(Status::Malformed); }
75        let owner_service = g.u64()?;
76        let object = g.reference()?;
77        if g.u64()? != 0 || object.id == 0 || object.generation == 0 || slot as usize >= MAX_GRANTS
78            || grants[slot as usize].is_some() { return Err(Status::Malformed); }
79        match kind {
80            1 if (slot <= 3 || slot == SLOT_COMPLETION) && owner_service == 0 && rights & !31 == 0
81                && (if slot == 0 || slot == 3 { rights & 3 == 1 } else { rights & 3 == 2 }) => {},
82            2 if slot == 4 && rights == 0 && owner_service == api::syscall::service::VFS as u64 => {},
83            3..=5 => return Err(Status::Unsupported),
84            GRANT_KIND_INPUT_FRONTEND if slot == SLOT_FRONTEND && rights == 0
85                && owner_service == api::syscall::service::INPUT as u64 => {},
86            _ => return Err(Status::Malformed),
87        }
88        grants[slot as usize] = Some(Grant { kind, rights, slot, owner_service, object, wire });
89    }
90    r.finish()?;
91    if cwd_len != 0 && grants[4].is_none() { return Err(Status::Denied); }
92    Ok(Context { argv: 48..48+argv_len, cwd: 48+argv_len..48+argv_len+cwd_len,
93        grants, digest: crate::sha256::sha256(bytes) })
94}

96impl Context {
97    /// The stdio projection input inspects (kind 1, slot 0): exactly the four
98    /// stdio grant records of slots 0..=3, never the completion or later slots.
99    pub(crate) fn stdio_projection(&self) -> Result<[u8; STDIO_PROJECTION_LEN], Status> {
100        let mut stdio = [0u8; STDIO_PROJECTION_LEN];
101        for (slot, bytes) in stdio.chunks_exact_mut(40).enumerate() {
102            bytes.copy_from_slice(&self.grants[slot].ok_or(Status::Denied)?.wire);
103        }
104        Ok(stdio)
105    }
106}

114    const OPERATION: ObjectRefV1 = ObjectRefV1 { id: 7, generation: 1 };
115    const OBJECT: ObjectRefV1 = ObjectRefV1 { id: 9, generation: 2 };
116
117    fn grant(kind: u16, rights: u16, slot: u16, owner: u64) -> [u8; 40] {
118        let mut g = [0u8; 40];
119        g[0..2].copy_from_slice(&kind.to_le_bytes());
120        g[2..4].copy_from_slice(&rights.to_le_bytes());
121        g[4..6].copy_from_slice(&slot.to_le_bytes());
122        g[8..16].copy_from_slice(&owner.to_le_bytes());
123        g[16..32].copy_from_slice(&OBJECT.to_bytes());
124        g
125    }
126    fn context(grants: &[[u8; 40]]) -> Vec<u8> {
127        let len = 48 + grants.len() * 40;
128        let mut bytes = Vec::new();
129        for word in [1u32, len as u32, 0, 0] { bytes.extend_from_slice(&word.to_le_bytes()); }
130        bytes.extend_from_slice(&0u16.to_le_bytes());
131        bytes.extend_from_slice(&(grants.len() as u16).to_le_bytes());
132        for word in [0u32, 0, 0] { bytes.extend_from_slice(&word.to_le_bytes()); }
133        bytes.extend_from_slice(&OPERATION.to_bytes());
134        for g in grants { bytes.extend_from_slice(g); }
135        bytes
136    }
137    fn check(grants: &[[u8; 40]]) -> Result<Context, Status> { validate(&context(grants), OPERATION) }

139    #[test]
140    fn frontend_grant_is_kind_6_in_slot_8_only() {
141        let parsed = check(&[grant(6, 0, 8, INPUT as u64)]).expect("kind 6 slot 8");
142        let g = parsed.grants[8].expect("slot 8 populated");
143        assert_eq!((g.kind, g.slot, g.rights, g.object), (6, 8, 0, OBJECT));
144        assert!(parsed.grants[..8].iter().all(Option::is_none));
145        // Wrong rights, owner or slot for kind 6.
146        assert_eq!(check(&[grant(6, 1, 8, INPUT as u64)]).err(), Some(Status::Malformed));
147        assert_eq!(check(&[grant(6, 0, 8, VFS as u64)]).err(), Some(Status::Malformed));
148        assert_eq!(check(&[grant(6, 0, 8, 0)]).err(), Some(Status::Malformed));
149        for slot in 0..8 {
150            assert_eq!(check(&[grant(6, 0, slot, INPUT as u64)]).err(), Some(Status::Malformed), "slot {slot}");
151        }
152        // Slot 8 admits nothing but kind 6; slots past 8 do not exist.
153        assert_eq!(check(&[grant(1, 1, 8, 0)]).err(), Some(Status::Malformed));
154        assert_eq!(check(&[grant(1, 2, 8, 0)]).err(), Some(Status::Malformed));
155        assert_eq!(check(&[grant(2, 0, 8, VFS as u64)]).err(), Some(Status::Malformed));
156        assert_eq!(check(&[grant(6, 0, 9, INPUT as u64)]).err(), Some(Status::Malformed));
157        assert_eq!(check(&[grant(7, 0, 8, INPUT as u64)]).err(), Some(Status::Malformed));
158        // Duplicate slot 8 is refused.
159        assert_eq!(check(&[grant(6, 0, 8, INPUT as u64), grant(6, 0, 8, INPUT as u64)]).err(), Some(Status::Malformed));
160    }

188    #[test]
189    fn completion_grant_is_kind_1_write_only_in_slot_5() {
190        for rights in [2u16, 6, 10, 18, 26, 30] {
191            let parsed = check(&[grant(1, rights, 5, 0)]).expect("kind 1 slot 5 writer");
192            let g = parsed.grants[5].expect("slot 5 populated");
193            assert_eq!((g.kind, g.slot, g.rights, g.object), (1, 5, rights, OBJECT));
194            assert!(parsed.grants.iter().enumerate().all(|(slot, g)| slot == 5 || g.is_none()));
195        }
196        // Read, bidirectional, directionless and out-of-mask rights are refused.
197        for rights in [0u16, 1, 3, 17, 29, 31, 34, 0x8002] {
198            assert_eq!(check(&[grant(1, rights, 5, 0)]).err(), Some(Status::Malformed), "rights {rights}");
199        }
200        // Only the kernel owns endpoint grants; kind 2 and kind 6 are not slot-5 kinds.
201        assert_eq!(check(&[grant(1, 2, 5, VFS as u64)]).err(), Some(Status::Malformed));
202        assert_eq!(check(&[grant(1, 2, 5, INPUT as u64)]).err(), Some(Status::Malformed));
203        assert_eq!(check(&[grant(2, 0, 5, VFS as u64)]).err(), Some(Status::Malformed));
204        assert_eq!(check(&[grant(6, 0, 5, INPUT as u64)]).err(), Some(Status::Malformed));
205        // Slots 6..=7 stay reserved for every endpoint direction.
206        for slot in 6..=7 {
207            for rights in [1u16, 2, 18, 30] {
208                assert_eq!(check(&[grant(1, rights, slot, 0)]).err(), Some(Status::Malformed), "slot {slot}");
209            }
210        }
211        // One completion writer per context.
212        assert_eq!(check(&[grant(1, 2, 5, 0), at(grant(1, 2, 5, 0), 40)]).err(), Some(Status::Malformed));
213    }

215    #[test]
216    fn stdio_projection_is_exactly_slots_0_to_3() {
217        let stdio = [at(grant(1, 17, 0, 0), 20), at(grant(1, 30, 1, 0), 21), at(grant(1, 30, 2, 0), 22),
218            at(grant(1, 17, 3, 0), 23)];
219        let plain = check(&stdio).expect("stdio only").stdio_projection().expect("projection");
220        assert_eq!(plain.len(), 160);
221        assert_eq!(plain.as_slice(), stdio.concat().as_slice());
222        // The completion writer, listed first or last and with the frontend, never
223        // enters the projection input inspects.
224        let completion = at(grant(1, 18, 5, 0), 25);
225        let frontend = at(grant(6, 0, 8, INPUT as u64), 26);
226        let mut first = Vec::from([completion]);
227        first.extend_from_slice(&stdio);
228        let mut last = Vec::from(stdio);
229        last.extend_from_slice(&[frontend, completion]);
230        for grants in [first, last] {
231            let parsed = check(&grants).expect("stdio + completion");
232            assert_eq!(parsed.grants[5].map(|g| g.object.id), Some(25));
233            assert_eq!(parsed.stdio_projection().expect("projection"), plain);
234        }
235        // A completion writer does not stand in for a missing stdio grant.
236        let partial = check(&[stdio[0], stdio[1], stdio[2], completion]).expect("three stdio + completion");
237        assert_eq!(partial.stdio_projection().err(), Some(Status::Denied));
238    }

241    fn nine_grants_fit_and_ten_do_not() {
242        let full = [grant(1, 1, 0, 0), grant(1, 2, 1, 0), grant(1, 2, 2, 0), grant(1, 1, 3, 0),
243            grant(6, 0, 8, INPUT as u64)];
244        let parsed = check(&full).expect("stdio + frontend");
245        assert!(parsed.grants[..4].iter().all(Option::is_some) && parsed.grants[8].is_some());
246        // A header claiming ten grants is refused before any grant is read.
247        let mut ten = context(&[grant(6, 0, 8, INPUT as u64); 10]);
248        assert_eq!(validate(&ten, OPERATION).err(), Some(Status::Malformed));
249        ten.truncate(48 + 9 * 40);
250        let len = ten.len() as u32;
251        ten[4..8].copy_from_slice(&len.to_le_bytes());
252        ten[18..20].copy_from_slice(&9u16.to_le_bytes());
253        // Nine records, but duplicated slot 8: still refused by slot uniqueness.
254        assert_eq!(validate(&ten, OPERATION).err(), Some(Status::Malformed));
255        assert_eq!(MAX_LEN, 9112);
