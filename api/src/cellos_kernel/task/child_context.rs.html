1//! Immutable, bounded launch context validation before scheduler publication.
2use abi::cell_session::{
3    ObjectRefV1, Reader, Status, CONTEXT_MAX, CONTEXT_MAX_GRANTS, GRANT_KIND_INPUT_FRONTEND,
4    SLOT_FRONTEND,
5};
6use core::ops::Range;
7
8pub(crate) const MAX_LEN: usize = CONTEXT_MAX;
9pub(crate) const MAX_GRANTS: usize = CONTEXT_MAX_GRANTS;
10/// Kind-1 completion-record writer slot (write-only, same rules as stdout).
11pub(crate) const SLOT_COMPLETION: u16 = 5;
12/// Kind-1/slot-0 inspection projection: the four stdio grant records.
13pub(crate) const STDIO_PROJECTION_LEN: usize = 4 * 40;
14#[derive(Clone, Copy, Debug)]
15pub(crate) struct Grant {
16    pub kind: u16,
17    pub rights: u16,
18    #[allow(dead_code)] // decoded from the grant wire; no reader yet
19    pub slot: u16,
20    #[allow(dead_code)] // decoded from the grant wire; no reader yet
21    pub owner_service: u64,
22    pub object: ObjectRefV1,
23    pub wire: [u8; 40],
24}
25pub(crate) struct Context {
26    pub argv: Range<usize>,
27    pub cwd: Range<usize>,
28    /// Indexed by slot: 0..=3 kernel endpoints (stdio), 4 VFS cwd, 5 kernel
29    /// completion writer, 6..=7 reserved, 8 input frontend.
30    pub grants: [Option<Grant>; MAX_GRANTS],
31    pub digest: [u8; 32],
32}
33
34pub(crate) fn validate(bytes: &[u8], operation: ObjectRefV1) -> Result<Context, Status> {
35    if !(48..=MAX_LEN).contains(&bytes.len()) {
36        return Err(Status::Malformed);
37    }
38    let mut r = Reader::new(bytes);
39    if r.u32()? != 1 {
40        return Err(Status::UnsupportedVersion);
41    }
42    if r.u32()? as usize != bytes.len() || r.u32()? != 0 {
43        return Err(Status::Malformed);
44    }
45    let argv_len = r.u32()? as usize;
46    let env_count = r.u16()? as usize;
47    let grant_count = r.u16()? as usize;
48    let cwd_len = r.u32()? as usize;
49    let env_len = r.u32()? as usize;
50    if r.u32()? != 0
51        || r.reference()? != operation
52        || argv_len > 512
53        || cwd_len > 4096
54        || env_count > 64
55        || env_len > 4096
56        || grant_count > MAX_GRANTS
57    {
58        return Err(Status::Malformed);
59    }
60    let argv = r.take(argv_len)?;
61    // The prefix is the established structured argv carrier, with no argv[0].
62    if !argv.is_empty() {
63        if !argv.starts_with(b"\0argv1\0") || argv.last() != Some(&0) {
64            return Err(Status::Malformed);
65        }
66        for arg in argv[7..].split_inclusive(|b| *b == 0) {
67            if arg.last() != Some(&0) || core::str::from_utf8(&arg[..arg.len() - 1]).is_err() {
68                return Err(Status::Malformed);
69            }
70        }
71    }
72    let cwd = r.take(cwd_len)?;
73    if cwd.contains(&0) {
74        return Err(Status::Malformed);
75    }
76    let mut env = Reader::new(r.take(env_len)?);
77    let mut previous: Option<&[u8]> = None;
78    for _ in 0..env_count {
79        let kl = env.u16()? as usize;
80        let vl = env.u16()? as usize;
81        let key = env.take(kl)?;
82        let value = env.take(vl)?;
83        if key.is_empty()
84            || key.contains(&0)
85            || key.contains(&b'=')
86            || value.contains(&0)
87            || core::str::from_utf8(key).is_err()
88            || core::str::from_utf8(value).is_err()
89            || previous.is_some_and(|p| p >= key)
90        {
91            return Err(Status::Malformed);
92        }
93        previous = Some(key);
94    }
95    env.finish()?;
96    let mut grants = [None; MAX_GRANTS];
97    for _ in 0..grant_count {
98        let wire: [u8; 40] = r.take(40)?.try_into().map_err(|_| Status::Malformed)?;
99        let mut g = Reader::new(&wire);
100        let kind = g.u16()?;
101        let rights = g.u16()?;
102        let slot = g.u16()?;
103        if g.u16()? != 0 {
104            return Err(Status::Malformed);
105        }
106        let owner_service = g.u64()?;
107        let object = g.reference()?;
108        if g.u64()? != 0
109            || object.id == 0
110            || object.generation == 0
111            || slot as usize >= MAX_GRANTS
112            || grants[slot as usize].is_some()
113        {
114            return Err(Status::Malformed);
115        }
116        match kind {
117            1 if (slot <= 3 || slot == SLOT_COMPLETION)
118                && owner_service == 0
119                && rights & !31 == 0
120                && (if slot == 0 || slot == 3 {
121                    rights & 3 == 1
122                } else {
123                    rights & 3 == 2
124                }) => {}
125            2 if slot == 4 && rights == 0 && owner_service == abi::syscall::service::VFS as u64 => {
126            }
127            3..=5 => return Err(Status::Unsupported),
128            GRANT_KIND_INPUT_FRONTEND
129                if slot == SLOT_FRONTEND
130                    && rights == 0
131                    && owner_service == abi::syscall::service::INPUT as u64 => {}
132            _ => return Err(Status::Malformed),
133        }
134        grants[slot as usize] = Some(Grant {
135            kind,
136            rights,
137            slot,
138            owner_service,
139            object,
140            wire,
141        });
142    }
143    r.finish()?;
144    if cwd_len != 0 && grants[4].is_none() {
145        return Err(Status::Denied);
146    }
147    Ok(Context {
148        argv: 48..48 + argv_len,
149        cwd: 48 + argv_len..48 + argv_len + cwd_len,
150        grants,
151        digest: crate::sha256::sha256(bytes),
152    })
153}

155impl Context {
156    /// The stdio projection input inspects (kind 1, slot 0): exactly the four
157    /// stdio grant records of slots 0..=3, never the completion or later slots.
158    pub(crate) fn stdio_projection(&self) -> Result<[u8; STDIO_PROJECTION_LEN], Status> {
159        let mut stdio = [0u8; STDIO_PROJECTION_LEN];
160        for (slot, bytes) in stdio.chunks_exact_mut(40).enumerate() {
161            bytes.copy_from_slice(&self.grants[slot].ok_or(Status::Denied)?.wire);
162        }
163        Ok(stdio)
164    }
165}

173    const OPERATION: ObjectRefV1 = ObjectRefV1 {
174        id: 7,
175        generation: 1,
176    };
177    const OBJECT: ObjectRefV1 = ObjectRefV1 {
178        id: 9,
179        generation: 2,
180    };
181
182    fn grant(kind: u16, rights: u16, slot: u16, owner: u64) -> [u8; 40] {
183        let mut g = [0u8; 40];
184        g[0..2].copy_from_slice(&kind.to_le_bytes());
185        g[2..4].copy_from_slice(&rights.to_le_bytes());
186        g[4..6].copy_from_slice(&slot.to_le_bytes());
187        g[8..16].copy_from_slice(&owner.to_le_bytes());
188        g[16..32].copy_from_slice(&OBJECT.to_bytes());
189        g
190    }
191    fn context(grants: &[[u8; 40]]) -> Vec<u8> {
192        let len = 48 + grants.len() * 40;
193        let mut bytes = Vec::new();
194        for word in [1u32, len as u32, 0, 0] {
195            bytes.extend_from_slice(&word.to_le_bytes());
196        }
197        bytes.extend_from_slice(&0u16.to_le_bytes());
198        bytes.extend_from_slice(&(grants.len() as u16).to_le_bytes());
199        for word in [0u32, 0, 0] {
200            bytes.extend_from_slice(&word.to_le_bytes());
201        }
202        bytes.extend_from_slice(&OPERATION.to_bytes());
203        for g in grants {
204            bytes.extend_from_slice(g);
205        }
206        bytes
207    }
208    fn check(grants: &[[u8; 40]]) -> Result<Context, Status> {
209        validate(&context(grants), OPERATION)
210    }

214        let parsed = check(&[grant(6, 0, 8, INPUT as u64)]).expect("kind 6 slot 8");
215        let g = parsed.grants[8].expect("slot 8 populated");
216        assert_eq!((g.kind, g.slot, g.rights, g.object), (6, 8, 0, OBJECT));
217        assert!(parsed.grants[..8].iter().all(Option::is_none));

301    fn completion_grant_is_kind_1_write_only_in_slot_5() {
302        for rights in [2u16, 6, 10, 18, 26, 30] {
303            let parsed = check(&[grant(1, rights, 5, 0)]).expect("kind 1 slot 5 writer");
304            let g = parsed.grants[5].expect("slot 5 populated");
305            assert_eq!((g.kind, g.slot, g.rights, g.object), (1, 5, rights, OBJECT));
306            assert!(parsed
307                .grants
308                .iter()
309                .enumerate()
310                .all(|(slot, g)| slot == 5 || g.is_none()));
311        }
312        // Read, bidirectional, directionless and out-of-mask rights are refused.
313        for rights in [0u16, 1, 3, 17, 29, 31, 34, 0x8002] {
314            assert_eq!(
315                check(&[grant(1, rights, 5, 0)]).err(),
316                Some(Status::Malformed),
317                "rights {rights}"
318            );
319        }
320        // Only the kernel owns endpoint grants; kind 2 and kind 6 are not slot-5 kinds.
321        assert_eq!(
322            check(&[grant(1, 2, 5, VFS as u64)]).err(),
323            Some(Status::Malformed)
324        );
325        assert_eq!(
326            check(&[grant(1, 2, 5, INPUT as u64)]).err(),
327            Some(Status::Malformed)
328        );
329        assert_eq!(
330            check(&[grant(2, 0, 5, VFS as u64)]).err(),
331            Some(Status::Malformed)
332        );
333        assert_eq!(
334            check(&[grant(6, 0, 5, INPUT as u64)]).err(),
335            Some(Status::Malformed)
336        );
337        // Slots 6..=7 stay reserved for every endpoint direction.
338        for slot in 6..=7 {
339            for rights in [1u16, 2, 18, 30] {
340                assert_eq!(
341                    check(&[grant(1, rights, slot, 0)]).err(),
342                    Some(Status::Malformed),
343                    "slot {slot}"
344                );
345            }
346        }
347        // One completion writer per context.
348        assert_eq!(
349            check(&[grant(1, 2, 5, 0), at(grant(1, 2, 5, 0), 40)]).err(),
350            Some(Status::Malformed)
351        );
352    }

354    #[test]
355    fn stdio_projection_is_exactly_slots_0_to_3() {
356        let stdio = [
357            at(grant(1, 17, 0, 0), 20),
358            at(grant(1, 30, 1, 0), 21),
359            at(grant(1, 30, 2, 0), 22),
360            at(grant(1, 17, 3, 0), 23),
361        ];
362        let plain = check(&stdio)
363            .expect("stdio only")
364            .stdio_projection()
365            .expect("projection");
366        assert_eq!(plain.len(), 160);
367        assert_eq!(plain.as_slice(), stdio.concat().as_slice());
368        // The completion writer, listed first or last and with the frontend, never
369        // enters the projection input inspects.
370        let completion = at(grant(1, 18, 5, 0), 25);
371        let frontend = at(grant(6, 0, 8, INPUT as u64), 26);
372        let mut first = Vec::from([completion]);
373        first.extend_from_slice(&stdio);
374        let mut last = Vec::from(stdio);
375        last.extend_from_slice(&[frontend, completion]);
376        for grants in [first, last] {
377            let parsed = check(&grants).expect("stdio + completion");
378            assert_eq!(parsed.grants[5].map(|g| g.object.id), Some(25));
379            assert_eq!(parsed.stdio_projection().expect("projection"), plain);
380        }
381        // A completion writer does not stand in for a missing stdio grant.
382        let partial =
383            check(&[stdio[0], stdio[1], stdio[2], completion]).expect("three stdio + completion");
384        assert_eq!(partial.stdio_projection().err(), Some(Status::Denied));
385    }

387    #[test]
388    fn nine_grants_fit_and_ten_do_not() {
389        let full = [
390            grant(1, 1, 0, 0),
391            grant(1, 2, 1, 0),
392            grant(1, 2, 2, 0),
393            grant(1, 1, 3, 0),
394            grant(6, 0, 8, INPUT as u64),
395        ];
396        let parsed = check(&full).expect("stdio + frontend");
397        assert!(parsed.grants[..4].iter().all(Option::is_some) && parsed.grants[8].is_some());
398        // A header claiming ten grants is refused before any grant is read.
399        let mut ten = context(&[grant(6, 0, 8, INPUT as u64); 10]);
400        assert_eq!(validate(&ten, OPERATION).err(), Some(Status::Malformed));
401        ten.truncate(48 + 9 * 40);
402        let len = ten.len() as u32;
403        ten[4..8].copy_from_slice(&len.to_le_bytes());
404        ten[18..20].copy_from_slice(&9u16.to_le_bytes());
405        // Nine records, but duplicated slot 8: still refused by slot uniqueness.
406        assert_eq!(validate(&ten, OPERATION).err(), Some(Status::Malformed));
407        assert_eq!(MAX_LEN, 9112);
408    }
