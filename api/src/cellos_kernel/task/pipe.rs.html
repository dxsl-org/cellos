1//! Kernel-owned pipes — one-way byte streams between tasks (ADR-0018 §2.1).
2//!
3//! A pipe is a fixed-capacity ring with two endpoint kinds (read/write). Handles
4//! live in the owning task's table, so a handle is usable by exactly the task that
5//! holds it: `PipeShare` duplicates one end into another task's table, and nothing
6//! else moves it. That is what makes a handle capability-like without adding an
7//! allowlist bit — the authority is ownership, and a task cannot share what it does
8//! not own.
9//!
10//! Semantics, chosen to match what a ported `pipe(2)`/`popen(3)` needs:
11//!
12//! * **Backpressure is real.** `PipeWrite` copies at most what fits and returns the
13//!   count; when the ring is full it parks until a reader drains it (or the deadline
14//!   expires). The ring never grows past its creation capacity.
15//! * **EOF is exact.** A zero-length write closes the writer end. Readers observe
16//!   `0` bytes only when the ring is empty *and* no writer end remains — including
17//!   when the last writer died, exited, or faulted, because task teardown closes its
18//!   endpoints.
19//! * **No post-close reads.** Once an end is closed, its handle is gone; the data
20//!   already in the ring stays readable until drained, which is what a pipe does.
21//! * **Deadlines, not poll.** Every blocking call takes a deadline so a cell can run
22//!   an event loop without `poll(2)`; a wait that times out returns 0 bytes for a
23//!   read and 0 for a write, and the caller re-checks.
24//!
25//! Locking: `PIPES` is a leaf lock taken under `SCHEDULER` (same order as the futex
26//! wait queues and the ready queues), never the other way round.
27
28use crate::sync::Spinlock;
29use alloc::collections::{BTreeMap, VecDeque};
30use alloc::vec::Vec;

32/// Default ring capacity when a caller asks for 0.
33pub(crate) const DEFAULT_CAPACITY: usize = 4096;
34/// Smallest ring worth creating.
35pub(crate) const MIN_CAPACITY: usize = 64;
36/// Largest ring: the whole pipe is kernel memory charged to the creator's quota, so
37/// this is a per-pipe ceiling, not a policy on how many pipes a cell may hold.
38pub(crate) const MAX_CAPACITY: usize = 64 * 1024;
39
40/// Which end of a pipe a handle names.
41#[derive(Clone, Copy, Debug, Eq, PartialEq, Ord, PartialOrd)]
42pub enum PipeEnd {
43    Read,
44    Write,
45}

47/// A handle as the ABI carries it: pipe identity plus end kind.
48#[derive(Clone, Copy, Debug, Eq, PartialEq, Ord, PartialOrd)]
49pub struct PipeHandle {
50    pub(crate) pipe: usize,
51    pub(crate) end: PipeEnd,
52}
53
54struct Pipe {
55    buffer: Vec<u8>,
56    head: usize,
57    len: usize,
58    readers: usize,
59    writers: usize,
60    read_waiters: VecDeque<usize>,
61    write_waiters: VecDeque<usize>,
62    /// Cell charged for the ring; the charge is refunded when the pipe is dropped.
63    owner_cell: usize,
64}
65
66impl Pipe {
67    fn capacity(&self) -> usize {
68        self.buffer.len()
69    }
70
71    fn is_empty(&self) -> bool {
72        self.len == 0
73    }
74
75    fn is_full(&self) -> bool {
76        self.len == self.buffer.len()
77    }
78
79    /// Copy `src` into the ring, returning how many bytes fit.
80    fn write(&mut self, src: &[u8]) -> usize {
81        let capacity = self.capacity();
82        let space = capacity - self.len;
83        let count = space.min(src.len());
84        for (offset, byte) in src.iter().take(count).enumerate() {
85            let index = (self.head + self.len + offset) % capacity;
86            self.buffer[index] = *byte;
87        }
88        self.len += count;
89        count
90    }
91
92    /// Copy up to `dst.len()` bytes out of the ring.
93    fn read(&mut self, dst: &mut [u8]) -> usize {
94        let capacity = self.capacity();
95        let count = self.len.min(dst.len());
96        for (offset, slot) in dst.iter_mut().take(count).enumerate() {
97            let index = (self.head + offset) % capacity;
98            *slot = self.buffer[index];
99        }
100        self.head = (self.head + count) % capacity;
101        self.len -= count;
102        count
103    }
104}

106static PIPES: Spinlock<BTreeMap<usize, Pipe>> = Spinlock::new(BTreeMap::new());
107static NEXT_PIPE_ID: Spinlock<usize> = Spinlock::new(1);
108
109/// Pipe rings are kernel objects. Charge their capacity explicitly to the
110/// creating cell, but allocate and free their storage as kernel memory so the
111/// allocator never silently adds a second per-cell charge or refunds a peer
112/// that happens to close the last end.
113struct PipeAllocationContext {
114    previous_cell: usize,
115}
116
117impl PipeAllocationContext {
118    fn enter() -> Self {
119        let previous_cell = crate::task::hart_local::current_cell_id();
120        crate::task::hart_local::set_current_cell_id(0);
121        Self { previous_cell }
122    }
123}
124
125impl Drop for PipeAllocationContext {
126    fn drop(&mut self) {
127        crate::task::hart_local::set_current_cell_id(self.previous_cell);
128    }
129}
130
131fn release_pipe(pipe: Pipe) {
132    let owner_cell = pipe.owner_cell;
133    let capacity = pipe.capacity();
134    let _kernel_allocation = PipeAllocationContext::enter();
135    drop(pipe);
136    crate::memory::cell_quota::refund(owner_cell, capacity);
137}
138
139fn next_pipe_id() -> usize {
140    let mut next = NEXT_PIPE_ID.lock();
141    let id = *next;
142    *next = next.saturating_add(1);
143    id
144}

146/// Encode a handle for the ABI: `pipe_id << 1 | end_bit`.
147impl PipeHandle {
148    pub(crate) fn to_raw(self) -> usize {
149        (self.pipe << 1) | matches!(self.end, PipeEnd::Write) as usize
150    }
151
152    /// Decode an ABI handle. A zero value is not a valid handle (pipe ids start
153    /// at 1), so it is rejected here rather than looked up.
154    pub(crate) fn from_raw(raw: usize) -> Option<Self> {
155        let pipe = raw >> 1;
156        if pipe == 0 {
157            return None;
158        }
159        Some(Self {
160            pipe,
161            end: if raw & 1 == 1 {
162                PipeEnd::Write
163            } else {
164                PipeEnd::Read
165            },
166        })
167    }
168}

170/// Create a pipe. The ring capacity is charged exactly once to `owner_cell`.
171pub(crate) fn create(
172    capacity: usize,
173    owner_cell: usize,
174) -> Result<(PipeHandle, PipeHandle), crate::task::syscall::SyscallError> {
175    let capacity = if capacity == 0 {
176        DEFAULT_CAPACITY
177    } else {
178        capacity.clamp(MIN_CAPACITY, MAX_CAPACITY)
179    };
180    if !crate::memory::cell_quota::charge(owner_cell, capacity) {
181        return Err(crate::task::syscall::SyscallError::OutOfMemory);
182    }
183
184    let _kernel_allocation = PipeAllocationContext::enter();
185    let mut buffer = Vec::new();
186    if buffer.try_reserve_exact(capacity).is_err() {
187        drop(_kernel_allocation);
188        crate::memory::cell_quota::refund(owner_cell, capacity);
189        return Err(crate::task::syscall::SyscallError::OutOfMemory);
190    }
191    buffer.resize(capacity, 0);
192    let pipe = Pipe {
193        buffer,
194        head: 0,
195        len: 0,
196        readers: 1,
197        writers: 1,
198        read_waiters: VecDeque::new(),
199        write_waiters: VecDeque::new(),
200        owner_cell,
201    };
202    let id = next_pipe_id();
203    PIPES.lock().insert(id, pipe);
204    Ok((
205        PipeHandle {
206            pipe: id,
207            end: PipeEnd::Read,
208        },
209        PipeHandle {
210            pipe: id,
211            end: PipeEnd::Write,
212        },
213    ))
214}

216/// Duplicate an existing end for another task. The endpoint count—not merely
217/// object liveness—defines EOF and broken-pipe behavior, so every shared handle
218/// must be represented here before it is installed in a task table.
219pub(crate) fn duplicate_end(handle: PipeHandle) -> bool {
220    let mut pipes = PIPES.lock();
221    let Some(pipe) = pipes.get_mut(&handle.pipe) else {
222        return false;
223    };
224    match handle.end {
225        PipeEnd::Read => pipe.readers = pipe.readers.saturating_add(1),
226        PipeEnd::Write => pipe.writers = pipe.writers.saturating_add(1),
227    }
228    true
229}

231/// Close one end. Returns true when the pipe object was dropped (last end).
232///
233/// Closing the last writer end makes readers observe EOF; closing the last reader
234/// end makes writers fail with a closed pipe.
235pub(crate) fn close_end(handle: PipeHandle, tid: usize) -> bool {
236    let mut pipes = PIPES.lock();
237    let Some(pipe) = pipes.get_mut(&handle.pipe) else {
238        return false;
239    };
240    match handle.end {
241        PipeEnd::Read => {
242            pipe.readers = pipe.readers.saturating_sub(1);
243            pipe.read_waiters.retain(|queued| *queued != tid);
244        }
245        PipeEnd::Write => {
246            pipe.writers = pipe.writers.saturating_sub(1);
247            pipe.write_waiters.retain(|queued| *queued != tid);
248        }
249    }
250    // Both ends gone: release the object and refund the ring.
251    if pipe.readers == 0 && pipe.writers == 0 {
252        let pipe = pipes
253            .remove(&handle.pipe)
254            .expect("pipe remained present while holding PIPES");
255        drop(pipes);
256        release_pipe(pipe);
257        return true;
258    }
259    false
260}

262/// Writer ends still attached to this pipe.
263fn writers_of(pipe: &Pipe) -> usize {
264    pipe.writers
265}

267/// Result of a write attempt.
268pub(crate) enum WriteOutcome {
269    /// Bytes accepted (0 with `closed` set means the reader end is gone).
270    Wrote(usize),
271    /// The ring was full; the caller should park and retry.
272    Full,
273    /// No reader end remains: this is the `EPIPE`-shaped case.
274    NoReader,
275}
276
277/// Try to write into the pipe.
278pub(crate) fn try_write(handle: PipeHandle, src: &[u8], tid: usize) -> Option<WriteOutcome> {
279    let mut pipes = PIPES.lock();
280    let pipe = pipes.get_mut(&handle.pipe)?;
281    if pipe.readers == 0 {
282        pipe.write_waiters.retain(|queued| *queued != tid);
283        return Some(WriteOutcome::NoReader);
284    }
285    if pipe.is_full() {
286        return Some(WriteOutcome::Full);
287    }
288    let written = pipe.write(src);
289    // A reader may now make progress.
290    let waiters: Vec<usize> = pipe.read_waiters.drain(..).collect();
291    drop(pipes);
292    for waiter in waiters {
293        crate::task::wake_task(waiter, 0);
294    }
295    Some(WriteOutcome::Wrote(written))
296}

298/// Result of a read attempt.
299pub(crate) enum ReadOutcome {
300    /// Bytes read; 0 with `eof` set means the stream ended.
301    Read(usize),
302    Eof,
303    /// The ring was empty but a writer end remains: park and retry.
304    Empty,
305}

307/// Try to read from the pipe.
308pub(crate) fn try_read(handle: PipeHandle, dst: &mut [u8], tid: usize) -> Option<ReadOutcome> {
309    let mut pipes = PIPES.lock();
310    let pipe = pipes.get_mut(&handle.pipe)?;
311    if !pipe.is_empty() {
312        let read = pipe.read(dst);
313        // A writer may now make progress.
314        let waiters: Vec<usize> = pipe.write_waiters.drain(..).collect();
315        drop(pipes);
316        for waiter in waiters {
317            crate::task::wake_task(waiter, 0);
318        }
319        return Some(ReadOutcome::Read(read));
320    }
321    if writers_of(pipe) == 0 {
322        pipe.read_waiters.retain(|queued| *queued != tid);
323        return Some(ReadOutcome::Eof);
324    }
325    Some(ReadOutcome::Empty)
326}

328/// Park `tid` as a reader (waiting for data or EOF).
329pub(crate) fn park_reader(handle: PipeHandle, tid: usize) {
330    let mut pipes = PIPES.lock();
331    if let Some(pipe) = pipes.get_mut(&handle.pipe) {
332        if !pipe.read_waiters.contains(&tid) {
333            pipe.read_waiters.push_back(tid);
334        }
335    }
336}

338/// Park `tid` as a writer (waiting for space or a reader).
339pub(crate) fn park_writer(handle: PipeHandle, tid: usize) {
340    let mut pipes = PIPES.lock();
341    if let Some(pipe) = pipes.get_mut(&handle.pipe) {
342        if !pipe.write_waiters.contains(&tid) {
343            pipe.write_waiters.push_back(tid);
344        }
345    }
346}

348/// Drop every waiter entry and endpoint a dying task owned.
349///
350/// Called from task teardown, which is what turns a faulted or exited writer into
351/// an EOF for its readers instead of a stream that hangs forever.
352pub(crate) fn on_task_leaves(tid: usize, owned: &[PipeHandle]) {
353    let mut to_wake = Vec::new();
354    {
355        let mut pipes = PIPES.lock();
356        for handle in owned {
357            let Some(pipe) = pipes.get_mut(&handle.pipe) else {
358                continue;
359            };
360            match handle.end {
361                PipeEnd::Read => {
362                    pipe.readers = pipe.readers.saturating_sub(1);
363                    pipe.read_waiters.retain(|queued| *queued != tid);
364                    if pipe.readers == 0 {
365                        // Writers lose their peer: wake them so they can fail.
366                        to_wake.extend(pipe.write_waiters.drain(..));
367                    }
368                }
369                PipeEnd::Write => {
370                    pipe.writers = pipe.writers.saturating_sub(1);
371                    pipe.write_waiters.retain(|queued| *queued != tid);
372                    if pipe.writers == 0 {
373                        // Readers see EOF: wake them to observe it.
374                        to_wake.extend(pipe.read_waiters.drain(..));
375                    }
376                }
377            }
378        }
379        // Release objects with no ends left.
380        let dead: Vec<usize> = pipes
381            .iter()
382            .filter(|(_, pipe)| pipe.readers == 0 && pipe.writers == 0)
383            .map(|(id, _)| *id)
384            .collect();
385        let released: Vec<Pipe> = dead
386            .into_iter()
387            .filter_map(|id| pipes.remove(&id))
388            .collect();
389        drop(pipes);
390        for pipe in released {
391            release_pipe(pipe);
392        }
393    }
394    for waiter in to_wake {
395        crate::task::wake_task(waiter, 0);
396    }
397}

399/// Deadline sweep arm: a parked reader/writer whose deadline elapsed.
400///
401/// The task is woken with a 0-byte result so the caller re-checks and can run an
402/// event loop without `poll(2)`.
403pub(crate) fn on_deadline(handle: PipeHandle, tid: usize, end: PipeEnd) {
404    let mut pipes = PIPES.lock();
405    if let Some(pipe) = pipes.get_mut(&handle.pipe) {
406        match end {
407            PipeEnd::Read => pipe.read_waiters.retain(|queued| *queued != tid),
408            PipeEnd::Write => pipe.write_waiters.retain(|queued| *queued != tid),
409        }
410    }
