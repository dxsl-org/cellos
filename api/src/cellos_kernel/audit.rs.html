1//! Kernel audit ring buffer — 256 KB in-memory event log.
2//!
3//! Records key kernel events (IPC, file, net, spawn, fault, exit) with a
4//! monotonic timestamp for post-mortem analysis.  A background Cell (or shell
5//! builtin) drains the ring and appends to `/data/kernel.log`.
6//!
7//! # Concurrency
8//! Two things can make a write concurrent with another producer: the timer ISR
9//! preempting a syscall-context `log_event()` call on the same hart, and a
10//! second hart logging at the same time.  S-mode interrupts are disabled for
11//! the duration (which also makes the lock below unreachable from an ISR, so it
12//! cannot self-deadlock), and `WRITE` serialises the hart-to-hart case: without
13//! it two harts read the same `head`, write into the same byte range, and store
14//! the same advanced `head`, so one record is silently overwritten.
15//!
16//! # Overflow
17//! When the ring is full the oldest whole records are evicted, counted in
18//! [`dropped_count`], and reported in-band by a `RecordsDropped` record written
19//! just ahead of the record that forced the eviction. No blocking, no corruption.
20//!
21//! # Record format
22//! ```
23//! [u64 mtime_ticks LE][u8 event_type][u8 payload_len][payload…]
24//! ```
25//! Total minimum: 10 bytes per event.
26
27use core::cell::UnsafeCell;
28use core::sync::atomic::{AtomicUsize, Ordering};
29
30const BUF_SIZE: usize = 256 * 1024; // must be a power of two
31const MASK: usize = BUF_SIZE - 1;
32
33/// Serialises producers. Taken with S-mode interrupts already disabled, so a
34/// same-hart ISR can never wait on it and a holder never blocks an interrupt.
35static WRITE: crate::sync::Spinlock<()> =
36    crate::sync::Spinlock::ranked((), crate::sync::rank::AUDIT_WRITE);
37
38/// Kernel audit event type byte.
39#[repr(u8)]
40#[allow(dead_code)] // reason: variants logged by different kernel subsystems
41pub enum AuditEvent {
42    IpcSend = 1,
43    IpcRecv = 2,
44    FileOpen = 3,
45    FileWrite = 4,
46    NetTx = 5,
47    NetRx = 6,
48    CellSpawn = 7,
49    CellFault = 8,
50    CellExit = 9,
51    /// A launch was refused before any task existed. Governed-spawn refusals
52    /// use [`encode_subject`] with the **spawner** as subject ([`Subject::KERNEL`]
53    /// for boot and trusted-init launches), `d0` = refusal code (1 malformed
54    /// manifest, 2 owner admission, 3 signature unverifiable, 4 unsigned in
55    /// release, 0 privileged manifest outside `/bin`), `d1` = manifest flags.
56    /// Domain-admission refusals still use `encode_u32x2(0, audit_code)`.
57    CellSpawnDenied = 10,
58    /// An RT-priority cell's `RecvTimeout` deadline elapsed before its awaited
59    /// message arrived — a missed control-loop cycle. Payload: `encode_u32x2(cell_id,
60    /// cumulative_miss_count)`. Observability only (no enforcement): makes RT misses
61    /// visible for post-mortem / tuning once real-hardware bench data is available.
62    RtDeadlineMiss = 11,
63    /// An RT-priority cell crossed the CPU-monopoly *warning* threshold (a fraction
64    /// of the watchdog budget) without yielding — an early signal BEFORE the hard
65    /// watchdog kill. Payload: `encode_u32x2(cell_id, run_ticks)`. One-shot per episode.
66    RtCpuOverrun = 12,
67    /// A cell that opted into liveness heartbeating missed its deadline — a silent hang
68    /// (deadlock / stuck loop) the CPU watchdog cannot detect. The kernel terminates it
69    /// for supervised restart. Payload: `encode_u32x2(cell_id, tid)`.
70    CellHung = 13,
71    /// A syscall was refused on an authority check. `d0`: the missing
72    /// allowlist bit; `0x100 | bit` when the bit was present but the edge
73    /// withholds its observation right; `u32::MAX` for an unknown opcode; or the
74    /// raw handle for a pipe-ownership refusal.
75    /// Payload: [`encode_subject`] with the caller as subject, `d1` = 0.
76    SyscallDenied = 14,
77    /// A cell's ELF image was measured at spawn (integrity measurement, IMA-style).
78    /// Payload: `encode_u32x2(tid, sha256_prefix_le_u32)`. The full digest and the
79    /// rolling aggregate live in [`crate::measurement_log`].
80    CellMeasure = 15,
81    /// The signed operator policy was loaded + verified at boot (P5).
82    /// Payload: `encode_u32x2(entry_count, 0)`.
83    PolicyLoaded = 16,
84    /// The operator policy failed signature verification or parsing — fail-closed.
85    /// Payload: `encode_u32x2(reason_code, 0)`.
86    PolicyInvalid = 17,
87    /// No operator policy present in VIFS1. Payload: `encode_u32x2(0, 0)`.
88    PolicyAbsent = 18,
89    /// Operator policy narrowed a cell's spawn-time caps. Payload:
90    /// `encode_u32x2(tid, dropped_flags)` (dropped_flags: bit0 block_io, bit1
91    /// network, bit2 spawn, bit3 hypervisor, bit4 pcie_driver, bit5 platform,
92    /// bit6 supervisor, bit7 mmio_devices changed, bit8 block_regions changed,
93    /// bit9 net_resolve, bit10 net_connect, bit11 net_listen, bit12 net_l2,
94    /// bit13 observe_tasks, bit14 observe_memory, bit15 vfs_mutate,
95    /// bit16 config_write, bit17 kms_ssh_sign, bit18 kms_ssh_admin).
96    CapNarrowedByPolicy = 19,
97    /// Runtime revocation: a SpawnCap holder stripped capabilities from a live
98    /// Cell incarnation (or was refused for HYPERVISOR). Payload:
99    /// [`encode_subject`] with the **caller** as subject, `d0` = target tid,
100    /// `d1` = `cap_mask`, where `cap_mask` matches
101    /// `abi::syscall::cap_mask` constants (bit0=block_io, bit1=network, bit2=spawn,
102    /// bit3=hypervisor, bits8-15=mmio_devices, bits16-23=block_regions).
103    CapRevoked = 20,
104    /// Cell binary signature verified at spawn (binary integrity gate).
105    /// Payload: `encode_u32x2(tid, 0)`.
106    CellSignatureVerified = 21,
107    /// Cell binary signature absent or invalid — spawn denied.
108    /// Payload: `encode_u32x2(tid, 0)`. Path is logged separately via `log::warn!`.
109    CellSignatureFailed = 22,
110    /// A cell was granted privileged path authority that survived policy narrowing.
111    /// Payload: `encode_u32x2(tid, mask)` (bit0 pcie_driver, bit1 platform,
112    /// bit2 supervisor, bit3 serial_port). These are the caps that can DMA
113    /// anywhere, orchestrate other cells, or reach hardware the kernel owns, so
114    /// the grant — not only its removal — is auditable.
115    PrivilegedCapGranted = 23,
116    /// The signed maintenance bypass was exercised: policy narrowing was skipped
117    /// for this spawn. Requires BOTH the `maintenance-mode` build feature and the
118    /// `MAINTENANCE_PERMITTED` flag in the signed policy. Payload:
119    /// `encode_u32x2(tid, 0)`; the path is logged separately via `log::warn!`.
120    PolicyMaintenanceBypass = 24,
121    /// A cell hit `MAX_THREADS_PER_CELL` and its `spawn_thread` was refused.
122    /// Payload: `encode_u32x2(cell_id, live_task_count)`. Refusing is the correct
123    /// outcome, but a cell hitting the cap repeatedly is either leaking threads or
124    /// probing for the allocator-fragmentation DoS this cap closes.
125    ///
126    /// Numbered 25 rather than 23: this variant and the two above were authored on
127    /// parallel branches that each claimed 23. The discriminant is the byte written
128    /// into the ring, so a collision would leave two unrelated events
129    /// indistinguishable to anything that decodes the log.
130    ThreadCapReached = 25,
131    /// A present, verified policy had NO entry for a path that mints privileged
132    /// (P-TRUST) authority, so that authority was stripped instead of kept.
133    /// Payload: `encode_u32x2(tid, mask)` (bit0 pcie_driver, bit1 platform,
134    /// bit2 supervisor) — the bits the request carried into the strip; `mask == 0`
135    /// means the spawner held none of them and only the coverage gap remains.
136    ///
137    /// This is a bake-time mistake surfacing at runtime: every such path is
138    /// supposed to be listed in the signed policy, and `scripts/sign-policy.py`
139    /// refuses to build a blob that omits one. Path via `log::warn!`, as for the
140    /// other policy events.
141    PolicyNoEntryStripped = 26,
142    /// A cell left policy holding coarse network rights, which the kernel exports
143    /// to service-net as attested `CALLER_FLAG_NET_*` trailer flags (Phase 04
144    /// S2-N). Payload: `encode_u32x2(tid, mask)` (bit0 net_resolve, bit1
145    /// net_connect, bit2 net_listen, bit3 net_l2). Recorded for the grant, not
146    /// only a narrowing, because `net_l2` puts raw frames on the NIC.
147    NetRightsGranted = 27,
148    /// Synthetic event recorded when ring buffer overflows and older records are evicted.
149    /// Payload: `encode_u32x2(evicted_count, cumulative_dropped)`.
150    RecordsDropped = 28,
151}

153struct AuditRing {
154    buf: UnsafeCell<[u8; BUF_SIZE]>,
155    head: AtomicUsize,
156    tail: AtomicUsize,
157    /// Whole records evicted to make room for newer ones.
158    dropped: AtomicUsize,
159}
160
161// SAFETY: every access to `buf`, and every move of `tail`, happens under
162// `WRITE` with local interrupts masked: producers (any hart, any context) and
163// the consumer in `drain` are serialised, and a same-hart ISR cannot interleave.
164unsafe impl Sync for AuditRing {}
165
166static RING: AuditRing = AuditRing {
167    // SAFETY: UnsafeCell<[u8; N]> in a static is always zero-initialised.
168    buf: UnsafeCell::new([0u8; BUF_SIZE]),
169    head: AtomicUsize::new(0),
170    tail: AtomicUsize::new(0),
171    dropped: AtomicUsize::new(0),
172};
173
174/// Bytes of the synthetic `RecordsDropped` record: header plus `encode_u32x2`.
175const DROP_RECORD_LEN: usize = 10 + 8;
176
177/// Who an authority record is about: the acting task and its Cell
178/// incarnation (architecture review H-1). [`Subject::KERNEL`] (all zero) marks
179/// a kernel- or boot-initiated action with no calling task.
180#[derive(Clone, Copy, Debug, PartialEq, Eq)]
181pub struct Subject {
182    pub cell_id: u64,
183    pub generation: u64,
184    pub tid: u64,
185}
186
187impl Subject {
188    pub const KERNEL: Subject = Subject {
189        cell_id: 0,
190        generation: 0,
191        tid: 0,
192    };
193
194    /// The subject for `tid` read under a scheduler the caller already holds;
195    /// a missing task keeps its tid with a zero incarnation.
196    pub fn of(sched: &crate::sched::scheduler::Scheduler, tid: usize) -> Subject {
197        sched.tasks.get(&tid).map_or(
198            Subject {
199                tid: tid as u64,
200                ..Subject::KERNEL
201            },
202            |task| Subject {
203                cell_id: task.cell_id.0,
204                generation: task.cell_generation,
205                tid: tid as u64,
206            },
207        )
208    }
209}

211/// Payload of an authority record (`SyscallDenied`, `CapRevoked`,
212/// `CellSpawnDenied`): `[cell u64][generation u64][tid u64][d0 u32][d1 u32]`,
213/// all little-endian — the subject, then two event-specific words.
214pub fn encode_subject(subject: Subject, d0: u32, d1: u32) -> [u8; 32] {
215    let mut out = [0u8; 32];
216    out[0..8].copy_from_slice(&subject.cell_id.to_le_bytes());
217    out[8..16].copy_from_slice(&subject.generation.to_le_bytes());
218    out[16..24].copy_from_slice(&subject.tid.to_le_bytes());
219    out[24..28].copy_from_slice(&d0.to_le_bytes());
220    out[28..32].copy_from_slice(&d1.to_le_bytes());
221    out
222}

224/// Write a kernel audit event to the ring.
225///
226/// Disables interrupts for the duration of the write so a same-hart ISR cannot
227/// preempt a partial record. When the ring is full the oldest whole records are
228/// evicted (and counted in [`dropped_count`]), so the ring always holds the most
229/// recent events: without a consumer, drop-on-full would keep only the first
230/// 256 KiB after boot and lose everything since.
231pub fn log_event(event: AuditEvent, payload: &[u8]) {
232    // Capture the outgoing interrupt state and mask interrupts on this hart.
233    // Two reasons: a same-hart ISR must not re-enter the write and wait on the
234    // lock below, and the record's bytes must be ordered before its publication.
235    let saved_interrupts = crate::hal::arch::save_and_disable_interrupts();
236
237    // The guard drops before interrupts return: a same-hart ISR logging in the
238    // gap would otherwise spin on a lock its own hart holds.
239    {
240        let _serialised = WRITE.lock();
241        let head = RING.head.load(Ordering::Relaxed);
242        let tail = RING.tail.load(Ordering::Relaxed);
243        let dropped = RING.dropped.load(Ordering::Relaxed);
244        // SAFETY: `WRITE` is held with interrupts masked (see `impl Sync`).
245        let buf = unsafe { &mut *RING.buf.get() };
246
247        let (head, tail, evicted) = push_record(buf, head, tail, event as u8, payload, dropped);
248        if evicted != 0 {
249            RING.tail.store(tail, Ordering::Release);
250            RING.dropped.fetch_add(evicted, Ordering::Relaxed);
251        }
252
253        // Publish the write — consumer sees it only after this Release store.
254        RING.head.store(head, Ordering::Release);
255    }
256
257    // SAFETY: `saved_interrupts` was captured on this hart by the call above.
258    unsafe { crate::hal::arch::restore_sstatus(saved_interrupts) };
259}

261/// Evict for, then write, one record at `head`; returns the new head, the new
262/// tail and how many records were evicted. `dropped` is the cumulative eviction
263/// count before this call. An eviction is reported by a `RecordsDropped` record
264/// (evicted, cumulative) ahead of the event, so a consumer sees the gap where it
265/// happened; room for it is reserved up front so it never evicts on its own.
266fn push_record(
267    buf: &mut [u8; BUF_SIZE],
268    mut head: usize,
269    tail: usize,
270    event: u8,
271    payload: &[u8],
272    dropped: usize,
273) -> (usize, usize, usize) {
274    let payload = &payload[..payload.len().min(255)];
275    let is_drop_event = event == AuditEvent::RecordsDropped as u8;
276    let reserve = if is_drop_event { 0 } else { DROP_RECORD_LEN };
277    let (tail, evicted) = evict_for(buf, head, tail, 10 + payload.len() + reserve);
278    if evicted != 0 && !is_drop_event {
279        let total = dropped.wrapping_add(evicted);
280        let drop_payload = encode_u32x2(evicted as u32, total as u32);
281        head = write_record_bytes(buf, head, AuditEvent::RecordsDropped as u8, &drop_payload);
282    }
283    (write_record_bytes(buf, head, event, payload), tail, evicted)
284}

286/// Write one record (payload at most 255 bytes) at `pos`; returns the position
287/// just past it.
288fn write_record_bytes(buf: &mut [u8; BUF_SIZE], pos: usize, event: u8, payload: &[u8]) -> usize {
289    debug_assert!(payload.len() <= 255);
290    let plen = payload.len() as u8;
291    #[cfg(target_arch = "riscv64")]
292    let mtime = hal::common::timer::read_mtime().to_le_bytes();
293    #[cfg(not(target_arch = "riscv64"))]
294    let mtime = 0u64.to_le_bytes();
295    let mut curr = pos;
296    for &b in mtime
297        .iter()
298        .chain(core::slice::from_ref(&event))
299        .chain(core::slice::from_ref(&plen))
300        .chain(payload)
301    {
302        buf[curr & MASK] = b;
303        curr = curr.wrapping_add(1);
304    }
305    curr
306}

308/// Length of the record starting at ring position `at`: the 8-byte timestamp,
309/// the event byte, the length byte at offset 9, then the payload.
310fn record_len_at(buf: &[u8; BUF_SIZE], at: usize) -> usize {
311    10 + buf[at.wrapping_add(9) & MASK] as usize
312}
313
314/// Advance `tail` by whole records until `record_len` more bytes fit; returns
315/// the new tail and how many records were evicted. `tail` is always a record
316/// boundary (only this and [`consume`] move it, by whole records), and a
317/// record plus its `RecordsDropped` notice (at most 283 bytes) always fits an
318/// empty ring.
319fn evict_for(
320    buf: &[u8; BUF_SIZE],
321    head: usize,
322    mut tail: usize,
323    record_len: usize,
324) -> (usize, usize) {
325    let mut evicted = 0;
326    while head.wrapping_sub(tail) + record_len > BUF_SIZE {
327        tail = tail.wrapping_add(record_len_at(buf, tail));
328        evicted += 1;
329    }
330    (tail, evicted)
331}

333/// Copy whole records from `tail` towards `head` into `out`; a record that does
334/// not fit in the remaining space stops the copy. Returns bytes copied.
335fn copy_whole_records(buf: &[u8; BUF_SIZE], head: usize, tail: usize, out: &mut [u8]) -> usize {
336    let mut copied = 0;
337    while tail.wrapping_add(copied) != head {
338        let len = record_len_at(buf, tail.wrapping_add(copied));
339        if copied + len > out.len() {
340            break;
341        }
342        for i in 0..len {
343            out[copied + i] = buf[tail.wrapping_add(copied + i) & MASK];
344        }
345        copied += len;
346    }
347    copied
348}

350/// Copy whole records, up to `out.len()` bytes, from the oldest one without
351/// removing them. Returns the ring position the copy started at and the bytes
352/// copied; hand both to [`consume`] once the copy has reached its reader, so a
353/// failed delivery loses no audit record.
354pub fn peek(out: &mut [u8]) -> (usize, usize) {
355    let saved_interrupts = crate::hal::arch::save_and_disable_interrupts();
356    let peeked = {
357        let _serialised = WRITE.lock();
358        let head = RING.head.load(Ordering::Relaxed);
359        let tail = RING.tail.load(Ordering::Relaxed);
360        // SAFETY: `WRITE` is held with interrupts masked (see `impl Sync`).
361        let buf = unsafe { &*RING.buf.get() };
362        (tail, copy_whole_records(buf, head, tail, out))
363    };
364    // SAFETY: `saved_interrupts` was captured on this hart by the call above.
365    unsafe { crate::hal::arch::restore_sstatus(saved_interrupts) };
366    peeked
367}
368
369/// Remove the `copied` bytes that a [`peek`] starting at `start` delivered.
370/// Serialised with producers because they may advance `tail` when they evict.
371pub fn consume(start: usize, copied: usize) {
372    let saved_interrupts = crate::hal::arch::save_and_disable_interrupts();
373    {
374        let _serialised = WRITE.lock();
375        let tail = RING.tail.load(Ordering::Relaxed);
376        RING.tail
377            .store(tail_after_consume(tail, start, copied), Ordering::Release);
378    }
379    // SAFETY: `saved_interrupts` was captured on this hart by the call above.
380    unsafe { crate::hal::arch::restore_sstatus(saved_interrupts) };
381}
382
383/// New tail once `[start, start + copied)` is consumed. Producers evict along
384/// the same whole-record chain the peek copied, so a tail they moved is either
385/// inside that range (finish at its end, a record boundary) or past it (keep).
386fn tail_after_consume(tail: usize, start: usize, copied: usize) -> usize {
387    if tail.wrapping_sub(start) < copied {
388        start.wrapping_add(copied)
389    } else {
390        tail
391    }
392}
393
394/// Number of records evicted to make room for newer ones since boot.
395pub fn dropped_count() -> usize {
396    RING.dropped.load(Ordering::Relaxed)
397}

399#[cfg(feature = "test-hooks")]
400pub(crate) fn snapshot() -> (usize, usize, usize) {
401    (
402        RING.head.load(Ordering::Acquire),
403        RING.tail.load(Ordering::Acquire),
404        RING.dropped.load(Ordering::Acquire),
405    )
406}

410/// Encode two `u32` values into a fixed 8-byte payload.
411pub fn encode_u32x2(a: u32, b: u32) -> [u8; 8] {
412    let mut out = [0u8; 8];
413    out[..4].copy_from_slice(&a.to_le_bytes());
414    out[4..].copy_from_slice(&b.to_le_bytes());
415    out
416}

418#[cfg(test)]
419mod tests {
420    use super::*;
421
422    const EVENT: u8 = AuditEvent::CellSpawn as u8;
423    const DROPPED: u8 = AuditEvent::RecordsDropped as u8;
424
425    /// Overfilling the ring evicts only the oldest whole records: the newest
426    /// survive, every eviction is announced by a `RecordsDropped` record just
427    /// ahead of the record that caused it, every drained byte parses as whole
428    /// records, and a short read stops at a record boundary.
429    #[test]
430    fn full_ring_evicts_oldest_whole_records_and_drains_whole_records() {
431        let mut buf = alloc::boxed::Box::new([0u8; BUF_SIZE]);
432        // Start near usize::MAX so positions wrap during the test.
433        let (mut head, mut tail) = (usize::MAX - 1000, usize::MAX - 1000);
434        let mut evicted = 0;
435        let mut notices = 0u32;
436        let mut written = 0u32;
437        // Varying lengths so record boundaries never line up with the ring size.
438        while written < 4000 {
439            let len = 1 + (written as usize * 37) % 200;
440            let mut payload = alloc::vec![0u8; len];
441            payload[..4.min(len)].copy_from_slice(&written.to_le_bytes()[..4.min(len)]);
442            let (new_head, new_tail, n) =
443                push_record(&mut buf, head, tail, EVENT, &payload, evicted);
444            (head, tail) = (new_head, new_tail);
445            if n != 0 {
446                notices += 1;
447            }
448            evicted += n;
449            written += 1;
450            assert!(head.wrapping_sub(tail) <= BUF_SIZE);
451        }
452        assert!(
453            evicted > 0,
454            "4000 records of up to 210 bytes overflow 256 KiB"
455        );
456
457        let mut out = alloc::vec![0u8; BUF_SIZE];
458        let first_len = record_len_at(&buf, tail);
459        assert_eq!(
460            copy_whole_records(&buf, head, tail, &mut out[..first_len + 3]),
461            first_len,
462            "short read stops at a boundary"
463        );
464        assert_eq!(
465            copy_whole_records(&buf, head, tail, &mut out[..first_len - 1]),
466            0,
467            "a record larger than the space is left"
468        );
469
470        let n = copy_whole_records(&buf, head, tail, &mut out);
471        assert_eq!(
472            n,
473            head.wrapping_sub(tail),
474            "an ample buffer drains everything"
475        );
476        let mut at = 0;
477        let mut survivors = 0u32;
478        let mut seen_notices = 0;
479        let mut last_total = 0u32;
480        let mut ids = alloc::vec::Vec::new();
481        while at < n {
482            let len = 10 + out[at + 9] as usize;
483            assert!(at + len <= n, "partial record");
484            let p = &out[at + 10..at + len];
485            if out[at + 8] == DROPPED {
486                let count = u32::from_le_bytes(p[..4].try_into().unwrap());
487                let total = u32::from_le_bytes(p[4..8].try_into().unwrap());
488                assert!(count > 0, "a notice reports at least one eviction");
489                assert!(total > last_total, "cumulative count only grows");
490                assert!(total as usize <= evicted);
491                assert!(
492                    at + len < n,
493                    "a notice precedes the record it made room for"
494                );
495                assert_eq!(out[at + len + 8], EVENT);
496                last_total = total;
497                seen_notices += 1;
498            } else {
499                assert_eq!(out[at + 8], EVENT);
500                if p.len() >= 4 {
501                    ids.push(u32::from_le_bytes(p[..4].try_into().unwrap()));
502                }
503            }
504            survivors += 1;
505            at += len;
506        }
507        assert!(
508            seen_notices > 0,
509            "surviving evictions are announced in-band"
510        );
511        assert_eq!(last_total as usize, evicted, "the newest notice is current");
512        assert_eq!(ids.last(), Some(&3999), "the newest record survives");
513        assert!(
514            ids.windows(2).all(|w| w[1] > w[0]),
515            "survivors stay in write order"
516        );
517        assert_eq!(
518            evicted as u32 + survivors,
519            4000 + notices,
520            "every record is either evicted or still in the ring"
521        );
522    }

524    /// Consuming a peek moves `tail` to the end of what was delivered, unless
525    /// producers already evicted past it meanwhile.
526    #[test]
527    fn consume_finishes_a_peek_without_rewinding_tail() {
528        let start = usize::MAX - 5;
529        assert_eq!(tail_after_consume(start, start, 20), start.wrapping_add(20));
530        let mid = start.wrapping_add(8);
531        assert_eq!(tail_after_consume(mid, start, 20), start.wrapping_add(20));
532        let past = start.wrapping_add(30);
533        assert_eq!(tail_after_consume(past, start, 20), past);
534        assert_eq!(tail_after_consume(start, start, 0), start);
535    }
