1//! Kernel producer of the machine facts record (syscall 433, ADR-0029 §3).
2//!
3//! [`query`] builds one canonical `api::machine_facts::ViMachineFactsV1` at
4//! call time: the bound x86 profile and its source, the ACPI root OEM fields,
5//! SMBIOS type 1, the DMAR/VT-d/hypervisor flags, and the multiset of
6//! registered PCI functions re-read from config space, sorted without BDF.
7//! The record ends in a SHA-256 of the bytes before it. It grants nothing and
8//! changes nothing; init compares it with the stored pin.
9
10use alloc::vec::Vec;
11
12#[cfg(target_arch = "x86_64")]
13use api::machine_facts as wire;
14
15/// The current record, or `None` when there are no facts: not x86, no
16/// profile bound yet, or more PCI functions than one record holds.
17pub(crate) fn query() -> Option<Vec<u8>> {
18    #[cfg(target_arch = "x86_64")]
19    {
20        let (board, facts) = crate::board::x86_machine_inputs()?;
21        encode(
22            board.slug,
23            &facts,
24            crate::dev::iommu::is_active(),
25            crate::dev::pcie_ecam::fingerprint_functions(),
26        )
27    }
28    #[cfg(not(target_arch = "x86_64"))]
29    {
30        None
31    }
32}
33
34/// Canonical record for these inputs. `pci` is `None` without an admitted
35/// ECAM window; its order does not matter. `None` when a value does not fit
36/// the record (more than `MACHINE_FACTS_PCI_MAX` functions, an invalid slug).
37#[cfg(target_arch = "x86_64")]
38fn encode(
39    slug: &str,
40    facts: &crate::board::X86BootFacts,
41    vtd_active: bool,
42    pci: Option<Vec<wire::PciFunctionId>>,
43) -> Option<Vec<u8>> {
44    let identity = &facts.identity;
45    let mut flags = 0u16;
46    let mut record = wire::ViMachineFactsV1 {
47        flags: 0,
48        profile_source: match facts.source {
49            crate::board::ProfileSource::CommandLine => wire::PROFILE_SOURCE_CMDLINE,
50            crate::board::ProfileSource::Default => wire::PROFILE_SOURCE_DEFAULT,
51        },
52        slug: wire::FactString::new(slug.as_bytes())?,
53        acpi_oem_revision: 0,
54        acpi_oem_table_id: [0; 8],
55        acpi_oem_id: [0; 6],
56        smbios_uuid: [0; 16],
57        smbios_manufacturer: wire::FactString::EMPTY,
58        smbios_product: wire::FactString::EMPTY,
59        pci: Vec::new(),
60    };
61    if let Some(oem) = identity.acpi_oem {
62        flags |= wire::FLAG_ACPI_OEM;
63        record.acpi_oem_id = oem.oem_id;
64        record.acpi_oem_table_id = oem.oem_table_id;
65        record.acpi_oem_revision = oem.oem_revision;
66    }
67    if let Some(system) = identity.smbios {
68        flags |= wire::FLAG_SMBIOS;
69        record.smbios_manufacturer = wire::FactString::new(system.manufacturer.as_bytes())?;
70        record.smbios_product = wire::FactString::new(system.product.as_bytes())?;
71        if let Some(uuid) = system.uuid {
72            flags |= wire::FLAG_SMBIOS_UUID;
73            record.smbios_uuid = uuid;
74        }
75    }
76    if facts.dmar_present {
77        flags |= wire::FLAG_DMAR_PRESENT;
78    }
79    if vtd_active {
80        flags |= wire::FLAG_VTD_ACTIVE;
81    }
82    if identity.hypervisor {
83        flags |= wire::FLAG_HYPERVISOR;
84    }
85    if let Some(mut functions) = pci {
86        if functions.len() > wire::MACHINE_FACTS_PCI_MAX {
87            return None;
88        }
89        flags |= wire::FLAG_PCI_ECAM;
90        functions.sort_unstable();
91        record.pci = functions;
92    }
93    record.flags = flags;
94    record.encode(crate::sha256::sha256).ok()
95}
