1//! Change the permission bits of an ALREADY-MAPPED page.
2//!
3//! Distinct from `paging::map_page`, which installs (or replaces) a translation:
4//! these entry points keep the existing VA→PA binding and rewrite only the
5//! access-rights bits, then invalidate the stale TLB entries.
6//!
7//! - `protect_page`: one page, one invalidation of that VA.
8//! - `protect_range`: `protect_page` per page of a contiguous range.
9//! - `protect_pages` (the W^X lowering pass): on x86_64 every leaf of the list
10//!   is written first under `KERNEL_ROOT`, then one acknowledged invalidation
11//!   covers each contiguous run of written pages before it returns, on the error
12//!   path too; the other arches keep `protect_page` per page.
13//!
14//! Contract shared by every arch leg:
15//! - An unmapped VA is an error (`InvalidAddress`), never a silent map.
16//! - The physical frame is read back from the live page table, so the caller
17//!   cannot accidentally re-point the page while "only" changing flags.
18//! - The per-VA TLB invalidate happens BEFORE returning. A caller that lowers
19//!   permissions may therefore assume the new rights are in force on the
20//!   calling hart the moment the call returns. A remote hart that does not
21//!   confirm within the probe bound fails the call closed
22//!   (`InvalidationUnconfirmed`): the PTE is rewritten, but the caller must
23//!   not report the change as in force.
24//! - Cross-hart / cross-core scope is architecture-specific today:
25//!   - RV64: a changed PTE is ordered locally, invalidated with `sfence.vma`,
26//!     then invalidated synchronously on online remote harts through SBI RFENCE.
27//!     Firmware without RFENCE keeps the kernel single-hart; two-hart runtime
28//!     evidence remains a separate gate.
29//!   - x86_64: all-PCID/global local invalidation followed by every online CPU's
30//!     acknowledged local invalidation; a missing acknowledgement is the
31//!     `InvalidationUnconfirmed` error.
32//!   - AArch64: `flush_tlb_page` broadcasts a stage-1 TLBI
33//!     (`tlbi vaae1is`, plus `vae2is` when EL2 is active) inside the required
34//!     barrier pair, but this repo still treats two-PE runtime proof as gated
35//!     evidence rather than marking D7 complete.
36//! - Bare-physical arches return `NotSupported` instead of pretending W^X
37//!   exists.
38//!
39//! Re-exported as `memory::paging::{protect_page, protect_pages, protect_range}`
40//! — it lives in its own file only to keep `paging.rs` from growing further.
41

47/// Rewrite the permission bits of one mapped 4 KiB page and invalidate its TLB entry.
48///
49/// `new_flags` REPLACES the old flags wholesale — it is not OR-ed in. Pass the
50/// complete desired set (including `VALID` / `USER` / `ACCESSED` / `DIRTY`),
51/// because a missing `VALID` bit unmaps the page rather than restricting it
52/// (riscv64, aarch64; x86_64 always keeps the page present, and keeps the
53/// frame's cache and global bits).
54///
55/// # Errors
56/// - `PageTableError::InvalidAddress` — `vaddr` is not currently mapped.
57/// - `PageTableError::NotSupported` — paging is inactive, or the target is a
58///   bare-physical arch (riscv32 / x86_32 / arm32) with no page tables.
59#[cfg(cellos_wide)]
60pub fn protect_page(vaddr: VAddr, new_flags: Flags) -> PagingResult<()> {
61    let page_va = vaddr & !(PAGE_SIZE - 1);
62    // Before any rewrite: a hart that holds the frame allocator may not wait
63    // for the flush, so it must not change the leaf either.
64    let mut cap =
65        crate::memory::frame::FlushCap::take().ok_or(PageTableError::InvalidationUnconfirmed)?;
66    let root_lock = KERNEL_ROOT.lock();
67    let root_phys = (*root_lock).ok_or(PageTableError::NotSupported)?;
68    rewrite_leaf(root_phys, page_va, new_flags)?;
69    let outcome = crate::memory::tlb_shootdown::flush_page(&mut cap, page_va);
70    drop(root_lock);
71    confirmed(outcome, page_va)
72}

74/// `outcome` as the protect result, logging an unconfirmed one once the
75/// caller's `KERNEL_ROOT` is released (it ranks above the console).
76#[cfg(cellos_wide)]
77fn confirmed(outcome: crate::memory::frame_run::FlushOutcome, page_va: VAddr) -> PagingResult<()> {
78    match outcome {
79        crate::memory::frame_run::FlushOutcome::Confirmed => Ok(()),
80        crate::memory::frame_run::FlushOutcome::Unconfirmed { cause, .. } => {
81            log::error!(
82                "[tlb] shared invalidation unconfirmed: permission change at {:#x} not in force everywhere ({:?})",
83                page_va,
84                cause
85            );
86            Err(PageTableError::InvalidationUnconfirmed)
87        }
88    }
89}

91/// The leaf rewrite of [`protect_page`] without its invalidation. The caller
92/// holds `KERNEL_ROOT` (whose value is `root_phys`) and owns the flush.
93///
94/// Reading the frame back from the table is what makes this a permission
95/// change rather than a remap: the caller has no say in the target frame.
96#[cfg(cellos_wide)]
97fn rewrite_leaf(root_phys: types::PhysAddr, page_va: VAddr, new_flags: Flags) -> PagingResult<()> {
98    use hal::PageTableTrait;
99
100    // SAFETY: root_phys is the live kernel root and the caller holds its lock.
101    let table = unsafe { &mut *crate::memory::paging::table_at(root_phys) };
102    table
103        .protect(page_va, new_flags)
104        .map_err(|error| match error {
105            types::ViError::NotFound => PageTableError::InvalidAddress,
106            // Every table on the path exists; an allocation request is a surprise
107            // walk, reported as it always was.
108            _ => PageTableError::OutOfMemory,
109        })
110}

112/// Rewrite the permission bits of every page in `pages`, in order, as one batch
113/// (the W^X lowering pass, `loader::wx::enforce`).
114///
115/// Concurrency contract (phase 07b R1, owner grant
116/// `s2-phase07b-flush-batching-grant.md`):
117/// - x86_64: `KERNEL_ROOT` is held from the first leaf write through the last
118///   flush, as [`protect_page`] holds it through its own. Every leaf is written
119///   first, then ONE `tlb_shootdown::flush_range` covers each contiguous run of
120///   written pages. `flush_range` returns `Confirmed` only after every online
121///   CPU acknowledged; an unconfirmed run is the `InvalidationUnconfirmed`
122///   error (every run is still flushed). So on `Ok` a lowering (W→R, X
123///   removal) is in force on every online CPU — before the Cell can be
124///   registered to run, and before a failed spawn's teardown can free a frame.
125/// - The error path flushes the pages written before the failure, too: no
126///   caller ever sees a rewritten leaf whose invalidation is still pending.
127/// - riscv64/aarch64 (and the bare-physical arches): [`protect_page`] per page
128///   in order, exactly the pass this replaces.
129///
130/// # Errors
131/// The first page that could not be rewritten, with the [`protect_page`] error.
132/// The pages before it keep their new flags (see [`protect_range`]).
133#[cfg(target_arch = "x86_64")]
134pub fn protect_pages(pages: &[(VAddr, Flags)]) -> Result<(), (VAddr, PageTableError)> {
135    let Some(&(first, _)) = pages.first() else {
136        return Ok(());
137    };
138    let mut cap = crate::memory::frame::FlushCap::take()
139        .ok_or((first, PageTableError::InvalidationUnconfirmed))?;
140    let root_lock = KERNEL_ROOT.lock();
141    let root_phys = (*root_lock).ok_or((first, PageTableError::NotSupported))?;
142    let mut unconfirmed = None;
143    let result = write_then_flush(
144        pages,
145        |page_va, flags| rewrite_leaf(root_phys, page_va, flags),
146        |start, size| match crate::memory::tlb_shootdown::flush_range(&mut cap, start, size) {
147            crate::memory::frame_run::FlushOutcome::Confirmed => Ok(()),
148            outcome => {
149                unconfirmed.get_or_insert(outcome);
150                Err(PageTableError::InvalidationUnconfirmed)
151            }
152        },
153    );
154    drop(root_lock);
155    if let (Some(outcome), Err((va, _))) = (unconfirmed, result) {
156        // Logs it; the error is already the result.
157        let _ = confirmed(outcome, va);
158    }
159    result
160}

162/// See the x86_64 [`protect_pages`]: these arches keep the per-page pass.
163#[cfg(not(target_arch = "x86_64"))]
164pub fn protect_pages(pages: &[(VAddr, Flags)]) -> Result<(), (VAddr, PageTableError)> {
165    for &(va, flags) in pages {
166        protect_page(va, flags).map_err(|error| (va, error))?;
167    }
168    Ok(())
169}

171/// Rewrite the permission bits of every *mapped* page of the `pages` pages
172/// from page-aligned `start`, then invalidate the whole range on every hart
173/// with ONE shared flush, before returning either way (one step of the MMIO
174/// lowering, `paging::restore_mmio_run`).
175///
176/// - An absent leaf is skipped, not an error, but the flush still covers it:
177///   a translation removed without one may still be cached.
178/// - A leaf that cannot be rewritten does not stop the pass: every other page
179///   is still lowered and flushed, so a failure never leaves more pages
180///   reachable than it must.
181/// - `KERNEL_ROOT` is held from the first leaf write through the flush, as
182///   [`protect_pages`] holds it.
183/// - `await_flush`: wait for every hart to confirm the flush. An unconfirmed
184///   flush clears it (and is logged), so the later steps of the same lowering
185///   only send their requests: the lowering has already failed, and a silent
186///   hart would otherwise cost each step a full probe bound.
187///
188/// Returns the pages rewritten.
189///
190/// # Errors
191/// The first page that could not be rewritten, else `(start,
192/// InvalidationUnconfirmed)` when the flush is not confirmed on every hart
193/// (without `await_flush`, whenever a remote hart was asked).
194#[cfg(cellos_wide)]
195pub fn protect_mapped_range(
196    start: VAddr,
197    pages: usize,
198    new_flags: Flags,
199    await_flush: &mut bool,
200) -> Result<usize, (VAddr, PageTableError)> {
201    debug_assert_eq!(start & (PAGE_SIZE - 1), 0);
202    if pages == 0 {
203        return Ok(0);
204    }
205    let mut cap = if *await_flush {
206        Some(
207            crate::memory::frame::FlushCap::take()
208                .ok_or((start, PageTableError::InvalidationUnconfirmed))?,
209        )
210    } else {
211        None
212    };
213    let root_lock = KERNEL_ROOT.lock();
214    let root_phys = (*root_lock).ok_or((start, PageTableError::NotSupported))?;
215    let mut unconfirmed = None;
216    let result = write_mapped_then_flush(
217        start,
218        pages,
219        |page_va| match rewrite_leaf(root_phys, page_va, new_flags) {
220            Ok(()) => Ok(true),
221            Err(PageTableError::InvalidAddress) => Ok(false),
222            Err(error) => Err(error),
223        },
224        |from, size| {
225            let outcome = match cap.as_mut() {
226                Some(cap) => crate::memory::tlb_shootdown::flush_range(cap, from, size),
227                None => {
228                    crate::memory::tlb_shootdown::flush_ranges_without_wait(&[(from, size)], false)
229                }
230            };
231            match outcome {
232                crate::memory::frame_run::FlushOutcome::Confirmed => Ok(()),
233                outcome => {
234                    unconfirmed = Some(outcome);
235                    Err(PageTableError::InvalidationUnconfirmed)
236                }
237            }
238        },
239    );
240    drop(root_lock);
241    if let (Some(outcome), true) = (unconfirmed, cap.is_some()) {
242        *await_flush = false;
243        // Logs it; the result already carries the failure.
244        let _ = confirmed(outcome, start);
245    }
246    result
247}

249/// Write the leaf of each of the `pages` pages from `start` (`write` returns
250/// `false` for an absent leaf), every page even after one failed, then call
251/// `flush` once over the whole range. Returns the pages written; the error is
252/// the first failed write, else the flush's (named by `start`).
253#[cfg(cellos_wide)]
254fn write_mapped_then_flush<E>(
255    start: VAddr,
256    pages: usize,
257    mut write: impl FnMut(VAddr) -> Result<bool, E>,
258    flush: impl FnOnce(VAddr, usize) -> Result<(), E>,
259) -> Result<usize, (VAddr, E)> {
260    let mut written = 0;
261    let mut failure = None;
262    for page_va in (0..pages).map(|page| start + page * PAGE_SIZE) {
263        match write(page_va) {
264            Ok(true) => written += 1,
265            Ok(false) => {}
266            Err(error) => {
267                failure.get_or_insert((page_va, error));
268            }
269        }
270    }
271    let flushed = flush(start, pages * PAGE_SIZE).map_err(|error| (start, error));
272    match failure {
273        Some(failure) => Err(failure),
274        None => flushed.map(|()| written),
275    }
276}

278/// Write each page's leaf in order (stopping at the first failure), then call
279/// `flush` once per contiguous run of the pages actually written, before
280/// returning either way — every run is flushed even after one failed. The
281/// error carries the caller's address for the page: a write failure first,
282/// else the start of the first run whose flush failed.
283#[cfg(target_arch = "x86_64")]
284fn write_then_flush<E>(
285    pages: &[(VAddr, Flags)],
286    mut write: impl FnMut(VAddr, Flags) -> Result<(), E>,
287    mut flush: impl FnMut(VAddr, usize) -> Result<(), E>,
288) -> Result<(), (VAddr, E)> {
289    let mut written = 0;
290    let mut failure = None;
291    for &(va, flags) in pages {
292        if let Err(error) = write(va & !(PAGE_SIZE - 1), flags) {
293            failure = Some((va, error));
294            break;
295        }
296        written += 1;
297    }
298    let written_pages = pages[..written]
299        .iter()
300        .map(|&(va, _)| va & !(PAGE_SIZE - 1));
301    let mut flush_failure = None;
302    for (start, size) in contiguous_runs(written_pages) {
303        if let Err(error) = flush(start, size) {
304            flush_failure.get_or_insert((start, error));
305        }
306    }
307    failure.or(flush_failure).map_or(Ok(()), Err)
308}

310/// Coalesce page addresses, in the order given, into maximal runs of
311/// consecutive pages: `(start, size_in_bytes)`. A page that does not directly
312/// follow the previous one starts a new run, so every page is in exactly one.
313#[cfg(target_arch = "x86_64")]
314fn contiguous_runs(pages: impl Iterator<Item = VAddr>) -> impl Iterator<Item = (VAddr, usize)> {
315    let mut pages = pages.peekable();
316    core::iter::from_fn(move || {
317        let start = pages.next()?;
318        let mut size = PAGE_SIZE;
319        while let Some(next) = start.checked_add(size) {
320            if pages.next_if_eq(&next).is_none() {
321                break;
322            }
323            size += PAGE_SIZE;
324        }
325        Some((start, size))
326    })
327}

330/// unavailable rather than silently believing it was applied.
331#[cfg(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm"))]
332pub fn protect_page(_vaddr: VAddr, _new_flags: Flags) -> PagingResult<()> {
333    Err(PageTableError::NotSupported)
334}

336/// Apply [`protect_page`] to `pages` consecutive 4 KiB pages starting at `start`.
337///
338/// Fails fast: the first page that is not mapped aborts the walk and the pages
339/// already lowered KEEP their new flags. Callers must therefore treat an error
340/// as "permissions are in an unknown but never MORE permissive state" — this is
341/// the safe direction for a W^X lowering pass and the reason no rollback exists.
342///
343/// # Errors
344/// Propagates the first [`protect_page`] failure.
345pub fn protect_range(start: VAddr, pages: usize, new_flags: Flags) -> PagingResult<()> {
346    for i in 0..pages {
347        protect_page(start + i * PAGE_SIZE, new_flags)?;
348    }
349    Ok(())
350}

352#[cfg(all(test, target_arch = "x86_64"))]
353mod tests {
354    use super::*;
355    use alloc::vec::Vec;

357    #[derive(Debug, PartialEq, Eq)]
358    enum Event {
359        Write(VAddr),
360        Flush(VAddr, usize),
361    }
362
363    fn ro() -> Flags {
364        Flags::from_bits(Flags::VALID | Flags::USER | Flags::READ)
365    }
366
367    fn page(n: usize) -> VAddr {
368        0x1_0000_0000 + n * PAGE_SIZE
369    }
370
371    /// Drive `write_then_flush` with a recorder; `fail_at` makes that page's
372    /// write fail.
373    fn run(
374        pages: &[(VAddr, Flags)],
375        fail_at: Option<VAddr>,
376    ) -> (Result<(), (VAddr, ())>, Vec<Event>) {
377        let events = core::cell::RefCell::new(Vec::new());
378        let result = write_then_flush(
379            pages,
380            |va, _| {
381                if Some(va) == fail_at {
382                    return Err(());
383                }
384                events.borrow_mut().push(Event::Write(va));
385                Ok(())
386            },
387            |start, size| {
388                events.borrow_mut().push(Event::Flush(start, size));
389                Ok(())
390            },
391        );
392        (result, events.into_inner())
393    }
394
395    /// Each written page is covered by exactly one flush issued after it.
396    fn covered_once_after_write(events: &[Event], written: &[VAddr]) {
397        for &va in written {
398            let write_at = events.iter().position(|e| *e == Event::Write(va)).unwrap();
399            let covering: Vec<usize> = events
400                .iter()
401                .enumerate()
402                .filter(|(_, e)| matches!(e, Event::Flush(s, n) if (*s..*s + *n).contains(&va)))
403                .map(|(i, _)| i)
404                .collect();
405            assert_eq!(covering.len(), 1, "page {va:#x} flushed once");
406            assert!(
407                covering[0] > write_at,
408                "page {va:#x} flushed after its write"
409            );
410        }
411    }

413    #[test]
414    fn contiguous_runs_coalesce_only_adjacent_pages() {
415        let runs = |pages: &[VAddr]| contiguous_runs(pages.iter().copied()).collect::<Vec<_>>();
416        assert_eq!(runs(&[]), []);
417        assert_eq!(runs(&[page(0)]), [(page(0), PAGE_SIZE)]);
418        // An ELF image: R-X 0..3, R-- 3..5 (adjacent), RW- after a one-page gap.
419        let image: Vec<VAddr> = (0..5).chain(6..9).map(page).collect();
420        assert_eq!(
421            runs(&image),
422            [(page(0), 5 * PAGE_SIZE), (page(6), 3 * PAGE_SIZE)]
423        );
424        // Order is kept: a descending pair is two runs, never a merged one.
425        assert_eq!(
426            runs(&[page(1), page(0)]),
427            [(page(1), PAGE_SIZE), (page(0), PAGE_SIZE)]
428        );
429        // A run may end exactly at the last page boundary below 2^64. A run
430        // that contained the last page would end at 2^64, which `flush_range`
431        // refuses as a wrapping range; Cell pages lie far below it (the
432        // coalescer's `checked_add` alone would not wrap there either).
433        let top = usize::MAX & !(PAGE_SIZE - 1);
434        assert_eq!(
435            runs(&[top - 2 * PAGE_SIZE, top - PAGE_SIZE]),
436            [(top - 2 * PAGE_SIZE, 2 * PAGE_SIZE)]
437        );
438    }

558    #[test]
559    fn unaligned_addresses_are_written_and_flushed_as_their_page() {
560        let pages = [(page(2) + 0x10, ro()), (page(3) + 0xfff, ro())];
561        let (result, events) = run(&pages, None);
562        assert_eq!(result, Ok(()));
563        assert_eq!(
564            events,
565            [
566                Event::Write(page(2)),
567                Event::Write(page(3)),
568                Event::Flush(page(2), 2 * PAGE_SIZE)
569            ]
570        );
571        // The error names the caller's address, not the rounded page.
572        let (result, _) = run(&pages, Some(page(3)));
573        assert_eq!(result, Err((page(3) + 0xfff, ())));
574    }
