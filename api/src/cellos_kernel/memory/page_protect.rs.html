1//! Change the permission bits of an ALREADY-MAPPED page.
2//!
3//! Distinct from `paging::map_page`, which installs (or replaces) a translation:
4//! these entry points keep the existing VA→PA binding and rewrite only the
5//! access-rights bits, then invalidate the stale TLB entry for that one VA.
6//!
7//! Contract shared by every arch leg:
8//! - An unmapped VA is an error (`InvalidAddress`), never a silent map.
9//! - The physical frame is read back from the live page table, so the caller
10//!   cannot accidentally re-point the page while "only" changing flags.
11//! - The per-VA TLB invalidate happens BEFORE returning. A caller that lowers
12//!   permissions may therefore assume the new rights are in force on the
13//!   calling hart the moment the call returns.
14//! - Cross-hart / cross-core scope is architecture-specific today:
15//!   - RV64: a changed PTE is ordered locally, invalidated with `sfence.vma`,
16//!     then invalidated synchronously on online remote harts through SBI RFENCE.
17//!     Firmware without RFENCE keeps the kernel single-hart; two-hart runtime
18//!     evidence remains a separate gate.
19//!   - x86_64: `invlpg` is local only; there is no SMP IPI shootdown path yet.
20//!   - AArch64: `flush_tlb_page` broadcasts a stage-1 TLBI
21//!     (`tlbi vaae1is`, plus `vae2is` when EL2 is active) inside the required
22//!     barrier pair, but this repo still treats two-PE runtime proof as gated
23//!     evidence rather than marking D7 complete.
24//! - Bare-physical arches return `NotSupported` instead of pretending W^X
25//!   exists.
26//!
27//! Re-exported as `memory::paging::{protect_page, protect_range}` — it lives in
28//! its own file only to keep `paging.rs` from growing further.
29
30use crate::memory::paging::{Flags, PagingResult, PAGE_SIZE};
31#[allow(unused_imports)] // used by the paged arches only
32use crate::memory::paging::{PageTableError, KERNEL_ROOT};
33use types::VAddr;

35/// Rewrite the permission bits of one mapped 4 KiB page and invalidate its TLB entry.
36///
37/// `new_flags` REPLACES the old flags wholesale — it is not OR-ed in. Pass the
38/// complete desired set (including `VALID` / `USER` / `ACCESSED` / `DIRTY`),
39/// because a missing `VALID` bit unmaps the page rather than restricting it.
40///
41/// # Errors
42/// - `PageTableError::InvalidAddress` — `vaddr` is not currently mapped.
43/// - `PageTableError::NotSupported` — paging is inactive, or the target is a
44///   bare-physical arch (riscv32 / x86_32 / arm32) with no page tables.
45#[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
46pub fn protect_page(vaddr: VAddr, new_flags: Flags) -> PagingResult<()> {
47    use hal::{PageTable, PageTableTrait};
48
49    let page_va = vaddr & !(PAGE_SIZE - 1);
50    let root_lock = KERNEL_ROOT.lock();
51    let root_phys = (*root_lock).ok_or(PageTableError::NotSupported)?;
52    // SAFETY: KERNEL_ROOT holds the physical address of the live root table,
53    // which is identity-mapped on both of these arches, and the lock we hold
54    // gives us exclusive access to it for the duration of this function.
55    let root_table = unsafe { &mut *(root_phys as *mut PageTable) };
56
57    // Reading the frame back from the table is what makes this a permission
58    // change rather than a remap: the caller has no say in the target frame.
59    let phys = root_table
60        .translate(page_va)
61        .ok_or(PageTableError::InvalidAddress)?
62        & !(PAGE_SIZE - 1);
63
64    // Every intermediate table on this path already exists (the page is mapped),
65    // so `map` never invokes the allocator; a None-returning closure documents
66    // that and turns any surprise walk into OutOfMemory instead of a silent alloc.
67    let mut no_alloc = || None;
68    root_table
69        .map(page_va, phys, new_flags, &mut no_alloc)
70        .map_err(|_| PageTableError::OutOfMemory)?;
71
72    crate::memory::tlb_shootdown::flush_page(page_va);
73    Ok(())
74}

76/// x86_64 leg: rewrite the leaf PTE's flag bits, preserving the frame address
77/// and the cache-attribute bits (PWT/PCD) already recorded for the page.
78#[cfg(target_arch = "x86_64")]
79pub fn protect_page(vaddr: VAddr, new_flags: Flags) -> PagingResult<()> {
80    use crate::memory::frame::phys_to_virt;
81    use hal::paging::{walk_create, walk_read, PTE_ADDR_MASK, PTE_PCD, PTE_PRESENT, PTE_PWT};
82
83    let page_va = vaddr & !(PAGE_SIZE - 1);
84    let root_lock = KERNEL_ROOT.lock();
85    let root_phys = (*root_lock).ok_or(PageTableError::NotSupported)?;
86
87    // SAFETY: root_phys is the kernel's live PML4; phys_to_virt maps it into the
88    // HHDM window, which is readable/writable from Ring 0 for the whole boot.
89    let existing = unsafe { walk_read(phys_to_virt(root_phys) as *const u64, page_va) }
90        .ok_or(PageTableError::InvalidAddress)?;
91    if existing & PTE_PRESENT == 0 {
92        return Err(PageTableError::InvalidAddress);
93    }
94
95    let cache_bits = existing & (PTE_PWT | PTE_PCD);
96    let new_pte = (existing & PTE_ADDR_MASK) | cache_bits | pte_bits_from_flags(new_flags);
97
98    let mut no_alloc = || None::<usize>;
99    // SAFETY: the same PML4, re-walked for a mutable leaf pointer. The page is
100    // present, so no intermediate table is missing and `no_alloc` is never called.
101    let pte_ptr =
102        unsafe { walk_create(phys_to_virt(root_phys) as *mut u64, page_va, &mut no_alloc) }
103            .ok_or(PageTableError::InvalidAddress)?;
104    // SAFETY: pte_ptr is the leaf PTE slot for page_va, obtained from the walker.
105    unsafe {
106        core::ptr::write_volatile(pte_ptr, new_pte);
107    }
108
109    crate::memory::tlb_shootdown::flush_page(page_va);
110    Ok(())
111}

113/// Translate generic `PageFlags` into x86_64 leaf-PTE bits.
114///
115/// `EXECUTE` is inverted into the NX bit: absence of EXECUTE means NX is SET.
116/// Shared with `paging::map_page` so a page created and a page re-protected
117/// cannot disagree about what a given `Flags` value means.
118#[cfg(target_arch = "x86_64")]
119pub(crate) fn pte_bits_from_flags(flags: Flags) -> u64 {
120    use hal::paging::{PTE_NX, PTE_PRESENT, PTE_USER, PTE_WRITABLE};
121    let bits = flags.bits();
122    let mut pte = PTE_PRESENT;
123    if bits & hal::PageFlags::WRITE != 0 {
124        pte |= PTE_WRITABLE;
125    }
126    if bits & hal::PageFlags::USER != 0 {
127        pte |= PTE_USER;
128    }
129    if bits & hal::PageFlags::EXECUTE == 0 {
130        pte |= PTE_NX;
131    }
132    pte
133}

135/// Bare-physical arches have no PTEs to protect; the caller must treat W^X as
136/// unavailable rather than silently believing it was applied.
137#[cfg(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm"))]
138pub fn protect_page(_vaddr: VAddr, _new_flags: Flags) -> PagingResult<()> {
139    Err(PageTableError::NotSupported)
140}

142/// Apply [`protect_page`] to `pages` consecutive 4 KiB pages starting at `start`.
143///
144/// Fails fast: the first page that is not mapped aborts the walk and the pages
145/// already lowered KEEP their new flags. Callers must therefore treat an error
146/// as "permissions are in an unknown but never MORE permissive state" — this is
147/// the safe direction for a W^X lowering pass and the reason no rollback exists.
148///
149/// # Errors
150/// Propagates the first [`protect_page`] failure.
151pub fn protect_range(start: VAddr, pages: usize, new_flags: Flags) -> PagingResult<()> {
152    for i in 0..pages {
153        protect_page(start + i * PAGE_SIZE, new_flags)?;
154    }
155    Ok(())
