1//! A cell image's segment frames and its VA slot (architecture remediation
2//! Phase 16: moved from `task/stack.rs`, next to the release contract its
3//! teardown follows).
4//!
5//! The image's frames, the kernel-root tables its mappings needed and its VA
6//! slot are released as one operation: every leaf and every detached table is
7//! invalidated by one shared flush, and the slot stays reserved until the
8//! frames behind it are freed — or, when a hart does not confirm, quarantined
9//! with them (census risk 2: a slot handed out again while a remote hart may
10//! still translate it would resolve the new image's VA to the old frame).
11
12use crate::memory::cell_quota::Charge;
13use crate::memory::frame_run::{Batch, FrameRun, KernelAlias, RunKind, VaHold};
14use crate::memory::paging::{self, PAGE_SIZE};
15use types::VAddr;

17/// Physical frames mapped for a cell's ELF segments, recorded as `(vaddr, frame)`
18/// at load time so they can be reclaimed when the cell dies.
19///
20/// `Stack::drop` only frees stacks; without this a cell's code/data frames leak
21/// on every death (a supervised service restarted repeatedly would grow to OOM).
22/// Segment frames are allocated exclusively for the cell by `load_segments`
23/// (IPC/shared buffers use separate frames), so freeing them on death is safe.
24#[derive(Debug)]
25pub struct CellSegments {
26    pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
27    /// ELF pages whose final mapping grants the Cell write permission.
28    writable_pages: alloc::vec::Vec<types::VAddr>,
29    /// VA slot this image owns (`va_alloc::alloc_cell_va` for PIE,
30    /// `va_alloc::reserve_cell_va` for fixed-VA images); `0` for kernel-built
31    /// test segments. Returned to the allocator's free list on drop.
32    va_slot: usize,
33    /// The image's BTI decision (`loader::gnu_property::arm`): its executable
34    /// pages are guarded (stage-1 GP). Carried here so a Tier-2 private root
35    /// built from these segments (`memory::address_space::create_cell_domain`)
36    /// guards them too, not only the shared-address-space mapping.
37    guarded_code: bool,
38    #[cfg(feature = "p01-trace")]
39    pub(crate) p01_obligation: Option<crate::p01_resources::Obligation>,
40}
41
42impl CellSegments {
43    pub fn new(pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>, va_slot: usize) -> Self {
44        Self {
45            pages,
46            writable_pages: alloc::vec::Vec::new(),
47            va_slot,
48            guarded_code: false,
49            #[cfg(feature = "p01-trace")]
50            p01_obligation: None,
51        }
52    }
53
54    /// Callers size both vectors exactly (`try_reserve_exact`), so construction
55    /// never reallocates: a shrink here would be an infallible allocation on the
56    /// path documented as infallible, and would move buffers p01 already adopted.
57    pub fn with_writable_pages(
58        pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
59        mut writable_pages: alloc::vec::Vec<types::VAddr>,
60        va_slot: usize,
61    ) -> Self {
62        // Sorted for the binary searches below; in place, so no allocation.
63        writable_pages.sort_unstable();
64        writable_pages.dedup();
65        Self {
66            pages,
67            writable_pages,
68            va_slot,
69            guarded_code: false,
70            #[cfg(feature = "p01-trace")]
71            p01_obligation: None,
72        }
73    }
74
75    /// Record the image's BTI decision (see the `guarded_code` field).
76    pub fn with_guarded_code(mut self, guarded_code: bool) -> Self {
77        self.guarded_code = guarded_code;
78        self
79    }

81    /// Whether the image's executable pages are BTI-guarded.
82    pub fn guarded_code(&self) -> bool {
83        self.guarded_code
84    }

86    pub fn pages(&self) -> &[(types::VAddr, types::PhysAddr)] {
87        &self.pages
88    }

90    /// Exact capacities of this task's exclusive segment metadata, not frames.
91    /// Publication calls this while holding SCHEDULER with the final identity.
92    #[cfg(feature = "p01-trace")]
93    pub(crate) fn p01_backing(&self, owner: crate::p01_measurement::Owner) {
94        crate::memory::heap::p01_owner::runtime_vec(
95            &self.pages,
96            owner,
97            crate::memory::heap::p01_owner::TASK_SEGMENTS,
98        );
99        crate::memory::heap::p01_owner::runtime_vec(
100            &self.writable_pages,
101            owner,
102            crate::memory::heap::p01_owner::TASK_SEGMENTS,
103        );
104    }
105
106    pub fn is_writable(&self, va: types::VAddr) -> bool {
107        self.writable_pages.binary_search(&va).is_ok()
108    }
109    /// Return the exclusive end of the writable page containing `ptr`.
110    pub fn writable_page_end_containing(&self, ptr: usize) -> Option<usize> {
111        let page = ptr & !(PAGE_SIZE - 1);
112        self.writable_pages
113            .binary_search(&page)
114            .ok()
115            .and_then(|_| page.checked_add(PAGE_SIZE))
116    }
117
118    /// Whether every page touched by this range is writable in this Cell image.
119    pub fn contains_writable_range(&self, ptr: usize, len: usize) -> bool {
120        let Some(end) = ptr.checked_add(len) else {
121            return false;
122        };
123        if len == 0 {
124            return false;
125        }
126        let mut page = ptr & !(PAGE_SIZE - 1);
127        while page < end {
128            if self.writable_pages.binary_search(&page).is_err() {
129                return false;
130            }
131            page = match page.checked_add(PAGE_SIZE) {
132                Some(next) => next,
133                None => return false,
134            };
135        }
136        true
137    }

144    /// Total physical bytes reserved for this cell's ELF segment frames.
145    pub fn allocated_bytes(&self) -> usize {
146        self.pages.len() * PAGE_SIZE
147    }
148
149    /// Unmap this cell's segment VAs immediately at death — WITHOUT freeing the
150    /// frames (those are freed lazily when the zombie is reaped, in `drop`).
151    ///
152    /// Frees the address space right away so (a) a respawn can reuse the fixed VA
153    /// and (b) the load-time overwrite guard (`load_segments`) only ever observes
154    /// LIVE cells' (and kernel MMIO) mappings, never a dead-but-unreaped cell's.
155    /// Locks only `KERNEL_ROOT` (a leaf), so it is safe under the SCHEDULER lock.
156    /// One shared invalidation covers every page, and it is not waited for
157    /// (census F4: the caller holds `SCHEDULER`): nothing is freed here, the
158    /// frames and the VA slot stay owned, and the drop invalidates every VA
159    /// again and frees them only once that flush confirms.
160    pub fn eager_unmap(&self) {
161        let mut batch = Batch::new();
162        self.unmap_into(&mut batch);
163        let _ = batch.complete_without_wait();
164    }
165
166    /// Unmap every VA that still resolves to this image's frame and record
167    /// every VA in `batch`.
168    fn unmap_into(&self, batch: &mut Batch) {
169        for &(vaddr, frame) in &self.pages {
170            // Only unmap a VA that still resolves to OUR frame (it won't if a
171            // respawn already re-pointed it — leave the new mapping intact).
172            if paging::virt_to_phys(vaddr) == Some(frame) {
173                let _ = paging::unmap_page(batch, vaddr);
174            }
175            // Invalidate even when a replacement mapping won the race: an old
176            // remote translation for this VA can still point at this frame.
177            batch.invalidate(vaddr);
178        }
179    }
180
181    /// The teardown up to its flush: unmap, detach every emptied table and
182    /// retire the backing (the last run holding the VA slot) into `batch`.
183    fn teardown_into(&self, batch: &mut Batch) {
184        self.unmap_into(batch);
185        for &(vaddr, _frame) in &self.pages {
186            paging::prune_empty_tables(vaddr).retire_into(batch);
187        }
188        self.retire_backing_into(batch);
189    }
190
191    /// Tear this image down as part of a larger operation: its frames, tables
192    /// and VA slot join `batch`, whose one shared flush covers every owner
193    /// retired into it, and are freed — or quarantined — when it finishes.
194    #[cfg(feature = "test-hooks")]
195    // reason: only the x86 silent-CPU fixture retires owners into a shared batch.
196    #[allow(dead_code)]
197    pub(crate) fn retire_into(self, batch: &mut Batch) {
198        let mut segments = core::mem::ManuallyDrop::new(self);
199        segments.teardown_into(batch);
200        #[cfg(feature = "p01-trace")]
201        debug_assert!(
202            segments.p01_obligation.is_none(),
203            "an image with a p01 obligation is retired only by its drop"
204        );
205        // The batch owns the frames and the slot now; only the metadata is
206        // dropped here. An image with no frames releases its slot itself, as
207        // the drop does.
208        let pages = core::mem::take(&mut segments.pages);
209        drop(core::mem::take(&mut segments.writable_pages));
210        if pages.is_empty() && segments.va_slot != 0 {
211            crate::loader::va_alloc::free_cell_va(segments.va_slot);
212        }
213    }
214
215    /// Hand every backing frame to `batch` as runs of physically adjacent
216    /// frames (the loader allocates them in order), the last one holding the
217    /// VA slot until the frames are free.
218    fn retire_backing_into(&self, batch: &mut Batch) {
219        let mut runs = self.pages.iter().map(|&(_, frame)| frame).peekable();
220        while let Some(base) = runs.next() {
221            let mut pages = 1;
222            while runs.next_if_eq(&(base + pages * PAGE_SIZE)).is_some() {
223                pages += 1;
224            }
225            let run = FrameRun::unpublished(
226                base,
227                pages,
228                Charge::KERNEL,
229                RunKind::Leaf,
230                KernelAlias::Intact,
231            );
232            batch.retire(match (runs.peek(), self.va_slot) {
233                (None, slot) if slot != 0 => run.with_va_hold(VaHold {
234                    release: release_cell_va,
235                    start: slot,
236                    end: slot,
237                }),
238                _ => run,
239            });
240        }
241    }
242}

244/// [`VaHold`] release for a cell image's VA slot.
245pub(crate) fn release_cell_va(slot: VAddr, _end: VAddr) {
246    crate::loader::va_alloc::free_cell_va(slot);
247}

249impl Drop for CellSegments {
250    fn drop(&mut self) {
251        // One batch: leaf and parent invalidations reach every hart once,
252        // before either backing or page-table frames return to the allocator.
253        // An unconfirmed flush quarantines them, and the VA slot with them.
254        let mut batch = Batch::new();
255        self.teardown_into(&mut batch);
256        #[allow(unused_variables)] // reason: only the p01-trace ledger reads it
257        let flushed = batch.complete();
258        #[cfg(feature = "p01-trace")]
259        if flushed.is_confirmed() {
260            crate::p01_resources::finish(self.p01_obligation.take(), 1);
261        }
262        #[cfg(feature = "p01-trace")]
263        if self.p01_obligation.is_some() {
264            crate::p01_resources::finish(self.p01_obligation.take(), 6);
265        }
266        if self.pages.is_empty() && self.va_slot != 0 {
267            crate::loader::va_alloc::free_cell_va(self.va_slot);
268        }
269    }
