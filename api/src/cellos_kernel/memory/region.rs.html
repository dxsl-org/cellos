1//! One record behind the five region tables (architecture remediation
2//! Phase 16, census §7.7).
3//!
4//! `PAGE_GRANT_TABLE` (`GrantAlloc`), `REG_GRANT_TABLE` (`GrantRegister`), the
5//! SHM handles (`ShmAlloc`), the DMA export leases (`EXPORTS`) and the domain
6//! grant records (carried by grant rows) each kept their own row type, with its
7//! own owner, rights and receiver fields and its own idea of who frees the
8//! frames. Each table now stores one [`Region`]: what kind it is, who owns it,
9//! what it allows, who receives it, what it holds in the pin registry, and its
10//! frames — a [`FrameRun`] the region owns until its release takes it, or the
11//! range of another region's frames it locks (a DMA export). The SHM regions
12//! live in the `GrantAlloc` table ([`RegionTable`]), where only the SHM
13//! syscalls see them. Every table keeps its lock, its rank and its key, so the
14//! syscall-visible ids (a grant's or an SHM frame's base address, an export id)
15//! do not change.
16//!
17//! A region's frames leave it only through [`Region::into_run`], which consumes
18//! the region: whoever took the row out of its table is the one releaser (the
19//! single-free rule, census risk 1). A region dropped while it still owns a run
20//! never frees it: the run is leaked, counted and logged (`FrameRun`'s drop).
21
22use crate::memory::frame_run::FrameRun;
23use crate::memory::pin::ReleaseKey;
24use alloc::collections::BTreeMap;
25use alloc::vec::Vec;
26use types::{CellId, GrantPerm};

28/// The lifecycle record a domain-backed grant row carries (owner and receiver
29/// private roots, `Live → Revoking → Revoked`), on targets with the
30/// private-root grant lifecycle.
31#[cfg(cellos_domain_grants)]
32pub(crate) type DomainRecord = alloc::sync::Arc<crate::task::domain_grant::DomainGrant>;
33
34/// Elsewhere no row can carry a lifecycle record: the slot is always `None`.
35#[cfg(not(cellos_domain_grants))]
36pub(crate) type DomainRecord = ();

38/// What a region is, and so which syscalls may name it.
39#[derive(Clone, Copy, Debug, Eq, PartialEq)]
40pub(crate) enum RegionKind {
41    /// An `ShmAlloc` frame (syscalls 20/21); id = the frame's address.
42    Shm,
43    /// A `GrantAlloc` (208) or `GrantRegister` (215) buffer, by its table;
44    /// id = the base address.
45    Grant,
46    /// An `ExportDmaRange` (437) lease; id = the export id.
47    DmaExport,
48}
49
50/// Who owns a region: a task and the cell incarnation it ran as.
51///
52/// `tid` is 0 for a region an incarnation owns rather than a task (SHM), and
53/// for a region nobody owns any more (an orphaned registered grant, or a
54/// domain row left ownerless while its revoke waits): no caller has tid 0, so
55/// neither matches an owner check.
56#[derive(Clone, Copy, Debug, Eq, PartialEq)]
57pub(crate) struct RegionOwner {
58    pub(crate) tid: usize,
59    pub(crate) cell: CellId,
60    pub(crate) generation: u64,
61}
62
63impl RegionOwner {
64    /// Nobody.
65    pub(crate) const NONE: Self = Self {
66        tid: 0,
67        cell: CellId(0),
68        generation: 0,
69    };
70
71    /// Task `tid` of incarnation `(cell, generation)`.
72    pub(crate) const fn task(tid: usize, cell: CellId, generation: u64) -> Self {
73        Self {
74            tid,
75            cell,
76            generation,
77        }
78    }
79
80    /// Incarnation `(cell, generation)` itself, no task.
81    pub(crate) const fn incarnation(cell: u64, generation: u64) -> Self {
82        Self {
83            tid: 0,
84            cell: CellId(cell),
85            generation,
86        }
87    }
88
89    /// `(cell, generation)`.
90    pub(crate) const fn binding(&self) -> (CellId, u64) {
91        (self.cell, self.generation)
92    }
93}

95/// Who receives a region; one receiver at most in every table.
96#[derive(Clone, Copy, Debug, Eq, PartialEq)]
97pub(crate) enum Receiver {
98    None,
99    /// The task a grant is shared with (`GrantShare`), with its permission.
100    Task {
101        tid: usize,
102        perm: GrantPerm,
103    },
104    /// The cell an SHM frame was handed to (`ShmMap`'s target).
105    Cell(u64),
106    /// The provider incarnation that holds a DMA export.
107    Holder {
108        cell: u64,
109        generation: u64,
110    },
111}
112
113impl Receiver {
114    /// The grantee task and its permission, if this is a grant receiver.
115    pub(crate) const fn task(&self) -> Option<(usize, GrantPerm)> {
116        match *self {
117            Receiver::Task { tid, perm } => Some((tid, perm)),
118            _ => None,
119        }
120    }
121}

123/// What a region itself holds in the pin registry (`memory::pin`).
124#[derive(Clone, Copy, Debug, Eq, PartialEq)]
125pub(crate) enum PinState {
126    /// No range lock of its own. An in-flight operation may still pin part of
127    /// the region (`pin::holder_of`); its release then withholds the run under
128    /// that operation's key (`pin::withhold_pinned_run`).
129    Unlocked,
130    /// The region is a range lock over frames another region owns, under
131    /// `key`, the key its quarantine holds are released by (a DMA export,
132    /// `pin::lock_dma_export`).
133    Locked(ReleaseKey),
134}
135
136/// A region's frames.
137enum Backing {
138    /// Owned: `Mapped` until the region's release takes the run.
139    Run(FrameRun),
140    /// `[base, base + size)` of frames another region owns.
141    Borrowed { base: usize },
142}
143
144/// One row of a region table. `X` is the table's own state, if it has any
145/// (a DMA export's lifecycle and device window).
146pub(crate) struct Region<X = ()> {
147    kind: RegionKind,
148    pub(crate) owner: RegionOwner,
149    /// Bytes the region was asked for (and charged): a grant's or an export's
150    /// frames cover them rounded up to whole pages; an SHM handle is one frame
151    /// whatever it was charged.
152    pub(crate) size: usize,
153    /// What the region allows: its owner (a grant or SHM frame is `ReadWrite`)
154    /// or its device holder (a DMA export).
155    pub(crate) rights: GrantPerm,
156    pub(crate) receiver: Receiver,
157    pub(crate) pin: PinState,
158    /// The private-root lifecycle record of a domain-backed grant.
159    pub(crate) domain: Option<DomainRecord>,
160    backing: Backing,
161    pub(crate) ext: X,
162}
163
164impl Region {
165    /// A `GrantAlloc` or `GrantRegister` row owning `run`, not yet shared.
166    pub(crate) fn grant(
167        owner: RegionOwner,
168        run: FrameRun,
169        size: usize,
170        domain: Option<DomainRecord>,
171    ) -> Self {
172        Self {
173            kind: RegionKind::Grant,
174            owner,
175            size,
176            rights: GrantPerm::ReadWrite,
177            receiver: Receiver::None,
178            pin: PinState::Unlocked,
179            domain,
180            backing: Backing::Run(run),
181            ext: (),
182        }
183    }
184
185    /// An `ShmAlloc` frame owned by an incarnation, not yet handed to a cell.
186    pub(crate) fn shm(owner: RegionOwner, run: FrameRun, size: usize) -> Self {
187        Self {
188            kind: RegionKind::Shm,
189            owner,
190            size,
191            rights: GrantPerm::ReadWrite,
192            receiver: Receiver::None,
193            pin: PinState::Unlocked,
194            domain: None,
195            backing: Backing::Run(run),
196            ext: (),
197        }
198    }
199}

201impl<X> Region<X> {
202    /// A DMA export: a range lock (`key`) over `[base, base + size)` of a
203    /// grant `owner` owns, held by `holder`.
204    pub(crate) fn export(
205        owner: RegionOwner,
206        base: usize,
207        size: usize,
208        rights: GrantPerm,
209        holder: Receiver,
210        key: ReleaseKey,
211        ext: X,
212    ) -> Self {
213        Self {
214            kind: RegionKind::DmaExport,
215            owner,
216            size,
217            rights,
218            receiver: holder,
219            pin: PinState::Locked(key),
220            domain: None,
221            backing: Backing::Borrowed { base },
222            ext,
223        }
224    }
225
226    /// The first address of the region's frames.
227    pub(crate) fn base(&self) -> usize {
228        match &self.backing {
229            Backing::Run(run) => run.base(),
230            Backing::Borrowed { base } => *base,
231        }
232    }
233
234    /// The grantee and its permission (`GrantShare`), if any.
235    pub(crate) const fn shared_to(&self) -> Option<(usize, GrantPerm)> {
236        self.receiver.task()
237    }
238
239    /// Share with `grantee` (`Some`) or with nobody.
240    pub(crate) fn set_shared_to(&mut self, grantee: Option<(usize, GrantPerm)>) {
241        self.receiver = match grantee {
242            Some((tid, perm)) => Receiver::Task { tid, perm },
243            None => Receiver::None,
244        };
245    }
246
247    /// Leave the region with neither owner nor receiver: no caller can act on
248    /// it any more, only the reaper that releases it.
249    pub(crate) fn orphan(&mut self) {
250        self.owner = RegionOwner::NONE;
251        self.receiver = Receiver::None;
252    }
253
254    /// The region's own frames, for the one caller whose removal took the row
255    /// to release; `None` for a region that only locks another's frames.
256    pub(crate) fn into_run(self) -> Option<FrameRun> {
257        match self.backing {
258            Backing::Run(run) => Some(run),
259            Backing::Borrowed { .. } => None,
260        }
261    }
262
263    /// Host tests: drop a row whose run names synthetic frames (no allocator
264    /// owns them) without counting a leak.
265    #[cfg(test)]
266    pub(crate) fn discard_synthetic(self) {
267        if let Some(run) = self.into_run() {
268            core::mem::forget(run);
269        }
270    }
271}

273/// One grant table's rows, keyed by id.
274///
275/// The `GrantAlloc` table also holds the SHM regions ([`RegionKind::Shm`],
276/// keyed by the frame's address): SHM rides on the grant table and its
277/// release path, and keeps its own syscalls. Only the `shm*` methods see those
278/// rows. Every other method is the grant view and skips them, so no grant path
279/// (`GrantShare`, `GrantSlice`, `GrantFree`, `GrantDma`, an export's ownership
280/// proof, the SAS output ledger, a reaper) can resolve, change or remove an SHM
281/// handle, and `ShmMap` resolves nothing but SHM handles. Two rows never share
282/// an id: each is the address of a frame its region owns.
283pub(crate) struct RegionTable {
284    rows: BTreeMap<usize, Region>,
285}
286
287fn is_grant(region: &Region) -> bool {
288    region.kind != RegionKind::Shm
289}
290
291impl RegionTable {
292    pub(crate) const fn new() -> Self {
293        Self {
294            rows: BTreeMap::new(),
295        }
296    }
297
298    /// The grant row `id`.
299    pub(crate) fn get(&self, id: &usize) -> Option<&Region> {
300        self.rows.get(id).filter(|region| is_grant(region))
301    }
302
303    pub(crate) fn get_mut(&mut self, id: &usize) -> Option<&mut Region> {
304        self.rows.get_mut(id).filter(|region| is_grant(region))
305    }
306
307    /// Take grant row `id` out of the table; an SHM row stays.
308    pub(crate) fn remove(&mut self, id: &usize) -> Option<Region> {
309        self.get(id)?;
310        self.rows.remove(id)
311    }
312
313    /// The grant rows.
314    pub(crate) fn iter(&self) -> impl Iterator<Item = (&usize, &Region)> {
315        self.rows.iter().filter(|(_, region)| is_grant(region))
316    }
317
318    pub(crate) fn iter_mut(&mut self) -> impl Iterator<Item = (&usize, &mut Region)> {
319        self.rows.iter_mut().filter(|(_, region)| is_grant(region))
320    }
321
322    pub(crate) fn values(&self) -> impl Iterator<Item = &Region> {
323        self.iter().map(|(_, region)| region)
324    }
325
326    pub(crate) fn values_mut(&mut self) -> impl Iterator<Item = &mut Region> {
327        self.iter_mut().map(|(_, region)| region)
328    }
329
330    /// Does any row, grant or SHM, hold `id`? Only the RV64 grant race
331    /// fixture reissues a grant at a chosen address.
332    #[cfg(any(test, all(feature = "getrandom-sas-test", target_arch = "riscv64")))]
333    pub(crate) fn occupied(&self, id: &usize) -> bool {
334        self.rows.contains_key(id)
335    }
336
337    /// Insert `region` at `id`, or hand it back (`Some`) if a row of any kind
338    /// holds `id`; that row stays. Each id is the address of a frame its row owns,
339    /// so an occupied id means two runs name the same frames: replacing the
340    /// row would drop its run (leaked) while the new row went on to free those
341    /// frames. A refused region's run is leaked and logged when the caller
342    /// drops it (`FrameRun`'s drop), never freed.
343    #[must_use = "a refused region is dropped with no table lock held"]
344    pub(crate) fn insert(&mut self, id: usize, region: Region) -> Option<Region> {
345        match self.rows.entry(id) {
346            alloc::collections::btree_map::Entry::Occupied(_) => Some(region),
347            alloc::collections::btree_map::Entry::Vacant(slot) => {
348                slot.insert(region);
349                None
350            }
351        }
352    }
353
354    /// The SHM region `handle`.
355    pub(crate) fn shm(&self, handle: &usize) -> Option<&Region> {
356        self.rows.get(handle).filter(|region| !is_grant(region))
357    }
358
359    pub(crate) fn shm_mut(&mut self, handle: &usize) -> Option<&mut Region> {
360        self.rows.get_mut(handle).filter(|region| !is_grant(region))
361    }
362
363    /// Take every SHM region `owner` owns out of the table.
364    pub(crate) fn take_shm_of(&mut self, owner: RegionOwner) -> Vec<Region> {
365        let handles: Vec<usize> = self
366            .rows
367            .iter()
368            .filter(|(_, region)| !is_grant(region) && region.owner == owner)
369            .map(|(&handle, _)| handle)
370            .collect();
371        handles
372            .iter()
373            .filter_map(|handle| self.rows.remove(handle))
374            .collect()
375    }
376}
