1//! Async Pinning Registry — memory regions an in-flight asynchronous operation
2//! may still be reading or writing.
3//!
4//! # Contract
5//!
6//! A region is *pinned* while an agent outside the owning cell's control — a
7//! device programmed through the IOMMU, a kernel ISR, or a service holding a raw
8//! pointer into the region — may still touch its frames. While a region is
9//! pinned:
10//!
11//! * the owner may not free it: `GrantFree` and `GrantUnregister` refuse rather
12//!   than hand the frames back to the allocator, and
13//! * if the owner dies, the frames are **quarantined** — withheld from the frame
14//!   allocator instead of returned to it.
15//!
16//! Quarantine is the third option between the two failure modes. Returning a
17//! pinned frame leaves a device writing into whatever cell is allocated it next;
18//! blocking the death until the pin clears breaks the never-die property, since
19//! init restarts every permanent service and a cell dying mid-operation is the
20//! ordinary case. Nothing here ever delays a death: marking is a bounded scan of
21//! a fixed table and takes no other lock.
22//!
23//! Quarantine ends only on an explicit driver acknowledgement
24//! ([`acknowledge`]) — never on a timer, never implicitly. Frames with no
25//! acknowledgement stay withheld for the life of the boot, which leaks memory
26//! but cannot corrupt another cell.
27//!
28//! # Lock order
29//!
30//! `REGISTRY` is a **leaf**: no other lock is acquired while it is held, and it
31//! is never held across `FRAME_ALLOCATOR`, `KERNEL_ROOT` or `SCHEDULER`. The
32//! grant teardown path therefore runs
33//!
34//! ```text
35//! PAGE_GRANT_TABLE / REG_GRANT_TABLE → REGISTRY → (all released) → FRAME_ALLOCATOR → KERNEL_ROOT
36//! ```
37//!
38//! `FRAME_ALLOCATOR` is the OUTER lock of the last pair, contrary to what the
39//! older comments on `reap_grants_for_task` and `yield_cpu` claimed:
40//! `free_grant_pages` takes `FRAME_ALLOCATOR` first and holds it across
41//! `unmap_page`/`map_page`, each of which takes `KERNEL_ROOT` internally.
42//! Neither is ever acquired while `SCHEDULER` is held — the watchdog defers its
43//! reap list precisely so `yield_cpu` can drain it after dropping `SCHEDULER`.
44

50/// Regions the kernel tracks as pinned at one time, across all cells.
51///
52/// A fixed table rather than a map: the count must not be driven by anything a
53/// caller supplies, and the teardown paths that scan it run on cell death.
54const MAX_PINNED_REGIONS: usize = 128;
55
56/// Per-owner ceiling, so one driver cell cannot starve the table.
57///
58/// Sized from the largest real consumer: the e1000 cell authorises 34 regions
59/// (two rings plus 16 TX and 16 RX buffers).
60const MAX_PINS_PER_TASK: usize = 48;
61
62/// Frame ranges the quarantine can hold at one time. Entries are only created
63/// by the reaper, one per grant a dying cell had pinned, so the bound is the
64/// number of grants that can be in flight across all dead cells at once.
65const MAX_QUARANTINED_REGIONS: usize = 64;
66pub(crate) const MAX_VFS_LEASES: usize = 32;

90/// Who holds a pin overlapping a queried region.
91#[derive(Debug, Clone, Copy)]
92pub struct PinHolder {
93    /// Page-aligned base of the pinned region.
94    pub base: usize,
95    /// Length of the pinned region in pages.
96    pub pages: usize,
97    /// Task id that owns the memory participating in the operation.
98    pub owner: usize,
99    /// Outstanding pin requests against this region. Released together by
100    /// [`acknowledge`]; there is no per-request release today.
101    pub holds: u32,
102    /// The owner has died and these frames are withheld from the allocator.
103    pub quarantined: bool,
104    /// The grant owner died; only the active holder may complete this lease.
105    pub pending_revoke: bool,
106    /// Exact VFS request generation when this hold belongs to a VFS lease.
107    pub request_generation: u64,
108    /// VFS task holding the lease, or 0 for owner-wide DMA pins.
109    pub holder_tid: usize,
110}

178/// Result of the reaper's single-transaction pin-to-quarantine transfer.
179///
180/// `Free` means no live pin protected the range at the transfer linearization
181/// point, so the reaper may return the frames to the allocator. `Full` means a
182/// live pin did protect them but the bounded quarantine has no slot; callers
183/// must leak rather than free in that case.
184#[derive(Clone, Copy, Debug, PartialEq, Eq)]
185pub(crate) enum FrameTransfer {
186    Free,
187    Withheld,
188    Full,
189}

235/// One lock covers both tables: they are only ever consulted together, and a
236/// single leaf lock removes any ordering question between them.
237static REGISTRY: Spinlock<Registry> = Spinlock::new(Registry {
238    pins: [EMPTY_PIN; MAX_PINNED_REGIONS],
239    cache_sync: [EMPTY_CACHE_SYNC; MAX_PINNED_REGIONS],
240    next_cache_token: 1,
241    vfs_leases: [EMPTY_VFS_LEASE; MAX_VFS_LEASES],
242    quarantine: [EMPTY_HOLD; MAX_QUARANTINED_REGIONS],
243});

245/// Page-aligned `(base, page_count)` covering `[base, base + len)`.
246fn span(base: usize, len: usize) -> Option<(usize, usize)> {
247    if len == 0 {
248        return None;
249    }
250    let start = base & !(PAGE_SIZE - 1);
251    let end = base.checked_add(len)?.checked_add(PAGE_SIZE - 1)? & !(PAGE_SIZE - 1);
252    Some((start, (end - start) / PAGE_SIZE))
253}
254
255fn overlaps(a: (usize, usize), b: (usize, usize)) -> bool {
256    a.0 < b.0 + b.1 * PAGE_SIZE && b.0 < a.0 + a.1 * PAGE_SIZE
257}

310fn hold_quarantined_frames(
311    reg: &mut Registry,
312    base: usize,
313    pages: usize,
314    release: ReleaseKey,
315) -> bool {
316    if pages == 0 {
317        return false;
318    }
319    match reg.quarantine.iter_mut().find(|entry| entry.pages == 0) {
320        Some(slot) => {
321            *slot = FrameHold {
322                base,
323                pages,
324                release,
325            };
326            true
327        }
328        None => false,
329    }
330}

332fn release_quarantined_frames<F>(reg: &mut Registry, mut matches: F) -> Vec<(usize, usize)>
333where
334    F: FnMut(ReleaseKey) -> bool,
335{
336    let mut released = Vec::new();
337    for hold in reg.quarantine.iter_mut() {
338        if hold.pages != 0 && matches(hold.release) {
339            released.push((hold.base, hold.pages));
340            *hold = EMPTY_HOLD;
341        }
342    }
343    released
344}

346/// Record `[base, base + len)` as participating in an in-flight operation owned
347/// by task `owner`, so the region cannot be freed underneath it.
348///
349/// Re-pinning a range already held by the same owner bumps its hold count
350/// instead of consuming a second slot.
351///
352/// # Errors
353/// [`PinError::InvalidRange`] for an empty or overflowing range,
354/// [`PinError::TaskLimit`] once the owner holds [`MAX_PINS_PER_TASK`] regions,
355/// [`PinError::TableFull`] when no slot is left. Callers must fail the operation
356/// closed: authorising a device against memory the kernel cannot protect is the
357/// use-after-free this registry exists to prevent.
358pub fn pin(base: usize, len: usize, owner: usize) -> Result<(), PinError> {
359    let (start, pages) = span(base, len).ok_or(PinError::InvalidRange)?;
360    let mut reg = REGISTRY.lock();
361    let mut owned = 0usize;
362    let mut free_slot = None;
363    let mut existing = None;
364    for (i, e) in reg.pins.iter().enumerate() {
365        if e.pages == 0 {
366            if free_slot.is_none() {
367                free_slot = Some(i);
368            }
369            continue;
370        }
371        if e.owner == owner {
372            owned += 1;
373            if e.base == start && e.pages == pages && !e.quarantined {
374                existing = Some(i);
375                break;
376            }
377        }
378    }
379    if let Some(i) = existing {
380        reg.pins[i].holds = reg.pins[i].holds.saturating_add(1);
381        return Ok(());
382    }
383    if owned >= MAX_PINS_PER_TASK {
384        return Err(PinError::TaskLimit);
385    }
386    let slot = free_slot.ok_or(PinError::TableFull)?;
387    reg.pins[slot] = PinEntry {
388        base: start,
389        pages,
390        owner,
391        holds: 1,
392        quarantined: false,
393    };
394    Ok(())
395}

397/// Reserve one exact cache synchronization pin and return its non-zero token.
398///
399/// The caller must prove Grant ownership and logical byte bounds before calling
400/// this leaf function. `complete_cache_sync` is the only normal release path.
401pub fn begin_cache_sync(base: usize, len: usize, owner: usize) -> Result<usize, PinError> {
402    let (start, pages) = span(base, len).ok_or(PinError::InvalidRange)?;
403    let mut reg = REGISTRY.lock();
404    let owned = reg
405        .pins
406        .iter()
407        .filter(|entry| entry.pages != 0 && entry.owner == owner)
408        .count()
409        + reg
410            .cache_sync
411            .iter()
412            .filter(|entry| entry.pages != 0 && entry.owner == owner)
413            .count();
414    if owned >= MAX_PINS_PER_TASK {
415        return Err(PinError::TaskLimit);
416    }
417    let slot = reg
418        .cache_sync
419        .iter()
420        .position(|entry| entry.pages == 0)
421        .ok_or(PinError::TableFull)?;
422    let mut token = reg.next_cache_token;
423    if token == 0 {
424        token = 1;
425    }
426    reg.next_cache_token = token.wrapping_add(1);
427    if reg.next_cache_token == 0 {
428        reg.next_cache_token = 1;
429    }
430    reg.cache_sync[slot] = CacheSyncEntry {
431        operation_base: base,
432        operation_len: len,
433        base: start,
434        pages,
435        owner,
436        token,
437        completing: false,
438        quarantined: false,
439    };
440    Ok(token)
441}

443/// Return the exact owned cache-sync span for `token` without releasing it.
444pub fn begin_cache_sync_completion(token: usize, owner: usize) -> Option<(usize, usize)> {
445    if token == 0 {
446        return None;
447    }
448    let mut reg = REGISTRY.lock();
449    reg.cache_sync
450        .iter_mut()
451        .find(|entry| {
452            entry.pages != 0 && entry.owner == owner && entry.token == token && !entry.completing
453        })
454        .and_then(|entry| {
455            entry.completing = true;
456            Some((entry.operation_base, entry.operation_len))
457        })
458}

460/// Drop only the cache-sync operation authenticated by `(owner, token)`.
461pub fn complete_cache_sync(token: usize, owner: usize) -> bool {
462    if token == 0 {
463        return false;
464    }
465    let mut reg = REGISTRY.lock();
466    let Some(entry) = reg.cache_sync.iter_mut().find(|entry| {
467        entry.pages != 0 && entry.owner == owner && entry.token == token && entry.completing
468    }) else {
469        return false;
470    };
471    *entry = EMPTY_CACHE_SYNC;
472    true
473}

492/// The first pin overlapping `[base, base + len)`, if any.
493///
494/// Teardown paths call this to decide whether frames may be released.
495pub fn holder_of(base: usize, len: usize) -> Option<PinHolder> {
496    let probe = span(base, len)?;
497    let reg = REGISTRY.lock();
498    reg.pins
499        .iter()
500        .find(|e| e.pages != 0 && overlaps(probe, (e.base, e.pages)))
501        .map(pin_holder_from_pin)
502        .or_else(|| {
503            reg.cache_sync
504                .iter()
505                .find(|entry| entry.pages != 0 && overlaps(probe, (entry.base, entry.pages)))
506                .map(pin_holder_from_cache_sync)
507        })
508        .or_else(|| {
509            reg.vfs_leases
510                .iter()
511                .find(|e| e.pages != 0 && overlaps(probe, (e.base, e.pages)))
512                .map(pin_holder_from_vfs_lease)
513        })
514}

516/// The overlapping VFS lease for `grant_owner`, if any.
517///
518/// Grant teardown uses this while holding a grant-table lock to distinguish a
519/// live VFS per-request lease from a plain shared registered grant. The lock
520/// order stays grant table -> pin registry because this function takes only the
521/// pin leaf.
522pub(crate) fn vfs_holder_of_owner(
523    base: usize,
524    len: usize,
525    grant_owner: usize,
526) -> Option<PinHolder> {
527    let probe = span(base, len)?;
528    let reg = REGISTRY.lock();
529    reg.vfs_leases
530        .iter()
531        .find(|e| {
532            e.pages != 0 && e.grant_owner == grant_owner && overlaps(probe, (e.base, e.pages))
533        })
534        .map(pin_holder_from_vfs_lease)
535}

537/// Mark every non-VFS pin owned by `tid` as quarantined and report how many.
538///
539/// Called from the grant reaper before the grant tables are swept. Marking does
540/// not free anything and does not wait: the dying task proceeds immediately.
541pub fn quarantine_task(tid: usize) -> usize {
542    let mut reg = REGISTRY.lock();
543    let mut marked = 0;
544    for e in reg.pins.iter_mut() {
545        if e.pages != 0 && e.owner == tid {
546            e.quarantined = true;
547            marked += 1;
548        }
549    }
550    for entry in reg.cache_sync.iter_mut() {
551        if entry.pages != 0 && entry.owner == tid {
552            entry.quarantined = true;
553            marked += 1;
554        }
555    }
556    // VFS leases are per-request and held by another task. Their owner death
557    // is marked by `mark_vfs_lease_pending_revoke` from the scheduler while it
558    // still has the holder+owner+generation context; broad owner-wide marking
559    // would let an unrelated holder release a live raw pointer.
560    marked
561}

563/// Take custody of `pages` frames at `base` that the reaper declined to free,
564/// charging them to the pin holder `owner` whose acknowledgement releases them.
565///
566/// Only the reaper calls this, and only for frames it was otherwise about to
567/// return to the allocator. That is what makes [`acknowledge`] safe to free
568/// blindly: a range that never came out of a grant — an MMIO window a driver
569/// authorised for DMA, or a grant whose ownership passed to a live grantee —
570/// never enters the quarantine and so is never handed to `deallocate_frame`.
571///
572/// Returns `false` when the quarantine is full. The caller must still not free
573/// the frames: they are leaked for the life of the boot, which is the safe
574/// direction.
575pub fn withhold_frames(base: usize, pages: usize, owner: usize) -> bool {
576    let mut reg = REGISTRY.lock();
577    hold_quarantined_frames(&mut reg, base, pages, ReleaseKey::Owner(owner))
578}

580/// Take custody of `pages` VFS-lease frames that the reaper declined to free.
581///
582/// The exact lease is checked while holding the registry lock before the
583/// quarantine entry is installed. A missing lease therefore cannot leave an
584/// orphaned [`ReleaseKey::VfsLease`] that nobody can release.
585pub fn withhold_vfs_frames(
586    base: usize,
587    pages: usize,
588    holder_tid: usize,
589    grant_owner: usize,
590    request_generation: u64,
591) -> bool {
592    let mut reg = REGISTRY.lock();
593    if !reg
594        .vfs_leases
595        .iter()
596        .any(|entry| same_vfs_lease(entry, holder_tid, grant_owner, request_generation))
597    {
598        return false;
599    }
600    hold_quarantined_frames(
601        &mut reg,
602        base,
603        pages,
604        ReleaseKey::VfsLease {
605            holder_tid,
606            grant_owner,
607            request_generation,
608        },
609    )
610}

612/// Atomically decide whether a dying owner's frames must be quarantined.
613///
614/// Lookup and custody transfer share the `REGISTRY` transaction. In
615/// particular, a VFS holder that completes its exact lease first removes that
616/// lease before this function can install its release key, and the reaper then
617/// receives [`FrameTransfer::Free`]. If this function wins, that same exact
618/// release key owns the quarantine entry until the holder releases it.
619pub(crate) fn withhold_pinned_frames(base: usize, pages: usize) -> FrameTransfer {
620    let Some(len) = pages.checked_mul(PAGE_SIZE) else {
621        return FrameTransfer::Free;
622    };
623    let Some(probe) = span(base, len) else {
624        return FrameTransfer::Free;
625    };
626    let mut reg = REGISTRY.lock();
627
628    if let Some(entry) = reg
629        .pins
630        .iter()
631        .find(|entry| entry.pages != 0 && overlaps(probe, (entry.base, entry.pages)))
632    {
633        let owner = entry.owner;
634        return if hold_quarantined_frames(&mut reg, base, pages, ReleaseKey::Owner(owner)) {
635            FrameTransfer::Withheld
636        } else {
637            FrameTransfer::Full
638        };
639    }
640
641    if reg
642        .cache_sync
643        .iter()
644        .any(|entry| entry.pages != 0 && overlaps(probe, (entry.base, entry.pages)))
645    {
646        return if hold_quarantined_frames(&mut reg, base, pages, ReleaseKey::CacheSyncBootLifetime)
647        {
648            FrameTransfer::Withheld
649        } else {
650            FrameTransfer::Full
651        };
652    }
653
654    let Some(entry) = reg
655        .vfs_leases
656        .iter()
657        .find(|entry| entry.pages != 0 && overlaps(probe, (entry.base, entry.pages)))
658    else {
659        return FrameTransfer::Free;
660    };
661    let release = ReleaseKey::VfsLease {
662        holder_tid: entry.holder_tid,
663        grant_owner: entry.grant_owner,
664        request_generation: entry.request_generation,
665    };
666    if hold_quarantined_frames(&mut reg, base, pages, release) {
667        FrameTransfer::Withheld
668    } else {
669        FrameTransfer::Full
670    }
671}

673/// IOMMU-driver acknowledgement for `tid`: drop its ordinary pins and hand back the
674/// `(base, pages)` frame ranges the reaper placed in quarantine on its behalf.
675///
676/// The caller returns those frames to the allocator. Safe to call before the
677/// reaper as well as after — an acknowledgement that arrives first simply drops
678/// ordinary pins, and the reaper then frees the frames itself. Cache-sync pins
679/// deliberately do not participate: a VideoCore response has no trustworthy
680/// driver acknowledgement and remains boot-lifetime after task death.
681pub fn acknowledge(tid: usize) -> Vec<(usize, usize)> {
682    let mut reg = REGISTRY.lock();
683    for e in reg.pins.iter_mut() {
684        if e.pages != 0 && e.owner == tid {
685            *e = EMPTY_PIN;
686        }
687    }
688    release_quarantined_frames(&mut reg, |release| release == ReleaseKey::Owner(tid))
689}
690
691pub fn pin_vfs_lease(
692    base: usize,
693    len: usize,
694    grant_owner: usize,
695    holder_tid: usize,
696    grant_id: usize,
697    request_generation: u64,
698) -> Result<(), VfsLeaseError> {
699    let (start, pages) = span(base, len).ok_or(VfsLeaseError::InvalidRange)?;
700    let mut reg = REGISTRY.lock();
701    let mut owned = 0usize;
702    let mut free_slot = None;
703    for (i, e) in reg.vfs_leases.iter().enumerate() {
704        if e.pages == 0 {
705            if free_slot.is_none() {
706                free_slot = Some(i);
707            }
708            continue;
709        }
710        if e.holder_tid == holder_tid {
711            owned += 1;
712        }
713        if same_vfs_lease(e, holder_tid, grant_owner, request_generation) {
714            return Err(VfsLeaseError::DuplicateLease);
715        }
716    }
717    if owned >= MAX_PINS_PER_TASK {
718        return Err(VfsLeaseError::TaskLimit);
719    }
720    let slot = free_slot.ok_or(VfsLeaseError::TableFull)?;
721    reg.vfs_leases[slot] = VfsLeaseEntry {
722        base: start,
723        pages,
724        grant_owner,
725        holder_tid,
726        grant_id,
727        request_generation,
728        quarantined: false,
729        pending_revoke: false,
730    };
731    Ok(())
732}
733
734/// Mark one live holder's lease revoked because its grant owner has died.
735///
736/// This preserves the pin and caller context until the holder completes or
737/// drops that exact request, while preventing another `GrantSlice` from being
738/// issued from the stale context.
739pub fn mark_vfs_lease_pending_revoke(
740    holder_tid: usize,
741    grant_owner: usize,
742    request_generation: u64,
743) -> bool {
744    let mut reg = REGISTRY.lock();
745    let Some(entry) = reg
746        .vfs_leases
747        .iter_mut()
748        .find(|entry| same_vfs_lease(entry, holder_tid, grant_owner, request_generation))
749    else {
750        return false;
751    };
752    entry.quarantined = true;
753    entry.pending_revoke = true;
754    true
755}
756
757/// Whether the exact live VFS lease has been revoked by owner death.
758pub fn vfs_lease_pending_revoke(
759    holder_tid: usize,
760    grant_owner: usize,
761    request_generation: u64,
762) -> bool {
763    let reg = REGISTRY.lock();
764    reg.vfs_leases
765        .iter()
766        .find(|entry| same_vfs_lease(entry, holder_tid, grant_owner, request_generation))
767        .is_some_and(|entry| entry.pending_revoke)
768}
769
770pub fn release_vfs_lease(
771    holder_tid: usize,
772    grant_owner: usize,
773    request_generation: u64,
774) -> Vec<(usize, usize)> {
775    let mut reg = REGISTRY.lock();
776    for e in reg.vfs_leases.iter_mut() {
777        if same_vfs_lease(e, holder_tid, grant_owner, request_generation) {
778            *e = EMPTY_VFS_LEASE;
779        }
780    }
781    release_quarantined_frames(&mut reg, |release| {
782        release
783            == ReleaseKey::VfsLease {
784                holder_tid,
785                grant_owner,
786                request_generation,
787            }
788    })
789}
790
791pub fn release_vfs_holder_leases(holder_tid: usize) -> Vec<(usize, usize)> {
792    let mut reg = REGISTRY.lock();
793    for e in reg.vfs_leases.iter_mut() {
794        if e.pages != 0 && e.holder_tid == holder_tid {
795            *e = EMPTY_VFS_LEASE;
796        }
797    }
798    release_quarantined_frames(&mut reg, |release| {
799        matches!(
800            release,
801            ReleaseKey::VfsLease {
802                holder_tid: lease_holder,
803                ..
804            } if lease_holder == holder_tid
805        )
806    })
807}
808
809pub fn find_vfs_lease(
810    holder_tid: usize,
811    grant_owner: usize,
812    request_generation: u64,
813) -> Option<(usize, usize, usize)> {
814    let reg = REGISTRY.lock();
815    reg.vfs_leases
816        .iter()
817        .find(|e| same_vfs_lease(e, holder_tid, grant_owner, request_generation))
818        .map(|e| (e.base, e.pages, e.grant_id))
819}

821/// Pages currently withheld from the frame allocator.
822pub fn quarantined_pages() -> usize {
823    let reg = REGISTRY.lock();
824    reg.quarantine.iter().map(|q| q.pages).sum()
825}
