1//! Private completion boundary for permission-lowering and unmap TLB maintenance.
2//!
3//! No caller may execute newly restricted code or recycle a retired VA/frame
4//! until this function returns. RV64 uses SBI RFENCE for every online remote
5//! hart; other targets retain their established local or broadcast HAL paths.
6
7use crate::memory::paging::PAGE_SIZE;
8use types::VAddr;
9
10#[cfg(feature = "test-hooks")]
11use core::sync::atomic::{AtomicBool, Ordering};
12
13/// Test-only negative-control switch. It is absent from production kernels.
14#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
15static TEST_SKIP_REMOTE_RFENCE: AtomicBool = AtomicBool::new(false);
16
17#[cfg(feature = "test-hooks")]
18static TEST_FLUSH_TRACKING: AtomicBool = AtomicBool::new(false);
19#[cfg(feature = "test-hooks")]
20static TEST_FLUSHED_PAGES: crate::sync::Spinlock<alloc::vec::Vec<VAddr>> =
21    crate::sync::Spinlock::new(alloc::vec::Vec::new());
22
23#[cfg(feature = "test-hooks")]
24pub(crate) fn begin_test_flush_observation() {
25    TEST_FLUSHED_PAGES.lock().clear();
26    TEST_FLUSH_TRACKING.store(true, Ordering::Release);
27}
28
29#[cfg(feature = "test-hooks")]
30pub(crate) fn test_flush_observed(vaddr: VAddr) -> bool {
31    let page = vaddr & !(PAGE_SIZE - 1);
32    TEST_FLUSHED_PAGES.lock().iter().any(|&entry| entry == page)
33}
34
35#[cfg(feature = "test-hooks")]
36pub(crate) fn finish_test_flush_observation() {
37    TEST_FLUSH_TRACKING.store(false, Ordering::Release);
38}
39
40/// Enable the test-only negative control around one known self-test operation.
41#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
42pub(crate) fn set_test_skip_remote_rfence(enabled: bool) {
43    TEST_SKIP_REMOTE_RFENCE.store(enabled, Ordering::Release);
44}
45
46/// Invalidate the translation for one changed page before its memory can run or reuse.
47#[inline]
48pub fn flush_page(vaddr: VAddr) {
49    let page = vaddr & !(PAGE_SIZE - 1);
50    #[cfg(feature = "test-hooks")]
51    if TEST_FLUSH_TRACKING.load(Ordering::Acquire) {
52        TEST_FLUSHED_PAGES.lock().push(page);
53    }
54    flush_range(page, PAGE_SIZE);
55}
56
57/// Invalidate a page-aligned range after its PTEs have changed.
58pub fn flush_range(start: VAddr, size: usize) {
59    #[cfg(target_arch = "riscv64")]
60    {
61        // Keep the PTE write visible to both the compiler and remote table walkers
62        // before the firmware asks those harts to execute SFENCE.VMA.
63        core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::Release);
64        // SAFETY: S-mode may order prior page-table stores with `fence`; this
65        // changes no memory and is required before remote invalidation.
66        unsafe {
67            core::arch::asm!("fence rw, rw", options(nostack));
68        }
69
70        hal::paging::flush_tlb_page(start);
71        let Some((remote_mask, remote_base)) = crate::task::smp::remote_online_sbi_target() else {
72            return;
73        };
74        #[cfg(feature = "test-hooks")]
75        if TEST_SKIP_REMOTE_RFENCE.load(Ordering::Acquire) {
76            return;
77        }
78        if let Err(error) = hal::sbi::sbi_remote_sfence_vma(remote_mask, remote_base, start, size) {
79            // Continuing would allow a remote hart to retain write access to a
80            // page that the current hart has already restricted or retired.
81            panic!("[tlb] RV64 RFENCE failed after PTE update: {}", error);
82        }
83    }
84
85    #[cfg(not(any(target_arch = "riscv64", target_arch = "riscv32")))]
86    {
87        let _ = size;
88        hal::paging::flush_tlb_page(start);
89    }
90
91    #[cfg(target_arch = "riscv32")]
92    {
93        // RV32 boards use the bare-physical SATP=0 contract and have no TLB.
94        let _ = (start, size);
95    }
96}
