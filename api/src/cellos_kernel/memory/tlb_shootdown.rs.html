1//! Private completion boundary for permission-lowering and unmap TLB maintenance.
2//!
3//! A frame is released only after its private tag is invalidated locally and
4//! every online hart (RISC-V via RFENCE/IPI, AArch64 via the GIC SGI) has
5//! acknowledged its own all-ASID flush. Targets with a single CPU have no
6//! remote to confirm and rely on the local flush alone.
7//!
8//! Confirming that acknowledgement is a **probe**, never a wait:
9//! [`confirm_tag_invalidation`] spends one bounded deadline, and a caller that
10//! does not get an acknowledgement hands its frames to
11//! [`crate::memory::deferred_release`], whose reaper confirms the tag later from
12//! the timer path. A release path that keeps waiting instead either eats the
13//! boot's time budget or — if it gave up sooner — frees a frame a remote hart can
14//! still resolve.
15

22/// Test-only negative control: pretend the peer's acknowledgement never arrives.
23#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
24static TEST_SKIP_REMOTE_RFENCE: AtomicBool = AtomicBool::new(false);

26/// Test-only silent-peer window: while armed, every tag invalidation is reported
27/// unconfirmed even though the request was still delivered and, on a two-hart
28/// boot, genuinely flushed.
29///
30/// The deferred-release fixture needs the *unconfirmed* state reachable without
31/// waiting for a real remote hart to stall mid-boot, and on a one-hart boot no
32/// remote exists to stall at all. Arming it exercises exactly the release-path
33/// branch that must retain frames instead of freeing or quarantining them;
34/// disarming it is what "the acknowledgement resumes" means for the reaper.
35#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
36static TEST_WITHHOLD_TAG_ACK: AtomicBool = AtomicBool::new(false);

39#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]

41    TEST_WITHHOLD_TAG_ACK.store(enabled, Ordering::Release);

86/// Why an invalidation could not be confirmed complete.
87#[derive(Clone, Copy, Debug, Eq, PartialEq)]
88pub enum FlushAckError {
89    /// A hart did not publish its completion before the deadline.
90    Timeout { hart: usize, epoch: usize },
91}
92
93/// Invalidate one private root's tag locally and confirm every online hart with
94/// **one** bounded attempt.
95///
96/// Unlike a current-root page flush this covers a root that is inactive on the
97/// caller (notably a non-current x86 PCID). RV64 remote harts acknowledge their
98/// local all-ASID flush before any frame behind this tag may be released.
99///
100/// A caller that gets `Err` must **not** widen the wait: a peer hart can be held
101/// non-preemptible for seconds, and a release path that keeps waiting either
102/// burns the boot's time budget (the 25 x 200 ms budget this replaces truncated
103/// two-hart boots) or — if it gave up sooner — would free a frame a remote hart
104/// can still resolve. Hand the frames to
105/// [`crate::memory::deferred_release`] instead and let its reaper confirm the
106/// invalidation from the timer path.
107pub fn confirm_tag_invalidation(asid: usize) -> Result<(), FlushAckError> {
108    local_tag_flush(asid);
109    let result = match await_remote_invalidation(tag_probe_ticks()) {
110        Err(error) => Err(error),
111        Ok(()) => {
112            // Test-only: the acknowledgement is withheld for a bounded window, so
113            // the release paths' unconfirmed branch is reachable without a real
114            // hart stalling mid-boot (and on a one-hart boot, where no remote
115            // exists to stall).
116            #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
117            if test_tag_ack_withheld() {
118                Err(FlushAckError::Timeout {
119                    hart: crate::task::hart_local::current_hart_id(),
120                    epoch: 0,
121                })
122            } else {
123                Ok(())
124            }
125            #[cfg(not(all(feature = "test-hooks", target_arch = "riscv64")))]
126            Ok(())
127        }
128    };
129    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
130    if let Err(error) = &result {
131        log::warn!(
132            "[tlb] tag {} invalidation unconfirmed: hart {} silent through the {} tick probe",
133            asid,
134            match error {
135                FlushAckError::Timeout { hart, .. } => hart,
136            },
137            tag_probe_ticks()
138        );
139    }
140    result
141}

143/// One non-blocking step of a deferred tag release: flush the tag locally and
144/// ask every online remote hart again.
145///
146/// It is the reaper's companion to [`confirm_tag_invalidation`]. A peer that took
147/// the IPI but never published its completion (or whose request was lost) only
148/// finishes when it is asked again, so entries alternate between re-issuing and
149/// checking. It never waits on a remote hart and costs one IPI per remote hart,
150/// which is what lets the timer ISR call it. The local tag flush is repeated on
151/// purpose: it is the cheapest half of the boundary, and repeating it keeps an
152/// entry safe even while the remote answer is missing.
153pub fn reissue_tag_invalidation(asid: usize) {
154    local_tag_flush(asid);
155    issue_remote_tag_flushes();
156}
157
158/// The local half of a tag invalidation, on its own.
159///
160/// The deferred reaper issues the remote half itself: it records the epoch each
161/// remote hart was asked for *per tag*, so completion is decided by that tag's
162/// own acknowledgement rather than by whether any hart owes any invalidation at
163/// all (which stays true for as long as unrelated teardowns keep asking).
164pub fn flush_tag_local(asid: usize) {
165    local_tag_flush(asid);
166}

168/// The first online remote hart that still owes an invalidation it was asked
169/// for, if any — a pure read of the acknowledgement machinery, no wait and no
170/// flush. `None` means every remote hart has published a completion at least as
171/// new as the last request, so a tag flushed before that request is safe.
172pub fn tag_invalidation_outstanding() -> Option<usize> {
173    // Test-only: inside the withheld window the tag reads unconfirmed even on a
174    // one-hart boot, where nothing remote can be outstanding.
175    #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
176    if test_tag_ack_withheld() {
177        return Some(crate::task::hart_local::current_hart_id());
178    }
179    remote_tag_flushes_outstanding()
180}

182/// Flush one private root's tag locally and publish the PTE stores before any
183/// remote hart is asked to do the same.
184fn local_tag_flush(asid: usize) {
185    // Publish PTE stores before requesting an invalidation on another hart.
186    #[cfg(target_arch = "riscv64")]
187    {
188        core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::Release);
189        // SAFETY: an S-mode fence orders prior page-table stores.
190        unsafe { core::arch::asm!("fence rw, rw", options(nostack)) };
191    }
192    #[cfg(target_arch = "aarch64")]
193    {
194        core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::Release);
195        // SAFETY: `dsb ishst` orders prior page-table stores before the IPI that
196        // asks a remote hart to invalidate them.
197        unsafe { core::arch::asm!("dsb ishst", options(nostack)) };
198    }
199    #[cfg(target_arch = "x86_64")]
200    core::sync::atomic::fence(core::sync::atomic::Ordering::SeqCst);
201    #[cfg(feature = "test-hooks")]
202    if TEST_FLUSH_TRACKING.load(Ordering::Acquire) {
203        TEST_FLUSHED_TAGS.lock().push(asid);
204    }
205    // Every leaf that differs between roots is non-global (`PageFlags::NON_GLOBAL`
206    // -> `PTE_nG`), so the retiring root's own tag is the whole invalidation
207    // scope. The leaves this deliberately leaves alone — the shared kernel
208    // ranges and the RAM identity entries the kernel root owns — are identical
209    // in every root, so a surviving entry of theirs resolves the same way.
210    hal::domain::flush_asid(asid);
211}

213/// Ask every online remote hart to invalidate locally and wait for that answer
214/// for at most `ticks`.
215///
216/// One request per hart with a fresh epoch: a delivered IPI can still be late
217/// (the target may be in a long non-interruptible stretch), so the epoch — not
218/// the firmware's return — is what proves the remote hart stopped using the
219/// translation.
220#[cfg(any(target_arch = "riscv64", target_arch = "aarch64", target_arch = "x86_64"))]
221fn await_remote_invalidation(ticks: u64) -> Result<(), FlushAckError> {
222    use crate::task::smp;
223    let me = crate::task::hart_local::current_hart_id();
224    for hart in smp::online_harts().filter(|hart| *hart != me) {
225        let epoch = smp::request_tlb_flush(hart);
226        let deadline = counter_now() + ticks;
227        while !smp::tlb_flush_completed(hart, epoch) {
228            if counter_now() > deadline {
229                return Err(FlushAckError::Timeout { hart, epoch });
230            }
231            // Spin, and only spin: this runs in whatever context the release
232            // happens to be in, where widening the interrupt window can admit a
233            // timer ISR into the middle of a teardown (observed as a hang in
234            // the admission fixture). A remote hart makes progress anyway —
235            // the emulator interleaves vCPUs — so the wait needs a budget, not
236            // a yield.
237            core::hint::spin_loop();
238            #[cfg(target_arch = "x86_64")]
239            smp::service_tlb_flush(me);
240        }
241    }
242    Ok(())
243}

245/// Targets without a second hart: the local flush above is the whole contract
246/// and there is no remote to confirm.
247#[cfg(not(any(target_arch = "riscv64", target_arch = "aarch64", target_arch = "x86_64")))]
248fn await_remote_invalidation(_ticks: u64) -> Result<(), FlushAckError> {
249    Ok(())
250}

252/// Free-running counter used for bounded probes.
253///
254/// It advances whether or not the caller's context takes interrupts, which is
255/// what a probe that must not wait needs.
256#[cfg(any(target_arch = "riscv64", target_arch = "aarch64", target_arch = "x86_64"))]
257#[inline]
258fn counter_now() -> u64 {
259    #[cfg(target_arch = "riscv64")]
260    {
261        hal::common::timer::read_mtime()
262    }
263    #[cfg(target_arch = "aarch64")]
264    {
265        hal::aarch64::timer::counter_now()
266    }
267    #[cfg(target_arch = "x86_64")]
268    { unsafe { core::arch::x86_64::_rdtsc() } }
269}

271#[cfg(any(target_arch = "riscv64", target_arch = "aarch64", target_arch = "x86_64"))]
272fn issue_remote_tag_flushes() {
273    let me = crate::task::hart_local::current_hart_id();
274    for hart in crate::task::smp::online_harts().filter(|hart| *hart != me) {
275        let _ = crate::task::smp::request_tlb_flush(hart);
276    }
277}

283/// The first online remote hart that still owes an invalidation, if any.
284#[cfg(any(target_arch = "riscv64", target_arch = "aarch64", target_arch = "x86_64"))]
285fn remote_tag_flushes_outstanding() -> Option<usize> {
286    let me = crate::task::hart_local::current_hart_id();
287    crate::task::smp::online_harts()

297/// How long [`confirm_tag_invalidation`] waits for a remote acknowledgement
298/// before the caller defers its frames. Generous on purpose: a slow ack costs
299/// one probe latency, a missing one costs isolation, and the probe is now the
300/// *only* wait — the reaper's retries happen without blocking anyone.
301#[cfg(target_arch = "riscv64")]
302const fn tag_probe_ticks() -> u64 {
303    20 * hal::common::timer::TICKS_PER_10MS
304}
305
306/// 200 ms in this platform's counter units — the budget for one probe.
307#[cfg(target_arch = "aarch64")]
308fn tag_probe_ticks() -> u64 {
309    hal::aarch64::timer::counter_frequency_hz() / 5
310}
311
312#[cfg(target_arch = "x86_64")]
313const fn tag_probe_ticks() -> u64 { 2_000_000_000 }
314
315#[cfg(not(any(target_arch = "riscv64", target_arch = "aarch64", target_arch = "x86_64")))]
316const fn tag_probe_ticks() -> u64 {
317    0

320/// Invalidate the translation for one changed page before its memory can run or reuse.
321#[inline]
322pub fn flush_page(vaddr: VAddr) {
323    let page = vaddr & !(PAGE_SIZE - 1);
324    #[cfg(feature = "test-hooks")]
325    if TEST_FLUSH_TRACKING.load(Ordering::Acquire) {
326        TEST_FLUSHED_PAGES.lock().push(page);
327    }
328    flush_range(page, PAGE_SIZE);
329}

331/// Invalidate a page-aligned range after its PTEs have changed.
332pub fn flush_range(start: VAddr, size: usize) {
333    debug_assert!(start & (PAGE_SIZE - 1) == 0);
334    debug_assert!(size & (PAGE_SIZE - 1) == 0);
335    let end = start
336        .checked_add(size)
337        .expect("TLB flush range must not wrap the address space");
338
339    #[cfg(target_arch = "x86_64")]
340    core::sync::atomic::fence(core::sync::atomic::Ordering::SeqCst);
341    #[cfg(target_arch = "riscv64")]
342    {
343        // Keep the PTE write visible to both the compiler and remote table walkers
344        // before the firmware asks those harts to execute SFENCE.VMA.
345        core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::Release);
346        // SAFETY: S-mode may order prior page-table stores with `fence`; this
347        // changes no memory and is required before remote invalidation.
348        unsafe {
349            core::arch::asm!("fence rw, rw", options(nostack));
350        }
351
352        for page in (start..end).step_by(PAGE_SIZE) {
353            hal::paging::flush_tlb_page(page);
354        }
355        let Some((remote_mask, remote_base)) = crate::task::smp::remote_online_sbi_target() else {
356            return;
357        };
358        #[cfg(feature = "test-hooks")]
359        if TEST_SKIP_REMOTE_RFENCE.load(Ordering::Acquire) {
360            return;
361        }
362        if let Err(error) = hal::sbi::sbi_remote_sfence_vma(remote_mask, remote_base, start, size) {
363            // Continuing would allow a remote hart to retain write access to a
364            // page that the current hart has already restricted or retired.
365            panic!("[tlb] RV64 RFENCE failed after PTE update: {}", error);
366        }
367    }
368
369    #[cfg(not(any(target_arch = "riscv64", target_arch = "riscv32", target_arch = "x86_64")))]
370    for page in (start..end).step_by(PAGE_SIZE) {
371        hal::paging::flush_tlb_page(page);
372    }
373    #[cfg(target_arch = "x86_64")]
374    {
375        let _ = end; // Overflow was checked even though invalidation covers every tag.
376        // Shared leaves may be cached under inactive PCIDs too. Invalidate all
377        // local tags/globals, then require every online peer's captured epoch
378        // before returning W^X success or permitting frame reuse.
379        crate::memory::paging::tlb_flush_all();
380        if let Err(error) = await_remote_invalidation(tag_probe_ticks()) {
381            panic!("[x86-smp] shared mapping invalidation unconfirmed: {:?}", error);
382        }
383    }
384
385    #[cfg(target_arch = "riscv32")]
386    {
387        let _ = end;
388    }
