1//! Private completion boundary for permission-lowering and unmap TLB maintenance.
2//!
3//! A frame is released only after its private tag is invalidated locally and
4//! every online hart (RISC-V via RFENCE/IPI, AArch64 via the GIC SGI) has
5//! acknowledged its own all-ASID flush. Targets with a single CPU have no
6//! remote to confirm and rely on the local flush alone.
7//!
8//! Confirming that acknowledgement is a **probe**, never a wait:
9//! [`confirm_tag_invalidation`] spends one bounded deadline, and a caller that
10//! does not get an acknowledgement hands its frames to
11//! [`crate::memory::deferred_release`], whose reaper confirms the tag later from
12//! the timer path. A release path that keeps waiting instead either eats the
13//! boot's time budget or — if it gave up sooner — frees a frame a remote hart can
14//! still resolve.
15

25/// Test-only silent-peer window: while armed, every tag invalidation is reported
26/// unconfirmed even though the request was still delivered and, on a two-hart
27/// boot, genuinely flushed.
28///
29/// The deferred-release fixture needs the *unconfirmed* state reachable without
30/// waiting for a real remote hart to stall mid-boot, and on a one-hart boot no
31/// remote exists to stall at all. Arming it exercises exactly the release-path
32/// branch that must retain frames instead of freeing or quarantining them;
33/// disarming it is what "the acknowledgement resumes" means for the reaper.
34/// Armed by the RV64 aspace fixture and the x86_64 grant fixtures.
35#[cfg(feature = "test-hooks")]
36static TEST_WITHHOLD_TAG_ACK: AtomicBool = AtomicBool::new(false);

84/// Record a shared flush's pages for the observation window.
85#[cfg(feature = "test-hooks")]
86fn observe_shared_flush(ranges: &[(VAddr, usize)], all: bool) {
87    if !TEST_FLUSH_TRACKING.load(Ordering::Acquire) {
88        return;
89    }
90    if all {
91        TEST_FLUSHED_ALL.store(true, Ordering::Release);
92        return;
93    }
94    let mut pages = TEST_FLUSHED_PAGES.lock();
95    #[cfg(feature = "p01-trace")]
96    let _semantic = crate::memory::heap::p01_owner::creation(
97        None,
98        crate::memory::heap::p01_owner::TEST_FIXTURE_SHARED,
99    );
100    for &(start, size) in ranges {
101        pages.extend((start..start.saturating_add(size)).step_by(PAGE_SIZE));
102    }
103}

118/// Why an invalidation could not be confirmed complete.
119#[derive(Clone, Copy, Debug, Eq, PartialEq)]
120pub enum FlushAckError {
121    /// A hart did not publish its completion before the deadline.
122    Timeout { hart: usize, epoch: usize },
123}
124
125/// Invalidate one private root's tag locally and confirm every online hart with
126/// **one** bounded attempt.
127///
128/// Unlike a current-root page flush this covers a root that is inactive on the
129/// caller (notably a non-current x86 PCID). RV64 remote harts acknowledge their
130/// local all-ASID flush before any frame behind this tag may be released.
131///
132/// A caller that gets `Err` must **not** widen the wait: a peer hart can be held
133/// non-preemptible for seconds, and a release path that keeps waiting either
134/// burns the boot's time budget (the 25 x 200 ms budget this replaces truncated
135/// two-hart boots) or — if it gave up sooner — would free a frame a remote hart
136/// can still resolve. Hand the frames to
137/// [`crate::memory::deferred_release`] instead and let its reaper confirm the
138/// invalidation from the timer path.
139///
140/// `Ok` carries the [`Invalidated`] proof for `asid`: on this path, the only
141/// way a private tag goes back to its pool.
142pub(crate) fn confirm_tag_invalidation(asid: usize) -> Result<Invalidated, FlushAckError> {
143    local_tag_flush(asid);
144    // `withheld`: the failure is the test seam's, not a silent hart's.
145    let mut asked = [0; crate::sched::smp::MAX_HARTS];
146    #[allow(unused_variables)] // reason: only the test-hooks log below reads it
147    let (result, withheld) = match await_remote_invalidation(tag_probe_ticks(), &mut asked) {
148        Err(error) => (Err(error), false),
149        Ok(()) => {
150            // Test-only: the acknowledgement is withheld for a bounded window, so
151            // the release paths' unconfirmed branch is reachable without a real
152            // hart stalling mid-boot (and on a one-hart boot, where no remote
153            // exists to stall).
154            #[cfg(feature = "test-hooks")]
155            if test_tag_ack_withheld() {
156                let forced = FlushAckError::Timeout {
157                    hart: crate::sched::hart_local::current_hart_id(),
158                    epoch: 0,
159                };
160                (Err(forced), true)
161            } else {
162                (Ok(()), false)
163            }
164            #[cfg(not(feature = "test-hooks"))]
165            (Ok(()), false)
166        }
167    };
168    // The stress gate (phase 07b) counts these lines on x86 as well. A forced
169    // failure names itself (`withheld (test)`), so a lane that excuses the
170    // withheld window excuses only those, and a "silent hart" line is always a
171    // real one.
172    #[cfg(all(
173        feature = "test-hooks",
174        any(target_arch = "riscv64", target_arch = "x86_64")
175    ))]
176    if let Err(error) = &result {
177        if withheld {
178            log::warn!("[tlb] tag {asid} invalidation unconfirmed: withheld (test)");
179        } else {
180            log::warn!(
181                "[tlb] tag {} invalidation unconfirmed: hart {} silent through the {} tick probe",
182                asid,
183                match error {
184                    FlushAckError::Timeout { hart, .. } => hart,
185                },
186                tag_probe_ticks()
187            );
188        }
189    }
190    result.map(|()| Invalidated::of(asid as u32))
191}

193/// One non-blocking step of a deferred tag release: flush the tag locally and
194/// ask every online remote hart again.
195///
196/// It is the reaper's companion to [`confirm_tag_invalidation`]. A peer that took
197/// the IPI but never published its completion (or whose request was lost) only
198/// finishes when it is asked again, so entries alternate between re-issuing and
199/// checking. It never waits on a remote hart and costs one IPI per remote hart,
200/// which is what lets the timer ISR call it. The local tag flush is repeated on
201/// purpose: it is the cheapest half of the boundary, and repeating it keeps an
202/// entry safe even while the remote answer is missing.
203pub fn reissue_tag_invalidation(asid: usize) {
204    local_tag_flush(asid);
205    issue_remote_tag_flushes();
206}
207
208/// The local half of a tag invalidation, on its own.
209///
210/// The deferred reaper issues the remote half itself: it records the epoch each
211/// remote hart was asked for *per tag*, so completion is decided by that tag's
212/// own acknowledgement rather than by whether any hart owes any invalidation at
213/// all (which stays true for as long as unrelated teardowns keep asking).
214pub fn flush_tag_local(asid: usize) {
215    local_tag_flush(asid);
216}

218/// The first online remote hart that still owes an invalidation it was asked
219/// for, if any — a pure read of the acknowledgement machinery, no wait and no
220/// flush. `None` means every remote hart has published a completion at least as
221/// new as the last request, so a tag flushed before that request is safe.
222pub fn tag_invalidation_outstanding() -> Option<usize> {
223    // Test-only: inside the withheld window the tag reads unconfirmed even on a
224    // one-hart boot, where nothing remote can be outstanding.
225    #[cfg(feature = "test-hooks")]
226    if test_tag_ack_withheld() {
227        return Some(crate::sched::hart_local::current_hart_id());
228    }
229    remote_tag_flushes_outstanding()
230}

232/// Flush one private root's tag locally and publish the PTE stores before any
233/// remote hart is asked to do the same.
234fn local_tag_flush(asid: usize) {
235    // Publish PTE stores before requesting an invalidation on another hart.
236    crate::arch::tlb::publish_pte_stores();
237    #[cfg(feature = "test-hooks")]
238    if TEST_FLUSH_TRACKING.load(Ordering::Acquire) {
239        let mut tags = TEST_FLUSHED_TAGS.lock();
240        #[cfg(feature = "p01-trace")]
241        let _semantic = crate::memory::heap::p01_owner::creation(
242            None,
243            crate::memory::heap::p01_owner::TEST_FIXTURE_SHARED,
244        );
245        tags.push(asid);
246    }
247    // Every leaf that differs between roots is non-global (`PageFlags::NON_GLOBAL`
248    // -> `PTE_nG`), so the retiring root's own tag is the whole invalidation
249    // scope. The leaves this deliberately leaves alone — the shared kernel
250    // ranges and the RAM identity entries the kernel root owns — are identical
251    // in every root, so a surviving entry of theirs resolves the same way.
252    hal::domain::flush_asid(asid);
253}

255/// Ask every online remote hart to invalidate locally and wait for that answer
256/// for at most `ticks`.
257///
258/// One request per hart with a fresh epoch: a delivered IPI can still be late
259/// (the target may be in a long non-interruptible stretch), so the epoch — not
260/// the firmware's return — is what proves the remote hart stopped using the
261/// translation.
262///
263/// Broadcast, then wait over the set (phase 07b): every remote request is
264/// published and its IPI sent before the first wait, and one deadline covers
265/// the whole set. The targets are the online harts at the broadcast; the error
266/// names the lowest-numbered hart still silent at the deadline, and `asked`
267/// receives the epoch every hart was asked for. Callers keep their fail-closed
268/// on that error (shared flushes quarantine their frames, tag releases defer
269/// theirs) — nothing here continues past an unconfirmed hart.
270#[cfg(cellos_wide)]
271fn await_remote_invalidation(ticks: u64, asked: &mut Epochs) -> Result<(), FlushAckError> {
272    use crate::sched::smp;
273    let me = crate::sched::hart_local::current_hart_id();
274    let epochs = asked;
275    let requested =
276        smp::RemoteEpochs::broadcast(smp::online_harts().filter(|hart| *hart != me), |hart| {
277            let epoch = smp::request_tlb_flush(hart);
278            epochs[hart] = epoch;
279            epoch
280        });
281    let deadline = counter_now() + ticks;
282    while let Some((hart, epoch)) = smp::tlb_flush_set_outstanding(&requested) {
283        if counter_now() > deadline {
284            return Err(FlushAckError::Timeout { hart, epoch });
285        }
286        // Spin, and only spin: this runs in whatever context the release
287        // happens to be in, where widening the interrupt window can admit a
288        // timer ISR into the middle of a teardown (observed as a hang in
289        // the admission fixture). A remote hart makes progress anyway —
290        // the emulator interleaves vCPUs — so the wait needs a budget, not
291        // a yield.
292        core::hint::spin_loop();
293        // Two or more harts can shoot each other down at once, each with
294        // interrupts possibly masked: answering our own requests here is what
295        // keeps a cycle of waiters from deadlocking. Every arch needs it now
296        // that hart 0 is a target too — hart 0 and hart 1 can each wait on the
297        // other. The hart is re-read on every pass instead of reusing `me`:
298        // callers run with interrupts either masked or not (a debug_assert
299        // would be wrong for the latter), and with them unmasked a timer ISR
300        // may switch this context away and resume it on another CPU.
301        // Servicing a stale id would publish a completion for a hart whose TLB
302        // this CPU never flushed — a forged acknowledgement — and would leave
303        // the CPU we now run on (possibly in `requested`) unanswered. The read
304        // and the service run with interrupts masked, so no switch can fall
305        // between them; `service_tlb_flush(current)` then flushes exactly the
306        // CPU it acknowledges for. Masking only narrows the interrupt window,
307        // it never widens it.
308        let flags = hal::arch::save_and_disable_interrupts();
309        smp::answer_tlb_flush_in_spin(crate::sched::hart_local::current_hart_id());
310        // SAFETY: restores exactly the interrupt state saved above.
311        unsafe { hal::arch::restore_sstatus(flags) };
312    }
313    Ok(())
314}

323/// Ask every online remote hart for a full local flush and return the epochs,
324/// without waiting for any of them (0 = not asked).
325#[cfg(cellos_wide)]
326pub fn request_remote_flushes() -> Epochs {
327    let me = crate::sched::hart_local::current_hart_id();
328    let mut epochs = [0; crate::sched::smp::MAX_HARTS];
329    for hart in crate::sched::smp::online_harts().filter(|hart| *hart != me) {
330        epochs[hart] = crate::sched::smp::request_tlb_flush(hart);
331    }
332    epochs
333}

341/// Free-running counter used for bounded probes.
342///
343/// It advances whether or not the caller's context takes interrupts, which is
344/// what a probe that must not wait needs.
345#[cfg(cellos_wide)]
346#[inline]
347fn counter_now() -> u64 {
348    #[cfg(target_arch = "riscv64")]
349    {
350        hal::common::timer::read_mtime()
351    }
352    #[cfg(target_arch = "aarch64")]
353    {
354        hal::aarch64::timer::counter_now()
355    }
356    #[cfg(target_arch = "x86_64")]
357    {
358        unsafe { core::arch::x86_64::_rdtsc() }
359    }
360}

388/// How long [`confirm_tag_invalidation`] waits for a remote acknowledgement
389/// before the caller defers its frames. Generous on purpose: a slow ack costs
390/// one probe latency, a missing one costs isolation, and the probe is now the
391/// *only* wait — the reaper's retries happen without blocking anyone.
392#[cfg(target_arch = "riscv64")]
393const fn tag_probe_ticks() -> u64 {
394    20 * hal::common::timer::TICKS_PER_10MS
395}

397/// 200 ms in this platform's counter units — the budget for one probe.
398#[cfg(target_arch = "aarch64")]
399fn tag_probe_ticks() -> u64 {
400    hal::aarch64::timer::counter_frequency_hz() / 5
401}

416/// Shared-root invalidations wider than this many pages flush everything
417/// locally instead of page by page (RV64 `sfence.vma`, AArch64 `tlbi vmalle1is`).
418const PAGE_FLUSH_LIMIT: usize = 64;
419
420/// Invalidate the translation for one changed page of the shared root on every
421/// online hart. See [`flush_ranges`].
422#[inline]
423pub fn flush_page(cap: &mut FlushCap, vaddr: VAddr) -> FlushOutcome {
424    flush_range(cap, vaddr & !(PAGE_SIZE - 1), PAGE_SIZE)
425}
426
427/// Invalidate a page-aligned range of the shared root. See [`flush_ranges`].
428pub fn flush_range(cap: &mut FlushCap, start: VAddr, size: usize) -> FlushOutcome {
429    flush_ranges(cap, &[(start, size)], false)
430}

432/// Invalidate page-aligned ranges of the shared (kernel) root on every online
433/// hart, or every translation when `all`, as **one** operation: one remote
434/// request per hart, one bounded probe.
435///
436/// Never panics and never waits past the probe bound. `Unconfirmed` carries the
437/// epochs each remote hart was asked for: the caller keeps every frame the
438/// changed translations named (`frame_run::Batch` quarantines them) and must
439/// not report a permission lowering as done. Callers hold no `FRAME_ALLOCATOR`
440/// (the `cap` proves it) and log the outcome themselves, because this may run
441/// under `KERNEL_ROOT`, which ranks above the console.
442///
443/// - x86_64: every PCID and global entry locally, then every online CPU's
444///   acknowledged full flush.
445/// - riscv64: `sfence.vma` per page (or all), then SBI RFENCE per range. An
446///   RFENCE error, or `all`, takes the IPI epoch path instead: every remote hart
447///   serves the request with a full local flush.
448/// - aarch64: broadcast `tlbi` per page (or `vmalle1is`); no acknowledgement
449///   exists to wait for.
450pub fn flush_ranges(cap: &mut FlushCap, ranges: &[(VAddr, usize)], all: bool) -> FlushOutcome {
451    let _ = cap;
452    #[cfg(feature = "test-hooks")]
453    assert!(
454        !crate::memory::frame::held_here(),
455        "[tlb] shared flush while this hart holds the frame allocator"
456    );
457    let all = all || wider_than_page_flush(ranges);
458    shared_flush(ranges, all, true)
459}
460
461/// The local half of [`flush_ranges`] plus the remote *requests*, without
462/// waiting for any answer: the result is `Unconfirmed` whenever a remote hart
463/// was asked. For a path that cannot wait (it holds the frame allocator); its
464/// frames are quarantined until the reaper sees the epochs published.
465pub fn flush_ranges_without_wait(ranges: &[(VAddr, usize)], all: bool) -> FlushOutcome {
466    let all = all || wider_than_page_flush(ranges);
467    shared_flush(ranges, all, false)
468}
469
470/// Does the batch cover more pages than a per-page flush should, or wrap?
471fn wider_than_page_flush(ranges: &[(VAddr, usize)]) -> bool {
472    let mut pages = 0usize;
473    for &(start, size) in ranges {
474        debug_assert!(start & (PAGE_SIZE - 1) == 0);
475        debug_assert!(size & (PAGE_SIZE - 1) == 0);
476        if start.checked_add(size).is_none() {
477            // A wrapping range is a caller bug; invalidate everything rather
478            // than trust its bounds.
479            return true;
480        }
481        pages = pages.saturating_add(size / PAGE_SIZE);
482    }
483    pages > PAGE_FLUSH_LIMIT
484}

486/// What remote harts still owe after the arch half of a shared flush
487/// (`crate::arch::tlb::shared_flush_local`) has run.
488pub(crate) enum RemoteWork {
489    /// Nothing: the arch's own instructions reached every hart (AArch64
490    /// broadcast `tlbi`, a completed RV64 RFENCE) or there is no other hart.
491    // reason: x86 always takes the epoch path.
492    #[allow(dead_code)]
493    None,
494    /// Every online remote hart must serve the IPI epoch with a full flush.
495    /// `rfence_failed`: RV64 tried RFENCE first and the firmware refused.
496    // reason: AArch64's broadcast `tlbi` never needs it.
497    #[allow(dead_code)]
498    EpochPath { rfence_failed: bool },
499}
500
501fn shared_flush(ranges: &[(VAddr, usize)], all: bool, wait: bool) -> FlushOutcome {
502    #[cfg(feature = "test-hooks")]
503    observe_shared_flush(ranges, all);
504    match crate::arch::tlb::shared_flush_local(ranges, all) {
505        RemoteWork::None => FlushOutcome::Confirmed,
506        RemoteWork::EpochPath { rfence_failed } => remote_full_flush(wait, rfence_failed),
507    }
508}

510/// Every online remote hart's full flush: one bounded probe when `wait`,
511/// otherwise only the requests (`Unconfirmed { NotAwaited }` whenever one was
512/// sent: nobody waited, so no hart is known to be silent).
513fn remote_full_flush(wait: bool, rfence_failed: bool) -> FlushOutcome {
514    if wait {
515        let mut epochs = [0; crate::sched::smp::MAX_HARTS];
516        return match await_remote_invalidation(tag_probe_ticks(), &mut epochs) {
517            Ok(()) => FlushOutcome::Confirmed,
518            Err(FlushAckError::Timeout { hart, .. }) => FlushOutcome::Unconfirmed {
519                epochs,
520                cause: if rfence_failed {
521                    UnconfirmedCause::RfenceFailed { hart }
522                } else {
523                    UnconfirmedCause::Timeout { hart }
524                },
525            },
526        };
527    }
528    let epochs = request_remote_flushes();
529    match epochs.iter().position(|epoch| *epoch != 0) {
530        None => FlushOutcome::Confirmed,
531        Some(hart) => FlushOutcome::Unconfirmed {
532            epochs,
533            cause: UnconfirmedCause::NotAwaited { hart: Some(hart) },
534        },
535    }
