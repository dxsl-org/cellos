1//! Per-cell quota accounts.
2//!
3//! - `QUOTA_LIMITS`: one atomic limit word per CellId slot, `UNREGISTERED` when
4//!   the slot holds no incarnation; `GUEST_LIMITS` beside it.
5//! - `IN_USE` (memory), `DMA_IN_USE` and `GUEST_IN_USE`: one atomic counter per
6//!   slot, packing the registered incarnation's generation above its live byte
7//!   count.
8//!
9//! A charge names its owner explicitly ([`Charge`]): one Cell incarnation, or
10//! the kernel. Nothing here locks or allocates, so a charge or refund is safe
11//! inside `GlobalAlloc::alloc` and in interrupt context, and every refund is
12//! checked against the generation it was charged to.
13
14use core::sync::atomic::{AtomicUsize, Ordering};
15use types::CellId;
16
17/// Maximum CellId tracked (index into IN_USE array).
18///
19/// 64 is the production ceiling: it sizes the per-cell counter array and the
20/// CellId space. The D5 per-request-server gate asks how many cells actually fit
21/// now that the boot memory map is read from the DTB, and at 64 this constant
22/// binds long before memory does — so `cell-scale-experiment` raises it for the
23/// measurement lane (scripts/qemu-cell-scale.sh) only.
24#[cfg(not(feature = "cell-scale-experiment"))]
25pub const MAX_CELLS: usize = 64;
26#[cfg(feature = "cell-scale-experiment")]
27pub const MAX_CELLS: usize = 4096;
28
29/// Default heap quota per Cell: 16 MiB.
30///
31/// Raised from 4 MiB to support heavier cells (DOOM needs ~10 MiB for WAD +
32/// zone heap; compositor and VFS also benefit from the headroom).  The quota
33/// is a runtime charge limit — no physical pages are pre-allocated.
34pub const DEFAULT_QUOTA_BYTES: usize = 16 * 1024 * 1024;
35
36// The constant above remains the hard default/ceiling used by limit tests.
37// Cold boot may lower the default for subsequently spawned cells only.
38static RUNTIME_DEFAULT_QUOTA_BYTES: AtomicUsize = AtomicUsize::new(DEFAULT_QUOTA_BYTES);
39
40pub fn default_quota_bytes() -> usize {
41    RUNTIME_DEFAULT_QUOTA_BYTES.load(Ordering::Acquire)
42}
43
44pub(crate) fn set_default_quota_bytes(bytes: usize) {
45    assert!((1024 * 1024..=DEFAULT_QUOTA_BYTES).contains(&bytes));
46    RUNTIME_DEFAULT_QUOTA_BYTES.store(bytes, Ordering::Release);
47}

49/// A quota owner: one Cell incarnation, or the kernel (cell 0).
50#[derive(Clone, Copy, Debug, Eq, PartialEq)]
51pub(crate) struct Charge {
52    cell: CellId,
53    generation: u64,
54}
55
56impl Charge {
57    /// The kernel: never limited and never counted.
58    pub(crate) const KERNEL: Charge = Charge::new(CellId(0), 0);
59
60    /// Incarnation `generation` of `cell`.
61    pub(crate) const fn new(cell: CellId, generation: u64) -> Self {
62        Self { cell, generation }
63    }
64
65    /// The incarnation `task` runs as.
66    pub(crate) fn of(task: &crate::task::Task) -> Self {
67        Self::new(task.cell_id, task.cell_generation)
68    }
69
70    /// The incarnation dispatched on this hart. Only syscall entry turns it
71    /// into an allocation charge (`memory::heap::ChargeScope`).
72    pub(crate) fn dispatched() -> Self {
73        Self::new(
74            CellId(crate::sched::hart_local::current_cell_id() as u64),
75            crate::sched::hart_local::current_cell_generation(),
76        )
77    }
78
79    pub(crate) fn cell(self) -> CellId {
80        self.cell
81    }
82
83    pub(crate) fn generation(self) -> u64 {
84        self.generation
85    }
86
87    /// The counter slot this charge names; `None` for the kernel and for a
88    /// CellId beyond the table, neither of which is counted.
89    fn slot(self) -> Option<usize> {
90        if self.cell == Self::KERNEL.cell {
91            return None;
92        }
93        usize::try_from(self.cell.0)
94            .ok()
95            .filter(|&id| id < MAX_CELLS)
96    }
97
98    /// The generation bits a counter word carries for this incarnation.
99    fn tag(self) -> usize {
100        (self.generation as usize & GENERATION_MASK) << GENERATION_SHIFT
101    }
102}

104/// Which counter a charge draws from.
105#[derive(Clone, Copy, Debug, Eq, PartialEq)]
106pub(crate) enum Account {
107    /// Kernel heap, and the frames billed beside it (thread stacks, SHM,
108    /// domain page tables).
109    Memory,
110    /// Bytes published for device DMA; bounded by the memory limit.
111    Dma,
112    /// A VM's guest RAM, nested or Stage-2 tables and control structures;
113    /// bounded by the guest limit.
114    Guest,
115}
116
117impl Account {
118    fn counters(self) -> &'static [AtomicUsize; MAX_CELLS] {
119        match self {
120            Self::Memory => &IN_USE,
121            Self::Dma => &DMA_IN_USE,
122            Self::Guest => &GUEST_IN_USE,
123        }
124    }
125}

127/// The limits an incarnation is registered with.
128#[derive(Clone, Copy, Debug, Eq, PartialEq)]
129pub(crate) struct Limits {
130    /// Bound on the memory account, and on the DMA account.
131    pub(crate) memory: usize,
132    /// Bound on the guest account; 0 refuses every VM.
133    pub(crate) guest: usize,
134}

136/// A charge was refused; nothing was taken.
137#[derive(Debug, Eq, PartialEq)]
138pub(crate) struct QuotaExceeded;

140/// `QUOTA_LIMITS` value of a slot with no registered incarnation.
141const UNREGISTERED: usize = usize::MAX;
142
143/// Limit per CellId slot. A registered limit is clamped to `BYTES_MASK`, so it
144/// never equals `UNREGISTERED`. Read lock-free by every charge.
145static QUOTA_LIMITS: [AtomicUsize; MAX_CELLS] =
146    [const { AtomicUsize::new(UNREGISTERED) }; MAX_CELLS];
147
148/// Live memory bytes per CellId slot.
149///
150/// Each word packs the registered incarnation's generation (bits
151/// `GENERATION_SHIFT..`) above its live byte count (`BYTES_MASK`). Every charge
152/// and refund compares the generation in the same CAS that adds or subtracts,
153/// so bytes charged to an earlier incarnation of a reused CellId can never move
154/// the current incarnation's counter.
155///
156/// Uses an inline `const { }` repeat-array seed (not a named `const`) — each
157/// slot is independently zero-initialized at compile time, so there is no
158/// shared-instance footgun (clippy::declare_interior_mutable_const) to worry
159/// about; a named `static` cannot be used here because `AtomicUsize` is not
160/// `Copy`, which the `[expr; N]` repeat form otherwise requires.
161static IN_USE: [AtomicUsize; MAX_CELLS] = [const { AtomicUsize::new(0) }; MAX_CELLS];
162
163/// Live DMA-published bytes per CellId slot, packed as `IN_USE`.
164/// DMA quota = 1× memory quota (validated 2026-06-22).
165static DMA_IN_USE: [AtomicUsize; MAX_CELLS] = [const { AtomicUsize::new(0) }; MAX_CELLS];
166
167/// Guest limit per CellId slot, read only while `QUOTA_LIMITS` holds the slot;
168/// 0 for a slot that holds no incarnation.
169static GUEST_LIMITS: [AtomicUsize; MAX_CELLS] = [const { AtomicUsize::new(0) }; MAX_CELLS];
170
171/// Live guest bytes per CellId slot, packed as `IN_USE`.
172static GUEST_IN_USE: [AtomicUsize; MAX_CELLS] = [const { AtomicUsize::new(0) }; MAX_CELLS];
173
174/// Low bits of a counter word: live bytes, at most 4 GiB - 1 per account.
175/// Every account's bound fits: the memory and DMA limits default to
176/// `DEFAULT_QUOTA_BYTES` (16 MiB), the largest guest budget is 1544 MiB
177/// (`catalog`), and a larger registered limit is clamped to `BYTES_MASK`.
178const GENERATION_SHIFT: u32 = 32;
179const BYTES_MASK: usize = (1 << GENERATION_SHIFT) - 1;
180/// The task generation bits a counter word keeps (the low 32). A generation is
181/// minted per incarnation (`tcb::next_cell_generation`; a thread takes its
182/// Cell's), so a stale charge or refund can alias a later incarnation of its
183/// CellId only after 2^32 incarnations.
184const GENERATION_MASK: usize = (1 << (usize::BITS - GENERATION_SHIFT)) - 1;
185const _: () = assert!(
186    usize::BITS == 64,
187    "counter words pack generation and bytes into 64 bits"
188);
189const _: () = assert!(
190    DEFAULT_QUOTA_BYTES <= BYTES_MASK && MAX_CELLS <= BYTES_MASK,
191    "a counter's byte field holds every default limit, and an owner token's \
192     low bits every CellId"
193);
194
195/// Owner token for allocations that are not charged to any Cell row.
196pub(crate) const UNCHARGED_OWNER: usize = 0;
197
198/// `word` with `size` live bytes removed and its generation kept.
199fn without_bytes(word: usize, size: usize) -> usize {
200    (word & !BYTES_MASK) | (word & BYTES_MASK).saturating_sub(size)
201}

203#[cfg(feature = "p01-trace")]
204struct BoundQuota {
205    ready: core::sync::atomic::AtomicBool,
206    cell: core::sync::atomic::AtomicU64,
207    generation: core::sync::atomic::AtomicU64,
208    retired: core::sync::atomic::AtomicBool,
209}
210#[cfg(feature = "p01-trace")]
211static P01_SOURCE: crate::p01_resources::Source = crate::p01_resources::Source::new();
212#[cfg(feature = "p01-trace")]
213static P01_SUBJECTS: [BoundQuota; 9] = [const {
214    BoundQuota {
215        ready: core::sync::atomic::AtomicBool::new(false),
216        cell: core::sync::atomic::AtomicU64::new(0),
217        generation: core::sync::atomic::AtomicU64::new(0),
218        retired: core::sync::atomic::AtomicBool::new(false),
219    }
220}; 9];
221
222#[cfg(feature = "p01-trace")]
223pub(crate) fn p01_static_bytes() -> usize {
224    core::mem::size_of_val(&P01_SOURCE) + core::mem::size_of_val(&P01_SUBJECTS)
225}
226/// Publication binds the quota row while the exact incarnation still owns it.
227/// This is a copy-only hook; it never locks or allocates.
228#[cfg(feature = "p01-trace")]
229pub(crate) fn p01_bind(subject: u8, cell: u64, generation: u64) {
230    let _write = P01_SOURCE.write();
231    let Some(row) = P01_SUBJECTS.get(subject as usize) else {
232        P01_SOURCE.fault();
233        return;
234    };
235    if cell == 0
236        || cell as usize >= MAX_CELLS
237        || generation == 0
238        || row.ready.load(Ordering::SeqCst)
239    {
240        P01_SOURCE.fault();
241        return;
242    }
243    row.cell.store(cell, Ordering::SeqCst);
244    row.generation.store(generation, Ordering::SeqCst);
245    row.ready.store(true, Ordering::SeqCst);
246}
247#[cfg(feature = "p01-trace")]
248pub(crate) fn p01_snapshot() -> crate::p01_resources::Metric {
249    let before = P01_SOURCE.before();
250    let mut charge = 0u64;
251    for (index, row) in P01_SUBJECTS.iter().enumerate() {
252        if !row.ready.load(Ordering::SeqCst) || row.retired.load(Ordering::SeqCst) {
253            continue;
254        }
255        let cell = row.cell.load(Ordering::SeqCst);
256        let generation = row.generation.load(Ordering::SeqCst);
257        // Worker rows, if ever present, describe the same Cell charge. Never
258        // multiply the execution-context counter by the number of task rows.
259        if P01_SUBJECTS[..index].iter().any(|prior| {
260            prior.ready.load(Ordering::SeqCst)
261                && !prior.retired.load(Ordering::SeqCst)
262                && prior.cell.load(Ordering::SeqCst) == cell
263                && prior.generation.load(Ordering::SeqCst) == generation
264        }) {
265            continue;
266        }
267        let value = (IN_USE[cell as usize].load(Ordering::SeqCst) & BYTES_MASK) as u64;
268        match charge.checked_add(value) {
269            Some(next) => charge = next,
270            None => P01_SOURCE.fault(),
271        }
272    }
273    // Retired means this execution-context charge row was reset/deregistered,
274    // never that deferred physical resources or historical objects were freed.
275    P01_SOURCE.metric(charge, before, 5)
276}

278/// Start slot `id`'s counters from zero under `tag`.
279fn reset_counters(cell_id: CellId, tag: usize) {
280    let id = cell_id.0 as usize;
281    #[cfg(feature = "p01-trace")]
282    let _write = P01_SOURCE.write();
283    #[cfg(feature = "p01-trace")]
284    if P01_SUBJECTS.iter().any(|row| {
285        row.ready.load(Ordering::SeqCst)
286            && !row.retired.load(Ordering::SeqCst)
287            && row.cell.load(Ordering::SeqCst) == cell_id.0
288    }) {
289        P01_SOURCE.fault();
290    }
291    IN_USE[id].store(tag, Ordering::Release);
292    DMA_IN_USE[id].store(tag, Ordering::Release);
293    GUEST_IN_USE[id].store(tag, Ordering::Release);
294}

296/// Register incarnation `generation` of `cell_id` with `limits`, replacing
297/// whatever the slot held: its counters restart from zero under the new
298/// generation. Allocates nothing and takes no lock.
299pub(crate) fn register(cell_id: CellId, generation: u64, limits: Limits) {
300    let charge = Charge::new(cell_id, generation);
301    let Some(id) = charge.slot() else {
302        return;
303    };
304    reset_counters(cell_id, charge.tag());
305    GUEST_LIMITS[id].store(limits.guest.min(BYTES_MASK), Ordering::Release);
306    QUOTA_LIMITS[id].store(limits.memory.min(BYTES_MASK), Ordering::Release);
307}
308
309/// Deregister a Cell on exit: the slot holds no incarnation, and its counters
310/// keep the retired generation at zero bytes, so a late refund is a no-op.
311pub(crate) fn deregister(cell_id: CellId) {
312    let Some(id) = Charge::new(cell_id, 0).slot() else {
313        return;
314    };
315    #[cfg(feature = "p01-trace")]
316    let _write = P01_SOURCE.write();
317    #[cfg(feature = "p01-trace")]
318    for row in &P01_SUBJECTS {
319        if row.ready.load(Ordering::SeqCst) && row.cell.load(Ordering::SeqCst) == cell_id.0 {
320            row.retired.store(true, Ordering::SeqCst);
321        }
322    }
323    QUOTA_LIMITS[id].store(UNREGISTERED, Ordering::Release);
324    GUEST_LIMITS[id].store(0, Ordering::Release);
325    for counters in [&IN_USE, &DMA_IN_USE, &GUEST_IN_USE] {
326        let _ = counters[id].fetch_update(Ordering::AcqRel, Ordering::Acquire, |word| {
327            Some(word & !BYTES_MASK)
328        });
329    }
330}

332/// Rollback owner for a quota row belonging to an unpublished cell.
333pub(crate) struct QuotaReservation {
334    cell_id: CellId,
335    committed: bool,
336}
337
338impl QuotaReservation {
339    /// Reserve the exact requested CellId for incarnation `generation`.
340    pub(crate) fn reserve(
341        cell_id: CellId,
342        generation: u64,
343        limits: Limits,
344    ) -> Result<Self, types::ViError> {
345        let charge = Charge::new(cell_id, generation);
346        let Some(id) = charge.slot() else {
347            return Err(types::ViError::PermissionDenied);
348        };
349        // The claim is the limit store itself, so two reservations of one
350        // CellId cannot both succeed.
351        if QUOTA_LIMITS[id]
352            .compare_exchange(
353                UNREGISTERED,
354                limits.memory.min(BYTES_MASK),
355                Ordering::AcqRel,
356                Ordering::Acquire,
357            )
358            .is_err()
359        {
360            return Err(types::ViError::AlreadyExists);
361        }
362        reset_counters(cell_id, charge.tag());
363        GUEST_LIMITS[id].store(limits.guest.min(BYTES_MASK), Ordering::Release);
364        Ok(Self {
365            cell_id,
366            committed: false,
367        })
368    }
369
370    /// Reserve the lowest vacant bounded identity. Cell identity is independent
371    /// from the monotonic task ID, so exited cells release their quota slot for
372    /// later publication rather than permanently exhausting `MAX_CELLS`.
373    pub(crate) fn reserve_next(generation: u64, limits: Limits) -> Result<Self, types::ViError> {
374        for raw in 1..MAX_CELLS {
375            match Self::reserve(CellId(raw as u64), generation, limits) {
376                Ok(reservation) => return Ok(reservation),
377                Err(types::ViError::AlreadyExists) => {}
378                Err(error) => return Err(error),
379            }
380        }
381        log::error!(
382            "Stack alloc failed: cell quota table exhausted (max={})",
383            MAX_CELLS
384        );
385        Err(types::ViError::OutOfMemory)
386    }
387
388    pub(crate) fn cell_id(&self) -> CellId {
389        self.cell_id
390    }
391
392    /// Transfer cleanup responsibility to the published task exit path.
393    pub(crate) fn commit(mut self) {
394        self.committed = true;
395    }
396}
397
398impl Drop for QuotaReservation {
399    fn drop(&mut self) {
400        if !self.committed {
401            deregister(self.cell_id);
402        }
403    }
404}

406/// Add `bytes` to `charge`'s `account`. Refused past the slot's limit, or when
407/// the slot holds a different incarnation. The kernel, a CellId beyond the
408/// table and an unregistered slot have no limit and are not counted
409/// (`Ok(false)`).
410fn add(charge: Charge, account: Account, bytes: usize) -> Result<bool, QuotaExceeded> {
411    let Some(id) = charge.slot() else {
412        return Ok(false);
413    };
414    let registered = QUOTA_LIMITS[id].load(Ordering::Acquire);
415    if registered == UNREGISTERED {
416        return Ok(false);
417    }
418    let limit = match account {
419        Account::Memory | Account::Dma => registered,
420        Account::Guest => GUEST_LIMITS[id].load(Ordering::Acquire),
421    };
422    let tag = charge.tag();
423    #[cfg(feature = "p01-trace")]
424    let _write = P01_SOURCE.write();
425    // checked_add refuses a hostile size near MAX instead of wrapping past the
426    // limit, and the byte count can never carry into the generation bits.
427    account.counters()[id]
428        .fetch_update(Ordering::AcqRel, Ordering::Acquire, |word| {
429            if word & !BYTES_MASK != tag {
430                return None;
431            }
432            let bytes = (word & BYTES_MASK)
433                .checked_add(bytes)
434                .filter(|&total| total <= limit)?;
435            Some(tag | bytes)
436        })
437        .map(|_| true)
438        .map_err(|_| QuotaExceeded)
439}

441/// Return `bytes` to `charge`'s `account`. A no-op unless the slot still holds
442/// that incarnation, so a refund never lands on a later incarnation of the
443/// CellId. Saturating: never underflows into the generation bits.
444pub(crate) fn refund(charge: Charge, account: Account, bytes: usize) {
445    let Some(id) = charge.slot() else {
446        return;
447    };
448    let tag = charge.tag();
449    #[cfg(feature = "p01-trace")]
450    let _write = P01_SOURCE.write();
451    let _ = account.counters()[id].fetch_update(Ordering::AcqRel, Ordering::Acquire, |word| {
452        (word & !BYTES_MASK == tag).then(|| without_bytes(word, bytes))
453    });
454}

456/// Bytes taken from one account of one incarnation, refunded once on drop, to
457/// that incarnation only (a no-op once the slot was re-registered).
458#[must_use = "dropping a reservation refunds it"]
459#[derive(Debug)]
460pub(crate) struct Reservation {
461    charge: Charge,
462    account: Account,
463    bytes: usize,
464}
465
466impl Reservation {
467    /// Take `bytes` from `charge`'s `account`, or refuse with nothing taken.
468    pub(crate) fn take(
469        charge: Charge,
470        account: Account,
471        bytes: usize,
472    ) -> Result<Self, QuotaExceeded> {
473        add(charge, account, bytes)?;
474        Ok(Self {
475            charge,
476            account,
477            bytes,
478        })
479    }
480
481    /// The kernel's budget for an object that draws frames over its life
482    /// (a kernel-owned VM's tables): every split succeeds and nothing is
483    /// counted.
484    pub(crate) const fn kernel(account: Account) -> Self {
485        Self {
486            charge: Charge::KERNEL,
487            account,
488            bytes: usize::MAX,
489        }
490    }
491
492    /// Draw `bytes` down into a reservation of their own, same owner and
493    /// account; `None`, changing nothing, when fewer remain.
494    pub(crate) fn split(&mut self, bytes: usize) -> Option<Self> {
495        self.bytes = self.bytes.checked_sub(bytes)?;
496        Some(Self {
497            charge: self.charge,
498            account: self.account,
499            bytes,
500        })
501    }
502
503    /// Leave the bytes charged with no owning object. They return only through
504    /// [`refund`] with the same charge and account, or at `deregister`.
505    pub(crate) fn keep(self) {
506        core::mem::forget(self);
507    }
508
509    /// Hand the bytes to an object that refunds them itself (a `FrameRun`
510    /// when its frames are freed): they stay charged, and the parts name the
511    /// refund.
512    pub(crate) fn into_parts(self) -> (Charge, Account, usize) {
513        let parts = (self.charge, self.account, self.bytes);
514        core::mem::forget(self);
515        parts
516    }
517}
518
519impl Drop for Reservation {
520    fn drop(&mut self) {
521        refund(self.charge, self.account, self.bytes);
522    }
523}

525/// Charge `size` heap bytes to `charge` and return the owner token the charge
526/// was recorded against, or `None` if it is refused — the caller
527/// (`QuotaAlloc::alloc`) must return `null_mut()` in that case.
528///
529/// The token names the exact incarnation (CellId + generation) that was
530/// charged; `UNCHARGED_OWNER` when the charge is not counted. Pass it to
531/// `refund_owner` so the refund reaches that incarnation and nothing else.
532pub(crate) fn charge_owner(charge: Charge, size: usize) -> Option<usize> {
533    match add(charge, Account::Memory, size) {
534        Ok(true) => Some(charge.tag() | charge.cell.0 as usize),
535        Ok(false) => Some(UNCHARGED_OWNER),
536        Err(QuotaExceeded) => None,
537    }
538}

540/// Refund `size` bytes to the row incarnation named by `owner` (from
541/// `charge_owner`).  Lock-free.  A no-op when the row has been re-registered
542/// since the charge: that incarnation's counter was reset and is not ours.
543pub(crate) fn refund_owner(owner: usize, size: usize) {
544    let cell_id_raw = owner & BYTES_MASK;
545    if cell_id_raw == 0 || cell_id_raw >= MAX_CELLS {
546        return;
547    }
548    #[cfg(feature = "p01-trace")]
549    let _write = P01_SOURCE.write();
550    let _ = IN_USE[cell_id_raw].fetch_update(Ordering::AcqRel, Ordering::Acquire, |word| {
551        (word & !BYTES_MASK == owner & !BYTES_MASK).then(|| without_bytes(word, size))
552    });
553}

555/// The registered memory limit of a Cell slot, or `None` when it has none.
556/// Lock-free; safe under any lock.
557pub fn limit(cell_id_raw: usize) -> Option<usize> {
558    QUOTA_LIMITS
559        .get(cell_id_raw)
560        .map(|limit| limit.load(Ordering::Acquire))
561        .filter(|&limit| limit != UNREGISTERED)
562}

564/// Current byte usage for a Cell (for diagnostics).
565pub fn in_use(cell_id: CellId) -> usize {
566    let id = cell_id.0 as usize;
567    if id < MAX_CELLS {
568        IN_USE[id].load(Ordering::Relaxed) & BYTES_MASK
569    } else {
570        0
571    }
572}
573
574/// Current guest-account usage for a Cell (for diagnostics and fixtures).
575pub fn guest_in_use(cell_id: CellId) -> usize {
576    GUEST_IN_USE
577        .get(cell_id.0 as usize)
578        .map_or(0, |word| word.load(Ordering::Acquire) & BYTES_MASK)
579}

591/// Copies quota state into heap-backed rows.
592///
593/// Heap-backed, never stack-resident: under `cell-scale-experiment` `MAX_CELLS`
594/// is 4096, so these four arrays are 64 KiB + 3 × 32 KiB = 160 KiB — more than
595/// the 64 KiB kernel stack (`__stack_bottom..__stack_top`, `kernel/linker.ld`).
596/// A stack copy wrote straight through every static below the stack — `PLATFORM`
597/// sits 5.5 KiB under `__stack_bottom` — so the boot self-tests ran on smashed
598/// state and the kernel panicked in `platform::with` ("platform::init not called
599/// before platform::with"); measured 2026-10-03. Returning by value moved the
600/// same 128 KiB through `StateSnapshot`, so the whole snapshot chain must stay
601/// heap-backed.
602pub(crate) fn snapshot() -> QuotaSnapshot {
603    let mut limits = alloc::vec![None; MAX_CELLS];
604    let mut heap = alloc::vec![0; MAX_CELLS];
605    let mut dma = alloc::vec![0; MAX_CELLS];
606    let mut guest = alloc::vec![0; MAX_CELLS];
607    for cell_id in 0..MAX_CELLS {
608        limits[cell_id] = limit(cell_id);
609        heap[cell_id] = IN_USE[cell_id].load(Ordering::Acquire) & BYTES_MASK;
610        dma[cell_id] = DMA_IN_USE[cell_id].load(Ordering::Acquire) & BYTES_MASK;
611        guest[cell_id] = GUEST_IN_USE[cell_id].load(Ordering::Acquire) & BYTES_MASK;
612    }
613    QuotaSnapshot {
614        limits,
615        heap,
616        dma,
617        guest,
618    }
619}

626#[cfg(feature = "test-hooks")]
627pub(crate) fn reusable_cell_id_contract() -> bool {
628    const GENERATION: u64 = 1;
629    let limits = Limits {
630        memory: DEFAULT_QUOTA_BYTES,
631        guest: 0,
632    };
633    let before = snapshot();
634    let mut held = alloc::vec::Vec::new();
635    loop {
636        match QuotaReservation::reserve_next(GENERATION, limits) {
637            Ok(reservation) => held.push(reservation),
638            Err(types::ViError::OutOfMemory) | Err(types::ViError::PermissionDenied) => break,
639            Err(_) => return false,
640        }
641    }
642    let Some(released) = held.pop() else {
643        return false;
644    };
645    let released_id = released.cell_id();
646    drop(released);
647    let reused = QuotaReservation::reserve_next(GENERATION, limits)
648        .map(|reservation| reservation.cell_id() == released_id)
649        .unwrap_or(false);
650    drop(held);
651    reused && snapshot() == before
652}

674    fn concurrent_workers_share_one_atomic_dma_quota() {
675        let _guard = crate::TEST_STATE_LOCK.lock();
676        const LIMIT: usize = 4096;
677        register(CELL, 1, limits(LIMIT));
678        let charge = Charge::new(CELL, 1);
679
680        let barrier = Arc::new(Barrier::new(2));
681        let successes = Arc::new(AtomicUsize::new(0));
682        let mut workers = alloc::vec::Vec::new();
683        for _ in 0..2 {
684            let barrier = Arc::clone(&barrier);
685            let successes = Arc::clone(&successes);
686            workers.push(thread::spawn(move || {
687                let reservation = Reservation::take(charge, Account::Dma, LIMIT);
688                if reservation.is_ok() {
689                    successes.fetch_add(1, Ordering::Relaxed);
690                }
691                barrier.wait();
692                drop(reservation);
693            }));
694        }
695        for worker in workers {
696            worker.join().expect("DMA quota worker");
697        }
698
699        assert_eq!(successes.load(Ordering::Relaxed), 1);
700        assert_eq!(dma_in_use(CELL), 0);
701        deregister(CELL);
702    }
