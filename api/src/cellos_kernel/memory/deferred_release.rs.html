1//! The private-root half of the release contract: frames a private root no
2//! longer maps, retained until that root's tag invalidation is confirmed.
3//!
4//! A release path must never wait synchronously for a remote acknowledgement:
5//! a peer hart can be held non-preemptible for seconds. A path whose bounded
6//! probe did not confirm its tag hands the frames (and, for a root teardown,
7//! the tag's lease) to [`defer_frames`] / [`defer_frames_and_tag`]. They
8//! become `Quarantined(TagUnconfirmed)` runs in the one frame-run quarantine
9//! (`memory::frame_run::PENDING`), whose hart-0 reaper frees them once every
10//! remote hart has published an epoch asked for after the unmap.
11//!
12//! Fail-closed, in this order:
13//!
14//! 1. a frame is released **only** after the invalidation that covers it is
15//!    acknowledged — never on a timeout, never silently;
16//! 2. nothing is abandoned: a hart that never answers keeps the frames
17//!    quarantined (counted, and logged once on escalation) and the tag reads
18//!    unconfirmed for as long as it does (census §7.5);
19//! 3. a wait that cannot be recorded (no heap) leaks its frames, counted, and
20//!    makes every tag read unconfirmed for the rest of the boot.
21//!
22//! [`tag_invalidation_unconfirmed`] is the one answer every dependent path asks
23//! (a grant drain, a re-publish over a retired VA): `false` is the only proof
24//! that no hart can still resolve a stale translation under the tag.
25

27use super::frame_run::{FlushOutcome, KernelBackend, RunKind, TagWait, PENDING};
28use super::tag_pool::TagLease;
29use alloc::vec::Vec;

31/// Retain `frames` until `asid`'s invalidation is confirmed.
32///
33/// The frames must already be detached from every structure that could hand
34/// them back to the allocator: this is the last owner. The tag stays live (the
35/// caller keeps its lease), so nothing can reuse it while the wait is pending.
36pub fn defer_frames(asid: usize, domain: u64, frames: Vec<OwnedFrame>, reason: &'static str) {
37    defer(asid, domain, frames, reason, None)
38}
39
40/// Retain `frames` **and** `tag`'s lease until its invalidation is confirmed.
41/// Used by a root teardown whose own confirmation failed: the reaper releases
42/// the lease, with its proof, only when the same wait completes. A wait that
43/// cannot be recorded drops the lease, which keeps the tag reserved for the
44/// boot.
45pub(crate) fn defer_frames_and_tag(
46    tag: TagLease,
47    domain: u64,
48    frames: Vec<OwnedFrame>,
49    reason: &'static str,
50) {
51    defer(tag.value() as usize, domain, frames, reason, Some(tag))
52}
53
54fn defer(
55    asid: usize,
56    domain: u64,
57    frames: Vec<OwnedFrame>,
58    reason: &'static str,
59    tag: Option<TagLease>,
60) {
61    #[cfg(feature = "p01-trace")]
62    let frames = {
63        let owner = crate::p01_resources::owner_for_domain(domain);
64        if owner.is_none()
65            && crate::memory::heap::p01_owner::current_attempt().is_some()
66            && !frames.is_empty()
67        {
68            // A selected prepublication retention has no exact domain binding.
69            crate::p01_retained::fault();
70        }
71        if let Some(owner) = owner {
72            // One wait per tag holds every retention behind it (stacks
73            // included), so no single retention's completion is observable.
74            crate::p01_resources::unknown(owner, 6);
75        }
76        let mut frames = frames;
77        for frame in &mut frames {
78            frame.p01_retain(owner, reason);
79        }
80        frames
81    };
82    let count = frames.len();
83    let mut runs = Vec::new();
84    if runs.try_reserve_exact(count).is_err() {
85        // No heap to carry the frames: never free them, and fail closed on
86        // every tag, as for any wait the quarantine cannot record.
87        for frame in frames {
88            core::mem::forget(frame);
89        }
90        PENDING.lose_frames(count, "no heap for the tag wait");
91        return;
92    }
93    runs.extend(
94        frames
95            .into_iter()
96            .map(|frame| frame.into_run(RunKind::PageTable)),
97    );
98    // Every online remote is asked again after the unmap: each serves its
99    // epoch with a full flush, which drops this tag's entries too.
100    let epochs = super::frame_run::tag_epochs(asid, FlushOutcome::Confirmed, &mut KernelBackend);
101    // The decision and the log are separated so the console never nests inside
102    // the quarantine lock.
103    match PENDING.defer_tag(asid, runs, tag, &epochs) {
104        TagWait::Queued => log::warn!(
105            "[aspace] deferred release queued: tag={} domain={} frames={} reason={} — the tag stays unconfirmed and every frame behind it is retained until an acknowledgement lands",
106            asid,
107            domain,
108            count,
109            reason
110        ),
111        TagWait::Extended { pages } => log::warn!(
112            "[aspace] deferred release extended: tag={} total_frames={} reason={}",
113            asid,
114            pages,
115            reason
116        ),
117        TagWait::Lost => {}
118    }
119}

121/// Is `asid`'s invalidation still unconfirmed?
122///
123/// True while a tag wait is pending in the frame-run quarantine, and for every
124/// tag once a tag wait could not be recorded. Every path that gates a reuse of
125/// the address space on a confirmed invalidation must ask this instead of
126/// waiting: a `false` answer is the only proof that no hart can still resolve
127/// a stale translation, and a missing acknowledgement must never look like one.
128pub fn tag_invalidation_unconfirmed(asid: usize) -> bool {
129    PENDING.tag_unconfirmed(asid, &KernelBackend)
130}

132/// Test-hooks view: frames retained behind tag `asid`.
133///
134/// A deferred release withholds its frames from the allocator, and no
135/// allocator-side count can see them. A fixture that accounts a deferred revoke
136/// by its grant backing alone (the `grant-gate` frames property) therefore
137/// needs this to tell a correct deferral from a leak.
138#[cfg(feature = "test-hooks")]
139#[cfg_attr(
140    not(any(cellos_domain_grants, target_arch = "riscv64")),
141    allow(dead_code) // reason: the grant fixtures and the RV64 aspace fixture assert on it
142)]
143pub(crate) fn deferred_release_frames_for(asid: usize) -> usize {
144    PENDING.tag_pages(asid)
145}
