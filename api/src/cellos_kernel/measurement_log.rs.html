1//! Per-Cell integrity measurement log (Linux IMA-style, TPM-free).
2//!
3//! On every `spawn_from_path`, the loader records `SHA-256(elf_bytes)` here
4//! BEFORE the cell is scheduled to run. Entries form an append-only log plus a
5//! rolling aggregate (`agg = SHA256(agg || entry_hash)`) — the single value a
6//! future DICE/EAT remote-attestation token signs to prove the exact software
7//! that ran on this device. See
8//! `docs/research/research-cell-security-permissions.md` §3.6.
9//!
10//! This is *measurement* (evidence), not *enforcement*: it does not block a
11//! cell. Signature-based rejection (Cell binary signing) is a separate,
12//! orthogonal gate. Pairing the two gives "measured + verified launch".
13
14use crate::sync::Spinlock;
15use alloc::string::String;
16use alloc::vec::Vec;
17
18/// Soft cap on retained entries — bounds kernel memory. Beyond this individual
19/// entries stop being appended (the aggregate still advances, so attestation
20/// integrity is preserved) and a warning is logged. Cell spawns are bounded in
21/// practice on G1.
22const MAX_ENTRIES: usize = 256;
23
24/// Most detailed entries one creator may retain, counting entries already
25/// committed through prepared publication and outstanding promises. A creator
26/// that reaches it through its OWN committed entries keeps going: `reserve`
27/// evicts that creator's oldest committed detailed entry (the aggregate, which
28/// already covers it, is untouched), so one creator can never occupy more than
29/// its share and a long-lived creator is never locked out. Only a creator whose
30/// outstanding promises alone reach the share is refused. Entries of other
31/// creators and ordinary (`commit_staged`) launches are never evicted.
32pub(crate) const CREATOR_SHARE: usize = MAX_ENTRIES / 4;
33
34/// Identity of the Cell that prepared a child: `(cell id, generation)`.
35pub type Creator = (u64, u64);
36
37/// Creator recorded on entries that no prepared publication produced.
38pub const NO_CREATOR: Creator = (0, 0);
39
40/// Per-creator count of OUTSTANDING promises. A row exists only while its count
41/// is non-zero, so the table never exceeds `MAX_ENTRIES` rows.
42struct Share {
43    creator: Creator,
44    promised: usize,
45}
46
47/// One measured cell: its TID, the SHA-256 of its ELF image, its path, and the
48/// creator that prepared it (`NO_CREATOR` for ordinary launches).
49pub struct MeasureEntry {
50    pub tid: u32,
51    pub hash: [u8; 32],
52    pub path: String,
53    pub creator: Creator,
54}
55
56struct Log {
57    entries: Vec<MeasureEntry>,
58    aggregate: [u8; 32],
59    /// Detailed-log slots promised to prepared (not yet published) children.
60    /// Invariants under `LOG`: `entries.len() + promises <= MAX_ENTRIES` and
61    /// `entries.capacity() >= entries.len() + promises`, so a reserved commit
62    /// appends without allocating and no ordinary launch can consume the slot.
63    promises: usize,
64    /// Per-creator share accounting for prepared publications.
65    shares: Vec<Share>,
66}
67
68static LOG: Spinlock<Log> = Spinlock::new(Log {
69    entries: Vec::new(),
70    aggregate: [0u8; 32],
71    promises: 0,
72    shares: Vec::new(),
73});
74
75/// Force-release this module's lock during fault teardown.
76///
77/// # Safety
78/// Single-hart kernel; called only from the fault/panic path with interrupts
79/// disabled. Force-unlocking an already-free Spinlock is a no-op.
80pub unsafe fn force_unlock_locks() {
81    LOG.force_unlock();
82}
83
84fn advance_aggregate(log: &mut Log, hash: &[u8; 32]) {
85    let mut buf = [0u8; 64];
86    buf[..32].copy_from_slice(&log.aggregate);
87    buf[32..].copy_from_slice(hash);
88    log.aggregate = crate::sha256::sha256(&buf);
89}

91#[cfg(cellos_p01_trace)]
92fn attribute_log_buffers(log: &Log) {
93    if log.entries.capacity() != 0 {
94        crate::memory::heap::p01_owner::shared_buffer(log.entries.as_ptr() as usize,
95            core::alloc::Layout::array::<MeasureEntry>(log.entries.capacity()).unwrap(),
96            crate::memory::heap::p01_owner::LOG_SHARED);
97    }
98}

100#[cfg(cellos_p01_trace)]
101fn attribute_path(path: &String) {
102    if path.capacity() != 0 {
103        crate::memory::heap::p01_owner::shared_buffer(path.as_ptr() as usize,
104            core::alloc::Layout::array::<u8>(path.capacity()).unwrap(),
105            crate::memory::heap::p01_owner::LOG_PATH);
106    }
107}

109fn emit_audit(tid: usize, hash: &[u8; 32]) {
110    let hp = u32::from_le_bytes([hash[0], hash[1], hash[2], hash[3]]);
111    crate::audit::log_event(
112        crate::audit::AuditEvent::CellMeasure,
113        &crate::audit::encode_u32x2(tid as u32, hp),
114    );
115}

117/// Commit a digest and owned label staged before atomic publication.
118///
119/// Ordinary launches never consume a slot promised to a prepared child: they
120/// append only while `len + promises < MAX_ENTRIES` and while capacity for the
121/// new entry *and* every outstanding promise can be reserved. Otherwise the
122/// aggregate-only fallback below applies, exactly as at the historical cap.
123pub(crate) fn commit_staged(tid: usize, path: String, hash: [u8; 32]) {
124    {
125        let mut log = LOG.lock();
126        #[cfg(cellos_p01_trace)]
127        let _semantic = crate::memory::heap::p01_owner::creation(None, crate::memory::heap::p01_owner::LOG_SHARED);
128        #[cfg(cellos_p01_trace)]
129        attribute_log_buffers(&log);
130        advance_aggregate(&mut log, &hash);
131
132        let promised = log.promises;
133        if log.entries.len() + promised < MAX_ENTRIES {
134            // Growth is fallible: `Vec::push` reallocates through the infallible
135            // allocator, and a capacity doubling that cannot be satisfied halts
136            // the kernel instead of dropping a measurement. Dropping the entry
137            // keeps the aggregate advancing, which is the property attestation
138            // depends on. Capacity for outstanding promises is kept in addition
139            // to this entry.
140            if log.entries.try_reserve(promised + 1).is_err() {
141                log::warn!(
142                    "[measure] no heap for entry {} — aggregate still advancing",
143                    log.entries.len()
144                );
145            } else {
146                #[cfg(cellos_p01_trace)]
147                attribute_path(&path);
148                log.entries.push(MeasureEntry {
149                    tid: tid as u32,
150                    hash,
151                    path,
152                    creator: NO_CREATOR,
153                });
154            }
155        } else {
156            log::warn!(
157                "[measure] log full ({} entries, {} promised) — aggregate still advancing",
158                log.entries.len(),
159                promised
160            );
161        }
162    }
163    emit_audit(tid, &hash);
164}

166/// Promise of one detailed log entry for a prepared, unpublished child.
167///
168/// `reserve` charges bounded capacity before any visible publication; `commit`
169/// consumes exactly that promise, appends and advances the aggregate together,
170/// and cannot allocate or fail. Dropping an uncommitted guard releases the
171/// promise without touching the aggregate or the log.
172pub(crate) struct PublicationReservation {
173    armed: bool,
174    creator: Creator,
175}
176
177impl PublicationReservation {
178    /// Approximate retained bytes one committed entry costs (excluding the path
179    /// bytes, which the staged label already owns).
180    pub(crate) const ENTRY_BYTES: usize = core::mem::size_of::<MeasureEntry>();
181
182    /// `creator` is charged one outstanding promise. At its share through its own
183    /// committed entries (or with the log full), its oldest committed entry is
184    /// evicted to make room; it is refused only when its promises alone reach the
185    /// share or when the capacity is held by other creators' entries. Eviction
186    /// happens only after every fallible step succeeded.
187    pub(crate) fn reserve(creator: Creator) -> Result<Self, types::ViError> {
188        let mut log = LOG.lock();
189        let slot = log.shares.iter().position(|share| share.creator == creator);
190        let promised = slot.map_or(0, |i| log.shares[i].promised);
191        if promised >= CREATOR_SHARE {
192            return Err(types::ViError::OutOfMemory);
193        }
194        let own = log.entries.iter().position(|entry| entry.creator == creator);
195        let committed = if own.is_some() {
196            log.entries.iter().filter(|entry| entry.creator == creator).count()
197        } else {
198            0
199        };
200        let wanted = log
201            .entries
202            .len()
203            .checked_add(log.promises)
204            .and_then(|n| n.checked_add(1))
205            .ok_or(types::ViError::OutOfMemory)?;
206        let evict = if wanted > MAX_ENTRIES || promised + committed >= CREATOR_SHARE {
207            // The detailed record is required for a prepared child. Room is made
208            // only from this creator's own oldest entry; with none, preparation is
209            // refused, never silently omitted.
210            Some(own.ok_or(types::ViError::OutOfMemory)?)
211        } else {
212            None
213        };
214        let additional = log.promises + 1;
215        #[cfg(cellos_p01_trace)]
216        let _semantic = crate::memory::heap::p01_owner::creation(None, crate::memory::heap::p01_owner::LOG_SHARED);
217        log.entries
218            .try_reserve(additional)
219            .map_err(|_| types::ViError::OutOfMemory)?;
220        match slot {
221            Some(i) => log.shares[i].promised += 1,
222            None => {
223                log.shares
224                    .try_reserve(1)
225                    .map_err(|_| types::ViError::OutOfMemory)?;
226                log.shares.push(Share { creator, promised: 1 });
227            }
228        }
229        if let Some(index) = evict {
230            // Capacity only shrinks the entry count; the aggregate is untouched.
231            drop(log.entries.remove(index));
232        }
233        log.promises += 1;
234        Ok(Self { armed: true, creator })
235    }
236
237    /// Publish the promised entry. The path was staged and owned before this
238    /// call; nothing here allocates.
239    pub(crate) fn commit(mut self, tid: usize, path: String, hash: [u8; 32]) {
240        {
241            let mut log = LOG.lock();
242            #[cfg(cellos_p01_trace)]
243            attribute_log_buffers(&log);
244            assert!(
245                log.promises != 0 && log.entries.len() < log.entries.capacity(),
246                "measurement publication promise lost its reserved slot"
247            );
248            advance_aggregate(&mut log, &hash);
249            log.promises -= 1;
250            // The promise becomes a committed entry: it stops counting as outstanding
251            // for the creator (no allocation: the row is only shrunk or swapped out).
252            if let Some(i) = log.shares.iter().position(|share| share.creator == self.creator) {
253                log.shares[i].promised = log.shares[i].promised.saturating_sub(1);
254                if log.shares[i].promised == 0 {
255                    log.shares.swap_remove(i);
256                }
257            }
258            #[cfg(cellos_p01_trace)]
259            attribute_path(&path);
260            log.entries.push(MeasureEntry {
261                tid: tid as u32,
262                hash,
263                path,
264                creator: self.creator,
265            });
266        }
267        self.armed = false;
268        emit_audit(tid, &hash);
269    }
270}
271
272impl Drop for PublicationReservation {
273    fn drop(&mut self) {
274        if self.armed {
275            let mut log = LOG.lock();
276            log.promises = log.promises.saturating_sub(1);
277            // The promise returns to the creator.
278            if let Some(i) = log.shares.iter().position(|share| share.creator == self.creator) {
279                log.shares[i].promised = log.shares[i].promised.saturating_sub(1);
280                if log.shares[i].promised == 0 {
281                    log.shares.swap_remove(i);
282                }
283            }
284        }
285    }
286}

288/// (entries, capacity, promises, maximum) for publication regression tests.
289#[cfg(feature = "test-hooks")]
290pub(crate) fn test_shape() -> (usize, usize, usize, usize) {
291    let log = LOG.lock();
292    (log.entries.len(), log.entries.capacity(), log.promises, MAX_ENTRIES)
293}

295/// `(outstanding promises, retained committed entries)` charged to `creator`.
296#[cfg(feature = "test-hooks")]
297pub(crate) fn test_creator_used(creator: Creator) -> (usize, usize) {
298    let log = LOG.lock();
299    (
300        log.shares.iter().find(|share| share.creator == creator).map_or(0, |share| share.promised),
301        log.entries.iter().filter(|entry| entry.creator == creator).count(),
302    )
303}
