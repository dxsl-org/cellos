1use core::cell::UnsafeCell;
2use core::mem::MaybeUninit;
3use core::sync::atomic::{AtomicBool, Ordering};
4
5/// Write-once storage for immutable data published during single-core boot.
6///
7/// Unlike a spinlock, publication uses no exclusive load/store instructions,
8/// so it remains usable before the MMU assigns Normal memory attributes.
9pub(super) struct BootOnce<T> {
10    initialized: AtomicBool,
11    value: UnsafeCell<MaybeUninit<T>>,
12}
13
14// SAFETY: initialization is restricted to the boot CPU before SMP starts;
15// publication is release/acquire and the value is immutable afterward.
16unsafe impl<T: Sync> Sync for BootOnce<T> {}
17
18impl<T> BootOnce<T> {
19    pub(super) const fn new() -> Self {
20        Self {
21            initialized: AtomicBool::new(false),
22            value: UnsafeCell::new(MaybeUninit::uninit()),
23        }
24    }
25
26    /// Publish the value exactly once before secondary CPUs or interrupts run.
27    ///
28    /// # Safety
29    /// The caller must guarantee exclusive boot-time access with no concurrent
30    /// readers or initializers. A repeated sequential call is rejected.
31    #[cfg(any(test, target_arch = "riscv64", target_arch = "aarch64"))]
32    pub(super) unsafe fn initialize(&self, value: T) {
33        assert!(
34            !self.initialized.load(Ordering::Relaxed),
35            "boot value initialized more than once"
36        );
37        // SAFETY: guaranteed by the caller; the release store publishes the
38        // completed write to later acquire readers.
39        unsafe { (*self.value.get()).write(value) };
40        self.initialized.store(true, Ordering::Release);
41    }
42
43    pub(super) fn get(&self) -> Option<&T> {
44        if !self.initialized.load(Ordering::Acquire) {
45            return None;
46        }
47        // SAFETY: an acquire load observed the release publication, and the
48        // value is never mutated after initialization.
49        Some(unsafe { (&*self.value.get()).assume_init_ref() })
50    }
51}

53#[cfg(test)]
54mod tests {
55    use super::BootOnce;
56
57    #[test]
58    fn publishes_one_immutable_value() {
59        let value = BootOnce::new();
60        assert!(value.get().is_none());
61
62        // SAFETY: this local test is single-threaded and initializes once.
63        unsafe { value.initialize(42_u64) };
64
65        assert_eq!(value.get(), Some(&42));
66        assert_eq!(value.get(), Some(&42));
67    }
68
69    #[test]
70    #[should_panic(expected = "boot value initialized more than once")]
71    fn rejects_reinitialization() {
72        let value = BootOnce::new();
73        // SAFETY: both calls are single-threaded; the second must panic before writing.
74        unsafe { value.initialize(1_u64) };
75        unsafe { value.initialize(2_u64) };
76    }
