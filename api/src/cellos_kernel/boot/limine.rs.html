237/// A response pointer must name a complete, aligned bootloader-owned object.
238#[cfg(target_arch = "x86_64")]
239pub fn mp_record_valid<T>(pointer: *const T, count: usize) -> bool {
240    let start = pointer as usize;
241    if start == 0 || start % core::mem::align_of::<T>() != 0 {
242        return false;
243    }
244    let Some(size) = core::mem::size_of::<T>().checked_mul(count) else { return false };
245    let Some(end) = start.checked_add(size) else { return false };
246    let Some(offset) = get_hhdm_offset() else { return false };
247    let Some(physical) = (start as u64).checked_sub(offset) else { return false };
248    let Some(physical_end) = (end as u64).checked_sub(offset) else { return false };
249    let Some(map) = get_memory_map() else { return false };
250    // Limine's memory-map response is already a mandatory trusted boot input.
251    unsafe {
252        (0..map.entry_count as usize).any(|index| {
253            let entry = &**map.entries.add(index);
254            entry.entry_type == 5
255                && physical >= entry.base
256                && entry.base.checked_add(entry.length).is_some_and(|limit| physical_end <= limit)
257        })
258    }
259}

261#[cfg(target_arch = "x86_64")]
262pub fn get_mp_response() -> Option<&'static LimineMpResponse> {
263    let response = unsafe { core::ptr::read_volatile(&raw const MP_REQUEST.response) };
264    mp_record_valid(response, 1).then(|| unsafe { &*response })
265}

315/// DTB (Device Tree Blob) request — Limine v8 protocol.
316/// The response provides the physical address of the DTB passed by firmware.
317/// On RISC-V UEFI boots this is the only reliable way to get the DTB address
318/// (the `a1` register contains the Limine boot info pointer, not the DTB).
319///
320/// Request GUID (after LIMINE_COMMON_MAGIC):
321///   0x0b40dca86177520e, 0xc8809c1e7bbbde33
322/// Verify against: https://github.com/limine-bootloader/limine/blob/v8.x-binary/limine.h
323#[repr(C)]
324pub struct LimineDtbRequest {
325    pub id: [u64; 4],
326    pub revision: u64,
327    pub response: *const LimineDtbResponse,
328}
329
330unsafe impl Send for LimineDtbRequest {}
331unsafe impl Sync for LimineDtbRequest {}

333#[repr(C)]

337    pub dtb_ptr: *const u8,

353/// Return the DTB physical address from the Limine DtbResponse, if present.
354///
355/// Returns `None` when:
356/// - not booted via Limine (OpenSBI direct boot), or
357/// - Limine did not populate a DTB response (non-RISC-V or no firmware DTB).
358///
359/// The kernel's `platform::init` calls this first and falls back to the `a1`
360/// register value when it returns `None`.
361pub fn get_dtb_ptr() -> Option<usize> {
362    // SAFETY: DTB_REQUEST is a Limine protocol request static. The response
363    // pointer is written by Limine before the kernel entry point is called,
364    // so no data race exists at the point this function is invoked (single
365    // hart, no other code running). Null check guards against absent response.
366    unsafe {
367        let response = DTB_REQUEST.response;
368        if response.is_null() {
369            return None;
370        }
371        let dtb = (*response).dtb_ptr;
372        if dtb.is_null() {
373            return None;
374        }
375        Some(dtb as usize)
376    }
377}

423/// Get the RSDP physical address from the Limine RSDP response.
424///
425/// Returns `None` when:
426/// - not x86_64 (compile-time), or
427/// - Limine did not populate an RSDP response (UEFI firmware has no ACPI),
428/// - the response address is null/zero.
429///
430/// The kernel's x86_64 boot block calls this before `init_kernel_paging_x86`
431/// to parse ACPI tables for MMIO base addresses.
432#[cfg(target_arch = "x86_64")]
433pub fn get_rsdp_ptr() -> Option<usize> {
434    // SAFETY: RSDP_REQUEST is a Limine protocol request static. The response
435    // pointer is written by Limine before the kernel entry point is called,
436    // so no data race exists at the point this function is invoked (single
437    // CPU, no other code running). Null check guards against absent response.
438    unsafe {
439        let response = RSDP_REQUEST.response;
440        if response.is_null() {
441            return None;
442        }
443        let addr = (*response).address as usize;
444        if addr == 0 {
445            None
446        } else {
447            Some(addr)
448        }
449    }
450}

452/// Limine memory map entry types
453#[derive(Debug, Clone, Copy, PartialEq, Eq)]
454#[repr(u64)]
455pub enum LimineMemoryType {
456    Usable = 0,
457    Reserved = 1,
458    AcpiReclaimable = 2,
459    AcpiNvs = 3,
460    BadMemory = 4,
461    BootloaderReclaimable = 5,
462    KernelAndModules = 6,
463    Framebuffer = 7,
464}
465
466impl LimineMemoryMapEntry {
467    pub fn memory_type(&self) -> LimineMemoryType {
468        match self.entry_type {
469            0 => LimineMemoryType::Usable,
470            1 => LimineMemoryType::Reserved,
471            2 => LimineMemoryType::AcpiReclaimable,
472            3 => LimineMemoryType::AcpiNvs,
473            4 => LimineMemoryType::BadMemory,
474            5 => LimineMemoryType::BootloaderReclaimable,
475            6 => LimineMemoryType::KernelAndModules,
476            7 => LimineMemoryType::Framebuffer,
477            _ => LimineMemoryType::Reserved,
478        }
479    }
