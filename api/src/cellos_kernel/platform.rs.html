1//! Platform peripheral discovery via DTB.
2//!
3//! The architecture's `crate::arch::boot::platform::init(dtb_ptr)` runs in
4//! `kmain` before any driver init (riscv64 parses the DTB, aarch64 takes the
5//! board backend's facts). All drivers read MMIO addresses via
6//! `platform::with(|p| p.xxx)`. riscv64 falls back to the SoC profile's
7//! defaults when the DTB is absent or a compatible node is not found.
8//!
9//! Call ordering invariant: `init` → `uart::init` → `hal::ARCH.init` (which calls
10//! `plic::init` internally). Any call to `with` before `init` panics.
11
12mod boot_once;
13
14use boot_once::BootOnce;
15
16// ── Public types ───────────────────────────────────────────────────────────────
17
18/// A VirtIO MMIO device found in the DTB.
19#[derive(Clone, Copy)]
20pub struct VirtioEntry {
21    pub base: usize,
22    pub irq: u32,
23}
24
25/// Platform peripheral layout populated once from the DTB at early boot.
26#[derive(Clone)]
27pub struct PlatformInfo {
28    pub uart_base: usize,
29    pub uart_irq: u32,
30    pub plic_base: usize,
31    /// Mapped region size for identity-map range covering all PLIC registers.
32    pub plic_size: usize,
33    pub clint_base: usize,
34    /// VirtIO MMIO slots from DTB (up to 8). `None` = slot unused.
35    pub virtio_mmio: [Option<VirtioEntry>; 8],
36    /// Goldfish RTC MMIO base (0 = not found in DTB, using default).
37    pub rtc_base: usize,
38}
39
40// ── Storage ────────────────────────────────────────────────────────────────────
41
42static PLATFORM: BootOnce<PlatformInfo> = BootOnce::new();
43
44/// Publish immutable platform data while the boot CPU is the only running core.
45///
46/// Spinlocks are forbidden on this path: ARM64 reaches it before the MMU has
47/// assigned Normal memory attributes, so LL/SC may abort on real hardware.
48#[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
49pub(crate) unsafe fn publish_boot(info: PlatformInfo) {
50    // SAFETY: every architecture calls `platform::init` once from `kmain`
51    // before interrupts, paging, or secondary-core startup.
52    unsafe { PLATFORM.initialize(info) };
53}
54
55// ── Public API ─────────────────────────────────────────────────────────────────
56
57/// Borrow the platform info. Panics if `init` was not called.
58pub fn with<R>(f: impl FnOnce(&PlatformInfo) -> R) -> R {
59    f(PLATFORM
60        .get()
61        .expect("[platform] platform::init not called before platform::with"))
