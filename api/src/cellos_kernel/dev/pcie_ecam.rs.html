1//! PCIe ECAM (Enhanced Configuration Access Mechanism) walker.
2//!
3//! The kernel fallback scans one admitted bus, probing functions per the
4//! header-type multi-function bit. Exposes a kernel-internal device snapshot.
5//!
6//! ECAM bases per machine:
7//!   x86_64      : runtime ACPI MCFG (no q35 fallback)
8//!   RISC-V virt : 0x3000_0000
9//!   AArch64     : none. The kernel discovers no config-space window there
10//!                 (devices are VirtIO MMIO), so it never touches PCI config space.
11//!
12//! The Platform Cell may map and enumerate the complete firmware-admitted bus
13//! range. Kernel fallback enumeration remains bus-0-only on non-x86 targets.
14//!
15//! Call ordering: `platform::init()` → `init_kernel_paging*()` → `pcie_ecam::init()`.
16

21/// Ethernet controller IDs that have a Driver Cell in this image, each with its
22/// owning cell.
23///
24/// This is the **family gate**: an Ethernet device is offered to a Driver Cell
25/// only when a cell drives its family, and every other Ethernet ID is still
26/// rejected **by name** at scan time (`[e1000] unsupported Ethernet
27/// vvvv:dddd; driver gate closed`). Widened in phase 04a of the x86 PC lane from
28/// "82540EM only" to "families that have a driver cell": the 8254x family
29/// (`/bin/e1000`) and the igb family (`/bin/igb`) — QEMU's 82576 model plus the
30/// datasheet i210/i211 SKUs the cell's `identity.rs` lists. Phase 08 (S3) adds
31/// the 82545EM to `/bin/e1000` and the 82574L to the new `/bin/e1000e`: the
32/// default NICs of VMware guests, and the latter is also q35's default NIC.
33/// Phase 09 (S3) adds VMware's paravirtual vmxnet3 (`/bin/vmxnet3`). PVSCSI
34/// (`15AD:07C0`) is a SCSI controller (class `01:00:00`), not gated here.
35///
36/// The table bounds what the *kernel* offers; each cell names its own IDs
37/// through the vendor:device query (424) and repeats the identity check before
38/// programming a register model, so no cell can bind a foreign family by
39/// accident. Every owner's `identity.rs` lists exactly its rows here.
40pub const ETHERNET_DRIVER_IDS: [(u16, u16, &str); 9] = [
41    (0x8086, 0x100E, "/bin/e1000"),      // 82540EM
42    (0x8086, 0x100F, "/bin/e1000"),      // 82545EM copper (QEMU e1000-82545em, VMware e1000)
43    (0x8086, 0x10D3, "/bin/e1000e"),     // 82574L (QEMU e1000e, q35 default, VMware e1000e)
44    (0x8086, 0x10C9, "/bin/igb"),        // 82576 — QEMU's igb model
45    (0x8086, 0x1533, "/bin/igb"),        // i210 copper with external flash
46    (0x8086, 0x1521, "/bin/igb"),        // I350 copper (server 4-port)
47    (0x1AF4, 0x1041, "/bin/virtio-net"), // virtio-net (modern), ADR-0031 phase 05
48    (0x1AF4, 0x1000, "/bin/virtio-net"), // virtio-net (transitional, modern caps required)
49    (0x15AD, 0x07B0, "/bin/vmxnet3"),    // VMware vmxnet3 (QEMU vmxnet3, VMware default PV NIC)
50];

52// Deliberately refused until the cell implements their prerequisites: flashless
53// i210 (0x157B/0x157C) and i211 (0x1539) need an **iNVM** read path, and
54// fibre/SerDes/SGMII (0x1536/0x1537/0x1538) need media-specific link setup; the
55// 82574LA (0x10F6) and every I219 are `Target` entries of `/bin/e1000e` that no
56// lane validates yet (phase 13). Widening this table without that work would
57// hand a device to a cell that cannot drive it, which is worse than refusing it
58// by name.

60/// Ethernet IDs served by the **legacy class-triple** query (`FindPcieDevice`,
61/// opcode 418).
62///
63/// `02:00:00` is one triple for every Ethernet family, so once a second family
64/// has a Driver Cell the class query can no longer name a device uniquely: it
65/// would hand the igb controller to `/bin/e1000`, which must then decline and
66/// release it before `/bin/igb` can claim it — a scheduling-dependent race. The
67/// fix is [`find_vendor_device`] (opcode 424), which matches exactly. `find_class`
68/// therefore keeps serving only the family that predates the identity query, the
69/// 82540EM, so 418's semantics stay byte-identical; every in-tree Ethernet cell,
70/// `/bin/e1000` included since phase 08 (H-S3-6), uses the vendor:device query.
71/// The family gate above still governs the scan warning and the vendor query.
72pub const ETHERNET_LEGACY_CLASS_IDS: [(u16, u16); 1] = [(0x8086, 0x100E)];

74/// The Driver Cell that owns Ethernet `(vendor_id, device_id)`, if any.
75pub fn ethernet_driver_cell(vendor_id: u16, device_id: u16) -> Option<&'static str> {
76    ETHERNET_DRIVER_IDS
77        .iter()
78        .find(|&&(vendor, device, _)| vendor == vendor_id && device == device_id)
79        .map(|&(_, _, cell)| cell)
80}

101/// Checked byte length for an inclusive MCFG bus range.
102pub fn ecam_window_size(bus_start: u8, bus_end: u8) -> Option<usize> {
103    let buses = (bus_end as usize)
104        .checked_sub(bus_start as usize)?
105        .checked_add(1)?;
106    buses.checked_mul(ECAM_BYTES_PER_BUS)
107}

145/// A Base Address Register decoded from PCI config space.
146#[derive(Clone, Copy, Debug)]
147pub enum Bar {
148    /// 32-bit MMIO BAR: base address and probed size.
149    Memory32 { addr: u32, size: u32 },
150    /// 64-bit MMIO BAR: base address (64-bit) and probed size. Consumes two slots.
151    Memory64 { addr: u64, size: u64 },
152    /// I/O port BAR (not mapped by this driver; skipped).
153    Io,
154    /// Unused / empty BAR slot.
155    None,
156}
157
158impl Bar {
159    /// Physical base address of this BAR, or 0 for I/O / empty.
160    pub fn base_addr(&self) -> u64 {
161        match self {
162            Bar::Memory32 { addr, .. } => *addr as u64,
163            Bar::Memory64 { addr, .. } => *addr,
164            _ => 0,
165        }
166    }
167}

185/// A discovered PCI function (device + function).
186#[derive(Clone, Debug)]
187pub struct PciDevice {
188    /// (bus, device, function).
189    pub bdf: (u8, u8, u8),
190    pub vendor_id: u16,
191    pub device_id: u16,
192    /// Class code (offset 0x0B).
193    pub class: u8,
194    /// Subclass (offset 0x0A).
195    pub subclass: u8,
196    /// Programming interface (offset 0x09).
197    pub prog_if: u8,
198    /// Up to 6 BARs (64-bit BARs consume two slots; slot N+1 is `Bar::None`).
199    pub bars: [Bar; 6],
200    /// MSI-X capability, if present in capability list.
201    pub msix: Option<MsixCap>,
202    /// PM capability, if present.
203    pub pm: Option<PmCap>,
204}

206impl PciDevice {
207    /// Wire BDF: `bus[15:8] | dev[7:3] | fun[2:0]`.
208    pub fn canonical_bdf(&self) -> u32 {
209        (self.bdf.0 as u32) << 8 | (self.bdf.1 as u32) << 3 | self.bdf.2 as u32
210    }
211
212    /// Class code `class << 16 | subclass << 8 | prog_if`.
213    pub fn class_code(&self) -> u32 {
214        (self.class as u32) << 16 | (self.subclass as u32) << 8 | self.prog_if as u32
215    }
216}

245/// Compute an MMIO address relative to an admitted inclusive bus range.
246#[inline(always)]
247fn config_addr_in_range(
248    ecam_base: usize,
249    bus_start: u8,
250    bus_end: u8,
251    bus: u8,
252    dev: u8,
253    fun: u8,
254    off: usize,
255) -> Option<*mut u8> {
256    if bus < bus_start || bus > bus_end || dev >= 32 || fun >= 8 || off >= 0x1000 {
257        return None;
258    }
259    let bus_offset = (bus as usize)
260        .checked_sub(bus_start as usize)?
261        .checked_mul(ECAM_BYTES_PER_BUS)?;
262    let offset = bus_offset
263        .checked_add((dev as usize) << 15)?
264        .checked_add((fun as usize) << 12)?
265        .checked_add(off)?;
266    Some(ecam_base.checked_add(offset)? as *mut u8)
267}

269/// Read vendor/device identity from one checked ECAM function.
270///
271/// # Safety
272/// The admitted ECAM window must be mapped for the kernel and remain live.
273#[allow(dead_code)]
274unsafe fn read_identity_in_range(
275    ecam_base: usize,
276    bus_start: u8,
277    bus_end: u8,
278    bus: u8,
279    dev: u8,
280    fun: u8,
281) -> Option<(u16, u16)> {
282    let ptr = config_addr_in_range(ecam_base, bus_start, bus_end, bus, dev, fun, CFG_VENDOR_ID)?;
283    // SAFETY: the checked function header lies in the caller-provided live ECAM window.
284    let vendor = unsafe { core::ptr::read_volatile(ptr.cast::<u16>()) };
285    // SAFETY: device ID immediately follows vendor ID in the same function header.
286    let device = unsafe { core::ptr::read_volatile(ptr.add(CFG_DEVICE_ID).cast::<u16>()) };
287    Some((vendor, device))
288}

297/// Read a u32 from PCI config space via ECAM MMIO.
298///
299/// # Safety
300/// `config_addr(ecam_base, bus, dev, fun, off)` must point into the ECAM window
301/// as the kernel maps it ([`ecam_window`]).  `off` must be
302/// 4-byte-aligned.
303unsafe fn read32(ecam_base: usize, bus: u8, dev: u8, fun: u8, off: usize) -> u32 {

305    let ptr = config_addr(ecam_base, bus, dev, fun, off) as *const u32;
306    // SAFETY: volatile prevents the compiler from optimising away hardware reads.
307    unsafe { core::ptr::read_volatile(ptr) }
308}

361/// Decode all BARs for a type-0 function.
362///
363/// The write-all-ones / read-back size-probe modifies the command register to
364/// disable memory decode before each probe and restores it after, preventing
365/// the device from responding to MMIO accesses during the transient probe window.
366///
367/// # Safety
368/// `ecam_base` must be the kernel's ECAM window base ([`ecam_window`]).
369unsafe fn decode_bars(ecam_base: usize, bus: u8, dev: u8, fun: u8) -> [Bar; 6] {
370    let mut bars = [Bar::None; 6];
371    let mut i = 0usize;
372    while i < 6 {
373        let off = CFG_BAR0 + i * 4;
374        // SAFETY: ECAM window is mapped for the kernel; off is within type-0 header.
375        let raw = unsafe { read32(ecam_base, bus, dev, fun, off) };
376        if raw & 1 == 1 {
377            // I/O BAR — skip.
378            bars[i] = Bar::Io;
379            i += 1;
380            continue;
381        }
382        let bar_type = (raw >> 1) & 0x3; // bits [2:1]
383        let addr32 = raw & !0xF;
384
385        if bar_type == 0x2 {
386            // 64-bit MMIO: base spans this slot + next.
387            if i + 1 >= 6 {
388                bars[i] = Bar::None;
389                i += 1;
390                continue;
391            }
392            let off_hi = CFG_BAR0 + (i + 1) * 4;
393            // SAFETY: ECAM window is mapped for the kernel.
394            let raw_hi = unsafe { read32(ecam_base, bus, dev, fun, off_hi) };
395            let addr64 = (addr32 as u64) | ((raw_hi as u64) << 32);
396            let size64 = unsafe { probe_bar_size64(ecam_base, bus, dev, fun, i) };
397            bars[i] = Bar::Memory64 {
398                addr: addr64,
399                size: size64,
400            };
401            bars[i + 1] = Bar::None; // second slot consumed
402            i += 2;
403        } else {
404            // 32-bit MMIO.
405            let size32 = unsafe { probe_bar_size32(ecam_base, bus, dev, fun, i) };
406            bars[i] = Bar::Memory32 {
407                addr: addr32,
408                size: size32,
409            };
410            i += 1;
411        }
412    }
413    bars
414}

416/// Probe size of a 32-bit MMIO BAR via the write-all-ones / read-back method.
417///
418/// Saves/restores both the command register (disabling memory decode during the
419/// probe) and the BAR value so the device is left in its original state.
420///
421/// # Safety
422/// `ecam_base` must be the kernel's ECAM window base ([`ecam_window`]).
423unsafe fn probe_bar_size32(ecam_base: usize, bus: u8, dev: u8, fun: u8, bar_idx: usize) -> u32 {
424    let off = CFG_BAR0 + bar_idx * 4;
425    // Save original values.
426    // SAFETY: ECAM window is mapped for the kernel.
427    let orig_cmd = unsafe { read16(ecam_base, bus, dev, fun, CFG_COMMAND) };
428    // SAFETY: ECAM window is mapped for the kernel.
429    let orig_bar = unsafe { read32(ecam_base, bus, dev, fun, off) };
430
431    // Disable memory decode before writing all-ones to BAR.
432    // This prevents the device from claiming MMIO during the probe window.
433    // SAFETY: writing command register disables decode only for this function.
434    unsafe {
435        write16(
436            ecam_base,
437            bus,
438            dev,
439            fun,
440            CFG_COMMAND,
441            orig_cmd & !CMD_MEM_SPACE,
442        );
443    }
444
445    // Write all-ones to BAR.
446    // SAFETY: ECAM window is mapped for the kernel; BAR write is bounded to this function.
447    unsafe {
448        write32(ecam_base, bus, dev, fun, off, 0xFFFF_FFFF);
449    }
450    // SAFETY: ECAM window is mapped for the kernel.
451    let readback = unsafe { read32(ecam_base, bus, dev, fun, off) };
452
453    // Restore BAR and command register.
454    // SAFETY: ECAM window is mapped for the kernel.
455    unsafe {
456        write32(ecam_base, bus, dev, fun, off, orig_bar);
457    }
458    unsafe {
459        write16(ecam_base, bus, dev, fun, CFG_COMMAND, orig_cmd);
460    }
461
462    // Size = ~(readback & mask) + 1 where mask = 0xFFFFFFF0 (clear lower 4 bits).
463    let mask = readback & 0xFFFF_FFF0;
464    if mask == 0 {
465        return 0;
466    }
467    (!mask).wrapping_add(1)
468}

470/// Probe size of a 64-bit MMIO BAR (both low and high dwords).
471///
472/// # Safety
473/// `ecam_base` must be the kernel's ECAM window base ([`ecam_window`]).
474unsafe fn probe_bar_size64(ecam_base: usize, bus: u8, dev: u8, fun: u8, bar_idx: usize) -> u64 {
475    let off_lo = CFG_BAR0 + bar_idx * 4;
476    let off_hi = CFG_BAR0 + (bar_idx + 1) * 4;
477
478    // SAFETY: ECAM window is mapped for the kernel.
479    let orig_cmd = unsafe { read16(ecam_base, bus, dev, fun, CFG_COMMAND) };
480    // SAFETY: ECAM window is mapped for the kernel.
481    let orig_lo = unsafe { read32(ecam_base, bus, dev, fun, off_lo) };
482    // SAFETY: ECAM window is mapped for the kernel.
483    let orig_hi = unsafe { read32(ecam_base, bus, dev, fun, off_hi) };
484
485    // Disable memory decode during probe.
486    // SAFETY: ECAM window is mapped for the kernel.
487    unsafe {
488        write16(
489            ecam_base,
490            bus,
491            dev,
492            fun,
493            CFG_COMMAND,
494            orig_cmd & !CMD_MEM_SPACE,
495        );
496    }
497
498    // Write all-ones to both halves.
499    // SAFETY: ECAM window is mapped for the kernel.
500    unsafe {
501        write32(ecam_base, bus, dev, fun, off_lo, 0xFFFF_FFFF);
502    }
503    unsafe {
504        write32(ecam_base, bus, dev, fun, off_hi, 0xFFFF_FFFF);
505    }
506
507    // SAFETY: ECAM window is mapped for the kernel.
508    let rb_lo = unsafe { read32(ecam_base, bus, dev, fun, off_lo) };
509    // SAFETY: ECAM window is mapped for the kernel.
510    let rb_hi = unsafe { read32(ecam_base, bus, dev, fun, off_hi) };
511
512    // Restore.
513    // SAFETY: ECAM window is mapped for the kernel.
514    unsafe {
515        write32(ecam_base, bus, dev, fun, off_lo, orig_lo);
516    }
517    unsafe {
518        write32(ecam_base, bus, dev, fun, off_hi, orig_hi);
519    }
520    unsafe {
521        write16(ecam_base, bus, dev, fun, CFG_COMMAND, orig_cmd);
522    }
523
524    let mask64 = ((rb_hi as u64) << 32) | ((rb_lo & 0xFFFF_FFF0) as u64);
525    if mask64 == 0 {
526        return 0;
527    }
528    (!mask64).wrapping_add(1)
529}

533/// Walk the capability list for a type-0 function; return MSI-X and PM caps.
534///
535/// # Safety
536/// `ecam_base` must be the kernel's ECAM window base ([`ecam_window`]).
537unsafe fn walk_caps(
538    ecam_base: usize,
539    bus: u8,
540    dev: u8,
541    fun: u8,
542) -> (Option<MsixCap>, Option<PmCap>) {
543    let mut msix = None;
544    let mut pm = None;
545
546    // Capability list is present only when status bit 4 is set.
547    // SAFETY: ECAM window is mapped for the kernel.
548    let status = unsafe { read16(ecam_base, bus, dev, fun, CFG_STATUS) };
549    if status & (1 << 4) == 0 {
550        return (None, None);
551    }
552
553    // SAFETY: ECAM window is mapped for the kernel.
554    let mut cap_ptr = unsafe { read8(ecam_base, bus, dev, fun, CFG_CAP_PTR) } & 0xFC;
555    let mut budget = 64u8; // guard against malformed circular lists
556
557    while cap_ptr != 0 && budget > 0 {
558        budget -= 1;
559        // SAFETY: ECAM window is mapped for the kernel.
560        let cap_id = unsafe { read8(ecam_base, bus, dev, fun, cap_ptr as usize) };
561        // SAFETY: ECAM window is mapped for the kernel.
562        let next = unsafe { read8(ecam_base, bus, dev, fun, cap_ptr as usize + 1) } & 0xFC;
563
564        match cap_id {
565            CAP_ID_MSIX => {
566                // SAFETY: ECAM window is mapped for the kernel.
567                let msg_ctrl = unsafe { read16(ecam_base, bus, dev, fun, cap_ptr as usize + 2) };
568                msix = Some(MsixCap {
569                    cap_offset: cap_ptr,
570                    msg_ctrl,
571                });
572            }
573            CAP_ID_PM => {
574                pm = Some(PmCap {
575                    cap_offset: cap_ptr,
576                });
577            }
578            _ => {}
579        }
580        cap_ptr = next;
581    }
582
583    (msix, pm)
584}

586/// BARs (type-0 headers only), MSI-X and PM capabilities of one function: the
587/// record the boot scan keeps, so a function the Platform Cell registers later
588/// is recorded exactly like a scanned one.
589///
590/// # Safety
591/// `ecam_base + (config_bus << 20)` must map the function's bus window.
592unsafe fn decode_function(
593    ecam_base: usize,
594    config_bus: u8,
595    dev: u8,
596    fun: u8,
597) -> ([Bar; 6], Option<MsixCap>, Option<PmCap>) {
598    // SAFETY (all reads below): the caller guarantees the bus window is mapped.
599    let hdr = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_HEADER_TYPE) } & 0x7F;
600    // Decode BARs for type-0 headers (endpoints). Skip bridges (type-1).
601    let bars = if hdr == 0 {
602        unsafe { decode_bars(ecam_base, config_bus, dev, fun) }
603    } else {
604        [Bar::None; 6]
605    };
606    let (msix, pm) = unsafe { walk_caps(ecam_base, config_bus, dev, fun) };
607    (bars, msix, pm)
608}

612/// Scan one bus of the ECAM window and populate the global device list.
613///
614/// # Safety
615/// `ecam_base + (config_bus << 20)` must point to the kernel's ECAM
616/// window for `reported_bus`. Nonzero-start allocations use `config_bus = 0`
617/// because their allocation base already denotes the first admitted bus.
618unsafe fn scan(ecam_base: usize, config_bus: u8, reported_bus: u8) {
619    let mut devices = PCI_DEVICES.lock();
620
621    for dev in 0u8..32 {
622        // SAFETY: the selected ECAM bus window is mapped for the kernel.
623        let vendor = unsafe { read16(ecam_base, config_bus, dev, 0, CFG_VENDOR_ID) };
624        if vendor == 0xFFFF {
625            continue;
626        }
627
628        // SAFETY: the selected ECAM bus window is mapped for the kernel.
629        let hdr_type = unsafe { read8(ecam_base, config_bus, dev, 0, CFG_HEADER_TYPE) };
630        let max_fun: u8 = if hdr_type & 0x80 != 0 { 8 } else { 1 };
631
632        for fun in 0u8..max_fun {
633            // SAFETY: the selected ECAM bus window is mapped for the kernel.
634            let vid = unsafe { read16(ecam_base, config_bus, dev, fun, CFG_VENDOR_ID) };
635            if vid == 0xFFFF {
636                continue;
637            }
638
639            // SAFETY: the selected ECAM bus window is mapped for the kernel.
640            let did = unsafe { read16(ecam_base, config_bus, dev, fun, CFG_DEVICE_ID) };
641            let prog_if = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_CLASS_PROG) };
642            let subclass = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_SUBCLASS) };
643            let class = unsafe { read8(ecam_base, config_bus, dev, fun, CFG_CLASS_CODE) };
644
645            // SAFETY: the selected ECAM bus window is mapped for the kernel.
646            let (bars, msix, pm) = unsafe { decode_function(ecam_base, config_bus, dev, fun) };
647
648            let bar0_addr = bars[0].base_addr();
649            log::info!(
650                "[pcie] {:02x}:{:02x}.{} vendor={:04x} device={:04x} \
651                 class={:02x}:{:02x}:{:02x} bar0={:#x}",
652                reported_bus,
653                dev,
654                fun,
655                vid,
656                did,
657                class,
658                subclass,
659                prog_if,
660                bar0_addr
661            );
662            if class == 0x02
663                && subclass == 0x00
664                && prog_if == 0x00
665                && !ethernet_id_has_driver(vid, did)
666            {
667                log::warn!(
668                    "[e1000] unsupported Ethernet {:04x}:{:04x}; driver gate closed",
669                    vid,
670                    did
671                );
672            }
673
674            #[cfg(feature = "p01-trace")]
675            let _semantic = crate::memory::heap::p01_owner::creation(
676                None,
677                crate::memory::heap::p01_owner::DRIVER_SHARED,
678            );
679            devices.push(PciDevice {
680                bdf: (reported_bus, dev, fun),
681                vendor_id: vid,
682                device_id: did,
683                class,
684                subclass,
685                prog_if,
686                bars,
687                msix,
688                pm,
689            });
690        }
691    }
692}

694/// The admitted ECAM window as the kernel reaches it: `(base, first bus, last
695/// bus)`; base 0 = none. Every kernel config-space access goes through this
696/// base.
697///
698/// On x86_64 the base is the kernel's own alias of the MCFG allocation
699/// (`paging::mmio_alias::ecam_kernel_alias`), never its identity window: the Platform
700/// Cell's claim turns those identity leaves USER, and under SMAP a supervisor
701/// access through a USER leaf faults.
702///
703/// Only a window the kernel maps and knows to be backed counts. AArch64 has none:
704/// QEMU `virt` moves ECAM above 4 GiB whenever highmem ECAM is on (the default for
705/// 64-bit CPUs), and the kernel neither reads the DTB `pcie` node nor scans there,
706/// so a config access at the board profile's low-ECAM guess (`0x3F00_0000`) hits
707/// nothing and takes a synchronous external abort.
708fn ecam_window() -> (usize, u8, u8) {
709    #[cfg(target_arch = "x86_64")]
710    let window = {
711        let (start, end) = ecam_bus_range_x86();
712        (
713            crate::memory::paging::mmio_alias::ecam_kernel_alias(),
714            start,
715            end,
716        )
717    };
718    #[cfg(target_arch = "riscv64")]
719    let window = (ECAM_BASE_RISCV, 0u8, 0u8);
720    #[cfg(not(any(target_arch = "x86_64", target_arch = "riscv64")))]
721    let window = (0usize, 0u8, 0u8);
722    window
723}

725/// Whether the kernel has a config-space window to quiesce or enable functions
726/// through. Without one, [`quiesce_function`] fails closed for every BDF.
727pub fn config_space_admitted() -> bool {
728    ecam_window().0 != 0
729}

731/// The admitted ECAM window as a physical byte range `(base, len)`, or `None`
732/// when the kernel has none. This is the only physical range the Platform
733/// Cell's `RequestMmio` bypass may claim.
734pub fn admitted_ecam_range() -> Option<(usize, usize)> {
735    let (kernel_base, bus_start, bus_end) = ecam_window();
736    if kernel_base == 0 {
737        return None;
738    }
739    // x86_64 reaches the window through an alias; the claim names the MCFG base.
740    #[cfg(target_arch = "x86_64")]
741    let base = ecam_base_x86();
742    #[cfg(not(target_arch = "x86_64"))]
743    let base = kernel_base;
744    let len = ecam_window_size(bus_start, bus_end)?;
745    base.checked_add(len).map(|_| (base, len))
746}

749    (
750        ((bdf >> 8) & 0xFF) as u8,
751        ((bdf >> 3) & 0x1F) as u8,
752        (bdf & 0x07) as u8,

756/// Whether canonical `bdf` names a function that answers in the admitted
757/// ECAM window (its vendor ID is not `0xFFFF`), the check 236 makes before it
758/// records one: the gate of a Platform Cell BAR registration (235) for a
759/// function without a device record. Without a window none answers.
760pub fn function_answers(bdf: u32) -> bool {
761    // SAFETY: the admitted ECAM window is mapped for the kernel for its life.
762    unsafe { answers_in_window(bdf, ecam_window()) }
763}

765/// [`function_answers`] against the window `(base, bus_start, bus_end)`.
766///
767/// # Safety
768/// A nonzero `base` must map config space for buses `bus_start..=bus_end`.
769unsafe fn answers_in_window(bdf: u32, (base, bus_start, bus_end): (usize, u8, u8)) -> bool {
770    if base == 0 || bdf > 0xFFFF {
771        return false;
772    }
773    let (bus, dev, fun) = split_bdf(bdf);
774    // SAFETY: the caller's window contract; the read is bounded to that window.
775    unsafe { read_identity_in_range(base, bus_start, bus_end, bus, dev, fun) }
776        .is_some_and(|(vendor, _)| vendor != 0xFFFF)
777}

779/// Enable MMIO decoding and bus mastering after the caller has established
780/// exclusive BDF ownership and an IOMMU mapping. Also clears the INTx Disable
781/// bit that [`quiesce_function`] set when a previous owner retired, so a new
782/// owner receives the function in the state it had before any quiescence.
783pub fn enable_bus_master(bdf: u32) {
784    let (bus, dev, fun) = split_bdf(bdf);
785    let (ecam_base, bus_start, bus_end) = ecam_window();
786    if ecam_base == 0 {
787        return;
788    }
789    let Some(command_ptr) =
790        config_addr_in_range(ecam_base, bus_start, bus_end, bus, dev, fun, CFG_COMMAND)
791    else {
792        log::warn!(
793            "[pcie] refusing bus-master enable outside ECAM range: {:02x}:{:02x}.{}",
794            bus,
795            dev,
796            fun
797        );
798        return;
799    };
800
801    // SAFETY: the complete admitted ECAM window is mapped for the kernel
802    // ([`ecam_window`]) before Driver Cells run, and command_ptr was checked
803    // against that exact range.
804    unsafe {
805        let command = core::ptr::read_volatile(command_ptr as *const u16);
806        core::ptr::write_volatile(
807            command_ptr as *mut u16,
808            (command | CMD_MEM_SPACE | CMD_BUS_MASTER) & !CMD_INTX_DISABLE,
809        );
810    }
811}

813/// Command register once a Driver Cell owns the function: memory decoding on,
814/// every other bit as it was. Bus mastering stays off until `GrantDma`
815/// ([`enable_bus_master`]), so a device the cell has not reset yet cannot DMA.
816const fn decoding_command(command: u16) -> u16 {
817    command | CMD_MEM_SPACE
818}

820/// Turn on memory decoding (only) for a function a Driver Cell has just
821/// claimed, so its first register access does not depend on firmware having
822/// enabled decoding. VMware's EFI leaves it off on a function it has no driver
823/// for (vmxnet3), and BAR reads then return all-ones.
824pub fn enable_memory_decode(bdf: u32) {
825    let (ecam_base, bus_start, bus_end) = ecam_window();
826    if ecam_base == 0 {
827        return;
828    }
829    // SAFETY: the admitted ECAM window is mapped for the kernel for its life.
830    if !unsafe { enable_memory_decode_in_window(ecam_base, bus_start, bus_end, bdf) } {
831        let (bus, dev, fun) = split_bdf(bdf);
832        log::warn!(
833            "[pcie] refusing memory-decode enable outside ECAM range: {bus:02x}:{dev:02x}.{fun}"
834        );
835    }
836}

838/// [`enable_memory_decode`] against an explicit ECAM window; `false` when the
839/// BDF lies outside it. An absent function (all-ones command) is left alone.
840///
841/// # Safety
842/// `ecam_base` must map config space for buses `bus_start..=bus_end`.
843unsafe fn enable_memory_decode_in_window(
844    ecam_base: usize,
845    bus_start: u8,
846    bus_end: u8,
847    bdf: u32,
848) -> bool {
849    if bdf > 0xFFFF {
850        return false;
851    }
852    let (bus, dev, fun) = split_bdf(bdf);
853    let Some(ptr) = config_addr_in_range(ecam_base, bus_start, bus_end, bus, dev, fun, CFG_COMMAND)
854    else {
855        return false;
856    };
857    // SAFETY: `ptr` was range-checked against the caller's window.
858    unsafe {
859        let command = core::ptr::read_volatile(ptr as *const u16);
860        if command != 0xFFFF && command & CMD_MEM_SPACE == 0 {
861            core::ptr::write_volatile(ptr as *mut u16, decoding_command(command));
862        }
863    }
864    true
865}

869/// Command register after quiescence: no new DMA, no INTx. Memory decoding is
870/// kept so a later reset (or the next driver) can still reach the BARs.
871const fn quiesced_command(command: u16) -> u16 {
872    (command & !CMD_BUS_MASTER) | CMD_INTX_DISABLE
873}
874
875/// MSI Message Control after quiescence: MSI Enable (bit 0) cleared.
876const fn quiesced_msi_control(control: u16) -> u16 {
877    control & !MSI_ENABLE
878}
879
880/// MSI-X Message Control after quiescence: MSI-X Enable (bit 15) cleared and
881/// Function Mask (bit 14) set, so no vector can fire even if re-enabled.
882const fn quiesced_msix_control(control: u16) -> u16 {
883    (control & !MSIX_ENABLE) | MSIX_FUNCTION_MASK
884}

886/// What `quiesce_function` left in config space.
887#[derive(Debug, Clone, Copy, PartialEq, Eq)]
888pub struct Quiesced {
889    /// Command register as read back.
890    pub command: u16,
891    /// The function has an MSI / MSI-X capability (now disabled).
892    pub msi: bool,
893    pub msix: bool,
894    /// Some register held a live value (Bus Master, INTx, MSI or MSI-X enabled)
895    /// before this call; `false` on an idempotent repeat.
896    pub changed: bool,
897}
898
899/// Why a function could not be shown quiescent; its owner must keep it.
900#[derive(Debug, Clone, Copy, PartialEq, Eq)]
901pub enum QuiesceError {
902    /// No ECAM window, or the BDF lies outside it.
903    OutOfRange,
904    /// A config write did not read back as written.
905    Readback { offset: u16, wrote: u16, read: u16 },
906}
907
908/// Stop a function a dead driver owned from starting DMA or raising
909/// interrupts: clear Bus Master, set INTx Disable, and turn off MSI and MSI-X
910/// (masking MSI-X), each confirmed by readback.
911///
912/// The capability list is walked here, bounded by the same 64-entry budget as
913/// the scan, through range-checked config addresses; no cached or
914/// driver-supplied offset is used.
915///
916/// Clearing Bus Master stops new requests but does not prove that writes the
917/// device already posted have drained. The caller still needs the IOMMU fence
918/// before it releases any frame.
919pub fn quiesce_function(bdf: u32) -> Result<Quiesced, QuiesceError> {
920    let (ecam_base, bus_start, bus_end) = ecam_window();
921    if ecam_base == 0 {
922        return Err(QuiesceError::OutOfRange);
923    }
924    // SAFETY: the admitted ECAM window is mapped for the kernel for its life.
925    unsafe { quiesce_in_window(ecam_base, bus_start, bus_end, bdf) }
926}
927
928/// [`quiesce_function`] against an explicit ECAM window.
929///
930/// # Safety
931/// `ecam_base` must map config space for buses `bus_start..=bus_end`.
932unsafe fn quiesce_in_window(
933    ecam_base: usize,
934    bus_start: u8,
935    bus_end: u8,
936    bdf: u32,
937) -> Result<Quiesced, QuiesceError> {
938    // A BDF is 16 bits; anything above would alias another function.
939    if bdf > 0xFFFF {
940        return Err(QuiesceError::OutOfRange);
941    }
942    let (bus, dev, fun) = split_bdf(bdf);
943    let addr = |off: usize| {
944        config_addr_in_range(ecam_base, bus_start, bus_end, bus, dev, fun, off)
945            .ok_or(QuiesceError::OutOfRange)
946    };
947    // SAFETY (all accesses below): every pointer comes from
948    // `config_addr_in_range` for the caller's window.
949    let read16 =
950        |off: usize| addr(off).map(|p| unsafe { core::ptr::read_volatile(p as *const u16) });
951    let read8 = |off: usize| addr(off).map(|p| unsafe { core::ptr::read_volatile(p as *const u8) });
952    let changed = core::cell::Cell::new(false);
953    let update = |off: usize, next: fn(u16) -> u16| -> Result<u16, QuiesceError> {
954        let ptr = addr(off)? as *mut u16;
955        let before = unsafe { core::ptr::read_volatile(ptr) };
956        let wrote = next(before);
957        if wrote != before {
958            changed.set(true);
959            unsafe { core::ptr::write_volatile(ptr, wrote) };
960        }
961        let read = unsafe { core::ptr::read_volatile(ptr) };
962        if read != wrote {
963            return Err(QuiesceError::Readback {
964                offset: off as u16,
965                wrote,
966                read,
967            });
968        }
969        Ok(read)
970    };
971
972    // No function answers here (all-ones vendor): nothing can master the bus
973    // or interrupt, and config writes would be dropped.
974    if read16(CFG_VENDOR_ID)? == 0xFFFF {
975        return Ok(Quiesced {
976            command: 0xFFFF,
977            msi: false,
978            msix: false,
979            changed: false,
980        });
981    }
982
983    // Function first: once Bus Master is off the device cannot start new DMA,
984    // whatever the interrupt capabilities below turn out to hold.
985    let command = update(CFG_COMMAND, quiesced_command)?;
986    let mut quiesced = Quiesced {
987        command,
988        msi: false,
989        msix: false,
990        changed: false,
991    };
992
993    if read16(CFG_STATUS)? & STATUS_CAP_LIST != 0 {
994        let mut cap = read8(CFG_CAP_PTR)? & 0xFC;
995        let mut budget = 64u8;
996        while cap != 0 && budget > 0 {
997            budget -= 1;
998            let at = cap as usize;
999            match read8(at)? {
1000                CAP_ID_MSI => {
1001                    quiesced.msi = true;
1002                    update(at + 2, quiesced_msi_control)?;
1003                }
1004                CAP_ID_MSIX => {
1005                    quiesced.msix = true;
1006                    update(at + 2, quiesced_msix_control)?;
1007                }
1008                _ => {}
1009            }
1010            cap = read8(at + 1)? & 0xFC;
1011        }
1012    }
1013    quiesced.changed = changed.get();
1014    Ok(quiesced)
1015}

1036/// Initialise the ECAM scanner.
1037///
1038/// Must be called after paging is active (ECAM window mapped) and before
1039/// Driver Cells are spawned (they call sys_find_pcie_device to discover their
1040/// endpoint). Idempotent; safe to call more than once (later calls are no-ops).
1041pub fn init() {
1042    // The admitted window as the kernel reaches it (`ecam_window`: the alias
1043    // of the validated ACPI MCFG window on x86_64, whose base denotes its
1044    // first bus; the identity window on RV64). AArch64 admits no window there
1045    // but scans the FDT-recorded base. The log shows the kernel's address.
1046    #[cfg(target_arch = "aarch64")]
1047    let (ecam_base, scan_bus) = (ecam_base_aarch64(), 0u8);
1048    #[cfg(not(target_arch = "aarch64"))]
1049    let (ecam_base, scan_bus, _) = ecam_window();
1050
1051    if ecam_base == 0 {
1052        log::info!("[pcie] ECAM: no PCIe on this architecture");
1053        return;
1054    }
1055
1056    // If already populated (e.g. called twice), skip re-scan.
1057    if !PCI_DEVICES.lock().is_empty() {
1058        log::info!("[pcie] ECAM: already scanned, skipping");
1059        return;
1060    }
1061    log::info!(
1062        "[pcie] ECAM fallback scan bus {} @ {:#x} (1 MiB window)",
1063        scan_bus,
1064        ecam_base
1065    );
1066
1067    // SAFETY: the first admitted ECAM bus window is mapped for the kernel
1068    // (identity on RV64, the kernel alias on x86_64) before this function is
1069    // called. The MCFG allocation base denotes `scan_bus`, so config-space
1070    // access is relative to local bus offset zero.
1071    unsafe {
1072        scan(ecam_base, 0, scan_bus);
1073    }
1074
1075    let count = PCI_DEVICES.lock().len();
1076    if count == 0 {
1077        log::warn!(
1078            "[pcie] ECAM scan found no devices on bus {} — check base and MMIO mapping",
1079            scan_bus
1080        );
1081    } else {
1082        log::info!("[pcie] ECAM scan complete: {} device(s) found", count);
1083    }
1084
1085    // Record every discovered function and its BAR windows, so a Driver Cell
1086    // that claims a function can call sys_request_mmio on its BARs.
1087    for dev in PCI_DEVICES.lock().iter() {
1088        let bdf = dev.canonical_bdf();
1089        crate::dev::ownership::register_pci_function(bdf, dev.class_code());
1090        for bar in &dev.bars {
1091            let base = bar.base_addr() as usize;
1092            if base == 0 {
1093                continue;
1094            }
1095            // Probe size is stored in Bar::Memory32/Memory64; use a safe default
1096            // of 16 KiB if the probed size is 0 (identity entry).
1097            let len = match bar {
1098                Bar::Memory32 { size, .. } => {
1099                    if *size == 0 {
1100                        0x4000
1101                    } else {
1102                        *size as usize
1103                    }
1104                }
1105                Bar::Memory64 { size, .. } => {
1106                    if *size == 0 {
1107                        0x4000
1108                    } else {
1109                        *size as usize
1110                    }
1111                }
1112                _ => continue,
1113            };
1114            crate::dev::ownership::register_pci_bar(bdf, base, len);
1115        }
1116    }
1117}

1119/// What [`register_device`] (syscall 236) did with a Platform Cell registration.
1120#[cfg_attr(not(target_arch = "x86_64"), allow(dead_code))]
1121#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1122pub enum Registration {
1123    /// The function is in `PCI_DEVICES`: recorded now, or earlier by the boot
1124    /// scan or a previous registration.
1125    Recorded,
1126    /// Recorded exactly like [`Registration::Recorded`], so the machine
1127    /// fingerprint (433) sees it as it sees a bus-0 function, but it is an
1128    /// Ethernet controller (`02:00:00`) no Driver Cell drives
1129    /// ([`ethernet_id_has_driver`]): the handout queries ([`find_class_where`],
1130    /// [`find_vendor_device`]) never offer it.
1131    RecordedWithoutDriver,
1132    /// No function answers at the BDF inside the admitted window (or there is
1133    /// no window); nothing is recorded.
1134    Absent,
1135}

1137/// Register a device announced by the Platform Cell via `sys_register_pci_device`.
1138///
1139/// On x86 the kernel records the function from config space, exactly as the
1140/// boot scan records a bus-0 function: identity, class code, every BAR and its
1141/// capabilities are re-read by BDF through the admitted ECAM window, so `cls`
1142/// is only compared (a mismatch is logged) and never stored. A function behind
1143/// a bridge (bus ≥ 1) therefore keeps a BAR other than BAR0 — VMware's and
1144/// ICH9's AHCI decode their ABAR at BAR5. Every function is recorded, an
1145/// Ethernet ID without a Driver Cell included (that one reports
1146/// [`Registration::RecordedWithoutDriver`]; the queries are the only gate).
1147///
1148/// The duplicate check, the BAR size probe and the insert run under one
1149/// `PCI_DEVICES` guard, so two registrations of one BDF cannot both probe it or
1150/// both insert it; a BDF already present (the kernel ECAM scan may have found
1151/// it) is not probed again. Other architectures keep the Platform Cell's class
1152/// and BAR0. A new record also records the function's class and BAR0 in its
1153/// ownership record ([`crate::dev::ownership`]) for `sys_request_mmio`.
1154#[must_use]
1155pub fn register_device(bdf: u32, cls: u32, bar0_base: usize, bar0_size: usize) -> Registration {
1156    // Canonical BDF wire layout everywhere (Platform Cell, FindPcieDevice,
1157    // GrantDma, iommu_x86): bus[15:8] | dev[7:3] | fun[2:0].
1158    let (bus, dev, fun) = split_bdf(bdf);
1159
1160    #[cfg(target_arch = "x86_64")]
1161    let (registration, added) = {
1162        let (ecam_base, bus_start, bus_end) = ecam_window();
1163        if ecam_base == 0 {
1164            return Registration::Absent;
1165        }
1166        let mut devices = PCI_DEVICES.lock();
1167        // SAFETY: x86 paging maps the validated MCFG allocation before the
1168        // Platform Cell can issue registration syscalls.
1169        unsafe {
1170            record_in_window(
1171                ecam_base,
1172                bus_start,
1173                bus_end,
1174                (bus, dev, fun),
1175                cls,
1176                &mut devices,
1177            )
1178        }
1179    };
1180
1181    #[cfg(not(target_arch = "x86_64"))]
1182    let (registration, added) = {
1183        let mut devices = PCI_DEVICES.lock();
1184        let added = !devices.iter().any(|d| d.bdf == (bus, dev, fun));
1185        if added {
1186            let (class, subclass, prog_if) = ((cls >> 16) as u8, (cls >> 8) as u8, cls as u8);
1187            let bars = platform_bar0(bar0_base, bar0_size);
1188            log::info!(
1189                "[pcie] platform-reg {:02x}:{:02x}.{} class={:02x}:{:02x}:{:02x} bar0={:#x}",
1190                bus,
1191                dev,
1192                fun,
1193                class,
1194                subclass,
1195                prog_if,
1196                bars[0].base_addr()
1197            );
1198            #[cfg(feature = "p01-trace")]
1199            let _semantic = crate::memory::heap::p01_owner::creation(
1200                None,
1201                crate::memory::heap::p01_owner::DRIVER_SHARED,
1202            );
1203            devices.push(PciDevice {
1204                bdf: (bus, dev, fun),
1205                vendor_id: 0,
1206                device_id: 0,
1207                class,
1208                subclass,
1209                prog_if,
1210                bars,
1211                msix: None,
1212                pm: None,
1213            });
1214        }
1215        (Registration::Recorded, added)
1216    };
1217
1218    if added {
1219        if let Some(dev) = device(bdf) {
1220            crate::dev::ownership::register_pci_function(bdf, dev.class_code());
1221        }
1222        if bar0_base != 0 && bar0_size != 0 {
1223            crate::dev::ownership::register_pci_bar(bdf, bar0_base, bar0_size);
1224        }
1225    }
1226    registration
1227}

1229/// The Platform Cell's BAR0 as a BAR set: the record of a registration on an
1230/// architecture where the kernel does not re-read the function.
1231#[cfg(any(not(target_arch = "x86_64"), test))]
1232fn platform_bar0(bar0_base: usize, bar0_size: usize) -> [Bar; 6] {
1233    let mut bars = [Bar::None; 6];
1234    if bar0_base != 0 {
1235        bars[0] = if bar0_size <= u32::MAX as usize {
1236            Bar::Memory32 {
1237                addr: bar0_base as u32,
1238                size: bar0_size as u32,
1239            }
1240        } else {
1241            Bar::Memory64 {
1242                addr: bar0_base as u64,
1243                size: bar0_size as u64,
1244            }
1245        };
1246    }
1247    bars
1248}

1250/// `(base, size)` of the first memory BAR with a non-zero base. Empty BARs
1251/// decode as `Memory32 { addr: 0, .. }`, so the base must be checked or the
1252/// first absent slot wins.
1253pub fn first_memory_bar(bars: &[Bar; 6]) -> Option<(u64, u64)> {
1254    bars.iter().find_map(|bar| match *bar {
1255        Bar::Memory32 { addr, size } if addr != 0 => Some((addr as u64, size as u64)),
1256        Bar::Memory64 { addr, size } if addr != 0 => Some((addr, size)),
1257        _ => None,
1258    })

1262/// one memory BAR, and none with a size but base 0. Memory decoding is turned
1263/// on at claim time only then; an unassigned BAR would decode at address 0.

1284/// Find the first supported device matching (class, subclass, prog_if).
1285///
1286/// Example: NVMe = `find_class(0x01, 0x08, 0x02)`.
1287///
1288/// The Ethernet class triple is one triple for every NIC family, so on x86 it is
1289/// offered only for the family that still owns it (see
1290/// [`ETHERNET_LEGACY_CLASS_IDS`]); a NIC cell whose controller shares the triple
1291/// uses [`find_vendor_device`] instead.
1292pub fn find_class(class: u8, subclass: u8, prog_if: u8) -> Option<PciDevice> {
1293    find_class_where(class, subclass, prog_if, |_| true)
1294}
1295
1296/// [`find_class`], skipping every match whose canonical BDF fails `usable`.
1297///
1298/// Syscall 418 passes "unowned or owned by the caller" (ADR-0031 CP1 §6), so a
1299/// second instance of one driver reaches the next identical function. `usable`
1300/// runs after the device list lock is released.
1301pub fn find_class_where(
1302    class: u8,
1303    subclass: u8,
1304    prog_if: u8,
1305    usable: impl Fn(u32) -> bool,
1306) -> Option<PciDevice> {
1307    let matches: Vec<PciDevice> = PCI_DEVICES
1308        .lock()
1309        .iter()
1310        .filter(|d| {
1311            if d.class != class || d.subclass != subclass || d.prog_if != prog_if {
1312                return false;
1313            }
1314            // Ethernet spans several Driver Cells (/bin/e1000, /bin/e1000e,
1315            // /bin/igb, /bin/virtio-net), so the class triple cannot name a
1316            // device uniquely. It is served only for the legacy family; every
1317            // cell claims its own controller by exact vendor:device
1318            // (`find_vendor_device`). Binding an endpoint by generic Ethernet
1319            // class would program an incompatible register model, so an ID with
1320            // no cell stays refused here and is named in `scan`'s warning.
1321            #[cfg(target_arch = "x86_64")]
1322            if class == 0x02 && subclass == 0x00 && prog_if == 0x00 {
1323                return ethernet_class_query_serves(d.vendor_id, d.device_id);
1324            }
1325            true
1326        })
1327        .cloned()
1328        .collect();
1329    matches.into_iter().find(|d| usable(d.canonical_bdf()))
1330}

1332/// Find the first device with an exact `(vendor_id, device_id)` whose canonical
1333/// BDF passes `usable` (syscall 424 passes "unowned or owned by the caller",
1334/// ADR-0031 CP1 §6; `usable` runs after the device list lock is released).
1335///
1336/// This is how a Driver Cell whose controller family shares a class triple with
1337/// another cell (the two Ethernet cells on `02:00:00`) claims *its* device: the
1338/// match names the device, so no sibling has to decline and release it first.
1339/// The Ethernet family gate still applies, so an Ethernet ID with no Driver Cell
1340/// is never offered even when its vendor:device pair is queried verbatim.
1341pub fn find_vendor_device(
1342    vendor_id: u16,
1343    device_id: u16,
1344    usable: impl Fn(u32) -> bool,
1345) -> Option<PciDevice> {
1346    let matches: Vec<PciDevice> = PCI_DEVICES
1347        .lock()
1348        .iter()
1349        .filter(|d| {
1350            if d.vendor_id != vendor_id || d.device_id != device_id {
1351                return false;
1352            }
1353            #[cfg(target_arch = "x86_64")]
1354            if d.class == 0x02 && d.subclass == 0x00 && d.prog_if == 0x00 {
1355                return ethernet_id_has_driver(d.vendor_id, d.device_id);
1356            }
1357            true
1358        })
1359        .cloned()
1360        .collect();
1361    matches.into_iter().find(|d| usable(d.canonical_bdf()))
1362}

1368        .iter()
