2//! Per-device ownership records (architecture remediation Phase 18).
3//!
4//! One [`DeviceRecord`] per device a Driver Cell can own: a PCI function, a
5//! VirtIO MMIO slot, or the DWC2 USB aperture. Each names the device's owner,
6//! its claimable windows, its interrupts and its DMA domain, so MMIO
7//! (`RequestMmio`), IRQ (435, `WaitIrq`, the MSI and VirtIO ISRs), DMA
8//! (`GrantDma`, DMA export) and route (416/417) checks answer from one record,
9//! and one release ([`release_for`]) frees it.
10//!
11//! The owner is the claiming task id, the key every check used before. A dead
12//! owner's records stay owned (quarantined) until its DMA teardown is
13//! acknowledged and its windows are released, so no replacement can claim the
14//! device, its windows or its vectors while hardware may still reach the old
15//! domain.
16//!
17//! The table holds a record for every device the boot scan, the Platform Cell
18//! or a claim names, however many there are: records are appended only at
19//! registration and claim (boot and syscall context, charged to the kernel)
20//! and never removed, so release, teardown (which may run in the timer/IPI
21//! ISR) and every check only touch records in place and never allocate.
22
23use crate::sync::Spinlock;
24use abi::block_route::{DeviceKey, BLOCK_DEVICE_MMIO, BLOCK_DEVICE_PCI};
25use alloc::vec::Vec;
26use core::sync::atomic::{AtomicU64, Ordering};

28/// Windows one record may hold: a PCI function's six BARs.
29const MAX_BARS: usize = 6;
30
31/// What a record describes.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33#[cfg_attr(
34    not(any(target_arch = "riscv64", target_arch = "aarch64")),
35    allow(dead_code)
36)]
37pub(crate) enum DeviceKind {
38    /// A PCI function; its class code `class << 16 | subclass << 8 | prog_if`
39    /// once the scan or a registration reported it.
40    Pci { class: Option<u32> },
41    /// A VirtIO MMIO transport slot.
42    VirtioSlot,
43    /// The DWC2 USB host aperture (the RPi3 LAN9514 sits behind it).
44    Dwc2,
45}
46
47/// The interrupt lines a record holds.
48#[derive(Clone, Copy, Debug, PartialEq, Eq)]
49pub(crate) enum IrqLines {
50    /// The fixed line of a VirtIO slot or of the DWC2 aperture.
51    #[cfg_attr(
52        not(any(target_arch = "riscv64", target_arch = "aarch64", test)),
53        allow(dead_code)
54    )]
55    Line(u8),
56    /// The MSI/MSI-X vectors 435 allocated to the owner.
57    #[cfg(target_arch = "x86_64")]
58    Msi(abi::pcie_function::PcieIrqAlloc),
59}
60
61impl IrqLines {
62    fn contains(self, irq: u8) -> bool {
63        match self {
64            Self::Line(line) => line == irq,
65            #[cfg(target_arch = "x86_64")]
66            Self::Msi(alloc) => irq >= alloc.first_irq && (irq - alloc.first_irq) < alloc.count,
67        }
68    }
69
70    /// Every line, in ascending order.
71    fn lines(self) -> impl Iterator<Item = u8> {
72        let (first, count) = match self {
73            Self::Line(line) => (line, 1),
74            #[cfg(target_arch = "x86_64")]
75            Self::Msi(alloc) => (alloc.first_irq, alloc.count),
76        };
77        (0..count).map(move |offset| first.wrapping_add(offset))
78    }
79}

81/// One device's ownership record.
82#[derive(Clone, Copy)]
83pub(crate) struct DeviceRecord {
84    key: DeviceKey,
85    kind: DeviceKind,
86    /// The claiming task; `None` when free.
87    owner: Option<usize>,
88    /// Claimable windows `(base, len)`; `len == 0` is an empty slot.
89    bars: [(usize, usize); MAX_BARS],
90    /// The device's interrupts: a VirtIO slot's or the DWC2 line, fixed at
91    /// registration; on x86_64 a PCI function's MSI vectors once 435
92    /// allocated them to its owner.
93    irq: Option<IrqLines>,
94    /// The tid of the NIC route bound to this device, while it has a
95    /// provider ([`set_nic_route`]); its lines are NIC lines while that tid
96    /// also owns the device.
97    nic_route: Option<usize>,
98}
99
100impl DeviceRecord {
101    const fn new(key: DeviceKey, kind: DeviceKind) -> Self {
102        Self {
103            key,
104            kind,
105            owner: None,
106            bars: [(0, 0); MAX_BARS],
107            irq: None,
108            nic_route: None,
109        }
110    }
111
112    fn owned_by(&self, tid: usize) -> bool {
113        self.owner == Some(tid)
114    }
115
116    fn holds_irq(&self, irq: u8) -> bool {
117        self.irq.is_some_and(|lines| lines.contains(irq))
118    }
119
120    /// Whether `[base, base + len)` is one of this record's windows: a BAR or
121    /// VirtIO slot claimed from its exact base, or any window inside the DWC2
122    /// aperture.
123    fn has_window(&self, base: usize, len: usize) -> bool {
124        len != 0
125            && self.bars.iter().any(|&(bar, bar_len)| {
126                bar_len != 0
127                    && match self.kind {
128                        DeviceKind::Dwc2 => {
129                            base >= bar
130                                && base
131                                    .checked_add(len)
132                                    .is_some_and(|end| end <= bar + bar_len)
133                        }
134                        _ => base == bar && len <= bar_len,
135                    }
136            })
137    }
138
139    fn add_window(&mut self, base: usize, len: usize) -> bool {
140        if self.bars.contains(&(base, len)) {
141            return true;
142        }
143        match self.bars.iter_mut().find(|bar| bar.1 == 0) {
144            Some(slot) => {
145                *slot = (base, len);
146                true
147            }
148            None => false,
149        }
150    }
151}

153struct Devices {
154    /// Append-only: a record's index never changes and none is removed.
155    records: Vec<DeviceRecord>,
156}
157
158impl Devices {
159    fn find(&mut self, key: DeviceKey) -> Option<&mut DeviceRecord> {
160        self.records.iter_mut().find(|record| record.key == key)
161    }
162
163    /// The record of `key`, appended as `kind` when absent. Boot or syscall
164    /// context only: the append may grow the table, charged to the kernel.
165    /// `None` only when the heap refuses that growth.
166    fn find_or_insert(&mut self, key: DeviceKey, kind: DeviceKind) -> Option<&mut DeviceRecord> {
167        let index = match self.records.iter().position(|record| record.key == key) {
168            Some(index) => index,
169            None => {
170                let _kernel = crate::memory::heap::ChargeScope::enter(
171                    crate::memory::cell_quota::Charge::KERNEL,
172                );
173                #[cfg(feature = "p01-trace")]
174                let _semantic = crate::memory::heap::p01_owner::creation(
175                    None,
176                    crate::memory::heap::p01_owner::DRIVER_SHARED,
177                );
178                self.records.try_reserve(1).ok()?;
179                self.records.push(DeviceRecord::new(key, kind));
180                self.records.len() - 1
181            }
182        };
183        self.records.get_mut(index)
184    }
185
186    /// Rewrite [`NIC_LINES`] from the records: every line of a device whose
187    /// owner is the provider of its NIC route. Called, under `DEVICES`, by
188    /// every change of an owner, a line or a NIC route provider, so the
189    /// words always describe the records. Never allocates (teardown may run
190    /// in an ISR).
191    fn publish_nic_lines(&self) {
192        let mut words = [0u64; NIC_LINE_WORDS];
193        for record in &self.records {
194            let Some(lines) = record.irq else {
195                continue;
196            };
197            if record.owner.is_some() && record.owner == record.nic_route {
198                lines
199                    .lines()
200                    .for_each(|line| words[usize::from(line / 64)] |= 1 << (line % 64));
201            }
202        }
203        for (word, bits) in NIC_LINES.iter().zip(words) {
204            word.store(bits, Ordering::Release);
205        }
206    }
207}

209/// Every device record. Lock order: after `SCHEDULER`, the grant tables,
210/// `ROLE_PUBLICATION`, `NIC_ROUTES` and `BLOCK_ROUTES` (route publication and
211/// clearing check ownership under them), before `RESOURCE_REGISTRY` and
212/// `PCIE_MSI_STATE`.
213static DEVICES: Spinlock<Devices> = Spinlock::ranked(
214    Devices {
215        records: Vec::new(),
216    },
217    crate::sync::rank::DEVICES,
218);
219
220const NIC_LINE_WORDS: usize = 256 / 64;
221
222/// The interrupt lines (x86_64 MSI vectors, riscv64/aarch64 VirtIO lines) of
223/// the devices behind a live NIC route, one bit per `u8` line: written only
224/// by [`Devices::publish_nic_lines`] under `DEVICES`, read lock-free by the
225/// MSI and VirtIO ISRs ([`is_nic_line`]).
226static NIC_LINES: [AtomicU64; NIC_LINE_WORDS] = [const { AtomicU64::new(0) }; NIC_LINE_WORDS];
227
228fn pci_key(bdf: u32) -> Option<DeviceKey> {
229    DeviceKey::new(BLOCK_DEVICE_PCI, bdf as u64)
230}
231
232fn mmio_key(base: usize) -> Option<DeviceKey> {
233    DeviceKey::new(BLOCK_DEVICE_MMIO, base as u64)
234}

236/// Logged, after `DEVICES` is released, when the heap refused `key` a record.
237fn no_record(key: DeviceKey) {
238    log::warn!(
239        "[devices] out of memory; {} has no ownership record",
240        crate::dev::driver_cell::DeviceName(key)
241    );
242}

244/// Record PCI function `bdf` with its class code (boot scan, 236).
245pub(crate) fn register_pci_function(bdf: u32, class: u32) {
246    let Some(key) = pci_key(bdf) else {
247        return;
248    };
249    let kind = DeviceKind::Pci { class: Some(class) };
250    let recorded = DEVICES
251        .lock()
252        .find_or_insert(key, kind)
253        .map(|record| record.kind = kind)
254        .is_some();
255    if !recorded {
256        no_record(key);
257    }
258}

260/// Add BAR window `[base, base + len)` to PCI function `bdf`'s record (boot
261/// scan, 236; 235 through [`register_platform_bar`]). Only a bounded
262/// conventional BAR is admitted
263/// ([`valid_pcie_bar_window`](crate::resource_registry::valid_pcie_bar_window)).
264pub(crate) fn register_pci_bar(bdf: u32, base: usize, len: usize) {
265    if !crate::resource_registry::valid_pcie_bar_window(base, len) {
266        log::warn!(
267            "[pcie] rejected invalid BAR window base={:#x} len={:#x}",
268            base,
269            len
270        );
271        return;
272    }
273    let Some(key) = pci_key(bdf) else {
274        log::warn!("[pcie] BAR {base:#x}+{len:#x} names no function (bdf={bdf:#x}); not recorded");
275        return;
276    };
277    let added = DEVICES
278        .lock()
279        .find_or_insert(key, DeviceKind::Pci { class: None })
280        .map(|record| record.add_window(base, len));
281    match added {
282        Some(true) => {}
283        Some(false) => log::warn!(
284            "[pcie] {}: no BAR slot left for {base:#x}+{len:#x}",
285            crate::dev::driver_cell::DeviceName(key)
286        ),
287        None => no_record(key),
288    }
289}

291/// A Platform Cell BAR registration (235): the window goes to `bdf`'s record
292/// when it has one, and creates a record only when `answers()` reports that
293/// the function answers in config space, so records track real functions.
294/// `false`, with nothing recorded, otherwise. Records are never removed, so a
295/// record found here is still there when the window is added.
296pub(crate) fn register_platform_bar(
297    bdf: u32,
298    base: usize,
299    len: usize,
300    answers: impl FnOnce() -> bool,
301) -> bool {
302    let Some(key) = pci_key(bdf) else {
303        return false;
304    };
305    let known = DEVICES.lock().find(key).is_some();
306    if !known && !answers() {
307        return false;
308    }
309    register_pci_bar(bdf, base, len);
310    true
311}

313/// Record the MMIO device at `base` (a VirtIO slot or the DWC2 aperture)
314/// with its claimable window and its fixed interrupt line, if it has one.
315#[cfg_attr(
316    not(any(target_arch = "riscv64", target_arch = "aarch64")),
317    allow(dead_code)
318)]
319pub(crate) fn register_mmio_device(base: usize, len: usize, kind: DeviceKind, line: Option<u8>) {
320    let Some(key) = mmio_key(base) else {
321        return;
322    };
323    let recorded = {
324        let mut devices = DEVICES.lock();
325        let recorded = devices
326            .find_or_insert(key, kind)
327            .map(|record| {
328                record.add_window(base, len);
329                record.irq = line.map(IrqLines::Line);
330            })
331            .is_some();
332        devices.publish_nic_lines();
333        recorded
334    };
335    if !recorded {
336        no_record(key);
337    }
338}

340/// Claim PCI function `bdf` for `tid` (the 418/424 handout). Repeated claims
341/// by the owner are idempotent; a live or quarantined owner is never
342/// displaced. A function without a record gets one.
343pub(crate) fn claim_pci(bdf: u32, tid: usize) -> bool {
344    let Some(key) = pci_key(bdf) else {
345        return false;
346    };
347    let claimed = {
348        let mut devices = DEVICES.lock();
349        let claimed = devices
350            .find_or_insert(key, DeviceKind::Pci { class: None })
351            .map(|record| match record.owner {
352                Some(owner) => owner == tid,
353                None => {
354                    record.owner = Some(tid);
355                    true
356                }
357            });
358        devices.publish_nic_lines();
359        claimed
360    };
361    claimed.unwrap_or_else(|| {
362        no_record(key);
363        false
364    })
365}

410/// What a `RequestMmio` claim found in the device records.
411#[derive(Clone, Copy, Debug, PartialEq, Eq)]
412pub(crate) enum WindowClaim {
413    /// No record of the path holds the window.
414    NoDevice,
415    /// The caller already owns the window's device.
416    Owned,
417    /// The caller now owns the window's MMIO device (its first claim of a
418    /// VirtIO slot or the DWC2 aperture); [`unclaim`] undoes it when the
419    /// window grant then fails.
420    Claimed(DeviceKey),
421    /// The window is a BAR of a PCI function the caller has not claimed
422    /// through 418/424 (free, or another task's).
423    NotOwner,
424    /// The window's MMIO device is another task's, live or awaiting its
425    /// teardown.
426    Held,
427}
428
429/// Claim the device whose window `[base, base + len)` is, for `tid`, before
430/// the window is granted: a PCI function's BAR only for its owner, a VirtIO
431/// slot (`dwc2 == false`) or the DWC2 aperture (`dwc2 == true`) for its owner
432/// or, when free, for its first claimant.
433pub(crate) fn claim_window(tid: usize, base: usize, len: usize, dwc2: bool) -> WindowClaim {
434    let mut devices = DEVICES.lock();
435    let claim = devices.claim_window(tid, base, len, dwc2);
436    if matches!(claim, WindowClaim::Claimed(_)) {
437        devices.publish_nic_lines();
438    }
439    claim
440}
441
442impl Devices {
443    fn claim_window(&mut self, tid: usize, base: usize, len: usize, dwc2: bool) -> WindowClaim {
444        let mut matching = self.records.iter_mut().filter(|record| {
445            (record.kind == DeviceKind::Dwc2) == dwc2 && record.has_window(base, len)
446        });
447        let Some(record) = matching.next() else {
448            return WindowClaim::NoDevice;
449        };
450        // Two records naming one window (a 235 registration repeating a
451        // scanned BAR under another BDF) answer for whichever the caller owns.
452        if !record.owned_by(tid) && matching.any(|other| other.owned_by(tid)) {
453            return WindowClaim::Owned;
454        }
455        match (record.owner, record.kind) {
456            (Some(owner), _) if owner == tid => WindowClaim::Owned,
457            (_, DeviceKind::Pci { .. }) => WindowClaim::NotOwner,
458            (Some(_), _) => WindowClaim::Held,
459            (None, _) => {
460                record.owner = Some(tid);
461                WindowClaim::Claimed(record.key)
462            }
463        }
464    }
465}

467/// Undo a [`WindowClaim::Claimed`] whose window grant failed: free `key`'s
468/// record if `tid` still owns it.
469pub(crate) fn unclaim(tid: usize, key: DeviceKey) {
470    let mut devices = DEVICES.lock();
471    if let Some(record) = devices.find(key).filter(|record| record.owned_by(tid)) {
472        record.owner = None;
473        devices.publish_nic_lines();
474    }
475}

477/// Which of an owner's records [`release_for`] frees.
478#[derive(Clone, Copy, Debug, PartialEq, Eq)]
479pub(crate) enum Scope {
480    /// Every record: the owner retired, its DMA teardown was acknowledged and
481    /// its windows were released.
482    Retired,
483    /// The records `PcieDriverCap` claims (PCI functions through 418/424,
484    /// VirtIO slots through `RequestMmio`): a runtime revoke of that
485    /// capability, after its DMA teardown and the release of its windows.
486    PcieDriver,
487}

489/// Free the records `tid` owns in `scope`: the one release of device
490/// ownership. A freed record keeps its windows and fixed line for the next
491/// claimant; MSI vectors were already taken out by `x86::release_msis`.
492pub(crate) fn release_for(tid: usize, scope: Scope) {
493    let mut devices = DEVICES.lock();
494    for record in devices.records.iter_mut() {
495        if record.owned_by(tid) && (scope == Scope::Retired || record.kind != DeviceKind::Dwc2) {
496            record.owner = None;
497        }
498    }
499    devices.publish_nic_lines();
500}

502/// Free the MMIO devices `tid` owns through one of the released `windows`
503/// (a capability revoke releases windows by class while the task lives).
504pub(crate) fn release_mmio_devices_in(tid: usize, windows: &[(usize, usize)]) {
505    let mut devices = DEVICES.lock();
506    for record in devices.records.iter_mut() {
507        if record.key.kind == BLOCK_DEVICE_MMIO
508            && record.owner == Some(tid)
509            && windows
510                .iter()
511                .any(|&(base, len)| record.has_window(base, len))
512        {
513            record.owner = None;
514        }
515    }
516    devices.publish_nic_lines();
517}

542/// Whether `irq` is a line of the device of a live NIC route, owned by that
543/// route's provider: the proof the MSI and VirtIO ISRs signal `NET_RX` on.
544/// Lock-free: one atomic load.
545pub(crate) fn is_nic_line(irq: u8) -> bool {
546    NIC_LINES[usize::from(irq / 64)].load(Ordering::Acquire) & (1 << (irq % 64)) != 0

560/// Every line of every device `tid` owns: the snapshot a
561/// `WaitCompletion(IRQ)` waits on.
562#[cfg_attr(not(target_arch = "x86_64"), allow(dead_code))]
563pub(crate) fn lines_owned_by(tid: usize) -> crate::dev::irq_wait::IrqLineSet {
564    let mut set = crate::dev::irq_wait::IrqLineSet::EMPTY;
565    for record in DEVICES.lock().records.iter() {
566        if let Some(lines) = record.irq.filter(|_| record.owned_by(tid)) {
567            lines.lines().for_each(|line| set.insert(line));
568        }
569    }
570    set
571}

573/// Consume `irq`'s pending flag only while `tid` still owns the line.
574///
575/// The ownership check and the swap share one `DEVICES` hold, and
576/// `x86::release_msis` drops the flags under it too, so a revoked (and possibly
577/// re-allocated) line is never consumed by its former owner.
578#[cfg_attr(not(target_arch = "x86_64"), allow(dead_code))]
579pub(crate) fn take_owned_line(tid: usize, irq: u8) -> bool {
580    let devices = DEVICES.lock();
581    let owned = devices
582        .records
583        .iter()
584        .any(|record| record.owned_by(tid) && record.holds_irq(irq));
585    owned && crate::dev::irq_wait::take_pending(irq)
586}

588/// Re-pend a line taken by [`take_owned_line`] whose record could not be
589/// delivered; a line revoked meanwhile is left alone.
590#[cfg_attr(not(target_arch = "x86_64"), allow(dead_code))]
591pub(crate) fn repend_owned_line(tid: usize, irq: u8) {
592    let devices = DEVICES.lock();
593    if devices
594        .records
595        .iter()

598        crate::dev::irq_wait::restore_pending(irq);
599    }
600}

608/// Make `tid` the owner of a VirtIO slot record at `base` with line `line`,
609/// displacing any owner; host tests only, under `TEST_STATE_LOCK`.
610#[cfg(test)]
611pub(crate) fn test_own_mmio(tid: usize, base: usize, line: u8) {
612    let mut devices = DEVICES.lock();
613    let record = devices
614        .find_or_insert(mmio_key(base).expect("valid base"), DeviceKind::VirtioSlot)
615        .expect("heap for a test record");
616    record.add_window(base, 0x200);
617    record.owner = Some(tid);
618    record.irq = Some(IrqLines::Line(line));
619    devices.publish_nic_lines();
620}

626    #[test]
627    fn a_claim_holds_until_its_owner_is_released() {
628        const BDF: u32 = 0x00_FE_ED;
629        const FIRST: usize = 0xA001;
630        const COMPETING: usize = 0xA002;
631        let _guard = crate::TEST_STATE_LOCK.lock();
632        release_for(FIRST, Scope::Retired);
633        release_for(COMPETING, Scope::Retired);
634        assert!(claim_pci(BDF, FIRST));
635        assert!(claim_pci(BDF, FIRST), "idempotent");
636        assert!(!claim_pci(BDF, COMPETING));
637        assert_eq!(pci_owner(BDF), Some(FIRST));
638        assert!(devices_of(FIRST)
639            .iter()
640            .any(|&(key, _)| key == pci_key(BDF).unwrap()));
641
642        release_for(FIRST, Scope::PcieDriver);
643        assert_eq!(pci_owner(BDF), None);
644        assert!(claim_pci(BDF, COMPETING));
645        assert_eq!(pci_owner(BDF), Some(COMPETING));
646        release_for(COMPETING, Scope::Retired);
647        assert!(!claim_pci(0x1_0000, FIRST), "not a BDF");
648    }

650    /// A machine with more functions than the old fixed table (256) loses
651    /// none: each keeps its record and class, and each is claimable.
652    #[test]
653    fn every_registered_function_keeps_a_claimable_record() {
654        const TID: usize = 0xA0C0;
655        const FUNCTIONS: u32 = 600;
656        const FIRST_BDF: u32 = 0xB000;
657        const CLASS: u32 = 0x02_00_00;
658        let _guard = crate::TEST_STATE_LOCK.lock();
659        for bdf in FIRST_BDF..FIRST_BDF + FUNCTIONS {
660            register_pci_function(bdf, CLASS);
661        }
662        for bdf in FIRST_BDF..FIRST_BDF + FUNCTIONS {
663            assert!(claim_pci(bdf, TID), "{bdf:#x} claimable");
664        }
665        let owned = devices_of(TID);
666        assert_eq!(owned.len(), FUNCTIONS as usize);
667        assert!(owned
668            .iter()
669            .all(|&(_, kind)| kind == DeviceKind::Pci { class: Some(CLASS) }));
670        release_for(TID, Scope::Retired);
671        assert!(devices_of(TID).is_empty());
672    }

674    /// A repeated function or BAR registration (a Platform Cell rescan, 235
675    /// after 236) adds no record and no window.
676    #[test]
677    fn repeated_registrations_add_nothing() {
678        const BDF: u32 = 0xB300;
679        let key = pci_key(BDF).unwrap();
680        let records_of = || {
681            let devices = DEVICES.lock();
682            let mut records = devices.records.iter().filter(|record| record.key == key);
683            (
684                records.clone().count(),
685                records.next().map(|record| record.bars),
686            )
687        };
688        let _guard = crate::TEST_STATE_LOCK.lock();
689        register_pci_function(BDF, 0x01_08_02);
690        register_pci_bar(BDF, 0xFD00_0000, 0x4000);
691        let (count, windows) = records_of();
692        assert_eq!(count, 1);
693        for _ in 0..3 {
694            register_pci_function(BDF, 0x01_08_02);
695            register_pci_bar(BDF, 0xFD00_0000, 0x4000);
696        }
697        assert_eq!(records_of(), (1, windows));
698    }

700    /// 235 creates a record only for a function that answers; a function
701    /// that already has one gains the window without a config-space probe.
702    #[test]
703    fn a_platform_bar_records_only_a_function_that_answers() {
704        const SILENT: u32 = 0xB410;
705        const ANSWERS: u32 = 0xB418;
706        const KNOWN: u32 = 0xB420;
707        let _guard = crate::TEST_STATE_LOCK.lock();
708        let has_record = |bdf| DEVICES.lock().find(pci_key(bdf).unwrap()).is_some();
709        assert!(!register_platform_bar(SILENT, 0xFD10_0000, 0x1000, || {
710            false
711        }));
712        assert!(!has_record(SILENT), "a silent function gets no record");
713        assert!(register_platform_bar(ANSWERS, 0xFD11_0000, 0x1000, || true));
714        assert!(has_record(ANSWERS));
715        assert!(is_device_window(0xFD11_0000, 0x1000));
716
717        register_pci_function(KNOWN, 0x02_00_00);
718        let probed = core::cell::Cell::new(false);
719        assert!(register_platform_bar(KNOWN, 0xFD12_0000, 0x1000, || {
720            probed.set(true);
721            false
722        }));
723        assert!(!probed.get(), "a known function is not probed");
724        assert!(is_device_window(0xFD12_0000, 0x1000));
725        assert!(
726            !register_platform_bar(0x1_0000, 0xFD13_0000, 0x1000, || true),
727            "not a BDF"

731    /// A `PcieDriverCap` revoke frees what that capability claimed and
732    /// leaves the DWC2 aperture, which `UsbDriverCap` claimed.
733    #[test]
734    fn a_pcie_driver_release_keeps_the_dwc2_aperture() {
735        const TID: usize = 0xA0D2;
736        const DWC2: usize = 0x7A10_0000;
737        const SLOT: usize = 0x7A20_0000;
738        let _guard = crate::TEST_STATE_LOCK.lock();
739        register_mmio_device(DWC2, 0x1_0000, DeviceKind::Dwc2, Some(0x2C));
740        assert_eq!(
741            claim_window(TID, DWC2 + 0x1000, 0x1000, true),
742            WindowClaim::Claimed(mmio_key(DWC2).unwrap())
743        );
744        test_own_mmio(TID, SLOT, 0x2D);
745        release_for(TID, Scope::PcieDriver);
746        assert!(irq_owned_by(TID, 0x2C), "DWC2 kept");
747        assert!(!irq_owned_by(TID, 0x2D), "VirtIO slot freed");
748        release_for(TID, Scope::Retired);
749        assert!(!irq_owned_by(TID, 0x2C));
750        assert_eq!(
751            claim_window(TID + 1, DWC2, 0x1000, true),
752            WindowClaim::Claimed(mmio_key(DWC2).unwrap()),
753            "free for the next claimant"
754        );
755        release_for(TID + 1, Scope::Retired);
756    }

772    /// The lock-free NIC line set follows every input under `DEVICES`: a line
773    /// is set only while its device's owner is its NIC route's provider, and
774    /// clears with the route or the owner (the x86_64 MSI allocation:
775    /// `x86::tests`).
776    #[test]
777    fn nic_lines_follow_the_route_the_owner_and_the_lines() {
778        const SLOT: usize = 0x7B00_0000;
779        const LINE: u8 = 0xC3;
780        const TID: usize = 0x7B01;
781        let _guard = crate::TEST_STATE_LOCK.lock();
782        let key = mmio_key(SLOT).unwrap();
783        test_own_mmio(TID, SLOT, LINE);
784        assert!(!is_nic_line(LINE), "owned, no route");
785        set_nic_route(key, Some(TID + 1));
786        assert!(!is_nic_line(LINE), "route of a task that does not own it");
787        set_nic_route(key, Some(TID));
788        assert!(is_nic_line(LINE));
789        assert!(!is_nic_line(LINE - 1) && !is_nic_line(LINE + 1));
790        set_nic_route(key, None);
791        assert!(!is_nic_line(LINE), "route lost its provider");
792        set_nic_route(key, Some(TID));
793        release_for(TID, Scope::PcieDriver);
794        assert!(!is_nic_line(LINE), "device released");
795        assert_eq!(
796            claim_window(TID, SLOT, 0x200, false),
797            WindowClaim::Claimed(key)
798        );
799        assert!(is_nic_line(LINE), "reclaimed by the route's provider");
800        set_nic_route(key, None);
801        release_for(TID, Scope::Retired);
802    }
