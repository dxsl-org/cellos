69/// Signal QEMU to exit with a success (0) or failure (1) code.
70///
71/// Only available under the `test-hooks` feature — never call this in
72/// production builds. The kernel integration-test harness uses this
73/// instead of parsing serial output for "PASS"/"FAIL" banners.
74///
75/// Device addresses: RISC-V = SiFive test 0x100000, ARM64 = semihosting,
76/// x86_64 = isa-debug-exit (iobase 0xF4).
77#[cfg(feature = "test-hooks")]
78pub fn qemu_exit(success: bool) -> ! {
79    use qemu_exit::QEMUExit;
80    #[cfg(target_arch = "riscv64")]
81    {
82        // SAFETY: 0x100000 is the SiFive test device address on the QEMU `virt`
83        // machine used by this test-only exit path.
84        unsafe {
85            qemu_exit::RISCV64::new(0x100000).exit(if success { 0 } else { 1 });
86        }
87    }
88    #[cfg(target_arch = "aarch64")]
89    {
90        qemu_exit::AArch64::new().exit(if success { 0 } else { 1 });
91    }
92    #[cfg(target_arch = "x86_64")]
93    {
94        // SAFETY: 0xF4 is the QEMU `isa-debug-exit` I/O port wired in the x86
95        // test harness (`-device isa-debug-exit,iobase=0xf4`); writing it exits
96        // QEMU. Constructing the port handle is `unsafe` (arbitrary port I/O).
97        unsafe {
98            qemu_exit::X86::new(0xF4, 0).exit(if success { 0 } else { 1 });
99        }
100    }
101    // Fallback for other arches: spin forever so the test times out clearly.
102    #[allow(clippy::empty_loop)]
103    // reason: on riscv/aarch64/x86_64 the arch-specific `.exit()` above diverges,
104    // so this loop is unreachable there; it only executes on a hypothetical arch
105    // with no qemu_exit backend.
106    #[allow(unreachable_code)]
107    loop {}
108}

122/// Kernel entry point called from HAL boot code
123#[no_mangle]
124pub extern "C" fn kmain(hartid: usize, dtb: usize) -> ! {
125    #[cfg(target_arch = "riscv64")]
126    task::smp::set_boot_physical_hart(hartid);
127    #[cfg(not(target_arch = "riscv64"))]
128    let _hartid = hartid;
129    let dtb = boot::effective_dtb(dtb);
130    cpu_features::detect(dtb);
131    // Parse DTB for MMIO bases before any driver or paging init.
132    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
133    crate::platform::init(dtb);
134    // Set runtime PLIC base before the later RV64 PLIC initialization consumes it.
135    #[cfg(target_arch = "riscv64")]
136    crate::platform::with(|p| hal::common::plic::set_plic_base(p.plic_base));
137    // 0. Initialize UART immediately for early logging
138    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
139    task::drivers::uart::init();
140    #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
141    crate::hal::uart_bcm_mini::init();
142    #[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
143    crate::hal::uart_pl011::init();
144    #[cfg(target_arch = "arm")]
145    crate::hal::uart_pl011::init();
146    #[cfg(target_arch = "x86_64")]
147    {
148        let board = crate::board::selected();
149        if !board.has_driver(cellos_boards::DriverId::Uart16550PortIo) {
150            panic!("[board] selected x86 board does not enable the early UART");
151        }
152        let soc = crate::board::selected_x86_64_soc();
153        crate::hal::uart_16550::configure(soc.com1.base, soc.com1.irq);
154        crate::hal::uart_16550::init();
155        crate::hal::uart_16550::puts(
156            "[x86-gate] configured 16550 ready: TX + polled RX; IRQ pending MADT\n",
157        );
158        if crate::hal::uart_16550::poll_input().is_some() {
159            crate::hal::uart_16550::puts("[x86-gate] COM1 polled RX observed\n");
160        }
161    }
162    #[cfg(target_arch = "x86")]
163    crate::hal::uart_16550::init();
164
165    // Set HHDM base for LAPIC/IOAPIC MMIO access AND for phys_to_virt.
166    // Limine maps RAM at HHDM_BASE+phys (no identity mapping of physical RAM).
167    // This must be called before FrameAllocator::new_from_map.
168    #[cfg(target_arch = "x86_64")]
169    {
170        let hhdm = crate::boot::limine::get_hhdm_offset().unwrap_or(0);
171        crate::hal::apic::set_hhdm_base(hhdm);
172        crate::memory::frame::set_phys_offset(hhdm as usize);
173        // Propagate the HHDM offset to the HAL PML4 walker so walk_create /
174        // walk_read can dereference physical PTE addresses via HHDM virtual ptrs.
175        crate::hal::paging::set_hhdm_offset(hhdm as usize);
176        // Initialise KASLR seed from HHDM entropy + RDTSC.
177        crate::memory::kaslr::init_kaslr(hhdm);
178    }
179
180    // 1. Initialize HAL (Architecture specific) - Early Trap Setup
181    // x86_64: LAPIC is deferred until after paging sets up the MMIO mapping
182    // (LAPIC phys 0xFEE00000 isn't in Limine's HHDM for MMIO regions).
183    #[cfg(target_arch = "x86_64")]
184    {
185        crate::hal::gdt::init();
186        crate::hal::idt::init();
187        crate::hal::cet::init_kernel_cet(); // LAYER2-CET-INIT
188        crate::hal::pku::init(); // LAYER2-PKU-INIT (requires IBT, checked inside)
189        crate::hal::syscall::init();
190        // apic::init_lapic() deferred — needs MMIO mapped via custom PML4
191    }
192    #[cfg(not(target_arch = "x86_64"))]
193    hal::ARCH.init();
194    // Define puts helper — arch-specific character output.
195    let puts = |s: &str| {
196        for c in s.bytes() {
197            #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
198            {
199                let _ = crate::hal::sbi::console_putchar(c);
200            }
201            #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
202            {
203                crate::hal::uart_bcm_mini::putchar(c);
204            }
205            #[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
206            {
207                crate::hal::uart_pl011::putchar(c);
208            }
209            #[cfg(target_arch = "arm")]
210            {
211                crate::hal::uart_pl011::putchar(c);
212            }
213            #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
214            {
215                crate::hal::uart_16550::putchar(c);
216            }
217        }
218    };
219
220    // Restore log_info helper
221    let log_info = |s: &str| {
222        puts("[INFO] ");
223        puts(s);
224        puts("\n");
225    };
226
227    // Stable banner — CI greps for this exact string.
228    puts("[Cellos] kernel boot v");
229    puts(env!("CARGO_PKG_VERSION"));
230    puts("\n");
231    puts("Kernel started (Hart: 0, DTB: ...)\n");
232    #[cfg(target_arch = "riscv64")]
233    if cpu_features::has_h_ext() {
234        puts("[cpu] H-extension: detected\n");
235    } else {
236        puts("[cpu] H-extension: not present\n");
237    }
238
239    // Parse bootloader information
240    let boot_info_result = boot::parse_bootloader_info();
241
242    // Check if Limine failed, if so, use fallback (SimpleBootInfo)
243    let boot_info: &dyn BootInfo = match &boot_info_result {
244        Ok(info) => info,
245        Err(_) => {
246            log_info("Limine not found, using QEMU/OpenSBI fallback");
247            // aarch64-virt sizes the kernel region from the linker end symbol —
248            // EMBEDDED_OVERRIDE images make the binary size unbounded.
249            boot::fallback_boot_info(dtb)
250        }
251    };
252    // Log physical base — non-default value confirms KASLR is active.
253    {
254        puts("[boot] kernel_phys_base=0x");
255        let mut base = boot_info.kernel_base();
256        let digits = b"0123456789abcdef";
257        let mut hex_buf = [b'0'; 16];
258        for i in (0..16).rev() {
259            hex_buf[i] = digits[base & 0xf];
260            base >>= 4;
261        }
262        if let Ok(s) = core::str::from_utf8(&hex_buf) {
263            puts(s);
264        }
265        puts("\n");
266    }
267
268    // Initialize kernel subsystems
269
270    // 1. Memory Management
271    // Get memory map from Boot Info (Converted to Cellos format)
272    let mmap_entries = boot_info.memory_map();
273
274    // Initialize frame allocator with the largest usable region
275    let frame_allocator = memory::frame::FrameAllocator::new_from_map(mmap_entries);
276    log::info!(
277        "[boot] allocator range {:#x}..{:#x} ({} bytes)",
278        frame_allocator.memory_start(),
279        frame_allocator.memory_end(),
280        frame_allocator.total_frames() * 4096
281    );
282
283    // Targets that already have usable memory attributes (or intentionally run
284    // without an MMU) can publish the allocator immediately. AArch64 must keep
285    // it local until paging is active because its Spinlock uses exclusive
286    // atomics that require Normal memory attributes; RV64 follows the same
287    // build-then-publish lifecycle.
288    #[cfg(any(
289        target_arch = "riscv32",
290        target_arch = "x86_64",
291        target_arch = "x86",
292        target_arch = "arm",
293    ))]
294    unsafe {
295        core::ptr::write(
296            &mut *memory::frame::FRAME_ALLOCATOR.lock(),
297            Some(frame_allocator),
298        );
299    }
300    #[cfg(any(
301        target_arch = "riscv32",
302        target_arch = "x86_64",
303        target_arch = "x86",
304        target_arch = "arm",
305    ))]
306    log_info("Frame allocator initialized");
307
308    // Limine does not guarantee HHDM mappings for firmware-owned memory. The
309    // parser requests a mapping for each RSDP/SDT range before dereferencing it.
310    // Child tables must stay inside ACPI reclaimable/NVS or firmware-reserved
311    // memory-map entries. Legacy BIOS ACPI records may live in the bounded
312    // windows supplied by the selected x86 platform profile. Some BIOSes,
313    // including q35, report other SDTs as Reserved, so table checksums remain
314    // the final trust gate.
315    #[cfg(target_arch = "x86_64")]
316    let acpi_info = {
317        let rsdp = crate::boot::limine::get_rsdp_ptr().unwrap_or(0);
318        let soc = crate::board::selected_x86_64_soc();
319        let mut map_physical = |physical: usize, length: usize| {
320            let end = physical.checked_add(length)?;
321            if length == 0 {
322                return None;
323            }
324            let legacy_firmware = soc.legacy_bios_window.contains(physical, length);
325            let legacy_rsdp = physical == rsdp && soc.legacy_rsdp_window.contains(physical, length);
326            let firmware_owned = mmap_entries.iter().any(|entry| {
327                let Some(entry_end) = entry.base.checked_add(entry.length) else {
328                    return false;
329                };
330                matches!(
331                    entry.ty,
332                    crate::boot::MemoryType::Reserved
333                        | crate::boot::MemoryType::AcpiReclaimable
334                        | crate::boot::MemoryType::AcpiNvs
335                ) && physical >= entry.base
336                    && end <= entry_end
337            });
338            if !legacy_firmware && !legacy_rsdp && !firmware_owned {
339                log::warn!(
340                    "[acpi] rejected non-firmware range {:#x}..{:#x}",
341                    physical,
342                    end
343                );
344                return None;
345            }
346            let mut allocator = memory::frame::FRAME_ALLOCATOR.lock();
347            let mut allocate = || {
348                allocator
349                    .as_mut()
350                    .and_then(|frames| frames.allocate_frame())
351            };
352            // SAFETY: the published allocator owns returned frames and the live
353            // Limine page tables are reachable through the configured HHDM.
354            let mapped = unsafe {
355                crate::hal::paging::map_hhdm_firmware_range(physical, length, &mut allocate)
356            };
357            if mapped {
358                Some(crate::memory::frame::phys_to_virt(physical))
359            } else {
360                None
361            }
362        };
363
364        let info = crate::acpi::parse(rsdp, &mut map_physical);
365        log::info!(
366            "[acpi] gates: madt={} hpet={} mcfg={}",
367            info.lapic_base != 0 && info.ioapic_base != 0,
368            info.hpet_base != 0,
369            info.ecam_base != 0
370        );
371        info
372    };
373
374    #[cfg(target_arch = "x86_64")]
375    let x86_ioapic_base = if crate::board::selected().has_driver(cellos_boards::DriverId::IoApic) {
376        acpi_info.ioapic_base
377    } else {
378        0
379    };
380    #[cfg(target_arch = "x86_64")]
381    let x86_lapic_base = if x86_ioapic_base != 0 {
382        acpi_info.lapic_base
383    } else {
384        0
385    };
386    #[cfg(target_arch = "x86_64")]
387    let x86_hpet_base = if crate::board::selected().has_driver(cellos_boards::DriverId::Hpet) {
388        acpi_info.hpet_base
389    } else {
390        0
391    };
392    #[cfg(target_arch = "x86_64")]
393    let x86_ecam_base = if crate::board::selected().has_driver(cellos_boards::DriverId::PcieEcam) {
394        acpi_info.ecam_base
395    } else {
396        0
397    };
398    #[cfg(target_arch = "x86_64")]
399    let x86_timer_ready = x86_lapic_base != 0 && x86_ioapic_base != 0 && x86_hpet_base != 0;
400
401    // 3. Paging (Virtual Memory) Setup
402    // x86_64 bring-up: Limine's PML4 already maps RAM via HHDM and the kernel
403    // at 0xFFFFFFFF80000000. We skip building + activating our own page tables
404    // until the full x86_64 port (Phase 09). init_kernel_paging uses physical
405    // addresses as virtual pointers, which would fault under Limine's paging.
406    #[cfg(not(any(
407        target_arch = "riscv32",
408        target_arch = "x86_64",
409        target_arch = "x86",
410        target_arch = "arm",
411    )))]
412    {
413        let mut frame_allocator = frame_allocator;
414        log_info("Initializing paging...");
415        let root_table_phys =
416            memory::paging::init_kernel_paging(&mut frame_allocator, mmap_entries)
417                .expect("Failed to initialize paging");
418        log_info("Paging initialized");
419        log_info("Activating paging...");
420        unsafe {
421            memory::paging::activate_paging(root_table_phys);
422        }
423        *memory::paging::KERNEL_ROOT.lock() = Some(root_table_phys);
424        #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
425        TEST_SAFE_ROOT.store(root_table_phys, core::sync::atomic::Ordering::Release);
426        unsafe {
427            core::ptr::write(
428                &mut *memory::frame::FRAME_ALLOCATOR.lock(),
429                Some(frame_allocator),
430            );
431        }
432        log_info("Paging activated");
433        log_info("Frame allocator initialized");
434        // Set sstatus.SUM=1 so S-mode (kernel) can access USER-mapped pages throughout
435        // the kernel lifetime. VirtIO/peripheral MMIO is mapped USER=1 for Driver Cells
436        // (U-mode). Without SUM=1 the kernel's tech-debt VirtIO drivers fault at early-boot
437        // MMIO init. In Cellos's SAS+LBI model security comes from Rust type safety, not
438        // hardware USER-bit separation for kernel-vs-cell — SUM=1 is safe and intentional.
439        #[cfg(target_arch = "riscv64")]
440        // SAFETY: csrs modifies sstatus.SUM (bit 18). Safe to set for kernel S-mode code.
441        unsafe {
442            core::arch::asm!("csrs sstatus, {sum}", sum = in(reg) 0x40000_usize, options(nostack));
443        }
444    }
445    #[cfg(target_arch = "x86_64")]
446    {
447        // Set runtime ECAM base from validated ACPI before PCIe scan. Zero
448        // keeps PCIe closed; there is no implicit q35 fallback.
449        crate::task::drivers::pcie_ecam::set_ecam_base_x86(x86_ecam_base as usize);
450
451        log_info("Initializing x86_64 paging (kernel PML4)...");
452        let root_table_phys = {
453            let mut locked_frame_allocator = memory::frame::FRAME_ALLOCATOR.lock();
454            memory::paging::init_kernel_paging_x86(
455                locked_frame_allocator
456                    .as_mut()
457                    .expect("Frame allocator not initialized"),
458                x86_ioapic_base,
459                x86_hpet_base,
460                x86_lapic_base,
461                x86_ecam_base,
462            )
463            .expect("Failed to initialize x86_64 kernel PML4")
464        };
465        log_info("x86_64 paging initialized");
466        log_info("Activating x86_64 paging (mov cr3)...");
467        // SAFETY: init_kernel_paging_x86 copied higher-half entries from Limine's PML4
468        // (preserving kernel text/data/HHDM) and identity-mapped MMIO, so the kernel
469        // continues executing after this CR3 switch without a triple-fault.
470        unsafe {
471            memory::paging::activate_paging(root_table_phys);
472        }
473        // Immediate configured port-I/O probe after activate_paging. If 'Q'
474        // appears, the CR3 switch returned to kmain with the UART mechanism live.
475        crate::hal::uart_16550::putchar(b'Q');
476        if x86_timer_ready {
477            crate::hal::apic::set_lapic_phys(x86_lapic_base);
478            crate::hal::apic::set_ioapic_phys(x86_ioapic_base);
479            crate::hal::apic::set_irq_overrides(
480                &acpi_info.irq_overrides,
481                acpi_info.ioapic_gsi_base,
482            );
483            crate::hal::set_hpet_base(x86_hpet_base as usize);
484            crate::hal::init_timers();
485            log_info("x86_64 timers initialized from ACPI (HPET + LAPIC)");
486        } else {
487            log::error!("[x86-gate] timer CLOSED: validated MADT + HPET required");
488        }
489
490        // Tier 3b x86 VMM P01: enter root-of-virtualization on the BSP.
491        // SVM first (TCG-testable); VMX only on genuine Intel (KVM/HW lane).
492        // Failure is non-fatal — the kernel runs fine without virt; the
493        // HypervisorCap gate simply stays closed (has_x86_virt() == false).
494        match cpu_features::x86_virt_kind() {
495            Some(cpu_features::X86Virt::Svm) => {
496                let hsave_pa = {
497                    let mut guard = memory::frame::FRAME_ALLOCATOR.lock();
498                    guard
499                        .as_mut()
500                        .expect("Frame allocator not initialized")
501                        .allocate_frame()
502                };
503                match hsave_pa {
504                    // SAFETY: hsave frame is freshly allocated, exclusively
505                    // owned, never freed (root operation lasts until reset),
506                    // and never mapped into any guest NPT (P02 invariant).
507                    Some(pa) => match unsafe { hal::svm::enable(pa as u64) } {
508                        Ok(()) => {
509                            cpu_features::latch_x86_root_active();
510                            log_info("x86 virt: root operation active (SVM)");
511                        }
512                        Err(_) => log_info("x86 virt: SVM present but firmware-locked; cap closed"),
513                    },
514                    None => log_info("x86 virt: OOM allocating HSAVE frame; cap closed"),
515                }
516            }
517            Some(cpu_features::X86Virt::Vmx) => {
518                let vmxon_pa = {
519                    let mut guard = memory::frame::FRAME_ALLOCATOR.lock();
520                    guard
521                        .as_mut()
522                        .expect("Frame allocator not initialized")
523                        .allocate_frame()
524                };
525                match vmxon_pa {
526                    Some(pa) => {
527                        let va = memory::frame::phys_to_virt(pa) as *mut u32;
528                        // SAFETY: vmxon frame freshly allocated + exclusively
529                        // owned + alive forever; va is its HHDM mapping.
530                        match unsafe { hal::vmx::enter_root(pa as u64, va) } {
531                            Ok(()) => {
532                                cpu_features::latch_x86_root_active();
533                                log_info("x86 virt: root operation active (VMX)");
534                            }
535                            Err(_) => log_info("x86 virt: VMX present but unavailable; cap closed"),
536                        }
537                    }
538                    None => log_info("x86 virt: OOM allocating VMXON frame; cap closed"),
539                }
540            }
541            None => log_info("x86 virt: not supported by CPU; cap closed"),
542        }
543    }
544    // Bare physical: RV32 Nano (SATP=0), x86_32 (CR0.PG=0), AArch32 (MMU off).
545    #[cfg(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm"))]
546    {
547        memory::paging::init_bare();
548        log_info("Paging: bare physical");
549    }
550
551    // 4. Heap Allocator (Global) - MUST be after paging but before any allocations
552    // 32 MiB = 8192 frames. Sized to hold:
553    //   - embedded RAM disk copy (~4 MiB), VirtIO GPU framebuffer (~4 MiB), cell ELFs + kernel structures
554    const HEAP_FRAMES: usize = 8_192;
555    let heap_start = {
556        let mut allocator_guard = memory::frame::FRAME_ALLOCATOR.lock();
557        let allocator = allocator_guard
558            .as_mut()

1076    // remaining Info chatter is per-spawn noise ([loader] SpawnFromPath, Spawn:,
1077    // ELF LOAD) that floods the UART and buries the shell prompt. WARN/ERROR still
1078    // surface real problems. Raise back to Info when debugging the spawn path.
1079
1080    // Enable interrupts before entering the idle loop.
1081    // RISC-V: set SPP=1 and SIE=1 in sstatus (0x102).
1082    // AArch64: clear DAIF.I bit to unmask IRQs.
1083    // x86_64: STI via ARCH.enable_interrupts().
1084    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1085    unsafe {
1086        // SAFETY: csrs sstatus SPP|SIE from S-mode — standard interrupt enable.
1087        core::arch::asm!("csrs sstatus, {0}", in(reg) 0x102usize);
1088    }
1089    #[cfg(target_arch = "aarch64")]
1090    unsafe {
1091        core::arch::asm!("msr daifclr, #2", options(nomem, nostack));
1092    }
1093    // x86_64, x86_32, AArch32: use the Arch trait's enable_interrupts().
1094    #[cfg(any(target_arch = "x86_64", target_arch = "x86", target_arch = "arm"))]
1095    crate::hal::ARCH.enable_interrupts();
1096
1097    log::set_max_level(log::LevelFilter::Warn);
1098
1099    // Probe 'Q': fires ONLY if no IRQ preempted the code between daifclr and here.
1100    // Q fires  → BCM2835 IRQ is not actually delivered to CPU on QEMU (pending but silent).
1101    // Q absent → IRQ fired immediately after daifclr (expected); check if H/G appear.
1102    #[cfg(feature = "board-rpi3")]
1103    crate::hal::uart_bcm_mini::probe_put(b'Q');
1104
1105    // board-rpi3: Print IRQ state snapshot immediately after enabling IRQs.
1106    // Format: "K<src_bit8><pend_bit1><src_nibble>" where src = CORE0_IRQ_SOURCE,
1107    // pend = IRQ_PENDING1 bit 1 (BCM2835 systimer C1 pending).
1108    // If G never fires but K shows pend=1 and src=0: QEMU does not route BCM2835→BCM2836.
1109    #[cfg(feature = "board-rpi3")]
1110    {
1111        let soc = hal_soc_bcm27xx::BCM2837;
1112        // SAFETY: both BCM controller apertures are identity-mapped before IRQs are enabled.
1113        let src_raw = unsafe {
1114            core::ptr::read_volatile((soc.mmio.local_controller_base + 0x60) as *const u32)
1115        };
1116        let pend =
1117            unsafe { core::ptr::read_volatile((soc.mmio.legacy_irq_base + 0x04) as *const u32) };
1118        let hex = |n: u32| -> u8 {
1119            if n < 10 {
1120                b'0' + n as u8
1121            } else {
1122                b'a' + n as u8 - 10
1123            }
1124        };
1125        crate::hal::uart_bcm_mini::probe_put(b'K');
1126        crate::hal::uart_bcm_mini::probe_put(if src_raw & soc.irq.local_gpu_mask != 0 {
1127            b'1'
1128        } else {
1129            b'0'
1130        });
1131        crate::hal::uart_bcm_mini::probe_put(if pend & (1 << soc.irq.system_timer_c1) != 0 {
1132            b'1'
1133        } else {
1134            b'0'
1135        });
1136        crate::hal::uart_bcm_mini::probe_put(hex(src_raw & 0xF));
1137    }
1138
1139    // Probe 'L': fires once per idle loop iteration (only first 3 on board-rpi3 to avoid flood).
1140    // If 'L' never appears after 'K', the code never reaches the idle loop.
1141    #[cfg(feature = "board-rpi3")]
1142    {
1143        static IDLE_COUNT: core::sync::atomic::AtomicUsize =
1144            core::sync::atomic::AtomicUsize::new(0);
1145        loop {
1146            let n = IDLE_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
1147            if n < 4 {
1148                crate::hal::uart_bcm_mini::probe_put(b'L');
1149            }
1150            crate::task::yield_cpu();
1151            crate::hal::ARCH.wait_for_interrupt();
1152        }
1153    }
1154    #[cfg(not(feature = "board-rpi3"))]
1155    loop {
1156        crate::task::yield_cpu();
1157        crate::hal::ARCH.wait_for_interrupt();
1158    }
1159}

1161/// Emit an allocation-free RV64 fault snapshot before the panic path.
1162#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
1163#[no_mangle]
1164pub extern "C" fn vi_rv64_kernel_fault_snapshot(satp: usize, stval: usize) {
1165    fn put_hex(value: usize) {
1166        const HEX: &[u8; 16] = b"0123456789abcdef";
1167        for shift in (0..16).rev().map(|n| n * 4) {
1168            let _ = crate::hal::sbi::console_putchar(HEX[(value >> shift) & 0xf]);
1169        }
1170    }
1171
1172    fn put_label(label: &[u8], value: usize) {
1173        for byte in label {
1174            let _ = crate::hal::sbi::console_putchar(*byte);
1175        }
1176        put_hex(value);
1177    }
1178
1179    unsafe fn read_pte(table_phys: usize, index: usize) -> usize {
1180        (table_phys as *const usize).add(index).read_volatile()
1181    }
1182
1183    fn next_table(entry: usize) -> usize {
1184        ((entry >> 10) & 0x003f_ffff_ffff_ffff) << 12
1185    }
1186
1187    for byte in b"\n[selftest] RV64-FAULT satp=0x" {
1188        let _ = crate::hal::sbi::console_putchar(*byte);
1189    }
1190    put_hex(satp);
1191    for byte in b" stval=0x" {
1192        let _ = crate::hal::sbi::console_putchar(*byte);
1193    }
1194    put_hex(stval);
1195
1196    let kernel_root = TEST_SAFE_ROOT.load(core::sync::atomic::Ordering::Acquire);
1197    if kernel_root != 0 {
1198        let kernel_satp = (8usize << 60) | (kernel_root >> 12);
1199        // SAFETY: this test-only panic snapshot switches to the known-safe
1200        // shared root so it can inspect private page-table memory by physical VA.
1201        unsafe {
1202            core::arch::asm!(
1203                "csrw satp, {satp}",
1204                "sfence.vma zero, zero",
1205                satp = in(reg) kernel_satp,
1206                options(nostack)
1207            );
1208        }
1209        let root_phys = (satp & ((1usize << 44) - 1)) << 12;
1210        let vpn2 = (stval >> 30) & 0x1ff;
1211        let vpn1 = (stval >> 21) & 0x1ff;
1212        let vpn0 = (stval >> 12) & 0x1ff;
1213        let pte2 = unsafe { read_pte(root_phys, vpn2) };
1214        let pte1 = if pte2 & 1 != 0 {
1215            unsafe { read_pte(next_table(pte2), vpn1) }
1216        } else {
1217            0
1218        };
1219        let pte0 = if pte1 & 1 != 0 {
1220            unsafe { read_pte(next_table(pte1), vpn0) }
1221        } else {
1222            0
1223        };
1224        put_label(b" pte2=0x", pte2);
1225        put_label(b" pte1=0x", pte1);
1226        put_label(b" pte0=0x", pte0);
1227    }
1228}

1230/// Non-recoverable kernel panic handler.
1231///
1232/// `current_cell_id` is allocation attribution, not an execution-mode proof:
1233/// kernel code can hold scheduler or other kernel locks while servicing a
1234/// Cell. Therefore this handler never schedules or retires a Cell. Only the
1235/// architecture trap path, after proving an interrupted U-mode context, may
1236/// enter the deferred retirement funnel.
1237#[panic_handler]
1238fn panic(info: &PanicInfo) -> ! {
1239    // True kernel panic: print diagnostics and halt.
1240    #[inline(always)]
1241    fn panic_putchar(c: u8) {
1242        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1243        {
1244            let _ = crate::hal::sbi::console_putchar(c);
1245        }
1246        #[cfg(all(target_arch = "aarch64", feature = "board-rpi3"))]
1247        {
1248            crate::hal::uart_bcm_mini::putchar(c);
1249        }
1250        #[cfg(all(target_arch = "aarch64", not(feature = "board-rpi3")))]
1251        {
1252            crate::hal::uart_pl011::putchar(c);
1253        }
1254        #[cfg(target_arch = "arm")]
1255        {
1256            crate::hal::uart_pl011::putchar(c);
1257        }
1258        #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
1259        {
1260            crate::hal::uart_16550::putchar(c);
1261        }
1262    }
1263    let puts = |s: &str| {
1264        for c in s.bytes() {
1265            panic_putchar(c);
1266        }
1267    };
1268    puts("\n[KERNEL PANIC] ");
1269    puts("Critical failure.\n");
1270    use core::fmt::Write;
1271    struct PanicWriter;
1272    impl core::fmt::Write for PanicWriter {
1273        fn write_str(&mut self, s: &str) -> core::fmt::Result {
1274            for c in s.bytes() {
1275                panic_putchar(c);
1276            }
1277            Ok(())
1278        }
1279    }
1280    let _ = writeln!(PanicWriter, "{}", info);
1281
1282    // Reboot or spin: RISC-V uses SBI SRST; ARM64 / x86_64 spin.
1283    puts("[KERNEL PANIC] halting...\n");
1284    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1285    crate::hal::sbi::system_reset(crate::hal::sbi::SBI_RESET_COLD_REBOOT, 0);
1286    #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
1287    loop {
1288        unsafe {
1289            core::arch::asm!("cli; hlt", options(nomem, nostack));
1290        }
1291    }
1292    // Fallback halt for all non-x86 arches (including riscv — unreachable after system_reset).
1293    #[cfg(not(any(target_arch = "x86_64", target_arch = "x86")))]
1294    loop {
1295        unsafe {
1296            core::arch::asm!("wfi", options(nomem, nostack));
1297        }
1298    }
1299}
