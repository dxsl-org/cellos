120/// Signal QEMU to exit with a success (0) or failure (1) code.
121///
122/// Only available under the `test-hooks` feature — never call this in
123/// production builds. The kernel integration-test harness uses this
124/// instead of parsing serial output for "PASS"/"FAIL" banners.
125///
126/// Device addresses: RISC-V = SiFive test 0x100000, ARM64 = semihosting,
127/// x86_64 = isa-debug-exit (iobase 0xF4).
128#[cfg(feature = "test-hooks")]
129pub fn qemu_exit(success: bool) -> ! {
130    use qemu_exit::QEMUExit;
131    #[cfg(target_arch = "riscv64")]
132    {
133        // SAFETY: 0x100000 is the SiFive test device address on the QEMU `virt`
134        // machine used by this test-only exit path.
135        unsafe {
136            qemu_exit::RISCV64::new(0x100000).exit(if success { 0 } else { 1 });
137        }
138    }
139    #[cfg(target_arch = "aarch64")]
140    {
141        qemu_exit::AArch64::new().exit(if success { 0 } else { 1 });
142    }
143    #[cfg(target_arch = "x86_64")]
144    {
145        // SAFETY: 0xF4 is the QEMU `isa-debug-exit` I/O port wired in the x86
146        // test harness (`-device isa-debug-exit,iobase=0xf4`); writing it exits
147        // QEMU. Constructing the port handle is `unsafe` (arbitrary port I/O).
148        unsafe {
149            qemu_exit::X86::new(0xF4, 0).exit(if success { 0 } else { 1 });
150        }
151    }
152    // Fallback for other arches: spin forever so the test times out clearly.
153    #[allow(clippy::empty_loop)]
154    // reason: on riscv/aarch64/x86_64 the arch-specific `.exit()` above diverges,
155    // so this loop is unreachable there; it only executes on a hypothetical arch
156    // with no qemu_exit backend.
157    #[allow(unreachable_code)]
158    loop {}
159}

173/// Kernel entry point called from HAL boot code
174#[no_mangle]
175pub extern "C" fn kmain(hartid: usize, dtb: usize) -> ! {
176    #[cfg(target_arch = "riscv64")]
177    crate::sched::smp::set_boot_physical_hart(hartid);
178    #[cfg(not(target_arch = "riscv64"))]
179    let _hartid = hartid;
180    let dtb = boot::effective_dtb(dtb);
181    cpu_features::detect(dtb);
182    // Parse DTB for MMIO bases before any driver or paging init.
183    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
184    crate::platform::init(dtb);
185    // Set runtime PLIC base before the later RV64 PLIC initialization consumes it.
186    #[cfg(target_arch = "riscv64")]
187    crate::platform::with(|p| hal::common::plic::set_plic_base(p.plic_base));
188    // 0. Initialize UART immediately for early logging
189    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
190    crate::dev::uart::init();
191    #[cfg(target_arch = "aarch64")]
192    crate::hal::backend::Board::console_init();
193    #[cfg(target_arch = "arm")]
194    crate::hal::uart_pl011::init();
195    #[cfg(target_arch = "x86_64")]
196    {
197        // No profile is bound before ACPI discovery; every catalog profile
198        // shares these facts (const-asserted in hal_soc_x86), so the console
199        // and COM wiring cannot depend on which profile is bound later.
200        let early = &hal_soc_x86::X86_EARLY_FACTS;
201        crate::hal::uart_16550::configure(early.com1.base, early.com1.irq);
202        crate::hal::uart_16550::init();
203        // Register every declared port: the console takes slot 0 and each extra
204        // port must answer a 16550 probe before it is offered, so a declared
205        // port this machine does not have produces a refusal rather than a
206        // cell writing into empty port space.
207        {
208            use crate::hal::uart_16550::{SerialPortSpec, MAX_PORTS};
209            let mut specs = [SerialPortSpec { base: 0, irq: 0 }; MAX_PORTS];
210            let declared = early.serial_ports.len().min(MAX_PORTS);
211            for (slot, port) in early.serial_ports.iter().take(declared).enumerate() {
212                specs[slot] = SerialPortSpec {
213                    base: port.base,
214                    irq: port.irq,
215                };
216            }
217            crate::hal::uart_16550::configure_all(&specs[..declared]);
218        }
219        // Emitted only after `init()`: `putchar` asserts a configured port. The
220        // bound profile is named by `board::bind_x86_profile` after ACPI.
221        crate::hal::uart_16550::puts(
222            "[x86-gate] configured 16550 ready: TX + polled RX; IRQ pending MADT\n",
223        );
224        if crate::hal::uart_16550::poll_input().is_some() {
225            crate::hal::uart_16550::puts("[x86-gate] COM1 polled RX observed\n");
226        }
227        // Register the log backend now, as RV64 does above: the console is
228        // configured and the boot stub installed GS (Spinlock needs it), and
229        // neither the logger nor its console lock allocates. Registering it at
230        // step 6 dropped every `log` record of the boot path before the heap
231        // (allocator ranges, firmware-range rejections, ACPI table findings).
232        crate::dev::uart::init();
233    }
234    #[cfg(target_arch = "x86")]
235    crate::hal::uart_16550::init();
236
237    // Set HHDM base for LAPIC/IOAPIC MMIO access AND for phys_to_virt.
238    // Limine maps RAM at HHDM_BASE+phys (no identity mapping of physical RAM).
239    // This must be called before FrameAllocator::new_from_map.
240    #[cfg(target_arch = "x86_64")]
241    {
242        let hhdm = crate::boot::limine::get_hhdm_offset().unwrap_or(0);
243        crate::hal::apic::set_hhdm_base(hhdm);
244        crate::memory::frame::set_phys_offset(hhdm as usize);
245        // Propagate the HHDM offset to the HAL PML4 walker so walk_create /
246        // walk_read can dereference physical PTE addresses via HHDM virtual ptrs.
247        crate::hal::paging::set_hhdm_offset(hhdm as usize);
248        // Initialise KASLR seed from HHDM entropy + RDTSC.
249        crate::memory::kaslr::init_kaslr(hhdm);
250    }
251
252    // 1. Initialize HAL (Architecture specific) - Early Trap Setup
253    // x86_64: LAPIC is deferred until after paging sets up the MMIO mapping
254    // (LAPIC phys 0xFEE00000 isn't in Limine's HHDM for MMIO regions).
255    #[cfg(target_arch = "x86_64")]
256    {
257        // SSE/FXSR first: every IDT entry runs FXSAVE/LDMXCSR, so a fault
258        // taken during gdt/idt init (including the IDT fixture's probes) would
259        // #UD in the entry path and triple-fault without CR4.OSFXSR.
260        crate::hal::init_sse();
261        // Unmap the guard pages around every NMI/#DF/#MC IST stack, and the one
262        // below the BSP boot stack, before the TSS and IST gates are loaded.
263        // The live root is Limine's; the kernel PML4, AP roots and domain roots
264        // later copy its higher-half entries, so they share these cleared PTEs.
265        if let Err(error) = unsafe { crate::hal::x86_64::ist::install_guards() } {
266            panic!("x86: IST/boot-stack guard pages unavailable: {error:?}");
267        }
268        crate::hal::gdt::init();
269        crate::hal::idt::init();
270        crate::hal::cet::init_kernel_cet(); // LAYER2-CET-INIT
271        crate::hal::pku::init(); // LAYER2-PKU-INIT (requires IBT, checked inside)
272        crate::hal::syscall::init();
273        // Opt-in SMEP/SMAP/UMIP profile: a requested protection the CPU
274        // lacks refuses boot rather than running unprotected.
275        assert!(
276            crate::hal::x86_64::protect::init_cpu(),
277            "x86: requested SMEP/SMAP/UMIP profile unsupported by CPU"
278        );
279        crate::hal::x86_64::publish_boot_cpu_policy();
280        // apic::init_lapic() deferred — needs MMIO mapped via custom PML4
281    }
282    #[cfg(not(target_arch = "x86_64"))]
283    hal::ARCH.init();
284    // Define puts helper — arch-specific character output.
285    let puts = |s: &str| {
286        for c in s.bytes() {
287            #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
288            {
289                let _ = crate::hal::sbi::console_putchar(c);
290            }
291            #[cfg(target_arch = "aarch64")]
292            {
293                crate::hal::backend::Board::early_putc(c);
294            }
295            #[cfg(target_arch = "arm")]
296            {
297                crate::hal::uart_pl011::putchar(c);
298            }
299            #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
300            {
301                crate::hal::uart_16550::putchar(c);
302            }
303        }
304    };
305
306    // Restore log_info helper
307    let log_info = |s: &str| {
308        puts("[INFO] ");
309        puts(s);
310        puts("\n");
311    };
312
313    // Stable banner — CI greps for this exact string.
314    puts("[Cellos] kernel boot v");
315    puts(env!("CARGO_PKG_VERSION"));
316    puts("\n");
317    puts("Kernel started (Hart: 0, DTB: ...)\n");
318    #[cfg(target_arch = "riscv64")]
319    if cpu_features::has_h_ext() {
320        puts("[cpu] H-extension: detected\n");
321    } else {
322        puts("[cpu] H-extension: not present\n");
323    }
324
325    // Parse bootloader information
326    let boot_info_result = boot::parse_bootloader_info();
327
328    // Check if Limine failed, if so, use fallback (SimpleBootInfo)
329    let boot_info: &dyn BootInfo = match &boot_info_result {
330        Ok(info) => info,
331        Err(_) => {
332            log_info("Limine not found, using QEMU/OpenSBI fallback");
333            // aarch64-virt sizes the kernel region from the linker end symbol —
334            // EMBEDDED_OVERRIDE images make the binary size unbounded.
335            boot::fallback_boot_info(dtb)
336        }
337    };
338    // Log physical base — non-default value confirms KASLR is active.
339    {
340        puts("[boot] kernel_phys_base=0x");
341        let mut base = boot_info.kernel_base();
342        let digits = b"0123456789abcdef";
343        let mut hex_buf = [b'0'; 16];
344        for i in (0..16).rev() {
345            hex_buf[i] = digits[base & 0xf];
346            base >>= 4;
347        }
348        if let Ok(s) = core::str::from_utf8(&hex_buf) {
349            puts(s);
350        }
351        puts("\n");
352    }
353
354    // Initialize kernel subsystems
355
356    // 1. Memory Management
357    // Get memory map from Boot Info (Converted to Cellos format)
358    let mmap_entries = boot_info.memory_map();
359
360    // Initialize the frame allocator with every usable range in the map.
361    let frame_allocator = memory::frame::FrameAllocator::new_from_map(mmap_entries);
362    log::info!(
363        "[boot] allocator: {} usable range(s), {} frames ({} MiB)",
364        frame_allocator.managed_ranges().count(),
365        frame_allocator.total_frames(),
366        frame_allocator.total_frames() * 4096 / (1024 * 1024)
367    );
368    for range in frame_allocator.managed_ranges() {
369        log::info!(
370            "[boot]   range {:#x}..{:#x} ({} MiB)",
371            range.start,
372            range.start + range.frames * 4096,
373            range.frames * 4096 / (1024 * 1024)
374        );
375    }
376
377    // Targets that already have usable memory attributes (or intentionally run
378    // without an MMU) can publish the allocator immediately. AArch64 must keep
379    // it local until paging is active because its Spinlock uses exclusive
380    // atomics that require Normal memory attributes; RV64 follows the same
381    // build-then-publish lifecycle.
382    #[cfg(any(
383        target_arch = "riscv32",
384        target_arch = "x86_64",
385        target_arch = "x86",
386        target_arch = "arm",
387    ))]
388    unsafe {
389        core::ptr::write(
390            &mut *memory::frame::FRAME_ALLOCATOR.lock(),
391            Some(frame_allocator),
392        );
393    }
394    #[cfg(any(
395        target_arch = "riscv32",
396        target_arch = "x86_64",
397        target_arch = "x86",
398        target_arch = "arm",
399    ))]
400    log_info("Frame allocator initialized");
401
402    // Limine does not guarantee HHDM mappings for firmware-owned memory. The
403    // ACPI and SMBIOS parsers request a mapping for each RSDP/SDT/SMBIOS range
404    // before dereferencing it. Those ranges must stay inside ACPI
405    // reclaimable/NVS or firmware-reserved memory-map entries. Legacy BIOS
406    // records may live in the bounded windows of `X86_EARLY_FACTS`, which every
407    // catalog profile shares. Some BIOSes, including q35, report other SDTs as
408    // Reserved, so checksums remain the final trust gate.
409    #[cfg(target_arch = "x86_64")]
410    let (acpi_info, smbios_system) = {
411        let rsdp = crate::boot::limine::get_rsdp_ptr().unwrap_or(0);
412        let early = &hal_soc_x86::X86_EARLY_FACTS;
413        let mut map_physical = |physical: usize, length: usize| {
414            let end = physical.checked_add(length)?;
415            if length == 0 {
416                return None;
417            }
418            let legacy_firmware = early.legacy_bios_window.contains(physical, length);
419            let legacy_rsdp =
420                physical == rsdp && early.legacy_rsdp_window.contains(physical, length);
421            let firmware_owned = mmap_entries.iter().any(|entry| {
422                let Some(entry_end) = entry.base.checked_add(entry.length) else {
423                    return false;
424                };
425                matches!(
426                    entry.ty,
427                    crate::boot::MemoryType::Reserved
428                        | crate::boot::MemoryType::AcpiReclaimable
429                        | crate::boot::MemoryType::AcpiNvs
430                ) && physical >= entry.base
431                    && end <= entry_end
432            });
433            if !legacy_firmware && !legacy_rsdp && !firmware_owned {
434                log::warn!(
435                    "[firmware] rejected non-firmware range {:#x}..{:#x}",
436                    physical,
437                    end
438                );
439                return None;
440            }
441            let mut allocator = memory::frame::FRAME_ALLOCATOR.lock();
442            let mut allocate = || {
443                allocator
444                    .as_mut()
445                    .and_then(|frames| frames.allocate_frame())
446            };
447            // SAFETY: the published allocator owns returned frames and the live
448            // Limine page tables are reachable through the configured HHDM.
449            let mapped = unsafe {
450                crate::hal::paging::map_hhdm_firmware_range(physical, length, &mut allocate)
451            };
452            if mapped {
453                Some(crate::memory::frame::phys_to_virt(physical))
454            } else {
455                None
456            }
457        };
458
459        let info = crate::acpi::parse(rsdp, &mut map_physical);
460        log::info!(
461            "[acpi] gates: madt={} hpet={} mcfg={} dmar={} fadt={} spcr={}",
462            info.lapic_base != 0 && info.ioapic_base != 0,
463            info.hpet_base != 0,
464            info.ecam_base != 0,
465            info.dmar_base != 0,
466            info.fadt.is_some(),
467            info.spcr.is_some()
468        );
469        crate::acpi::with_topology(|topo| {
470            log::info!(
471                "[acpi] topology: cpus={} ioapics={} nmis={} mcfg={} drhd={} rmrr={} ir={} x2apic_opt_out={}",
472                topo.cpus.len(),
473                topo.ioapics.len(),
474                topo.lapic_nmis.len(),
475                topo.mcfg.len(),
476                topo.drhds.len(),
477                topo.rmrrs.len(),
478                topo.dmar_intr_remap(),
479                topo.dmar_x2apic_opt_out()
480            )
481        });
482        // Power facts (`\_S5_`, reset register, GPE blocks) need the same
483        // firmware-range mapper; the G1 private lane keeps its own contract.
484        #[cfg(not(feature = "g1-cpu-accounting"))]
485        crate::acpi::sci::init(&info, rsdp, &mut map_physical);
486        // SMBIOS entry points and tables get the same firmware-range admission
487        // as ACPI tables: the legacy BIOS window or firmware-owned memory.
488        let (smbios_32, smbios_64) = crate::boot::limine::get_smbios_entry_points();
489        let smbios = crate::smbios::discover(smbios_32, smbios_64, &mut map_physical);
490        (info, smbios)
491    };
492    #[cfg(target_arch = "x86_64")]
493    let platform_identity = crate::board::PlatformIdentity {
494        acpi_oem: acpi_info.root_oem,
495        smbios: smbios_system,
496        hypervisor: cpu_features::hypervisor_present(),
497    };
498    #[cfg(target_arch = "x86_64")]
499    crate::board::report_identity(&platform_identity);
500    // Bind the profile once, before the first `has_driver` gate. The operator
501    // names it on the Limine command line; detected facts may only reject it.
502    // The facts are kept with the profile for the machine fingerprint (433).
503    #[cfg(target_arch = "x86_64")]
504    {
505        let cmdline = crate::boot::limine::get_executable_cmdline(crate::board::CMDLINE_MAX + 1)
506            .unwrap_or(&[]);
507        match crate::board::choose_x86_profile(cmdline, &platform_identity) {
508            Ok((board, source)) => crate::board::bind_x86_profile(
509                board,
510                crate::board::X86BootFacts {
511                    identity: platform_identity,
512                    source,
513                    dmar_present: acpi_info.dmar_present,
514                },
515            ),
516            Err(error) => panic!("[board] {}", error),
517        }
518        let console_opt = crate::board::console_option(cmdline).unwrap_or(None);
519        let _ = crate::dev::console_select::select_and_apply_console(console_opt, &acpi_info);
520    }
521
522    #[cfg(target_arch = "x86_64")]
523    let x86_ioapic_base = if crate::board::selected().has_driver(cellos_boards::DriverId::IoApic) {
524        acpi_info.ioapic_base
525    } else {
526        0
527    };
528    #[cfg(target_arch = "x86_64")]
529    let x86_lapic_base = if x86_ioapic_base != 0 {
530        acpi_info.lapic_base
531    } else {
532        0
533    };
534    #[cfg(target_arch = "x86_64")]
535    let x86_hpet_base = if crate::board::selected().has_driver(cellos_boards::DriverId::Hpet) {
536        acpi_info.hpet_base
537    } else {
538        0
539    };
540    #[cfg(target_arch = "x86_64")]
541    let x86_ecam_base = if crate::board::selected().has_driver(cellos_boards::DriverId::PcieEcam) {
542        acpi_info.ecam_base
543    } else {
544        0
545    };
546    #[cfg(target_arch = "x86_64")]
547    let x86_ecam_len = if x86_ecam_base == 0 {
548        0
549    } else {
550        crate::dev::pcie_ecam::ecam_window_size(acpi_info.ecam_bus_start, acpi_info.ecam_bus_end)
551            .expect("validated ACPI MCFG range has a bounded ECAM window")
552    };
553    // The timer gate opens only after `init_timers` finds a clocksource
554    // (phase 02 of plan 261103-2218); HPET is one candidate, not a requirement.
555    #[cfg(target_arch = "x86_64")]
556    let x86_interrupts_ready = x86_lapic_base != 0 && x86_ioapic_base != 0;
557    #[cfg(target_arch = "x86_64")]
558    #[cfg_attr(feature = "g1-cpu-accounting", allow(unused_assignments))]
559    // G1 assigns or panics on every path
560    let mut x86_timer_ready = false;
561
562    // 3. Paging (Virtual Memory) Setup
563    // x86_64 bring-up: Limine's PML4 already maps RAM via HHDM and the kernel
564    // at 0xFFFFFFFF80000000. We skip building + activating our own page tables
565    // until the full x86_64 port (Phase 09). init_kernel_paging uses physical
566    // addresses as virtual pointers, which would fault under Limine's paging.
567    #[cfg(not(any(
568        target_arch = "riscv32",
569        target_arch = "x86_64",
570        target_arch = "x86",
571        target_arch = "arm",
572    )))]
573    {
574        let mut frame_allocator = frame_allocator;
575        log_info("Initializing paging...");
576        let root_table_phys =
577            memory::paging::init_kernel_paging(&mut frame_allocator, mmap_entries)
578                .expect("Failed to initialize paging");
579        log_info("Paging initialized");
580        log_info("Activating paging...");
581        unsafe {
582            memory::paging::activate_paging(root_table_phys);
583        }
584        *memory::paging::KERNEL_ROOT.lock() = Some(root_table_phys);
585        #[cfg(target_arch = "aarch64")]
586        crate::hal::domain::record_kernel_ttbr0(root_table_phys);
587        #[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
588        TEST_SAFE_ROOT.store(root_table_phys, core::sync::atomic::Ordering::Release);
589        unsafe {
590            core::ptr::write(
591                &mut *memory::frame::FRAME_ALLOCATOR.lock(),
592                Some(frame_allocator),
593            );
594        }
595        log_info("Paging activated");
596        log_info("Frame allocator initialized");
597        // Set sstatus.SUM=1 so S-mode (kernel) can access USER-mapped pages throughout
598        // the kernel lifetime. VirtIO/peripheral MMIO is mapped USER=1 for Driver Cells
599        // (U-mode). Without SUM=1 the kernel's tech-debt VirtIO drivers fault at early-boot
600        // MMIO init. In Cellos's SAS+LBI model security comes from Rust type safety, not
601        // hardware USER-bit separation for kernel-vs-cell — SUM=1 is safe and intentional.
602        #[cfg(target_arch = "riscv64")]
603        // SAFETY: csrs modifies sstatus.SUM (bit 18). Safe to set for kernel S-mode code.
604        unsafe {
605            core::arch::asm!("csrs sstatus, {sum}", sum = in(reg) 0x40000_usize, options(nostack));
606        }
607    }
608    #[cfg(target_arch = "x86_64")]
609    {
610        // Set runtime ECAM base from validated ACPI before PCIe scan. Zero
611        // keeps PCIe closed; there is no implicit q35 fallback.
612        crate::dev::pcie_ecam::set_ecam_base_x86(x86_ecam_base as usize);
613        crate::dev::pcie_ecam::set_ecam_bus_range_x86(
614            acpi_info.ecam_bus_start,
615            acpi_info.ecam_bus_end,
616        );
617
618        log_info("Initializing x86_64 paging (kernel PML4)...");
619        let root_table_phys = {
620            let mut locked_frame_allocator = memory::frame::FRAME_ALLOCATOR.lock();
621            memory::paging::init_kernel_paging_x86(
622                locked_frame_allocator
623                    .as_mut()
624                    .expect("Frame allocator not initialized"),
625                x86_ioapic_base,
626                x86_hpet_base,
627                x86_lapic_base,
628                x86_ecam_base,
629                x86_ecam_len,
630            )
631            .expect("Failed to initialize x86_64 kernel PML4")
632        };
633        log_info("x86_64 paging initialized");
634        log_info("Activating x86_64 paging (mov cr3)...");
635        // SAFETY: init_kernel_paging_x86 copied higher-half entries from Limine's PML4
636        // (preserving kernel text/data/HHDM) and identity-mapped MMIO, so the kernel
637        // continues executing after this CR3 switch without a triple-fault.
638        unsafe {
639            memory::paging::activate_paging(root_table_phys);
640        }
641        // Immediate configured port-I/O probe after activate_paging. If 'Q'
642        // appears, the CR3 switch returned to kmain with the UART mechanism live.
643        crate::hal::uart_16550::putchar(b'Q');
644        // The G1 private lane keeps its HPET-only contract.
645        #[cfg(feature = "g1-cpu-accounting")]
646        let x86_timer_candidates = x86_interrupts_ready && x86_hpet_base != 0;
647        #[cfg(not(feature = "g1-cpu-accounting"))]
648        let x86_timer_candidates = x86_interrupts_ready;
649        if x86_timer_candidates {
650            crate::hal::apic::set_lapic_phys(x86_lapic_base);
651            crate::hal::apic::set_irq_overrides(
652                &acpi_info.irq_overrides,
653                &acpi_info.irq_override_flags,
654            );
655            register_x86_ioapics();
656            crate::hal::set_hpet_base(x86_hpet_base as usize);
657            #[cfg(not(feature = "g1-cpu-accounting"))]
658            {
659                offer_x86_clock_candidates(&acpi_info);
660                match crate::hal::init_timers() {
661                    Some(setup) => {
662                        x86_timer_ready = true;
663                        log::info!(
664                            "[x86-gate] timer: clock={} clockevent={:?} apic={}",
665                            setup.source.name(),
666                            setup.clockevent,
667                            if setup.x2apic { "x2apic" } else { "xapic" }
668                        );
669                    }
670                    None => log::error!(
671                        "[x86-gate] timer CLOSED: no clocksource (TSC, kvmclock, HPET or PM timer)"
672                    ),
673                }
674            }
675            #[cfg(feature = "g1-cpu-accounting")]
676            {
677                let vendor = core::arch::x86_64::__cpuid(0);
678                assert!(
679                    vendor.ebx == u32::from_le_bytes(*b"Genu")
680                        && vendor.edx == u32::from_le_bytes(*b"ineI")
681                        && vendor.ecx == u32::from_le_bytes(*b"ntel"),
682                    "G1 private lane requires GenuineIntel CPUID"
683                );
684                let calibration = crate::hal::init_timers_observed()
685                    .expect("G1 requires actual HPET/LAPIC timer initialization");
686                let clock = crate::hal::hpet::checked_sample()
687                    .expect("G1 requires a checked shared 64-bit HPET");
688                let initial = calibration
689                    .ticks_per_ms
690                    .checked_mul(10)
691                    .and_then(|count| u32::try_from(count).ok())
692                    .expect("G1 rejects overflowed LAPIC calibration");
693                assert!(
694                    calibration.hpet_used
695                        && initial >= 10_000
696                        && initial == calibration.initial_count
697                        && calibration.lvt & (1 << 16) == 0
698                        && (calibration.lvt >> 17) & 3 == 1,
699                    "G1 rejects LAPIC fallback, stopped or masked timer"
700                );
701                // The G1 lane's boot receipt: emitted raw and allocation-free,
702                // starting its own line, without a log-level tag.
703                struct G1BootWriter;
704                impl core::fmt::Write for G1BootWriter {
705                    fn write_str(&mut self, text: &str) -> core::fmt::Result {
706                        for byte in text.bytes() {
707                            crate::hal::uart_16550::putchar(byte);
708                        }
709                        Ok(())
710                    }
711                }
712                core::fmt::write(&mut G1BootWriter, format_args!(
713                    "\nG1-CLOCK-BOOT raw={} period_fs={} base={} caps={} config={} ticks_per_ms={} initial={} lvt={}\n",
714                    clock.raw_counter, clock.period_fs, clock.base, clock.capabilities,
715                    clock.configuration, calibration.ticks_per_ms,
716                    calibration.initial_count, calibration.lvt,
717                )).expect("G1 boot receipt formatting");
718                x86_timer_ready = true;
719                log_info("x86_64 timers initialized from ACPI (HPET + LAPIC)");
720            }
721        } else {
722            // Fail-closed decision: it must reach the console.
723            log::error!("[x86-gate] timer CLOSED: validated MADT required");
724            #[cfg(feature = "g1-cpu-accounting")]
725            panic!("G1 private measurement requires validated MADT and HPET");
726        }
727
728        // Tier 3 x86 VMM P01: enter root-of-virtualization on the BSP.
729        // SVM first (TCG-testable); VMX only on genuine Intel (KVM/HW lane).
730        // Failure is non-fatal — the kernel runs fine without virt; the
731        // HypervisorCap gate simply stays closed (has_x86_virt() == false).
732        // The two backends differ in how a refusal surfaces, and that decides
733        // what may be attempted under a hypervisor:
734        //   * VMXON **faults** (#GP) when the host grants the bits but refuses
735        //     nested operation, so VMX is never attempted under a hypervisor —
736        //     an optional capability must not take the boot down.
737        //   * SVM's enable path only writes EFER.SVME and VM_HSAVE_PA, both
738        //     gated on the CPUID bit, and reports failure instead of faulting.
739        //     QEMU advertises CPUID.1:ECX[31] on every builtin CPU model while
740        //     still emulating SVM root operation under TCG, so gating SVM on
741        //     the hypervisor bit would close the capability on the only lane
742        //     that can exercise it (the x86 hypervisor smoke lane).
743        let _under_hypervisor = cpu_features::hypervisor_present();
744        // Root operation is per-CPU: the same helper runs on each AP before it
745        // goes online (crate::sched::smp::x86_secondary_main).
746        if cpu_features::x86_virt_kind() == Some(cpu_features::X86Virt::Vmx) {
747            // VMXON faults (#GP) rather than reporting failure when a
748            // precondition is unmet; logged first, a fault stays attributable.
749            log::info!(
750                "[kernel] x86 virt: VMX preconditions CPUID={} FEATURE_CONTROL={:#x} VMX_BASIC_revid={:#x}",
751                hal::vmx::supported(),
752                hal::vmx::feature_control(),
753                hal::vmx::vmxon_revision()
754            );
755        }
756        match cpu_features::enter_x86_root_on_current_cpu() {
757            Ok(kind) => {
758                cpu_features::latch_x86_root_active();
759                log_info(match kind {
760                    cpu_features::X86Virt::Svm => "x86 virt: root operation active (SVM)",
761                    cpu_features::X86Virt::Vmx => "x86 virt: root operation active (VMX)",
762                });
763            }
764            Err(reason) => {
765                puts("[INFO] x86 virt: ");
766                puts(reason);
767                puts("; cap closed\n");
768            }
769        }
770    }
771    // Bare physical: RV32 Nano (SATP=0), x86_32 (CR0.PG=0), AArch32 (MMU off).
772    #[cfg(any(target_arch = "riscv32", target_arch = "x86", target_arch = "arm"))]
773    {
774        memory::paging::init_bare();
775        log_info("Paging: bare physical");
776    }
777
778    // 4. Heap Allocator (Global) - MUST be after paging but before any allocations
779    // 4 MiB = 1024 frames (reduced from 32 MiB / 8192 frames under Phase 01 memory footprint reduction).
780    // Sized to hold kernel structures, scheduler state, IPC buffers, and cell signing scratch.
781    #[cfg(target_pointer_width = "64")]
782    const HEAP_FRAMES: usize = 8_192; // 32 MiB
783    #[cfg(target_pointer_width = "32")]
784    const HEAP_FRAMES: usize = 1_024; // 4 MiB
785    let heap_start = {
786        let mut allocator_guard = memory::frame::FRAME_ALLOCATOR.lock();
787        let allocator = allocator_guard
788            .as_mut()
789            .expect("Frame allocator not initialized");
790        // One contiguous reservation: the heap below is initialized over
791        // `HEAP_FRAMES * 4096` bytes starting at this frame, so a scattered
792        // allocation would write over frames this kernel does not own. Refusing
793        // here is the only safe outcome — never a smaller or fragmented heap.
794        // The run is taken in the bitmap's range, right after the bitmap and the
795        // early frames there: the shared `[bitmap, heap)` registration below
796        // depends on that order.
797        memory::frame::reserve_boot_heap(allocator, HEAP_FRAMES).expect(
798            "OOM: no contiguous run for the boot heap in the frame bitmap's range — \
799             refusing to boot with a fragmented heap",
800        )
801    };
802    let heap_size = HEAP_FRAMES * 4096;
803    // On x86_64, phys_to_virt adds HHDM offset (Limine maps RAM at HHDM+phys).
804    // On RISC-V, phys_to_virt returns phys unchanged (identity-mapped before paging).
805    let heap_virt = memory::frame::phys_to_virt(heap_start);
806    unsafe {
807        memory::heap::init_heap(heap_virt, heap_size);
808    }
809    log_info("Heap initialized");
810
811    memory::rt_heap::init();
812    log_info("RT heap initialized");
813
814    // 5. Hardware Abstraction Layer (HAL) Initialization
815    // GDT/IDT/SYSCALL already done at step 1. Initialize PLIC for RISC-V external IRQs.
816    #[cfg(target_arch = "riscv64")]
817    if crate::board::active().has_driver(cellos_boards::DriverId::PlicSifive) {
818        if let Some((context, irqs, irq_count)) = crate::platform::riscv_plic_init_data() {
819            crate::hal::common::plic::init(context, &irqs[..irq_count]);
820        } else {
821            log::warn!("[plic] no active RV64 context mapping; external IRQs stay disabled");
822        }
823    } else {
824        log::info!("[plic] disabled by active board descriptor");
825    }
826    log_info("HAL initialized (PLIC enabled)");
827
828    // 6. Logger & Drivers & FS
829    // Registers the log backend on AArch64/ARM; RV64 and x86_64 registered it
830    // at entry, so the call is a no-op there.
831    crate::dev::uart::init();
832    #[cfg(target_arch = "aarch64")]
833    {
834        use hal::backend::{ArmEl2Monitor, El2Monitor};
835        if El2Monitor::PRESENT {
836            let current_el: u64;
837            unsafe {
838                core::arch::asm!("mrs {el}, CurrentEL", el = out(reg) current_el,
839                                 options(nomem, nostack));
840            }
841            log::info!(
842                "[pi-monitor] entry EL={} host EL={} HVC ready={}",
843                El2Monitor::entry_el(),
844                current_el >> 2,
845                El2Monitor::is_ready()
846            );
847            let (booted_el2, hvc_result, op0_seen) = El2Monitor::probe_status();
848            log::info!(
849                "[pi-monitor] probe booted_el2={} hvc_result=0x{:x} op0_seen={}",
850                booted_el2,
851                hvc_result,
852                op0_seen
853            );
854            crate::hypervisor::registry::pi_monitor_smoke();
855        }
856    }
857
858    // Phase 07: reserve the warm-snapshot capture's scratch workspace from the
859    // frame allocator — the capture stages its own stack and buffers there, so
860    // they can never be inside a captured run.  After the frame allocator and
861    // the heap (and now the log backend), before any cell or capture can run.
862    // A no-op unless the build is `snapshot-qualified`; a build without a
863    // workspace refuses every capture.
864    boot::reserve_snapshot_scratch();
865    #[cfg(target_arch = "aarch64")]
866    if matches!(
867        <hal::backend::Board as hal::backend::ArmPlatform>::DTB_POLICY,
868        hal::backend::DtbPolicy::Optional
869    ) {
870        let (start, end) = boot::fallback_dtb_ram_range();
871        log::info!("[boot] DTB RAM range {:#x}..{:#x}", start, end);
872    }
873    #[cfg(target_arch = "riscv64")]
874    crate::dev::uart::init_input();
875    #[cfg(target_arch = "aarch64")]
876    if crate::hal::backend::Board::CONSOLE_RX_IRQ {
877        // IRQ delivery must start only after the heap-backed queue exists.
878        crate::dev::uart::init_input();
879        crate::hal::backend::Board::console_enable_rx_irq();
880    }
881    // RV32 Nano / x86_64 bring-up: skip VirtIO MMIO probing (PCIe transport not yet ported).
882    // x86_64 gets VirtIO via the PCI path in virtio_pci::init() below.
883    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
884    crate::dev::init();
885    // x86_64: load embedded kernel_fs.img into RAM so EarlyLoader can serve ELFs from it.
886    // VirtIO devices on q35 are on PCIe — probed via virtio_pci::init() after ECAM scan.
887    #[cfg(target_arch = "x86_64")]
888    {
889        crate::dev::ramdisk::init_driver();
890        crate::dev::ps2_keyboard::init();
891        // COM1 TX remains available without ACPI. RX IRQ requires the timer
892        // gate (validated MADT plus a clocksource) and an I/O APIC that serves
893        // the UART's GSI, because it is routed through IOAPIC/LAPIC.
894        if !x86_timer_ready {
895            log::warn!("[x86-gate] configured UART IRQ CLOSED: interrupt/timer unavailable");
896        } else if !crate::hal::uart_16550::init_input_irq() {
897            log::warn!("[x86-gate] configured UART IRQ CLOSED: no I/O APIC serves its GSI");
898        }
899        // Initialise the RX buffer that vi_handle_uart_irq() writes into.
900        crate::dev::uart::init_input();
901        // Phase 06: name every declared serial port and whether the machine
902        // actually answered the 16550 probe. The console is slot 0; the extras
903        // are polled, so an absent one is a refusal, never a silent write.
904        {
905            let soc = crate::board::selected_x86_64_soc();
906            for (index, port) in soc.serial_ports.iter().enumerate() {
907                match crate::hal::uart_16550::port_base_at(index) {
908                    Some(base) => log::info!(
909                        "[x86-gate] serial port {} base={:#x} irq={} usable",
910                        index,
911                        base,
912                        port.irq
913                    ),
914                    None => log::warn!(
915                        "[x86-gate] serial port {} base={:#x} irq={} absent (16550 probe failed)",
916                        index,
917                        port.base,
918                        port.irq
919                    ),
920                }
921            }
922            if soc.serial_ports.len() > crate::hal::uart_16550::MAX_PORTS {
923                log::warn!(
924                    "[x86-gate] {} declared serial ports exceed the {} this kernel carries",
925                    soc.serial_ports.len(),
926                    crate::hal::uart_16550::MAX_PORTS
927                );
928            }
929        }
930        log_info("x86_64: ramdisk + UART RX IRQ initialised");
931    }
932
933    // PCIe ECAM scan + NVMe + e1000 + VirtIO PCI init.
934    // ARM64 virt uses VirtIO MMIO (not PCIe); accessing 0x3F000000 on QEMU
935    // virt 7+ triggers a Synchronous External Abort — skip on aarch64.
936    // x86_64: the complete validated ACPI MCFG window is identity-mapped;
937    // the early kernel fallback scans its first admitted bus before Platform.
938    #[cfg(any(target_arch = "riscv64", target_arch = "x86_64"))]
939    {
940        #[cfg(target_arch = "x86_64")]
941        if x86_ecam_base != 0 {
942            // Early first-admitted-bus fallback: seed kernel PCI identity and
943            // activate VT-d before any DMA-capable Driver Cell can run. The
944            // Platform Cell receives the full validated MCFG range via argv
945            // later and performs authoritative multi-bus enumeration.
946            crate::dev::pcie_ecam::init();
947            crate::dev::iommu::init(crate::dev::iommu::DmaIsolationInput {
948                vtd_base: acpi_info.dmar_base,
949                vtd_units: acpi_info.dmar_units,
950                vtd_include_pci_all: acpi_info.dmar_include_pci_all,
951            });
952            crate::dev::iommu::activate_isolation();
953            // FADT IAPC_BOOT_ARCH bit 3: firmware says MSI must not be enabled.
954            crate::dev::pcie_msi::init(
955                acpi_info
956                    .fadt
957                    .as_ref()
958                    .is_some_and(|fadt| fadt.iapc_boot_arch & (1 << 3) != 0),
959            );
960        } else {
961            log::error!("[x86-gate] PCIe CLOSED: validated MCFG required");
962        }
963        #[cfg(target_arch = "riscv64")]
964        crate::dev::iommu::set_deferred_init_pending();
965        // VirtIO PCI block init removed (G2 loader redesign phase 06). x86 block I/O
966        // is served by the NVMe Driver Cell (F4); the kernel drives no block hardware.
967        // activate_isolation() is now called inside iommu::try_deferred_init() once the
968        // IOMMU device has been registered by the Platform Cell. The call below is a no-op
969        // (IOMMU not yet initialized at this point in boot).
970        #[cfg(target_arch = "riscv64")]
971        crate::dev::iommu::activate_isolation();
972    }
973
974    // Attempt warm boot from snapshot before any cell initialization.
975    // RV32 Nano / x86_64 skip: no VirtIO block in bring-up.
976    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
977    if snapshot::try_restore() {
978        // try_restore() called yield_cpu() and should not return in a successful
979        // warm boot.  If we reach here, fall through to cold boot as a safety net.
980        log::warn!("[boot] snapshot restore returned unexpectedly → cold boot");
981    }
982
983    // Cross-check the on-disk MBR against the compiled-in partition layout
984    // (warn-only — surfaces image/kernel drift at boot instead of as silent
985    // corruption later).
986    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
987    crate::loader::disk_layout::verify_mbr();
988
989    // Probe the cell bootstrap table so SpawnFromPath works during init.
990    // RV32 Nano / x86_64 bring-up: no disk.
991    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
992    match crate::loader::early::EarlyLoader::probe() {
993        Ok(()) => puts("[loader] cell bootstrap table loaded\n"),
994        Err(_) => {
995            puts("[loader] WARN: cell table not found — disk image may lack bootstrap section\n")
996        }
997    }
998
999    // RV32 Nano: no FAT32 FS in bring-up.
1000    // x86_64 uses the ramdisk-backed embedded FS to serve cell ELFs via VIFS1.
1001    #[cfg(any(
1002        target_arch = "riscv64",
1003        target_arch = "aarch64",
1004        target_arch = "x86_64"
1005    ))]
1006    fs::init();
1007
1008    #[cfg(any(
1009        target_arch = "riscv64",
1010        target_arch = "aarch64",
1011        target_arch = "x86_64"
1012    ))]
1013    system_config::load_from_vifs1();
1014
1015    // Load + verify the signed operator policy (P5b) NOW: after VIFS1 is mounted,
1016    // before any cap-bearing cell spawns. Absent → dev-permissive (this G1 build);
1017    // invalid → fail-closed. Phase 04 folds policy::lookup into the spawn grant.
1018    #[cfg(any(
1019        target_arch = "riscv64",
1020        target_arch = "aarch64",
1021        target_arch = "x86_64"
1022    ))]
1023    policy::load_from_vifs1();
1024
1025    // Phase 20: hot-migration state-transfer self-test.
1026    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
1027    crate::cell::state_stash::self_test();
1028
1029    log_info("Kernel subsystems initialized successfully.");
1030
1031    // 7. Initialize Scheduler
1032    log_info("Initializing scheduler...");
1033    task::init();
1034    log_info("Scheduler initialized");
1035    #[cfg(feature = "test-hooks")]
1036    if crate::sched::scheduler::death_subscription_quota_self_test() {

1555            // Only meaningful when SVM root operation was entered at boot.

1557            if cpu_features::has_x86_virt() {
1558                crate::hypervisor::svm_registry::x86_smoke();
1559            } else {
1560                log_info("X86-VMM-SMOKE: SKIP (no SVM/VMX root)");
1561            }
1562        }

1564        if cpu_features::has_x86_virt() {
1565            crate::hypervisor::svm_registry::x86_mmio_smoke();
1566        } else {
1567            log_info("X86-MMIO-SMOKE: SKIP (no SVM/VMX root)");
1568        }

1570        if let Some(fb) = boot_info.framebuffer() {
1571            let hhdm = boot_info.hhdm_offset();
1572            let phys_addr = if fb.addr >= hhdm && hhdm > 0 {
1573                fb.addr - hhdm
1574            } else {
1575                fb.addr
1576            };
1577            let width = fb.width as u16;
1578            let height = fb.height as u16;
1579            let pitch = fb.pitch as usize;
1580            let size = pitch.wrapping_mul(height as usize);
1581            let pages = size.div_ceil(4096);
1582            let owner = types::CellId(0);
1583            match crate::memory::paging::map_display_framebuffer_user(phys_addr, pages) {
1584                Ok(_) => {
1585                    let _ = crate::resource_registry::reserve_display_framebuffer(
1586                        owner, phys_addr, size, width, height, pitch,
1587                    );
1588                    let _ = crate::resource_registry::activate_display_framebuffer(
1589                        owner, phys_addr, size, width, height, pitch,
1590                    );
1591                    log_info("[boot] x86_64 Limine linear framebuffer mapped and registered");
1592                    crate::dev::fb_console::init(fb.addr, width as usize, height as usize, pitch);
1593                    unsafe {
1594                        let ptr = fb.addr as *mut u32;
1595                        let stride = pitch / 4;
1596                        let total = stride * (height as usize);
1597                        // Dark slate background (0xFF1E1E2E)
1598                        for i in 0..total {
1599                            *ptr.add(i) = 0xFF1E1E2E;
1600                        }
1601                        // Top banner bar (40px height, Cyan 0xFF00A8E8)
1602                        for y in 0..40usize {
1603                            for x in 0..(width as usize) {
1604                                *ptr.add(y * stride + x) = 0xFF00A8E8;
1605                            }
1606                        }
1607                    }
1608                }
1609                Err(_) => {
1610                    log_info("[boot] failed to map linear framebuffer");
1611                }
1612            }
1613        }

1615        // Spawn the Platform Cell before init (PCIe ECAM scanner).
1616        // RISC-V uses the compile-time GPEX window (0x3000_0000).
1617        // x86_64 hands the validated runtime ACPI-MCFG base and bus range via argv.
1618        // Failure is non-fatal: kernel-side PCI_DEVICES stays empty; Driver Cells
1619        // that rely on sys_find_pcie_device will simply not find their device.
1620        #[cfg(target_arch = "riscv64")]
1621        match crate::loader::spawn_from_path(
1622            "/bin/platform",
1623            crate::loader::SpawnRequest::governed_boot(),
1624        ) {
1625            Ok(_) => log_info("Platform Cell spawned (PCIe ECAM scanner)"),
1626            Err(_) => log_info("Platform Cell absent — PCIe BARs will not be pre-registered"),
1627        }
1628
1629        #[cfg(target_arch = "x86_64")]
1630        if x86_ecam_base != 0 {
1631            let argv_str = alloc::format!(
1632                "--ecam-base={:#x} --bus-start={} --bus-end={}",
1633                x86_ecam_base,
1634                acpi_info.ecam_bus_start,
1635                acpi_info.ecam_bus_end,
1636            );
1637            match crate::loader::spawn_from_path(
1638                "/bin/platform",
1639                crate::loader::SpawnRequest::governed_boot().with_argv(argv_str.into_bytes()),
1640            ) {
1641                Ok(_) => log_info("Platform Cell spawned (x86 PCIe ECAM scanner)"),
1642                Err(_) => log_info("Platform Cell absent — PCIe BARs will not be pre-registered"),
1643            }
1644        }

1646        // `aligned_elf::bytes` borrows an already aligned embedded image and only
1647        // materializes an aligned copy when the linker did not provide one. Do not
1648        // unconditionally duplicate init on the kernel heap at this boot boundary.
1649        #[cfg(feature = "test-hooks")]
1650        {
1651            log_info("ATOMIC_PUBLICATION_AP-15: arming trusted init");
1652            crate::loader::atomic_publication_tests::arm_trusted_success();
1653            log_info("ATOMIC_PUBLICATION_AP-15: armed for trusted init");
1654        }
1655        log_info("Publishing trusted init");
1656        #[cfg(feature = "p01-trace")]
1657        let init_result = crate::loader::spawn_bootstrap_init(INIT_ELF);
1658        #[cfg(not(feature = "p01-trace"))]
1659        let init_result = crate::loader::spawn_trusted_init(INIT_ELF);
1660        match init_result {
1661            Ok(tid) => {
1662                #[cfg(feature = "test-hooks")]
1663                crate::loader::atomic_publication_tests::finish_trusted_success(tid);
1664                log::info!(
1665                    "Successfully spawned init with complete root authority (tid={})",
1666                    tid
1667                );
1668                #[cfg(target_arch = "aarch64")]
1669                crate::hal::backend::Board::diag_probe(b'V');
1670            }
1671            Err(_e) => {
1672                log_info("Failed to spawn init");
1673                // Failure occurs before init reaches any ready queue.
1674                #[cfg(target_arch = "aarch64")]
1675                crate::hal::backend::Board::diag_probe(b'F');
1676                #[cfg(feature = "test-hooks")]
1677                panic!("atomic-publication trusted-init success contract did not publish init");
1678            }
1679        }
1680    }

1682    // Ring-3 smoke test: spawn a minimal U-mode task that logs and exits.
1683    // RISC-V only — task writes RISC-V machine code directly.
1684    // Expected serial output: "Hi from U-mode!\n" followed by task exit.
1685    #[cfg(all(target_arch = "riscv64", feature = "test-hooks"))]
1686    match task::user_hello::spawn() {
1687        Ok(tid) => {
1688            puts("[task] spawning user_hello at ");
1689            // Print tid as decimal (max 20 digits for usize)
1690            let mut buf = [0u8; 20];
1691            let mut n = tid;
1692            let mut i = 20usize;
1693            if n == 0 {
1694                i -= 1;
1695                buf[i] = b'0';
1696            } else {
1697                while n > 0 {
1698                    i -= 1;
1699                    buf[i] = b'0' + (n % 10) as u8;
1700                    n /= 10;
1701                }
1702            }
1703            let _ = core::str::from_utf8(&buf[i..]).map(puts);
1704            puts("\n");
1705            let _ = tid; // suppress unused warning
1706        }
1707        Err(_) => log_info("[task] user_hello spawn failed"),
1708    }
1709
1710    // Deliberately underflow a U-mode test task into the upper guard page. The
1711    // fault handler must terminate only that task and let boot continue.
1712    #[cfg(all(target_arch = "riscv64", feature = "test-hooks"))]
1713    if task::stack_overflow_probe::spawn().is_err() {
1714        log_info("[stack-guard] deliberate overflow probe spawn FAIL");
1715    }
1716
1717    log_info("Kernel initialization complete. Entering idle loop.");
1718
1719    // 9. Start multitasking
1720    log_info("Starting scheduler...");
1721
1722    // Quiet the shared console for interactive use. Kernel bring-up is done; the
1723    // remaining Info chatter is per-spawn noise ([loader] SpawnFromPath, Spawn:,
1724    // ELF LOAD) that floods the UART and buries the shell prompt. WARN/ERROR still
1725    // surface real problems. Raise back to Info when debugging the spawn path.
1726
1727    // Enable interrupts before entering the idle loop.
1728    // RISC-V: set SPP=1 and SIE=1 in sstatus (0x102).
1729    // AArch64: clear DAIF.I bit to unmask IRQs.
1730    // x86_64: STI via ARCH.enable_interrupts().
1731    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1732    unsafe {
1733        // SAFETY: csrs sstatus SPP|SIE from S-mode — standard interrupt enable.
1734        core::arch::asm!("csrs sstatus, {0}", in(reg) 0x102usize);
1735    }
1736    #[cfg(target_arch = "aarch64")]
1737    unsafe {
1738        core::arch::asm!("msr daifclr, #2", options(nomem, nostack));
1739    }
1740    // x86_64, x86_32, AArch32: use the Arch trait's enable_interrupts().
1741    #[cfg(any(target_arch = "x86_64", target_arch = "x86", target_arch = "arm"))]
1742    crate::hal::ARCH.enable_interrupts();
1743
1744    // Test images keep kernel Info live.
1745    //
1746    // The quieting above exists for the interactive console, and production keeps
1747    // it. A test image cannot: the witnesses a domain-entry lane reads are emitted
1748    // *after* init and the cells are running, which is the point of the lane. On
1749    // AArch64 the boot happens to never reach this line with a task pending (the
1750    // interrupt enable just above drops straight into the scheduler), so its lane
1751    // has always read Info-level witnesses (`S22-AARCH64-DOMAIN-LIVE`,
1752    // `DOMAIN-FRAME-RELEASE`). On x86_64 the boot does continue into the idle loop,
1753    // so the same class of witnesses (`S22-X86-DOMAIN-LIVE`,
1754    // `S22-X86-DOMAIN-TEARDOWN`, `[selftest] DOMAIN-FRAME-RELEASE`) would be
1755    // silenced here. `test-hooks` only — no production image is affected.
1756    //
1757    // It *replaces* the quieting rather than undoing it afterwards: which of the
1758    // two statements ran first used to be a race — kmain's tail executes only
1759    // after the scheduler hands the boot context back — and a witness emitted
1760    // before the quieting landed was dropped, so the same image could lose or keep
1761    // a marker between runs.
1762    // AArch64 test images are in the same position as x86_64's, and for a
1763    // sharper reason: on one CPU the boot happens to never reach this line with a
1764    // task pending, so the witnesses were emitted *before* the quieting by luck.
1765    // The moment a second hart is online the boot does reach it and the
1766    // `S22-AARCH64-DOMAIN-*` witnesses disappear — a lane reading them fails on
1767    // an image that is working. Quieting is a console policy for interactive and
1768    // production boots; a test image is read by its own lane.
1769    // An explicit [logging] is authoritative for the whole cold boot. Only old
1770    // images / omitted sections retain this later interactive/test default.
1771    if !system_config::logging_explicit() {
1772        #[cfg(all(
1773            feature = "test-hooks",
1774            any(target_arch = "x86_64", target_arch = "aarch64")
1775        ))]
1776        log::set_max_level(log::LevelFilter::Info);
1777        #[cfg(not(all(
1778            feature = "test-hooks",
1779            any(target_arch = "x86_64", target_arch = "aarch64")
1780        )))]
1781        log::set_max_level(log::LevelFilter::Warn);
1782    }
1783
1784    // Probe 'Q': fires ONLY if no IRQ preempted the code between daifclr and here.
1785    // Q fires  → BCM2835 IRQ is not actually delivered to CPU on QEMU (pending but silent).
1786    // Q absent → IRQ fired immediately after daifclr (expected); check if H/G appear.
1787    #[cfg(target_arch = "aarch64")]
1788    crate::hal::backend::Board::diag_probe(b'Q');
1789
1790    // Board IRQ-state snapshot immediately after enabling IRQs (RPi3: the
1791    // "K<src_bit8><pend_bit1><src_nibble>" probe; a no-op elsewhere).
1792    #[cfg(target_arch = "aarch64")]
1793    crate::hal::backend::Board::irq_snapshot_probe();
1794
1795    // Probe 'L': fires once per idle loop iteration, for the board's first
1796    // `IDLE_PROBES` iterations only (RPi3: 4) to avoid flood.
1797    // If 'L' never appears after 'K', the code never reaches the idle loop.
1798    #[cfg(target_arch = "aarch64")]
1799    if <hal::backend::Board as hal::backend::ArmPlatform>::IDLE_PROBES != 0 {
1800        static IDLE_COUNT: core::sync::atomic::AtomicUsize =
1801            core::sync::atomic::AtomicUsize::new(0);
1802        loop {
1803            let n = IDLE_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
1804            // reason: `IDLE_PROBES` is a per-board const; on boards with no idle
1805            // probe it is 0, and this whole block is then dead.
1806            #[allow(clippy::absurd_extreme_comparisons)]
1807            if n < <hal::backend::Board as hal::backend::ArmPlatform>::IDLE_PROBES {
1808                crate::hal::backend::Board::diag_probe(b'L');
1809            }
1810            crate::task::yield_cpu();
1811            crate::hal::ARCH.wait_for_interrupt();
1812        }
1813    }
1814    loop {
1815        crate::task::yield_cpu();
1816        crate::hal::ARCH.wait_for_interrupt();
1817    }
1818}

1820/// Emit an allocation-free RV64 fault snapshot before the panic path.
1821#[cfg(all(feature = "test-hooks", target_arch = "riscv64"))]
1822#[no_mangle]
1823pub extern "C" fn vi_rv64_kernel_fault_snapshot(satp: usize, stval: usize) {
1824    fn put_hex(value: usize) {
1825        const HEX: &[u8; 16] = b"0123456789abcdef";
1826        for shift in (0..16).rev().map(|n| n * 4) {
1827            let _ = crate::hal::sbi::console_putchar(HEX[(value >> shift) & 0xf]);
1828        }
1829    }
1830
1831    fn put_label(label: &[u8], value: usize) {
1832        for byte in label {
1833            let _ = crate::hal::sbi::console_putchar(*byte);
1834        }
1835        put_hex(value);
1836    }
1837
1838    unsafe fn read_pte(table_phys: usize, index: usize) -> usize {
1839        (table_phys as *const usize).add(index).read_volatile()
1840    }
1841
1842    fn next_table(entry: usize) -> usize {
1843        ((entry >> 10) & 0x003f_ffff_ffff_ffff) << 12
1844    }
1845
1846    for byte in b"\n[selftest] RV64-FAULT satp=0x" {
1847        let _ = crate::hal::sbi::console_putchar(*byte);
1848    }
1849    put_hex(satp);
1850    for byte in b" stval=0x" {
1851        let _ = crate::hal::sbi::console_putchar(*byte);
1852    }
1853    put_hex(stval);
1854
1855    let kernel_root = TEST_SAFE_ROOT.load(core::sync::atomic::Ordering::Acquire);
1856    if kernel_root != 0 {
1857        let kernel_satp = (8usize << 60) | (kernel_root >> 12);
1858        // SAFETY: this test-only panic snapshot switches to the known-safe
1859        // shared root so it can inspect private page-table memory by physical VA.
1860        unsafe {
1861            core::arch::asm!(
1862                "csrw satp, {satp}",
1863                "sfence.vma zero, zero",
1864                satp = in(reg) kernel_satp,
1865                options(nostack)
1866            );
1867        }
1868        let root_phys = (satp & ((1usize << 44) - 1)) << 12;
1869        let vpn2 = (stval >> 30) & 0x1ff;
1870        let vpn1 = (stval >> 21) & 0x1ff;
1871        let vpn0 = (stval >> 12) & 0x1ff;
1872        let pte2 = unsafe { read_pte(root_phys, vpn2) };
1873        let pte1 = if pte2 & 1 != 0 {
1874            unsafe { read_pte(next_table(pte2), vpn1) }
1875        } else {
1876            0
1877        };
1878        let pte0 = if pte1 & 1 != 0 {
1879            unsafe { read_pte(next_table(pte1), vpn0) }
1880        } else {
1881            0
1882        };
1883        put_label(b" pte2=0x", pte2);
1884        put_label(b" pte1=0x", pte1);
1885        put_label(b" pte0=0x", pte0);
1886    }
1887}

1889/// Non-recoverable kernel panic handler.
1890///
1891/// `current_cell_id` is allocation attribution, not an execution-mode proof:
1892/// kernel code can hold scheduler or other kernel locks while servicing a
1893/// Cell. Therefore this handler never schedules or retires a Cell. Only the
1894/// architecture trap path, after proving an interrupted U-mode context, may
1895/// enter the deferred retirement funnel.
1896#[cfg(not(all(test, not(target_os = "none"))))]
1897#[panic_handler]
1898fn panic(info: &PanicInfo) -> ! {
1899    // True kernel panic: print diagnostics and halt.
1900    #[inline(always)]
1901    fn panic_putchar(c: u8) {
1902        #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1903        {
1904            let _ = crate::hal::sbi::console_putchar(c);
1905        }
1906        #[cfg(target_arch = "aarch64")]
1907        {
1908            crate::hal::backend::Board::early_putc(c);
1909        }
1910        #[cfg(target_arch = "arm")]
1911        {
1912            crate::hal::uart_pl011::putchar(c);
1913        }
1914        #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
1915        {
1916            crate::hal::uart_16550::putchar(c);
1917        }
1918    }
1919    let puts = |s: &str| {
1920        for c in s.bytes() {
1921            panic_putchar(c);
1922        }
1923    };
1924    puts("\n[KERNEL PANIC] ");
1925    puts("Critical failure.\n");
1926    use core::fmt::Write;
1927    struct PanicWriter;
1928    impl core::fmt::Write for PanicWriter {
1929        fn write_str(&mut self, s: &str) -> core::fmt::Result {
1930            for c in s.bytes() {
1931                panic_putchar(c);
1932            }
1933            Ok(())
1934        }
1935    }
1936    let _ = writeln!(PanicWriter, "{}", info);
1937
1938    // Reboot or spin: RISC-V uses SBI SRST; ARM64 / x86_64 spin.
1939    puts("[KERNEL PANIC] halting...\n");
1940    #[cfg(any(target_arch = "riscv64", target_arch = "riscv32"))]
1941    crate::hal::sbi::system_reset(crate::hal::sbi::SBI_RESET_COLD_REBOOT, 0);
1942    #[cfg(any(target_arch = "x86_64", target_arch = "x86"))]
1943    loop {
1944        unsafe {
1945            core::arch::asm!("cli; hlt", options(nomem, nostack));
1946        }
1947    }
1948    // Fallback halt for all non-x86 arches (including riscv — unreachable after system_reset).
1949    #[cfg(not(any(target_arch = "x86_64", target_arch = "x86")))]
1950    loop {
1951        unsafe {
1952            core::arch::asm!("wfi", options(nomem, nostack));
1953        }
1954    }
1955}

1958///
1959/// Runs after the kernel PML4 is active: `init_kernel_paging_x86` identity-maps
1960/// the legacy (last-entry) unit, and every other unit is mapped here. APIC

1976        "[x86-gate] ioapics registered={} declared={}",
1977        registered,
1978        declared
1979    );
1980}

1982/// Hand the HAL the clocksource candidates `init_timers` may choose besides
1983/// HPET and the TSC:
1984/// - the FADT PM timer, when it is a port-I/O block on a non-hardware-reduced
1985///   platform;
1986/// - one zeroed frame for kvmclock, when KVM advertises it.
1987///
1988/// The frame is never freed: once registered, the host writes into it until
1989/// reset.
1990#[cfg(all(target_arch = "x86_64", not(feature = "g1-cpu-accounting")))]
1991fn offer_x86_clock_candidates(acpi_info: &crate::acpi::AcpiInfo) {
1992    if let Some(fadt) = acpi_info.fadt.filter(|fadt| !fadt.hw_reduced()) {
1993        let timer = fadt.pm_timer;
1994        match u16::try_from(timer.address) {
1995            Ok(port) if timer.space_id == crate::acpi::Gas::SYSTEM_IO && port != 0 => {
1996                crate::hal::set_pm_timer(port, if fadt.pm_timer_32bit() { 32 } else { 24 });
1997            }
1998            _ if timer.is_present() => log::warn!(
1999                "[x86-gate] PM timer at space {} address {:#x} not used (port I/O only)",
2000                timer.space_id,
2001                timer.address
2002            ),
2003            _ => {}
2004        }
2005    }
2006    if crate::hal::clock::kvmclock::available() {
2007        let frame = memory::frame::FRAME_ALLOCATOR
2008            .lock()
2009            .as_mut()
2010            .and_then(|frames| frames.allocate_frame());
2011        if let Some(phys) = frame {
2012            let virt = memory::frame::phys_to_virt(phys);
2013            // SAFETY: a freshly allocated, HHDM-mapped frame owned by this call.
2014            unsafe { core::ptr::write_bytes(virt as *mut u8, 0, 4096) };
2015            crate::hal::clock::set_pvclock_area(virt, phys as u64);
2016        }
2017    }
