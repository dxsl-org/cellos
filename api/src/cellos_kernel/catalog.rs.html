2//! The cell catalog: every authority the kernel keys on an install path.
3//!
4//! One row per path holds what used to live in four separate match tables:
5//! - `request` — authority the path asks for on top of its manifest, because no
6//!   manifest bit exists for it (P-TRUST driver authority, network rights, Q,
7//!   KSSH). Still intersected with the ceiling and the operator policy.
8//! - `boot` — the ceiling of a kernel/boot-initiated spawn (`Spawner::Root`).
9//! - `user_target` — the child ceiling of a reviewed launch edge
10//!   (`loader::launch_profile`); which callers may use it is edge logic, not data.
11//! - `trusted_core` — keeps its caps under a `DenyAll` policy (field recovery).
12//!
13//! Rules that are not per path stay as functions here: the user-app sandbox
14//! prefixes, the S2-T/S2-W observe/mutate request, and the legacy
15//! manifest-less fallback. `scripts/catalog.generated.json` is exported from
16//! this module by a host test; `scripts/sign-policy.py` reads that file instead
17//! of parsing Rust.
18//!
19//! **Invariant — every ceiling is per path, never a union.** A union of rows is
20//! a superset of every row and would admit everything while looking like a
21//! restriction (`loader::boot_ceiling::self_test` pins this). An unknown path has
22//! no boot row and gets `CapSet::EMPTY` (fail-closed).
23
24use crate::resource_registry::{DEV_DISPLAY, DEV_GPIO, DEV_I2C, DEV_SPI, DEV_UART};
25use crate::task::cap::CapSet;
26
27/// One install path's authority rows. See the module documentation.
28#[derive(Copy, Clone, Debug)]
29pub struct Cell {
30    pub path: &'static str,
31    pub request: CapSet,
32    pub boot: Option<CapSet>,
33    pub user_target: Option<CapSet>,
34    pub trusted_core: bool,
35    /// Guest-memory budget in bytes for the path's VMs (guest RAM, nested or
36    /// Stage-2 tables, VM control structures). Kernel-only: not exported to
37    /// `scripts/catalog.generated.json`, so the manifest and signed policy do
38    /// not carry it.
39    pub guest_memory: usize,
40}
41
42const ROW: Cell = Cell {
43    path: "",
44    request: CapSet::EMPTY,
45    boot: None,
46    user_target: None,
47    trusted_core: false,
48    guest_memory: 0,
49};
50
51const MIB: usize = 1024 * 1024;

53/// Install-path prefixes of the user-app sandbox: empty boot and target ceiling,
54/// and no S2-T/S2-W request (annex D-N2, D-TW1).
55pub const SANDBOX_PREFIXES: [&str; 3] = ["/apps/", "/data/bin/", "/home/"];
56
57pub fn is_sandbox(path: &str) -> bool {
58    SANDBOX_PREFIXES
59        .iter()
60        .any(|prefix| path.starts_with(prefix))
61}

65/// The pre-annex task/memory observation and VFS-mutation rights (annex
66/// S2-T/S2-W §3, D-TW1). Every non-sandbox row keeps them.
67const LEGACY: CapSet = CapSet::LEGACY_OBSERVE_MUTATE;
68/// VFS block regions after request ∩ ceiling ∩ policy: P1 | P4 | SRV | cell-store.
69/// A `0b111` ceiling would zero the cell-store bit before policy runs.
70pub const VFS_REGIONS: u8 = 0b1111;
71const CONSOLE_MMIO: u8 = DEV_GPIO | DEV_UART;
72const ROOT_MMIO: u8 = CONSOLE_MMIO | DEV_I2C | DEV_SPI | DEV_DISPLAY;
73/// `pcie_driver` + the display MMIO class: a PCIe driver claims a BAR and authorises DMA.
74const PCIE: CapSet = CapSet {
75    pcie_driver: true,
76    mmio_devices: DEV_DISPLAY,
77    ..CapSet::EMPTY
78};
79/// Outbound network client: `Resolve` plus outbound sockets (annex S2-N §5).
80const NET_CLIENT: CapSet = CapSet {
81    net_resolve: true,
82    net_connect: true,
83    ..CapSet::EMPTY
84};
85/// Inbound-only server: listening sockets, no DNS and no outbound connect.
86const NET_LISTENER: CapSet = CapSet {
87    net_listen: true,
88    ..CapSet::EMPTY
89};
90/// Client and server in one Cell (`nc -l`, the cluster broker).
91const NET_PEER: CapSet = CapSet {
92    net_resolve: true,
93    net_connect: true,
94    net_listen: true,
95    ..CapSet::EMPTY
96};
97const NET_L2: CapSet = CapSet {
98    net_l2: true,
99    ..CapSet::EMPTY
100};

102/// Field-wise OR. Used to layer a row's request and to write rows as
103/// "named set + extra rights".
104pub const fn union(a: CapSet, b: CapSet) -> CapSet {
105    CapSet {
106        block_io: a.block_io | b.block_io,
107        network: a.network | b.network,
108        spawn: a.spawn | b.spawn,
109        hypervisor: a.hypervisor | b.hypervisor,
110        mmio_devices: a.mmio_devices | b.mmio_devices,
111        block_regions: a.block_regions | b.block_regions,
112        pcie_driver: a.pcie_driver | b.pcie_driver,
113        platform: a.platform | b.platform,
114        supervisor: a.supervisor | b.supervisor,
115        usb_driver: a.usb_driver | b.usb_driver,
116        serial_port: a.serial_port | b.serial_port,
117        net_resolve: a.net_resolve | b.net_resolve,
118        net_connect: a.net_connect | b.net_connect,
119        net_listen: a.net_listen | b.net_listen,
120        net_l2: a.net_l2 | b.net_l2,
121        observe_tasks: a.observe_tasks | b.observe_tasks,
122        observe_memory: a.observe_memory | b.observe_memory,
123        vfs_mutate: a.vfs_mutate | b.vfs_mutate,
124        config_write: a.config_write | b.config_write,
125        kms_ssh_sign: a.kms_ssh_sign | b.kms_ssh_sign,
126        kms_ssh_admin: a.kms_ssh_admin | b.kms_ssh_admin,
127    }
128}

130const fn legacy_with(extra: CapSet) -> CapSet {
131    union(LEGACY, extra)
132}
133
134const fn mmio(mask: u8) -> CapSet {
135    CapSet {
136        mmio_devices: mask,
137        ..LEGACY
138    }
139}

141/// A capability-free boot cell that is also a reviewed launch target.
142const fn plain(path: &'static str) -> Cell {
143    Cell {
144        path,
145        boot: Some(LEGACY),
146        user_target: Some(LEGACY),
147        ..ROW
148    }
149}

151/// A capability-free reviewed launch target that is not a boot cell.
152const fn target(path: &'static str) -> Cell {
153    Cell {
154        path,
155        user_target: Some(LEGACY),
156        ..ROW
157    }
158}

160/// A capability-free boot cell that is not a reviewed launch target.
161const fn boot_only(path: &'static str) -> Cell {
162    Cell {
163        path,
164        boot: Some(LEGACY),
165        ..ROW
166    }

169/// A PCIe Driver Cell: requests and may hold `pcie_driver` + `DEV_DISPLAY`.
170const fn pcie_driver(path: &'static str) -> Cell {
171    Cell {
172        path,
173        request: PCIE,
174        boot: Some(legacy_with(PCIE)),
175        ..ROW
176    }

179/// An outbound network client launched by a reviewed edge.
180const fn net_client(path: &'static str) -> Cell {
181    Cell {
182        path,
183        request: NET_CLIENT,
184        user_target: Some(legacy_with(NET_CLIENT)),
185        ..ROW
186    }

189/// A Cell whose reviewed edge keeps lifecycle authority (`spawn`).

643pub const TEST_HOOK_CELLS: &[Cell] = &[
644    pcie_driver("/bin/edu-holder"),
645    boot_only("/bin/export-client"),
646];

648/// Private S2-T/S2-W positive control (annex §6, E1): the probe binary the
649/// sandbox copy at `/data/bin/s2tw-probe` runs, on a reviewed edge with X/M/W.
650#[cfg(feature = "phase04-driver-profile")]
651pub const PHASE04_DRIVER_CELLS: &[Cell] = &[target("/bin/s2tw-probe")];

653/// Every row compiled into this kernel, production rows first.
654pub fn rows() -> impl Iterator<Item = &'static Cell> {
655    let rows = CELLS.iter();
656    #[cfg(feature = "test-hooks")]
657    let rows = rows.chain(TEST_HOOK_CELLS.iter());
658    #[cfg(feature = "phase04-driver-profile")]
659    let rows = rows.chain(PHASE04_DRIVER_CELLS.iter());
660    rows
661}
662
663pub fn find(path: &str) -> Option<&'static Cell> {
664    rows().find(|row| row.path == path)
665}
666
667/// Authority `path` requests on top of its manifest: the row's `request`, plus
668/// the S2-T/S2-W observe/mutate rights for every non-sandbox path (annex §3).
669pub fn request(path: &str) -> CapSet {
670    let row = find(path).map_or(CapSet::EMPTY, |row| row.request);
671    if is_sandbox(path) {
672        row
673    } else {
674        union(row, LEGACY)
675    }
676}
677
678/// Boot ceiling row, `Some(EMPTY)` for the sandbox, `None` for an unknown path.
679pub fn boot(path: &str) -> Option<CapSet> {
680    match find(path).and_then(|row| row.boot) {
681        Some(caps) => Some(caps),
682        None if is_sandbox(path) => Some(CapSet::EMPTY),
683        None => None,
684    }
685}
686
687/// Reviewed launch-target ceiling, `Some(EMPTY)` for the sandbox, else `None`.
688pub fn user_target(path: &str) -> Option<CapSet> {
689    match find(path).and_then(|row| row.user_target) {
690        Some(caps) => Some(caps),
691        None if is_sandbox(path) => Some(CapSet::EMPTY),
692        None => None,
693    }
694}

696/// Install paths whose bytes are another program's signed image, so the image
697/// name would claim that program's launch edges. `/bin/cellbox-ssh` is the
698/// signed `cellbox` file run as the network-facing SSH client; it must not act
699/// as the operator shell (KSSH K7: the sign holder never reaches the admin Cell).
700pub const IDENTITY_ALIASES: &[&str] = &["/bin/cellbox-ssh"];
701
702/// Whether a Cell launched by the kernel from `path` may be named by its signed
703/// image (Phase 06): a `/bin/` install path (the same rule that admits a
704/// privileged manifest, `governed_spawn`) that is not an alias. Callers must
705/// also have resolved `path` in the kernel (Path, Pinned or boot route); a
706/// caller-supplied ELF's advisory path never qualifies.
707pub fn names_launch_identity(path: &str) -> bool {
708    path.starts_with("/bin/") && !IDENTITY_ALIASES.contains(&path)
709}
710
711pub fn is_trusted_core(path: &str) -> bool {
712    find(path).is_some_and(|row| row.trusted_core)
713}
714
715/// Legacy grants for a cell without a `__ViCell_manifest`, still bounded by the
716/// launch request. Suffix match on `/bin/` paths only (historic behaviour).
717pub fn legacy_manifestless(path: &str) -> CapSet {
718    let mut c = CapSet::EMPTY;
719    if path.starts_with("/bin/") {
720        if path.ends_with("/bin/vfs") {
721            c.block_io = true;
722            c.block_regions = 0b111; // legacy fallback: P1 + P4 + SRV
723        }
724        if path.ends_with("/bin/net") {
725            c.network = true;
726        }
727        if path.ends_with("/bin/cellbox") || path.ends_with("/bin/init") {
728            c.spawn = true;
729        }
730    }
731    c
732}

734// ── PCI functions each driver program may claim (H-3) ───────────────────────
735
736/// One PCI match a driver program may query with 418 (class triple) or 424
737/// (vendor:device). `prog_if: None` admits every programming interface.
738#[derive(Copy, Clone, Debug, PartialEq, Eq)]
739pub enum PciMatch {
740    Class {
741        class: u8,
742        subclass: u8,
743        prog_if: Option<u8>,
744    },
745    Id {
746        vendor: u16,
747        device: u16,
748    },
749}
750
751const fn id(vendor: u16, device: u16) -> PciMatch {
752    PciMatch::Id { vendor, device }
753}
754
755const VIRTIO: u16 = 0x1AF4;
756const VMWARE: u16 = 0x15AD;
757const INTEL: u16 = 0x8086;
758
759/// PCI functions per **measured program name** (`__ViCell_image`, the name the
760/// kernel attests for the caller's Cell — never an install path or task name).
761/// Holding `pcie_driver` is necessary but no longer sufficient: 418/424 also
762/// refuse a query outside the caller's row, so one driver cannot claim another
763/// driver's controller by asking first. A program with no row claims nothing.
764pub const PCI_DRIVERS: &[(&str, &[PciMatch])] = &[
765    (
766        "nvme",
767        &[PciMatch::Class {
768            class: 0x01,
769            subclass: 0x08,
770            prog_if: Some(0x02),
771        }],
772    ),
773    // AHCI, plus the two non-AHCI SATA prog-ifs it probes only to log a refusal.
774    (
775        "ahci",
776        &[PciMatch::Class {
777            class: 0x01,
778            subclass: 0x06,
779            prog_if: None,
780        }],
781    ),
782    (
783        "xhci",
784        &[PciMatch::Class {
785            class: 0x0C,
786            subclass: 0x03,
787            prog_if: Some(0x30),
788        }],
789    ),
790    ("virtio-blk", &[id(VIRTIO, 0x1042), id(VIRTIO, 0x1001)]),
791    ("virtio-scsi", &[id(VIRTIO, 0x1048), id(VIRTIO, 0x1004)]),
792    ("virtio-net", &[id(VIRTIO, 0x1041), id(VIRTIO, 0x1000)]),
793    ("vmxnet3", &[id(VMWARE, 0x07B0)]),
794    ("pvscsi", &[id(VMWARE, 0x07C0)]),
795    ("e1000", &[id(INTEL, 0x100E), id(INTEL, 0x100F)]),
796    ("e1000e", &[id(INTEL, 0x10D3)]),
797    (
798        "igb",
799        &[id(INTEL, 0x10C9), id(INTEL, 0x1533), id(INTEL, 0x1521)],
800    ),
801    // ADR-0033 CP1 §7 QEMU `edu` holder; only test-hooks kernels give it
802    // `pcie_driver`, so the row is inert elsewhere.
803    ("edu-holder", &[id(0x1234, 0x11E8)]),
804];
805
806fn pci_row(program: &str) -> &'static [PciMatch] {
807    PCI_DRIVERS
808        .iter()
809        .find(|(name, _)| *name == program)
810        .map_or(&[], |(_, matches)| *matches)
811}
812
813/// Whether `program` may query the class triple with 418.
814pub fn pci_class_allowed(program: &str, class: u8, subclass: u8, prog_if: u8) -> bool {
815    pci_row(program).iter().any(|m| {
816        matches!(*m, PciMatch::Class { class: c, subclass: s, prog_if: p }
817            if c == class && s == subclass && p.is_none_or(|p| p == prog_if))
818    })
819}
820
821/// Whether `program` may query `vendor:device` with 424.
822pub fn pci_id_allowed(program: &str, vendor: u16, device: u16) -> bool {
823    pci_row(program).contains(&PciMatch::Id { vendor, device })
824}

830    /// Every Ethernet ID the kernel hands out is claimable by the program
831    /// installed at the path the table names, and by no other NIC program.

842                assert!(
843                    !pci_id_allowed(other, vendor, device),
844                    "{other} may claim {path}'s device"
845                );
846            }
847        }
848    }

851    /// sandbox copies of a signed program do not.
