57/// Write-once holder of the profile bound at boot.
58#[cfg(target_arch = "x86_64")]
59struct ProfileCell(core::sync::atomic::AtomicPtr<BoardDescriptor>);
60
61#[cfg(target_arch = "x86_64")]
62impl ProfileCell {
63    const fn new() -> Self {
64        Self(core::sync::atomic::AtomicPtr::new(core::ptr::null_mut()))
65    }
66
67    /// Stores `board` if nothing is bound yet; `false` if a profile already is.
68    fn bind(&self, board: &'static BoardDescriptor) -> bool {
69        use core::sync::atomic::Ordering;
70        self.0
71            .compare_exchange(
72                core::ptr::null_mut(),
73                (board as *const BoardDescriptor).cast_mut(),
74                Ordering::AcqRel,
75                Ordering::Acquire,
76            )
77            .is_ok()
78    }
79
80    fn get(&self) -> Option<&'static BoardDescriptor> {
81        let board = self.0.load(core::sync::atomic::Ordering::Acquire);
82        // SAFETY: `bind` is the only store, and it stores only pointers taken
83        // from `&'static BoardDescriptor`; the pointee is never written.
84        unsafe { board.as_ref() }
85    }
86}

120#[cfg(target_arch = "x86_64")]
121pub(crate) fn x86_soc_profile(board: &BoardDescriptor) -> &'static hal_soc_x86::X86PlatformProfile {
122    use cellos_boards::SocId;

136/// Why the boot command line was refused.
137#[cfg(target_arch = "x86_64")]
138#[derive(Clone, Copy, Debug, Eq, PartialEq)]
139pub(crate) enum CmdlineError {
140    /// Longer than [`CMDLINE_MAX`].
141    TooLong,
142    /// A byte outside printable ASCII and tab.
143    NotPrintableAscii,
144    /// `cellos.profile` with an empty value, or a value that is not a slug
145    /// (`[a-z0-9_-]`, at most 64 bytes).
146    InvalidProfile,
147    /// `cellos.profile` given more than once.
148    DuplicateProfile,
149}
150
151/// The `cellos.profile=` value on `cmdline`, or `None` when it is absent.
152///
153/// Tokens are separated by spaces or tabs. Every other token is ignored, but
154/// the whole line must be printable ASCII and at most [`CMDLINE_MAX`] bytes.
155#[cfg(target_arch = "x86_64")]
156pub(crate) fn profile_option(cmdline: &[u8]) -> Result<Option<&str>, CmdlineError> {
157    if cmdline.len() > CMDLINE_MAX {
158        return Err(CmdlineError::TooLong);
159    }
160    if !cmdline
161        .iter()
162        .all(|byte| matches!(byte, b'\t' | 0x20..=0x7E))
163    {
164        return Err(CmdlineError::NotPrintableAscii);
165    }
166    let text = core::str::from_utf8(cmdline).map_err(|_| CmdlineError::NotPrintableAscii)?;
167    let mut found = None;
168    for token in text.split([' ', '\t']).filter(|token| !token.is_empty()) {
169        let value = match token.split_once('=') {
170            Some((PROFILE_KEY, value)) => value,
171            None if token == PROFILE_KEY => "",
172            _ => continue,
173        };
174        let is_slug = !value.is_empty()
175            && value.len() <= PROFILE_SLUG_MAX
176            && value
177                .bytes()
178                .all(|byte| matches!(byte, b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_'));
179        if !is_slug {
180            return Err(CmdlineError::InvalidProfile);
181        }
182        if found.replace(value).is_some() {
183            return Err(CmdlineError::DuplicateProfile);
184        }
185    }
186    Ok(found)
187}

192/// The `cellos.console=` value on `cmdline`, or `None` when it is absent.
193#[cfg(target_arch = "x86_64")]
194pub(crate) fn console_option(cmdline: &[u8]) -> Result<Option<&str>, CmdlineError> {
195    if cmdline.len() > CMDLINE_MAX {
196        return Err(CmdlineError::TooLong);
197    }
198    if !cmdline
199        .iter()
200        .all(|byte| matches!(byte, b'\t' | 0x20..=0x7E))
201    {
202        return Err(CmdlineError::NotPrintableAscii);
203    }
204    let text = core::str::from_utf8(cmdline).map_err(|_| CmdlineError::NotPrintableAscii)?;
205    let mut found = None;
206    for token in text.split([' ', '\t']).filter(|token| !token.is_empty()) {
207        let value = match token.split_once('=') {
208            Some((CONSOLE_KEY, value)) => value,
209            None if token == CONSOLE_KEY => "",
210            _ => continue,
211        };
212        if value.is_empty() || value.len() > 64 {
213            continue;
214        }
215        if found.replace(value).is_some() {
216            return Err(CmdlineError::DuplicateProfile);
217        }
218    }
219    Ok(found)
220}

230/// Why no x86 profile could be bound. Every variant stops boot.
231#[cfg(target_arch = "x86_64")]
232#[derive(Clone, Copy, Debug, Eq, PartialEq)]
233pub(crate) enum BindError<'a> {
234    Cmdline(CmdlineError),
235    /// The named slug is not in `cellos_boards::x86_catalog()`.
236    UnknownProfile(&'a str),
237    /// The named profile weakens DMA isolation, and CPUID reports no
238    /// hypervisor to make that consistent with the machine.
239    NeedsHypervisor(&'static str),
240}
241
242#[cfg(target_arch = "x86_64")]
243impl core::fmt::Display for BindError<'_> {
244    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
245        match self {
246            Self::Cmdline(error) => write!(f, "boot command line refused: {:?}", error),
247            Self::UnknownProfile(slug) => {
248                write!(f, "unknown cellos.profile={slug}; this image binds only:")?;
249                for board in cellos_boards::x86_catalog() {
250                    write!(f, " {}", board.slug)?;
251                }
252                Ok(())
253            }
254            Self::NeedsHypervisor(slug) => write!(
255                f,
256                "cellos.profile={slug} does not require DMA isolation and is admitted only \
257                 under a hypervisor, but CPUID reports none"
258            ),
259        }
260    }
261}

263/// Applies the x86 binding rules (ADR-0029 §2) to the boot command line and
264/// the detected identity.
265///
266/// 1. A `cellos.profile=` slug selects that catalog entry; an unknown slug is
267///    an error.
268/// 2. Without one, [`DEFAULT_X86_PROFILE`] applies.
269/// 3. A profile that does not require DMA isolation is refused unless CPUID
270///    reports a hypervisor. Detected facts only reject; they never choose.
271#[cfg(target_arch = "x86_64")]
272pub(crate) fn choose_x86_profile<'a>(
273    cmdline: &'a [u8],
274    identity: &PlatformIdentity,
275) -> Result<(&'static BoardDescriptor, ProfileSource), BindError<'a>> {
276    let (slug, source) = match profile_option(cmdline).map_err(BindError::Cmdline)? {
277        Some(slug) => (slug, ProfileSource::CommandLine),
278        None => (DEFAULT_X86_PROFILE, ProfileSource::Default),
279    };
280    let board = cellos_boards::find_by_slug(cellos_boards::x86_catalog(), slug)
281        .ok_or(BindError::UnknownProfile(slug))?;
282    if board.dma_isolation != cellos_boards::DmaIsolation::Required && !identity.hypervisor {
283        return Err(BindError::NeedsHypervisor(board.slug));
284    }
285    Ok((board, source))
286}

288/// Binds `board` as the x86 profile for the rest of the boot, keeps `facts`
289/// for the machine fingerprint, and names the profile on the console.
290///
291/// # Panics
292/// If `board` fails validation or a profile is already bound.
293#[cfg(target_arch = "x86_64")]
294pub(crate) fn bind_x86_profile(board: &'static BoardDescriptor, facts: X86BootFacts) {
295    if let Err(error) = board.validate_for(Architecture::X86_64) {
296        invalid_descriptor(error);
297    }
298    let soc = x86_soc_profile(board);
299    if !board.has_driver(cellos_boards::DriverId::Uart16550PortIo) {
300        panic!(
301            "[board] x86 profile {} does not enable the early UART",
302            board.slug
303        );
304    }
305    {
306        let mut kept = X86_MACHINE_FACTS.lock();
307        if kept.is_some() || !BOUND_X86_PROFILE.bind(board) {
308            panic!("[board] x86 profile bound twice");
309        }
310        *kept = Some(facts);
311    }
312    let source = facts.source;
313    log::info!(
314        "[x86-gate] board={} soc-profile={} source={} dma_isolation={:?}",
315        board.slug,
316        soc.slug,
317        match source {
318            ProfileSource::CommandLine => "cmdline",
319            ProfileSource::Default => "default",
320        },
321        board.dma_isolation
322    );
323    // A weaker-than-Required profile is only reachable by naming it under a
324    // hypervisor (`choose_x86_profile`); say so on the console in operator terms.
325    if board.dma_isolation != cellos_boards::DmaIsolation::Required {
326        log::warn!("[x86-gate] dma isolation optional (operator-selected VM profile)");
327    }
328}

357/// One log line: `acpi_oem=… smbios=… hv=0|1`. Firmware strings are printed
358/// quoted, with bytes outside printable ASCII, `"` and `\` escaped, so they
359/// cannot break the line or forge another record.
360#[cfg(target_arch = "x86_64")]
361impl core::fmt::Display for PlatformIdentity {
362    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
363        match &self.acpi_oem {
364            Some(oem) => write!(
365                f,
366                "acpi_oem={} acpi_table={} acpi_rev={:#x}",
367                Quoted(&oem.oem_id),
368                Quoted(&oem.oem_table_id),
369                oem.oem_revision
370            )?,
371            None => f.write_str("acpi_oem=none")?,
372        }
373        match &self.smbios {
374            Some(system) => {
375                write!(
376                    f,
377                    " smbios=manufacturer={} product={} uuid=",
378                    Quoted(system.manufacturer.as_bytes()),
379                    Quoted(system.product.as_bytes())
380                )?;
381                match system.uuid {
382                    Some(uuid) => {
383                        for (index, byte) in uuid.iter().enumerate() {
384                            if matches!(index, 4 | 6 | 8 | 10) {
385                                f.write_str("-")?;
386                            }
387                            write!(f, "{:02x}", byte)?;
388                        }
389                    }
390                    None => f.write_str("none")?,
391                }
392            }
393            None => f.write_str(" smbios=none")?,
394        }
395        write!(f, " hv={}", u8::from(self.hypervisor))
396    }
397}

405/// Firmware bytes rendered as a quoted, escaped ASCII string.
406#[cfg(target_arch = "x86_64")]
407struct Quoted<'a>(&'a [u8]);
408
409#[cfg(target_arch = "x86_64")]
410impl core::fmt::Display for Quoted<'_> {
411    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
412        f.write_str("\"")?;
413        for &byte in self.0 {
414            match byte {
415                b'"' | b'\\' => write!(f, "\\{}", byte as char)?,
416                0x20..=0x7E => write!(f, "{}", byte as char)?,
417                _ => write!(f, "\\x{:02x}", byte)?,
418            }
419        }
420        f.write_str("\"")
421    }
422}

575            (b"quiet \xff", CmdlineError::NotPrintableAscii),
576        ] {
577            assert_eq!(profile_option(cmdline), Err(error), "{cmdline:?}");
578        }
579        let slug = alloc::format!("cellos.profile={}", "a".repeat(65));
580        assert_eq!(
581            profile_option(slug.as_bytes()),
582            Err(CmdlineError::InvalidProfile)
583        );
584        let mut long = alloc::vec![b' '; CMDLINE_MAX];
585        assert_eq!(profile_option(&long), Ok(None));
586        long.push(b' ');
587        assert_eq!(profile_option(&long), Err(CmdlineError::TooLong));
588    }

590    #[test]
591    fn binding_rules_default_strict_and_never_let_facts_choose() {
592        // No option: the strict profile, whatever was detected.
593        for hypervisor in [false, true] {
594            let (board, source) = choose_x86_profile(b"quiet", &facts(hypervisor)).unwrap();
595            assert_eq!(board.slug, DEFAULT_X86_PROFILE);
596            assert_eq!(board.dma_isolation, DmaIsolation::Required);
597            assert_eq!(source, ProfileSource::Default);
598        }
599        // The weak profile needs both the operator's choice and a hypervisor.
600        let (board, source) =
601            choose_x86_profile(b"cellos.profile=qemu-q35-x86_64", &facts(true)).unwrap();
602        assert_eq!(board.slug, "qemu-q35-x86_64");
603        assert_eq!(source, ProfileSource::CommandLine);
604        assert_eq!(
605            choose_x86_profile(b"cellos.profile=qemu-q35-x86_64", &facts(false)),
606            Err(BindError::NeedsHypervisor("qemu-q35-x86_64"))
607        );
608        // ADR-0031 D-VM: the generic VM profile follows the same rule.
609        let (board, _) = choose_x86_profile(b"cellos.profile=x86_64-vm", &facts(true)).unwrap();
610        assert_eq!(board.slug, "x86_64-vm");
611        assert_eq!(board.dma_isolation, DmaIsolation::Optional);
612        assert_eq!(
613            choose_x86_profile(b"cellos.profile=x86_64-vm", &facts(false)),
614            Err(BindError::NeedsHypervisor("x86_64-vm"))
615        );
616        // Naming the strict profile needs no detected fact.
617        let (board, source) =
618            choose_x86_profile(b"cellos.profile=x86_64-pc", &facts(false)).unwrap();
619        assert_eq!(board.slug, "x86_64-pc");
620        assert_eq!(source, ProfileSource::CommandLine);
621        // Unknown and malformed selections fail; they never fall back.
622        assert_eq!(
623            choose_x86_profile(b"cellos.profile=qemu-virt-riscv64", &facts(true)),
624            Err(BindError::UnknownProfile("qemu-virt-riscv64"))
625        );
626        assert_eq!(
627            choose_x86_profile(b"cellos.profile=", &facts(true)),
628            Err(BindError::Cmdline(CmdlineError::InvalidProfile))
629        );
630    }
