2//! Owner trust anchor (Anchor 3) — Ed25519 signature verification for owner A/B admission records.
3//!
4//! Distinct from Fleet Policy anchor (`policy.rs`) and Publisher/Cell anchor (`signing.rs`).
5//! The device holds only the owner public key; owner private signing occurs offline.
6
7/// Dev Owner Ed25519 public key — derived from deterministic seed `[0x4F; 32]`.
8#[cfg(feature = "dev-signing-key")]
9const DEV_OWNER_SIGNER_PUBKEY: [u8; 32] = [
10    0x00, 0xe3, 0xc5, 0x6b, 0x91, 0xab, 0x0a, 0x01, 0x71, 0x74, 0xb9, 0x66, 0x45, 0xea, 0xf9, 0x28,
11    0x36, 0x6c, 0xdb, 0xae, 0x1e, 0x87, 0xfd, 0x21, 0xbf, 0x86, 0x66, 0x1d, 0x86, 0xf3, 0xe7, 0xef,
12];
13
14#[cfg(feature = "dev-signing-key")]
15const OWNER_SIGNER_PUBKEY: [u8; 32] = DEV_OWNER_SIGNER_PUBKEY;
16
17#[cfg(not(feature = "dev-signing-key"))]
18const OWNER_SIGNER_PUBKEY: [u8; 32] = [0u8; 32]; // TODO(prod): provisioned owner key
19#[allow(dead_code)]
20pub fn owner_signer_pubkey() -> &'static [u8; 32] {
21    &OWNER_SIGNER_PUBKEY
22}
23
24/// Verify an Ed25519 signature against the owner trust anchor.
25pub fn verify_owner_signature(payload: &[u8], sig: &[u8; 64]) -> bool {
26    crate::ed25519::verify(&OWNER_SIGNER_PUBKEY, payload, sig)
27}
