1//! Volatile access to the queue's DMA allocation.
2
3use crate::ring::DMA_PAGES;
4use core::fmt;
5use ostd::dma::DmaBuf;
6use ostd::syscall::SyscallError;
7
8/// [`DMA_PAGES`] page-aligned pages authorized for one device, laid out per
9/// [`crate::ring`]. Never freed: the device may hold the address until reset.
10///
11/// Accessors take offsets from the region map `crate::ring` proves to lie
12/// inside the allocation, and multi-byte fields at their natural alignment;
13/// an offset past the end panics instead of touching foreign memory.
14pub struct DmaArea {
15    buf: DmaBuf,
16    iova: u64,
17}
18
19/// Why the DMA area could not be set up.
20#[derive(Debug)]
21pub enum DmaError {
22    Alloc,
23    Authorize(SyscallError),
24}
25
26impl fmt::Display for DmaError {
27    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
28        match self {
29            Self::Alloc => write!(f, "DMA allocation of {DMA_PAGES} pages failed"),
30            Self::Authorize(error) => write!(f, "DMA authorization failed: {error:?}"),
31        }
32    }
33}
34
35impl DmaArea {
36    /// Allocates the area and authorizes it for `bdf`, keeping the returned
37    /// device-visible address. A buffer the device was never authorized for
38    /// is returned to the kernel on failure.
39    pub fn new(bdf: u32) -> Result<Self, DmaError> {
40        let buf = DmaBuf::alloc(DMA_PAGES).ok_or(DmaError::Alloc)?;
41        match buf.authorize(bdf) {
42            Ok(iova) => Ok(Self { buf, iova }),
43            Err(error) => {
44                buf.free();
45                Err(DmaError::Authorize(error))
46            }
47        }
48    }
49
50    /// Device-visible address of byte `off`.
51    pub fn iova(&self, off: usize) -> u64 {
52        self.iova + off as u64
53    }
54
55    fn at(&self, off: usize, len: usize) -> *mut u8 {
56        assert!(off
57            .checked_add(len)
58            .is_some_and(|end| end <= self.buf.size()));
59        self.buf.virt().wrapping_add(off)
60    }
61
62    pub fn write_bytes(&self, off: usize, bytes: &[u8]) {
63        let base = self.at(off, bytes.len());
64        for (i, &byte) in bytes.iter().enumerate() {
65            // SAFETY: inside this allocation (checked by `at`).
66            unsafe { base.add(i).write_volatile(byte) };
67        }
68    }
69
70    pub fn fill_zero(&self, off: usize, len: usize) {
71        let base = self.at(off, len);
72        for i in 0..len {
73            // SAFETY: inside this allocation (checked by `at`).
74            unsafe { base.add(i).write_volatile(0) };
75        }
76    }
77
78    /// Volatile: the device may have written these bytes.
79    pub fn read_bytes(&self, off: usize, out: &mut [u8]) {
80        let base = self.at(off, out.len());
81        for (i, byte) in out.iter_mut().enumerate() {
82            // SAFETY: inside this allocation (checked by `at`).
83            *byte = unsafe { base.add(i).read_volatile() };
84        }
85    }
86
87    pub fn write_u16(&self, off: usize, value: u16) {
88        // SAFETY: in bounds (checked by `at`); callers pass 2-aligned fields.
89        unsafe { self.at(off, 2).cast::<u16>().write_volatile(value.to_le()) };
90    }
91
92    pub fn write_u32(&self, off: usize, value: u32) {
93        // SAFETY: in bounds (checked by `at`); callers pass 4-aligned fields.
94        unsafe { self.at(off, 4).cast::<u32>().write_volatile(value.to_le()) };
95    }
96
97    pub fn write_u64(&self, off: usize, value: u64) {
98        // SAFETY: in bounds (checked by `at`); callers pass 8-aligned fields.
99        unsafe { self.at(off, 8).cast::<u64>().write_volatile(value.to_le()) };
100    }
101
102    pub fn read_u16(&self, off: usize) -> u16 {
103        // SAFETY: in bounds (checked by `at`); callers pass 2-aligned fields.
104        u16::from_le(unsafe { self.at(off, 2).cast::<u16>().read_volatile() })
105    }
106
107    pub fn read_u32(&self, off: usize) -> u32 {
108        // SAFETY: in bounds (checked by `at`); callers pass 4-aligned fields.
109        u32::from_le(unsafe { self.at(off, 4).cast::<u32>().read_volatile() })
110    }
111}
