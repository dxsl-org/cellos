1//! virtio-scsi request/response wire format (virtio 1.2 §5.6.6.1) and LUN
2//! re-encoding. Pure: no device access, host-tested.
3//!
4//! Everything read back from the device is untrusted; decoding never panics.
5
6use scsi_block::{Completion, Sense};
7
8/// `lun[8] + id(le64) + task_attr + prio + crn`, before `cdb[cdb_size]`.
9pub const REQ_FIXED: usize = 19;
10/// `sense_len(le32) + residual(le32) + status_qualifier(le16) + status +
11/// response`, before `sense[sense_size]`.
12pub const RESP_FIXED: usize = 12;
13/// `VIRTIO_SCSI_S_SIMPLE`.
14const TASK_ATTR_SIMPLE: u8 = 0;
15/// `VIRTIO_SCSI_S_OK`: the command reached the LUN and `status` is valid.
16pub const RESPONSE_OK: u8 = 0;
17/// Highest LUN the virtio-scsi flat-space LUN field can carry (14 bits).
18const MAX_LUN: u16 = 0x3FFF;
19
20/// Why the request queue produced no SAM status for a command.
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub enum VscsiError {
23    /// The 8-byte SAM LUN uses an addressing method virtio-scsi cannot carry.
24    UnsupportedLun([u8; 8]),
25    /// The CDB is longer than the device's `cdb_size`.
26    CdbTooLong(usize),
27    /// The data phase exceeds the bounce buffer.
28    DataTooLarge(usize),
29    /// Non-OK virtio-scsi `response` (e.g. 3 = BAD_TARGET).
30    Response(u8),
31    /// No completion within the I/O budget; the queue is now poisoned.
32    Timeout,
33    /// The used ring returned something never submitted; the queue is now
34    /// poisoned.
35    BadCompletion,
36    /// An earlier timeout or bad completion left the queue unusable.
37    Poisoned,
38}
39
40impl VscsiError {
41    /// Whether the device was (or may have been) notified of the command.
42    pub fn issued(&self) -> bool {
43        matches!(
44            self,
45            Self::Response(_) | Self::Timeout | Self::BadCompletion
46        )
47    }
48
49    /// Whether the queue refuses every later command after this error.
50    pub fn poisons(&self) -> bool {
51        matches!(self, Self::Timeout | Self::BadCompletion | Self::Poisoned)
52    }
53}
54
55/// Logical unit number of a single-level SAM LUN in peripheral (bus 0) or
56/// flat addressing; `None` for any other form.
57pub fn lun_number(sam: [u8; 8]) -> Option<u16> {
58    if sam[2..] != [0; 6] {
59        return None;
60    }
61    match sam[0] >> 6 {
62        0 if sam[0] == 0 => Some(u16::from(sam[1])),
63        1 => Some((u16::from(sam[0] & 0x3F) << 8) | u16::from(sam[1])),
64        _ => None,
65    }
66}
67
68/// virtio-scsi `lun` field for `sam` on `target`:
69/// `[1, target, 0x40 | lun >> 8, lun & 0xff, 0, 0, 0, 0]`.
70pub fn virtio_lun(target: u8, sam: [u8; 8]) -> Result<[u8; 8], VscsiError> {
71    let lun = lun_number(sam)
72        .filter(|&lun| lun <= MAX_LUN)
73        .ok_or(VscsiError::UnsupportedLun(sam))?;
74    Ok([1, target, 0x40 | (lun >> 8) as u8, lun as u8, 0, 0, 0, 0])
75}

77/// Encodes `virtio_scsi_req_cmd`'s device-readable part into `out` with the
78/// CDB zero-padded to `cdb_size`; returns the encoded length.
79///
80/// `out` must hold `REQ_FIXED + cdb_size` bytes (the caller sizes it from
81/// validated limits).
82pub fn encode_req(
83    out: &mut [u8],
84    lun: [u8; 8],
85    tag: u64,
86    cdb: &[u8],
87    cdb_size: usize,
88) -> Result<usize, VscsiError> {
89    if cdb.len() > cdb_size {
90        return Err(VscsiError::CdbTooLong(cdb.len()));
91    }
92    let len = REQ_FIXED + cdb_size;
93    let out = &mut out[..len];
94    out[..8].copy_from_slice(&lun);
95    out[8..16].copy_from_slice(&tag.to_le_bytes());
96    out[16] = TASK_ATTR_SIMPLE;
97    out[17] = 0; // prio
98    out[18] = 0; // crn
99    out[REQ_FIXED..REQ_FIXED + cdb.len()].copy_from_slice(cdb);
100    out[REQ_FIXED + cdb.len()..].fill(0);
101    Ok(len)
102}

104/// Decodes `virtio_scsi_resp_cmd` (`RESP_FIXED + sense_size` bytes in
105/// `resp`) into a [`Completion`]. Sense is decoded from
106/// `min(sense_len, sense bytes present)` bytes when `sense_len > 0`.
107pub fn decode_resp(resp: &[u8]) -> Result<Completion, VscsiError> {
108    let Some(fixed) = resp.get(..RESP_FIXED) else {
109        return Err(VscsiError::BadCompletion);
110    };
111    let word =
112        |at: usize| u32::from_le_bytes([fixed[at], fixed[at + 1], fixed[at + 2], fixed[at + 3]]);
113    let response = fixed[11];
114    if response != RESPONSE_OK {
115        return Err(VscsiError::Response(response));
116    }
117    let sense_area = &resp[RESP_FIXED..];
118    let sense_len = (word(0) as usize).min(sense_area.len());
119    let sense = (sense_len > 0)
120        .then(|| Sense::decode(&sense_area[..sense_len]))
121        .flatten();
122    Ok(Completion {
123        status: fixed[10],
124        sense,
125        residual: word(4),
126    })
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132    use scsi_block::{sense_key, status};

134    #[test]
135    fn peripheral_and_flat_luns_map_to_virtio_layout() {
136        assert_eq!(virtio_lun(0, [0; 8]), Ok([1, 0, 0x40, 0, 0, 0, 0, 0]));
137        assert_eq!(
138            virtio_lun(3, [0, 7, 0, 0, 0, 0, 0, 0]),
139            Ok([1, 3, 0x40, 7, 0, 0, 0, 0])
140        );
141        assert_eq!(
142            virtio_lun(1, [0x41, 0x23, 0, 0, 0, 0, 0, 0]),
143            Ok([1, 1, 0x41, 0x23, 0, 0, 0, 0])
144        );
145        assert_eq!(lun_number([0x7F, 0xFF, 0, 0, 0, 0, 0, 0]), Some(MAX_LUN));
146    }
147
148    #[test]
149    fn unsupported_luns_are_refused() {
150        for sam in [
151            [0x01, 0, 0, 0, 0, 0, 0, 0], // peripheral, bus 1
152            [0x80, 1, 0, 0, 0, 0, 0, 0], // logical unit addressing
153            [0xC0, 1, 0, 0, 0, 0, 0, 0], // extended addressing
154            [0, 1, 0, 1, 0, 0, 0, 0],    // second level present
155        ] {
156            assert_eq!(virtio_lun(0, sam), Err(VscsiError::UnsupportedLun(sam)));
157        }
158    }
159
160    #[test]
161    fn request_header_is_padded_to_cdb_size() {
162        let mut out = [0xAAu8; 64];
163        let lun = [1, 2, 0x40, 5, 0, 0, 0, 0];
164        let len = encode_req(
165            &mut out,
166            lun,
167            0x0102_0304_0506_0708,
168            &[0x12, 0, 0, 0, 36, 0],
169            32,
170        )
171        .unwrap();
172        assert_eq!(len, REQ_FIXED + 32);
173        assert_eq!(&out[..8], &lun);
174        assert_eq!(&out[8..16], &[8, 7, 6, 5, 4, 3, 2, 1]);
175        assert_eq!(&out[16..19], &[0, 0, 0]);
176        assert_eq!(&out[19..25], &[0x12, 0, 0, 0, 36, 0]);
177        assert!(out[25..len].iter().all(|&b| b == 0));
178        assert_eq!(out[len], 0xAA);
179        assert_eq!(
180            encode_req(&mut out, lun, 0, &[0; 17], 16),
181            Err(VscsiError::CdbTooLong(17))
182        );
183    }
184
185    fn resp(sense_len: u32, residual: u32, status: u8, response: u8, sense: &[u8]) -> [u8; 40] {
186        let mut out = [0u8; 40];
187        out[..4].copy_from_slice(&sense_len.to_le_bytes());
188        out[4..8].copy_from_slice(&residual.to_le_bytes());
189        out[10] = status;
190        out[11] = response;
191        out[RESP_FIXED..RESP_FIXED + sense.len()].copy_from_slice(sense);
192        out
193    }

195    #[test]
196    fn good_and_check_condition_responses_decode() {
197        let done = decode_resp(&resp(0, 16, status::GOOD, 0, &[])).unwrap();
198        assert_eq!(
199            (done.status, done.sense, done.residual),
200            (status::GOOD, None, 16)
201        );
202
203        let fixed = [
204            0x70,
205            0,
206            sense_key::UNIT_ATTENTION,
207            0,
208            0,
209            0,
210            0,
211            10,
212            0,
213            0,
214            0,
215            0,
216            0x29,
217            0,
218        ];
219        let done = decode_resp(&resp(200, 0, status::CHECK_CONDITION, 0, &fixed)).unwrap();
220        let sense = done.sense.unwrap();
221        assert_eq!((sense.key, sense.asc), (sense_key::UNIT_ATTENTION, 0x29));
222    }

224    #[test]
225    fn non_ok_response_and_truncated_buffer_are_errors() {
226        let bad_target = decode_resp(&resp(0, 0, 0, 3, &[]));
227        assert_eq!(bad_target, Err(VscsiError::Response(3)));
228        assert!(VscsiError::Response(3).issued());
229        assert!(!VscsiError::Poisoned.issued());
230        assert!(VscsiError::Timeout.poisons() && !VscsiError::Response(3).poisons());
231        assert_eq!(decode_resp(&[0; 11]), Err(VscsiError::BadCompletion));
232    }
