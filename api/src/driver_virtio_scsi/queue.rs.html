1//! The request virtqueue: a minimal polled split virtqueue that runs one
2//! SCSI command at a time and implements [`ScsiExecutor`].
3//!
4//! Every command reuses descriptors `0..CHAIN_MAX` and the fixed request,
5//! response and bounce areas of [`crate::ring`]: header (device-readable),
6//! data-out, response (device-writable), data-in — in that order, as the
7//! virtio-scsi request layout requires. A timeout or a nonsensical used-ring
8//! entry poisons the queue: the device may still own the descriptors, so no
9//! later command is submitted.
10
11use crate::bars::Bars;
12use crate::config::Limits;
13use crate::dma::DmaArea;
14use crate::proto::{self, VscsiError, RESP_FIXED};
15use crate::ring::{self, RingLayout, AVAIL_F_NO_INTERRUPT, CHAIN_MAX, DATA_LEN, DATA_OFF};
16use crate::ring::{DESC_F_NEXT, DESC_F_WRITE, REQ_LEN, REQ_OFF, RESP_LEN, RESP_OFF};
17use crate::wait::{Deadline, IO_BUDGET_MS};
18use core::sync::atomic::{fence, Ordering};
19use ostd::syscall::sys_get_time_ms;
20use scsi_block::{Completion, DataPhase, ScsiExecutor};
21use virtio_drivers::transport::Transport;
22use virtio_pci_transport::{InitFailure, VirtioPciTransport};

24/// Queue 0 is control, 1 event, 2.. request; only the first request queue
25/// is used.
26pub const REQUEST_QUEUE: u16 = 2;
27/// Head descriptor of every chain.
28const HEAD: u16 = 0;
29
30/// One chain segment: offset into the DMA area, length, flags.
31type Segment = (usize, usize, u16);
32
33pub struct RequestQueue {
34    // Declared before `_bars`: dropping the transport resets the device
35    // while its BARs are still claimed.
36    transport: VirtioPciTransport,
37    _bars: Bars,
38    dma: DmaArea,
39    ring: RingLayout,
40    limits: Limits,
41    avail_idx: u16,
42    last_used: u16,
43    next_tag: u64,
44    poisoned: bool,
45}
46
47impl RequestQueue {
48    /// Zeroes the rings, programs [`REQUEST_QUEUE`] with `size` entries in
49    /// `dma` and sets `DRIVER_OK`.
50    ///
51    /// Precondition: `transport` has negotiated features (`FEATURES_OK`);
52    /// `size` comes from [`ring::queue_size`] for this queue.
53    pub fn start(
54        transport: VirtioPciTransport,
55        bars: Bars,
56        dma: DmaArea,
57        size: u16,
58        limits: Limits,
59    ) -> Result<Self, InitFailure> {
60        let ring = RingLayout::new(size);
61        dma.fill_zero(0, ring.end);
62        dma.write_u16(ring.avail, AVAIL_F_NO_INTERRUPT);
63        let mut queue = Self {
64            transport,
65            _bars: bars,
66            dma,
67            ring,
68            limits,
69            avail_idx: 0,
70            last_used: 0,
71            next_tag: 0,
72            poisoned: false,
73        };
74        let addr = |off: usize| queue.dma.iova(off) as usize;
75        let (desc, avail, used) = (addr(0), addr(ring.avail), addr(ring.used));
76        let transport = &mut queue.transport;
77        transport.queue_set(REQUEST_QUEUE, u32::from(size), desc, avail, used);
78        transport.finish_init();
79        match transport.init_failure() {
80            Some(failure) => Err(failure),
81            None => Ok(queue),
82        }
83    }
84
85    fn submit(&mut self, chain: &[Segment]) {
86        let dma = &self.dma;
87        for (i, &(off, len, flags)) in chain.iter().enumerate() {
88            let next = i + 1 < chain.len();
89            let desc = self.ring.desc(i as u16);
90            dma.write_u64(desc, dma.iova(off));
91            dma.write_u32(desc + 8, len as u32);
92            dma.write_u16(desc + 12, flags | if next { DESC_F_NEXT } else { 0 });
93            dma.write_u16(desc + 14, if next { i as u16 + 1 } else { 0 });
94        }
95        dma.write_u16(self.ring.avail_entry(self.avail_idx), HEAD);
96        self.avail_idx = self.avail_idx.wrapping_add(1);
97        // Descriptors, buffers and the ring entry before the index.
98        fence(Ordering::Release);
99        self.dma.write_u16(self.ring.avail_idx(), self.avail_idx);
100        // The index before the notify write.
101        fence(Ordering::SeqCst);
102        self.transport.notify(REQUEST_QUEUE);
103    }
104
105    /// Polls the used ring for the one in-flight chain within the budget.
106    fn wait(&mut self) -> Result<(), VscsiError> {
107        let mut deadline = Deadline::start(IO_BUDGET_MS, sys_get_time_ms());
108        let used_idx = loop {
109            let idx = self.dma.read_u16(self.ring.used_idx());
110            if idx != self.last_used {
111                break idx;
112            }
113            if !deadline.tick() {
114                core::hint::spin_loop();
115                continue;
116            }
117            ostd::task::yield_now();
118            if deadline.expired(sys_get_time_ms()) {
119                self.poisoned = true;
120                return Err(VscsiError::Timeout);
121            }
122        };
123        // The used element and every device write after the index.
124        fence(Ordering::Acquire);
125        let id = self.dma.read_u32(self.ring.used_entry(self.last_used));
126        if ring::completions(used_idx, self.last_used) != 1 || id != u32::from(HEAD) {
127            self.poisoned = true;
128            return Err(VscsiError::BadCompletion);
129        }
130        self.last_used = used_idx;
131        Ok(())
132    }
133}
134
135impl ScsiExecutor for RequestQueue {
136    type Error = VscsiError;
137
138    fn execute(
139        &mut self,
140        target: u8,
141        lun: [u8; 8],
142        cdb: &[u8],
143        data: DataPhase<'_>,
144    ) -> Result<Completion, VscsiError> {
145        if self.poisoned {
146            return Err(VscsiError::Poisoned);
147        }
148        let lun = proto::virtio_lun(target, lun)?;
149        let data_len = data.len();
150        if data_len > DATA_LEN {
151            return Err(VscsiError::DataTooLarge(data_len));
152        }
153        let mut req = [0u8; REQ_LEN];
154        let req_len = proto::encode_req(&mut req, lun, self.next_tag, cdb, self.limits.cdb_size)?;
155        self.next_tag = self.next_tag.wrapping_add(1);
156        let resp_len = RESP_FIXED + self.limits.sense_size;
157        self.dma.write_bytes(REQ_OFF, &req[..req_len]);
158        self.dma.fill_zero(RESP_OFF, resp_len);
159
160        let mut chain: [Segment; CHAIN_MAX] = [(0, 0, 0); CHAIN_MAX];
161        let mut len = 0;
162        let mut push = |segment: Segment| {
163            chain[len] = segment;
164            len += 1;
165        };
166        push((REQ_OFF, req_len, 0));
167        if let DataPhase::Out(buf) = &data {
168            if !buf.is_empty() {
169                self.dma.write_bytes(DATA_OFF, buf);
170                push((DATA_OFF, data_len, 0));
171            }
172        }
173        push((RESP_OFF, resp_len, DESC_F_WRITE));
174        if matches!(data, DataPhase::In(_)) && data_len > 0 {
175            push((DATA_OFF, data_len, DESC_F_WRITE));
176        }
177        self.submit(&chain[..len]);
178        self.wait()?;
179
180        if let DataPhase::In(buf) = data {
181            self.dma.read_bytes(DATA_OFF, buf);
182        }
183        let mut resp = [0u8; RESP_LEN];
184        self.dma.read_bytes(RESP_OFF, &mut resp[..resp_len]);
185        proto::decode_resp(&resp[..resp_len])
186    }
