1//! virtio-scsi Driver Cell — Tier-1 Privileged Driver Cell (x86_64).
2//!
3//! Claims the first QEMU `virtio-scsi-pci` controller (modern 1af4:1048,
4//! then transitional 1af4:1004), brings it up over the cell-side virtio PCI
5//! transport with a polled request virtqueue in authorized DMA memory, finds
6//! the first direct-access disk LUN through `scsi-block`, publishes the block
7//! route of its BDF via `sys_register_block_device(BlockDevice::Pci { .. })`
8//! and serves the `DrvRequest` sector-I/O protocol (same wire format as the
9//! virtio-blk/NVMe cells) to the registered VFS only.
10//!
11//! Interrupts are not routed (phase 04b deferred): completion is polled with
12//! a time budget.
13//!
14//! # Law 4 exception
15//! `src/dma.rs` (DMA memory) and `src/config.rs` (device configuration
16//! reads) use `unsafe`; every block carries a `// SAFETY:` comment.
17
18// Test builds only: a hosted `cargo test` gets std and libtest's entry, so the
19// pure modules (`proto`, `ring`, `config`, `wait`) run as host unit tests.
20#![cfg_attr(not(test), no_std)]
21#![cfg_attr(not(test), no_main)]

47/// Tid of the registered VFS once a request from it was accepted. Tids are
48/// never reissued, so a cached match stays valid for the life of that VFS.
49static VFS_TID: AtomicUsize = AtomicUsize::new(0);
50
51/// Whether `sender_tid` is the registered VFS, the only accepted block client.
52fn sender_is_vfs(sender_tid: usize) -> bool {
53    let cached = VFS_TID.load(Ordering::Relaxed);
54    if cached != 0 && cached == sender_tid {
55        return true;
56    }
57    if ostd::service::sender_is_live_provider(api::syscall::service::VFS, sender_tid) {
58        VFS_TID.store(sender_tid, Ordering::Relaxed);
59        return true;
60    }
61    false
62}

64/// PCI requester ID printed as `bb:dd.f`.
65struct Bdf(u32);
66
67impl fmt::Display for Bdf {
68    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
69        let bdf = self.0;
70        write!(
71            f,
72            "{:02x}:{:02x}.{:x}",
73            (bdf >> 8) & 0xFF,
74            (bdf >> 3) & 0x1F,
75            bdf & 0x7
76        )
77    }
78}

80/// Discovery through registration; exits the cell on any failure.
81fn start() {
82    let Some(info) = init::discover() else {
83        // Nothing to claim: no controller, or no PcieDriverCap.
84        ostd::io::println("[virtio-scsi] no controller - exiting");
85        sys_exit(0);
86    };
87    let bdf = Bdf(info.bdf);
88    let record = match sys_query_pcie_function(info.bdf) {
89        Ok(record) => record,
90        Err(error) => {
91            let _ = print_fmt(format_args!(
92                "[virtio-scsi] query {bdf} failed: {error:?}\n"
93            ));
94            sys_exit(1)
95        }
96    };
97    let (queue, limits) = match init::bring_up(&record) {
98        Ok(up) => up,
99        // Refused at construction, or at negotiation when the offer changed
100        // across the reset.
101        Err(
102            InitError::Transport(TransportError::AccessPlatformMissing)
103            | InitError::Negotiation(Some(InitFailure::AccessPlatformMissing)),
104        ) => {
105            let _ = print_fmt(format_args!(
106                "[virtio-scsi] refused {bdf}: VIRTIO_F_ACCESS_PLATFORM required (DMA remapped)\n"
107            ));
108            sys_exit(0)
109        }
110        Err(error) => {
111            let _ = print_fmt(format_args!("[virtio-scsi] init {bdf} failed: {error}\n"));
112            sys_exit(1)
113        }
114    };
115    let Some(disk) = scan::first_disk(queue, &limits) else {
116        let _ = print_fmt(format_args!("[virtio-scsi] no disk on {bdf} - exiting\n"));
117        sys_exit(0)
118    };
119    if disk.block_size() as usize != SECTOR_SIZE {
120        let _ = print_fmt(format_args!(
121            "[virtio-scsi] unsupported block size {} on {bdf} (block protocol needs {SECTOR_SIZE}) - exiting\n",
122            disk.block_size()
123        ));
124        drop(disk); // resets the device before the cell goes away
125        sys_exit(0)
126    }
127    let target = disk.target();
128    let lun = proto::lun_number(disk.lun()).unwrap_or_default();
129    let blocks = disk.block_count();
130
131    // Publish state before registration: VFS may route a request as soon as
132    // the syscall succeeds.
133    *STATE.lock() = Some(disk);
134    if let Err(error) = sys_register_block_device(BlockDevice::Pci {
135        bdf: info.bdf as u16,
136    }) {
137        let disk = STATE.lock().take();
138        drop(disk);
139        let _ = print_fmt(format_args!(
140            "[virtio-scsi] block driver registration failed: {error:?} ({bdf})\n"
141        ));
142        sys_exit(1);
143    }
144    let _ = print_fmt(format_args!(
145        "[virtio-scsi] pci {bdf} {:04x}:{:04x} target={target} lun={lun} blocks={blocks} irq=polled reason=04b-deferred\n",
146        record.vendor_id, record.device_id
147    ));
148    ostd::io::println("[virtio-scsi] ready");
149}

151fn handler(_ctx: &mut AppContext, event: AppEvent) {
152    match event {
153        AppEvent::Init => start(),
154
155        // VFS speaks the raw DrvRequest protocol (no 0xAC App-SDK envelope),
156        // so requests may arrive as RawMessage; accept Message too — layout is
157        // identical. Only the registered VFS may issue block requests; any
158        // other sender gets `NotIssued` without touching the device.
159        AppEvent::Message { sender_tid, data } | AppEvent::RawMessage { sender_tid, data } => {
160            let mut reply = [0u8; dispatch::REPLY_SIZE];
161            let len = if !sender_is_vfs(sender_tid) {
162                let _ = print_fmt(format_args!(
163                    "[virtio-scsi] refused block request from tid={sender_tid}: sender is not the registered VFS\n"
164                ));
165                reply[0] = BlockOutcome::NotIssued.to_byte();
166                1
167            } else if let Some(disk) = STATE.lock().as_mut() {
168                dispatch::handle(disk, data.as_ref(), &mut reply)
169            } else {
170                reply[0] = BlockOutcome::NotIssued.to_byte(); // not initialised
171                1
172            };
173            let _ = ostd::ipc::reply_to(sender_tid, &reply[..len]);
174        }
175
176        AppEvent::Shutdown | AppEvent::ShutdownWith { .. } => sys_exit(0),
177
178        _ => {}
179    }
180}
181
182ostd::run_app!(handler);
