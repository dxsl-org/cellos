1//! What the hardware-independent `xe` backend asks of the GPU and of the kernel.
2//!
3//! [`crate::backend::XeSession`] keeps the uAPI bookkeeping (handles, VMs, queues) and
4//! calls a [`GpuHw`] for every effect that reaches the device: PPGTT leaves, GuC contexts,
5//! ring submission, and the client-memory export leases of CP1 D-G1 option B. The driver
6//! Cell implements it over MMIO, its DMA pool and syscalls 438/439; host tests implement it
7//! with recording fakes. Keeping the effects behind one trait is what lets the uAPI rules be
8//! tested byte for byte without a GPU.
9
10use crate::backend::XeError;
11use crate::pte::PatIndex;
12
13/// A client export as `MapDmaExport` (438) described it (`ViDmaExportV1`, CP1 §5.2).
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub struct ExportRecord {
16    pub export_id: u64,
17    /// The exporting Cell: must equal the identity of the request that carried the id.
18    pub owner_cell_id: u64,
19    pub owner_generation: u64,
20    /// Base of the kernel-chosen IOVA window in the driver's VT-d domain (4 KiB aligned).
21    pub iova: u64,
22    /// Bytes the window covers.
23    pub len: u64,
24    /// `EXPORT_DEVICE_WRITE`: clear means the remapper lets the device only read.
25    pub device_write: bool,
26}
27
28/// Why a 438/439 call failed, mirroring `ostd::syscall::DmaExportError`.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum ExportError {
31    PermissionDenied,
32    NotFound,
33    InvalidInput,
34    NotSupported,
35    OutOfMemory,
36    /// 439: the window drain is not acknowledged yet; the kernel retries it on its own.
37    TryAgain,
38    /// 438: the window was published but its invalidation was not acknowledged.
39    Unconfirmed,
40    /// The kernel wrote a record the v1 decoder refuses.
41    Malformed,
42}
43
44/// The holder half of the export syscalls (CP1 §5.2-5.3).
45pub trait ExportMapper {
46    /// `MapDmaExport` (438): map `export_id` into the driver's device domain. Repeat
47    /// calls return the same record.
48    fn map_export(&mut self, export_id: u64) -> Result<ExportRecord, ExportError>;
49    /// `RevokeDmaExport` (439) as the lease holder. Called only after the GPU can no
50    /// longer reach the window through any PPGTT this driver wrote.
51    fn revoke_export(&mut self, export_id: u64) -> Result<(), ExportError>;
52}
53
54/// A hardware address space (one per-client PPGTT), named by the [`GpuHw`] that made it.
55#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
56pub struct HwVm(pub u32);
57
58/// A hardware execution context (one GuC context with its LRC and ring).
59#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
60pub struct HwQueue(pub u32);
61
62/// Device effects of the `xe` uAPI.
63///
64/// Every method that reports success has completed its effect on the device side (leaves
65/// written, H2G message published); nothing is queued for later.
66pub trait GpuHw: ExportMapper {
67    /// Allocate an empty per-client PPGTT.
68    fn vm_create(&mut self) -> Result<HwVm, XeError>;
69    /// Write 4 KiB leaves for `[va, va + len)` pointing at device addresses
70    /// `[dma, dma + len)`. A failure leaves no leaf of this call behind.
71    fn vm_map(
72        &mut self,
73        vm: HwVm,
74        va: u64,
75        dma: u64,
76        len: u64,
77        pat: PatIndex,
78        writable: bool,
79    ) -> Result<(), XeError>;
80    /// Write `XE_PTE_NULL` leaves for `[va, va + len)`: reads return zero, writes drop.
81    fn vm_map_null(&mut self, vm: HwVm, va: u64, len: u64, pat: PatIndex) -> Result<(), XeError>;
82    /// Clear every present leaf in `[va, va + len)` and invalidate the GPU TLBs, so the
83    /// range is unreachable when this returns.
84    fn vm_unmap(&mut self, vm: HwVm, va: u64, len: u64);
85    /// Free the PPGTT's tables. The caller has unmapped every range and destroyed every
86    /// queue that used the VM.
87    fn vm_destroy(&mut self, vm: HwVm);
88    /// Create and register a GuC context of `engine_class` running in `vm`.
89    fn queue_create(&mut self, vm: HwVm, engine_class: u16) -> Result<HwQueue, XeError>;
90    /// Run the batch at GPU VA `batch_va` on `queue`. `Err(TryAgain)` when the ring has
91    /// no room for another job.
92    fn queue_submit(&mut self, queue: HwQueue, batch_va: u64) -> Result<(), XeError>;
93    /// The `DRM_XE_EXEC_QUEUE_BAN_FLAG_*` bits the GuC reported for `queue`, 0 if none.
94    fn queue_ban(&mut self, queue: HwQueue) -> u32;
95    /// Stop and deregister the context and free its LRC.
96    fn queue_destroy(&mut self, queue: HwQueue);
97}
