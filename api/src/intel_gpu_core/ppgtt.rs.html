1//! Per-client GPU page tables (PPGTT) with 4 KiB leaves, mirroring the level
2//! arithmetic of `xe_pt.c` (`XE_PTE_SHIFT + XE_PDE_SHIFT * level`) and the
3//! `va_bits` / `vm_max_level` platform values of `xe_pci.c` (all MIT).
4//!
5//! Table memory is owned by the caller through [`TableStore`]; this module never
6//! allocates. Unmapping leaves empty intermediate tables in place: they are
7//! reclaimed by [`Ppgtt::destroy`] with the whole VM, matching the lifetime of a client.
8
9use crate::pte::{self, EncodeError, PatIndex, ADDR_MASK_4K, PAGE_PRESENT, PDE_PS_LEAF};
10use alloc::vec::Vec;
11
12pub const ENTRIES: usize = 512;
13
14/// Backing store for 4 KiB, 512-entry tables addressed by physical address.
15pub trait TableStore {
16    /// A zeroed, 4 KiB-aligned table, or `None` when out of memory.
17    fn alloc_table(&mut self) -> Option<u64>;
18    fn read(&self, table: u64, index: usize) -> u64;
19    fn write(&mut self, table: u64, index: usize, value: u64);
20    /// Return a table [`Self::alloc_table`] handed out; no entry of it is referenced.
21    fn free_table(&mut self, table: u64);
22}
23
24/// Address-space shape of one platform.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub struct Layout {
27    pub va_bits: u32,
28    /// Level of the root table (`vm_max_level`).
29    pub max_level: u32,
30    /// Device DMA width (`dma_mask_size`).
31    pub dma_bits: u32,
32}
33
34impl Layout {
35    /// Panther Lake / Wildcat Lake (`ptl_desc`: va_bits 48, vm_max_level 4, dma 46).
36    pub const PTL: Self = Self {
37        va_bits: 48,
38        max_level: 4,
39        dma_bits: 46,
40    };
41
42    fn is_valid(&self) -> bool {
43        (1..=4).contains(&self.max_level)
44            && self.va_bits > 12 + 9 * self.max_level - 9
45            && self.va_bits <= 12 + 9 * (self.max_level + 1)
46            && self.dma_bits <= 52
47    }
48}
49
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub enum MapError {
52    BadLayout,
53    /// Virtual address not 4 KiB aligned or outside `va_bits`.
54    BadVirtual,
55    Encode(EncodeError),
56    /// A leaf (4 KiB or larger) already covers this address.
57    AlreadyMapped,
58    NotMapped,
59    OutOfTables,
60}
61
62pub struct Ppgtt {
63    layout: Layout,
64    root: u64,
65}
66
67fn index(va: u64, level: u32) -> usize {
68    ((va >> (12 + 9 * level)) as usize) & (ENTRIES - 1)
69}
70
71impl Ppgtt {
72    pub fn new(store: &mut impl TableStore, layout: Layout) -> Result<Self, MapError> {
73        if !layout.is_valid() {
74            return Err(MapError::BadLayout);
75        }
76        let root = store.alloc_table().ok_or(MapError::OutOfTables)?;
77        Ok(Self { layout, root })
78    }
79
80    /// Physical address of the root table, for the GuC context descriptor.
81    pub fn root(&self) -> u64 {
82        self.root
83    }
84
85    fn check_va(&self, va: u64) -> Result<(), MapError> {
86        if !va.is_multiple_of(pte::level_span(0)) || va >> self.layout.va_bits != 0 {
87            return Err(MapError::BadVirtual);
88        }
89        Ok(())
90    }
91
92    /// Walk to the level-0 table for `va`, creating tables when `create` is set.
93    fn leaf_table(
94        &self,
95        store: &mut impl TableStore,
96        va: u64,
97        create: bool,
98    ) -> Result<Option<u64>, MapError> {
99        let mut table = self.root;
100        for level in (1..=self.layout.max_level).rev() {
101            let slot = index(va, level);
102            let entry = store.read(table, slot);
103            if entry & PAGE_PRESENT == 0 {
104                if !create {
105                    return Ok(None);
106                }
107                let next = store.alloc_table().ok_or(MapError::OutOfTables)?;
108                let pde = pte::encode_table(next, PatIndex::WB, self.layout.dma_bits)
109                    .map_err(MapError::Encode)?;
110                store.write(table, slot, pde);
111                table = next;
112            } else if level <= 2 && entry & PDE_PS_LEAF != 0 {
113                return Err(MapError::AlreadyMapped);
114            } else {
115                table = entry & ADDR_MASK_4K;
116            }
117        }
118        Ok(Some(table))
119    }
120
121    pub fn map_4k(
122        &mut self,
123        store: &mut impl TableStore,
124        va: u64,
125        phys: u64,
126        pat: PatIndex,
127        writable: bool,
128    ) -> Result<(), MapError> {
129        let leaf = pte::encode_leaf(phys, 0, pat, writable, self.layout.dma_bits)
130            .map_err(MapError::Encode)?;
131        self.install_4k(store, va, leaf)
132    }
133
134    /// Map `va` to a `NULL` leaf ([`pte::encode_null_leaf`]).
135    pub fn map_null_4k(
136        &mut self,
137        store: &mut impl TableStore,
138        va: u64,
139        pat: PatIndex,
140    ) -> Result<(), MapError> {
141        self.install_4k(store, va, pte::encode_null_leaf(pat))
142    }
143
144    fn install_4k(
145        &mut self,
146        store: &mut impl TableStore,
147        va: u64,
148        leaf: u64,
149    ) -> Result<(), MapError> {
150        self.check_va(va)?;
151        let table = self
152            .leaf_table(store, va, true)?
153            .expect("created on demand");
154        let slot = index(va, 0);
155        if store.read(table, slot) & PAGE_PRESENT != 0 {
156            return Err(MapError::AlreadyMapped);
157        }
158        store.write(table, slot, leaf);
159        Ok(())
160    }
161
162    /// Clear the leaf for `va` and return the physical page it pointed at. The
163    /// caller must invalidate the GPU TLB before reusing that page.
164    pub fn unmap_4k(&mut self, store: &mut impl TableStore, va: u64) -> Result<u64, MapError> {
165        self.check_va(va)?;
166        let table = self
167            .leaf_table(store, va, false)?
168            .ok_or(MapError::NotMapped)?;
169        let slot = index(va, 0);
170        let entry = store.read(table, slot);
171        if entry & PAGE_PRESENT == 0 {
172            return Err(MapError::NotMapped);
173        }
174        store.write(table, slot, 0);
175        Ok(entry & ADDR_MASK_4K)
176    }
177
178    /// Physical address `va` maps to, if any.
179    pub fn translate(&self, store: &mut impl TableStore, va: u64) -> Result<Option<u64>, MapError> {
180        let page = va & !(pte::level_span(0) - 1);
181        self.check_va(page)?;
182        let Some(table) = self.leaf_table(store, page, false)? else {
183            return Ok(None);
184        };
185        let entry = store.read(table, index(page, 0));
186        Ok((entry & PAGE_PRESENT != 0).then_some((entry & ADDR_MASK_4K) | (va & 0xfff)))
187    }
188
189    /// Free every table of this address space, root last. The caller has cleared the
190    /// leaves and invalidated the GPU TLBs, and no context still points at the root.
191    pub fn destroy(self, store: &mut impl TableStore) {
192        // (table, level) pairs still to visit; a level-1+ entry without the PS bit
193        // points at a child table.
194        let mut pending = Vec::from([(self.root, self.layout.max_level)]);
195        let mut order = Vec::new();
196        while let Some((table, level)) = pending.pop() {
197            order.push(table);
198            if level == 0 {
199                continue;
200            }
201            for slot in 0..ENTRIES {
202                let entry = store.read(table, slot);
203                let huge_leaf = level <= 2 && entry & PDE_PS_LEAF != 0;
204                if entry & PAGE_PRESENT != 0 && !huge_leaf {
205                    pending.push((entry & ADDR_MASK_4K, level - 1));
206                }
207            }
208        }
209        // Children before parents, so no freed table is ever read again.
210        for table in order.into_iter().rev() {
211            store.free_table(table);
212        }
213    }
214}
