1//! Payloads of `DRM_IOCTL_XE_DEVICE_QUERY` for Mesa `anv`.
2//!
3//! Upstream reference: `drivers/gpu/drm/xe/xe_query.c` and `xe_drm.h` (MIT). Every layout
4//! below is the uapi struct byte for byte. The values come from a [`DeviceInfo`], which the
5//! driver Cell fills from the probed GT registers; the device has one GT and system memory
6//! only.
7
8use crate::uapi::ioctl::EINVAL;
9use alloc::vec;
10use alloc::vec::Vec;
11
12pub const DRM_XE_DEVICE_QUERY_ENGINES: u32 = 0;
13pub const DRM_XE_DEVICE_QUERY_MEM_REGIONS: u32 = 1;
14pub const DRM_XE_DEVICE_QUERY_CONFIG: u32 = 2;
15pub const DRM_XE_DEVICE_QUERY_GT_LIST: u32 = 3;
16pub const DRM_XE_DEVICE_QUERY_HWCONFIG: u32 = 4;
17pub const DRM_XE_DEVICE_QUERY_GT_TOPOLOGY: u32 = 5;
18
19pub const DRM_XE_TOPO_DSS_GEOMETRY: u16 = 1;
20pub const DRM_XE_TOPO_DSS_COMPUTE: u16 = 2;
21pub const DRM_XE_TOPO_SIMD16_EU_PER_DSS: u16 = 5;
22
23/// `sizeof(struct drm_xe_engine)`.
24const ENGINE_SIZE: usize = 32;
25/// `sizeof(struct drm_xe_mem_region)`.
26const MEM_REGION_SIZE: usize = 88;
27/// `sizeof(struct drm_xe_gt)`.
28const GT_SIZE: usize = 96;
29/// `DRM_XE_QUERY_CONFIG_MAX_EXEC_QUEUE_PRIORITY + 1`.
30const CONFIG_PARAMS: usize = 5;
31/// `sizeof(xe_dss_mask_t)`: `XE_MAX_DSS_FUSE_BITS` (128) rounded to `unsigned long`s.
32const DSS_MASK_BYTES: usize = 16;
33/// `sizeof(xe_eu_mask_t)`: `XE_MAX_EU_FUSE_BITS` (32) rounded to one `unsigned long`.
34const EU_MASK_BYTES: usize = 8;
35
36/// What `DEVICE_QUERY` and `DRM_IOCTL_VERSION` report about the GPU.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub struct DeviceInfo {
39    /// `drm_version.desc`.
40    pub name: &'static str,
41    /// GMD_ID graphics architecture (`drm_xe_gt.ip_ver_major`).
42    pub ip_ver_major: u16,
43    /// GMD_ID graphics release (`drm_xe_gt.ip_ver_minor`).
44    pub ip_ver_minor: u16,
45    /// GMD_ID revision (`drm_xe_gt.ip_ver_rev`).
46    pub ip_ver_rev: u16,
47    /// Command streamer timestamp frequency (`drm_xe_gt.reference_clock`).
48    pub reference_clock_hz: u32,
49    /// Geometry DSS fuse mask (`DRM_XE_TOPO_DSS_GEOMETRY`).
50    pub dss_geometry: u128,
51    /// Compute DSS fuse mask (`DRM_XE_TOPO_DSS_COMPUTE`).
52    pub dss_compute: u128,
53    /// SIMD16 EUs enabled in each DSS (`DRM_XE_TOPO_SIMD16_EU_PER_DSS`).
54    pub eu_per_dss: u32,
55    /// `drm_xe_mem_region.total_size` of the system-memory region: the bytes this
56    /// device can make GPU-visible for clients.
57    pub sysmem_bytes: u64,
58}
59
60/// The bytes `query_id` copies to `drm_xe_device_query.data`, or `EINVAL` for an unknown id.
61pub fn query_payload(device: &DeviceInfo, query_id: u32) -> Result<Vec<u8>, i32> {
62    match query_id {
63        DRM_XE_DEVICE_QUERY_ENGINES => {
64            // One RENDER engine, instance 0 on GT 0: all-zero `drm_xe_engine`.
65            let mut out = vec![0u8; 8 + ENGINE_SIZE];
66            put(&mut out, 0, &1u32.to_le_bytes());
67            Ok(out)
68        }
69        DRM_XE_DEVICE_QUERY_MEM_REGIONS => {
70            let mut out = vec![0u8; 8 + MEM_REGION_SIZE];
71            put(&mut out, 0, &1u32.to_le_bytes());
72            // mem_class SYSMEM (0), instance 0
73            put(&mut out, 8 + 4, &4096u32.to_le_bytes()); // min_page_size
74            put(&mut out, 8 + 8, &device.sysmem_bytes.to_le_bytes()); // total_size
75            Ok(out)
76        }
77        DRM_XE_DEVICE_QUERY_CONFIG => {
78            let mut out = vec![0u8; 8 + CONFIG_PARAMS * 8];
79            put(&mut out, 0, &(CONFIG_PARAMS as u32).to_le_bytes());
80            // info[0] REV_AND_DEVICE_ID and info[1] FLAGS stay 0.
81            put(&mut out, 8 + 2 * 8, &4096u64.to_le_bytes()); // MIN_ALIGNMENT
82            put(&mut out, 8 + 3 * 8, &48u64.to_le_bytes()); // VA_BITS
83            put(&mut out, 8 + 4 * 8, &2u64.to_le_bytes()); // MAX_EXEC_QUEUE_PRIORITY
84            Ok(out)
85        }
86        DRM_XE_DEVICE_QUERY_GT_LIST => {
87            let mut out = vec![0u8; 8 + GT_SIZE];
88            put(&mut out, 0, &1u32.to_le_bytes());
89            let gt = 8;
90            // type MAIN (0), tile 0, gt_id 0
91            put(&mut out, gt + 12, &device.reference_clock_hz.to_le_bytes());
92            put(&mut out, gt + 16, &1u64.to_le_bytes()); // near_mem_regions: sysmem
93            put(&mut out, gt + 32, &device.ip_ver_major.to_le_bytes());
94            put(&mut out, gt + 34, &device.ip_ver_minor.to_le_bytes());
95            put(&mut out, gt + 36, &device.ip_ver_rev.to_le_bytes());
96            Ok(out)
97        }
98        DRM_XE_DEVICE_QUERY_GT_TOPOLOGY => {
99            // Entries are packed: each mask follows its 8-byte header directly.
100            let mut out = Vec::with_capacity(3 * 8 + 2 * DSS_MASK_BYTES + EU_MASK_BYTES);
101            let eu = u64::from(device.eu_per_dss).to_le_bytes();
102            for (kind, mask) in [
103                (
104                    DRM_XE_TOPO_DSS_GEOMETRY,
105                    &device.dss_geometry.to_le_bytes()[..],
106                ),
107                (
108                    DRM_XE_TOPO_DSS_COMPUTE,
109                    &device.dss_compute.to_le_bytes()[..],
110                ),
111                (DRM_XE_TOPO_SIMD16_EU_PER_DSS, &eu[..]),
112            ] {
113                out.extend_from_slice(&0u16.to_le_bytes()); // gt_id
114                out.extend_from_slice(&kind.to_le_bytes());
115                out.extend_from_slice(&(mask.len() as u32).to_le_bytes()); // num_bytes
116                out.extend_from_slice(mask);
117            }
118            Ok(out)
119        }
120        _ => Err(EINVAL),
121    }
122}
123
124fn put(out: &mut [u8], offset: usize, bytes: &[u8]) {
125    out[offset..offset + bytes.len()].copy_from_slice(bytes);
