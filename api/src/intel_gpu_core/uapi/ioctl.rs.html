1//! DRM and Linux Xe ioctl definitions and request decoding.
2//!
3//! Upstream reference: `include/uapi/drm/drm.h`, `include/uapi/drm/xe_drm.h` and
4//! `include/uapi/asm-generic/ioctl.h` (MIT / GPL-2.0 WITH Linux-syscall-note).
5//! Struct sizes are the LP64 layouts every Cellos target uses.
6
7pub const DRM_IOCTL_BASE: u32 = 0x64; // 'd'
8pub const DRM_COMMAND_BASE: u32 = 0x40;
9
10// ── Linux ioctl encoding: nr 7:0, type 15:8, size 29:16, dir 31:30 ─────────────
11
12pub const IOC_WRITE: u32 = 1;
13pub const IOC_READ: u32 = 2;
14
15#[inline]
16pub const fn ioctl_nr(cmd: u32) -> u32 {
17    cmd & 0xFF
18}
19
20#[inline]
21pub const fn ioctl_type(cmd: u32) -> u32 {
22    (cmd >> 8) & 0xFF
23}
24
25#[inline]
26pub const fn ioctl_size(cmd: u32) -> usize {
27    ((cmd >> 16) & 0x3FFF) as usize
28}
29
30#[inline]
31pub const fn ioctl_dir(cmd: u32) -> u32 {
32    cmd >> 30
33}
34
35/// `DRM_IOWR(nr, struct)` for a struct of `size` bytes.
36pub const fn drm_iowr(nr: u32, size: usize) -> u32 {
37    ((IOC_READ | IOC_WRITE) << 30) | ((size as u32) << 16) | (DRM_IOCTL_BASE << 8) | nr
38}
39
40/// `DRM_IOW(nr, struct)` for a struct of `size` bytes.
41pub const fn drm_iow(nr: u32, size: usize) -> u32 {
42    (IOC_WRITE << 30) | ((size as u32) << 16) | (DRM_IOCTL_BASE << 8) | nr
43}
