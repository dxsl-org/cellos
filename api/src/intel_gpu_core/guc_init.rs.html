1//! GuC boot parameters (`GUC_CTL_*`) and the WOPCM partition, computed before the
2//! firmware transfer.
3//!
4//! Upstream reference (MIT): `drivers/gpu/drm/xe/xe_guc.c` (`guc_ctl_log_params_flags`,
5//! `guc_ctl_feature_flags`, `guc_ctl_debug_flags`, `guc_ctl_ads_flags`, `guc_ctl_devid`,
6//! `guc_init_params_post_hwconfig`, `guc_write_params`), `xe_guc_fwif.h` (`GUC_CTL_*`
7//! bit layouts), `xe_guc_log.h` (buffer sizes of a non-debug build) and `xe_wopcm.c`
8//! (`xe_wopcm_init`, `__check_layout`, `__wopcm_regs_locked`, `__wopcm_init_regs`).
9
10/// `GUC_CTL_MAX_DWORDS`: parameters written to `SOFT_SCRATCH(1..=14)`.
11pub const GUC_CTL_MAX_DWORDS: usize = 14;
12pub const GUC_CTL_LOG_PARAMS: usize = 0;
13pub const GUC_CTL_WA: usize = 1;
14pub const GUC_CTL_FEATURE: usize = 2;
15pub const GUC_CTL_DEBUG: usize = 3;
16pub const GUC_CTL_ADS: usize = 4;
17pub const GUC_CTL_DEVID: usize = 5;
18
19pub const GUC_LOG_VALID: u32 = 1 << 0;
20pub const GUC_LOG_NOTIFY_ON_HALF_FULL: u32 = 1 << 1;
21pub const GUC_LOG_CAPTURE_ALLOC_UNITS: u32 = 1 << 2;
22pub const GUC_LOG_LOG_ALLOC_UNITS: u32 = 1 << 3;
23/// `GUC_CTL_ENABLE_LITE_RESTORE`.
24pub const GUC_CTL_ENABLE_LITE_RESTORE: u32 = 1 << 4;
25/// `GUC_CTL_ENABLE_SLPC`.
26pub const GUC_CTL_ENABLE_SLPC: u32 = 1 << 2;
27/// `GUC_LOG_DISABLED` in `GUC_CTL_DEBUG`.
28pub const GUC_LOG_DISABLED: u32 = 1 << 6;
29
30/// `XE_GUC_LOG_CRASH_DUMP_BUFFER_SIZE` (non-debug build).
31pub const LOG_CRASH_DUMP_SIZE: u32 = 16 * 1024;
32/// `XE_GUC_LOG_EVENT_DATA_BUFFER_SIZE` (non-debug build).
33pub const LOG_EVENT_DATA_SIZE: u32 = 64 * 1024;
34/// `XE_GUC_LOG_STATE_CAPTURE_BUFFER_SIZE` (non-debug build).
35pub const LOG_STATE_CAPTURE_SIZE: u32 = 1024 * 1024;
36/// `GUC_LOG_SIZE`: a 4 KiB header page plus the three buffers.
37pub const GUC_LOG_SIZE: u32 =
38    4096 + LOG_EVENT_DATA_SIZE + LOG_CRASH_DUMP_SIZE + LOG_STATE_CAPTURE_SIZE;
39
40/// `guc_ctl_log_params_flags` for the non-debug sizes: the crash and event buffers are
41/// not 1 MiB multiples, so they count in 4 KiB units; the capture buffer counts in 1 MiB
42/// units (`GUC_LOG_CAPTURE_ALLOC_UNITS`).
43pub const fn log_params(log_ggtt: u32) -> u32 {
44    const LOG_UNIT: u32 = 4096;
45    const CAPTURE_UNIT: u32 = 1024 * 1024;
46    GUC_LOG_VALID
47        | GUC_LOG_NOTIFY_ON_HALF_FULL
48        | GUC_LOG_CAPTURE_ALLOC_UNITS
49        | (LOG_CRASH_DUMP_SIZE / LOG_UNIT - 1) << 4
50        | (LOG_EVENT_DATA_SIZE / LOG_UNIT - 1) << 6
51        | (LOG_STATE_CAPTURE_SIZE / CAPTURE_UNIT - 1) << 10
52        | (log_ggtt >> 12) << 12
53}
54
55/// `guc_ctl_feature_flags` for this driver: lite restore only. SLPC stays off because the
56/// driver does not run GuC power management (`xe_guc_pc.c`), as upstream does on
57/// `skip_guc_pc` platforms; `GUC_CTL_DISABLE_MULTI_QUEUE` is not needed from GuC 70.66 on
58/// and the pinned release is 70.72.1.
59pub const fn feature_flags() -> u32 {
60    GUC_CTL_ENABLE_LITE_RESTORE
61}
62
63/// `guc_ctl_debug_flags` at `guc_log_level` 1 (non-verbose): verbose logging disabled.
64pub const fn debug_flags() -> u32 {
65    GUC_LOG_DISABLED
66}
67
68/// `guc_ctl_ads_flags`: the ADS page number in `GUC_ADS_ADDR` (bits 21:1).
69pub const fn ads_flags(ads_ggtt: u32) -> u32 {
70    (ads_ggtt >> 12) << 1
71}
72
73/// `guc_ctl_devid`.
74pub const fn devid(device_id: u16, revision: u8) -> u32 {
75    (device_id as u32) << 16 | revision as u32
76}
77
78/// The parameter block `guc_write_params` copies to `SOFT_SCRATCH(1 + i)`.
79///
80/// `GUC_CTL_WA` is 0: `guc_ctl_wa_flags` derives it from the `XE_GT_WA` rule tables
81/// (`xe_wa_oob.rules`), which this driver does not carry. That is also the value upstream
82/// boots its first (hwconfig) load with (`guc_init_params`).
83pub fn params(
84    log_ggtt: u32,
85    ads_ggtt: u32,
86    device_id: u16,
87    revision: u8,
88) -> [u32; GUC_CTL_MAX_DWORDS] {
89    let mut p = [0u32; GUC_CTL_MAX_DWORDS];
90    p[GUC_CTL_LOG_PARAMS] = log_params(log_ggtt);
91    p[GUC_CTL_FEATURE] = feature_flags();
92    p[GUC_CTL_DEBUG] = debug_flags();
93    p[GUC_CTL_ADS] = ads_flags(ads_ggtt);
94    p[GUC_CTL_WA] = 0;
95    p[GUC_CTL_DEVID] = devid(device_id, revision);
96    p
97}

99/// `LNL_WOPCM_SIZE`: graphics 20 and later (`xe_wopcm_size`, `platform >= XE_LUNARLAKE`).
100pub const LNL_WOPCM_SIZE: u32 = 8 << 20;
101/// `max_wopcm_size` for every platform this driver serves.
102pub const MAX_WOPCM_SIZE: u32 = 8 << 20;
103/// `WOPCM_RESERVED_SIZE`: reserved above the HuC image.
104pub const WOPCM_RESERVED_SIZE: u32 = 16 * 1024;
105/// `GUC_WOPCM_RESERVED`: reserved at the start of the GuC region.
106pub const GUC_WOPCM_RESERVED: u32 = 16 * 1024;
107/// `GUC_WOPCM_STACK_RESERVED`.
108pub const GUC_WOPCM_STACK_RESERVED: u32 = 8 * 1024;
109/// `WOPCM_HW_CTX_RESERVED`: reserved at the top for hardware contexts.
110pub const WOPCM_HW_CTX_RESERVED: u32 = 32 * 1024 + 4 * 1024;
111/// `GUC_WOPCM_OFFSET_SHIFT`: the GuC base is 16 KiB aligned.
112pub const GUC_WOPCM_OFFSET_ALIGNMENT: u32 = 1 << 14;
113/// `GUC_WOPCM_SIZE_MASK` (bits 31:12) of `GUC_WOPCM_SIZE` (0xc050).
114pub const GUC_WOPCM_SIZE_MASK: u32 = 0xffff_f000;
115/// `GUC_WOPCM_SIZE_LOCKED`.
116pub const GUC_WOPCM_SIZE_LOCKED: u32 = 1 << 0;
117/// `GUC_WOPCM_OFFSET_MASK` (bits 31:14) of `DMA_GUC_WOPCM_OFFSET` (0xc340).
118pub const GUC_WOPCM_OFFSET_MASK: u32 = 0xffff_c000;
119/// `GUC_WOPCM_OFFSET_VALID`.
120pub const GUC_WOPCM_OFFSET_VALID: u32 = 1 << 0;

122/// The GuC's part of WOPCM.
123#[derive(Clone, Copy, Debug, PartialEq, Eq)]
124pub struct WopcmLayout {
125    pub base: u32,
126    pub size: u32,
127}
128
129/// Why the WOPCM cannot hold the firmware (`-E2BIG` upstream).
130#[derive(Clone, Copy, Debug, PartialEq, Eq)]
131pub enum WopcmError {
132    /// The GuC region overlaps the hardware-context reservation.
133    InvalidLayout,
134    /// The GuC region cannot hold the image plus its reserved space and stack.
135    NoSpaceForGuc,
136    /// The region below the GuC base cannot hold the HuC image plus its reservation.
137    NoSpaceForHuc,
138}
139
140/// `__check_layout`.
141pub fn check_layout(
142    wopcm_size: u32,
143    layout: WopcmLayout,
144    guc_fw_size: u32,
145    huc_fw_size: u32,
146) -> Result<(), WopcmError> {
147    let size = wopcm_size - WOPCM_HW_CTX_RESERVED;
148    if layout.base >= size || layout.size > size - layout.base {
149        return Err(WopcmError::InvalidLayout);
150    }
151    if layout.size < guc_fw_size + GUC_WOPCM_RESERVED + GUC_WOPCM_STACK_RESERVED {
152        return Err(WopcmError::NoSpaceForGuc);
153    }
154    if layout.base < huc_fw_size + WOPCM_RESERVED_SIZE {
155        return Err(WopcmError::NoSpaceForHuc);
156    }
157    Ok(())
158}

160/// The partition `xe_wopcm_init` computes when the registers are not locked yet: the GuC
161/// base right above the HuC image and its reservation, the rest (minus the hardware
162/// context reservation) to the GuC.
163pub fn compute_layout(
164    wopcm_size: u32,
165    guc_fw_size: u32,
166    huc_fw_size: u32,
167) -> Result<WopcmLayout, WopcmError> {
168    let ctx_rsvd = WOPCM_HW_CTX_RESERVED;
169    let base = (huc_fw_size + WOPCM_RESERVED_SIZE).next_multiple_of(GUC_WOPCM_OFFSET_ALIGNMENT);
170    let base = base.min(wopcm_size - ctx_rsvd);
171    let size = (wopcm_size - ctx_rsvd - base) & GUC_WOPCM_SIZE_MASK;
172    let layout = WopcmLayout { base, size };
173    check_layout(wopcm_size, layout, guc_fw_size, huc_fw_size)?;
174    Ok(layout)
175}

177/// `__wopcm_regs_locked`: the partition firmware (BIOS/IFWI) or an earlier load already
178/// locked, from the raw `DMA_GUC_WOPCM_OFFSET` and `GUC_WOPCM_SIZE` values.
179pub fn locked_layout(reg_offset: u32, reg_size: u32) -> Option<WopcmLayout> {
180    (reg_size & GUC_WOPCM_SIZE_LOCKED != 0 && reg_offset & GUC_WOPCM_OFFSET_VALID != 0).then_some(
181        WopcmLayout {
182            base: reg_offset & GUC_WOPCM_OFFSET_MASK,
183            size: reg_size & GUC_WOPCM_SIZE_MASK,
184        },
185    )
186}
