1//! The Linux `xe` uAPI subset Mesa `anv` issues for compute (ADR-0033 G1), decoded at the
2//! uapi struct offsets over one open DRM file's [`XeSession`] and [`Syncobjs`].
3//!
4//! Upstream reference: `include/uapi/drm/drm.h`, `include/uapi/drm/xe_drm.h`,
5//! `drivers/gpu/drm/drm_ioctl.c` and the `xe_*_ioctl` argument checks in
6//! `drivers/gpu/drm/xe/{xe_bo,xe_vm,xe_exec_queue,xe_exec}.c` (MIT). Requests are matched
7//! on the full Linux encoding (direction, type, number, struct size). The driver Cell's
8//! IPC service and the host harnesses run this one decoder, so a request means the same
9//! thing on both.
10

12pub mod query;

25/// The calling process's memory, for the user pointers embedded in ioctl structs.
26pub trait UserMemory {
27    /// Copy `out.len()` bytes from user address `addr`; `Err(EFAULT)` if not readable.
28    fn read(&self, addr: u64, out: &mut [u8]) -> Result<(), i32>;
29    /// Copy `data` to user address `addr`; `Err(EFAULT)` if not writable.
30    fn write(&mut self, addr: u64, data: &[u8]) -> Result<(), i32>;
31}

33/// The kernel-attested Cell incarnation that sent a request.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub struct Sender {
36    pub cell_id: u64,
37    pub generation: u64,
38}

40/// Everything one ioctl may use besides its struct and user memory.
41pub struct IoctlContext<'a, H: GpuHw> {
42    /// Who sent the request: the owner every attached export must have.
43    pub sender: Sender,
44    /// Export ids attached to the request (`SEG_EXPORT`, [`crate::wire`]).
45    pub exports: &'a [u64],
46    pub hw: &'a mut H,
47}

67    /// Serve one ioctl. `arg` is the request's struct, exactly `ioctl_size(request)` bytes;
68    /// like `drm_ioctl`, it may be updated even when an error is returned.
69    pub fn ioctl<H: GpuHw>(
70        &mut self,
71        device: &DeviceInfo,
72        request: u32,
73        arg: &mut [u8],
74        mem: &mut impl UserMemory,
75        ctx: &mut IoctlContext<'_, H>,
76    ) -> Result<(), i32> {
77        if arg.len() != ioctl_size(request) {
78            return Err(EINVAL);
79        }
80        // Only an object creation takes memory; an attachment anywhere else is a
81        // malformed request, not something to ignore.
82        if request != DRM_IOCTL_XE_GEM_CREATE && !ctx.exports.is_empty() {
83            return Err(EINVAL);
84        }
85        let hw = &mut *ctx.hw;
86        match request {
87            DRM_IOCTL_VERSION => version(device, arg, mem),
88            DRM_IOCTL_GET_CAP => {
89                match u64_at(arg, 0) {
90                    DRM_CAP_SYNCOBJ | DRM_CAP_SYNCOBJ_TIMELINE => put(arg, 8, &1u64.to_le_bytes()),
91                    _ => return Err(EINVAL),
92                }
93                Ok(())
94            }
95            DRM_IOCTL_XE_DEVICE_QUERY => {
96                zeroed(arg, 0..8)?;
97                zeroed(arg, 24..40)?;
98                let payload = query_payload(device, u32_at(arg, 8))?;
99                let size = u32::try_from(payload.len()).map_err(|_| EINVAL)?;
100                match u32_at(arg, 12) {
101                    0 => put(arg, 12, &size.to_le_bytes()),
102                    n if n != size => return Err(EINVAL),
103                    _ => mem.write(u64_at(arg, 16), &payload)?,
104                }
105                Ok(())
106            }
107            DRM_IOCTL_XE_GEM_CREATE => self.gem_create(arg, ctx),
108            DRM_IOCTL_XE_GEM_MMAP_OFFSET => {
109                zeroed(arg, 0..8)?;
110                zeroed(arg, 12..16)?; // PCI_BARRIER offsets are not served
111                zeroed(arg, 24..40)?;
112                let offset = self
113                    .session
114                    .gem_mmap_offset(u32_at(arg, 8))
115                    .map_err(xe_to_errno)?;
116                put(arg, 16, &offset.to_le_bytes());
117                Ok(())
118            }
119            DRM_IOCTL_GEM_CLOSE => self
120                .session
121                .gem_close(u32_at(arg, 0), hw)
122                .map_err(xe_to_errno),
123            DRM_IOCTL_XE_VM_CREATE => {
124                zeroed(arg, 0..8)?;
125                zeroed(arg, 16..32)?;
126                let flags = u32_at(arg, 8);
127                if flags & !0xF != 0 {
128                    return Err(EINVAL);
129                }
130                let vm_id = self.session.vm_create(flags, hw).map_err(xe_to_errno)?;
131                put(arg, 12, &vm_id.to_le_bytes());
132                Ok(())
133            }
134            DRM_IOCTL_XE_VM_DESTROY => {
135                zeroed(arg, 4..24)?;
136                self.session
137                    .vm_destroy(u32_at(arg, 0), hw)
138                    .map_err(xe_to_errno)
139            }
140            DRM_IOCTL_XE_VM_BIND => self.vm_bind(arg, mem, hw),
141            DRM_IOCTL_XE_EXEC_QUEUE_CREATE => self.exec_queue_create(arg, mem, hw),
142            DRM_IOCTL_XE_EXEC_QUEUE_DESTROY => {
143                zeroed(arg, 4..24)?;
144                self.session
145                    .exec_queue_destroy(u32_at(arg, 0), &mut self.syncobjs, hw)
146                    .map_err(xe_to_errno)
147            }
148            DRM_IOCTL_XE_EXEC_QUEUE_GET_PROPERTY => {
149                zeroed(arg, 0..8)?;
150                zeroed(arg, 24..40)?;
151                if u32_at(arg, 12) != XE_EXEC_QUEUE_PROPERTY_BAN {
152                    return Err(EINVAL);
153                }
154                let ban = self
155                    .session
156                    .exec_queue_ban(u32_at(arg, 8), hw)
157                    .map_err(xe_to_errno)?;
158                put(arg, 16, &u64::from(ban).to_le_bytes());
159                Ok(())
160            }
161            DRM_IOCTL_XE_EXEC => self.exec(arg, hw),
162            DRM_IOCTL_SYNCOBJ_CREATE => {
163                let handle = self.syncobjs.create(u32_at(arg, 4)).map_err(|_| EINVAL)?;
164                put(arg, 0, &handle.to_le_bytes());
165                Ok(())
166            }
167            DRM_IOCTL_SYNCOBJ_DESTROY => {
168                zeroed(arg, 4..8)?;
169                self.syncobjs.destroy(u32_at(arg, 0)).map_err(|_| EINVAL)
170            }
171            _ => Err(EINVAL),
172        }
173    }

175    /// `drm_xe_gem_create` over the request's one export attachment (CP1 D-G1 option B).
176    ///
177    /// Struct checks run before the export is mapped, so a malformed request has no
178    /// kernel-side effect. Without an attachment there is no memory to place the object
179    /// in, which `xe` reports for a placement it cannot serve: `-EINVAL`.
180    fn gem_create<H: GpuHw>(
181        &mut self,
182        arg: &mut [u8],
183        ctx: &mut IoctlContext<'_, H>,
184    ) -> Result<(), i32> {
185        zeroed(arg, 0..8)?;
186        zeroed(arg, 34..56)?;
187        // Bit 0 is the only region `MEM_REGIONS` reports (sysmem instance 0).
188        if u32_at(arg, 16) != 1 || u32_at(arg, 20) & !0xF != 0 {
189            return Err(EINVAL);
190        }
191        let size = usize::try_from(u64_at(arg, 8)).map_err(|_| EINVAL)?;
192        let &[export_id] = ctx.exports else {
193            return Err(EINVAL);
194        };
195        let record = ctx.hw.map_export(export_id).map_err(export_errno)?;
196        if record.export_id != export_id {
197            return Err(EIO);
198        }
199        // Confused-deputy check (CP1 §5.2): the lease must belong to the Cell incarnation
200        // that sent this request, not to whichever client's id it carries.
201        if (record.owner_cell_id, record.owner_generation)
202            != (ctx.sender.cell_id, ctx.sender.generation)
203        {
204            return Err(EPERM);
205        }
206        let handle = self
207            .session
208            .gem_create(size, &record, u32_at(arg, 24), u16_at(arg, 32))
209            .map_err(xe_to_errno)?;
210        put(arg, 28, &handle.to_le_bytes());
211        Ok(())
212    }

214    /// `drm_xe_vm_bind`: one inline op, or `num_binds` ops at `vector_of_binds`.
215    fn vm_bind(
216        &mut self,
217        arg: &[u8],
218        mem: &impl UserMemory,
219        hw: &mut impl GpuHw,
220    ) -> Result<(), i32> {
221        zeroed(arg, 0..8)?; // extensions
222        zeroed(arg, 16..20)?; // pad
223        zeroed(arg, 104..108)?; // pad2
224        zeroed(arg, 120..136)?; // reserved
225        let num_syncs = u32_at(arg, 108) as usize;
226        if num_syncs > MAX_EXEC_SYNCS {
227            return Err(EINVAL);
228        }
229        let vm_id = u32_at(arg, 8);
230        let num_binds = u32_at(arg, 20) as usize;
231        let ops_bytes = match num_binds {
232            0 => Vec::new(),
233            1 => arg[24..24 + VM_BIND_OP_SIZE].to_vec(),
234            n => {
235                let len = n.checked_mul(VM_BIND_OP_SIZE).ok_or(EINVAL)?;
236                let mut bytes = vec![0u8; len];
237                mem.read(u64_at(arg, 24), &mut bytes)?;
238                bytes
239            }
240        };
241        let ops = ops_bytes
242            .chunks_exact(VM_BIND_OP_SIZE)
243            .map(decode_bind_op)
244            .collect::<Result<Vec<_>, _>>()?;
245        // Bind queues of class VM_BIND are not created by this driver: a named queue is
246        // either unknown or not a bind queue (`xe_vm_bind_ioctl`).
247        let queue = u32_at(arg, 12);
248        if queue != 0 {
249            return Err(if self.session.exec_queues.contains_key(&queue) {
250                EINVAL
251            } else {
252                ENOENT
253            });
254        }
255        // Binds are applied synchronously, but in-syncs cannot be waited on and out-syncs
256        // are not wired to the device's TLB invalidation completion yet: refuse rather
257        // than signal something that has not happened.
258        if num_syncs != 0 {
259            return Err(EOPNOTSUPP);
260        }
261        self.session.vm_bind(vm_id, &ops, hw).map_err(xe_to_errno)
262    }

264    fn exec_queue_create(
265        &mut self,
266        arg: &mut [u8],
267        mem: &impl UserMemory,
268        hw: &mut impl GpuHw,
269    ) -> Result<(), i32> {
270        zeroed(arg, 0..8)?;
271        zeroed(arg, 32..48)?;
272        // Queues are single-engine; `flags` may carry only LOW_LATENCY_HINT.
273        if u16_at(arg, 8) != 1 || u16_at(arg, 10) != 1 || u32_at(arg, 16) & !1 != 0 {
274            return Err(EINVAL);
275        }
276        let mut instance = [0u8; EXEC_QUEUE_INSTANCE_SIZE];
277        mem.read(u64_at(arg, 24), &mut instance)?;
278        // Only the engine `ENGINES` reports exists: RENDER instance 0 on GT 0.
279        if u16_at(&instance, 0) != XE_ENGINE_CLASS_RENDER || instance[2..6] != [0; 4] {
280            return Err(EINVAL);
281        }
282        let q_id = self
283            .session
284            .exec_queue_create(
285                u32_at(arg, 12),
286                XE_ENGINE_CLASS_RENDER,
287                XE_EXEC_QUEUE_PRIORITY_NORMAL,
288                hw,
289            )
290            .map_err(xe_to_errno)?;
291        put(arg, 20, &q_id.to_le_bytes());
292        Ok(())
293    }

295    /// `drm_xe_exec`, with the checks in `xe_exec_ioctl` order: arguments, queue lookup,
296    /// width, ban, then the work.
297    fn exec(&mut self, arg: &[u8], hw: &mut impl GpuHw) -> Result<(), i32> {
298        zeroed(arg, 0..8)?;
299        zeroed(arg, 34..56)?;
300        let num_syncs = u32_at(arg, 12) as usize;
301        if num_syncs > MAX_EXEC_SYNCS {
302            return Err(EINVAL);
303        }
304        let queue = u32_at(arg, 8);
305        let ban = self
306            .session
307            .exec_queue_ban(queue, hw)
308            .map_err(xe_to_errno)?;
309        let num_batch_buffer = u16_at(arg, 32);
310        // Every queue is width 1.
311        if num_batch_buffer > 1 {
312            return Err(EINVAL);
313        }
314        if ban != 0 {
315            return Err(ECANCELED);
316        }
317        // In/out syncobjs are not wired to job completion yet; accepting them would leave
318        // Mesa waiting forever on fences nothing signals.
319        if num_syncs != 0 {
320            return Err(EOPNOTSUPP);
321        }
322        // Zero batches only signal syncs; without syncs there is nothing to do.
323        if num_batch_buffer == 0 {
324            return Ok(());
325        }
326        self.session
327            .exec(queue, u64_at(arg, 24), None, hw)
328            .map(|_| ())
329            .map_err(xe_to_errno)
330    }
331}

333/// One `drm_xe_vm_bind_op`, with the per-op struct checks of `vm_bind_ioctl_check_args`.
334fn decode_bind_op(op: &[u8]) -> Result<BindOp, i32> {
335    zeroed(op, 0..8)?; // extensions
336    zeroed(op, 14..16)?; // pad
337    zeroed(op, 52..80)?; // pad2, reserved
338    let kind = u32_at(op, 40);
339    // `prefetch_region && op != PREFETCH`.
340    if u32_at(op, 48) != 0 && kind != XE_VM_BIND_OP_PREFETCH {
341        return Err(EINVAL);
342    }
343    Ok(BindOp {
344        op: kind,
345        flags: u32_at(op, 44),
346        gem_handle: u32_at(op, 8),
347        obj_offset: u64_at(op, 16),
348        addr: u64_at(op, 32),
349        range: u64_at(op, 24),
350        pat_index: u16_at(op, 12),
351    })
352}

354/// `drm_version`: numbers in place, strings through `drm_copy_field` (truncate to the
355/// caller's buffer, always report the full length).
356fn version(device: &DeviceInfo, arg: &mut [u8], mem: &mut impl UserMemory) -> Result<(), i32> {
357    put(arg, 0, &1i32.to_le_bytes()); // version_major
358    put(arg, 4, &0i32.to_le_bytes()); // version_minor
359    put(arg, 8, &0i32.to_le_bytes()); // version_patchlevel
360    for (len_at, value) in [(16, "xe"), (32, ""), (48, device.name)] {
361        let cap = usize::try_from(u64_at(arg, len_at)).unwrap_or(usize::MAX);
362        let ptr = u64_at(arg, len_at + 8);
363        let n = value.len().min(cap);
364        if n != 0 && ptr != 0 {
365            mem.write(ptr, &value.as_bytes()[..n])?;
366        }
367        put(arg, len_at, &(value.len() as u64).to_le_bytes());
368    }
369    Ok(())
370}

372fn zeroed(arg: &[u8], range: core::ops::Range<usize>) -> Result<(), i32> {
373    if arg[range].iter().all(|&b| b == 0) {
374        Ok(())
375    } else {
376        Err(EINVAL)
377    }
378}

380fn put(arg: &mut [u8], offset: usize, bytes: &[u8]) {
381    arg[offset..offset + bytes.len()].copy_from_slice(bytes);
382}

384/// Little-endian `u16` at `offset` of a uapi struct.
385pub fn u16_at(arg: &[u8], offset: usize) -> u16 {
386    u16::from_le_bytes([arg[offset], arg[offset + 1]])
387}
388
389/// Little-endian `u32` at `offset` of a uapi struct.
390pub fn u32_at(arg: &[u8], offset: usize) -> u32 {
391    let mut bytes = [0u8; 4];
392    bytes.copy_from_slice(&arg[offset..offset + 4]);
393    u32::from_le_bytes(bytes)
394}
395
396/// Little-endian `u64` at `offset` of a uapi struct.
397pub fn u64_at(arg: &[u8], offset: usize) -> u64 {
398    let mut bytes = [0u8; 8];
399    bytes.copy_from_slice(&arg[offset..offset + 8]);
400    u64::from_le_bytes(bytes)
401}

403fn xe_to_errno(err: XeError) -> i32 {
404    match err {
405        XeError::Invalid => EINVAL,
406        XeError::NoEntry => ENOENT,
407        XeError::OutOfMemory => ENOMEM,
408        XeError::Canceled => ECANCELED,
409        XeError::PermissionDenied => EPERM,
410        XeError::NotSupported => EOPNOTSUPP,
411        XeError::TryAgain => EAGAIN,
412        XeError::Io => EIO,
413    }
414}

416/// What a refused 438 means to the client: an unknown or dead lease names no memory,
417/// a lease held for another driver is not this client's to use.
418fn export_errno(err: ExportError) -> i32 {
419    match err {
420        ExportError::NotFound => ENOENT,
421        ExportError::PermissionDenied => EPERM,
422        ExportError::InvalidInput => EINVAL,
423        ExportError::NotSupported => EOPNOTSUPP,
424        ExportError::OutOfMemory => ENOMEM,
425        ExportError::TryAgain | ExportError::Unconfirmed | ExportError::Malformed => EIO,
426    }
427}
