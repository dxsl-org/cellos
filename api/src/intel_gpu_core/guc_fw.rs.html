1//! GuC firmware CSS header (`drivers/gpu/drm/xe/abi/uc_fw_abi.h`, MIT) and the
2//! size/version rules of `parse_css_header` / `guc_read_css_info` in
3//! `drivers/gpu/drm/xe/xe_uc_fw.c`.
4//!
5//! Only the 128-byte public header is read. The uCode and RSA bytes are located,
6//! never interpreted (the firmware licence forbids reverse engineering).
7
8/// Size of `struct uc_css_header`.
9pub const CSS_HEADER_LEN: usize = 128;
10
11/// Oldest GuC release `xe` accepts (`guc_read_css_info`).
12pub const MIN_GUC_RELEASE: Version = Version::new(70, 29, 2);
13
14#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
15pub struct Version {
16    pub major: u8,
17    pub minor: u8,
18    pub patch: u8,
19}
20
21impl Version {
22    pub const fn new(major: u8, minor: u8, patch: u8) -> Self {
23        Self {
24            major,
25            minor,
26            patch,
27        }
28    }
29
30    /// Decode the `CSS_SW_VERSION_UC_{MAJOR,MINOR,PATCH}` fields (bits 23:16, 15:8, 7:0).
31    pub const fn from_css(word: u32) -> Self {
32        Self::new((word >> 16) as u8, (word >> 8) as u8, word as u8)
33    }
34}
35
36/// `CSS_UKERNEL_INFO_BUILDTYPE`.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum BuildType {
39    Production,
40    PreProduction,
41    Debug,
42    Unknown(u8),
43}
44
45/// Located parts of a CSS-format GuC image. Offsets are byte offsets into the blob.
46#[derive(Clone, Copy, Debug, PartialEq, Eq)]
47pub struct GucImage {
48    pub release: Version,
49    /// Submission interface version (`submission_version`).
50    pub compatibility: Version,
51    pub build_type: BuildType,
52    pub private_data_size: u32,
53    pub ucode_offset: usize,
54    pub ucode_len: usize,
55    pub rsa_offset: usize,
56    pub rsa_len: usize,
57}
58
59#[derive(Clone, Copy, Debug, PartialEq, Eq)]
60pub enum FwError {
61    /// Blob shorter than the CSS header (`-ENODATA` upstream).
62    TooShortForHeader,
63    /// `header_size_dw` minus the key/modulus/exponent sizes is not 128 bytes
64    /// (`-EPROTO` upstream), including arithmetic underflow.
65    BadHeaderSize,
66    /// `size_dw` smaller than `header_size_dw`.
67    BadTotalSize,
68    /// Blob shorter than header + uCode + RSA (`-ENOEXEC` upstream).
69    Truncated,
70    /// Release older than [`MIN_GUC_RELEASE`].
71    ReleaseTooOld(Version),
72    /// Release differs from the pinned one (ADR-0033 §5: refuse other versions).
73    NotPinned { found: Version, pinned: Version },
74}
75
76fn word(blob: &[u8], byte_off: usize) -> u32 {
77    let b = &blob[byte_off..byte_off + 4];
78    u32::from_le_bytes([b[0], b[1], b[2], b[3]])
79}
80
81/// Parse and size-check a GuC CSS image.
82pub fn parse(blob: &[u8]) -> Result<GucImage, FwError> {
83    if blob.len() < CSS_HEADER_LEN {
84        return Err(FwError::TooShortForHeader);
85    }
86    let header_dw = word(blob, 4);
87    let size_dw = word(blob, 24);
88    let (key_dw, modulus_dw, exponent_dw) = (word(blob, 28), word(blob, 32), word(blob, 36));
89
90    let css_dw = header_dw
91        .checked_sub(key_dw)
92        .and_then(|v| v.checked_sub(modulus_dw))
93        .and_then(|v| v.checked_sub(exponent_dw))
94        .ok_or(FwError::BadHeaderSize)?;
95    if css_dw as usize * 4 != CSS_HEADER_LEN {
96        return Err(FwError::BadHeaderSize);
97    }
98    let ucode_len = size_dw
99        .checked_sub(header_dw)
100        .ok_or(FwError::BadTotalSize)? as usize
101        * 4;
102    let rsa_len = key_dw as usize * 4;
103    let needed = CSS_HEADER_LEN
104        .checked_add(ucode_len)
105        .and_then(|v| v.checked_add(rsa_len))
106        .ok_or(FwError::Truncated)?;
107    if blob.len() < needed {
108        return Err(FwError::Truncated);
109    }
110
111    let release = Version::from_css(word(blob, 64));
112    if release < MIN_GUC_RELEASE {
113        return Err(FwError::ReleaseTooOld(release));
114    }
115    let build_type = match (word(blob, 124) >> 2) & 0x3 {
116        0 => BuildType::Production,
117        1 => BuildType::PreProduction,
118        2 => BuildType::Debug,
119        other => BuildType::Unknown(other as u8),
120    };
121    Ok(GucImage {
122        release,
123        compatibility: Version::from_css(word(blob, 68)),
124        build_type,
125        private_data_size: word(blob, 120),
126        ucode_offset: CSS_HEADER_LEN,
127        ucode_len,
128        rsa_offset: CSS_HEADER_LEN + ucode_len,
129        rsa_len,
130    })
131}
132
133/// Accept exactly the pinned release (stricter than upstream's major-match rule,
134/// as ADR-0033 §5 requires).
135pub fn check_pinned(image: &GucImage, pinned: Version) -> Result<(), FwError> {
136    if image.release == pinned {
137        Ok(())
138    } else {
139        Err(FwError::NotPinned {
140            found: image.release,
141            pinned,
142        })
143    }
144}
