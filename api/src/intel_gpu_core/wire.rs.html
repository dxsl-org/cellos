1//! `XE_GPU` IPC transport (service id 21, CP1 D-G8): one DRM file operation per request,
2//! the `xe` ioctl carried unchanged.
3//!
4//! The transport adds no GPU semantics (ADR-0033 G1, "no new GPU API"): the request code
5//! and the struct bytes are Linux's. The one thing IPC cannot do in place is touch the
6//! client's memory through the user pointers inside a struct, so the client declares the
7//! user ranges a request may touch:
8//! - an `IN` segment carries its bytes to the driver, which may read only inside it;
9//! - an `OUT` segment is a range the driver may write; the reply returns every write and the
10//!   client applies it only inside a range it declared.
11//! - an `EXPORT` segment is not memory at all: it attaches one client-memory export lease
12//!   (`ExportDmaRange`, syscall 437) to the request, the way an fd rides beside an ioctl.
13//!   Its descriptor is `{addr = export id, len = 0, flags = SEG_EXPORT}` and nothing else;
14//!   the driver maps it with `MapDmaExport` (438) and checks that the lease's owner is the
15//!   request's attested sender. `GEM_CREATE` carries exactly one (CP1 D-G1 option B); every
16//!   other request carries none.
17//!
18//! Anything outside the declared segments fails with `EFAULT`, as a bad user pointer does.
19//!
20//! Request (little-endian): `op u8, segment_count u8, arg_len u16, file u32, request u32,
21//! reserved u32 (0)`, then `segment_count` × `{addr u64, len u32, flags u32}`, the `arg`
22//! bytes, and the bytes of each `IN` segment in order; the counts give the frame length.
23//! Reply: `errno u32 (0 = success), value u32, arg_len u16, write_count u8, reserved u8 (0),
24//! reserved u32 (0)`, then `write_count` × `{addr u64, len u32, reserved u32 (0)}`, the `arg`
25//! bytes and the bytes of each write in order. The layout is recorded in
26//! `docs/abi-boundary-records.md`.

34/// Fixed header of a request and of a reply.
35pub const HEADER_LEN: usize = 16;
36/// One segment (request) or write (reply) descriptor.
37pub const DESCRIPTOR_LEN: usize = 16;
38/// Segments one request may declare. `VERSION` needs three (`VM_BIND` and `EXEC` two at
39/// most, `GEM_CREATE` one attachment); no served ioctl needs more.
40pub const MAX_SEGMENTS: usize = 4;
41/// Writes one reply may carry.
42pub const MAX_WRITES: usize = 8;
43/// Largest request: an `IPC_BUF_SIZE` (4096) receive buffer minus the 32-byte
44/// caller-identity trailer the kernel writes at its end.
45pub const MAX_REQUEST_LEN: usize = 4096 - 32;
46/// Largest reply: one `IPC_BUF_SIZE` IPC payload.
47pub const MAX_REPLY_LEN: usize = 4096;
48
49/// Open a DRM file; the reply `value` is its file id.
50pub const OP_OPEN: u8 = 1;
51/// Close the DRM file named by `file` (`postclose`).
52pub const OP_CLOSE: u8 = 2;
53/// Run `request` on the DRM file named by `file`.
54pub const OP_IOCTL: u8 = 3;
55
56/// The driver may read the segment; its bytes follow the `arg`.
57pub const SEG_IN: u32 = 1 << 0;
58/// The driver may write the segment.
59pub const SEG_OUT: u32 = 1 << 1;
60/// The descriptor attaches export lease `addr`; `len` is 0 and no other flag is set.
61pub const SEG_EXPORT: u32 = 1 << 2;
62/// Largest export id the kernel hands out (`1..=isize::MAX`, CP1 §5.1).
63pub const MAX_EXPORT_ID: u64 = i64::MAX as u64;
64
65/// A user range a request may touch.
66#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
67pub struct Segment {
68    pub addr: u64,
69    pub len: u32,
70    pub flags: u32,
71}
72
73impl Segment {
74    /// An attachment of export lease `export_id`.
75    pub const fn export(export_id: u64) -> Self {
76        Self {
77            addr: export_id,
78            len: 0,
79            flags: SEG_EXPORT,
80        }
81    }
82
83    /// Whether this descriptor is well formed: a non-empty mix of `IN`/`OUT`, or an export
84    /// attachment with a valid id, zero length and no other flag.
85    fn is_valid(&self) -> bool {
86        if self.flags & SEG_EXPORT != 0 {
87            return self.flags == SEG_EXPORT
88                && self.len == 0
89                && (1..=MAX_EXPORT_ID).contains(&self.addr);
90        }
91        self.flags != 0 && self.flags & !(SEG_IN | SEG_OUT) == 0
92    }
93
94    fn covers(&self, addr: u64, len: usize, flag: u32) -> bool {
95        let Some(offset) = addr.checked_sub(self.addr) else {
96            return false;
97        };
98        self.flags & flag != 0
99            && offset
100                .checked_add(len as u64)
101                .is_some_and(|end| end <= u64::from(self.len))
102    }
103}
