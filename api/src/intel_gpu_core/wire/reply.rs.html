8/// Build a reply into `out` (at least [`super::MAX_REPLY_LEN`] bytes) and return its length.
9pub fn encode_reply(
10    out: &mut [u8],
11    errno: i32,
12    value: u32,
13    arg: &[u8],
14    mem: Option<&SegmentMemory<'_>>,
15) -> usize {
16    let writes = mem.map_or(&[][..], |m| &m.writes[..]);
17    let data = mem.map_or(&[][..], |m| &m.data[..]);
18    out[..HEADER_LEN].fill(0);
19    out[0..4].copy_from_slice(&(errno as u32).to_le_bytes());
20    out[4..8].copy_from_slice(&value.to_le_bytes());
21    out[8..10].copy_from_slice(&(arg.len() as u16).to_le_bytes());
22    out[10] = writes.len() as u8;
23    let mut at = HEADER_LEN;
24    for &(addr, len) in writes {
25        out[at..at + DESCRIPTOR_LEN].fill(0);
26        out[at..at + 8].copy_from_slice(&addr.to_le_bytes());
27        out[at + 8..at + 12].copy_from_slice(&(len as u32).to_le_bytes());
28        at += DESCRIPTOR_LEN;
29    }
30    out[at..at + arg.len()].copy_from_slice(arg);
31    at += arg.len();
32    out[at..at + data.len()].copy_from_slice(data);
33    at + data.len()
34}

36/// A decoded reply; borrows the received frame.
37#[derive(Debug)]
38pub struct Reply<'a> {
39    /// `0` on success, else the positive errno.
40    pub errno: i32,
41    pub value: u32,
42    pub arg: &'a [u8],
43    writes: Vec<(u64, &'a [u8])>,
44}
45
46/// Decode a reply to a request that declared `segments`. A write outside the declared
47/// `OUT` ranges is a broken reply and is refused whole (`EFAULT`), never partly applied.
48pub fn decode_reply<'a>(frame: &'a [u8], segments: &[Segment]) -> Result<Reply<'a>, i32> {
49    let count = *frame.get(10).ok_or(EINVAL)? as usize;
50    if frame.len() < HEADER_LEN || frame[11] != 0 || u32_at(frame, 12) != 0 || count > MAX_WRITES {
51        return Err(EINVAL);
52    }
53    let arg_len = u16_at(frame, 8) as usize;
54    let mut at = HEADER_LEN;
55    let mut spans = Vec::with_capacity(count);
56    for _ in 0..count {
57        let d = frame.get(at..at + DESCRIPTOR_LEN).ok_or(EINVAL)?;
58        if u32_at(d, 12) != 0 {
59            return Err(EINVAL);
60        }
61        spans.push((u64_at(d, 0), u32_at(d, 8) as usize));
62        at += DESCRIPTOR_LEN;
63    }
64    let arg = frame.get(at..at + arg_len).ok_or(EINVAL)?;
65    at += arg_len;
66    let mut writes = Vec::with_capacity(count);
67    for (addr, len) in spans {
68        if !segments.iter().any(|s| s.covers(addr, len, SEG_OUT)) {
69            return Err(EFAULT);
70        }
71        writes.push((addr, frame.get(at..at + len).ok_or(EINVAL)?));
72        at += len;
73    }
74    if at != frame.len() {
75        return Err(EINVAL);
76    }
77    Ok(Reply {
78        errno: u32_at(frame, 0) as i32,
79        value: u32_at(frame, 4),
80        arg,
81        writes,
82    })
83}
84
85impl Reply<'_> {
86    /// Apply the driver's user-memory writes to the client's memory.
87    pub fn apply_writes(&self, mem: &mut impl UserMemory) -> Result<(), i32> {
88        for &(addr, bytes) in &self.writes {
89            mem.write(addr, bytes)?;
90        }
91        Ok(())
92    }
