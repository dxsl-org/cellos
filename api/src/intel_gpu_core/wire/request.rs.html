12/// A decoded request; borrows the received frame.
13#[derive(Debug)]
14pub struct Request<'a> {
15    pub op: u8,
16    pub file: u32,
17    pub request: u32,
18    pub arg: &'a [u8],
19    segments: Vec<Segment>,
20    in_data: &'a [u8],
21}
22
23/// Build a request into `out` and return its length. `IN` segment bytes are read from the
24/// caller's own memory, so an unreadable range fails here with the caller's `EFAULT`.
25pub fn encode_request(
26    out: &mut [u8],
27    op: u8,
28    file: u32,
29    request: u32,
30    arg: &[u8],
31    segments: &[Segment],
32    mem: &impl UserMemory,
33) -> Result<usize, i32> {
34    if segments.len() > MAX_SEGMENTS || !segments.iter().all(Segment::is_valid) {
35        return Err(EINVAL);
36    }
37    let arg_len = u16::try_from(arg.len()).map_err(|_| EINVAL)?;
38    let mut len = HEADER_LEN + segments.len() * DESCRIPTOR_LEN;
39    out.get_mut(..len).ok_or(EINVAL)?.fill(0);
40    out[0] = op;
41    out[1] = segments.len() as u8;
42    out[2..4].copy_from_slice(&arg_len.to_le_bytes());
43    out[4..8].copy_from_slice(&file.to_le_bytes());
44    out[8..12].copy_from_slice(&request.to_le_bytes());
45    for (i, s) in segments.iter().enumerate() {
46        let at = HEADER_LEN + i * DESCRIPTOR_LEN;
47        out[at..at + 8].copy_from_slice(&s.addr.to_le_bytes());
48        out[at + 8..at + 12].copy_from_slice(&s.len.to_le_bytes());
49        out[at + 12..at + 16].copy_from_slice(&s.flags.to_le_bytes());
50    }
51    let limit = out.len().min(MAX_REQUEST_LEN);
52    let append = |len: &mut usize, bytes: usize| -> Result<core::ops::Range<usize>, i32> {
53        let end = len
54            .checked_add(bytes)
55            .filter(|&e| e <= limit)
56            .ok_or(ENOMEM)?;
57        let range = *len..end;
58        *len = end;
59        Ok(range)
60    };
61    let range = append(&mut len, arg.len())?;
62    out[range].copy_from_slice(arg);
63    for s in segments.iter().filter(|s| s.flags & SEG_IN != 0) {
64        let range = append(&mut len, s.len as usize)?;
65        mem.read(s.addr, &mut out[range])?;
66    }
67    Ok(len)
68}

70/// Decode the request at the start of `buf`. A receive reports no length, so the header's
71/// counts define the frame; it must lie within `buf` and [`MAX_REQUEST_LEN`]. `EINVAL` for
72/// any malformed frame.
73pub fn decode_request(buf: &[u8]) -> Result<Request<'_>, i32> {
74    let frame = &buf[..buf.len().min(MAX_REQUEST_LEN)];
75    if frame.len() < HEADER_LEN || u32_at(frame, 12) != 0 {
76        return Err(EINVAL);
77    }
78    let count = frame[1] as usize;
79    let arg_len = u16_at(frame, 2) as usize;
80    if count > MAX_SEGMENTS {
81        return Err(EINVAL);
82    }
83    let mut at = HEADER_LEN;
84    let mut segments = Vec::with_capacity(count);
85    let mut in_len = 0usize;
86    for _ in 0..count {
87        let segment = Segment {
88            addr: u64_at(frame.get(at..at + DESCRIPTOR_LEN).ok_or(EINVAL)?, 0),
89            len: u32_at(frame, at + 8),
90            flags: u32_at(frame, at + 12),
91        };
92        if !segment.is_valid() {
93            return Err(EINVAL);
94        }
95        if segment.flags & SEG_IN != 0 {
96            in_len = in_len.checked_add(segment.len as usize).ok_or(EINVAL)?;
97        }
98        segments.push(segment);
99        at += DESCRIPTOR_LEN;
100    }
101    let arg_end = at.checked_add(arg_len).ok_or(EINVAL)?;
102    let end = arg_end
103        .checked_add(in_len)
104        .filter(|&end| end <= frame.len())
105        .ok_or(EINVAL)?;
106    Ok(Request {
107        op: frame[0],
108        file: u32_at(frame, 4),
109        request: u32_at(frame, 8),
110        arg: &frame[at..arg_end],
111        segments,
112        in_data: &frame[arg_end..end],
113    })
114}

116impl Request<'_> {
117    /// The client memory this request exposes, for [`crate::uapi::DrmFile::ioctl`].
118    pub fn memory(&self) -> SegmentMemory<'_> {
119        SegmentMemory {
120            segments: &self.segments,
121            in_data: self.in_data,
122            writes: Vec::new(),
123            data: Vec::new(),
124            // Every write costs its descriptor and bytes in a reply that also carries the
125            // header and the echoed arg.
126            budget: MAX_REPLY_LEN.saturating_sub(HEADER_LEN + self.arg.len()),
127        }
128    }
129
130    /// The export ids attached to this request, in declaration order.
131    pub fn exports(&self) -> Vec<u64> {
132        self.segments
133            .iter()
134            .filter(|s| s.flags == SEG_EXPORT)
135            .map(|s| s.addr)
136            .collect()
137    }
138}

140/// Driver-side [`UserMemory`]: reads from `IN` segments, records writes to `OUT` segments.
141pub struct SegmentMemory<'a> {
142    segments: &'a [Segment],
143    in_data: &'a [u8],
144    /// `(addr, len)` of each recorded write, in order; their bytes are in `data`.
145    pub(super) writes: Vec<(u64, usize)>,
146    pub(super) data: Vec<u8>,
147    /// Reply bytes still free for write descriptors and data.
148    budget: usize,
149}
150
151impl UserMemory for SegmentMemory<'_> {
152    fn read(&self, addr: u64, out: &mut [u8]) -> Result<(), i32> {
153        let mut base = 0usize;
154        for s in self.segments.iter().filter(|s| s.flags & SEG_IN != 0) {
155            if s.covers(addr, out.len(), SEG_IN) {
156                let start = base + (addr - s.addr) as usize;
157                out.copy_from_slice(&self.in_data[start..start + out.len()]);
158                return Ok(());
159            }
160            base += s.len as usize;
161        }
162        Err(EFAULT)
163    }
164
165    fn write(&mut self, addr: u64, data: &[u8]) -> Result<(), i32> {
166        if !self
167            .segments
168            .iter()
169            .any(|s| s.covers(addr, data.len(), SEG_OUT))
170        {
171            return Err(EFAULT);
172        }
173        let cost = DESCRIPTOR_LEN + data.len();
174        if self.writes.len() == MAX_WRITES || cost > self.budget {
175            return Err(ENOMEM);
176        }
177        self.budget -= cost;
178        self.writes.push((addr, data.len()));
179        self.data.extend_from_slice(data);
180        Ok(())
181    }
182}
