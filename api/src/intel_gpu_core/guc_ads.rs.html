1//! The GuC Additional Data Structures (ADS) blob for a GT on which the driver submits only
2//! to render engine instance 0.
3//!
4//! Upstream reference (MIT): `drivers/gpu/drm/xe/xe_guc_ads.c` (`struct __guc_ads_blob`,
5//! the offset helpers, `guc_policies_init`, `fill_engine_enable_masks`,
6//! `guc_golden_lrc_init`, `guc_mapping_table_init`, `guc_doorbell_init`,
7//! `xe_guc_ads_populate_minimal`, `xe_guc_ads_populate`), `xe_guc_fwif.h` (`struct
8//! guc_ads`, `guc_policies`, `guc_gt_system_info`, `guc_engine_usage`,
9//! `guc_um_init_params`), `abi/guc_capture_abi.h` (`guc_mmio_reg_set`) and `xe_lrc.c`
10//! (`xe_gt_lrc_size`, `xe_lrc_engine_state_size`).
11//!
12//! What the pinned GuC 70.x requires, and why each field is here:
13//! - `scheduler_policies`, `gt_system_info` and `private_data` pointers, with the policies
14//!   marked valid: `xe_guc_ads_populate_minimal` writes exactly these, and upstream boots the
15//!   firmware with that minimal blob on every probe, so the firmware is known to accept
16//!   everything else being zero;
17//! - a private-data area of the CSS header's `private_data_size`: GuC scratch it owns;
18//! - for submission, the engine-enable mask and the logical-to-physical mapping table of
19//!   the render engine (`xe_guc_ads_populate`), else the GuC knows no engine to run a
20//!   context on, and the golden-context address and engine-state size of the render
21//!   class (`guc_golden_lrc_init`), which the GuC reads when it resets an engine;
22//! - the doorbell count per SQIDI from `DIST_DBS_POPULATED` (`guc_doorbell_init`).
23//!
24//! Left zero, as in the minimal blob: the engine register save/restore lists
25//! (`guc_mmio_reg_state_init`), the error-capture lists, the workaround KLVs and the
26//! golden context contents (upstream records them after the first submission,
27//! `xe_gt_record_default_lrcs`). Consequence: after a GuC engine reset the engine restarts
28//! from zeroed state without workaround registers restored.
29
30use crate::guc_abi::{GUC_MAX_ENGINE_CLASSES, GUC_MAX_INSTANCES_PER_CLASS, GUC_RENDER_CLASS};
31use crate::lrc::{PPHWSP_SIZE, RENDER_IMAGE_SIZE};
32use alloc::vec;
33use alloc::vec::Vec;
34
35const PAGE: usize = 4096;
36
37/// `sizeof(struct guc_ads)`.
38pub const GUC_ADS_SIZE: usize =
39    16 * 32 * 8 + 5 * 4 + 2 * 16 * 4 + 2 * 4 + 2 * 2 * 16 * 4 + 2 * 4 + 3 * 4 + 11 * 4;
40/// `offsetof(struct guc_ads, scheduler_policies)`.
41pub const ADS_SCHEDULER_POLICIES: usize = 16 * 32 * 8 + 4;
42/// `offsetof(struct guc_ads, gt_system_info)`.
43pub const ADS_GT_SYSTEM_INFO: usize = ADS_SCHEDULER_POLICIES + 4;
44/// `offsetof(struct guc_ads, golden_context_lrca)`.
45pub const ADS_GOLDEN_CONTEXT_LRCA: usize = ADS_GT_SYSTEM_INFO + 3 * 4;
46/// `offsetof(struct guc_ads, eng_state_size)`.
47pub const ADS_ENG_STATE_SIZE: usize = ADS_GOLDEN_CONTEXT_LRCA + 16 * 4;
48/// `offsetof(struct guc_ads, private_data)`.
49pub const ADS_PRIVATE_DATA: usize = ADS_ENG_STATE_SIZE + 16 * 4;
50
51/// `sizeof(struct guc_policies)`.
52pub const POLICIES_SIZE: usize = 16 * 4 + 4 * 4 + 4 * 4;
53/// `offsetof(struct __guc_ads_blob, policies)`.
54pub const POLICIES_OFFSET: usize = GUC_ADS_SIZE;
55/// `sizeof(struct guc_gt_system_info)`.
56pub const SYSTEM_INFO_SIZE: usize = 16 * 32 + 16 * 4 + 16 * 4;
57/// `offsetof(struct __guc_ads_blob, system_info)`.
58pub const SYSTEM_INFO_OFFSET: usize = POLICIES_OFFSET + POLICIES_SIZE;
59/// `sizeof(struct guc_engine_usage)`.
60pub const ENGINE_USAGE_SIZE: usize = 16 * 32 * 32;
61/// `sizeof(struct guc_um_init_params)`.
62pub const UM_INIT_PARAMS_SIZE: usize = 8 + 6 * 4 + 3 * 32;
63/// `guc_ads_regset_offset`: where the dynamic part starts.
64pub const REGSET_OFFSET: usize =
65    SYSTEM_INFO_OFFSET + SYSTEM_INFO_SIZE + ENGINE_USAGE_SIZE + UM_INIT_PARAMS_SIZE;
66
67/// `xe_gt_lrc_size(gt, XE_ENGINE_CLASS_RENDER)` for graphics 20.xx-30.xx: PPHWSP plus the
68/// engine image, no indirect ring state.
69pub const RENDER_LRC_SIZE: usize = (PPHWSP_SIZE + RENDER_IMAGE_SIZE) as usize;
70/// `xe_lrc_reg_size` for graphics 12.50 and later.
71pub const LRC_REG_SIZE: usize = 96 * 4;
72/// `xe_lrc_engine_state_size(gt, XE_ENGINE_CLASS_RENDER)`.
73pub const RENDER_ENGINE_STATE_SIZE: usize = RENDER_IMAGE_SIZE as usize - LRC_REG_SIZE;
74
75/// `guc_ads_golden_lrc_offset` with no register save/restore lists.
76pub const GOLDEN_LRC_OFFSET: usize = REGSET_OFFSET.next_multiple_of(PAGE);
77/// `guc_ads_private_data_offset`: after the golden render context; the workaround KLV and
78/// capture areas are empty.
79pub const PRIVATE_DATA_OFFSET: usize = (GOLDEN_LRC_OFFSET + RENDER_LRC_SIZE).next_multiple_of(PAGE);
80
81/// `GLOBAL_POLICY_DEFAULT_DPC_PROMOTE_TIME_US`.
82pub const DPC_PROMOTE_TIME_US: u32 = 500_000;
83/// `GLOBAL_POLICY_MAX_NUM_WI`.
84pub const MAX_NUM_WORK_ITEMS: u32 = 15;
85/// `GUC_GENERIC_GT_SYSINFO_DOORBELL_COUNT_PER_SQIDI`.
86pub const SYSINFO_DOORBELL_COUNT_PER_SQIDI: usize = 2;
87/// `DOORBELLS_PER_SQIDI_MASK` (bits 23:16) of `DIST_DBS_POPULATED` (0xd08).
88pub const DOORBELLS_PER_SQIDI_SHIFT: u32 = 16;
89
90/// `guc_ads_size`: the blob plus the firmware's private data, page aligned.
91pub const fn ads_size(private_data_size: u32) -> usize {
92    PRIVATE_DATA_OFFSET + (private_data_size as usize).next_multiple_of(PAGE)
93}
94
95/// Build the ADS blob that will sit at GGTT address `ggtt`.
96///
97/// `render_instances` is the mask of render engine instances the GuC may schedule on
98/// (bit 0 = RCS0); `dist_dbs_populated` is the raw `DIST_DBS_POPULATED` register.
99pub fn build(
100    ggtt: u32,
101    private_data_size: u32,
102    render_instances: u32,
103    dist_dbs_populated: u32,
104) -> Vec<u8> {
105    let mut blob = vec![0u8; ads_size(private_data_size)];
106    let mut put =
107        |offset: usize, value: u32| blob[offset..offset + 4].copy_from_slice(&value.to_le_bytes());
108
109    // `guc_policies_init`.
110    put(POLICIES_OFFSET + 16 * 4, DPC_PROMOTE_TIME_US);
111    put(POLICIES_OFFSET + 16 * 4 + 8, MAX_NUM_WORK_ITEMS);
112    put(POLICIES_OFFSET + 16 * 4 + 12, 0); // global_flags
113    put(POLICIES_OFFSET + 16 * 4 + 4, 1); // is_valid
114
115    // `fill_engine_enable_masks`: only the render class is driven.
116    let masks = SYSTEM_INFO_OFFSET + GUC_MAX_ENGINE_CLASSES * GUC_MAX_INSTANCES_PER_CLASS;
117    put(masks + GUC_RENDER_CLASS as usize * 4, render_instances);
118
119    // `guc_doorbell_init` (graphics 12+, integrated).
120    let sysinfo = masks + GUC_MAX_ENGINE_CLASSES * 4;
121    put(
122        sysinfo + SYSINFO_DOORBELL_COUNT_PER_SQIDI * 4,
123        (dist_dbs_populated >> DOORBELLS_PER_SQIDI_SHIFT & 0xff) + 1,
124    );
125
126    // `guc_golden_lrc_init` for the enabled class.
127    if render_instances != 0 {
128        put(
129            ADS_ENG_STATE_SIZE + GUC_RENDER_CLASS as usize * 4,
130            RENDER_ENGINE_STATE_SIZE as u32,
131        );
132        put(
133            ADS_GOLDEN_CONTEXT_LRCA + GUC_RENDER_CLASS as usize * 4,
134            ggtt + GOLDEN_LRC_OFFSET as u32,
135        );
136    }
137
138    put(ADS_SCHEDULER_POLICIES, ggtt + POLICIES_OFFSET as u32);
139    put(ADS_GT_SYSTEM_INFO, ggtt + SYSTEM_INFO_OFFSET as u32);
140    put(ADS_PRIVATE_DATA, ggtt + PRIVATE_DATA_OFFSET as u32);
141
142    // `guc_mapping_table_init`: every slot invalid (`GUC_MAX_INSTANCES_PER_CLASS`) except
143    // the enabled render instances, which map to themselves.
144    let table = &mut blob[SYSTEM_INFO_OFFSET..masks];
145    table.fill(GUC_MAX_INSTANCES_PER_CLASS as u8);
146    for instance in 0..GUC_MAX_INSTANCES_PER_CLASS {
147        if render_instances >> instance & 1 != 0 {
148            table[GUC_RENDER_CLASS as usize * GUC_MAX_INSTANCES_PER_CLASS + instance] =
149                instance as u8;
150        }
151    }
152    blob
153}
