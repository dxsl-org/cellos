1//! Server half of the `XE_GPU` transport ([`crate::wire`]): DRM files per client, keyed by
2//! the kernel-stamped sender tid.
3//!
4//! A client is one sending thread of one Cell incarnation. Its files are reachable only
5//! through its own tid, so file ids and every object id inside a file are private to it
6//! (two clients may both hold file 1 and VM 1). The incarnation (`cell_id`, `generation`)
7//! is recorded with the tid: a tid the kernel reuses for another Cell or a restarted one
8//! finds the previous owner's files released, never inherited. The same incarnation is the
9//! owner every export attached to the client's requests must have.
10
11use crate::hw::GpuHw;
12use crate::uapi::ioctl::{EBADF, EINVAL, EMFILE, ENFILE};
13use crate::uapi::query::DeviceInfo;
14use crate::uapi::{DrmFile, IoctlContext, Sender};
15use crate::wire::{self, OP_CLOSE, OP_IOCTL, OP_OPEN};
16use alloc::collections::BTreeMap;
17
18/// Clients served at once (`ENFILE` beyond).
19pub const MAX_CLIENTS: usize = 64;
20/// DRM files one client may hold open (`EMFILE` beyond).
21pub const MAX_FILES_PER_CLIENT: usize = 16;
22
23/// The kernel-attested sender of one request.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub struct Caller {
26    pub tid: usize,
27    pub cell_id: u64,
28    pub generation: u64,
29}
30
31struct Client {
32    cell_id: u64,
33    generation: u64,
34    next_file: u32,
35    files: BTreeMap<u32, DrmFile>,
36}
37
38/// Every client's DRM files and the device they are served for.
39pub struct XeServer {
40    device: DeviceInfo,
41    clients: BTreeMap<usize, Client>,
42}
43
44impl XeServer {
45    pub fn new(device: DeviceInfo) -> Self {
46        Self {
47            device,
48            clients: BTreeMap::new(),
49        }
50    }
51
52    /// Serve one request frame from `caller` on `hw`, writing the reply into `reply` (at
53    /// least [`wire::MAX_REPLY_LEN`] bytes). Returns the reply length.
54    pub fn handle(
55        &mut self,
56        caller: Caller,
57        frame: &[u8],
58        reply: &mut [u8],
59        hw: &mut impl GpuHw,
60    ) -> usize {
61        let request = match wire::decode_request(frame) {
62            Ok(request) => request,
63            Err(errno) => return wire::encode_reply(reply, errno, 0, &[], None),
64        };
65        self.evict_stale(caller, hw);
66        let exports = request.exports();
67        // Open and close carry no memory: an attachment there is malformed.
68        if request.op != OP_IOCTL && !exports.is_empty() {
69            return wire::encode_reply(reply, EINVAL, 0, &[], None);
70        }
71        match request.op {
72            OP_OPEN => match self.open(caller) {
73                Ok(file) => wire::encode_reply(reply, 0, file, &[], None),
74                Err(errno) => wire::encode_reply(reply, errno, 0, &[], None),
75            },
76            OP_CLOSE => {
77                let errno = self.close(caller.tid, request.file, hw).err().unwrap_or(0);
78                wire::encode_reply(reply, errno, 0, &[], None)
79            }
80            OP_IOCTL => {
81                let Some(file) = self
82                    .clients
83                    .get_mut(&caller.tid)
84                    .and_then(|c| c.files.get_mut(&request.file))
85                else {
86                    return wire::encode_reply(reply, EBADF, 0, &[], None);
87                };
88                let mut arg = request.arg.to_vec();
89                let mut mem = request.memory();
90                let mut ctx = IoctlContext {
91                    sender: Sender {
92                        cell_id: caller.cell_id,
93                        generation: caller.generation,
94                    },
95                    exports: &exports,
96                    hw,
97                };
98                let errno = file
99                    .ioctl(&self.device, request.request, &mut arg, &mut mem, &mut ctx)
100                    .err()
101                    .unwrap_or(0);
102                // `drm_ioctl` copies the struct back whether or not the handler failed.
103                wire::encode_reply(reply, errno, 0, &arg, Some(&mem))
104            }
105            _ => wire::encode_reply(reply, EINVAL, 0, &[], None),
106        }
107    }

109    /// Release every file of the client that sent from `tid` (its thread or Cell died):
110    /// contexts stopped, GPU mappings cleared, export leases revoked. Returns the number of
111    /// files released.
112    pub fn release_client(&mut self, tid: usize, hw: &mut impl GpuHw) -> usize {
113        self.clients
114            .remove(&tid)
115            .map_or(0, |client| release_files(client, hw))
116    }
117
118    /// Clients currently holding at least one open file.
119    pub fn client_count(&self) -> usize {
120        self.clients.len()
121    }
122
123    fn evict_stale(&mut self, caller: Caller, hw: &mut impl GpuHw) {
124        let stale = self
125            .clients
126            .get(&caller.tid)
127            .is_some_and(|c| (c.cell_id, c.generation) != (caller.cell_id, caller.generation));
128        if stale {
129            self.release_client(caller.tid, hw);
130        }
131    }
132
133    fn open(&mut self, caller: Caller) -> Result<u32, i32> {
134        if !self.clients.contains_key(&caller.tid) && self.clients.len() >= MAX_CLIENTS {
135            return Err(ENFILE);
136        }
137        let client = self.clients.entry(caller.tid).or_insert(Client {
138            cell_id: caller.cell_id,
139            generation: caller.generation,
140            next_file: 1,
141            files: BTreeMap::new(),
142        });
143        if client.files.len() >= MAX_FILES_PER_CLIENT {
144            return Err(EMFILE);
145        }
146        let file = client.next_file;
147        client.next_file = file.checked_add(1).ok_or(EMFILE)?;
148        client.files.insert(file, DrmFile::new());
149        Ok(file)
150    }
151
152    fn close(&mut self, tid: usize, file: u32, hw: &mut impl GpuHw) -> Result<(), i32> {
153        let client = self.clients.get_mut(&tid).ok_or(EBADF)?;
154        let mut drm = client.files.remove(&file).ok_or(EBADF)?;
155        drm.release(hw);
156        if client.files.is_empty() {
157            self.clients.remove(&tid);
158        }
159        Ok(())
160    }
161}
162
163fn release_files(client: Client, hw: &mut impl GpuHw) -> usize {
164    let count = client.files.len();
165    for mut file in client.files.into_values() {
166        file.release(hw);
167    }
168    count
169}
