1//! GuC command transport (CT): message headers from
2//! `drivers/gpu/drm/xe/abi/guc_communication_ctb_abi.h` and
3//! `abi/guc_messages_abi.h`, ring arithmetic from `h2g_write` / `g2h_read` in
4//! `drivers/gpu/drm/xe/xe_guc_ct.c` (all MIT).
5//!
6//! The rings are dword arrays shared with the GuC. The caller owns the shared
7//! memory and the descriptor (head/tail/status words) and must order its writes:
8//! message dwords become visible before the descriptor `tail` that publishes them.
9
10/// `struct guc_ct_buffer_desc` is 16 dwords.
11pub const DESC_DWORDS: usize = 16;
12pub const DESC_HEAD: usize = 0;
13pub const DESC_TAIL: usize = 1;
14pub const DESC_STATUS: usize = 2;
15pub const STATUS_OVERFLOW: u32 = 1 << 0;
16pub const STATUS_UNDERFLOW: u32 = 1 << 1;
17pub const STATUS_MISMATCH: u32 = 1 << 2;
18pub const STATUS_DISABLED: u32 = 1 << 3;
19
20/// `CTB_H2G_BUFFER_SIZE` (16 KiB) and `CTB_G2H_BUFFER_SIZE` (128 KiB) in dwords.
21pub const H2G_DWORDS: u32 = 16 * 1024 / 4;
22pub const G2H_DWORDS: u32 = 128 * 1024 / 4;
23/// `GUC_CTB_MAX_DWORDS`: payload dwords after the CT header.
24pub const MAX_MSG_DWORDS: usize = 255;

26/// `GUC_HXG_TYPE_*`.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub enum HxgType {
29    Request = 0,
30    Event = 1,
31    FastRequest = 2,
32    NoResponseBusy = 3,
33    NoResponseRetry = 5,
34    ResponseFailure = 6,
35    ResponseSuccess = 7,
36}
37
38impl HxgType {
39    fn from_bits(bits: u32) -> Option<Self> {
40        Some(match bits {
41            0 => Self::Request,
42            1 => Self::Event,
43            2 => Self::FastRequest,
44            3 => Self::NoResponseBusy,
45            5 => Self::NoResponseRetry,
46            6 => Self::ResponseFailure,
47            7 => Self::ResponseSuccess,
48            _ => return None,
49        })
50    }
51}

53/// CT header: FENCE 31:16, FORMAT 15:12 (= HXG, 0), NUM_DWORDS 7:0.
54pub const fn ct_header(fence: u16, num_dwords: u8) -> u32 {
55    (fence as u32) << 16 | num_dwords as u32
56}
57
58/// Host-originated HXG header: TYPE 30:28, ACTION|DATA0 27:0 (`MAKE_ACTION`).
59pub const fn hxg_header(kind: HxgType, action: u32) -> u32 {
60    (kind as u32) << 28 | (action & 0x0fff_ffff)
61}
62
63/// A decoded HXG header.
64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub struct Hxg {
66    pub from_guc: bool,
67    pub kind: HxgType,
68    /// AUX 27:0 (action + data0 for requests/events, data0 for responses).
69    pub aux: u32,
70}
71
72pub fn decode_hxg(word: u32) -> Option<Hxg> {
73    Some(Hxg {
74        from_guc: word >> 31 == 1,
75        kind: HxgType::from_bits((word >> 28) & 7)?,
76        aux: word & 0x0fff_ffff,
77    })
78}

80#[derive(Clone, Copy, Debug, PartialEq, Eq)]
81pub enum CtError {
82    /// Message empty or longer than [`MAX_MSG_DWORDS`].
83    BadLength,
84    /// Head or tail outside the ring (`GUC_CTB_STATUS_OVERFLOW`).
85    Overflow,
86    /// A G2H header announces more dwords than were published (`UNDERFLOW`).
87    Underflow,
88    /// Not enough free space; retry after the GuC advances `head`.
89    NoSpace,
90    /// The message would cross the end: the tail was zero-filled and reset to 0.
91    /// Publish the new tail, then retry (upstream returns `-EAGAIN`).
92    Wrapped { new_tail: u32 },
93    /// FORMAT field is not HXG.
94    BadFormat,
95}
96
97/// Free dwords for a producer at `tail` (`CIRC_SPACE(tail, head, size)`).
98pub fn space(size: u32, head: u32, tail: u32) -> u32 {
99    head.wrapping_sub(tail.wrapping_add(1)) & (size - 1)
100}

102/// Write one H2G message. `action[0]` is the action word, the rest its data.
103/// Returns the new tail to publish in the descriptor.
104pub fn h2g_write(
105    ring: &mut [u32],
106    head: u32,
107    tail: u32,
108    fence: u16,
109    kind: HxgType,
110    action: &[u32],
111) -> Result<u32, CtError> {
112    let size = ring.len() as u32;
113    debug_assert!(size.is_power_of_two());
114    if action.is_empty() || action.len() > MAX_MSG_DWORDS {
115        return Err(CtError::BadLength);
116    }
117    if head >= size || tail >= size {
118        return Err(CtError::Overflow);
119    }
120    let full = action.len() as u32 + 1;
121    if full > space(size, head, tail) {
122        return Err(CtError::NoSpace);
123    }
124    if tail + full > size {
125        ring[tail as usize..].fill(0);
126        return Err(CtError::Wrapped { new_tail: 0 });
127    }
128    let t = tail as usize;
129    ring[t] = ct_header(fence, action.len() as u8);
130    ring[t + 1] = hxg_header(kind, action[0]);
131    ring[t + 2..t + full as usize].copy_from_slice(&action[1..]);
132    Ok((tail + full) % size)
133}

135/// Read one G2H message into `out` (header excluded). Returns `None` when the ring
136/// is empty, else the fence, the payload length and the new head to publish.
137pub fn g2h_read(
138    ring: &[u32],
139    head: u32,
140    tail: u32,
141    out: &mut [u32; MAX_MSG_DWORDS],
142) -> Result<Option<(u16, usize, u32)>, CtError> {
143    let size = ring.len() as u32;
144    debug_assert!(size.is_power_of_two());
145    if head >= size || tail >= size {
146        return Err(CtError::Overflow);
147    }
148    if head == tail {
149        return Ok(None);
150    }
151    let available = tail.wrapping_sub(head) & (size - 1);
152    let header = ring[head as usize];
153    if (header >> 12) & 0xf != 0 {
154        return Err(CtError::BadFormat);
155    }
156    let len = (header & 0xff) as usize;
157    if len as u32 + 1 > available {
158        return Err(CtError::Underflow);
159    }
160    for (i, slot) in out.iter_mut().take(len).enumerate() {
161        *slot = ring[((head + 1 + i as u32) % size) as usize];
162    }
163    Ok(Some((
164        (header >> 16) as u16,
165        len,
166        (head + 1 + len as u32) % size,
167    )))
168}
