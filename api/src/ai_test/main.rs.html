1//! QEMU oracle for the unified AI inference service (`/bin/ai`) — Spec 24 G2 Level A.
2//!
3//! The oracle drives the *public* SDK path a real application uses (`AiClient` over typed IPC) and
4//! compares the result against the golden values produced by `scripts/gen-ai-test-model.py`, whose
5//! reference forward pass shares no code with the engine:
6//!
7//! 1. `Describe` must report the deployed model, the CPU backend, and this build's limits.
8//! 2. A greedy generation of the golden prompt must return the golden token ids, exactly.
9//! 3. `embed` must return the golden vector within `1e-3` per component.
10//! 4. A session abandoned mid-stream must not wedge the service: the next caller still gets a
11//!    session and a full generation.
12//!
13//! Prints exactly one `[ai-test] PASS` line on success; any deviation prints `[ai-test] FAIL …`
14//! and exits non-zero so the QEMU harness cannot mistake a partial run for a pass.
15
16#![no_std]
17#![no_main]
18#![forbid(unsafe_code)]

37/// Golden values from the fixture generator (same file the deployed model was built from).
38const GOLDEN: &str = include_str!("../../../../models/tiny-llama-64.golden.txt");
39
40/// Poll round trips allowed for one 8-token generation. The service advances four model steps per
41/// poll, so a 7-token prompt plus 8 generated tokens needs about four.
42const MAX_POLLS: usize = 32;
43
44/// Embedding tolerance per component (f32 engine vs the f64 reference).
45const EMBED_TOLERANCE: f32 = 1e-3;
46
47ostd::cell_main!(cell_main);
48
49fn cell_main() {
50    println("[ai-test] AI inference oracle starting");
51    let golden = match Golden::parse(GOLDEN) {
52        Some(golden) => golden,
53        None => fail("golden fixture is unreadable"),
54    };
55
56    let mut client = AiClient::new(ai_sdk::ostd_transport::OstdTransport::new());
57    let info = match client.describe() {
58        Ok(info) => info,
59        Err(AiClientError::NoService) => fail("service::AI is not registered"),
60        Err(error) => fail_with("describe", error),
61    };
62    print("[ai-test] model=");
63    println(info.model.as_str());
64    print("[ai-test] vocab=");
65    print_usize(info.vocab_size as usize);
66    print(" context=");
67    print_usize(info.context_tokens as usize);
68    print(" sessions=");
69    print_usize(info.max_sessions as usize);
70    println("");
71    if info.active_backend != backend::CPU {
72        fail("service did not report the CPU backend");
73    }
74    if info.vocab_size == 0 {
75        fail("service has no model resident");
76    }
77
78    // 1. Greedy generation must reproduce the reference token ids exactly.
79    let params = InferParams::greedy(golden.prompt.as_str(), golden.greedy_ids.len() as u16);
80    let generation = match client.generate(&params, MAX_POLLS) {
81        Ok(generation) => generation,
82        Err(error) => fail_with("generate", error),
83    };
84    if generation.ids != golden.greedy_ids {
85        print("[ai-test] expected tokens ");
86        print_usize(golden.greedy_ids.len());
87        print(" got ");
88        print_usize(generation.ids.len());
89        println("");
90        fail("greedy token ids differ from the reference");
91    }
92    print("[ai-test] greedy ids matched: ");
93    print_usize(generation.ids.len());
94    print(" tokens over ");
95    print_usize(generation.polls);
96    println(" polls");
97
98    // 2. The embedding path must match the reference vector.
99    let values = match client.embed(golden.embed_text.as_str()) {
100        Ok(values) => values,
101        Err(error) => fail_with("embed", error),
102    };
103    if values.len() != golden.embed_values.len() {
104        fail("embedding dimension differs from the reference");
105    }
106    let mut worst = 0.0f32;
107    for (actual, expected) in values.iter().zip(&golden.embed_values) {
108        let delta = (actual - expected).abs();
109        if delta > worst {
110            worst = delta;
111        }
112    }
113    if worst > EMBED_TOLERANCE {
114        fail("embedding vector differs from the reference");
115    }
116    print("[ai-test] embedding matched: ");
117    print_usize(values.len());
118    println(" dims");
119
120    // 3. An abandoned session must not wedge the service.
121    let abandoned = match client.submit(&params) {
122        Ok(request_id) => request_id,
123        Err(error) => fail_with("submit", error),
124    };
125    match client.poll(abandoned, ai_proto::MAX_TOKENS_PER_POLL) {
126        Ok(_) => {}
127        Err(error) => fail_with("poll", error),
128    }
129    if client.cancel(abandoned).is_err() {
130        fail("cancelling an abandoned session failed");
131    }
132    let after = match client.generate(&params, MAX_POLLS) {
133        Ok(generation) => generation,
134        Err(error) => fail_with("generate", error),
135    };
136    if after.ids != golden.greedy_ids {
137        fail("the service did not recover after an abandoned session");
138    }
139    println("[ai-test] abandoned session released; service still serving");
140
141    // 4. The ratified streaming surface: `AiClient::prompt` returns a token stream. Only the
142    //    mock-transport host tests exercise it today, so drive it against the real service here.
143    let future = client.prompt(&params);
144    let mut future = core::pin::pin!(future);
145    let waker = core::task::Waker::noop();
146    let mut context = core::task::Context::from_waker(waker);
147    let mut stream = match future.as_mut().poll(&mut context) {
148        core::task::Poll::Ready(Ok(stream)) => stream,
149        core::task::Poll::Ready(Err(error)) => fail_with("prompt", error),
150        core::task::Poll::Pending => fail("prompt did not resolve on its first poll"),
151    };
152    let mut streamed = Vec::new();
153    for item in &mut stream {
154        match item {
155            Ok(token) => streamed.push(token.id),
156            Err(error) => fail_with("prompt stream", error),
157        }
158    }
159    if streamed != golden.greedy_ids {
160        fail("the token stream did not reproduce the reference ids");
161    }
162    if stream.text() != after.text {
163        fail("the token stream text differs from the drained generation text");
164    }
165    print("[ai-test] prompt stream matched: ");
166    print_usize(streamed.len());
167    println(" tokens");
168
169    println("[ai-test] PASS");
170    sys_exit(0);
171}

173/// Print a failure line and exit non-zero: the harness must never read a partial run as a pass.
174fn fail(reason: &str) -> ! {
175    println("[ai-test] FAIL");
176    println(reason);
177    sys_exit(1);
178}
179
180/// Fail with the client error's typed cause, so the serial log names the exact refusal.
181fn fail_with(context: &str, error: AiClientError) -> ! {
182    ostd::io::print("[ai-test] FAIL ");
183    println(context);
184    ostd::io::print_fmt(format_args!("[ai-test] cause: {error:?}\n"));
185    sys_exit(1);
186}

188/// The golden expectations the deployed model must reproduce.
189struct Golden {
190    prompt: String,
191    greedy_ids: Vec<u32>,
192    embed_text: String,
193    embed_values: Vec<f32>,
194}
195
196impl Golden {
197    /// Parse `key=value` lines, ignoring comments.
198    fn parse(text: &str) -> Option<Self> {
199        let mut golden = Self {
200            prompt: String::new(),
201            greedy_ids: Vec::new(),
202            embed_text: String::new(),
203            embed_values: Vec::new(),
204        };
205        for line in text.lines() {
206            if line.is_empty() || line.starts_with('#') {
207                continue;
208            }
209            let (key, value) = line.split_once('=')?;
210            match key {
211                "prompt" => golden.prompt = String::from(value),
212                "greedy_ids" => golden.greedy_ids = parse_ids(value)?,
213                "embed_text" => golden.embed_text = String::from(value),
214                "embed_values" => golden.embed_values = parse_values(value)?,
215                _ => {}
216            }
217        }
218        if golden.greedy_ids.is_empty() || golden.embed_values.is_empty() {
219            return None;
220        }
221        Some(golden)
222    }
223}
224
225fn parse_ids(value: &str) -> Option<Vec<u32>> {
226    let mut ids = Vec::new();
227    for item in value.split(',') {
228        if item.is_empty() {
229            continue;
230        }
231        ids.push(item.parse::<u32>().ok()?);
232    }
233    Some(ids)
234}
235
236fn parse_values(value: &str) -> Option<Vec<f32>> {
237    let mut values = Vec::new();
238    for item in value.split(',') {
239        if item.is_empty() {
240            continue;
241        }
242        values.push(item.parse::<f32>().ok()?);
243    }
244    Some(values)
245}
