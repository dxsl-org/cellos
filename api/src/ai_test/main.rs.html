1//! QEMU oracle for the unified AI inference service (`/bin/ai`) — Spec 24 G2 Level A.
2//!
3//! The oracle drives the *public* SDK path a real application uses (`AiClient` over typed IPC) and
4//! compares the result against the golden values produced by `scripts/gen-ai-test-model.py`, whose
5//! reference forward pass shares no code with the engine:
6//!
7//! 1. `Describe` must report the deployed model, the CPU backend, and this build's limits.
8//! 2. A greedy generation of the golden prompt must return the golden token ids, exactly.
9//! 3. `embed` must return the golden vector within `1e-3` per component.
10//! 4. A session abandoned mid-stream must not wedge the service: the next caller still gets a
11//!    session and a full generation.
12//!
13//! Prints exactly one `[ai-test] PASS` line on success; any deviation prints `[ai-test] FAIL …`
14//! and exits non-zero so the QEMU harness cannot mistake a partial run for a pass.
15
16#![no_std]
17#![no_main]
18#![forbid(unsafe_code)]

36/// Golden values from the fixture generator (same file the deployed model was built from).
37const GOLDEN: &str = include_str!("../../../../models/tiny-llama-64.golden.txt");
38
39/// Poll round trips allowed for one 8-token generation. The service advances four model steps per
40/// poll, so a 7-token prompt plus 8 generated tokens needs about four.
41const MAX_POLLS: usize = 32;
42
43/// Embedding tolerance per component (f32 engine vs the f64 reference).
44const EMBED_TOLERANCE: f32 = 1e-3;
45
46ostd::cell_main!(cell_main);
47
48fn cell_main() {
49    println("[ai-test] AI inference oracle starting");
50    let golden = match Golden::parse(GOLDEN) {
51        Some(golden) => golden,
52        None => fail("golden fixture is unreadable"),
53    };
54
55    let mut client = AiClient::new(ai_sdk::ostd_transport::OstdTransport::new());
56    let info = match client.describe() {
57        Ok(info) => info,
58        Err(AiClientError::NoService) => fail("service::AI is not registered"),
59        Err(error) => fail_with("describe", error),
60    };
61    print("[ai-test] model=");
62        println(info.model.as_str());
63    print("[ai-test] vocab=");
64        print_usize(info.vocab_size as usize);
65    print(" context=");
66        print_usize(info.context_tokens as usize);
67    print(" sessions=");
68        print_usize(info.max_sessions as usize);
69    println("");
70    if info.active_backend != backend::CPU {
71        fail("service did not report the CPU backend");
72    }
73    if info.vocab_size == 0 {
74        fail("service has no model resident");
75    }
76
77    // 1. Greedy generation must reproduce the reference token ids exactly.
78    let params = InferParams::greedy(golden.prompt.as_str(), golden.greedy_ids.len() as u16);
79    let generation = match client.generate(&params, MAX_POLLS) {
80        Ok(generation) => generation,
81        Err(error) => fail_with("generate", error),
82    };
83    if generation.ids != golden.greedy_ids {
84        print("[ai-test] expected tokens ");
85        print_usize(golden.greedy_ids.len());
86        print(" got ");
87        print_usize(generation.ids.len());
88        println("");
89        fail("greedy token ids differ from the reference");
90    }
91    print("[ai-test] greedy ids matched: ");
92        print_usize(generation.ids.len());
93    print(" tokens over ");
94        print_usize(generation.polls);
95    println(" polls");
96
97    // 2. The embedding path must match the reference vector.
98    let values = match client.embed(golden.embed_text.as_str()) {
99        Ok(values) => values,
100        Err(error) => fail_with("embed", error),
101    };
102    if values.len() != golden.embed_values.len() {
103        fail("embedding dimension differs from the reference");
104    }
105    let mut worst = 0.0f32;
106    for (actual, expected) in values.iter().zip(&golden.embed_values) {
107        let delta = (actual - expected).abs();
108        if delta > worst {
109            worst = delta;
110        }
111    }
112    if worst > EMBED_TOLERANCE {
113        fail("embedding vector differs from the reference");
114    }
115    print("[ai-test] embedding matched: ");
116        print_usize(values.len());
117    println(" dims");
118
119    // 3. An abandoned session must not wedge the service.
120    let abandoned = match client.submit(&params) {
121        Ok(request_id) => request_id,
122        Err(error) => fail_with("submit", error),
123    };
124    match client.poll(abandoned, ai_proto::MAX_TOKENS_PER_POLL) {
125        Ok(_) => {}
126        Err(error) => fail_with("poll", error),
127    }
128    if client.cancel(abandoned).is_err() {
129        fail("cancelling an abandoned session failed");
130    }
131    let after = match client.generate(&params, MAX_POLLS) {
132        Ok(generation) => generation,
133        Err(error) => fail_with("generate", error),
134    };
135    if after.ids != golden.greedy_ids {
136        fail("the service did not recover after an abandoned session");
137    }
138    println("[ai-test] abandoned session released; service still serving");
139
140    println("[ai-test] PASS");
141    sys_exit(0);
142}

144/// Print a failure line and exit non-zero: the harness must never read a partial run as a pass.
145fn fail(reason: &str) -> ! {
146    println("[ai-test] FAIL");
147    println(reason);
148    sys_exit(1);
149}
150
151/// Fail with the client error's typed cause, so the serial log names the exact refusal.
152fn fail_with(context: &str, error: AiClientError) -> ! {
153    ostd::io::print("[ai-test] FAIL ");
154    println(context);
155    ostd::io::print_fmt(format_args!("[ai-test] cause: {error:?}\n"));
156    sys_exit(1);
157}

159/// The golden expectations the deployed model must reproduce.
160struct Golden {
161    prompt: String,
162    greedy_ids: Vec<u32>,
163    embed_text: String,
164    embed_values: Vec<f32>,
165}
166
167impl Golden {
168    /// Parse `key=value` lines, ignoring comments.
169    fn parse(text: &str) -> Option<Self> {
170        let mut golden = Self {
171            prompt: String::new(),
172            greedy_ids: Vec::new(),
173            embed_text: String::new(),
174            embed_values: Vec::new(),
175        };
176        for line in text.lines() {
177            if line.is_empty() || line.starts_with('#') {
178                continue;
179            }
180            let (key, value) = line.split_once('=')?;
181            match key {
182                "prompt" => golden.prompt = String::from(value),
183                "greedy_ids" => golden.greedy_ids = parse_ids(value)?,
184                "embed_text" => golden.embed_text = String::from(value),
185                "embed_values" => golden.embed_values = parse_values(value)?,
186                _ => {}
187            }
188        }
189        if golden.greedy_ids.is_empty() || golden.embed_values.is_empty() {
190            return None;
191        }
192        Some(golden)
193    }
194}
195
196fn parse_ids(value: &str) -> Option<Vec<u32>> {
197    let mut ids = Vec::new();
198    for item in value.split(',') {
199        if item.is_empty() {
200            continue;
201        }
202        ids.push(item.parse::<u32>().ok()?);
203    }
204    Some(ids)
205}
206
207fn parse_values(value: &str) -> Option<Vec<f32>> {
208    let mut values = Vec::new();
209    for item in value.split(',') {
210        if item.is_empty() {
211            continue;
212        }
213        values.push(item.parse::<f32>().ok()?);
214    }
215    Some(values)
216}
