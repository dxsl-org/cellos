27/// Scheduler ticks between mailbox checks while a TcpReady wait is outstanding:
28/// one tick (10 ms) bounds how late a waker is seen.
29const WAKE_POLL_TICKS: u64 = 1;
30
31/// How one TcpReady exchange with the net service ended.
32enum ReadyCall {
33    /// The net service's reply, of this many bytes.
34    Reply(usize),
35    /// A message, normally a `REACTOR_WAKE` from `notify`, reached the mailbox first.
36    Interrupted,
37    /// The budget ran out, or the operation expired, without a reply.
38    Elapsed,
39}
40
41/// One TcpReady exact operation bounded by `budget` scheduler ticks.
42///
43/// The net service parks a waiting request until readiness or its own
44/// deadline, so between short parks the mailbox is checked and any message
45/// ends the call, as the wildcard receive this replaces did: a waker is never
46/// held behind the readiness wait. Dropping the unsettled call cancels it; a
47/// readiness query has no effects to reconcile.
48fn ready_call(
49    net_tid: usize,
50    request: &[u8],
51    reply: &mut [u8],
52    budget: u64,
53) -> Result<ReadyCall, IpcError> {
54    let started = sys_get_scheduler_ticks().ok_or(IpcError::InvalidOperation)?;
55    let elapsed = || sys_get_scheduler_ticks().map_or(budget, |now| now.wrapping_sub(started));
56    let call = loop {
57        match PendingCall::submit(net_tid, request) {
58            Ok(call) => break call,
59            Err(IpcSubmitError::Busy) if elapsed() < budget => sys_yield(),
60            Err(IpcSubmitError::Busy) => return Ok(ReadyCall::Elapsed),
61            Err(IpcSubmitError::PeerGone) => return Err(IpcError::PeerGone),
62            Err(IpcSubmitError::InvalidRequest) => return Err(IpcError::Send),
63        }
64    };
65    loop {
66        if let Some(done) = call.wait_and_take(reply, WAKE_POLL_TICKS, 1)? {
67            return match done.terminal {
68                IpcTerminal::Reply => Ok(ReadyCall::Reply(done.len)),
69                IpcTerminal::PeerGone => Err(IpcError::PeerGone),
70                IpcTerminal::PreDispatchTimeout
71                | IpcTerminal::Indeterminate
72                | IpcTerminal::Cancelled => Ok(ReadyCall::Elapsed),
73            };
74        }
75        let mut mail = [0u8; 64];
76        if let SyscallResult::Ok(sender) = sys_try_recv(0, &mut mail) {
77            if sender > 0 {
78                return Ok(ReadyCall::Interrupted);
79            }
80        }
81        if elapsed() >= budget {
82            return Ok(ReadyCall::Elapsed);
83        }
84    }
85}

87/// An observed I/O readiness event.
88#[derive(Debug, Clone, Copy, PartialEq, Eq)]
89pub struct Event {
90    /// Opaque token / key assigned by the caller.
91    pub key: usize,
92    /// Whether the file/socket is readable.
93    pub readable: bool,
94    /// Whether the file/socket is writable.
95    pub writable: bool,
96}
97
98impl Event {
99    /// Create a new event with arbitrary readability and writability.
100    pub const fn new(key: usize, readable: bool, writable: bool) -> Self {
101        Self {
102            key,
103            readable,
104            writable,
105        }
106    }
107
108    /// An event signaling readability.
109    pub const fn readable(key: usize) -> Self {
110        Self {
111            key,
112            readable: true,
113            writable: false,
114        }
115    }
116
117    /// An event signaling writability.
118    pub const fn writable(key: usize) -> Self {
119        Self {
120            key,
121            readable: false,
122            writable: true,
123        }
124    }
125
126    /// An event signaling both readability and writability.
127    pub const fn all(key: usize) -> Self {
128        Self {
129            key,
130            readable: true,
131            writable: true,
132        }
133    }
134
135    /// An empty event.
136    pub const fn none(key: usize) -> Self {
137        Self {
138            key,
139            readable: false,
140            writable: false,
141        }
142    }
143
144    /// Convert event flags into CellOS `net_ready` mask.
145    pub fn to_net_mask(&self) -> u8 {
146        let mut mask = 0;
147        if self.readable {
148            mask |= net_ready::READ | net_ready::ACCEPT;
149        }
150        if self.writable {
151            mask |= net_ready::WRITE;
152        }
153        mask
154    }
155}

157/// A container for returned events.
158#[derive(Debug, Default)]
159pub struct Events {
160    list: Vec<Event>,
161}
162
163impl Events {
164    /// Create an empty event list.
165    pub fn new() -> Self {
166        Self { list: Vec::new() }
167    }
168
169    /// Create an empty event list with pre-allocated capacity.
170    pub fn with_capacity(capacity: usize) -> Self {
171        Self {
172            list: Vec::with_capacity(capacity),
173        }
174    }
175
176    /// Clear all events in the list.
177    pub fn clear(&mut self) {
178        self.list.clear();
179    }
180
181    /// Check if the event list is empty.
182    pub fn is_empty(&self) -> bool {
183        self.list.is_empty()
184    }
185
186    /// Return the number of events.
187    pub fn len(&self) -> usize {
188        self.list.len()
189    }
190
191    /// Iterate over observed events.
192    pub fn iter(&self) -> core::slice::Iter<'_, Event> {
193        self.list.iter()
194    }
195
196    /// Push an event into the list.
197    pub fn push(&mut self, event: Event) {
198        self.list.push(event);
199    }
200}

202/// Errors returned by `Poller` operations.
203#[derive(Debug, Clone, Copy, PartialEq, Eq)]
204pub enum PollerError {
205    /// The requested resource was already registered.
206    AlreadyExists,
207    /// The requested resource was not found.
208    NotFound,
209    /// Network service is unavailable.
210    NetServiceUnavailable,
211    /// Invalid arguments or interest mask.
212    InvalidInput,
213    /// IPC transport or kernel error.
214    IpcError,
215}

217/// Poller managing socket interest registrations and multiplexing readiness events.
218pub struct Poller {
219    /// Registered socket interests: socket_cap -> (key, net_mask).
220    interests: Mutex<BTreeMap<u32, (usize, u8)>>,
221}
222
223impl Poller {
224    /// Create a new Poller instance.
225    pub fn new() -> Result<Self, PollerError> {
226        Ok(Self {
227            interests: Mutex::new(BTreeMap::new()),
228        })
229    }
230
231    /// Register a socket capability and its desired interest.
232    pub fn add(&self, socket_cap: u32, event: Event) -> Result<(), PollerError> {
233        let mut lock = self.interests.lock();
234        if lock.contains_key(&socket_cap) {
235            return Err(PollerError::AlreadyExists);
236        }
237        let mask = event.to_net_mask();
238        if mask == 0 {
239            return Err(PollerError::InvalidInput);
240        }
241        lock.insert(socket_cap, (event.key, mask));
242        Ok(())
243    }
244
245    /// Modify the interest or key for an already-registered socket capability.
246    pub fn modify(&self, socket_cap: u32, event: Event) -> Result<(), PollerError> {
247        let mut lock = self.interests.lock();
248        let entry = lock.get_mut(&socket_cap).ok_or(PollerError::NotFound)?;
249        let mask = event.to_net_mask();
250        if mask == 0 {
251            return Err(PollerError::InvalidInput);
252        }
253        *entry = (event.key, mask);
254        Ok(())
255    }
256
257    /// Deregister a socket capability from the poller.
258    pub fn delete(&self, socket_cap: u32) -> Result<(), PollerError> {
259        let mut lock = self.interests.lock();
260        lock.remove(&socket_cap).ok_or(PollerError::NotFound)?;
261        Ok(())
262    }
263
264    /// Interrupt any ongoing or future `wait()` call on this reactor.
265    ///
266    /// Issues a kernel-coalesced `REACTOR_WAKE` (0x12) message to self (target=0).
267    pub fn notify(&self) -> Result<(), PollerError> {
268        let wake = [REACTOR_WAKE_OPCODE];
269        match sys_try_send(0, &wake) {
270            SyscallResult::Ok(_) => Ok(()),
271            SyscallResult::Err(_) => Err(PollerError::IpcError),
272        }
273    }
274
275    /// Wait for readiness events across registered sockets.
276    pub fn wait(
277        &self,
278        events: &mut Events,
279        timeout: Option<Duration>,
280    ) -> Result<usize, PollerError> {
281        events.clear();
282
283        // 1. Drain any pending same-cell notify wakeup
284        let mut drain_buf = [0u8; 64];
285        if let SyscallResult::Ok(sender) = sys_try_recv(0, &mut drain_buf) {
286            if sender > 0 && drain_buf.first() == Some(&REACTOR_WAKE_OPCODE) {
287                // Notified: return empty events (interrupted wait)
288                return Ok(0);
289            }
290        }
291
292        // 2. Snapshot current interests
293        let snapshot: Vec<TcpInterest> = {
294            let lock = self.interests.lock();
295            lock.iter()
296                .map(|(&cap_id, &(_, mask))| TcpInterest { cap_id, mask })
297                .collect()
298        };
299
300        if snapshot.is_empty() {
301            // No sockets registered: park for timeout if any, otherwise return immediately
302            if let Some(d) = timeout {
303                if !d.is_zero() {
304                    let ticks = (d.as_millis() / 10).max(1) as u64;
305                    let mut dummy = [0u8; 8];
306                    let _ = sys_recv_timeout(0, &mut dummy, ticks);
307                }
308            }
309            return Ok(0);
310        }
311
312        // 3. Resolve Net Service
313        let net_tid = provider_tid(service::NET).ok_or(PollerError::NetServiceUnavailable)?;
314
315        // 4. Encode interests into packed buffer
316        let mut packed = [0u8; 256 * NET_READY_RECORD_BYTES];
317        let Some(encoded_interests) = encode_tcp_interests(&snapshot, &mut packed) else {
318            return Err(PollerError::InvalidInput);
319        };
320
321        let wait_flag = timeout != Some(Duration::ZERO);
322        let request = NetRequest::TcpReady {
323            interests: encoded_interests,
324            cursor: 0,
325            wait: wait_flag,
326        };
327
328        let mut req_buf = [0u8; 1536];
329        let Ok(req_bytes) = wire::ipc::encode(&request, &mut req_buf) else {
330            return Err(PollerError::InvalidInput);
331        };
332
333        // 5. Exchange the request for its reply, interruptible by notify
334        let timeout_ticks = timeout
335            .map(|d| (d.as_millis() / 10).max(1) as u64)
336            .unwrap_or(3000); // Default 30s timeout if None
337
338        let mut resp_buf = [0u8; 1536];
339        let len = match ready_call(net_tid, req_bytes, &mut resp_buf, timeout_ticks) {
340            Ok(ReadyCall::Reply(len)) => len,
341            // Woken by notify (or other mail), or the timeout expired.
342            Ok(ReadyCall::Interrupted | ReadyCall::Elapsed) => return Ok(0),
343            Err(_) => return Err(PollerError::IpcError),
344        };
345
346        // Decode NetResponse
347        let Ok(response) = wire::ipc::decode::<NetResponse<'_>>(&resp_buf[..len]) else {
348            return Err(PollerError::IpcError);
349        };
350
351        match response {
352            NetResponse::TcpReady {
353                events: ready_slice,
354                ..
355            } => {
356                let lock = self.interests.lock();
357                for chunk in ready_slice.chunks_exact(NET_READY_RECORD_BYTES) {
358                    if let Some(ev) = TcpReadyEvent::decode(chunk) {
359                        if let Some(&(key, _)) = lock.get(&ev.cap_id) {
360                            let readable = ev.ready & (net_ready::READ | net_ready::ACCEPT) != 0;
361                            let writable = ev.ready & net_ready::WRITE != 0;
362                            if readable || writable {
363                                events.push(Event::new(key, readable, writable));
364                            }
365                        }
366                    }
367                }
368                Ok(events.len())
369            }
370            NetResponse::NotReady => Ok(0),
371            _ => Ok(0),
372        }
373    }
374}

376#[cfg(test)]
377mod tests {
378    use super::*;
379
380    #[test]
381    fn test_event_mask_conversions() {
382        let r = Event::readable(10);
383        assert!(r.readable);
384        assert!(!r.writable);
385        assert_eq!(r.to_net_mask(), net_ready::READ | net_ready::ACCEPT);
386
387        let w = Event::writable(20);
388        assert!(!w.readable);
389        assert!(w.writable);
390        assert_eq!(w.to_net_mask(), net_ready::WRITE);
391
392        let all = Event::all(30);
393        assert!(all.readable && all.writable);
394        assert_eq!(
395            all.to_net_mask(),
396            net_ready::READ | net_ready::ACCEPT | net_ready::WRITE
397        );
398    }
399
400    #[test]
401    fn test_poller_registration_lifecycle() {
402        let poller = Poller::new().expect("poller creation");
403        let socket_cap = 1001;
404
405        // Add socket interest
406        assert_eq!(poller.add(socket_cap, Event::readable(1)), Ok(()));
407
408        // Duplicate add must fail with AlreadyExists
409        assert_eq!(
410            poller.add(socket_cap, Event::writable(1)),
411            Err(PollerError::AlreadyExists)
412        );
413
414        // Modify existing interest
415        assert_eq!(poller.modify(socket_cap, Event::all(2)), Ok(()));
416
417        // Delete existing socket
418        assert_eq!(poller.delete(socket_cap), Ok(()));
419
420        // Delete non-existent socket must fail with NotFound
421        assert_eq!(poller.delete(socket_cap), Err(PollerError::NotFound));
422    }

424    #[test]
425    fn test_events_container() {
426        let mut events = Events::with_capacity(16);

430        events.push(Event::readable(1));
