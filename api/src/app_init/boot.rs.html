12// QEMU-virt VirtIO drivers. BCM2837 exposes no VirtIO MMIO window, so probing
13// them on board-rpi3 is a guaranteed data abort at 0x0A000000; storage is the
14// in-kernel Arasan SDHCI path and networking is the LAN9514 via /bin/dwc2-usb.
15#[cfg(not(feature = "board-rpi3"))]
16const VIRTIO_BLOCK_DRIVER: &str = "/bin/block";
17#[cfg(not(feature = "board-rpi3"))]
18const VIRTIO_NET_DRIVER: &str = "/bin/virtio-net";

20pub(crate) fn start_block_drivers() {
21    #[cfg(not(feature = "board-rpi3"))]
22    let _ = sys_spawn_from_path(VIRTIO_BLOCK_DRIVER);
23    let _ = sys_spawn_from_path("/bin/nvme");
24    for _ in 0..400 {
25        if sys_lookup_service(service::BLOCK_DRIVER).is_some() {
26            break;
27        }
28        ostd::task::yield_now();
29    }
30}

32pub(crate) fn prepare_service(path: &str) {
33    if path == "/bin/net" {
34        #[cfg(not(feature = "board-rpi3"))]
35        let _ = sys_spawn_from_path(VIRTIO_NET_DRIVER);
36        let _ = sys_spawn_from_path("/bin/e1000");
37        let _ = sys_spawn_from_path("/bin/dwc2-usb");
38        if sys_lookup_service(service::BLOCK_DRIVER).is_none() {
39            let _ = sys_spawn_from_path("/bin/nvme");
40        }
41        for _ in 0..4 {
42            ostd::task::yield_now();
43        }
44    }
45    if path == "/bin/compositor" {
46        let _ = sys_spawn_from_path(DISPLAY_DRIVER_PATH);
47        for _ in 0..4 {
48            ostd::task::yield_now();
49        }
50    }
51}

53pub(crate) fn spawn_hypervisor() -> Option<usize> {
54    match sys_spawn_from_path("/bin/hypervisor") {
55        SyscallResult::Ok(tid) => {
56            let _ = sys_notify_on_exit(tid);
57            if let SyscallResult::Err(_) =
58                sys_register_service(api::hypervisor::HYPERVISOR_SERVICE_ID, tid)
59            {
60                ostd::io::println("Init: hypervisor service registration failed.");
61                None
62            } else {
63                Some(tid)
64            }
65        }
66        _ => None,
67    }
68}

70pub(crate) fn spawn_optional_services() -> Option<usize> {
71    // fb-console mirrors the kernel user log to the display, so it stays on for
72    // RPi3: a board whose only console is the serial header shows nothing on
73    // HDMI otherwise. Bitmap text on a compositor surface, not a TTY.
74    #[cfg(not(feature = "hypervisor-min"))]
75    match sys_spawn_from_path("/bin/fb-console") {
76        SyscallResult::Ok(_) => ostd::io::println("Init: fb-console spawned."),
77        SyscallResult::Err(_) => ostd::io::println("Init: fb-console spawn failed."),
78    }
79    // The desktop shell is not auto-started on RPi3: it is packaged in the
80    // image and launched on demand (`desktop &`), which keeps the board's
81    // console and compositor free during device bring-up.
82    #[cfg(all(not(feature = "hypervisor-min"), not(feature = "board-rpi3")))]
83    match sys_spawn_from_path("/bin/desktop") {
84        SyscallResult::Ok(_) => ostd::io::println("Init: desktop spawned."),
85        SyscallResult::Err(_) => ostd::io::println("Init: desktop spawn failed."),
86    }
87
88    #[cfg(not(feature = "board-rpi3"))]
89    let hypervisor_tid = spawn_hypervisor();
90    #[cfg(feature = "board-rpi3")]
91    let hypervisor_tid = None;
92
93    #[cfg(all(not(feature = "hypervisor-min"), not(feature = "board-rpi3")))]
94    {
95        let _ = sys_spawn_from_path("/bin/silo-test");
96        // Spec 24 G2 Level A oracle: exercises the inference service over typed IPC.
97        let _ = sys_spawn_from_path("/bin/ai-test");
98        let _ = sys_spawn_from_path("/bin/vfs-test");
99        let _ = sys_spawn_from_path("/bin/srv-test");
100        let _ = sys_spawn_from_path("/bin/std-smoke");
101    }
102    hypervisor_tid
103}
