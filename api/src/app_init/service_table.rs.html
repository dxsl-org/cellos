4#[derive(Clone, Copy)]
5pub(crate) enum RestartPolicy {
6    Permanent,
7    /// Restarted only while it has not exited cleanly; both profiles' shells use
8    /// it, so it is not gated on the profile any more.
9    Transient,
10    #[allow(dead_code)]
11    Temporary,
12}

14#[derive(Clone, Copy)]
15pub(crate) enum Registration {
16    /// The shell starts unregistered in every profile; the server and embedded
17    /// tables register every other service they launch.
18    #[allow(dead_code)]
19    None,
20    Init(u16),
21    #[cfg(feature = "development-silo-provider")]
22    SelfReady(u16),
23}

25pub(crate) struct Service {

27    pub(crate) registration: Registration,
28    pub(crate) policy: RestartPolicy,

34impl Service {
35    const fn new(path: &'static str, registration: Registration, policy: RestartPolicy) -> Self {
36        Self {
37            path,
38            registration,
39            policy,
40            tid: None,
41            restart_count: 0,
42            window_start: 0,
43        }
44    }
45
46    pub(crate) const fn service_id(&self) -> Option<u16> {
47        match self.registration {
48            Registration::None => None,
49            Registration::Init(id) => Some(id),
50            #[cfg(feature = "development-silo-provider")]
51            Registration::SelfReady(id) => Some(id),
52        }
53    }
54}

56/// Services every image carries: storage, the network service and a shell on a
57/// serial console. This is the embedded-first floor -- Cellos boots to a prompt
58/// on a board with nothing but a UART, and everything else is an option.
59pub(crate) const BASE_SERVICES: usize = 3;

61/// One term per option, so adding an option means adding one `#[cfg]` entry to
62/// the table below and one term here. A missing term is a compile error, not a
63/// silently truncated table: `configured()` must produce exactly this many.
64pub(crate) const SERVICE_COUNT: usize = BASE_SERVICES
65    + cfg!(feature = "input") as usize
66    + cfg!(feature = "ai") as usize * 2
67    + cfg!(feature = "ui") as usize
68    + (cfg!(feature = "ui") && !cfg!(feature = "board-rpi3")) as usize
69    + cfg!(any(
70        feature = "supervisor",
71        feature = "hostile-backend-recovery"
72    )) as usize
73    + cfg!(feature = "c2c-broker") as usize
74    + cfg!(feature = "development-silo-provider") as usize;

154pub(crate) fn spawn(service: &mut Service) -> Option<usize> {
155    let tid = match sys_spawn_from_path(service.path) {
156        SyscallResult::Ok(tid) => tid,
157        _ => {
158            service.tid = None;
159            return None;
160        }
161    };
162    service.tid = Some(tid);
163    if let Registration::Init(service_id) = service.registration {
164        let _ = sys_register_service(service_id, tid);
165    }
166    Some(tid)
167}
168/// Scheduler ticks (10 ms slices) — the clock the kernel itself uses for `RecvTimeout`
169/// deadlines, and the one every `*_TICKS` constant in this crate assumes.
170///
171/// `sys_get_time()` is `GetTime` **op 0**, the raw architected counter (10 MHz `mtime` on
172/// RV64), so a window compared against it is a thousand times too small: the restart
173/// budget rolled on every exit and `init` never gave up on a crash-looping service.
174pub(crate) fn now_ticks() -> u64 {
175    ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0)
176}

178#[cfg(feature = "development-silo-provider")]
179pub(crate) fn wait_for_exact_registration(service_id: u16, expected_tid: usize) -> bool {
180    const READY_TIMEOUT_TICKS: u64 = 5_000;
181    let started = now_ticks();
182    loop {
183        match ostd::syscall::sys_lookup_service(service_id) {
184            Some(tid) => return tid == expected_tid,
185            None if now_ticks().wrapping_sub(started) >= READY_TIMEOUT_TICKS => {
186                return false;
187            }
188            None => ostd::task::yield_now(),
189        }
190    }
191}
