1//! The machine-pin gate as seen from init (ADR-0029 §3, Phase 13 amendment).
2//!
3//! Init launches the `machine-pin` Cell right after the VFS (x86 only),
4//! registers it as `MACHINE_PIN`, and asks it `MachinePinRequest::Gate` as a
5//! client. Init answers no request itself. Safe mode is init's to apply: it
6//! starts only the static safe set (the `input` service and the `cellbox`
7//! console shell) and tells the pin Cell the shell's tid one way
8//! (`MachinePinRequest::ConsoleShell`), the only sender whose `Confirm` the
9//! Cell honours. Without an answer the gate fails closed: safe mode.
10
11use alloc::format;
12use alloc::string::String;
13use alloc::vec::Vec;
14use ostd::io::println;
15use ostd::ipc::{IpcTerminal, PendingCall};
16use wire::machine_pin::{MachinePinRequest, MachinePinResponse, PinState};
17
18use crate::service_table::{self, Service};
19
20/// The safe set, by plan entry name (CP1 Q6).
21const SAFE_SERVICE: &str = "input";
22const SAFE_SHELL: &str = "cellbox";
23/// One `Gate` call: 31 rounds of 100 scheduler ticks, just past the kernel's
24/// 30 s operation deadline, so an expired submission is resubmitted.
25const GATE_ROUND_TICKS: u64 = 100;
26const GATE_ROUNDS: usize = 31;
27/// Submissions before the gate fails closed (about two minutes in all): the
28/// Cell answers only after its first VFS reads and writes of the pin store.
29const GATE_ATTEMPTS: usize = 4;
30
31/// What the pin Cell decided for this boot.
32pub(crate) struct Decision {
33    /// `None` off x86, or when the Cell could not be started or gave no answer.
34    pub(crate) state: Option<PinState>,
35    pub(crate) safe_mode: bool,
36}
37
38/// The safe-mode banner's reason for `state`.
39fn reason(state: Option<PinState>) -> &'static str {
40    match state {
41        Some(PinState::Mismatch) => "machine pin mismatch",
42        Some(PinState::RecordUnreadable) => "machine pin record unreadable",
43        Some(PinState::StoreUnavailable) => "machine pin store unavailable",
44        Some(_) => "machine facts unavailable",
45        None => "machine pin unavailable",
46    }
47}
48
49/// Launch the pin Cell at `services[index]` and ask it for this boot's gate.
50/// `None` index: not x86, no pin and nothing withheld.
51pub(crate) fn gate(services: &mut [Service], index: Option<usize>) -> Decision {
52    let Some(index) = index else {
53        println("Init: machine pin not applicable on this architecture.");
54        return Decision {
55            state: None,
56            safe_mode: false,
57        };
58    };
59    let unavailable = Decision {
60        state: None,
61        safe_mode: true,
62    };
63    let Some(pin) = service_table::spawn(&mut services[index]) else {
64        println("Init: machine-pin spawn/registration failed — failing closed.");
65        return unavailable;
66    };
67    println("Init: machine-pin launched.");
68    match ask_gate(pin) {
69        Some((state, safe_mode)) => Decision {
70            state: Some(state),
71            safe_mode,
72        },
73        None => {
74            println("Init: machine-pin gave no gate answer — failing closed.");
75            unavailable
76        }
77    }
78}
79
80/// `MachinePinRequest::Gate`, resubmitted while the Cell lives and the kernel
81/// expired an earlier submission before it was answered.
82fn ask_gate(pin: usize) -> Option<(PinState, bool)> {
83    let request = MachinePinRequest::Gate.encode();
84    let mut reply = [0u8; wire::ipc::IPC_BUF_SIZE];
85    for _ in 0..GATE_ATTEMPTS {
86        let pending = PendingCall::submit(pin, &request).ok()?;
87        if let Ok(Some(completion)) =
88            pending.wait_and_take(&mut reply, GATE_ROUND_TICKS, GATE_ROUNDS)
89        {
90            if completion.terminal != IpcTerminal::Reply {
91                if service_table::task_alive(pin) != Some(true) {
92                    return None;
93                }
94                continue;
95            }
96            return match MachinePinResponse::decode(&reply[..completion.len]) {
97                Some(MachinePinResponse::Gate { state, safe_mode }) => Some((state, safe_mode)),
98                _ => None,
99            };
100        }
101        // Still outstanding: dropping the handle cancels it before resubmitting.
102        if service_table::task_alive(pin) != Some(true) {
103            return None;
104        }
105    }
106    None
107}

109/// The safe set's plan indices and the enabled required entries it withholds.
110/// The shell is looked up among the autoload entries (`autoload_offset..`),
111/// the service among the configured services (`1..autoload_offset`).
112pub(crate) fn safe_set(
113    services: &[Service],
114    autoload_offset: usize,
115    autoload_end: usize,
116) -> (Option<usize>, Option<usize>, Vec<String>) {
117    let enabled = |index: &usize, name: &str| {
118        services[*index].spec.enabled && services[*index].spec.name == name
119    };
120    let service =
121        (1..autoload_offset.min(services.len())).find(|index| enabled(index, SAFE_SERVICE));
122    let shell = (autoload_offset..autoload_end.min(services.len()))
123        .find(|index| enabled(index, SAFE_SHELL));
124    let withheld = services
125        .iter()
126        .enumerate()
127        .filter(|(index, entry)| {
128            (1..autoload_end.min(services.len())).contains(index)
129                && Some(*index) != service
130                && Some(*index) != shell
131                && entry.spec.enabled
132                && entry.spec.required
133                && entry.tid.is_none()
134        })
135        .map(|(_, entry)| entry.spec.name.clone())
136        .collect();
137    (service, shell, withheld)
138}

140/// The safe-mode console init tells the pin Cell about, and re-tells after a
141/// restart of either.
142pub(crate) struct SafeConsole {
143    pub(crate) pin: usize,
144    pub(crate) shell: usize,
145}
146
147impl SafeConsole {
148    /// One-way `ConsoleShell` to the live pin Cell naming the live shell.
149    pub(crate) fn announce(&self, services: &[Service]) {
150        let (Some(pin), Some(shell)) = (services[self.pin].tid, services[self.shell].tid) else {
151            return;
152        };
153        let notice = MachinePinRequest::ConsoleShell { tid: shell as u64 }.encode();
154        if ostd::ipc::post(pin, &notice).is_err() {
155            println("Init: machine pin: console shell notice undeliverable.");
156        }
157    }
158}

160/// Start only the safe set, printing why. Boot-critical drivers and the VFS
161/// started before the gate keep running under supervision.
162pub(crate) fn enter_safe_mode(
163    services: &mut [Service],
164    autoload_offset: usize,
165    autoload_end: usize,
166    decision: &Decision,
167    pin: Option<usize>,
168) -> Option<SafeConsole> {
169    println(&format!(
170        "Init: SAFE MODE — {}; non-boot services withheld. At the console run 'machine pin --confirm' then reboot.",
171        reason(decision.state)
172    ));
173    let (service, shell, withheld) = safe_set(services, autoload_offset, autoload_end);
174    if !withheld.is_empty() {
175        println(&format!(
176            "Init: SAFE MODE — required services withheld: {}",
177            withheld.join(", ")
178        ));
179    }
180    for index in [service, shell].into_iter().flatten() {
181        match service_table::spawn(&mut services[index]) {
182            Some(_) => println(&format!("Init: {} launched.", services[index].spec.name)),
183            None => println(&format!(
184                "Init: {} spawn/registration failed.",
185                services[index].spec.name
186            )),
187        }
188    }
189    let Some(shell) = shell else {
190        println("Init: SAFE MODE — no console in plan; supervising storage only.");
191        return None;
192    };
193    let console = SafeConsole { pin: pin?, shell };
194    console.announce(services);
195    Some(console)
196}

198#[cfg(test)]
199mod tests {
200    use super::*;
201    use alloc::vec;
202
203    #[test]
204    fn safe_set_is_input_and_the_cellbox_shell_from_the_shipped_plan() {
205        let plan =
206            cellos_boot_config::parse_cells(include_bytes!("../../../../config/services.toml"))
207                .unwrap();
208        let autoload =
209            cellos_boot_config::parse_cells(include_bytes!("../../../../config/autoload.toml"))
210                .unwrap();
211        let autoload_offset = 1 + plan.cells.len();
212        let mut services = vec![Service::bootstrap_vfs()];
213        services.extend(plan.cells.into_iter().map(Service::new));
214        services.extend(autoload.cells.into_iter().map(Service::new));
215        let autoload_end = services.len();
216        // Init's fixed rows after the plan are never part of the safe set.
217        services.push(Service::machine_pin(2));
218        services.push(Service::activator());
219        let (service, shell, withheld) = safe_set(&services, autoload_offset, autoload_end);
220        let service = service.expect("input in the plan");
221        let shell = shell.expect("cellbox in autoload");
222        assert_eq!(services[service].spec.name, "input");
223        assert_eq!(
224            (
225                services[shell].spec.name.as_str(),
226                services[shell].spec.args.as_slice()
227            ),
228            ("cellbox", &[String::from("sh")][..])
229        );
230        assert!(shell >= autoload_offset);
231        for name in ["net", "silo"] {
232            assert!(
233                withheld.iter().any(|entry| entry == name),
234                "{name} withheld: {withheld:?}"
235            );
236        }
237        assert!(!withheld.iter().any(|entry| {
238            ["input", "cellbox", "vfs", "machine-pin", "activator"].contains(&entry.as_str())
239        }));
240    }
241
242    #[test]
243    fn the_banner_names_the_gate_outcome_and_fails_closed_without_one() {
244        assert_eq!(reason(Some(PinState::Mismatch)), "machine pin mismatch");
245        assert_eq!(
246            reason(Some(PinState::StoreUnavailable)),
247            "machine pin store unavailable"
248        );
249        assert_eq!(
250            reason(Some(PinState::FactsUnavailable)),
251            "machine facts unavailable"
252        );
253        assert_eq!(reason(None), "machine pin unavailable");
254    }
