1//! Init's fixed-target activation endpoint. Readiness uses exact operations,
2//! so waiting for a provider never consumes init's requests or death records.
3use api::caller_identity::{CallerIdentity, CALLER_IDENTITY_LEN};

13const PAYLOAD_BYTES: usize = 64;
14const MESSAGE_BYTES: usize = PAYLOAD_BYTES + CALLER_IDENTITY_LEN;
15const DEFERRED_MESSAGES: usize = 32;
16const READY_TICKS: u64 = 300;
17const LEGACY_TICKS: u64 = 50;
18// A finite polling backstop also bounds a broken/stalled scheduler clock.
19const POLL_LIMIT: usize = 100_000;
20
21#[derive(Clone, Copy)]
22pub(crate) struct Message {
23    pub sender: usize,
24    pub bytes: [u8; MESSAGE_BYTES],
25    pub operation: Option<ipc::IpcOpId>,
26    pub identity: Option<CallerIdentity>,
27    // A deferred request must attest/watch while its receive context is current.
28    watched: Option<Owner>,
29    watch_prepared: bool,
30}
31
32impl Message {
33    pub fn receive(mask: usize, blocking: bool) -> Option<Self> {
34        let mut bytes = [0u8; MESSAGE_BYTES];
35        let result = if blocking { syscall::sys_recv_attested(mask, &mut bytes) }
36            else { syscall::sys_try_recv_attested(mask, &mut bytes) };
37        // Capture before another receive can replace the kernel's request context.
38        let operation = ipc::current();
39        match result {
40            SyscallResult::Ok(sender) if sender != 0 && sender != usize::MAX => Some(Self {
41                sender, operation, identity: CallerIdentity::from_recv_buf(&bytes), bytes,
42                watched: None, watch_prepared: false,
43            }),
44            _ => None,
45        }
46    }
47
48    fn principal(&self) -> Option<Principal> {
49        let identity = self.identity?;
50        (identity.sender_tid == self.sender as u64 && identity.cell_id != 0 && identity.generation != 0)
51            .then_some(Principal { cell_id: identity.cell_id, generation: identity.generation })
52    }
53
54    pub fn reason(&self) -> u64 { u64::from_le_bytes(self.bytes[..8].try_into().unwrap()) }
55}

57pub(crate) struct Activator {
58    state: State,
59    deferred: [Option<Message>; DEFERRED_MESSAGES],
60    deferred_len: usize,
61    // Legacy VFS cannot cancel a dispatched operation. Do not confuse a late
62    // grant-read reply with another read on that same provider incarnation.
63    poisoned_vfs: Option<usize>,
64}
65
66impl Activator {
67    pub const fn new() -> Self {
68        Self { state: State::new(), deferred: [None; DEFERRED_MESSAGES], deferred_len: 0, poisoned_vfs: None }
69    }
70
71    pub fn next_message(&mut self) -> Option<Message> {
72        if self.deferred_len == 0 { return Message::receive(0, true); }
73        let message = self.deferred[0].take();
74        self.deferred.copy_within(1..self.deferred_len, 0);
75        self.deferred_len -= 1;
76        self.deferred[self.deferred_len] = None;
77        message
78    }
79
80    fn watch(&self, principal: Principal, sender: usize) -> Option<Owner> {
81        if sender == 0 || task_alive(sender) != Some(true) { return None; }
82        if !matches!(syscall::sys_notify_on_exit(sender), SyscallResult::Ok(0)) { return None; }
83        Some(Owner { principal, root_tid: sender, watch: sender as u64 })
84    }
85
86    fn defer(&mut self, mut message: Message) {
87        if matches!(Request::decode(&message.bytes[..FRAME_BYTES]), Some(Request::Acquire { .. })) {
88            message.watch_prepared = true;
89            message.watched = message.principal().and_then(|principal| self.watch(principal, message.sender));
90        }
91        // The exchange checks capacity BEFORE receiving, so no message is lost.
92        self.deferred[self.deferred_len] = Some(message);
93        self.deferred_len += 1;
94    }
95
96    /// Returns true for an activation frame or a known demand-engine death.
97    /// Only the kernel-written ABSENCE of a trailer plus independently observed
98    /// death can reclaim resources. An ordinary caller cannot forge an exit.
99    pub fn handle(&mut self, mut message: Message) -> bool {
100        if let Some(request) = Request::decode(&message.bytes[..FRAME_BYTES]) {
101            let engine = request.engine();
102            let Some(principal) = message.principal() else {
103                self.cancel_provisional(&message);
104                deliver(&message, Response::Error { engine, failure: Failure::Denied });
105                return true;
106            };
107            let result = match request {
108                Request::Acquire { .. } => self.acquire(&mut message, principal, engine),
109                Request::Release { lease, .. } => {
110                    match self.state.release(principal, engine, lease) {
111                        Ok(effects) => { self.apply(effects); Ok(Response::Released { engine }) }
112                        Err(failure) => Err(failure),
113                    }
114                }
115            };
116            if let Err(failure) = result { deliver(&message, Response::Error { engine, failure }); }
117            else if let Ok(response @ Response::Released { .. }) = result { deliver(&message, response); }
118            return true;
119        }
120        self.cancel_provisional(&message);
121        if message.identity.is_some() || task_alive(message.sender) != Some(false) { return false; }
122        if let Some(effects) = self.state.engine_dead(message.sender) {
123            self.apply(effects);
124            if self.state.watches_root(message.sender) {
125                let effects = self.state.owner_dead(message.sender);
126                self.apply(effects);
127            }
128            ostd::io::println("Init: demand engine exited — leases revoked, no restart.");
129            return true;
130        }
131        if self.state.watches_root(message.sender) {
132            let effects = self.state.owner_dead(message.sender);
133            self.apply(effects);
134        }
135        false
136    }
137
138    fn cancel_provisional(&self, _message: &Message) {}
139
140    fn acquire(&mut self, message: &mut Message, principal: Principal, engine: Engine) -> Result<Response, Failure> {
141        // Revocation happens before preflight, including an independently
142        // replaced service registry row. Never return an old provider's lease.
143        if let Some(provider) = self.state.provider(engine) {
144            if task_alive(provider.tid) == Some(false) {
145                if let Some(effects) = self.state.engine_dead(provider.tid) { self.apply(effects); }
146            } else if provider.phase == Phase::Ready && syscall::sys_lookup_service(service_id(engine)) != Some(provider.tid) {
147                let effects = self.state.retire(engine);
148                self.apply(effects);
149            }
150        }
151        if let Err(failure) = self.state.preflight(principal, engine) {
152            self.cancel_provisional(message);
153            return Err(failure);
154        }
155        let installed = self.state.owner(principal);
156        let owner = match installed {
157            Some(owner) => {
158                self.cancel_provisional(message);
159                owner
160            }
161            None => {
162                let watched = if message.watch_prepared { message.watched } else { self.watch(principal, message.sender) };
163                let Some(owner) = watched else { return Err(Failure::Denied); };
164                message.watched = Some(owner);
165                owner
166            }
167        };
168        let result = self.acquire_watched(message, owner, engine);
169        // Failed first demand never retains an otherwise-unused owner watch.
170        if result.is_err() { self.cancel_provisional(message); }
171        result
172    }
173
174    fn acquire_watched(&mut self, message: &Message, owner: Owner, engine: Engine) -> Result<Response, Failure> {
175        if task_alive(owner.root_tid) != Some(true) { return Err(Failure::Denied); }
176        if self.state.provider(engine).is_none() { self.start_engine(engine)?; }
177        let provider = self.state.provider(engine).ok_or(Failure::Failed)?;
178        if provider.phase != Phase::Ready || task_alive(provider.tid) != Some(true)
179            || syscall::sys_lookup_service(service_id(engine)) != Some(provider.tid)
180        {
181            let effects = self.state.retire(engine);
182            self.apply(effects);
183            return Err(Failure::Failed);
184        }
185        let root_alive = task_alive(owner.root_tid);
186        if root_alive != Some(true) {
187            // A dead root revokes its demand; an uncertain liveness query must
188            // not revoke existing demand. Either case retires an unused start.
189            let effects = if root_alive == Some(false) { self.state.owner_dead(owner.root_tid) }
190                else { self.state.unused() };
191            self.apply(effects);
192            return Err(Failure::Denied);
193        }
194        let (lease, newly_allocated) = match self.state.acquire(owner, engine) {
195            Ok(lease) => lease,
196            Err(failure) => {
197                let effects = self.state.unused();
198                self.apply(effects);
199                return Err(failure);
200            }
201        };
202        let response = Response::Ready { engine, lease: lease.token, tid: provider.tid as u64 };
203        if !deliver(message, response) {
204            if newly_allocated {
205                if let Ok(effects) = self.state.release(owner.principal, engine, lease.token) { self.apply(effects); }
206            }
207            // Reply cancellation is not a reason to revoke previously delivered
208            // demand or another owner's share of this engine.
209            return Err(Failure::Failed);
210        }
211        Ok(response)
212    }

214    fn start_engine(&mut self, engine: Engine) -> Result<(), Failure> {
215        let tid = match syscall::sys_spawn_from_path_raw(engine.path()) {
216            SyscallResult::Ok(tid) if tid != 0 => tid,
217            _ => self.spawn_via_vfs(engine)?,
218        };
219        let provider = match self.state.start(engine, tid) {
220            Ok(provider) => provider,
221            Err(failure) => { let _ = syscall::sys_force_exit(tid); return Err(failure); }
222        };
223        let ready = matches!(syscall::sys_notify_on_exit(tid), SyscallResult::Ok(0))
224            && readiness(provider)
225            && task_alive(tid) == Some(true)
226            && matches!(syscall::sys_register_service(service_id(engine), tid), SyscallResult::Ok(0))
227            && syscall::sys_lookup_service(service_id(engine)) == Some(tid)
228            && task_alive(tid) == Some(true);
229        if !ready || !self.state.ready(provider) {
230            let effects = self.state.retire(engine);
231            self.apply(effects);
232            return Err(Failure::Failed);
233        }
234        ostd::io::println(match engine {
235            Engine::JavaScript => "Init: demand QuickJS ready (Tier 2).",
236            Engine::Pdf => "Init: demand PDF ready (Tier 2).",
237        });
238        Ok(())
239    }
240
241    fn apply(&mut self, effects: Effects) {
242        let _ = &effects.cancel;
243        for provider in effects.retire.into_iter().flatten() {
244            if matches!(syscall::sys_force_exit(provider.tid), SyscallResult::Ok(0))
245                || task_alive(provider.tid) == Some(false)
246            {
247                self.state.terminated(provider);
248                ostd::io::println("Init: unused demand engine stopped.");
249            }
250            // Failed termination retains Retiring until authenticated death.
251        }
252    }

254    /// Same Stat -> shared Grant -> ReadFileGrant -> SpawnFromElf route as
255    /// ostd::sys_spawn_from_path, with bounded waits and retained init messages.
256    fn spawn_via_vfs(&mut self, engine: Engine) -> Result<usize, Failure> {
257        let vfs = syscall::sys_lookup_service(service::VFS).ok_or(Failure::Unavailable)?;
258        if self.poisoned_vfs == Some(vfs) { return Err(Failure::Unavailable); }
259        let stat = self.vfs_exchange(vfs, &VfsRequest::Stat(engine.path()))?;
260        let size = match api::ipc::decode::<VfsResponse<'_>>(&stat[..PAYLOAD_BYTES]) {
261            Ok(VfsResponse::Stat { size, is_dir: false }) if size > 0 => usize::try_from(size).map_err(|_| Failure::Failed)?,
262            _ => return Err(Failure::Unavailable),
263        };
264        let grant = syscall::sys_grant_alloc(size).ok_or(Failure::Busy)?;
265        if !syscall::sys_grant_share(grant, vfs, 2) {
266            syscall::sys_grant_free(grant);
267            return Err(Failure::Denied);
268        }
269        let read = self.vfs_exchange(vfs, &VfsRequest::ReadFileGrant { path: engine.path(), grant, max: size });
270        let attempt = match read {
271            Ok(bytes) => match api::ipc::decode::<VfsResponse<'_>>(&bytes[..PAYLOAD_BYTES]) {
272                Ok(VfsResponse::GrantDone { bytes }) if bytes > 0 && bytes <= size => {
273                    match syscall::sys_spawn_from_elf(grant, bytes, engine.path()) {
274                        SyscallResult::Ok(tid) if tid != 0 => Ok(tid),
275                        _ => Err(Failure::Failed),
276                    }
277                }
278                _ => Err(Failure::Failed),
279            },
280            Err(failure) => Err(failure),
281        };
282        syscall::sys_grant_free(grant);
283        attempt
284    }
285
286    fn vfs_exchange(&mut self, peer: usize, request: &VfsRequest<'_>) -> Result<[u8; MESSAGE_BYTES], Failure> {
287        let mut send = [0u8; 512];
288        let bytes = api::ipc::encode(request, &mut send).map_err(|_| Failure::Failed)?;
289        let started = now_ticks();
290        let mut admitted = false;
291        for _ in 0..POLL_LIMIT {
292            if now_ticks().wrapping_sub(started) >= LEGACY_TICKS { break; }
293            if !admitted {
294                admitted = matches!(syscall::sys_try_send(peer, bytes), SyscallResult::Ok(0));
295            } else {
296                // A masked receive can still yield ANY pending death record.
297                // If full, leave the next message in the kernel and fail safely.
298                if self.deferred_len == DEFERRED_MESSAGES { break; }
299                if let Some(message) = Message::receive(peer, false) {
300                    if message.sender == peer && message.principal().is_some() { return Ok(message.bytes); }
301                    self.defer(message);
302                }
303            }
304            ostd::task::yield_now();
305        }
306        if admitted { self.poisoned_vfs = Some(peer); }
307        Err(Failure::Unavailable)
308    }
309}

311pub(crate) fn service_id(engine: Engine) -> u16 {
312    match engine { Engine::JavaScript => service::OCEL_JS, Engine::Pdf => service::OCEL_PDF }
313}
314
315fn deliver(message: &Message, response: Response) -> bool {
316    let frame = response.encode();
317    if let Some(operation) = message.operation { return ipc::reply(operation, &frame).is_ok(); }
318    let started = now_ticks();
319    for _ in 0..POLL_LIMIT {
320        if matches!(syscall::sys_try_send(message.sender, &frame), SyscallResult::Ok(0)) { return true; }
321        if now_ticks().wrapping_sub(started) >= LEGACY_TICKS { break; }
322        ostd::task::yield_now();
323    }
324    false
325}

327fn readiness(provider: Provider) -> bool {
328    let mut send = [0u8; 16];
329    let encoded = match provider.engine {
330        Engine::JavaScript => api::ipc::encode(&dom_arena::OcelJsRequest::Ping, &mut send),
331        Engine::Pdf => api::ipc::encode(&ocel_pdf_proto::Request::Ping, &mut send),
332    };
333    let Ok(encoded) = encoded else { return false; };
334    let Ok(operation) = ipc::submit(provider.tid, encoded) else { return false; };
335    let mut reply = [0u8; api::ipc::IPC_BUF_SIZE];
336    let started = now_ticks();
337    for _ in 0..POLL_LIMIT {
338        match ipc::take(operation, &mut reply) {
339            Ok(IpcTakeResult::Terminal { status: IpcTerminal::Reply, len }) => {
340                return match provider.engine {
341                    Engine::JavaScript => matches!(api::ipc::decode::<dom_arena::OcelJsReply>(&reply[..len]), Ok(dom_arena::OcelJsReply::Ready)),
342                    Engine::Pdf => matches!(api::ipc::decode::<ocel_pdf_proto::Response>(&reply[..len]), Ok(ocel_pdf_proto::Response::Ready)),
343                };
344            }
345            Ok(IpcTakeResult::Pending) => {
346                if now_ticks().wrapping_sub(started) >= READY_TICKS { break; }
347                // A finite one-tick park never touches legacy receive messages.
348                ipc::wait(1);
349            }
350            Ok(IpcTakeResult::Terminal { .. }) => return false,
351            Err(_) => break,
352        }
353    }
354    let _ = ipc::cancel(operation);
355    let _ = ipc::take(operation, &mut reply);
356    false
357}

359/// Absence from a truncated process table is NOT proof of death. Most images
360/// fit the stack buffer; larger tables grow only for a liveness query, with a
361/// hard bound and fail-closed uncertainty instead of forged cleanup authority.
362pub(crate) fn task_alive(tid: usize) -> Option<bool> {
363    let mut stack = [ProcessInfo::default(); 64];
364    let count = syscall::sys_get_procs(&mut stack).ok()?;
365    if let Some(row) = stack.iter().take(count).find(|row| row.id == tid) { return Some(row.state != 3); }
366    if count < stack.len() { return Some(false); }
367    let mut rows = alloc::vec![ProcessInfo::default(); 128];
368    loop {
369        let count = syscall::sys_get_procs(&mut rows).ok()?;
370        if let Some(row) = rows.iter().take(count).find(|row| row.id == tid) { return Some(row.state != 3); }
371        if count < rows.len() { return Some(false); }
372        if rows.len() >= 8192 { return None; }
373        rows.resize(rows.len() * 2, ProcessInfo::default());
374    }
