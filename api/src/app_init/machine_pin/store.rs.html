1//! The pin store: `/srv/.cellos-machine/pin` on CellosFS `/srv` (P5).
2//!
3//! The VFS admits only the live `MACHINE_PIN` provider (init) to this
4//! namespace. No durability is claimed: a torn write reads back as an
5//! unreadable record, which is safe mode, never an automatic re-pin.
6//!
7//! The VFS answers the same IO error for an absent path and for a failed
8//! backend, so a failed `Stat` never proves absence. Absence is concluded only
9//! from a successful, complete listing of the parent directory that lacks the
10//! entry; every other failure is `Unavailable`, which never writes.
11
12use alloc::vec::Vec;
13use api::machine_facts::MACHINE_FACTS_CAPACITY;
14use ostd::clients::VfsClient;
15use ostd::{ViError, ViResult};
16
17pub(crate) const PIN_DIR: &str = "/srv/.cellos-machine";
18pub(crate) const PIN_PATH: &str = "/srv/.cellos-machine/pin";
19const SRV: &str = "/srv";
20
21/// A `ListDir` reply at least this long may have been cut at the reply
22/// buffer, so it cannot prove that an entry is missing.
23const LISTING_COMPLETE_BELOW: usize = 3_800;
24
25/// What reading the pin found.
26#[derive(Debug, PartialEq, Eq)]
27pub(crate) enum PinRead {
28    /// The file's bytes, at most `MACHINE_FACTS_CAPACITY`.
29    Present(Vec<u8>),
30    /// Definitive absence on a healthy store: a complete listing of the
31    /// parent directory without the entry.
32    Absent,
33    /// A file larger than any record.
34    Oversized,
35    /// The store could not answer.
36    Unavailable(&'static str),
37}
38
39/// Storage the gate needs. The VFS implements it; host tests use a fake.
40pub(crate) trait PinStore {
41    fn read(&mut self) -> PinRead;
42    /// Write `bytes`, then read them back; `true` only on an exact match.
43    fn write(&mut self, bytes: &[u8]) -> bool;
44}
45
46/// The VFS operations the store uses, so the absence rule is host-testable.
47pub(crate) trait VfsOps {
48    fn stat(&mut self, path: &str) -> ViResult<(u64, bool)>;
49    fn list_dir(&mut self, path: &str) -> ViResult<Vec<u8>>;
50    fn read_file(&mut self, path: &str, max: usize) -> ViResult<Vec<u8>>;
51    fn mkdir(&mut self, path: &str) -> ViResult<()>;
52    fn write_file(&mut self, path: &str, bytes: &[u8]) -> ViResult<()>;
53}
54
55impl VfsOps for VfsClient {
56    fn stat(&mut self, path: &str) -> ViResult<(u64, bool)> {
57        VfsClient::stat(self, path)
58    }
59    fn list_dir(&mut self, path: &str) -> ViResult<Vec<u8>> {
60        VfsClient::list_dir(self, path)
61    }
62    fn read_file(&mut self, path: &str, max: usize) -> ViResult<Vec<u8>> {
63        self.read_file_bounded(path, max)
64    }
65    fn mkdir(&mut self, path: &str) -> ViResult<()> {
66        VfsClient::mkdir(self, path)
67    }
68    fn write_file(&mut self, path: &str, bytes: &[u8]) -> ViResult<()> {
69        VfsClient::write_file(self, path, bytes)
70    }
71}

73/// Whether a complete CellosFS `ListDir` reply (`f:`/`d:` kind prefix, then
74/// the name, one per line) names `name` with either kind; `None` when the
75/// reply may be truncated.
76fn listing_names(listing: &[u8], name: &[u8]) -> Option<bool> {
77    if listing.len() >= LISTING_COMPLETE_BELOW {
78        return None;
79    }
80    Some(listing.split(|byte| *byte == b'\n').any(|entry| {
81        entry.len() == name.len() + 2 && matches!(&entry[..2], b"f:" | b"d:") && &entry[2..] == name
82    }))
83}

85/// The pin's state as seen through `ops`. Writes nothing.
86pub(crate) fn read_pin(ops: &mut impl VfsOps) -> PinRead {
87    // `/srv` answers as a directory only when its CellosFS volume opened.
88    if !matches!(ops.stat(SRV), Ok((_, true))) {
89        return PinRead::Unavailable("/srv not mounted");
90    }
91    match ops.stat(PIN_DIR) {
92        Ok((_, true)) => {}
93        Ok((_, false)) => return PinRead::Unavailable("pin directory is a file"),
94        Err(_) => {
95            return match ops
96                .list_dir(SRV)
97                .ok()
98                .and_then(|names| listing_names(&names, b".cellos-machine"))
99            {
100                Some(false) => PinRead::Absent,
101                Some(true) => PinRead::Unavailable("pin directory listed but not statable"),
102                None => PinRead::Unavailable("/srv not listable"),
103            };
104        }
105    }
106    match ops
107        .list_dir(PIN_DIR)
108        .ok()
109        .and_then(|names| listing_names(&names, b"pin"))
110    {
111        Some(false) => PinRead::Absent,
112        Some(true) => match ops.read_file(PIN_PATH, MACHINE_FACTS_CAPACITY) {
113            Ok(bytes) => PinRead::Present(bytes),
114            Err(ViError::OutOfMemory) => PinRead::Oversized,
115            Err(_) => PinRead::Unavailable("pin listed but unreadable"),
116        },
117        None => PinRead::Unavailable("pin directory unlistable"),
118    }
119}

121/// Create the directory if needed, write `bytes`, read them back.
122pub(crate) fn write_pin(ops: &mut impl VfsOps, bytes: &[u8]) -> bool {
123    if !matches!(ops.stat(PIN_DIR), Ok((_, true))) && ops.mkdir(PIN_DIR).is_err() {
124        return false;
125    }
126    ops.write_file(PIN_PATH, bytes).is_ok()
127        && matches!(ops.read_file(PIN_PATH, MACHINE_FACTS_CAPACITY), Ok(back) if back == bytes)
128}

130/// [`PinStore`] over the VFS, as the live `MACHINE_PIN` provider.
131pub(crate) struct VfsPinStore {
132    vfs: VfsClient,
133}
134
135impl VfsPinStore {
136    pub(crate) fn new() -> Self {
137        Self {
138            vfs: VfsClient::new(),
139        }
140    }
141}
142
143impl PinStore for VfsPinStore {
144    fn read(&mut self) -> PinRead {
145        read_pin(&mut self.vfs)
146    }
147
148    fn write(&mut self, bytes: &[u8]) -> bool {
149        write_pin(&mut self.vfs, bytes)
150    }
