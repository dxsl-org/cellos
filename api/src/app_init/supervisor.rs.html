9const MAX_RESTARTS_PER_WINDOW: u32 = 5;
10const RESTART_WINDOW_TICKS: u64 = 1000;
11
12pub(crate) fn run(services: &mut [Service], mut hypervisor_tid: Option<usize>) -> ! {
13    let mut buffer = [0u8; 16];
14    let mut hypervisor_restarts = 0u32;
15    let mut hypervisor_last_restart = 0u64;
16    loop {
17        let dead = match sys_recv(0, &mut buffer) {
18            SyscallResult::Ok(tid) => tid,
19            _ => {
20                ostd::task::yield_now();
21                continue;
22            }
23        };
24        #[cfg(not(feature = "hypervisor-min"))]
25        let reason = u64::from_le_bytes(buffer[..8].try_into().unwrap());
26
27        if hypervisor_tid == Some(dead) {
28            relay_hypervisor_exit(dead);
29            let now = sys_get_time();
30            if now.saturating_sub(hypervisor_last_restart) > RESTART_WINDOW_TICKS {
31                hypervisor_restarts = 0;
32            }
33            hypervisor_last_restart = now;
34            if hypervisor_restarts >= MAX_RESTARTS_PER_WINDOW {
35                println("Init: hypervisor restart storm — giving up.");
36                hypervisor_tid = None;
37                continue;
38            }
39            hypervisor_restarts += 1;
40            println("Init: hypervisor died — respawning...");
41            hypervisor_tid = crate::boot::spawn_hypervisor();
42            if hypervisor_tid.is_some() {
43                println("Init: hypervisor restarted.");
44            } else {
45                println("Init: hypervisor restart FAILED.");
46            }
47            continue;
48        }
49
50        #[cfg(feature = "development-silo-provider")]
51        if services
52            .iter()
53            .any(|candidate| candidate.tid == Some(dead) && candidate.path == "/bin/kms")
54        {
55            let silo_ready = services
56                .iter()
57                .find(|candidate| candidate.path == "/bin/silo")
58                .and_then(|silo| silo.tid)
59                .is_some_and(|tid| service_table::wait_for_exact_registration(service::SILO, tid));
60            if !silo_ready {
61                if let Some(kms) = services
62                    .iter_mut()
63                    .find(|candidate| candidate.tid == Some(dead))
64                {
65                    kms.tid = None;
66                }
67                println("Init: KMS restart blocked — exact Silo instance not ready.");
68                continue;
69            }
70        }
71
72        let Some(service) = services
73            .iter_mut()
74            .find(|service| service.tid == Some(dead))
75        else {
76            continue;
77        };
78        let should_restart = match service.policy {
79            RestartPolicy::Temporary => false,
80            #[cfg(not(feature = "hypervisor-min"))]
81            RestartPolicy::Transient => reason != 0,
82            RestartPolicy::Permanent => true,
83        };
84        if !should_restart {
85            println("Init: service exited cleanly — policy says no restart.");
86            service.tid = None;
87            continue;
88        }
89
90        let now = sys_get_time();
91        if now.wrapping_sub(service.window_start) > RESTART_WINDOW_TICKS {
92            service.window_start = now;
93            service.restart_count = 0;
94        }
95        if service.restart_count >= MAX_RESTARTS_PER_WINDOW {
96            println("Init: restart storm — giving up on this service (escalate).");
97            service.tid = None;
98            continue;
99        }
100        service.restart_count += 1;
101
102        println("Init: service died — restarting...");
103        if let Some(tid) = service_table::spawn(service) {
104            let _ = sys_notify_on_exit(tid);
105            println("Init: service restarted.");
106        } else {
107            println("Init: service restart FAILED.");
108        }
109    }
110}

112fn relay_hypervisor_exit(dead: usize) {
113    let mut message = [0u8; 1 + core::mem::size_of::<usize>()];
114    message[0] = 0xE1;
115    message[1..].copy_from_slice(&dead.to_le_bytes());
116    if let Some(supervisor) = sys_lookup_service(service::SUPERVISOR) {
117        let _ = sys_send(supervisor, &message);
118    }
119    println("Init: hypervisor exited — scanout cleanup relayed.");
120}
