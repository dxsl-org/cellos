9const MAX_RESTARTS_PER_WINDOW: u32 = 5;
10const RESTART_WINDOW_TICKS: u64 = 1000;
11
12pub(crate) fn run(services: &mut [Service], mut hypervisor_tid: Option<usize>) -> ! {
13    let mut buffer = [0u8; 16];
14    loop {
15        let dead = match sys_recv(0, &mut buffer) {
16            SyscallResult::Ok(tid) => tid,
17            _ => {
18                ostd::task::yield_now();
19                continue;
20            }
21        };
22        #[cfg(not(feature = "hypervisor-min"))]
23        let reason = u64::from_le_bytes(buffer[..8].try_into().unwrap());
24
25        if hypervisor_tid == Some(dead) {
26            relay_hypervisor_exit(dead);
27            hypervisor_tid = None;
28            continue;
29        }
30
31        #[cfg(feature = "development-silo-provider")]
32        if services
33            .iter()
34            .any(|candidate| candidate.tid == Some(dead) && candidate.path == "/bin/kms")
35        {
36            let silo_ready = services
37                .iter()
38                .find(|candidate| candidate.path == "/bin/silo")
39                .and_then(|silo| silo.tid)
40                .is_some_and(|tid| service_table::wait_for_exact_registration(service::SILO, tid));
41            if !silo_ready {
42                if let Some(kms) = services
43                    .iter_mut()
44                    .find(|candidate| candidate.tid == Some(dead))
45                {
46                    kms.tid = None;
47                }
48                println("Init: KMS restart blocked — exact Silo instance not ready.");
49                continue;
50            }
51        }
52
53        let Some(service) = services
54            .iter_mut()
55            .find(|service| service.tid == Some(dead))
56        else {
57            continue;
58        };
59        let should_restart = match service.policy {
60            RestartPolicy::Temporary => false,
61            #[cfg(not(feature = "hypervisor-min"))]
62            RestartPolicy::Transient => reason != 0,
63            RestartPolicy::Permanent => true,
64        };
65        if !should_restart {
66            println("Init: service exited cleanly — policy says no restart.");
67            service.tid = None;
68            continue;
69        }
70
71        let now = sys_get_time();
72        if now.wrapping_sub(service.window_start) > RESTART_WINDOW_TICKS {
73            service.window_start = now;
74            service.restart_count = 0;
75        }
76        if service.restart_count >= MAX_RESTARTS_PER_WINDOW {
77            println("Init: restart storm — giving up on this service (escalate).");
78            service.tid = None;
79            continue;
80        }
81        service.restart_count += 1;
82
83        println("Init: service died — restarting...");
84        if let Some(tid) = service_table::spawn(service) {
85            let _ = sys_notify_on_exit(tid);
86            println("Init: service restarted.");
87        } else {
88            println("Init: service restart FAILED.");
89        }
90    }
91}

93fn relay_hypervisor_exit(dead: usize) {
94    let mut message = [0u8; 1 + core::mem::size_of::<usize>()];
95    message[0] = 0xE1;
96    message[1..].copy_from_slice(&dead.to_le_bytes());
97    if let Some(supervisor) = sys_lookup_service(service::SUPERVISOR) {
98        let _ = sys_send(supervisor, &message);
99    }
100    println("Init: hypervisor exited — scanout cleanup relayed.");
101}
