10/// Registry publication is not VFS application readiness. Only an actual
11/// successful, operation-correlated RPC proves the bootstrap server can serve.
12pub(crate) fn wait_for_vfs(tid: usize, timeout: u64) -> bool {
13    let started = crate::service_table::now_ticks();
14    let mut send = [0u8; 512];
15    let mut recv = [0u8; 512];
16    for _ in 0..10_000 {
17        if ostd::syscall::sys_lookup_service(api::syscall::service::VFS) != Some(tid)
18            || crate::activation::task_alive(tid) == Some(false)
19        {
20            return false;
21        }
22        if matches!(ostd::ipc::service_call_typed_bounded::<_, VfsResponse<'_>>(
23            tid, &VfsRequest::Stat("/"), &mut send, &mut recv, 50,
24        ), Ok(VfsResponse::Stat { is_dir: true, .. })) {
25            return true;
26        }
27        if crate::service_table::now_ticks().wrapping_sub(started) >= timeout {
28            return false;
29        }
30        ostd::task::yield_now();
31    }
32    false
33}

35/// Read and validate both files before any configured cell can launch.
36pub(crate) fn load() -> Result<(CellConfig, CellConfig), alloc::string::String> {
37    let mut vfs = VfsClient::new();
38    let services_bytes = read_required(&mut vfs, SERVICES_PATH)?;
39    let services = parse_cells(&services_bytes)
40        .map_err(|error| format!("{SERVICES_PATH}: {error}"))?;
41
42    // Stat must precede the read: a file known to be present but unreadable,
43    // oversized, empty or malformed must not become the optional empty plan.
44    // The existing VFS protocol uses IO for absent paths as well as backend stat
45    // failures; it cannot distinguish those two cases at this boundary.
46    let mut send = [0u8; 512];
47    let mut recv = [0u8; 512];
48    let tid = ostd::syscall::sys_lookup_service(api::syscall::service::VFS)
49        .ok_or_else(|| format!("{AUTOLOAD_PATH}: VFS unavailable"))?;
50    let response = ostd::ipc::service_call_typed_bounded::<_, VfsResponse<'_>>(
51        tid, &VfsRequest::Stat(AUTOLOAD_PATH), &mut send, &mut recv, 50,
52    ).map_err(|error| format!("{AUTOLOAD_PATH}: stat RPC failed: {error:?}"))?;
53    let autoload = match response {
54        VfsResponse::Stat { size, is_dir: false } if size <= MAX_CONFIG_BYTES as u64 => {
55            let bytes = read_required(&mut vfs, AUTOLOAD_PATH)?;
56            parse_cells(&bytes).map_err(|error| format!("{AUTOLOAD_PATH}: {error}"))?
57        }
58        VfsResponse::Err(1) => {
59            ostd::io::println("Init: autoload.toml absent — optional autoload is explicitly empty.");
60            CellConfig::default()
61        }
62        other => return Err(format!("{AUTOLOAD_PATH}: invalid stat response: {other:?}")),
63    };
64    validate_plan(&services, &autoload).map_err(|error| format!("boot plan: {error}"))?;
65    Ok((services, autoload))
66}

68fn read_required(vfs: &mut VfsClient, path: &str) -> Result<Vec<u8>, alloc::string::String> {
69    let expected_size = match vfs.stat(path) {
70        Ok((size, false)) if size <= MAX_CONFIG_BYTES as u64 => size as usize,
71        Ok(_) => return Err(format!("{path}: not a bounded configuration file")),
72        Err(error) => return Err(format!("{path}: required file unavailable: {error:?}")),
73    };
74    let bytes = vfs.read_file_bounded(path, MAX_CONFIG_BYTES)
75        .map_err(|error| format!("{path}: bounded VFS read failed: {error:?}"))?;
76    if bytes.len() != expected_size {
77        return Err(format!("{path}: truncated or inconsistent configuration (expected {expected_size} bytes, read {})", bytes.len()));
78    }
79    Ok(bytes)
