71fn cell_main() {
72    println("Init: Starting Cellos Orchestrator...");
73    boot::start_block_drivers();
74    let mut services = Vec::with_capacity(cellos_boot_config::MAX_CELLS + 1);
75    services.push(service_table::Service::bootstrap_vfs());
76    match service_table::spawn(&mut services[0]) {
77        Some(_) => {}
78        None => {
79            println("Init: required VFS bootstrap failed — boot stopped.");
80            return;
81        }
82    };
83    let (service_config, autoload_config) = match configuration::load() {
84        Ok(config) => config,
85        Err(error) => {
86            println(&alloc::format!("Init: configuration failure — {error}"));
87            supervisor::run(&mut services);
88        }
89    };
90    let service_order = match ordered_indices(&service_config, &CellConfig::default()) {
91        Ok(order) => order,
92        Err(error) => {
93            println(&alloc::format!("Init: service ordering failure — {error}"));
94            supervisor::run(&mut services);
95        }
96    };
97    let autoload_order = match ordered_indices(&autoload_config, &service_config) {
98        Ok(order) => order,
99        Err(error) => {
100            println(&alloc::format!("Init: autoload ordering failure — {error}"));
101            supervisor::run(&mut services);
102        }
103    };
104    let autoload_offset = 1 + service_config.cells.len();
105    services.extend(service_config.cells.into_iter().map(service_table::Service::new));
106    services.extend(autoload_config.cells.into_iter().map(service_table::Service::new));
107    if !launch_order(&mut services, &service_order, 1) {
108        println("Init: required service failed — autoload and OS ready blocked.");
109        supervisor::run(&mut services);
110    }
111
112    let init_tid = {
113        let mut procs = [api::syscall::ProcessInfo::default(); 16];
114        let mut tid = 2;
115        if let Ok(count) = ostd::syscall::sys_get_procs(&mut procs) {
116            for proc in procs.iter().take(count) {
117                if &proc.name[..4] == b"init" && (proc.name[4] == 0 || proc.name[4] == b' ') {
118                    tid = proc.id;
119                    break;
120                }
121            }
122        }
123        tid
124    };
125    if !matches!(
126        ostd::syscall::sys_register_service(api::syscall::service::OCEL_ACTIVATOR, init_tid),
127        ostd::syscall::SyscallResult::Ok(0)
128    ) {
129        println("Init: Ocel activator registration failed — autoload blocked.");
130        supervisor::run(&mut services);
131    }
132
133    // Test fixtures retain their reviewed boot lanes. Production optional cells,
134    // including shell and hypervisor autostart, come exclusively from the plan.
135    boot::spawn_test_fixtures();
136    if !required_services_available(&services[..autoload_offset]) {
137        println("Init: required service lost before autoload — OS ready blocked.");
138        supervisor::run(&mut services);
139    }
140    println("Init: services spawned.");
141    println("Init: service registry verified (init registration is publication, not application readiness).");
142    println("Init: OS ready.");
143    if !launch_order(&mut services, &autoload_order, autoload_offset)
144        || !required_services_available(&services)
145    {
146        println("Init: required cell unavailable — remaining autoload blocked.");
147        supervisor::run(&mut services);
148    }
149    println("Init: supervising services (auto-restart on crash)...");
150    supervisor::run(&mut services)
151}

153fn required_services_available(services: &[service_table::Service]) -> bool {
154    services.iter().all(|service| {
155        !service.spec.enabled || !service.spec.required || service.available()
156    })
157}
158
159fn launch_order(services: &mut [service_table::Service], order: &[usize], offset: usize) -> bool {
160    for &entry in order {
161        if services.iter().any(|service| {
162            service.spec.enabled && service.spec.required && service.tid.is_some() && !service.available()
163        }) {
164            println("Init: required service lost during launch — remaining cells blocked.");
165            return false;
166        }
167        let index = offset + entry;
168        if !service_table::dependencies_available(services, index) {
169            println(&alloc::format!("Init: {} blocked — dependency unavailable.", services[index].spec.name));
170            if services[index].spec.required { return false; }
171            continue;
172        }
173        if service_table::spawn(&mut services[index]).is_none() {
174            println(&alloc::format!("Init: {} spawn/registration failed.", services[index].spec.name));
175            if services[index].spec.required { return false; }
176        } else {
177            println(&alloc::format!("Init: {} launched.", services[index].spec.name));
178        }
179        ostd::task::yield_now();
180    }
181    true
182}
