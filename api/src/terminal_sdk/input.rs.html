2//! Incremental epoch-bound event/control framing, independent of any GUI toolkit.
3use alloc::string::String;
4use api::cell_session::{Reader,Status};
5#[derive(Clone,PartialEq,Eq)]pub struct Event{pub sequence:u64,pub epoch:u64,pub kind:EventKind}
6#[derive(Clone,PartialEq,Eq)]pub enum EventKind{Key{code:u32,modifiers:u32,pressed:u8},Text(String),PasteBegin,PasteEnd,Resize{rows:u16,cols:u16},Interrupt,Eof,Lost(u32),
7    /// Owner notice to the session controller: draw the fixed first-trust prompt for this
8    /// input-validated descriptor (graphic ASCII only) and stop relaying the child's output.
9    TrustPrompt{host:String,port:u16,algorithm:String,fingerprint:String},
10    /// Owner notice to the controller: the trust capture ended; resume relaying.
11    TrustPromptEnd{accepted:bool}}
12// Keystrokes and typed text can be secrets: Debug names the variant (and the
13// non-secret structural fields) but NEVER the Key code/modifiers or Text bytes.
14impl core::fmt::Debug for EventKind{
15    fn fmt(&self,f:&mut core::fmt::Formatter<'_>)->core::fmt::Result{
16        match self{
17            Self::Key{..}=>f.write_str("Key(<redacted>)"),
18            Self::Text(text)=>write!(f,"Text(<redacted {} bytes>)",text.len()),
19            Self::PasteBegin=>f.write_str("PasteBegin"),Self::PasteEnd=>f.write_str("PasteEnd"),
20            Self::Resize{rows,cols}=>f.debug_struct("Resize").field("rows",rows).field("cols",cols).finish(),
21            Self::Interrupt=>f.write_str("Interrupt"),Self::Eof=>f.write_str("Eof"),
22            Self::Lost(reason)=>f.debug_tuple("Lost").field(reason).finish(),
23            Self::TrustPrompt{host,port,algorithm,fingerprint}=>f.debug_struct("TrustPrompt").field("host",host).field("port",port).field("algorithm",algorithm).field("fingerprint",fingerprint).finish(),
24            Self::TrustPromptEnd{accepted}=>f.debug_struct("TrustPromptEnd").field("accepted",accepted).finish(),
25        }
26    }
27}
28impl core::fmt::Debug for Event{
29    fn fmt(&self,f:&mut core::fmt::Formatter<'_>)->core::fmt::Result{
30        f.debug_struct("Event").field("sequence",&self.sequence).field("epoch",&self.epoch).field("kind",&self.kind).finish()
31    }
32}
33impl Event{
34    pub fn decode(frame:&[u8])->Result<Self,Status>{
35        if frame.len()<24{return Err(Status::Malformed)}let mut r=Reader::new(frame);let sequence=r.u64()?;let epoch=r.u64()?;let kind=r.u16()?;let len=r.u16()? as usize;if r.u32()?!=0||sequence==0||epoch==0||len>512||frame.len()!=24+len{return Err(Status::Malformed)}let payload=r.take(len)?;r.finish()?;
36        api::services::session::validate_event(kind,payload)?;
37        let mut r=Reader::new(payload);let kind=match kind{
38            1=>{let code=r.u32()?;let modifiers=r.u32()?;let pressed=r.u8()?;if r.take(7)?.iter().any(|b|*b!=0){return Err(Status::Malformed)}EventKind::Key{code,modifiers,pressed}},
39            2=>EventKind::Text(String::from(core::str::from_utf8(payload).map_err(|_|Status::UnsupportedInput)?)),
40            3=>EventKind::PasteBegin,4=>EventKind::PasteEnd,5=>{let rows=r.u16()?;let cols=r.u16()?;EventKind::Resize{rows,cols}},6=>EventKind::Interrupt,7=>EventKind::Eof,8=>EventKind::Lost(r.u32()?),
41            9=>{let p=api::cell_session::trust::TrustPromptV1::decode(payload)?;EventKind::TrustPrompt{host:String::from(p.host),port:p.port,algorithm:String::from(p.algorithm),fingerprint:String::from(p.fingerprint)}},
42            10=>EventKind::TrustPromptEnd{accepted:api::cell_session::trust::decode_prompt_end(payload)?},
43            _=>return Err(Status::UnsupportedInput),
44        };Ok(Self{sequence,epoch,kind})
45    }
46    pub fn is_control(&self)->bool{matches!(self.kind,EventKind::Resize{..}|EventKind::Interrupt|EventKind::Eof|EventKind::Lost(_)|EventKind::TrustPrompt{..}|EventKind::TrustPromptEnd{..})}
47    /// An input-owner notice for the session controller (rides the control ring, is never
48    /// acknowledged by the reader and never delivered by `next_event`/`next_control`).
49    pub fn is_owner_notice(&self)->bool{matches!(self.kind,EventKind::TrustPrompt{..}|EventKind::TrustPromptEnd{..})}
50    /// Wipe any typed text this event carries. The SDK scrubs everything it retains; a caller
51    /// that receives an event from `next_event`/`next_control` owns it and SHOULD call this
52    /// once the text has been consumed (keystrokes can be secrets).
53    pub fn scrub(&mut self){self.kind.scrub()}
54}
55impl EventKind{
56    /// Zero the text of a `Text` event in place (its allocation is kept, the string becomes empty).
57    pub fn scrub(&mut self){if let Self::Text(text)=self{ostd::session::zeroize_string(text)}}
58}
59/// Owner notices buffered per session (input queues at most eight per session).
60pub const NOTICE_MAX:usize=8;
61/// At most one 536-byte frame is buffered; chunk boundaries never split semantics.
62pub struct ControlDecoder{bytes:[u8;536],used:usize,expected:usize}
63impl Default for ControlDecoder{fn default()->Self{Self{bytes:[0;536],used:0,expected:24}}}
64impl ControlDecoder{
65    pub fn clear(&mut self){ostd::session::zeroize(&mut self.bytes[..self.used]);self.used=0;self.expected=24;}
66    pub fn is_partial(&self)->bool{self.used!=0}
67    /// Owner notices of `epoch` go to `notices` (at most [`NOTICE_MAX`]); other controls to
68    /// `out`. A notice of another epoch belongs to an ended foreground and is dropped. Other
69    /// controls of another epoch are `WrongEpoch` on a reader's own ring; with `drop_stale`
70    /// (the controller reading its own ring while a foreground child is the reader, so
71    /// nothing current can be on it but notices) they are dropped instead.
72    pub fn feed(&mut self,mut bytes:&[u8],epoch:u64,out:&mut alloc::collections::VecDeque<Event>,notices:&mut alloc::collections::VecDeque<Event>,drop_stale:bool)->Result<(),Status>{
73        while !bytes.is_empty(){let n=(self.expected-self.used).min(bytes.len());self.bytes[self.used..self.used+n].copy_from_slice(&bytes[..n]);self.used+=n;bytes=&bytes[n..];if self.used!=self.expected{continue}
74            if self.expected==24{let len=u16::from_le_bytes(self.bytes[18..20].try_into().map_err(|_|Status::Malformed)?) as usize;if len>512{self.clear();return Err(Status::Malformed)}self.expected=24+len;if self.used!=self.expected{continue}}
75            let event=match Event::decode(&self.bytes[..self.expected]){Ok(event)=>event,Err(error)=>{self.clear();return Err(error)}};self.clear();
76            if !event.is_control(){return Err(Status::Malformed)}
77            if event.is_owner_notice(){if event.epoch==epoch{if notices.len()>=NOTICE_MAX{return Err(Status::QuotaExceeded)}notices.try_reserve(1).map_err(|_|Status::QuotaExceeded)?;notices.push_back(event);}continue}
78            if event.epoch!=epoch{if drop_stale{continue}return Err(Status::WrongEpoch)}if out.len()>=128{return Err(Status::QuotaExceeded)}insert_ordered(out,event)?;
79        }Ok(())
80    }
81}
82pub(crate)fn insert_ordered(out:&mut alloc::collections::VecDeque<Event>,event:Event)->Result<(),Status>{
83    if out.iter().any(|e|e.sequence==event.sequence){return Err(Status::Malformed)}let index=out.iter().position(|e|e.sequence>event.sequence).unwrap_or(out.len());out.try_reserve(1).map_err(|_|Status::QuotaExceeded)?;out.insert(index,event);Ok(())
84}
