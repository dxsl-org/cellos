2//! Frontend host (SessionV1 selectors 23–28, Phase 05 S5-3 annex option C): a Cell that
3//! draws a terminal (the GUI Terminal, the private Phase04 driver) launches a shell that
4//! is the **controller of its own session** and attaches to it only as a frontend.
5//!
6//! The host never creates or owns a session. It
7//! 1. creates the two output rings it drains under its holder operation (`open`);
8//! 2. per launch, creates a shell operation, registers an input-owned frontend
9//!    capability (`RegisterFrontend`) under it, prepares the shell under that same
10//!    operation with the stdout/stderr *writer* ends (kind 1, slots 1/2) and the frontend
11//!    (kind 6, slot 8) as context grants, binds the frontend to exactly that prepared
12//!    child (`BindFrontendChild`), binds the input receipt to the kernel's kind-6 slot and
13//!    commits;
14//! 3. waits (`QueryFrontend`) until the shell called `CreateForFrontend` and the frontend
15//!    is Attached, then feeds normalized input through the returned attachment
16//!    (`SubmitEvents`/paste/`SubmitRaw`/`EndInput`/`Resize`, expected epoch 0: the input
17//!    owner substitutes the session's current epoch for a frontend attachment);
18//! 4. drains the output rings (the shell relays its own and its children's output there);
19//! 5. may release the frontend (`ReleaseFrontend`: the shell keeps a Detached session) and
20//!    re-attach a fresh frontend with the holder-bound `resume` capability
21//!    (`ReattachFrontend`), or close (cancel the shell, release the operation).
22//!
23//! No GUI/viUI dependency: callers render the drained bytes. Every method is
24//! non-blocking except [`FrontendHost::launch`]/[`FrontendHost::reattach`] (bounded owner
25//! RPCs), [`FrontendHost::wait_attached`] (caller-bounded) and [`FrontendHost::close`]
26//! (bounded reaping).

74/// Map one `QueryFrontend` answer to the host state and, when Attached, the link and
75/// the current `resume` capability.
76pub(crate) fn frontend_state(status:&FrontendStatusV1)->Result<(FrontendState,Option<(Link,ObjectRefV1)>),Status>{
77    Ok(match status.state{
78        FRONTEND_STATE_PENDING=>(FrontendState::Pending,None),
79        FRONTEND_STATE_BOUND=>(FrontendState::Bound,None),
80        FRONTEND_STATE_ATTACHED=>{
81            if status.session.is_zero()||status.attachment.is_zero()||status.resume.is_zero(){return Err(Status::Malformed)}
82            (FrontendState::Attached,Some((Link{session:status.session,attachment:status.attachment},status.resume)))
83        }
84        FRONTEND_STATE_CLOSED=>(FrontendState::Closed,None),
85        FRONTEND_STATE_LOST=>(FrontendState::Lost,None),
86        _=>return Err(Status::Malformed),
87    })
88}

105/// Decode a `QueryChild` result body: `state:u32, reason:u32, exit_code:i32,

109    let mut r=Reader::new(body);let state=r.u32()?;let reason=r.u32()?;let code=r.i32()?;if r.u32()?!=0{return Err(Status::Malformed)}let _sequence=r.u64()?;r.finish()?;

113/// Non-blocking drain: "nothing yet" and a closed pipe both read as zero bytes;
114/// a count the stream could not have produced is a protocol error.
115pub(crate) fn classify_drain(result:Result<usize,Status>,capacity:usize)->Result<usize,Status>{
116    match result{
117        Ok(n) if n<=capacity.min(DRAIN_MAX)=>Ok(n),
118        Ok(_)=>Err(Status::Malformed),
119        Err(Status::WouldBlock|Status::TimedOut|Status::Eof)=>Ok(0),
120        Err(e)=>Err(e),
121    }
122}

124#[derive(Clone,Copy,Debug,PartialEq,Eq)]
125pub(crate) enum ReapStep{Done,Retry{sleep:bool},Fail(Status)}
126/// One iteration of a bounded wait (`attempt` is zero based): `Ok(true)` is done,
127/// `Ok(false)` and `WouldBlock`/`TimedOut` retry.
128pub(crate) fn reap_step(attempt:u32,result:Result<bool,Status>)->ReapStep{
129    match result{
130        Ok(true)=>ReapStep::Done,
131        Ok(false)|Err(Status::WouldBlock|Status::TimedOut)=>if attempt+1>=REAP_ATTEMPTS{ReapStep::Fail(Status::TimedOut)}else{ReapStep::Retry{sleep:attempt>=REAP_YIELDS}},
132        Err(e)=>ReapStep::Fail(e),
133    }
134}

136/// Exact context length of the shell launch (argv/environment encodings of
137/// `ChildContext`, no cwd, three grants), computed before any reference exists.
138pub(crate) fn launch_context_len(argv:&[&str],environment:&[(&str,&str)])->Result<usize,Status>{
139    let argv_len=argv.iter().try_fold(7usize,|n,a|n.checked_add(a.len()+1)).ok_or(Status::Malformed)?;
140    let env_len=environment.iter().try_fold(0usize,|n,(k,v)|n.checked_add(4+k.len()+v.len())).ok_or(Status::Malformed)?;
141    if argv_len>512||env_len>4096||environment.len()>64{return Err(Status::QuotaExceeded)}
142    Ok(48+argv_len+env_len+3*40)
143}

196    /// Register a frontend and launch `path` (the shell) bound to it, under a fresh shell
197    /// operation. One live shell per host: a previous shell must have been observed
198    /// exited ([`Self::poll_child`]); its operation is released first. The shell starts
199    /// without a cwd grant (a cwd grant would path-seal it).
200    pub fn launch(&mut self,path:&str,argv:&[&str],environment:&[(&str,&str)])->Result<(),Status>{
201        if self.closed{return Err(Status::Revoked)}
202        if let Some(previous)=self.shell.as_ref(){
203            if previous.child.is_some()&&previous.exit.is_none(){return Err(Status::QuotaExceeded)}
204            self.retire_shell()?;
205        }
206        let operation=Operation::create(Some(self.operation.reference()))?;
207        self.shell=Some(Shell{operation,child:None,exit:None});
208        let result=self.launch_shell(path,argv,environment);
209        if result.is_err(){let _=operation.cancel(5);let _=self.retire_shell();}
210        result
211    }

213    fn launch_shell(&mut self,path:&str,argv:&[&str],environment:&[(&str,&str)])->Result<(),Status>{
214        let operation=self.shell.as_ref().ok_or(Status::NotFound)?.operation;
215        let frontend=self.register(self.rows,self.cols)?;self.frontend=Some(frontend);self.state=FrontendState::Pending;
216        let mut prepared=operation.prepare_child(path,launch_context_len(argv,environment)?)?;

218        let grants=launch_grants(stdout.reference().raw(),stderr.reference().raw(),frontend)?;

226    }

228    /// Release an ended (or never committed) shell's frontend and operation. The input

241    /// One `QueryFrontend`; caches the attachment and the current `resume` capability.
242    pub fn poll_frontend(&mut self)->Result<FrontendState,Status>{
243        if self.closed{return Err(Status::Revoked)}
244        if self.state==FrontendState::Detached{return Ok(FrontendState::Detached)}
245        let frontend=self.frontend.ok_or(Status::Stale)?;
246        let mut reply=[0u8;HEADER_LEN+FrontendStatusV1::LEN];let n=self.input(InputOp::QueryFrontend,&frontend.to_bytes(),&mut reply)?;
247        let (state,attached)=frontend_state(&FrontendStatusV1::decode(&reply[HEADER_LEN..n])?)?;
248        match attached{Some((link,resume))=>{self.link=Some(link);self.resume=Some(resume);}None=>{self.link=None;if matches!(state,FrontendState::Closed|FrontendState::Lost){self.resume=None;}}}
249        self.state=state;Ok(state)
250    }

252    /// Poll until the shell attached this frontend. `Revoked` when the frontend closed or
253    /// was lost, `Eof` when the shell ended before attaching, `TimedOut` after `timeout_ms`.
254    pub fn wait_attached(&mut self,timeout_ms:u64)->Result<(),Status>{
255        let end=ostd::session::monotonic_ns()?.saturating_add(timeout_ms.saturating_mul(1_000_000));let mut spins=0u32;
256        loop{
257            match self.poll_frontend()?{
258                FrontendState::Attached=>return Ok(()),
259                FrontendState::Closed|FrontendState::Lost|FrontendState::Detached=>return Err(Status::Revoked),
260                FrontendState::Pending|FrontendState::Bound=>{}
261            }
262            spins=spins.wrapping_add(1);
263            if spins%16==0&&self.poll_child()?==ChildPoll::Exited{return Err(Status::Eof)}
264            if ostd::session::monotonic_ns()?>=end{return Err(Status::TimedOut)}
265            let _=ostd::syscall::sys_set_timer(1);
266        }
267    }

348    /// Exit code of a shell that ended by itself; `None` while running, when cancelled or
349    /// faulted, or before [`Self::poll_child`] observed the exit.
350    pub fn exit_code(&self)->Option<i32>{self.shell.as_ref().and_then(|s|s.exit).and_then(|(_,code)|code)}

352    /// Cancel the live shell's operation. The exit is then observed by [`Self::poll_child`].
353    pub fn cancel_child(&mut self)->Result<(),Status>{
354        if self.closed{return Err(Status::Revoked)}
355        match self.shell.as_ref(){Some(shell) if shell.child.is_some()&&shell.exit.is_none()=>shell.operation.cancel(CANCEL_REASON),_=>Err(Status::NotFound)}
356    }

358    /// Detach this frontend (`ReleaseFrontend`). The shell keeps its session (Detached,
359    /// `Lost(4)` to its reader); [`Self::reattach`] resumes it while this host lives.
360    pub fn release(&mut self)->Result<(),Status>{
361        if self.closed{return Err(Status::Revoked)}
362        if self.state!=FrontendState::Attached||self.resume.is_none(){return Err(Status::WrongSession)}
363        let frontend=self.frontend.ok_or(Status::Stale)?;self.input(InputOp::ReleaseFrontend,&frontend.to_bytes(),&mut[0u8;HEADER_LEN])?;
364        self.frontend=None;self.link=None;self.state=FrontendState::Detached;Ok(())
365    }
366
367    /// Register a fresh frontend of `rows`×`cols` and resume the Detached session with the
368    /// holder-bound `resume` capability. The old attachment stays stale; the owner reissues
369    /// `resume`, read back with `QueryFrontend`.
370    pub fn reattach(&mut self,rows:u16,cols:u16)->Result<(),Status>{
371        if self.closed{return Err(Status::Revoked)}
372        if self.state!=FrontendState::Detached{return Err(Status::WrongSession)}
373        let resume=self.resume.ok_or(Status::Stale)?;api::services::session::validate_dimensions(rows,cols)?;
374        let frontend=self.register(rows,cols)?;
375        let mut body=[0u8;ReattachFrontendV1::LEN];ReattachFrontendV1{frontend,resume}.encode(&mut body)?;
376        let mut reply=[0u8;HEADER_LEN+ReattachResultV1::LEN];
377        let result=match self.input(InputOp::ReattachFrontend,&body,&mut reply){Ok(n)=>ReattachResultV1::decode(&reply[HEADER_LEN..n]),Err(status)=>Err(status)};
378        let attached=match result{
379            Ok(attached)=>attached,
380            Err(status)=>{let _=self.input(InputOp::ReleaseFrontend,&frontend.to_bytes(),&mut[0u8;HEADER_LEN]);return Err(status)}
381        };
382        self.frontend=Some(frontend);self.rows=rows;self.cols=cols;self.state=FrontendState::Bound;self.resume=None;
383        self.link=Some(Link{session:attached.session,attachment:attached.attachment});
384        match self.poll_frontend()?{
385            FrontendState::Attached if self.link==Some(Link{session:attached.session,attachment:attached.attachment})=>Ok(()),
386            FrontendState::Attached=>Err(Status::WrongSession),
387            _=>Err(Status::Stale),
388        }
389    }
390
391    /// Idempotent: cancels a live shell (bounded reaping), releases its operation, then
392    /// the holder operation (the output rings with it).
393    pub fn close(&mut self)->Result<(),Status>{
394        if self.closed{return Ok(())}
395        let mut first:Result<(),Status>=Ok(());
396        let live=self.shell.as_ref().is_some_and(|s|s.child.is_some()&&s.exit.is_none());
397        if let Some(shell)=self.shell.as_ref(){let _=shell.operation.cancel(CANCEL_REASON);}
398        if live{
399            let mut attempt=0u32;
400            first=loop{
401                match reap_step(attempt,self.poll_child().map(|p|p==ChildPoll::Exited)){
402                    ReapStep::Done=>break Ok(()),
403                    ReapStep::Retry{sleep}=>{ostd::syscall::sys_yield();if sleep{let _=ostd::syscall::sys_set_timer(1);}}
404                    ReapStep::Fail(status)=>break Err(status),
405                }
406                attempt+=1;
407            };
408        }
409        let retired=self.retire_shell();
410        let _=self.operation.cancel(CANCEL_REASON);
411        let closed=crate::session::close_operation_bounded(self.operation.reference());
412        self.closed=true;self.state=FrontendState::Closed;
413        first.and(retired).and(closed)
414    }
415}
416impl Drop for FrontendHost{fn drop(&mut self){let _=self.close();}}
