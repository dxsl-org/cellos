10#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
11pub enum KeySymbol {
12    Character(char), Enter, Backspace, Tab, Escape, Up, Down, Right, Left,
13    Home, End, Insert, Delete, PageUp, PageDown, F(u8),
14}
15
16impl fmt::Display for KeySymbol {
17    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
18        match self {
19            Self::Character(c) => write!(f, "{}", c),
20            Self::F(n) => write!(f, "F{}", n),
21            other => write!(f, "{:?}", other),
22        }
23    }
24}

26#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
27pub struct Modifiers { pub shift: bool, pub alt: bool, pub control: bool }
28
29impl Modifiers {
30    pub const NONE: Self = Self { shift: false, alt: false, control: false };
31    pub const SHIFT: Self = Self { shift: true, ..Self::NONE };
32    pub const ALT: Self = Self { alt: true, ..Self::NONE };
33    pub const CONTROL: Self = Self { control: true, ..Self::NONE };
34
35    pub fn csi_parameter(&self) -> u8 {
36        1 + self.shift as u8 + 2 * self.alt as u8 + 4 * self.control as u8
37    }
38    pub fn from_csi_parameter(m: u8) -> Result<Self, TerminalError> {
39        if !(2..=8).contains(&m) { return Err(TerminalError::UnsupportedInput); }
40        let n = m - 1;
41        Ok(Self { shift: n & 1 != 0, alt: n & 2 != 0, control: n & 4 != 0 })
42    }
43    pub fn is_empty(&self) -> bool { !self.shift && !self.alt && !self.control }
44    pub fn from_bits(bits: u32) -> Result<Self, TerminalError> {
45        if bits & !7 != 0 { return Err(TerminalError::UnsupportedInput); }
46        Ok(Self { shift: bits & 1 != 0, control: bits & 2 != 0, alt: bits & 4 != 0 })
47    }
48    pub fn bits(self) -> u32 {
49        self.shift as u32 | ((self.control as u32) << 1) | ((self.alt as u32) << 2)
50    }
51}

53#[derive(Debug, Clone, Copy, PartialEq, Eq)]
54#[repr(u8)]
55pub enum KeyState { Release = 0, Press = 1, Repeat = 2 }

57/// Text and Key are mutually exclusive for a single logical action. A Paste is
58/// one completely validated batch, never an incrementally delivered prefix.
59#[derive(Debug, Clone, PartialEq, Eq)]
60pub enum KeyAction {
61    Text(String),
62    Key { symbol: KeySymbol, modifiers: Modifiers, state: KeyState },
63    Paste(String),
64}
65
66impl KeySymbol {
67    pub fn from_code(code: u32) -> Result<Self, TerminalError> {
68        Ok(match code {
69            0x20..=0x10ffff if code != 0x7f => Self::Character(
70                char::from_u32(code).ok_or(TerminalError::UnsupportedInput)?),
71            0x110001 => Self::Enter, 0x110002 => Self::Escape,
72            0x110003 => Self::Backspace, 0x110004 => Self::Tab,
73            0x110005 => Self::Delete, 0x110006 => Self::Insert,
74            0x110010 => Self::Up, 0x110011 => Self::Down,
75            0x110012 => Self::Left, 0x110013 => Self::Right,
76            0x110014 => Self::Home, 0x110015 => Self::End,
77            0x110016 => Self::PageUp, 0x110017 => Self::PageDown,
78            0x110020..=0x11002b => Self::F((code - 0x110020 + 1) as u8),
79            _ => return Err(TerminalError::UnsupportedInput),
80        })
81    }
82    pub fn code(self) -> Result<u32, TerminalError> {
83        Ok(match self {
84            Self::Character(c) if c >= ' ' && c != '\u{7f}' => c as u32,
85            Self::Enter => 0x110001, Self::Escape => 0x110002,
86            Self::Backspace => 0x110003, Self::Tab => 0x110004,
87            Self::Delete => 0x110005, Self::Insert => 0x110006,
88            Self::Up => 0x110010, Self::Down => 0x110011,
89            Self::Left => 0x110012, Self::Right => 0x110013,
90            Self::Home => 0x110014, Self::End => 0x110015,
91            Self::PageUp => 0x110016, Self::PageDown => 0x110017,
92            Self::F(n @ 1..=12) => 0x110020 + n as u32 - 1,
93            _ => return Err(TerminalError::UnsupportedInput),
94        })
95    }
96}
97
98impl KeyAction {
99    pub fn from_key(code: u32, modifiers: u32, pressed: u8) -> Result<Self, TerminalError> {
100        let symbol = KeySymbol::from_code(code)?;
101        let modifiers = Modifiers::from_bits(modifiers)?;
102        let state = match pressed {
103            0 => KeyState::Release, 1 => KeyState::Press, 2 => KeyState::Repeat,
104            _ => return Err(TerminalError::UnsupportedInput),
105        };
106        validate_key(symbol, modifiers)?;
107        Ok(Self::Key { symbol, modifiers, state })
108    }
109    pub fn key_fields(&self) -> Result<(u32, u32, u8), TerminalError> {
110        match self {
111            Self::Key { symbol, modifiers, state } => {
112                validate_key(*symbol, *modifiers)?;
113                Ok((symbol.code()?, modifiers.bits(), *state as u8))
114            }
115            _ => Err(TerminalError::UnsupportedInput),
116        }
117    }
118}

120pub fn validate_text(bytes: &[u8]) -> Result<&str, TerminalError> {
121    api::services::session::validate_text(bytes).map_err(codec_error)?;
122    core::str::from_utf8(bytes).map_err(|_| TerminalError::UnsupportedInput)
123}
124
125pub fn validate_paste(bytes: &[u8]) -> Result<&str, TerminalError> {
126    api::services::session::validate_paste(bytes).map_err(codec_error)?;
127    core::str::from_utf8(bytes).map_err(|_| TerminalError::UnsupportedInput)
128}

130fn codec_error(status: api::cell_session::Status) -> TerminalError {
131    match status {
132        api::cell_session::Status::QuotaExceeded => TerminalError::BufferOverflow,
133        api::cell_session::Status::Malformed => TerminalError::InvalidInput,
134        _ => TerminalError::UnsupportedInput,
135    }
136}
137
138fn validate_key(symbol: KeySymbol, mods: Modifiers) -> Result<(), TerminalError> {
139    api::services::session::encode_key(symbol.code()?, mods.bits(), 0, &mut [0; 16])
140        .map_err(codec_error)?;
141    Ok(())
142}

144/// Validate the entire action before appending anything. Paste is bracketed by default.
145pub fn encode_key_action(action: &KeyAction, out: &mut Vec<u8>) -> Result<(), TerminalError> {
146    encode_key_action_with_paste(action, true, out)
147}
148
149pub fn encode_key_action_with_paste(action: &KeyAction, bracketed: bool, out: &mut Vec<u8>) -> Result<(), TerminalError> {
150    match action {
151        KeyAction::Text(text) => { validate_text(text.as_bytes())?; out.extend_from_slice(text.as_bytes()); }
152        KeyAction::Paste(text) => {
153            validate_paste(text.as_bytes())?;
154            if bracketed { out.extend_from_slice(b"\x1b[200~"); }
155            out.extend_from_slice(text.as_bytes());
156            if bracketed { out.extend_from_slice(b"\x1b[201~"); }
157        }
158        KeyAction::Key { symbol, modifiers, state } => {
159            let mut bytes = ostd::session::Scrub::<16>::new();
160            let count = api::services::session::encode_key(
161                symbol.code()?, modifiers.bits(), *state as u8, &mut bytes,
162            ).map_err(codec_error)?;
163            out.extend_from_slice(&bytes[..count]);
164        }
165    }
166    Ok(())
167}

169/// Receiver-monotonic wrapper around the one shared ABI grammar. The parser's
170/// paste markers remain internal until the complete <=64KiB batch validates.
171/// Call `expire` at `deadline_ns` even if the frontend has gone quiet.
172#[derive(Debug)]
173pub struct Decoder {
174    inner: api::services::session::SerialDecoder,
175    last_now: Option<u64>,
176    in_paste: bool,
177    paste: Vec<u8>,
178}
179
180impl Default for Decoder { fn default() -> Self { Self::new() } }
181
182impl Decoder {
183    pub const MAX_CONTROL_LEN: usize = 256;
184    pub const MAX_UTF8_LEN: usize = 4;
185    pub const MAX_PASTE_LEN: usize = 65536;
186    pub const ESC_TIMEOUT_NS: u64 = 30_000_000;
187
188    pub fn new() -> Self {
189        Self { inner: api::services::session::SerialDecoder::new(),
190            last_now: None, in_paste: false, paste: Vec::new() }
191    }
192    pub fn deadline_ns(&self) -> Option<u64> { self.inner.deadline_ns() }
193    fn clock(&mut self, now: u64) -> Result<(), TerminalError> {
194        if self.last_now.is_some_and(|last| now < last) { return Err(TerminalError::InvalidInput); }
195        self.last_now = Some(now);
196        Ok(())
197    }
198    fn reset(&mut self) {
199        // A fresh decoder (not `reset`) also overwrites the pending-sequence bytes.
200        self.inner = api::services::session::SerialDecoder::new(); self.in_paste = false;
201        ostd::session::zeroize_vec(&mut self.paste);
202    }
203    pub fn feed(&mut self, input: &[u8], now_ns: u64, out: &mut Vec<KeyAction>) -> Result<(), TerminalError> {
204        self.expire(now_ns, out)?;
205        let mut frame = ostd::session::Scrub::<{ api::services::session::EVENT_MAX }>::new();
206        for &byte in input {
207            let result = self.inner.feed(byte, now_ns, &mut frame).map_err(codec_error)
208                .and_then(|count| self.frame(&frame[..count], out));
209            if let Err(error) = result { self.reset(); return Err(error); }
210        }
211        Ok(())
212    }
213    pub fn expire(&mut self, now_ns: u64, out: &mut Vec<KeyAction>) -> Result<(), TerminalError> {
214        self.clock(now_ns)?;
215        let mut frame = ostd::session::Scrub::<{ api::services::session::EVENT_MAX }>::new();
216        let result = self.inner.expire(now_ns, &mut frame).map_err(codec_error)
217            .and_then(|count| self.frame(&frame[..count], out));
218        if result.is_err() { self.reset(); }
219        result
220    }
221    /// EOF never completes a missing scalar, control or paste terminator.
222    pub fn flush(&mut self, out: &mut Vec<KeyAction>) -> Result<(), TerminalError> {
223        let mut frame = ostd::session::Scrub::<{ api::services::session::EVENT_MAX }>::new();
224        let result = self.inner.finish(&mut frame).map_err(codec_error)
225            .and_then(|count| self.frame(&frame[..count], out))
226            .and_then(|()| if self.in_paste { Err(TerminalError::UnsupportedInput) } else { Ok(()) });
227        if result.is_err() { self.reset(); }
228        result
229    }
230    fn frame(&mut self, mut frames: &[u8], out: &mut Vec<KeyAction>) -> Result<(), TerminalError> {
231        while !frames.is_empty() {
232            let header = frames.get(..24).ok_or(TerminalError::InvalidInput)?;
233            let payload_len = u16::from_le_bytes([header[18], header[19]]) as usize;
234            let length = 24 + payload_len;
235            let frame = frames.get(..length).ok_or(TerminalError::InvalidInput)?;
236            self.one_frame(frame, out)?;
237            frames = &frames[length..];
238        }
239        Ok(())
240    }
241    fn one_frame(&mut self, frame: &[u8], out: &mut Vec<KeyAction>) -> Result<(), TerminalError> {
242        use api::services::session::event_kind;
243        // Internal decoder frames have no owner sequence/epoch. Paste text may
244        // contain HT/CR/LF, unlike ordinary TEXT; validate it as a whole batch.
245        let mut reader = api::cell_session::Reader::new(frame);
246        let _sequence = reader.u64().map_err(codec_error)?;
247        let _epoch = reader.u64().map_err(codec_error)?;
248        let kind = reader.u16().map_err(codec_error)?;
249        let length = reader.u16().map_err(codec_error)? as usize;
250        if reader.u32().map_err(codec_error)? != 0 { return Err(TerminalError::InvalidInput); }
251        let payload = reader.take(length).map_err(codec_error)?;
252        reader.finish().map_err(codec_error)?;
253        match kind {
254            event_kind::KEY if !self.in_paste => {
255                let mut key = api::cell_session::Reader::new(payload);
256                let code = key.u32().map_err(codec_error)?;
257                let modifiers = key.u32().map_err(codec_error)?;
258                let pressed = key.u8().map_err(codec_error)?;
259                if key.take(7).map_err(codec_error)?.iter().any(|&b| b != 0) {
260                    return Err(TerminalError::InvalidInput);
261                }
262                key.finish().map_err(codec_error)?;
263                out.push(KeyAction::from_key(code, modifiers, pressed)?);
264            }
265            event_kind::TEXT if self.in_paste => {
266                let next = self.paste.len().checked_add(payload.len()).ok_or(TerminalError::BufferOverflow)?;
267                if next > Self::MAX_PASTE_LEN { return Err(TerminalError::BufferOverflow); }
268                self.paste.extend_from_slice(payload);
269            }
270            event_kind::TEXT => {
271                out.push(KeyAction::Text(String::from(validate_text(payload)?)));
272            }
273            event_kind::PASTE_BEGIN if !self.in_paste && payload.is_empty() => {
274                self.in_paste = true;
275            }
276            event_kind::PASTE_END if self.in_paste && payload.is_empty() => {
277                validate_paste(&self.paste)?;
278                let text = match String::from_utf8(core::mem::take(&mut self.paste)) {
279                    Ok(text) => text,
280                    Err(error) => {
281                        let mut bytes = error.into_bytes();
282                        ostd::session::zeroize_vec(&mut bytes);
283                        return Err(TerminalError::UnsupportedInput);
284                    }
285                };
286                self.in_paste = false;
287                out.push(KeyAction::Paste(text));
288            }
289            _ => return Err(TerminalError::UnsupportedInput),
290        }
291        Ok(())
292    }
293}
294impl Drop for Decoder { fn drop(&mut self) { self.reset(); } }

296impl KeyAction {
297    /// Wipe the text of a `Text`/`Paste` action in place once it has been consumed.
298    pub fn scrub(&mut self) {
299        if let Self::Text(text) | Self::Paste(text) = self { ostd::session::zeroize_string(text); }
