2//! Raw-mode input for an interactive shell that is itself a session's foreground child.
3//!
4//! In canonical mode the input owner turns Ctrl-C into a cancellation of the committed
5//! foreground child's Operation and emits no `Interrupt` event. A shell that runs as that
6//! child would therefore be cancelled by every Ctrl-C. A shell that holds the raw-mode
7//! lease receives Ctrl-C as the byte `0x03` on stdin instead, and decides for itself what
8//! to stop. [`RawInput`] is the SDK's persistent form of that lease plus an event source
9//! that gives the shell the same [`Event`] stream the canonical path produces:
10//!
11//! - printable input -> `Text`; Enter/Backspace/arrows/... -> `Key`;
12//! - Ctrl-C (`0x03`) -> `Interrupt`;
13//! - a bracketed paste (the lease also requests the bracketed-paste bit) -> `PasteBegin`,
14//!   one `Text` with the whole paste, `PasteEnd`, so pasted newlines stay inert text;
15//! - resize / end-of-input arrive on the control endpoint exactly as in canonical mode;
16//! - stdin end of file -> `Eof`; an undecodable sequence -> `Lost`.
17//!
18//! No input-service change is involved: the lease is an ordinary `SetMode` the foreground
19//! reader is allowed to take, and it is restored by the owner if the holder dies.
20use alloc::collections::VecDeque;
21use alloc::vec::Vec;
22use api::cell_session::{ObjectRefV1,Status};
23use crate::codec::{Decoder,KeyAction,KeyState,KeySymbol,Modifiers};
24use crate::input::{Event,EventKind};
25use crate::session::TerminalSession;
26
27/// Raw routing plus bracketed paste (`TerminalMode` bits 0 and 4).
28pub const RAW_SHELL_MODE:u32=1|16;
29/// Reason reported by a `Lost` event raised for input the codec cannot decode.
30pub const LOST_UNDECODABLE:u32=1;
31
32/// A held raw-mode lease and the incremental decoder for the bytes it routes to stdin.
33pub struct RawInput{lease:Option<ObjectRefV1>,decoder:Decoder,pending:VecDeque<Event>,sequence:u64,epoch:u64}
34
35impl RawInput{
36    /// Take the raw-mode lease on `session` (the foreground reader's `SetMode`). The
37    /// session must have no unread typed input; `WouldBlock` means retry.
38    pub fn enter(session:&mut TerminalSession)->Result<Self,Status>{
39        let (lease,_)=session.acquire_mode(RAW_SHELL_MODE,RAW_SHELL_MODE)?;
40        Ok(Self{lease:Some(lease),decoder:Decoder::new(),pending:VecDeque::new(),sequence:0,epoch:session.epoch()})
41    }
42    /// Release the lease (idempotent). Buffered input is wiped.
43    pub fn leave(&mut self,session:&mut TerminalSession)->Result<(),Status>{
44        self.scrub();
45        match self.lease.take(){Some(lease)=>session.release_mode(lease),None=>Ok(())}
46    }
47    fn scrub(&mut self){for event in self.pending.iter_mut(){event.scrub()}self.pending.clear();}
48
49    /// Next input event. `deadline_ns` follows `TerminalSession::next_event`: `0` polls
50    /// without blocking (`WouldBlock`), `u64::MAX` waits forever, anything else is an
51    /// absolute monotonic deadline (`TimedOut`). Control events (resize, end of input) are
52    /// delivered before newly read bytes.
53    pub fn next_event(&mut self,session:&mut TerminalSession,deadline_ns:u64)->Result<Event,Status>{
54        loop{
55            if let Some(event)=self.pending.pop_front(){return Ok(event)}
56            match session.next_control(0){
57                Ok(event)=>return Ok(event),
58                Err(Status::WouldBlock)=>{},
59                Err(status)=>return Err(status),
60            }
61            let now=ostd::session::monotonic_ns()?;
62            let mut actions:Vec<KeyAction>=Vec::new();
63            let expired=self.decoder.expire(now,&mut actions);
64            let mut bytes=ostd::session::Scrub::<512>::new();
65            let mut lost=expired.is_err();
66            match session.read_input(&mut bytes,0){
67                Ok(0)=>return Err(Status::Malformed),
68                Ok(count)=>{if self.decoder.feed(&bytes[..count],now,&mut actions).is_err(){lost=true}},
69                Err(Status::WouldBlock)=>{},
70                Err(Status::Eof)=>{
71                    if self.decoder.flush(&mut actions).is_err(){lost=true}
72                    self.queue(actions,lost);
73                    self.push_kind(EventKind::Eof);
74                    continue;
75                }
76                Err(status)=>return Err(status),
77            }
78            self.queue(actions,lost);
79            if !self.pending.is_empty(){continue}
80            if deadline_ns==0{return Err(Status::WouldBlock)}
81            if deadline_ns!=u64::MAX&&ostd::session::monotonic_ns()?>=deadline_ns{return Err(Status::TimedOut)}
82            let _=ostd::syscall::sys_set_timer(1);
83        }
84    }
85    fn queue(&mut self,actions:Vec<KeyAction>,lost:bool){
86        for action in actions{self.convert_action(action)}
87        if lost{self.push_lost()}
88    }
89    fn convert_action(&mut self,action:KeyAction){
90        let mut kinds=Vec::new();
91        convert(action,&mut kinds);
92        for kind in kinds{self.push_kind(kind)}
93    }
94    fn push_lost(&mut self){self.push_kind(EventKind::Lost(LOST_UNDECODABLE))}
95    fn push_kind(&mut self,kind:EventKind){
96        self.sequence=self.sequence.saturating_add(1);
97        self.pending.push_back(Event{sequence:self.sequence,epoch:self.epoch,kind});
98    }
99}
100impl Drop for RawInput{fn drop(&mut self){self.scrub()}}

102/// True for Ctrl-C: the control-modified `c` with no other modifier, pressed or repeated.
103pub fn is_interrupt(symbol:KeySymbol,modifiers:Modifiers,state:KeyState)->bool{
104    matches!(symbol,KeySymbol::Character('c'|'C'))
105        &&modifiers==Modifiers::CONTROL
106        &&matches!(state,KeyState::Press|KeyState::Repeat)
107}
108
109/// One decoded action as the event kinds the canonical path would have produced.
110pub fn convert(action:KeyAction,out:&mut Vec<EventKind>){
111    match action{
112        KeyAction::Text(text)=>out.push(EventKind::Text(text)),
113        KeyAction::Paste(text)=>{
114            out.push(EventKind::PasteBegin);
115            out.push(EventKind::Text(text));
116            out.push(EventKind::PasteEnd);
117        }
118        KeyAction::Key{symbol,modifiers,state}=>{
119            if is_interrupt(symbol,modifiers,state){out.push(EventKind::Interrupt);return}
120            // Keys without a normalized code (a bare control byte) are dropped, like the
121            // canonical path drops what it cannot name.
122            if let Ok(code)=symbol.code(){
123                out.push(EventKind::Key{code,modifiers:modifiers.bits(),pressed:state as u8});
124            }
125        }
126    }

129#[cfg(test)]
130mod tests{
131    extern crate std;
132    use super::*;
133    use std::string::String;
134
135    fn decode(bytes:&[u8])->Vec<EventKind>{
136        let mut decoder=Decoder::new();
137        let mut actions=Vec::new();
138        decoder.feed(bytes,0,&mut actions).unwrap();
139        decoder.flush(&mut actions).unwrap();
140        let mut kinds=Vec::new();
141        for action in actions{convert(action,&mut kinds)}
142        kinds
143    }
144
145    #[test]
146    fn ctrl_c_byte_is_an_interrupt_and_other_controls_are_not(){
147        assert_eq!(decode(b"\x03"),[EventKind::Interrupt]);
148        // Ctrl-D, Ctrl-A are ordinary control keys the shell ignores.
149        let kinds=decode(b"\x04\x01");
150        assert_eq!(kinds.len(),2);
151        assert!(kinds.iter().all(|k|matches!(k,EventKind::Key{modifiers:2,..})));
152        // Alt-C and Ctrl-Shift-C are not Ctrl-C.
153        assert!(matches!(decode(b"\x1bc")[..],[EventKind::Key{modifiers:4,..}]));
154    }
155
156    #[test]
157    fn typed_text_enter_backspace_and_arrows_match_the_canonical_codes(){
158        assert_eq!(decode(b"ls\r"),[
159            EventKind::Text(String::from("l")),
160            EventKind::Text(String::from("s")),
161            EventKind::Key{code:0x110001,modifiers:0,pressed:1},
162        ]);
163        assert_eq!(decode(b"\x7f"),[EventKind::Key{code:0x110003,modifiers:0,pressed:1}]);
164        assert_eq!(decode(b"\x1b[A\x1b[B"),[
165            EventKind::Key{code:0x110010,modifiers:0,pressed:1},
166            EventKind::Key{code:0x110011,modifiers:0,pressed:1},
167        ]);
168    }
169
170    #[test]
171    fn a_bracketed_paste_is_inert_text_between_markers(){
172        assert_eq!(decode(b"\x1b[200~echo a\nrm b\x1b[201~"),[
173            EventKind::PasteBegin,
174            EventKind::Text(String::from("echo a\nrm b")),
175            EventKind::PasteEnd,
176        ]);
177        // A control byte inside a paste is refused with the whole paste, never executed
178        // and never taken for Ctrl-C.
179        let mut decoder=Decoder::new();
180        let mut actions=Vec::new();
181        assert!(decoder.feed(b"\x1b[200~a\x03b\x1b[201~",0,&mut actions).is_err());
182        assert!(actions.is_empty());
183    }
184
185    #[test]
186    fn the_lease_requests_raw_and_bracketed_paste_only(){
187        assert_eq!(RAW_SHELL_MODE,0b10001);
188        assert_eq!(RAW_SHELL_MODE&2,0,"echo stays with the shell's own echo");
189    }
