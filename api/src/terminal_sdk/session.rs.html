127    pub fn next_event(&mut self,deadline_ns:u64)->Result<Event,Status>{
128        if self.mode&1!=0{return Err(Status::WrongDirection)}
129        loop{
130            self.drain_control()?;
131            // Partial preceding controls cannot be overtaken by typed events.
132            if self.decoder.is_partial(){
133                if deadline_ns==0{return Err(Status::WouldBlock)}
134                if deadline_ns!=u64::MAX&&ostd::session::monotonic_ns()?>=deadline_ns{return Err(Status::TimedOut)}

136            }
137            if self.events.front().is_some_and(|event|!event.is_control()){return self.deliver_event()}
138            // Control may follow typed records still in the owner log. Fetch
139            // a nonblocking batch before choosing the next owner sequence.
140            let now=ostd::session::monotonic_ns()?;
141            if self.events.is_empty()&&deadline_ns!=0&&deadline_ns!=u64::MAX&&now>=deadline_ns{return Err(Status::TimedOut)}

143            let mut body=[0u8;56];let mut w=Writer::new(&mut body);w.reference(self.reference.0)?;w.reference(self.foreground.0)?;w.u64(self.cursor)?;w.u16(64)?;w.u16(0)?;w.u32(0)?;w.u64(bounded)?;

223/// CloseOperation retried while WouldBlock (receipts are acknowledged
224/// asynchronously by input/VFS). Bounded: first yields, then one-tick sleeps.
225pub(crate) fn close_operation_bounded(operation:OperationRef)->Result<(),Status>{
226    let mut last=Status::WouldBlock;
227    for attempt in 0..64u32{
228        match Operation::from_ref(operation).close(){
229            Ok(())=>return Ok(()),
230            Err(Status::WouldBlock)=>{ostd::syscall::sys_yield();if attempt>=8{let _=ostd::syscall::sys_set_timer(1);}}
231            Err(e)=>{last=e;break}
232        }
233    }
234    // Gave up (or definite refusal): never let correlation lanes accumulate.
235    ostd::session::retire_operation(operation.raw());
236    Err(last)
237}

272    fn launch_foreground(&mut self,path:&str,argv:&[&str],cwd:&[u8],environment:&[(&str,&str)],cwd_source:Option<ObjectRefV1>,with_completion:bool)->Result<ForegroundChild,Status>{
273        // The completion stream lives under the session's own operation, not the child's:
274        // cancelling the child (Ctrl-C) cancels its operation and every endpoint in it,
275        // and the record written before that must stay readable.
276        let completion=if with_completion{Some(Operation::from_ref(self.operation).create_stream(4096)?)}else{None};
277        let release=|completion:Option<(Endpoint,Endpoint)>|if let Some((reader,writer))=completion{let _=reader.close();let _=writer.close();};
278        let operation=match Operation::create(Some(self.operation)){Ok(operation)=>operation,Err(status)=>{release(completion);return Err(status)}};
279        let result=self.prepare_foreground(operation,path,argv,cwd,environment,cwd_source,completion);
280        if result.is_err(){
281            // Every failure path (including a late one after publication) cancels,
282            // then releases the child operation so the session's child quota and the
283            // SDK correlation lanes cannot leak; the completion stream goes with it.
284            let _=operation.cancel(5);
285            let _=close_operation_bounded(operation.reference());
286            release(completion);
287        }
288        result
289    }
290    /// Release a reaped, restored foreground's child operation. Input acknowledges its
291    /// receipt asynchronously and VFS settles its cwd receipt when [`Self::finish_foreground`]
292    /// drops the cwd lease, so CloseOperation answers WouldBlock until then; retry for a
293    /// bounded number of yields. Correlation lanes for the operation are dropped on
294    /// success and on giving up. The completion reader (an endpoint of the session's
295    /// operation) is closed first: read the record before this.
296    pub fn close_foreground(&mut self,child:&ForegroundChild)->Result<(),Status>{
297        // A repeated close finds the reader already gone; the operation close decides.
298        if let Some(reader)=child.completion{let _=reader.close();}
299        close_operation_bounded(child.operation)
300    }

342/// Exact immutable-context length and grant count of a foreground launch (stdio, optional
343/// cwd, optional completion writer), computed before any authority reference exists.
344pub(crate) fn foreground_context_len(argv:&[&str],cwd:&[u8],environment:&[(&str,&str)],with_completion:bool)->Result<(usize,usize),Status>{
345    let argv_len=argv.iter().try_fold(7usize,|n,a|n.checked_add(a.len()+1)).ok_or(Status::Malformed)?;
346    let env_len=environment.iter().try_fold(0usize,|n,(k,v)|n.checked_add(4+k.len()+v.len())).ok_or(Status::Malformed)?;
347    if argv_len>512||cwd.len()>4096||env_len>4096||environment.len()>64{return Err(Status::QuotaExceeded)}
348    let grant_count=4+usize::from(!cwd.is_empty())+usize::from(with_completion);
349    Ok((48+argv_len+cwd.len()+env_len+grant_count*40,grant_count))
350}
