2//! Shell-owned byte pipelines: children whose fd 0/1/2 are delegated stream
3//! endpoints, never the keyboard owner.
4//!
5//! A [`Pipeline`] is a tree of kernel Operations below one root (a child of the
6//! session Operation). The kernel cancels the Operation that holds a producer's
7//! endpoint, and everything below it, when the last reader of a stream closes, and
8//! an endpoint can only be delegated to a child prepared at or below the Operation
9//! it lives in. The tree is laid out so that exactly the right things stop:
10//!
11//! - the root holds what the shell reads (output and diagnostics) and is never
12//!   cancelled by a stage;
13//! - stage `i` runs in its own stage Operation, so a stage that was stopped never
14//!   takes a neighbour with it;
15//! - the stream between stage `i` and `i + 1` lives in ring Operation `i`, and the
16//!   ring Operations nest outward toward the last stage (`ring n-2` is a child of the
17//!   root, `ring i` a child of `ring i + 1`); stage `i + 1` and every ring and stage
18//!   upstream of it sit below `ring i`. Losing the consumer of a stream therefore
19//!   cancels everything upstream of it (nothing is producing for nobody), never a
20//!   stage downstream, which still reads what was written and ends normally;
21//! - the stream the shell feeds stage 0 through lives in stage 0's own Operation.
22//!
23//! [`Pipeline::spawn`] prepares and commits one child inside its stage Operation
24//! through the same `prepare_child -> delegate -> write_context -> commit`
25//! sequence a foreground child uses, but without `authorize_input`, input-service
26//! receipts or a foreground lease: the child can never own the terminal. Only the
27//! VFS prepared-cwd reservation is bound, exactly as `spawn_foreground` does.
28//!
29//! Ownership: the caller keeps every endpoint copy it passes to `spawn` and
30//! must close it afterwards (the child holds its own delegated copies). EOF at a
31//! reader is the close of the last writer copy, so callers must close their
32//! writer copies once the stages that need them are spawned. A reader passed as
33//! stdin is moved to the child at commit (the kernel consumes the caller's copy).
34use api::cell_session::{ChildRefV1,KernelOp,ObjectRefV1,Reader,Writer,Status,VfsOp,HEADER_LEN};
35use api::service_binding::ServiceBinding;
36use ostd::child_context::{ChildContext,Grant,GrantSlot};
37use crate::stream::{Endpoint,EndpointRights,Operation,OperationRef,ReceiptRef,ok};

39/// Maximum immutable-context argv / cwd / environment sizes (ABI limits).
40const ARGV_MAX:usize=512;
41const CWD_MAX:usize=4096;
42const ENV_MAX:usize=4096;
43const ENV_ENTRIES_MAX:usize=64;
44const PATH_MAX:usize=4096;
45/// Ring capacity of the empty stdin stream (the kernel minimum).
46const PLACEHOLDER_CAPACITY:u32=64;
47/// Ring capacity of every pipeline stream (one transfer unit is 512 bytes).
48const STREAM_CAPACITY:u32=4096;
49/// Most stages one pipeline may have (the shell's own pipeline limit).
50pub const MAX_STAGES:usize=16;

52/// Everything needed to launch one pipeline stage.
53#[derive(Debug,Clone,Copy)]
54pub struct PipeSpec<'a>{
55    pub path:&'a str,
56    pub argv:&'a[&'a str],
57    /// Raw cwd bytes; empty means no cwd grant. Non-empty requires either
58    /// `cwd_source` or a UTF-8 path the VFS can open.
59    pub cwd:&'a[u8],
60    pub environment:&'a[(&'a str,&'a str)],
61    /// Existing VFS directory grant to derive the child's cwd from (never an
62    /// ambient path lookup when present).
63    pub cwd_source:Option<ObjectRefV1>,
64}

66/// Pure pre-effect validation. Returns the exact immutable context length.
67/// Size violations are `QuotaExceeded`, malformed bytes `Malformed`, a path-only
68/// cwd that is not UTF-8 `UnsupportedInput` (same classes as `spawn_foreground`).
69pub(crate) fn validate(spec:&PipeSpec<'_>)->Result<usize,Status>{
70    if spec.path.is_empty()||spec.path.len()>PATH_MAX||spec.path.as_bytes().contains(&0){return Err(Status::Malformed)}
71    let argv_len=spec.argv.iter().try_fold(7usize,|n,a|n.checked_add(a.len()+1)).ok_or(Status::Malformed)?;
72    let env_len=spec.environment.iter().try_fold(0usize,|n,(k,v)|n.checked_add(4+k.len()+v.len())).ok_or(Status::Malformed)?;
73    if argv_len>ARGV_MAX||spec.cwd.len()>CWD_MAX||env_len>ENV_MAX||spec.environment.len()>ENV_ENTRIES_MAX{return Err(Status::QuotaExceeded)}
74    if spec.argv.iter().any(|a|a.as_bytes().contains(&0))||spec.cwd.contains(&0){return Err(Status::Malformed)}
75    if !spec.cwd.is_empty()&&spec.cwd_source.is_none()&&core::str::from_utf8(spec.cwd).is_err(){return Err(Status::UnsupportedInput)}
76    let grant_count=3+usize::from(!spec.cwd.is_empty());
77    Ok(48+argv_len+spec.cwd.len()+env_len+grant_count*40)
78}

80/// Kernel child state: terminal record complete (`QueryChild` `state`).
81const CHILD_TERMINAL:u32=3;
82/// `QueryChild` result body after the result header: state, reason, exit code,
83/// reserved, terminal sequence.
84const CHILD_STATUS_LEN:usize=24;
85/// Upper bound on one [`PipeChild::wait`]. A child that never reaches its terminal
86/// record inside it yields `Status::TimedOut`; it is never reported as exited.
87const WAIT_BUDGET_NS:u64=30_000_000_000;
88/// Polls that only yield before the wait starts sleeping one tick per poll.
89const WAIT_YIELDS:u32=8;
90
91/// The kernel's terminal record of a committed child, as its preparer sees it.
92#[derive(Debug,Clone,Copy,PartialEq,Eq)]
93pub struct ChildStatus{
94    /// 1 live, 2 retiring, 3 terminal (the only state with a final `exit_code`).
95    pub state:u32,
96    pub reason:u32,
97    pub exit_code:i32,
98    pub terminal_sequence:u64,
99}
100impl ChildStatus{
101    pub fn is_terminal(&self)->bool{self.state==CHILD_TERMINAL}
102    /// Decode the 24-byte `QueryChild` body (after the 24-byte result header).
103    /// Any other length, a nonzero reserved word or an unknown state is `Malformed`.
104    pub(crate) fn decode(body:&[u8])->Result<Self,Status>{
105        if body.len()!=CHILD_STATUS_LEN{return Err(Status::Malformed)}
106        let mut r=Reader::new(body);
107        let state=r.u32()?;let reason=r.u32()?;let exit_code=r.i32()?;let reserved=r.u32()?;let terminal_sequence=r.u64()?;
108        r.finish()?;
109        if reserved!=0||state>CHILD_TERMINAL{return Err(Status::Malformed)}
110        Ok(Self{state,reason,exit_code,terminal_sequence})
111    }
112}

114/// How [`PipeChild::wait`] spends the gap before poll number `attempt` (0-based).
115#[derive(Debug,Clone,Copy,PartialEq,Eq)]
116enum Pause{Yield,Tick}
117fn pause_for(attempt:u32)->Pause{if attempt<WAIT_YIELDS{Pause::Yield}else{Pause::Tick}}
118/// Absolute deadline for a wait that starts at `now_ns`; `Malformed` on overflow.
119fn wait_deadline(now_ns:u64)->Result<u64,Status>{now_ns.checked_add(WAIT_BUDGET_NS).ok_or(Status::Malformed)}

121/// A committed pipeline stage. Keep it alive until the stage is reaped and drop it
122/// before closing its stage Operation: it keeps a dynamically admitted cwd
123/// directory open, and dropping it lets VFS settle the child's cwd receipt.
124pub struct PipeChild{child:ChildRefV1,operation:OperationRef,_cwd:Option<CwdLease>}
125impl PipeChild{
126    pub fn child(&self)->ChildRefV1{self.child}
127    pub fn root_tid(&self)->usize{self.child.root_tid as usize}
128    /// One `QueryChild` (non-mutating, idempotent): the kernel's current record of
129    /// this child. Only the preparer may ask, and only until the pipeline Operation
130    /// is closed. Kernel refusals surface as their own status (`WrongOwner`,
131    /// `Stale`); a reply that is not exactly header plus 24 bytes is `Malformed`.
132    pub fn query(&self)->Result<ChildStatus,Status>{query_child(self.operation,self.child)}
133    /// Wait until the kernel's terminal record of the child exists and return its
134    /// exit code. Polls [`PipeChild::query`] (a few yields, then one-tick sleeps)
135    /// while the child is live or retiring, for at most 30 seconds. Never guesses:
136    /// `Status::TimedOut` when the budget ends first (the child may still run, so
137    /// cancel the pipeline operation if it must stop), `Status::Unsupported` when
138    /// the clock is unavailable, `Stale`/`WrongOwner`/other kernel statuses as
139    /// reported. Transient `WouldBlock`/`TimedOut` answers are polled again.
140    /// Idempotent: the record stays readable until the operation is closed.
141    pub fn wait(&self)->Result<i32,Status>{self.wait_status().map(|status|status.exit_code)}
142    /// [`PipeChild::wait`] returning the whole terminal record: `reason` is nonzero
143    /// when the kernel stopped the child (a cancelled Operation) instead of the
144    /// child exiting by itself, and `exit_code` is then meaningless.
145    pub fn wait_status(&self)->Result<ChildStatus,Status>{wait_child_status(self.operation,self.child)}
146}

148/// One `QueryChild` of `child` under its preparing `operation`.
149pub(crate) fn query_child(operation:OperationRef,child:ChildRefV1)->Result<ChildStatus,Status>{
150    let mut reply=[0u8;HEADER_LEN+CHILD_STATUS_LEN];
151    let total=crate::stream::call(operation,KernelOp::QueryChild,false,&child.to_bytes(),&mut reply)?;
152    ChildStatus::decode(reply.get(HEADER_LEN..total).ok_or(Status::Malformed)?)
153}
154
155/// Bounded poll of [`query_child`] until the terminal record exists (see [`PipeChild::wait`]).
156pub(crate) fn wait_child_status(operation:OperationRef,child:ChildRefV1)->Result<ChildStatus,Status>{
157    let deadline=wait_deadline(ostd::session::monotonic_ns()?)?;
158    let mut attempt=0u32;
159    loop{
160        match query_child(operation,child){
161            Ok(status) if status.is_terminal()=>return Ok(status),
162            Ok(_)|Err(Status::WouldBlock|Status::TimedOut)=>{}
163            Err(e)=>return Err(e),
164        }
165        if ostd::session::monotonic_ns()?>=deadline{return Err(Status::TimedOut)}
166        match pause_for(attempt){
167            Pause::Yield=>ostd::syscall::sys_yield(),
168            Pause::Tick=>{let _=ostd::syscall::sys_set_timer(1);}
169        }
170        attempt=attempt.saturating_add(1);
171    }
172}

174/// Where an Operation of a pipeline hangs.
175#[derive(Debug,Clone,Copy,PartialEq,Eq)]
176enum Parent{Root,Ring(usize)}
177/// The parent of every ring Operation (index = ring) and stage Operation (index =
178/// stage) of a `stages`-stage pipeline. Ring `i` carries the stream from stage `i`
179/// to stage `i + 1`; see the module documentation for why they nest outward.
180#[derive(Debug,PartialEq,Eq)]
181struct Plan{rings:alloc::vec::Vec<Parent>,stages:alloc::vec::Vec<Parent>}
182fn layout_plan(stages:usize)->Plan{
183    let rings=(0..stages.saturating_sub(1)).map(|ring|if ring+2==stages{Parent::Root}else{Parent::Ring(ring+1)}).collect();
184    let stage_parents=(0..stages).map(|stage|if stages==1{Parent::Root}else{Parent::Ring(stage.saturating_sub(1))}).collect();
185    Plan{rings,stages:stage_parents}
186}

188/// The root Operation of a pipeline and the ring and stage Operations below it.
189pub struct Pipeline{operation:Operation,rings:alloc::vec::Vec<Operation>,stages:alloc::vec::Vec<Operation>,created:alloc::vec::Vec<Operation>}
190impl Pipeline{
191    /// One root Operation per pipeline, a child of the session operation.
192    pub fn create(parent:OperationRef)->Result<Pipeline,Status>{
193        Ok(Pipeline{operation:Operation::create(Some(parent))?,rings:alloc::vec::Vec::new(),stages:alloc::vec::Vec::new(),created:alloc::vec::Vec::new()})
194    }
195    pub fn operation(&self)->OperationRef{self.operation.reference()}
196    /// `(reader, writer)` in the root Operation, the same order as
197    /// `Operation::create_stream`: for the ends the shell reads (output and
198    /// diagnostics). The root is never cancelled by a stage.
199    pub fn stream(&self)->Result<(Endpoint,Endpoint),Status>{self.operation.create_stream(STREAM_CAPACITY)}
200    /// Open the ring and stage Operations for `stages` stages (at least one, at most
201    /// sixteen). Call once, before any stream or spawn.
202    pub fn layout(&mut self,stages:usize)->Result<(),Status>{
203        if stages==0||stages>MAX_STAGES||!self.stages.is_empty(){return Err(Status::Malformed)}
204        let plan=layout_plan(stages);
205        let mut rings:alloc::vec::Vec<Option<Operation>>=alloc::vec![None;plan.rings.len()];
206        // Outermost ring first: ring `stages - 2` below the root, ring `i` below ring `i + 1`.
207        for ring in (0..plan.rings.len()).rev(){
208            let parent=match plan.rings[ring]{Parent::Root=>self.operation,Parent::Ring(outer)=>rings[outer].ok_or(Status::Malformed)?};
209            let operation=Operation::create(Some(parent.reference()))?;
210            self.created.push(operation);rings[ring]=Some(operation);
211        }
212        self.rings=rings.into_iter().flatten().collect();
213        for parent in plan.stages{
214            let parent=match parent{Parent::Root=>self.operation,Parent::Ring(ring)=>self.rings[ring]};
215            let operation=Operation::create(Some(parent.reference()))?;
216            self.created.push(operation);self.stages.push(operation);
217        }
218        Ok(())
219    }
220    /// `(reader, writer)` of the stream between stage `ring` and stage `ring + 1`,
221    /// created in that ring's Operation.
222    pub fn link(&self,ring:usize)->Result<(Endpoint,Endpoint),Status>{
223        self.rings.get(ring).ok_or(Status::Malformed)?.create_stream(STREAM_CAPACITY)
224    }
225    /// `(reader, writer)` created inside stage `stage`'s own Operation: the stream
226    /// the shell feeds that stage through. When the stage stops reading the kernel
227    /// cancels only that stage's Operation.
228    pub fn stage_stream(&self,stage:usize)->Result<(Endpoint,Endpoint),Status>{
229        self.stages.get(stage).ok_or(Status::Malformed)?.create_stream(STREAM_CAPACITY)
230    }
231    /// Cancel the whole pipeline (root and everything below it).
232    pub fn cancel(&self,reason:u32)->Result<(),Status>{self.operation.cancel(reason)}
233    /// Release every Operation, children before parents, then the root (bounded retry
234    /// on `WouldBlock`; correlation lanes are retired on success and on giving up).
235    /// Drop every reaped [`PipeChild`] first: VFS settles a stage's cwd receipt, which
236    /// its Operation's release waits for, only when the owner contacts it.
237    /// The first failure is returned after every release was attempted.
238    pub fn close(self)->Result<(),Status>{
239        let mut first=Ok(());
240        for operation in self.created.iter().rev(){
241            if let Err(e)=close_bounded(operation.reference()){if first.is_ok(){first=Err(e)}}
242        }
243        let root=close_bounded(self.operation.reference());
244        first.and(root)
245    }
246
247    /// Prepare and commit one child in stage Operation `stage` whose fd0/fd1/fd2 are
248    /// delegated copies of the given endpoints (stdin with reader rights,
249    /// stdout/stderr with writer rights). `None` for `stdin` gives the child an
250    /// empty stream: its first read sees end of file.
251    ///
252    /// The caller keeps its own endpoint copies and must close them; this never
253    /// closes a caller endpoint. Failure before commit publishes nothing (the
254    /// prepared child is aborted and every endpoint made here is closed). A
255    /// failure at or after commit may have published a child, so it cancels the
256    /// stage Operation (reason 5, the `spawn_foreground` reason) and returns the
257    /// error; the Operation itself is released by [`Pipeline::close`].
258    pub fn spawn(&self,stage:usize,spec:&PipeSpec<'_>,stdin:Option<Endpoint>,stdout:Endpoint,stderr:Endpoint)->Result<PipeChild,Status>{
259        let operation=*self.stages.get(stage).ok_or(Status::Malformed)?;
260        let context_len=validate(spec)?;
261        let mut committing=false;
262        let result=Self::prepare(operation,spec,context_len,stdin,stdout,stderr,&mut committing);
263        match result{
264            Err(_) if committing=>{let _=operation.cancel(5);}
265            // Before commit, a stage whose Operation the kernel cancelled lost its
266            // consumer while it was being prepared (the cancellation aborts the
267            // prepared child, so any step after it fails): report that as `Cancelled`.
268            Err(status) if status!=Status::Cancelled&&operation.query().is_ok_and(|state|state.state!=0)=>return Err(Status::Cancelled),
269            _=>{}
270        }
271        result
272    }

274    fn prepare(operation:Operation,spec:&PipeSpec<'_>,context_len:usize,stdin:Option<Endpoint>,stdout:Endpoint,stderr:Endpoint,committing:&mut bool)->Result<PipeChild,Status>{
275        // Endpoint copies owned by this call, declared before `prepared` so an
276        // unpublished child is aborted first. `empty_reader` (the reader of the empty
277        // stdin stream) outlives the commit; `delegated` is closed only while the
278        // child is unpublished.
279        let mut empty_reader=Closer::new();let mut delegated=Closer::new();
280        let mut prepared=operation.prepare_child(spec.path,context_len)?;
281        let stdin_source=match stdin{
282            Some(e)=>e,
283            None=>{
284                // Closing the last reader of a stream stops its producers, so the
285                // reader stays open until the child holds its own copy. The writer is
286                // closed now: the child reads end of file at once.
287                let(reader,writer)=operation.create_stream(PLACEHOLDER_CAPACITY)?;
288                empty_reader.push(reader);
289                writer.close()?;
290                reader
291            }
292        };
293        let stdin_copy=stdin_source.delegate(&prepared,EndpointRights::READ_ONLY)?;delegated.push(stdin_copy);
294        let stdout_copy=stdout.delegate(&prepared,EndpointRights::WRITE_DELEGATE)?;delegated.push(stdout_copy);
295        let stderr_copy=stderr.delegate(&prepared,EndpointRights::WRITE_DELEGATE)?;delegated.push(stderr_copy);
296
297        let mut cwd_lease=None;let mut reservation=None;let mut child_cwd=None;
298        let vfs_owner=if spec.cwd.is_empty(){None}else{Some(ostd::session::service_binding(api::syscall::service::VFS)?)};
299        if let Some(owner)=vfs_owner{
300            let lease=cwd_lease.insert(CwdLease::new(owner,operation.reference()));
301            let source=match spec.cwd_source{
302                Some(source)=>source,
303                None=>{
304                    let path=core::str::from_utf8(spec.cwd).map_err(|_|Status::UnsupportedInput)?;
305                    // The dynamically opened directory is granted under the caller's
306                    // own principal generation (held by every endpoint it owns).
307                    let generation=stderr.inspect()?.holder.generation;
308                    let mut send=[0u8;4096];let mut recv=[0u8;4096];
309                    let request=api::ipc::VfsRequest::OpenRootDir{path};
310                    let response=ostd::ipc::service_call_typed_bounded::<_,api::ipc::VfsResponse<'_>>(owner.tid as usize,&request,&mut send,&mut recv,100).map_err(|_|Status::Indeterminate)?;
311                    let id=match response{api::ipc::VfsResponse::DirHandle(handle)=>handle.0,_=>return Err(Status::Denied)};
312                    // From here the lease closes the handle on every exit.
313                    lease.dir=Some(api::dir_handles::ViDirHandle(id));
314                    if ostd::session::service_binding(api::syscall::service::VFS)?!=owner{return Err(Status::Stale)}
315                    ObjectRefV1{id,generation}
316                }
317            };
318            let mut body=[0u8;32];let mut w=Writer::new(&mut body);w.reference(prepared.reference().raw())?;w.reference(source)?;
319            let mut reply=[0u8;56];let n=vfs(owner,operation.reference(),VfsOp::ReserveCwd,&body,&mut reply)?;
320            let mut r=Reader::new(&reply[24..n]);reservation=Some(r.reference()?);child_cwd=Some(r.reference()?);r.finish()?;
321        }
322
323        let endpoint_grant=|slot:GrantSlot,copy:Endpoint,rights:EndpointRights|Grant::endpoint(slot,copy.reference().raw(),rights.bits() as u16);
324        let mut grants=[endpoint_grant(GrantSlot::Stdin,stdin_copy,EndpointRights::READ_ONLY)?,endpoint_grant(GrantSlot::Stdout,stdout_copy,EndpointRights::WRITE_DELEGATE)?,endpoint_grant(GrantSlot::Stderr,stderr_copy,EndpointRights::WRITE_DELEGATE)?,Grant{kind:1,rights:0,slot:GrantSlot::Stdin,owner_service:0,object:ObjectRefV1::ZERO}];
325        let mut count=3;
326        if let Some(grant)=child_cwd{grants[3]=Grant::cwd(grant)?;count=4;}
327        let context=ChildContext::new(operation.reference().raw(),spec.argv,spec.cwd,spec.environment,&grants[..count])?;
328        if context.as_bytes().len()!=context_len{return Err(Status::Malformed)}
329        prepared.write_context(&context)?;
330        if let(Some(owner),Some(reservation))=(vfs_owner,reservation){
331            let mut body=[0u8;32];let mut w=Writer::new(&mut body);w.reference(reservation)?;w.reference(prepared.reference().raw())?;
332            let mut reply=[0u8;40];let n=vfs(owner,operation.reference(),VfsOp::FinalizeCwd,&body,&mut reply)?;
333            let receipt=ReceiptRef::from_raw(ObjectRefV1::from_bytes(&reply[24..n])?)?;
334            if let Some(lease)=cwd_lease.as_mut(){lease.receipt=Some(receipt);}
335            prepared.bind_receipt(2,4,owner,receipt)?;
336        }
337        // From the commit call on, a child may be published (even when the call
338        // reports an error); the delegated copies then belong to it.
339        *committing=true;delegated.disarm();
340        let child=prepared.commit()?;
341        // The child holds its own reader of the empty stream now.
342        empty_reader.close_all();
343        Ok(PipeChild{child,operation:operation.reference(),_cwd:cwd_lease})
344    }
345}

347/// Closes every pushed endpoint on drop or on demand; `disarm` forgets them.
348struct Closer{list:[Option<Endpoint>;4],len:usize}
349impl Closer{
350    fn new()->Self{Self{list:[None;4],len:0}}
351    fn push(&mut self,endpoint:Endpoint){if self.len<self.list.len(){self.list[self.len]=Some(endpoint);self.len+=1;}}
352    fn close_all(&mut self){for slot in self.list.iter_mut().take(self.len){if let Some(endpoint)=slot.take(){let _=endpoint.close();}}self.len=0;}
353    fn disarm(&mut self){self.list=[None;4];self.len=0;}
354}
355impl Drop for Closer{fn drop(&mut self){self.close_all()}}

357fn vfs(owner:ServiceBinding,operation:OperationRef,op:VfsOp,body:&[u8],reply:&mut[u8])->Result<usize,Status>{
358    let header=ostd::session::service_call(owner,api::syscall::service::VFS,operation.raw(),op as u16,true,body,reply)?;ok(header.status)?;Ok(header.total_len as usize)
359}

361/// CloseOperation retried while `WouldBlock` (receipt owners settle receipts
362/// asynchronously): eight yields, then one-tick sleeps, 64 attempts at most.
363/// Correlation lanes are retired on success and on giving up.
364fn close_bounded(operation:OperationRef)->Result<(),Status>{
365    let mut last=Status::WouldBlock;
366    for attempt in 0..64u32{
367        match Operation::from_ref(operation).close(){
368            Ok(())=>return Ok(()),
369            Err(Status::WouldBlock)=>{ostd::syscall::sys_yield();if attempt>=8{let _=ostd::syscall::sys_set_timer(1);}}
370            Err(e)=>{last=e;break}
371        }
372    }
373    ostd::session::retire_operation(operation.raw());
374    Err(last)
375}

377/// The owner's side of a child's VFS cwd reservation, dropped once the child is
378/// reaped. VFS does not poll a running child that never contacts it: it settles an
379/// owner's reservations when that owner sends a request, and closing the child's
380/// Operation waits for the settled receipt. Dropping the lease is that request: it
381/// closes the dynamically admitted source directory, or else queries the receipt
382/// (`ResolveCwd`, action 0). Cleanup only; VFS still validates every handle operation.
383pub(crate) struct CwdLease{owner:ServiceBinding,operation:OperationRef,pub(crate) dir:Option<api::dir_handles::ViDirHandle>,pub(crate) receipt:Option<ReceiptRef>}
384impl CwdLease{pub(crate) fn new(owner:ServiceBinding,operation:OperationRef)->Self{Self{owner,operation,dir:None,receipt:None}}}
385impl Drop for CwdLease{fn drop(&mut self){
386    if ostd::session::service_binding(api::syscall::service::VFS).ok()!=Some(self.owner){return}
387    if let Some(dir)=self.dir{
388        let mut send=[0u8;4096];let mut recv=[0u8;4096];
389        let _=ostd::ipc::service_call_typed_bounded::<_,api::ipc::VfsResponse<'_>>(self.owner.tid as usize,&api::ipc::VfsRequest::CloseDir{dir},&mut send,&mut recv,100);
390    }else if let Some(receipt)=self.receipt{
391        let mut body=[0u8;24];let mut w=Writer::new(&mut body);
392        if w.reference(receipt.raw()).and_then(|()|w.u32(0)).and_then(|()|w.u32(0)).is_ok(){
393            let _=ostd::session::service_call(self.owner,api::syscall::service::VFS,self.operation.raw(),VfsOp::ResolveCwd as u16,false,&body,&mut[0u8;HEADER_LEN+24]);
394        }
395    }
396}}

462    /// Whether `node` hangs (at any depth) below ring `ring`.
463    fn below_ring(plan:&Plan,mut parent:Parent,ring:usize)->bool{
464        loop{
465            match parent{
466                Parent::Root=>return false,
467                Parent::Ring(at) if at==ring=>return true,
468                Parent::Ring(at)=>parent=plan.rings[at],
469            }
470        }
471    }

476    #[test]
477    fn rings_nest_outward_toward_the_last_stage(){
478        let plan=layout_plan(4);
479        // Ring 2 (stream into the last stage) hangs from the root, ring 1 from ring 2, ring 0 from ring 1.
480        assert_eq!(plan.rings,std::vec![Parent::Ring(1),Parent::Ring(2),Parent::Root]);
481        assert_eq!(plan.stages,std::vec![Parent::Ring(0),Parent::Ring(0),Parent::Ring(1),Parent::Ring(2)]);
482    }
483    #[test]
484    fn losing_a_consumer_cancels_only_what_is_upstream_of_it(){
485        for stages in 2..=MAX_STAGES{
486            let plan=layout_plan(stages);
487            for ring in 0..stages-1{
488                for stage in 0..stages{
489                    // Cancelling ring `ring` (consumer stage `ring + 1` lost its reader) stops the
490                    // consumer and every stage before it, never a stage after it.
491                    assert_eq!(below_ring(&plan,plan.stages[stage],ring),stage<=ring+1,"{stages} stages, ring {ring}, stage {stage}");
492                }
493                // No ring below it is a later one: nothing downstream hangs from this ring.
494                for outer in ring+1..stages-1{assert!(!below_ring(&plan,plan.rings[outer],ring))}
495            }
496        }
497    }
498    #[test]
499    fn every_stage_is_below_every_ring_it_touches_and_the_root_is_never_cancelled_by_one(){
500        for stages in 2..=MAX_STAGES{
501            let plan=layout_plan(stages);
502            for stage in 0..stages{
503                // A stage reads ring `stage - 1` and writes ring `stage`: delegation needs it below both.
504                if stage>0{assert!(below_ring(&plan,plan.stages[stage],stage-1))}
505                if stage+1<stages{assert!(below_ring(&plan,plan.stages[stage],stage))}
506            }
507            // Every ring chain ends at the root, so no cancelled ring can reach it.
508            assert_eq!(plan.rings.iter().filter(|p|**p==Parent::Root).count(),1);
509        }
510    }

517    fn status_body(state:u32,reason:u32,exit_code:i32,reserved:u32,sequence:u64)->[u8;24]{
518        let mut body=[0u8;24];let mut w=Writer::new(&mut body);
519        w.u32(state).unwrap();w.u32(reason).unwrap();w.i32(exit_code).unwrap();w.u32(reserved).unwrap();w.u64(sequence).unwrap();
520        body
521    }
522    #[test]
523    fn child_status_decodes_the_24_byte_record(){
524        let body=status_body(3,5,-7,0,0x0102_0304_0506_0708);
525        // Fixed little-endian layout: state, reason, exit code, reserved, sequence.
526        assert_eq!(&body[..12],&[3,0,0,0,5,0,0,0,0xf9,0xff,0xff,0xff]);
527        assert_eq!(&body[16..],&[8,7,6,5,4,3,2,1]);
528        let status=ChildStatus::decode(&body).unwrap();
529        assert_eq!(status,ChildStatus{state:3,reason:5,exit_code:-7,terminal_sequence:0x0102_0304_0506_0708});
530        assert!(status.is_terminal());
531        assert!(!ChildStatus::decode(&status_body(1,0,0,0,0)).unwrap().is_terminal());
532        assert!(!ChildStatus::decode(&status_body(2,0,0,0,0)).unwrap().is_terminal());
533    }
