2//! Shell-owned byte pipelines: children whose fd 0/1/2 are delegated stream
3//! endpoints, never the keyboard owner.
4//!
5//! A [`Pipeline`] is a tree of kernel Operations below one root (a child of the
6//! session Operation). The kernel cancels the Operation that holds a producer's
7//! endpoint, and everything below it, when the last reader of a stream closes, and
8//! an endpoint can only be delegated to a child prepared at or below the Operation
9//! it lives in. The tree is laid out so that exactly the right things stop:
10//!
11//! - the root holds what the shell reads (output and diagnostics) and is never
12//!   cancelled by a stage;
13//! - stage `i` runs in its own stage Operation, so a stage that was stopped never
14//!   takes a neighbour with it;
15//! - the stream between stage `i` and `i + 1` lives in ring Operation `i`, and the
16//!   ring Operations nest outward toward the last stage (`ring n-2` is a child of the
17//!   root, `ring i` a child of `ring i + 1`); stage `i + 1` and every ring and stage
18//!   upstream of it sit below `ring i`. Losing the consumer of a stream therefore
19//!   cancels everything upstream of it (nothing is producing for nobody), never a
20//!   stage downstream, which still reads what was written and ends normally;
21//! - the stream the shell feeds stage 0 through lives in stage 0's own Operation.
22//!
23//! [`Pipeline::spawn`] prepares and commits one child inside its stage Operation
24//! through the same `prepare_child -> delegate -> write_context -> commit`
25//! sequence a foreground child uses, but without `authorize_input`, input-service
26//! receipts or a foreground lease: the child can never own the terminal. Only the
27//! VFS prepared-cwd reservation is bound, exactly as `spawn_foreground` does.
28//!
29//! Ownership: the caller keeps every endpoint copy it passes to `spawn` and
30//! must close it afterwards (the child holds its own delegated copies). EOF at a
31//! reader is the close of the last writer copy, so callers must close their
32//! writer copies once the stages that need them are spawned. A reader passed as
33//! stdin is moved to the child at commit (the kernel consumes the caller's copy).
34use crate::stream::{ok, Endpoint, EndpointRights, Operation, OperationRef, ReceiptRef};
35use abi::cell_session::{
36    ChildRefV1, KernelOp, ObjectRefV1, Reader, Status, VfsOp, Writer, HEADER_LEN,
37};
38use abi::service_binding::ServiceBinding;
39use ostd::child_context::{ChildContext, Grant, GrantSlot};

41/// Maximum immutable-context argv / cwd / environment sizes (ABI limits).
42const ARGV_MAX: usize = 512;
43const CWD_MAX: usize = 4096;
44const ENV_MAX: usize = 4096;
45const ENV_ENTRIES_MAX: usize = 64;
46const PATH_MAX: usize = 4096;
47/// Ring capacity of the empty stdin stream (the kernel minimum).
48const PLACEHOLDER_CAPACITY: u32 = 64;
49/// Ring capacity of every pipeline stream (one transfer unit is 512 bytes).
50const STREAM_CAPACITY: u32 = 4096;
51/// Most stages one pipeline may have (the shell's own pipeline limit).
52pub const MAX_STAGES: usize = 16;

54/// Everything needed to launch one pipeline stage.
55#[derive(Debug, Clone, Copy)]
56pub struct PipeSpec<'a> {
57    pub path: &'a str,
58    pub argv: &'a [&'a str],
59    /// Raw cwd bytes; empty means no cwd grant. Non-empty requires either
60    /// `cwd_source` or a UTF-8 path the VFS can open.
61    pub cwd: &'a [u8],
62    pub environment: &'a [(&'a str, &'a str)],
63    /// Existing VFS directory grant to derive the child's cwd from (never an
64    /// ambient path lookup when present).
65    pub cwd_source: Option<ObjectRefV1>,
66    /// Give the stage a write-only completion-record stream in slot 5 (what a foreground
67    /// child gets from `spawn_foreground_with_completion`): the stream is created in the
68    /// pipeline's root Operation, so a stage stopped by the kernel never takes a record it
69    /// already wrote away, and the reader end stays in the returned [`PipeChild`]
70    /// ([`PipeChild::read_completion`]). The child takes it with [`crate::completion_writer`].
71    pub completion: bool,
72}

74/// Pure pre-effect validation. Returns the exact immutable context length.
75/// Size violations are `QuotaExceeded`, malformed bytes `Malformed`, a path-only
76/// cwd that is not UTF-8 `UnsupportedInput` (same classes as `spawn_foreground`).
77pub(crate) fn validate(spec: &PipeSpec<'_>) -> Result<usize, Status> {
78    if spec.path.is_empty() || spec.path.len() > PATH_MAX || spec.path.as_bytes().contains(&0) {
79        return Err(Status::Malformed);
80    }
81    let argv_len = spec
82        .argv
83        .iter()
84        .try_fold(7usize, |n, a| n.checked_add(a.len() + 1))
85        .ok_or(Status::Malformed)?;
86    let env_len = spec
87        .environment
88        .iter()
89        .try_fold(0usize, |n, (k, v)| n.checked_add(4 + k.len() + v.len()))
90        .ok_or(Status::Malformed)?;
91    if argv_len > ARGV_MAX
92        || spec.cwd.len() > CWD_MAX
93        || env_len > ENV_MAX
94        || spec.environment.len() > ENV_ENTRIES_MAX
95    {
96        return Err(Status::QuotaExceeded);
97    }
98    if spec.argv.iter().any(|a| a.as_bytes().contains(&0)) || spec.cwd.contains(&0) {
99        return Err(Status::Malformed);
100    }
101    if !spec.cwd.is_empty() && spec.cwd_source.is_none() && core::str::from_utf8(spec.cwd).is_err()
102    {
103        return Err(Status::UnsupportedInput);
104    }
105    let grant_count = 3 + usize::from(!spec.cwd.is_empty()) + usize::from(spec.completion);
106    Ok(48 + argv_len + spec.cwd.len() + env_len + grant_count * 40)
107}

109/// Kernel child state: terminal record complete (`QueryChild` `state`).
110const CHILD_TERMINAL: u32 = 3;
111/// `QueryChild` result body after the result header: state, reason, exit code,
112/// reserved, terminal sequence.
113const CHILD_STATUS_LEN: usize = 24;
114/// Upper bound on one [`PipeChild::wait`]. A child that never reaches its terminal
115/// record inside it yields `Status::TimedOut`; it is never reported as exited.
116const WAIT_BUDGET_NS: u64 = 30_000_000_000;
117/// Polls that only yield before the wait starts sleeping one tick per poll.
118const WAIT_YIELDS: u32 = 8;
119
120/// The kernel's terminal record of a committed child, as its preparer sees it.
121#[derive(Debug, Clone, Copy, PartialEq, Eq)]
122pub struct ChildStatus {
123    /// 1 live, 2 retiring, 3 terminal (the only state with a final `exit_code`).
124    pub state: u32,
125    pub reason: u32,
126    pub exit_code: i32,
127    pub terminal_sequence: u64,
128}
129impl ChildStatus {
130    pub fn is_terminal(&self) -> bool {
131        self.state == CHILD_TERMINAL
132    }
133    /// Decode the 24-byte `QueryChild` body (after the 24-byte result header).
134    /// Any other length, a nonzero reserved word or an unknown state is `Malformed`.
135    pub(crate) fn decode(body: &[u8]) -> Result<Self, Status> {
136        if body.len() != CHILD_STATUS_LEN {
137            return Err(Status::Malformed);
138        }
139        let mut r = Reader::new(body);
140        let state = r.u32()?;
141        let reason = r.u32()?;
142        let exit_code = r.i32()?;
143        let reserved = r.u32()?;
144        let terminal_sequence = r.u64()?;
145        r.finish()?;
146        if reserved != 0 || state > CHILD_TERMINAL {
147            return Err(Status::Malformed);
148        }
149        Ok(Self {
150            state,
151            reason,
152            exit_code,
153            terminal_sequence,
154        })
155    }
156}

158/// How [`PipeChild::wait`] spends the gap before poll number `attempt` (0-based).
159#[derive(Debug, Clone, Copy, PartialEq, Eq)]
160enum Pause {
161    Yield,
162    Tick,
163}
164fn pause_for(attempt: u32) -> Pause {
165    if attempt < WAIT_YIELDS {
166        Pause::Yield
167    } else {
168        Pause::Tick
169    }
170}
171/// Absolute deadline for a wait that starts at `now_ns`; `Malformed` on overflow.
172fn wait_deadline(now_ns: u64) -> Result<u64, Status> {
173    now_ns.checked_add(WAIT_BUDGET_NS).ok_or(Status::Malformed)
174}

176/// A committed pipeline stage. Keep it alive until the stage is reaped and drop it
177/// before closing its stage Operation: it keeps a dynamically admitted cwd
178/// directory open, and dropping it lets VFS settle the child's cwd receipt.
179pub struct PipeChild {
180    child: ChildRefV1,
181    operation: OperationRef,
182    _cwd: Option<CwdLease>,
183    completion: Option<Endpoint>,
184}
185impl PipeChild {
186    pub fn child(&self) -> ChildRefV1 {
187        self.child
188    }
189    pub fn root_tid(&self) -> usize {
190        self.child.root_tid as usize
191    }
192    /// One `QueryChild` (non-mutating, idempotent): the kernel's current record of
193    /// this child. Only the preparer may ask, and only until the pipeline Operation
194    /// is closed. Kernel refusals surface as their own status (`WrongOwner`,
195    /// `Stale`); a reply that is not exactly header plus 24 bytes is `Malformed`.
196    pub fn query(&self) -> Result<ChildStatus, Status> {
197        query_child(self.operation, self.child)
198    }
199    /// Wait until the kernel's terminal record of the child exists and return its
200    /// exit code. Polls [`PipeChild::query`] (a few yields, then one-tick sleeps)
201    /// while the child is live or retiring, for at most 30 seconds. Never guesses:
202    /// `Status::TimedOut` when the budget ends first (the child may still run, so
203    /// cancel the pipeline operation if it must stop), `Status::Unsupported` when
204    /// the clock is unavailable, `Stale`/`WrongOwner`/other kernel statuses as
205    /// reported. Transient `WouldBlock`/`TimedOut` answers are polled again.
206    /// Idempotent: the record stays readable until the operation is closed.
207    pub fn wait(&self) -> Result<i32, Status> {
208        self.wait_status().map(|status| status.exit_code)
209    }
210    /// [`PipeChild::wait`] returning the whole terminal record: `reason` is nonzero
211    /// when the kernel stopped the child (a cancelled Operation) instead of the
212    /// child exiting by itself, and `exit_code` is then meaningless.
213    pub fn wait_status(&self) -> Result<ChildStatus, Status> {
214        wait_child_status(self.operation, self.child)
215    }
216    /// Nonblocking drain of the stage's completion stream (see [`PipeSpec::completion`]):
217    /// copies everything currently buffered, up to `out.len()`, and returns the byte count.
218    /// Nothing buffered, writer end-of-stream, and a stage spawned without a completion
219    /// stream all read as `Ok(0)`; only real stream errors fail. Read after the stage is
220    /// reaped and before [`Pipeline::close`], which releases the stream with its Operation.
221    pub fn read_completion(&self, out: &mut [u8]) -> Result<usize, Status> {
222        let Some(reader) = self.completion else {
223            return Ok(0);
224        };
225        let mut total = 0;
226        while total < out.len() {
227            match reader.read(&mut out[total..], 0) {
228                Ok(n) => total += n,
229                Err(Status::WouldBlock | Status::Eof) => break,
230                Err(e) => return Err(e),
231            }
232        }
233        Ok(total)
234    }
235}

237/// One `QueryChild` of `child` under its preparing `operation`.
238pub(crate) fn query_child(
239    operation: OperationRef,
240    child: ChildRefV1,
241) -> Result<ChildStatus, Status> {
242    let mut reply = [0u8; HEADER_LEN + CHILD_STATUS_LEN];
243    let total = crate::stream::call(
244        operation,
245        KernelOp::QueryChild,
246        false,
247        &child.to_bytes(),
248        &mut reply,
249    )?;
250    ChildStatus::decode(reply.get(HEADER_LEN..total).ok_or(Status::Malformed)?)
251}
252
253/// Bounded poll of [`query_child`] until the terminal record exists (see [`PipeChild::wait`]).
254pub(crate) fn wait_child_status(
255    operation: OperationRef,
256    child: ChildRefV1,
257) -> Result<ChildStatus, Status> {
258    let deadline = wait_deadline(ostd::session::monotonic_ns()?)?;
259    let mut attempt = 0u32;
260    loop {
261        match query_child(operation, child) {
262            Ok(status) if status.is_terminal() => return Ok(status),
263            Ok(_) | Err(Status::WouldBlock | Status::TimedOut) => {}
264            Err(e) => return Err(e),
265        }
266        if ostd::session::monotonic_ns()? >= deadline {
267            return Err(Status::TimedOut);
268        }
269        match pause_for(attempt) {
270            Pause::Yield => ostd::syscall::sys_yield(),
271            Pause::Tick => {
272                let _ = ostd::syscall::sys_set_timer(1);
273            }
274        }
275        attempt = attempt.saturating_add(1);
276    }
277}

279/// Where an Operation of a pipeline hangs.
280#[derive(Debug, Clone, Copy, PartialEq, Eq)]
281enum Parent {
282    Root,
283    Ring(usize),
284}
285/// The parent of every ring Operation (index = ring) and stage Operation (index =
286/// stage) of a `stages`-stage pipeline. Ring `i` carries the stream from stage `i`
287/// to stage `i + 1`; see the module documentation for why they nest outward.
288#[derive(Debug, PartialEq, Eq)]
289struct Plan {
290    rings: alloc::vec::Vec<Parent>,
291    stages: alloc::vec::Vec<Parent>,
292}
293fn layout_plan(stages: usize) -> Plan {
294    let rings = (0..stages.saturating_sub(1))
295        .map(|ring| {
296            if ring + 2 == stages {
297                Parent::Root
298            } else {
299                Parent::Ring(ring + 1)
300            }
301        })
302        .collect();
303    let stage_parents = (0..stages)
304        .map(|stage| {
305            if stages == 1 {
306                Parent::Root
307            } else {
308                Parent::Ring(stage.saturating_sub(1))
309            }
310        })
311        .collect();
312    Plan {
313        rings,
314        stages: stage_parents,
315    }
316}

318/// The root Operation of a pipeline and the ring and stage Operations below it.
319pub struct Pipeline {
320    operation: Operation,
321    rings: alloc::vec::Vec<Operation>,
322    stages: alloc::vec::Vec<Operation>,
323    created: alloc::vec::Vec<Operation>,
324}
325impl Pipeline {
326    /// One root Operation per pipeline, a child of the session operation.
327    pub fn create(parent: OperationRef) -> Result<Pipeline, Status> {
328        Ok(Pipeline {
329            operation: Operation::create(Some(parent))?,
330            rings: alloc::vec::Vec::new(),
331            stages: alloc::vec::Vec::new(),
332            created: alloc::vec::Vec::new(),
333        })
334    }
335    pub fn operation(&self) -> OperationRef {
336        self.operation.reference()
337    }
338    /// `(reader, writer)` in the root Operation, the same order as
339    /// `Operation::create_stream`: for the ends the shell reads (output and
340    /// diagnostics). The root is never cancelled by a stage.
341    pub fn stream(&self) -> Result<(Endpoint, Endpoint), Status> {
342        self.operation.create_stream(STREAM_CAPACITY)
343    }
344    /// Open the ring and stage Operations for `stages` stages (at least one, at most
345    /// sixteen). Call once, before any stream or spawn.
346    pub fn layout(&mut self, stages: usize) -> Result<(), Status> {
347        if stages == 0 || stages > MAX_STAGES || !self.stages.is_empty() {
348            return Err(Status::Malformed);
349        }
350        let plan = layout_plan(stages);
351        let mut rings: alloc::vec::Vec<Option<Operation>> = alloc::vec![None;plan.rings.len()];
352        // Outermost ring first: ring `stages - 2` below the root, ring `i` below ring `i + 1`.
353        for ring in (0..plan.rings.len()).rev() {
354            let parent = match plan.rings[ring] {
355                Parent::Root => self.operation,
356                Parent::Ring(outer) => rings[outer].ok_or(Status::Malformed)?,
357            };
358            let operation = Operation::create(Some(parent.reference()))?;
359            self.created.push(operation);
360            rings[ring] = Some(operation);
361        }
362        self.rings = rings.into_iter().flatten().collect();
363        for parent in plan.stages {
364            let parent = match parent {
365                Parent::Root => self.operation,
366                Parent::Ring(ring) => self.rings[ring],
367            };
368            let operation = Operation::create(Some(parent.reference()))?;
369            self.created.push(operation);
370            self.stages.push(operation);
371        }
372        Ok(())
373    }
374    /// `(reader, writer)` of the stream between stage `ring` and stage `ring + 1`,
375    /// created in that ring's Operation.
376    pub fn link(&self, ring: usize) -> Result<(Endpoint, Endpoint), Status> {
377        self.rings
378            .get(ring)
379            .ok_or(Status::Malformed)?
380            .create_stream(STREAM_CAPACITY)
381    }
382    /// `(reader, writer)` created inside stage `stage`'s own Operation: the stream
383    /// the shell feeds that stage through. When the stage stops reading the kernel
384    /// cancels only that stage's Operation.
385    pub fn stage_stream(&self, stage: usize) -> Result<(Endpoint, Endpoint), Status> {
386        self.stages
387            .get(stage)
388            .ok_or(Status::Malformed)?
389            .create_stream(STREAM_CAPACITY)
390    }
391    /// Cancel the whole pipeline (root and everything below it).
392    pub fn cancel(&self, reason: u32) -> Result<(), Status> {
393        self.operation.cancel(reason)
394    }
395    /// Release every Operation, children before parents, then the root (bounded retry
396    /// on `WouldBlock`; correlation lanes are retired on success and on giving up).
397    /// Drop every reaped [`PipeChild`] first: VFS settles a stage's cwd receipt, which
398    /// its Operation's release waits for, only when the owner contacts it.
399    /// The first failure is returned after every release was attempted.
400    pub fn close(self) -> Result<(), Status> {
401        let mut first = Ok(());
402        for operation in self.created.iter().rev() {
403            if let Err(e) = close_bounded(operation.reference()) {
404                if first.is_ok() {
405                    first = Err(e)
406                }
407            }
408        }
409        let root = close_bounded(self.operation.reference());
410        first.and(root)
411    }

413    /// Prepare and commit one child in stage Operation `stage` whose fd0/fd1/fd2 are
414    /// delegated copies of the given endpoints (stdin with reader rights,
415    /// stdout/stderr with writer rights). `None` for `stdin` gives the child an
416    /// empty stream: its first read sees end of file.
417    ///
418    /// The caller keeps its own endpoint copies and must close them; this never
419    /// closes a caller endpoint. Failure before commit publishes nothing (the
420    /// prepared child is aborted and every endpoint made here is closed). A
421    /// failure at or after commit may have published a child, so it cancels the
422    /// stage Operation (reason 5, the `spawn_foreground` reason) and returns the
423    /// error; the Operation itself is released by [`Pipeline::close`].
424    pub fn spawn(
425        &self,
426        stage: usize,
427        spec: &PipeSpec<'_>,
428        stdin: Option<Endpoint>,
429        stdout: Endpoint,
430        stderr: Endpoint,
431    ) -> Result<PipeChild, Status> {
432        let operation = *self.stages.get(stage).ok_or(Status::Malformed)?;
433        let context_len = validate(spec)?;
434        // The completion stream hangs from the root, which no stage can cancel.
435        let completion = if spec.completion {
436            Some(self.operation.create_stream(STREAM_CAPACITY)?)
437        } else {
438            None
439        };
440        let mut committing = false;
441        let result = Self::prepare(
442            operation,
443            spec,
444            context_len,
445            (stdin, stdout, stderr),
446            completion,
447            &mut committing,
448        );
449        if result.is_err() {
450            if let Some((reader, writer)) = completion {
451                let _ = reader.close();
452                let _ = writer.close();
453            }
454        }
455        match result {
456            Err(_) if committing => {
457                let _ = operation.cancel(5);
458            }
459            // Before commit, a stage whose Operation the kernel cancelled lost its
460            // consumer while it was being prepared (the cancellation aborts the
461            // prepared child, so any step after it fails): report that as `Cancelled`.
462            Err(status)
463                if status != Status::Cancelled
464                    && operation.query().is_ok_and(|state| state.state != 0) =>
465            {
466                return Err(Status::Cancelled)
467            }
468            _ => {}
469        }
470        result
471    }

473    fn prepare(
474        operation: Operation,
475        spec: &PipeSpec<'_>,
476        context_len: usize,
477        stdio: (Option<Endpoint>, Endpoint, Endpoint),
478        completion: Option<(Endpoint, Endpoint)>,
479        committing: &mut bool,
480    ) -> Result<PipeChild, Status> {
481        let (stdin, stdout, stderr) = stdio;
482        // Endpoint copies owned by this call, declared before `prepared` so an
483        // unpublished child is aborted first. `empty_reader` (the reader of the empty
484        // stdin stream) outlives the commit; `delegated` is closed only while the
485        // child is unpublished.
486        let mut empty_reader = Closer::new();
487        let mut delegated = Closer::new();
488        let mut prepared = operation.prepare_child_resolving(spec.path, context_len)?;
489        let stdin_source = match stdin {
490            Some(e) => e,
491            None => {
492                // Closing the last reader of a stream stops its producers, so the
493                // reader stays open until the child holds its own copy. The writer is
494                // closed now: the child reads end of file at once.
495                let (reader, writer) = operation.create_stream(PLACEHOLDER_CAPACITY)?;
496                empty_reader.push(reader);
497                writer.close()?;
498                reader
499            }
500        };
501        let stdin_copy = stdin_source.delegate(&prepared, EndpointRights::READ_ONLY)?;
502        delegated.push(stdin_copy);
503        let stdout_copy = stdout.delegate(&prepared, EndpointRights::WRITE_DELEGATE)?;
504        delegated.push(stdout_copy);
505        let stderr_copy = stderr.delegate(&prepared, EndpointRights::WRITE_DELEGATE)?;
506        delegated.push(stderr_copy);
507        // Completion stream: the writer is delegated write-only; the reader stays here.
508        let completion_copy = match completion {
509            Some((_, writer)) => {
510                let copy = writer.delegate(&prepared, EndpointRights::WRITE_ONLY)?;
511                delegated.push(copy);
512                Some(copy)
513            }
514            None => None,
515        };
516
517        let mut cwd_lease = None;
518        let mut reservation = None;
519        let mut child_cwd = None;
520        let vfs_owner = if spec.cwd.is_empty() {
521            None
522        } else {
523            Some(ostd::session::service_binding(abi::syscall::service::VFS)?)
524        };
525        if let Some(owner) = vfs_owner {
526            let lease = cwd_lease.insert(CwdLease::new(owner, operation.reference()));
527            let source = match spec.cwd_source {
528                Some(source) => source,
529                None => {
530                    let path =
531                        core::str::from_utf8(spec.cwd).map_err(|_| Status::UnsupportedInput)?;
532                    // The dynamically opened directory is granted under the caller's
533                    // own principal generation (held by every endpoint it owns).
534                    let generation = stderr.inspect()?.holder.generation;
535                    let mut send = [0u8; 4096];
536                    let mut recv = [0u8; 4096];
537                    let request = wire::ipc::VfsRequest::OpenRootDir { path };
538                    let response = ostd::ipc::service_call_typed_bounded::<
539                        _,
540                        wire::ipc::VfsResponse<'_>,
541                    >(
542                        owner.tid as usize, &request, &mut send, &mut recv, 100
543                    )
544                    .map_err(|_| Status::Indeterminate)?;
545                    let id = match response {
546                        wire::ipc::VfsResponse::DirHandle(handle) => handle.0,
547                        _ => return Err(Status::Denied),
548                    };
549                    // From here the lease closes the handle on every exit.
550                    lease.dir = Some(abi::dir_handles::ViDirHandle(id));
551                    if ostd::session::service_binding(abi::syscall::service::VFS)? != owner {
552                        return Err(Status::Stale);
553                    }
554                    ObjectRefV1 { id, generation }
555                }
556            };
557            let mut body = [0u8; 32];
558            let mut w = Writer::new(&mut body);
559            w.reference(prepared.reference().raw())?;
560            w.reference(source)?;
561            let mut reply = [0u8; 56];
562            let n = vfs(
563                owner,
564                operation.reference(),
565                VfsOp::ReserveCwd,
566                &body,
567                &mut reply,
568            )?;
569            let mut r = Reader::new(&reply[24..n]);
570            reservation = Some(r.reference()?);
571            child_cwd = Some(r.reference()?);
572            r.finish()?;
573        }
574
575        let endpoint_grant = |slot: GrantSlot, copy: Endpoint, rights: EndpointRights| {
576            Grant::endpoint(slot, copy.reference().raw(), rights.bits() as u16)
577        };
578        let mut grants = [
579            endpoint_grant(GrantSlot::Stdin, stdin_copy, EndpointRights::READ_ONLY)?,
580            endpoint_grant(
581                GrantSlot::Stdout,
582                stdout_copy,
583                EndpointRights::WRITE_DELEGATE,
584            )?,
585            endpoint_grant(
586                GrantSlot::Stderr,
587                stderr_copy,
588                EndpointRights::WRITE_DELEGATE,
589            )?,
590            Grant {
591                kind: 1,
592                rights: 0,
593                slot: GrantSlot::Stdin,
594                owner_service: 0,
595                object: ObjectRefV1::ZERO,
596            },
597            Grant {
598                kind: 1,
599                rights: 0,
600                slot: GrantSlot::Stdin,
601                owner_service: 0,
602                object: ObjectRefV1::ZERO,
603            },
604        ];
605        // Ascending slot order: stdio (0..=2), cwd (4), completion (5).
606        let mut count = 3;
607        if let Some(grant) = child_cwd {
608            grants[count] = Grant::cwd(grant)?;
609            count += 1;
610        }
611        if let Some(copy) = completion_copy {
612            grants[count] =
613                endpoint_grant(GrantSlot::Completion, copy, EndpointRights::WRITE_ONLY)?;
614            count += 1;
615        }
616        let context = ChildContext::new(
617            operation.reference().raw(),
618            spec.argv,
619            spec.cwd,
620            spec.environment,
621            &grants[..count],
622        )?;
623        if context.as_bytes().len() != context_len {
624            return Err(Status::Malformed);
625        }
626        prepared.write_context(&context)?;
627        if let (Some(owner), Some(reservation)) = (vfs_owner, reservation) {
628            let mut body = [0u8; 32];
629            let mut w = Writer::new(&mut body);
630            w.reference(reservation)?;
631            w.reference(prepared.reference().raw())?;
632            let mut reply = [0u8; 40];
633            let n = vfs(
634                owner,
635                operation.reference(),
636                VfsOp::FinalizeCwd,
637                &body,
638                &mut reply,
639            )?;
640            let receipt = ReceiptRef::from_raw(ObjectRefV1::from_bytes(&reply[24..n])?)?;
641            if let Some(lease) = cwd_lease.as_mut() {
642                lease.receipt = Some(receipt);
643            }
644            prepared.bind_receipt(2, 4, owner, receipt)?;
645        }
646        // From the commit call on, a child may be published (even when the call
647        // reports an error); the delegated copies then belong to it.
648        *committing = true;
649        delegated.disarm();
650        let child = prepared.commit()?;
651        // The child holds its own reader of the empty stream now.
652        empty_reader.close_all();
653        // The child holds its own writer now: it is the sole completion writer.
654        if let Some((_, writer)) = completion {
655            let _ = writer.close();
656        }
657        Ok(PipeChild {
658            child,
659            operation: operation.reference(),
660            _cwd: cwd_lease,
661            completion: completion.map(|(reader, _)| reader),
662        })
663    }
664}

666/// Closes every pushed endpoint on drop or on demand; `disarm` forgets them.
667struct Closer {
668    list: [Option<Endpoint>; 4],
669    len: usize,
670}
671impl Closer {
672    fn new() -> Self {
673        Self {
674            list: [None; 4],
675            len: 0,
676        }
677    }
678    fn push(&mut self, endpoint: Endpoint) {
679        if self.len < self.list.len() {
680            self.list[self.len] = Some(endpoint);
681            self.len += 1;
682        }
683    }
684    fn close_all(&mut self) {
685        for slot in self.list.iter_mut().take(self.len) {
686            if let Some(endpoint) = slot.take() {
687                let _ = endpoint.close();
688            }
689        }
690        self.len = 0;
691    }
692    fn disarm(&mut self) {
693        self.list = [None; 4];
694        self.len = 0;
695    }
696}
697impl Drop for Closer {
698    fn drop(&mut self) {
699        self.close_all()
700    }
701}

703fn vfs(
704    owner: ServiceBinding,
705    operation: OperationRef,
706    op: VfsOp,
707    body: &[u8],
708    reply: &mut [u8],
709) -> Result<usize, Status> {
710    let header = ostd::session::service_call(
711        owner,
712        abi::syscall::service::VFS,
713        operation.raw(),
714        op as u16,
715        true,
716        body,
717        reply,
718    )?;
719    ok(header.status)?;
720    Ok(header.total_len as usize)
721}

723/// CloseOperation retried while `WouldBlock` (receipt owners settle receipts
724/// asynchronously): eight yields, then one-tick sleeps, 64 attempts at most.
725/// Correlation lanes are retired on success and on giving up.
726fn close_bounded(operation: OperationRef) -> Result<(), Status> {
727    let mut last = Status::WouldBlock;
728    for attempt in 0..64u32 {
729        match Operation::from_ref(operation).close() {
730            Ok(()) => return Ok(()),
731            Err(Status::WouldBlock) => {
732                ostd::syscall::sys_yield();
733                if attempt >= 8 {
734                    let _ = ostd::syscall::sys_set_timer(1);
735                }
736            }
737            Err(e) => {
738                last = e;
739                break;
740            }
741        }
742    }
743    ostd::session::retire_operation(operation.raw());
744    Err(last)
745}

747/// The owner's side of a child's VFS cwd reservation, dropped once the child is
748/// reaped. VFS does not poll a running child that never contacts it: it settles an
749/// owner's reservations when that owner sends a request, and closing the child's
750/// Operation waits for the settled receipt. Dropping the lease is that request: it
751/// closes the dynamically admitted source directory, or else queries the receipt
752/// (`ResolveCwd`, action 0). Cleanup only; VFS still validates every handle operation.
753pub(crate) struct CwdLease {
754    owner: ServiceBinding,
755    operation: OperationRef,
756    pub(crate) dir: Option<abi::dir_handles::ViDirHandle>,
757    pub(crate) receipt: Option<ReceiptRef>,
758}
759impl CwdLease {
760    pub(crate) fn new(owner: ServiceBinding, operation: OperationRef) -> Self {
761        Self {
762            owner,
763            operation,
764            dir: None,
765            receipt: None,
766        }
767    }
768}
769impl Drop for CwdLease {
770    fn drop(&mut self) {
771        if ostd::session::service_binding(abi::syscall::service::VFS).ok() != Some(self.owner) {
772            return;
773        }
774        if let Some(dir) = self.dir {
775            let mut send = [0u8; 4096];
776            let mut recv = [0u8; 4096];
777            let _ = ostd::ipc::service_call_typed_bounded::<_, wire::ipc::VfsResponse<'_>>(
778                self.owner.tid as usize,
779                &wire::ipc::VfsRequest::CloseDir { dir },
780                &mut send,
781                &mut recv,
782                100,
783            );
784        } else if let Some(receipt) = self.receipt {
785            let mut body = [0u8; 24];
786            let mut w = Writer::new(&mut body);
787            if w.reference(receipt.raw())
788                .and_then(|()| w.u32(0))
789                .and_then(|()| w.u32(0))
790                .is_ok()
791            {
792                let _ = ostd::session::service_call(
793                    self.owner,
794                    abi::syscall::service::VFS,
795                    self.operation.raw(),
796                    VfsOp::ResolveCwd as u16,
797                    false,
798                    &body,
799                    &mut [0u8; HEADER_LEN + 24],
800                );
801            }
802        }
803    }
804}
