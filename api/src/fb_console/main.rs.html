1//! fb-console — mirrors kernel user-log output to the HDMI display.
2//!
3//! Creates a full-screen background surface in the compositor and renders
4//! incoming UTF-8 log bytes with antialiased JetBrains Mono at 16px. Wraps at
5//! the right edge and scrolls at the bottom. No UART I/O or VT escape parser.
6//!
7//! Requires `ReadLog` capability (allowlist bit 54).
8

10#![cfg_attr(not(test), no_main)]
11#![forbid(unsafe_code)]

36// Foreground/background colours (BGRA).
37const FG: [u8; 4] = [0xCC, 0xCC, 0xCC, 0xFF]; // light grey
38const BG: [u8; 4] = [0x00, 0x00, 0x00, 0xFF]; // black
39const FONT_PX: f32 = 16.0;
40const FACE: FontFace = FontFace::MonoRegular;

42fn cell_extent(metric: f32) -> usize {
43    let whole = metric as usize;
44    (whole + usize::from(metric > whole as f32)).max(1)
45}

52fn cell_main() {
53    #[cfg(target_os = "none")]
54    init_custom_heap();
55    ostd::io::println("[fb-console] starting");
56    let (width, height) = ostd::syscall::sys_get_resolution();
57    let width = width as usize;
58    let height = height as usize;
59
60    let comp = wait_for_compositor();
61    ostd::io::println("[fb-console] compositor found");
62    let mut surf = match ViSurface::create_background(
63        comp,
64        width as u32,
65        height as u32,
66        PixelFormat::Bgra8888,
67    ) {
68        Ok(s) => s,
69        Err(_) => {
70            sys_exit(1);
71        }
72    };
73    ostd::io::println("[fb-console] background surface created");
74
75    // Clear to background.
76    let px = surf.pixels_mut();
77    for chunk in px.chunks_exact_mut(4) {
78        chunk.copy_from_slice(&BG);
79    }
80    surf.damage_all();
81    ostd::io::println("[fb-console] initial damage submitted");
82
83    let mut console = Console::new(&surf);
84    let mut decoder = Utf8Decoder::new();
85    let mut buf = [0u8; 256];
86    let mut first_log_batch = true;
87
88    loop {
89        let n = sys_read_log(&mut buf);
90        if n == 0 {
91            sys_yield();
92            continue;
93        }
94        if first_log_batch {
95            first_log_batch = false;
96            ostd::io::println("[fb-console] first log batch received");
97        }
98
99        let mut dirty = false;
100        for &byte in &buf[..n] {
101            for c in decoder.push(byte).into_iter().flatten() {
102                dirty |= console.write_char(&mut surf, c);
103            }
104        }
105        if dirty {
106            surf.damage_all();
107        }
108    }
109}

111/// A measured fixed-cell grid. A full final cell wraps only when the next base
112/// character arrives, so a combining mark from a later read still rewrites it.
113struct Console {
114    fonts: TextFonts,
115    width: usize,
116    height: usize,
117    cell_width: usize,
118    cell_height: usize,
119    cols: usize,
120    rows: usize,
121    col: usize,
122    row: usize,
123    // A cell retains at most 32 UTF-8 bytes, including its base and marks.
124    cluster: String,
125}
126
127impl Console {
128    fn new(surf: &ViSurface) -> Self {
129        let mut fonts = TextFonts::new();
130        let cell_width = cell_extent(fonts.measure(FACE, "M", FONT_PX));
131        let cell_height = cell_extent(fonts.line_height(FACE, FONT_PX));
132        let width = surf.width() as usize;
133        let height = surf.height() as usize;
134        Self {
135            fonts,
136            width,
137            height,
138            cell_width,
139            cell_height,
140            cols: (width / cell_width).max(1),
141            rows: (height / cell_height).max(1),
142            col: 0,
143            row: 0,
144            cluster: String::with_capacity(32),
145        }
146    }
147
148    fn write_char(&mut self, surf: &mut ViSurface, c: char) -> bool {
149        match c {
150            '\n' => {
151                self.newline(surf);
152                true
153            }
154            '\r' => {
155                self.cluster.clear();
156                self.col = 0;
157                false
158            }
159            '\t' => {
160                self.cluster.clear();
161                // Expand to the next eight-column tab stop, including wrapping.
162                if self.col == self.cols {
163                    self.newline(surf);
164                }
165                let spaces = 8 - self.col % 8;
166                for _ in 0..spaces {
167                    self.write_char(surf, ' ');
168                }
169                self.cluster.clear();
170                true
171            }
172            '\u{8}' | '\u{7f}' => {
173                self.cluster.clear();
174                // The shell emits BS, space, BS to erase a cell.
175                if self.col > 0 {
176                    self.col -= 1;
177                } else if self.row > 0 {
178                    self.row -= 1;
179                    self.col = self.cols - 1;
180                }
181                false
182            }
183            c if c.is_control() => {
184                self.cluster.clear();
185                false
186            }
187            c if is_combining_mark(c) && !self.cluster.is_empty() => {
188                if !append_combining_mark(&mut self.cluster, c) {
189                    return false;
190                }
191                self.paint_cluster(surf, self.col - 1);
192                true
193            }
194            _ => {
195                if self.col == self.cols {
196                    self.newline(surf);
197                }
198                self.cluster.clear();
199                self.cluster.push(c);
200                self.paint_cluster(surf, self.col);
201                self.col += 1;
202                true
203            }
204        }
205    }
206
207    fn newline(&mut self, surf: &mut ViSurface) {
208        self.cluster.clear();
209        self.col = 0;
210        self.row += 1;
211        if self.row >= self.rows {
212            self.scroll_up(surf);
213            self.row = self.rows - 1;
214        }
215    }
216
217    fn paint_cluster(&mut self, surf: &mut ViSurface, col: usize) {
218        let x = col * self.cell_width;
219        let y = self.row * self.cell_height;
220        let right = (x + self.cell_width).min(self.width);
221        let bottom = (y + self.cell_height).min(self.height);
222        let stride = surf.stride();
223        let px = surf.pixels_mut();
224        // Clear the whole old cell before composing/repainting, not just the
225        // new glyph's ink: accents and CR/backspace overwrites leave no residue.
226        for row in y..bottom {
227            let start = row * stride + x * 4;
228            let end = row * stride + right * 4;
229            for pixel in px[start..end].chunks_exact_mut(4) {
230                pixel.copy_from_slice(&BG);
231            }
232        }
233        // TextFonts composes NFC before painting, including Vietnamese marks
234        // received in a different sys_read_log chunk from their base character.
235        self.fonts.draw_text(
236            px,
237            self.width as u32,
238            self.height as u32,
239            stride,
240            x as f32,
241            y as f32,
242            &self.cluster,
243            FACE,
244            FONT_PX,
245            FG,
246            (x as i32, y as i32, right as i32, bottom as i32),
247        );
248    }
249
250    fn scroll_up(&self, surf: &mut ViSurface) {
251        let stride = surf.stride();
252        let row_height = self.cell_height.min(self.height);
253        let keep_height = (self.rows - 1) * self.cell_height;
254        let keep_bytes = keep_height * stride;
255        let px = surf.pixels_mut();
256        px.copy_within(row_height * stride..row_height * stride + keep_bytes, 0);
257        // Also clear any partial grid row below the last complete cell.
258        for pixel in px[keep_bytes..self.height * stride].chunks_exact_mut(4) {
259            pixel.copy_from_slice(&BG);
260        }
261    }
262}

264fn append_combining_mark(cluster: &mut String, c: char) -> bool {
265    const MAX_CLUSTER_BYTES: usize = 32;
266    if cluster.len() + c.len_utf8() > MAX_CLUSTER_BYTES {
267        return false;
268    }
269    cluster.push(c);
270    true
271}

273/// Streaming UTF-8, retaining incomplete sequences across log reads. Malformed
274/// input replaces each maximal invalid prefix with U+FFFD and reprocesses the
275/// first non-continuation byte, so an interrupted sequence never eats CR/LF.
276struct Utf8Decoder {
277    bytes: [u8; 4],
278    len: usize,
279    expected: usize,
280}
281
282impl Utf8Decoder {
283    const fn new() -> Self {
284        Self { bytes: [0; 4], len: 0, expected: 0 }
285    }
286
287    fn push(&mut self, byte: u8) -> [Option<char>; 2] {
288        let mut replacement = None;
289        if self.len > 0 {
290            let continuation = (0x80..=0xbf).contains(&byte)
291                && (self.len != 1
292                    || match self.bytes[0] {
293                        0xe0 => byte >= 0xa0,
294                        0xed => byte <= 0x9f,
295                        0xf0 => byte >= 0x90,
296                        0xf4 => byte <= 0x8f,
297                        _ => true,
298                    });
299            if continuation {
300                self.bytes[self.len] = byte;
301                self.len += 1;
302                if self.len == self.expected {
303                    let c = core::str::from_utf8(&self.bytes[..self.len])
304                        .ok()
305                        .and_then(|text| text.chars().next())
306                        .unwrap_or('\u{fffd}');
307                    self.len = 0;
308                    return [Some(c), None];
309                }
310                return [None, None];
311            }
312            self.len = 0;
313            replacement = Some('\u{fffd}');
314        }
315
316        let c = match byte {
317            0x00..=0x7f => Some(byte as char),
318            0xc2..=0xdf | 0xe0..=0xef | 0xf0..=0xf4 => {
319                self.bytes[0] = byte;
320                self.len = 1;
321                self.expected = match byte {
322                    0xc2..=0xdf => 2,
323                    0xe0..=0xef => 3,
324                    _ => 4,
325                };
326                None
327            }
328            _ => Some('\u{fffd}'),
329        };
330        [replacement, c]
331    }
332}

335mod tests {
336    use super::Utf8Decoder;

339    fn decode(chunks: &[&[u8]]) -> String {
340        let mut decoder = Utf8Decoder::new();
341        let mut text = String::new();
342        for chunk in chunks {
343            for &byte in *chunk {
344                for c in decoder.push(byte).into_iter().flatten() {
345                    text.push(c);
346                }
347            }
348        }
349        text
350    }

352    #[test]
353    fn combining_cluster_stays_bounded_and_preserves_vietnamese() {
354        let mut cluster = String::from("a");
355        assert!(super::append_combining_mark(&mut cluster, '\u{302}'));
356        assert!(super::append_combining_mark(&mut cluster, '\u{301}'));
357        assert_eq!(ostd::typography::normalized_chars(&cluster).collect::<String>(), "ấ");
358        for _ in 0..10_000 {
359            super::append_combining_mark(&mut cluster, '\u{301}');
360        }
361        assert!(cluster.len() <= 32);
362        let retained = cluster.clone();
363        assert!(!super::append_combining_mark(&mut cluster, '\u{301}'));
364        assert_eq!(cluster, retained);
365    }

367    #[test]
368    fn unicode_survives_every_chunk_boundary() {
369        let text = "Tiếng Việt: á ề a\u{302}\u{301} 🐈\r\n";
370        let bytes = text.as_bytes();
371        for split in 0..=bytes.len() {
372            assert_eq!(decode(&[&bytes[..split], &[], &bytes[split..]]), text);
373        }
374
375        let mut decoder = Utf8Decoder::new();
376        let mut decoded = String::new();
377        for &byte in bytes {
378            for c in decoder.push(byte).into_iter().flatten() {
379                decoded.push(c);
380            }
381        }
382        assert_eq!(decoded, text);
383    }

385    #[test]
386    fn incomplete_sequence_waits_for_later_bytes() {
387        let mut decoder = Utf8Decoder::new();
388        assert_eq!(decoder.push(0xf0), [None, None]);
389        assert_eq!(decoder.push(0x9f), [None, None]);
390        assert_eq!(decoder.push(0x90), [None, None]);
391        assert_eq!(decoder.push(0x88), [Some('🐈'), None]);
392        assert_eq!(decoder.push(b'A'), [None, Some('A')]);
393    }

395    #[test]
396    fn interrupted_prefix_preserves_carriage_return_and_line_feed() {
397        assert_eq!(decode(&[&[0xe1], &[0x80, b'\r'], b"\n"]), "\u{fffd}\r\n");
398        assert_eq!(decode(&[&[0xf0, 0x9f], &[0x90, b'\n']]), "\u{fffd}\n");
399        assert_eq!(decode(&[&[0xe2, b'\r', b'\n']]), "\u{fffd}\r\n");

402    #[test]
403    fn malformed_utf8_replaces_maximal_invalid_prefixes() {
404        let cases: &[(&[u8], &str)] = &[
405            (&[0x80], "\u{fffd}"),
406            (&[0xc0, 0xaf], "\u{fffd}\u{fffd}"),
407            (&[0xe0, 0x80, 0x80], "\u{fffd}\u{fffd}\u{fffd}"),
408            (&[0xed, 0xa0, 0x80], "\u{fffd}\u{fffd}\u{fffd}"),
409            (&[0xf4, 0x90, 0x80, 0x80], "\u{fffd}\u{fffd}\u{fffd}\u{fffd}"),
410            (&[0xf5, 0xff], "\u{fffd}\u{fffd}"),
411            (&[0xe2, 0x82, b'A'], "\u{fffd}A"),
412            (&[0xe2, 0x82, 0xc3, 0xa9], "\u{fffd}é"),
413        ];
414        for &(bytes, expected) in cases {
415            for split in 0..=bytes.len() {
416                assert_eq!(decode(&[&bytes[..split], &bytes[split..]]), expected);
417            }
418        }
