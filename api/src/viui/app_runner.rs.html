2//! Minimal tick-based app runner for ViUI v2.
3//!
4//! # Repaint strategy
5//!
6//! Two independent dirty flags:
7//! - `layout_dirty`: set by consumed events or `mark_dirty()`. Triggers full
8//!   layout + Signal re-subscribe (updates captured bounds) + full repaint.
9//! - `dirty_region`: set by Signal subscriptions. Triggers partial repaint with
10//!   the cached layout — no layout pass, O(m) paint where m = dirty widgets.
11//!
12//! Path A (structural): event consumed → layout → re-subscribe → mark_all
13//! Path B (signal-only): signal fires → mark(widget_bounds) → partial repaint
14//!
15//! # Animation
16//!
17//! Add `Box<dyn Animatable>` via `add_animation()`. On every `tick_with_dt()`
18//! call all active animations are advanced by `dt_ms` before event processing.
19//! Animations that change a `Signal<f32>` trigger Path B automatically.
20//!
21//! # Usage
22//!
23//! ```rust,ignore
24//! let count = Signal::new(0i32);
25//! let label = Label::new(count.map(|n| format!("Count: {n}")));
26//! let cnt2  = count.clone();
27//! let btn   = Button::new("Increment", move || cnt2.update(|n| *n += 1));
28//! let root  = vstack!(label, btn);
29//!
30//! let mut app = ViApp::new(Box::new(root), Box::new(renderer));
31//! loop {
32//!     app.tick(&collect_input_events());
33//!     // or: app.tick_with_dt(&events, elapsed_ms);
34//! }
35//! ```

53// ─── Focus state ─────────────────────────────────────────────────────────────

55/// Tracks Tab-order focus across all focusable widgets in the tree.
56///
57/// `list` is rebuilt after every layout pass (bounds are stale until then).
58/// `idx` is clamped or cleared whenever the list shrinks.
59struct FocusState {
60    /// Screen rects of all focusable widgets, in tree (tab) order.
61    list: alloc::vec::Vec<crate::layout::Rect>,
62    /// Currently focused index into `list`, or `None` when no widget is focused.
63    idx: Option<usize>,
64}
65
66impl FocusState {
67    fn new() -> Self {
68        Self {
69            list: alloc::vec::Vec::new(),
70            idx: None,
71        }
72    }
73
74    /// Advance or reverse focus by one step, wrapping around.
75    fn advance(&mut self, reverse: bool) {
76        if self.list.is_empty() {
77            self.idx = None;
78            return;
79        }
80        self.idx = Some(match self.idx {
81            None => {
82                if reverse {
83                    self.list.len() - 1
84                } else {
85                    0
86                }
87            }
88            Some(i) => {
89                if reverse {
90                    if i == 0 {
91                        self.list.len() - 1
92                    } else {
93                        i - 1
94                    }
95                } else {
96                    (i + 1) % self.list.len()
97                }
98            }
99        });
100    }
101
102    /// Focus the widget containing a pointer press, clearing focus on empty space.
103    fn focus_at(&mut self, point: crate::layout::Point) -> bool {
104        let next = self.list.iter().position(|bounds| bounds.contains(point));
105        let changed = self.idx != next;
106        self.idx = next;
107        changed
108    }
109
110    /// Returns the screen rect of the currently focused widget, if any.
111    fn current_bounds(&self) -> Option<crate::layout::Rect> {
112        self.idx.and_then(|i| self.list.get(i).copied())
113    }
114}

116// ─── Key repeat ──────────────────────────────────────────────────────────────

118/// Delay before the first software repeat fires (milliseconds).
119const KEY_REPEAT_DELAY_MS: u64 = 500;
120
121/// Interval between subsequent software repeat events (milliseconds).
122const KEY_REPEAT_INTERVAL_MS: u64 = 50;
123
124/// Tracks which key is currently held for software key-repeat injection.
125///
126/// # Hardware vs software repeat
127/// The input service already sends `state=Repeated` events for hardware
128/// auto-repeat (see `api::input::KeyState::Repeated`). `input_bridge::parse_input_message`
129/// treats those identically to `Pressed`, so hardware repeat already works without
130/// this struct. `KeyRepeatState` provides *software* repeat for platforms that do NOT
131/// forward hardware repeat (serial consoles, touch keyboards, synthetic input in tests).
132/// When hardware repeat arrives it naturally resets `held_ms` via the Pressed arm, so
133/// the two paths do not fight.
134struct KeyRepeatState {
135    /// The key currently held, or `None` when no key is pressed.
136    held_key: Option<KeyCode>,
137    /// Modifier state captured at the moment the key was pressed.
138    held_mods: Modifiers,
139    /// Accumulated milliseconds since the key was first pressed (reset on new press).
140    held_ms: u64,
141    /// Accumulated milliseconds since the last repeat event fired (reset on each fire).
142    since_last_ms: u64,
143}
144
145impl KeyRepeatState {
146    const fn new() -> Self {
147        Self {
148            held_key: None,
149            held_mods: Modifiers {
150                shift: false,
151                ctrl: false,
152                alt: false,
153            },
154            held_ms: 0,
155            since_last_ms: 0,
156        }
157    }
158}

160/// Tick-based app runner.
161///
162/// Call `tick()` or `tick_with_dt()` from the cell's main loop, passing any
163/// buffered input events and — for animation — the milliseconds elapsed since
164/// the last call.
165pub struct ViApp {
166    root: Box<dyn ViNode>,
167    renderer: Box<dyn ViRenderer>,
168    font_ctx: FontContext,
169    animations: Vec<Box<dyn Animatable>>,
170    dirty_region: DirtyRegion,
171    dirty_handles: Vec<SubscriptionHandle>,
172    layout_dirty: bool,
173    /// Active design-token set. Widgets read colors/spacing from `cx.theme`.
174    theme: Box<dyn ViTheme>,
175    /// Software key-repeat state. See `KeyRepeatState` for rationale.
176    key_repeat: KeyRepeatState,
177    /// Tab-order focus state. Rebuilt after every layout pass.
178    focus: FocusState,
179    /// Last known mouse position, updated on every MouseMove.
180    /// Injected into MousePress/MouseRelease events whose pos is (0,0) because
181    /// the input service encodes button events without position (caller must track).
182    mouse_pos: crate::layout::Point,
183    /// Stack of overlay widgets (dialogs, drop-down popups). Rendered and
184    /// dispatched on top of `root`; topmost entry wins events first.
185    overlays: Vec<OverlayEntry>,
186    /// Deferred overlay mutations queued by widgets during `event()`.
187    /// Drained once per frame after all input events are processed.
188    action_queue: OverlayActionQueue,
189    /// Active toast notifications, in chronological order.
190    toast_entries: VecDeque<crate::node_widgets::toast::ToastEntry>,
191}

193impl ViApp {
194    /// Create a new app with no scalable font (8×8 bitmap fallback).
195    ///
196    /// The first tick always renders a full frame.
197    pub fn new(root: Box<dyn ViNode>, renderer: Box<dyn ViRenderer>) -> Self {
198        let dirty_region: DirtyRegion = Rc::new(RefCell::new(DirtyRect::new()));
199        Self {
200            root,
201            renderer,
202            font_ctx: FontContext::no_font(),
203            animations: Vec::new(),
204            dirty_region,
205            dirty_handles: Vec::new(),
206            layout_dirty: true,
207            theme: Box::new(DarkTheme),
208            key_repeat: KeyRepeatState::new(),
209            focus: FocusState::new(),
210            mouse_pos: crate::layout::Point::new(0.0, 0.0),
211            overlays: Vec::new(),
212            action_queue: new_action_queue(),
213            toast_entries: VecDeque::new(),
214        }
215    }

217    // ── Overlay API ──────────────────────────────────────────────────────────

219    /// Return a clone of the action queue.
220    ///
221    /// Pass this clone to `Dialog` / `DropDown` constructors so they can push
222    /// `OverlayAction` items without holding a borrow on `ViApp`.
223    pub fn action_queue(&self) -> OverlayActionQueue {
224        Rc::clone(&self.action_queue)
225    }
226
227    /// Push a widget as a blocking or non-blocking overlay.
228    ///
229    /// `blocking = true` — modal: all input is consumed by the overlay, the
230    /// root does not see events while it is on screen (e.g. Dialog).
231    /// `blocking = false` — non-modal: input passes through when not consumed
232    /// by the overlay (e.g. DropDown popup).
233    pub fn push_overlay(&mut self, widget: Box<dyn ViNode>, blocking: bool) {
234        self.overlays.push(OverlayEntry {
235            widget,
236            blocking,
237            dismiss_outside: !blocking,
238            anchor_bounds: None,
239        });
240        self.layout_dirty = true;
241    }
242
243    /// Pop the topmost overlay. No-op when the overlay stack is empty.
244    pub fn pop_overlay(&mut self) {
245        self.overlays.pop();
246        self.layout_dirty = true;
247    }
248
249    /// Show a transient toast notification.
250    ///
251    /// Toasts are rendered above all content and auto-dismiss after
252    /// `config.duration_ms` milliseconds (`0` = never auto-dismiss).
253    pub fn show_toast(&mut self, config: crate::node_widgets::toast::ToastConfig) {
254        let widget = crate::node_widgets::toast::Toast::new(&config);
255        self.toast_entries
256            .push_back(crate::node_widgets::toast::ToastEntry {
257                config,
258                elapsed_ms: 0,
259                widget,
260            });
261        self.layout_dirty = true;
262    }

264    /// Override the default 8×8 bitmap font with a scalable TTF font.
265    ///
266    /// `font_bytes`: raw TTF/OTF data (use `include_bytes!` for embedded fonts).
267    /// `size_px`: default text height in pixels.
268    /// Returns `self` unchanged when the font bytes are invalid.
269    pub fn with_font(mut self, font_bytes: &[u8], size_px: f32) -> Self {
270        if let Some(ctx) = FontContext::with_font(font_bytes, size_px) {
271            self.font_ctx = ctx;
272        }
273        self
274    }

276    /// Switch the active theme. The next frame will render with the new tokens.
277    pub fn set_theme<T: ViTheme + 'static>(&mut self, theme: T) {
278        self.theme = Box::new(theme);
279    }

281    /// Register an animation that is advanced every `tick_with_dt()` call.
282    ///
283    /// Animations typically wrap an `AnimatedSignal<f32>` and drive a widget
284    /// signal — no widget code changes are required.
285    pub fn add_animation(&mut self, anim: Box<dyn Animatable>) {
286        self.animations.push(anim);
287    }
288
289    /// Process events and render if dirty. Returns `true` if a frame was rendered.
290    ///
291    /// Equivalent to `tick_with_dt(events, 0)` — animations are not advanced.
292    pub fn tick(&mut self, events: &[Event]) -> bool {
293        self.tick_with_dt(events, 0)
294    }
295
296    /// Process events, advance animations by `dt_ms` milliseconds, and render
297    /// if dirty. Returns `true` if a frame was rendered.
298    ///
299    /// `dt_ms`: milliseconds elapsed since the last call. Pass `0` to skip
300    /// animation advancement (useful for initial render or event-only ticks).
301    pub fn tick_with_dt(&mut self, events: &[Event], dt_ms: u32) -> bool {
302        // ── Advance animations ────────────────────────────────────────────────
303        // Animations update Signal<f32> values which trigger dirty-region marks
304        // via their subscriptions — no explicit dirty flag needed here.
305        if dt_ms > 0 {
306            for anim in &mut self.animations {
307                anim.tick(dt_ms);
308            }
309        }
310
311        // ── Drain action queue from previous frame ────────────────────────────
312        // Widgets may have pushed actions during the last tick's event pass.
313        // Drain before the new event pass so actions are visible immediately.
314        self.drain_action_queue();
315
316        // ── Process input events ──────────────────────────────────────────────
317        for e in events {
318            // Tab focus cycling — consumed by the focus system, not dispatched to root.
319            if let Event::KeyPress {
320                key: KeyCode::Tab,
321                modifiers,
322            } = e
323            {
324                self.focus.advance(modifiers.shift);
325                self.layout_dirty = true;
326                continue;
327            }
328
329            // Enter activation — route directly to the focused widget via activate_at().
330            // When a blocking overlay is active, activate_at on the overlay is more
331            // appropriate; skip root activation in that case.
332            if let Event::KeyPress {
333                key: KeyCode::Enter,
334                ..
335            } = e
336            {
337                let top_blocking = self.overlays.last().map(|o| o.blocking).unwrap_or(false);
338                if !top_blocking {
339                    if let Some(b) = self.focus.current_bounds() {
340                        if self.root.activate_at(b) {
341                            self.layout_dirty = true;
342                        }
343                    }
344                }
345                continue;
346            }
347
348            // Track mouse position for button-event injection.
349            // parse_mouse_button in input_bridge sets pos=(0,0) because the input
350            // service encodes button events without coordinates — we inject the last
351            // known position here so hit-testing in Button/Slider works correctly.
352            if let Event::MouseMove { pos } = e {
353                self.mouse_pos = *pos;
354            }
355
356            // Patch mouse events that arrive with pos=(0,0) — substitute the last
357            // tracked position so widget hit-testing is correct.
358            // parse_mouse_button and parse_mouse_scroll in input_bridge set pos=(0,0)
359            // because the input service encodes those events without coordinates.
360            let patched;
361            let ev: &Event = match e {
362                Event::MousePress { pos, button } if pos.x == 0.0 && pos.y == 0.0 => {
363                    patched = Event::MousePress {
364                        pos: self.mouse_pos,
365                        button: *button,
366                    };
367                    &patched
368                }
369                Event::MouseRelease { pos, button } if pos.x == 0.0 && pos.y == 0.0 => {
370                    patched = Event::MouseRelease {
371                        pos: self.mouse_pos,
372                        button: *button,
373                    };
374                    &patched
375                }
376                Event::Scroll { pos, delta_y } if pos.x == 0.0 && pos.y == 0.0 => {
377                    patched = Event::Scroll {
378                        pos: self.mouse_pos,
379                        delta_y: *delta_y,
380                    };
381                    &patched
382                }
383                other => other,
384            };
385
386            if let Event::MousePress {
387                pos,
388                button: crate::event::MouseButton::Left,
389            } = ev
390            {
391                if self.focus.focus_at(*pos) {
392                    self.layout_dirty = true;
393                }
394            }
395
396            // Track held key for software repeat injection (see KeyRepeatState).
397            match ev {
398                Event::KeyPress { key, modifiers } => {
399                    self.key_repeat.held_key = Some(*key);
400                    self.key_repeat.held_mods = *modifiers;
401                    self.key_repeat.held_ms = 0;
402                    self.key_repeat.since_last_ms = 0;
403                }
404                Event::KeyRelease { .. } => {
405                    self.key_repeat.held_key = None;
406                }
407                _ => {}
408            }
409
410            // Route event: topmost overlay wins, then falls through if non-blocking.
411            // Snapshot overlay metadata before the mutable borrow for event().
412            let overlay_meta = self
413                .overlays
414                .last()
415                .map(|t| (t.blocking, t.dismiss_outside, t.widget.bounds()));
416
417            let consumed = if let Some(top) = self.overlays.last_mut() {
418                let top_blocking = top.blocking;
419                let overlay_consumed = top.widget.event(ev);
420                if top_blocking {
421                    // Blocking overlay: never dispatch to root.
422                    overlay_consumed
423                } else if !overlay_consumed {
424                    // Non-blocking overlay did not consume — dispatch to root.
425                    self.root.event(ev)
426                } else {
427                    true
428                }
429            } else {
430                self.root.event(ev)
431            };
432
433            // Enforce dismiss_outside: if the topmost non-blocking overlay has
434            // dismiss_outside=true and a press/touch landed outside its bounds,
435            // pop it now. The popup's own event() intentionally returns false for
436            // outside clicks (it cannot pop itself during the borrow), so this is
437            // the only place the dismiss is enforced.
438            if let Some((blocking, dismiss_outside, overlay_bounds)) = overlay_meta {
439                if !blocking && dismiss_outside {
440                    let outside_press = match ev {
441                        Event::MousePress { pos, .. } => !overlay_bounds.contains(*pos),
442                        Event::TouchBegin { pos, .. } => !overlay_bounds.contains(*pos),
443                        _ => false,
444                    };
445                    if outside_press {
446                        self.action_queue.borrow_mut().push(OverlayAction::Pop);
447                    }
448                }
449            }
450
451            if consumed {
452                self.layout_dirty = true;
453            }
454        }
455
456        // ── Software key-repeat injection ─────────────────────────────────────
457        // Only runs when time is advancing (dt_ms > 0) and a key is held.
458        // Platforms that send hardware repeat events (api::input::KeyState::Repeated)
459        // already inject repeats via input_bridge, so this path fires only when
460        // held_ms crosses the thresholds without hardware-repeat interruption.
461        if dt_ms > 0 {
462            if let Some(key) = self.key_repeat.held_key {
463                self.key_repeat.held_ms += dt_ms as u64;
464                self.key_repeat.since_last_ms += dt_ms as u64;
465
466                if self.key_repeat.held_ms >= KEY_REPEAT_DELAY_MS
467                    && self.key_repeat.since_last_ms >= KEY_REPEAT_INTERVAL_MS
468                {
469                    self.key_repeat.since_last_ms = 0;
470                    let repeat_event = Event::KeyPress {
471                        key,
472                        modifiers: self.key_repeat.held_mods,
473                    };
474                    let consumed = if let Some(top) = self.overlays.last_mut() {
475                        let top_blocking = top.blocking;
476                        let overlay_consumed = top.widget.event(&repeat_event);
477                        if top_blocking {
478                            overlay_consumed
479                        } else if !overlay_consumed {
480                            self.root.event(&repeat_event)
481                        } else {
482                            true
483                        }
484                    } else {
485                        self.root.event(&repeat_event)
486                    };
487                    if consumed {
488                        self.layout_dirty = true;
489                    }
490                }
491            }
492        }
493
494        // ── Drain action queue (widgets may have queued actions this frame) ────
495        self.drain_action_queue();
496
497        // ── Advance toast timers ──────────────────────────────────────────────
498        if dt_ms > 0 && !self.toast_entries.is_empty() {
499            let prev_len = self.toast_entries.len();
500            for entry in &mut self.toast_entries {
501                entry.elapsed_ms = entry.elapsed_ms.saturating_add(dt_ms);
502            }
503            self.toast_entries.retain(|entry| {
504                entry.config.duration_ms == 0 || entry.elapsed_ms < entry.config.duration_ms
505            });
506            if self.toast_entries.len() != prev_len {
507                self.layout_dirty = true;
508            }
509        }
510
511        // ── Path A: structural change → full layout + re-subscribe ────────────
512        if self.layout_dirty {
513            self.layout_dirty = false;
514            let (w, h) = self.renderer.size();
515            let screen = Constraints::root(Size::new(w as f32, h as f32));
516
517            // Layout root widget.
518            self.root.layout(screen);
519
520            // Layout all overlay widgets (they need fresh screen constraints).
521            for entry in &mut self.overlays {
522                entry.widget.layout(screen);
523            }
524
525            // Layout toast widgets.
526            for entry in &mut self.toast_entries {
527                entry.widget.layout(screen);
528            }
529
530            // Rebuild focus list.  When a blocking overlay is active, focus is
531            // confined to that overlay's focusable bounds.
532            if self.overlays.last().map(|e| e.blocking).unwrap_or(false) {
533                self.focus.list = self
534                    .overlays
535                    .last_mut()
536                    .map(|e| e.widget.collect_focusable_bounds())
537                    .unwrap_or_default();
538            } else {
539                self.focus.list = self.root.collect_focusable_bounds();
540            }
541            // Clamp idx if the list shrank (e.g. widget removed).
542            if let Some(i) = self.focus.idx {
543                if i >= self.focus.list.len() {
544                    self.focus.idx = None;
545                }
546            }
547
548            // Rebuild signal subscriptions for root + overlays.
549            self.dirty_handles = self
550                .root
551                .collect_dirty_handles(Rc::clone(&self.dirty_region));
552            for entry in &mut self.overlays {
553                let handles = entry
554                    .widget
555                    .collect_dirty_handles(Rc::clone(&self.dirty_region));
556                self.dirty_handles.extend(handles);
557            }
558
559            self.dirty_region.borrow_mut().mark_all(w as f32, h as f32);
560        }
561
562        // ── Path B: render accumulated dirty region ───────────────────────────
563        let damage = self.dirty_region.borrow_mut().take();
564        if damage.is_none() {
565            return false;
566        }
567
568        // Snapshot focus bounds before the borrow split — Option<Rect> is Copy.
569        let focus_bounds = self.focus.current_bounds();
570
571        // Split borrows: renderer, font_ctx, theme, root, overlays, and
572        // toast_entries are all disjoint struct fields.
573        let renderer = &mut self.renderer;
574        let font_ctx = &mut self.font_ctx;
575        let theme = &*self.theme;
576        let root = &self.root;
577        let overlays = &self.overlays;
578        let toast_entries = &self.toast_entries;
579
580        renderer.render(damage, &mut |canvas| {
581            let mut cx = RenderCtx {
582                canvas,
583                font: font_ctx,
584                theme,
585            };
586            root.paint(&mut cx);
587
588            // Paint overlays bottom-to-top.
589            for entry in overlays.iter() {
590                if entry.blocking {
591                    // Dim the background with a semi-transparent scrim.
592                    let (sw, sh) = (cx.canvas.width() as f32, cx.canvas.height() as f32);
593                    cx.canvas.fill_rect(
594                        crate::layout::Rect::new(0.0, 0.0, sw, sh),
595                        crate::canvas::Color::bgra(0, 0, 0, 160),
596                    );
597                }
598                entry.widget.paint(&mut cx);
599            }
600
601            // Paint toasts above all overlays.
602            for entry in toast_entries.iter() {
603                entry.widget.paint(&mut cx);
604            }
605
606            // Post-paint focus ring — drawn on top of all widget content.
607            if let Some(b) = focus_bounds {
608                cx.canvas.draw_rect_border(b, cx.theme.accent(), 2.0);
609            }
610        });
611        true
612    }
613
614    // ── Internal helpers ─────────────────────────────────────────────────────
615
616    /// Drain the action queue and apply each `OverlayAction` to `self`.
617    ///
618    /// Called twice per frame: once before event processing (to apply actions
619    /// from the previous frame) and once after (to apply actions queued this
620    /// frame). The double drain ensures single-frame round-trip latency.
621    fn drain_action_queue(&mut self) {
622        // Collect into a local vec to release the borrow on action_queue before
623        // mutating `self.overlays` / `self.toast_entries`.
624        let actions: Vec<OverlayAction> = self.action_queue.borrow_mut().drain(..).collect();
625        for action in actions {
626            match action {
627                OverlayAction::Push(entry) => {
628                    self.overlays.push(entry);
629                    self.layout_dirty = true;
630                }
631                OverlayAction::Pop => {
632                    self.overlays.pop();
633                    self.layout_dirty = true;
634                }
635                OverlayAction::PopAll => {
636                    self.overlays.clear();
637                    self.layout_dirty = true;
638                }
639                OverlayAction::ShowToast(config) => {
640                    let widget = crate::node_widgets::toast::Toast::new(&config);
641                    self.toast_entries
642                        .push_back(crate::node_widgets::toast::ToastEntry {
643                            config,
644                            elapsed_ms: 0,
645                            widget,
646                        });
647                    self.layout_dirty = true;
648                }
649            }
650        }
651    }
652
653    /// Force a full repaint on the next `tick()` call.
654    pub fn mark_dirty(&mut self) {
655        self.layout_dirty = true;
656    }

675    struct ProbeStats {
676        layouts: Cell<u32>,
677        paints: Cell<u32>,
678        events: Cell<u32>,
679    }
680
681    struct ProbeNode {
682        signal: Signal<u8>,
683        stats: Rc<ProbeStats>,
684        bounds: Rect,
685        consumes_events: bool,
686    }
687
688    impl ProbeNode {
689        fn new(signal: Signal<u8>, stats: Rc<ProbeStats>, consumes_events: bool) -> Self {
690            Self {
691                signal,
692                stats,
693                bounds: Rect::ZERO,
694                consumes_events,
695            }
696        }
697    }
698
699    impl ViNode for ProbeNode {
700        fn layout(&mut self, constraints: Constraints) -> Size {
701            self.stats.layouts.set(self.stats.layouts.get() + 1);
702            self.bounds = Rect::from_origin_size(constraints.origin, constraints.max);
703            constraints.max
704        }
705
706        fn bounds(&self) -> Rect {
707            self.bounds
708        }
709
710        fn paint(&self, _cx: &mut RenderCtx<'_>) {
711            self.stats.paints.set(self.stats.paints.get() + 1);
712        }
713
714        fn event(&mut self, _event: &Event) -> bool {
715            self.stats.events.set(self.stats.events.get() + 1);
716            self.consumes_events
717        }
718
719        fn collect_dirty_handles(&mut self, region: DirtyRegion) -> Vec<SubscriptionHandle> {
720            let bounds = self.bounds;
721            alloc::vec![self
722                .signal
723                .subscribe(move || region.borrow_mut().mark(bounds))]
724        }
725    }
726
727    fn stats() -> Rc<ProbeStats> {
728        Rc::new(ProbeStats {
729            layouts: Cell::new(0),
730            paints: Cell::new(0),
731            events: Cell::new(0),
732        })
733    }
734
735    fn app_with_probe(signal: Signal<u8>, stats: Rc<ProbeStats>, consumes_events: bool) -> ViApp {
736        ViApp::new(
737            Box::new(ProbeNode::new(signal, stats, consumes_events)),
738            Box::new(ViSurfaceRenderer::new(32, 24)),
739        )
740    }

742    #[test]
743    fn pointer_press_focuses_the_widget_under_the_cursor() {
744        let mut focus = FocusState::new();
745        focus.list = alloc::vec![
746            Rect::new(0.0, 0.0, 10.0, 10.0),
747            Rect::new(20.0, 0.0, 10.0, 10.0),
748        ];
749
750        assert!(focus.focus_at(Point::new(25.0, 5.0)));
751        assert_eq!(focus.current_bounds(), Some(focus.list[1]));
752        assert!(focus.focus_at(Point::new(15.0, 5.0)));
753        assert_eq!(focus.current_bounds(), None);
754    }
755    #[test]
756    fn initial_tick_renders_once_and_idle_tick_skips_work() {
757        let signal = Signal::new(0);
758        let stats = stats();
759        let mut app = app_with_probe(signal, Rc::clone(&stats), false);
760
761        assert!(app.tick(&[]));
762        assert_eq!(stats.layouts.get(), 1);
763        assert_eq!(stats.paints.get(), 1);
764
765        assert!(!app.tick(&[]));
766        assert_eq!(stats.layouts.get(), 1);
767        assert_eq!(stats.paints.get(), 1);
768    }

770    #[test]
771    fn consumed_event_relayouts_and_repaints_the_headless_app() {
772        let signal = Signal::new(0);
773        let stats = stats();
774        let mut app = app_with_probe(signal, Rc::clone(&stats), true);
775        app.tick(&[]);
776
777        assert!(app.tick(&[Event::Char('x')]));
778        assert_eq!(stats.events.get(), 1);
779        assert_eq!(stats.layouts.get(), 2);
780        assert_eq!(stats.paints.get(), 2);
781    }
