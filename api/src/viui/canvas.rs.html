11/// Packed BGRA colour (matches VirtIO GPU wire format and compositor blending).
12#[derive(Copy, Clone, Default, Debug, PartialEq, Eq)]
13pub struct Color(pub u32);
14
15impl Color {
16    pub const TRANSPARENT: Self = Self(0x0000_0000);
17    pub const BLACK: Self = Self(0xFF00_0000);
18    pub const WHITE: Self = Self(0xFFFF_FFFF);
19    pub const RED: Self = Self(0xFF00_00FF);
20    pub const GREEN: Self = Self(0xFF00_FF00);
21    pub const BLUE: Self = Self(0xFFFF_0000);
22    pub const GRAY: Self = Self(0xFF80_8080);
23    pub const DARK_GRAY: Self = Self(0xFF40_4040);
24    pub const YELLOW: Self = Self(0xFF00_FFFF); // BGRA: B=0 G=FF R=FF A=FF
25    pub const CYAN: Self = Self(0xFFFF_FF00); // BGRA: B=FF G=FF R=0 A=FF
26    pub const MAGENTA: Self = Self(0xFFFF_00FF); // BGRA: B=FF G=0 R=FF A=FF
27
28    /// Construct from BGRA channels.
29    pub const fn bgra(b: u8, g: u8, r: u8, a: u8) -> Self {
30        Self((b as u32) | ((g as u32) << 8) | ((r as u32) << 16) | ((a as u32) << 24))
31    }
32
33    /// Construct from RGB channels (fully opaque).
34    pub const fn rgb(r: u8, g: u8, b: u8) -> Self {
35        Self::bgra(b, g, r, 0xFF)
36    }
37    /// Construct from RGBA channels.
38    pub const fn rgba(r: u8, g: u8, b: u8, a: u8) -> Self {
39        Self::bgra(b, g, r, a)
40    }
41
42    /// Return packed BGRA bytes in little-endian order.
43    pub const fn as_bgra_bytes(self) -> [u8; 4] {
44        [
45            (self.0 & 0xFF) as u8,
46            ((self.0 >> 8) & 0xFF) as u8,
47            ((self.0 >> 16) & 0xFF) as u8,
48            ((self.0 >> 24) & 0xFF) as u8,
49        ]
50    }
51
52    #[inline]
53    pub fn b(self) -> u8 {
54        (self.0 & 0xFF) as u8
55    }
56    #[inline]
57    pub fn g(self) -> u8 {
58        (self.0 >> 8 & 0xFF) as u8
59    }
60    #[inline]
61    pub fn r(self) -> u8 {
62        (self.0 >> 16 & 0xFF) as u8
63    }
64    #[inline]
65    pub fn a(self) -> u8 {
66        (self.0 >> 24 & 0xFF) as u8
67    }
68
69    pub fn with_alpha(self, a: u8) -> Self {
70        Self((self.0 & 0x00FF_FFFF) | ((a as u32) << 24))
71    }
72
73    /// Alpha-blend `src` over `dst` (both BGRA). Uses `src.a` as blend factor.
74    #[inline]
75    pub fn blend_over(src: Color, dst: Color) -> Color {
76        let sa = src.a() as u32;
77        if sa == 0 {
78            return dst;
79        }
80        if sa == 255 {
81            return src;
82        }
83        let inv = 255 - sa;
84        Color::bgra(
85            ((src.b() as u32 * sa + dst.b() as u32 * inv) / 255) as u8,
86            ((src.g() as u32 * sa + dst.g() as u32 * inv) / 255) as u8,
87            ((src.r() as u32 * sa + dst.r() as u32 * inv) / 255) as u8,
88            255,
89        )
90    }
91}

122/// Abstract pixel drawing surface.
123///
124/// All coordinates are screen-space (absolute pixels).
125/// Clip stack: operations outside the current clip rect are silently discarded.
126pub trait ViCanvas {
127    /// Fill `rect` with a solid colour.
128    fn fill_rect(&mut self, rect: Rect, color: Color);
129
130    /// Draw a line from `a` to `b` (Bresenham integer algorithm in P02).
131    fn draw_line(&mut self, a: Point, b: Point, color: Color);
132
133    /// Draw text starting at `pos` (top-left). `style.size_px == 0` uses bitmap 8×8.
134    fn draw_text(&mut self, pos: Point, text: &str, style: TextStyle);
135
136    /// Draw scalable text at `pos` (top-left) using a GlyphAtlas.
137    ///
138    /// Backends must implement scalable text explicitly; command-only GPU
139    /// canvases record text but do not promise glyph rendering.
140    fn draw_text_scaled(
141        &mut self,
142        pos: Point,
143        text: &str,
144        px: f32,
145        color: Color,
146        atlas: &mut ostd::font_atlas::GlyphAtlas,
147    );
148
149    /// Blit raw BGRA pixels from `pixels` into `dest`.
150    ///
151    /// `src_stride`: bytes per row in the source buffer.
152    fn draw_image(&mut self, dest: Rect, pixels: &[u8], src_stride: u32);
153
154    /// Push a clip rectangle (intersects with any existing clip).
155    fn clip_push(&mut self, rect: Rect);
156
157    /// Pop the most recently pushed clip rectangle.
158    fn clip_pop(&mut self);
159
160    /// Current clip rectangle, or `None` if no clip is active.
161    fn clip_rect(&self) -> Option<Rect>;
162
163    /// Raw pixel buffer — used by `PaintCx::paint_text()` for glyph atlas blitting.
164    fn pixels_mut(&mut self) -> &mut [u8];
165
166    /// Bytes per row in the pixel buffer.
167    fn stride(&self) -> u32;
168
169    fn width(&self) -> u32;
170    fn height(&self) -> u32;
171
172    /// Draw a rectangular border using 4 `fill_rect` calls.
173    ///
174    /// Draws `thickness`-wide edges on all four sides of `rect`.
175    /// The default implementation works for all `ViCanvas` implementors.
176    fn draw_rect_border(&mut self, rect: Rect, color: Color, thickness: f32) {
177        let t = thickness;
178        // top edge
179        self.fill_rect(
180            Rect {
181                x: rect.x,
182                y: rect.y,
183                w: rect.w,
184                h: t,
185            },
186            color,
187        );
188        // bottom edge
189        self.fill_rect(
190            Rect {
191                x: rect.x,
192                y: rect.y + rect.h - t,
193                w: rect.w,
194                h: t,
195            },
196            color,
197        );
198        // left edge
199        self.fill_rect(
200            Rect {
201                x: rect.x,
202                y: rect.y,
203                w: t,
204                h: rect.h,
205            },
206            color,
207        );
208        // right edge
209        self.fill_rect(
210            Rect {
211                x: rect.x + rect.w - t,
212                y: rect.y,
213                w: t,
214                h: rect.h,
215            },
216            color,
217        );
218    }
219
220    /// Fill `rect` with every corner rounded by `radius` px.
221    ///
222    /// The default draws square corners; rasterizing backends override it.
223    fn fill_round_rect(&mut self, rect: Rect, radius: u32, color: Color) {
224        let _ = radius;
225        self.fill_rect(rect, color);
226    }
227
228    /// 1px outline covering exactly the boundary pixels of
229    /// [`Self::fill_round_rect`] for the same `rect` and `radius`.
230    ///
231    /// The default draws square corners; rasterizing backends override it.
232    fn stroke_round_rect(&mut self, rect: Rect, radius: u32, color: Color) {
233        let _ = radius;
234        self.draw_rect_border(rect, color, 1.0);
235    }
236}

240const CLIP_STACK_DEPTH: usize = 16;
241
242/// Software rasterizer that writes directly into a BGRA8888 framebuffer slice.
243///
244/// `'fb` ties the canvas to the caller's borrow of the compositor surface buffer.
245/// `stride` must be `>= width * 4`; mismatches cause silent pixel corruption.
246pub struct FramebufferCanvas<'fb> {
247    pixels: &'fb mut [u8],
248    stride: u32,
249    width: u32,
250    height: u32,
251    clip_stack: [Rect; CLIP_STACK_DEPTH],
252    clip_depth: usize,
253    // Integer (x0,y0,x1,y1) shadow of clip_stack — avoids f32 casts in put_pixel hot path.
254    clip_stack_i: [(i32, i32, i32, i32); CLIP_STACK_DEPTH],
255}
256
257impl<'fb> FramebufferCanvas<'fb> {
258    pub fn new(pixels: &'fb mut [u8], stride: u32, width: u32, height: u32) -> Self {
259        let full = Rect {
260            x: 0.0,
261            y: 0.0,
262            w: width as f32,
263            h: height as f32,
264        };
265        let mut clip_stack = [Rect::default(); CLIP_STACK_DEPTH];
266        clip_stack[0] = full;
267        let mut clip_stack_i = [(0i32, 0i32, 0i32, 0i32); CLIP_STACK_DEPTH];
268        clip_stack_i[0] = (0, 0, width as i32, height as i32);
269        Self {
270            pixels,
271            stride,
272            width,
273            height,
274            clip_stack,
275            clip_depth: 0,
276            clip_stack_i,
277        }
278    }
279
280    #[inline]
281    fn active_clip(&self) -> Rect {
282        self.clip_stack[self.clip_depth]
283    }
284
285    /// Write one pixel with src-over alpha blending.
286    #[inline]
287    fn put_pixel(&mut self, x: i32, y: i32, color: Color) {
288        if x < 0 || y < 0 {
289            return;
290        }
291        let (px, py) = (x as u32, y as u32);
292        if px >= self.width || py >= self.height {
293            return;
294        }
295        let (cx0, cy0, cx1, cy1) = self.clip_stack_i[self.clip_depth];
296        if (px as i32) < cx0 || (px as i32) >= cx1 {
297            return;
298        }
299        if (py as i32) < cy0 || (py as i32) >= cy1 {
300            return;
301        }
302        let off = (py * self.stride + px * 4) as usize;
303        if off + 3 >= self.pixels.len() {
304            return;
305        }
306        // u32 load: LLVM emits single LDR on aligned targets
307        let dst = Color(u32::from_le_bytes([
308            self.pixels[off],
309            self.pixels[off + 1],
310            self.pixels[off + 2],
311            self.pixels[off + 3],
312        ]));
313        let out = Color::blend_over(color, dst);
314        // u32 store: LLVM emits single STR on aligned targets
315        self.pixels[off..off + 4].copy_from_slice(&out.0.to_le_bytes());
316    }
317
318    /// `linear::Canvas` over this framebuffer, clipped to the active clip.
319    fn linear(&mut self) -> linear::Canvas<'_> {
320        let (x0, y0, x1, y1) = self.clip_stack_i[self.clip_depth];
321        linear::Canvas::new(self.pixels, self.width, self.height, self.stride as usize).clip_to(
322            x0,
323            y0,
324            (x1 - x0).max(0) as u32,
325            (y1 - y0).max(0) as u32,
326        )
327    }
328
329    /// Snap `rect` to whole pixels the way `fill_rect` does.
330    fn round_shape(rect: Rect, radius: u32) -> Option<RoundRect> {
331        let (x0, y0) = (rect.x as i32, rect.y as i32);
332        let (x1, y1) = ((rect.x + rect.w) as i32, (rect.y + rect.h) as i32);
333        (x1 > x0 && y1 > y0).then(|| {
334            RoundRect::new(
335                x0,
336                y0,
337                (x1 - x0) as u32,
338                (y1 - y0) as u32,
339                Radii::all(radius),
340            )
341        })
342    }
343}

345impl<'fb> ViCanvas for FramebufferCanvas<'fb> {
346    fn fill_rect(&mut self, rect: Rect, color: Color) {
347        let clip = self.active_clip();
348        let clipped = match rect.intersect(&clip) {
349            Some(r) => r,
350            None => return,
351        };
352
353        let x0 = clipped.x as i32;
354        let y0 = clipped.y as i32;
355        let x1 = (clipped.x + clipped.w) as i32;
356        let y1 = (clipped.y + clipped.h) as i32;
357
358        if color.a() == 255 {
359            // Fast path: no blending — precompute bytes once, write as u32 per pixel
360            let pixel = color.0.to_le_bytes();
361            for y in y0..y1 {
362                if y < 0 || y as u32 >= self.height {
363                    continue;
364                }
365                let row_off = (y as u32 * self.stride) as usize;
366                for x in x0..x1 {
367                    if x < 0 || x as u32 >= self.width {
368                        continue;
369                    }
370                    let off = row_off + (x as usize) * 4;
371                    if off + 3 < self.pixels.len() {
372                        self.pixels[off..off + 4].copy_from_slice(&pixel);
373                    }
374                }
375            }
376        } else {
377            // Alpha blend inline — coords are pre-clipped so put_pixel bounds checks are
378            // redundant. Precompute channels once; compute row_off once per row.
379            let sa = color.a() as u32;
380            let inv = 255 - sa;
381            let cb = color.b() as u32;
382            let cg = color.g() as u32;
383            let cr = color.r() as u32;
384            for y in y0..y1 {
385                let row_off = y as usize * self.stride as usize;
386                for x in x0..x1 {
387                    let off = row_off + x as usize * 4;
388                    if off + 3 >= self.pixels.len() {
389                        continue;
390                    }
391                    let db = self.pixels[off] as u32;
392                    let dg = self.pixels[off + 1] as u32;
393                    let dr = self.pixels[off + 2] as u32;
394                    let out: u32 = ((cb * sa + db * inv) / 255)
395                        | (((cg * sa + dg * inv) / 255) << 8)
396                        | (((cr * sa + dr * inv) / 255) << 16)
397                        | (0xFF << 24);
398                    self.pixels[off..off + 4].copy_from_slice(&out.to_le_bytes());
399                }
400            }
401        }
402    }

404    fn draw_line(&mut self, a: Point, b: Point, color: Color) {
405        // Bresenham integer line
406        let mut x0 = a.x as i32;
407        let mut y0 = a.y as i32;
408        let x1 = b.x as i32;
409        let y1 = b.y as i32;
410        let dx = (x1 - x0).abs();
411        let dy = -(y1 - y0).abs();
412        let sx = if x0 < x1 { 1i32 } else { -1 };
413        let sy = if y0 < y1 { 1i32 } else { -1 };
414        let mut err = dx + dy;
415        loop {
416            self.put_pixel(x0, y0, color);
417            if x0 == x1 && y0 == y1 {
418                break;
419            }
420            let e2 = 2 * err;
421            if e2 >= dy {
422                err += dy;
423                x0 += sx;
424            }
425            if e2 <= dx {
426                err += dx;
427                y0 += sy;
428            }
429        }
430    }

432    fn draw_text(&mut self, pos: Point, text: &str, style: TextStyle) {
433        // Reads FONT8X8 directly — ostd::font::draw_text uses a different byte-order
434        // convention and cannot be used here without producing incorrect colours.
435        use ostd::font::FONT8X8;
436        let mut cx = pos.x as i32;
437        let cy = pos.y as i32;
438        let w = self.width as i32;
439        let h = self.height as i32;
440        let opaque = style.color.a() == 255;
441        // Precompute bytes once — avoids b()/g()/r() extraction inside glyph loop
442        let pixel = style.color.0.to_le_bytes();
443        let clip = self.active_clip();
444
445        for ch in text.chars() {
446            // A — skip char entirely outside screen bounds
447            if cx >= w || cx + 8 <= 0 || cy >= h || cy + 8 <= 0 {
448                cx += 8;
449                continue;
450            }
451
452            // FONT8X8 covers 0x20..=0x7E (95 glyphs); index 0 = space (0x20)
453            let code = ch as u32;
454            let idx = if (0x20..=0x7E).contains(&code) {
455                (code - 0x20) as usize
456            } else {
457                (b'?' - 0x20) as usize
458            };
459            let glyph = &FONT8X8[idx];
460
461            // Fast path: char fully in screen bounds + clip + opaque — no per-pixel checks, no blend
462            let fully_in_bounds = cx >= 0 && cx + 8 <= w && cy >= 0 && cy + 8 <= h;
463            let fully_in_clip = (cx as f32) >= clip.x
464                && (cx + 8) as f32 <= clip.x + clip.w
465                && (cy as f32) >= clip.y
466                && (cy + 8) as f32 <= clip.y + clip.h;
467
468            if opaque && fully_in_bounds && fully_in_clip {
469                for (row, &bits) in glyph.iter().enumerate() {
470                    // C — skip empty glyph rows (space, punctuation)
471                    if bits == 0 {
472                        continue;
473                    }
474                    // B — row byte offset computed once per row
475                    let row_off = (cy as usize + row) * self.stride as usize;
476                    for col in 0..8usize {
477                        // FONT8X8 is LSB-first: bit 0 = leftmost pixel, bit 7 = rightmost
478                        if bits & (1u8 << col) != 0 {
479                            let off = row_off + (cx as usize + col) * 4;
480                            self.pixels[off..off + 4].copy_from_slice(&pixel);
481                        }
482                    }
483                }
484            } else {
485                // Slow path — boundary chars, semi-transparent, or partial clip
486                for row in 0..8i32 {
487                    let bits = glyph[row as usize];
488                    if bits == 0 {
489                        continue;
490                    } // C — empty row skip even in slow path
491                    for col in 0..8i32 {
492                        if bits & (1u8 << col) != 0 {
493                            self.put_pixel(cx + col, cy + row, style.color);
494                        }
495                    }
496                }
497            }
498            cx += 8;
499        }
500    }

530    fn draw_image(&mut self, dest: Rect, pixels: &[u8], src_stride: u32) {
531        let clip = self.active_clip();
532        let clipped = match dest.intersect(&clip) {
533            Some(r) => r,
534            None => return,
535        };
536        let dx_off = (clipped.x - dest.x) as i32;
537        let dy_off = (clipped.y - dest.y) as i32;
538        let dst_x0 = clipped.x as i32;
539        let dst_y0 = clipped.y as i32;
540        let dst_x1 = (clipped.x + clipped.w) as i32;
541        let dst_y1 = (clipped.y + clipped.h) as i32;
542
543        for dy in 0..(dst_y1 - dst_y0) {
544            let sy = (dy_off + dy) as usize;
545            let dst_y = dst_y0 + dy;
546            if dst_y < 0 || dst_y as u32 >= self.height {
547                continue;
548            }
549            let dst_row_off = dst_y as usize * self.stride as usize;
550            for dx in 0..(dst_x1 - dst_x0) {
551                let sx = (dx_off + dx) as usize;
552                let src_off = sy * src_stride as usize + sx * 4;
553                if src_off + 3 >= pixels.len() {
554                    continue;
555                }
556                let dst_off = dst_row_off + (dst_x0 + dx) as usize * 4;
557                if dst_off + 3 >= self.pixels.len() {
558                    continue;
559                }
560                let sa = pixels[src_off + 3];
561                if sa == 0 {
562                    continue;
563                }
564                if sa == 255 {
565                    // Opaque source: direct 4-byte copy, zero blend arithmetic.
566                    self.pixels[dst_off..dst_off + 4]
567                        .copy_from_slice(&pixels[src_off..src_off + 4]);
568                } else {
569                    // Partial alpha: inline blend over destination.
570                    let sa32 = sa as u32;
571                    let inv = 255 - sa32;
572                    let sb = pixels[src_off] as u32;
573                    let sg = pixels[src_off + 1] as u32;
574                    let sr = pixels[src_off + 2] as u32;
575                    let db = self.pixels[dst_off] as u32;
576                    let dg = self.pixels[dst_off + 1] as u32;
577                    let dr = self.pixels[dst_off + 2] as u32;
578                    let out: u32 = ((sb * sa32 + db * inv) / 255)
579                        | (((sg * sa32 + dg * inv) / 255) << 8)
580                        | (((sr * sa32 + dr * inv) / 255) << 16)
581                        | (0xFF << 24);
582                    self.pixels[dst_off..dst_off + 4].copy_from_slice(&out.to_le_bytes());
583                }
584            }
585        }
586    }

588    fn clip_push(&mut self, rect: Rect) {
589        let current = self.active_clip();
590        let new_clip = current.intersect(&rect).unwrap_or(Rect::ZERO);
591        if self.clip_depth + 1 < CLIP_STACK_DEPTH {
592            self.clip_depth += 1;
593            self.clip_stack[self.clip_depth] = new_clip;
594            self.clip_stack_i[self.clip_depth] = (
595                new_clip.x as i32,
596                new_clip.y as i32,
597                (new_clip.x + new_clip.w) as i32,
598                (new_clip.y + new_clip.h) as i32,
599            );
600        }
601    }
602
603    fn clip_pop(&mut self) {
604        if self.clip_depth > 0 {
605            self.clip_depth -= 1;
606        }
607    }

632    fn stroke_round_rect(&mut self, rect: Rect, radius: u32, color: Color) {

651    fn pixel(pixels: &[u8], x: u32, y: u32) -> Color {
652        let off = ((y * W + x) * 4) as usize;
653        Color(u32::from_le_bytes([
654            pixels[off],
655            pixels[off + 1],
656            pixels[off + 2],
657            pixels[off + 3],
658        ]))
659    }
660
661    fn canvas_with(draw: impl FnOnce(&mut FramebufferCanvas<'_>)) -> alloc::vec::Vec<u8> {
662        let mut pixels = alloc::vec![0u8; (W * H * 4) as usize];
663        let mut canvas = FramebufferCanvas::new(&mut pixels, W * 4, W, H);
664        canvas.fill_rect(Rect::new(0.0, 0.0, W as f32, H as f32), BG);
665        draw(&mut canvas);
666        pixels
667    }
668
669    #[test]
670    fn rounded_fill_skips_corners_and_stroke_matches_linear_outline() {
671        let ink = Color::rgb(200, 100, 50);
672        let (x, y, w, h) = (4u32, 3u32, 20u32, 14u32);
673        let rect = Rect::new(x as f32, y as f32, w as f32, h as f32);
674
675        let filled = canvas_with(|c| c.fill_round_rect(rect, PANEL_RADIUS, ink));
676        assert_eq!(pixel(&filled, x, y), BG);
677        assert_eq!(pixel(&filled, x + w - 1, y + h - 1), BG);
678        assert_eq!(pixel(&filled, x + w / 2, y), ink);
679        assert_eq!(pixel(&filled, x + w / 2, y + h / 2), ink);
680
681        let stroked = canvas_with(|c| c.stroke_round_rect(rect, PANEL_RADIUS, ink));
682        let shape = RoundRect::new(x as i32, y as i32, w, h, Radii::all(PANEL_RADIUS));
683        for py in 0..H {
684            for px in 0..W {
685                let on_outline = py >= y && py < y + h && px >= x && {
686                    let ((l0, l1), (r0, r1)) = shape.outline(py - y);
687                    let col = px - x;
688                    (l0..l1).contains(&col) || (r0..r1).contains(&col)
689                };
690                let expected = if on_outline { ink } else { BG };
691                assert_eq!(pixel(&stroked, px, py), expected, "({px}, {py})");
692            }
693        }
