2//! ViRenderer trait — abstract rendering backend for ViUI v2.
3//!
4//! # Backend selection
5//!
6//! G1: `FramebufferRenderer` (CPU software rasterizer, default)
7//! G2+: GPU backend implementing the same trait; widget code is unchanged.
8//!
9//! # Object safety
10//!
11//! `ViRenderer` is object-safe; use `Box<dyn ViRenderer>` to store a
12//! heap-allocated renderer when the concrete type is unknown at compile time.
13//!
14//! # Lifetime note
15//!
16//! `FramebufferCanvas<'fb>` borrows pixels from `ViSurface`. The closure pattern
17//! in `render()` confines that borrow to the stack frame, avoiding any
18//! self-referential struct or lifetime gymnastics.
19

30// ─── ViRenderer ──────────────────────────────────────────────────────────────
31
32/// Abstract rendering backend.
33///
34/// # Contract
35///
36/// - Call `render()` once per frame, after collecting dirty rects via `DirtyRect`.
37/// - All painting must happen inside the `draw` closure — canvas is invalid after return.
38/// - `damage` is advisory to drawing but is submitted precisely by compatible backends.
39pub trait ViRenderer {
40    /// Run a paint closure with exclusive canvas access, then submit the frame.
41    ///
42    /// `damage`: the screen region that changed this frame. `None` = full surface.
43    fn render(&mut self, damage: Option<Rect>, draw: &mut dyn FnMut(&mut dyn ViCanvas));
44
45    /// Surface dimensions in pixels as `(width, height)`.
46    fn size(&self) -> (u32, u32);
47}

49// ─── FramebufferRenderer ─────────────────────────────────────────────────────
50
51/// G1 CPU renderer wrapping a `ViSurface` + `FramebufferCanvas`.
52///
53/// The `FramebufferCanvas<'fb>` borrow is confined to the `render()` call
54/// stack frame — no heap allocation, no unsafe, no self-referential struct.
55pub struct FramebufferRenderer {
56    surf: SurfaceHandle,
57}
58
59/// Single-threaded access to a Grant-backed compositor surface.
60///
61/// The handle lets a [`FramebufferRenderer`] and a managed lifecycle runner
62/// operate on the same surface without duplicating ownership of its Grant.
63pub(crate) type SurfaceHandle = Rc<RefCell<ViSurface>>;
64
65/// Wrap a surface for sharing between ViUI's renderer and lifecycle adapter.
66pub(crate) fn shared_surface(surf: ViSurface) -> SurfaceHandle {
67    Rc::new(RefCell::new(surf))
68}
69
70impl FramebufferRenderer {
71    pub fn new(surf: ViSurface) -> Self {
72        Self::from_surface_handle(shared_surface(surf))
73    }
74
75    /// Create a renderer for an existing managed-surface handle.
76    pub(crate) fn from_surface_handle(surf: SurfaceHandle) -> Self {
77        Self { surf }
78    }
79
80    /// Unwrap the inner `ViSurface` (e.g. for IPC cleanup after app exit).
81    ///
82    /// # Panics
83    /// Panics when another lifecycle or renderer handle is still alive.
84    pub fn into_surf(self) -> ViSurface {
85        match Rc::try_unwrap(self.surf) {
86            Ok(surface) => surface.into_inner(),
87            Err(_) => panic!("FramebufferRenderer surface is still shared"),
88        }
89    }
90}

92impl ViRenderer for FramebufferRenderer {
93    fn render(&mut self, damage: Option<Rect>, draw: &mut dyn FnMut(&mut dyn ViCanvas)) {
94        let mut surf = self.surf.borrow_mut();
95        let (w, h) = (surf.width(), surf.height());
96        {
97            let stride = surf.stride() as u32;
98            let pixels = surf.pixels_mut();
99            let mut canvas = FramebufferCanvas::new(pixels, stride, w, h);
100            draw(&mut canvas);
101        }
102        match damage {
103            None => surf.damage_all(),
104            Some(damage) => {
105                if let Some(rect) = clipped_damage(damage, w, h) {
106                    surf.damage(rect);
107                }
108            }
109        }
110    }
111
112    fn size(&self) -> (u32, u32) {
113        let surf = self.surf.borrow();
114        (surf.width(), surf.height())
115    }
116}

118/// Convert ViUI float bounds into a clipped, outward-rounded surface rectangle.
119fn clipped_damage(damage: Rect, width: u32, height: u32) -> Option<SurfaceRect> {
120    if damage.w <= 0.0
121        || damage.h <= 0.0
122        || !damage.x.is_finite()
123        || !damage.y.is_finite()
124        || !damage.w.is_finite()
125        || !damage.h.is_finite()
126    {
127        return None;
128    }
129    let right = damage.x + damage.w;
130    let bottom = damage.y + damage.h;
131    if !right.is_finite() || !bottom.is_finite() {
132        return None;
133    }
134    let left = damage.x.max(0.0).min(width as f32);
135    let top = damage.y.max(0.0).min(height as f32);
136    let right = right.max(0.0).min(width as f32);
137    let bottom = bottom.max(0.0).min(height as f32);
138    let x = floor_nonnegative(left);
139    let y = floor_nonnegative(top);
140    let right = ceil_nonnegative(right);
141    let bottom = ceil_nonnegative(bottom);
142    (right > x && bottom > y).then_some(SurfaceRect {
143        x: x as i32,
144        y: y as i32,
145        w: right - x,
146        h: bottom - y,
147    })
148}
149
150/// Convert a bounded non-negative finite float to its integer floor in `no_std`.
151fn floor_nonnegative(value: f32) -> u32 {
152    value as u32
153}
154
155/// Convert a bounded non-negative finite float to its integer ceiling in `no_std`.
156fn ceil_nonnegative(value: f32) -> u32 {
157    let truncated = floor_nonnegative(value);
158    truncated.saturating_add(u32::from((truncated as f32) < value))
159}

161#[cfg(test)]
162mod tests {
163    use super::clipped_damage;
164    use crate::layout::Rect;
165
166    #[test]
167    fn damage_rounds_outward() {
168        let rect = clipped_damage(Rect::new(1.2, 2.8, 3.1, 4.01), 20, 20).unwrap();
169        assert_eq!((rect.x, rect.y, rect.w, rect.h), (1, 2, 4, 5));
170    }
171
172    #[test]
173    fn damage_clips_to_surface() {
174        let rect = clipped_damage(Rect::new(-3.5, 8.2, 7.1, 4.1), 10, 10).unwrap();
175        assert_eq!((rect.x, rect.y, rect.w, rect.h), (0, 8, 4, 2));
176    }
177
178    #[test]
179    fn damage_ignores_empty_offscreen_and_non_finite_rects() {
180        assert!(clipped_damage(Rect::new(1.0, 1.0, 0.0, 2.0), 10, 10).is_none());
181        assert!(clipped_damage(Rect::new(12.0, 1.0, 2.0, 2.0), 10, 10).is_none());
182        assert!(clipped_damage(Rect::new(f32::NAN, 1.0, 2.0, 2.0), 10, 10).is_none());
183    }
