2//! Shared scalable typography for layout and paint. GUI defaults to bundled Inter 16px.
3
4use ostd::font_atlas::GlyphAtlas;
5
6pub struct FontContext {
7    pub atlas: GlyphAtlas,
8    pub size_px: f32,
9}
10
11impl Default for FontContext {
12    fn default() -> Self {
13        Self {
14            atlas: GlyphAtlas::from_static(ostd::typography::INTER_REGULAR)
15                .expect("bundled Inter must be a valid static TrueType font"),
16            size_px: 16.0,
17        }
18    }
19}
20
21impl FontContext {
22    /// Load a custom font used by both layout and paint.
23    pub fn with_font(font_bytes: &[u8], size_px: f32) -> Option<Self> {
24        if !ostd::font_atlas::supported_size(size_px) {
25            return None;
26        }
27        GlyphAtlas::new(font_bytes).map(|atlas| Self { atlas, size_px })
28    }
29
30    pub fn measure(&mut self, text: &str) -> f32 {
31        self.atlas.measure(text, self.size_px)
32    }
33
34    pub fn line_height(&self) -> f32 {
35        self.atlas.line_height(self.size_px)
36    }
37
38    /// Populate original UTF-8 cluster boundaries with normalized, kerned pen advances.
39    /// Storage is reused across layouts. Vietnamese combining sequences are indivisible.
40    pub fn cursor_positions(&mut self, text: &str, positions: &mut alloc::vec::Vec<(usize, f32)>) {
41        positions.clear();
42        positions.push((0, 0.0));
43        let mut start = 0;
44        let mut pen = 0.0;
45        let mut previous = None;
46        for end in text.char_indices()
47            .filter_map(|(i, ch)| if i > 0 && !ostd::typography::is_combining_mark(ch) { Some(i) } else { None })
48            .chain(core::iter::once(text.len()))
49        {
50            for ch in ostd::typography::normalized_chars(&text[start..end]) {
51                if let Some(left) = previous { pen += self.atlas.kerning(left, ch, self.size_px); }
52                pen += self.atlas.advance(ch, self.size_px);
53                previous = Some(ch);
54            }
55            if end > 0 { positions.push((end, pen)); }
56            start = end;
57        }
58    }
59
60    pub fn hit_position(positions: &[(usize, f32)], x: f32) -> usize {
61        for pair in positions.windows(2) {
62            if x < (pair[0].1 + pair[1].1) * 0.5 { return pair[0].0; }
63        }
64        positions.last().map_or(0, |p| p.0)
65    }
66}

68#[cfg(test)]
69mod tests {
70    use super::FontContext;

76    #[test]
77    fn proportional_layout_updates_for_equal_byte_length_text() {
78        let mut font = FontContext::default();
79        let text = Signal::new(alloc::string::String::from("iii"));
80        let mut label = crate::node_widgets::label::Label::new(text.clone());
81        let constraints = Constraints::root(Size::new(300.0, 100.0));
82        let narrow = label.layout(constraints, &mut font);
83        text.set(alloc::string::String::from("WWW"));
84        let wide = label.layout(constraints, &mut font);
85        assert!(wide.w > narrow.w);
86        assert_eq!(wide.w, font.measure("WWW"));
87        assert_eq!(wide.h, font.line_height());
88    }
89
90    #[test]
91    fn custom_size_and_v1_layout_share_font_metrics() {
92        use crate::widget::ViWidget;
93        let mut font = FontContext::with_font(ostd::typography::INTER_REGULAR, 24.0).unwrap();
94        let label = crate::widgets::label::Label::new("Tiếng Việt");
95        let node = label.layout(Constraints::root(Size::new(500.0, 100.0)), &mut font);
96        assert_eq!(node.bounds.w, font.measure("Tiếng Việt"));
97        assert_eq!(node.bounds.h, font.line_height());
98    }

100    #[test]
101    fn cursor_boundaries_match_nfc_advances_without_splitting_vietnamese_marks() {
102        let mut font = FontContext::default();
103        let mut positions = alloc::vec::Vec::new();
104        let text = "e\u{0302}\u{0301} Wi";
105        font.cursor_positions(text, &mut positions);
106        assert_eq!(positions[1].0, "e\u{0302}\u{0301}".len());
107        assert_eq!(positions[1].1, font.measure("ế"));
108        assert_eq!(positions.last().unwrap().1, font.measure(text));
109        for &(byte, x) in &positions {
110            assert_eq!(FontContext::hit_position(&positions, x), byte);
111        }
112    }

114    #[test]
115    fn canvas_text_is_normalized_antialiased_clipped_and_bgra() {
116        let mut font = FontContext::default();
117        let mut red = alloc::vec![0; 100 * 40 * 4];
118        let mut blue = red.clone();
119        for (pixels, text, color) in [
120            (&mut red, "ế", Color::rgb(255, 0, 0)),
121            (&mut blue, "e\u{0302}\u{0301}", Color::rgb(0, 0, 255)),
122        ] {
123            let mut canvas = FramebufferCanvas::new(pixels, 400, 100, 40);
124            canvas.clip_push(Rect::new(4.0, 0.0, 10.0, 40.0));
125            canvas.draw_text_scaled(Point::new(2.0, 2.0), text, 16.0, color, &mut font.atlas);
126        }
127        let mut partial_coverage = false;
128        let mut painted = false;
129        for (index, (r, b)) in red.chunks_exact(4).zip(blue.chunks_exact(4)).enumerate() {
130            assert_eq!(r[3], b[3]);
131            assert_eq!(r[2], b[0]);
132            assert_eq!(r[0], 0);
133            assert_eq!(b[2], 0);
134            if index % 100 < 4 || index % 100 >= 14 { assert_eq!(r[3], 0); }
135            painted |= r[3] > 0;
136            partial_coverage |= r[3] > 0 && r[3] < 255;
137        }
138        assert!(painted && partial_coverage);
139    }
