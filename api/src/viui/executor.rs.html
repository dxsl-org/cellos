2//! `CommandExecutor` trait + `CpuExecutor` — CPU playback backend.
3//!
4//! `CommandExecutor` decouples the command recorder (`GpuCanvas`) from the
5//! actual drawing strategy, letting `GpuRenderer` drive either a CPU
6//! rasterizer today or a hardware GPU backend in G2+.
7//!
8//! # Damage-rect filtering
9//! When `damage` is `Some(rect)`, commands whose bounding rect does not
10//! intersect `rect` are skipped. `None` = full repaint (skip nothing).
11//! Over-estimation of bounding rects is safe; under-estimation causes
12//! missing pixels. `GpuCmd::bounding_rect()` always over-estimates.
13
14use crate::canvas::{FramebufferCanvas, ViCanvas};
15use crate::gpu_cmd::{GpuCmd, GpuCommandBuffer, RecordedCmd};
16use crate::layout::Rect;
17use ostd::display::ViSurface;
18
19// ─── CommandExecutor ─────────────────────────────────────────────────────────
20
21/// Executes a recorded `GpuCommandBuffer`, optionally constrained to a damage rect.
22pub trait CommandExecutor {
23    /// Replay all commands in `buf`, skipping those outside `damage` if set.
24    fn execute(&mut self, buf: &GpuCommandBuffer, damage: Option<Rect>);
25}

27// ─── CpuExecutor ─────────────────────────────────────────────────────────────
28
29/// G1 CPU executor: replays `GpuCmd`s via `FramebufferCanvas` + `ViSurface`.
30///
31/// Produces identical output to `FramebufferRenderer` but skips commands
32/// outside the supplied damage rect, reducing CPU rasterization work.
33pub struct CpuExecutor {
34    surf: ViSurface,
35    font: crate::font_context::FontContext,
36}
37
38impl CpuExecutor {
39    pub fn new(surf: ViSurface) -> Self {
40        Self { surf, font: crate::font_context::FontContext::default() }
41    }
42
43    /// Unwrap the inner `ViSurface` (e.g. for IPC cleanup after app exit).
44    pub fn into_surf(self) -> ViSurface {
45        self.surf
46    }
47}
48
49impl CommandExecutor for CpuExecutor {
50    fn execute(&mut self, buf: &GpuCommandBuffer, damage: Option<Rect>) {
51        let stride = self.surf.stride() as u32;
52        let (w, h) = (self.surf.width(), self.surf.height());
53        let pixels = self.surf.pixels_mut();
54        let mut canvas = FramebufferCanvas::new(pixels, stride, w, h);
55        execute_commands(&mut canvas, &mut self.font, buf, damage);
56
57        // G1: always damage_all; G2+ can flip only the damage rect.
58        self.surf.damage_all();
59    }
60}
61
62/// CPU playback shared by the surface executor and pixel-buffer regressions.
63fn execute_commands(canvas: &mut FramebufferCanvas<'_>, font: &mut crate::font_context::FontContext, buf: &GpuCommandBuffer, damage: Option<Rect>) {
64    if let Some(rect) = damage { canvas.clip_push(rect); }
65    for RecordedCmd { cmd, bounds } in buf.recorded_slice() {
66        if let Some(damage_rect) = damage {
67            if let Some(b) = bounds {
68                if !b.intersects(damage_rect) { continue; }
69            }
70        }
71        match cmd {
72            GpuCmd::FillRect { rect, color } => canvas.fill_rect(*rect, *color),
73            GpuCmd::DrawLine { a, b, color } => canvas.draw_line(*a, *b, *color),
74            GpuCmd::DrawText { pos, text, color, size_px } => {
75                canvas.draw_text_scaled(*pos, text, *size_px, *color, &mut font.atlas);
76            }
77            GpuCmd::DrawImage { dest, pixels, src_stride } => canvas.draw_image(*dest, pixels, *src_stride),
78            GpuCmd::DrawTextShort { pos, bytes, len, color, size_px } => {
79                let text = core::str::from_utf8(&bytes[..*len as usize]).expect("recorded text is valid UTF-8");
80                canvas.draw_text_scaled(*pos, text, *size_px, *color, &mut font.atlas);
81            }
82            GpuCmd::ClipPush { rect } => canvas.clip_push(*rect),
83            GpuCmd::ClipPop => canvas.clip_pop(),
84        }
85    }
86    if damage.is_some() { canvas.clip_pop(); }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92    use crate::canvas::Color;
93    use crate::gpu_canvas::GpuCanvas;
94    use crate::layout::Point;
95
96    #[test]
97    fn recorded_cpu_text_preserves_fractional_sizes_and_clipping() {
98        let mut font = crate::font_context::FontContext::default();
99        let long_text = "W".repeat(130);
100        for text in ["Wiế", long_text.as_str()] {
101            for px in [16.5, 0.5] {
102                let mut commands = GpuCommandBuffer::new();
103                {
104                    let mut recorder = GpuCanvas::new(&mut commands, 160, 50);
105                    recorder.clip_push(Rect::new(3.0, 0.0, 80.0, 50.0));
106                    recorder.draw_text_scaled(Point::new(1.0, 1.0), text, px, Color::WHITE, &mut font.atlas);
107                    recorder.clip_pop();
108                }
109                let mut recorded = alloc::vec![0; 160 * 50 * 4];
110                let mut direct = recorded.clone();
111                let mut playback = FramebufferCanvas::new(&mut recorded, 640, 160, 50);
112                execute_commands(&mut playback, &mut font, &commands, None);
113                let mut expected = FramebufferCanvas::new(&mut direct, 640, 160, 50);
114                expected.clip_push(Rect::new(3.0, 0.0, 80.0, 50.0));
115                expected.draw_text_scaled(Point::new(1.0, 1.0), text, px, Color::WHITE, &mut font.atlas);
116                assert_eq!(recorded, direct, "recorded playback differs at {px}px");
117            }
118        }
119    }
