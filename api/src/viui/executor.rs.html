2//! `CommandExecutor` trait + `CpuExecutor` — CPU playback backend.
3//!
4//! `CommandExecutor` decouples the command recorder (`GpuCanvas`) from the
5//! actual drawing strategy, letting `GpuRenderer` drive either a CPU
6//! rasterizer today or a hardware GPU backend in G2+.
7//!
8//! # Damage-rect filtering
9//! When `damage` is `Some(rect)`, commands whose bounding rect does not
10//! intersect `rect` are skipped. `None` = full repaint (skip nothing).
11//! Over-estimation of bounding rects is safe; under-estimation causes
12//! missing pixels. `GpuCmd::bounding_rect()` always over-estimates.
13
14use crate::canvas::{FramebufferCanvas, ViCanvas};
15use crate::gpu_cmd::{GpuCmd, GpuCommandBuffer, RecordedCmd};
16use crate::layout::Rect;
17use ostd::display::ViSurface;

19// ─── CommandExecutor ─────────────────────────────────────────────────────────
20
21/// Executes a recorded `GpuCommandBuffer`, optionally constrained to a damage rect.
22pub trait CommandExecutor {
23    /// Replay all commands in `buf`, skipping those outside `damage` if set.
24    fn execute(&mut self, buf: &GpuCommandBuffer, damage: Option<Rect>);
25}
26
27// ─── CpuExecutor ─────────────────────────────────────────────────────────────
28
29/// G1 CPU executor: replays `GpuCmd`s via `FramebufferCanvas` + `ViSurface`.
30///
31/// Produces identical output to `FramebufferRenderer` but skips commands
32/// outside the supplied damage rect, reducing CPU rasterization work.
33pub struct CpuExecutor {
34    surf: ViSurface,
35    font: crate::font_context::FontContext,
36}
37
38impl CpuExecutor {
39    pub fn new(surf: ViSurface) -> Self {
40        Self {
41            surf,
42            font: crate::font_context::FontContext::default(),
43        }
44    }
45
46    /// Unwrap the inner `ViSurface` (e.g. for IPC cleanup after app exit).
47    pub fn into_surf(self) -> ViSurface {
48        self.surf
49    }
50}
51
52impl CommandExecutor for CpuExecutor {
53    fn execute(&mut self, buf: &GpuCommandBuffer, damage: Option<Rect>) {
54        let stride = self.surf.stride() as u32;
55        let (w, h) = (self.surf.width(), self.surf.height());
56        let pixels = self.surf.pixels_mut();
57        let mut canvas = FramebufferCanvas::new(pixels, stride, w, h);
58        execute_commands(&mut canvas, &mut self.font, buf, damage);
59
60        // G1: always damage_all; G2+ can flip only the damage rect.
61        self.surf.damage_all();
62    }
63}
64
65/// CPU playback shared by the surface executor and pixel-buffer regressions.
66fn execute_commands(
67    canvas: &mut FramebufferCanvas<'_>,
68    font: &mut crate::font_context::FontContext,
69    buf: &GpuCommandBuffer,
70    damage: Option<Rect>,
71) {
72    if let Some(rect) = damage {
73        canvas.clip_push(rect);
74    }
75    for RecordedCmd { cmd, bounds } in buf.recorded_slice() {
76        if let Some(damage_rect) = damage {
77            if let Some(b) = bounds {
78                if !b.intersects(damage_rect) {
79                    continue;
80                }
81            }
82        }
83        match cmd {
84            GpuCmd::FillRect { rect, color } => canvas.fill_rect(*rect, *color),
85            GpuCmd::FillRoundRect {
86                rect,
87                radius,
88                color,
89            } => canvas.fill_round_rect(*rect, *radius, *color),
90            GpuCmd::StrokeRoundRect {
91                rect,
92                radius,
93                color,
94            } => canvas.stroke_round_rect(*rect, *radius, *color),
95            GpuCmd::DrawLine { a, b, color } => canvas.draw_line(*a, *b, *color),
96            GpuCmd::DrawText {
97                pos,
98                text,
99                color,
100                size_px,
101            } => {
102                canvas.draw_text_scaled(*pos, text, *size_px, *color, &mut font.atlas);
103            }
104            GpuCmd::DrawImage {
105                dest,
106                pixels,
107                src_stride,
108            } => canvas.draw_image(*dest, pixels, *src_stride),
109            GpuCmd::DrawTextShort {
110                pos,
111                bytes,
112                len,
113                color,
114                size_px,
115            } => {
116                let text = core::str::from_utf8(&bytes[..*len as usize])
117                    .expect("recorded text is valid UTF-8");
118                canvas.draw_text_scaled(*pos, text, *size_px, *color, &mut font.atlas);
119            }
120            GpuCmd::ClipPush { rect } => canvas.clip_push(*rect),
121            GpuCmd::ClipPop => canvas.clip_pop(),
122        }
123    }
124    if damage.is_some() {
125        canvas.clip_pop();
126    }
127}

136    #[test]
137    fn recorded_cpu_text_preserves_fractional_sizes_and_clipping() {
138        let mut font = crate::font_context::FontContext::default();
139        let long_text = "W".repeat(130);
140        for text in ["Wiế", long_text.as_str()] {
141            for px in [16.5, 0.5] {
142                let mut commands = GpuCommandBuffer::new();
143                {
144                    let mut recorder = GpuCanvas::new(&mut commands, 160, 50);
145                    recorder.clip_push(Rect::new(3.0, 0.0, 80.0, 50.0));
146                    recorder.draw_text_scaled(
147                        Point::new(1.0, 1.0),
148                        text,
149                        px,
150                        Color::WHITE,
151                        &mut font.atlas,
152                    );
153                    recorder.clip_pop();
154                }
155                let mut recorded = alloc::vec![0; 160 * 50 * 4];
156                let mut direct = recorded.clone();
157                let mut playback = FramebufferCanvas::new(&mut recorded, 640, 160, 50);
158                execute_commands(&mut playback, &mut font, &commands, None);
159                let mut expected = FramebufferCanvas::new(&mut direct, 640, 160, 50);
160                expected.clip_push(Rect::new(3.0, 0.0, 80.0, 50.0));
161                expected.draw_text_scaled(
162                    Point::new(1.0, 1.0),
163                    text,
164                    px,
165                    Color::WHITE,
166                    &mut font.atlas,
167                );
168                assert_eq!(recorded, direct, "recorded playback differs at {px}px");
169            }
170        }
171    }

173    #[test]
174    fn recorded_round_rects_replay_pixel_identical_to_direct_drawing() {
175        use ostd::display::linear::PANEL_RADIUS;
176
177        let (w, h) = (48u32, 32u32);
178        let rect = Rect::new(4.0, 3.0, 36.0, 24.0);
179        // Clip cuts through the bottom-right corner: recorded geometry must
180        // stay unclipped so the visible arc is unchanged.
181        let clip = Rect::new(0.0, 0.0, 30.0, 20.0);
182        let (fill, ink) = (Color::rgb(30, 60, 90), Color::rgb(200, 100, 50));
183        let draw = |canvas: &mut dyn ViCanvas| {
184            canvas.fill_rect(Rect::new(0.0, 0.0, w as f32, h as f32), Color::WHITE);
185            canvas.fill_round_rect(rect, PANEL_RADIUS, fill);
186            canvas.stroke_round_rect(rect, PANEL_RADIUS, ink);
187            canvas.clip_push(clip);
188            canvas.fill_round_rect(rect, PANEL_RADIUS, ink);
189            canvas.stroke_round_rect(rect, PANEL_RADIUS, fill);
190            canvas.clip_pop();
191        };
192
193        let mut commands = GpuCommandBuffer::new();
194        draw(&mut GpuCanvas::new(&mut commands, w, h));
195        let mut recorded = alloc::vec![0; (w * h * 4) as usize];
196        let mut direct = recorded.clone();
197        let mut font = crate::font_context::FontContext::default();
198        execute_commands(
199            &mut FramebufferCanvas::new(&mut recorded, w * 4, w, h),
200            &mut font,
201            &commands,
202            None,
203        );
204        draw(&mut FramebufferCanvas::new(&mut direct, w * 4, w, h));
205        assert_eq!(recorded, direct);
206    }
