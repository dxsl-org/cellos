1//! TextEdit widget — single-line text input with keyboard editing.
2
3use alloc::{string::String, vec::Vec};
4use core::cell::{Cell, RefCell};
5
6use crate::event::{Event, EventCx, EventStatus, KeyCode, MouseButton};
7use crate::layout::{Constraints, LayoutNode, Length, Point, Rect, Size};
8use crate::widget::{PaintCx, ViWidget, WidgetId};
9use ostd::display::linear::CONTROL_RADIUS;
10
11const PAD: f32 = 4.0;
12
13pub struct TextEdit {
14    pub id: WidgetId,
15    pub text: String,
16    cursor: usize, // original UTF-8 byte boundary
17    focused: bool,
18    width: Length,
19    bounds: Cell<Rect>,
20    cursor_positions: RefCell<Vec<(usize, f32)>>,
21}
22
23impl TextEdit {
24    pub fn new(id: WidgetId) -> Self {
25        Self {
26            id,
27            text: String::new(),
28            cursor: 0,
29            focused: false,
30            width: Length::Fill,
31            bounds: Cell::new(Rect::ZERO),
32            cursor_positions: RefCell::new(Vec::new()),
33        }
34    }
35
36    pub fn with_text(mut self, text: impl Into<String>) -> Self {
37        self.text = text.into();
38        self.cursor = self.text.len();
39        self
40    }
41
42    pub fn with_width(mut self, w: Length) -> Self {
43        self.width = w;
44        self
45    }
46
47    fn next_cursor(&self) -> usize {
48        self.text[self.cursor..]
49            .char_indices()
50            .skip(1)
51            .find(|(_, ch)| !ostd::typography::is_combining_mark(*ch))
52            .map_or(self.text.len(), |(i, _)| self.cursor + i)
53    }
54
55    fn previous_cursor(&self) -> usize {
56        self.text[..self.cursor]
57            .char_indices()
58            .rev()
59            .find(|(_, ch)| !ostd::typography::is_combining_mark(*ch))
60            .map_or(0, |(i, _)| i)
61    }
62
63    fn insert_char(&mut self, ch: char) {
64        self.text.insert(self.cursor, ch);
65        self.cursor += ch.len_utf8();
66    }
67
68    fn delete_before_cursor(&mut self) {
69        let start = self.previous_cursor();
70        self.text.replace_range(start..self.cursor, "");
71        self.cursor = start;
72    }
73
74    fn delete_at_cursor(&mut self) {
75        let end = self.next_cursor();
76        self.text.replace_range(self.cursor..end, "");
77    }
78}
79
80impl ViWidget for TextEdit {
81    fn layout(
82        &self,
83        constraints: Constraints,
84        font: &mut crate::font_context::FontContext,
85    ) -> LayoutNode {
86        font.cursor_positions(&self.text, &mut self.cursor_positions.borrow_mut());
87        let w = match self.width {
88            Length::Fill | Length::FillPortion(_) => constraints.max.w,
89            Length::Fixed(v) => v,
90            Length::Shrink => (font.measure(&self.text) + font.measure("  ") + PAD * 2.0).max(40.0),
91        };
92        let size = constraints.constrain(Size {
93            w,
94            h: font.line_height() + PAD * 2.0,
95        });
96        let bounds = Rect::from_origin_size(constraints.origin, size);
97        self.bounds.set(bounds);
98        LayoutNode::leaf(bounds)
99    }
100
101    fn paint(&self, cx: &mut PaintCx) {
102        let bounds = self.bounds.get().translate(cx.origin.x, cx.origin.y);
103        let bg = if self.focused {
104            cx.theme.input_focused_bg()
105        } else {
106            cx.theme.input_bg()
107        };
108        cx.canvas.fill_round_rect(bounds, CONTROL_RADIUS, bg);
109
110        let border = if self.focused {
111            cx.theme.input_focused_border()
112        } else {
113            cx.theme.border()
114        };
115        cx.canvas.stroke_round_rect(bounds, CONTROL_RADIUS, border);
116
117        let text_pos = Point::new(bounds.x + PAD, bounds.y + PAD);
118        cx.canvas.clip_push(bounds);
119        let text_color = cx.theme.text_primary();
120        cx.draw_text(text_pos, &self.text, text_color, 0.0);
121
122        // Cursor
123        if self.focused {
124            let cx_x = bounds.x + PAD + cx.font.measure(&self.text[..self.cursor]);
125            let cy0 = bounds.y + PAD;
126            let cy1 = bounds.y + PAD + cx.font.line_height();
127            cx.canvas
128                .draw_line(Point::new(cx_x, cy0), Point::new(cx_x, cy1), text_color);
129        }
130        cx.canvas.clip_pop();
131    }
132
133    fn event(&mut self, cx: &mut EventCx, e: &Event) -> EventStatus {
134        let bounds = cx.bounds();
135        match e {
136            Event::MousePress {
137                pos,
138                button: MouseButton::Left,
139            } => {
140                if bounds.contains(*pos) {
141                    if !self.focused {
142                        self.focused = true;
143                        cx.set_focus(self.id);
144                        cx.mark_dirty();
145                    }
146                    // Move cursor to click position
147                    let rel_x = (pos.x - bounds.x - PAD).max(0.0);
148                    self.cursor = crate::font_context::FontContext::hit_position(
149                        &self.cursor_positions.borrow(),
150                        rel_x,
151                    );
152                    cx.mark_dirty();
153                    EventStatus::Consumed
154                } else {
155                    if self.focused {
156                        self.focused = false;
157                        cx.release_focus();
158                        cx.mark_dirty();
159                    }
160                    EventStatus::Ignored
161                }
162            }
163            Event::Char(ch) if self.focused => {
164                if !ch.is_control() {
165                    self.insert_char(*ch);
166                    cx.mark_dirty();
167                    EventStatus::Consumed
168                } else {
169                    EventStatus::Ignored
170                }
171            }
172            Event::KeyPress { key, .. } if self.focused => {
173                match key {
174                    KeyCode::Backspace => {
175                        self.delete_before_cursor();
176                        cx.mark_dirty();
177                    }
178                    KeyCode::Delete => {
179                        self.delete_at_cursor();
180                        cx.mark_dirty();
181                    }
182                    KeyCode::Left => {
183                        if self.cursor > 0 {
184                            self.cursor = self.previous_cursor();
185                            cx.mark_dirty();
186                        }
187                    }
188                    KeyCode::Right => {
189                        if self.cursor < self.text.len() {
190                            self.cursor = self.next_cursor();
191                            cx.mark_dirty();
192                        }
193                    }
194                    KeyCode::Home => {
195                        self.cursor = 0;
196                        cx.mark_dirty();
197                    }
198                    KeyCode::End => {
199                        self.cursor = self.text.len();
200                        cx.mark_dirty();
201                    }
202                    KeyCode::Escape => {
203                        self.focused = false;
204                        cx.release_focus();
205                        cx.mark_dirty();
206                    }
207                    _ => return EventStatus::Ignored,
208                }
209                EventStatus::Consumed
210            }
211            Event::Blur => {
212                self.focused = false;
213                cx.mark_dirty();
214                EventStatus::Ignored
215            }
216            _ => EventStatus::Ignored,
217        }
218    }
219}

222mod tests {
223    use super::*;
224    use crate::font_context::FontContext;

226    #[test]
227    fn click_and_cursor_use_measured_v1_layout() {
228        let mut edit = TextEdit::new(WidgetId::ROOT).with_text("Wiii");
229        let mut font = FontContext::default();
230        let layout = edit.layout(Constraints::root(Size::new(200.0, 80.0)), &mut font);
231        let mut state = crate::state_store::WidgetStateStore::new();
232        let mut focus = crate::state_store::FocusManager::new();
233        let mut cx = EventCx {
234            state: &mut state,
235            focus: &mut focus,
236            widget_id: edit.id,
237            layout: crate::layout::LayoutView(&layout),
238            needs_repaint: false,
239        };
240        edit.event(
241            &mut cx,
242            &Event::MousePress {
243                pos: Point::new(PAD + font.measure("W"), 6.0),
244                button: MouseButton::Left,
245            },
246        );
247        assert_eq!(edit.cursor, 1);
248        edit.event(&mut cx, &Event::Char('x'));
249        assert_eq!(edit.text, "Wxiii");
250        assert_eq!(layout.bounds.h, font.line_height() + PAD * 2.0);
251    }

253    #[test]
254    fn focused_field_text_and_cursor_use_theme_tokens() {
255        use crate::node_widgets::style::test_support::{fills, lines, record, strokes, text_color};
256        use crate::theme::{ViTheme, DARK_THEME, LIGHT_THEME};
257
258        for theme in [&LIGHT_THEME as &dyn ViTheme, &DARK_THEME] {
259            let mut edit = TextEdit::new(WidgetId::ROOT).with_text("cellos");
260            edit.focused = true;
261            let mut font = FontContext::default();
262            edit.layout(Constraints::root(Size::new(200.0, 80.0)), &mut font);
263            let buf =
264                record(|canvas| edit.paint(&mut PaintCx::with_theme(canvas, theme, &mut font)));
265            let painted = fills(&buf);
266            assert_eq!(painted[0].1, theme.input_focused_bg());
267            assert_eq!(strokes(&buf), [theme.input_focused_border()]);
268            assert_eq!(text_color(&buf, "cellos"), Some(theme.text_primary()));
269            assert_eq!(lines(&buf), [theme.text_primary()]);
270        }
271    }
