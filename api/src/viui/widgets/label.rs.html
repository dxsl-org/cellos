1//! Label widget — static text display.
2
3use alloc::string::String;
4
5use crate::canvas::Color;
6use crate::event::{Event, EventCx, EventStatus};
7use crate::layout::{Constraints, LayoutNode, Rect, Size};
8use crate::widget::{PaintCx, ViWidget};
9
10pub struct Label {
11    pub text: String,
12    /// Text colour; `None` uses the theme's `text_primary`.
13    pub color: Option<Color>,
14    /// Explicit size in px; `0` inherits the paint context's font size.
15    pub size_px: u16,
16    bounds: core::cell::Cell<Rect>,
17}
18
19impl Label {
20    pub fn new(text: impl Into<String>) -> Self {
21        Self {
22            text: text.into(),
23            color: None,
24            size_px: 0,
25            bounds: core::cell::Cell::new(Rect::ZERO),
26        }
27    }
28
29    pub fn with_color(mut self, color: Color) -> Self {
30        self.color = Some(color);
31        self
32    }
33
34    /// Measure with the font and size used to paint this label.
35    pub fn measure(&self, font: &mut crate::font_context::FontContext) -> Size {
36        let px = if self.size_px > 0 {
37            self.size_px as f32
38        } else {
39            font.size_px
40        };
41        Size {
42            w: font.atlas.measure(&self.text, px),
43            h: font.atlas.line_height(px),
44        }
45    }
46}
47
48impl ViWidget for Label {
49    fn layout(
50        &self,
51        constraints: Constraints,
52        font: &mut crate::font_context::FontContext,
53    ) -> LayoutNode {
54        let desired = self.measure(font);
55        let size = constraints.constrain(desired);
56        let bounds = Rect::from_origin_size(constraints.origin, size);
57        self.bounds.set(bounds);
58        LayoutNode::leaf(bounds)
59    }
60
61    fn paint(&self, cx: &mut PaintCx) {
62        let pos = self.bounds.get().origin().offset(cx.origin.x, cx.origin.y);
63        let color = self.color.unwrap_or_else(|| cx.theme.text_primary());
64        cx.draw_text(pos, &self.text, color, self.size_px as f32);
65    }
66
67    fn event(&mut self, _cx: &mut EventCx, _e: &Event) -> EventStatus {
68        EventStatus::Ignored
69    }
70}

72#[cfg(test)]
73mod tests {
74    use super::Label;
75
76    use crate::canvas::Color;
77    use crate::font_context::FontContext;
78    use crate::layout::{Constraints, Size};
79    use crate::node_widgets::style::test_support::{record, text_color};
80    use crate::theme::{ViTheme, DARK_THEME, LIGHT_THEME};
81    use crate::widget::{PaintCx, ViWidget};
82
83    fn paint(label: &Label, theme: &dyn ViTheme) -> crate::gpu_cmd::GpuCommandBuffer {
84        let mut font = FontContext::default();
85        label.layout(Constraints::root(Size::new(200.0, 40.0)), &mut font);
86        record(|canvas| label.paint(&mut PaintCx::with_theme(canvas, theme, &mut font)))
87    }
88
89    #[test]
90    fn text_follows_the_theme_unless_overridden() {
91        for theme in [&LIGHT_THEME as &dyn ViTheme, &DARK_THEME] {
92            let buf = paint(&Label::new("Hello"), theme);
93            assert_eq!(text_color(&buf, "Hello"), Some(theme.text_primary()));
94            let red = Color::rgb(200, 0, 0);
95            let buf = paint(&Label::new("Hello").with_color(red), theme);
96            assert_eq!(text_color(&buf, "Hello"), Some(red));
97        }
98    }
