14/// An individual formatted span of text within a rich text view.
15#[derive(Clone, Debug, PartialEq)]
16pub struct TextSpan {
17    pub text: String,
18    pub color: Option<Color>,
19    pub bg_color: Option<Color>,
20    pub bold: bool,
21    pub italic: bool,
22    pub code: bool,
23    pub size_px: Option<f32>,
24    pub link: Option<String>,
25}
26
27impl TextSpan {
28    pub fn plain(text: impl Into<String>) -> Self {
29        Self {
30            text: text.into(),
31            color: None,
32            bg_color: None,
33            bold: false,
34            italic: false,
35            code: false,
36            size_px: None,
37            link: None,
38        }
39    }
40
41    pub fn bold(text: impl Into<String>) -> Self {
42        let mut s = Self::plain(text);
43        s.bold = true;
44        s
45    }
46
47    pub fn italic(text: impl Into<String>) -> Self {
48        let mut s = Self::plain(text);
49        s.italic = true;
50        s
51    }
52
53    pub fn code(text: impl Into<String>) -> Self {
54        Self {
55            text: text.into(),
56            color: Some(Color::CYAN),
57            bg_color: Some(Color::bgra(45, 38, 30, 255)), // Subtle dark highlight
58            bold: false,
59            italic: false,
60            code: true,
61            size_px: None,
62            link: None,
63        }
64    }
65
66    pub fn link(text: impl Into<String>, url: impl Into<String>) -> Self {
67        Self {
68            text: text.into(),
69            color: Some(Color::bgra(240, 160, 60, 255)), // Vibrant accent link blue
70            bg_color: None,
71            bold: false,
72            italic: false,
73            code: false,
74            size_px: None,
75            link: Some(url.into()),
76        }
77    }
78
79    pub fn with_color(mut self, color: Color) -> Self {
80        self.color = Some(color);
81        self
82    }
83
84    pub fn with_bg_color(mut self, bg: Color) -> Self {
85        self.bg_color = Some(bg);
86        self
87    }
88
89    pub fn with_size(mut self, size_px: f32) -> Self {
90        self.size_px = Some(size_px);
91        self
92    }
93}

106/// RichTextView displays styled text spans with automatic word wrapping.
107pub struct RichTextView {
108    pub spans: Vec<TextSpan>,
109    pub line_spacing: f32,
110    pub default_color: Option<Color>,
111    bounds: core::cell::Cell<Rect>,
112    layout_cache: RefCell<Vec<LaidOutWord>>,
113}

115impl RichTextView {

120            default_color: None,

123        }
124    }

144    /// Internal layout calculation for wrapping spans within a given max width.
145    pub(crate) fn layout_spans_into(
146        spans: &[TextSpan],
147        origin: Point,
148        max_w: f32,
149        line_spacing: f32,
150        default_color: Color,
151        font: &mut crate::font_context::FontContext,
152        out_words: &mut Vec<LaidOutWord>,
153    ) -> Size {
154        out_words.clear();
155
156        let base_font_size = font.size_px;
157        let mut cur_x = 0.0f32;
158        let mut cur_y = 0.0f32;
159        let mut max_row_w = 0.0f32;
160        let mut cur_row_h = font.atlas.line_height(base_font_size);
161
162        for span in spans {
163            let size_px = span.size_px.unwrap_or(base_font_size);
164            let line_h = font.atlas.line_height(size_px);
165            if line_h > cur_row_h {
166                cur_row_h = line_h;
167            }
168
169            let text_color = span.color.unwrap_or(default_color);
170            let is_link = span.link.is_some();
171
172            // Handle newline or split words
173            for paragraph_part in span.text.split('\n') {
174                let words: Vec<&str> = paragraph_part.split(' ').collect();
175                let word_count = words.len();
176
177                for (idx, &word) in words.iter().enumerate() {
178                    if word.is_empty() {
179                        if idx + 1 < word_count {
180                            // Empty space between multiple spaces
181                            let space_w = font.atlas.measure(" ", size_px);
182                            cur_x += space_w;
183                        }
184                        continue;
185                    }
186
187                    let word_w = font.atlas.measure(word, size_px);
188                    let space_w = if idx + 1 < word_count {
189                        font.atlas.measure(" ", size_px)
190                    } else {
191                        0.0
192                    };
193
194                    // Line wrap check
195                    if cur_x > 0.0 && (cur_x + word_w) > max_w {
196                        cur_x = 0.0;
197                        cur_y += cur_row_h + line_spacing;
198                        cur_row_h = line_h;
199                    }
200
201                    let word_rect = Rect::new(
202                        origin.x + cur_x,
203                        origin.y + cur_y,
204                        word_w,
205                        line_h,
206                    );
207
208                    out_words.push(LaidOutWord {
209                        rect: word_rect,
210                        text: word.to_string(),
211                        color: text_color,
212                        bg_color: span.bg_color,
213                        size_px,
214                        _is_code: span.code,
215                        is_link,
216                    });
217
218                    cur_x += word_w + space_w;
219                    if cur_x > max_row_w {
220                        max_row_w = cur_x;
221                    }
222                }
223            }
224        }
225
226        let total_h = if out_words.is_empty() {
227            0.0
228        } else {
229            cur_y + cur_row_h
230        };
231
232        Size {
233            w: max_row_w.min(max_w),
234            h: total_h,
235        }
236    }
237}

239impl ViWidget for RichTextView {
240    fn layout(&self, constraints: Constraints, font: &mut crate::font_context::FontContext) -> LayoutNode {
241        let mut words = self.layout_cache.borrow_mut();
242        let fallback_color = self.default_color.unwrap_or(Color::WHITE);
243        let desired = Self::layout_spans_into(
244            &self.spans,
245            constraints.origin,
246            constraints.max.w,
247            self.line_spacing,
248            fallback_color,
249            font,
250            &mut words,
251        );
252
253        let size = constraints.constrain(desired);
254        let bounds = Rect::from_origin_size(constraints.origin, size);
255        self.bounds.set(bounds);
256        LayoutNode::leaf(bounds)
257    }
258
259    fn paint(&self, cx: &mut PaintCx) {
260        let words = self.layout_cache.borrow();
261        for word in words.iter() {
262            let screen_rect = cx.screen_rect(word.rect);
263
264            // Draw background badge (e.g. for inline code)
265            if let Some(bg) = word.bg_color {
266                let pad = 2.0;
267                let badge_rect = Rect::new(
268                    screen_rect.x - pad,
269                    screen_rect.y,
270                    screen_rect.w + pad * 2.0,
271                    screen_rect.h,
272                );
273                cx.canvas.fill_rect(badge_rect, bg);
274            }
275
276            // Draw text
277            let pos = screen_rect.origin();
278            cx.draw_text(pos, &word.text, word.color, word.size_px);
279
280            // Draw link underline
281            if word.is_link {
282                let underline_y = screen_rect.y + screen_rect.h - 1.0;
283                let p1 = Point::new(screen_rect.x, underline_y);
284                let p2 = Point::new(screen_rect.x + screen_rect.w, underline_y);
285                cx.canvas.draw_line(p1, p2, word.color);
286            }
287        }
288    }
289
290    fn event(&mut self, _cx: &mut EventCx, _e: &Event) -> EventStatus {
291        EventStatus::Ignored
292    }
293}

300    #[test]
301    fn test_text_span_constructors() {
302        let s1 = TextSpan::plain("Hello");
303        assert_eq!(s1.text, "Hello");
304        assert!(!s1.bold);
305        assert!(!s1.code);
306
307        let s2 = TextSpan::bold("World");
308        assert!(s2.bold);
309
310        let s3 = TextSpan::code("cargo check");
311        assert!(s3.code);
312        assert_eq!(s3.color, Some(Color::CYAN));
313
314        let s4 = TextSpan::link("CellOS", "https://cellos.org");
315        assert_eq!(s4.link.as_deref(), Some("https://cellos.org"));
316    }
317
318    #[test]
319    fn test_rich_text_wrapping() {
320        let mut font = FontContext::default();
321        let spans = alloc::vec![
322            TextSpan::plain("The quick brown fox jumps over the lazy dog."),
323            TextSpan::code("fn main()"),
324        ];
325        let view = RichTextView::new(spans);
326
327        // Narrow constraint should wrap into multiple rows
328        let constraints = Constraints::new(Point::ZERO, Size::new(100.0, 500.0));
329        let node = view.layout(constraints, &mut font);
330
331        assert!(node.bounds.w <= 100.0);
332        assert!(node.bounds.h > font.line_height());
333        assert!(view.layout_cache.borrow().len() >= 9);
334    }
