14/// An individual formatted span of text within a rich text view.
15///
16/// `None` colours resolve against the active theme at paint time: text uses
17/// the view's default (theme `text_primary`), links `reader_link`, and code
18/// spans get a `button_normal` chip.
19#[derive(Clone, Debug, PartialEq)]
20pub struct TextSpan {
21    pub text: String,
22    pub color: Option<Color>,
23    pub bg_color: Option<Color>,
24    pub bold: bool,
25    pub italic: bool,
26    pub code: bool,
27    pub size_px: Option<f32>,
28    pub link: Option<String>,
29}
30
31impl TextSpan {
32    pub fn plain(text: impl Into<String>) -> Self {
33        Self {
34            text: text.into(),
35            color: None,
36            bg_color: None,
37            bold: false,
38            italic: false,
39            code: false,
40            size_px: None,
41            link: None,
42        }
43    }
44
45    pub fn bold(text: impl Into<String>) -> Self {
46        let mut s = Self::plain(text);
47        s.bold = true;
48        s
49    }
50
51    pub fn italic(text: impl Into<String>) -> Self {
52        let mut s = Self::plain(text);
53        s.italic = true;
54        s
55    }
56
57    pub fn code(text: impl Into<String>) -> Self {
58        let mut s = Self::plain(text);
59        s.code = true;
60        s
61    }
62
63    pub fn link(text: impl Into<String>, url: impl Into<String>) -> Self {
64        let mut s = Self::plain(text);
65        s.link = Some(url.into());
66        s
67    }
68
69    pub fn with_color(mut self, color: Color) -> Self {
70        self.color = Some(color);
71        self
72    }
73
74    pub fn with_bg_color(mut self, bg: Color) -> Self {
75        self.bg_color = Some(bg);
76        self
77    }
78
79    pub fn with_size(mut self, size_px: f32) -> Self {
80        self.size_px = Some(size_px);
81        self
82    }
83}

97/// RichTextView displays styled text spans with automatic word wrapping.
98pub struct RichTextView {
99    pub spans: Vec<TextSpan>,
100    pub line_spacing: f32,
101    /// Colour for spans without their own; `None` uses the theme's `text_primary`.
102    pub default_color: Option<Color>,
103    bounds: core::cell::Cell<Rect>,
104    layout_cache: RefCell<Vec<LaidOutWord>>,
105}

136    /// Internal layout calculation for wrapping spans within a given max width.
137    pub(crate) fn layout_spans_into(
138        spans: &[TextSpan],
139        origin: Point,
140        max_w: f32,
141        line_spacing: f32,
142        font: &mut crate::font_context::FontContext,
143        out_words: &mut Vec<LaidOutWord>,
144    ) -> Size {
145        out_words.clear();
146
147        let base_font_size = font.size_px;
148        let mut cur_x = 0.0f32;
149        let mut cur_y = 0.0f32;
150        let mut max_row_w = 0.0f32;
151        let mut cur_row_h = font.atlas.line_height(base_font_size);
152
153        for span in spans {
154            let size_px = span.size_px.unwrap_or(base_font_size);
155            let line_h = font.atlas.line_height(size_px);
156            if line_h > cur_row_h {
157                cur_row_h = line_h;
158            }
159
160            let is_link = span.link.is_some();
161
162            // Handle newline or split words
163            for paragraph_part in span.text.split('\n') {
164                let words: Vec<&str> = paragraph_part.split(' ').collect();
165                let word_count = words.len();
166
167                for (idx, &word) in words.iter().enumerate() {
168                    if word.is_empty() {
169                        if idx + 1 < word_count {
170                            // Empty space between multiple spaces
171                            let space_w = font.atlas.measure(" ", size_px);
172                            cur_x += space_w;
173                        }
174                        continue;
175                    }
176
177                    let word_w = font.atlas.measure(word, size_px);
178                    let space_w = if idx + 1 < word_count {
179                        font.atlas.measure(" ", size_px)
180                    } else {
181                        0.0
182                    };
183
184                    // Line wrap check
185                    if cur_x > 0.0 && (cur_x + word_w) > max_w {
186                        cur_x = 0.0;
187                        cur_y += cur_row_h + line_spacing;
188                        cur_row_h = line_h;
189                    }
190
191                    let word_rect = Rect::new(origin.x + cur_x, origin.y + cur_y, word_w, line_h);
192
193                    out_words.push(LaidOutWord {
194                        rect: word_rect,
195                        text: word.to_string(),
196                        color: span.color,
197                        bg_color: span.bg_color,
198                        size_px,
199                        is_code: span.code,
200                        is_link,
201                    });
202
203                    cur_x += word_w + space_w;
204                    if cur_x > max_row_w {
205                        max_row_w = cur_x;
206                    }
207                }
208            }
209        }
210
211        let total_h = if out_words.is_empty() {
212            0.0
213        } else {
214            cur_y + cur_row_h
215        };
216
217        Size {
218            w: max_row_w.min(max_w),
219            h: total_h,
220        }
221    }

223    /// Paint laid-out words; `fallback` colours spans without their own.
224    pub(crate) fn paint_words(words: &[LaidOutWord], fallback: Color, cx: &mut PaintCx) {
225        for word in words {
226            let screen_rect = cx.screen_rect(word.rect);
227            let color = word.color.unwrap_or(if word.is_link {
228                cx.theme.reader_link()
229            } else {
230                fallback
231            });
232
233            // Background chip (explicit, or the themed inline-code chip)
234            let bg = word
235                .bg_color
236                .or_else(|| word.is_code.then(|| cx.theme.button_normal()));
237            if let Some(bg) = bg {
238                let pad = 2.0;
239                let badge_rect = Rect::new(
240                    screen_rect.x - pad,
241                    screen_rect.y,
242                    screen_rect.w + pad * 2.0,
243                    screen_rect.h,
244                );
245                cx.canvas.fill_rect(badge_rect, bg);
246            }
247
248            cx.draw_text(screen_rect.origin(), &word.text, color, word.size_px);
249
250            if word.is_link {
251                let underline_y = screen_rect.y + screen_rect.h - 1.0;
252                let p1 = Point::new(screen_rect.x, underline_y);
253                let p2 = Point::new(screen_rect.x + screen_rect.w, underline_y);
254                cx.canvas.draw_line(p1, p2, color);
255            }
256        }
257    }
258}

260impl ViWidget for RichTextView {
261    fn layout(
262        &self,
263        constraints: Constraints,
264        font: &mut crate::font_context::FontContext,
265    ) -> LayoutNode {
266        let mut words = self.layout_cache.borrow_mut();
267        let desired = Self::layout_spans_into(
268            &self.spans,
269            constraints.origin,
270            constraints.max.w,
271            self.line_spacing,
272            font,
273            &mut words,
274        );
275
276        let size = constraints.constrain(desired);
277        let bounds = Rect::from_origin_size(constraints.origin, size);
278        self.bounds.set(bounds);
279        LayoutNode::leaf(bounds)
280    }
281
282    fn paint(&self, cx: &mut PaintCx) {
283        let words = self.layout_cache.borrow();
284        let fallback = self
285            .default_color
286            .unwrap_or_else(|| cx.theme.text_primary());
287        Self::paint_words(&words, fallback, cx);
288    }
289
290    fn event(&mut self, _cx: &mut EventCx, _e: &Event) -> EventStatus {
291        EventStatus::Ignored
292    }
293}

300    #[test]
301    fn test_text_span_constructors() {
302        let s1 = TextSpan::plain("Hello");
303        assert_eq!(s1.text, "Hello");
304        assert!(!s1.bold);
305        assert!(!s1.code);
306
307        let s2 = TextSpan::bold("World");
308        assert!(s2.bold);
309
310        let s3 = TextSpan::code("cargo check");
311        assert!(s3.code);
312        assert_eq!(s3.color, None);
313
314        let s4 = TextSpan::link("CellOS", "https://cellos.org");
315        assert_eq!(s4.link.as_deref(), Some("https://cellos.org"));
316    }

318    #[test]
319    fn unstyled_spans_resolve_colours_from_the_theme() {
320        use crate::node_widgets::style::test_support::{fills, lines, record, text_color};
321        use crate::theme::{ViTheme, DARK_THEME, LIGHT_THEME};
322
323        for theme in [&LIGHT_THEME as &dyn ViTheme, &DARK_THEME] {
324            let view = RichTextView::new(alloc::vec![
325                TextSpan::plain("plain "),
326                TextSpan::code("code "),
327                TextSpan::link("link", "https://cellos.org"),
328            ]);
329            let mut font = FontContext::default();
330            view.layout(
331                Constraints::new(Point::ZERO, Size::new(400.0, 100.0)),
332                &mut font,
333            );
334            let buf =
335                record(|canvas| view.paint(&mut PaintCx::with_theme(canvas, theme, &mut font)));
336            assert_eq!(text_color(&buf, "plain"), Some(theme.text_primary()));
337            assert_eq!(text_color(&buf, "code"), Some(theme.text_primary()));
338            assert_eq!(fills(&buf)[0].1, theme.button_normal());
339            assert_eq!(text_color(&buf, "link"), Some(theme.reader_link()));
340            assert_eq!(lines(&buf), [theme.reader_link()]);
341
342            let red = Color::rgb(200, 0, 0);
343            let view =
344                RichTextView::new(alloc::vec![TextSpan::plain("plain")]).with_default_color(red);
345            view.layout(
346                Constraints::new(Point::ZERO, Size::new(400.0, 100.0)),
347                &mut font,
348            );
349            let buf =
350                record(|canvas| view.paint(&mut PaintCx::with_theme(canvas, theme, &mut font)));
351            assert_eq!(text_color(&buf, "plain"), Some(red));
352        }
353    }

355    #[test]
356    fn test_rich_text_wrapping() {
357        let mut font = FontContext::default();
358        let spans = alloc::vec![
359            TextSpan::plain("The quick brown fox jumps over the lazy dog."),
360            TextSpan::code("fn main()"),
361        ];
362        let view = RichTextView::new(spans);
363
364        // Narrow constraint should wrap into multiple rows
365        let constraints = Constraints::new(Point::ZERO, Size::new(100.0, 500.0));
366        let node = view.layout(constraints, &mut font);
367
368        assert!(node.bounds.w <= 100.0);
369        assert!(node.bounds.h > font.line_height());
370        assert!(view.layout_cache.borrow().len() >= 9);
371    }
