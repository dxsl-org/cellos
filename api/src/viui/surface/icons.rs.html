1//! ViSurface wrappers over the CellOS Linear icon set
2//! (`ostd::display::linear::icons`).
3
4use ostd::display::ViSurface;
5
6pub use ostd::display::linear::icons::{paint_icon, LineIcon, ICON_SIZE};
7
8use super::{bgra, linear_canvas};
9use crate::canvas::Color;
10
11/// Draw `icon` with its grid's top-left at `(x, y)`.
12pub fn draw_icon(surf: &mut ViSurface, icon: LineIcon, x: i32, y: i32, color: Color) {
13    paint_icon(&mut linear_canvas(surf), icon, x, y, bgra(color));
14}
15
16/// Draw `icon` centred in the box `(x, y, w, h)`, snapped to whole pixels.
17pub fn draw_icon_centered(
18    surf: &mut ViSurface,
19    icon: LineIcon,
20    (x, y, w, h): (i32, i32, u32, u32),
21    color: Color,
22) {
23    let left = x + (w.saturating_sub(ICON_SIZE) / 2) as i32;
24    let top = y + (h.saturating_sub(ICON_SIZE) / 2) as i32;
25    draw_icon(surf, icon, left, top, color);
26}
