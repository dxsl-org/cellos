1//! The GuC command transport (CT) as the driver runs it: two buffer objects in device
2//! memory, registered and enabled over MMIO, then H2G messages written into the H2G ring
3//! and G2H events drained from the G2H ring by polling.
4//!
5//! Upstream reference (MIT): `drivers/gpu/drm/xe/xe_guc_ct.c` (`xe_guc_ct_init` object
6//! layout, `guc_ct_ctb_h2g_register`, `guc_ct_ctb_g2h_register`,
7//! `guc_ct_control_toggle`, `__xe_guc_ct_start`, `h2g_write`, `__guc_ct_send_locked`,
8//! `g2h_read`), `abi/guc_communication_ctb_abi.h` (descriptor and message formats).
9//! The ring arithmetic is [`intel_gpu_core::ct`]. Interrupts are not used: the driver
10//! drains the G2H ring whenever it needs an answer or touches a queue.
11

21};
22use intel_gpu_core::guc_abi::{

27};

29/// `xe_guc_ct_queue_proc_time_jiffies`: a full 16 KiB H2G ring takes about 4 s to drain,
30/// so a sender waits that long for space before declaring the channel dead.
31pub const H2G_SPACE_TIMEOUT_US: u64 = 4_000_000;
32
33/// Why a CT operation failed.
34#[derive(Clone, Copy, Debug, PartialEq, Eq)]
35pub enum CtFailure {
36    /// The buffers could not be allocated.
37    NoMemory,
38    /// The MMIO registration or enable failed.
39    Setup(GucError),
40    /// A descriptor reports an error status, or a pointer is out of range.
41    Broken(u32),
42    /// The GuC did not make room in the H2G ring in time.
43    NoSpace,
44}
45
46/// One enabled CT channel.
47pub struct GucCt {
48    h2g: Region,
49    g2h: Region,
50    /// CPU copy of the H2G ring; the ring code writes here and changed dwords are copied
51    /// to device memory before the descriptor tail publishes them.
52    shadow: Vec<u32>,
53    tail: u32,
54    g2h_head: u32,
55    next_fence: u16,
56}
57
58impl GucCt {
59    /// `xe_guc_ct_init` + `__xe_guc_ct_start(needs_register)`: allocate both objects
60    /// (descriptors at 0 and 2 KiB, rings at 4 KiB), register their GGTT addresses and
61    /// sizes with `HOST2GUC_SELF_CFG`, then enable with `HOST2GUC_CONTROL_CTB`.
62    pub fn enable(
63        pool: &mut Pool,
64        mem: &mut impl DmaMemory,
65        regs: &mut impl RegisterFile,
66        ggtt: &mut Ggtt,
67        clock: &mut impl Clock,
68    ) -> Result<Self, CtFailure> {
69        let pages = |bytes: u32| (CTB_BUFFER_OFFSET + bytes) as usize / PAGE;
70        let h2g = pool
71            .alloc(mem, regs, ggtt, pages(CTB_H2G_BUFFER_SIZE))
72            .ok_or(CtFailure::NoMemory)?;
73        let g2h = pool
74            .alloc(mem, regs, ggtt, pages(CTB_G2H_BUFFER_SIZE))
75            .ok_or(CtFailure::NoMemory)?;
76        let self_cfg = [
77            (KLV_H2G_CTB_DESCRIPTOR_ADDR, 2, u64::from(h2g.ggtt)),
78            (KLV_H2G_CTB_ADDR, 2, u64::from(h2g.ggtt + CTB_BUFFER_OFFSET)),
79            (KLV_H2G_CTB_SIZE, 1, u64::from(CTB_H2G_BUFFER_SIZE)),
80            (
81                KLV_G2H_CTB_DESCRIPTOR_ADDR,
82                2,
83                u64::from(g2h.ggtt + CTB_G2H_DESC_OFFSET),
84            ),
85            (KLV_G2H_CTB_ADDR, 2, u64::from(g2h.ggtt + CTB_BUFFER_OFFSET)),
86            (KLV_G2H_CTB_SIZE, 1, u64::from(CTB_G2H_BUFFER_SIZE)),
87        ];
88        for (key, len, value) in self_cfg {
89            // `guc_self_cfg`: the GuC answers with the number of KLVs it accepted, 1.
90            match mmio_send(regs, clock, &guc_abi::self_cfg(key, len, value)) {
91                Ok(1) => {}
92                Ok(n) => return Err(CtFailure::Setup(GucError::MmioProtocol(n))),
93                Err(e) => return Err(CtFailure::Setup(e)),
94            }
95        }
96        match mmio_send(regs, clock, &guc_abi::control_ctb(true)) {
97            Ok(0) => {}
98            Ok(n) => return Err(CtFailure::Setup(GucError::MmioProtocol(n))),
99            Err(e) => return Err(CtFailure::Setup(e)),
100        }
101        Ok(Self {
102            h2g,
103            g2h,
104            shadow: vec![0; ct::H2G_DWORDS as usize],
105            tail: 0,
106            g2h_head: 0,
107            next_fence: 0,
108        })
109    }
110
111    /// GGTT addresses of the H2G and G2H objects.
112    #[cfg(test)]
113    pub fn ggtt(&self) -> (u32, u32) {
114        (self.h2g.ggtt, self.g2h.ggtt)
115    }
116
117    fn h2g_desc(&self, field: usize) -> u64 {
118        self.h2g.iova + (field * 4) as u64
119    }
120
121    fn g2h_desc(&self, field: usize) -> u64 {
122        self.g2h.iova + u64::from(CTB_G2H_DESC_OFFSET) + (field * 4) as u64
123    }
124
125    /// `__guc_ct_send_locked` for a message no response is expected to (a
126    /// `FAST_REQUEST`, fence marked untracked): write it, publish the tail, notify.
127    pub fn send(
128        &mut self,
129        mem: &mut impl DmaMemory,
130        regs: &mut impl RegisterFile,
131        clock: &mut impl Clock,
132        action: &[u32],
133    ) -> Result<(), CtFailure> {
134        let fence = self.next_fence & CT_SEQNO_MASK | CT_SEQNO_UNTRACKED;
135        self.next_fence = self.next_fence.wrapping_add(1);
136        let deadline = clock.now_us().saturating_add(H2G_SPACE_TIMEOUT_US);
137        loop {
138            let status = read_u32(mem, self.h2g_desc(DESC_STATUS)).ok_or(CtFailure::NoMemory)?;
139            if status != 0 {
140                return Err(CtFailure::Broken(status));
141            }
142            let head = read_u32(mem, self.h2g_desc(DESC_HEAD)).ok_or(CtFailure::NoMemory)?;
143            let old = self.tail;
144            match ct::h2g_write(
145                &mut self.shadow,
146                head,
147                old,
148                fence,
149                HxgType::FastRequest,
150                action,
151            ) {
152                Ok(tail) => {
153                    self.publish(mem, old, tail)?;
154                    return notify(regs).map_err(CtFailure::Setup);
155                }
156                // The message would cross the end: the rest was zero-filled (NOPs); publish
157                // the wrapped tail and try again from 0.
158                Err(CtError::Wrapped { new_tail }) => {
159                    self.publish(mem, old, new_tail)?;
160                }
161                Err(CtError::NoSpace) if clock.now_us() < deadline => clock.relax(1_000),
162                Err(CtError::NoSpace) => return Err(CtFailure::NoSpace),
163                Err(_) => return Err(CtFailure::Broken(head)),
164            }
165        }
166    }
167
168    /// Copy the shadow dwords `[old, new)` (or `[old, end)` on a wrap to 0) to the ring,
169    /// then move the descriptor tail: the GuC must never see a tail ahead of its data.
170    fn publish(&mut self, mem: &mut impl DmaMemory, old: u32, new: u32) -> Result<(), CtFailure> {
171        let end = if new > old { new } else { ct::H2G_DWORDS };
172        let ring = self.h2g.iova + u64::from(CTB_BUFFER_OFFSET);
173        if !write_u32s(
174            mem,
175            ring + u64::from(old) * 4,
176            &self.shadow[old as usize..end as usize],
177        ) || !write_u32s(mem, self.h2g_desc(DESC_TAIL), &[new])
178        {
179            return Err(CtFailure::NoMemory);
180        }
181        self.tail = new;
182        Ok(())
183    }
184
185    /// `g2h_read` until the ring is empty: decode every message and hand back the events.
186    pub fn receive(&mut self, mem: &mut impl DmaMemory) -> Result<Vec<G2hEvent>, CtFailure> {
187        let size = ct::G2H_DWORDS;
188        let ring = self.g2h.iova + u64::from(CTB_BUFFER_OFFSET);
189        let status = read_u32(mem, self.g2h_desc(DESC_STATUS)).ok_or(CtFailure::NoMemory)?;
190        if status != 0 {
191            return Err(CtFailure::Broken(status));
192        }
193        let tail = read_u32(mem, self.g2h_desc(DESC_TAIL)).ok_or(CtFailure::NoMemory)?;
194        if tail >= size {
195            return Err(CtFailure::Broken(tail));
196        }
197        let mut events = Vec::new();
198        while self.g2h_head != tail {
199            let available = tail.wrapping_sub(self.g2h_head) & (size - 1);
200            let at = |i: u32| ring + u64::from((self.g2h_head + i) % size) * 4;
201            let header = read_u32(mem, at(0)).ok_or(CtFailure::NoMemory)?;
202            let len = header & 0xff;
203            if header >> 12 & 0xf != 0 || len as usize > MAX_MSG_DWORDS || len + 1 > available {
204                return Err(CtFailure::Broken(header));
205            }
206            let mut hxg = Vec::with_capacity(len as usize);
207            for i in 1..=len {
208                hxg.push(read_u32(mem, at(i)).ok_or(CtFailure::NoMemory)?);
209            }
210            events.push(decode_g2h(&hxg));
211            self.g2h_head = (self.g2h_head + 1 + len) % size;
212        }
213        write_u32s(mem, self.g2h_desc(DESC_HEAD), &[self.g2h_head])
214            .then_some(events)
215            .ok_or(CtFailure::NoMemory)
216    }
