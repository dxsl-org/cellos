1//! Intel GT forcewake acquisition and release.
2//!
3//! Upstream reference: `drivers/gpu/drm/xe/xe_force_wake.c` (MIT): `__domain_wait` waits
4//! for the ack with `xe_mmio_wait32` for `XE_FORCE_WAKE_ACK_TIMEOUT_MS` (50 ms).
5
6use crate::clock::Clock;
7use crate::regs::{
8    wait32, RegisterFile, FORCEWAKE_ACK_BIT, FORCEWAKE_ACK_GT, FORCEWAKE_CLEAR, FORCEWAKE_GT,
9    FORCEWAKE_REQ,
10};
11use types::{ViError, ViResult};
12
13/// `XE_FORCE_WAKE_ACK_TIMEOUT_MS` in microseconds.
14pub const FORCEWAKE_ACK_TIMEOUT_US: u64 = 50_000;
15
16/// Acquire GT forcewake domain, preventing the GT from entering sleep states.
17pub fn forcewake_gt_acquire(regs: &mut impl RegisterFile, clock: &mut impl Clock) -> ViResult<()> {
18    regs.write32(FORCEWAKE_GT, FORCEWAKE_REQ)?;
19    wait_ack(regs, clock, true)
20}
21
22/// Release GT forcewake domain, allowing the GT to enter power-saving states.
23pub fn forcewake_gt_release(regs: &mut impl RegisterFile, clock: &mut impl Clock) -> ViResult<()> {
24    regs.write32(FORCEWAKE_GT, FORCEWAKE_CLEAR)?;
25    wait_ack(regs, clock, false)
26}
27
28fn wait_ack(regs: &impl RegisterFile, clock: &mut impl Clock, awake: bool) -> ViResult<()> {
29    let expected = if awake { FORCEWAKE_ACK_BIT } else { 0 };
30    let ack = wait32(
31        regs,
32        clock,
33        FORCEWAKE_ACK_GT,
34        FORCEWAKE_ACK_BIT,
35        expected,
36        FORCEWAKE_ACK_TIMEOUT_US,
37    )
38    .map_err(|_| ViError::IO)?;
39    // An unpowered or unmapped BAR reads all-ones, which would otherwise pass as an ack.
40    if ack == u32::MAX {
41        return Err(ViError::IO);
42    }
43    Ok(())
44}
