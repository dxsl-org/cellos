1//! Intel GuC firmware load: WOPCM partition, reset, parameter block, transfer setup,
2//! uCode DMA, boot wait, and the MMIO host-to-GuC channel.
3//!
4//! Upstream reference (MIT): `drivers/gpu/drm/xe/xe_guc.c` (`xe_guc_reset`,
5//! `guc_write_params`, `guc_prepare_xfer`, `guc_xfer_rsa`, `guc_load_done`,
6//! `guc_wait_ucode`, `__xe_guc_upload`, `xe_guc_mmio_send_recv`, `xe_guc_notify`),
7//! `xe_uc_fw.c` (`uc_fw_xfer`), `xe_wopcm.c` (`xe_wopcm_init`), `xe_uc.c` (order:
8//! WOPCM at init, then sanitize/reset, then upload), `regs/xe_guc_regs.h` and
9//! `abi/guc_errors_abi.h`.
10//!
11//! Every wait is bounded by time through [`Clock`], with upstream's limits: GDRST 5 ms,
12//! uCode DMA 100 ms, GuC boot polled every 10 ms for up to 3 s (`GUC_LOAD_TIMEOUT_SEC` of a
13//! non-debug build; a load above 200 ms is reported as slow), MMIO replies 50 ms.
14

35pub const GUC_STATUS_BOOTROM_MASK: u32 = 0x7F << 1;
36/// `GS_UKERNEL_MASK`, bits 15:8.
37pub const GUC_STATUS_UKERNEL_MASK: u32 = 0xFF << 8;
38/// `XE_GUC_LOAD_STATUS_READY` in the microkernel field.
39pub const GUC_STATUS_UKERNEL_READY: u32 = 0xF0 << 8;
40/// `GUC_WOPCM_SIZE`: size in bits 31:12, `GUC_WOPCM_SIZE_LOCKED` in bit 0.

42/// `GUC_SHIM_CONTROL`, programmed before the uCode DMA.
43pub const GUC_SHIM_CONTROL: usize = 0x00C064;

45pub const GUC_ENABLE_READ_CACHE_FOR_SRAM_DATA: u32 = 1 << 9;
46pub const GUC_ENABLE_READ_CACHE_FOR_WOPCM_DATA: u32 = 1 << 10;

64pub const GUC_DMA_CTRL: usize = 0x00C314;

103/// Why the GuC did not come up, or the MMIO channel failed.
104#[derive(Clone, Copy, Debug, PartialEq, Eq)]
105pub enum GucError {
106    /// An MMIO access failed.
107    Mmio,
108    /// The WOPCM cannot hold the image.
109    Wopcm(WopcmError),
110    /// The WOPCM registers did not read back locked with the written values
111    /// (`xe_mmio_write32_and_verify`); the value read.
112    WopcmNotLocked(u32),
113    /// `GRDOM_GUC` did not clear within 5 ms; the last `GDRST`.
114    ResetTimeout(u32),
115    /// The reset finished but the MIA core is not held in reset; `GUC_STATUS`.
116    NotInReset(u32),
117    /// An RSA key shorter than the scratch registers (`xe_uc_fw_copy_rsa` short copy).
118    RsaTooShort,
119    /// The image or its GGTT address does not fit the DMA registers.
120    BadImage,
121    /// `START_DMA` did not clear within 100 ms; `DMA_CTRL`.
122    DmaTimeout(u32),
123    /// `guc_load_done` saw a terminal failure code; `GUC_STATUS`.
124    LoadFailed(u32),
125    /// No terminal state within 3 s; the last `GUC_STATUS`.
126    LoadTimeout(u32),
127    /// No MMIO reply in time; the last `VF_SW_FLAG(0)`.
128    MmioTimeout(u32),
129    /// The GuC answered `RESPONSE_FAILURE`.
130    MmioFailure { hint: u32, error: u32 },
131    /// The GuC answered with something no request expects (`-EPROTO`).
132    MmioProtocol(u32),
133}

135impl From<WaitError> for GucError {
136    fn from(e: WaitError) -> Self {
137        match e {
138            WaitError::Mmio => GucError::Mmio,
139            WaitError::TimedOut(v) => GucError::MmioTimeout(v),
140        }
141    }
142}

144fn mmio<T>(r: types::ViResult<T>) -> Result<T, GucError> {
145    r.map_err(|_| GucError::Mmio)
146}

148/// `xe_wopcm_init` for a platform without HuC firmware: keep a partition firmware already
149/// locked if it fits the image, else compute one, write it, and require the hardware to
150/// report both registers locked with exactly those values (`__wopcm_init_regs`).
151pub fn init_wopcm(regs: &mut impl RegisterFile, guc_fw_size: u32) -> Result<WopcmLayout, GucError> {
152    let reg_offset = mmio(regs.read32(DMA_GUC_WOPCM_OFFSET))?;
153    let reg_size = mmio(regs.read32(GUC_WOPCM_SIZE))?;
154    if let Some(layout) = locked_layout(reg_offset, reg_size) {
155        // Preprogrammed by BIOS/IFWI: check against the largest WOPCM (`max_wopcm_size`).
156        check_layout(MAX_WOPCM_SIZE, layout, guc_fw_size, 0).map_err(GucError::Wopcm)?;
157        return Ok(layout);
158    }
159    let layout = compute_layout(LNL_WOPCM_SIZE, guc_fw_size, 0).map_err(GucError::Wopcm)?;
160
161    mmio(regs.write32(GUC_WOPCM_SIZE, layout.size))?;
162    let size = mmio(regs.read32(GUC_WOPCM_SIZE))?;
163    if size & (GUC_WOPCM_SIZE_MASK | GUC_WOPCM_SIZE_LOCKED) != layout.size | GUC_WOPCM_SIZE_LOCKED {
164        return Err(GucError::WopcmNotLocked(size));
165    }
166    // No HuC image is loaded, so `HUC_LOADING_AGENT_GUC` stays clear.
167    mmio(regs.write32(DMA_GUC_WOPCM_OFFSET, layout.base))?;
168    let base = mmio(regs.read32(DMA_GUC_WOPCM_OFFSET))?;
169    if base & (GUC_WOPCM_OFFSET_MASK | GUC_WOPCM_OFFSET_VALID)
170        != layout.base | GUC_WOPCM_OFFSET_VALID
171    {
172        return Err(GucError::WopcmNotLocked(base));
173    }
174    Ok(layout)
175}

177/// `xe_guc_reset`: reset the GuC domain, wait up to 5 ms for the bit to self-clear, then
178/// require the microcontroller held in reset.
179pub fn reset_guc(regs: &mut impl RegisterFile, clock: &mut impl Clock) -> Result<(), GucError> {
180    mmio(regs.write32(GDRST, GRDOM_GUC))?;
181    wait32(regs, clock, GDRST, GRDOM_GUC, 0, GDRST_TIMEOUT_US).map_err(|e| match e {
182        WaitError::TimedOut(v) => GucError::ResetTimeout(v),
183        WaitError::Mmio => GucError::Mmio,
184    })?;
185    let status = mmio(regs.read32(GUC_STATUS))?;
186    if status & GUC_STATUS_MIA_IN_RESET == 0 {
187        return Err(GucError::NotInReset(status));
188    }
189    Ok(())
190}

192/// `guc_write_params`: `SOFT_SCRATCH(0)` cleared, the block in `SOFT_SCRATCH(1..=14)`.
193pub fn write_params(
194    regs: &mut impl RegisterFile,
195    params: &[u32; GUC_CTL_MAX_DWORDS],
196) -> Result<(), GucError> {
197    mmio(regs.write32(SOFT_SCRATCH_0, 0))?;
198    for (i, &p) in params.iter().enumerate() {
199        mmio(regs.write32(SOFT_SCRATCH_0 + 4 * (1 + i), p))?;
200    }
201    Ok(())
202}

204/// `guc_prepare_xfer` for graphics 20+: shim caching and MOCS index, doorbells on, ARAT
205/// interrupts unmasked.
206pub fn prepare_xfer(regs: &mut impl RegisterFile) -> Result<(), GucError> {
207    let shim = GUC_ENABLE_READ_CACHE_LOGIC
208        | GUC_ENABLE_READ_CACHE_FOR_SRAM_DATA
209        | GUC_ENABLE_READ_CACHE_FOR_WOPCM_DATA
210        | GUC_ENABLE_MIA_CLOCK_GATING
211        | XE2_MOCS_UC_INDEX << GUC_MOCS_INDEX_SHIFT;
212    mmio(regs.write32(GUC_SHIM_CONTROL, shim))?;
213    mmio(regs.write32(GT_PM_CONFIG, GT_DOORBELL_ENABLE))?;
214    let mask = mmio(regs.read32(PMINTRMSK))?;
215    mmio(regs.write32(PMINTRMSK, mask & !ARAT_EXPIRED_INTRMSK))
216}

218/// `guc_xfer_rsa`: the boot ROM reads a key larger than the scratch registers from GGTT;
219/// a smaller one must fill all of them (`xe_uc_fw_copy_rsa` short copy is `-ENOMEM`).
220pub fn transfer_rsa(
221    regs: &mut impl RegisterFile,
222    rsa: &[u8],
223    rsa_ggtt: u64,
224) -> Result<(), GucError> {
225    const SCRATCH_BYTES: usize = UOS_RSA_SCRATCH_COUNT * 4;
226    if rsa.len() > SCRATCH_BYTES {
227        let addr = u32::try_from(rsa_ggtt).map_err(|_| GucError::BadImage)?;
228        return mmio(regs.write32(UOS_RSA_SCRATCH_0, addr));
229    }
230    if rsa.len() < SCRATCH_BYTES {
231        return Err(GucError::RsaTooShort);
232    }
233    for (i, word) in rsa.chunks_exact(4).enumerate() {
234        let word = u32::from_le_bytes([word[0], word[1], word[2], word[3]]);
235        mmio(regs.write32(UOS_RSA_SCRATCH_0 + i * 4, word))?;
236    }
237    Ok(())
238}

240/// `uc_fw_xfer`: DMA the CSS header plus uCode (`size` bytes at GGTT address `src_ggtt`)
241/// into WOPCM at `dst_wopcm`, waiting up to 100 ms for the engine.
242pub fn upload_guc_firmware(
243    regs: &mut impl RegisterFile,
244    clock: &mut impl Clock,
245    src_ggtt: u64,
246    dst_wopcm: u32,
247    size: usize,
248) -> Result<(), GucError> {
249    if size == 0 || !size.is_multiple_of(4) {
250        return Err(GucError::BadImage);
251    }
252    let size = u32::try_from(size).map_err(|_| GucError::BadImage)?;
253    // The high dword carries the address-space selector in bits 20:16.
254    let src_high = u32::try_from(src_ggtt >> 32)
255        .ok()
256        .filter(|high| *high <= 0xFFFF)
257        .ok_or(GucError::BadImage)?;
258
259    mmio(regs.write32(GUC_DMA_ADDR_0_LOW, src_ggtt as u32))?;
260    mmio(regs.write32(GUC_DMA_ADDR_0_HIGH, src_high | DMA_ADDRESS_SPACE_GGTT))?;
261    mmio(regs.write32(GUC_DMA_ADDR_1_LOW, dst_wopcm))?;
262    mmio(regs.write32(GUC_DMA_ADDR_1_HIGH, DMA_ADDRESS_SPACE_WOPCM))?;
263    mmio(regs.write32(GUC_DMA_COPY_SIZE, size))?;
264
265    let flags = GUC_DMA_CTRL_UOS_MOVE | GUC_DMA_CTRL_START;
266    mmio(regs.write32(GUC_DMA_CTRL, masked_write(flags, flags)))?;
267    let start = u32::from(GUC_DMA_CTRL_START);
268    let done = wait32(regs, clock, GUC_DMA_CTRL, start, 0, DMA_TIMEOUT_US);
269    // Upstream clears the move bit whether or not the copy finished.
270    mmio(regs.write32(GUC_DMA_CTRL, masked_write(0, GUC_DMA_CTRL_UOS_MOVE)))?;
271    done.map(|_| ()).map_err(|e| match e {
272        WaitError::TimedOut(v) => GucError::DmaTimeout(v),
273        WaitError::Mmio => GucError::Mmio,
274    })
275}

277/// What `guc_load_done` makes of one `GUC_STATUS` value.
278#[derive(Clone, Copy, Debug, PartialEq, Eq)]
279pub enum LoadState {
280    Running,
281    Failed,
282    InProgress,
283}
284
285/// `guc_load_done`: terminal microkernel codes (`XE_GUC_LOAD_STATUS_*`), then terminal
286/// boot ROM codes (`XE_BOOTROM_STATUS_*`), else still loading.
287pub fn guc_load_done(status: u32) -> LoadState {
288    let ukernel = (status & GUC_STATUS_UKERNEL_MASK) >> 8;
289    let bootrom = (status & GUC_STATUS_BOOTROM_MASK) >> 1;
290    if status & GUC_STATUS_UKERNEL_MASK == GUC_STATUS_UKERNEL_READY {
291        return LoadState::Running;
292    }
293    let ukernel_failed = matches!(
294        ukernel,
295        0x02..=0x04 | 0x07 | 0x08 | 0x60 | 0x70 | 0x71 | 0x73..=0x76
296    );
297    let bootrom_failed = matches!(
298        bootrom,
299        0x13 | 0x2B | 0x50 | 0x73 | 0x74 | 0x75 | 0x77 | 0x79 | 0x7A | 0x7E
300    );
301    if ukernel_failed || bootrom_failed {
302        LoadState::Failed
303    } else {
304        LoadState::InProgress
305    }
306}

308/// `guc_wait_ucode`: read `GUC_STATUS` every 10 ms until `guc_load_done` reports a terminal
309/// state, for at most 3 s. Returns how long the load took in microseconds.
310pub fn wait_guc_ukernel_running(
311    regs: &impl RegisterFile,
312    clock: &mut impl Clock,
313) -> Result<u64, GucError> {
314    let start = clock.now_us();
315    loop {
316        let status = mmio(regs.read32(GUC_STATUS))?;
317        match guc_load_done(status) {
318            LoadState::Running => return Ok(clock.now_us() - start),
319            LoadState::Failed => return Err(GucError::LoadFailed(status)),
320            LoadState::InProgress => {}
321        }
322        let elapsed = clock.now_us() - start;
323        if elapsed >= LOAD_TIMEOUT_US {
324            return Err(GucError::LoadTimeout(status));
325        }
326        clock.relax(LOAD_POLL_US.min(LOAD_TIMEOUT_US - elapsed));
327    }
328}

330/// `xe_guc_mmio_send_recv` on the primary GT: request dwords into `VF_SW_FLAG(0..n)`,
331/// a posting read of the last flag, notify, then wait for a reply with the GuC origin bit.
332/// Returns the reply's `DATA0`.
333pub fn mmio_send(
334    regs: &mut impl RegisterFile,
335    clock: &mut impl Clock,
336    request: &[u32],
337) -> Result<u32, GucError> {
338    debug_assert!(!request.is_empty() && request.len() <= VF_SW_FLAG_COUNT);
339    let started = clock.now_us();
340    let mut sleep = 1_000u64;
341    loop {
342        for (i, &word) in request.iter().enumerate() {
343            mmio(regs.write32(VF_SW_FLAG_0 + 4 * i, word))?;
344        }
345        mmio(regs.read32(VF_SW_FLAG_0 + 4 * (VF_SW_FLAG_COUNT - 1)))?;
346        notify(regs)?;
347
348        let origin = intel_gpu_core::guc_abi::HXG_ORIGIN_GUC;
349        let mut header = wait32(
350            regs,
351            clock,
352            VF_SW_FLAG_0,
353            origin,
354            origin,
355            MMIO_REPLY_TIMEOUT_US,
356        )?;
357        if decode_mmio_reply(header) == MmioReply::Busy {
358            // SUCCESS (7) and FAILURE (6) differ in one bit; wait for both upper type bits.
359            let final_bits = 6 << 28;
360            header = wait32(
361                regs,
362                clock,
363                VF_SW_FLAG_0,
364                final_bits,
365                final_bits,
366                MMIO_BUSY_TIMEOUT_US,
367            )?;
368        }
369        match decode_mmio_reply(header) {
370            MmioReply::Success(data0) => return Ok(data0),
371            MmioReply::Retry if clock.now_us() - started < MMIO_RETRY_LIMIT_US => {
372                clock.relax(sleep);
373                sleep = (sleep * 2).min(256_000);
374            }
375            MmioReply::Retry => return Err(GucError::MmioTimeout(header)),
376            MmioReply::Failure { hint, error } => {
377                return Err(GucError::MmioFailure { hint, error })
378            }
379            _ => return Err(GucError::MmioProtocol(header)),
380        }
381    }
382}
