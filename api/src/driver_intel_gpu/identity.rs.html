1//! Intel GT identity and capability discovery from hardware registers.
2//!
3//! Upstream reference: `drivers/gpu/drm/xe/{xe_pci,xe_gt_topology,xe_gt_clock}.c` (MIT).
4
5use crate::regs::{
6    RegisterFile, GMD_ID, RPM_CONFIG0, XE2_GT_COMPUTE_DSS_2, XE2_GT_GEOMETRY_DSS_1,
7    XE2_GT_GEOMETRY_DSS_2, XEHPC_GT_COMPUTE_DSS_ENABLE_EXT, XEHP_GT_COMPUTE_DSS_ENABLE,
8    XELP_EU_ENABLE, XELP_GT_GEOMETRY_DSS_ENABLE,
9};
10pub const VENDOR_INTEL: u16 = 0x8086;
11use intel_gpu_core::uapi::query::DeviceInfo;
12use types::{ViError, ViResult};

14/// Recognized Intel graphics generation.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum GfxArch {
17    /// Xe2 generation (Lunar Lake, Gfx version 20.04).
18    Xe2,
19    /// Xe3 generation (Panther Lake / Wildcat Lake, Gfx version 30.00).
20    Xe3,
21}
22
23/// Discovered hardware GT identity and topology.
24#[derive(Clone, Debug, PartialEq, Eq)]
25pub struct GtIdentity {
26    pub arch: GfxArch,
27    pub ip_ver_major: u32,
28    pub ip_ver_minor: u32,
29    pub ip_ver_rev: u32,
30    /// Geometry DSS enable bits across the three Xe2/Xe3 fuse registers (96 bits).
31    pub dss_geometry_mask: u128,
32    /// Compute DSS enable bits across the three Xe2/Xe3 fuse registers (96 bits).
33    pub dss_compute_mask: u128,
34    /// SIMD16 EUs per DSS: on Xe2 and later one fuse bit is one EU (`load_eu_mask`).
35    pub eu_enable_mask: u32,
36    pub ref_clock_hz: u64,
37}
38
39impl GtIdentity {
40    pub const fn name(&self) -> &'static str {
41        match self.arch {
42            GfxArch::Xe2 => "Intel Xe2 (LNL)",
43            GfxArch::Xe3 => "Intel Xe3 (PTL/WCL)",
44        }
45    }
46
47    /// What `DEVICE_QUERY` reports for this GT. `sysmem_bytes` is the system memory the
48    /// driver can make GPU-visible for clients.
49    pub fn device_info(&self, sysmem_bytes: u64) -> DeviceInfo {
50        DeviceInfo {
51            name: self.name(),
52            // GMD_ID fields are 10/8/6 bits wide, so these never truncate.
53            ip_ver_major: self.ip_ver_major as u16,
54            ip_ver_minor: self.ip_ver_minor as u16,
55            ip_ver_rev: self.ip_ver_rev as u16,
56            // `read_ref_clock` yields one of four crystal rates, all below 2^32.
57            reference_clock_hz: self.ref_clock_hz as u32,
58            dss_geometry: self.dss_geometry_mask,
59            dss_compute: self.dss_compute_mask,
60            eu_per_dss: self.eu_enable_mask,
61            sysmem_bytes,
62        }
63    }
64}
65
66/// Decode the raw 32-bit `GMD_ID` register: arch 31:22, release 21:14, revid 5:0.
67pub fn parse_gmd_id(raw: u32) -> ViResult<(GfxArch, u32, u32, u32)> {
68    let major = raw >> 22;
69    let minor = (raw >> 14) & 0xFF;
70    let rev = raw & 0x3F;
71
72    let arch = match major {
73        20 => GfxArch::Xe2,
74        30 => GfxArch::Xe3,
75        _ => return Err(ViError::NotSupported),
76    };
77
78    Ok((arch, major, minor, rev))
79}

81/// Read the crystal reference clock in Hz from `RPM_CONFIG0` bits 5:3 (`xe_gt_clock.c`).
82pub fn read_ref_clock(regs: &impl RegisterFile) -> ViResult<u64> {
83    let rpm = regs.read32(RPM_CONFIG0)?;
84    match (rpm >> 3) & 0x7 {
85        0 => Ok(24_000_000),
86        1 => Ok(19_200_000),
87        2 => Ok(38_400_000),
88        3 => Ok(25_000_000),
89        _ => Err(ViError::NotSupported),
90    }
91}

93/// Fuse topology of one GT (`xe_gt_topology.c`).
94#[derive(Clone, Copy, Debug, PartialEq, Eq)]
95pub struct Topology {
96    pub dss_geometry: u128,
97    pub dss_compute: u128,
98    pub eu_per_dss: u32,
99}
100
101/// Read the geometry and compute DSS masks and the EU enable mask.
102pub fn read_topology(regs: &impl RegisterFile) -> ViResult<Topology> {
103    let geometry = [
104        XELP_GT_GEOMETRY_DSS_ENABLE,
105        XE2_GT_GEOMETRY_DSS_1,
106        XE2_GT_GEOMETRY_DSS_2,
107    ];
108    let compute = [
109        XEHP_GT_COMPUTE_DSS_ENABLE,
110        XEHPC_GT_COMPUTE_DSS_ENABLE_EXT,
111        XE2_GT_COMPUTE_DSS_2,
112    ];
113    Ok(Topology {
114        dss_geometry: read_mask(regs, &geometry)?,
115        dss_compute: read_mask(regs, &compute)?,
116        eu_per_dss: regs.read32(XELP_EU_ENABLE)?,
117    })
118}
119
120/// Concatenate fuse registers, the first one holding bits 31:0.
121fn read_mask(regs: &impl RegisterFile, words: &[usize; 3]) -> ViResult<u128> {
122    let mut mask = 0u128;
123    for (word, &reg) in words.iter().enumerate() {
124        mask |= u128::from(regs.read32(reg)?) << (32 * word);
125    }
126    Ok(mask)
127}
128
129/// Complete GT discovery probe.
130pub fn probe_gt(regs: &impl RegisterFile) -> ViResult<GtIdentity> {
131    let raw_gmd = regs.read32(GMD_ID)?;
132    let (arch, major, minor, rev) = parse_gmd_id(raw_gmd)?;
133    let topology = read_topology(regs)?;
134    let ref_clock_hz = read_ref_clock(regs)?;
135
136    Ok(GtIdentity {
137        arch,
138        ip_ver_major: major,
139        ip_ver_minor: minor,
140        ip_ver_rev: rev,
141        dss_geometry_mask: topology.dss_geometry,
142        dss_compute_mask: topology.dss_compute,
143        eu_enable_mask: topology.eu_per_dss,
144        ref_clock_hz,
145    })
