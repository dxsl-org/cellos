1//! GPU bring-up: forcewake -> GT identity -> GuC firmware -> GGTT -> WOPCM -> GuC reset ->
2//! log, ADS and parameters -> transfer setup -> RSA -> uCode DMA -> boot -> CT enabled.
3//!
4//! Upstream reference: the `xe_device_probe` / `xe_uc_init` / `xe_uc_load_hw` order in
5//! `drivers/gpu/drm/xe/{xe_device,xe_uc,xe_guc,xe_wopcm}.c` (MIT). Every failure is a typed
6//! [`BringUpError`]; nothing short of a running GuC with an enabled CT is reported ready.
7

25/// The GuC release ADR-0033 §5 pins (`research/firmware-pin.md`) for PTL/WCL and LNL.
26pub const PINNED_GUC: Version = Version::new(70, 72, 1);
27
28/// The linux-firmware GuC image for `arch`.
29pub const fn guc_firmware_path(arch: GfxArch) -> &'static str {
30    match arch {
31        GfxArch::Xe2 => "/lib/firmware/xe/lnl_guc_70.bin",
32        GfxArch::Xe3 => "/lib/firmware/xe/ptl_guc_70.bin",
33    }
34}

36/// What bring-up needs from the system besides MMIO: the firmware file in memory the
37/// GPU can read.
38pub trait Platform {
39    /// Read the file at `path`; its length, or `Ok(None)` when the image does not carry it.
40    fn read_firmware(&mut self, path: &str) -> Result<Option<usize>, ViError>;
41    /// Run `f` over the bytes [`Self::read_firmware`] loaded.
42    fn with_firmware<T>(&mut self, f: impl FnOnce(&[u8]) -> T) -> T;
43    /// Authorize the GPU to read the loaded firmware; returns its page-aligned device address.
44    fn map_firmware(&mut self) -> Result<u64, ViError>;
45}

47/// The PCI identity the GuC parameter block carries (`guc_ctl_devid`).
48#[derive(Clone, Copy, Debug, PartialEq, Eq)]
49pub struct PciIdentity {
50    pub device_id: u16,
51    /// PCI revision. The kernel's `PcieDeviceInfo` record does not carry it, so the Cell
52    /// passes 0 until it does.
53    pub revision: u8,
54}

56/// Why the GPU did not come up.
57#[derive(Clone, Copy, Debug, PartialEq, Eq)]
58pub enum BringUpError {
59    /// Forcewake was not acknowledged, or the GT identity is unreadable or unsupported.
60    Probe(ViError),
61    /// The image carries no GuC firmware at this path.
62    FirmwareUnavailable(&'static str),
63    /// The firmware file could not be read (I/O, or larger than the driver accepts).
64    FirmwareRead(ViError),
65    /// The file is not a valid GuC image, or not the pinned release.
66    FirmwareRejected(FwError),
67    /// The firmware could not be made device-readable or mapped into the GGTT.
68    FirmwareMap(ViError),
69    /// No device memory for the GuC log, ADS or CT buffers.
70    NoMemory,
71    /// The GuC did not partition, reset, take the DMA, or reach its running state.
72    GucLoad(GucError),
73    /// The GuC runs but its command transport could not be enabled.
74    GucCt(CtFailure),
75}

77/// A GPU whose GuC is running with its CT channel enabled.
78pub struct Ready<R: RegisterFile, C: Clock, M: DmaMemory> {
79    pub controller: IntelGpuController<R, C>,
80    pub mem: M,
81    pub ggtt: Ggtt,
82    pub pool: Pool,
83    pub ct: GucCt,
84    /// GGTT address of the GuC image.
85    pub guc_ggtt: u64,
86    pub wopcm: WopcmLayout,
87    /// How long the GuC took to boot, in microseconds (upstream warns above 200 ms).
88    pub load_us: u64,
89}

91/// Bring the GPU behind `regs` up. On failure after forcewake, forcewake is released again.
92pub fn bring_up<R: RegisterFile, C: Clock, M: DmaMemory>(
93    regs: R,
94    clock: C,
95    mut mem: M,
96    pci: PciIdentity,
97    platform: &mut impl Platform,
98) -> Result<Ready<R, C, M>, BringUpError> {
99    let mut controller = IntelGpuController::probe(regs, clock).map_err(BringUpError::Probe)?;
100    let mut ggtt = Ggtt::new();
101    let mut pool = Pool::default();
102    match load_guc(
103        &mut controller,
104        &mut mem,
105        &mut ggtt,
106        &mut pool,
107        pci,
108        platform,
109    ) {
110        Ok(loaded) => Ok(Ready {
111            controller,
112            mem,
113            ggtt,
114            pool,
115            ct: loaded.ct,
116            guc_ggtt: loaded.guc_ggtt,
117            wopcm: loaded.wopcm,
118            load_us: loaded.load_us,
119        }),
120        Err(error) => {
121            let _ = controller.shutdown();
122            Err(error)
123        }
124    }
125}

127struct Loaded {
128    ct: GucCt,
129    guc_ggtt: u64,
130    wopcm: WopcmLayout,
131    load_us: u64,
132}
133
134fn load_guc<R: RegisterFile, C: Clock, M: DmaMemory>(
135    controller: &mut IntelGpuController<R, C>,
136    mem: &mut M,
137    ggtt: &mut Ggtt,
138    pool: &mut Pool,
139    pci: PciIdentity,
140    platform: &mut impl Platform,
141) -> Result<Loaded, BringUpError> {
142    let (image, guc_ggtt) = map_firmware(controller, ggtt, platform)?;
143    let regs = &mut controller.regs;
144    let clock = &mut controller.clock;
145    let upload_size = CSS_HEADER_LEN + image.ucode_len;
146    let gucerr = BringUpError::GucLoad;
147
148    // `xe_uc_init` → `xe_wopcm_init`: the partition must hold header plus uCode.
149    let guc_fw_size = u32::try_from(upload_size).map_err(|_| gucerr(GucError::BadImage))?;
150    let wopcm = init_wopcm(regs, guc_fw_size).map_err(gucerr)?;
151    // `xe_uc_sanitize_reset` → `xe_guc_reset`.
152    reset_guc(regs, clock).map_err(gucerr)?;
153
154    // `xe_guc_upload` → `xe_guc_ads_populate`; the log object (`xe_guc_log_init`).
155    let log = pool
156        .alloc(mem, regs, ggtt, (GUC_LOG_SIZE as usize).div_ceil(PAGE))
157        .ok_or(BringUpError::NoMemory)?;
158    let ads_size = guc_ads::ads_size(image.private_data_size);
159    let ads = pool
160        .alloc(mem, regs, ggtt, ads_size.div_ceil(PAGE))
161        .ok_or(BringUpError::NoMemory)?;
162    let doorbells = regs
163        .read32(DIST_DBS_POPULATED)
164        .map_err(|_| gucerr(GucError::Mmio))?;
165    let blob = guc_ads::build(ads.ggtt, image.private_data_size, 1, doorbells);
166    if !mem.write(ads.iova, &blob) {
167        return Err(BringUpError::NoMemory);
168    }
169
170    // `__xe_guc_upload`: parameters, transfer setup, RSA, DMA at 8 KiB, boot wait.
171    let params = guc_init::params(log.ggtt, ads.ggtt, pci.device_id, pci.revision);
172    write_params(regs, &params).map_err(gucerr)?;
173    prepare_xfer(regs).map_err(gucerr)?;
174    let rsa = image.rsa_offset..image.rsa_offset + image.rsa_len;
175    platform
176        .with_firmware(|blob| transfer_rsa(regs, &blob[rsa], guc_ggtt + image.rsa_offset as u64))
177        .map_err(gucerr)?;
178    upload_guc_firmware(regs, clock, guc_ggtt, GUC_WOPCM_LOAD_OFFSET, upload_size)
179        .map_err(gucerr)?;
180    let load_us = wait_guc_ukernel_running(regs, clock).map_err(gucerr)?;
181
182    // `xe_guc_enable_communication` → `xe_guc_ct_enable`. GuC-to-host interrupts stay
183    // off: the driver polls the G2H ring.
184    let ct = GucCt::enable(pool, mem, regs, ggtt, clock).map_err(BringUpError::GucCt)?;
185    // The ADS and log regions are never freed: the GuC uses them for as long as it runs.
186    Ok(Loaded {
187        ct,
188        guc_ggtt,
189        wopcm,
190        load_us,
191    })
192}

194/// Read, validate and GGTT-map the pinned GuC image.
195fn map_firmware<R: RegisterFile, C: Clock>(
196    controller: &mut IntelGpuController<R, C>,
197    ggtt: &mut Ggtt,
198    platform: &mut impl Platform,
199) -> Result<(GucImage, u64), BringUpError> {
200    let path = guc_firmware_path(controller.identity.arch);
201    let len = platform
202        .read_firmware(path)
203        .map_err(BringUpError::FirmwareRead)?
204        .ok_or(BringUpError::FirmwareUnavailable(path))?;
205    // Validate before the GPU may read a single byte of the file.
206    let image = platform
207        .with_firmware(|blob| {
208            let image = guc_fw::parse(blob)?;
209            guc_fw::check_pinned(&image, PINNED_GUC).map(|()| image)
210        })
211        .map_err(BringUpError::FirmwareRejected)?;
212    let iova = platform.map_firmware().map_err(BringUpError::FirmwareMap)?;
213    let guc_ggtt = ggtt
214        .map(&mut controller.regs, iova, len, PatIndex::WB)
215        .map_err(BringUpError::FirmwareMap)?;
216    Ok((image, guc_ggtt))
217}
