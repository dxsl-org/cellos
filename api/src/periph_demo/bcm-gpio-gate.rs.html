1use driver_gpio_bcm::BcmGpio;
2use hal_gpio::{Edge, PinDir, ViGpio};
3use ostd::io::println;
4use ostd::syscall::sys_recv_timeout;
5use types::ViError;
6
7const OUTPUT_PIN: u8 = 17;
8const INPUT_PIN: u8 = 27;
9const INPUT_MASK: u32 = 1 << INPUT_PIN;
10
11/// Run the RPi3 GPIO17-to-GPIO27 physical loopback gate.
12///
13/// Returns `false` only when BCM GPIO is not allowlisted, allowing QEMU to run
14/// the existing PL061 gate. Once BCM ownership is available, failures stay visible.
15pub fn run() -> bool {
16    let mut gpio = match BcmGpio::open() {
17        Ok(gpio) => gpio,
18        Err(ViError::PermissionDenied) => {
19            println("[phase03-gpio] BCM GPIO permission denied; trying PL061 fallback");
20            return false;
21        }
22        Err(_) => {
23            println("[phase03-gpio] FAIL: BCM GPIO open");
24            return true;
25        }
26    };
27
28    let setup = gpio
29        .set_direction(OUTPUT_PIN, PinDir::Output)
30        .and_then(|_| gpio.set_direction(INPUT_PIN, PinDir::Input))
31        .and_then(|_| gpio.write_pin(OUTPUT_PIN, false))
32        .and_then(|_| gpio.disable_irq(INPUT_PIN))
33        .and_then(|_| gpio.clear_eds(0, INPUT_MASK));
34    let precheck = gpio.read_eds(0);
35    if setup.is_err() || precheck.is_err() || precheck.is_ok_and(|eds| eds & INPUT_MASK != 0) {
36        println("[phase03-gpio] FAIL: negative precheck");
37        return true;
38    }
39    println("[phase03-gpio] negative precheck PASS");
40
41    let armed = gpio.enable_edge_irq(INPUT_PIN, Edge::Rising).is_ok();
42    let toggled = armed && gpio.write_pin(OUTPUT_PIN, true).is_ok();
43    if !toggled {
44        let _ = gpio.clear_eds(0, INPUT_MASK);
45        let _ = gpio.disable_irq(INPUT_PIN);
46        let _ = gpio.write_pin(OUTPUT_PIN, false);
47        println("[phase03-gpio] FAIL: arm/toggle");
48        return true;
49    }
50
51    let mut detected = false;
52    for _ in 0..50 {
53        if gpio.read_eds(0).is_ok_and(|eds| eds & INPUT_MASK != 0) {
54            detected = true;
55            break;
56        }
57        let mut message = [0u8; 16];
58        let _ = sys_recv_timeout(0, &mut message, 1);
59    }
60    let _ = gpio.clear_eds(0, INPUT_MASK);
61    let _ = gpio.disable_irq(INPUT_PIN);
62    let _ = gpio.write_pin(OUTPUT_PIN, false);
63
64    if detected {
65        println("[phase03-gpio] PASS: GPIO17->GPIO27 rising edge detected");
66    } else {
67        println("[phase03-gpio] FAIL: connect physical pin 11 to pin 13");
68    }
69    true
70}
