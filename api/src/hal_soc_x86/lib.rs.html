3//! Immutable x86 machine facts owned by `hal/soc/x86`.
4//!
5//! ACPI-discovered LAPIC, IOAPIC, HPET, and PCIe ECAM addresses deliberately
6//! do not appear here: invalid or missing firmware must keep those gates closed.
7
8#[derive(Clone, Copy, Debug, Eq, PartialEq)]
9/// A half-open physical or port-address range.
10pub struct AddressRange {
11    pub base: usize,
12    pub size: usize,
13}
14
15impl AddressRange {
16    /// Returns the exclusive end, or `None` when the range overflows.
17    pub const fn end(self) -> Option<usize> {
18        self.base.checked_add(self.size)
19    }
20
21    /// Reports whether the non-empty candidate range is fully contained.
22    pub const fn contains(self, base: usize, size: usize) -> bool {
23        let Some(end) = base.checked_add(size) else {
24            return false;
25        };
26        let Some(limit) = self.end() else {
27            return false;
28        };
29        size != 0 && base >= self.base && end <= limit
30    }
31}

40/// How an RS485 transceiver takes its direction, as declared by the machine.
41///
42/// Neither variant is claimed as working by any board today: QEMU models no
43/// DE/RE line, so the timing each mechanism implies can only be measured on
44/// hardware (phase 07). The declaration exists so a machine with a transceiver
45/// has one place to state it, and so a descriptor that states it wrongly is a
46/// validation error rather than a silent assumption.
47#[derive(Clone, Copy, Debug, Eq, PartialEq)]
48pub enum Rs485Direction {
49    /// The transceiver derives direction from RTS, so software only has to keep
50    /// RTS asserted for the whole transmit burst.
51    RtsAuto,
52    /// Software drives an explicit DE/RE line: port-I/O offset `offset` from the
53    /// port base carries it.
54    Gpio { offset: u16 },
55}

57#[derive(Clone, Copy, Debug, Eq, PartialEq)]
58/// RS485 facts for one declared serial port.
59pub struct Rs485Port {
60    /// Index into [`X86PlatformProfile::serial_ports`].
61    pub port: u8,
62    pub direction: Rs485Direction,
63    /// Transmit-to-receive turnaround the transceiver requires, in nanoseconds.
64    /// Zero is rejected: a transceiver that needs no turnaround does not exist,
65    /// and a zero here would read as "timing not considered".
66    pub turnaround_ns: u32,
67}

69#[derive(Clone, Copy, Debug, Eq, PartialEq)]
70/// Immutable x86 machine facts consumed before ACPI discovery succeeds.
71pub struct X86PlatformProfile {
72    pub slug: &'static str,
73    /// The console port. Always `serial_ports[0]`; kept as its own field because
74    /// every console path (early boot, panic, log) must not index a list.
75    pub com1: PortIoDevice,
76    /// Every 16550 port this machine declares, console first. Ports the machine
77    /// does not actually expose fail the 16550 probe and are refused by name
78    /// rather than assumed present.
79    pub serial_ports: &'static [PortIoDevice],
80    /// RS485 transceiver facts, one entry per port that has one.
81    pub rs485: &'static [Rs485Port],
82    pub legacy_bios_window: AddressRange,
83    pub legacy_rsdp_window: AddressRange,
84}
85
86impl X86PlatformProfile {
87    /// Validates port wiring and bounded firmware-window relationships.
88    pub fn validate(self) -> Result<(), ValidationError> {
89        if self.com1.base == 0 {
90            return Err(ValidationError::ZeroPortBase);
91        }
92        if self.com1.irq >= 16 {
93            return Err(ValidationError::InvalidIsaIrq);
94        }
95        if self.serial_ports.is_empty() {
96            return Err(ValidationError::NoSerialPorts);
97        }
98        if self.serial_ports[0] != self.com1 {
99            return Err(ValidationError::ConsoleNotFirstSerialPort);
100        }
101        for (index, port) in self.serial_ports.iter().enumerate() {
102            if port.base == 0 {
103                return Err(ValidationError::ZeroPortBase);
104            }
105            if port.irq >= 16 {
106                return Err(ValidationError::InvalidIsaIrq);
107            }
108            if self.serial_ports[..index].iter().any(|p| p.base == port.base) {
109                return Err(ValidationError::DuplicateSerialPort);
110            }
111        }
112        for rs485 in self.rs485 {
113            if rs485.port as usize >= self.serial_ports.len() {
114                return Err(ValidationError::Rs485PortOutOfRange);
115            }
116            if rs485.turnaround_ns == 0 {
117                return Err(ValidationError::Rs485ZeroTurnaround);
118            }
119        }
120        if self.legacy_bios_window.size == 0 || self.legacy_rsdp_window.size == 0 {
121            return Err(ValidationError::ZeroSizedFirmwareWindow);
122        }
123        let Some(bios_end) = self.legacy_bios_window.end() else {
124            return Err(ValidationError::OverflowingFirmwareWindow);
125        };
126        let Some(rsdp_end) = self.legacy_rsdp_window.end() else {
127            return Err(ValidationError::OverflowingFirmwareWindow);
128        };
129        if self.legacy_bios_window.base < self.legacy_rsdp_window.base || bios_end > rsdp_end {
130            return Err(ValidationError::BiosWindowOutsideRsdpWindow);
131        }
132        Ok(())
133    }
134
135    /// The port at `index`, or `None` when the profile declares fewer ports.
136    pub const fn serial_port(self, index: usize) -> Option<PortIoDevice> {
137        if index < self.serial_ports.len() {
138            Some(self.serial_ports[index])
139        } else {
140            None
141        }
142    }
143}

145#[derive(Clone, Copy, Debug, Eq, PartialEq)]
146/// Structural failures that make an x86 platform profile unsafe to consume.
147pub enum ValidationError {
148    ZeroPortBase,
149    InvalidIsaIrq,
150    ZeroSizedFirmwareWindow,
151    OverflowingFirmwareWindow,
152    BiosWindowOutsideRsdpWindow,
153    /// A profile with no declared serial ports has no console.
154    NoSerialPorts,
155    /// The console must be the first declared port.
156    ConsoleNotFirstSerialPort,
157    /// Two declared ports share one base address.
158    DuplicateSerialPort,
159    /// An RS485 declaration names a port the profile does not declare.
160    Rs485PortOutOfRange,
161    /// An RS485 declaration omits the transceiver turnaround.
162    Rs485ZeroTurnaround,
163}

165/// The four standard PC COM ports, console first.
166///
167/// Both x86 profiles declare them: the addresses are the legacy ISA layout, and
168/// a machine that exposes fewer fails the probe on the missing ones (named, not
169/// assumed). A machine with non-standard port addresses needs its own profile —
170/// the HCL record for that machine, not a guess here.

182        irq: 4,
