3//! Immutable x86 machine facts owned by `hal/soc/x86`.
4//!
5//! ACPI-discovered LAPIC, IOAPIC, HPET, and PCIe ECAM addresses deliberately
6//! do not appear here: invalid or missing firmware must keep those gates closed.
7
8#[derive(Clone, Copy, Debug, Eq, PartialEq)]
9/// A half-open physical or port-address range.
10pub struct AddressRange {
11    pub base: usize,
12    pub size: usize,
13}
14
15impl AddressRange {
16    /// Returns the exclusive end, or `None` when the range overflows.
17    pub const fn end(self) -> Option<usize> {
18        self.base.checked_add(self.size)
19    }
20
21    /// Reports whether the non-empty candidate range is fully contained.
22    pub const fn contains(self, base: usize, size: usize) -> bool {
23        let Some(end) = base.checked_add(size) else {
24            return false;
25        };
26        let Some(limit) = self.end() else {
27            return false;
28        };
29        size != 0 && base >= self.base && end <= limit
30    }
31}

34/// Static port-I/O wiring for one legacy device.
35pub struct PortIoDevice {
36    pub base: u16,
37    pub irq: u8,

40#[derive(Clone, Copy, Debug, Eq, PartialEq)]
41/// Immutable x86 machine facts consumed before ACPI discovery succeeds.
42pub struct X86PlatformProfile {
43    pub slug: &'static str,
44    pub com1: PortIoDevice,
45    pub legacy_bios_window: AddressRange,
46    pub legacy_rsdp_window: AddressRange,
47}
48
49impl X86PlatformProfile {
50    /// Validates port wiring and bounded firmware-window relationships.
51    pub fn validate(self) -> Result<(), ValidationError> {
52        if self.com1.base == 0 {
53            return Err(ValidationError::ZeroPortBase);
54        }
55        if self.com1.irq >= 16 {
56            return Err(ValidationError::InvalidIsaIrq);
57        }
58        if self.legacy_bios_window.size == 0 || self.legacy_rsdp_window.size == 0 {
59            return Err(ValidationError::ZeroSizedFirmwareWindow);
60        }
61        let Some(bios_end) = self.legacy_bios_window.end() else {
62            return Err(ValidationError::OverflowingFirmwareWindow);
63        };
64        let Some(rsdp_end) = self.legacy_rsdp_window.end() else {
65            return Err(ValidationError::OverflowingFirmwareWindow);
66        };
67        if self.legacy_bios_window.base < self.legacy_rsdp_window.base || bios_end > rsdp_end {
68            return Err(ValidationError::BiosWindowOutsideRsdpWindow);
69        }
70        Ok(())
71    }
72}
73
74#[derive(Clone, Copy, Debug, Eq, PartialEq)]
75/// Structural failures that make an x86 platform profile unsafe to consume.
76pub enum ValidationError {
77    ZeroPortBase,
78    InvalidIsaIrq,
79    ZeroSizedFirmwareWindow,
80    OverflowingFirmwareWindow,
81    BiosWindowOutsideRsdpWindow,
82}
