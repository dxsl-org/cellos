30fn cell_main() {
31    let argv = ostd::args();
32    let role = argv.first().map(|arg| arg.as_str()).unwrap_or("");
33    ostd::io::println(&alloc::format!(
34        "[bench-probe] Started with role: '{}'",
35        role
36    ));
37    if role.starts_with("hotswap-cached-inc:") {
38        scenarios::hotswap_supervisor::run_cached_sender_probe(role);
39    }
40    if role.starts_with("native-stateful-cached-inc:") {
41        scenarios::native_stateful::run_cached_sender_probe(role);
42    }
43    match role {
44        "c2c-client" => scenarios::c2c_broker_oracle::run_client(),
45        "load" => scenarios::rt_load::run_load(),
46        "rt-probe" => scenarios::preempt_latency::run_probe(),
47        "ctl-loop" => scenarios::control_loop::run_control_loop(),
48        "ipc-echo" => {
49            let mut buf = [0xa5u8; 64];
50            loop {
51                buf.fill(0xa5);
52                let sender = match ostd::syscall::sys_recv(0, &mut buf) {
53                    ostd::syscall::SyscallResult::Ok(sid) if sid != 0 => sid,
54                    _ => continue,
55                };
56                let valid_request = buf[0] == 0x42 && buf[1..].iter().all(|&byte| byte == 0);
57                let reply = if valid_request { 0 } else { 1 };
58                let _ = ostd::syscall::sys_send(sender, &[reply]);
59            }
60        }
61        "fastpath-echo" => {
62            let mut addr_buf = [0u8; 8];
63            let sender = match ostd::syscall::sys_recv(0, &mut addr_buf) {
64                ostd::syscall::SyscallResult::Ok(sid) if sid != 0 => sid,
65                _ => return,
66            };
67            let handle = u64::from_le_bytes(addr_buf);
68            let client = match ostd::ring_channel::ChannelClient::connect(handle) {
69                Some(c) => c,
70                None => return,
71            };
72            // Send ACK to orchestrator
73            let _ = ostd::syscall::sys_send(sender, &[0x55]);
74
75            let endpoint = client.endpoint();
76            let mut req_buf = [0u8; 64];
77            let mut resp_buf = [0u8; 1];
78
79            loop {
80                let meta = match endpoint.rx.recv_blocking(&mut req_buf) {
81                    Ok(m) => m,
82                    Err(_) => break,
83                };
84                let valid = req_buf[0] == 0x42 && req_buf[1..meta.len].iter().all(|&b| b == 0);
85                resp_buf[0] = if valid { 0 } else { 1 };
86                if endpoint.tx.send_blocking(&resp_buf, meta.seq, 0).is_err() {
87                    break;
88                }
89            }
90        }
91        "resp-echo" => scenarios::vfs_getfile_breakdown::run_resp_echo(),
92        "smp-worker" => scenarios::smp::run_worker(),
93        "heartbeat-peer" => scenarios::smp::run_heartbeat_peer(),
94        _ => {} // unknown role: exit cleanly
95    }
96}
