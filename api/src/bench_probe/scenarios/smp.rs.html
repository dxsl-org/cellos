1//! SMP throughput scenarios — validate 2-hart work-stealing on QEMU.
2//!
3//! All spawn/notify/force_exit helpers run only in the orchestrator context
4//! (which has SpawnCap).  Only `run_worker` is called from bench-probe.
5
6use crate::framework::timer::NS_PER_TICK;
7use alloc::format;
8use api::task::TaskPriority;
9use core::hint::black_box;
10use ostd::{
11    io::println,
12    syscall::{
13        sys_exit, sys_force_exit, sys_get_time, sys_heartbeat, sys_notify_on_exit, sys_recv,
14        sys_send, sys_set_spawn_args, sys_spawn_pinned, SyscallResult,
15    },
16    task::yield_now,
17};
18
19/// Iterations per worker run.  Calibrated for ≥1 ms on QEMU TCG.
20pub const SMP_WORKER_ITERS: u64 = 500_000;
21
22const PROBE_PATH: &str = "/bin/bench-probe";
23const CAVEAT: &str = " [QEMU-TCG: 2 hart-threads; real-HW shows true 2× parallelism]";
24
25/// Optimizer-resistant arithmetic workload shared by worker and orchestrator.
26fn compute(iters: u64) -> u64 {
27    let mut acc = 0u64;
28    for i in 0..iters {
29        acc = acc.wrapping_add(i.wrapping_mul(3));
30    }
31    black_box(acc)
32}
33
34/// `smp-worker` bench-probe role entry — runs compute, prints, exits.
35pub fn run_worker() -> ! {
36    let acc = compute(SMP_WORKER_ITERS);
37    println(&format!("[smp] worker done acc={}", acc));
38    sys_exit(0);
39}

41/// Wait for one start message, then deliberately miss a heartbeat deadline.
42/// A caller's next blocking send must resume with an error when the watchdog
43/// kills this peer through the ordinary `exit_task` path.
44#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
45pub fn run_heartbeat_peer() -> ! {
46    let mut start = [0u8; 1];
47    let _ = sys_recv(0, &mut start);
48    sys_heartbeat(5);
49    loop {
50        yield_now();
51    }
52}

54// ── Orchestrator-only helpers ─────────────────────────────────────────────────
55
56fn spawn_worker() -> Result<usize, ()> {
57    if !sys_set_spawn_args("smp-worker") {
58        return Err(());
59    }
60    match sys_spawn_pinned(PROBE_PATH, TaskPriority::Normal as u8, 0) {
61        SyscallResult::Ok(tid) if tid != 0 => Ok(tid),
62        _ => Err(()),
63    }
64}
65
66fn recv_exit(tid: usize) -> Result<(), ()> {
67    let mut buf = [0u8; 8];
68    match sys_recv(tid, &mut buf) {
69        SyscallResult::Ok(sender) if sender == tid => Ok(()),
70        _ => Err(()),
71    }
72}
73
74fn wait_exit(tid: usize) -> Result<(), ()> {
75    if !matches!(sys_notify_on_exit(tid), SyscallResult::Ok(_)) {
76        return Err(());
77    }
78    recv_exit(tid)
79}
80
81fn force_exit(tid: usize) -> bool {
82    matches!(sys_force_exit(tid), SyscallResult::Ok(_))
83}

85/// Prove caller-visible dead-peer error plus queued ForceExit notification.
86pub fn run_peer_death_guard() -> ! {
87    if !sys_set_spawn_args("heartbeat-peer") {
88        println("[peer-death-runtime] FAIL — heartbeat child argument staging");
89        sys_exit(1);
90    }
91    let blocked_peer = match sys_spawn_pinned(PROBE_PATH, TaskPriority::Normal as u8, 0) {
92        SyscallResult::Ok(tid) => tid,
93        _ => {
94            println("[peer-death-runtime] FAIL — heartbeat child spawn");
95            sys_exit(1);
96        }
97    };
98    for _ in 0..20 {
99        yield_now();
100    }
101    if !matches!(sys_send(blocked_peer, &[1]), SyscallResult::Ok(_)) {
102        println("[peer-death-runtime] FAIL — heartbeat child start");
103        let _ = sys_force_exit(blocked_peer);
104        sys_exit(1);
105    }
106    if !matches!(sys_send(blocked_peer, &[2]), SyscallResult::Err(_)) {
107        println("[peer-death-runtime] FAIL — blocked send did not return an error");
108        let _ = sys_force_exit(blocked_peer);
109        sys_exit(1);
110    }
111
112    if !sys_set_spawn_args("load") {
113        println("[peer-death-runtime] FAIL — force-exit child argument staging");
114        sys_exit(1);
115    }
116    let forced_tid = match sys_spawn_pinned(PROBE_PATH, TaskPriority::Normal as u8, 0) {
117        SyscallResult::Ok(tid) => tid,
118        _ => {
119            println("[peer-death-runtime] FAIL — ForceExit child spawn");
120            sys_exit(1);
121        }
122    };
123    if !matches!(sys_notify_on_exit(forced_tid), SyscallResult::Ok(_)) {
124        println("[peer-death-runtime] FAIL — ForceExit watch");
125        let _ = sys_force_exit(forced_tid);
126        sys_exit(1);
127    }
128    for _ in 0..20 {
129        yield_now();
130    }
131    if !matches!(sys_force_exit(forced_tid), SyscallResult::Ok(_)) {
132        println("[peer-death-runtime] FAIL — ForceExit denied");
133        sys_exit(1);
134    }
135    let _ = recv_exit(forced_tid);
136
137    println("[peer-death-runtime] PASS (blocked-send error + ForceExit notification)");
138    sys_exit(0);
139}

145    pub n: u32,
146    pub value: u64,
147    pub passed: bool,
148}

150pub struct SmpFailure {

153    pub teardown_failed: bool,

156pub enum SmpOutcome {

158    Invalid(SmpFailure),
159}

161fn invalid(name: &'static str, stage: &'static str, teardown_failed: bool) -> SmpOutcome {
162    SmpOutcome::Invalid(SmpFailure {

164        stage,
165        teardown_failed,

169// ── Scenario 1: spawn_rate ────────────────────────────────────────────────────
170
171fn measure_spawn_rate() -> SmpOutcome {
172    const N: u64 = 8;
173    const TARGET: u64 = 10;
174    const NAME: &str = "smp_spawn_rate";
175
176    let t0 = sys_get_time();
177    for _ in 0..N {
178        let tid = match spawn_worker() {
179            Ok(tid) => tid,
180            Err(()) => return invalid(NAME, "setup", false),
181        };
182        if wait_exit(tid).is_err() {
183            return invalid(NAME, "measure", !force_exit(tid));
184        }
185    }
186    let dt_ns = sys_get_time()
187        .saturating_sub(t0)
188        .saturating_mul(NS_PER_TICK);
189    if dt_ns == 0 {
190        return invalid(NAME, "measure", false);
191    }
192    let per_sec = N.saturating_mul(1_000_000_000) / dt_ns;
193    let passed = per_sec >= TARGET;
194    println(&format!(
195        "[smp] spawn_rate {}: {}/sec (target ≥{}/sec){}",
196        if passed { "PASS" } else { "FAIL" },
197        per_sec,
198        TARGET,
199        CAVEAT
200    ));
201    SmpOutcome::Metric(SmpMetric {
202        name: NAME,
203        n: N as u32,
204        value: per_sec,
205        passed,
206    })
207}

209// ── Scenario 2: ipc_throughput ────────────────────────────────────────────────
210
211fn measure_ipc_throughput() -> SmpOutcome {
212    const MSGS: u64 = 1_000;
213    const TARGET: u64 = 5_000;
214    const NAME: &str = "smp_ipc_throughput";
215
216    if !sys_set_spawn_args("ipc-echo") {
217        return invalid(NAME, "setup", false);
218    }
219    let echo_tid = match sys_spawn_pinned(PROBE_PATH, TaskPriority::Normal as u8, 0) {
220        SyscallResult::Ok(tid) if tid != 0 => tid,
221        _ => return invalid(NAME, "setup", false),
222    };
223    for _ in 0..20 {
224        yield_now();
225    }
226
227    let mut request = [0u8; 64];
228    request[0] = 0x42;
229    let mut reply = [0u8; 8];
230    let t0 = sys_get_time();
231    let mut operation_failed = false;
232    for _ in 0..MSGS {
233        reply.fill(0xa5);
234        if !matches!(sys_send(echo_tid, &request), SyscallResult::Ok(_)) {
235            operation_failed = true;
236            break;
237        }
238        match sys_recv(echo_tid, &mut reply) {
239            SyscallResult::Ok(sender) if sender == echo_tid => {}
240            _ => {
241                operation_failed = true;
242                break;
243            }
244        }
245        if reply[0] != 0 || reply[1..].iter().any(|&byte| byte != 0xa5) {
246            operation_failed = true;
247            break;
248        }
249    }
250    let dt_ns = sys_get_time()
251        .saturating_sub(t0)
252        .saturating_mul(NS_PER_TICK);
253    let teardown_failed = !force_exit(echo_tid);
254    if operation_failed || dt_ns == 0 {
255        return invalid(NAME, "measure", teardown_failed);
256    }
257    if teardown_failed {
258        return invalid(NAME, "teardown", false);
259    }
260
261    let per_sec = MSGS.saturating_mul(1_000_000_000) / dt_ns;
262    let passed = per_sec >= TARGET;
263    println(&format!(
264        "[smp] ipc_throughput {}: {}/sec (target ≥{}/sec){}",
265        if passed { "PASS" } else { "FAIL" },
266        per_sec,
267        TARGET,
268        CAVEAT
269    ));
270    SmpOutcome::Metric(SmpMetric {
271        name: NAME,
272        n: MSGS as u32,
273        value: per_sec,
274        passed,
275    })
276}

278// ── Scenario 3: work_distribution ─────────────────────────────────────────────
279
280fn measure_work_distribution() -> SmpOutcome {
281    const NAME: &str = "smp_work_distribution";
282
283    let t0 = sys_get_time();
284    let tid1 = match spawn_worker() {
285        Ok(tid) => tid,
286        Err(()) => return invalid(NAME, "setup", false),
287    };
288    if wait_exit(tid1).is_err() {
289        return invalid(NAME, "measure", !force_exit(tid1));
290    }
291    let t_single = sys_get_time().saturating_sub(t0);
292
293    let t1 = sys_get_time();
294    let tid2 = match spawn_worker() {
295        Ok(tid) => tid,
296        Err(()) => return invalid(NAME, "setup", false),
297    };
298    if !matches!(sys_notify_on_exit(tid2), SyscallResult::Ok(_)) {
299        return invalid(NAME, "measure", !force_exit(tid2));
300    }
301    let _ = compute(SMP_WORKER_ITERS);
302    if recv_exit(tid2).is_err() {
303        return invalid(NAME, "measure", !force_exit(tid2));
304    }
305    let t_parallel = sys_get_time().saturating_sub(t1);
306    if t_single == 0 || t_parallel == 0 {
307        return invalid(NAME, "measure", false);
308    }
309
310    let scale_x100 = t_single.saturating_mul(200) / t_parallel;
311    let passed = scale_x100 >= 140;
312    println(&format!(
313        "[smp] work_distribution {}: scale={}.{:02}x T1={}t Tp={}t (target ≥1.40x){}",
314        if passed { "PASS" } else { "FAIL" },
315        scale_x100 / 100,
316        scale_x100 % 100,
317        t_single,
318        t_parallel,
319        CAVEAT
320    ));
321    SmpOutcome::Metric(SmpMetric {
322        name: NAME,
323        n: 2,
324        value: scale_x100,
325        passed,
326    })
327}
