1//! Watch `init`'s crash-storm budget engage.
2//!
3//! `init` restarts a `Permanent` service and gives up on it after five restarts inside one
4//! window (`MAX_RESTARTS_PER_WINDOW`, Spec 12 §4.3). The window is a scheduler-tick count; when
5//! it was compared against `GetTime` op 0 — the raw architected counter, 10 MHz `mtime` on RV64
6//! — it was ~0.1 ms wide, so the budget rolled on every exit and a crash-looping service was
7//! restarted forever.
8//!
9//! This scenario makes the storm real: it force-exits the service cell six times in a row and
10//! requires that `init` stops bringing it back. It runs in the orchestrator context, which is
11//! the one holding `SpawnCap` (`ForceExit` is gated on that capability).
12//!
13//! One warm-up exit plus one window of quiet anchors `init`'s window before the storm, so the
14//! first storm death is the roll and all six land in a single window (see the comment in
15//! `run`; without it a host that reaches the shell faster than one window reports a verdict it
16//! cannot support).
17//!
18//! Markers: `[init-giveup] PASS` / `[init-giveup] SKIP` / `[init-giveup] FAIL — <reason>`.
19
20use alloc::format;
21use api::syscall::service;
22use ostd::{
23    io::println,
24    syscall::{
25        sys_exit, sys_force_exit, sys_get_scheduler_ticks, sys_lookup_service, SyscallResult,
26    },
27    task::yield_now,
28};
29
30/// The service `init` supervises that this scenario kills. `/bin/config` is `Permanent`, holds
31/// no block/network capability (so `ForceExit` is allowed), and registers a well-known id — which
32/// is what makes the give-up observable from userspace.
33const TARGET_SERVICE: u16 = service::CONFIG;
34const TARGET_NAME: &str = "/bin/config";
35
36/// Abnormal exits to deliver. `init` checks `restart_count >= MAX_RESTARTS_PER_WINDOW` *before*
37/// incrementing, so five restarts are allowed and the sixth abnormal exit is the one it refuses.
38const KILLS: u32 = 6;
39/// How long one restart may take before the scenario calls it a failure.
40const RESTART_WAIT_TICKS: u64 = 300;
41/// How long the service must stay down after the last kill for the give-up to count.
42const ABSENT_TICKS: u64 = 200;
43/// The window `init` measures its budget in (`RESTART_WINDOW_TICKS`).
44///
45/// A storm has to fit inside one window: five restarts, then the sixth abnormal exit that
46/// exhausts the budget. When the host cannot restart the service that fast — a slow QEMU-TCG
47/// runner needs seconds per respawn — the window legitimately rolls and `init` is *right* to
48/// keep restarting, so the scenario reports `SKIP` with its measurement instead of a verdict
49/// it cannot support.
50const WINDOW_TICKS: u64 = 1000;
51/// Extra quiet ticks on top of one window before the storm starts.
52///
53/// `init` anchors its window when it *processes* a death, a few ticks after the
54/// exit this scenario delivered — and on a loaded runner the observed
55/// kill→process latency reaches 85 ticks. Without the margin the storm's first
56/// death can land exactly on the boundary (`window_age=1000`, measured on CI
57/// 2026-09-26), where the roll slips to the *second* death and the storm still
58/// loses one count. 200 ticks covers that latency with room to spare.
59const ANCHOR_MARGIN_TICKS: u64 = 200;
60
61fn now() -> u64 {
62    sys_get_scheduler_ticks().unwrap_or(0)
63}
64
65fn fail(why: &str) -> ! {
66    println(&format!("[init-giveup] FAIL — {why}"));
67    sys_exit(1);
68}
69
70/// Force one exit, then wait for `init` to bring the service back. Returns the
71/// tick at which the exit was delivered.
72fn force_exit_and_await_restart(label: &str) -> u64 {
73    let Some(tid) = sys_lookup_service(TARGET_SERVICE) else {
74        fail(&format!(
75            "{label}: {TARGET_NAME} is already absent — init gave up before its budget was spent"
76        ));
77    };
78    if !matches!(sys_force_exit(tid), SyscallResult::Ok(_)) {
79        fail(&format!("{label}: ForceExit refused for tid {tid}"));
80    }
81    let killed_at = now();
82    println(&format!("[init-giveup] {label}: tid={tid}"));
83
84    let deadline = killed_at + RESTART_WAIT_TICKS;
85    while now() < deadline {
86        match sys_lookup_service(TARGET_SERVICE) {
87            Some(new_tid) if new_tid != tid => {
88                println(&format!(
89                    "[init-giveup] {label}: restarted tid={new_tid} after {} ticks",
90                    now() - killed_at
91                ));
92                break;
93            }
94            _ => yield_now(),
95        }
96    }
97    killed_at
98}
99
100pub fn run() -> ! {
101    println(&format!(
102        "[init-giveup] START: {KILLS} forced exits on {TARGET_NAME}"
103    ));
104
105    // Anchor `init`'s window before the storm.
106    //
107    // The budget is a rate: five restarts per 1000-tick window, and the window
108    // rolls at the first death *after* it expires. A service that has not died
109    // since boot still carries `window_start = 0`, so a first death younger than
110    // one window rolls nothing — and then the roll lands *inside* the storm, one
111    // restart is charged to the previous window, the count never reaches the
112    // limit, and this witness reports a FAIL its host cannot support. Measured
113    // on CI (2026-09-26): boot + command in under 1000 ticks, six restarts in
114    // ~150 ticks, no give-up; locally the same storm runs with the first death
115    // at tick 1642 and the give-up lands exactly at the sixth.
116    //
117    // One warm-up exit plus one full window of quiet (plus a margin for the
118    // kill→process latency `init` adds on top of our own clock) makes the anchor
119    // certain: `window_start <= warmup_death < storm_start - WINDOW_TICKS`, so
120    // the storm's first death is guaranteed to be the roll and all six exits fit
121    // one window. A death *inside* the quiet window moves the anchor, so the
122    // window restarts from the last absence observed — the observed time is never
123    // earlier than init's own anchor, so the guarantee survives.
124    let quiet_ticks = WINDOW_TICKS + ANCHOR_MARGIN_TICKS;
125    let mut window_start = force_exit_and_await_restart("warmup");
126    while now().wrapping_sub(window_start) <= quiet_ticks {
127        if sys_lookup_service(TARGET_SERVICE).is_none() {
128            window_start = now();
129        }
130        yield_now();
131    }
132
133    let mut first_exit_tick = 0u64;
134    let mut last_exit_tick = 0u64;
135    for kill in 1..=KILLS {
136        let killed_at = force_exit_and_await_restart(&format!("kill {kill}/{KILLS}"));
137        if kill == 1 {
138            first_exit_tick = killed_at;
139        }
140        last_exit_tick = killed_at;
141    }
142
143    // If the six exits did not fit inside one window, `init` was right to keep restarting and
144    // this host simply cannot host the experiment. Say so with the measurement rather than
145    // reporting a verdict the run cannot support.
146    let span = last_exit_tick.saturating_sub(first_exit_tick);
147    if span > WINDOW_TICKS {
148        println(&format!(
149            "[init-giveup] SKIP — {KILLS} exits spanned {span} ticks, beyond init's {WINDOW_TICKS}-tick window; this host still restarted {TARGET_NAME} correctly"
150        ));
151        sys_exit(0);
152    }
153
154    // The budget is spent: the service must stay down for a whole window's worth of ticks.
155    let deadline = now() + ABSENT_TICKS;
156    while now() < deadline {
157        if let Some(tid) = sys_lookup_service(TARGET_SERVICE) {
158            fail(&format!(
159                "init restarted {TARGET_NAME} again (tid={tid}) — the crash-storm budget never engaged"
160            ));
161        }
162        yield_now();
163    }
164
165    println(&format!(
166        "[init-giveup] PASS (init gave up on {TARGET_NAME} after {KILLS} abnormal exits and left it down)"
167    ));
168    sys_exit(0);
169}
