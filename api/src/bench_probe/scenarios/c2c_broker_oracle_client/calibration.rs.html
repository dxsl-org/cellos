64pub fn run(broker_tid: usize, config: ClientConfig) -> ClientSummary {
65    let sample_count = config.request_count.min(BROKER_CALIBRATION_SAMPLES);
66    let mut summary = ClientSummary {
67        attempted: sample_count,
68        ..ClientSummary::default()
69    };
70    if sample_count == 0 {
71        return summary;
72    }
73    let mut samples = Samples::new(sample_count as usize);
74    let mut tx = [0u8; IPC_BUF_SIZE];
75    let mut rx = [0u8; IPC_BUF_SIZE];
76    let total_calls = BROKER_CALIBRATION_WARMUP.saturating_add(sample_count);
77    for offset in 0..total_calls {
78        let sequence = config.base_sequence + offset as u64;
79        let outcome = measure_call(broker_tid, sequence, &mut tx, &mut rx);
80        if offset < BROKER_CALIBRATION_WARMUP {
81            if !matches!(outcome, CallOutcome::Sample(_)) {
82                summary.warmup_failures = summary.warmup_failures.saturating_add(1);
83            }
84            continue;
85        }
86        match outcome {
87            CallOutcome::Sample(sample) => {
88                summary.success = summary.success.saturating_add(1);
89                samples.push(sample);
90            }
91            CallOutcome::Busy => summary.busy = summary.busy.saturating_add(1),
92            CallOutcome::Indeterminate => {
93                summary.indeterminate = summary.indeterminate.saturating_add(1)
94            }
95            CallOutcome::Correlation => summary.correlation = summary.correlation.saturating_add(1),
96            CallOutcome::InvalidTiming => {
97                summary.success = summary.success.saturating_add(1);
98                summary.timing_invalid = summary.timing_invalid.saturating_add(1);
99            }
100        }
101    }
102    if samples.total.len() == sample_count as usize {
103        let (p50, p99) = percentiles(&mut samples.total);
104        summary.latency_p50_ns = p50;
105        summary.latency_ns = p99;
106        summary.send_latency_ns = percentiles(&mut samples.send).1;
107        summary.reply_wait_ns = percentiles(&mut samples.reply_wait).1;
108        summary.worker_latency_ns = percentiles(&mut samples.worker).1;
109        summary.reply_pump_latency_ns = percentiles(&mut samples.reply_pump).1;
110        summary.client_wake_latency_ns = percentiles(&mut samples.client_wake).1;
111    }
112    summary
113}

115fn measure_call(
116    broker_tid: usize,
117    sequence: u64,
118    tx: &mut [u8; IPC_BUF_SIZE],
119    rx: &mut [u8; IPC_BUF_SIZE],
120) -> CallOutcome {
121    let Ok(len) = encode_echo_request(sequence, ECHO_BODY, tx) else {
122        return CallOutcome::Indeterminate;
123    };
124    let start = sys_get_time();
125    if !matches!(sys_send(broker_tid, &tx[..len]), SyscallResult::Ok(_)) {
126        return CallOutcome::Indeterminate;
127    }
128    let sent = sys_get_time();
129    if !super::super::c2c_broker_oracle::recv_from_broker(broker_tid, rx) {
130        return CallOutcome::Indeterminate;
131    }
132    let received = sys_get_time();
133    let Ok(reply) = decode_reply_frame(rx) else {
134        return CallOutcome::Correlation;
135    };
136    if reply.client_sequence != sequence {
137        return CallOutcome::Correlation;
138    }
139    match reply.status {
140        ReplyStatus::Busy => return CallOutcome::Busy,
141        ReplyStatus::Indeterminate | ReplyStatus::NotSupported => {
142            return CallOutcome::Indeterminate
143        }
144        ReplyStatus::Success => {}
145    }
146    let Ok(timestamps) = decode_timed_echo_reply(reply.payload, ECHO_BODY) else {
147        return CallOutcome::Correlation;
148    };
149    let worker = timestamps.worker_done_ticks;
150    let reply_sent = timestamps.reply_send_ticks;
151    if sent < start || worker < sent || reply_sent < worker || received < reply_sent {
152        return CallOutcome::InvalidTiming;
153    }
154    CallOutcome::Sample(TimedSample {
155        total: ticks_to_ns(received - start),
156        send: ticks_to_ns(sent - start),
157        reply_wait: ticks_to_ns(received - sent),
158        worker: ticks_to_ns(worker - sent),
159        reply_pump: ticks_to_ns(reply_sent - worker),
160        client_wake: ticks_to_ns(received - reply_sent),
161    })
162}

164fn percentiles(samples: &mut [u64]) -> (u64, u64) {
165    samples.sort_unstable();
166    let last = samples.len().saturating_sub(1);
167    let p50 = (samples.len() / 2).min(last);
168    let p99 = ((samples.len() * 99) / 100).min(last);
169    (samples[p50], samples[p99])
170}
