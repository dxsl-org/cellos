9pub fn update_summary(
10    summary: &mut ClientSummary,
11    sequence: u64,
12    rx: &[u8],
13    latency_ns: Option<u64>,
14    config: ClientConfig,
15) {
16    match decode_reply_frame(rx) {
17        Ok(reply) => {
18            summary.latency_ns = latency_ns.unwrap_or(0);
19            count_reply(
20                summary,
21                reply.status,
22                reply.client_sequence == sequence,
23                payload_matches(config, reply.payload),
24            );
25        }
26        Err(_) => summary.correlation += 1,
27    }
28}

30pub fn payload_matches(config: ClientConfig, payload: &[u8]) -> bool {
31    match config.mode {
32        ClientMode::HoldAsync => {
33            payload.len() == 3
34                && payload[0] == service_net_broker::bench_oracle::OP_HOLD
35                && payload[1..3] == config.hold_turns.to_le_bytes()
36        }
37        _ => service_net_broker::bench_oracle::decode_timed_echo_reply(payload, ECHO_BODY).is_ok(),
38    }
39}

41pub fn wait_config() -> (usize, ClientConfig) {
42    let mut buf = [0u8; CONFIG_BYTES];
43    loop {
44        if let SyscallResult::Ok(sender) = sys_recv(0, &mut buf) {
45            if let Some(config) = decode_config(&buf) {
46                return (sender, config);
47            }
48        }
49        yield_now();
50    }
51}

53pub fn wait_broker() -> Option<usize> {
54    for _ in 0..200 {
55        if let Some(broker_tid) = sys_lookup_service(api::syscall::service::NET_BROKER) {
56            return Some(broker_tid);
57        }
58        yield_now();
59    }
60    None
61}

73pub fn finish(parent_tid: usize, summary: ClientSummary, code: usize) -> ! {
74    let mut out = [0u8; SUMMARY_BYTES];
75    encode_summary(summary, &mut out);
76    let _ = sys_send(parent_tid, &out);
77    sys_exit(code)
78}

80pub(super) fn count_reply(
81    summary: &mut ClientSummary,
82    status: ReplyStatus,
83    sequence_ok: bool,
84    payload_ok: bool,
85) {
86    if !sequence_ok {
87        summary.correlation += 1;
88        return;
89    }
90    match status {
91        ReplyStatus::Success if payload_ok => summary.success += 1,
92        ReplyStatus::Success => summary.correlation += 1,
93        ReplyStatus::Busy => summary.busy += 1,
94        ReplyStatus::Indeterminate => summary.indeterminate += 1,
95    }
96}

98fn wait_signal(parent_tid: usize, buf: &mut [u8], accept: fn(&[u8]) -> bool) {
99    while !matches!(sys_recv(parent_tid, buf), SyscallResult::Ok(_)) || !accept(buf) {
100        yield_now();
101    }
