32pub fn snapshot_delta(before: OracleSnapshot, after: OracleSnapshot) -> SnapshotDelta {
33    SnapshotDelta {
34        accepted: after
35            .accepted
36            .saturating_sub(before.accepted)
37            .saturating_sub(1),
38        completed: after
39            .completed
40            .saturating_sub(before.completed)
41            .saturating_sub(1),
42        busy: after.busy.saturating_sub(before.busy),
43        terminal: after.terminal.saturating_sub(before.terminal),
44        indeterminate: after.indeterminate.saturating_sub(before.indeterminate),
45        duplicate: after.duplicate.saturating_sub(before.duplicate),
46        stale_reply: after.stale_reply.saturating_sub(before.stale_reply),
47        heartbeat_miss: after.heartbeat_miss.saturating_sub(before.heartbeat_miss),
48        watchdog_expired: after
49            .watchdog_expired
50            .saturating_sub(before.watchdog_expired),
51        network_progress: after
52            .network_progress
53            .saturating_sub(before.network_progress),
54    }
55}

57pub fn percentile_pair(samples: &mut [u64], len: usize) -> (u64, u64) {
58    samples[..len].sort_unstable();
59    let p50_idx = len.saturating_sub(1) / 2;
60    let p99_idx = ((len * 99).saturating_sub(1) / 100).min(len.saturating_sub(1));
61    (samples[p50_idx], samples[p99_idx])
62}

64pub fn aggregate(summaries: &[ClientSummary]) -> ClientSummary {
65    let mut total = ClientSummary::default();
66    for summary in summaries {
67        total.attempted = total.attempted.saturating_add(summary.attempted);
68        total.success = total.success.saturating_add(summary.success);
69        total.busy = total.busy.saturating_add(summary.busy);
70        total.indeterminate = total.indeterminate.saturating_add(summary.indeterminate);
71        total.correlation = total.correlation.saturating_add(summary.correlation);
72    }
73    total
74}
