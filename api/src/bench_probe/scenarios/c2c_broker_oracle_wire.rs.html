52pub fn encode_config(config: ClientConfig, out: &mut [u8; CONFIG_BYTES]) {
53    out[0] = TAG_CONFIG;
54    out[1] = config.mode as u8;
55    out[2..4].copy_from_slice(&config.request_count.to_le_bytes());
56    out[4..12].copy_from_slice(&config.base_sequence.to_le_bytes());
57    out[12..14].copy_from_slice(&config.hold_turns.to_le_bytes());
58    out[14] = (config.ack_posts as u8)
59        | ((config.wait_for_start as u8) << 1)
60        | ((config.wait_for_drain as u8) << 2);
61}

63pub fn decode_config(buf: &[u8]) -> Option<ClientConfig> {
64    if buf.first().copied()? != TAG_CONFIG || buf.len() < CONFIG_BYTES {
65        return None;
66    }
67    let mode = match buf[1] {
68        1 => ClientMode::EchoSync,
69        2 => ClientMode::EchoAsync,
70        3 => ClientMode::HoldAsync,
71        4 => ClientMode::EchoCalibrate,
72        _ => return None,
73    };
74    Some(ClientConfig {
75        mode,
76        request_count: u16::from_le_bytes(buf[2..4].try_into().ok()?),
77        base_sequence: u64::from_le_bytes(buf[4..12].try_into().ok()?),
78        hold_turns: u16::from_le_bytes(buf[12..14].try_into().ok()?),
79        ack_posts: buf[14] & 0x01 != 0,
80        wait_for_start: buf[14] & 0x02 != 0,
81        wait_for_drain: buf[14] & 0x04 != 0,
82    })
83}
