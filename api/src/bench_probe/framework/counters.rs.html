1//! Kernel benchmark-counter deltas around one scenario (raw opcode 504).
2//!
3//! `bench` reads the kernel's versioned counter record before and after each
4//! scenario and prints one `<scenario>.counters` JSON line. The counters are
5//! global (every task, not just this scenario's cells) and the delta spans the
6//! whole scenario: setup, warmup, measurement, teardown and the closing read.
7//! A kernel without the `bench-counters` feature refuses the opcode; every value
8//! is then reported as `"unknown"`, never as zero.
9
10use alloc::{format, string::String};
11use ostd::{io::println, syscall::sys_bench_counters};
12
13const RECORD_VERSION: u64 = 1;
14
15/// Record words 1..=7 in kernel order (word 0 is the version).
16const FIELDS: [&str; 7] = [
17    "syscalls",
18    "domain_syscall_cr3",
19    "root_programmings",
20    "context_switches",
21    "ipc_bytes_copied",
22    "heap_allocs",
23    "heap_alloc_bytes",
24];
25
26/// Cost-accounting labels; the collector checks them against its contract.
27const LABELS: &[(&str, &str, &str)] = &[
28    ("context_switch", "sas", "sched"),
29    ("ipc_send_recv", "sas", "copy"),
30    ("ipc_send_recv_domain", "domain", "copy"),
31    ("ipc_fastpath", "sas", "ring"),
32    ("syscall_yield", "sas", "syscall"),
33    ("memory_footprint", "sas", "footprint"),
34    ("preempt_latency", "sas", "sched"),
35    ("control_loop", "sas", "sched"),
36    ("ipc_send_recv_idle", "sas", "copy"),
37    ("syscall_yield_idle", "sas", "syscall"),
38    ("ipc_send_recv_load", "sas", "copy"),
39    ("syscall_yield_load", "sas", "syscall"),
40    ("smp_spawn_rate", "sas", "spawn"),
41    ("smp_ipc_throughput", "sas", "copy"),
42    ("smp_work_distribution", "sas", "sched"),
43    ("stage_encode_request_x1000", "sas", "codec"),
44    ("stage_decode_reply_x1000", "sas", "codec"),
45    ("stage_ecall_roundtrip_x1000", "sas", "syscall"),
46    ("total_typed_roundtrip_x1000", "sas", "copy"),
47];
48
49/// One counter record, or `None` when the kernel refuses the opcode.
50pub type Snapshot = Option<[u64; 8]>;
51
52pub fn read() -> Snapshot {
53    let mut record = [0u64; 8];
54    match sys_bench_counters(&mut record) {
55        Ok(()) if record[0] == RECORD_VERSION => Some(record),
56        _ => None,
57    }
58}
59
60/// Read the closing record and print `scenario`'s delta since `before`.
61pub fn emit(scenario: &str, before: Snapshot) {
62    let after = read();
63    let (tier, path) = LABELS
64        .iter()
65        .find(|(name, _, _)| *name == scenario)
66        .map_or(("unknown", "unknown"), |&(_, tier, path)| (tier, path));
67    let mut line = format!(
68        "{{\"name\":\"{}.counters\",\"tier\":\"{}\",\"path\":\"{}\"",
69        scenario, tier, path
70    );
71    for (index, field) in FIELDS.iter().enumerate() {
72        let delta = match (before, after) {
73            (Some(start), Some(end)) => end[index + 1].checked_sub(start[index + 1]),
74            _ => None,
75        };
76        let value = match delta {
77            Some(value) => format!("{}", value),
78            None => String::from("\"unknown\""),
79        };
80        line.push_str(&format!(",\"{}\":{}", field, value));
81    }
82    line.push('}');
83    println(&line);
84}
