5/// Page Table Entry Flags
6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7pub struct PageFlags(pub usize);
8
9impl PageFlags {
10    // Standard flags (mapping to typical HW bits)
11    pub const VALID: usize = 1 << 0;
12    pub const READ: usize = 1 << 1;
13    pub const WRITE: usize = 1 << 2;
14    pub const EXECUTE: usize = 1 << 3;
15    pub const USER: usize = 1 << 4;
16    /// Global mapping: kept in every translation context.
17    ///
18    /// x86_64 sets `G` (bit 8); RISC-V copies it into the Sv39 `G` bit;
19    /// AArch64 ignores it (its leaves are global unless `NON_GLOBAL`).
20    pub const GLOBAL: usize = 1 << 5;
21    pub const ACCESSED: usize = 1 << 6;
22    pub const DIRTY: usize = 1 << 7;
23    /// Device MMIO mapping — use non-cacheable Device-nGnRnE attributes (AArch64 MAIR index 0).
24    /// On RISC-V and x86_64 this flag is ignored; x86_64 MMIO asks for
25    /// [`Self::UNCACHED`] instead.
26    pub const DEVICE: usize = 1 << 8;
27    /// Address-space-scoped (non-global) mapping: the leaf belongs to the
28    /// architectural tag in use, not to every root.
29    ///
30    /// AArch64 translates this to `PTE_nG` (bit 11). Without it the leaf is
31    /// **global**, which has two consequences that matter for private roots:
32    /// an ASID-targeted invalidation (`tlbi aside1is`) never clears it, and the
33    /// translation stays usable after `TTBR0_EL1` is reprogrammed to a different
34    /// root. Private-root-only leaves therefore carry this bit so the tag-release
35    /// contract can be satisfied by targeted invalidation alone.
36    ///
37    /// RISC-V and x86_64 ignore it. Bit 9 is free in both: the x86_64 leaf
38    /// translator composes only `P`/`RW`/`US`/`NX`/`G`/`PCD` from the bits
39    /// documented for it and never reads any other bit, and on RISC-V the
40    /// software bitfield is copied
41    /// verbatim into the Sv39 leaf where bit 9 is `RSW[1]` — reserved for
42    /// supervisor software, ignored by the hardware for permission checks and
43    /// never consulted by `translate`/`unmap` (which mask the output address).
44    pub const NON_GLOBAL: usize = 1 << 9;
45
46    /// AArch64 BTI guarded page: the stage-1 leaf carries `GP` (bit 50), so an
47    /// indirect branch into it must land on a `BTI` landing pad (FEAT_BTI).
48    ///
49    /// Only meaningful together with `EXECUTE`, and only set by the loader for
50    /// executable pages of an image whose `GNU_PROPERTY_AARCH64_FEATURE_1_AND`
51    /// note declares BTI, on a CPU that implements FEAT_BTI (bit 50 is RES0
52    /// without it). x86_64 ignores it; RISC-V clears it, because bit 10 of an
53    /// Sv39 leaf is the low bit of the PPN.
54    pub const GUARDED: usize = 1 << 10;
55
56    /// Uncached mapping: x86_64 sets `PCD` (cache-disable, bit 4) on the leaf.
57    ///
58    /// The cache attribute of x86_64 MMIO on the shared kernel root. AArch64
59    /// ignores it (device memory is [`Self::DEVICE`]); RISC-V clears it,
60    /// because bit 11 of an Sv39 leaf is a PPN bit.
61    pub const UNCACHED: usize = 1 << 11;
62
63    pub const R_W_X: usize = Self::READ | Self::WRITE | Self::EXECUTE;
64
65    pub fn from_bits(bits: usize) -> Self {
66        Self(bits)
67    }
68
69    pub fn bits(&self) -> usize {
70        self.0
71    }
72
73    pub fn is_valid(&self) -> bool {
74        (self.0 & Self::VALID) != 0
75    }
76    pub fn is_writeable(&self) -> bool {
77        (self.0 & Self::WRITE) != 0
78    }
79    // ... add more as needed
80}

82/// One leaf translation as the hardware walker sees it, in neutral terms.
83///
84/// `pa` is the leaf's output address. For a 4 KiB leaf it is the frame on
85/// every architecture. For a larger (block/huge) leaf, RISC-V adds the offset
86/// of the walked address inside the leaf, x86_64 reports bits 51:12 of the
87/// entry and AArch64 bits 47:12, as the walkers this type replaced did.
88#[derive(Debug, Clone, Copy, PartialEq, Eq)]
89pub struct Leaf {
90    pub pa: PhysAddr,
91    /// The entry is valid (present).
92    pub valid: bool,
93    /// Loads are permitted. Always true on x86_64 and AArch64.
94    pub read: bool,
95    /// Stores are permitted.
96    pub write: bool,
97    /// User mode may access the page.
98    pub user: bool,
99    /// Instruction fetch is permitted at the privilege the leaf grants
100    /// (AArch64: `UXN` for a user leaf, `PXN` otherwise).
101    pub exec: bool,
102}

104/// Which intermediate tables a new mapping makes user-reachable.
105///
106/// Only x86_64 carries a user bit on intermediate entries (U/S is ANDed down
107/// the walk). RISC-V non-leaf entries have none and AArch64 leaves `APTable`
108/// zero, so both treat the two policies alike.
109#[derive(Debug, Clone, Copy, PartialEq, Eq)]
110pub enum TablePolicy {
111    /// A new intermediate table is user-reachable only under a `USER` leaf,
112    /// and a `USER` leaf upgrades every existing table on its walk, so a
113    /// supervisor leaf is also refused at the table level. Private roots.
114    FollowLeaf,
115    /// Every new intermediate table is user-reachable and an existing one is
116    /// left as it is: the leaf's own U/S bit is the only privilege check. The
117    /// shared kernel root.
118    UserReachable,
119}

121/// Generic Page Table Trait
122pub trait PageTableTrait {
123    /// Initialize paging (e.g., allocate root table).
124    /// Typically returns the Physical Address of the root table.
125    fn init(&mut self) -> ViResult<PhysAddr>;
126
127    /// Map a virtual page to a physical frame.
128    /// `alloc_fn` is a closure to allocate new frames for intermediate tables.
129    fn map(
130        &mut self,
131        virt: VAddr,
132        phys: PhysAddr,
133        flags: PageFlags,
134        alloc_fn: &mut dyn FnMut() -> Option<PhysAddr>,
135    ) -> ViResult<()>;
136
137    /// [`map`](Self::map) with an explicit intermediate-table policy; `map`
138    /// itself is [`TablePolicy::FollowLeaf`].
139    fn map_with(
140        &mut self,
141        virt: VAddr,
142        phys: PhysAddr,
143        flags: PageFlags,
144        tables: TablePolicy,
145        alloc_fn: &mut dyn FnMut() -> Option<PhysAddr>,
146    ) -> ViResult<()> {
147        // Only x86_64 has a table-level user bit (see `TablePolicy`).
148        let _ = tables;
149        self.map(virt, phys, flags, alloc_fn)
150    }
151
152    /// Rewrite the permissions of the 4 KiB page mapped at `virt`, keeping the
153    /// frame it maps. `Err(NotFound)` when nothing maps `virt`.
154    ///
155    /// The default re-maps the translated frame with `flags`, which replace
156    /// the old flags wholesale: without `VALID` the page is unmapped. x86_64
157    /// keeps the frame and the cache and global bits of the old leaf and
158    /// always sets `P`. Never allocates. No TLB invalidation beyond what
159    /// [`map`](Self::map) does; the caller owns it.
160    fn protect(&mut self, virt: VAddr, flags: PageFlags) -> ViResult<()> {
161        let phys = self.translate(virt).ok_or(ViError::NotFound)? & !0xFFF;
162        self.map(virt, phys, flags, &mut || None)
163    }
164
165    /// Unmap a virtual page.
166    fn unmap(&mut self, virt: VAddr) -> ViResult<()>;
167
168    /// Translate virtual address to physical address.
169    fn translate(&self, virt: VAddr) -> Option<PhysAddr>;
170
171    /// The leaf translating `virt`, or `None` when no valid leaf does.
172    fn leaf(&self, virt: VAddr) -> Option<Leaf>;
173
174    /// Identity map a region (Virtual = Physical).
175    fn identity_map(
176        &mut self,
177        start: PhysAddr,
178        end: PhysAddr,
179        flags: PageFlags,
180        alloc_fn: &mut dyn FnMut() -> Option<PhysAddr>,
181    ) -> ViResult<()> {
182        let mut addr = start;
183        // Assume generic page size 4KB for default identity map loop,
184        // implementation can override for huge pages optimization.
185        while addr < end {
186            self.map(addr, addr, flags, alloc_fn)?;
187            addr += 4096;
188        }
189        Ok(())
190    }
191
192    /// Activate this page table (load into CR3/SATP).
193    /// # Safety
194    /// Must ensure code/stack is mapped.
195    unsafe fn activate(&self);
196}
