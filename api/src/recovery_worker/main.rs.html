1//! `recovery-worker`: serves the demo counter but owns no business state. It
2//! attaches to its authority with the one-shot nonce its parent staged, refuses
3//! to serve unless the restored schema is its own, and commits every mutation at
4//! the authority before replying (CP1 §4.2). Registration (Ready) is the
5//! authority's decision, never the worker's.
6

48/// Attach round trip with the authority, admission included.
49const ATTACH_TICKS: u64 = 500;
50
51fn recovery_failed(why: &str) -> ! {
52    println(&format!("[recovery-worker] RecoveryFailed: {why}"));
53    ostd::syscall::sys_exit(EXIT_RECOVERY_FAILED)
54}
55
56fn staged_nonce() -> Option<u64> {
57    let args = ostd::args();
58    let value = args.iter().find_map(|arg| arg.strip_prefix("nonce="))?;
59    u64::from_str_radix(value, 16).ok()
60}
61
62fn cell_main() {
63    let Some(nonce) = staged_nonce() else {
64        recovery_failed("no attach nonce")
65    };
66    let Some(authority) = common::wait_lookup(AUTHORITY_SERVICE, 500) else {
67        recovery_failed("authority unavailable")
68    };
69    let attached = common::call(authority, Msg::Attach { nonce }, ATTACH_TICKS);
70    let (epoch, revision, counter) = match attached {
71        Some(Msg::Attached {
72            epoch,
73            revision,
74            schema,
75            counter,
76        }) => {
77            if schema != SCHEMA {
78                // One-way: the authority fences and answers nothing.
79                common::post(authority, Msg::RestoreFailed { epoch });
80                recovery_failed(&format!("schema {schema} is not {SCHEMA}"));
81            }
82            (epoch, revision, counter)
83        }
84        Some(Msg::Refused(refusal)) => recovery_failed(&format!("attach refused {refusal:?}")),
85        _ => recovery_failed("no attach reply"),
86    };
87    // One-way: publication is the authority's decision and it answers nothing.
88    if !common::post(authority, Msg::Restored { epoch, revision }) {
89        recovery_failed("authority unreachable");
90    }
91    println(&format!(
92        "[recovery-worker] restored epoch={epoch} revision={revision} counter={counter}"
93    ));
94    serve::run(serve::Worker {
95        authority,
96        epoch,
97        revision,
98        counter,
99    });
100}
