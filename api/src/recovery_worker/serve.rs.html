1//! The worker's request loop. Every mutation is looked up and then committed at
2//! the authority under the worker's epoch before the client sees a reply, so a
3//! crash at any point leaves either no effect or a committed, reconcilable one.
4//! Each client request is answered exactly once, through its own [`Incoming`].
5
6use alloc::format;
7use ostd::io::println;
8use ostd::ipc::Incoming;
9use recovery_demo::ledger::apply;
10use recovery_demo::wire::{Binding, Fault, Msg, Refusal};
11use recovery_demo::EXIT_FENCED;
12
13use crate::common;
14
15/// Ticks for one authority round trip, admission included, before failing the
16/// request closed.
17const AUTHORITY_TICKS: u64 = 500;
18/// Ticks a stalled mutation waits for the client's `Go`.
19const STALL_TICKS: u64 = 2_000;
20
21pub struct Worker {
22    pub authority: usize,
23    pub epoch: u64,
24    pub revision: u64,
25    pub counter: u64,
26}
27
28pub fn run(mut worker: Worker) -> ! {
29    loop {
30        let Some(rx) = common::recv_attested(0) else {
31            continue;
32        };
33        if rx.sender == worker.authority {
34            continue;
35        }
36        let Some((cell, generation)) = rx.principal() else {
37            continue;
38        };
39        let rights = rx.identity.map_or(0, |id| id.flags);
40        match rx.msg {
41            Some(Msg::Get) => {
42                common::reply(
43                    &rx.request,
44                    Msg::Value {
45                        counter: worker.counter,
46                        revision: worker.revision,
47                        epoch: worker.epoch,
48                    },
49                );
50            }
51            Some(Msg::Mutate {
52                request_id,
53                method,
54                amount,
55                fault,
56            }) => {
57                let binding = Binding {
58                    caller_cell: cell,
59                    caller_generation: generation,
60                    request_id,
61                    method,
62                    rights,
63                    payload: amount,
64                };
65                worker.mutate(rx.request, binding, fault);
66            }
67            #[cfg(feature = "fault-inject")]
68            Some(Msg::Crash { mode, arg }) => {
69                common::reply(&rx.request, Msg::Ack);
70                crate::fault::scenario(mode, arg);
71            }
72            _ => {}
73        }
74    }
75}
76
77impl Worker {
78    fn call(&self, msg: Msg) -> Option<Msg> {
79        common::call(self.authority, msg, AUTHORITY_TICKS)
80    }
81
82    /// `request` is the client's `Mutate`. On the stall path `Staged` answers
83    /// it and the outcome answers the client's later `Go` instead.
84    fn mutate(&mut self, mut request: Incoming, binding: Binding, fault: Fault) {
85        let request_id = binding.request_id;
86        match self.call(Msg::Lookup {
87            epoch: self.epoch,
88            binding,
89        }) {
90            Some(Msg::Present { revision, result }) => {
91                common::reply(
92                    &request,
93                    Msg::Done {
94                        request_id,
95                        result,
96                        revision,
97                        replayed: true,
98                    },
99                );
100                return;
101            }
102            Some(Msg::Absent) => {}
103            Some(Msg::Refused(refusal)) => return self.refuse(&request, request_id, refusal),
104            _ => return self.refuse(&request, request_id, Refusal::RecoveryFailed),
105        }
106        let Some(new_state) = apply(binding.method, self.counter, binding.payload) else {
107            return self.refuse(&request, request_id, Refusal::InvalidMutation);
108        };
109        self.fault_point(fault, Fault::BeforeCommit);
110        if cfg!(feature = "fault-inject") && fault == Fault::Stall {
111            let client = request.sender();
112            common::reply(&request, Msg::Staged { request_id });
113            match common::recv_request_from(client, STALL_TICKS) {
114                Some((Some(Msg::Go), go)) => request = go,
115                Some((_, other)) => return self.refuse(&other, request_id, Refusal::Denied),
116                None => {
117                    // No request is left to answer: tell the client one-way.
118                    common::post(
119                        client,
120                        Msg::Failed {
121                            request_id,
122                            refusal: Refusal::Denied,
123                        },
124                    );
125                    return;
126                }
127            }
128        }
129        let commit = Msg::Commit {
130            epoch: self.epoch,
131            expected_revision: self.revision,
132            binding,
133            new_state,
134        };
135        match self.call(commit) {
136            Some(Msg::Committed { revision, result }) => {
137                self.revision = revision;
138                self.counter = result;
139                self.fault_point(fault, Fault::AfterCommit);
140                common::reply(
141                    &request,
142                    Msg::Done {
143                        request_id,
144                        result,
145                        revision,
146                        replayed: false,
147                    },
148                );
149            }
150            Some(Msg::Present { revision, result }) => {
151                common::reply(
152                    &request,
153                    Msg::Done {
154                        request_id,
155                        result,
156                        revision,
157                        replayed: true,
158                    },
159                );
160            }
161            Some(Msg::Refused(refusal)) => self.refuse(&request, request_id, refusal),
162            _ => self.refuse(&request, request_id, Refusal::RecoveryFailed),
163        }
164    }
165
166    /// A fenced writer reports the refusal and exits: it can never commit again.
167    fn refuse(&self, request: &Incoming, request_id: u64, refusal: Refusal) {
168        common::reply(
169            request,
170            Msg::Failed {
171                request_id,
172                refusal,
173            },
174        );
175        if refusal == Refusal::StaleEpoch {
176            println(&format!(
177                "[recovery-worker] epoch {} fenced: exiting",
178                self.epoch
179            ));
180            ostd::syscall::sys_exit(EXIT_FENCED);
181        }
182    }
183
184    #[cfg(feature = "fault-inject")]
185    fn fault_point(&self, requested: Fault, here: Fault) {
186        if requested == here {
187            println(&format!("[recovery-worker] injected fault {here:?}"));
188            crate::fault::crash();
189        }
190    }
191
192    #[cfg(not(feature = "fault-inject"))]
193    fn fault_point(&self, _requested: Fault, _here: Fault) {}
194}
