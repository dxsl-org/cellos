2//! Non-production control protocol for hostile backend recovery images.
3//!
4//! This module is available only through the `hostile-backend-recovery` feature.

6/// Opcode for a request to terminate a supervised hypervisor backend.
7pub const KILL_REQUEST_OPCODE: u8 = 0x04;
8/// Opcode for the bounded response to a backend kill request.
9pub const KILL_RESPONSE_OPCODE: u8 = 0x05;
10/// Exact encoded size of a backend kill request.
11pub const KILL_REQUEST_LEN: usize = 3;
12/// Exact encoded size of a backend kill response.
13pub const KILL_RESPONSE_LEN: usize = 2;
14
15/// The requested backend was terminated.
16pub const KILL_STATUS_OK: u8 = 0x00;
17/// The request did not match the exact wire contract.
18pub const KILL_STATUS_INVALID_REQUEST: u8 = 0x01;
19/// The request sender was not the hypervisor Cell.
20pub const KILL_STATUS_REJECTED_CALLER: u8 = 0x02;
21/// The requested service is not a killable hostile-test backend.
22pub const KILL_STATUS_SERVICE_NOT_ALLOWED: u8 = 0x03;
23/// No current provider is registered for the requested service.
24pub const KILL_STATUS_SERVICE_NOT_FOUND: u8 = 0x04;
25/// The SupervisorCap-backed kill operation failed.
26pub const KILL_STATUS_KILL_FAILED: u8 = 0x05;
27
28/// Stable prefix for host-authored backend disconnect evidence.
29pub const DISCONNECT_LOG_MARKER: &str = "HOSTILE_BACKEND_DISCONNECT";
30
31/// Encode a strict backend kill request as `[opcode, service_id_le]`.
32pub const fn encode_kill_request(service_id: u16) -> [u8; KILL_REQUEST_LEN] {
33    let service_id = service_id.to_le_bytes();
34    [KILL_REQUEST_OPCODE, service_id[0], service_id[1]]
35}
36
37/// Parse a canonical backend kill request and return its service ID.
38///
39/// IPC receive buffers may be larger than the three-byte frame, so canonical
40/// zero padding is accepted. A missing prefix, mismatched opcode, or any
41/// non-zero trailing byte is rejected.
42pub fn parse_kill_request(bytes: &[u8]) -> Option<u16> {
43    if bytes.len() < KILL_REQUEST_LEN
44        || bytes[0] != KILL_REQUEST_OPCODE
45        || bytes[KILL_REQUEST_LEN..].iter().any(|&byte| byte != 0)
46    {
47        return None;
48    }
49    Some(u16::from_le_bytes([bytes[1], bytes[2]]))
50}
51
52/// Encode a strict backend kill response as `[opcode, status]`.
53pub const fn encode_kill_response(status: u8) -> [u8; KILL_RESPONSE_LEN] {
54    [KILL_RESPONSE_OPCODE, status]
55}
56
57/// Parse a canonical backend kill response and return its status byte.
58///
59/// IPC receive buffers may be larger than the two-byte frame, so canonical
60/// zero padding is accepted. A missing prefix, mismatched opcode, or any
61/// non-zero trailing byte is rejected.
62pub fn parse_kill_response(bytes: &[u8]) -> Option<u8> {
63    if bytes.len() < KILL_RESPONSE_LEN
64        || bytes[0] != KILL_RESPONSE_OPCODE
65        || bytes[KILL_RESPONSE_LEN..].iter().any(|&byte| byte != 0)
66    {
67        return None;
68    }
69    Some(bytes[1])
70}

72#[cfg(test)]
73mod tests {
74    use super::{
75        encode_kill_request, encode_kill_response, parse_kill_request, parse_kill_response,
76        KILL_REQUEST_OPCODE, KILL_RESPONSE_OPCODE,
77    };
78
79    #[test]
80    fn request_is_strict_little_endian_with_canonical_padding() {
81        assert_eq!(
82            encode_kill_request(0x1234),
83            [KILL_REQUEST_OPCODE, 0x34, 0x12]
84        );
85        let mut padded = [0u8; 4096];
86        padded[..3].copy_from_slice(&encode_kill_request(0x1234));
87        assert_eq!(parse_kill_request(&padded), Some(0x1234));
88        assert_eq!(parse_kill_request(&[KILL_REQUEST_OPCODE, 0x34]), None);
89        padded[3] = 1;
90        assert_eq!(parse_kill_request(&padded), None);
91    }
92
93    #[test]
94    fn response_is_strict_with_canonical_padding() {
95        assert_eq!(encode_kill_response(7), [KILL_RESPONSE_OPCODE, 7]);
96        let mut padded = [0u8; 8];
97        padded[..2].copy_from_slice(&encode_kill_response(7));
98        assert_eq!(parse_kill_response(&padded), Some(7));
99        assert_eq!(parse_kill_response(&[KILL_RESPONSE_OPCODE]), None);
100        padded[2] = 1;
101        assert_eq!(parse_kill_response(&padded), None);
102    }
103}
