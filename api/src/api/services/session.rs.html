64#[derive(Clone, Copy, Debug, PartialEq, Eq)]
65pub struct EventHeaderV1 {
66    pub sequence: u64,
67    pub epoch: u64,
68    pub kind: u16,
69    pub payload_len: u16,
70    pub flags: u32,
71}
72impl EventHeaderV1 {
73    /// Decode one complete frame, including canonical payload validation.
74    pub fn decode(record: &[u8]) -> Result<Self, Status> {
75        let mut r = Reader::new(record);
76        let header = Self { sequence: r.u64()?, epoch: r.u64()?, kind: r.u16()?, payload_len: r.u16()?, flags: r.u32()? };
77        if header.flags != 0 { return Err(Status::InvalidFlags); }
78        validate_event(header.kind, r.take(header.payload_len as usize)?)?;
79        r.finish()?;
80        Ok(header)
81    }
82    /// Encode the fixed header only; payload is written separately.
83    pub fn encode(&self, out: &mut [u8]) -> Result<(), Status> {
84        if self.flags != 0 { return Err(Status::InvalidFlags); }
85        if self.payload_len as usize > TRANSFER_MAX { return Err(Status::Malformed); }
86        if !(event_kind::KEY..=event_kind::TRUST_PROMPT_END).contains(&self.kind) { return Err(Status::UnsupportedInput); }
87        let mut w = Writer::new(out.get_mut(..EVENT_HEADER_LEN).ok_or(Status::BufferTooSmall)?);
88        w.u64(self.sequence)?; w.u64(self.epoch)?; w.u16(self.kind)?;
89        w.u16(self.payload_len)?; w.u32(0)
90    }
91}

93pub fn validate_dimensions(rows: u16, cols: u16) -> Result<(), Status> {
94    if rows == 0 || rows > 256 || cols == 0 || cols > 512
95        || u32::from(rows).checked_mul(u32::from(cols)).ok_or(Status::Malformed)? > 131072 {
96        Err(Status::Malformed)
97    } else { Ok(()) }
98}
99fn valid_key(code: u32) -> bool {
100    (code >= 0x20 && char::from_u32(code).is_some() && code != 0x7f)
101        || matches!(code, key_code::ENTER | key_code::ESCAPE | key_code::BACKSPACE |
102            key_code::TAB | key_code::DELETE | key_code::INSERT | key_code::UP |
103            key_code::DOWN | key_code::LEFT | key_code::RIGHT | key_code::HOME |
104            key_code::END | key_code::PAGE_UP | key_code::PAGE_DOWN)
105        || (key_code::F1..=key_code::F12).contains(&code)
106}
107/// Validate one normalized event. Owners separately restrict which event kinds
108/// a frontend may submit and assign all sequence/epoch values themselves.
109pub fn validate_event(kind: u16, payload: &[u8]) -> Result<(), Status> {
110    use event_kind::*;
111    match kind {
112        KEY => {
113            let mut r = Reader::new(payload);
114            let code = r.u32()?;
115            let mods = r.u32()?;
116            let pressed = r.u8()?;
117            if !valid_key(code) || mods & !modifiers::ALL != 0 { return Err(Status::UnsupportedInput); }
118            if pressed > 2 || r.take(7)?.iter().any(|v| *v != 0) { return Err(Status::Malformed); }
119            r.finish()?;
120            encode_key(code, mods, pressed, &mut [0; 16]).map(|_| ())
121        }
122        TEXT => {
123            if payload.is_empty() || payload.len() > TRANSFER_MAX { return Err(Status::Malformed); }
124            validate_paste(payload)
125        }
126        PASTE_BEGIN | PASTE_END | INTERRUPT | EOF => {
127            if payload.is_empty() { Ok(()) } else { Err(Status::Malformed) }
128        }
129        RESIZE => {
130            let mut r = Reader::new(payload);
131            validate_dimensions(r.u16()?, r.u16()?)?;
132            r.finish()
133        }
134        LOST => {
135            // 1..=3 routine input loss; 4 FrontendLost and 5 FrontendAttached (S5-3 annex §3).
136            let mut r = Reader::new(payload);
137            let reason = r.u32()?;
138            if !(1..=3).contains(&reason) && reason != crate::cell_session::frontend::LOST_FRONTEND
139                && reason != crate::cell_session::frontend::FRONTEND_ATTACHED { return Err(Status::Malformed); }
140            r.finish()
141        }
142        TRUST_PROMPT => crate::cell_session::trust::TrustPromptV1::decode(payload).map(|_| ()),
143        TRUST_PROMPT_END => crate::cell_session::trust::decode_prompt_end(payload).map(|_| ()),
144        _ => Err(Status::UnsupportedInput),
145    }
146}
147/// Printable committed Text outside a paste batch.
148pub fn validate_text(bytes: &[u8]) -> Result<(), Status> {
149    if bytes.is_empty() || bytes.len() > TRANSFER_MAX { return Err(Status::Malformed); }
150    let text = core::str::from_utf8(bytes).map_err(|_| Status::UnsupportedInput)?;
151    if text.chars().any(|c| c <= '\u{1f}' || c == '\u{7f}') {
152        Err(Status::UnsupportedInput)
153    } else { Ok(()) }
154}
155/// Whole paste admission precedes any PasteBegin/Text/PasteEnd delivery.
156pub fn validate_paste(bytes: &[u8]) -> Result<(), Status> {
157    if bytes.len() > PASTE_MAX { return Err(Status::QuotaExceeded); }
158    let text = core::str::from_utf8(bytes).map_err(|_| Status::UnsupportedInput)?;
159    if text.chars().any(|c| (c <= '\u{1f}' && !matches!(c, '\t' | '\r' | '\n')) || c == '\u{7f}') {
160        Err(Status::UnsupportedInput)
161    } else { Ok(()) }
162}

164/// Encode one frozen logical Key into raw bytes. Release emits zero bytes;
165/// unsupported combinations never write a prefix to the caller's buffer.
166pub fn encode_key(code: u32, mods: u32, pressed: u8, out: &mut [u8]) -> Result<usize, Status> {
167    use key_code::*;
168    if !valid_key(code) || mods & !modifiers::ALL != 0 { return Err(Status::UnsupportedInput); }
169    if pressed > 2 { return Err(Status::Malformed); }
170    let mut bytes = [0u8; 16];
171    let mut w = Writer::new(&mut bytes);
172    let alt = mods & modifiers::ALT != 0;
173    let ctrl = mods & modifiers::CONTROL != 0;
174    let shift = mods & modifiers::SHIFT != 0;
175    if code <= 0x10ffff {
176        if !alt && !ctrl { return Err(Status::UnsupportedInput); }
177        if alt { w.u8(0x1b)?; }
178        if ctrl {
179            let byte = match code {
180                0x61..=0x7a => (code - 0x60) as u8,
181                0x41..=0x5a => (code - 0x40) as u8,
182                0x20 | 0x40 => 0, 0x5b => 0x1b, 0x5c => 0x1c,
183                0x5d => 0x1d, 0x5e => 0x1e, 0x5f => 0x1f, 0x3f => 0x7f,
184                _ => return Err(Status::UnsupportedInput),
185            };
186            w.u8(byte)?;
187        } else {
188            let mut utf8 = [0; 4];
189            w.bytes(char::from_u32(code).ok_or(Status::UnsupportedInput)?.encode_utf8(&mut utf8).as_bytes())?;
190        }
191    } else if matches!(code, ENTER | BACKSPACE | ESCAPE | TAB) {
192        if ctrl { return Err(Status::UnsupportedInput); }
193        if alt { w.u8(0x1b)?; }
194        match code {
195            ENTER => w.u8(b'\r')?, BACKSPACE => w.u8(0x7f)?, ESCAPE => w.u8(0x1b)?,
196            TAB if shift => w.bytes(b"\x1b[Z")?, TAB => w.u8(b'\t')?, _ => unreachable!(),
197        }
198    } else {
199        let parameter = 1 + u8::from(shift) + 2 * u8::from(alt) + 4 * u8::from(ctrl);
200        let final_byte = match code {
201            UP => Some(b'A'), DOWN => Some(b'B'), RIGHT => Some(b'C'), LEFT => Some(b'D'),
202            HOME => Some(b'H'), END => Some(b'F'), F1 => Some(b'P'),
203            0x110021 => Some(b'Q'), 0x110022 => Some(b'R'), 0x110023 => Some(b'S'), _ => None,
204        };
205        if let Some(final_byte) = final_byte {
206            if parameter == 1 && (F1..=F1 + 3).contains(&code) { w.bytes(b"\x1bO")?; }
207            else { w.bytes(b"\x1b[")?; }
208            if parameter != 1 { w.bytes(&[b'1', b';', b'0' + parameter])?; }
209            w.u8(final_byte)?;
210        } else {
211            let number: u8 = match code {
212                INSERT => 2, DELETE => 3, PAGE_UP => 5, PAGE_DOWN => 6,
213                0x110024 => 15, 0x110025 => 17, 0x110026 => 18, 0x110027 => 19,
214                0x110028 => 20, 0x110029 => 21, 0x11002a => 23, F12 => 24,
215                _ => return Err(Status::UnsupportedInput),
216            };
217            w.bytes(b"\x1b[")?;
218            if number >= 10 { w.u8(b'0' + number / 10)?; }
219            w.u8(b'0' + number % 10)?;
220            if parameter != 1 { w.bytes(&[b';', b'0' + parameter])?; }
221            w.u8(b'~')?;
222        }
223    }
224    let len = w.position();
225    if pressed == 0 { return Ok(0); }
226    out.get_mut(..len).ok_or(Status::BufferTooSmall)?.copy_from_slice(&bytes[..len]);
227    Ok(len)
228}

230fn event(kind: u16, payload: &[u8], out: &mut [u8]) -> Result<usize, Status> {
231    let len = EVENT_HEADER_LEN + payload.len();
232    let out = out.get_mut(..len).ok_or(Status::BufferTooSmall)?;
233    EventHeaderV1 { sequence: 0, epoch: 0, kind, payload_len: payload.len() as u16, flags: 0 }.encode(out)?;
234    out[EVENT_HEADER_LEN..].copy_from_slice(payload);
235    Ok(len)
236}
237fn key_event(code: u32, mods: u32, out: &mut [u8]) -> Result<usize, Status> {
238    let mut payload = [0; 16];
239    let mut w = Writer::new(&mut payload);
240    w.u32(code)?; w.u32(mods)?; w.u8(1)?; w.zero(7)?;
241    event(event_kind::KEY, &payload, out)
242}
243fn scalar_event(bytes: &[u8], alt: bool, out: &mut [u8]) -> Result<usize, Status> {
244    if bytes.len() == 1 {
245        let (code, mods) = match bytes[0] {
246            0 => (0x20, modifiers::CONTROL),
247            b'\t' => (key_code::TAB, 0), b'\r' => (key_code::ENTER, 0),
248            0x1b => (key_code::ESCAPE, 0), 0x7f => (key_code::BACKSPACE, 0),
249            1..=26 => (u32::from(bytes[0]) + 0x60, modifiers::CONTROL),
250            0x1c..=0x1f => (u32::from(bytes[0]) + 0x40, modifiers::CONTROL),
251            _ => (u32::from(bytes[0]), 0),
252        };
253        if mods != 0 || alt || bytes[0] < 0x20 || bytes[0] == 0x7f {
254            return key_event(code, mods | if alt { modifiers::ALT } else { 0 }, out);
255        }
256    }
257    let text = core::str::from_utf8(bytes).map_err(|_| Status::UnsupportedInput)?;
258    let code = text.chars().next().ok_or(Status::UnsupportedInput)? as u32;
259    if alt { key_event(code, modifiers::ALT, out) }
260    else { validate_text(bytes)?; event(event_kind::TEXT, bytes, out) }
261}

263/// Incremental bounded serial/SSH grammar. Returns complete normalized frames
264/// (sequence/epoch zero). Paste markers are framing: the owner must stage and
265/// validate the entire paste before delivery. All timestamps are nanoseconds.
266#[derive(Debug)]
267pub struct SerialDecoder {
268    pending: [u8; CONTROL_MAX],
269    len: usize,
270    esc_started: Option<u64>,
271    paste: bool,
272    last_now: Option<u64>,
273}
274impl Default for SerialDecoder { fn default() -> Self { Self::new() } }
275impl SerialDecoder {
276    pub const fn new() -> Self { Self { pending: [0; CONTROL_MAX], len: 0, esc_started: None, paste: false, last_now: None } }
277    pub fn reset(&mut self) { self.wipe_pending(); self.esc_started = None; self.paste = false; self.last_now = None; }
278    /// Zero every pending byte (volatile: partial sequences may be pasted secrets), then forget them.
279    fn wipe_pending(&mut self) {
280        // Safe Rust only (this crate denies `unsafe`): write zeros, then make the
281        // buffer observable so the stores cannot be elided as dead.
282        self.pending.fill(0);
283        core::hint::black_box(&mut self.pending);
284        core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::SeqCst);
285        self.len = 0;
286    }
287    /// Every completed or aborted sequence ends here: no partial-sequence byte survives it.
288    /// The paste flag is framing state across many sequences and is left to PasteEnd/reset/finish.
289    fn clear_action(&mut self) { self.wipe_pending(); self.esc_started = None; }
290    pub fn deadline_ns(&self) -> Option<u64> {
291        self.esc_started.and_then(|start| start.checked_add(ESC_TIMEOUT_NS))
292    }
293    pub fn feed(&mut self, byte: u8, now_ns: u64, out: &mut [u8]) -> Result<usize, Status> {
294        let prefix = self.expire(now_ns, out)?;
295        let out = out.get_mut(prefix..).ok_or(Status::BufferTooSmall)?;
296        // A second ESC may be the Alt prefix of Shift+Tab.
297        if self.len >= 2 && self.pending[..2] == [0x1b, 0x1b] {
298            if self.len == 2 && byte == b'[' {
299                self.pending[2] = byte;
300                self.len = 3;
301                return Ok(prefix);
302            }
303            if self.len == 3 && byte == b'Z' {
304                let n = key_event(key_code::TAB, modifiers::ALT | modifiers::SHIFT, out)?;
305                self.clear_action();
306                return Ok(prefix + n);
307            }
308            let had_bracket = self.len == 3;
309            let mut n = key_event(key_code::ESCAPE, modifiers::ALT, out)?;
310            self.clear_action();
311            if had_bracket { n += scalar_event(b"[", false, &mut out[n..])?; }
312            n += self.feed(byte, now_ns, &mut out[n..])?;
313            return Ok(prefix + n);
314        }
315        if self.len == CONTROL_MAX { self.clear_action(); return Err(Status::UnsupportedInput); }
316        self.pending[self.len] = byte;
317        self.len += 1;
318        if self.len == 1 && byte == 0x1b {
319            if now_ns.checked_add(ESC_TIMEOUT_NS).is_none() {
320                self.clear_action();
321                return Err(Status::Malformed);
322            }
323            self.esc_started = Some(now_ns);
324            return Ok(prefix);
325        }
326        let result = self.decode_pending(out);
327        if !matches!(result, Ok(0) | Err(Status::BufferTooSmall)) { self.clear_action(); }
328        result.map(|n| prefix + n)
329    }
330    fn decode_pending(&mut self, out: &mut [u8]) -> Result<usize, Status> {
331        let bytes = &self.pending[..self.len];
332        if bytes[0] != 0x1b {
333            if self.paste {
334                match core::str::from_utf8(bytes) {
335                    Ok(_) => { validate_paste(bytes)?; return event(event_kind::TEXT, bytes, out); }
336                    Err(error) if error.error_len().is_none() && bytes.len() < 4 => return Ok(0),
337                    Err(_) => return Err(Status::UnsupportedInput),
338                }
339            }
340            return match core::str::from_utf8(bytes) {
341                Ok(_) => scalar_event(bytes, false, out),
342                Err(error) if error.error_len().is_none() && bytes.len() < 4 => Ok(0),
343                Err(_) => Err(Status::UnsupportedInput),
344            };
345        }
346        if self.len == 1 { return Ok(0); }
347        if self.paste {
348            if b"\x1b[201~".starts_with(bytes) {
349                if self.len < 6 { return Ok(0); }
350                let len = event(event_kind::PASTE_END, &[], out)?;
351                self.paste = false;
352                return Ok(len);
353            }
354            return Err(Status::UnsupportedInput);
355        }
356        if bytes == b"\x1b\x1b" { return Ok(0); }
357        if bytes == b"\x1b[200~" {
358            let len = event(event_kind::PASTE_BEGIN, &[], out)?;
359            self.paste = true;
360            return Ok(len);
361        }
362        if bytes[1] != b'[' && bytes[1] != b'O' {
363            return match core::str::from_utf8(&bytes[1..]) {
364                Ok(_) => scalar_event(&bytes[1..], true, out),
365                Err(error) if error.error_len().is_none() && self.len < 5 => Ok(0),
366                Err(_) => Err(Status::UnsupportedInput),
367            };
368        }
369        if self.len == 2 { return Ok(0); }
370        if self.len == 3 && bytes[1] == b'O' {
371            let code = match byte_final(bytes[2]) {
372                Some(code) if (key_code::F1..=key_code::F1 + 3).contains(&code) => code,
373                _ => return Err(Status::UnsupportedInput),
374            };
375            return key_event(code, 0, out);
376        }
377        let last = bytes[self.len - 1];
378        if last.is_ascii_digit() || last == b';' { return Ok(0); }
379        if bytes[1] != b'[' { return Err(Status::UnsupportedInput); }
380        if bytes == b"\x1b[Z" { return key_event(key_code::TAB, modifiers::SHIFT, out); }
381        let parameters = &bytes[2..self.len - 1];
382        let (number, mods) = parse_parameters(parameters)?;
383        let code = if last == b'~' {
384            match number {
385                2 => key_code::INSERT, 3 => key_code::DELETE,
386                5 => key_code::PAGE_UP, 6 => key_code::PAGE_DOWN,
387                15 => key_code::F1 + 4, 17 => key_code::F1 + 5,
388                18 => key_code::F1 + 6, 19 => key_code::F1 + 7,
389                20 => key_code::F1 + 8, 21 => key_code::F1 + 9,
390                23 => key_code::F1 + 10, 24 => key_code::F12,
391                _ => return Err(Status::UnsupportedInput),
392            }
393        } else {
394            let code = byte_final(last).ok_or(Status::UnsupportedInput)?;
395            if !parameters.is_empty() && (number != 1 || mods == 0) { return Err(Status::UnsupportedInput); }
396            if parameters.is_empty() && (key_code::F1..=key_code::F1 + 3).contains(&code) { return Err(Status::UnsupportedInput); }
397            code
398        };
399        key_event(code, mods, out)
400    }
401    /// Resolve an expired ESC prefix. Incomplete longer controls fail explicitly.
402    pub fn expire(&mut self, now_ns: u64, out: &mut [u8]) -> Result<usize, Status> {
403        if self.last_now.is_some_and(|last| now_ns < last) {
404            self.reset();
405            return Err(Status::Malformed);
406        }
407        self.last_now = Some(now_ns);
408        let Some(deadline) = self.deadline_ns() else { return Ok(0); };
409        if now_ns < deadline { return Ok(0); }
410        self.finish_action(out)
411    }
412    fn finish_action(&mut self, out: &mut [u8]) -> Result<usize, Status> {
413        let result = if self.paste { Err(Status::UnsupportedInput) }
414            else if self.len == 1 { key_event(key_code::ESCAPE, 0, out) }
415            else if self.len == 2 && matches!(self.pending[1], b'[' | b'O') { scalar_event(&self.pending[1..2], true, out) }
416            else if self.len >= 2 && self.pending[..2] == [0x1b, 0x1b] {
417                let n = key_event(key_code::ESCAPE, modifiers::ALT, out)?;
418                if self.len == 3 { scalar_event(b"[", false, &mut out[n..]).map(|tail| n + tail) } else { Ok(n) }
419            }
420            else { Err(Status::UnsupportedInput) };
421        if !matches!(result, Err(Status::BufferTooSmall)) { self.clear_action(); }
422        result
423    }
424    /// End-of-source resolution; incomplete UTF-8/control/paste is unsupported.
425    pub fn finish(&mut self, out: &mut [u8]) -> Result<usize, Status> {
426        if self.paste { self.reset(); return Err(Status::UnsupportedInput); }
427        if self.len == 0 { return Ok(0); }
428        if self.esc_started.is_some() { self.finish_action(out) }
429        else { self.clear_action(); Err(Status::UnsupportedInput) }
430    }
431}
432fn byte_final(byte: u8) -> Option<u32> {
433    use key_code::*;
434    Some(match byte { b'A' => UP, b'B' => DOWN, b'C' => RIGHT, b'D' => LEFT,
435        b'H' => HOME, b'F' => END, b'P' => F1, b'Q' => F1 + 1,
436        b'R' => F1 + 2, b'S' => F1 + 3, _ => return None })
437}
438fn parse_parameters(bytes: &[u8]) -> Result<(u16, u32), Status> {
439    if bytes.is_empty() { return Ok((1, 0)); }
440    let mut parts = bytes.split(|byte| *byte == b';');
441    let number_bytes = parts.next().ok_or(Status::UnsupportedInput)?;
442    if number_bytes.is_empty() || number_bytes[0] == b'0' { return Err(Status::UnsupportedInput); }
443    let mut number = 0u16;
444    for byte in number_bytes {
445        if !byte.is_ascii_digit() { return Err(Status::UnsupportedInput); }
446        number = number.checked_mul(10).and_then(|n| n.checked_add(u16::from(*byte - b'0'))).ok_or(Status::UnsupportedInput)?;
447    }
448    let mods = if let Some(parameter) = parts.next() {
449        if parameter.len() != 1 || !(b'2'..=b'8').contains(&parameter[0]) { return Err(Status::UnsupportedInput); }
450        let value = parameter[0] - b'1';
451        u32::from(value & 1) | if value & 2 != 0 { modifiers::ALT } else { 0 }
452            | if value & 4 != 0 { modifiers::CONTROL } else { 0 }
453    } else { 0 };
454    if parts.next().is_some() { return Err(Status::UnsupportedInput); }
455    Ok((number, mods))
456}

604    #[test]
605    fn no_partial_sequence_byte_survives_a_completed_or_rejected_sequence() {
606        let mut frame = [0; EVENT_MAX];
607        // Incomplete sequences retain their bytes; completion wipes every one of them.
608        for complete in [&b"\x1b[1;5A"[..], b"\x1b[24;8~", b"\x1bOP", b"\x1b[Z", b"\x1bx", b"\x1b\x1bx",
609                         b"\x1b\x1b[Z", b"\x1b\x1b[x", "é".as_bytes(), "😀".as_bytes(), b"a"] {
610            let mut decoder = SerialDecoder::new();
611            let (last, head) = complete.split_last().unwrap();
612            for byte in head { assert!(decoder.feed(*byte, 0, &mut frame).is_ok()); }
613            if !head.is_empty() && head != b"\x1b\x1b" { assert!(decoder.pending[..head.len().min(CONTROL_MAX)].iter().any(|b| *b != 0)); }
614            assert!(decoder.feed(*last, 0, &mut frame).unwrap() > 0, "{complete:?}");
615            assert_wiped(&decoder, "completed");
616            assert_eq!(decoder.deadline_ns(), None);
617        }
618        // Rejected sequences wipe too.
619        for rejected in [&b"\x1b[1A"[..], b"\x1b[01;2A", b"\x1b[1;1A", b"\x1bOZ", b"\x1b[9~", b"\xff", b"\xc3("] {
620            let mut decoder = SerialDecoder::new();
621            assert!(feed_all(&mut decoder, rejected, 0).is_err(), "{rejected:?}");
622            assert_wiped(&decoder, "rejected");
623        }
624        // Control overflow: the full pending buffer is wiped by the rejection.
625        let mut decoder = SerialDecoder::new();
626        assert_eq!(feed_all(&mut decoder, b"\x1b[", 0), Ok(0));
627        for _ in 0..CONTROL_MAX - 2 { assert_eq!(decoder.feed(b'1', 0, &mut frame), Ok(0)); }
628        assert!(decoder.pending.iter().all(|b| *b != 0));
629        assert_eq!(decoder.feed(b'1', 0, &mut frame), Err(Status::UnsupportedInput));
630        assert_wiped(&decoder, "overflow");
631        // ESC deadline resolution (expire and finish) and reset.
632        let mut decoder = SerialDecoder::new();
633        assert_eq!(decoder.feed(0x1b, 0, &mut frame), Ok(0));
634        assert_eq!(decoder.expire(ESC_TIMEOUT_NS, &mut frame), Ok(40));
635        assert_wiped(&decoder, "expired ESC");
636        assert_eq!(feed_all(&mut decoder, b"\x1b[1", ESC_TIMEOUT_NS), Ok(0));
637        assert!(decoder.finish(&mut frame).is_err());
638        assert_wiped(&decoder, "finish");
639        assert_eq!(feed_all(&mut decoder, b"\xc3", ESC_TIMEOUT_NS), Ok(0));
640        assert!(decoder.finish(&mut frame).is_err());
641        assert_wiped(&decoder, "finish partial scalar");
642        assert_eq!(feed_all(&mut decoder, b"\x1b[1;", ESC_TIMEOUT_NS), Ok(0));
643        decoder.reset();
644        assert_wiped(&decoder, "reset");
645        assert_eq!(decoder.feed(0x1b, 10, &mut frame), Ok(0));
646        assert_eq!(decoder.expire(9, &mut frame), Err(Status::Malformed));
647        assert_wiped(&decoder, "backwards clock");
648    }
