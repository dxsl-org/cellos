2//! Supervisor wire, carried inside the App SDK envelope
3//! ([`super::app_envelope`]): `[MAGIC, KIND_MESSAGE, op, …]`.
4//!
5//! - `OP_HOTSWAP`: `{ target_service: [u8; 64], new_elf: [u8; 128] }`, 193 B.
6//! - `OP_SNAPSHOT`: opcode only, 1 B.
7//! - `OP_STATUS` reply: `[OP_STATUS, phase, result]`, 3 B.
8//! - `OP_POWER`: `[OP_POWER, action]`, 2 B; action is the `SystemPower` a0.
9//!
10//! `OP_POWER` has the same byte as the feature-gated hostile-backend
11//! `KILL_REQUEST_OPCODE`; the envelope keeps them apart, since kill requests
12//! are raw frames.
13
14use super::app_envelope::{KIND_MESSAGE, MAGIC};
15
16pub const OP_HOTSWAP: u8 = 0x01;
17pub const OP_SNAPSHOT: u8 = 0x02;
18pub const OP_STATUS: u8 = 0x03;
19pub const OP_POWER: u8 = 0x04;
20
21/// Status phases (`OP_STATUS` byte 1).
22pub const PHASE_REFUSED: u8 = 0;
23pub const PHASE_SNAPSHOT: u8 = 1;
24/// A refused `OP_POWER`; success has no reply.
25pub const PHASE_POWER: u8 = 2;
26pub const PHASE_HOTSWAP_DONE: u8 = 6;
27/// Hot-swap failed; the result byte is a [`HotswapError`].
28pub const PHASE_HOTSWAP_ERROR: u8 = 0xFF;
29
30/// Status results (`OP_STATUS` byte 2) outside [`PHASE_HOTSWAP_ERROR`].
31pub const STATUS_OK: u8 = 0x00;
32pub const STATUS_UNAVAILABLE: u8 = 0x01;
33pub const STATUS_REJECTED_CALLER: u8 = 0xFD;
34pub const STATUS_SERVICE_NOT_FOUND: u8 = 0xFE;
35pub const STATUS_INVALID_REQUEST: u8 = 0xFF;
36
37pub const SVC_NAME_LEN: usize = 64;
38pub const ELF_PATH_LEN: usize = 128;
39pub const HOTSWAP_REQUEST_LEN: usize = 1 + SVC_NAME_LEN + ELF_PATH_LEN;
40pub const SNAPSHOT_REQUEST_LEN: usize = 1;
41pub const POWER_REQUEST_LEN: usize = 2;
42pub const STATUS_REPLY_LEN: usize = 3;
43
44pub const POWER_ACTION_REBOOT: u8 = 1;
45pub const POWER_ACTION_POWEROFF: u8 = 2;
46
47/// The only power request the kernel sends: the ACPI power button.
48pub const POWER_BUTTON_REQUEST: [u8; 4] = [MAGIC, KIND_MESSAGE, OP_POWER, POWER_ACTION_POWEROFF];
49/// Sender id of kernel-originated requests (the power button).
50pub const KERNEL_SENDER: usize = 0;
51
52/// Init → supervisor raw notice that a cell owner exited:
53/// `[OWNER_EXITED_NOTICE, owner_tid usize LE]`, sent only by init (tid 1).
54pub const OWNER_EXITED_NOTICE: u8 = 0xE1;
55/// Supervisor → compositor relay of [`OWNER_EXITED_NOTICE`], same layout.
56pub const OWNER_EXITED_RELAY: u8 = 0xE2;
57pub const OWNER_EXITED_LEN: usize = 1 + core::mem::size_of::<usize>();

59/// Result byte of a `PHASE_HOTSWAP_ERROR` status.
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61#[repr(u8)]
62pub enum HotswapError {
63    /// Target service name is not recognized or not currently live.
64    ServiceNotFound = 0,
65    /// Freeze syscall rejected (permission denied or tid invalid).
66    FreezeFailed = 1,
67    /// Unable to send Snapshot IPC to old cell.
68    SnapshotIpcFailed = 2,
69    /// Old cell did not stash state within the timeout.
70    SnapshotTimeout = 3,
71    /// Spawn of new ELF failed (path missing, no SpawnCap, loader error).
72    SpawnFailed = 4,
73    /// Unable to send Restore IPC to new cell.
74    RestoreIpcFailed = 5,
75    /// New cell did not call sys_hotswap_ready() within the timeout.
76    ReadyTimeout = 6,
77    /// Service mapping changed before it could be paused for snapshotting.
78    PauseFailed = 7,
79    /// Replacement became ready but could not be published as the provider.
80    RegisterFailed = 8,
81}
82
83impl HotswapError {
84    pub const fn from_u8(code: u8) -> Option<Self> {
85        Some(match code {
86            0 => Self::ServiceNotFound,
87            1 => Self::FreezeFailed,
88            2 => Self::SnapshotIpcFailed,
89            3 => Self::SnapshotTimeout,
90            4 => Self::SpawnFailed,
91            5 => Self::RestoreIpcFailed,
92            6 => Self::ReadyTimeout,
93            7 => Self::PauseFailed,
94            8 => Self::RegisterFailed,
95            _ => return None,
96        })
97    }
98}
