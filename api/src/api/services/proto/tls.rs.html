2//! Raw TLS frames to the net service. Every request starts `[op][cap u64 LE]`.
3//!
4//! - `CONNECT`: `[op][cap=0][ip 4][port u16][hostname_len u16][hostname]` →
5//!   reply `cap u64`, [`CONNECT_FAILED`] on failure.
6//! - `SEND`: `[op][cap][len u16][data]` → reply `u32` count, [`SEND_ERROR`].
7//! - `RECV`: `[op][cap][want u32]` → reply `[len u16][data]`; len 0 is EOF,
8//!   [`RECV_ERROR`] an error.
9//! - `CLOSE`: `[op][cap]` → `[CLOSE_OK]` or `[CLOSE_FAILED]`; an empty reply
10//!   is a parse error.
11
12pub const CLOSE: u8 = 0x15;
13pub const CONNECT: u8 = 0x30;
14pub const SEND: u8 = 0x31;
15pub const RECV: u8 = 0x32;
16
17pub const HEADER_LEN: usize = 9;
18pub const CONNECT_HEADER_LEN: usize = 17;
19pub const SEND_HEADER_LEN: usize = 11;
20pub const RECV_REQUEST_LEN: usize = 13;
21pub const MAX_HOSTNAME: usize = 495;
22pub const MAX_SEND: usize = 501;
23pub const MAX_RECV_DATA: usize = 4094;
24
25pub const CONNECT_FAILED: u64 = 0;
26pub const SEND_ERROR: u32 = u32::MAX;
27pub const RECV_EOF: u16 = 0;
28pub const RECV_ERROR: u16 = u16::MAX;
29pub const CLOSE_OK: u8 = 0x00;
30pub const CLOSE_FAILED: u8 = 0xFF;
