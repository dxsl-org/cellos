2//! Deciding whether a byte string may name something inside a directory handle.
3//!
4//! ## Why this works on raw bytes
5//! Every rule below is applied to the bytes exactly as they arrived, before any
6//! normalisation, case folding, or separator rewriting. Normalisation is where
7//! traversal holes come from: a check that runs against a cleaned-up copy is a
8//! check against a string the backend will never see, and the two disagree
9//! precisely on the inputs an attacker is choosing. Decoding to UTF-8 happens
10//! *last* and only to hand the caller a `&str`; it never changes a byte, so the
11//! decision has already been made by then.
12//!
13//! ## What a component may not be
14//! A component names one entry in one directory. It therefore may not contain a
15//! separator of any kind, may not be `.` or `..`, may not be empty, and may not
16//! carry control bytes. With separators excluded, `..` as a whole component is
17//! the only remaining way to name a parent, so rejecting it closes the set.
18//!
19//! Backslash is refused alongside `/` even though no backend here treats it as a
20//! separator: the cost is one comparison, and the alternative is that adding a
21//! backend which does treat it as one silently opens an escape.
22
23/// Longest component a directory handle will resolve.
24///
25/// Matches the widest limit among the mounted backends, so a name this layer
26/// accepts can never be truncated into a different name further down.
27pub const MAX_DIR_NAME_LEN: usize = 255;
28
29/// Longest absolute path accepted when a root handle is acquired.
30pub const MAX_DIR_PATH_LEN: usize = 1024;
31
32/// Why a name was refused. Every variant is a refusal; none is recoverable by
33/// rewriting the name on the service's behalf.
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub enum DirNameError {
36    /// No bytes at all. Names the directory itself, which no `*At` operation
37    /// means, and which several backends would resolve as the directory.
38    Empty,
39    /// Longer than the limit for its kind.
40    TooLong,
41    /// Contains `/` or `\`, so it spans more than one directory.
42    Separator,
43    /// Is `..`, the one remaining way to name a parent once separators are out.
44    Traversal,
45    /// Is `.`, which names the directory the handle already refers to.
46    CurrentDir,
47    /// Carries a byte below `0x20` or `0x7F`, including NUL. A backend reached
48    /// through a C interface would read a NUL as the end of the name, so the
49    /// name it stores and the name checked here would differ.
50    Control,
51    /// Not valid UTF-8. Rejected rather than replaced: an overlong encoding of
52    /// `/` is invalid UTF-8 and must not become a separator, and lossy decoding
53    /// would turn it into a name nobody asked for.
54    NotUtf8,
55    /// An absolute path that does not begin at the root.
56    NotAbsolute,
57}
58
59/// Check one component of a path, as raw bytes, and return it as `&str`.
60///
61/// The returned string is the same bytes, unchanged — this function never
62/// rewrites, trims, or folds anything.
63///
64/// # Errors
65/// See [`DirNameError`]. A component that passes cannot name anything outside
66/// the directory it is resolved against.
67pub fn validate_dir_component(raw: &[u8]) -> Result<&str, DirNameError> {
68    if raw.is_empty() {
69        return Err(DirNameError::Empty);
70    }
71    if raw.len() > MAX_DIR_NAME_LEN {
72        return Err(DirNameError::TooLong);
73    }
74    if raw == b".." {
75        return Err(DirNameError::Traversal);
76    }
77    if raw == b"." {
78        return Err(DirNameError::CurrentDir);
79    }
80    for &b in raw {
81        if b == b'/' || b == b'\\' {
82            return Err(DirNameError::Separator);
83        }
84        if b < 0x20 || b == 0x7F {
85            return Err(DirNameError::Control);
86        }
87    }
88    core::str::from_utf8(raw).map_err(|_| DirNameError::NotUtf8)
89}

91/// Check an absolute path offered as the root of a new handle, as raw bytes.
92///
93/// Root acquisition is the one place a path string still appears, so the same
94/// byte-level rules apply to every component of it. A path that passes contains
95/// no traversal at all, which means the handle it produces refers to exactly the
96/// directory the caller named and the resolution rules above hold from then on.
97///
98/// # Errors
99/// [`DirNameError::NotAbsolute`] when the path does not start at the root;
100/// otherwise whatever the offending component fails on, with an empty component
101/// (from `//` or a trailing `/`) reported as [`DirNameError::Empty`]. `/` itself
102/// is accepted and has no components.
103pub fn validate_dir_path(raw: &[u8]) -> Result<&str, DirNameError> {
104    if raw.is_empty() {
105        return Err(DirNameError::Empty);
106    }
107    if raw.len() > MAX_DIR_PATH_LEN {
108        return Err(DirNameError::TooLong);
109    }
110    if raw[0] != b'/' {
111        return Err(DirNameError::NotAbsolute);
112    }
113    if raw != b"/" {
114        for component in raw[1..].split(|&b| b == b'/') {
115            validate_dir_component(component)?;
116        }
117    }
118    core::str::from_utf8(raw).map_err(|_| DirNameError::NotUtf8)
119}

121/// Join a validated component onto a validated directory path.
122///
123/// Only sound for inputs that came through [`validate_dir_component`] and
124/// [`validate_dir_path`]: with no separator in `name` the result has exactly one
125/// more component than `dir`, and that component is `name`.
126pub fn join_component(dir: &str, name: &str) -> alloc::string::String {
127    let mut out = alloc::string::String::with_capacity(dir.len() + 1 + name.len());
128    out.push_str(dir.trim_end_matches('/'));
129    out.push('/');
130    out.push_str(name);
131    out
132}
