1//! Typed IPC message enums for ViCell services.
2//!
3//! Both kernel and Cell crates link to `libs/api`, so these types are shared
4//! across the IPC boundary without any unsafe casting.
5//!
6//! ## Wire format
7//! `postcard::to_slice` serializes into a caller-provided `[u8; IPC_BUF_SIZE]`
8//! stack buffer.  The receiver calls `postcard::from_bytes` to deserialize.
9//! Discriminants are 1-byte varints (< 127 variants); slice lengths are varint-
10//! prefixed.  Total overhead per message is minimal compared to the ecall trap.
11//!
12//! ## Lifetime contract
13//! Types that borrow (`&'a str`, `&'a [u8]`) point into the caller's buffer.
14//! The decoded value MUST be consumed before the buffer is reused.
15
16use serde::{Deserialize, Serialize};
17
18use crate::cell_session::ObjectRefV1;

20/// IPC payload buffer size.  Must be large enough for the largest serialized
21/// message in the system — currently a VFS Write with ~900-byte content
22/// (~916 bytes encoded).  4 KiB gives comfortable headroom.
23pub const IPC_BUF_SIZE: usize = 4096;
24/// Maximum inline TCP data that leaves conservative postcard framing headroom.
25pub const NET_TCP_INLINE_DATA_MAX: usize = IPC_BUF_SIZE - 256;
26
27/// Largest single grant chunk a `ReadHandleGrant` / `WriteHandleGrant` may move.
28///
29/// Both sides must agree: the caller (hypervisor cell) batches one guest block
30/// request into chunks of at most this size, and the VFS rejects anything
31/// larger. Grants are backed by *contiguous* frames, so the cap trades round
32/// trips against the allocator's ability to find a contiguous run — 16 pages,
33/// with callers free to fall back to smaller chunks when allocation fails.
34pub const VFS_GRANT_CHUNK: usize = 64 * 1024;

38/// Requests sent to the VFS service (`/bin/vfs`).
39#[derive(Debug, Serialize, Deserialize)]
40pub enum VfsRequest<'a> {
41    /// Read file content from a VFS-managed path.
42    GetFile(&'a str),
43    /// List directory entries as a newline-separated string.
44    ListDir(&'a str),
45    /// Stat a path — returns size + is_dir flag.
46    Stat(&'a str),
47    /// Write (create/overwrite) a file.
48    Write { path: &'a str, content: &'a [u8] },
49    /// Append bytes to an existing file (or create it).
50    Append { path: &'a str, content: &'a [u8] },
51    /// Create a directory.
52    Mkdir(&'a str),
53    /// Remove an empty directory.
54    Rmdir(&'a str),
55    /// Delete a file.
56    Unlink(&'a str),
57    /// Remove a directory tree recursively.
58    RmdirRecursive(&'a str),
59    /// Start a non-blocking file read.  Returns `PendingHandle(id)` immediately;
60    /// call `Poll { handle: id }` to retrieve data when ready.
61    ReadAsync { path: &'a str },
62    /// Poll a pending read for completion.
63    Poll { handle: u32 },
64    /// Zero-copy large read: VFS reads `size` bytes at `offset` from the file
65    /// identified by `cap` directly into the caller's pre-allocated grant buffer.
66    /// The caller allocates the grant, GrantShare's it RW to VFS, and then waits
67    /// for `GrantDone`. VFS copies at most `min(size, grant_len, 4096, data_len -
68    /// offset)` bytes and replies only after the copy completes.
69    ReadGrant {
70        cap: u64,
71        offset: u64,
72        size: usize,
73        grant: usize,
74    },
75    /// Zero-copy large write: VFS reads `bytes` bytes from the caller's grant
76    /// buffer and writes them at `offset` through a VFS-issued file-handle ID in
77    /// `cap`. GrantDone is sent only after the write commits (write-through on
78    /// FAT32) — F14 invariant.
79    WriteGrant {
80        cap: u64,
81        offset: u64,
82        grant: usize,
83        bytes: usize,
84    },
85    /// Zero-copy full-file read by PATH into the caller's grant (G2 loader redesign).
86    /// Unlike `ReadGrant` (cap + 4 KB page at a time), this resolves `path` through
87    /// the VFS mount table (so it reaches the `/bin` cell-store overlay) and copies
88    /// the ENTIRE file into the caller's pre-shared grant in one shot, replying
89    /// `GrantDone { bytes }`. `grant` must be owned by the caller and GrantShare'd
90    /// RW to VFS. Short grants are valid: VFS copies `min(file_len, max, grant_len)`
91    /// bytes, so a file that grew after the caller's Stat cannot overflow the grant.
92    /// Used to read a cell ELF for `sys_spawn_from_elf`.
93    ReadFileGrant {
94        path: &'a str,
95        grant: usize,
96        max: usize,
97    },
98
99    // ── Directory capabilities ────────────────────────────────────────────────
100    //
101    // Appended at the END and never reordered. Postcard encodes the variant
102    // index as a varint, so inserting anywhere above would renumber every
103    // variant after the insertion point: an old sender and a new receiver would
104    // then agree on the bytes and disagree on the meaning, which is a silently
105    // wrong operation rather than a decode error. This system has shipped that
106    // bug once already.
107    //
108    // Each of these names a directory the service issued to this caller and a
109    // single entry inside it. The name cannot express a path outside the
110    // directory, so there is no path for the service to authorize — see
111    // `docs/specs/09c-vfs-directory-capabilities-adr.md`.
112    /// Acquire a handle to an absolute directory path.
113    ///
114    /// The one remaining operation that names a path, and the bootstrap for
115    /// every other handle: a cell acquires the directories it needs and then
116    /// sends `SealPaths`, after which no path string from it is served.
117    OpenRootDir { path: &'a str },
118    /// Derive a handle to `name` inside `dir`, narrower than `dir` by
119    /// construction. Revoking `dir` revokes it.
120    OpenDir {
121        dir: crate::dir_handles::ViDirHandle,
122        name: &'a str,
123    },
124    /// Read the whole of `name` inside `dir`.
125    ReadAt {
126        dir: crate::dir_handles::ViDirHandle,
127        name: &'a str,
128    },
129    /// Create or overwrite `name` inside `dir`.
130    WriteAt {
131        dir: crate::dir_handles::ViDirHandle,
132        name: &'a str,
133        content: &'a [u8],
134    },
135    /// Size and kind of `name` inside `dir`.
136    StatAt {
137        dir: crate::dir_handles::ViDirHandle,
138        name: &'a str,
139    },
140    /// Entries of `dir` itself.
141    ///
142    /// Takes no name: listing a subdirectory means holding a handle to it, which
143    /// is what `OpenDir` is for. A name here would be a second resolution path
144    /// to keep in step with the first.
145    ListAt {
146        dir: crate::dir_handles::ViDirHandle,
147    },
148    /// Delete `name` inside `dir`.
149    UnlinkAt {
150        dir: crate::dir_handles::ViDirHandle,
151        name: &'a str,
152    },
153    /// Give up `dir`, and with it every handle derived from `dir`.
154    ///
155    /// Revocation is transitive on purpose: a derived handle that outlived the
156    /// handle it came from would let a cell keep access forever by deriving a
157    /// subdirectory and dropping the original.
158    CloseDir {
159        dir: crate::dir_handles::ViDirHandle,
160    },
161    /// Give up the ability to name a path, permanently, for this cell.
162    ///
163    /// Every path-addressed request from the sender is refused from here on. The
164    /// transition is one-way and survives for the life of the cell: an operation
165    /// that could undo it would make the guarantee advisory.
166    SealPaths,
167    /// Open the file `name` inside `dir` and receive a service-local file
168    /// handle for bounded inline reads.
169    OpenFileAt {
170        dir: crate::dir_handles::ViDirHandle,
171        name: &'a str,
172    },
173    /// Read at most `max` bytes from `file`, starting at `offset`.
174    ReadFileHandle {
175        file: crate::vfs_file_handles::ViVfsFileHandle,
176        offset: u64,
177        max: u32,
178    },
179    /// Give up `file`.
180    CloseFile {
181        file: crate::vfs_file_handles::ViVfsFileHandle,
182    },
183    /// Bounded zero-copy read into a grant buffer using a VFS handle.
184    ReadHandleGrant {
185        file: crate::vfs_file_handles::ViVfsFileHandle,
186        offset: u64,
187        size: usize,
188        grant: usize,
189    },
190    /// Bounded zero-copy write from a grant buffer using a VFS handle.
191    WriteHandleGrant {
192        file: crate::vfs_file_handles::ViVfsFileHandle,
193        offset: u64,
194        grant: usize,
195        bytes: usize,
196    },
197    /// Flush dirty pages for a file handle to the block device.
198    SyncHandle {
199        file: crate::vfs_file_handles::ViVfsFileHandle,
200    },
201    /// Atomic rename of a file or directory from `old` to `new`.
202    Rename { old: &'a str, new: &'a str },
203    /// Owner-authenticated prepared-cwd protocol; no filesystem mutation rights.
204    PreparedChildV1(&'a [u8]),
205    /// One page of the entries of `dir` itself, starting at `cursor`.
206    ///
207    /// `cursor` 0 starts a listing; every later page passes the `next` value of
208    /// the previous [`VfsResponse::DirPage`] unchanged (it is opaque). Answered
209    /// with `DirPage`, whose `entries` are packed per [`vfs_dir_page`]. Unlike
210    /// `ListAt`, a listing that does not fit one reply is never cut short
211    /// silently: only the page with [`vfs_dir_page::COMPLETE`] ends it.
212    ListAtPage {
213        dir: crate::dir_handles::ViDirHandle,
214        cursor: u32,
215    },
216    /// Current size of `file`, and whether a content or binding mutation reached
217    /// its path after it was opened. Answered with `FileState`.
218    FileHandleState {
219        file: crate::vfs_file_handles::ViVfsFileHandle,
220    },
221
222    // ── D03 conditional save (Phase 08). Appended; never reorder. ─────────────
223    /// Observe `path` for a later conditional publication. Answered with
224    /// `Observed`, or `Err(6)` on a backend without conditional save.
225    Observe { path: &'a str },
226    /// Open a staging transaction that can only publish at `dst`. Answered with
227    /// `SaveId`. Nothing visible at `dst` changes until `SaveCommit`.
228    SaveBegin { dst: &'a str },
229    /// Append staged bytes; `offset` must equal the bytes staged so far and
230    /// `bytes` is at most [`VFS_SAVE_CHUNK`].
231    SaveChunk { id: u32, offset: u64, bytes: &'a [u8] },
232    /// Seal the staged bytes: the service reads them back and requires exactly
233    /// `len` bytes with this SHA-256. No further chunks are accepted.
234    SaveSeal { id: u32, len: u64, sha256: [u8; 32] },
235    /// Publish the sealed bytes at `dst` only if it still matches `expect`.
236    /// Answered with `SaveOutcome`; a repeated commit or query answers the
237    /// recorded outcome and never runs the publication again.
238    SaveCommit { id: u32, expect: SaveExpect },
239    /// Outcome of this transaction's commit, if one was attempted.
240    SaveQuery { id: u32 },
241    /// Forget the transaction, removing any unpublished staged bytes.
242    SaveAbort { id: u32 },
243    /// Phase 09 (abi-vfs-telemetry §4, read-only part): look up the configured SSH
244    /// host identifier and port in the protected host-trust store and compare the
245    /// complete public key. Answered with [`VfsResponse::Trust`]. Needs the caller's
246    /// attested outbound-connect right; never reveals store contents.
247    TrustLookup { host: &'a str, port: u16, algorithm: &'a str, key: &'a [u8] },
248    /// Phase 09 S9-4c first trust: for a host and port that [`Self::TrustLookup`] reports
249    /// Unknown, the service registers a 60-second challenge with the input owner bound to
250    /// the caller, the exact key and the store's current state, and answers
251    /// [`VfsResponse::TrustChallenged`] (or [`VfsResponse::Trust`] with the refusal). The
252    /// caller then runs `BeginTrustConfirmation` with `confirmation` on `session` at
253    /// `epoch`, its own terminal session. Needs the attested outbound-connect right.
254    TrustChallenge { host: &'a str, port: u16, algorithm: &'a str, key: &'a [u8], session: ObjectRefV1, epoch: u64 },
255    /// Phase 09 S9-4c: complete the caller's own `challenge`. Answered with
256    /// [`VfsResponse::Trust`]: `PENDING` while undecided; after an accepted confirmation
257    /// the store is updated and the answer is `KNOWN_EXACT` (or `MISMATCH` when another
258    /// writer stored a different key first); `REFUSED`, `EXPIRED`, `DENIED` or
259    /// `UNAVAILABLE` otherwise. Any final answer ends the challenge.
260    TrustCommit { challenge: ObjectRefV1 },
261}

268/// The destination state a conditional save was confirmed against.
269#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
270pub enum SaveExpect {
271    Absent,
272    /// Exact file identity: `inode` is never reused on the volume and
273    /// `generation` advances on every mutation, even a byte-identical one.
274    Present { inode: u64, generation: u64 },
275}

277/// `Observe` reply.
278#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
279pub enum Observation {
280    Absent,
281    Present { inode: u64, generation: u64, size: u64, regular: bool },
282}
283
284impl Observation {
285    /// The expectation that binds a save to exactly this observation; `None`
286    /// for a non-regular object, which a save never replaces.
287    pub fn expect(self) -> Option<SaveExpect> {
288        match self {
289            Self::Absent => Some(SaveExpect::Absent),
290            Self::Present { inode, generation, regular: true, .. } =>
291                Some(SaveExpect::Present { inode, generation }),
292            Self::Present { .. } => None,
293        }
294    }
295}

330/// Result of `SaveCommit` / `SaveQuery`.
331#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
332pub enum SaveOutcome {
333    /// Published and synced through the backend's barriers. This is not a
334    /// power-loss guarantee.
335    Committed { inode: u64, generation: u64 },
336    /// `dst` no longer matched the expectation; nothing changed.
337    Conflict,
338    /// Proven not published; `dst` is unchanged. Reason per [`save_refusal`].
339    NotPublished(u8),
340    /// The publication may or may not have reached storage.
341    Indeterminate,
342    /// Published, and the backend's durability protocol proved the new bytes survive
343    /// power loss (before this reply: old or new whole, never torn). No backend
344    /// returns it yet: CellosFS reports durability unavailable until the separate
345    /// crash-safety plan (`.agents/261025-cellosfs-crash-safety/plan.md`) lands.
346    CommittedDurable { inode: u64, generation: u64 },
347}
348
349impl SaveOutcome {
350    /// The published object for either successful outcome.
351    pub fn published(self) -> Option<(u64, u64)> {
352        match self {
353            Self::Committed { inode, generation } | Self::CommittedDurable { inode, generation } => {
354                Some((inode, generation))
355            }
356            Self::Conflict | Self::NotPublished(_) | Self::Indeterminate => None,
357        }
358    }
359}

361impl VfsRequest<'_> {
362    /// Whether this request names its target with a path string.
363    ///
364    /// A sealed cell is refused every request for which this is true. Matched
365    /// exhaustively on purpose: a new variant does not compile until someone has
366    /// decided which side of the boundary it falls on, and the answer defaults
367    /// to nothing.
368    pub fn is_path_addressed(&self) -> bool {
369        match self {
370            Self::GetFile(_)
371            | Self::ListDir(_)
372            | Self::Stat(_)
373            | Self::Write { .. }
374            | Self::Append { .. }
375            | Self::Mkdir(_)
376            | Self::Rmdir(_)
377            | Self::Unlink(_)
378            | Self::RmdirRecursive(_)
379            | Self::ReadAsync { .. }
380            | Self::ReadFileGrant { .. }
381            | Self::OpenRootDir { .. }
382            | Self::Rename { .. } => true,
383            // `Poll`, `ReadGrant` and `WriteGrant` carry a handle the service
384            // issued, not a path. They stay reachable so a sealed cell can drain
385            // work it started before sealing; the path recorded against the
386            // handle is still re-authorized on every use.
387            Self::Poll { .. }
388            | Self::ReadGrant { .. }
389            | Self::WriteGrant { .. }
390            | Self::OpenDir { .. }
391            | Self::ReadAt { .. }
392            | Self::WriteAt { .. }
393            | Self::StatAt { .. }
394            | Self::ListAt { .. }
395            | Self::UnlinkAt { .. }
396            | Self::CloseDir { .. }
397            | Self::SealPaths
398            | Self::OpenFileAt { .. }
399            | Self::ReadFileHandle { .. }
400            | Self::CloseFile { .. }
401            | Self::ReadHandleGrant { .. }
402            | Self::WriteHandleGrant { .. }
403            | Self::SyncHandle { .. }
404            | Self::PreparedChildV1(_)
405            | Self::ListAtPage { .. }
406            | Self::FileHandleState { .. } => false,
407            // A save names its destination once, at `SaveBegin`; the rest carry
408            // the service-issued transaction id.
409            Self::Observe { .. } | Self::SaveBegin { .. } => true,
410            Self::SaveChunk { .. }
411            | Self::SaveSeal { .. }
412            | Self::SaveCommit { .. }
413            | Self::SaveQuery { .. }
414            | Self::SaveAbort { .. }
415            // Names a host, never a path; the store location is the service's own.
416            | Self::TrustLookup { .. }
417            | Self::TrustChallenge { .. }
418            | Self::TrustCommit { .. } => false,
419        }
420    }
421
422    /// Whether this request requires the caller to hold VfsMutate authority.
423    ///
424    /// Matched exhaustively so any future variant must be classified.
425    pub fn requires_mutation_authority(&self) -> bool {
426        match self {
427            Self::Write { .. }
428            | Self::Append { .. }
429            | Self::Mkdir(_)
430            | Self::Rmdir(_)
431            | Self::Unlink(_)
432            | Self::RmdirRecursive(_)
433            | Self::WriteGrant { .. }
434            | Self::WriteAt { .. }
435            | Self::UnlinkAt { .. }
436            | Self::WriteHandleGrant { .. }
437            | Self::SyncHandle { .. }
438            | Self::Rename { .. }
439            | Self::SaveBegin { .. }
440            | Self::SaveChunk { .. }
441            | Self::SaveSeal { .. }
442            | Self::SaveCommit { .. }
443            | Self::SaveAbort { .. } => true,
444
445            Self::GetFile(_)
446            | Self::ListDir(_)
447            | Self::Stat(_)
448            | Self::ReadAsync { .. }
449            | Self::Poll { .. }
450            | Self::ReadGrant { .. }
451            | Self::ReadFileGrant { .. }
452            | Self::OpenRootDir { .. }
453            | Self::OpenDir { .. }
454            | Self::ReadAt { .. }
455            | Self::StatAt { .. }
456            | Self::ListAt { .. }
457            | Self::CloseDir { .. }
458            | Self::SealPaths
459            | Self::OpenFileAt { .. }
460            | Self::ReadFileHandle { .. }
461            | Self::CloseFile { .. }
462            | Self::ReadHandleGrant { .. }
463            | Self::PreparedChildV1(_)
464            | Self::ListAtPage { .. }
465            | Self::FileHandleState { .. }
466            | Self::Observe { .. }
467            | Self::SaveQuery { .. }
468            | Self::TrustLookup { .. }
469            // The store is written on the service's own authority after an
470            // owner-attested confirmation, never with the caller's.
471            | Self::TrustChallenge { .. }
472            | Self::TrustCommit { .. } => false,
473        }
474    }

477/// Responses from the VFS service.
478#[derive(Debug, Serialize, Deserialize)]
479pub enum VfsResponse<'a> {
480    /// File or directory listing bytes (copied into response buffer).
481    Data(&'a [u8]),
482    /// Zero-copy file access: raw pointer + length in the shared address space.
483    /// Used by binary loaders that need to map the ELF directly from VFS memory.
484    DataPtr { ptr: u64, len: u64 },
485    /// Stat result: (file_size, is_dir).
486    Stat { size: u64, is_dir: bool },
487    /// Successful write, mkdir, unlink, etc.
488    Ok,
489    /// Error — opaque error code (`types::ViError` discriminant).
490    Err(u8),
491    /// Async read accepted; poll this handle for completion.
492    PendingHandle(u32),
493    /// Read still in progress — call Poll again after yielding.
494    Pending,
495    /// Zero-copy I/O complete: `bytes` is the number of bytes transferred.
496    GrantDone { bytes: usize },
497    /// A directory handle the service has issued to the caller.
498    ///
499    /// Appended at the END for the same reason as the request variants: an
500    /// insertion above would renumber the rest and turn an old reply into a
501    /// different one that still decodes.
502    DirHandle(crate::dir_handles::ViDirHandle),
503    /// A file handle the service has issued to the caller.
504    FileHandle(crate::vfs_file_handles::ViVfsFileHandle),
505    /// Complete borrowed V1 prepared-cwd result record.
506    PreparedChildV1(&'a [u8]),
507    /// One `ListAtPage` reply: whole entries packed per [`vfs_dir_page`],
508    /// the cursor for the following page, and [`vfs_dir_page`] flag bits.
509    /// On the page that carries `COMPLETE`, `next` is 0 and means nothing.
510    DirPage {
511        next: u32,
512        flags: u8,
513        entries: &'a [u8],
514    },
515    /// `FileHandleState` reply.
516    FileState { size: u64, modified: bool },
517    /// `Observe` reply.
518    Observed(Observation),
519    /// `SaveBegin` reply: the transaction id.
520    SaveId(u32),
521    /// `SaveCommit` / `SaveQuery` reply.
522    SaveOutcome(SaveOutcome),
523    /// `TrustLookup` reply: one of [`ssh_trust`].
524    Trust(u8),
525    /// `TrustChallenge` reply: the input-issued confirmation to begin, and the
526    /// service-issued challenge to commit.
527    TrustChallenged { confirmation: ObjectRefV1, challenge: ObjectRefV1 },
528}

530/// Packed entry format and flag bits of [`VfsResponse::DirPage`].
531///
532/// Each entry is `kind:u8, len:u8, name[len]`. `name` is one directory
533/// component that passes [`crate::dir_name::validate_dir_component`], so it is
534/// valid UTF-8, 1–255 bytes, and can be passed back to `OpenDir`/`OpenFileAt`
535/// unchanged. A page holds only whole entries.
536pub mod vfs_dir_page {
537    use crate::dir_name::{validate_dir_component, DirNameError};
538
539    /// Set on the final page of a listing.
540    pub const COMPLETE: u8 = 1 << 0;
541    /// A child of the listed directory was created, removed or renamed after the
542    /// directory handle was opened, so pages may not form one consistent view.
543    /// Sticky for the life of the handle.
544    pub const CHANGED: u8 = 1 << 1;
545    /// Bytes before the name in one packed entry.
546    pub const ENTRY_HEADER_LEN: usize = 2;
547
548    /// What a directory entry is.
549    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
550    pub enum EntryKind {
551        File,
552        Dir,
553        /// The root of another mount: listing or reading it leaves this mount.
554        Mount,
555        /// Neither a regular file nor a directory.
556        Other,
557    }
558
559    impl EntryKind {
560        pub const fn wire(self) -> u8 {
561            match self {
562                Self::File => 0,
563                Self::Dir => 1,
564                Self::Mount => 2,
565                Self::Other => 3,
566            }
567        }
568
569        pub const fn from_wire(byte: u8) -> Option<Self> {
570            match byte {
571                0 => Some(Self::File),
572                1 => Some(Self::Dir),
573                2 => Some(Self::Mount),
574                3 => Some(Self::Other),
575                _ => None,
576            }
577        }
578    }
579
580    /// Append one entry to `out` at `pos`.
581    ///
582    /// Returns the position after the entry, or `Ok(None)` when it does not fit
583    /// (nothing is written then).
584    ///
585    /// # Errors
586    /// The name is not a valid directory component.
587    pub fn push_entry(
588        out: &mut [u8],
589        pos: usize,
590        kind: EntryKind,
591        name: &[u8],
592    ) -> Result<Option<usize>, DirNameError> {
593        let name = validate_dir_component(name)?.as_bytes();
594        let end = pos + ENTRY_HEADER_LEN + name.len();
595        if end > out.len() {
596            return Ok(None);
597        }
598        out[pos] = kind.wire();
599        // validate_dir_component bounds the name to MAX_DIR_NAME_LEN (255).
600        out[pos + 1] = name.len() as u8;
601        out[pos + ENTRY_HEADER_LEN..end].copy_from_slice(name);
602        Ok(Some(end))
603    }
604
605    /// One decoded entry, borrowing the page bytes.
606    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
607    pub struct Entry<'a> {
608        pub kind: EntryKind,
609        pub name: &'a str,
610    }
611
612    /// The page bytes do not follow the packed format.
613    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
614    pub struct MalformedPage;
615
616    /// Decode the entries of one page in order.
617    pub fn entries(page: &[u8]) -> Entries<'_> {
618        Entries { rest: page }
619    }
620
621    /// Iterator over packed entries; yields one error and then stops.
622    pub struct Entries<'a> {
623        rest: &'a [u8],
624    }
625
626    impl<'a> Iterator for Entries<'a> {
627        type Item = Result<Entry<'a>, MalformedPage>;
628
629        fn next(&mut self) -> Option<Self::Item> {
630            let rest = self.rest;
631            if rest.is_empty() {
632                return None;
633            }
634            self.rest = &[];
635            let [kind, len, tail @ ..] = rest else {
636                return Some(Err(MalformedPage));
637            };
638            let len = usize::from(*len);
639            let (Some(kind), Some(name)) = (EntryKind::from_wire(*kind), tail.get(..len)) else {
640                return Some(Err(MalformedPage));
641            };
642            let Ok(name) = validate_dir_component(name) else {
643                return Some(Err(MalformedPage));
644            };
645            self.rest = &tail[len..];
646            Some(Ok(Entry { kind, name }))
647        }
648    }
649}

666/// Raw reactor wakeup opcode emitted within the same cell: `[0x12]`.

669/// Fixed length of a raw `NET_READY` frame in bytes.
670pub const NET_READY_FRAME_BYTES: usize = 6;
671
672/// Packed TCP readiness record: cap_id (u32 LE) followed by one flags byte.
673/// Unlike borrowed structured slices, `&[u8]` deserializes without allocating.
674pub const NET_READY_RECORD_BYTES: usize = 5;

676/// Attested-owner-scoped TCP interest; cap ids are validated by the net cell.
677#[derive(Debug, Clone, Copy)]
678pub struct TcpInterest {
679    pub cap_id: u32,
680    pub mask: u8,
681}

683/// One level-triggered socket result within a paginated readiness batch.
684#[derive(Debug, Clone, Copy)]
685pub struct TcpReadyEvent {
686    pub cap_id: u32,
687    pub ready: u8,
688}
689
690impl TcpInterest {
691    pub fn decode(record: &[u8]) -> Option<Self> {
692        if record.len() != NET_READY_RECORD_BYTES { return None; }
693        Some(Self {
694            cap_id: u32::from_le_bytes(record[..4].try_into().ok()?),
695            mask: record[4],
696        })
697    }
698}
699
700impl TcpReadyEvent {
701    pub fn decode(record: &[u8]) -> Option<Self> {
702        if record.len() != NET_READY_RECORD_BYTES { return None; }
703        Some(Self {
704            cap_id: u32::from_le_bytes(record[..4].try_into().ok()?),
705            ready: record[4],
706        })
707    }
708}

710/// Pack interests into caller-owned storage for `NetRequest::TcpReady`.
711pub fn encode_tcp_interests<'a>(
712    interests: &[TcpInterest],
713    output: &'a mut [u8],
714) -> Option<&'a [u8]> {
715    let len = interests.len().checked_mul(NET_READY_RECORD_BYTES)?;
716    if output.len() < len { return None; }
717    for (interest, chunk) in interests.iter().zip(output[..len].chunks_exact_mut(NET_READY_RECORD_BYTES)) {
718        chunk[..4].copy_from_slice(&interest.cap_id.to_le_bytes());
719        chunk[4] = interest.mask;
720    }
721    Some(&output[..len])
722}

967// ── Serialization helpers ─────────────────────────────────────────────────────
968
969/// Serialize `msg` into `buf`.
970///
971/// Returns a slice of the bytes actually written.
972///
973/// # Errors
974/// Returns `postcard::Error` if the encoded message exceeds `buf.len()`.
975pub fn encode<'a, T: Serialize>(msg: &T, buf: &'a mut [u8]) -> postcard::Result<&'a mut [u8]> {
976    postcard::to_slice(msg, buf)
977}
978
979/// Deserialize a typed message from the START of `buf`, tolerating trailing bytes.
980///
981/// The IPC receive buffer is fixed at 512 bytes; the sender may have encoded a
982/// shorter message.  `take_from_bytes` reads exactly what the message needs and
983/// ignores the rest — unlike `from_bytes` which requires the entire slice to be
984/// consumed.
985///
986/// For types with borrowed fields (`&str`, `&[u8]`), the returned value borrows
987/// from `buf` — it must be consumed before `buf` is reused or overwritten.
988pub fn decode<'de, T: Deserialize<'de>>(buf: &'de [u8]) -> postcard::Result<T> {
989    let (msg, _remaining) = postcard::take_from_bytes(buf)?;
990    Ok(msg)
991}

1109        assert_eq!(encode(&VfsResponse::Observed(Observation::Absent), &mut out).unwrap(), b"\x0d\x00");
1110        // Every step that changes storage needs mutation authority; only the
1111        // destination-naming steps are path-addressed.
1112        let seal = VfsRequest::SaveSeal { id: 1, len: 0, sha256: [0; 32] };
1113        assert!(seal.requires_mutation_authority() && !seal.is_path_addressed());
1114        let begin = VfsRequest::SaveBegin { dst: "a" };
1115        assert!(begin.requires_mutation_authority() && begin.is_path_addressed());
1116        let observe = VfsRequest::Observe { path: "a" };
1117        assert!(!observe.requires_mutation_authority() && observe.is_path_addressed());
1118        assert!(!VfsRequest::SaveQuery { id: 1 }.requires_mutation_authority());
1119        // A directory observation never yields a replaceable expectation.
1120        let dir = Observation::Present { inode: 2, generation: 1, size: 0, regular: false };
1121        assert_eq!(dir.expect(), None);
1122    }
