1//! Constants for [`super::manifest::CellManifest`] — magic/version, protection
2//! classes, legacy isolation-tier aliases, and capability flag bits. Split out
3//! of `manifest.rs` to keep that file under the 200-LOC law; re-exported from
4//! `manifest.rs` so callers see one surface
5//! (`api::manifest::MANIFEST_FLAG_*` etc. keep working unchanged).
6
7/// Magic value identifying a valid manifest (`0x5649_4345`, "VICE" as a u32).
8pub const MANIFEST_MAGIC: u32 = 0x5649_4345;
9
10/// Current manifest layout version.  Bump on any field addition or reorder.
11pub const MANIFEST_VERSION: u8 = 2;
12
13/// The v1 layout version (8-byte record).  Recognised by `from_bytes` for upcast.
14pub const MANIFEST_VERSION_V1: u8 = 1;
15
16// ─── Protection classes / legacy isolation-tier names ─────────────────────────
17// The on-disk byte is a FLOOR request for the x86 PKU protection domain: a
18// higher number means MORE isolation / LESS authority. A cell may always raise
19// its own class (self-restriction); it may NOT lower it below the floor the
20// loader derives from its capabilities (see `kernel/src/loader.rs`). This is why
21// declaring a class is safe without a ceiling check for the raise direction, and
22// gated for the lower.
23
24/// Trusted-core domain (PKU key 0 — no fencing).  Only reachable if the cell's
25/// caps already authorise it (the loader floors non-privileged cells above this).
26pub const TIER_TRUSTED_CORE: u8 = 0;
27/// Canonical alias for `TIER_TRUSTED_CORE`.
28pub const PROTECTION_CLASS_TRUSTED_CORE: u8 = TIER_TRUSTED_CORE;
29/// Standard Rust cell (PKU key 1).  The default authority floor for a plain cell.
30pub const TIER_STANDARD: u8 = 1;
31/// Canonical alias for `TIER_STANDARD`.
32pub const PROTECTION_CLASS_STANDARD: u8 = TIER_STANDARD;
33/// Tier-1b C/FFI cell (PKU key 2).  Fences untrusted FFI code from the cell's Rust
34/// data — the key the v1 `TODO(pku-ffi)` could not reach.
35pub const TIER_TIER1B_FFI: u8 = 2;
36/// Canonical alias for `TIER_TIER1B_FFI`.
37pub const PROTECTION_CLASS_FFI: u8 = TIER_TIER1B_FFI;
38/// Untrusted domain (maps to the most-isolated available key).
39pub const TIER_UNTRUSTED: u8 = 3;
40/// Canonical alias for `TIER_UNTRUSTED`.
41pub const PROTECTION_CLASS_UNTRUSTED: u8 = TIER_UNTRUSTED;
42/// Sentinel meaning "no explicit tier requested — apply the caller's floor
43/// policy (the legacy `is_trusted` heuristic)."  Set automatically on upcast from
44/// a v1 manifest (which had no tier field), AND baked in by the tier-less
45/// constructors (`CellManifest::new`/`with_parts`, used by `declare_manifest!`'s
46/// back-compat macro forms) into a native v2 record — so it is a valid value in
47/// BOTH v1-upcast and native-v2 manifests, never an error on its own.
48pub const TIER_LEGACY: u8 = 0xFF;
49/// Canonical alias for `TIER_LEGACY`.
50pub const PROTECTION_CLASS_LEGACY: u8 = TIER_LEGACY;
51
52// ─── Capability flags (u16 in v2; low 8 bits are bit-identical to v1) ─────────
53
54/// Raw block-device access (BlkRead/BlkWrite/BlkFlush).  Grants `BlockIoCap`.
55pub const MANIFEST_FLAG_BLOCK_IO: u16 = 1 << 0;
56/// Network transmit/receive (NetTx/NetRx).  Grants `NetworkCap`.
57pub const MANIFEST_FLAG_NETWORK: u16 = 1 << 1;
58/// Cell spawning and lifecycle control entry points (SpawnFromPath/SpawnPinned). Grants `SpawnCap`.
59pub const MANIFEST_FLAG_SPAWN: u16 = 1 << 2;
60/// GPIO pin control (ViGpio driver cell).  MMIO range via `sys_request_mmio`.
61pub const MANIFEST_FLAG_GPIO: u16 = 1 << 3;
62/// UART serial access (ViUart driver cell).  MMIO range via `sys_request_mmio`.
63pub const MANIFEST_FLAG_UART: u16 = 1 << 4;
64/// RISC-V H-extension (hypervisor) CSR access for VMM cells.  Grants `HypervisorCap`
65/// only when the CPU also reports H-ext at boot.
66pub const MANIFEST_FLAG_HYPERVISOR: u16 = 1 << 5;
67/// Block-I/O sector range grant: MBR partition P1 (FAT32, `api::disk`).
68pub const MANIFEST_FLAG_PART_DATA: u16 = 1 << 6;
69/// Block-I/O sector range grant: MBR partition P4 (littlefs, `api::disk`).
70pub const MANIFEST_FLAG_PART_LFS: u16 = 1 << 7;
71/// CAN bus controller MMIO (v2 — freed by the u16 widening).  Grants the CAN
72/// device class via `sys_request_mmio`.
73pub const MANIFEST_FLAG_CAN: u16 = 1 << 8;
74/// ADC controller MMIO (v2).  Grants the ADC device class via `sys_request_mmio`.
75pub const MANIFEST_FLAG_ADC: u16 = 1 << 9;
76/// I2C controller MMIO (v2).  Grants the I2C device class via `sys_request_mmio`.
77pub const MANIFEST_FLAG_I2C: u16 = 1 << 10;
78/// SPI controller MMIO (v2).  Grants the SPI device class via `sys_request_mmio`.
79pub const MANIFEST_FLAG_SPI: u16 = 1 << 11;
80
81/// Bitmask of all defined flags. `from_bytes` rejects manifests setting any bit
82/// outside this mask; the loader classifies such a record as malformed and
83/// denies it before task creation.
84pub const MANIFEST_FLAGS_MASK: u16 = MANIFEST_FLAG_BLOCK_IO
85    | MANIFEST_FLAG_NETWORK
86    | MANIFEST_FLAG_SPAWN
87    | MANIFEST_FLAG_GPIO
88    | MANIFEST_FLAG_UART
89    | MANIFEST_FLAG_HYPERVISOR
90    | MANIFEST_FLAG_PART_DATA
91    | MANIFEST_FLAG_PART_LFS
92    | MANIFEST_FLAG_CAN
93    | MANIFEST_FLAG_ADC
94    | MANIFEST_FLAG_I2C
95    | MANIFEST_FLAG_SPI;
