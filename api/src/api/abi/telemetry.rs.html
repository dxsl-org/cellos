2//! Cell-aware telemetry snapshot ABI (`TelemetrySnapshot = 430`).
3//!
4//! Phase 06 S6-1 (`.agents/261020-1750-cellbox-cli/phase-06-entry-dossier.md` §5,
5//! approved CP1 2026-10-23). One call writes, in one capture, a header followed by
6//! whole Cell rows and then whole task rows into a caller buffer:
7//!
8//! ```text
9//! TelemetryHeaderV1 (96) | TelemetryCellV1 (96) × emitted_cells | TelemetryTaskV1 (88) × emitted_tasks
10//! ```
11//!
12//! Identity is `(provider_epoch, cell_id, generation)` for a Cell incarnation and
13//! `(provider_epoch, tid)` for a task: TIDs are never reissued within one boot and
14//! generations are global, so neither needs a separate incarnation field (DV-1/DV-2).
15//!
16//! Units: every `*_ns` field is nanoseconds on the monotonic clock `GetTime(0)`
17//! exposes, so `cpu_ns` deltas divide directly by `capture_start_ns` deltas.
18//! `cpu_ns` is exact on-CPU time (charged at every scheduler pass), and a Cell's
19//! `cpu_ns` includes the CPU of its already-exited threads for this incarnation.
20//! Memory: `quota_charged_bytes` is the Cell's whole quota charge (kernel heap in
21//! its context, worker stacks, pipes, session objects), not a heap size, and task
22//! `user_stack_bytes`/`segment_bytes` are informational — never sum them into the
23//! Cell (DV-4).
24//!
25//! Encoding is explicit little-endian at fixed offsets; the `repr(C)` structs are
26//! the documented layout, never reinterpreted from raw bytes. A V1 row is never a
27//! V2 row: a later layout gets a new version number.
28
29/// Header `version` written by this layout.
30pub const TELEMETRY_VERSION: u32 = 1;
31/// Exact byte size of [`TelemetryHeaderV1`].
32pub const HEADER_LEN: usize = 96;
33/// Exact byte size of [`TelemetryCellV1`].
34pub const CELL_LEN: usize = 96;
35/// Exact byte size of [`TelemetryTaskV1`].
36pub const TASK_LEN: usize = 88;
37/// Required alignment of the caller buffer.
38pub const BUFFER_ALIGN: usize = 8;
39/// Bytes of a row's `name`; longer names are cut at a UTF-8 boundary and flagged.
40pub const NAME_LEN: usize = 32;
41/// Header `scope`: every Cell and task in the system (the only scope offered, DV-6).
42pub const SCOPE_SYSTEM: u32 = 1;
43
44/// `a2` request flags. Every other bit must be zero.
45pub mod request {
46    /// Emit task rows after the Cell rows.
47    pub const TASKS: usize = 1 << 0;
48    /// Fill the system-memory block (also needs the caller's `observe_memory` right).
49    pub const SYSTEM_MEMORY: usize = 1 << 1;
50    /// Every defined request bit.
51    pub const ALL: usize = TASKS | SYSTEM_MEMORY;
52}
53
54/// [`TelemetryHeaderV1::flags`].
55pub mod header_flags {
56    /// Every requested row was emitted; no row was dropped for lack of space.
57    pub const COMPLETE: u32 = 1 << 0;
58    /// The monotonic clock was available: `cpu_ns` and the capture times are valid.
59    pub const CPU_AVAILABLE: u32 = 1 << 1;
60    /// The `mem_*` fields hold the global frame-allocator snapshot.
61    pub const SYSMEM_PRESENT: u32 = 1 << 2;
62    /// System memory was requested but the caller lacks `observe_memory`; `mem_*` are 0.
63    pub const SYSMEM_DENIED: u32 = 1 << 3;
64    /// Memory counters are lock-free and were read during the capture interval, so
65    /// they are not instant-consistent with membership and CPU.
66    pub const MEMORY_INTERVAL: u32 = 1 << 4;
67    /// Every defined header flag.
68    pub const ALL: u32 =
69        COMPLETE | CPU_AVAILABLE | SYSMEM_PRESENT | SYSMEM_DENIED | MEMORY_INTERVAL;
70}
71
72/// [`TelemetryCellV1::flags`].
73pub mod cell_flags {
74    pub const CPU_AVAILABLE: u32 = 1 << 0;
75    pub const QUOTA_AVAILABLE: u32 = 1 << 1;
76    pub const LIMIT_AVAILABLE: u32 = 1 << 2;
77    pub const NAME_TRUNCATED: u32 = 1 << 3;
78    pub const ALL: u32 = CPU_AVAILABLE | QUOTA_AVAILABLE | LIMIT_AVAILABLE | NAME_TRUNCATED;
79}

81/// [`TelemetryTaskV1::flags`].
82pub mod task_flags {
83    pub const CPU_AVAILABLE: u32 = 1 << 0;
84    pub const NAME_TRUNCATED: u32 = 1 << 1;
85    /// The task's user stack is charged to its Cell's quota (worker threads), so
86    /// it is already inside the Cell's `quota_charged_bytes`.
87    pub const STACK_IN_QUOTA: u32 = 1 << 2;
88    pub const ALL: u32 = CPU_AVAILABLE | NAME_TRUNCATED | STACK_IN_QUOTA;
89}

164/// Why kernel-written bytes are not a V1 snapshot.
165#[derive(Clone, Copy, Debug, PartialEq, Eq)]
166pub enum TelemetryDecodeError {
167    /// Fewer bytes than a header.
168    Short,
169    /// `version` is not [`TELEMETRY_VERSION`].
170    Version(u32),
171    /// `scope` is not [`SCOPE_SYSTEM`].
172    Scope(u32),
173    /// The byte count does not match the emitted row counts exactly.
174    Length,
175    /// A count, flag, state, reserved field or ordering rule is violated.
176    Malformed,
177}

179/// Cut `name` to [`NAME_LEN`] bytes at a UTF-8 boundary; `true` when it was cut.
180pub fn encode_name(name: &str) -> ([u8; NAME_LEN], bool) {
181    let mut out = [0u8; NAME_LEN];
182    let mut end = name.len().min(NAME_LEN);
183    while !name.is_char_boundary(end) {
184        end -= 1;
185    }
186    out[..end].copy_from_slice(&name.as_bytes()[..end]);
187    (out, end < name.len())
188}
189
190/// The NUL-terminated prefix of a row name.
191pub fn name_bytes(name: &[u8; NAME_LEN]) -> &[u8] {
192    let end = name.iter().position(|&b| b == 0).unwrap_or(NAME_LEN);
193    &name[..end]
194}

196fn put_u32(out: &mut [u8], at: usize, value: u32) {
197    out[at..at + 4].copy_from_slice(&value.to_le_bytes());
198}
199fn put_u64(out: &mut [u8], at: usize, value: u64) {
200    out[at..at + 8].copy_from_slice(&value.to_le_bytes());
201}
202fn get_u32(bytes: &[u8], at: usize) -> u32 {
203    let mut word = [0u8; 4];
204    word.copy_from_slice(&bytes[at..at + 4]);
205    u32::from_le_bytes(word)
206}
207fn get_u64(bytes: &[u8], at: usize) -> u64 {
208    let mut word = [0u8; 8];
209    word.copy_from_slice(&bytes[at..at + 8]);
210    u64::from_le_bytes(word)
211}
212fn get_name(bytes: &[u8], at: usize) -> [u8; NAME_LEN] {
213    let mut name = [0u8; NAME_LEN];
214    name.copy_from_slice(&bytes[at..at + NAME_LEN]);
215    name
216}

218impl TelemetryHeaderV1 {
219    pub fn to_bytes(&self) -> [u8; HEADER_LEN] {
220        let mut out = [0u8; HEADER_LEN];
221        put_u32(&mut out, 0, self.version);
222        put_u32(&mut out, 4, self.flags);
223        put_u64(&mut out, 8, self.provider_epoch);
224        put_u64(&mut out, 16, self.sample_id);
225        put_u64(&mut out, 24, self.capture_start_ns);
226        put_u64(&mut out, 32, self.capture_end_ns);
227        put_u32(&mut out, 40, self.online_cpus);
228        put_u32(&mut out, 44, self.scope);
229        put_u32(&mut out, 48, self.total_cells);
230        put_u32(&mut out, 52, self.total_tasks);
231        put_u32(&mut out, 56, self.emitted_cells);
232        put_u32(&mut out, 60, self.emitted_tasks);
233        put_u64(&mut out, 64, self.mem_total_frames);
234        put_u64(&mut out, 72, self.mem_used_frames);
235        put_u64(&mut out, 80, self.mem_free_frames);
236        put_u64(&mut out, 88, self.mem_page_size);
237        out
238    }
239
240    pub fn read(bytes: &[u8; HEADER_LEN]) -> Self {
241        Self {
242            version: get_u32(bytes, 0),
243            flags: get_u32(bytes, 4),
244            provider_epoch: get_u64(bytes, 8),
245            sample_id: get_u64(bytes, 16),
246            capture_start_ns: get_u64(bytes, 24),
247            capture_end_ns: get_u64(bytes, 32),
248            online_cpus: get_u32(bytes, 40),
249            scope: get_u32(bytes, 44),
250            total_cells: get_u32(bytes, 48),
251            total_tasks: get_u32(bytes, 52),
252            emitted_cells: get_u32(bytes, 56),
253            emitted_tasks: get_u32(bytes, 60),
254            mem_total_frames: get_u64(bytes, 64),
255            mem_used_frames: get_u64(bytes, 72),
256            mem_free_frames: get_u64(bytes, 80),
257            mem_page_size: get_u64(bytes, 88),
258        }
259    }
260
261    pub const fn has(&self, flag: u32) -> bool {
262        self.flags & flag != 0
263    }
264
265    /// Exact byte count of the snapshot this header describes.
266    pub fn snapshot_len(&self) -> Option<usize> {
267        (self.emitted_cells as usize)
268            .checked_mul(CELL_LEN)?
269            .checked_add((self.emitted_tasks as usize).checked_mul(TASK_LEN)?)?
270            .checked_add(HEADER_LEN)
271    }
272
273    fn validate(&self) -> Result<(), TelemetryDecodeError> {
274        use header_flags::*;
275        if self.version != TELEMETRY_VERSION {
276            return Err(TelemetryDecodeError::Version(self.version));
277        }
278        if self.scope != SCOPE_SYSTEM {
279            return Err(TelemetryDecodeError::Scope(self.scope));
280        }
281        let memory = [
282            self.mem_total_frames,
283            self.mem_used_frames,
284            self.mem_free_frames,
285            self.mem_page_size,
286        ];
287        let malformed = self.flags & !ALL != 0
288            || self.provider_epoch == 0
289            || self.sample_id == 0
290            || self.emitted_cells > self.total_cells
291            || self.emitted_tasks > self.total_tasks
292            || (self.has(COMPLETE) && self.emitted_cells != self.total_cells)
293            || (self.has(SYSMEM_PRESENT) && self.has(SYSMEM_DENIED))
294            || (!self.has(SYSMEM_PRESENT) && memory.iter().any(|&word| word != 0))
295            || (self.has(CPU_AVAILABLE) && self.capture_end_ns < self.capture_start_ns);
296        if malformed {
297            Err(TelemetryDecodeError::Malformed)
298        } else {
299            Ok(())
300        }
301    }
302}

387/// A validated view over kernel-written snapshot bytes.
388#[derive(Clone, Copy, Debug)]
389pub struct TelemetrySnapshot<'a> {
390    header: TelemetryHeaderV1,
391    cells: &'a [u8],
392    tasks: &'a [u8],
393}
394
395impl<'a> TelemetrySnapshot<'a> {
396    /// Parse exactly the bytes `TelemetrySnapshot` reported writing.
397    ///
398    /// Rejects anything but a whole V1 snapshot: wrong version or scope, a byte
399    /// count that disagrees with the emitted counts, unknown flag bits, invalid
400    /// states, a nonzero reserved field, or rows out of the documented order
401    /// (Cells by generation, tasks by `(generation, tid)`).
402    pub fn parse(bytes: &'a [u8]) -> Result<Self, TelemetryDecodeError> {
403        let head: &[u8; HEADER_LEN] = bytes
404            .get(..HEADER_LEN)
405            .and_then(|head| head.try_into().ok())
406            .ok_or(TelemetryDecodeError::Short)?;
407        let header = TelemetryHeaderV1::read(head);
408        header.validate()?;
409        if header.snapshot_len() != Some(bytes.len()) {
410            return Err(TelemetryDecodeError::Length);
411        }
412        let split = HEADER_LEN + header.emitted_cells as usize * CELL_LEN;
413        let snapshot = Self {
414            header,
415            cells: &bytes[HEADER_LEN..split],
416            tasks: &bytes[split..],
417        };
418        let mut previous = 0u64;
419        for cell in snapshot.cells() {
420            if !cell.valid() || cell.generation <= previous {
421                return Err(TelemetryDecodeError::Malformed);
422            }
423            previous = cell.generation;
424        }
425        let mut previous = (0u64, 0u64);
426        for task in snapshot.tasks() {
427            let key = (task.generation, task.tid);
428            if !task.valid() || key <= previous {
429                return Err(TelemetryDecodeError::Malformed);
430            }
431            previous = key;
432        }
433        Ok(snapshot)
434    }
435
436    pub fn header(&self) -> &TelemetryHeaderV1 {
437        &self.header
438    }
439
440    pub fn cells(&self) -> impl Iterator<Item = TelemetryCellV1> + 'a {
441        self.cells
442            .chunks_exact(CELL_LEN)
443            .filter_map(|row| row.try_into().ok())
444            .map(TelemetryCellV1::read)
445    }
446
447    pub fn tasks(&self) -> impl Iterator<Item = TelemetryTaskV1> + 'a {
448        self.tasks
449            .chunks_exact(TASK_LEN)
450            .filter_map(|row| row.try_into().ok())
451            .map(TelemetryTaskV1::read)
452    }
453}

599        );
600        let bytes = encode(&header(0, 2), &[], &[task(4, 20), task(4, 11)]);
601        assert_eq!(
602            TelemetrySnapshot::parse(&bytes).err(),
603            Some(TelemetryDecodeError::Malformed)
604        );
605        let mut odd = cell(4);

629    fn names_are_cut_on_a_char_boundary_and_flagged() {
630        assert_eq!(name_bytes(&encode_name("top").0), b"top");
631        assert!(!encode_name("top").1);
632        let long = "é".repeat(17); // 34 bytes; 16 whole chars fit in 32
633        let (name, cut) = encode_name(&long);
634        assert!(cut);
635        assert_eq!(name_bytes(&name).len(), 32);
636        assert!(core::str::from_utf8(name_bytes(&name)).is_ok());
637        // A two-byte char straddling byte 32 is dropped whole, never split.
638        let (name, cut) = encode_name(&alloc::format!("{}é", "x".repeat(31)));
639        assert!(cut);
640        assert_eq!(name_bytes(&name), "x".repeat(31).as_bytes());
641    }
