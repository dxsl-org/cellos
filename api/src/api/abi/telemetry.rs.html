2//! Cell-aware telemetry snapshot ABI (`TelemetrySnapshot = 430`).
3//!
4//! Phase 06 S6-1 (`.agents/261020-1750-cellbox-cli/phase-06-entry-dossier.md` §5,
5//! approved CP1 2026-10-23). One call writes, in one capture, a header followed by
6//! whole Cell rows and then whole task rows into a caller buffer:
7//!
8//! ```text
9//! TelemetryHeaderV1 (96) | TelemetryCellV1 (96) × emitted_cells | TelemetryTaskV1 (88) × emitted_tasks
10//! ```
11//!
12//! Identity is `(provider_epoch, cell_id, generation)` for a Cell incarnation and
13//! `(provider_epoch, tid)` for a task: TIDs are never reissued within one boot and
14//! generations are global, so neither needs a separate incarnation field (DV-1/DV-2).
15//!
16//! Units: every `*_ns` field is nanoseconds on the monotonic clock `GetTime(0)`
17//! exposes, so `cpu_ns` deltas divide directly by `capture_start_ns` deltas.
18//! `cpu_ns` is exact on-CPU time (charged at every scheduler pass), and a Cell's
19//! `cpu_ns` includes the CPU of its already-exited threads for this incarnation.
20//! Memory: `quota_charged_bytes` is the Cell's whole quota charge (kernel heap in
21//! its context, worker stacks, pipes, session objects), not a heap size, and task
22//! `user_stack_bytes`/`segment_bytes` are informational — never sum them into the
23//! Cell (DV-4).
24//!
25//! Encoding is explicit little-endian at fixed offsets; the `repr(C)` structs are
26//! the documented layout, never reinterpreted from raw bytes. A V1 row is never a
27//! V2 row: a later layout gets a new version number.
28
29/// Header `version` written by this layout.
30pub const TELEMETRY_VERSION: u32 = 1;
31/// Exact byte size of [`TelemetryHeaderV1`].
32pub const HEADER_LEN: usize = 96;
33/// Exact byte size of [`TelemetryCellV1`].
34pub const CELL_LEN: usize = 96;
35/// Exact byte size of [`TelemetryTaskV1`].
36pub const TASK_LEN: usize = 88;
37/// Required alignment of the caller buffer.
38pub const BUFFER_ALIGN: usize = 8;
39/// Bytes of a row's `name`; longer names are cut at a UTF-8 boundary and flagged.
40pub const NAME_LEN: usize = 32;
41/// Header `scope`: every Cell and task in the system (the only scope offered, DV-6).
42pub const SCOPE_SYSTEM: u32 = 1;
43
44/// `a2` request flags. Every other bit must be zero.
45pub mod request {
46    /// Emit task rows after the Cell rows.
47    pub const TASKS: usize = 1 << 0;
48    /// Fill the system-memory block (also needs the caller's `observe_memory` right).
49    pub const SYSTEM_MEMORY: usize = 1 << 1;
50    /// Every defined request bit.
51    pub const ALL: usize = TASKS | SYSTEM_MEMORY;
52}
53
54/// [`TelemetryHeaderV1::flags`].
55pub mod header_flags {
56    /// Every requested row was emitted; no row was dropped for lack of space.
57    pub const COMPLETE: u32 = 1 << 0;
58    /// The monotonic clock was available: `cpu_ns` and the capture times are valid.
59    pub const CPU_AVAILABLE: u32 = 1 << 1;
60    /// The `mem_*` fields hold the global frame-allocator snapshot.
61    pub const SYSMEM_PRESENT: u32 = 1 << 2;
62    /// System memory was requested but the caller lacks `observe_memory`; `mem_*` are 0.
63    pub const SYSMEM_DENIED: u32 = 1 << 3;
64    /// Memory counters are lock-free and were read during the capture interval, so
65    /// they are not instant-consistent with membership and CPU.
66    pub const MEMORY_INTERVAL: u32 = 1 << 4;
67    /// Every defined header flag.
68    pub const ALL: u32 = COMPLETE | CPU_AVAILABLE | SYSMEM_PRESENT | SYSMEM_DENIED | MEMORY_INTERVAL;
69}
70
71/// [`TelemetryCellV1::flags`].
72pub mod cell_flags {
73    pub const CPU_AVAILABLE: u32 = 1 << 0;
74    pub const QUOTA_AVAILABLE: u32 = 1 << 1;
75    pub const LIMIT_AVAILABLE: u32 = 1 << 2;
76    pub const NAME_TRUNCATED: u32 = 1 << 3;
77    pub const ALL: u32 = CPU_AVAILABLE | QUOTA_AVAILABLE | LIMIT_AVAILABLE | NAME_TRUNCATED;
78}

80/// [`TelemetryTaskV1::flags`].
81pub mod task_flags {
82    pub const CPU_AVAILABLE: u32 = 1 << 0;
83    pub const NAME_TRUNCATED: u32 = 1 << 1;
84    /// The task's user stack is charged to its Cell's quota (worker threads), so
85    /// it is already inside the Cell's `quota_charged_bytes`.
86    pub const STACK_IN_QUOTA: u32 = 1 << 2;
87    pub const ALL: u32 = CPU_AVAILABLE | NAME_TRUNCATED | STACK_IN_QUOTA;
88}

163/// Why kernel-written bytes are not a V1 snapshot.
164#[derive(Clone, Copy, Debug, PartialEq, Eq)]
165pub enum TelemetryDecodeError {
166    /// Fewer bytes than a header.
167    Short,
168    /// `version` is not [`TELEMETRY_VERSION`].
169    Version(u32),
170    /// `scope` is not [`SCOPE_SYSTEM`].
171    Scope(u32),
172    /// The byte count does not match the emitted row counts exactly.
173    Length,
174    /// A count, flag, state, reserved field or ordering rule is violated.
175    Malformed,
176}

178/// Cut `name` to [`NAME_LEN`] bytes at a UTF-8 boundary; `true` when it was cut.
179pub fn encode_name(name: &str) -> ([u8; NAME_LEN], bool) {
180    let mut out = [0u8; NAME_LEN];
181    let mut end = name.len().min(NAME_LEN);
182    while !name.is_char_boundary(end) {
183        end -= 1;
184    }
185    out[..end].copy_from_slice(&name.as_bytes()[..end]);
186    (out, end < name.len())
187}
188
189/// The NUL-terminated prefix of a row name.
190pub fn name_bytes(name: &[u8; NAME_LEN]) -> &[u8] {
191    let end = name.iter().position(|&b| b == 0).unwrap_or(NAME_LEN);
192    &name[..end]
193}

195fn put_u32(out: &mut [u8], at: usize, value: u32) {
196    out[at..at + 4].copy_from_slice(&value.to_le_bytes());
197}
198fn put_u64(out: &mut [u8], at: usize, value: u64) {
199    out[at..at + 8].copy_from_slice(&value.to_le_bytes());
200}
201fn get_u32(bytes: &[u8], at: usize) -> u32 {
202    let mut word = [0u8; 4];
203    word.copy_from_slice(&bytes[at..at + 4]);
204    u32::from_le_bytes(word)
205}
206fn get_u64(bytes: &[u8], at: usize) -> u64 {
207    let mut word = [0u8; 8];
208    word.copy_from_slice(&bytes[at..at + 8]);
209    u64::from_le_bytes(word)
210}
211fn get_name(bytes: &[u8], at: usize) -> [u8; NAME_LEN] {
212    let mut name = [0u8; NAME_LEN];
213    name.copy_from_slice(&bytes[at..at + NAME_LEN]);
214    name
215}

217impl TelemetryHeaderV1 {
218    pub fn to_bytes(&self) -> [u8; HEADER_LEN] {
219        let mut out = [0u8; HEADER_LEN];
220        put_u32(&mut out, 0, self.version);
221        put_u32(&mut out, 4, self.flags);
222        put_u64(&mut out, 8, self.provider_epoch);
223        put_u64(&mut out, 16, self.sample_id);
224        put_u64(&mut out, 24, self.capture_start_ns);
225        put_u64(&mut out, 32, self.capture_end_ns);
226        put_u32(&mut out, 40, self.online_cpus);
227        put_u32(&mut out, 44, self.scope);
228        put_u32(&mut out, 48, self.total_cells);
229        put_u32(&mut out, 52, self.total_tasks);
230        put_u32(&mut out, 56, self.emitted_cells);
231        put_u32(&mut out, 60, self.emitted_tasks);
232        put_u64(&mut out, 64, self.mem_total_frames);
233        put_u64(&mut out, 72, self.mem_used_frames);
234        put_u64(&mut out, 80, self.mem_free_frames);
235        put_u64(&mut out, 88, self.mem_page_size);
236        out
237    }
238
239    pub fn read(bytes: &[u8; HEADER_LEN]) -> Self {
240        Self {
241            version: get_u32(bytes, 0),
242            flags: get_u32(bytes, 4),
243            provider_epoch: get_u64(bytes, 8),
244            sample_id: get_u64(bytes, 16),
245            capture_start_ns: get_u64(bytes, 24),
246            capture_end_ns: get_u64(bytes, 32),
247            online_cpus: get_u32(bytes, 40),
248            scope: get_u32(bytes, 44),
249            total_cells: get_u32(bytes, 48),
250            total_tasks: get_u32(bytes, 52),
251            emitted_cells: get_u32(bytes, 56),
252            emitted_tasks: get_u32(bytes, 60),
253            mem_total_frames: get_u64(bytes, 64),
254            mem_used_frames: get_u64(bytes, 72),
255            mem_free_frames: get_u64(bytes, 80),
256            mem_page_size: get_u64(bytes, 88),
257        }
258    }
259
260    pub const fn has(&self, flag: u32) -> bool {
261        self.flags & flag != 0
262    }
263
264    /// Exact byte count of the snapshot this header describes.
265    pub fn snapshot_len(&self) -> Option<usize> {
266        (self.emitted_cells as usize)
267            .checked_mul(CELL_LEN)?
268            .checked_add((self.emitted_tasks as usize).checked_mul(TASK_LEN)?)?
269            .checked_add(HEADER_LEN)
270    }
271
272    fn validate(&self) -> Result<(), TelemetryDecodeError> {
273        use header_flags::*;
274        if self.version != TELEMETRY_VERSION {
275            return Err(TelemetryDecodeError::Version(self.version));
276        }
277        if self.scope != SCOPE_SYSTEM {
278            return Err(TelemetryDecodeError::Scope(self.scope));
279        }
280        let memory = [self.mem_total_frames, self.mem_used_frames, self.mem_free_frames, self.mem_page_size];
281        let malformed = self.flags & !ALL != 0
282            || self.provider_epoch == 0
283            || self.sample_id == 0
284            || self.emitted_cells > self.total_cells
285            || self.emitted_tasks > self.total_tasks
286            || (self.has(COMPLETE) && self.emitted_cells != self.total_cells)
287            || (self.has(SYSMEM_PRESENT) && self.has(SYSMEM_DENIED))
288            || (!self.has(SYSMEM_PRESENT) && memory.iter().any(|&word| word != 0))
289            || (self.has(CPU_AVAILABLE) && self.capture_end_ns < self.capture_start_ns);
290        if malformed { Err(TelemetryDecodeError::Malformed) } else { Ok(()) }
291    }
292}

377/// A validated view over kernel-written snapshot bytes.
378#[derive(Clone, Copy, Debug)]
379pub struct TelemetrySnapshot<'a> {
380    header: TelemetryHeaderV1,
381    cells: &'a [u8],
382    tasks: &'a [u8],
383}
384
385impl<'a> TelemetrySnapshot<'a> {
386    /// Parse exactly the bytes `TelemetrySnapshot` reported writing.
387    ///
388    /// Rejects anything but a whole V1 snapshot: wrong version or scope, a byte
389    /// count that disagrees with the emitted counts, unknown flag bits, invalid
390    /// states, a nonzero reserved field, or rows out of the documented order
391    /// (Cells by generation, tasks by `(generation, tid)`).
392    pub fn parse(bytes: &'a [u8]) -> Result<Self, TelemetryDecodeError> {
393        let head: &[u8; HEADER_LEN] = bytes
394            .get(..HEADER_LEN)
395            .and_then(|head| head.try_into().ok())
396            .ok_or(TelemetryDecodeError::Short)?;
397        let header = TelemetryHeaderV1::read(head);
398        header.validate()?;
399        if header.snapshot_len() != Some(bytes.len()) {
400            return Err(TelemetryDecodeError::Length);
401        }
402        let split = HEADER_LEN + header.emitted_cells as usize * CELL_LEN;
403        let snapshot = Self { header, cells: &bytes[HEADER_LEN..split], tasks: &bytes[split..] };
404        let mut previous = 0u64;
405        for cell in snapshot.cells() {
406            if !cell.valid() || cell.generation <= previous {
407                return Err(TelemetryDecodeError::Malformed);
408            }
409            previous = cell.generation;
410        }
411        let mut previous = (0u64, 0u64);
412        for task in snapshot.tasks() {
413            let key = (task.generation, task.tid);
414            if !task.valid() || key <= previous {
415                return Err(TelemetryDecodeError::Malformed);
416            }
417            previous = key;
418        }
419        Ok(snapshot)
420    }
421
422    pub fn header(&self) -> &TelemetryHeaderV1 {
423        &self.header
424    }
425
426    pub fn cells(&self) -> impl Iterator<Item = TelemetryCellV1> + 'a {
427        self.cells
428            .chunks_exact(CELL_LEN)
429            .filter_map(|row| row.try_into().ok())
430            .map(TelemetryCellV1::read)
431    }
432
433    pub fn tasks(&self) -> impl Iterator<Item = TelemetryTaskV1> + 'a {
434        self.tasks
435            .chunks_exact(TASK_LEN)
436            .filter_map(|row| row.try_into().ok())
437            .map(TelemetryTaskV1::read)
438    }
439}

555    fn names_are_cut_on_a_char_boundary_and_flagged() {
556        assert_eq!(name_bytes(&encode_name("top").0), b"top");
557        assert!(!encode_name("top").1);
558        let long = "é".repeat(17); // 34 bytes; 16 whole chars fit in 32
559        let (name, cut) = encode_name(&long);
560        assert!(cut);
561        assert_eq!(name_bytes(&name).len(), 32);
562        assert!(core::str::from_utf8(name_bytes(&name)).is_ok());

564        let (name, cut) = encode_name(&alloc::format!("{}é", "x".repeat(31)));
565        assert!(cut);
566        assert_eq!(name_bytes(&name), "x".repeat(31).as_bytes());
