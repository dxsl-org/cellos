2//! The kernel's statement about where a cell's directory-handle set came from.
3//!
4//! [`super::caller_identity`] answers "which cell is calling me?" in a fixed
5//! 32-byte trailer written past the payload of a recv buffer. A handle set is
6//! variable-length and would not fit there, and padding it into the trailer
7//! would mean either a hard truncation or reserving the tail of every recv
8//! buffer for a field almost no message needs.
9//!
10//! So this record is not pushed; it is pulled. The filesystem service, having
11//! learned a caller's cell id from the attested trailer, asks the kernel for
12//! that cell's record and the kernel writes this structure into the service's
13//! own buffer during the service's own syscall. Its trust basis is identical to
14//! the trailer's — produced by the kernel from live scheduler state, never
15//! relayed through a message any cell composed — while the size question
16//! disappears, because the service supplies a buffer it sized itself and an
17//! undersized one is an error rather than a silent truncation.
18//!
19//! ## Invariants
20//! - A record with `count > 0` always names a spawner. "These handles came from
21//!   somewhere" is not a claim a service can check.
22//! - `cell_id` of `0` is never a cell (the kernel uses it for its own
23//!   allocations) and is rejected at parse time.
24//! - The kernel states only provenance. Whether the spawner genuinely held these
25//!   handles is a question about the filesystem service's own table, and only it
26//!   can answer it.
27
28use super::dir_handles::{DirHandleSet, MAX_SPAWN_DIR_HANDLES};
29
30/// Encoded size of a [`ViDirHandleAttestation`].
31pub const DIR_ATTESTATION_LEN: usize = 48 + 8 * MAX_SPAWN_DIR_HANDLES;
32
33/// Layout version of the encoded record.
34pub const DIR_ATTESTATION_VERSION: u32 = 1;
35
36/// Tag distinguishing a kernel-written record from whatever the buffer held.
37///
38/// Not a security control — in a single address space a cell that can write
39/// another's buffer can write anything. It exists so a service can tell "the
40/// kernel filled this in" from "my buffer still holds the last reply".
41const MAGIC: u32 = 0x5649_4448; // "VIDH"
42
43/// What the kernel says about a cell's inherited directory handles.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub struct ViDirHandleAttestation {
46    /// The cell the record describes.
47    pub cell_id: u64,
48    /// That cell's generation, so a service cannot bind a dead cell's set to the
49    /// respawned cell that took its id.
50    pub generation: u64,
51    /// The cell that named the set at spawn. `0` only when the set is empty.
52    pub spawner_cell_id: u64,
53    /// The spawner's generation at the moment of the spawn.
54    pub spawner_generation: u64,
55    /// The handles the spawner named. Unvalidated authority: the kernel asserts
56    /// only that this spawner named them.
57    pub set: DirHandleSet,
58}
59
60impl ViDirHandleAttestation {
61    /// Encode into the fixed-size record the kernel writes.
62    pub fn to_bytes(&self) -> [u8; DIR_ATTESTATION_LEN] {
63        let mut out = [0u8; DIR_ATTESTATION_LEN];
64        out[0..4].copy_from_slice(&MAGIC.to_le_bytes());
65        out[4..8].copy_from_slice(&DIR_ATTESTATION_VERSION.to_le_bytes());
66        out[8..16].copy_from_slice(&self.cell_id.to_le_bytes());
67        out[16..24].copy_from_slice(&self.generation.to_le_bytes());
68        out[24..32].copy_from_slice(&self.spawner_cell_id.to_le_bytes());
69        out[32..40].copy_from_slice(&self.spawner_generation.to_le_bytes());
70        out[40..44].copy_from_slice(&(self.set.len() as u32).to_le_bytes());
71        // [44..48] reserved: zero today so a flag word can be added later
72        // without moving the handle array.
73        for (i, h) in self.set.as_slice().iter().enumerate() {
74            let at = 48 + i * 8;
75            out[at..at + 8].copy_from_slice(&h.to_le_bytes());
76        }
77        out
78    }
79
80    /// Parse a record the kernel wrote into a service-owned buffer.
81    ///
82    /// Returns `None` for a short buffer, an untagged or wrong-version record, a
83    /// `cell_id` of `0`, a count past [`MAX_SPAWN_DIR_HANDLES`], a non-empty set
84    /// with no spawner, or any handle the set rebuilds as invalid. Every one of
85    /// those means "no attested set", and the only correct response to that is to
86    /// bind nothing.
87    pub fn from_bytes(bytes: &[u8]) -> Option<Self> {
88        if bytes.len() < DIR_ATTESTATION_LEN {
89            return None;
90        }
91        let word32 = |at: usize| -> u32 {
92            let mut b = [0u8; 4];
93            b.copy_from_slice(&bytes[at..at + 4]);
94            u32::from_le_bytes(b)
95        };
96        let word64 = |at: usize| -> u64 {
97            let mut b = [0u8; 8];
98            b.copy_from_slice(&bytes[at..at + 8]);
99            u64::from_le_bytes(b)
100        };
101        if word32(0) != MAGIC || word32(4) != DIR_ATTESTATION_VERSION {
102            return None;
103        }
104        let cell_id = word64(8);
105        if cell_id == 0 {
106            return None;
107        }
108        let count = word32(40) as usize;
109        if count > MAX_SPAWN_DIR_HANDLES {
110            return None;
111        }
112        let spawner_cell_id = word64(24);
113        if count > 0 && spawner_cell_id == 0 {
114            return None; // a set nobody is named as the source of
115        }
116        let mut carrier = super::dir_handles::ViSpawnDirHandles::EMPTY;
117        carrier.count = count as u32;
118        for i in 0..count {
119            carrier.handles[i] = word64(48 + i * 8);
120        }
121        Some(Self {
122            cell_id,
123            generation: word64(16),
124            spawner_cell_id,
125            spawner_generation: word64(32),
126            set: DirHandleSet::from_carrier(&carrier).ok()?,
127        })
128    }
129}
