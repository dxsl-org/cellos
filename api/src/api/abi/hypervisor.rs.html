2//! ARM64 EL2 VMM syscall ABI — stable kernel↔cell contract.
3//!
4//! ⚠️ **Law 1**: this file is part of the stable ABI between kernel and Cells.
5//! Any changes require 2× user confirmation.  `VERSION = 2` (Tier 3b x86 P04
6//! appended discriminants 8-11).  To add new exit types, add variants at new
7//! explicit discriminant values only — never change existing discriminants,
8//! field names, or field types.
9//!
10//! # Confidential-computing neutrality (frozen invariants — `review-cc-neutral-abi-freeze.md`)
11//! The VERSION=2 freeze must not preclude TDX / SEV-SNP / ARM-CCA guests later:
12//! - **I1 (size envelope is the freeze boundary):** `size_of::<ViVmExit>()` is
13//!   pinned by the `Hvc { regs: [u64; 8] }` variant (~80 B) and asserted below.
14//!   A future CC exit MUST carry a *shared-region reference* (e.g. a GHCB GPA +
15//!   metadata), NEVER an inline guest register file — a TDX `TDG.VP.VMCALL`
16//!   register dump (~13 GPRs, ~104 B) would overflow the envelope and break every
17//!   VERSION=2 cell's `validate_user_buf`.
18//! - **I2 (field provenance):** every field is a value the guest EXPLICITLY
19//!   delivered (ISV=1 syndrome / IOIO qualifier / GHCB / TDVMCALL). `Hvc.regs`
20//!   are published hypercall args only; no variant carries guest RIP or raw
21//!   instruction bytes.
22//! - **I3 (append-only for CC):** a CC attested-launch / sysreg path is always a
23//!   NEW variant at discriminant 12+, never a reshape of PortIn/PortOut/Msr.
24
25/// VM exit reason written by `sys_run_vcpu` into the caller-provided out-param.
26///
27/// `#[repr(C, u8)]` guarantees a stable ABI: the `u8` discriminant precedes each
28/// variant's payload in memory, and the total size equals the largest variant
29/// padded to alignment.  The kernel writes via `*mut ViVmExit` (SAS: kernel and
30/// cell share the same virtual address space, so the pointer is valid in both).
31///
32/// **Frozen at VERSION 1.** Never modify existing variant fields.
33#[repr(C, u8)]
34#[derive(Debug, Clone, Copy)]
35pub enum ViVmExit {
36    /// Stage-2 data-abort (read) — unmapped MMIO IPA; ISV=1 guaranteed.
37    MmioRead { ipa: u64, size: u8, reg: u8 } = 0,
38    /// Stage-2 data-abort (write) — unmapped MMIO IPA; ISV=1 guaranteed.
39    MmioWrite { ipa: u64, size: u8, val: u64 } = 1,
40    /// HVC instruction — covers PSCI calls and general hypercall ABI.
41    Hvc { imm: u16, regs: [u64; 8] } = 2,
42    /// WFI instruction — guest idle; hypervisor may inject a virtual IRQ.
43    Wfi = 3,
44    /// System-register access (EC=0x18) — timer register emulation (P05+).
45    SysReg {
46        op0: u8,
47        op1: u8,
48        crn: u8,
49        crm: u8,
50        op2: u8,
51        rt: u8,
52        is_write: bool,
53    } = 4,
54    /// `budget_ns` budget expired — no guest fault; re-enter after servicing IPC.
55    Preempted = 5,
56    /// Guest requested shutdown — PSCI SYSTEM_OFF / CPU_OFF (P05+).
57    Shutdown = 6,
58    /// Unrecognized exception class — includes S1PTW=1 stage-1 walk faults.
59    /// Treat as fatal guest fault: log `ec`/`iss` and halt the VM.
60    Unknown { ec: u32, iss: u32 } = 7,
61
62    // ── x86 (SVM/VT-x) exits — appended in Tier 3b P04, VERSION 2 ────────────
63    /// x86 `IN` from an I/O port. `reg` is reserved (guest `IN` always targets
64    /// (E)AX); kept for symmetry with `MmioRead`.
65    PortIn { port: u16, size: u8, reg: u8 } = 8,
66    /// x86 `OUT` to an I/O port. `val` holds the low `size` bytes written.
67    PortOut { port: u16, size: u8, val: u32 } = 9,
68    /// x86 `HLT` — guest idle; the hypervisor may inject an IRQ (P05).
69    Hlt = 10,
70    /// x86 RDMSR/WRMSR. `index` = ECX; `val` = EDX:EAX on a write.
71    Msr {
72        index: u32,
73        is_write: bool,
74        val: u64,
75    } = 11,
76}
77
78impl ViVmExit {
79    /// ABI version — increment when adding new discriminant values.
80    pub const VERSION: u32 = 2;
81}
82
83// I1: the x86 variants (≤ 16 B payload) must NOT grow the enum past the
84// `Hvc { regs: [u64; 8] }` envelope, or `validate_user_buf(size_of::<ViVmExit>())`
85// in the run_vcpu syscall path breaks for every existing cell. Pin it.
86const _: () = assert!(core::mem::size_of::<ViVmExit>() == 80);
