3/// System Call Identifiers (The Contract)
4///
5/// These IDs must match between Kernel and User (libs/ostd).
6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7#[repr(usize)]
8pub enum ViSyscall {
9    // === IPC (0-9) ===
10    Send = 0,
11    /// Blocking receive.
12    ///
13    /// ABI: a0 = mask, a1 = buf_ptr, a2 = buf_len, a3 = flags → sender tid.
14    ///
15    /// a3 was unused (every pre-existing caller passes 0). Passing
16    /// [`crate::caller_identity::RECV_ATTEST_CALLER`] additionally makes the
17    /// kernel write a [`crate::caller_identity::CallerIdentity`] into the LAST
18    /// `CALLER_IDENTITY_LEN` bytes of `buf`, after the payload — the only
19    /// unforgeable answer to "which cell called me?" available to a service.
20    /// Opting in reserves that tail; see that module for the invariants.
21    Recv = 1,
22    Call = 2,
23    Reply = 3,
24    /// Non-blocking send: delivers immediately if target is in `Recv`, otherwise
25    /// drops the message and returns without blocking the caller.
26    /// Prevents deadlock when the focused cell is busy (e.g. reading UART).
27    TrySend = 4,
28    /// Non-blocking receive: returns 0 immediately when no message is queued
29    /// (vs `Recv`, which parks the task until a message arrives).
30    TryRecv = 7,
31
32    // === Process Management (10-49) ===
33    Exit = 60, // Linux compat usually, but we define our own space
34    /// Force-terminate another task by TID.  Non-blocking: caller continues.
35    /// Requires `SpawnCap` on caller; rejected on system cells (block_io/network_cap).
36    /// ABI: a0 = target_tid → 0 on success, usize::MAX on error.
37    ForceExit = 61,
38    Spawn = 5,
39    Exec = 6, // Deprecated/Legacy
40    /// Spawn a cell from caller-supplied ELF bytes already resident in memory.
41    ///
42    /// The kernel still routes the request through the normal admission gate,
43    /// but approval is by an exact launch profile on the mem route, not by the
44    /// caller holding broad lifecycle authority. Phase 04 intentionally leaves
45    /// this route fail-closed for shell/user cells.
46    SpawnFromMem = 10,
47    /// Spawn a cell by reading its ELF from a VFS path.
48    ///
49    /// ABI: a0 = path_ptr, a1 = path_len → tid (>0) on success.
50    /// Phase 04 approval is by an exact kernel launch profile for the tuple
51    /// `(caller identity, SpawnFromPath, path)`. This is distinct from
52    /// lifecycle operations that still require ambient `SpawnCap`.
53    SpawnFromPath = 12,
54    /// Open a file by path, returning a kernel capability ID.
55    /// ABI: a0 = path_ptr, a1 = path_len → CapId (> 0) on success.
56    OpenCap = 13,
57    /// Read bytes from a cap-backed file.
58    /// ABI: a0 = cap_id, a1 = buf_ptr, a2 = buf_len → bytes_read.
59    ReadCap = 14,
60    /// Revoke a capability (close).
61    /// ABI: a0 = cap_id → 0 on success.
62    CloseCap = 15,
63    /// Seek a cap-backed file cursor.
64    /// ABI: a0 = cap_id, a1 = offset (i64 bits), a2 = whence (0=Start,1=Current,2=End)
65    ///      → new absolute position on success.
66    /// Requires the same allowlist bit as ReadCap (bit 16).
67    SeekCap = 228,
68    /// Write bytes into a cap-backed file at the current cursor position.
69    /// ABI: a0 = cap_id, a1 = buf_ptr, a2 = buf_len → bytes_written.
70    /// Requires WriteCap allowlist bit (bit 45).
71    WriteCap = 229,
72    /// Query the size of a cap-backed file in bytes.
73    /// ABI: a0 = cap_id → file_size on success. Does NOT move the cursor.
74    /// Shares the ReadCap allowlist bit (bit 16).
75    StatCap = 230,
76    /// Truncate a cap-backed file to exactly `len` bytes.
77    /// ABI: a0 = cap_id, a1 = len → 0 on success.
78    /// Returns an error if `len > current_size`; use `WriteCap` to extend.
79    /// Requires TruncateCap allowlist bit (bit 46).
80    TruncateCap = 231,
81    /// Flush all dirty pages for a cap-backed file to the underlying block device (fsync).
82    /// ABI: a0 = cap_id → 0 on success.
83    /// No-op on write-through filesystems (RamDisk). Hooks into device flush on NVMe (G2).
84    /// Shares the WriteCap allowlist bit (bit 45).
85    SyncCap = 232,
86    /// Map a PCIe device's DMA range into the IOMMU for the calling Cell.
87    /// ABI: a0 = bdf: u32, a1 = phys: u64, a2 = size: usize → IOVA (== phys for SAS) on success.
88    /// Caller must own the BDF via Resource Registry; range must be within Cell memory quota.
89    GrantDma = 233,
90    /// Block the calling Driver Cell until hardware IRQ `irq_num` fires.
91    /// The kernel ISR sets an atomic flag (no lock); the scheduler sweep transitions the task
92    /// to Ready. Lost-wakeup safe: an IRQ that fires between the Cell's queue-check and this
93    /// call is not missed (take_pending before park). Requires PcieDriverCap or PlatformCap.
94    /// ABI: a0 = irq_num (u8), a1 = mmio_base (VirtIO MMIO slot base for InterruptACK, or 0).
95    WaitIrq = 234,
96    /// Register a PCIe device BAR in the kernel ECAM BAR table (Platform Cell only).
97    /// Called by the Platform Cell after scanning ECAM config space; populates the allowlist
98    /// that Driver Cells consult via `sys_request_mmio`. Singleton PlatformCap required.
99    /// ABI: a0 = bdf (u32), a1 = base (usize), a2 = len (usize) → 0 on success.
100    RegisterPcieBar = 235,
101    /// Register a discovered PCI device with class/BAR info (Platform Cell only).
102    ///
103    /// Called by the Platform Cell after ECAM scan; populates kernel `PCI_DEVICES` so
104    /// `sys_find_pcie_device` queries work without a kernel-side ECAM scan.
105    /// `bdf`       = (bus<<16)|(dev<<8)|fun
106    /// `cls`       = (class<<16)|(subclass<<8)|prog_if
107    /// `bar0_base` = physical base of BAR0 (or 0 if absent)
108    /// `bar0_size` = probed BAR0 size in bytes (or 0)
109    /// Requires singleton PlatformCap (allowlist bit 53).
110    RegisterPciDevice = 236,
111    /// Drain up to `max` bytes from the kernel user-log ring buffer into `buf`.
112    ///
113    /// Returns the number of bytes actually copied (may be 0 if no new output).
114    /// The ring buffer is written by `sys_log` (cell stdout) and `print_user_log`
115    /// (kernel's USER: prefixed UART path) — gives the fb-console cell a live
116    /// mirror of everything that appears on the serial console.
117    ///
118    /// ABI: a0 = buf_ptr, a1 = max → bytes_copied.
119    /// Gated by ReadLogCap (allowlist bit 54) — declare with `ReadLog` in manifest.
120    ReadLog = 237,
121    /// Spawn a cell from ELF bytes supplied in a caller-owned Grant region.
122    ///
123    /// The userspace spawner (init/supervisor/shell) reads a cell's ELF from VFS
124    /// into a Grant, then calls this. The kernel applies the IDENTICAL spawn trust
125    /// gate as `SpawnFromPath` (Ed25519 signature, manifest-privilege, capability
126    /// intersection, policy, syscall allowlist, cluster, quota, measurement) over
127    /// the grant bytes — trust is over the bytes, so the source is irrelevant.
128    /// Lets the kernel loader stop reading a filesystem for post-boot spawns
129    /// (G2 loader redesign). Like `SpawnFromPath`, Phase 04 approval is by the
130    /// exact kernel launch profile for `(caller identity, SpawnFromElf,
131    /// path_hint)` rather than by broad lifecycle `SpawnCap`.
132    ///
133    /// ABI: a0 = grant_id, a1 = len (bytes), a2 = path_hint_ptr, a3 = path_hint_len
134    ///      → tid (>0) on success. `path_hint` is advisory (drives the /bin/
135    ///      privilege check, policy lookup, measurement label); a caller lying
136    ///      about it can only LOSE privilege, never gain it.
137    SpawnFromElf = 238,
138    /// Return extended process telemetry rows into a caller-owned buffer.
139    ///
140    /// ABI: a0 = buf_ptr, a1 = row_capacity. Returns rows written.
141    /// Preserves `GetProcs` v1 layout/opcode by exposing a separate v2 row type.
142    GetProcs2 = 239,
143    /// Return one fixed-width global physical-frame allocator snapshot.
144    ///
145    /// ABI: a0 = out_ptr, a1 = out_len. On success writes exactly one
146    /// [`ViMemInfoV1`] and returns its byte size. This aggregate telemetry is
147    /// opt-in because used/free capacity is observable across cells.
148    MemInfo = 243,
149    /// Spawn a cell pinned to a specific hardware core.
150    ///
151    /// ABI: a0 = path_ptr, a1 = path_len, a2 = priority: u8, a3 = core_id: usize.
152    /// On single-core systems core_id must be 0; any other value returns NotSupported.
153    /// Phase 04 approval is by the exact kernel launch profile for
154    /// `(caller identity, SpawnPinned, path)`; lifecycle `SpawnCap` still gates
155    /// `ForceExit`, `NotifyOnExit`, `RegisterService`, and snapshot control.
156    SpawnPinned = 16,
157    /// Serialize all allocated physical frames to the snapshot sector range.
158    /// Triggers a warm-boot-capable snapshot.  Returns frame count on success.
159    Snapshot = 420,
160    Wait = 8,      // Wait for task
161    Yield = 104,   // Linux sched_yield is 24, but we use 104 in current code
162    SetTimer = 35, // Added SetTimer
163    ShmAlloc = 20, // Allocate Shared Memory
164    ShmMap = 21,   // Map Shared Memory to Task
165
166    // === Logging (50-59) ===
167    Log = 11, // Current implementation uses 11
168
169    // === Filesystem (100-199) ===
170    Open = 101,
171    Read = 102,
172    Close = 103,
173    ReadDir = 105,
174    Write = 109,
175    Seek = 106,
176    FileOp = 107, // Rename, Remove
177
178    // === Time/System (120-129) ===
179    GetTime = 120,
180
181    // === GPU / Display (Phase 16) ===
182    /// Flush a pixel rectangle from a cell-provided buffer to the GPU framebuffer.
183    /// ABI: a0 = data_ptr, a1 = data_len, a2 = x, a3 = y (w+h embedded in len)
184    GpuFlush = 300,
185    /// Set or move the VirtIO GPU hardware cursor.
186    /// ABI: a0 = op (0=set sprite, 1=move), a1 = data_ptr (op=0: 64×64 BGRA sprite),
187    ///      a2 = xy ((x<<16)|y), a3 = hot ((hot_x<<16)|hot_y, op=0 only).
188    GpuCursor = 301,
189    /// Query the VirtIO GPU's current scanout resolution.
190    /// ABI: no args → returns (width as u32) << 32 | (height as u32).
191    /// Falls back to (1280, 800) if GPU not yet initialized.
192    GpuGetResolution = 302,

194    // === Network ===
195    /// Transmit one Ethernet frame through the kernel VirtIO NIC.
196    /// ABI: a0 = frame_ptr, a1 = frame_len → 1 on success, 0 on failure.
197    NetTx = 310,
198    /// Receive one pending Ethernet frame from the kernel VirtIO NIC.
199    /// ABI: a0 = buf_ptr, a1 = buf_len → bytes written (0 if none ready).
200    NetRx = 311,

202    // === Advanced IPC (Phase 20) ===
203    /// Recv with a deadline in monotonic ticks.
204    /// ABI: a0 = mask, a1 = buf_ptr, a2 = buf_len, a3 = timeout_ticks → sender_id or ViError::Timeout.
205    RecvTimeout = 201,
206    /// Gather-send: one IPC message assembled from up to 8 non-contiguous buffers.
207    /// ABI: a0 = target, a1 = iovec_ptr ([ptr:usize, len:usize] × count), a2 = iovec_count.
208    SendGather = 202,
209    /// Scatter-receive: one IPC message written into up to 8 non-contiguous buffers.
210    /// ABI: a0 = mask, a1 = iovec_ptr, a2 = iovec_count → sender_id.
211    RecvScatter = 203,
212    /// Register the caller as a watcher of `watched_tid`. When that task exits OR
213    /// faults, the kernel delivers a death notification to the caller's pending
214    /// `Recv` (`sender_id` = the dead tid, payload = the exit/fault reason), so a
215    /// supervisor can wait-any across many children with a single recv loop.
216    /// Requires `SpawnCap`. ABI: a0 = watched_tid → 0 on success, usize::MAX on error.
217    NotifyOnExit = 204,
218    /// Register `tid` as the current provider of well-known `service_id`. SpawnCap
219    /// owns the namespace. Narrow kernel-only exceptions may authorize one fixed
220    /// service ID and `tid=0` self-registration; they grant no general registration
221    /// authority. ABI: a0 = service_id (u16), a1 = tid → 0 / usize::MAX.
222    RegisterService = 205,
223    /// Resolve a well-known `service_id` to its current provider tid. Open to all cells so
224    /// any client can reconnect after a service restart. ABI: a0 = service_id (u16) →
225    /// provider tid (> 0), or 0 if no live provider is currently registered.
226    LookupService = 206,
227    /// Liveness heartbeat: the caller asserts it is alive and will call again within
228    /// `interval_ticks` 10 ms scheduler ticks. If it misses that deadline the kernel
229    /// terminates it as HUNG — catching a silent hang (deadlock / stuck loop) that the
230    /// CPU-monopoly watchdog cannot see — so the supervisor restarts it. `interval_ticks
231    /// = 0` disables the heartbeat. Open to all cells. ABI: a0 = interval_ticks → 0.
232    Heartbeat = 207,
233
234    // === Zero-Copy Grant (Storage 2.0, Phase 01) ===
235    /// Allocate a kernel-managed Grant region of up to 4096 pages (16 MB).
236    /// ABI: a0 = size (rounded up to pages, max 16777216) → base_paddr (> 0) on success, 0 on OOM.
237    /// Requires GrantCap (allowlist bit 39).
238    GrantAlloc = 208,
239    /// Share Grant access with `target_task`.
240    /// ABI: a0 = grant_id, a1 = target_task_id, a2 = GrantPerm (0=RO, 1=WO, 2=RW) → 0 on success.
241    /// In the current SAS build those permission bits are a protocol contract,
242    /// not Tier 2 hardware isolation against a malicious peer.
243    /// Requires GrantCap (bit 39).
244    GrantShare = 209,
245    /// Return the raw pointer to a Grant region the caller has access to.
246    /// ABI: a0 = grant_id, a1 = optional `size_out` pointer → ptr (usize, same
247    /// as grant_id for identity-mapped SAS) on success.
248    ///
249    /// `size_out` may be null. When non-null it must point to one aligned
250    /// `usize` slot fully contained in the caller's own user stack; segment
251    /// memory and arbitrary user mappings are rejected. An invalid `size_out`
252    /// pointer makes the syscall fail with `InvalidInput`.
253    ///
254    /// Returns usize::MAX on permission denied or not found.
255    GrantSlice = 210,
256    /// Release a Grant region: unmaps its pages and frees its frames.
257    /// ABI: a0 = grant_id → 0 on success.
258    GrantFree = 211,
259    /// Start an async sector read into a Grant buffer (poll-based stub; Phase 04 for true async).
260    /// ABI: a0 = sector, a1 = grant_id → async_id (1 = immediately complete) on success.
261    /// Requires BlockIoCap (bit 36) — same gate as raw block-I/O opcodes 500/501.
262    BlkReadAsync = 212,
263    /// Request exclusive MMIO ownership for a peripheral Driver Cell.
264    /// ABI: a0 = base (physical MMIO address), a1 = len → 0 on success.
265    /// Requires the corresponding manifest flag (GPIO or UART) to be set.
266    /// The kernel checks the Resource Registry allowlist; rejects unknown ranges.
267    RequestMmio = 213,
268    /// Fill caller buffer with VirtIO-RNG entropy (true hardware randomness).
269    /// Required for TLS key generation — mtime-seeded PRNG is cryptographically broken.
270    /// ABI: a0 = buf_ptr, a1 = len (max 64 per call, one VirtIO descriptor) → bytes written.
271    /// Returns 0 if no VirtIO-RNG device is present.
272    GetRandom = 214,
273    /// Allocate a persistent, pre-pinned Grant buffer for the cell's lifetime.
274    /// Unlike GrantAlloc, the buffer is not freed until GrantUnregister or cell exit.
275    /// ABI: a0 = size → reg_id (physical base, > 0) on success, 0 on OOM.
276    /// Requires GrantCap (allowlist bit 39).
277    GrantRegister = 215,
278    /// Release a registered buffer allocated via GrantRegister.
279    /// ABI: a0 = reg_id → 0 on success.
280    /// Requires GrantCap (allowlist bit 39).
281    GrantUnregister = 216,
282    /// Block until one or more bits in `mask` fire, or `timeout_ticks` 10ms ticks elapse.
283    /// `timeout_ticks = 0` means block indefinitely.
284    /// Returns the fired event bits (>0) on wake, or 0 on timeout.
285    /// Bit 0 = NET_RX (NIC frame available). See `api::syscall::events`.
286    /// ABI: a0 = mask (u32), a1 = timeout_ticks (u64 lo, a2 = hi) → fired_mask (usize).
287    /// Allowlist-gated (bit 42). Requires `WaitForEvent` in `declare_syscalls!`.
288    WaitForEvent = 217,
289
290    /// Play raw PCM frames on the VirtIO sound device's output stream (blocking).
291    /// PCM format is fixed: signed 16-bit LE, 2 channels, 44100 Hz (frame = 4 bytes).
292    /// The kernel configures + starts the output stream on first use.
293    /// ABI: a0 = pcm_ptr, a1 = pcm_len (bytes) → frames written (0 on no device/error).
294    /// Allowlist-gated (bit 47). Requires `AudioPlay` in `declare_syscalls!`.
295    AudioPlay = 218,
296
297    /// Revoke one or more capabilities from a live running cell (runtime revocation).
298    ///
299    /// The caller must hold `SpawnCap`.  System cells (`block_io_cap` / `network_cap`
300    /// holders) are protected — revoking I/O caps from a running service mid-flight
301    /// would corrupt driver state; use the supervisor freeze/pause/replacement flow instead.
302    ///
303    /// `cap_mask` bit layout (see `cap_mask` module constants):
304    ///   bit 0 → block_io · bit 1 → network · bit 2 → spawn · bit 3 → hypervisor
305    ///   bits  8-15 → mmio_devices sub-mask (matches resource_registry::DEV_*)
306    ///   bits 16-23 → block_regions sub-mask (partition access bits)
307    ///
308    /// ABI: a0 = target_tid, a1 = cap_mask (u32) → 0 on success.
309    /// Returns `PermissionDenied` if caller lacks SpawnCap or target is a system cell.
310    /// Always permitted past the syscall allowlist (SpawnCap is the authority gate).
311    CapRevoke = 219,

313    // === Hypervisor (220-225) — HypervisorCap gated (allowlist bit 44) ===
314    // All six syscalls require `hypervisor = true` in the cell manifest AND the
315    // kernel to have booted at EL2 (ARM64) or with H-ext (RISC-V).  A cell
316    // without HypervisorCap calling any of these receives `PermissionDenied`.
317    /// Allocate guest RAM and a Stage-2 table; returns opaque `vm_id`.
318    /// ABI: a0 = guest_pages → vm_id (>0) on success, 0 on OOM / unsupported.
319    CreateVm = 220,
320    /// Create a vCPU in `vm_id` with initial PC `entry_pc`; returns `vcpu_id`.
321    /// ABI: a0 = vm_id, a1 = entry_pc → vcpu_id (>0).
322    CreateVcpu = 221,
323    /// Map guest IPA range `[ipa, ipa+size)` to kernel-managed frames in `vm_id`.
324    /// ABI: a0 = vm_id, a1 = ipa, a2 = size (bytes), a3 = writable:bool → 0.
325    MapGuestMemory = 222,
326    /// World-switch into vCPU until exit or `budget_ns` ns expire; write `ViVmExit`.
327    /// ABI: a0 = vm_id, a1 = vcpu_id, a2 = budget_ns, a3 = out_ptr (*mut ViVmExit) → 0.
328    RunVcpu = 223,
329    /// Read (write=0) or write (write=1) vCPU registers x0-x30 + sp + pc (32×u64).
330    /// ABI: a0 = vm_id, a1 = vcpu_id, a2 = buf_ptr, a3 = write:bool → 0.
331    VcpuRegs = 224,
332    /// Inject a GICv2 virtual interrupt into vCPU (0 ≤ intid ≤ 1019).
333    /// ABI: a0 = vm_id, a1 = vcpu_id, a2 = intid → 0.
334    InjectIrq = 225,
335    /// Copy `len` bytes from caller's `src_ptr` into guest physical RAM at `gpa`.
336    /// Kernel validates `src_ptr + len` lies within the caller's mapped address space
337    /// and `gpa + len` lies within the VM's carved guest-RAM region.
338    /// ABI: a0 = vm_id, a1 = gpa, a2 = src_ptr, a3 = len → bytes_written.
339    WriteGuestMemory = 226,
340    /// Copy `len` bytes from guest physical RAM at `gpa` into caller's `dst_ptr`.
341    /// Kernel bounds-checks both sides: `dst_ptr + len` within caller's address space

343    /// ABI: a0 = vm_id, a1 = gpa, a2 = dst_ptr, a3 = len → bytes_read.
344    ReadGuestMemory = 227,

409    // === Directory-handle inheritance (appended last; see abi.rs rules) ===
410    /// 240: Name the directory handles the caller's NEXT spawn passes to its child.
411    ///
412    /// ABI: a0 = ptr to [`crate::dir_handles::ViSpawnDirHandles`] (0 clears a
413    /// previously named set), a1..a3 reserved → 0 on success.
414    ///
415    /// Staged on the caller's own task rather than passed as a spawn argument
416    /// because `SpawnFromElf` and `SpawnPinned` already use all four argument
417    /// registers, and a carrier that reached only half the spawn entry points
418    /// would leave the other half silently granting nothing. The set is consumed
419    /// and cleared by the next spawn the caller makes, whether that spawn
420    /// succeeds or fails, so a set can never attach to a later unrelated spawn.
421    ///
422    /// A malformed set (unknown version, over-long count, zero or repeated
423    /// handle) is refused here rather than trimmed — see
424    /// [`crate::dir_handles::DirHandleSetError`].
425    ///
426    /// Privileged: requires lifecycle `SpawnCap`. Exact launch-profile approval
427    /// is evaluated later by the actual spawn syscalls; this carrier syscall is
428    /// the staging authority for the caller's next child launch.
429    /// Always permitted past the syscall allowlist, matching
430    /// `RegisterService`/`CapRevoke`.
431    SpawnSetDirs = 240,
432    /// 241: Read the kernel's record of a cell's inherited directory handles.
433    ///
434    /// ABI: a0 = cell_id, a1 = buf_ptr, a2 = buf_len → bytes written
435    /// ([`crate::dir_attestation::DIR_ATTESTATION_LEN`]).
436    ///
437    /// The kernel states which spawner named the set and what the set was. It
438    /// does NOT state that the spawner held those handles — that is a question
439    /// about the filesystem service's own table, and only the service can answer
440    /// it. A buffer shorter than the record is an error, never a partial write.
441    ///
442    /// Restricted to the registered provider of `service::VFS` and to a cell
443    /// asking about itself. Always permitted past the syscall allowlist; the
444    /// identity check at dispatch is the gate.
445    QueryDirHandles = 241,
446    /// 242: Wait for one asynchronous result on the caller's completion queue.
447    ///
448    /// ABI: a0 = source mask (see [`crate::completion::source`]),
449    /// a1 = timeout_ticks_lo,
450    /// a2 = timeout_ticks_hi, a3 = pointer to a
451    /// [`crate::completion::COMPLETION_LEN`]-byte buffer → 1 when a completion
452    /// was written there, 0 when the wait ended with nothing.
453    ///
454    /// Calling this **is** the submission: it reserves a slot on the caller's
455    /// own queue, from the caller's own context, so the completion the source
456    /// later appends can never fail to find somewhere to land. A queue with no
457    /// free slot refuses here, where the caller can act on it, rather than in
458    /// the interrupt handler that is trying to report a result.
459    ///
460    /// Source-specific deadline semantics are deliberate:
461    /// - `NET_RX`: zero waits indefinitely; a finite deadline that expires
462    ///   releases the reservation and reports 0 with no record.
463    /// - `TIMER`: a finite deadline is required, and expiry writes a completion
464    ///   whose source is `TIMER` and whose result is 0.
465    ///
466    /// Distinct from [`Self::WaitForEvent`], which reports only *that* an event
467    /// bit fired. This reports *which submission* finished and with what result,
468    /// which is what lets one parked thread serve several outstanding
469    /// operations. Both exist; neither replaces the other.
470    WaitCompletion = 242,
471    /// 244: Attest the root task currently owning a Cell principal.
472    ///
473    /// ABI: a0 = cell_id, a1 = generation, a2 = output pointer, a3 = output
474    /// length. The record is available only to the registered VFS provider while
475    /// it is handling that exact principal.
476    ResolveCellOwner = 244,
477    /// 245: Atomically attest and subscribe to a Cell root's death.
478    ///
479    /// Uses the same arguments as `ResolveCellOwner`; returns an opaque non-zero
480    /// watch token and writes the owner record on success.
481    WatchCellOwner = 245,
482    /// 246: Cancel exactly one live Cell-owner watch. ABI: a0 = token.
483    CancelCellOwnerWatch = 246,
484    /// 247: RV32-safe root-owner attestation.
485    ///
486    /// ABI: a0 = [`crate::cell_owner::CellOwnerRequest`] pointer, a1 = request
487    /// length, a2 = owner-record output pointer, a3 = output length. This is an
488    /// additive record ABI; 244 remains unchanged for existing 64-bit callers.
489    ResolveCellOwnerRecord = 247,
490    /// 248: RV32-safe atomic root-owner watch.
491    ///
492    /// Uses the same fixed request and owner response records as 247. Its
493    /// positive token result remains bounded to `isize::MAX`.
494    WatchCellOwnerRecord = 248,
495    /// Begin cache synchronization for an owned Grant subrange.
496    ///
497    /// ABI: a0 = grant_id, a1 = byte offset, a2 = byte length → non-zero
498    /// operation token. The matching [`Self::GrantCacheSyncComplete`] is the
499    /// only operation that can release the pin established here.
500    GrantCacheSyncBegin = 249,
501    /// Complete one exact cache synchronization operation.
502    ///
503    /// ABI: a0 = token → 0. Foreign, stale, and duplicate tokens fail closed.
504    GrantCacheSyncComplete = 250,
505    /// Register the VideoCore-owned scanout allocation for the display cell.
506    ///
507    /// ABI: a0 = physical base, a1 = byte size, a2 = `(width << 16) | height`,
508    /// a3 = pitch. The kernel validates the reserved range before mapping it.
509    RegisterDisplayFramebuffer = 251,
510}

512/// Bit constants for the `cap_mask` argument of `ViSyscall::CapRevoke`.
513///
514/// The 32-bit mask encodes which capability classes to strip from the target cell.
515/// Boolean caps occupy the low 4 bits; parameterized caps use two 8-bit sub-fields.
516pub mod cap_mask {
517    /// Revoke raw block-device access (BlkRead/BlkWrite/BlkFlush).
518    pub const BLOCK_IO: u32 = 1 << 0;
519    /// Revoke network transmit/receive (NetTx/NetRx).
520    pub const NETWORK: u32 = 1 << 1;
521    /// Revoke lifecycle spawn authority (`ForceExit`, `NotifyOnExit`,
522    /// `RegisterService`, snapshot control, and any exact launch
523    /// profile that explicitly requires lifecycle authority).
524    pub const SPAWN: u32 = 1 << 2;
525    /// Revoke RISC-V H-extension / ARM64 EL2 hypervisor access.
526    pub const HYPERVISOR: u32 = 1 << 3;
527
528    /// Bit offset of the `mmio_devices` sub-field within the mask.
529    pub const MMIO_SHIFT: u32 = 8;
530    /// Mask selecting the `mmio_devices` 8-bit sub-field (bits 8-15).
531    /// Each bit within this byte corresponds to a `resource_registry::DEV_*` flag.
532    pub const MMIO_MASK: u32 = 0xFF << MMIO_SHIFT;
533
534    /// Bit offset of the `block_regions` sub-field within the mask.
535    pub const BLKREGION_SHIFT: u32 = 16;
536    /// Mask selecting the `block_regions` 8-bit sub-field (bits 16-23).
537    pub const BLKREGION_MASK: u32 = 0xFF << BLKREGION_SHIFT;
538
539    /// Revoke all capability classes at once.
540    pub const ALL: u32 = BLOCK_IO | NETWORK | SPAWN | HYPERVISOR | MMIO_MASK | BLKREGION_MASK;

543/// Compact bitset of permitted syscalls, stored as a `u64`.
544///
545/// Used to build the allowlist embedded in the ELF `__ViCell_syscalls` section
546/// via [`declare_syscalls!`].  All methods are `const` so the value can be
547/// computed at compile time.
548///
549/// ```ignore
550/// const MY_ALLOWED: SyscallSet = SyscallSet::EMPTY
551///     .with(ViSyscall::Send)
552///     .with(ViSyscall::Recv)
553///     .with(ViSyscall::Log);
554/// ```
555#[derive(Copy, Clone, Debug, PartialEq, Eq)]
556pub struct SyscallSet(pub u64);
557
558impl SyscallSet {
559    /// Empty set — all syscalls denied (not useful on its own; combine with `with`).
560    pub const EMPTY: Self = Self(0);
561    /// Permit-all sentinel — the default for cells without a `__ViCell_syscalls` section.
562    pub const ALL: Self = Self(u64::MAX);
563
564    /// Return a new `SyscallSet` with `syscall` added.
565    ///
566    /// If `syscall` has no allowlist bit (always-permitted syscalls like `Yield`
567    /// or `Exit`) the set is returned unchanged.
568    pub const fn with(self, syscall: ViSyscall) -> Self {
569        match syscall.allowlist_bit() {
570            Some(bit) => Self(self.0 | (1u64 << bit)),
571            None => self,
572        }
573    }
574
575    /// Raw bit-mask value to embed in `__ViCell_syscalls`.
576    pub const fn bits(self) -> u64 {
577        self.0
578    }
579
580    /// Returns `true` if `syscall` is permitted by this set.
581    ///
582    /// Always-permitted syscalls (`Yield`, `Exit`, …) return `true` regardless
583    /// of the stored bits.
584    pub const fn permits(self, syscall: ViSyscall) -> bool {
585        match syscall.allowlist_bit() {
586            Some(bit) => (self.0 >> bit) & 1 == 1,
587            None => true,
588        }
589    }
590}

592/// Embed a syscall allowlist into the current Cell's ELF binary.
593///
594/// Places a `u64` bitset into the `__ViCell_syscalls` ELF section.  The kernel
595/// reads it at spawn time and enforces it on every syscall — any call whose bit
596/// is not set is rejected with `PermissionDenied` before the handler runs.
597///
598/// Cells that do **not** call this macro default to permit-all (backwards
599/// compatible).  Cells that call it are restricted to exactly the listed
600/// syscalls plus always-permitted ones (`Yield`, `Exit`, `ForceExit`, …).
601///
602/// # Usage
603/// ```ignore
604/// api::declare_syscalls![Send, Recv, TryRecv, Log, Heartbeat, LookupService];
605/// ```
606#[macro_export]
607macro_rules! declare_syscalls {
608    ($($syscall:ident),* $(,)?) => {
609        #[used]
610        #[link_section = "__ViCell_syscalls"]
611        pub static VICELL_SYSCALLS: u64 = const {
612            let mut mask: u64 = 0u64;
613            $(
614                if let Some(bit) = $crate::syscall::ViSyscall::$syscall.allowlist_bit() {
615                    mask |= 1u64 << bit;
616                }
617            )*
618            mask
619        };
620    };
621}

625impl ViSyscall {
626    /// Stable bit index for the per-Cell syscall allowlist stored in
627    /// `Task::syscall_allowlist`.
628    ///
629    /// Bit indices are independent of raw opcode values so they remain stable
630    /// even if opcodes are renumbered.  Returns `None` for syscalls that are
631    /// always permitted (Yield, Exit) — filtering them would prevent a Cell
632    /// from ever yielding or cleanly shutting down.
633    ///
634    /// Bit 36 is reserved for raw block-I/O syscalls (opcodes 500-503) which
635    /// bypass `ViSyscall::from()` and must be checked separately at dispatch.
636    pub const fn allowlist_bit(self) -> Option<u8> {
637        match self {
638            Self::Send => Some(0),
639            Self::TrySend => Some(0), // same allowlist bit as Send (subset of Send)
640            Self::Recv => Some(1),
641            Self::TryRecv => Some(2),
642            Self::Reply => Some(3),
643            Self::Call => Some(4),
644            Self::Spawn => Some(5),
645            Self::SpawnFromMem => Some(6),
646            // Exact launch-profile route: path-based spawn authorization lives in
647            // the kernel profile table, but the allowlist still names this family
648            // of outward launch operations separately from lifecycle SpawnCap.
649            Self::SpawnFromPath => Some(7),
650            // SpawnFromElf shares the same outward-launch allowlist bit (7) as
651            // SpawnFromPath: both are exact launch-profile routes for reviewed
652            // path-based child launches.
653            Self::SpawnFromElf => Some(7),
654            // SpawnPinned keeps its own allowlist bit because affinity/pinned
655            // launches are a distinct external surface, even though Phase 04
656            // still routes approval through exact kernel launch profiles.
657            Self::SpawnPinned => Some(8),
658            Self::Wait => Some(9),
659            Self::Log => Some(10),
660            Self::SetTimer => Some(11),
661            Self::ShmAlloc => Some(12),
662            Self::ShmMap => Some(13),
663            Self::GetProcs => Some(14),
664            Self::OpenCap => Some(15),
665            Self::ReadCap | Self::SeekCap | Self::StatCap => Some(16),
666            Self::CloseCap => Some(17),
667            Self::WriteCap | Self::SyncCap => Some(45),
668            Self::TruncateCap => Some(46),
669            Self::Open => Some(18),
670            Self::Read => Some(19),
671            Self::Write => Some(20),
672            Self::Close => Some(21),
673            Self::ReadDir => Some(22),
674            Self::Seek => Some(23),
675            Self::FileOp => Some(24),
676            Self::GetTime => Some(25),
677            Self::GpuFlush => Some(26),
678            Self::GpuCursor => Some(43),
679            Self::GpuGetResolution => Some(44),
680            Self::NetTx => Some(27),
681            Self::NetRx => Some(28),
682            Self::RecvTimeout => Some(29),
683            Self::SendGather => Some(30),
684            Self::RecvScatter => Some(31),
685            // Bit 32 remains the compatibility bucket shared by the live ready
686            // signal and snapshot control after retiring opcode 400.
687            Self::HotSwapReady => Some(HOTSWAP_STATE_TRANSFER_BIT),
688            Self::StateStash => Some(33),
689            Self::StateRestore => Some(34),
690            // StateStashClear: same allowlist bit as StateRestore (both are stash housekeeping).
691            Self::StateStashClear => Some(34),
692            Self::Exec => Some(35),
693            // Snapshot: privileged warm-boot operation; reuses compatibility bit 32
694            // (allowlist-only here; the kernel authority gate requires SupervisorCap).
695            Self::Snapshot => Some(HOTSWAP_STATE_TRANSFER_BIT),
696            // LookupService is an open syscall (any client resolves a service endpoint).
697            Self::LookupService => Some(37),
698            // Heartbeat is an open syscall (any cell asserts its own liveness).
699            Self::Heartbeat => Some(38),
700            // GrantCap (bit 39): cells that need zero-copy large-file I/O via Grant API.
701            Self::GrantAlloc
702            | Self::GrantShare
703            | Self::GrantSlice
704            | Self::GrantFree
705            | Self::GrantRegister
706            | Self::GrantUnregister => Some(39),
707            // BlkReadAsync reuses BlockIoCap (bit 36) — same authority as raw block I/O.
708            Self::BlkReadAsync => Some(36),
709            // RequestMmio: gated by GPIO (bit 40) or UART (bit 41) manifest flags.
710            // The kernel re-checks the manifest and allowlist at dispatch; here we
711            // assign bit 40 as the generic "peripheral MMIO" allowlist bit.
712            Self::RequestMmio => Some(40),
713            // GetRandom: any cell that needs entropy (TLS, crypto) declares this.
714            // Bit 41 was previously allocated as a UART MMIO alias; repurposing for entropy
715            // is safe because UART MMIO is enforced at dispatch by resource-registry check,
716            // not by this bit alone.
717            Self::GetRandom => Some(41),
718            // WaitForEvent: IRQ-driven sleep (net RX waker, Phase 04 SMP).
719            Self::WaitForEvent => Some(42),
720            // WaitCompletion shares the WaitForEvent allowlist bit deliberately:
721            // same event-source authority, different return path. A fresh bit
722            // would deny the call to older cells for no security benefit.
723            // It shares bit 42 deliberately: a
724            // fresh bit would deny the call to every cell whose
725            // `__ViCell_syscalls` section was generated before that bit
726            // existed, and the two calls gate the same thing.
727            Self::WaitCompletion => Some(42),
728            // AudioPlay: VirtIO sound output (bit 47 — next free after TruncateCap 46).
729            Self::AudioPlay => Some(47),
730            // HypervisorCap (bit 44): all 6 VMM syscalls share one bit.
731            // Bit 43 is already assigned to GpuCursor; 44 is the next free slot.
732            Self::CreateVm
733            | Self::CreateVcpu
734            | Self::MapGuestMemory
735            | Self::RunVcpu
736            | Self::VcpuRegs
737            | Self::InjectIrq
738            | Self::WriteGuestMemory
739            | Self::ReadGuestMemory => Some(44),
740            // GrantDma (bit 48): PCIe DMA authorization. Cells that need to drive
741            // hardware DMA (Driver Cells) declare this capability in their manifest.
742            Self::GrantDma => Some(48),
743            // SupervisorOp (bit 49): freeze/resume/kill/query a live Cell.
744            // Gated by SupervisorCap which is ONLY granted via direct kernel TCB write.
745            Self::FreezeCell
746            | Self::ResumeCell
747            | Self::KillCell
748            | Self::QueryHotswapReady
749            | Self::PauseService => Some(49),
750            // DriverRegistration (bit 50): announce as the active block/NIC driver,
751            // and discover PCIe device BARs. All gated by PcieDriverCap.
752            Self::RegisterBlockDriver | Self::RegisterNicDriver | Self::FindPcieDevice => Some(50),
753            // WaitIrq (bit 51): block a Driver Cell until a hardware IRQ fires.
754            // Gated by PcieDriverCap or PlatformCap — only Driver Cells may wait on IRQs.
755            Self::WaitIrq => Some(51),
756            // RegisterPcieBar (bit 52): Platform Cell announces a discovered BAR.
757            // Singleton PlatformCap is the gate — only one trusted Platform Cell may call this.
758            Self::RegisterPcieBar => Some(52),
759            // RegisterPciDevice (bit 53): Platform Cell announces a PCI device with class info.
760            // Populates kernel PCI_DEVICES so find_class() queries work without kernel ECAM scan.
761            Self::RegisterPciDevice => Some(53),
762            // ReadLog (bit 54): read the kernel user-log ring buffer.
763            // Gated so random cells cannot snoop on other cells' stdout.
764            // Declare with `ReadLog` in the cell manifest to enable.
765            Self::ReadLog => Some(54),
766            // GetProcs2 (bit 55): richer task telemetry, opt-in like ReadLog.
767            Self::GetProcs2 => Some(55),
768            // MemInfo (bit 56): global allocator telemetry is a cross-cell side channel.
769            Self::MemInfo => Some(56),
770            // SpawnReplacement (bit 57): supervisor-only replacement spawn via a
771            // kernel-recorded frozen-cell ceiling. This stays separate from bit 49
772            // so older supervisor-op allowlists do not silently gain it.
773            Self::SpawnReplacement => Some(57),
774            // Cache synchronization is one authority split into an exact
775            // begin/complete lifecycle. A caller that cannot complete must
776            // leave the grant quarantined until task death/reboot.
777            Self::GrantCacheSyncBegin | Self::GrantCacheSyncComplete => Some(58),
778            // A firmware framebuffer is distinct from peripheral MMIO and
779            // therefore receives a dedicated, opt-in authority bit.
780            Self::RegisterDisplayFramebuffer => Some(59),
781            // Yield, Exit, and ForceExit are always permitted — a Cell must be able
782            // to yield the CPU, exit cleanly, and force-terminate unresponsive tasks
783            // regardless of its allowlist.  SpawnCap is the authority gate for ForceExit.
784            // NotifyOnExit, RegisterService, and CapRevoke are always permitted past
785            // the allowlist; dispatch applies SpawnCap or a narrower operation-specific
786            // capability gate.
787            // SpawnSetDirs (SpawnCap) and QueryDirHandles (VFS-provider or self)
788            // join them: giving them bits would silently deny every cell whose
789            // `__ViCell_syscalls` section was generated before those bits existed,
790            // and the authority check at dispatch is the real gate either way.
791            Self::Yield
792            | Self::Exit
793            | Self::ForceExit
794            | Self::NotifyOnExit
795            | Self::RegisterService
796            | Self::CapRevoke
797            | Self::SpawnSetDirs
798            | Self::QueryDirHandles
799            | Self::ResolveCellOwner
800            | Self::WatchCellOwner
801            | Self::CancelCellOwnerWatch
802            | Self::ResolveCellOwnerRecord
803            | Self::WatchCellOwnerRecord
804            | Self::Unknown => None,
805        }
806    }
