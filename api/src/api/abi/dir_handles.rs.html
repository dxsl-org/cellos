2//! Directory-handle inheritance across a spawn.
3//!
4//! A directory handle is issued, validated and revoked by the filesystem
5//! service. The kernel carries a set of them from a spawner to the cell it
6//! spawns, and later states on its own authority which spawner supplied that
7//! set. It never interprets a handle: it cannot tell a live one from a revoked
8//! one and holds no opinion about what any of them refer to.
9//!
10//! ## Why the kernel is only a courier
11//! Two records of the same authority drift, and drift in this direction widens
12//! authority silently instead of failing to compile. The filesystem service
13//! stays the only place a handle means anything — it checks the spawner
14//! genuinely held every handle in the set before binding any of them to the
15//! child.
16//!
17//! ## Invariants
18//! - A set is bounded at [`MAX_SPAWN_DIR_HANDLES`] by the carrier's own shape,
19//!   so no caller-supplied count ever sizes an allocation.
20//! - Validation is all-or-nothing ([`DirHandleSet::from_carrier`]). A set that
21//!   is silently narrowed is indistinguishable from the set the spawner meant to
22//!   pass, so an over-long or malformed set fails the spawn instead.
23//! - `0` is not a handle. It is what a zeroed struct carries, and accepting it
24//!   would make "no set" and "a set of one unknown thing" the same value.
25//! - A non-empty set always names the spawner it came from. A set with no
26//!   attested source is worth nothing to the service that must check it.
27
28/// Largest directory-handle set a spawn may carry.
29///
30/// The bound is structural: the carrier holds its handles inline, so a caller
31/// cannot drive a kernel-side allocation with a count it chooses.
32pub const MAX_SPAWN_DIR_HANDLES: usize = 8;
33
34/// Layout version of [`ViSpawnDirHandles`]. A mismatch is rejected rather than
35/// interpreted, so a future field can never be read as a handle.
36pub const SPAWN_DIR_HANDLES_VERSION: u32 = 1;
37
38/// A filesystem-service-issued directory handle, opaque to the kernel.
39///
40/// Deliberately not `CapId`: a kernel capability and a service-issued handle
41/// share a representation and nothing else, and passing one where the other
42/// belongs must be a compile error on a boundary whose purpose is that authority
43/// cannot be mistaken.
44///
45/// The serde impls carry that distinction onto the wire as well, at no cost in
46/// bytes: postcard encodes a newtype struct as its inner value, so the encoding
47/// is identical to a bare `u64` while `VfsRequest` still refuses a `CapId`.
48#[repr(transparent)]
49#[derive(
50    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize, serde::Deserialize,
51)]
52pub struct ViDirHandle(pub u64);
53
54/// Caller-supplied carrier naming the directory handles a spawn should pass on.
55#[repr(C)]
56#[derive(Debug, Clone, Copy)]
57pub struct ViSpawnDirHandles {
58    /// Must equal [`SPAWN_DIR_HANDLES_VERSION`].
59    pub version: u32,
60    /// Number of leading entries of `handles` that are in use.
61    pub count: u32,
62    /// Handles to pass on; entries at or past `count` are ignored.
63    pub handles: [u64; MAX_SPAWN_DIR_HANDLES],
64}
65
66impl ViSpawnDirHandles {
67    /// A carrier naming no handles — what a spawner that inherits nothing sends.
68    pub const EMPTY: Self = Self {
69        version: SPAWN_DIR_HANDLES_VERSION,
70        count: 0,
71        handles: [0; MAX_SPAWN_DIR_HANDLES],
72    };
73
74    /// Build a carrier from the handles a spawner wants its child to receive.
75    ///
76    /// # Errors
77    /// [`DirHandleSetError::TooMany`] when more than [`MAX_SPAWN_DIR_HANDLES`]
78    /// handles are supplied. Truncating instead would hand the child a narrower
79    /// set than the spawner asked for without either side being told.
80    pub fn new(handles: &[ViDirHandle]) -> Result<Self, DirHandleSetError> {
81        if handles.len() > MAX_SPAWN_DIR_HANDLES {
82            return Err(DirHandleSetError::TooMany);
83        }
84        let mut out = Self::EMPTY;
85        out.count = handles.len() as u32;
86        for (slot, h) in out.handles.iter_mut().zip(handles) {
87            *slot = h.0;
88        }
89        Ok(out)
90    }
91}
92
93/// Why a carrier was refused. Every variant fails the spawn outright.
94#[derive(Debug, Clone, Copy, PartialEq, Eq)]
95pub enum DirHandleSetError {
96    /// `version` is not [`SPAWN_DIR_HANDLES_VERSION`].
97    UnsupportedVersion,
98    /// More handles than [`MAX_SPAWN_DIR_HANDLES`].
99    TooMany,
100    /// A handle was `0`, which is the absent value and never a handle.
101    ZeroHandle,
102    /// The same handle appears twice, so the set does not say what it looks like
103    /// it says about how many distinct directories the child receives.
104    Duplicate,
105}
106
107/// A validated directory-handle set, as recorded against a task.
108#[derive(Debug, Clone, Copy, PartialEq, Eq)]
109pub struct DirHandleSet {
110    count: u8,
111    handles: [u64; MAX_SPAWN_DIR_HANDLES],
112}
113
114impl DirHandleSet {
115    /// The set a cell spawned without a carrier receives.
116    pub const EMPTY: Self = Self {
117        count: 0,
118        handles: [0; MAX_SPAWN_DIR_HANDLES],
119    };
120
121    /// Validate a caller-supplied carrier.
122    ///
123    /// # Errors
124    /// See [`DirHandleSetError`]. Nothing is accepted partially: a carrier with
125    /// one bad entry yields no set at all.
126    pub fn from_carrier(carrier: &ViSpawnDirHandles) -> Result<Self, DirHandleSetError> {
127        if carrier.version != SPAWN_DIR_HANDLES_VERSION {
128            return Err(DirHandleSetError::UnsupportedVersion);
129        }
130        let count = carrier.count as usize;
131        if count > MAX_SPAWN_DIR_HANDLES {
132            return Err(DirHandleSetError::TooMany);
133        }
134        let mut out = Self::EMPTY;
135        for i in 0..count {
136            let h = carrier.handles[i];
137            if h == 0 {
138                return Err(DirHandleSetError::ZeroHandle);
139            }
140            if out.handles[..i].contains(&h) {
141                return Err(DirHandleSetError::Duplicate);
142            }
143            out.handles[i] = h;
144        }
145        out.count = count as u8;
146        Ok(out)
147    }
148
149    /// The handles in the set, in the order the spawner named them.
150    pub fn as_slice(&self) -> &[u64] {
151        &self.handles[..self.count as usize]
152    }
153
154    pub fn len(&self) -> usize {
155        self.count as usize
156    }
157
158    pub fn is_empty(&self) -> bool {
159        self.count == 0
160    }
161}
162
163impl Default for DirHandleSet {
164    fn default() -> Self {
165        Self::EMPTY
166    }
167}

169/// What a task received from the cell that spawned it.
170///
171/// The spawner is recorded alongside the set because a set with no named source
172/// cannot be checked: the filesystem service's question is "did *this* cell hold
173/// *these* handles", and it has no way to ask it about a set that arrived
174/// anonymously.
175#[derive(Debug, Clone, Copy, PartialEq, Eq)]
176pub struct InheritedDirHandles {
177    /// Cell that named the set. `0` when nothing was inherited.
178    pub spawner_cell_id: u64,
179    /// The spawner's generation at the moment of the spawn, so a set cannot be
180    /// credited to a later cell that reused the id.
181    pub spawner_generation: u64,
182    /// The handles named. Empty whenever `spawner_cell_id` is `0`.
183    pub set: DirHandleSet,
184}
185
186impl InheritedDirHandles {
187    /// A task spawned without a carrier, and every task created before boot
188    /// reaches userspace.
189    pub const NONE: Self = Self {
190        spawner_cell_id: 0,
191        spawner_generation: 0,
192        set: DirHandleSet::EMPTY,
193    };
194}
195
196impl Default for InheritedDirHandles {
197    fn default() -> Self {
198        Self::NONE
199    }
