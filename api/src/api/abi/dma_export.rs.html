2//! Client-memory DMA export and mapping records (`ExportDmaRange = 437`,
3//! `MapDmaExport = 438`, `RevokeDmaExport = 439`); ADR-0033, CP1 §5.
4//!
5//! [`ViDmaExportV1`] is a class K record (`docs/abi-boundary-records.md`):
6//! syscall 438 with an exact out length of [`DMA_EXPORT_RECORD_LEN`] (48 bytes),
7//! version exactly 1, little-endian in declared field order, every reserved byte
8//! and unknown flag bit zero. Producer: the kernel. Consumer: the Driver Cell
9//! that holds the lease and owns the BDF.
10//!
11//! Every refusal of 437-439 returns one of the `ERR_*` sentinels below; a
12//! successful 437 returns an export id in `1..=isize::MAX`, which never
13//! collides with them.
14
15use types::ViError;
16
17/// The only record version 438 writes and a v1 consumer accepts.
18pub const DMA_EXPORT_VERSION: u16 = 1;
19/// Exact `out_len` syscall 438 accepts and bytes it writes.
20pub const DMA_EXPORT_RECORD_LEN: usize = 48;
21
22/// Maximum bytes a single export may cover: 4 GiB (2^32).
23pub const EXPORT_MAX_BYTES: u64 = 1 << 32;
24
25/// `flags` bit: device is granted write access to the exported memory.
26/// If clear, the export is device-read-only (`R`-only VT-d leaves).
27pub const EXPORT_DEVICE_WRITE: u16 = 1 << 0;
28
29/// All known flag bits in version 1.
30pub const EXPORT_FLAGS_KNOWN: u16 = EXPORT_DEVICE_WRITE;
31
32/// Holder not eligible, range not owned, caller not the lease holder, or
33/// 438 without allowlist bit 48 / `PcieDriverCap` / BDF ownership.
34pub const ERR_PERMISSION_DENIED: usize = usize::MAX;
35/// 439: the window invalidation was not acknowledged; the lease stays
36/// `Revoking` with the owner range locked and the kernel retries.
37pub const ERR_TRY_AGAIN: usize = usize::MAX - 1;
38/// No live provider (437), or an unknown, revoked or owner-dead export (438),
39/// or a caller that is neither owner nor holder (439, no existence oracle).
40pub const ERR_FILE_NOT_FOUND: usize = usize::MAX - 2;
41/// Malformed arguments: alignment, length, flags, service or BDF bits,
42/// `out_len != 48`, non-zero reserved registers.
43pub const ERR_INVALID_INPUT: usize = usize::MAX - 3;
44/// Reserved for the record family; 438 refuses a short buffer as invalid input.
45pub const ERR_BUFFER_TOO_SMALL: usize = usize::MAX - 4;
46/// No translating remapper for the export or the BDF: exports never fall back
47/// to identity DMA.
48pub const ERR_NOT_SUPPORTED: usize = usize::MAX - 5;
49/// Lease table, IOVA window space, page tables or the holder's DMA quota.
50pub const ERR_OUT_OF_MEMORY: usize = usize::MAX - 6;
51/// 438: the window was published but its invalidation was not acknowledged;
52/// only revoke or death cleans it.
53pub const ERR_UNKNOWN: usize = usize::MAX - 7;
54
55/// Export record returned by `MapDmaExport` (syscall 438).
56#[repr(C)]
57#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
58pub struct ViDmaExportV1 {
59    /// Exactly [`DMA_EXPORT_VERSION`].
60    pub version: u16,
61    /// Echo of the export flags ([`EXPORT_DEVICE_WRITE`]).
62    pub flags: u16,
63    /// Zero.
64    pub reserved0: u32,
65    /// Echo of the export id.
66    pub export_id: u64,
67    /// The exporting Cell. The driver must match it against the caller identity
68    /// of the request that carried the id (confused-deputy check).
69    pub owner_cell_id: u64,
70    /// Generation of the exporting Cell.
71    pub owner_generation: u64,
72    /// Device-visible window base in the holder's domain, 4 KiB aligned.
73    pub iova: u64,
74    /// Bytes covered, echo of the export length.
75    pub len: u64,
76}
77
78impl ViDmaExportV1 {
79    /// A version-1 record with zero reserved bytes.
80    pub const fn new(
81        flags: u16,
82        export_id: u64,
83        owner_cell_id: u64,
84        owner_generation: u64,
85        iova: u64,
86        len: u64,
87    ) -> Self {
88        Self {
89            version: DMA_EXPORT_VERSION,
90            flags,
91            reserved0: 0,
92            export_id,
93            owner_cell_id,
94            owner_generation,
95            iova,
96            len,
97        }
98    }
99
100    /// Serialize in declared field order, little-endian.
101    pub fn encode(&self, out: &mut [u8; DMA_EXPORT_RECORD_LEN]) {
102        out[0..2].copy_from_slice(&self.version.to_le_bytes());
103        out[2..4].copy_from_slice(&self.flags.to_le_bytes());
104        out[4..8].copy_from_slice(&self.reserved0.to_le_bytes());
105        out[8..16].copy_from_slice(&self.export_id.to_le_bytes());
106        out[16..24].copy_from_slice(&self.owner_cell_id.to_le_bytes());
107        out[24..32].copy_from_slice(&self.owner_generation.to_le_bytes());
108        out[32..40].copy_from_slice(&self.iova.to_le_bytes());
109        out[40..48].copy_from_slice(&self.len.to_le_bytes());
110    }
111
112    /// Decode exactly [`DMA_EXPORT_RECORD_LEN`] bytes, refusing any other
113    /// length, a version other than 1, unknown flag bits or non-zero reserved
114    /// bytes with [`ViError::InvalidInput`].
115    pub fn decode(buf: &[u8]) -> Result<Self, ViError> {
116        let buf: &[u8; DMA_EXPORT_RECORD_LEN] =
117            buf.try_into().map_err(|_| ViError::InvalidInput)?;
118        let u64_at = |at: usize| {
119            let mut word = [0u8; 8];
120            word.copy_from_slice(&buf[at..at + 8]);
121            u64::from_le_bytes(word)
122        };
123        let version = u16::from_le_bytes([buf[0], buf[1]]);
124        let flags = u16::from_le_bytes([buf[2], buf[3]]);
125        let reserved0 = u32::from_le_bytes([buf[4], buf[5], buf[6], buf[7]]);
126        if version != DMA_EXPORT_VERSION || flags & !EXPORT_FLAGS_KNOWN != 0 || reserved0 != 0 {
127            return Err(ViError::InvalidInput);
128        }
129        Ok(Self {
130            version,
131            flags,
132            reserved0,
133            export_id: u64_at(8),
134            owner_cell_id: u64_at(16),
135            owner_generation: u64_at(24),
136            iova: u64_at(32),
137            len: u64_at(40),
138        })
139    }
140}
