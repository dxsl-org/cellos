2//! Kernel-attested caller identity for service IPC.
3//!
4//! A service cannot derive who called it. `sys_recv` returns a **tid**, and a tid
5//! is not a cell: a thread carries its own tid while inheriting its parent cell's
6//! `CellId`, so `CellId(tid)` misattributes every thread. Anything the request
7//! payload claims about the caller is chosen by the caller and therefore worth
8//! nothing to an authorization decision.
9//!
10//! So the kernel states it. A receiver opts in by passing
11//! [`RECV_ATTEST_CALLER`] as the fourth argument of `ViSyscall::Recv`; the kernel
12//! then writes a [`CallerIdentity`] trailer into the **last
13//! [`CALLER_IDENTITY_LEN`] bytes** of that receiver's own recv buffer, after the
14//! message payload has been copied.
15//!
16//! ## Why the tail of the buffer, and not a new message field
17//! Postcard byte 0 is the request discriminant; a trailer at a fixed offset
18//! past the payload is invisible to existing parsers on both kernel and cell sides.
19//! The trailer reports sender diagnostics; VFS obtains root lifetime endpoints
20//! separately via `ResolveCellOwner` or `WatchCellOwner`.
21//! ## Invariants
22//! - Opting in **reserves** the tail: a receiver that passes the flag must treat
23//!   `buf[buf_len - CALLER_IDENTITY_LEN ..]` as kernel-owned and must not expect
24//!   payload bytes there. Cell recv buffers are `IPC_BUF_SIZE` (4096) and the
25//!   largest message in the system is under 1 KiB, so the reserved tail costs no
26//!   usable payload today.
27//! - The kernel writes the trailer **after** copying the payload, so a sender
28//!   that pads its message out to the full buffer cannot pre-place a forged
29//!   trailer: its bytes there are overwritten.
30//! - Absent or garbage tail → [`CallerIdentity::from_trailer`] returns
31//!   `None`, which the receiver MUST treat as "unknown caller" and deny. There is
32//!   no permissive interpretation of a missing trailer.
33//! - A `cell_id` of 0 is never a cell (the kernel uses it for its own
34//!   allocations), so it is rejected at parse time rather than deep in a policy
35//!   table.
36
37/// Fourth `ViSyscall::Recv` argument requesting a caller-identity trailer.
38///
39/// `0` (what every pre-existing caller passes) means "no trailer" — the kernel
40/// then leaves the receiver's buffer exactly as before, so opting in is the only
41/// way to change any receiver's observable behaviour.
42pub const RECV_ATTEST_CALLER: usize = 1;
43
44/// Caller identity flag indicating the calling Cell has declared VfsMutate authority.
45pub const CALLER_FLAG_VFS_MUTATE: u32 = 1 << 0;
46
47/// Size in bytes of the identity trailer reserved at the end of a recv buffer.
48pub const CALLER_IDENTITY_LEN: usize = 32;
49/// Tag distinguishing a kernel-written trailer from leftover payload bytes.
50///
51/// Not a security control — a cell that can write another cell's recv buffer in a
52/// single address space can write anything, including the service's policy table.
53/// It exists so a receiver that opted in can tell "kernel wrote identity here"
54/// from "previous, longer message left bytes here".
55const MAGIC: u32 = 0x5649_4349; // "VICI"
56
57/// Who the kernel says called.
58///
59/// Construct only from a kernel-written trailer ([`CallerIdentity::from_trailer`])
60/// or, inside the kernel, from live scheduler state — never from request bytes.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub struct CallerIdentity {
63    /// Flag word unpacked from trailer bytes 4..8 (e.g. [`CALLER_FLAG_VFS_MUTATE`]).
64    pub flags: u32,
65    /// Owning cell of the calling task. A thread reports its parent cell here.
66    pub cell_id: u64,
67    /// Monotonic epoch assigned when the cell was created, inherited by its threads.
68    pub generation: u64,
69    /// Task id that actually sent the message. Diagnostics only.
70    pub sender_tid: u64,
71}
72
73impl CallerIdentity {
74    /// Serialize into the fixed-size trailer the kernel writes.
75    pub fn to_trailer(&self) -> [u8; CALLER_IDENTITY_LEN] {
76        let mut out = [0u8; CALLER_IDENTITY_LEN];
77        out[0..4].copy_from_slice(&MAGIC.to_le_bytes());
78        out[4..8].copy_from_slice(&self.flags.to_le_bytes());
79        out[8..16].copy_from_slice(&self.cell_id.to_le_bytes());
80        out[16..24].copy_from_slice(&self.generation.to_le_bytes());
81        out[24..32].copy_from_slice(&self.sender_tid.to_le_bytes());
82        out
83    }
84
85    /// Parse the trailer out of a recv buffer that opted in with
86    /// [`RECV_ATTEST_CALLER`].
87    ///
88    /// Returns `None` when the buffer is too small to hold a trailer, when the
89    /// tail is not a kernel-written trailer, or when the attested `cell_id` is 0.
90    /// Every one of those is "identity unknown", and the only correct response to
91    /// an unknown identity is to deny the request.
92    pub fn from_recv_buf(buf: &[u8]) -> Option<Self> {
93        let start = buf.len().checked_sub(CALLER_IDENTITY_LEN)?;
94        Self::from_trailer(&buf[start..])
95    }
96
97    /// Parse a trailer from exactly the trailer bytes.
98    ///
99    /// Split out from [`Self::from_recv_buf`] so the kernel and host tests can
100    /// round-trip the encoding without constructing a whole recv buffer.
101    pub fn from_trailer(bytes: &[u8]) -> Option<Self> {
102        if bytes.len() < CALLER_IDENTITY_LEN {
103            return None;
104        }
105        let word = |at: usize| -> u64 {
106            let mut b = [0u8; 8];
107            b.copy_from_slice(&bytes[at..at + 8]);
108            u64::from_le_bytes(b)
109        };
110        let mut tag = [0u8; 4];
111        tag.copy_from_slice(&bytes[0..4]);
112        if u32::from_le_bytes(tag) != MAGIC {
113            return None;
114        }
115        let mut flag_bytes = [0u8; 4];
116        flag_bytes.copy_from_slice(&bytes[4..8]);
117        let flags = u32::from_le_bytes(flag_bytes);
118        let cell_id = word(8);
119        if cell_id == 0 {
120            return None; // kernel/unattributable — never a cell
121        }
122        Some(Self {
123            flags,
124            cell_id,
125            generation: word(16),
126            sender_tid: word(24),
127        })
128    }
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134
135    const SAMPLE: CallerIdentity = CallerIdentity {
136        flags: 0,
137        cell_id: 7,
138        generation: 42,
139        sender_tid: 9,
140    };
141
142    #[test]
143    fn trailer_round_trips() {
144        assert_eq!(
145            CallerIdentity::from_trailer(&SAMPLE.to_trailer()),
146            Some(SAMPLE)
147        );
148    }
149    #[test]
150    fn flags_round_trip() {
151        let mut sample = SAMPLE;
152        sample.flags = CALLER_FLAG_VFS_MUTATE;
153        assert_eq!(
154            CallerIdentity::from_trailer(&sample.to_trailer()),
155            Some(sample)
156        );
157    }
158
159    #[test]
160    fn untagged_tail_is_not_an_identity() {
161        // A recv buffer whose tail still holds bytes of a previous message.
162        assert_eq!(
163            CallerIdentity::from_trailer(&[0xABu8; CALLER_IDENTITY_LEN]),
164            None
165        );
166        assert_eq!(
167            CallerIdentity::from_trailer(&[0u8; CALLER_IDENTITY_LEN]),
168            None
169        );
170    }
171
172    #[test]
173    fn cell_zero_is_rejected() {
174        let mut trailer = SAMPLE.to_trailer();
175        trailer[8..16].copy_from_slice(&0u64.to_le_bytes());
176        assert_eq!(CallerIdentity::from_trailer(&trailer), None);
177    }
178
179    #[test]
180    fn short_buffer_yields_no_identity() {
181        assert_eq!(CallerIdentity::from_trailer(&[0u8; 8]), None);
182        assert_eq!(CallerIdentity::from_recv_buf(&[0u8; 4]), None);
183    }
184
185    #[test]
186    fn identity_is_read_from_the_tail_not_the_head() {
187        let mut buf = [0u8; 512];
188        let start = buf.len() - CALLER_IDENTITY_LEN;
189        buf[start..].copy_from_slice(&SAMPLE.to_trailer());
190        // A payload that happens to contain the tag at the head must not be
191        // mistaken for the trailer.
192        buf[0..CALLER_IDENTITY_LEN].copy_from_slice(&SAMPLE.to_trailer());
193        assert_eq!(CallerIdentity::from_recv_buf(&buf), Some(SAMPLE));
194    }
195}
