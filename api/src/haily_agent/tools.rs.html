1//! Sandboxed Tool Routing and Execution for Haily Agent.
2
3extern crate alloc;
4
5use alloc::format;
6use alloc::string::String;
7use agent_proto::{AgentToolRequest, AgentToolResponse};
8use ostd::syscall::{sys_recv, sys_send, SyscallResult};
9
10/// Registry of tool cell TIDs.
11#[derive(Debug, Default, Clone, Copy)]
12pub struct ToolRouter {
13    pub fs_cell: usize,
14    pub sys_cell: usize,
15    pub spawn_cell: usize,
16}
17
18impl ToolRouter {
19    /// Route a tool name to the responsible tool cell TID.
20    pub fn route(&self, tool_name: &str) -> usize {
21        match tool_name {
22            "read_file" | "write_file" | "list_dir" => self.fs_cell,
23            "list_cells" | "sys_info" | "lookup_service" => self.sys_cell,
24            "spawn_cell" | "kill_cell" => self.spawn_cell,
25            _ => 0,
26        }
27    }
28
29    /// Check if this tool performs mutating side effects requiring approval.
30    pub fn requires_approval(tool_name: &str) -> bool {
31        matches!(tool_name, "write_file" | "spawn_cell" | "kill_cell")
32    }
33
34    /// Execute a tool invocation via IPC to the target tool cell.
35    pub fn execute(&self, name: &str, args_json: &str) -> Result<String, String> {
36        let cell_tid = self.route(name);
37        if cell_tid == 0 {
38            return Err(format!("tool '{}' unavailable (cell not bound)", name));
39        }
40
41        let req = AgentToolRequest::Invoke {
42            name,
43            args_json,
44        };
45        let mut out = [0u8; 4096];
46        let encoded = postcard::to_slice(&req, &mut out)
47            .map_err(|_| String::from("tool request payload exceeded IPC buffer"))?;
48
49        match sys_send(cell_tid, encoded) {
50            SyscallResult::Ok(_) => {}
51            _ => return Err(format!("IPC send to tool cell '{}' failed", name)),
52        }
53
54        let mut buf = [0u8; 4096];
55        loop {
56            match sys_recv(0, &mut buf) {
57                SyscallResult::Ok(sender) if sender == cell_tid => {
58                    return match postcard::take_from_bytes::<AgentToolResponse>(&buf) {
59                        Ok((AgentToolResponse::Ok { result_json }, _)) => Ok(result_json),
60                        Ok((AgentToolResponse::Err { message }, _)) => Err(message),
61                        Err(_) => Err(String::from("corrupted AgentToolResponse")),
62                    };
63                }
64                SyscallResult::Ok(_) => continue,
65                _ => return Err(format!("no reply received from tool cell '{}'", name)),
66            }
67        }
68    }
69}
