1//! One renderer service loop for the trusted and copied-only domain entries.
2
3use crate::Clock;
4use alloc::vec::Vec;
5use ostd::syscall::{sys_exit, sys_recv, SyscallResult};
6use render_core::{
7    render_tile, Output, RenderError, Request, Response, Scene, COPIED_TILE_MAX_BYTES,
8};
9
10/// The domain entry instantiates only `run::<false>`, eliminating its grant path
11/// at monomorphization as well as withholding the grant syscall in its manifest.
12pub fn run<const ALLOW_SHARED: bool>() -> ! {
13    let clock = match Clock::new() {
14        Ok(clock) => clock,
15        Err(error) => {
16            ostd::io::println(&alloc::format!("c2c-render-worker: {}", error));
17            sys_exit(1);
18        }
19    };
20    let scene = Scene::demo();
21    let scene_hash = scene.fingerprint();
22    let mut owner = None;
23    let mut copied = Vec::with_capacity(COPIED_TILE_MAX_BYTES);
24    let mut incoming = [0u8; api::ipc::IPC_BUF_SIZE];
25    let mut outgoing = [0u8; api::ipc::IPC_BUF_SIZE];
26    loop {
27        incoming.fill(0);
28        let sender = match sys_recv(owner.unwrap_or(0), &mut incoming) {
29            SyscallResult::Ok(sender) if sender > 0 => sender,
30            result => {
31                ostd::io::println(&alloc::format!(
32                    "c2c-render-worker: receive failed: {:?}",
33                    result
34                ));
35                sys_exit(1);
36            }
37        };
38        // Only kernel-owned operations are served. A raw mailbox message cannot
39        // acquire a grant write window or become a renderer reply.
40        let Some(operation) = ostd::ipc::current() else {
41            continue;
42        };
43        if owner.is_some_and(|expected| expected != sender) {
44            continue;
45        }
46        let request = postcard::take_from_bytes::<Request>(&incoming)
47            .ok()
48            .filter(|(_, tail)| tail.iter().all(|byte| *byte == 0))
49            .map(|(request, _)| request);
50        let mut stopping = false;
51        let response = match request {
52            Some(Request::Ready) if owner.is_none() || owner == Some(sender) => {
53                owner = Some(sender);
54                Response::Ready { scene_hash }
55            }
56            Some(Request::Stop) if owner == Some(sender) => {
57                stopping = true;
58                Response::Stopped
59            }
60            Some(Request::Render {
61                config,
62                tile,
63                scene_hash: requested,
64                output,
65            }) if owner == Some(sender) => {
66                // Refuse all Shared outputs at the domain boundary, before any
67                // handle conversion, grant access, validation or rendering.
68                if !ALLOW_SHARED && matches!(output, Output::Shared { .. }) {
69                    Response::Error(RenderError::InvalidOutput)
70                } else if requested != scene_hash {
71                    Response::Error(RenderError::SceneMismatch)
72                } else {
73                    match tile.validated(config).and_then(|tile| {
74                        let bytes = tile.rgb_bytes()?;
75                        let (stats, compute_ns, pixels, processor_id_before, processor_id_after) =
76                            match output {
77                            Output::Copied => {
78                                if bytes > COPIED_TILE_MAX_BYTES {
79                                    return Err(RenderError::InvalidOutput);
80                                }
81                                copied.resize(bytes, 0);
82                                let processor_id_before = ostd::system_info::processor_id();
83                                let started = clock.now_ns();
84                                let stats = render_tile(&scene, config, tile, &mut copied)?;
85                                let compute_ns = clock.elapsed_ns(started);
86                                let processor_id_after = ostd::system_info::processor_id();
87                                (
88                                    stats,
89                                    compute_ns,
90                                    core::mem::take(&mut copied),
91                                    processor_id_before,
92                                    processor_id_after,
93                                )
94                            }
95                            Output::Shared {
96                                grant_id,
97                                bytes: declared,
98                            } => {
99                                // A compile-time boundary also excludes this
100                                // branch from the copied-only instantiation.
101                                if !ALLOW_SHARED {
102                                    return Err(RenderError::InvalidOutput);
103                                }
104                                if declared as usize != bytes {
105                                    return Err(RenderError::OutputLength);
106                                }
107                                let grant_id = usize::try_from(grant_id)
108                                    .map_err(|_| RenderError::InvalidOutput)?;
109                                // The coordinator is blocked on this exact operation;
110                                // the closure ends before reply and retains no mapping.
111                                let (stats, duration, processor_id_before, processor_id_after) =
112                                    ostd::grant::with_shared_bytes_mut(grant_id, |buffer| {
113                                        let target = buffer
114                                            .get_mut(..bytes)
115                                            .ok_or(RenderError::OutputLength)?;
116                                        let processor_id_before = ostd::system_info::processor_id();
117                                        let started = clock.now_ns();
118                                        let stats = render_tile(&scene, config, tile, target)?;
119                                        let duration = clock.elapsed_ns(started);
120                                        let processor_id_after = ostd::system_info::processor_id();
121                                        Ok::<_, RenderError>((
122                                            stats,
123                                            duration,
124                                            processor_id_before,
125                                            processor_id_after,
126                                        ))
127                                    })
128                                    .ok_or(RenderError::InvalidOutput)??;
129                                (
130                                    stats,
131                                    duration,
132                                    Vec::new(),
133                                    processor_id_before,
134                                    processor_id_after,
135                                )
136                            }
137                        };
138                        Ok(Response::Done {
139                            tile_id: tile.id,
140                            scene_hash,
141                            stats,
142                            compute_ns,
143                            pixels,
144                            processor_id_before,
145                            processor_id_after,
146                        })
147                    }) {
148                        Ok(response) => response,
149                        Err(error) => Response::Error(error),
150                    }
151                }
152            }
153            _ => Response::Error(RenderError::InvalidOutput),
154        };
155        let sent = api::ipc::encode(&response, &mut outgoing)
156            .map_err(|_| ostd::ipc::IpcError::Encode)
157            .and_then(|encoded| ostd::ipc::reply(operation, encoded));
158        if let Response::Done { pixels, .. } = response {
159            if !pixels.is_empty() {
160                copied = pixels;
161            }
162        }
163        if let Err(error) = sent {
164            // No further writes after an expired/dead caller operation.
165            ostd::io::println(&alloc::format!(
166                "c2c-render-worker: reply failed: {:?}",
167                error
168            ));
169            sys_exit(1);
170        }
171        if stopping {
172            sys_exit(0);
173        }
174    }
175}
