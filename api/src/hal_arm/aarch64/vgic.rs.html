1//! GICv2 Virtual CPU Interface Control (GICH) MMIO register access for EL2 VMM.
2//!
3//! The GICH base comes from the board (`ArmResources::VGIC`); boards without a
4//! GICv2 virtualization interface never reach this module. Functions that touch
5//! GICH are `unsafe` — the caller must be at EL2 with GICH MMIO accessible. The
6//! list-register planning ([`plan_entry_lrs`], [`sort_exit_lrs`]) and the timer
7//! line ([`vtimer_line_asserted`]) are pure, so they compile and are tested on
8//! the host.
9//!
10//! # GICH_LR encoding for a Group 0 Pending virtual IRQ
11//! `[30]=Group1=0, [29:28]=Pending=0b01, [27:23]=Priority=0, [9:0]=VIntID`
12//! → value = `0x1000_0000 | intid`
13//!
14//! Group 0, because a GICv2 guest enables its CPU interface with `GICC_CTLR = 1`:
15//! on the virtual interface (`GICV_CTLR`) bit 0 is EnableGrp0, so a Group 1 list
16//! register is never signalled to a Linux guest. KVM loads GICv2 guests' LRs as
17//! Group 0 for the same reason.
18//!
19//! # GICV Stage-2 passthrough
20//! The caller maps the guest's GICC IPA onto the board's GICV window read-write
21//! in the Stage-2 table so guest GICC reads/writes hit real GICV hardware
22//! without causing a VM exit. The guest must be able to write it: its
23//! `GICC_CTLR`/`GICC_PMR` enable the interface and `GICC_EOIR` completes each
24//! interrupt. Those writes only reach the running vCPU's state in GICH
25//! (`GICH_VMCR`, `GICH_APR`, the LRs), which [`CpuIfState`] saves on every exit
26//! and restores on every entry, so no other vCPU sees them.
27
28const GICH_HCR: usize = 0x000;
29const GICH_VTR: usize = 0x004;
30const GICH_VMCR: usize = 0x008;
31const GICH_ELRSR0: usize = 0x030;
32const GICH_APR: usize = 0x0F0;
33const GICH_LR: usize = 0x100; // LR[n] at +n*4
34
35const LR_PENDING: u32 = 1 << 28;
36const LR_ACTIVE: u32 = 1 << 29;
37const LR_VINTID: u32 = 0x3FF;
38/// GIC-400 GICH_VMCR reset value: interface disabled, VPMR=0, VBPR0=2, VBPR1=3.
39const VMCR_RESET: u32 = 0x004C_0000;
40
41/// Maximum list registers to use; QEMU GICv2 has 4.
42pub const MAX_LRS: usize = 4;
43
44#[inline]
45unsafe fn rd32(addr: usize) -> u32 {
46    // SAFETY: caller guarantees EL2 + valid MMIO address.
47    core::ptr::read_volatile(addr as *const u32)
48}
49
50#[inline]
51unsafe fn wr32(addr: usize, val: u32) {
52    // SAFETY: caller guarantees EL2 + valid MMIO address.
53    core::ptr::write_volatile(addr as *mut u32, val);
54}

56/// Enable GICH (set GICH_HCR.EN = 1).
57///
58/// # Safety
59/// Caller must be at EL2; the GICH window at `gich` must be mapped.
60pub unsafe fn enable(gich: usize) {
61    let cur = rd32(gich + GICH_HCR);
62    wr32(gich + GICH_HCR, cur | 1);
63}
64
65/// Return the number of list registers (GICH_VTR[4:0] + 1), capped at MAX_LRS.
66///
67/// # Safety
68/// Caller must be at EL2; the GICH window at `gich` must be mapped.
69pub unsafe fn lr_count(gich: usize) -> usize {
70    let vtr = rd32(gich + GICH_VTR);
71    (((vtr & 0x1F) + 1) as usize).min(MAX_LRS)
72}
73
74/// The guest's virtual timer line: `CNTV_CTL` ENABLE=1 and IMASK=0 (ISTATUS is
75/// ignored: it is a snapshot from the last exit), and the guest's virtual count
76/// `now` has reached `CNTV_CVAL` (unsigned compare, as the architecture
77/// defines the timer condition).
78pub fn vtimer_line_asserted(ctl: u64, cval: u64, now: u64) -> bool {
79    ctl & 0b11 == 0b01 && now >= cval
80}

82/// Plan the list registers for one guest entry, using at most `n_lrs`.
83///
84/// The `active` LRs saved at the last exit come first, verbatim. Then pending
85/// INTIDs from `next_pending` fill the free slots as Group 0 Pending; one that
86/// matches an LR already planned sets that LR's Pending bit instead, because
87/// two LRs must never carry the same virtual INTID. Pending INTIDs are pulled
88/// only while a slot is free, so the rest stay in the caller's pending set.
89/// Returns the LR values and how many are in use.
90pub fn plan_entry_lrs(
91    active: &[u32],
92    n_lrs: usize,
93    mut next_pending: impl FnMut() -> Option<u32>,
94) -> ([u32; MAX_LRS], usize) {
95    let n_lrs = n_lrs.min(MAX_LRS);
96    let mut lrs = [0u32; MAX_LRS];
97    let mut used = active.len().min(n_lrs);
98    lrs[..used].copy_from_slice(&active[..used]);
99    while used < n_lrs {
100        let Some(intid) = next_pending() else {
101            break;
102        };
103        let intid = intid & LR_VINTID;
104        if let Some(lr) = lrs[..used].iter_mut().find(|lr| **lr & LR_VINTID == intid) {
105            *lr |= LR_PENDING;
106        } else {
107            lrs[used] = LR_PENDING | intid;
108            used += 1;
109        }
110    }
111    (lrs, used)
112}

114/// Sort the LRs read back after an exit.
115///
116/// Empty LRs (`elrsr` bit set) are skipped. Active and Pending+Active LRs are
117/// kept verbatim for the next entry, because the guest is still handling them.
118/// Pending-only LRs go back through `repend`, except for level-sensitive
119/// INTIDs (`is_level`): the caller re-samples those lines at entry, so a line
120/// the guest has since deasserted is not delivered again. For the same reason
121/// a kept Active level LR loses its Pending bit. Returns the kept Active LRs
122/// and their count.
123pub fn sort_exit_lrs(
124    lrs: &[u32],
125    elrsr: u32,
126    is_level: impl Fn(u32) -> bool,
127    mut repend: impl FnMut(u32),
128) -> ([u32; MAX_LRS], usize) {
129    let mut active = [0u32; MAX_LRS];
130    let mut n_active = 0;
131    for (n, &lr) in lrs.iter().enumerate().take(MAX_LRS) {
132        if (elrsr >> n) & 1 != 0 {
133            continue;
134        }
135        let intid = lr & LR_VINTID;
136        if lr & LR_ACTIVE != 0 {
137            active[n_active] = if is_level(intid) {
138                lr & !LR_PENDING
139            } else {
140                lr
141            };
142            n_active += 1;
143        } else if lr & LR_PENDING != 0 && !is_level(intid) {
144            repend(intid);
145        }
146    }
147    (active, n_active)
148}

150/// One vCPU's virtual CPU interface state. It lives in GICH only while that
151/// vCPU runs: [`CpuIfState::load`] installs it before entry and
152/// [`CpuIfState::save`] takes it back after exit.
153#[derive(Clone, Copy)]
154pub struct CpuIfState {
155    /// Guest `GICV_CTLR`/`PMR`/`BPR` view (GICH_VMCR).
156    vmcr: u32,
157    /// Guest active priorities (GICH_APR).
158    apr: u32,
159    /// LRs that still held an Active (or Pending+Active) interrupt at exit.
160    /// Reloading them as plain Pending would deliver an interrupt the guest is
161    /// still handling a second time.
162    active: [u32; MAX_LRS],
163    n_active: usize,
164    /// Implemented LRs (GICH_VTR), read at the first `load`; 0 until then.
165    n_lrs: usize,
166}
167
168impl Default for CpuIfState {
169    fn default() -> Self {
170        Self::new()
171    }
172}
173
174impl CpuIfState {
175    /// Interface state of a vCPU that has not run yet.
176    pub const fn new() -> Self {
177        Self {
178            vmcr: VMCR_RESET,
179            apr: 0,
180            active: [0; MAX_LRS],
181            n_active: 0,
182            n_lrs: 0,
183        }
184    }
185
186    /// Install this vCPU's interface state in the GICH window at `gich`, with
187    /// LRs planned by [`plan_entry_lrs`]; every other implemented LR is
188    /// cleared. Returns the number of LRs in use, which the matching
189    /// [`CpuIfState::save`] needs.
190    ///
191    /// Precondition: INTIDs from `next_pending` are ≤ 1019 (validated by the
192    /// syscall layer, m3).
193    ///
194    /// # Safety
195    /// Caller must be at EL2 with the GICH window at `gich` mapped and no vCPU
196    /// running.
197    pub unsafe fn load(&mut self, gich: usize, next_pending: impl FnMut() -> Option<u32>) -> usize {
198        if self.n_lrs == 0 {
199            self.n_lrs = lr_count(gich);
200        }
201        let (lrs, used) = plan_entry_lrs(&self.active[..self.n_active], self.n_lrs, next_pending);
202        self.n_active = 0;
203        wr32(gich + GICH_VMCR, self.vmcr);
204        wr32(gich + GICH_APR, self.apr);
205        for (n, lr) in lrs.iter().enumerate().take(self.n_lrs) {
206            wr32(gich + GICH_LR + n * 4, if n < used { *lr } else { 0 });
207        }
208        used
209    }
210
211    /// Take this vCPU's interface state back from the GICH window at `gich`
212    /// after exit and clear the `used` LRs; the LRs are sorted by
213    /// [`sort_exit_lrs`] with `is_level` and `repend`.
214    ///
215    /// # Safety
216    /// Caller must be at EL2 with the GICH window at `gich` mapped and no vCPU
217    /// running; `used` is the value the matching `load` returned.
218    pub unsafe fn save(
219        &mut self,
220        gich: usize,
221        used: usize,
222        is_level: impl Fn(u32) -> bool,
223        repend: impl FnMut(u32),
224    ) {
225        let used = used.min(self.n_lrs);
226        let elrsr = rd32(gich + GICH_ELRSR0);
227        let mut lrs = [0u32; MAX_LRS];
228        for (n, lr) in lrs.iter_mut().enumerate().take(used) {
229            *lr = rd32(gich + GICH_LR + n * 4);
230            wr32(gich + GICH_LR + n * 4, 0);
231        }
232        (self.active, self.n_active) = sort_exit_lrs(&lrs[..used], elrsr, is_level, repend);
233        self.vmcr = rd32(gich + GICH_VMCR);
234        self.apr = rd32(gich + GICH_APR);
235    }
236}
