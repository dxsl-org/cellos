1//! ARM64 Stage-2 EL2 system-register helpers. Pi's EL1 host invokes the
2//! retained EL2 monitor via HVC; QEMU virt's EL2 host writes registers directly.
3//! A page-table update must be visible before any TLBI or guest entry.
4//!
5//! # Register layout
6//!
7//! ## VTCR_EL2 (40-bit IPA, 4 KB granule, SL0=1 start at Level 1)
8//! | Field | Bits | Value |
9//! |-------|------|-------|
10//! | T0SZ  | [5:0] | 24 (IPA = 2^40 = 1 TiB) |
11//! | SL0   | [7:6] | 0b01 (start at L1) |
12//! | IRGN0 | [9:8] | 0b01 (Normal WB-WA inner) |
13//! | ORGN0 | [11:10] | 0b01 (Normal WB-WA outer) |
14//! | SH0   | [13:12] | 0b11 (Inner-shareable) |
15//! | PS    | [18:16] | 0b010 (40-bit PA size) |
16//! | VS    | [19] | 0 (8-bit VMIDs) |
17//! | RES1  | [31] | 1 |
18//!
19//! Computed: `0x80023558`
20//!
21//! ## VTTBR_EL2
22//! bits[63:56] = RES0 (`VTCR_EL2.VS` = 0)
23//! bits[55:48] = VMID (≥ 1; host is VMID 0; see [`VMID_COUNT`])
24//! bits[47:1]  = BADDR (root PA[47:13] for 8 KB-aligned root)
25//!
26//! ## HCR_EL2.VM (bit 0)
27//! Setting this bit enables Stage-2 translation for EL0/EL1.
28//! Must be cleared before tearing down the Stage-2 table.
29
30/// VTCR_EL2 value for 40-bit IPA, 4 KB granule, Level-1 start (SL0=1),
31/// Normal-WB-WA Inner+Outer-shareable, bit[31]=RES1.
32///
33/// T0SZ=24 | SL0=1 | IRGN0=01 | ORGN0=01 | SH0=11 | PS=010 | VS=0 | RES1
34#[cfg(target_arch = "aarch64")]
35const VTCR_VALUE: u64 = 0x8002_3558;
36
37/// `VTCR_EL2.VS`: 16-bit VMIDs. [`VTCR_VALUE`] leaves it clear, which is what
38/// [`VMID_COUNT`] assumes (not every core implements 16-bit VMIDs).
39#[cfg(target_arch = "aarch64")]
40const VTCR_VS: u64 = 1 << 19;
41#[cfg(target_arch = "aarch64")]
42const _: () = assert!(VTCR_VALUE & VTCR_VS == 0);
43
44/// VMIDs the Stage-2 configuration tells apart. With `VTCR_EL2.VS` clear a
45/// VMID is `VTTBR_EL2[55:48]`: VMIDs 256 apart share TLB tags (256 aliases the
46/// host's VMID 0), and a VMID above 255 sets RES0 bits of `VTTBR_EL2`. VMID 0
47/// is the host's, so guests get `1..VMID_COUNT`. The kernel's tag pool hands
48/// them out, and reissues one only after every TLB entry tagged with it was
49/// invalidated on every CPU and no CPU still names it in `VTTBR_EL2`.
50pub const VMID_COUNT: usize = 1 << 8;
51

55/// Arm Stage-2 translation for the given `vmid` and root page-table at `root_pa`.
56///
57/// Programs VTCR_EL2 and VTTBR_EL2 and issues a full Stage-2 TLB invalidation.
58/// It does NOT set HCR_EL2.VM: the host's Cells keep running at EL0 after this
59/// returns, and with VM=1 their EL1&0 stage-1 walks are themselves Stage-2
60/// translated through the guest map — the first post-syscall instruction fetch
61/// then walks garbage and the Cell dies on a wild PC (observed as an
62/// instruction abort at return-address | 0x06000000 on QEMU 8.2). VM is owned
63/// exclusively by the world-switch: `vcpu_enter_guest` sets it on guest entry
64/// and `vt_vcpu_trap` clears it on every guest exit (vcpu.rs "CRITICAL" note).
65///
66/// # Safety
67/// - Must be called with an active EL2 host, or the Pi EL1 monitor initialized.
68/// - `root_pa` must be 8 KB-aligned (concatenated 2 × L1 tables, VTCR.SL0=1).
69/// - The Stage-2 page table must be fully populated; the entry `dsb ishst`
70///   publishes WB descriptor stores to the Inner-shareable walkers.
71/// - `vmid` must be ≥ 1 (VMID 0 is reserved for the EL2 host) and below
72///   [`VMID_COUNT`].
73/// - This function must NOT be called while a vCPU is running (VTTBR races).
74#[cfg(target_arch = "aarch64")]
75pub unsafe fn monitor_enable_stage2(vmid: u16, root_pa: u64) {
76    debug_assert_eq!(root_pa % (2 * 4096), 0, "S2 root must be 8 KB-aligned");
77    debug_assert!(vmid >= 1, "VMID 0 is reserved for the EL2 host");
78    debug_assert!(
79        usize::from(vmid) < VMID_COUNT,
80        "VMID {vmid} is wider than 8 bits"
81    );
82
83    let vttbr = ((vmid as u64) << 48) | root_pa;
84
85    // SAFETY: caller guarantees EL2 context and well-formed root PA.
86    unsafe {
87        core::arch::asm!(
88            // Stage-2 descriptors were populated by EL1 (Pi) or EL2 (virt).
89            // Complete their WB stores before programming the walk and TLBI.
90            "dsb ishst",
91            // 1. Program VTCR_EL2 (S2 translation parameters).
92            "msr vtcr_el2,  {vtcr}",
93            // 2. Program VTTBR_EL2 (VMID + S2 root PA).
94            "msr vttbr_el2, {vttbr}",
95            "isb",
96            // 3. Invalidate all Stage-1 + Stage-2 TLB entries for this VMID.
97            //    HCR_EL2.VM stays 0 — vcpu_enter_guest sets it per guest entry.
98            "tlbi vmalls12e1is",
99            "dsb ish",
100            "isb",
101            vtcr  = in(reg) VTCR_VALUE,
102            vttbr = in(reg) vttbr,
103            options(nostack),
104        );
105    }
106}

108/// Disable Stage-2 translation (clear HCR_EL2.VM=0), flush the S2 TLB for the
109/// VMID in this CPU's `VTTBR_EL2`, then clear `VTTBR_EL2` so this CPU names
110/// no Stage-2 root.
111///
112/// Call during VM teardown before freeing the Stage-2 page table frames.
113///
114/// # Safety
115/// Must be called from EL2 with no vCPU currently running.
116#[cfg(target_arch = "aarch64")]
117pub unsafe fn monitor_disable_stage2() {
118    // SAFETY: EL2 context; clearing HCR_EL2.VM disables guest translation.
119    unsafe {
120        core::arch::asm!(
121            "mrs {tmp}, hcr_el2",
122            "bic {tmp}, {tmp}, #1",   // clear VM bit
123            "msr hcr_el2, {tmp}",
124            "isb",
125            "tlbi vmalls12e1is",
126            "dsb ish",
127            "msr vttbr_el2, xzr",
128            "isb",
129            tmp = out(reg) _,
130            options(nomem, nostack),
131        );
132    }
133}

135/// Point this CPU's Stage-2 walk at `vmid`'s root for the guest entry that
136/// follows: `VTCR_EL2` and `VTTBR_EL2` only. `VTTBR_EL2` is per CPU, and a vCPU
137/// may enter on any CPU, so every entry programs it; without this a vCPU ran
138/// against whichever root that CPU last named. No TLB invalidation: entries
139/// are tagged by VMID, and the VMID's stale entries were invalidated when its
140/// VM was created ([`monitor_enable_stage2`]).
141///
142/// # Safety
143/// Must be called from EL2 with no vCPU running on this CPU, under the same
144/// conditions as [`monitor_enable_stage2`].
145#[cfg(target_arch = "aarch64")]
146pub unsafe fn monitor_load_stage2(vmid: u16, root_pa: u64) {
147    debug_assert_eq!(root_pa % (2 * 4096), 0, "S2 root must be 8 KB-aligned");
148    debug_assert!(vmid >= 1, "VMID 0 is reserved for the EL2 host");
149    debug_assert!(
150        usize::from(vmid) < VMID_COUNT,
151        "VMID {vmid} is wider than 8 bits"
152    );
153    let vttbr = ((vmid as u64) << 48) | root_pa;
154    // SAFETY: caller guarantees EL2 context and a live, well-formed root.
155    unsafe {
156        core::arch::asm!(
157            "dsb ishst",
158            "msr vtcr_el2,  {vtcr}",
159            "msr vttbr_el2, {vttbr}",
160            "isb",
161            vtcr  = in(reg) VTCR_VALUE,
162            vttbr = in(reg) vttbr,
163            options(nostack),
164        );
165    }
166}

168/// Clear this CPU's `VTTBR_EL2`, so it names no Stage-2 root once its guest
169/// has exited: a root freed later is then named by no CPU.
170///
171/// # Safety
172/// Must be called from EL2 with no vCPU running on this CPU (`HCR_EL2.VM`
173/// clear).
174#[cfg(target_arch = "aarch64")]
175pub unsafe fn monitor_clear_stage2_root() {
176    // SAFETY: EL2 context; VTTBR_EL2 is unused while HCR_EL2.VM is clear.
177    unsafe {
178        core::arch::asm!("msr vttbr_el2, xzr", "isb", options(nomem, nostack));
179    }
180}

182/// Invalidate all Stage-1 + Stage-2 TLB entries for all VMIDs (broadcast).
183///
184/// Use after bulk Stage-2 table changes or during VM teardown. `TLBI ALLE1IS`
185/// covers every VMID of the EL1&0 regime; `TLBI VMALLS12E1IS` would cover only
186/// the VMID in this CPU's `VTTBR_EL2`, which outside a guest run names no VM.
187///
188/// # Safety
189/// Must be called from EL2.
190#[cfg(target_arch = "aarch64")]
191pub unsafe fn monitor_s2_tlb_flush_all() {
192    // SAFETY: EL2 broadcast TLB invalidation; no memory mutation.
193    unsafe {
194        core::arch::asm!(
195            "dsb ishst",
196            "tlbi alle1is",
197            "dsb ish",
198            "isb",
199            options(nostack),
200        );
201    }
202}

204/// This CPU's `VTTBR_EL2` (VMID and Stage-2 root); 0 when it names no root.
205///
206/// # Safety
207/// Must be called from EL2 (an EL2 host; the Pi kernel runs at EL1).
208#[cfg(target_arch = "aarch64")]
209pub unsafe fn monitor_read_vttbr() -> u64 {
210    let vttbr: u64;
211    // SAFETY: EL2 context; reading VTTBR_EL2 has no side effects.
212    unsafe {
213        core::arch::asm!("mrs {v}, vttbr_el2", v = out(reg) vttbr, options(nomem, nostack));
214    }
215    vttbr
216}

218/// Bind an S2 root through the private Pi HVC gateway or directly at an EL2 host.
219///
220/// # Safety
221/// Must be called with an active EL2 host, or after the Pi EL1 monitor
222/// initialized. `root_pa` must be 8 KB-aligned, `vmid` must be ≥ 1 (VMID 0 is
223/// reserved for the EL2 host) and below [`VMID_COUNT`], the Stage-2 page table
224/// must be fully populated and flushed to RAM, and no vCPU may be running
225/// (VTTBR race).
226#[cfg(target_arch = "aarch64")]
227pub unsafe fn enable_stage2(vmid: u16, root_pa: u64) {
228    if El2Monitor::PRESENT {
229        assert!(El2Monitor::is_ready(), "EL2 monitor unavailable");
230        assert_eq!(El2Monitor::gateway(1, vmid as u64, root_pa, 0, 0), 0);
231    } else {
232        unsafe {
233            monitor_enable_stage2(vmid, root_pa);
234        }
235    }
236}

238/// Program this CPU's `VTTBR_EL2` for one guest entry on an EL2 host
239/// ([`monitor_load_stage2`]). The Pi monitor does the same inside its run op,
240/// so this is for EL2 hosts only.
241///
242/// # Safety
243/// As [`monitor_load_stage2`]; never called through the Pi gateway.
244#[cfg(target_arch = "aarch64")]
245pub unsafe fn load_stage2(vmid: u16, root_pa: u64) {
246    debug_assert!(
247        !El2Monitor::PRESENT,
248        "the Pi monitor loads VTTBR_EL2 itself"
249    );
250    unsafe {
251        monitor_load_stage2(vmid, root_pa);
252    }
253}

255/// Clear this CPU's `VTTBR_EL2` through the private Pi HVC gateway or
256/// directly at an EL2 host ([`monitor_clear_stage2_root`]).
257///
258/// # Safety
259/// Must be called with an active EL2 host, or after the Pi EL1 monitor
260/// initialized, and with no vCPU running on this CPU.
261#[cfg(target_arch = "aarch64")]
262pub unsafe fn clear_stage2_root() {
263    if El2Monitor::PRESENT {
264        assert!(El2Monitor::is_ready(), "EL2 monitor unavailable");
265        assert_eq!(El2Monitor::gateway(7, 0, 0, 0, 0), 0);
266    } else {
267        unsafe {
268            monitor_clear_stage2_root();
269        }
270    }
271}

273/// Unbind the S2 root through the private Pi HVC gateway or directly at an EL2 host.
274///
275/// # Safety
276/// Must be called with an active EL2 host, or after the Pi EL1 monitor
277/// initialized, and with no vCPU running (clearing `HCR_EL2.VM` under a live
278/// guest would drop its translation).
279#[cfg(target_arch = "aarch64")]
280pub unsafe fn disable_stage2() {
281    if El2Monitor::PRESENT {
282        assert!(El2Monitor::is_ready(), "EL2 monitor unavailable");
283        assert_eq!(El2Monitor::gateway(2, 0, 0, 0, 0), 0);
284    } else {
285        unsafe {
286            monitor_disable_stage2();
287        }
288    }
289}

291/// Invalidate every Stage-2 entry through the private Pi HVC gateway or
292/// directly at an EL2 host.
293///
294/// # Safety
295/// Must be called with an active EL2 host, or after the Pi EL1 monitor
296/// initialized.
297#[cfg(target_arch = "aarch64")]
298pub unsafe fn s2_tlb_flush_all() {
299    if El2Monitor::PRESENT {
300        assert!(El2Monitor::is_ready(), "EL2 monitor unavailable");
301        assert_eq!(El2Monitor::gateway(3, 0, 0, 0, 0), 0);
302    } else {
303        unsafe {
304            monitor_s2_tlb_flush_all();
305        }
306    }
