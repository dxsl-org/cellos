1//! ARM64 Stage-2 EL2 system-register helpers. Pi's EL1 host invokes the
2//! retained EL2 monitor via HVC; QEMU virt's EL2 host writes registers directly.
3//! A page-table update must be visible before any TLBI or guest entry.
4//!
5//! # Register layout
6//!
7//! ## VTCR_EL2 (40-bit IPA, 4 KB granule, SL0=1 start at Level 1)
8//! | Field | Bits | Value |
9//! |-------|------|-------|
10//! | T0SZ  | [5:0] | 24 (IPA = 2^40 = 1 TiB) |
11//! | SL0   | [7:6] | 0b01 (start at L1) |
12//! | IRGN0 | [9:8] | 0b01 (Normal WB-WA inner) |
13//! | ORGN0 | [11:10] | 0b01 (Normal WB-WA outer) |
14//! | SH0   | [13:12] | 0b11 (Inner-shareable) |
15//! | PS    | [18:16] | 0b010 (40-bit PA size) |
16//! | RES1  | [31] | 1 |
17//!
18//! Computed: `0x80023558`
19//!
20//! ## VTTBR_EL2
21//! bits[63:48] = VMID (≥ 1; host is VMID 0)
22//! bits[47:1]  = BADDR (root PA[47:13] for 8 KB-aligned root)
23//!
24//! ## HCR_EL2.VM (bit 0)
25//! Setting this bit enables Stage-2 translation for EL0/EL1.
26//! Must be cleared before tearing down the Stage-2 table.
27
28/// VTCR_EL2 value for 40-bit IPA, 4 KB granule, Level-1 start (SL0=1),
29/// Normal-WB-WA Inner+Outer-shareable, bit[31]=RES1.
30///
31/// T0SZ=24 | SL0=1 | IRGN0=01 | ORGN0=01 | SH0=11 | PS=010 | RES1
32#[cfg(target_arch = "aarch64")]
33const VTCR_VALUE: u64 = 0x8002_3558;
34
35/// Arm Stage-2 translation for the given `vmid` and root page-table at `root_pa`.
36///
37/// Programs VTCR_EL2 and VTTBR_EL2 and issues a full Stage-2 TLB invalidation.
38/// It does NOT set HCR_EL2.VM: the host's Cells keep running at EL0 after this
39/// returns, and with VM=1 their EL1&0 stage-1 walks are themselves Stage-2
40/// translated through the guest map — the first post-syscall instruction fetch
41/// then walks garbage and the Cell dies on a wild PC (observed as an
42/// instruction abort at return-address | 0x06000000 on QEMU 8.2). VM is owned
43/// exclusively by the world-switch: `vcpu_enter_guest` sets it on guest entry
44/// and `vt_vcpu_trap` clears it on every guest exit (vcpu.rs "CRITICAL" note).
45///
46/// # Safety
47/// - Must be called with an active EL2 host, or the Pi EL1 monitor initialized.
48/// - `root_pa` must be 8 KB-aligned (concatenated 2 × L1 tables, VTCR.SL0=1).
49/// - The Stage-2 page table must be fully populated and all writes flushed to
50///   RAM before this function is called.
51/// - `vmid` must be ≥ 1 (VMID 0 is reserved for the EL2 host).
52/// - This function must NOT be called while a vCPU is running (VTTBR races).
53#[cfg(target_arch = "aarch64")]
54pub unsafe fn monitor_enable_stage2(vmid: u16, root_pa: u64) {
55    debug_assert_eq!(root_pa % (2 * 4096), 0, "S2 root must be 8 KB-aligned");
56    debug_assert!(vmid >= 1, "VMID 0 is reserved for the EL2 host");
57
58    let vttbr = ((vmid as u64) << 48) | root_pa;
59
60    // SAFETY: caller guarantees EL2 context and well-formed root PA.
61    unsafe {
62        core::arch::asm!(
63            // 1. Program VTCR_EL2 (S2 translation parameters).
64            "msr vtcr_el2,  {vtcr}",
65            // 2. Program VTTBR_EL2 (VMID + S2 root PA).
66            "msr vttbr_el2, {vttbr}",
67            "isb",
68            // 3. Invalidate all Stage-1 + Stage-2 TLB entries for this VMID.
69            //    HCR_EL2.VM stays 0 — vcpu_enter_guest sets it per guest entry.
70            "tlbi vmalls12e1is",
71            "dsb ish",
72            "isb",
73            vtcr  = in(reg) VTCR_VALUE,
74            vttbr = in(reg) vttbr,
75            options(nomem, nostack),
76        );
77    }
78}

80/// Disable Stage-2 translation (clear HCR_EL2.VM=0) and flush the S2 TLB.
81///
82/// Call during VM teardown before freeing the Stage-2 page table frames.
83///
84/// # Safety
85/// Must be called from EL2 with no vCPU currently running.
86#[cfg(target_arch = "aarch64")]
87pub unsafe fn monitor_disable_stage2() {
88    // SAFETY: EL2 context; clearing HCR_EL2.VM disables guest translation.
89    unsafe {
90        core::arch::asm!(
91            "mrs {tmp}, hcr_el2",
92            "bic {tmp}, {tmp}, #1",   // clear VM bit
93            "msr hcr_el2, {tmp}",
94            "isb",
95            "tlbi vmalls12e1is",
96            "dsb ish",
97            "isb",
98            tmp = out(reg) _,
99            options(nomem, nostack),
100        );
101    }
102}

104/// Invalidate all Stage-1 + Stage-2 TLB entries for all VMIDs (broadcast).
105///
106/// Use after bulk Stage-2 table changes or during VM teardown.
107///
108/// # Safety
109/// Must be called from EL2.
110#[cfg(target_arch = "aarch64")]
111pub unsafe fn monitor_s2_tlb_flush_all() {
112    // SAFETY: EL2 broadcast TLB invalidation; no memory mutation.
113    unsafe {
114        core::arch::asm!(
115            "tlbi vmalls12e1is",
116            "dsb ish",
117            "isb",
118            options(nomem, nostack),
119        );
120    }
121}

123/// Invalidate the Stage-2 TLB entry for a single guest IPA.
124///
125/// Encodes `ipa >> 12` into the `TLBI IPAS2E1IS` operand as required by the
126/// ARM architecture (D8-7204: Xt = IPA[47:12]).  Also issues `TLBI VMALLE1IS`
127/// to flush any Stage-1 entries cached during the S2 walk.
128///
129/// # Safety
130/// Must be called from EL2 after updating the Stage-2 descriptor for `ipa`.
131#[cfg(target_arch = "aarch64")]
132pub unsafe fn monitor_s2_tlb_flush_ipa(ipa: u64) {
133    let encoded = ipa >> 12; // TLBI IPAS2E1IS encodes IPA[47:12] in Xt
134                             // SAFETY: EL2 TLB maintenance; encoded IPA operand is correct per ARM DDI.
135    unsafe {
136        core::arch::asm!(
137            "tlbi ipas2e1is, {x}",  // invalidate Stage-2 entry for this IPA
138            "dsb ish",
139            "tlbi vmalle1is",       // flush Stage-1 entries cached during S2 walk
140            "dsb ish",
141            "isb",
142            x = in(reg) encoded,
143            options(nomem, nostack),
144        );
145    }
146}

148/// Bind an S2 root through the private Pi HVC gateway or directly at an EL2 host.
149///
150/// # Safety
151/// Must be called with an active EL2 host, or after the Pi EL1 monitor
152/// initialized. `root_pa` must be 8 KB-aligned, `vmid` must be ≥ 1 (VMID 0 is
153/// reserved for the EL2 host), the Stage-2 page table must be fully populated
154/// and flushed to RAM, and no vCPU may be running (VTTBR race).
155#[cfg(target_arch = "aarch64")]
156pub unsafe fn enable_stage2(vmid: u16, root_pa: u64) {
157    #[cfg(feature = "board-rpi3")]
158    {
159        assert!(super::monitor::is_ready(), "EL2 monitor unavailable");
160        assert_eq!(super::monitor::gateway(1, vmid as u64, root_pa, 0, 0), 0);
161    }
162    #[cfg(not(feature = "board-rpi3"))]
163    unsafe {
164        monitor_enable_stage2(vmid, root_pa);
165    }
166}

168/// Unbind the S2 root through the private Pi HVC gateway or directly at an EL2 host.
169///
170/// # Safety
171/// Must be called with an active EL2 host, or after the Pi EL1 monitor
172/// initialized, and with no vCPU running (clearing `HCR_EL2.VM` under a live
173/// guest would drop its translation).
174#[cfg(target_arch = "aarch64")]
175pub unsafe fn disable_stage2() {
176    #[cfg(feature = "board-rpi3")]
177    {
178        assert!(super::monitor::is_ready(), "EL2 monitor unavailable");
179        assert_eq!(super::monitor::gateway(2, 0, 0, 0, 0), 0);
180    }
181    #[cfg(not(feature = "board-rpi3"))]
182    unsafe {
183        monitor_disable_stage2();
184    }
185}
