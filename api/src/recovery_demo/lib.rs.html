1//! Phase 09B software provider crash/restore witness (CP1 option B,
2//! `.agents/261030-1945-ginux-lessons-hardening/phase-09b-cp1.md`).
3//!
4//! Three Cells: `recovery-authority` owns the committed state, the writer
5//! epoch and the request-binding ledger and is the lifecycle parent of
6//! `recovery-worker`, which serves clients and commits every mutation at the
7//! authority before replying; `recovery-client` drives the evidence scenarios.
8//! All state is RAM-only and survives a worker crash only.
9

11#![forbid(unsafe_code)]

23/// Cell-private fixture service ids (same pattern as init's 0x7A01 fixture);
24/// no `libs/api` constant. Init registers the authority from autoload config.
25pub const AUTHORITY_SERVICE: u16 = 0x7A0A;
26/// Published by the authority only after a replacement restored and validated.
27pub const WORKER_SERVICE: u16 = 0x7A0B;
28pub const WORKER_PATH: &str = "/bin/recovery-worker";

30/// Worker exit code after a refused or invalid restore: never restarted.

43/// `Crash.mode` scenarios executed by a fault-inject worker.
44pub mod crash {
45    /// A helper thread spins on another hart while the root thread faults.
46    pub const SPIN: u8 = 1;
47    /// A helper faults while the root thread waits in a VFS read whose lease
48    /// the `recovery-fixture` VFS withholds (missing borrower acknowledgement).
49    pub const HOLD_LEASE: u8 = 2;
50    /// Allocate and map an untracked SHM frame, then fault.
51    pub const SHM_EXPORT: u8 = 3;
52    /// Set the lock word in the authority's writable grant, then fault.
53    pub const LOCK_AND_DIE: u8 = 4;
54}
55
