1//! Fixed 64-byte frames exchanged by the recovery-demo client, worker and
2//! authority. Same-build, provider-private records (not kernel ABI): byte 0 is
3//! the tag, byte 1 the version (exactly 1), byte 2 a per-tag code, byte 3 is
4//! reserved zero, bytes 4..8 a little-endian `u32`, then seven `u64` words.
5//! Decoding refuses a short buffer, another version, a non-zero reserved byte
6//! and an unknown tag before anything acts on the frame.
7
8pub const FRAME_LEN: usize = 64;
9const VERSION: u8 = 1;
10
11/// Request binding committed atomically with its mutation (CP1 §4.2).
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub struct Binding {
14    pub caller_cell: u64,
15    pub caller_generation: u64,
16    pub request_id: u64,
17    pub method: u8,
18    pub rights: u32,
19    /// Canonical payload: the operand itself, so no digest is needed.
20    pub payload: u64,
21}
22
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum Refusal {
25    Conflict = 1,
26    StaleEpoch = 2,
27    RevisionMismatch = 3,
28    RecoveryFailed = 4,
29    QuotaExceeded = 5,
30    Denied = 6,
31    InvalidMutation = 7,
32}
33
34/// Client-requested crash point inside the worker (fault-inject builds act on it).
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36pub enum Fault {
37    None = 0,
38    BeforeCommit = 1,
39    AfterCommit = 2,
40    /// Stage the mutation, wait for `Go`, then commit (delayed old writer).
41    Stall = 3,
42}

44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum Msg {
46    // worker → authority
47    Attach {
48        nonce: u64,
49    },
50    Restored {
51        epoch: u64,
52        revision: u64,
53    },
54    RestoreFailed {
55        epoch: u64,
56    },
57    Commit {
58        epoch: u64,
59        expected_revision: u64,
60        binding: Binding,
61        new_state: u64,
62    },
63    Lookup {
64        epoch: u64,
65        binding: Binding,
66    },
67    // authority → worker
68    Attached {
69        epoch: u64,
70        revision: u64,
71        schema: u32,
72        counter: u64,
73    },
74    Committed {
75        revision: u64,
76        result: u64,
77    },
78    Present {
79        revision: u64,
80        result: u64,
81    },
82    Absent,
83    Refused(Refusal),
84    // client → authority
85    Report,
86    Fence,
87    Corrupt {
88        kind: u8,
89    },
90    Heal,
91    /// Share a fresh authority-owned page read-write with the current worker.
92    Lend,
93    // authority → client
94    Status {
95        state: u8,
96        worker: u64,
97        epoch: u64,
98        revision: u64,
99        counter: u64,
100        restarts: u64,
101        ledger: u64,
102        lock_word: u64,
103    },
104    Lent {
105        grant: u64,
106    },
107    // client → worker
108    Mutate {
109        request_id: u64,
110        method: u8,
111        amount: u64,
112        fault: Fault,
113    },
114    Get,
115    Go,
116    Crash {
117        mode: u8,
118        arg: u64,
119    },
120    // worker → client
121    Done {
122        request_id: u64,
123        result: u64,
124        revision: u64,
125        replayed: bool,
126    },
127    Value {
128        counter: u64,
129        revision: u64,
130        epoch: u64,
131    },
132    Staged {
133        request_id: u64,
134    },
135    Failed {
136        request_id: u64,
137        refusal: Refusal,
138    },
139    Ack,
140}

142struct Raw {
143    tag: u8,
144    code: u8,
145    half: u32,
146    w: [u64; 7],
147}
148impl Raw {
149    fn new(tag: u8) -> Self {
150        Self {
151            tag,
152            code: 0,
153            half: 0,
154            w: [0; 7],
155        }
156    }
157}
158
159fn refusal(code: u8) -> Option<Refusal> {
160    use Refusal::*;
161    [
162        Conflict,
163        StaleEpoch,
164        RevisionMismatch,
165        RecoveryFailed,
166        QuotaExceeded,
167        Denied,
168        InvalidMutation,
169    ]
170    .into_iter()
171    .find(|r| *r as u8 == code)
172}
173
174fn fault(code: u8) -> Option<Fault> {
175    [
176        Fault::None,
177        Fault::BeforeCommit,
178        Fault::AfterCommit,
179        Fault::Stall,
180    ]
181    .into_iter()
182    .find(|f| *f as u8 == code)
183}
184
185fn put_binding(r: &mut Raw, b: &Binding, first: usize) {
186    r.code = b.method;
187    r.half = b.rights;
188    r.w[first..first + 4].copy_from_slice(&[
189        b.caller_cell,
190        b.caller_generation,
191        b.request_id,
192        b.payload,
193    ]);
194}
195
196fn binding(r: &Raw, first: usize) -> Binding {
197    Binding {
198        caller_cell: r.w[first],
199        caller_generation: r.w[first + 1],
200        request_id: r.w[first + 2],
201        method: r.code,
202        rights: r.half,
203        payload: r.w[first + 3],
204    }
205}

207impl Msg {
208    pub fn encode(&self) -> [u8; FRAME_LEN] {
209        let r = match *self {
210            Msg::Attach { nonce } => Raw {
211                w: [nonce, 0, 0, 0, 0, 0, 0],
212                ..Raw::new(0x11)
213            },
214            Msg::Restored { epoch, revision } => Raw {
215                w: [epoch, revision, 0, 0, 0, 0, 0],
216                ..Raw::new(0x12)
217            },
218            Msg::RestoreFailed { epoch } => Raw {
219                w: [epoch, 0, 0, 0, 0, 0, 0],
220                ..Raw::new(0x13)
221            },
222            Msg::Commit {
223                epoch,
224                expected_revision,
225                binding,
226                new_state,
227            } => {
228                let mut r = Raw {
229                    w: [epoch, expected_revision, 0, 0, 0, 0, new_state],
230                    ..Raw::new(0x14)
231                };
232                put_binding(&mut r, &binding, 2);
233                r
234            }
235            Msg::Lookup { epoch, binding } => {
236                let mut r = Raw {
237                    w: [epoch, 0, 0, 0, 0, 0, 0],
238                    ..Raw::new(0x15)
239                };
240                put_binding(&mut r, &binding, 2);
241                r
242            }
243            Msg::Attached {
244                epoch,
245                revision,
246                schema,
247                counter,
248            } => Raw {
249                half: schema,
250                w: [epoch, revision, counter, 0, 0, 0, 0],
251                ..Raw::new(0x21)
252            },
253            Msg::Committed { revision, result } => Raw {
254                w: [revision, result, 0, 0, 0, 0, 0],
255                ..Raw::new(0x22)
256            },
257            Msg::Present { revision, result } => Raw {
258                w: [revision, result, 0, 0, 0, 0, 0],
259                ..Raw::new(0x23)
260            },
261            Msg::Absent => Raw::new(0x24),
262            Msg::Refused(code) => Raw {
263                code: code as u8,
264                ..Raw::new(0x25)
265            },
266            Msg::Report => Raw::new(0x31),
267            Msg::Fence => Raw::new(0x32),
268            Msg::Corrupt { kind } => Raw {
269                code: kind,
270                ..Raw::new(0x33)
271            },
272            Msg::Heal => Raw::new(0x34),
273            Msg::Lend => Raw::new(0x35),
274            Msg::Lent { grant } => Raw {
275                w: [grant, 0, 0, 0, 0, 0, 0],
276                ..Raw::new(0x42)
277            },
278            Msg::Status {
279                state,
280                worker,
281                epoch,
282                revision,
283                counter,
284                restarts,
285                ledger,
286                lock_word,
287            } => Raw {
288                code: state,
289                w: [
290                    worker, epoch, revision, counter, restarts, ledger, lock_word,
291                ],
292                ..Raw::new(0x41)
293            },
294            Msg::Mutate {
295                request_id,
296                method,
297                amount,
298                fault,
299            } => Raw {
300                code: method,
301                half: fault as u32,
302                w: [request_id, amount, 0, 0, 0, 0, 0],
303                ..Raw::new(0x51)
304            },
305            Msg::Get => Raw::new(0x52),
306            Msg::Go => Raw::new(0x53),
307            Msg::Crash { mode, arg } => Raw {
308                code: mode,
309                w: [arg, 0, 0, 0, 0, 0, 0],
310                ..Raw::new(0x54)
311            },
312            Msg::Done {
313                request_id,
314                result,
315                revision,
316                replayed,
317            } => Raw {
318                code: replayed as u8,
319                w: [request_id, result, revision, 0, 0, 0, 0],
320                ..Raw::new(0x61)
321            },
322            Msg::Value {
323                counter,
324                revision,
325                epoch,
326            } => Raw {
327                w: [counter, revision, epoch, 0, 0, 0, 0],
328                ..Raw::new(0x62)
329            },
330            Msg::Staged { request_id } => Raw {
331                w: [request_id, 0, 0, 0, 0, 0, 0],
332                ..Raw::new(0x63)
333            },
334            Msg::Failed {
335                request_id,
336                refusal,
337            } => Raw {
338                code: refusal as u8,
339                w: [request_id, 0, 0, 0, 0, 0, 0],
340                ..Raw::new(0x64)
341            },
342            Msg::Ack => Raw::new(0x65),
343        };
344        let mut out = [0u8; FRAME_LEN];
345        out[0] = r.tag;
346        out[1] = VERSION;
347        out[2] = r.code;
348        out[4..8].copy_from_slice(&r.half.to_le_bytes());
349        for (i, word) in r.w.iter().enumerate() {
350            out[8 + i * 8..16 + i * 8].copy_from_slice(&word.to_le_bytes());
351        }
352        out
353    }
354
355    pub fn decode(bytes: &[u8]) -> Option<Msg> {
356        let bytes = bytes.get(..FRAME_LEN)?;
357        if bytes[1] != VERSION || bytes[3] != 0 {
358            return None;
359        }
360        let mut r = Raw::new(bytes[0]);
361        r.code = bytes[2];
362        r.half = u32::from_le_bytes(bytes[4..8].try_into().ok()?);
363        for i in 0..7 {
364            r.w[i] = u64::from_le_bytes(bytes[8 + i * 8..16 + i * 8].try_into().ok()?);
365        }
366        let w = r.w;
367        Some(match r.tag {
368            0x11 => Msg::Attach { nonce: w[0] },
369            0x12 => Msg::Restored {
370                epoch: w[0],
371                revision: w[1],
372            },
373            0x13 => Msg::RestoreFailed { epoch: w[0] },
374            0x14 => Msg::Commit {
375                epoch: w[0],
376                expected_revision: w[1],
377                binding: binding(&r, 2),
378                new_state: w[6],
379            },
380            0x15 => Msg::Lookup {
381                epoch: w[0],
382                binding: binding(&r, 2),
383            },
384            0x21 => Msg::Attached {
385                epoch: w[0],
386                revision: w[1],
387                schema: r.half,
388                counter: w[2],
389            },
390            0x22 => Msg::Committed {
391                revision: w[0],
392                result: w[1],
393            },
394            0x23 => Msg::Present {
395                revision: w[0],
396                result: w[1],
397            },
398            0x24 => Msg::Absent,
399            0x25 => Msg::Refused(refusal(r.code)?),
400            0x31 => Msg::Report,
401            0x32 => Msg::Fence,
402            0x33 => Msg::Corrupt { kind: r.code },
403            0x34 => Msg::Heal,
404            0x35 => Msg::Lend,
405            0x42 => Msg::Lent { grant: w[0] },
406            0x41 => Msg::Status {
407                state: r.code,
408                worker: w[0],
409                epoch: w[1],
410                revision: w[2],
411                counter: w[3],
412                restarts: w[4],
413                ledger: w[5],
414                lock_word: w[6],
415            },
416            0x51 => Msg::Mutate {
417                request_id: w[0],
418                method: r.code,
419                amount: w[1],
420                fault: u8::try_from(r.half).ok().and_then(fault)?,
421            },
422            0x52 => Msg::Get,
423            0x53 => Msg::Go,
424            0x54 => Msg::Crash {
425                mode: r.code,
426                arg: w[0],
427            },
428            0x61 => Msg::Done {
429                request_id: w[0],
430                result: w[1],
431                revision: w[2],
432                replayed: r.code != 0,
433            },
434            0x62 => Msg::Value {
435                counter: w[0],
436                revision: w[1],
437                epoch: w[2],
438            },
439            0x63 => Msg::Staged { request_id: w[0] },
440            0x64 => Msg::Failed {
441                request_id: w[0],
442                refusal: refusal(r.code)?,
443            },
444            0x65 => Msg::Ack,
445            _ => return None,
446        })
447    }
