1//! Tier-2 provider of the Phase-02 cross-tier fixture.
2//!
3//! Runs in a **private root** (Tier 2 Paged Domain), launched by `init` under the
4//! `tier2-rpc-entry` feature, and is capability-free: its manifest declares
5//! `PROTECTION_CLASS_UNTRUSTED` and no capability bits, so admission rests on the
6//! class alone. It serves exactly the requests the driver sends and then exits, so
7//! the driver can witness that a descriptor naming it is refused once it is gone.
8//!
9//! Two things it proves by existing:
10//!
11//! 1. A private-root Cell can receive a copied typed request from a Tier-1 Cell and
12//!    answer it — the Tier-2 side of the exchange, over the same `Send`/`Recv`
13//!    surface every other Cell uses (`copy_to_user` into this root, `copy_from_user`
14//!    out of it).
15//! 2. A private-root Cell can call a **named Tier-1 service** through the shipped
16//!    SDK: on the echo request it resolves VFS with `ServiceRef` (which since
17//!    Phase-02 slice A resolves the `LookupServiceBound` binding) and performs a
18//!    typed `Stat("/")`, reporting the result back inside its reply.
19//!
20//! It does **not** register a service: `RegisterService` is `SpawnCap`-gated and a
21//! private-root Cell holds no capabilities, so the fixture is tid-addressed and the
22//! driver witnesses that the service id resolves to nothing.
23
24#![no_std]
25#![no_main]
26#![forbid(unsafe_code)]

54/// How many requests this provider serves before exiting. The driver sends this
55/// many typed requests, in this order, and nothing else can reach it.
56const SERVED_BUDGET: usize = 2;
57
58fn cell_main() {
59    println("[tier2-rpc] provider-start tier2");
60    let mut request_buf = [0u8; IPC_BUF_SIZE];
61    let mut reply_buf = [0u8; IPC_BUF_SIZE];
62    let mut served = 0usize;
63
64    while served < SERVED_BUDGET {
65        // Blocking masked receive: the driver is the only cell that sends here.
66        let sender = match sys_recv(0, &mut request_buf) {
67            SyscallResult::Ok(sender) => sender,
68            SyscallResult::Err(_) => {
69                println("[tier2-rpc] provider-recv-failed");
70                sys_exit(1);
71            }
72        };
73        let Some(request) = decode_request(&request_buf) else {
74            println("[tier2-rpc] provider-undecodable-request");
75            sys_exit(1);
76        };
77        served += 1;
78
79        let response = match request.method {
80            METHOD_ECHO => {
81                // The Tier-2→Tier-1 half: a private-root Cell calling the named VFS
82                // service through the shipped SDK, over copied typed IPC.
83                let (vfs_stat_ok, vfs_root_is_dir) = stat_root_through_sdk();
84                println(&format!(
85                    "[tier2-rpc] provider-echo nonce={} len={} vfs_ok={}",
86                    request.nonce,
87                    request.payload.len(),
88                    vfs_stat_ok
89                ));
90                EchoResponse {
91                    method: METHOD_ECHO,
92                    nonce: request.nonce,
93                    payload_len: request.payload.len() as u32,
94                    payload_checksum: checksum(request.payload),
95                    vfs_stat_ok,
96                    vfs_root_is_dir,
97                    error: None,
98                }
99            }
100            METHOD_UNAUTHORIZED => {
101                // Refused without a side effect, and the refusal names itself.
102                println("[tier2-rpc] provider-refused-unauthorized-method");
103                EchoResponse {
104                    method: METHOD_UNAUTHORIZED,
105                    nonce: request.nonce,
106                    payload_len: 0,
107                    payload_checksum: 0,
108                    vfs_stat_ok: false,
109                    vfs_root_is_dir: false,
110                    error: Some(UNAUTHORIZED),
111                }
112            }
113            other => {
114                println(&format!("[tier2-rpc] provider-refused-method {other}"));
115                EchoResponse {
116                    method: other,
117                    nonce: request.nonce,
118                    payload_len: 0,
119                    payload_checksum: 0,
120                    vfs_stat_ok: false,
121                    vfs_root_is_dir: false,
122                    error: Some(UNAUTHORIZED),
123                }
124            }
125        };
126
127        let Some(len) = encode_response(&response, &mut reply_buf) else {
128            println("[tier2-rpc] provider-response-does-not-fit");
129            sys_exit(1);
130        };
131        if !matches!(sys_send(sender, &reply_buf[..len]), SyscallResult::Ok(_)) {
132            println("[tier2-rpc] provider-reply-failed");
133            sys_exit(1);
134        }
135    }
136
137    println(&format!("[tier2-rpc] provider-served {served}"));
138    println("[tier2-rpc] provider-exiting");
139    sys_exit(0)
140}

142/// Ask the named Tier-1 VFS service about `/` through the SDK and report whether the
143/// call completed and what it said. A refusal is not fatal to the fixture: it is
144/// reported in the reply, so the driver sees which half of the exchange moved.
145fn stat_root_through_sdk() -> (bool, bool) {
146    let mut vfs: ServiceRef<{ service::VFS }> = ServiceRef::new();
147    let mut response_buffer = [0u8; IPC_BUF_SIZE];
148    match vfs.call::<VfsRequest, VfsResponse>(&VfsRequest::Stat("/"), &mut response_buffer) {
149        Ok(VfsResponse::Stat { is_dir, .. }) => {
150            println(&format!("[tier2-rpc] provider-vfs-stat ok is_dir={is_dir}"));
151            (true, is_dir)
152        }
153        Ok(other) => {
154            println(&format!(
155                "[tier2-rpc] provider-vfs-stat unexpected {other:?}"
156            ));
157            (true, false)
158        }
159        Err(error) => {
160            println(&format!("[tier2-rpc] provider-vfs-stat failed {error:?}"));
161            (false, false)
162        }
163    }
164}
