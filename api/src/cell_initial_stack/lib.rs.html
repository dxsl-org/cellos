1//! The Cell initial stack, v1 (CP1 `cp1-cell-initial-stack-auxv.md`, AX-2…AX-5).
2//!
3//! The loader writes this block just below the top of every Cell's root user
4//! stack and enters the Cell with `sp` at its first byte. Layout, low to high
5//! address, all words 64-bit little-endian:
6//!
7//! ```text
8//! sp →  argc = 0
9//!       argv terminator (NULL)
10//!       envp terminator (NULL)
11//!       auxv pairs: AT_PHDR, AT_PHENT, AT_PHNUM, AT_PAGESZ, AT_BASE, AT_ENTRY,
12//!                   [AT_RANDOM], then AT_NULL
13//!       program-header copy (phnum × 56 bytes, 8-byte aligned)
14//!       [16 entropy bytes]
15//!       padding up to the stack top
16//! ```
17//!
18//! `sp` is 16-byte aligned on every architecture. `AT_PHDR` and `AT_RANDOM`
19//! hold absolute addresses inside the block. `AT_BASE` carries the Cell's load
20//! base (Cellos has no interpreter). Pointers are computed from `top`, so the
21//! block is position-dependent on the stack it is written to and nothing else.
22
23#![no_std]
24#![forbid(unsafe_code)]
25
26/// ELF64 program-header entry size.
27pub const PHENT: usize = 56;
28/// AX-5: a Cell with more program headers is refused at spawn.
29pub const MAX_PHDRS: usize = 16;
30/// Page size reported in `AT_PAGESZ`.
31pub const PAGE_SIZE: u64 = 4096;
32
33pub const AT_NULL: u64 = 0;
34pub const AT_PHDR: u64 = 3;
35pub const AT_PHENT: u64 = 4;
36pub const AT_PHNUM: u64 = 5;
37pub const AT_PAGESZ: u64 = 6;
38pub const AT_BASE: u64 = 7;
39pub const AT_ENTRY: u64 = 9;
40pub const AT_RANDOM: u64 = 25;
41
42const HEAD_WORDS: usize = 3; // argc, argv NULL, envp NULL
43const MAX_AUX_PAIRS: usize = 8; // seven entries + AT_NULL
44const RANDOM_BYTES: usize = 16;
45
46/// Upper bound of the block size, rounded to 16 bytes.
47pub const MAX_BYTES: usize =
48    (((HEAD_WORDS + 2 * MAX_AUX_PAIRS) * 8 + MAX_PHDRS * PHENT + RANDOM_BYTES) + 15) & !15;
49
50/// What the loader knows about the image when it builds the block.
51pub struct Image<'a> {
52    /// The ELF program-header table, `phnum × PHENT` bytes, copied verbatim.
53    pub phdrs: &'a [u8],
54    /// Runtime load base of the image (0 for a non-PIE image).
55    pub load_base: u64,
56    /// Runtime entry address (`e_entry + load_base`).
57    pub entry: u64,
58    /// Kernel entropy; `None` omits `AT_RANDOM` (AX-4).
59    pub random: Option<[u8; RANDOM_BYTES]>,
60}
61
62/// The block to copy to `[sp, top)`.
63pub struct Block {
64    bytes: [u8; MAX_BYTES],
65    len: usize,
66    sp: u64,
67}
68
69impl Block {
70    /// Entry stack pointer: the address of `argc`.
71    pub fn sp(&self) -> u64 {
72        self.sp
73    }
74
75    /// Bytes to write at `sp`; their length is `top - sp`.
76    pub fn bytes(&self) -> &[u8] {
77        &self.bytes[..self.len]
78    }
79}
80
81#[derive(Debug, PartialEq, Eq)]
82pub enum Error {
83    /// The table is not a whole number of 56-byte entries.
84    BadPhdrTable,
85    /// More than [`MAX_PHDRS`] program headers (AX-5).
86    TooManyPhdrs,
87    /// `top` is not 16-byte aligned or too low to hold the block.
88    BadTop,
89}
90
91/// Lay out the block for a stack whose highest address (exclusive) is `top`.
92pub fn build(top: u64, image: &Image<'_>) -> Result<Block, Error> {
93    if !image.phdrs.len().is_multiple_of(PHENT) {
94        return Err(Error::BadPhdrTable);
95    }
96    let phnum = image.phdrs.len() / PHENT;
97    if phnum > MAX_PHDRS {
98        return Err(Error::TooManyPhdrs);
99    }
100    let aux_pairs = 7 + usize::from(image.random.is_some());
101    let words_len = (HEAD_WORDS + 2 * aux_pairs) * 8;
102    let phdr_off = words_len;
103    let random_off = phdr_off + image.phdrs.len();
104    let used = random_off
105        + if image.random.is_some() {
106            RANDOM_BYTES
107        } else {
108            0
109        };
110    let len = (used + 15) & !15;
111    if !top.is_multiple_of(16) || top < len as u64 {
112        return Err(Error::BadTop);
113    }
114    let sp = top - len as u64;
115
116    let mut bytes = [0u8; MAX_BYTES];
117    let mut at = 0;
118    let mut word = |value: u64| {
119        bytes[at..at + 8].copy_from_slice(&value.to_le_bytes());
120        at += 8;
121    };
122    word(0); // argc
123    word(0); // argv terminator
124    word(0); // envp terminator
125    for (key, value) in [
126        (AT_PHDR, sp + phdr_off as u64),
127        (AT_PHENT, PHENT as u64),
128        (AT_PHNUM, phnum as u64),
129        (AT_PAGESZ, PAGE_SIZE),
130        (AT_BASE, image.load_base),
131        (AT_ENTRY, image.entry),
132    ] {
133        word(key);
134        word(value);
135    }
136    if image.random.is_some() {
137        word(AT_RANDOM);
138        word(sp + random_off as u64);
139    }
140    word(AT_NULL);
141    word(0);
142
143    bytes[phdr_off..random_off].copy_from_slice(image.phdrs);
144    if let Some(random) = image.random {
145        bytes[random_off..random_off + RANDOM_BYTES].copy_from_slice(&random);
146    }
147    Ok(Block { bytes, len, sp })
148}
