3/// Sense key values (SPC-4 table 49); the key is the low nibble.

20    pub const DATA_PROTECT: u8 = 0x7;
21    /// Command aborted by the device server.

25/// ASC 0x04 "logical unit not ready".
26const ASC_NOT_READY: u8 = 0x04;
27/// ASCQ 0x01 under ASC 0x04: "in process of becoming ready".
28const ASCQ_BECOMING_READY: u8 = 0x01;
29
30/// Decoded sense: the three fields the disk layer acts on.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub struct Sense {
33    /// Sense key (low nibble only; see [`sense_key`]).
34    pub key: u8,
35    /// Additional sense code; 0 when the device did not supply one.
36    pub asc: u8,
37    /// Additional sense code qualifier; 0 when the device did not supply one.
38    pub ascq: u8,
39}
40
41/// How a caller should react to a CHECK CONDITION carrying a given [`Sense`].
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub enum SenseClass {
44    /// Transient: reissuing the same command may succeed (UNIT ATTENTION, or
45    /// NOT READY / "becoming ready").
46    Retry,
47    /// Reissuing the same command will not help.
48    Fatal,
49}
50
51impl Sense {
52    /// Decodes fixed-format (response code 0x70/0x71) or descriptor-format
53    /// (0x72/0x73) sense data.
54    ///
55    /// Returns `None` when the response code is unknown or `bytes` is too
56    /// short to hold the sense key (3 bytes fixed, 2 bytes descriptor). In
57    /// fixed format, ASC/ASCQ are taken only when both the buffer and the
58    /// ADDITIONAL SENSE LENGTH (byte 7) cover bytes 12–13; otherwise they read
59    /// as 0. Descriptor format needs 4 bytes for ASC/ASCQ, with the same
60    /// fallback.
61    pub fn decode(bytes: &[u8]) -> Option<Self> {
62        match bytes.first()? & 0x7F {
63            0x70 | 0x71 => Self::decode_fixed(bytes),
64            0x72 | 0x73 => Self::decode_descriptor(bytes),
65            _ => None,
66        }
67    }
68
69    fn decode_fixed(bytes: &[u8]) -> Option<Self> {
70        let key = bytes.get(2)? & 0x0F;
71        // Bytes 0..8 are always present in a well-formed record; byte 7 counts
72        // what follows it, so ASC/ASCQ (12, 13) need an additional length >= 6.
73        let covered = bytes.get(7).is_some_and(|&add| usize::from(add) >= 6);
74        let (asc, ascq) = match (covered, bytes.get(12), bytes.get(13)) {
75            (true, Some(&asc), Some(&ascq)) => (asc, ascq),
76            _ => (0, 0),
77        };
78        Some(Self { key, asc, ascq })
79    }
80
81    fn decode_descriptor(bytes: &[u8]) -> Option<Self> {
82        let key = bytes.get(1)? & 0x0F;
83        let (asc, ascq) = match (bytes.get(2), bytes.get(3)) {
84            (Some(&asc), Some(&ascq)) => (asc, ascq),
85            _ => (0, 0),
86        };
87        Some(Self { key, asc, ascq })
88    }
89
90    /// Classifies this sense for retry decisions.
91    ///
92    /// [`SenseClass::Retry`]: UNIT ATTENTION, and NOT READY with ASC/ASCQ
93    /// 04h/01h ("becoming ready"). Everything else — including MEDIUM ERROR,
94    /// HARDWARE ERROR, ILLEGAL REQUEST, DATA PROTECT, and NOT READY for any
95    /// other reason (no medium, manual intervention) — is
96    /// [`SenseClass::Fatal`].
97    pub fn class(&self) -> SenseClass {
98        match (self.key, self.asc, self.ascq) {
99            (sense_key::UNIT_ATTENTION, _, _) => SenseClass::Retry,
100            (sense_key::NOT_READY, ASC_NOT_READY, ASCQ_BECOMING_READY) => SenseClass::Retry,
101            _ => SenseClass::Fatal,
102        }
103    }
104}
