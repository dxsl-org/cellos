1//! Error types. Device-supplied input never panics; it lands in one of these.
2
3use crate::sense::Sense;
4
5/// A device response that does not match the SPC/SBC layout.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub enum ParseError {
8    /// INQUIRY returned fewer than the 5-byte standard header.
9    InquiryTooShort(usize),
10    /// REPORT LUNS returned fewer than the 8-byte list header.
11    ReportLunsTooShort(usize),
12    /// READ CAPACITY(16) returned fewer than the 12 bytes holding the
13    /// last LBA and block length.
14    CapacityTooShort(usize),
15    /// READ CAPACITY(16) reported a last LBA of `u64::MAX`, so the block
16    /// count does not fit in `u64`.
17    CapacityOverflow,
18    /// READ CAPACITY(16) reported a logical block length that is not a power
19    /// of two of at least 512 bytes.
20    BadBlockSize(u32),
21}

23/// Errors from probing or using a [`crate::ScsiDisk`].
24///
25/// `T` is the executor's transport error ([`crate::ScsiExecutor::Error`]).
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub enum ScsiError<T> {
28    /// The executor failed to obtain a SAM status.
29    Transport(T),
30    /// CHECK CONDITION with decoded sense; use [`Sense::class`] to decide on
31    /// a retry.
32    CheckCondition(Sense),
33    /// CHECK CONDITION without decodable sense data.
34    CheckConditionNoSense,
35    /// A SAM status other than GOOD or CHECK CONDITION (see
36    /// [`crate::status`]).
37    Status(u8),
38    /// The device reported a residual larger than the data buffer.
39    BadResidual {
40        /// Data buffer length in bytes.
41        len: usize,
42        /// Residual the device reported.
43        residual: u32,
44    },
45    /// READ/WRITE transferred fewer bytes than requested.
46    ShortTransfer {
47        /// Bytes requested.
48        expected: usize,
49        /// Bytes actually transferred.
50        transferred: usize,
51    },
52    /// Device data-in failed to parse.
53    Parse(ParseError),
54    /// No reported LUN answered INQUIRY as a connected direct-access disk.
55    NoDisk,
56    /// TEST UNIT READY still returned Retry-class sense after the
57    /// caller-supplied number of retries; carries the last sense.
58    NotReady(Sense),
59    /// [`crate::ProbeConfig::max_transfer_bytes`] is smaller than one block.
60    TransferLimitBelowBlockSize {
61        /// Configured limit in bytes.
62        max_transfer_bytes: u32,
63        /// Device logical block length.
64        block_size: u32,
65    },
66    /// Buffer length is not a multiple of the block size.
67    Misaligned {
68        /// Buffer length in bytes.
69        len: usize,
70        /// Device logical block length.
71        block_size: u32,
72    },
73    /// The request extends past the last block.
74    OutOfRange {
75        /// First block requested.
76        lba: u64,
77        /// Number of blocks requested.
78        blocks: u64,
79        /// Device block count.
80        block_count: u64,
81    },
82    /// The request exceeds the per-command transfer limit.
83    TooLarge {
84        /// Number of blocks requested.
85        blocks: u64,
86        /// Most blocks one command may carry.
87        max_blocks: u32,
88    },
89}
90
91impl<T> From<ParseError> for ScsiError<T> {
92    fn from(err: ParseError) -> Self {
93        ScsiError::Parse(err)
94    }
