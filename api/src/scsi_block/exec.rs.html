6/// SAM-5 status codes the disk layer distinguishes.
7pub mod status {
8    /// Command completed successfully.
9    pub const GOOD: u8 = 0x00;
10    /// Command failed; sense data describes why.
11    pub const CHECK_CONDITION: u8 = 0x02;
12    /// Logical unit is busy.
13    pub const BUSY: u8 = 0x08;
14    /// Logical unit is reserved by another initiator.
15    pub const RESERVATION_CONFLICT: u8 = 0x18;
16    /// Task set is full.
17    pub const TASK_SET_FULL: u8 = 0x28;
18    /// Command aborted by another initiator.
19    pub const TASK_ABORTED: u8 = 0x40;
20}
21
22/// The 8-byte SAM LUN of logical unit 0; also used when REPORT LUNS is
23/// unsupported.
24pub const LUN0: [u8; 8] = [0; 8];

26/// The data phase of one command, carrying the buffer the transport DMAs
27/// into or out of.
28#[derive(Debug)]
29pub enum DataPhase<'a> {
30    /// No data transfer.
31    None,
32    /// Device-to-host transfer into the buffer.
33    In(&'a mut [u8]),
34    /// Host-to-device transfer from the buffer.
35    Out(&'a [u8]),
36}
37
38impl DataPhase<'_> {
39    /// Buffer length in bytes; 0 for [`DataPhase::None`].
40    pub fn len(&self) -> usize {
41        match self {
42            DataPhase::None => 0,
43            DataPhase::In(buf) => buf.len(),
44            DataPhase::Out(buf) => buf.len(),
45        }
46    }
47
48    /// Whether the phase transfers no bytes.
49    pub fn is_empty(&self) -> bool {
50        self.len() == 0
51    }
52}
53
54/// What the device returned for one command.
55#[derive(Clone, Copy, Debug, PartialEq, Eq)]
56pub struct Completion {
57    /// SAM status byte (see [`status`]).
58    pub status: u8,
59    /// Sense decoded with [`Sense::decode`] when the transport received sense
60    /// bytes it could decode; `None` otherwise.
61    pub sense: Option<Sense>,
62    /// Bytes of the data phase NOT transferred. Device-reported and
63    /// untrusted: the disk layer rejects values above the buffer length.
64    pub residual: u32,
65}
66
67/// Executes SCSI commands over one transport (virtio-scsi request queue,
68/// PVSCSI ring, ...).
69pub trait ScsiExecutor {
70    /// Transport failure: queue full, device gone, host/target-level
71    /// response codes, or anything else that prevented a SAM status.
72    type Error;
73
74    /// Issues `cdb` to `lun` on `target` and blocks until it completes.
75    ///
76    /// `lun` is the 8-byte SAM LUN as returned by REPORT LUNS; the transport
77    /// re-encodes it if its wire format differs. Returns `Ok` whenever the
78    /// device produced a SAM status, including CHECK CONDITION; `Err` only
79    /// when no status was obtained.
80    fn execute(
81        &mut self,
82        target: u8,
83        lun: [u8; 8],
84        cdb: &[u8],
85        data: DataPhase<'_>,
86    ) -> Result<Completion, Self::Error>;
87}
88
89/// Lets a caller probe several targets with one executor (`&mut exec`)
90/// without giving it up when a probe fails.
91impl<T: ScsiExecutor + ?Sized> ScsiExecutor for &mut T {
92    type Error = T::Error;
93
94    fn execute(
95        &mut self,
96        target: u8,
97        lun: [u8; 8],
98        cdb: &[u8],
99        data: DataPhase<'_>,
100    ) -> Result<Completion, Self::Error> {
101        (**self).execute(target, lun, cdb, data)
102    }
103}
104
105/// Runs one command and maps non-GOOD outcomes to [`ScsiError`].
106///
107/// Returns the number of bytes actually transferred (buffer length minus
108/// residual).
109pub(crate) fn run<E: ScsiExecutor>(
110    exec: &mut E,
111    target: u8,
112    lun: [u8; 8],
113    cdb: &[u8],
114    data: DataPhase<'_>,
115) -> Result<usize, ScsiError<E::Error>> {
116    let len = data.len();
117    let done = exec
118        .execute(target, lun, cdb, data)
119        .map_err(ScsiError::Transport)?;
120    match (done.status, done.sense) {
121        (status::GOOD, _) => {}
122        (status::CHECK_CONDITION, Some(sense)) => return Err(ScsiError::CheckCondition(sense)),
123        (status::CHECK_CONDITION, None) => return Err(ScsiError::CheckConditionNoSense),
124        (other, _) => return Err(ScsiError::Status(other)),
125    }
126    usize::try_from(done.residual)
127        .ok()
128        .and_then(|residual| len.checked_sub(residual))
129        .ok_or(ScsiError::BadResidual {
130            len,
131            residual: done.residual,
132        })
133}
