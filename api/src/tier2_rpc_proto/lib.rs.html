1//! Wire contract for the Phase-02 cross-tier fixture (`/bin/tier2-rpc-provider`
2//! ↔ `/bin/tier2-rpc-driver`).
3//!
4//! One postcard record each way, so both cells cannot drift from one shape. The
5//! fixture is deliberately **tid-addressed**: a private-root Cell cannot register a
6//! service today (`RegisterService` stays `SpawnCap`-gated), so `init` hands the
7//! driver the provider's tid through the reviewed argv stash and no registry entry
8//! exists for [`SERVICE_ID`]. That id is carried only so the two cells name the
9//! same number, and the driver witnesses that it resolves to nothing.
10//!
11//! Not a public ABI: nothing outside the fixture's two cells links this crate.
12
13#![no_std]
14#![forbid(unsafe_code)]
15
16use serde::{Deserialize, Serialize};
17
18/// The fixture's service id. **Not registered** — see the module docs.
19pub const SERVICE_ID: u16 = 0x7A01;
20
21/// Served method: echo the request body and report what the Tier-2 cell saw when it
22/// called the named Tier-1 VFS service.
23pub const METHOD_ECHO: u8 = 1;
24
25/// A method the provider deliberately does not authorize. It must answer with a
26/// refusal and leave no side effect.
27pub const METHOD_UNAUTHORIZED: u8 = 9;
28
29/// Request the driver sends and the provider serves.
30#[derive(Serialize, Deserialize, Debug, PartialEq, Eq)]
31pub struct EchoRequest<'a> {
32    /// One of [`METHOD_ECHO`] / [`METHOD_UNAUTHORIZED`].
33    pub method: u8,
34    /// Caller-chosen nonce, echoed so a stale or reordered reply cannot pass.
35    pub nonce: u64,
36    /// Nontrivial payload: the provider echoes its length and checksum.
37    pub payload: &'a [u8],
38}
39
40/// What the provider answers.
41#[derive(Serialize, Deserialize, Debug, PartialEq, Eq)]
42pub struct EchoResponse<'a> {
43    /// The method this answer belongs to.
44    pub method: u8,
45    /// Echo of the request nonce.
46    pub nonce: u64,
47    /// Length and checksum of the payload the provider actually received.
48    pub payload_len: u32,
49    pub payload_checksum: u32,
50    /// Whether the Tier-2 cell's own typed call to the named Tier-1 VFS service
51    /// completed, and what that service said about the root directory.
52    pub vfs_stat_ok: bool,
53    pub vfs_root_is_dir: bool,
54    /// Non-empty only when the provider refused the request.
55    pub error: Option<&'a str>,
56}
57
58/// The refusal a provider returns for a method it does not authorize.
59pub const UNAUTHORIZED: &str = "unauthorized-method";
60
61/// FNV-1a over the payload: any truncation, reorder or substitution in the copied
62/// body changes it, so the driver can tell an echo from a plausible-looking reply.
63pub fn checksum(bytes: &[u8]) -> u32 {
64    let mut hash: u32 = 0x811c_9dc5;
65    for byte in bytes {
66        hash ^= *byte as u32;
67        hash = hash.wrapping_mul(0x0100_0193);
68    }
69    hash
70}
71
72/// Encode a request into `out`; `None` when it does not fit.
73pub fn encode_request(request: &EchoRequest<'_>, out: &mut [u8]) -> Option<usize> {
74    postcard::to_slice(request, out).ok().map(|used| used.len())
75}
76
77/// Decode a request from `bytes`; `None` when the bytes are not this record.
78pub fn decode_request(bytes: &[u8]) -> Option<EchoRequest<'_>> {
79    postcard::from_bytes(bytes).ok()
80}
81
82/// Encode a response into `out`; `None` when it does not fit.
83pub fn encode_response(response: &EchoResponse<'_>, out: &mut [u8]) -> Option<usize> {
84    postcard::to_slice(response, out)
85        .ok()
86        .map(|used| used.len())
87}
88
89/// Decode a response from `bytes`; `None` when the bytes are not this record.
90pub fn decode_response(bytes: &[u8]) -> Option<EchoResponse<'_>> {
91    postcard::from_bytes(bytes).ok()
92}
