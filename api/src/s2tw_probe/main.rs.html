1//! Private Phase04 S2-T/S2-W fixture (`phase-04-s2tw-annex.md` §6). One binary is
2//! packaged at a sandbox path (`/data/bin/s2tw-probe`, EMPTY launch ceiling) and at
3//! a private `/bin/s2tw-probe` fixture row (`CapSet::LEGACY_OBSERVE_MUTATE`); the
4//! output must differ only because of the path it was launched from.
5//!
6//! It declares `GetProcs`, `GetProcs2`, `TelemetrySnapshot`, `MemInfo` and
7//! `VfsMutate`, tries each right once and writes, to its delegated stdout (the
8//! shell's host stream, never the UART):
9//!
10//! ```text
11//! S2TW getprocs <r>
12//! S2TW getprocs2 <r>
13//! S2TW telemetry <r>
14//! S2TW meminfo <r>
15//! S2TW vfs-write <r>
16//! S2TW DONE
17//! ```
18//!
19//! `<r>` is `denied` for a refusal and `other:<detail>` for anything else:
20//! - telemetry: the syscall ABI encodes every refusal as the one generic error
21//!   sentinel, and each call passes a valid, non-empty buffer, so an error return
22//!   is the dispatch gate (`check_allowlist`) refusing it. `telemetry` is the
23//!   Phase 06 S6-1 `TelemetrySnapshot` (X right, shared bit 55); an admitted
24//!   snapshot must also decode, or it reads `other:malformed-<why>`;
25//! - `vfs-write`: `WriteAt` of `/tmp/s2tw-probe-<tag>` through the cwd directory
26//!   handle the shell delegated, answered `Err(ERR_DENIED)` by the VFS. `<tag>` is
27//!   the first argument (`x` when absent). The handle route keeps the request
28//!   reachable for this path-sealed child, so the only denial left is the missing
29//!   attested `CALLER_FLAG_VFS_MUTATE`.
30//!
31//! With the first argument `trunc` it instead makes one capture with task rows requested
32//! into a 192-byte buffer (room for the header and one Cell row only) and writes one line,
33//! `S2TW telemetry-trunc <r>`: `other:cells-E/T-tasks-e/U-complete-C` counts the whole
34//! rows `TelemetrySnapshot::parse` decoded (E Cells, e tasks) against the header totals
35//! (T, U) with the `COMPLETE` flag (C = 1 or 0); `denied` or `other:<detail>` otherwise.
36//! On any multi-Cell system it is the provider reporting truncation explicitly
37//! (Phase 06 S6-5, D65-8 T-a).
38//!
39//! Exit statuses: 0 every line written; 2 missing stdout grant or bad tag; 3 stdout
40//! failure. Nothing is retried. Never part of an ordinary image.

95const OK: i32 = 0;
96const USAGE: i32 = 2;
97const IO: i32 = 3;
98/// No deadline: block until the reader takes the bytes, closes or cancels.
99const FOREVER: u64 = u64::MAX;
100/// The VFS wire code for `PermissionDenied` (`cells/services/vfs/src/paths.rs`).
101const VFS_ERR_DENIED: u8 = 3;
102/// Directory the write targets; the shell's cwd must be it or an ancestor.
103const TARGET_DIR: &str = "/tmp";
104const ROWS: usize = 64;
105/// 16 KiB snapshot buffer: whole rows only; a fuller system reports truncation.
106const TELEMETRY_WORDS: usize = 2048;
107/// 192 bytes: the 96-byte header and one 96-byte Cell row (D65-8 T-a).
108const TRUNC_WORDS: usize = (api::telemetry::HEADER_LEN + api::telemetry::CELL_LEN) / 8;

110/// The committed endpoint a parent delegated into `slot`.
111fn granted(context: &ChildContext, slot: GrantSlot) -> Option<Endpoint> {
112    let operation = OperationRef::from_raw(context.operation()).ok()?;
113    let grant = context.grants().find(|g| g.slot == slot && g.kind == 1)?;
114    Some(Endpoint::new(EndpointRef::from_raw(grant.object).ok()?, operation))
115}
116
117/// Write every byte of `data`; the stream accepts at most 512 per call.
118fn write_all(out: Endpoint, mut data: &[u8]) -> Result<(), i32> {
119    while !data.is_empty() {
120        match out.write(data, FOREVER) {
121            Ok(count) if count > 0 && count <= data.len() => data = &data[count..],
122            _ => return Err(IO),
123        }
124    }
125    Ok(())
126}

128fn telemetry<E>(result: Result<usize, E>) -> String {
129    match result {
130        Ok(rows) => format!("other:rows-{rows}"),
131        Err(_) => String::from("denied"),
132    }
133}

135fn get_procs() -> String {
136    let mut rows: Vec<ProcessInfo> = Vec::new();
137    rows.resize(ROWS, ProcessInfo::default());
138    telemetry(sys_get_procs(&mut rows))
139}

143    rows.resize(ROWS, ProcessInfoV2::default());
144    telemetry(sys_get_procs2(&mut rows))

147fn telemetry_snapshot() -> String {
148    let mut out: Vec<u64> = Vec::new();
149    out.resize(TELEMETRY_WORDS, 0);
150    match sys_telemetry_snapshot(&mut out, request::TASKS) {
151        Ok(written) => match TelemetrySnapshot::parse(telemetry_bytes(&out, written)) {
152            Ok(snapshot) => {
153                let header = snapshot.header();
154                format!("other:cells-{}-tasks-{}", header.total_cells, header.total_tasks)
155            }
156            Err(why) => format!("other:malformed-{why:?}"),
157        },
158        Err(SyscallError::PermissionDenied) => String::from("denied"),
159        Err(error) => format!("other:error-{error:?}"),
160    }
161}

163/// `other:cells-E/T-tasks-e/U-complete-C` of snapshot `bytes`: the whole rows the parsed
164/// view holds against the header totals, or `other:malformed-<why>`.
165fn trunc_result(bytes: &[u8]) -> String {
166    match TelemetrySnapshot::parse(bytes) {
167        Ok(snapshot) => {
168            let header = snapshot.header();
169            format!(
170                "other:cells-{}/{}-tasks-{}/{}-complete-{}",
171                snapshot.cells().count(),
172                header.total_cells,
173                snapshot.tasks().count(),
174                header.total_tasks,
175                u8::from(header.has(header_flags::COMPLETE))
176            )
177        }
178        Err(why) => format!("other:malformed-{why:?}"),
179    }
180}

182/// One capture with task rows requested into a header + one Cell row buffer.
183fn telemetry_trunc() -> String {
184    let mut out = [0u64; TRUNC_WORDS];
185    match sys_telemetry_snapshot(&mut out, request::TASKS) {
186        Ok(written) => trunc_result(telemetry_bytes(&out, written)),
187        Err(SyscallError::PermissionDenied) => String::from("denied"),
188        Err(error) => format!("other:error-{error:?}"),
189    }
190}

199/// Components below the cwd that lead to [`TARGET_DIR`], or `None` when the cwd is
200/// neither the target nor one of its ancestors.
201fn components_to_target(cwd: &str) -> Option<Vec<&'static str>> {
202    let cwd = cwd.trim_end_matches('/');
203    let rest = TARGET_DIR.strip_prefix(cwd)?;
204    if !rest.is_empty() && !rest.starts_with('/') {
205        return None;
206    }
207    Some(rest.split('/').filter(|c| !c.is_empty()).collect())
208}

210/// Detail for an `other:` result (never a refusal).
211fn vfs_detail(response: &VfsResponse<'_>) -> String {
212    match response {
213        VfsResponse::Err(code) => format!("vfs-err-{code}"),
214        VfsResponse::Ok => String::from("ok"),
215        VfsResponse::DirHandle(handle) => format!("dir-{}", handle.0),
216        _ => String::from("unexpected-response"),
217    }
218}

220/// Create `/tmp/s2tw-probe-<tag>` through the delegated cwd handle.
221fn vfs_write(context: &ChildContext, tag: &str) -> String {
222    let Some(cwd_grant) = context.cwd_grant() else {
223        return String::from("other:no-cwd-grant");
224    };
225    let Ok(cwd) = core::str::from_utf8(context.cwd()) else {
226        return String::from("other:cwd-not-utf8");
227    };
228    let Some(components) = components_to_target(cwd) else {
229        return format!("other:cwd-outside-{TARGET_DIR}");
230    };
231    let mut vfs = VfsRef::new();
232    let mut dir = ViDirHandle(cwd_grant.id);
233    for name in components {
234        let mut reply = [0u8; IPC_BUF_SIZE];
235        match vfs.call::<VfsRequest, VfsResponse>(&VfsRequest::OpenDir { dir, name }, &mut reply) {
236            Ok(VfsResponse::DirHandle(handle)) => dir = handle,
237            Ok(other) => return format!("other:opendir-{}", vfs_detail(&other)),
238            Err(error) => return format!("other:opendir-ipc-{error:?}"),
239        }
240    }
241    let name = format!("s2tw-probe-{tag}");
242    let content = format!("s2tw-{tag}\n");
243    let request = VfsRequest::WriteAt { dir, name: &name, content: content.as_bytes() };
244    let mut reply = [0u8; IPC_BUF_SIZE];
245    match vfs.call::<VfsRequest, VfsResponse>(&request, &mut reply) {
246        Ok(VfsResponse::Err(VFS_ERR_DENIED)) => String::from("denied"),
247        Ok(other) => format!("other:{}", vfs_detail(&other)),
248        Err(error) => format!("other:ipc-{error:?}"),
249    }
250}

252fn run() -> i32 {
253    let context = match ChildContext::read_own() {
254        Ok(context) => context,
255        Err(_) => return USAGE,
256    };
257    let Some(output) = granted(&context, GrantSlot::Stdout) else {
258        return USAGE;
259    };
260    let tag = context.argv().next().unwrap_or("x");
261    if tag.is_empty() || !tag.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'-') {
262        return USAGE;
263    }
264    if tag == "trunc" {
265        return match write_all(output, format!("S2TW telemetry-trunc {}\n", telemetry_trunc()).as_bytes()) {
266            Ok(()) => OK,
267            Err(code) => code,
268        };
269    }
270    let results = [
271        ("getprocs", get_procs()),
272        ("getprocs2", get_procs2()),
273        ("telemetry", telemetry_snapshot()),
274        ("meminfo", mem_info()),
275        ("vfs-write", vfs_write(&context, tag)),
276    ];
277    for (right, result) in &results {
278        if let Err(code) = write_all(output, format!("S2TW {right} {result}\n").as_bytes()) {
279            return code;
280        }
281    }
282    match write_all(output, b"S2TW DONE\n") {
283        Ok(()) => OK,
284        Err(code) => code,
285    }
286}
287
288fn cell_main() {
289    sys_exit(run() as u32 as usize);
290}

315            .to_bytes()
316    }

318    #[test]
319    fn truncated_capture_reports_whole_rows_against_totals() {
320        let bytes = [&header(5, 1, 0).to_bytes()[..], &cell()[..]].concat();
321        assert_eq!(trunc_result(&bytes), "other:cells-1/5-tasks-0/40-complete-0");
322    }
323
324    #[test]
325    fn bytes_that_disagree_with_the_emitted_counts_are_malformed() {
326        // A partial row past the one emitted Cell.
327        let bytes = [&header(5, 1, 0).to_bytes()[..], &cell()[..], &[0u8; 8][..]].concat();
328        assert_eq!(trunc_result(&bytes), "other:malformed-Length");
329        // COMPLETE with fewer emitted than total Cells.
330        let bytes = [&header(5, 1, header_flags::COMPLETE).to_bytes()[..], &cell()[..]].concat();
331        assert_eq!(trunc_result(&bytes), "other:malformed-Malformed");
332    }
