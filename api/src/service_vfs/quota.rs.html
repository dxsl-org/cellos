1//! Per-cell disk quota enforcement for the VFS service.
2//!
3//! Phase 13 tracks bytes-on-disk per `CellId` and rejects writes that would
4//! push a cell over its quota.  Default quota is `DEFAULT_QUOTA_BYTES`.
5
6//! ## Who gets credited on delete
7//! Usage is charged to the cell that wrote a path, so it must be released to that
8//! same cell — not to whoever deletes the file. Releasing to the deleter let any
9//! cell mint quota for itself by deleting another cell's files, and left the real
10//! writer permanently charged for bytes that are gone. [`QuotaTracker`] therefore
11//! remembers the charged writer per path.
12//!
13//! Bound on that record: if two cells write the same path, the first writer stays
14//! recorded and is credited for the whole file when it is deleted. That
15//! misallocates between two cells that already share a file, but it preserves the
16//! property that matters — **deleting a file never credits the deleter**.
17//! The record is in-memory, so it does not survive a hot-swap of this cell; after
18//! one, a delete credits nobody, which errs toward over-charging rather than
19//! toward handing out free quota.
20
21use alloc::collections::BTreeMap;
22use alloc::string::{String, ToString};
23use types::{CellId, ViError, ViResult};
24
25/// Default per-cell quota: 32 MB.
26const DEFAULT_QUOTA_BYTES: u64 = 32 * 1024 * 1024;
27
28/// Per-cell quota and usage tracker.
29///
30/// Keyed by `CellId` alone, without the caller generation: a respawned service
31/// inherits its predecessor's usage because the bytes are still on disk.
32#[derive(Default)]
33pub struct QuotaTracker {
34    used: BTreeMap<u64, u64>,
35    /// Path → the cell whose ledger that path's bytes were charged to.
36    writers: BTreeMap<String, CellId>,
37    limit: u64,
38}
39
40impl QuotaTracker {
41    pub fn new() -> Self {
42        Self {
43            used: BTreeMap::new(),
44            writers: BTreeMap::new(),
45            limit: DEFAULT_QUOTA_BYTES,
46        }
47    }
48
49    /// Create a tracker with a custom byte limit.
50    ///
51    /// Used by `test-hooks` builds to exercise quota enforcement with a tiny
52    /// limit — writing the full 32 MB production limit over the 512-byte IPC
53    /// path would take ~67k messages, far too slow for a QEMU integration test.
54    #[cfg(feature = "test-hooks")]
55    pub fn with_limit(limit: u64) -> Self {
56        Self {
57            used: BTreeMap::new(),
58            writers: BTreeMap::new(),
59            limit,
60        }
61    }
62
63    /// Check whether `owner` can afford `bytes` without exceeding the quota.
64    ///
65    /// Does not mutate state — use before attempting the write, then call `charge`
66    /// only if the actual disk write succeeds.
67    pub fn can_charge(&self, owner: CellId, bytes: u64) -> bool {
68        let used = self.used.get(&owner.0).copied().unwrap_or(0);
69        used.saturating_add(bytes) <= self.limit
70    }
71
72    /// Charge `bytes` to `owner`.  Returns `Err(PermissionDenied)` if quota exceeded.
73    pub fn charge(&mut self, owner: CellId, bytes: u64) -> ViResult<()> {
74        let used = self.used.entry(owner.0).or_insert(0);
75        if *used + bytes > self.limit {
76            return Err(ViError::PermissionDenied);
77        }
78        *used += bytes;
79        Ok(())
80    }
81
82    /// Release `bytes` from `owner` (on file delete or close-after-write).
83    ///
84    /// Prefer [`Self::release_path`]: this releases from whichever cell is named,
85    /// so a caller that passes the requesting cell instead of the charged one
86    /// hands out quota that was never charged.
87    pub fn release(&mut self, owner: CellId, bytes: u64) {
88        if let Some(used) = self.used.get_mut(&owner.0) {
89            *used = used.saturating_sub(bytes);
90        }
91    }
92
93    /// Record `owner` as the cell charged for `path`, replacing any earlier
94    /// record.
95    ///
96    /// For a write that replaces the whole file: the previous contents have just
97    /// been released to the previous writer, so the new writer owns all the bytes.
98    pub fn set_writer(&mut self, path: &str, owner: CellId) {
99        self.writers.insert(path.to_string(), owner);
100    }
101
102    /// Record `owner` as the cell charged for `path` only if no cell is recorded
103    /// yet.
104    ///
105    /// For an append, which leaves earlier bytes charged where they were. See the
106    /// module note on files written by more than one cell.
107    pub fn record_writer(&mut self, path: &str, owner: CellId) {
108        self.writers.entry(path.to_string()).or_insert(owner);
109    }
110
111    /// The cell charged for `path`, if VFS charged anyone.
112    ///
113    /// `None` for a file that predates this cell's current instance (seeded at
114    /// boot, or written before a hot-swap) — nothing was charged for it here, so
115    /// nothing may be released for it either.
116    pub fn writer_of(&self, path: &str) -> Option<CellId> {
117        self.writers.get(path).copied()
118    }
119
120    /// Release `bytes` for `path` from the cell that was charged, and forget the
121    /// record.
122    ///
123    /// A path with no recorded writer releases nothing. That is deliberate: the
124    /// alternative — crediting the requesting cell — is the bug this replaces.
125    pub fn release_path(&mut self, path: &str, bytes: u64) {
126        if let Some(owner) = self.writers.remove(path) {
127            self.release(owner, bytes);
128        }
129    }
130
131    /// Return bytes used by `owner`.
132    #[allow(dead_code)] // reason: consumed by the quota-report shell builtin planned in M2.1 follow-up; exercised by test-hooks builds
133    pub fn used(&self, owner: CellId) -> u64 {
134        self.used.get(&owner.0).copied().unwrap_or(0)
135    }
136
137    /// Number of cells with recorded usage (for state-transfer sizing).
138    pub fn entry_count(&self) -> usize {
139        self.used.len()
140    }
141
142    /// Return all (cell_id, bytes_used) pairs for serialisation.
143    pub fn all_entries(&self) -> alloc::vec::Vec<(u64, u64)> {
144        self.used.iter().map(|(&k, &v)| (k, v)).collect()
145    }
146
147    /// Restore a previously serialised usage entry (called during hot-swap deserialise).
148    pub fn restore(&mut self, owner: CellId, bytes: u64) {
149        self.used.insert(owner.0, bytes);
150    }
151}
