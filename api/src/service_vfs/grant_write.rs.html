1//! Grant-backed writes through service-issued file handles.
2//!
3//! The wire carries an opaque `u64`, but only an entry in `FileHandleTable`
4//! gives it a path and an owner. Resolve that authority and re-check policy
5//! before resolving the caller's grant, then commit before acknowledging it.
6
7use alloc::string::ToString;
8
9use alloc::vec::Vec;
10
11use api::ipc::VfsResponse;
12use api::vfs_file_handles::ViVfsFileHandle;
13
14use crate::caller::Caller;
15use crate::manager::VfsManager;
16use crate::paths::{ERR_DENIED, ERR_IO, ERR_QUOTA};
17
18const MAX_GRANT_WRITE: usize = 4096;
19
20pub(crate) fn write<'a>(
21    vfs: &mut VfsManager,
22    caller: Caller,
23    cap: u64,
24    offset: u64,
25    grant: usize,
26    bytes: usize,
27) -> VfsResponse<'a> {
28    // This lookup makes an unknown cap and one held by another caller exactly
29    // the same failure. It also rejects a handle concurrently being closed or
30    // read through; no stale authority reaches grant memory.
31    let Some(path) = vfs.files.path_of(caller, ViVfsFileHandle(cap)).map(ToString::to_string)
32    else {
33        return VfsResponse::Err(ERR_DENIED);
34    };
35    if !vfs.access.can_write(caller, &path) {
36        return VfsResponse::Err(ERR_DENIED);
37    }
38    let Some((file_len, false)) = vfs.stat(&path) else {
39        return VfsResponse::Err(ERR_IO);
40    };
41    let Ok(file_len) = usize::try_from(file_len) else {
42        return VfsResponse::Err(ERR_IO);
43    };
44    let Ok(offset) = usize::try_from(offset) else {
45        return VfsResponse::Err(ERR_IO);
46    };
47    if bytes > MAX_GRANT_WRITE || offset > file_len {
48        return VfsResponse::Err(ERR_IO);
49    }
50    let Some(end) = offset.checked_add(bytes) else {
51        return VfsResponse::Err(ERR_IO);
52    };
53    let new_len = file_len.max(end);
54    let refunded_to_caller = if vfs.quota.writer_of(&path) == Some(caller.cell) {
55        file_len as u64
56    } else {
57        0
58    };
59    let net_charge = (new_len as u64).saturating_sub(refunded_to_caller);
60    if net_charge > 0 && !vfs.quota.can_charge(caller.cell, net_charge) {
61        return VfsResponse::Err(ERR_QUOTA);
62    }
63    if bytes == 0 {
64        return VfsResponse::GrantDone { bytes: 0 };
65    }
66
67    let Some((grant_ptr, grant_len)) = ostd::syscall::sys_grant_slice_with_len(grant) else {
68        return VfsResponse::Err(ERR_IO);
69    };
70    if bytes > grant_len {
71        return VfsResponse::Err(ERR_IO);
72    }
73
74    let old = vfs.read_to_vec(&path);
75    if old.len() != file_len {
76        return VfsResponse::Err(ERR_IO);
77    }
78    let mut committed = Vec::with_capacity(new_len);
79    committed.extend_from_slice(&old);
80    committed.resize(new_len, 0);
81    // SAFETY: `grant_ptr` is kernel-validated for `grant_len` bytes and `bytes`
82    // is capped by it above. `committed[offset..end]` is within its new length.
83    unsafe {
84        core::ptr::copy_nonoverlapping(grant_ptr, committed[offset..end].as_mut_ptr(), bytes);
85    }
86
87    if !vfs.write(&path, &committed) {
88        return VfsResponse::Err(ERR_IO);
89    }
90    // Backends report success only after their synchronous write commits. Charge
91    // after that commit, then acknowledge the exact visible byte count.
92    vfs.quota.release_path(&path, file_len as u64);
93    let _ = vfs.quota.charge(caller.cell, new_len as u64);
94    vfs.quota.set_writer(&path, caller.cell);
95    VfsResponse::GrantDone { bytes }
96}
