1//! Grant-backed writes through service-issued file handles.
2//!
3//! The wire carries an opaque `u64`, but only an entry in `FileHandleTable`
4//! gives it a path and an owner. Resolve that authority and re-check policy
5//! before copying through the leased GrantSlice adapter, then commit before
6//! acknowledging it.
7
8use alloc::string::ToString;
9
10use alloc::vec::Vec;
11
12use api::ipc::VfsResponse;
13use api::vfs_file_handles::ViVfsFileHandle;
14
15use crate::caller::Caller;
16use crate::manager::VfsManager;
17use crate::paths::{ERR_DENIED, ERR_IO, ERR_QUOTA};
18
19const MAX_GRANT_WRITE: usize = 4096;
20
21pub(crate) fn write<'a>(
22    vfs: &mut VfsManager,
23    caller: Caller,
24    cap: u64,
25    offset: u64,
26    grant: usize,
27    bytes: usize,
28) -> VfsResponse<'a> {
29    // This lookup makes an unknown cap and one held by another caller exactly
30    // the same failure. It also rejects a handle concurrently being closed or
31    // read through; no stale authority reaches grant memory.
32    let Some(path) = vfs
33        .files
34        .path_of(caller, ViVfsFileHandle(cap))
35        .map(ToString::to_string)
36    else {
37        return VfsResponse::Err(ERR_DENIED);
38    };
39    if crate::access::is_guest_disk_path(&path) || !vfs.access.can_write(caller, &path) {
40        return VfsResponse::Err(ERR_DENIED);
41    }
42    let Some((file_len, false)) = vfs.stat(&path) else {
43        return VfsResponse::Err(ERR_IO);
44    };
45    let Ok(file_len) = usize::try_from(file_len) else {
46        return VfsResponse::Err(ERR_IO);
47    };
48    let Ok(offset) = usize::try_from(offset) else {
49        return VfsResponse::Err(ERR_IO);
50    };
51    if bytes > MAX_GRANT_WRITE || offset > file_len {
52        return VfsResponse::Err(ERR_IO);
53    }
54    let Some(end) = offset.checked_add(bytes) else {
55        return VfsResponse::Err(ERR_IO);
56    };
57    let new_len = file_len.max(end);
58    let refunded_to_caller = if vfs.quota.writer_of(&path) == Some(caller.cell) {
59        file_len as u64
60    } else {
61        0
62    };
63    let net_charge = (new_len as u64).saturating_sub(refunded_to_caller);
64    if net_charge > 0 && !vfs.quota.can_charge(caller.cell, net_charge) {
65        return VfsResponse::Err(ERR_QUOTA);
66    }
67    if bytes == 0 {
68        return VfsResponse::GrantDone { bytes: 0 };
69    }
70
71    let old = vfs.read_to_vec(&path);
72    if old.len() != file_len {
73        return VfsResponse::Err(ERR_IO);
74    }
75    let mut committed = Vec::with_capacity(new_len);
76    committed.extend_from_slice(&old);
77    committed.resize(new_len, 0);
78    // GrantSlice resolution and VFS lease publication are one kernel grant-table
79    // transaction. Copy into only the requested file range; a missing or short
80    // grant is an I/O error and the backend remains untouched.
81    if ostd::syscall::sys_grant_copy_to_slice(grant, &mut committed[offset..end]) != Some(bytes) {
82        return VfsResponse::Err(ERR_IO);
83    }
84
85    if !vfs.write(&path, &committed) {
86        return VfsResponse::Err(ERR_IO);
87    }
88    // Backends report success only after their synchronous write commits. Charge
89    // after that commit, then acknowledge the exact visible byte count.
90    vfs.quota.release_path(&path, file_len as u64);
91    let _ = vfs.quota.charge(caller.cell, new_len as u64);
92    vfs.quota.set_writer(&path, caller.cell);
93    VfsResponse::GrantDone { bytes }
94}
