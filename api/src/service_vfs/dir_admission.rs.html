1//! Settling what a cell inherited before anything it sent is served.
2//!
3//! The kernel states provenance — which cell named a set of handle values for
4//! this one — and nothing more. It cannot tell whether the spawner was entitled
5//! to them, because a cell acquires handles from this service after it starts
6//! and only this service knows what it holds. So the record is pulled here, once
7//! per cell generation, and checked against the table by
8//! [`crate::dirs::DirTable::bind_inherited`].
9//!
10//! Ordering matters more than it looks. Both answers this produces — what the
11//! cell holds, and whether it may still name a path — have to be in place before
12//! the cell's first request, or a cell whose spawner placed it in the capability
13//! world would get one path-string operation in ahead of the seal.
14
15use api::dir_attestation::DIR_ATTESTATION_LEN;
16
17use crate::caller::Caller;
18use crate::dirs::bind::BindOutcome;
19use crate::dirs::lifecycle::Contact;
20use crate::manager::VfsManager;
21
22/// Apply generation-replacement cleanup before any attestation syscall.
23pub(crate) fn prepare_contact(vfs: &mut VfsManager, caller: Caller) -> bool {
24    let contact = vfs.dirs.on_contact(caller);
25    let Contact::NeedsAttestation {
26        replaced_owner,
27        revoked_dir_ids,
28    } = contact
29    else {
30        return false;
31    };
32    if let Some(owner) = replaced_owner {
33        let _ = vfs.files.purge_owner(owner);
34    }
35    let _ = vfs.files.revoke_by_parent_dirs(&revoked_dir_ids);
36    // Marked before the query, not after: one attempt per cell generation either
37    // way, so a cell cannot make the service re-query by failing repeatedly.
38    vfs.dirs.mark_attested(caller);
39    true
40}
41
42/// Settle what the kernel says about `caller`. Idempotent per cell generation.
43pub fn admit(vfs: &mut VfsManager, caller: Caller) {
44    if !prepare_contact(vfs, caller) {
45        return;
46    }
47    let mut buf = [0u8; DIR_ATTESTATION_LEN];
48    let Some(record) = ostd::syscall::sys_query_dir_handles(caller.cell.0, &mut buf) else {
49        return;
50    };
51    match vfs.dirs.bind_inherited(caller, &record) {
52        BindOutcome::NothingNamed => {}
53        // The record could not be attributed to this caller, so there is no set
54        // to act on. The cell keeps the authority an unmigrated cell has; it
55        // gains nothing.
56        BindOutcome::NotThisCell => {
57            ostd::io::println("[vfs] discarding a provenance record for a different cell");
58        }
59        BindOutcome::Bound(_) => {
60            vfs.dirs.seal(caller);
61        }
62        BindOutcome::Refused => {
63            vfs.dirs.seal(caller);
64            ostd::io::println(
65                "[vfs] refused an inherited handle set: the spawner did not hold all of it",
66            );
67        }
68    }
69}
