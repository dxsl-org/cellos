1//! First trust (Phase 09 S9-4c; abi-vfs-telemetry §4; D79-11/12).
2//!
3//! `TrustChallenge`: for a host and port the store does not hold, the service records a
4//! challenge bound to the caller principal, the exact key, the store's current
5//! observation and a 60-second deadline, and registers its immutable descriptor with the
6//! input owner (`RegisterTrustChallenge`). The input owner, not the caller, then runs the
7//! confirmation on the caller's terminal session.
8//!
9//! `TrustCommit`: the caller's own challenge only. The owner-attested decision
10//! (`CollectTrustDecision`) must accept and match every binding of the record (challenge,
11//! caller, session and epoch, descriptor SHA-256); then the store is re-observed and,
12//! still Unknown, extended by one `interactive` line and published through the D03
13//! machinery against that observation. Another writer's line for the same host and port
14//! wins: the same key answers KnownExact, a different one Mismatch, and nothing is ever
15//! overwritten. A publication reports `Committed` (synced, not crash-proof; S9-4a).
16//! Every final answer ends the challenge.

44/// Challenge lifetime: the input owner's ceiling, never extended.
45const LIFETIME_NS: u64 = tr::EXPIRY_MAX_NS;
46/// Open challenges per caller principal, and in total (§4: four per worker).
47const MAX_PER_CALLER: usize = 1;
48const MAX_CHALLENGES: usize = 4;
49const TRUST_DIR: &str = "/srv/.cellos-trust";
50
51struct Challenge {
52    challenge: Ref,
53    owner: Caller,
54    host: String,
55    port: u16,
56    algorithm: String,
57    key: Vec<u8>,
58    /// The store as it was when the challenge was issued.
59    observation: Observation,
60    expiry: u64,
61    confirmation: Ref,
62    session: Ref,
63    epoch: u64,
64    /// SHA-256 of the descriptor registered with the input owner.
65    digest: [u8; 32],
66}

91/// `SHA256:` and the unpadded base64 of the blob's digest, exactly as `vissh` prints it.
92fn fingerprint(key: &[u8]) -> String {
93    let mut s = String::from("SHA256:");
94    base64_encode(&Sha256::digest(key), false, &mut s);
95    s
96}

98/// The store's observation and its answer for this host, port and key.
99fn current(
100    vfs: &VfsManager,
101    host: &str,
102    port: u16,
103    algorithm: &str,
104    key: &[u8],
105) -> Option<(Observation, Vec<u8>, u8)> {
106    match vfs.observe(STORE_PATH)? {
107        Observation::Absent => Some((Observation::Absent, Vec::new(), ssh_trust::UNKNOWN)),
108        observation @ Observation::Present {
109            size,
110            regular: true,
111            ..
112        } => {
113            let bytes = vfs.read_to_vec(STORE_PATH);
114            if bytes.len() as u64 != size {
115                return None;
116            }
117            let code = lookup(&bytes, host, port, algorithm, key);
118            Some((observation, bytes, code))
119        }
120        Observation::Present { .. } => None,
121    }
122}

124#[allow(clippy::too_many_arguments)]
125pub fn challenge(
126    vfs: &mut VfsManager,
127    caller: Caller,
128    host: &str,
129    port: u16,
130    algorithm: &str,
131    key: &[u8],
132    session: Ref,
133    epoch: u64,
134) -> VfsResponse<'static> {
135    if caller.flags & CALLER_FLAG_NET_CONNECT == 0 {
136        return VfsResponse::Trust(ssh_trust::DENIED);
137    }
138    // Only what a store line can hold is ever offered to the user.
139    if !canonical_host(host)
140        || port == 0
141        || algorithm != "ssh-ed25519"
142        || key.is_empty()
143        || key.len() > MAX_BLOB
144        || !blob_names(key, algorithm)
145        || session.is_zero()
146    {
147        return VfsResponse::Err(0xFF);
148    }
149    let Some((observation, _, code)) = current(vfs, host, port, algorithm, key) else {
150        return VfsResponse::Trust(ssh_trust::UNAVAILABLE);
151    };
152    // KnownExact needs no prompt; a changed key never prompts.
153    if code != ssh_trust::UNKNOWN {
154        return VfsResponse::Trust(code);
155    }
156    let Ok(now) = owner::now_ns() else {
157        return VfsResponse::Trust(ssh_trust::UNAVAILABLE);
158    };
159    let table = &mut vfs.trust_challenges;
160    // A caller's own expired challenge gives way; any other is released one lifetime later.
161    table.records.retain(|r| {
162        now < r.expiry.saturating_add(LIFETIME_NS) && !(r.owner == caller && now >= r.expiry)
163    });
164    if table.records.iter().filter(|r| r.owner == caller).count() >= MAX_PER_CALLER
165        || table.records.len() >= MAX_CHALLENGES
166    {
167        return VfsResponse::Err(ERR_QUOTA);
168    }
169    let challenge = Ref {
170        id: table.next,
171        generation: 1,
172    };
173    let Some(next) = table.next.checked_add(1) else {
174        return VfsResponse::Err(ERR_QUOTA);
175    };
176    let expiry = now.saturating_add(LIFETIME_NS);
177    let fingerprint = fingerprint(key);
178    let descriptor = TrustChallengeV1 {
179        challenge,
180        target: principal(caller),
181        expiry_ns: expiry,
182        prompt: TrustPromptV1 {
183            port,
184            host,
185            algorithm,
186            fingerprint: &fingerprint,
187        },
188    };
189    let mut bytes = [0u8; TrustChallengeV1::MAX_LEN];
190    let Ok(len) = descriptor.encode(&mut bytes) else {
191        return VfsResponse::Err(0xFF);
192    };
193    let digest: [u8; 32] = Sha256::digest(&bytes[..len]).into();
194    let Ok(confirmation) = owner::register(&descriptor) else {
195        return VfsResponse::Trust(ssh_trust::UNAVAILABLE);
196    };
197    table.next = next;
198    table.records.push(Challenge {
199        challenge,
200        owner: caller,
201        host: String::from(host),
202        port,
203        algorithm: String::from(algorithm),
204        key: key.to_vec(),
205        observation,
206        expiry,
207        confirmation,
208        session,
209        epoch,
210        digest,
211    });
212    VfsResponse::TrustChallenged {
213        confirmation,
214        challenge,
215    }
216}

218pub fn commit(vfs: &mut VfsManager, caller: Caller, challenge: Ref) -> VfsResponse<'static> {
219    if caller.flags & CALLER_FLAG_NET_CONNECT == 0 {
220        return VfsResponse::Trust(ssh_trust::DENIED);
221    }
222    // Only the caller's own challenge: a guessed or another caller's ref is refused.
223    let Some(n) = vfs
224        .trust_challenges
225        .records
226        .iter()
227        .position(|r| r.challenge == challenge && r.owner == caller)
228    else {
229        return VfsResponse::Trust(ssh_trust::DENIED);
230    };
231    let decision = match owner::collect(vfs.trust_challenges.records[n].confirmation) {
232        Ok(decision) => decision,
233        // Uncertain transport: the decision is still the owner's to give.
234        Err(Status::Indeterminate | Status::WouldBlock | Status::TimedOut) => {
235            return VfsResponse::Trust(ssh_trust::PENDING)
236        }
237        Err(_) => {
238            vfs.trust_challenges.records.swap_remove(n);
239            return VfsResponse::Trust(ssh_trust::UNAVAILABLE);
240        }
241    };
242    if !decision.is_final() {
243        return VfsResponse::Trust(ssh_trust::PENDING);
244    }
245    let record = vfs.trust_challenges.records.swap_remove(n);
246    let code = match decision.decision {
247        tr::DECISION_ACCEPT => {
248            // The owner's reply must bind exactly this record: challenge, caller, the
249            // session and epoch the caller named, and the descriptor this service registered.
250            let bound = decision.challenge == record.challenge
251                && decision.target == principal(caller)
252                && decision.session == record.session
253                && decision.epoch == record.epoch
254                && decision.descriptor_sha256 == record.digest;
255            match owner::now_ns() {
256                _ if !bound => ssh_trust::REFUSED,
257                Ok(now) if now < record.expiry => persist(vfs, &record),
258                _ => ssh_trust::EXPIRED,
259            }
260        }
261        tr::DECISION_EXPIRED => ssh_trust::EXPIRED,
262        _ => ssh_trust::REFUSED,
263    };
264    VfsResponse::Trust(code)
265}

267/// Add the record's line to the store: at most two publication attempts, each against
268/// the observation it was built from.
269fn persist(vfs: &mut VfsManager, r: &Challenge) -> u8 {
270    let mut expected = r.observation;
271    for _ in 0..2 {
272        let Some((observation, current, code)) =
273            current(vfs, &r.host, r.port, &r.algorithm, &r.key)
274        else {
275            return ssh_trust::UNAVAILABLE;
276        };
277        // Changed since the challenge (or the last attempt): another writer may have
278        // stored this host and port meanwhile; its line is never overwritten.
279        if observation != expected && code != ssh_trust::UNKNOWN {
280            return code;
281        }
282        let mut bytes = if observation == Observation::Absent {
283            String::from(HEADER) + "\n"
284        } else {
285            match String::from_utf8(current) {
286                Ok(text) => text,
287                Err(_) => return ssh_trust::UNAVAILABLE,
288            }
289        };
290        bytes.push_str(&r.host);
291        bytes.push_str(&alloc::format!(" {} {} ", r.port, r.algorithm));
292        base64_encode(&r.key, true, &mut bytes);
293        bytes.push_str(" interactive\n");
294        // The store's own bounds (entries, blob, total size) hold for the new revision too.
295        if parse(bytes.as_bytes()).is_err() {
296            return ssh_trust::UNAVAILABLE;
297        }
298        match vfs.stat(TRUST_DIR) {
299            Some((_, true)) => {}
300            None if vfs.mkdir(TRUST_DIR) => {}
301            _ => return ssh_trust::UNAVAILABLE,
302        }
303        let Some(expect) = observation.expect() else {
304            return ssh_trust::UNAVAILABLE;
305        };
306        match crate::save::publish_owned(vfs, STORE_PATH, bytes.as_bytes(), expect) {
307            outcome if outcome.published().is_some() => return ssh_trust::KNOWN_EXACT,
308            SaveOutcome::Conflict => expected = observation,
309            _ => return ssh_trust::UNAVAILABLE,
310        }
311    }
312    ssh_trust::UNAVAILABLE
