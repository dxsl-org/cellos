1//! First trust (Phase 09 S9-4c; abi-vfs-telemetry §4; D79-11/12).
2//!
3//! `TrustChallenge`: for a host and port the store does not hold, the service records a
4//! challenge bound to the caller principal, the exact key, the store's current
5//! observation and a 60-second deadline, and registers its immutable descriptor with the
6//! input owner (`RegisterTrustChallenge`). The input owner, not the caller, then runs the
7//! confirmation on the caller's terminal session.
8//!
9//! `TrustCommit`: the caller's own challenge only. The owner-attested decision
10//! (`CollectTrustDecision`) must accept and match every binding of the record (challenge,
11//! caller, session and epoch, descriptor SHA-256); then the store is re-observed and,
12//! still Unknown, extended by one `interactive` line and published through the D03
13//! machinery against that observation. Another writer's line for the same host and port
14//! wins: the same key answers KnownExact, a different one Mismatch, and nothing is ever
15//! overwritten. A publication reports `Committed` (synced, not crash-proof; S9-4a).
16//! Every final answer ends the challenge.

39/// Challenge lifetime: the input owner's ceiling, never extended.
40const LIFETIME_NS: u64 = tr::EXPIRY_MAX_NS;
41/// Open challenges per caller principal, and in total (§4: four per worker).
42const MAX_PER_CALLER: usize = 1;
43const MAX_CHALLENGES: usize = 4;
44const TRUST_DIR: &str = "/srv/.cellos-trust";
45
46struct Challenge {
47    challenge: Ref,
48    owner: Caller,
49    host: String,
50    port: u16,
51    algorithm: String,
52    key: Vec<u8>,
53    /// The store as it was when the challenge was issued.
54    observation: Observation,
55    expiry: u64,
56    confirmation: Ref,
57    session: Ref,
58    epoch: u64,
59    /// SHA-256 of the descriptor registered with the input owner.
60    digest: [u8; 32],
61}

63/// Open challenges (not carried across a VFS restart: the input owner binds the old
64/// VFS principal, so its decisions can never be collected by a successor).
65pub struct Challenges {
66    records: Vec<Challenge>,
67    next: u64,
68}

80/// `SHA256:` and the unpadded base64 of the blob's digest, exactly as `vissh` prints it.
81fn fingerprint(key: &[u8]) -> String {
82    let mut s = String::from("SHA256:");
83    base64_encode(&Sha256::digest(key), false, &mut s);
84    s
85}

87/// The store's observation and its answer for this host, port and key.
88fn current(vfs: &VfsManager, host: &str, port: u16, algorithm: &str, key: &[u8]) -> Option<(Observation, Vec<u8>, u8)> {
89    match vfs.observe(STORE_PATH)? {
90        Observation::Absent => Some((Observation::Absent, Vec::new(), ssh_trust::UNKNOWN)),
91        observation @ Observation::Present { size, regular: true, .. } => {
92            let bytes = vfs.read_to_vec(STORE_PATH);
93            if bytes.len() as u64 != size {
94                return None;
95            }
96            let code = lookup(&bytes, host, port, algorithm, key);
97            Some((observation, bytes, code))
98        }
99        Observation::Present { .. } => None,
100    }
101}

103#[allow(clippy::too_many_arguments)]
104pub fn challenge(
105    vfs: &mut VfsManager, caller: Caller, host: &str, port: u16, algorithm: &str, key: &[u8], session: Ref, epoch: u64,
106) -> VfsResponse<'static> {
107    if caller.flags & CALLER_FLAG_NET_CONNECT == 0 {
108        return VfsResponse::Trust(ssh_trust::DENIED);
109    }
110    // Only what a store line can hold is ever offered to the user.
111    if !canonical_host(host) || port == 0 || algorithm != "ssh-ed25519" || key.is_empty() || key.len() > MAX_BLOB
112        || !blob_names(key, algorithm) || session.is_zero()
113    {
114        return VfsResponse::Err(0xFF);
115    }
116    let Some((observation, _, code)) = current(vfs, host, port, algorithm, key) else {
117        return VfsResponse::Trust(ssh_trust::UNAVAILABLE);
118    };
119    // KnownExact needs no prompt; a changed key never prompts.
120    if code != ssh_trust::UNKNOWN {
121        return VfsResponse::Trust(code);
122    }
123    let Ok(now) = owner::now_ns() else { return VfsResponse::Trust(ssh_trust::UNAVAILABLE) };
124    let table = &mut vfs.trust_challenges;
125    // A caller's own expired challenge gives way; any other is released one lifetime later.
126    table.records.retain(|r| now < r.expiry.saturating_add(LIFETIME_NS) && !(r.owner == caller && now >= r.expiry));
127    if table.records.iter().filter(|r| r.owner == caller).count() >= MAX_PER_CALLER || table.records.len() >= MAX_CHALLENGES {
128        return VfsResponse::Err(ERR_QUOTA);
129    }
130    let challenge = Ref { id: table.next, generation: 1 };
131    let Some(next) = table.next.checked_add(1) else { return VfsResponse::Err(ERR_QUOTA) };
132    let expiry = now.saturating_add(LIFETIME_NS);
133    let fingerprint = fingerprint(key);
134    let descriptor = TrustChallengeV1 {
135        challenge, target: principal(caller), expiry_ns: expiry,
136        prompt: TrustPromptV1 { port, host, algorithm, fingerprint: &fingerprint },
137    };
138    let mut bytes = [0u8; TrustChallengeV1::MAX_LEN];
139    let Ok(len) = descriptor.encode(&mut bytes) else { return VfsResponse::Err(0xFF) };
140    let digest: [u8; 32] = Sha256::digest(&bytes[..len]).into();
141    let Ok(confirmation) = owner::register(&descriptor) else { return VfsResponse::Trust(ssh_trust::UNAVAILABLE) };
142    table.next = next;
143    table.records.push(Challenge {
144        challenge, owner: caller, host: String::from(host), port, algorithm: String::from(algorithm), key: key.to_vec(),
145        observation, expiry, confirmation, session, epoch, digest,
146    });
147    VfsResponse::TrustChallenged { confirmation, challenge }
148}

150pub fn commit(vfs: &mut VfsManager, caller: Caller, challenge: Ref) -> VfsResponse<'static> {
151    if caller.flags & CALLER_FLAG_NET_CONNECT == 0 {
152        return VfsResponse::Trust(ssh_trust::DENIED);
153    }
154    // Only the caller's own challenge: a guessed or another caller's ref is refused.
155    let Some(n) = vfs.trust_challenges.records.iter().position(|r| r.challenge == challenge && r.owner == caller) else {
156        return VfsResponse::Trust(ssh_trust::DENIED);
157    };
158    let decision = match owner::collect(vfs.trust_challenges.records[n].confirmation) {
159        Ok(decision) => decision,
160        // Uncertain transport: the decision is still the owner's to give.
161        Err(Status::Indeterminate | Status::WouldBlock | Status::TimedOut) => return VfsResponse::Trust(ssh_trust::PENDING),
162        Err(_) => {
163            vfs.trust_challenges.records.swap_remove(n);
164            return VfsResponse::Trust(ssh_trust::UNAVAILABLE);
165        }
166    };
167    if !decision.is_final() {
168        return VfsResponse::Trust(ssh_trust::PENDING);
169    }
170    let record = vfs.trust_challenges.records.swap_remove(n);
171    let code = match decision.decision {
172        tr::DECISION_ACCEPT => {
173            // The owner's reply must bind exactly this record: challenge, caller, the
174            // session and epoch the caller named, and the descriptor this service registered.
175            let bound = decision.challenge == record.challenge && decision.target == principal(caller)
176                && decision.session == record.session && decision.epoch == record.epoch
177                && decision.descriptor_sha256 == record.digest;
178            match owner::now_ns() {
179                _ if !bound => ssh_trust::REFUSED,
180                Ok(now) if now < record.expiry => persist(vfs, &record),
181                _ => ssh_trust::EXPIRED,
182            }
183        }
184        tr::DECISION_EXPIRED => ssh_trust::EXPIRED,
185        _ => ssh_trust::REFUSED,
186    };
187    VfsResponse::Trust(code)
188}

190/// Add the record's line to the store: at most two publication attempts, each against
191/// the observation it was built from.
192fn persist(vfs: &mut VfsManager, r: &Challenge) -> u8 {
193    let mut expected = r.observation;
194    for _ in 0..2 {
195        let Some((observation, current, code)) = current(vfs, &r.host, r.port, &r.algorithm, &r.key) else {
196            return ssh_trust::UNAVAILABLE;
197        };
198        // Changed since the challenge (or the last attempt): another writer may have
199        // stored this host and port meanwhile; its line is never overwritten.
200        if observation != expected && code != ssh_trust::UNKNOWN {
201            return code;
202        }
203        let mut bytes = if observation == Observation::Absent { String::from(HEADER) + "\n" } else {
204            match String::from_utf8(current) { Ok(text) => text, Err(_) => return ssh_trust::UNAVAILABLE }
205        };
206        bytes.push_str(&r.host);
207        bytes.push_str(&alloc::format!(" {} {} ", r.port, r.algorithm));
208        base64_encode(&r.key, true, &mut bytes);
209        bytes.push_str(" interactive\n");
210        // The store's own bounds (entries, blob, total size) hold for the new revision too.
211        if parse(bytes.as_bytes()).is_err() {
212            return ssh_trust::UNAVAILABLE;
213        }
214        match vfs.stat(TRUST_DIR) {
215            Some((_, true)) => {}
216            None if vfs.mkdir(TRUST_DIR) => {}
217            _ => return ssh_trust::UNAVAILABLE,
218        }
219        let Some(expect) = observation.expect() else { return ssh_trust::UNAVAILABLE };
220        match crate::save::publish_owned(vfs, STORE_PATH, bytes.as_bytes(), expect) {
221            outcome if outcome.published().is_some() => return ssh_trust::KNOWN_EXACT,
222            SaveOutcome::Conflict => expected = observation,
223            _ => return ssh_trust::UNAVAILABLE,
224        }
225    }
226    ssh_trust::UNAVAILABLE
227}
