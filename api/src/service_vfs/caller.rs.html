1//! Who VFS is serving, according to the kernel.
2//!
3//! VFS used to build a `CellId` out of the sender tid. That was wrong twice: a
4//! thread reports its own tid while belonging to its parent cell, so its quota
5//! and its handles landed under a `CellId` matching no cell at all; and it made
6//! identity a property of the transport rather than something vouched for. Both
7//! disappear once the kernel states the identity — see `api::caller_identity`.
8//!
9//! Every authorization decision and every quota charge in this cell reads a
10//! [`Caller`], and a [`Caller`] can only be built from an attested identity.
11
12use api::caller_identity::CallerIdentity;
13use types::CellId;
14
15/// A caller VFS is willing to act for.
16///
17/// Two callers are the same principal only when both the cell and its generation
18/// match: a respawned cell that reused a dead cell's `CellId` is a *different*
19/// principal, and must not inherit the dead cell's handles, pending reads, or
20/// quota credit.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub struct Caller {
23    /// Owning cell of the calling task — the cell, never the thread.
24    pub cell: CellId,
25    /// Cell epoch from the attestation. Distinguishes a successor cell from the
26    /// one it replaced.
27    pub generation: u64,
28}
29
30impl Caller {
31    /// Adopt a kernel-attested identity.
32    ///
33    /// The only constructor outside tests, which is the point: there is no way to
34    /// obtain a `Caller` from request bytes.
35    pub fn from_attested(id: CallerIdentity) -> Self {
36        Self {
37            cell: CellId(id.cell_id),
38            generation: id.generation,
39        }
40    }
41
42    /// Whether durable per-caller state (open handles, pending reads) may be
43    /// recorded against this caller.
44    ///
45    /// Requires an attested generation: without one, a later caller under the
46    /// same `CellId` would be indistinguishable from this one, which is exactly
47    /// the confusion the generation exists to prevent. Paths that only consult
48    /// path-keyed policy do not need this.
49    pub fn may_own_state(&self) -> bool {
50        self.cell.0 != 0 && self.generation != 0
51    }
52}
53
54#[cfg(test)]
55mod tests {
56    use super::*;
57
58    #[test]
59    fn attested_identity_becomes_the_cell_not_the_sending_thread() {
60        let caller = Caller::from_attested(CallerIdentity {
61            cell_id: 4,
62            generation: 9,
63            sender_tid: 77, // a thread of cell 4
64        });
65        assert_eq!(caller.cell, CellId(4));
66        assert!(caller.may_own_state());
67    }
68
69    #[test]
70    fn same_cell_different_generation_is_a_different_principal() {
71        let old = Caller {
72            cell: CellId(4),
73            generation: 1,
74        };
75        let respawned = Caller {
76            cell: CellId(4),
77            generation: 2,
78        };
79        assert_ne!(old, respawned);
80    }
81
82    #[test]
83    fn unattested_generation_may_not_own_state() {
84        let caller = Caller {
85            cell: CellId(4),
86            generation: 0,
87        };
88        assert!(!caller.may_own_state());
89    }
