1//! Who VFS is serving, according to the kernel.
2//!
3//! VFS used to build a `CellId` out of the sender tid. That was wrong twice: a
4//! thread reports its own tid while belonging to its parent cell, so its quota
5//! and its handles landed under a `CellId` matching no cell at all; and it made
6//! identity a property of the transport rather than something vouched for. Both
7//! disappear once the kernel states the identity — see `api::caller_identity`.
8//!
9//! Every authorization decision and every quota charge in this cell reads a
10//! [`Caller`], and a [`Caller`] can only be built from an attested identity.
11
12use api::caller_identity::CallerIdentity;
13use types::CellId;
14
15/// A caller VFS is willing to act for.
16///
17/// Two callers are the same principal only when both the cell and its generation
18/// match: a respawned cell that reused a dead cell's `CellId` is a *different*
19/// principal, and must not inherit the dead cell's handles, pending reads, or
20/// quota credit.
21#[derive(Debug, Clone, Copy)]
22pub struct Caller {
23    /// Owning cell of the calling task — the cell, never the thread.
24    pub cell: CellId,
25    /// Cell epoch from the attestation. Distinguishes a successor cell from the
26    /// one it replaced.
27    pub generation: u64,
28    /// Sending thread the kernel attested for this request.
29    pub sender_tid: u64,
30}
31
32impl PartialEq for Caller {
33    fn eq(&self, other: &Self) -> bool {
34        self.cell == other.cell && self.generation == other.generation
35    }
36}
37
38impl Eq for Caller {}
39
40impl Caller {
41    /// Build a cell-generation principal when no current sender thread is part
42    /// of the authority being modeled.
43    pub const fn principal(cell: CellId, generation: u64) -> Self {
44        Self {
45            cell,
46            generation,
47            sender_tid: 0,
48        }
49    }
50
51    /// Adopt a kernel-attested identity.
52    ///
53    /// The only constructor outside tests, which is the point: there is no way to
54    /// obtain a `Caller` from request bytes.
55    pub fn from_attested(id: CallerIdentity) -> Self {
56        Self {
57            cell: CellId(id.cell_id),
58            generation: id.generation,
59            sender_tid: id.sender_tid,
60        }
61    }
62
63    /// Whether durable per-caller state (open handles, pending reads) may be
64    /// recorded against this caller.
65    ///
66    /// Requires an attested generation: without one, a later caller under the
67    /// same `CellId` would be indistinguishable from this one, which is exactly
68    /// the confusion the generation exists to prevent. Paths that only consult
69    /// path-keyed policy do not need this.
70    pub fn may_own_state(&self) -> bool {
71        self.cell.0 != 0 && self.generation != 0
72    }
73}

75#[cfg(test)]
76mod tests {
77    use super::*;
78
79    #[test]
80    fn attested_identity_becomes_the_cell_not_the_sending_thread() {
81        let caller = Caller::from_attested(CallerIdentity {
82            cell_id: 4,
83            generation: 9,
84            sender_tid: 77, // a thread of cell 4
85        });
86        assert_eq!(caller.cell, CellId(4));
87        assert!(caller.may_own_state());
88    }
89
90    #[test]
91    fn same_cell_different_generation_is_a_different_principal() {
92        let old = Caller {
93            cell: CellId(4),
94            generation: 1,
95            sender_tid: 70,
96        };
97        let respawned = Caller {
98            cell: CellId(4),
99            generation: 2,
100            sender_tid: 71,
101        };
102        assert_ne!(old, respawned);
103    }
104
105    #[test]
106    fn same_cell_generation_different_threads_is_the_same_principal() {
107        let left = Caller {
108            cell: CellId(4),
109            generation: 2,
110            sender_tid: 70,
111        };
112        let right = Caller {
113            cell: CellId(4),
114            generation: 2,
115            sender_tid: 71,
116        };
117        assert_eq!(left, right);
118        assert!(left.may_own_state());
119    }
120
121    #[test]
122    fn unattested_generation_may_not_own_state() {
123        let caller = Caller {
124            cell: CellId(4),
125            generation: 0,
126            sender_tid: 77,
127        };
128        assert!(!caller.may_own_state());
129    }
