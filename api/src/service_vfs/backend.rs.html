1//! `FsBackend` — the contract every mounted filesystem implements.
2//!
3//! Contract: all methods receive the ABSOLUTE VFS path (e.g. `/data/log.txt`),
4//! never a pre-stripped relative path. Each backend owns the stripping of its
5//! mount prefix because rejection semantics are backend-specific (FAT rejects
6//! empty-rel writes and `..` components; RamFS resolves the whole tree). This
7//! mirrors the pre-MountTable dispatch exactly — do not centralize stripping
8//! without re-validating every path edge case.
9//!
10//! Mutability: read ops take `&self`; mutating ops take `&mut self` because
11//! RamFS mutates its tree (FAT uses fatfs interior mutability and ignores it).
12
13use alloc::vec::Vec;
14use api::ipc::vfs_dir_page::{push_entry, EntryKind};
15
16/// Cursors a backend hands out stay below this bound. The bits above it belong
17/// to the layers that union several sources into one listing:
18/// `BinOverlay` uses bit 30 and `VfsManager` bit 31.
19pub(crate) const BACKEND_CURSOR_LIMIT: u32 = 1 << 30;
20
21/// One page written by [`FsBackend::list_page`].
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub struct ListPage {
24    /// Bytes of packed `api::ipc::vfs_dir_page` entries written to `out`.
25    pub written: usize,
26    /// Cursor of the first entry that did not fit; `None` when this page ends
27    /// the listing.
28    pub next: Option<u32>,
29}
30
31/// Page an ordered entry sequence: skip `start` entries, then pack whole
32/// entries into `out` until one does not fit.
33///
34/// `None` (never a short page) when iteration fails, a name is not a valid
35/// directory component, `start` lies past the end (a stale cursor), or the
36/// sequence outgrows [`BACKEND_CURSOR_LIMIT`].
37pub(crate) fn fill_page<N: AsRef<[u8]>>(
38    entries: impl Iterator<Item = Result<(EntryKind, N), ()>>,
39    start: u32,
40    out: &mut [u8],
41) -> Option<ListPage> {
42    let mut index = 0u32;
43    let mut written = 0;
44    for entry in entries {
45        let (kind, name) = entry.ok()?;
46        if index >= start {
47            match push_entry(out, written, kind, name.as_ref()) {
48                Ok(Some(end)) => written = end,
49                Ok(None) => return Some(ListPage { written, next: Some(index) }),
50                Err(_) => return None,
51            }
52        }
53        index += 1;
54        if index >= BACKEND_CURSOR_LIMIT {
55            return None;
56        }
57    }
58    (index >= start).then_some(ListPage { written, next: None })
59}

61/// Operational state only: healthy does not certify a durability barrier.
62#[derive(Clone, Copy, Debug, PartialEq, Eq)]
63pub enum BackendHealth {
64    Ready,
65    Unavailable,
66    Poisoned,
67}

69/// `Send` bound (Law 7): backends live inside the `GLOBAL_VFS` static Mutex,
70/// which requires its contents to be `Send` for the static to be `Sync`.
71pub trait FsBackend: Send {
72    fn health(&self) -> BackendHealth {
73        BackendHealth::Ready
74    }
75    /// SAS zero-copy pointer to the file's bytes. Only in-memory backends
76    /// return `Some` — pointers must stay valid while the VFS cell lives.
77    /// Disk backends return `None`; callers fall back to the copy path.
78    fn get_file_ptr(&self, path: &str) -> Option<(usize, usize)>;
79
80    /// List directory entries as `d:name\n` / `f:name\n` lines into `out`.
81    /// Returns bytes written; 0 when the path is missing or not a directory.
82    fn list(&self, path: &str, out: &mut [u8]) -> usize;
83
84    /// Page the entries of directory `path` into `out`, starting at entry
85    /// index `start` (see [`fill_page`]). Entry order is the backend's own and
86    /// must be stable while the directory is unchanged.
87    ///
88    /// `None` when `path` is missing or not a directory, iteration failed, a
89    /// name is not a valid component, or `start` is stale. Never a silent stop.
90    fn list_page(&self, path: &str, start: u32, out: &mut [u8]) -> Option<ListPage>;
91
92    /// `(size, is_dir)`, or `None` when the path does not exist.
93    fn stat(&self, path: &str) -> Option<(u64, bool)>;
94
95    /// File size for quota accounting. 0 when absent (callers charge net delta).
96    fn file_size(&self, path: &str) -> u64;
97
98    /// Whole-file copy for the async-read handle table. Empty when absent.
99    fn read_to_vec(&self, path: &str) -> Vec<u8>;
100
101    /// Create-or-truncate write. Authorization is the dispatcher's job
102    /// (AccessTable); backends may still enforce structural rules.
103    fn write(&mut self, path: &str, content: &[u8]) -> bool;
104    /// Positional bounded read. Backends that do not support it return 0.
105    fn read_at(&self, path: &str, offset: u64, buf: &mut [u8]) -> usize {
106        let _ = (path, offset, buf);
107        0
108    }
109
110    /// Positional bounded write. Backends that do not support it return false.
111    fn write_at(&mut self, path: &str, offset: u64, content: &[u8]) -> bool {
112        let _ = (path, offset, content);
113        false
114    }
115
116    /// Flush dirty pages for a path to the block device.
117    fn sync(&mut self, path: &str) -> bool {
118        let _ = path;
119        false
120    }
121
122    fn append(&mut self, path: &str, content: &[u8]) -> bool;
123    /// mkdir -p semantics on FAT; single-level create on RamFS.
124    fn mkdir(&mut self, path: &str) -> bool;
125
126    /// Remove an EMPTY directory (POSIX ENOTDIR/ENOTEMPTY semantics).
127    fn rmdir(&mut self, path: &str) -> bool;
128
129    /// Remove a regular FILE (directories are rejected).
130    fn unlink(&mut self, path: &str) -> bool;
131
132    /// Recursive delete (`rm -r`). Backends that do not support it return false.
133    fn rmdir_recursive(&mut self, path: &str) -> bool;
134
135    /// Atomic no-replace rename from `old` to `new`.
136    ///
137    /// Backends that do not support it return false; the `/srv` backend
138    /// (CellosFS Native) implements it.
139    fn rename_no_replace(&mut self, old: &str, new: &str) -> bool {
140        let _ = (old, new);
141        false
142    }
143
144    // ── D03 conditional save. Defaults: unsupported (FAT32, bootfs, RamFS). ──
145
146    /// Staging object path for transaction `id`, inside this backend; `None`
147    /// when the backend has no conditional publication.
148    fn stage_path(&self, id: u32) -> Option<alloc::string::String> {
149        let _ = id;
150        None
151    }
152
153    /// Exact identity of `path`; `None` on error or when unsupported.
154    fn observe(&self, path: &str) -> Option<api::ipc::Observation> {
155        let _ = path;
156        None
157    }
158
159    /// Create the empty staging object exclusively. Not synced: staged bytes
160    /// become persistent only with the commit's sync.
161    fn stage_create(&mut self, stage: &str) -> bool {
162        let _ = stage;
163        false
164    }
165
166    /// Positional write into a staging object, not synced.
167    fn stage_write(&mut self, stage: &str, offset: u64, bytes: &[u8]) -> bool {
168        let _ = (stage, offset, bytes);
169        false
170    }
171
172    /// Atomically check that `stage` is still `sealed` and `dst` still matches
173    /// `expect`, then publish `stage` at `dst` and sync.
174    fn publish_if(&mut self, stage: &str, sealed: (u64, u64), dst: &str,
175        expect: api::ipc::SaveExpect) -> api::ipc::SaveOutcome
176    {
177        let _ = (stage, sealed, dst, expect);
178        api::ipc::SaveOutcome::NotPublished(api::ipc::save_refusal::REFUSED)
179    }
180}
