1//! `/bin` overlay backend — unions the kernel VIFS1 ramdisk (`BootFsProxy`) with
2//! the on-disk FAT cell-store, so `/bin/<cell>` resolves whether the ELF is a
3//! bootstrap cell embedded in `kernel_fs.img` OR a non-bootstrap cell migrated to
4//! the disk cell-store (G2 loader redesign, phase 03).
5//!
6//! Read ops try VIFS1 first (bootstrap / trusted-core wins on name collision),
7//! then fall back to the cell-store. `list` unions both. Strictly read-only:
8//! `/bin` is never writable, so every mutating op fails `Unsupported`.
9
10use alloc::vec::Vec;
11
12use crate::backend::{FsBackend, ListPage};
13use crate::backend_bootfs::BootFsProxy;
14use crate::backend_fat::FatBackend;
15use crate::blk_router::DeviceRef;
16use crate::fs_error::FsError;
17
18/// Cursor bit marking a position in the cell-store half of a `/bin` listing.
19const STORE_PHASE: u32 = 1 << 30;
20
21pub struct BinOverlay {
22    boot: BootFsProxy,
23    store: FatBackend,
24}
25
26impl BinOverlay {
27    /// `store_base_lba`/`store_sectors` = `abi::disk::PART_CELLSTORE_*` on the
28    /// primary disk `dev`. The cell-store FAT is mounted with prefix `/bin` so
29    /// `/bin/<cell>` strips to `<cell>` at the FAT root — matching the layout
30    /// gen_disk writes into the cell-store.
31    pub fn new(dev: DeviceRef, store_base_lba: u64, store_sectors: u64) -> Self {
32        Self {
33            boot: BootFsProxy::new(),
34            store: FatBackend::mount("/bin", dev, store_base_lba, store_sectors),
35        }
36    }
37
38    /// The cell-store's stat; a cell-store that never mounted holds nothing.
39    fn store_stat(&self, path: &str) -> Result<Option<(u64, bool)>, FsError> {
40        match self.store.stat(path) {
41            Err(FsError::Unavailable) => Ok(None),
42            result => result,
43        }
44    }
45}
46
47/// VIFS1's answer unless it failed or is `empty`; then the cell-store's. An
48/// empty VIFS1 answer stands when the cell-store fails as well.
49fn boot_first<T>(
50    boot: Result<T, FsError>,
51    empty: impl Fn(&T) -> bool,
52    store: impl FnOnce() -> Result<T, FsError>,
53) -> Result<T, FsError> {
54    match boot {
55        Ok(value) if !empty(&value) => Ok(value),
56        Ok(value) => store().or(Ok(value)),
57        Err(_) => store(),
58    }
59}
60
61impl FsBackend for BinOverlay {
62    fn get_file_ptr(&self, path: &str) -> Option<(usize, usize)> {
63        self.boot
64            .get_file_ptr(path)
65            .or_else(|| self.store.get_file_ptr(path))
66    }
67
68    fn list(&self, path: &str, out: &mut [u8]) -> Result<usize, FsError> {
69        // Union: VIFS1 entries first, then cell-store entries into the remainder.
70        // Bootstrap and P2-only cell sets are disjoint by construction, so no
71        // dedup is needed (a rare collision would list the name twice — benign).
72        // Each half alone may be missing; only both missing (or a half that
73        // failed to read) fails the listing.
74        let n = match self.boot.list(path, out) {
75            Ok(n) => Some(n),
76            Err(FsError::NotFound | FsError::NotADirectory) => None,
77            Err(error) => return Err(error),
78        };
79        let at = n.unwrap_or(0);
80        if at >= out.len() {
81            return Ok(at);
82        }
83        match (n, self.store.list(path, &mut out[at..])) {
84            (_, Ok(m)) => Ok(at + m),
85            (Some(n), Err(FsError::NotFound | FsError::NotADirectory | FsError::Unavailable)) => {
86                Ok(n)
87            }
88            (_, Err(error)) => Err(error),
89        }
90    }
91
92    /// VIFS1 entries, then cell-store entries whose name VIFS1 does not
93    /// already list (VIFS1 wins on a collision, as it does for reads).
94    fn list_page(&self, path: &str, start: u32, out: &mut [u8]) -> Result<ListPage, FsError> {
95        let boot_dir = matches!(self.boot.stat(path)?, Some((_, true)));
96        let store_dir = matches!(self.store_stat(path)?, Some((_, true)));
97        if !boot_dir && !store_dir {
98            return Err(FsError::NotFound);
99        }
100        let mut written = 0;
101        let store_start = if start & STORE_PHASE == 0 {
102            if boot_dir {
103                let page = self.boot.list_page(path, start, out)?;
104                if page.next.is_some() {
105                    return Ok(page);
106                }
107                written = page.written;
108            } else if start != 0 {
109                return Err(FsError::InvalidPath);
110            }
111            0
112        } else {
113            start & !STORE_PHASE
114        };
115        if !store_dir {
116            if store_start != 0 {
117                return Err(FsError::InvalidPath);
118            }
119            return Ok(ListPage {
120                written,
121                next: None,
122            });
123        }
124        let shadowed = if boot_dir {
125            self.boot.names(path)?
126        } else {
127            Vec::new()
128        };
129        let page =
130            self.store
131                .list_page_filtered(path, store_start, &mut out[written..], &|name| {
132                    shadowed.iter().any(|boot| boot.as_ref() == name)
133                })?;
134        Ok(ListPage {
135            written: written + page.written,
136            next: page.next.map(|next| next | STORE_PHASE),
137        })
138    }
139
140    fn stat(&self, path: &str) -> Result<Option<(u64, bool)>, FsError> {
141        match self.boot.stat(path) {
142            Ok(Some(hit)) => Ok(Some(hit)),
143            Ok(None) => self.store_stat(path),
144            // Found in the cell-store is still an answer; absent is not.
145            Err(error) => self.store_stat(path)?.map(Some).ok_or(error),
146        }
147    }
148
149    fn file_size(&self, path: &str) -> Result<u64, FsError> {
150        boot_first(
151            self.boot.file_size(path),
152            |s| *s == 0,
153            || self.store.file_size(path),
154        )
155    }
156
157    fn read_to_vec(&self, path: &str) -> Result<Vec<u8>, FsError> {
158        boot_first(self.boot.read_to_vec(path), Vec::is_empty, || {
159            self.store.read_to_vec(path)
160        })
161    }
162
163    /// Positional read: VIFS1 answers first, then the on-disk cell-store — the same precedence
164    /// `read_to_vec` uses, so the overlay never serves a cell-store copy of a bootstrap file.
165    fn read_at(&self, path: &str, offset: u64, buf: &mut [u8]) -> Result<usize, FsError> {
166        let boot = self.boot.read_at(path, offset, buf);
167        boot_first(boot, |n| *n == 0, || self.store.read_at(path, offset, buf))
168    }
169
170    fn write(&mut self, _path: &str, _content: &[u8]) -> Result<(), FsError> {
171        Err(FsError::Unsupported)
172    }
173    fn append(&mut self, _path: &str, _content: &[u8]) -> Result<(), FsError> {
174        Err(FsError::Unsupported)
175    }
176    fn mkdir(&mut self, _path: &str) -> Result<(), FsError> {
177        Err(FsError::Unsupported)
178    }
179    fn rmdir(&mut self, _path: &str) -> Result<(), FsError> {
180        Err(FsError::Unsupported)
181    }
182    fn unlink(&mut self, _path: &str) -> Result<(), FsError> {
183        Err(FsError::Unsupported)
184    }
185    fn rmdir_recursive(&mut self, _path: &str) -> Result<(), FsError> {
186        Err(FsError::Unsupported)
187    }
188}
