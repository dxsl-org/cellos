10/// Most file handles one caller generation may hold at once.
11pub const MAX_FILE_HANDLES_PER_CALLER: usize = 32;
12
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub enum FileState {
15    Open,
16    InFlightSyncRead,
17    Tombstoned,
18    Closed,
19}
20
21pub struct FileEntry {
22    pub owner: Caller,
23    pub path: String,
24    pub parent_dir: u64,
25    pub state: FileState,
26    pub _lease: Option<crate::namespace::ServiceHandle>,
27}
28
29#[derive(Debug, Clone, Copy, PartialEq, Eq)]
30pub enum FileHandleError {
31    UnknownHandle,
32    TooManyHandles,
33    Exhausted,
34}
35
36pub struct FileHandleTable {
37    entries: BTreeMap<u64, FileEntry>,
38    counts: BTreeMap<(u64, u64), usize>,
39    next: u64,
40    exhausted: bool,
41}
42
43impl FileHandleTable {
44    pub fn new() -> Self {
45        Self {
46            entries: BTreeMap::new(),
47            counts: BTreeMap::new(),
48            next: 1,
49            exhausted: false,
50        }
51    }
52
53    #[allow(dead_code)]
54    pub fn insert(
55        &mut self,
56        owner: Caller,
57        path: &str,
58        parent_dir: u64,
59    ) -> Result<ViVfsFileHandle, FileHandleError> {
60        self.insert_leased(owner, path, parent_dir, None)
61    }
62
63    pub fn insert_leased(
64        &mut self,
65        owner: Caller,
66        path: &str,
67        parent_dir: u64,
68        lease: Option<crate::namespace::ServiceHandle>,
69    ) -> Result<ViVfsFileHandle, FileHandleError> {
70        let key = owner_counts::key(owner);
71        if self.counts.get(&key).copied().unwrap_or(0) >= MAX_FILE_HANDLES_PER_CALLER {
72            return Err(FileHandleError::TooManyHandles);
73        }
74        if self.exhausted || self.next == 0 {
75            return Err(FileHandleError::Exhausted);
76        }
77
78        let id = self.next;
79        self.next = self.next.checked_add(1).unwrap_or_else(|| {
80            self.exhausted = true;
81            self.next
82        });
83        self.entries.insert(
84            id,
85            FileEntry {
86                owner,
87                path: path.to_string(),
88                parent_dir,
89                state: FileState::Open,
90                _lease: lease,
91            },
92        );
93        *self.counts.entry(key).or_insert(0) += 1;
94        Ok(ViVfsFileHandle(id))
95    }
96
97    /// Resolve an open file handle for its owner without exposing entries held
98    /// by another caller. Grant writes use this to recover the authorized path
99    /// before resolving caller-controlled grant memory.
100    pub fn path_of(&self, caller: Caller, file: ViVfsFileHandle) -> Option<&str> {
101        self.entries
102            .get(&file.0)
103            .filter(|entry| entry.owner == caller && entry.state == FileState::Open)
104            .map(|entry| entry.path.as_str())
105    }
106
107    pub fn begin_sync_read(
108        &mut self,
109        caller: Caller,
110        file: ViVfsFileHandle,
111    ) -> Result<String, FileHandleError> {
112        let entry = self
113            .entries
114            .get_mut(&file.0)
115            .filter(|entry| entry.owner == caller)
116            .ok_or(FileHandleError::UnknownHandle)?;
117        match entry.state {
118            FileState::Open => {
119                entry.state = FileState::InFlightSyncRead;
120                Ok(entry.path.clone())
121            }
122            FileState::InFlightSyncRead | FileState::Tombstoned | FileState::Closed => {
123                Err(FileHandleError::UnknownHandle)
124            }
125        }
126    }
127
128    pub fn finish_sync_read(&mut self, caller: Caller, file: ViVfsFileHandle) -> bool {
129        let Some(entry) = self
130            .entries
131            .get_mut(&file.0)
132            .filter(|entry| entry.owner == caller)
133        else {
134            return false;
135        };
136        if entry.state != FileState::InFlightSyncRead {
137            return false;
138        }
139        entry.state = FileState::Open;
140        true
141    }
142
143    pub fn close(&mut self, caller: Caller, file: ViVfsFileHandle) -> bool {
144        self.remove_if_owned(caller, file.0, FileState::Closed)
145    }
146
147    pub fn revoke_by_parent_dirs(&mut self, dir_ids: &[u64]) -> usize {
148        let doomed: Vec<u64> = self
149            .entries
150            .iter()
151            .filter(|(_, entry)| dir_ids.contains(&entry.parent_dir))
152            .map(|(id, _)| *id)
153            .collect();
154        self.remove_ids(&doomed, FileState::Tombstoned)
155    }
156
157    pub fn purge_owner(&mut self, caller: Caller) -> usize {
158        let doomed: Vec<u64> = self
159            .entries
160            .iter()
161            .filter(|(_, entry)| entry.owner == caller)
162            .map(|(id, _)| *id)
163            .collect();
164        self.remove_ids(&doomed, FileState::Closed)
165    }

167    #[cfg(any(test, feature = "test-hooks"))]
168    pub(crate) fn contains(&self, file: ViVfsFileHandle) -> bool {
169        self.entries.contains_key(&file.0)
170    }
171
172    #[cfg(any(test, feature = "test-hooks"))]
173    pub(crate) fn held_by(&self, caller: Caller) -> usize {
174        self.counts
175            .get(&owner_counts::key(caller))
176            .copied()
177            .unwrap_or(0)
178    }
179
180    #[cfg(any(test, feature = "test-hooks"))]
181    pub(crate) fn set_next_for_test(&mut self, next: u64, exhausted: bool) {
182        self.next = next;
183        self.exhausted = exhausted;
184    }
185
186    fn remove_if_owned(&mut self, caller: Caller, id: u64, state: FileState) -> bool {
187        let Some(entry) = self.entries.get(&id) else {
188            return false;
189        };
190        if entry.owner != caller {
191            return false;
192        }
193        self.remove_id(id, caller, state).is_some()
194    }
195
196    fn remove_ids(&mut self, ids: &[u64], state: FileState) -> usize {
197        ids.iter()
198            .filter_map(|id| {
199                let owner = self.entries.get(id).map(|entry| entry.owner)?;
200                self.remove_id(*id, owner, state)
201            })
202            .count()
203    }
204
205    fn remove_id(&mut self, id: u64, owner: Caller, state: FileState) -> Option<()> {
206        let mut removed = self.entries.remove(&id)?;
207        removed.state = state;
208        owner_counts::decrement(&mut self.counts, owner);
209        Some(())
210    }
211}
