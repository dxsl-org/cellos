73// Narrow syscall allowlist — kernel enforces this at dispatch (Phase 27).
74// BootFS proxy (/bin via the kernel initramfs VIFS1): Open/Close/ReadDir for
75// listing (all synchronous), OpenCap/ReadCap/CloseCap for file reads — the FD
76// `Read` syscall is deliberately ABSENT: it is an async transformation that
77// requires the caller to park immediately, which a service dispatch loop
78// cannot do (see backend_bootfs.rs::read_to_vec).
79api::declare_syscalls![
80    Send,
81    SendGather,
82    Recv,
83    TryRecv,
84    Log,
85    Heartbeat,
86    GrantAlloc,
87    GrantShare,
88    GrantSlice,
89    GrantFree,
90    BlkReadAsync,
91    GrantRegister,
92    GrantUnregister,
93    StateStash,
94    StateRestore,
95    Open,
96    Close,
97    ReadDir,
98    OpenCap,
99    ReadCap,
100    // SeekCap shares ReadCap's allowlist bit (bit 16), so positional reads add no authority; it
101    // is the difference between a chunked read costing O(file) and O(file^2).
102    SeekCap,
103    CloseCap,
104    // Prepared-cwd commits (`prepared_cwd`): the own service binding (LookupServiceBound)
105    // and the kernel SessionV1 requests (`ostd::session::kernel_request`; CellSessionV1 has
106    // no allowlist bit, it is listed for the record). The bounded exact-operation block
107    // exchange (`blk_router`: IpcSubmit/IpcTake/IpcWait/IpcCancel share Send/Recv's bits)
108    // and its scheduler-tick budget and the grant-hold clock (GetTime).
109    LookupServiceBound,
110    IpcSubmit,
111    IpcTake,
112    IpcWait,
113    IpcCancel,
114    GetTime,
115    CellSessionV1,
116    // NOTE: deliberately NO SetTimer. VFS never calls it — a "SetTimer (bit 11)
117    // denied for tid <vfs>" kernel warn on x86 is the CANARY for the known x86
118    // syscall-redispatch corruption (syscall number read as user CS = 0x23 = 35
119    // with a pointer as the tick count; see TODO #9 / x86 q35 P02). Allowing it
120    // turns that corruption into an unbounded sleep that hangs the boot.
121];

124// the host test build has no image-provided heap to declare.

126// The size is named because the sector cache in `page_cache` has to fit inside

139#[cfg(target_os = "none")]
140fn vfs_main() {
141    init_custom_heap();
142    println("VFS Service v0.2: RamFS + mkdir/rmdir/unlink IPC (typed postcard)");
143    // VfsManager::new() mounts all backends, including the FAT volume on the
144    // VirtIO disk (which logs its own success/fallback status).
145    let vfs = VfsManager::new();
146    // Whether the idle reconcile cadence still has kernel state to poll; see the loop.
147    let idle_work = prepared_cwd::idle_work(&vfs);
148    #[cfg(feature = "test-hooks")]
149    {
150        file_handles::selftest::run();
151        access::selftest::run();
152        manager::read_path_selftest::run();
153        save::selftest::run();
154    }
155    *GLOBAL_VFS.lock() = Some(vfs);
156    liveness::arm();
157    let config = ostd::service::ServeConfig {
158        heartbeat_ticks: liveness::VFS_HEARTBEAT_TICKS,
159        ..ostd::service::ServeConfig::with_heartbeat()
160    }
161    .watching(ostd::service::WatchPolicy::Every, MAX_WATCHED_OWNERS);
162    ostd::service::serve(
163        config,
164        &mut Vfs {
165            idle_work,
166            idle_turns: 0,
167        },
168    );
169}

171/// Owners VFS watches at once: unbounded, like every live Cell that may hold
172/// VFS state; only a refused kernel watch refuses a caller.
173#[cfg(target_os = "none")]
174const MAX_WATCHED_OWNERS: usize = usize::MAX;
175
176/// The request loop's own state; the file system lives in `GLOBAL_VFS`.
177#[cfg(target_os = "none")]
178struct Vfs {
179    /// Whether bounded preparations are still unsettled in the kernel. They can
180    /// change without a message (they expire, commit or end), so only then does
181    /// the loop poll and reconcile; otherwise every change arrives as a request
182    /// or an owner death and the loop parks. Recomputed wherever state changes.
183    idle_work: bool,
184    idle_turns: u32,
185}
186
187#[cfg(target_os = "none")]
188impl ostd::service::Handler for Vfs {
189    fn on_request(&mut self, request: ostd::service::Request<'_>) {
190        let caller = caller::Caller::from_attested(*request.caller());
191        let mut encoded = [0u8; api::ipc::IPC_BUF_SIZE];
192        let encoded_len = {
193            let mut resp_buf = [0u8; api::ipc::IPC_BUF_SIZE];
194            let mut guard = GLOBAL_VFS.lock();
195            let vfs = guard
196                .as_mut()
197                .expect("VFS initialized before serving requests");
198            vfs.note_watched_owner(caller);
199            let response =
200                dispatch::handle_request(vfs, request.payload(), Some(caller), &mut resp_buf);
201            self.idle_work = prepared_cwd::idle_work(vfs);
202            api::ipc::encode(&response, &mut encoded)
203                .map(|bytes| bytes.len())
204                .unwrap_or(0)
205        };
206        // `GLOBAL_VFS` is unlocked before replying to the caller.
207        let _ = request.reply(&encoded[..encoded_len]);
208    }
209
210    /// No owner watch, no durable state: the request is refused.
211    fn on_unwatched(&mut self, request: ostd::service::Request<'_>) {
212        let mut encoded = [0u8; api::ipc::IPC_BUF_SIZE];
213        let denied = api::ipc::VfsResponse::Err(api::services::proto::vfs::VfsError::Denied as u8);
214        let len = api::ipc::encode(&denied, &mut encoded)
215            .map(|bytes| bytes.len())
216            .unwrap_or(0);
217        let _ = request.reply(&encoded[..len]);
218    }
219
220    fn on_owner_death(&mut self, owner: ostd::service::Owner) {
221        if let Some(vfs) = GLOBAL_VFS.lock().as_mut() {
222            let _ = vfs.owner_died(owner.cell_id, owner.generation);
223            prepared_cwd::reconcile(vfs);
224            self.idle_work = prepared_cwd::idle_work(vfs);
225        }
226    }
227
228    fn on_idle(&mut self) -> ostd::service::Wait {
229        if self.idle_work {
230            ostd::service::Wait::Poll
231        } else {
232            ostd::service::Wait::Block
233        }
234    }
235
236    /// A poll found nothing: count it toward the reconcile cadence. VFS holds no
237    /// SetTimer authority (see the allowlist), so the cadence counts empty polls
238    /// instead of wall time; requests reconcile only the caller's own
239    /// reservations (`dir_admission::admit`).
240    fn on_empty_poll(&mut self) {
241        const IDLE_RECONCILE_EVERY: u32 = 128;
242        self.idle_turns = self.idle_turns.wrapping_add(1);
243        if self.idle_turns.is_multiple_of(IDLE_RECONCILE_EVERY) {
244            if let Some(vfs) = GLOBAL_VFS.lock().as_mut() {
245                prepared_cwd::reconcile(vfs);
246                self.idle_work = prepared_cwd::idle_work(vfs);
247            }
248        }
249        ostd::task::yield_now();
250    }
251}
