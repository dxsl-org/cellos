60// Declares block-I/O capability; the kernel grants BlockIoCap at spawn.

64    block_io = true,
65    network = false,
66    spawn = false,
67    part_data = true,
68    part_lfs = true

71// Narrow syscall allowlist — kernel enforces this at dispatch (Phase 27).
72// BootFS proxy (/bin via the kernel initramfs VIFS1): Open/Close/ReadDir for
73// listing (all synchronous), OpenCap/ReadCap/CloseCap for file reads — the FD
74// `Read` syscall is deliberately ABSENT: it is an async transformation that
75// requires the caller to park immediately, which a service dispatch loop
76// cannot do (see backend_bootfs.rs::read_to_vec).
77api::declare_syscalls![
78    Send,
79    SendGather,
80    Recv,
81    TryRecv,
82    Log,
83    Heartbeat,
84    GrantAlloc,
85    GrantShare,
86    GrantSlice,
87    GrantFree,
88    BlkReadAsync,
89    GrantRegister,
90    GrantUnregister,
91    StateStash,
92    StateRestore,
93    Open,
94    Close,
95    ReadDir,
96    OpenCap,
97    ReadCap,
98    // SeekCap shares ReadCap's allowlist bit (bit 16), so positional reads add no authority; it
99    // is the difference between a chunked read costing O(file) and O(file^2).
100    SeekCap,
101    CloseCap,
102    // First-trust confirmation (S9-4b): VFS calls service-input over the authenticated
103    // SessionV1 transport (`ostd::session::register_trust_challenge` /
104    // `collect_trust_decision`): the issuer binding (LookupServiceBound), the bounded IPC exchange (IpcSubmit/IpcTake/IpcWait/IpcCancel
105    // share Send/Recv's bits), its scheduler-tick budget and the monotonic challenge
106    // deadline (GetTime). CellSessionV1 has no allowlist bit; it is listed for the record.
107    LookupServiceBound,
108    IpcSubmit,
109    IpcTake,
110    IpcWait,
111    IpcCancel,
112    GetTime,
113    CellSessionV1,
114    // NOTE: deliberately NO SetTimer. VFS never calls it — a "SetTimer (bit 11)
115    // denied for tid <vfs>" kernel warn on x86 is the CANARY for the known x86
116    // syscall-redispatch corruption (syscall number read as user CS = 0x23 = 35
117    // with a pointer as the tick count; see TODO #9 / x86 q35 P02). Allowing it
118    // turns that corruption into an unbounded sleep that hangs the boot.
119];

121// `ostd::heap` (and therefore the macro) exists only for the bare-metal target;
122// the host test build has no image-provided heap to declare.
123//
124// The size is named because the sector cache in `page_cache` has to fit inside
125// it: the cache declares its budget as a fraction of this, and a cache that
126// cannot fit kills the cell as soon as it fills (`OOM: cell heap exhausted`).
127pub(crate) const VFS_HEAP_BYTES: usize = 16 * 1024 * 1024;

129#[cfg(target_os = "none")]

137#[cfg(target_os = "none")]
138fn vfs_main() {
139    init_custom_heap();
140    println("VFS Service v0.2: RamFS + mkdir/rmdir/unlink IPC (typed postcard)");
141    // VfsManager::new() mounts all backends, including the FAT volume on the
142    // VirtIO disk (which logs its own success/fallback status).
143    let vfs = VfsManager::new();
144    // Whether the idle reconcile cadence still has kernel state to poll; see the loop.
145    let mut idle_work = prepared_cwd::idle_work(&vfs);
146    #[cfg(feature = "test-hooks")]
147    {
148        file_handles::selftest::run();
149        access::selftest::run();
150        manager::read_path_selftest::run();
151        save::selftest::run();
152    }
153    *GLOBAL_VFS.lock() = Some(vfs);
154    let mut buf = [0u8; api::ipc::IPC_BUF_SIZE];
155    // Full reconciliation runs once per this many idle turns. VFS holds no SetTimer
156    // authority (deliberately withheld, see the allowlist; GetTime is only the trust-call
157    // budget), so the cadence counts idle receive attempts instead of wall time; requests
158    // reconcile only the caller's own reservations (`dir_admission::admit`).
159    const IDLE_RECONCILE_EVERY: u32 = 128;
160    let mut idle_turns: u32 = 0;
161
162    loop {
163        // The kernel writes caller identity in the receive-buffer trailer.
164        // A sender TID is reply routing only, never a Cell principal.
165        //
166        // Only bounded preparations can change without a message (they expire, commit
167        // or end in the kernel), so the nonblocking receive and its reconcile cadence run
168        // only while `prepared_cwd::idle_work` says one is unsettled. Otherwise every
169        // change arrives as a request or an owner-death notification, and the loop
170        // parks in the blocking receive instead of yield-spinning. `idle_work` is
171        // recomputed wherever the loop changes VFS state.
172        let received = if idle_work {
173            ostd::syscall::sys_try_recv_attested(0, &mut buf)
174        } else {
175            ostd::syscall::sys_recv_attested(0, &mut buf)
176        };
177        match received {
178            ostd::syscall::SyscallResult::Ok(sender) if sender > 0 => {
179                let operation = ostd::ipc::current();
180                let Some(identity) = api::caller_identity::CallerIdentity::from_recv_buf(&buf)
181                else {
182                    let cancellations = GLOBAL_VFS
183                        .lock()
184                        .as_mut()
185                        .map(|vfs| {
186                            let _ = vfs.handle_unattributed_owner_death(sender);
187                            prepared_cwd::reconcile(vfs);
188                            idle_work = prepared_cwd::idle_work(vfs);
189                            vfs.take_owner_watch_cancellations()
190                        })
191                        .unwrap_or_default();
192                    for token in cancellations {
193                        ostd::syscall::sys_cancel_cell_owner_watch(token);
194                    }
195                    buf = [0u8; api::ipc::IPC_BUF_SIZE];
196                    continue;
197                };
198                let caller = caller::Caller::from_attested(identity);
199                // The atomic kernel operation verifies this exact receive context
200                // and registers root death before VFS can persist any state.
201                let Some((owner, token)) =
202                    ostd::syscall::sys_watch_cell_owner(caller.cell.0, caller.generation)
203                else {
204                    let mut encoded = [0u8; api::ipc::IPC_BUF_SIZE];
205                    let len = api::ipc::encode(&api::ipc::VfsResponse::Err(3), &mut encoded)
206                        .map(|bytes| bytes.len())
207                        .unwrap_or(0);
208                    if let Some(operation) = operation {
209                        let _ = ostd::ipc::reply(operation, &encoded[..len]);
210                    } else {
211                        let _ = ostd::ipc::post(sender, &encoded[..len]);
212                    }
213                    buf = [0u8; api::ipc::IPC_BUF_SIZE];
214                    continue;
215                };
216                let mut encoded = [0u8; api::ipc::IPC_BUF_SIZE];
217                let (encoded_len, cancellations) = {
218                    let mut resp_buf = [0u8; api::ipc::IPC_BUF_SIZE];
219                    let mut guard = GLOBAL_VFS.lock();
220                    let vfs = guard
221                        .as_mut()
222                        .expect("VFS initialized before serving requests");
223                    vfs.install_owner_watch(caller, owner.root_tid as usize, token);
224                    let response = dispatch::handle_request(vfs, &buf, Some(caller), &mut resp_buf);
225                    let len = api::ipc::encode(&response, &mut encoded)
226                        .map(|bytes| bytes.len())
227                        .unwrap_or(0);
228                    idle_work = prepared_cwd::idle_work(vfs);
229                    (len, vfs.take_owner_watch_cancellations())
230                };
231                for cancelled in cancellations {
232                    ostd::syscall::sys_cancel_cell_owner_watch(cancelled);
233                }
234                // `GLOBAL_VFS` is unlocked before replying to the caller.
235                if let Some(operation) = operation {
236                    let _ = ostd::ipc::reply(operation, &encoded[..encoded_len]);
237                } else {
238                    let _ = ostd::ipc::post(sender, &encoded[..encoded_len]);
239                }
240                buf = [0u8; api::ipc::IPC_BUF_SIZE];
241            }
242            _ => {
243                if idle_work {
244                    idle_turns = idle_turns.wrapping_add(1);
245                    if idle_turns.is_multiple_of(IDLE_RECONCILE_EVERY) {
246                        if let Some(vfs) = GLOBAL_VFS.lock().as_mut() {
247                            prepared_cwd::reconcile(vfs);
248                            idle_work = prepared_cwd::idle_work(vfs);
249                        }
250                    }
251                }
252                ostd::task::yield_now();
253            }
254        }
255    }
