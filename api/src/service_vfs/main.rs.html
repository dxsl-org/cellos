53// Declares block-I/O capability; the kernel grants BlockIoCap at spawn.
54// part_data/part_lfs scope the raw block syscalls to P1 (FAT32) + P4
55// (littlefs) — P2 cell-table and P3 snapshot stay kernel-only (P03 design).

57    block_io = true,
58    network = false,
59    spawn = false,
60    part_data = true,
61    part_lfs = true

64// Narrow syscall allowlist — kernel enforces this at dispatch (Phase 27).
65// BootFS proxy (/bin via the kernel initramfs VIFS1): Open/Close/ReadDir for
66// listing (all synchronous), OpenCap/ReadCap/CloseCap for file reads — the FD
67// `Read` syscall is deliberately ABSENT: it is an async transformation that
68// requires the caller to park immediately, which a service dispatch loop
69// cannot do (see backend_bootfs.rs::read_to_vec).
70api::declare_syscalls![
71    Send,
72    Recv,
73    TryRecv,
74    Reply,
75    Log,
76    Heartbeat,
77    LookupService,
78    GrantAlloc,
79    GrantShare,
80    GrantSlice,
81    GrantFree,
82    BlkReadAsync,
83    GrantRegister,
84    GrantUnregister,
85    StateStash,
86    StateRestore,
87    Open,
88    Close,
89    ReadDir,
90    OpenCap,
91    ReadCap,
92    CloseCap,
93    // NOTE: deliberately NO SetTimer. VFS never calls it — a "SetTimer (bit 11)
94    // denied for tid <vfs>" kernel warn on x86 is the CANARY for the known x86
95    // syscall-redispatch corruption (syscall number read as user CS = 0x23 = 35
96    // with a pointer as the tick count; see TODO #9 / x86 q35 P02). Allowing it
97    // turns that corruption into an unbounded sleep that hangs the boot.
98];

100#[no_mangle]
101pub fn main() {
102    println("VFS Service v0.2: RamFS + mkdir/rmdir/unlink IPC (typed postcard)");
103    // VfsManager::new() mounts all backends, including the FAT volume on the
104    // VirtIO disk (which logs its own success/fallback status).
105    let vfs = VfsManager::new();
106    #[cfg(feature = "test-hooks")]
107    {
108        file_handles::selftest::run();
109        access::selftest::run();
110    }
111    *GLOBAL_VFS.lock() = Some(vfs);
112
113    // Register the fast-IPC handler so trusted Cells can bypass ecall for VFS reads.
114    // The kernel records the VFS cell's ID at spawn time so it can clear this
115    // pointer if VFS crashes — see loader.rs fast_ipc::set_vfs_handler_cell call.
116    ostd::fast_ipc::register_vfs(vfs_fast_handler);
117    let mut buf = [0u8; api::ipc::IPC_BUF_SIZE];
118
119    loop {
120        // `sys_recv_attested` asks the kernel to state which cell sent the
121        // message, in the tail of `buf`.  VFS cannot derive that itself: `sender`
122        // is a tid, and a thread has its own tid while belonging to its parent
123        // cell, so a cell id built from that tid named a cell that does not exist
124        // and charged its quota to a ledger row nothing owned.
125        match ostd::syscall::sys_recv_attested(0, &mut buf) {
126            ostd::syscall::SyscallResult::Ok(sender) if sender > 0 => {
127                let Some(identity) = api::caller_identity::CallerIdentity::from_recv_buf(&buf)
128                else {
129                    let cancellations = GLOBAL_VFS
130                        .lock()
131                        .as_mut()
132                        .map(|vfs| {
133                            let _ = vfs.handle_unattributed_owner_death(sender);
134                            vfs.take_owner_watch_cancellations()
135                        })
136                        .unwrap_or_default();
137                    for token in cancellations {
138                        ostd::syscall::sys_cancel_cell_owner_watch(token);
139                    }
140                    buf = [0u8; api::ipc::IPC_BUF_SIZE];
141                    continue;
142                };
143                let caller = caller::Caller::from_attested(identity);
144                // The atomic kernel operation verifies this exact receive context
145                // and registers root death before VFS can persist any state.
146                let Some((owner, token)) =
147                    ostd::syscall::sys_watch_cell_owner(caller.cell.0, caller.generation)
148                else {
149                    let mut encoded = [0u8; api::ipc::IPC_BUF_SIZE];
150                    let len = api::ipc::encode(&api::ipc::VfsResponse::Err(3), &mut encoded)
151                        .map(|bytes| bytes.len())
152                        .unwrap_or(0);
153                    ostd::syscall::sys_send(sender, &encoded[..len]);
154                    buf = [0u8; api::ipc::IPC_BUF_SIZE];
155                    continue;
156                };
157                let mut encoded = [0u8; api::ipc::IPC_BUF_SIZE];
158                let (encoded_len, cancellations) = {
159                    let mut resp_buf = [0u8; api::ipc::IPC_BUF_SIZE];
160                    let mut guard = GLOBAL_VFS.lock();
161                    let vfs = guard
162                        .as_mut()
163                        .expect("VFS initialized before serving requests");
164                    vfs.install_owner_watch(caller, owner.root_tid as usize, token);
165                    let response = dispatch::handle_request(vfs, &buf, Some(caller), &mut resp_buf);
166                    let len = api::ipc::encode(&response, &mut encoded)
167                        .map(|bytes| bytes.len())
168                        .unwrap_or(0);
169                    (len, vfs.take_owner_watch_cancellations())
170                };
171                for cancelled in cancellations {
172                    ostd::syscall::sys_cancel_cell_owner_watch(cancelled);
173                }
174                // `GLOBAL_VFS` is unlocked for every kernel call and send.
175                ostd::syscall::sys_send(sender, &encoded[..encoded_len]);
176                buf = [0u8; api::ipc::IPC_BUF_SIZE];
177            }
178            _ => ostd::task::yield_now(),
179        }
180    }
181}
