8fn strip_mnt_sd_prefix(path: &str) -> Option<&str> {
9    if path.is_char_boundary(8) && path[..8].eq_ignore_ascii_case("/mnt/sd/") {
10        Some(&path[8..])
11    } else if path.is_char_boundary(7) && path[..7].eq_ignore_ascii_case("/mnt/sd") {
12        let rest = &path[7..];
13        if rest.is_empty() {
14            Some("")
15        } else if rest.starts_with('/') {
16            Some(&rest[1..])
17        } else {
18            None
19        }
20    } else {
21        None
22    }
23}

25fn is_sfn_alias(name: &str) -> bool {
26    let Some(dot) = name.rfind('.') else {
27        return false;
28    };
29    let (base, ext) = (&name[..dot], &name[dot + 1..]);
30    if !ext.eq_ignore_ascii_case("img") || base.len() > 8 || !base.contains('~') {
31        return false;
32    }
33    let Some(tilde) = base.find('~') else {
34        return false;
35    };
36    let prefix = &base[..tilde];
37    let suffix = &base[tilde + 1..];
38    let prefix_ok = prefix.eq_ignore_ascii_case("guest")
39        || prefix.eq_ignore_ascii_case("guest_")
40        || prefix.eq_ignore_ascii_case("guestd");
41    let suffix_ok = !suffix.is_empty() && suffix.chars().all(|c| c.is_ascii_digit());
42    prefix_ok && suffix_ok
43}

45pub(crate) fn is_guest_disk_path(path: &str) -> bool {
46    let Some(rel) = strip_mnt_sd_prefix(path) else {
47        return false;
48    };
49    if rel.is_empty() || rel.contains('/') {
50        return false;
51    }
52    rel.eq_ignore_ascii_case(GUEST_DISK_BASENAME)
53        || rel.eq_ignore_ascii_case(GUEST_DISK_83_BASENAME)
54        || rel.eq_ignore_ascii_case(GUEST_DISK_83_ALT_BASENAME)
55        || is_sfn_alias(rel)
56}

58pub(crate) fn contains_guest_disk(path: &str) -> bool {
59    if is_guest_disk_path(path) {
60        return true;
61    }
62    let trimmed = path.strip_suffix('/').unwrap_or(path);
63    if trimmed.is_empty()
64        || trimmed.eq_ignore_ascii_case("/mnt")
65        || trimmed.eq_ignore_ascii_case("/mnt/sd")
66    {
67        return true;
68    }
69    if let Some(rel) = strip_mnt_sd_prefix(path) {
70        let first_comp = rel.split('/').next().unwrap_or("");
71        if !first_comp.is_empty() && is_guest_disk_path(&alloc::format!("/mnt/sd/{first_comp}")) {
72            return true;
73        }
74    }
75    false
76}

78pub(crate) fn live_hypervisor_matches(caller: Caller, lookup: fn(u16) -> Option<usize>) -> bool {
79    if caller.cell.0 == 0 || caller.generation == 0 || caller.sender_tid == 0 {
80        return false;
81    }
82    usize::try_from(caller.sender_tid)
83        .ok()
84        .and_then(|sender| {
85            lookup(api::hypervisor::HYPERVISOR_SERVICE_ID).filter(|live| *live == sender)
86        })
87        .is_some()
88}
