1//! `/srv/sshd/**`: the inbound sshd's host-key seed and `authorized_keys`.
2//!
3//! Only a Cell whose **kernel-measured image** is a signature-verified image
4//! named `sshd` may read, stat, list, create, write, rename or unlink anything
5//! here; every other Cell is refused, and no caller may remove the namespace
6//! or an ancestor of it.
7//!
8//! Why the image and not a service id or a path: a well-known service id is
9//! bound by whoever registers it, and the path a Cell was spawned from is
10//! chosen by its spawner, so neither says which *program* is calling. The
11//! kernel measures the image at admission and reports, per live `(cell,
12//! generation)`, the anchor that verified its signature and the name in its
13//! signed `__ViCell_image` section (`abi::cell_image`). An unsigned image
14//! carries no name, so it never matches, whatever path it came from.
15//!
16//! Anchors: the release anchor always qualifies; the dev anchor qualifies
17//! because only a dev-anchor kernel can report it (`kernel/src/signing.rs`
18//! refuses to build a release kernel that would). Under the dev posture the
19//! rule is spoofable: the dev seed is public, so any Cell that can spawn from
20//! bytes can sign an image named `sshd`. The private profile anchor is not
21//! admitted.
22//!
23//! The lookup is always for the exact `(cell, generation)` of the request; a
24//! respawned Cell under the same `CellId` is asked about afresh, and a stale
25//! generation resolves to nothing. Nothing is cached. The fast path cannot make
26//! the syscall (it runs in the caller's context), so it refuses outright.
27//! A raw-block holder or an offline attacker reads the volume directly; this
28//! rule stops in-system Cells only.
29
30use crate::caller::Caller;
31use abi::cell_image::{CellImage, ANCHOR_DEV, ANCHOR_RELEASE};
32
33const NAMESPACE: &str = "/srv/sshd";
34const PREFIX: &str = "/srv/sshd/";
35/// The name `scripts/sign-cell.py` emits for the `service-sshd` artifact.
36const PROGRAM: &str = "sshd";
37
38/// Kernel query for the image identity of a `(cell, generation)`.
39pub(crate) type ImageLookup = fn(u64, u64) -> Option<CellImage>;
40
41/// Whether `prefix` names one of the two rule rows that guard the namespace.
42pub(super) fn is_srv_owner_store_rule(prefix: &str) -> bool {
43    matches!(prefix, NAMESPACE | PREFIX)
44}
45
46/// Whether `path` is the namespace root or lies below it.
47pub(super) fn is_srv_owner_namespace_path(path: &str) -> bool {
48    path == NAMESPACE || path.starts_with(PREFIX)
49}
50
51/// Whether removing `path` would remove the namespace: `path` is inside it or
52/// is one of its ancestors.
53pub(super) fn contains_srv_owner_namespace(path: &str) -> bool {
54    if is_srv_owner_namespace_path(path) {
55        return true;
56    }
57    let ancestor = path.strip_suffix('/').unwrap_or(path);
58    NAMESPACE
59        .strip_prefix(ancestor)
60        .is_some_and(|remainder| remainder.starts_with('/'))
61}
62
63/// Whether the kernel vouches that `caller`'s exact generation runs a signed
64/// `sshd` image under an admitted anchor. The record must echo the principal
65/// asked about.
66pub(super) fn srv_owner_matches(caller: Caller, lookup: ImageLookup) -> bool {
67    if caller.cell.0 == 0 || caller.generation == 0 {
68        return false;
69    }
70    lookup(caller.cell.0, caller.generation)
71        .filter(|image| image.cell_id == caller.cell.0 && image.generation == caller.generation)
72        .is_some_and(|image| {
73            matches!(image.identity.anchor(), ANCHOR_RELEASE | ANCHOR_DEV)
74                && image.identity.is_signed_program(PROGRAM)
75        })
76}
