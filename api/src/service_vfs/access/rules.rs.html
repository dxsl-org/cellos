1//! The policy data behind [`crate::access::AccessTable`].
2//!
3//! Two tables, consulted in this order:
4//!
5//! 1. [`EXACT_RULES`] — decisions for one whole path.
6//! 2. [`PREFIX_RULES`] — first matching prefix wins; the fallback.
7//!
8//! Rules are keyed by **filesystem path**, never by `CellId`. A `CellId` is
9//! `CellId(tid)`, so a service that init respawns after a crash comes back under
10//! a different one; a table keyed that way could not be written down ahead of
11//! time and would point at the wrong cell after the first restart.
12//!
13//! Nothing here is keyed by *cell* either. Scoping a rule to "only the net cell"
14//! needs a binding from the calling cell to which program it is running, and the
15//! only candidate today — the cell name — is the last component of the
16//! `path_hint` its spawner passed, which the spawner chooses freely. A cell
17//! spawned as `path_hint = "/bin/vfs"` would inherit whatever the rule granted
18//! `/bin/vfs`. Cell-scoped rules wait for an identity the kernel vouches for
19//! (a signature bound to the path, or an attested measurement label).
20
21/// Read/write decision for a path or a prefix.
22pub struct PathRule {
23    /// The path this rule applies to: a full path in [`EXACT_RULES`], a prefix in
24    /// [`PREFIX_RULES`].
25    pub prefix: &'static str,
26    /// True if any cell may read here.
27    pub allow_read_all: bool,
28    /// True if any cell may write here.
29    pub allow_write_all: bool,
30}
31
32/// Whole-path rules, checked before any prefix rule.
33///
34/// Empty: every path that needs a narrower rule than its prefix gives also needs
35/// to name *which* cell it is narrower for, and see the module note on why that
36/// binding does not exist yet. The lookup ships now so adding a row later is a
37/// data change rather than a control-flow change.
38pub static EXACT_RULES: &[PathRule] = &[PathRule {
39    prefix: "/srv/cellos",
40    allow_read_all: false,
41    allow_write_all: false,
42}];
43
44/// Prefix rules, first match wins. Ordered specific → general; `/` is last and
45/// matches every absolute path, so a path that reaches it is decided by it.
46///
47/// Deliberately broad on reads. Narrowing reads and boot-time reads are the same
48/// set right now (the loader reads every cell ELF through `/bin/`, the shell
49/// lists `/srv/`), so a narrow table would fail closed on the boot path and the
50/// repair would be to reopen all of it. Reads narrow per prefix, once each
51/// prefix's real readers are known.
52pub static PREFIX_RULES: &[PathRule] = &[
53    // The loader reads cell ELFs through here on every spawn.
54    PathRule {
55        prefix: "/bin/",
56        allow_read_all: true,
57        allow_write_all: false,
58    },
59    PathRule {
60        prefix: "/data/",
61        allow_read_all: true,
62        allow_write_all: true,
63    },
64    PathRule {
65        prefix: "/tmp/",
66        allow_read_all: true,
67        allow_write_all: true,
68    },
69    // FAT32 interop volume.
70    PathRule {
71        prefix: "/mnt/sd/",
72        allow_read_all: true,
73        allow_write_all: true,
74    },
75    // RedoxFS service volume. Writable like /data: without this rule every /srv
76    // write fell through to the read-only "/" rule and failed before reaching the
77    // backend.
78    PathRule {
79        prefix: "/srv/cellos/",
80        allow_read_all: false,
81        allow_write_all: false,
82    },
83    PathRule {
84        prefix: "/srv/",
85        allow_read_all: true,
86        allow_write_all: true,
87    },
88    // Root: readable, read-only. Reads stay open because the ramfs root holds
89    // paths outside every prefix above that cells read at startup (the cluster
90    // config at /etc/cellos/cluster.cfg is one), and denying them here would
91    // break those cells with an opaque PermissionDenied.
92    PathRule {
93        prefix: "/",
94        allow_read_all: true,
95        allow_write_all: false,
96    },
97];
