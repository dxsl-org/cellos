1//! Path authorization for the VFS service.
2//!
3//! Path-prefix rules rather than POSIX mode bits: in a Single Address Space OS a
4//! cell is the only principal, so there is no uid/gid to carry. The rule data and
5//! the reasoning behind its shape live in [`rules`].
6//!
7//! Deny by default. A path matching no rule is refused, and so is a caller whose
8//! identity the kernel did not vouch for — that check happens at the request
9//! boundary in `dispatch`, before a path is ever looked at, because a caller with
10//! no identity has no business reaching a policy table at all.
11//!
12//! These rules are VFS-internal and deliberately NOT driven by the cell manifest
13//! the kernel reads at spawn (`__ViCell_manifest`, source of `BlockIoCap` /
14//! `NetworkCap` / `SpawnCap`). Per-cell VFS path grants are a separate concern.

43/// Path rules, evaluated whole-path first and by prefix second.
44pub struct AccessTable {
45    exact: &'static [PathRule],
46    prefixes: &'static [PathRule],
47    service_lookup: fn(u16) -> Option<usize>,
48    image_lookup: ImageLookup,
49}
50
51impl AccessTable {
52    /// Initialize with the shipped rule tables.
53    pub fn new() -> Self {
54        Self::with_service_lookup(default_service_lookup)
55    }
56
57    pub(crate) fn with_service_lookup(service_lookup: fn(u16) -> Option<usize>) -> Self {
58        Self::with_lookups(service_lookup, default_image_lookup)
59    }
60
61    /// Both kernel queries replaceable, so a host test can stand in for the
62    /// service registry and for `ResolveCellImage`.
63    pub(crate) fn with_lookups(
64        service_lookup: fn(u16) -> Option<usize>,
65        image_lookup: ImageLookup,
66    ) -> Self {
67        Self {
68            exact: rules::EXACT_RULES,
69            prefixes: rules::PREFIX_RULES,
70            service_lookup,
71            image_lookup,
72        }
73    }
74
75    /// A table with replacement rule data, so a test can change policy under
76    /// a live handle.
77    #[cfg(test)]
78    pub(crate) fn with_rules(exact: &'static [PathRule], prefixes: &'static [PathRule]) -> Self {
79        Self {
80            exact,
81            prefixes,
82            service_lookup: default_service_lookup,
83            image_lookup: default_image_lookup,
84        }
85    }
86
87    /// Whether `caller` may write to `path`.
88    ///
89    /// `/srv/cellos/kms` and `/srv/.cellos-machine` are narrower than the broad
90    /// `/srv/` prefix: only the live KMS, respectively `MACHINE_PIN`, service
91    /// instance may touch them. The mutate flag is required everywhere, the
92    /// pin store included: the `machine-pin` Cell declares `VfsMutate` and
93    /// holds W on its boot edge (Phase 13), so no caller is exempt.
94    pub fn can_write(&self, caller: Caller, path: &str) -> bool {
95        caller.may_mutate() && self.decide(caller, path, AccessKind::Write)
96    }
97
98    /// Whether `caller` may write to the guest system disk image.
99    ///
100    /// Restricted to the live registered hypervisor provider; other cells cannot
101    /// touch the VM disk backend directly.
102    pub fn is_live_hypervisor(&self, caller: Caller) -> bool {
103        live_hypervisor_matches(caller, self.service_lookup)
104    }
105
106    /// Whether `caller` may read `path`.
107    pub fn can_read(&self, caller: Caller, path: &str) -> bool {
108        self.decide(caller, path, AccessKind::Read)
109    }
110
111    /// Whether `caller` may recursively remove `path` and its descendants.
112    pub fn can_remove_tree(&self, caller: Caller, path: &str) -> bool {
113        caller.may_mutate()
114            && is_canonical_policy_path(path)
115            && self.can_write(caller, path)
116            && !contains_provider_namespace(path)
117            && !contains_guest_disk(path)
118    }
119
120    /// Whether `caller` may remove `path` as a directory.
121    pub fn can_remove_dir(&self, caller: Caller, path: &str) -> bool {
122        caller.may_mutate()
123            && is_canonical_policy_path(path)
124            && self.can_write(caller, path)
125            && !contains_provider_namespace(path)
126            && !contains_guest_disk(path)
127    }
128
129    fn decide(&self, caller: Caller, path: &str, kind: AccessKind) -> bool {
130        // Runtime boot configuration is operator input, not a writable mailbox.
131        // FAT names are case-insensitive; protect both the subtree and ancestors
132        // against replacement, rename and recursive removal by ordinary cells.
133        if matches!(kind, AccessKind::Write) && touches_boot_configuration(path) {
134            return false;
135        }
136        if !is_canonical_policy_path(path) {
137            return false;
138        }
139        if is_guest_disk_path(path) {
140            return match kind {
141                AccessKind::Read => true,
142                AccessKind::Write => self.is_live_hypervisor(caller),
143            };
144        }
145        match self.rule_for(path) {
146            Some(rule) if is_kms_store_rule(rule.prefix) => {
147                live_kms_matches(caller, self.service_lookup)
148            }
149            Some(rule) if is_machine_pin_store_rule(rule.prefix) => {
150                live_machine_pin_matches(caller, self.service_lookup)
151            }
152            Some(rule) if is_srv_owner_store_rule(rule.prefix) => {
153                srv_owner_matches(caller, self.image_lookup)
154            }
155            Some(rule) => match kind {
156                AccessKind::Read => rule.allow_read_all,
157                AccessKind::Write => rule.allow_write_all,
158            },
159            None => false,
160        }
161    }
162
163    fn rule_for(&self, path: &str) -> Option<&'static PathRule> {
164        self.exact
165            .iter()
166            .find(|rule| rule.prefix == path)
167            .or_else(|| {
168                self.prefixes
169                    .iter()
170                    .find(|rule| path.starts_with(rule.prefix))
171            })
172    }
173}

181/// Whether removing `path` would remove a provider-held namespace: `path` is
182/// inside one or is an ancestor of one. No caller, the provider included, may.
183fn contains_provider_namespace(path: &str) -> bool {
184    contains_kms_namespace(path)
185        || contains_machine_pin_namespace(path)
186        || contains_srv_owner_namespace(path)
187}
188
189fn touches_boot_configuration(path: &str) -> bool {
190    const ROOT: &str = "/mnt/sd/etc/cellos";
191    if path == "/" {
192        return true;
193    }
194    let path = path.strip_suffix('/').unwrap_or(path);
195    if path.len() <= ROOT.len() {
196        ROOT.get(..path.len())
197            .is_some_and(|prefix| prefix.eq_ignore_ascii_case(path))
198            && (path.len() == ROOT.len() || ROOT.as_bytes()[path.len()] == b'/')
199    } else {
200        path.get(..ROOT.len())
201            .is_some_and(|prefix| prefix.eq_ignore_ascii_case(ROOT))
202            && path.as_bytes()[ROOT.len()] == b'/'
203    }
204}

212#[cfg(target_os = "none")]
213fn default_service_lookup(service_id: u16) -> Option<usize> {
214    ostd::service::provider_tid(service_id)
215}

217#[cfg(not(target_os = "none"))]
