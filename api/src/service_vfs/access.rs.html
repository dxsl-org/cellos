1//! Path authorization for the VFS service.
2//!
3//! Path-prefix rules rather than POSIX mode bits: in a Single Address Space OS a
4//! cell is the only principal, so there is no uid/gid to carry. The rule data and
5//! the reasoning behind its shape live in [`rules`].
6//!
7//! Deny by default. A path matching no rule is refused, and so is a caller whose
8//! identity the kernel did not vouch for — that check happens at the request
9//! boundary in `dispatch`, before a path is ever looked at, because a caller with
10//! no identity has no business reaching a policy table at all.
11//!
12//! These rules are VFS-internal and deliberately NOT driven by the cell manifest
13//! the kernel reads at spawn (`__ViCell_manifest`, source of `BlockIoCap` /
14//! `NetworkCap` / `SpawnCap`). Per-cell VFS path grants are a separate concern.
15
16mod rules;
17
18use crate::caller::Caller;
19pub use rules::PathRule;
20
21/// Path rules, evaluated whole-path first and by prefix second.
22pub struct AccessTable {
23    exact: &'static [PathRule],
24    prefixes: &'static [PathRule],
25}
26
27impl AccessTable {
28    /// Initialize with the shipped rule tables.
29    pub fn new() -> Self {
30        Self {
31            exact: rules::EXACT_RULES,
32            prefixes: rules::PREFIX_RULES,
33        }
34    }
35
36    /// Whether `caller` may write to `path`.
37    ///
38    /// `caller` is accepted for every decision so that adding a cell-scoped rule
39    /// is a change to [`rules`] alone; no rule discriminates on it today (see the
40    /// [`rules`] module note).
41    pub fn can_write(&self, _caller: Caller, path: &str) -> bool {
42        self.decide(path)
43            .map(|r| r.allow_write_all)
44            .unwrap_or(false)
45    }
46
47    /// Whether `caller` may read `path`.
48    ///
49    /// Every read op in `dispatch` is gated on this. Returns `false` when no rule
50    /// matches — a relative path, for instance, matches no prefix including `/`.
51    pub fn can_read(&self, _caller: Caller, path: &str) -> bool {
52        self.decide(path).map(|r| r.allow_read_all).unwrap_or(false)
53    }
54
55    /// The rule governing `path`: its whole-path rule if it has one, otherwise the
56    /// first prefix rule it matches, otherwise nothing (deny).
57    fn decide(&self, path: &str) -> Option<&'static PathRule> {
58        self.exact
59            .iter()
60            .find(|rule| rule.prefix == path)
61            .or_else(|| {
62                self.prefixes
63                    .iter()
64                    .find(|rule| path.starts_with(rule.prefix))
65            })
66    }
67}
68
69impl Default for AccessTable {
70    fn default() -> Self {
71        Self::new()
72    }
73}

75#[cfg(test)]
76mod tests {
77    use super::*;
78    use types::CellId;
79
80    const CELL: Caller = Caller {
81        cell: CellId(5),
82        generation: 1,
83    };
84
85    #[test]
86    fn reads_are_allowed_across_the_shipped_prefixes() {
87        let table = AccessTable::new();
88        for path in ["/bin/shell", "/data/x", "/tmp/x", "/mnt/sd/x", "/srv/x"] {
89            assert!(table.can_read(CELL, path), "{path} should be readable");
90        }
91    }
92
93    #[test]
94    fn a_path_matching_no_rule_is_denied_both_ways() {
95        let table = AccessTable::new();
96        // No leading slash → matches no prefix, not even "/".
97        assert!(!table.can_read(CELL, "etc/shadow"));
98        assert!(!table.can_write(CELL, "etc/shadow"));
99        assert!(!table.can_read(CELL, ""));
100    }
101
102    #[test]
103    fn bin_is_readable_but_not_writable() {
104        let table = AccessTable::new();
105        assert!(table.can_read(CELL, "/bin/vfs"));
106        assert!(!table.can_write(CELL, "/bin/vfs"));
107    }
108
109    #[test]
110    fn root_is_read_only() {
111        let table = AccessTable::new();
112        assert!(table.can_read(CELL, "/motd"));
113        assert!(!table.can_write(CELL, "/motd"));
114    }
115
116    #[test]
117    fn a_whole_path_rule_overrides_the_prefix_it_sits_under() {
118        static EXACT: &[PathRule] = &[PathRule {
119            prefix: "/data/secret",
120            allow_read_all: false,
121            allow_write_all: false,
122        }];
123        let table = AccessTable {
124            exact: EXACT,
125            prefixes: rules::PREFIX_RULES,
126        };
127        assert!(!table.can_read(CELL, "/data/secret"));
128        // Only the exact path is affected; its neighbours still follow /data/.
129        assert!(table.can_read(CELL, "/data/secretive"));
130        assert!(table.can_read(CELL, "/data/other"));
131    }
132}
