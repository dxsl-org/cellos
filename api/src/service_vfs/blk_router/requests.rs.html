1//! Sector read/write/flush over the [`Router`]: DrvRequest wire encoding and
2//! the per-request outcome rules (08A).
3
4use alloc::format;
5use types::block_wire::{
6    self, BlockOutcome, ReadRunRequest, MAX_REPLY_SIZE, MAX_RUN_SECTORS, SECTOR_SIZE,
7};
8
9use super::router::{Exchange, Router, Target};
10use super::{BlockSys, DeviceRef};

12/// Bytes of the largest `OP_READ_RUN`.
13const RUN_BYTES: usize = MAX_RUN_SECTORS as usize * SECTOR_SIZE;
14
15impl Router {
16    /// Send a write/flush request and decode its one-byte status reply.
17    fn exchange_status(
18        &mut self,
19        sys: &mut impl BlockSys,
20        i: usize,
21        tid: usize,
22        req: &[u8],
23    ) -> BlockOutcome {
24        let mut reply = [0u8; 1];
25        match self.exchange(sys, i, tid, req, &mut reply) {
26            Exchange::Refused | Exchange::Stalled { dispatched: false } => BlockOutcome::NotIssued,
27            // An empty reply is no status: the request may have run.
28            Exchange::Replied(0) => BlockOutcome::Indeterminate,
29            Exchange::Replied(_) => BlockOutcome::from_byte(reply[0]),
30            // Taken but never answered within the bound: it may yet run.
31            Exchange::Stalled { dispatched: true } => BlockOutcome::Indeterminate,
32            // The provider may have issued the command before it died; a write
33            // read as success would make the write-through page cache believe a
34            // sector reached the device.
35            Exchange::Missing => {
36                let name = &self.devices()[i].name;
37                sys.log(&format!(
38                    "[blk-router] {name}: driver died before replying; outcome indeterminate"
39                ));
40                BlockOutcome::Indeterminate
41            }
42        }
43    }

45    /// Read one sector. A death notification is never accepted as data.
46    pub fn read(
47        &mut self,
48        sys: &mut impl BlockSys,
49        dev: DeviceRef,
50        lba: u64,
51        buf: &mut [u8; SECTOR_SIZE],
52    ) -> bool {
53        match self.target(sys, dev) {
54            Target::Kernel => sys.kernel_read(lba, buf),
55            Target::Refused => false,
56            Target::Provider(i, tid) => {
57                // Request: [op u16][lba u64]; reply: [status][512 data].
58                let mut req = [0u8; 10];
59                req[0..2].copy_from_slice(&block_wire::OP_READ.to_le_bytes());
60                req[2..10].copy_from_slice(&lba.to_le_bytes());
61                let mut reply = [0u8; 1 + SECTOR_SIZE];
62                if !matches!(
63                    self.exchange(sys, i, tid, &req, &mut reply),
64                    Exchange::Replied(len) if len == reply.len()
65                ) || !BlockOutcome::from_byte(reply[0]).is_completed()
66                {
67                    return false;
68                }
69                buf.copy_from_slice(&reply[1..]);
70                true
71            }
72        }
73    }

75    /// Read `buf.len() / 512` whole sectors starting at `lba`: one
76    /// `OP_READ_RUN` per [`MAX_RUN_SECTORS`] for a routed device.
77    pub fn read_sectors(
78        &mut self,
79        sys: &mut impl BlockSys,
80        dev: DeviceRef,
81        lba: u64,
82        buf: &mut [u8],
83    ) -> bool {
84        if !buf.len().is_multiple_of(SECTOR_SIZE) {
85            return false;
86        }
87        buf.chunks_mut(RUN_BYTES).enumerate().all(|(k, chunk)| {
88            let first = lba + (k * usize::from(MAX_RUN_SECTORS)) as u64;
89            match self.target(sys, dev) {
90                Target::Refused => false,
91                Target::Provider(i, tid) => self.read_run(sys, i, tid, first, chunk),
92                Target::Kernel => {
93                    chunk
94                        .chunks_exact_mut(SECTOR_SIZE)
95                        .enumerate()
96                        .all(|(n, sector)| {
97                            let sector: &mut [u8; SECTOR_SIZE] =
98                                sector.try_into().expect("whole sector");
99                            sys.kernel_read(first + n as u64, sector)
100                        })
101                }
102            }
103        })
104    }

106    /// One `OP_READ_RUN`; the sectors are copied out only when it completed.
107    fn read_run(
108        &mut self,
109        sys: &mut impl BlockSys,
110        i: usize,
111        tid: usize,
112        lba: u64,
113        buf: &mut [u8],
114    ) -> bool {
115        let request = ReadRunRequest {
116            sector: lba,
117            count: (buf.len() / SECTOR_SIZE) as u16,
118        };
119        let mut reply = [0u8; MAX_REPLY_SIZE];
120        let reply = &mut reply[..request.reply_len()];
121        let expected = reply.len();
122        if !matches!(
123            self.exchange(sys, i, tid, &request.encode(), reply),
124            Exchange::Replied(len) if len == expected
125        ) || !BlockOutcome::from_byte(reply[0]).is_completed()
126        {
127            return false;
128        }
129        buf.copy_from_slice(&reply[1..]);
130        true
131    }

133    /// Write one sector. `Indeterminate` means the write may have been issued
134    /// and must not be replayed; the kernel path reports only success or failure,
135    /// so its failure is `Indeterminate`.
136    pub fn write(
137        &mut self,
138        sys: &mut impl BlockSys,
139        dev: DeviceRef,
140        lba: u64,
141        data: &[u8; SECTOR_SIZE],
142    ) -> BlockOutcome {
143        let outcome = match self.target(sys, dev) {
144            Target::Kernel if sys.kernel_write(lba, data) => BlockOutcome::Completed,
145            Target::Kernel => BlockOutcome::Indeterminate,
146            Target::Refused => BlockOutcome::NotIssued,
147            Target::Provider(i, tid) => {
148                // Request: [op u16][lba u64][512 data].
149                let mut req = [0u8; 10 + SECTOR_SIZE];
150                req[0..2].copy_from_slice(&block_wire::OP_WRITE.to_le_bytes());
151                req[2..10].copy_from_slice(&lba.to_le_bytes());
152                req[10..].copy_from_slice(data);
153                self.exchange_status(sys, i, tid, &req)
154            }
155        };
156        if outcome == BlockOutcome::Indeterminate {
157            sys.log(&format!(
158                "[blk-router] {}: write lba={} indeterminate; not replayed",
159                self.name(dev),
160                lba
161            ));
162        }
163        outcome
164    }

166    /// Flush the device's volatile write cache. Outcome rules as [`Self::write`].
167    pub fn flush(&mut self, sys: &mut impl BlockSys, dev: DeviceRef) -> BlockOutcome {
168        let outcome = match self.target(sys, dev) {
169            Target::Kernel if sys.kernel_flush() => BlockOutcome::Completed,
170            Target::Kernel => BlockOutcome::Indeterminate,
171            Target::Refused => BlockOutcome::NotIssued,
172            Target::Provider(i, tid) => {
173                self.exchange_status(sys, i, tid, &block_wire::OP_FLUSH.to_le_bytes())
174            }
175        };
176        if !outcome.is_completed() {
177            sys.log(&format!(
178                "[blk-router] {}: flush {:?}",
179                self.name(dev),
180                outcome
181            ));
182        }
183        outcome
184    }
