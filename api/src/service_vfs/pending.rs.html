1//! Pending async read table for the VFS service.
2//!
3//! Implements the server side of the two-opcode non-blocking read protocol:
4//!   1. `ReadAsync { path }` → VFS reads file data synchronously (disk is still
5//!      blocking), stores it under a handle owned by the requesting cell, and
6//!      returns `PendingHandle(id)`.
7//!   2. `Poll { handle: id }` → returns `Data(bytes)` (always ready with
8//!      synchronous backend) or `Err` if the handle is stale/consumed or
9//!      belongs to another cell.
10//!
11//! The protocol is correct regardless of the backend being synchronous: the
12//! caller-side loop with `yield_now()` cooperates correctly with the scheduler,
13//! and the API shape is ready for a future interrupt-driven block driver.
14//!
15//! Handle IDs are allocated sequentially from 1 and are therefore trivially
16//! guessable. They are not secrets: a slot's confidentiality rests entirely on
17//! the owner comparison in [`PendingTable::poll`], never on a caller failing to
18//! guess an ID.
19
20use alloc::collections::BTreeMap;
21use alloc::string::{String, ToString};
22use alloc::vec::Vec;
23
24use crate::caller::Caller;
25
26/// One pending async read slot.
27pub struct PendingRead {
28    /// The caller that issued the `ReadAsync`, and the only caller permitted to
29    /// poll this slot.  Carries the cell generation, so a cell that reappears
30    /// under a recycled `CellId` cannot poll its predecessor's slots.
31    pub owner: Caller,
32    /// Path the data was read from.  Kept so `Poll` can re-authorize instead of
33    /// trusting an open-time decision that a rule change may have invalidated.
34    pub path: String,
35    /// Pre-read file contents.  Data is available immediately with the current
36    /// synchronous VirtIO block backend.
37    pub data: Vec<u8>,
38}
39
40/// Table of pending reads keyed by opaque handle ID.
41///
42/// One table serves every client cell, so the keyspace is shared: an entry is
43/// reachable only by the cell recorded as its owner.
44pub struct PendingTable {
45    slots: BTreeMap<u32, PendingRead>,
46    next_id: u32,
47}
48
49impl PendingTable {
50    pub fn new() -> Self {
51        Self {
52            slots: BTreeMap::new(),
53            next_id: 1,
54        }
55    }
56
57    /// Insert pre-read data on behalf of `owner` and return its handle.
58    ///
59    /// The handle is usable only by `owner`; no other cell can poll it.
60    pub fn insert(&mut self, owner: Caller, path: &str, data: Vec<u8>) -> u32 {
61        let id = self.next_id;
62        self.next_id = self.next_id.wrapping_add(1).max(1); // skip 0
63        self.slots.insert(
64            id,
65            PendingRead {
66                owner,
67                path: path.to_string(),
68                data,
69            },
70        );
71        id
72    }
73
74    /// The path a slot was filled from, if `caller` owns that slot.
75    ///
76    /// Lets `Poll` re-check the path rules before handing over data.  A slot
77    /// belonging to another cell reads as absent, same as a stale handle.
78    pub fn owned_path(&self, caller: Caller, handle: u32) -> Option<&str> {
79        self.slots
80            .get(&handle)
81            .filter(|slot| slot.owner == caller)
82            .map(|slot| slot.path.as_str())
83    }
84
85    /// Consume the data for `handle` on behalf of `caller`.
86    ///
87    /// Returns `None` when the handle is stale (already polled), was never
88    /// issued, or is owned by a different cell.  Those three cases are
89    /// deliberately indistinguishable: a distinguishable "wrong owner" reply
90    /// would turn the sequential handle space into an existence oracle.
91    ///
92    /// A rejected poll leaves the slot untouched.  Removing first and comparing
93    /// afterwards would let any cell destroy another cell's pending read by
94    /// sweeping the handle space, even while receiving none of the data.
95    pub fn poll(&mut self, caller: Caller, handle: u32) -> Option<Vec<u8>> {
96        match self.slots.get(&handle) {
97            Some(slot) if slot.owner == caller => self.slots.remove(&handle).map(|s| s.data),
98            _ => None,
99        }
100    }
101}
102
103impl Default for PendingTable {
104    fn default() -> Self {
105        Self::new()
106    }
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112    use alloc::vec;
113
114    use types::CellId;
115
116    const CELL_A: Caller = Caller {
117        cell: CellId(11),
118        generation: 1,
119    };
120    const CELL_B: Caller = Caller {
121        cell: CellId(22),
122        generation: 1,
123    };
124    /// Cell 22 again after a respawn that reused its id: a different principal.
125    const CELL_B_RESPAWNED: Caller = Caller {
126        cell: CellId(22),
127        generation: 2,
128    };
129
130    #[test]
131    fn poll_returns_data_to_the_issuing_cell() {
132        let mut table = PendingTable::new();
133        let handle = table.insert(CELL_B, "/data/b", vec![1u8, 2, 3]);
134        assert_eq!(table.poll(CELL_B, handle), Some(vec![1u8, 2, 3]));
135        // A successful poll consumes the slot.
136        assert_eq!(table.poll(CELL_B, handle), None);
137    }
138
139    #[test]
140    fn poll_rejects_another_cells_handle() {
141        let mut table = PendingTable::new();
142        let handle = table.insert(CELL_B, "/data/b", vec![0xAA; 8]);
143        assert_eq!(table.poll(CELL_A, handle), None);
144    }
145
146    #[test]
147    fn handle_sweep_neither_reads_nor_destroys_another_cells_slot() {
148        let mut table = PendingTable::new();
149        let contents = vec![0xBE; 16];
150        let handle = table.insert(CELL_B, "/data/b", contents.clone());
151
152        // Cell A sweeps the low handle space the way an attacker would.
153        for probe in 1..=64u32 {
154            assert_eq!(
155                table.poll(CELL_A, probe),
156                None,
157                "handle {probe} readable by a non-owner"
158            );
159        }
160
161        // The owner's slot survived the sweep.
162        assert_eq!(table.poll(CELL_B, handle), Some(contents));
163    }
164
165    #[test]
166    fn a_respawned_cell_cannot_poll_its_predecessors_slot() {
167        let mut table = PendingTable::new();
168        let handle = table.insert(CELL_B, "/data/b", vec![0xCD; 4]);
169        assert_eq!(table.poll(CELL_B_RESPAWNED, handle), None);
170        // Refused, not consumed.
171        assert_eq!(table.poll(CELL_B, handle), Some(vec![0xCD; 4]));
172    }
173
174    #[test]
175    fn owned_path_is_visible_to_the_issuing_cell_only() {
176        let mut table = PendingTable::new();
177        let handle = table.insert(CELL_B, "/data/b", vec![1u8]);
178        assert_eq!(table.owned_path(CELL_B, handle), Some("/data/b"));
179        assert_eq!(table.owned_path(CELL_A, handle), None);
180        assert_eq!(table.owned_path(CELL_B_RESPAWNED, handle), None);
181    }
182
183    #[test]
184    fn poll_rejects_a_handle_that_was_never_issued() {
185        let mut table = PendingTable::new();
186        let handle = table.insert(CELL_A, "/data/a", vec![7u8]);
187        assert_eq!(table.poll(CELL_A, handle.wrapping_add(1)), None);
188    }
