1//! How long a directory handle lasts, and what takes it away.
2//!
3//! ## A handle dies with the cell that holds it
4//! An entry belongs to a [`Caller`], which carries the cell's generation, so a
5//! successor cell under a recycled id is a different holder and reaches nothing
6//! its predecessor had. Seeing a higher generation proves the predecessor is
7//! gone, and its entries are purged then rather than left to be filtered out on
8//! every lookup — a dead cell's authority should not sit in the table waiting
9//! for a lookup that never comes.
10//!
11//! That is also why a replacement instance inherits nothing across a hot-swap.
12//! The generation exists to tell a respawned cell from the one it replaced, and
13//! honouring a predecessor's handles would discard that distinction at the one
14//! moment it matters. A cell that transfers state re-acquires its handles on the
15//! way back up; what it cannot do is silently continue with a dead instance's
16//! authority.
17//!
18//! ## Revocation is transitive
19//! A derived handle is a narrower share of the handle it came from, so it cannot
20//! outlive it: authority surviving the withdrawal of the authority it came from
21//! would make revocation advisory, and a cell could keep access indefinitely by
22//! deriving a subdirectory and dropping the original. Every entry records the
23//! handle it was derived from and revocation walks that graph — across cells,
24//! because a set inherited at spawn records the spawner's handle as its parent.
25//!
26//! The cost is real and accepted: a child can lose access because of a decision
27//! about its parent, the same shape as a spawner's capability ceiling bounding a
28//! child it will never see again.

36/// What the service still owes `caller` before serving it.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum Contact {
39    /// Nothing outstanding.
40    Ready,
41    /// The kernel has not yet been asked what this cell inherited.
42    NeedsAttestation,
43}
44
45impl DirTable {
46    /// Register contact with `caller` and report what is still owed.
47    ///
48    /// A message arriving under a *lower* generation than the one on record
49    /// cannot come from a live cell; it is left unregistered, which means it
50    /// holds nothing and can bind nothing.
51    pub fn on_contact(&mut self, caller: Caller) -> Contact {
52        let cell = caller.cell.0;
53        match self.cells.get(&cell) {
54            Some(state) if state.generation == caller.generation => {
55                if state.attested {
56                    Contact::Ready
57                } else {
58                    Contact::NeedsAttestation
59                }
60            }
61            Some(state) if state.generation < caller.generation => {
62                self.purge_cell(cell);
63                self.insert_cell_state(caller);
64                Contact::NeedsAttestation
65            }
66            Some(_) => Contact::Ready,
67            None => {
68                self.insert_cell_state(caller);
69                Contact::NeedsAttestation
70            }
71        }
72    }
73
74    /// Record that the kernel has been asked about `caller`, whatever it said.
75    pub fn mark_attested(&mut self, caller: Caller) {
76        if let Some(state) = self.state_mut(caller) {
77            state.attested = true;
78        }
79    }
80
81    /// Whether path-addressed requests from `caller` must be refused.
82    pub fn is_sealed(&self, caller: Caller) -> bool {
83        self.state(caller).is_some_and(|s| s.sealed)
84    }
85
86    /// Whether this cell generation has been seen before.
87    ///
88    /// The fast-IPC path uses this: it cannot make the attestation syscall, so a
89    /// cell it has never seen might be one that should already be sealed. It
90    /// declines those and lets the ecall path decide.
91    pub fn has_met(&self, caller: Caller) -> bool {
92        self.state(caller).is_some()
93    }
94
95    /// Refuse every path-addressed request from `caller` from now on.
96    ///
97    /// One-way for the life of the cell generation. Returns `false` only for a
98    /// caller with no registered state, which cannot happen after
99    /// [`Self::on_contact`].
100    pub fn seal(&mut self, caller: Caller) -> bool {
101        match self.state_mut(caller) {
102            Some(state) => {
103                state.sealed = true;
104                true
105            }
106            None => false,
107        }
108    }
109
110    /// Revoke `dir` and everything derived from it. Returns how many entries went.
111    ///
112    /// # Errors
113    /// [`DirError::UnknownHandle`] when `dir` is not this caller's.
114    pub fn revoke(&mut self, caller: Caller, dir: ViDirHandle) -> Result<usize, DirError> {
115        if self.owned(caller, dir).is_none() {
116            return Err(DirError::UnknownHandle);
117        }
118        Ok(self.revoke_ids(&[dir.0]))
119    }
120
121    pub(crate) fn state(&self, caller: Caller) -> Option<&CellState> {
122        self.cells
123            .get(&caller.cell.0)
124            .filter(|s| s.generation == caller.generation)
125    }
126
127    pub(crate) fn state_mut(&mut self, caller: Caller) -> Option<&mut CellState> {
128        self.cells
129            .get_mut(&caller.cell.0)
130            .filter(|s| s.generation == caller.generation)
131    }
132
133    fn insert_cell_state(&mut self, caller: Caller) {
134        self.cells.insert(
135            caller.cell.0,
136            CellState {
137                generation: caller.generation,
138                sealed: false,
139                attested: false,
140                handles: 0,
141            },
142        );
143    }
144
145    /// Remove `roots` and, repeatedly, everything whose parent has gone.
146    ///
147    /// The fixpoint loop is not an optimisation target: derivation chains are a
148    /// handful of entries deep and revocation is rare.
149    pub(crate) fn revoke_ids(&mut self, roots: &[u64]) -> usize {
150        let mut doomed: Vec<u64> = roots.to_vec();
151        let mut cursor = 0;
152        while cursor < doomed.len() {
153            let id = doomed[cursor];
154            cursor += 1;
155            let children: Vec<u64> = self
156                .entries
157                .iter()
158                .filter(|(child, e)| e.parent == Some(id) && !doomed.contains(child))
159                .map(|(child, _)| *child)
160                .collect();
161            doomed.extend(children);
162        }
163        let mut removed = 0;
164        for id in doomed {
165            if let Some(entry) = self.entries.remove(&id) {
166                removed += 1;
167                if let Some(state) = self.cells.get_mut(&entry.owner.cell.0) {
168                    state.handles = state.handles.saturating_sub(1);
169                }
170            }
171        }
172        removed
173    }
174
175    /// Revoke everything `cell` holds, at any generation, and forget the cell.
176    pub(crate) fn purge_cell(&mut self, cell: u64) -> usize {
177        let roots: Vec<u64> = self
178            .entries
179            .iter()
180            .filter(|(_, e)| e.owner.cell.0 == cell)
181            .map(|(id, _)| *id)
182            .collect();
183        let removed = self.revoke_ids(&roots);
184        self.cells.remove(&cell);
185        removed
186    }
