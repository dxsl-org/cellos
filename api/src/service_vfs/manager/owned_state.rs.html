4impl VfsManager {
5    /// Install a root-lifetime watch before dispatch can create durable state.
6    /// A reused CellId retires only the predecessor principal, never a broad
7    /// CellId row, and its token is cancelled after the VFS lock is released.
8    pub fn install_owner_watch(&mut self, caller: Caller, root_tid: usize, token: u64) {
9        let stale: alloc::vec::Vec<(u64, u64)> = self
10            .watched_owners
11            .keys()
12            .filter(|(cell_id, generation)| {
13                *cell_id == caller.cell.0 && *generation != caller.generation
14            })
15            .copied()
16            .collect();
17        for key in stale {
18            if let Some(previous) = self.watched_owners.remove(&key) {
19                self.purge_owned_state(previous.principal);
20                self.cancelled_owner_watch_tokens.push(previous.token);
21            }
22        }
23        let key = (caller.cell.0, caller.generation);
24        if self.watched_owners.contains_key(&key) {
25            self.cancelled_owner_watch_tokens.push(token);
26            return;
27        }
28        self.watched_owners.insert(
29            key,
30            WatchedOwner {
31                principal: caller,
32                root_tid,
33                token,
34            },
35        );
36    }
37
38    /// A death notification has no caller trailer. It is attributable only when
39    /// its root TID matches a live, tokenized local owner record.
40    pub fn handle_unattributed_owner_death(&mut self, root_tid: usize) -> bool {
41        let matches: alloc::vec::Vec<(u64, u64)> = self
42            .watched_owners
43            .iter()
44            .filter_map(|(&key, owner)| (owner.root_tid == root_tid).then_some(key))
45            .collect();
46        let mut purged = false;
47        for key in matches {
48            if let Some(owner) = self.watched_owners.remove(&key) {
49                self.purge_owned_state(owner.principal);
50                self.cancelled_owner_watch_tokens.push(owner.token);
51                purged = true;
52            }
53        }
54        purged
55    }
56
57    pub fn take_owner_watch_cancellations(&mut self) -> alloc::vec::Vec<u64> {
58        core::mem::take(&mut self.cancelled_owner_watch_tokens)
59    }
60
61    fn purge_owned_state(&mut self, caller: Caller) -> usize {
62        let dirs = self.dirs.purge_owner(caller);
63        let revoked_files = self.files.revoke_by_parent_dirs(&dirs.revoked_ids);
64        let files = self.files.purge_owner(caller);
65        let handles = self.handles.purge_owner(caller);
66        let pending = self.pending.purge_owner(caller);
67        dirs.count + revoked_files + files + handles + pending
68    }
