1//! Backend routing and change marks for D03 conditional save. Authorization,
2//! quota and the transaction table live in `crate::save`.
3
4use alloc::string::String;
5
6use api::ipc::{Observation, SaveExpect, SaveOutcome};
7
8use crate::manager::VfsManager;
9
10impl VfsManager {
11    /// Staging path for transaction `id` on the backend that owns `dst`;
12    /// `None` when that backend has no conditional publication.
13    pub fn stage_path(&self, dst: &str, id: u32) -> Option<String> {
14        self.mounts.backend(dst)?.stage_path(id)
15    }
16
17    pub fn observe(&self, path: &str) -> Option<Observation> {
18        self.mounts.backend(path)?.observe(path)
19    }
20
21    pub fn stage_create(&mut self, stage: &str) -> bool {
22        self.mounts.backend_mut(stage).is_some_and(|b| b.stage_create(stage))
23    }
24
25    pub fn stage_write(&mut self, stage: &str, offset: u64, bytes: &[u8]) -> bool {
26        self.mounts.backend_mut(stage).is_some_and(|b| b.stage_write(stage, offset, bytes))
27    }
28
29    /// Remove an unpublished staging object (synced like any unlink).
30    pub fn stage_remove(&mut self, stage: &str) -> bool {
31        self.mounts.backend_mut(stage).is_some_and(|b| b.unlink(stage))
32    }
33
34    pub fn publish_if(&mut self, stage: &str, sealed: (u64, u64), dst: &str, expect: SaveExpect) -> SaveOutcome {
35        let created = self.would_create(dst);
36        let outcome = match self.mounts.backend_mut(dst) {
37            Some(backend) => backend.publish_if(stage, sealed, dst, expect),
38            None => SaveOutcome::NotPublished(api::ipc::save_refusal::REFUSED),
39        };
40        // An indeterminate publication may have reached storage, so it marks too.
41        if outcome.published().is_some() || outcome == SaveOutcome::Indeterminate {
42            self.mark_content(dst, created);
43            self.mark_binding(stage);
44        }
45        outcome
46    }
47}
