1//! Paged directory listing across the mount table (`ListAtPage`).
2//!
3//! A listing is the backend's own entries followed by the entries the mount
4//! table adds: mount points and mount ancestors below the directory, and the
5//! public configuration names under `/etc`. Unlike `list_dir`, nothing is capped
6//! or dropped: whatever does not fit one page is on the next one.
7
8use alloc::vec::Vec;
9use api::dir_name::join_component;
10use api::ipc::vfs_dir_page::{push_entry, EntryKind, ENTRY_HEADER_LEN};
11
12use crate::backend::{FsBackend, ListPage};
13use crate::manager::VfsManager;
14
15/// Cursor bit marking a position among the mount-table entries; backend
16/// cursors stay below `backend::BACKEND_CURSOR_LIMIT`.
17const EXTRA_PHASE: u32 = 1 << 31;
18
19/// One entry the mount table contributes to a listing.
20struct Extra {
21    name: &'static str,
22    kind: EntryKind,
23    /// The backend lists this name itself; only its kind is overridden.
24    in_backend: bool,
25}
26
27impl VfsManager {
28    /// One page of directory `path`, starting at `cursor`. `None` for anything
29    /// the wire reports as `ERR_IO`: not a listable directory, a backend
30    /// iteration error, an invalid name, or a stale cursor.
31    pub fn list_page(&self, path: &str, cursor: u32, out: &mut [u8]) -> Option<ListPage> {
32        let backend = self.mounts.backend(path)?;
33        let extras = self.listing_extras(path, backend);
34        let (mut written, extra_start) = if cursor & EXTRA_PHASE == 0 {
35            let page = if matches!(backend.stat(path), Some((_, true))) {
36                backend.list_page(path, cursor, out)?
37            } else if cursor == 0 && self.is_synthetic_dir(path) {
38                ListPage { written: 0, next: None }
39            } else {
40                return None;
41            };
42            if page.next.is_some_and(|next| next & EXTRA_PHASE != 0) {
43                return None;
44            }
45            mark_mount_kinds(&mut out[..page.written], &extras);
46            if page.next.is_some() {
47                return (page.written > 0).then_some(page);
48            }
49            (page.written, 0)
50        } else {
51            (0, (cursor & !EXTRA_PHASE) as usize)
52        };
53        let pending: Vec<&Extra> = extras.iter().filter(|extra| !extra.in_backend).collect();
54        if extra_start > pending.len() {
55            return None;
56        }
57        for (index, extra) in pending.iter().enumerate().skip(extra_start) {
58            match push_entry(out, written, extra.kind, extra.name.as_bytes()) {
59                Ok(Some(end)) => written = end,
60                Ok(None) => {
61                    let next = Some(EXTRA_PHASE | index as u32);
62                    return (written > 0).then_some(ListPage { written, next });
63                }
64                Err(_) => return None,
65            }
66        }
67        Some(ListPage { written, next: None })
68    }
69
70    /// A directory the mount table supplies although no backend stores it.
71    fn is_synthetic_dir(&self, path: &str) -> bool {
72        self.is_mount_ancestor(path) || matches!(path, "/etc" | "/etc/cellos")
73    }
74
75    /// Mount children of `path` (kind `Mount` for a mount point, `Dir` for a
76    /// directory that only leads to one), then the configuration names, the
77    /// same set `list_dir` adds.
78    fn listing_extras(&self, path: &str, backend: &dyn FsBackend) -> Vec<Extra> {
79        let mut extras: Vec<Extra> = Vec::new();
80        for name in self.mounts.direct_mount_children(path) {
81            let child = join_component(path, name);
82            let kind = if self.mounts.is_mount_point(&child) { EntryKind::Mount } else { EntryKind::Dir };
83            extras.push(Extra { name, kind, in_backend: backend.stat(&child).is_some() });
84        }
85        let configured: &[&'static str] = match path {
86            "/" => &["etc"],
87            "/etc" => &["cellos"],
88            "/etc/cellos" => &["system.toml", "services.toml", "autoload.toml"],
89            _ => &[],
90        };
91        for name in configured {
92            if extras.iter().any(|extra| extra.name == *name) {
93                continue;
94            }
95            let child = join_component(path, name);
96            let Some((_, directory)) = self.stat(&child) else {
97                continue;
98            };
99            let kind = if directory { EntryKind::Dir } else { EntryKind::File };
100            extras.push(Extra { name, kind, in_backend: backend.stat(&child).is_some() });
101        }
102        extras
103    }
104}

106/// Report backend entries that are mount points as `Mount`: the backend sees
107/// its own directory there, but every request below it goes to another mount.
108fn mark_mount_kinds(page: &mut [u8], extras: &[Extra]) {
109    let mut pos = 0;
110    while pos + ENTRY_HEADER_LEN <= page.len() {
111        let end = pos + ENTRY_HEADER_LEN + usize::from(page[pos + 1]);
112        let Some(name) = page.get(pos + ENTRY_HEADER_LEN..end) else {
113            return;
114        };
115        if let Some(extra) = extras.iter().find(|extra| extra.in_backend && extra.name.as_bytes() == name) {
116            page[pos] = extra.kind.wire();
117        }
118        pos = end;
119    }
