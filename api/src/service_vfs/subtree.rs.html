1//! Enumerating a directory tree's files before it is deleted.
2//!
3//! `FsBackend::rmdir_recursive` returns only a bool, so the bytes it frees have to
4//! be measured while the tree still exists. Quota must also be released *per
5//! file*, because two files in one tree can have been charged to two different
6//! cells — so the walk yields paths and sizes, not a single total.
7
8use alloc::string::String;
9use alloc::vec::Vec;
10
11use crate::manager::VfsManager;
12
13/// Every regular file under `path`, as `(full path, size in bytes)`.
14///
15/// Bounded to `depth` levels of recursion and to a 512-byte listing buffer per
16/// level, which is adequate for the directory sizes this filesystem carries. A
17/// tree deeper or wider than those bounds is under-reported, which under-releases
18/// quota — the safe direction, since over-releasing would hand out free quota.
19pub fn files_under(vfs: &VfsManager, path: &str, depth: u8) -> Vec<(String, u64)> {
20    let mut out = Vec::new();
21    collect(vfs, path, depth, &mut out);
22    out
23}
24
25fn collect(vfs: &VfsManager, path: &str, depth: u8, out: &mut Vec<(String, u64)>) {
26    if depth == 0 {
27        return;
28    }
29    let mut scratch = [0u8; 512];
30    let n = vfs.list_dir(path, &mut scratch);
31    let listing = core::str::from_utf8(&scratch[..n]).unwrap_or("");
32    let base = path.trim_end_matches('/');
33    for line in listing.split('\n') {
34        if let Some(name) = line.strip_prefix("f:") {
35            let child = join(base, name);
36            let size = vfs.file_size(&child);
37            out.push((child, size));
38        } else if let Some(name) = line.strip_prefix("d:") {
39            collect(vfs, &join(base, name), depth - 1, out);
40        }
41    }
42}
43
44fn join(base: &str, name: &str) -> String {
45    let mut child = String::with_capacity(base.len() + 1 + name.len());
46    child.push_str(base);
47    child.push('/');
48    child.push_str(name);
49    child
50}
