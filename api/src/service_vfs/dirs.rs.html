1//! Directory handles, and the service's authority over them.
2//!
3//! A handle is an entry in this table naming one resolved directory and the cell
4//! allowed to use it. Nothing outside this table gives a handle meaning: the
5//! kernel carries handle values between cells and states where they came from,
6//! but it cannot tell a live handle from a revoked one, so every question about
7//! what a handle permits is answered here.
8//!
9//! ## Why a cell cannot name what it does not hold
10//! A handle-addressed request carries a directory and one component. The
11//! component is checked as raw bytes before anything is joined to it
12//! (`api::dir_name`), so it cannot contain a separator or be `..`. The resolved
13//! path is therefore always exactly one level below a directory the caller was
14//! given. There is no path outside the handle the caller can express, which is
15//! the difference between this and an access check.
16//!
17//! Lifetime, sealing and revocation live in [`lifecycle`]; binding an inherited
18//! set to a child lives in [`bind`].
19
20pub mod bind;
21pub mod lifecycle;

32/// Most handles one cell may hold at once.
33///
34/// Bounds the table against a cell that opens directories in a loop. Well above
35/// `api::dir_handles::MAX_SPAWN_DIR_HANDLES`, so a cell can always derive from
36/// everything it inherited.
37pub const MAX_HANDLES_PER_CELL: usize = 32;
38
39/// One issued handle.
40pub(crate) struct DirEntry {
41    pub(crate) owner: Caller,
42    /// Absolute directory this handle refers to, already validated.
43    pub(crate) path: String,
44    /// Handle this one was derived from; `None` for a directly acquired root.
45    /// Revoking the parent revokes this entry with it.
46    pub(crate) parent: Option<u64>,
47    /// A prepared-child reservation is charged and linked, but never usable.
48    pub(crate) published: bool,
49    /// A child of `path` was created, removed or renamed after this handle was
50    /// issued; reported as `vfs_dir_page::CHANGED` on every later page.
51    pub(crate) changed: bool,
52}
53
54/// Per-cell state that outlives any individual handle.
55pub(crate) struct CellState {
56    pub(crate) generation: u64,
57    /// Set once and never cleared: path-addressed requests are refused from
58    /// here on.
59    pub(crate) sealed: bool,
60    /// Whether the kernel's provenance record for this cell has been consumed.
61    /// One attempt per cell generation, successful or not — a record that could
62    /// be re-read is a record that could be bound twice.
63    pub(crate) attested: bool,
64    pub(crate) handles: usize,
65}
66
67/// Why a handle-addressed request was refused.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum DirError {
70    /// No such handle, or one belonging to another cell. The two are
71    /// deliberately the same answer: sweeping handle values must reveal nothing
72    /// about what other cells hold.
73    UnknownHandle,
74    /// The name may not be resolved inside a directory.
75    BadName(DirNameError),
76    /// The caller's handle limit or the issuer's non-reusable ID space is exhausted.
77    TooManyHandles,
78}

94/// The service's handle table.
95pub struct DirTable {
96    pub(crate) entries: BTreeMap<u64, DirEntry>,
97    pub(crate) cells: BTreeMap<u64, CellState>,
98    next: u64,
99    pub(crate) prepared: crate::prepared_cwd::PreparedCwd,
100}
101
102impl DirTable {
103    pub fn new() -> Self {
104        Self {
105            entries: BTreeMap::new(),
106            cells: BTreeMap::new(),
107            // 0 is never a handle: it is what a zeroed spawn carrier holds, and
108            // issuing it would make "no handle" and "this handle" the same value.
109            next: 1,
110            prepared: crate::prepared_cwd::PreparedCwd::default(),
111        }
112    }
113
114    /// Issue a handle for an already-validated absolute directory path.
115    ///
116    /// # Errors
117    /// [`DirError::TooManyHandles`] when the caller or issuer has no handle capacity, and
118    /// [`DirError::UnknownHandle`] for a caller with no registered state, which
119    /// cannot happen after `on_contact`.
120    pub fn open_root(&mut self, caller: Caller, path: &str) -> Result<ViDirHandle, DirError> {
121        self.insert(caller, String::from(path), None)
122    }
123
124    /// The directory `dir` refers to, if `caller` holds it.
125    pub fn dir_path(&self, caller: Caller, dir: ViDirHandle) -> Option<&str> {
126        self.owned(caller, dir).map(|e| e.path.as_str())
127    }
128
129    /// The absolute path of `name` inside `dir`.
130    ///
131    /// # Errors
132    /// [`DirError::UnknownHandle`] or [`DirError::BadName`]; the name is checked
133    /// as raw bytes before anything is joined, so a refusal happens before the
134    /// backend sees a string at all.
135    pub fn resolve(
136        &self,
137        caller: Caller,
138        dir: ViDirHandle,
139        name: &str,
140    ) -> Result<String, DirError> {
141        let entry = self.owned(caller, dir).ok_or(DirError::UnknownHandle)?;
142        let checked = validate_dir_component(name.as_bytes()).map_err(DirError::BadName)?;
143        Ok(join_component(&entry.path, checked))
144    }
145
146    /// How many handles `caller` holds.
147    pub fn held_by(&self, caller: Caller) -> usize {
148        self.state(caller).map_or(0, |s| s.handles)
149    }
150
151    /// Mark every handle on directory `path` (see `paths::may_alias`) as changed.
152    pub(crate) fn mark_changed(&mut self, path: &str) {
153        for entry in self.entries.values_mut().filter(|e| may_alias(&e.path, path)) {
154            entry.changed = true;
155        }
156    }
157
158    /// Mark every handle to `path` or to a directory below it as changed.
159    pub(crate) fn mark_changed_within(&mut self, path: &str) {
160        for entry in self.entries.values_mut().filter(|e| may_alias_within(&e.path, path)) {
161            entry.changed = true;
162        }
163    }
164
165    /// Whether a handle on directory `path` exists, i.e. whether a change to
166    /// its entries would be observed.
167    pub(crate) fn watches(&self, path: &str) -> bool {
168        self.entries.values().any(|e| may_alias(&e.path, path))
169    }
170
171    pub(crate) fn owned(&self, caller: Caller, dir: ViDirHandle) -> Option<&DirEntry> {
172        self.entries.get(&dir.0).filter(|e| e.owner == caller && e.published)
173    }
174
175    pub(crate) fn insert(
176        &mut self,
177        owner: Caller,
178        path: String,
179        parent: Option<u64>,
180    ) -> Result<ViDirHandle, DirError> {
181        let next = self.next.checked_add(1);
182        let state = self.state_mut(owner).ok_or(DirError::UnknownHandle)?;
183        if state.handles >= MAX_HANDLES_PER_CELL {
184            return Err(DirError::TooManyHandles);
185        }
186        let next = next.ok_or(DirError::TooManyHandles)?;
187        state.handles += 1;
188        let id = self.next;
189        self.next = next;
190        self.entries.insert(
191            id,
192            DirEntry {
193                owner,
194                path,
195                parent,
196                published: true,
197                changed: false,
198            },
199        );
200        Ok(ViDirHandle(id))
201    }
202}

204impl Default for DirTable {

210#[cfg(test)]
211mod tests {
212    use super::*;
213    use api::dir_attestation::ViDirHandleAttestation;
214    use api::dir_handles::{DirHandleSet, ViSpawnDirHandles};
215    use types::CellId;
216
217    #[test]
218    fn exhausted_directory_ids_preserve_existing_owner_and_counts() {
219        let owner = Caller::principal(CellId(11), 1);
220        let peer = Caller::principal(CellId(12), 1);
221        let mut table = DirTable::new();
222        table.on_contact(owner);
223        table.on_contact(peer);
224        table.next = u64::MAX - 1;
225        let live = table.open_root(owner, "/data/owner").unwrap();
226        assert_eq!(table.open_root(peer, "/data/peer"), Err(DirError::TooManyHandles));
227        assert_eq!(table.dir_path(owner, live), Some("/data/owner"));
228        assert_eq!(table.dir_path(peer, live), None);
229        assert_eq!(table.held_by(owner), 1);
230        assert_eq!(table.held_by(peer), 0);
231        table.revoke(owner, live).unwrap();
232        assert_eq!(table.open_root(peer, "/data/again"), Err(DirError::TooManyHandles));
233        assert_eq!(table.held_by(peer), 0);
234    }
235
236    #[test]
237    fn inherited_set_rolls_back_if_ticket_space_exhausts_mid_bind() {
238        let owner = Caller::principal(CellId(21), 1);
239        let child = Caller::principal(CellId(22), 1);
240        let mut table = DirTable::new();
241        table.on_contact(owner);
242        table.on_contact(child);
243        let first = table.open_root(owner, "/data/first").unwrap();
244        let second = table.open_root(owner, "/data/second").unwrap();
245        let record = ViDirHandleAttestation {
246            cell_id: child.cell.0,
247            generation: child.generation,
248            spawner_cell_id: owner.cell.0,
249            spawner_generation: owner.generation,
250            set: DirHandleSet::from_carrier(
251                &ViSpawnDirHandles::new(&[first, second]).unwrap()).unwrap(),
252        };
253        table.next = u64::MAX - 1;
254        assert_eq!(table.bind_inherited(child, &record), bind::BindOutcome::Refused);
255        assert_eq!(table.held_by(child), 0);
256        assert_eq!(table.entries.len(), 2);
257        assert_eq!(table.dir_path(owner, first), Some("/data/first"));
258        assert_eq!(table.dir_path(owner, second), Some("/data/second"));
259    }
