1//! Directory handles, and the service's authority over them.
2//!
3//! A handle is an entry in this table naming one resolved directory and the cell
4//! allowed to use it. Nothing outside this table gives a handle meaning: the
5//! kernel carries handle values between cells and states where they came from,
6//! but it cannot tell a live handle from a revoked one, so every question about
7//! what a handle permits is answered here.
8//!
9//! ## Why a cell cannot name what it does not hold
10//! A handle-addressed request carries a directory and one component. The
11//! component is checked as raw bytes before anything is joined to it
12//! (`api::dir_name`), so it cannot contain a separator or be `..`. The resolved
13//! path is therefore always exactly one level below a directory the caller was
14//! given. There is no path outside the handle the caller can express, which is
15//! the difference between this and an access check.
16//!
17//! Lifetime, sealing and revocation live in [`lifecycle`]; binding an inherited
18//! set to a child lives in [`bind`].
19
20pub mod bind;
21pub mod lifecycle;

31/// Most handles one cell may hold at once.
32///
33/// Bounds the table against a cell that opens directories in a loop. Well above
34/// `api::dir_handles::MAX_SPAWN_DIR_HANDLES`, so a cell can always derive from
35/// everything it inherited.
36pub const MAX_HANDLES_PER_CELL: usize = 32;
37
38/// One issued handle.
39pub(crate) struct DirEntry {
40    pub(crate) owner: Caller,
41    /// Absolute directory this handle refers to, already validated.
42    pub(crate) path: String,
43    /// Handle this one was derived from; `None` for a directly acquired root.
44    /// Revoking the parent revokes this entry with it.
45    pub(crate) parent: Option<u64>,
46}
47
48/// Per-cell state that outlives any individual handle.
49pub(crate) struct CellState {
50    pub(crate) generation: u64,
51    /// Set once and never cleared: path-addressed requests are refused from
52    /// here on.
53    pub(crate) sealed: bool,
54    /// Whether the kernel's provenance record for this cell has been consumed.
55    /// One attempt per cell generation, successful or not — a record that could
56    /// be re-read is a record that could be bound twice.
57    pub(crate) attested: bool,
58    pub(crate) handles: usize,
59}
60
61/// Why a handle-addressed request was refused.
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub enum DirError {
64    /// No such handle, or one belonging to another cell. The two are
65    /// deliberately the same answer: sweeping handle values must reveal nothing
66    /// about what other cells hold.
67    UnknownHandle,
68    /// The name may not be resolved inside a directory.
69    BadName(DirNameError),
70    /// The caller holds [`MAX_HANDLES_PER_CELL`] already.
71    TooManyHandles,
72}

74pub(crate) struct RevokeOutcome {
75    pub(crate) count: usize,
76    pub(crate) revoked_ids: Vec<u64>,
77}

79impl RevokeOutcome {

85    }

88/// The service's handle table.
89pub struct DirTable {
90    pub(crate) entries: BTreeMap<u64, DirEntry>,
91    pub(crate) cells: BTreeMap<u64, CellState>,
92    next: u64,
93}
94
95impl DirTable {
96    pub fn new() -> Self {
97        Self {
98            entries: BTreeMap::new(),
99            cells: BTreeMap::new(),
100            // 0 is never a handle: it is what a zeroed spawn carrier holds, and
101            // issuing it would make "no handle" and "this handle" the same value.
102            next: 1,
103        }
104    }
105
106    /// Issue a handle for an already-validated absolute directory path.
107    ///
108    /// # Errors
109    /// [`DirError::TooManyHandles`] when the caller is at its limit, and
110    /// [`DirError::UnknownHandle`] for a caller with no registered state, which
111    /// cannot happen after `on_contact`.
112    pub fn open_root(&mut self, caller: Caller, path: &str) -> Result<ViDirHandle, DirError> {
113        self.insert(caller, String::from(path), None)
114    }
115
116    /// The directory `dir` refers to, if `caller` holds it.
117    pub fn dir_path(&self, caller: Caller, dir: ViDirHandle) -> Option<&str> {
118        self.owned(caller, dir).map(|e| e.path.as_str())
119    }
120
121    /// The absolute path of `name` inside `dir`.
122    ///
123    /// # Errors
124    /// [`DirError::UnknownHandle`] or [`DirError::BadName`]; the name is checked
125    /// as raw bytes before anything is joined, so a refusal happens before the
126    /// backend sees a string at all.
127    pub fn resolve(
128        &self,
129        caller: Caller,
130        dir: ViDirHandle,
131        name: &str,
132    ) -> Result<String, DirError> {
133        let entry = self.owned(caller, dir).ok_or(DirError::UnknownHandle)?;
134        let checked = validate_dir_component(name.as_bytes()).map_err(DirError::BadName)?;
135        Ok(join_component(&entry.path, checked))
136    }
137
138    /// How many handles `caller` holds.
139    pub fn held_by(&self, caller: Caller) -> usize {
140        self.state(caller).map_or(0, |s| s.handles)
141    }
142
143    pub(crate) fn owned(&self, caller: Caller, dir: ViDirHandle) -> Option<&DirEntry> {
144        self.entries.get(&dir.0).filter(|e| e.owner == caller)
145    }
146
147    pub(crate) fn insert(
148        &mut self,
149        owner: Caller,
150        path: String,
151        parent: Option<u64>,
152    ) -> Result<ViDirHandle, DirError> {
153        let state = self.state_mut(owner).ok_or(DirError::UnknownHandle)?;
154        if state.handles >= MAX_HANDLES_PER_CELL {
155            return Err(DirError::TooManyHandles);
156        }
157        state.handles += 1;
158        let id = self.next;
159        self.next = self.next.saturating_add(1);
160        self.entries.insert(
161            id,
162            DirEntry {
163                owner,
164                path,
165                parent,
166            },
167        );
168        Ok(ViDirHandle(id))
169    }
170}
171
172impl Default for DirTable {
173    fn default() -> Self {
174        Self::new()
175    }
