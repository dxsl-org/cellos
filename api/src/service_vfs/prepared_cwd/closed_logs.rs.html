1//! Retirement of a live creator's per-Operation request logs.
2//!
3//! A log fences two ID sequences for one `(creator, Operation)` scope: the creator's
4//! requests to VFS and VFS's own kernel mutations in that Operation. While the
5//! Operation exists, dropping the log would let a retransmitted request ID run again
6//! and would restart `kernel_next` against the kernel's ledger. Once the kernel has
7//! removed the Operation, no request in the scope can be admitted again: ReserveCwd
8//! resolves the prepared record (removed with the Operation), Commit/Resolve need a
9//! reservation, and Acknowledge/Query need the log. Operation references are never
10//! reused, so a removed scope cannot be named again.
11use alloc::vec::Vec;
12use api::cell_session::{KernelOp, ObjectRefV1, Status, Writer, HEADER_LEN};
13use types::CellId;
14
15use super::{kernel_query, scope, PreparedCwd, Scope, LOG_CHARGE};
16use crate::caller::Caller;
17use crate::manager::VfsManager;
18
19/// Logs that only the Operation's removal can still change, with the kernel ID they
20/// were acknowledged through: no reservation names the scope, no outcome is retained,
21/// and every kernel mutation was resolved and acknowledged. At least one kernel
22/// mutation is required, because the removal probe asks about VFS's own ledger in the
23/// Operation and the kernel creates that ledger with the first mutation.
24pub(super) fn quiescent(state: &PreparedCwd, only: Option<Caller>) -> Vec<(Scope, u64)> {
25    state
26        .logs
27        .iter()
28        .filter(|(key, log)| {
29            only.is_none_or(|c| key.0 == c.cell.0 && key.1 == c.generation)
30                && log.outcomes.is_empty()
31                && log.kernel_pending.is_empty()
32                && log.kernel_next > 1
33                && log.kernel_acked == log.kernel_next - 1
34                && !state
35                    .reservations
36                    .values()
37                    .any(|value| scope(value.owner, value.operation) == **key)
38        })
39        .map(|(key, log)| (*key, log.kernel_acked))
40        .collect()
41}

43/// The kernel keeps VFS's request ledger for an Operation until `remove_operation`.
44/// Re-acknowledging an already-acknowledged ID changes nothing on a live ledger and
45/// answers `UnknownRequest` once the ledger is gone. Any other answer, including a
46/// transport failure, keeps the log.
47fn operation_removed(key: Scope, acknowledged: u64) -> bool {
48    let mut body = [0u8; 24];
49    let mut w = Writer::new(&mut body);
50    let operation = ObjectRefV1 {
51        id: key.2,
52        generation: key.3,
53    };
54    if w.reference(operation).is_err() || w.u64(acknowledged).is_err() {
55        return false;
56    }
57    let mut out = [0u8; HEADER_LEN];
58    kernel_query(KernelOp::AcknowledgeRequests, &body, &mut out) == Err(Status::UnknownRequest)
59}

61/// Drop one log and return its charge to the creator.
62pub(super) fn drop_log(state: &mut PreparedCwd, vfs: &mut VfsManager, key: Scope) -> bool {
63    if state.logs.remove(&key).is_none() {
64        return false;
65    }
66    vfs.quota.release(CellId(key.0), LOG_CHARGE);
67    true
68}

70/// Retire every quiescent log (of `only`, when given) whose Operation the kernel has
71/// removed. Costs one kernel query per quiescent log.
72pub(super) fn retire_closed(state: &mut PreparedCwd, vfs: &mut VfsManager, only: Option<Caller>) {
73    for (key, acknowledged) in quiescent(state, only) {
74        if operation_removed(key, acknowledged) {
75            drop_log(state, vfs, key);
76        }
77    }
