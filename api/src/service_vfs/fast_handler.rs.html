1//! Fast-IPC handler: serves VfsRequest::GetFile without ecall overhead.
2
3use ostd::prelude::Mutex;
4
5use crate::manager::VfsManager;
6
7pub static GLOBAL_VFS: Mutex<Option<VfsManager>> = Mutex::new(None);
8
9/// Fast-IPC handler: serves VfsRequest::GetFile without ecall overhead.
10///
11/// Authorized exactly like the ecall path. It has to be: `GetFile` replies with a
12/// raw `DataPtr`, which in a single address space is permanent read authority that
13/// cannot be revoked once handed out — so an ungated fast path would make the gate
14/// on the ecall path decorative. `caller` comes from the kernel
15/// (`kernel::fast_ipc::call_vfs` resolves it from live scheduler state), never from
16/// an argument this cell's client controls; `None` means unattributable, which is
17/// refused.
18///
19/// A cell this service has never served over the ecall path is declined with a
20/// zero-length reply, which `call_vfs` callers treat as "fast path unavailable"
21/// and retry as an ordinary syscall.
22///
23/// # Safety
24/// Called with S-mode interrupts disabled (guaranteed by `ostd::fast_ipc::call_vfs`).
25pub unsafe fn vfs_fast_handler(
26    caller: Option<api::caller_identity::CallerIdentity>,
27    req: &api::ipc::VfsRequest<'_>,
28    out: &mut [u8; api::ipc::IPC_BUF_SIZE],
29) -> usize {
30    let resp = match caller.map(crate::caller::Caller::from_attested) {
31        None => api::ipc::VfsResponse::Err(3),
32        Some(caller) => match req {
33            api::ipc::VfsRequest::GetFile(path) => {
34                if let Some(vfs) = GLOBAL_VFS.lock().as_ref() {
35                    if !vfs.dirs.has_met(caller) {
36                        return 0;
37                    }
38                    if vfs.dirs.is_sealed(caller) || !vfs.access.can_read_fast(caller, path) {
39                        api::ipc::VfsResponse::Err(3)
40                    } else if let Some((ptr, len)) = vfs.get_file_ptr(path) {
41                        api::ipc::VfsResponse::DataPtr {
42                            ptr: ptr as u64,
43                            len: len as u64,
44                        }
45                    } else {
46                        api::ipc::VfsResponse::Err(1)
47                    }
48                } else {
49                    api::ipc::VfsResponse::Err(0xFF)
50                }
51            }
52            _ => api::ipc::VfsResponse::Err(0xFE),
53        },
54    };
55    api::ipc::encode(&resp, out).map(|s| s.len()).unwrap_or(0)
56}
