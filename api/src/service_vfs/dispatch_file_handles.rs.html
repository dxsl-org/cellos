11pub fn open_file<'a>(
12    vfs: &mut VfsManager,
13    caller: Caller,
14    dir: ViDirHandle,
15    name: &str,
16) -> VfsResponse<'a> {
17    if !caller.may_own_state() {
18        return VfsResponse::Err(ERR_DENIED);
19    }
20    let path = match resolve(vfs, caller, dir, name) {
21        Ok(path) => path,
22        Err(code) => return VfsResponse::Err(code),
23    };
24    if !vfs.access.can_read(caller, &path) {
25        return VfsResponse::Err(ERR_DENIED);
26    }
27    match vfs.stat(&path) {
28        Some((_, false)) => {}
29        Some((_, true)) | None => return VfsResponse::Err(ERR_IO),
30    }
31    let lease = if path == "/srv" || path.starts_with("/srv/") {
32        let Ok(k) = crate::namespace::NamespaceKey::parse(&path) else {
33            return VfsResponse::Err(ERR_DENIED);
34        };
35        let Ok(l) = vfs.ledger.acquire_service_handle(&k) else {
36            return VfsResponse::Err(ERR_DENIED);
37        };
38        Some(l)
39    } else {
40        None
41    };
42    match vfs.files.insert_leased(caller, &path, dir.0, lease) {
43        Ok(file) => VfsResponse::FileHandle(file),
44        Err(err) => VfsResponse::Err(file_err(err)),
45    }
46}
47pub fn read_file<'a>(
48    vfs: &mut VfsManager,
49    caller: Caller,
50    file: ViVfsFileHandle,
51    offset: u64,
52    max: u32,
53    resp_buf: &'a mut [u8; IPC_BUF_SIZE],
54) -> VfsResponse<'a> {
55    let path = match vfs.files.begin_sync_read(caller, file) {
56        Ok(path) => path,
57        Err(err) => return VfsResponse::Err(file_err(err)),
58    };
59    if !vfs.access.can_read(caller, &path) {
60        let _ = vfs.files.finish_sync_read(caller, file);
61        return VfsResponse::Err(ERR_DENIED);
62    }
63    let data = vfs.read_to_vec(&path);
64    if vfs.stat(&path).is_none() {
65        let _ = vfs.files.finish_sync_read(caller, file);
66        return VfsResponse::Err(ERR_IO);
67    }
68    let start = match usize::try_from(offset) {
69        Ok(start) => start,
70        Err(_) => {
71            let _ = vfs.files.finish_sync_read(caller, file);
72            return VfsResponse::Err(ERR_IO);
73        }
74    };
75    if max == 0 || start >= data.len() {
76        let _ = vfs.files.finish_sync_read(caller, file);
77        return VfsResponse::Data(&[]);
78    }
79    let n = usize::try_from(max)
80        .unwrap_or(MAX_INLINE_PAYLOAD)
81        .min(MAX_INLINE_PAYLOAD)
82        .min(data.len() - start);
83    resp_buf[..n].copy_from_slice(&data[start..start + n]);
84    let _ = vfs.files.finish_sync_read(caller, file);
85    VfsResponse::Data(&resp_buf[..n])
86}

88pub fn close_file<'a>(
89    vfs: &mut VfsManager,
90    caller: Caller,
91    file: ViVfsFileHandle,
92) -> VfsResponse<'a> {
93    if vfs.files.close(caller, file) {
94        VfsResponse::Ok
95    } else {
96        VfsResponse::Err(ERR_HANDLE)
97    }
98}

100fn file_err(err: FileHandleError) -> u8 {
101    match err {
102        FileHandleError::UnknownHandle => ERR_HANDLE,
103        FileHandleError::TooManyHandles | FileHandleError::Exhausted => ERR_QUOTA,
104    }
