1//! Typed IPC request dispatch — decodes one `VfsRequest` and produces the
2//! `VfsResponse`, routing filesystem ops through the MountTable.
3//!
4//! Cross-cutting policy lives here, not in backends: AccessTable authorization,
5//! quota accounting (net-delta on overwrite), async-read pending table, and
6//! zero-copy grant I/O.
7
8use api::services::proto::vfs::VfsError;
9
10use crate::caller::Caller;
11use crate::manager::VfsManager;
12
13/// Handle one decoded request on behalf of a caller the kernel has attested.
14///
15/// `caller` is `None` when the kernel did not vouch for the sender — an
16/// unattested delivery path, or a sender that died before it could be resolved.
17/// Every request is then refused: a service that guesses at an identity is a
18/// service with no authorization at all, and "some cell that owns nothing" is not
19/// a safe guess, because owning nothing still reads everything that is unowned.
20///
21/// `resp_buf` backs `VfsResponse::Data` payloads, so the response borrows it;
22/// callers encode before reusing the buffer.
23pub fn handle_request<'a>(
24    vfs: &mut VfsManager,
25    buf: &[u8],
26    caller: Option<Caller>,
27    resp_buf: &'a mut [u8; api::ipc::IPC_BUF_SIZE],
28) -> api::ipc::VfsResponse<'a> {
29    let Some(caller) = caller else {
30        return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
31    };
32
33    // Settle what the kernel says this cell inherited before acting on anything
34    // it sent. The answer decides both what it holds and whether it may still
35    // name a path, and a cell that reached its first request before the second
36    // answer landed would be one path-string operation ahead of the seal.
37    crate::dir_admission::admit(vfs, caller);
38    if !crate::prepared_cwd::ready(vfs, caller) {
39        return api::ipc::VfsResponse::Err(VfsError::Io as u8);
40    }
41
42    // Decode typed request; `take_from_bytes` tolerates trailing bytes in the
43    // receive buffer — both zeros left by a previous message and the kernel's
44    // caller-identity trailer at the very end.
45    let req = match api::ipc::decode::<api::ipc::VfsRequest>(buf) {
46        Ok(r) => r,
47        Err(_) => return api::ipc::VfsResponse::Err(VfsError::Malformed as u8),
48    };
49
50    if let api::ipc::VfsRequest::PreparedChildV1(record) = req {
51        return crate::prepared_cwd::handle(vfs, caller, record, resp_buf);
52    }
53
54    // Centrally deny any mutating request from a caller lacking VfsMutate authority
55    // BEFORE path sealing, handle resolution, or backend work. The only exception
56    // is init's machine pin store: `Write`/`Mkdir` inside `/srv/.cellos-machine`
57    // from the live `MACHINE_PIN` provider (see `AccessTable::is_machine_pin_writer`).
58    if req.requires_mutation_authority()
59        && !caller.may_mutate()
60        && !machine_pin_write(vfs, caller, &req)
61    {
62        return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
63    }
64    // A cell that has given up path strings is refused here, before the request
65    // reaches an arm that could serve it. Refusing at the entry rather than in
66    // each arm is what makes the guarantee hold for every path-addressed
67    // operation, including any added later.
68    if req.is_path_addressed() && vfs.dirs.is_sealed(caller) {
69        return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
70    }
71
72    // A source may not disappear between issuer sealing and kernel commit.
73    // Abort-if-uncommitted linearizes against commit before the revoke walks
74    // the existing directory lineage.
75    if let api::ipc::VfsRequest::CloseDir { dir } = req {
76        if !crate::prepared_cwd::before_revoke(vfs, caller, dir) {
77            return api::ipc::VfsResponse::Err(VfsError::Io as u8);
78        }
79    }
80
81    // Phase 12 (CP1 W3): the SSH host-trust store moved to KMS. The three trust variants
82    // stay as tombstones so every later request keeps its byte; they are unsupported.
83    if matches!(
84        req,
85        api::ipc::VfsRequest::TrustLookup { .. }
86            | api::ipc::VfsRequest::TrustChallenge { .. }
87            | api::ipc::VfsRequest::TrustCommit { .. }
88    ) {
89        return api::ipc::VfsResponse::Err(VfsError::Unsupported as u8);
90    }
91    if let Some(resp) = crate::save::handle(vfs, caller, &req) {
92        return resp;
93    }
94    if let Some(resp) = crate::dispatch_paths::handle_path_request(vfs, caller, &req) {
95        return resp;
96    }
97
98    match req {
99        api::ipc::VfsRequest::ListDir(p) => {
100            if !vfs.access.can_read(caller, p) {
101                return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
102            }
103            if !vfs.healthy(p) {
104                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
105            }
106            if !matches!(vfs.stat(p), Ok(Some((_, true)))) {
107                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
108            }
109            let listed = vfs.list_dir(p, resp_buf);
110            if !vfs.healthy(p) {
111                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
112            }
113            let Ok(n) = listed else {
114                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
115            };
116            api::ipc::VfsResponse::Data(&resp_buf[..n])
117        }
118        api::ipc::VfsRequest::ReadAsync { path } => {
119            // Authorize BEFORE reading: the read happens now (the disk backend is
120            // still blocking) and `Poll` only hands over what was already read, so
121            // this is the only point where the path is known.
122            if !vfs.access.can_read(caller, path) {
123                return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
124            }
125            // A handle records durable state against the caller, so it needs a
126            // caller that cannot be confused with a later one.
127            if !caller.may_own_state() {
128                return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
129            }
130            if !vfs.healthy(path) || !matches!(vfs.stat(path), Ok(Some((_, false)))) {
131                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
132            }
133            let lease = if path.starts_with("/srv") {
134                let Ok(k) = crate::namespace::NamespaceKey::parse(path) else {
135                    return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
136                };
137                let Ok(l) = vfs.ledger.acquire_service_handle(&k) else {
138                    return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
139                };
140                Some(l)
141            } else {
142                None
143            };
144            let data = vfs.read_to_vec(path);
145            if !vfs.healthy(path) {
146                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
147            }
148            // A read that failed part way is an error, never a short handle.
149            let Ok(data) = data else {
150                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
151            };
152            let handle = vfs.pending.insert(caller, path, data, lease);
153            api::ipc::VfsResponse::PendingHandle(handle)
154        }
155
156        api::ipc::VfsRequest::Poll { handle } => {
157            // With a synchronous backend data is always ready on first poll.
158            //
159            // Two independent checks, in this order.  Ownership first: a handle
160            // owned by another cell is refused with the same code as a stale one,
161            // so sweeping the sequential handle space reveals nothing about other
162            // cells.  Then the path rules again — the slot stores the path it was
163            // filled from, because the authorization at `ReadAsync` time proves
164            // only what policy said then, and a slot can outlive a rule change.
165            let readable = match vfs.pending.owned_path(caller, handle) {
166                Some(path) => vfs.access.can_read(caller, path),
167                None => return api::ipc::VfsResponse::Err(VfsError::Handle as u8),
168            };
169            if !readable {
170                return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
171            }
172            if !vfs
173                .pending
174                .owned_path(caller, handle)
175                .is_some_and(|path| vfs.healthy(path))
176            {
177                let _ = vfs.pending.poll(caller, handle);
178                return api::ipc::VfsResponse::Err(VfsError::Io as u8);
179            }
180            match vfs.pending.poll(caller, handle) {
181                Some(data) => {
182                    // Cap at 480, not resp_buf.len(): the reply must still fit
183                    // the 512-byte IPC frame AFTER the postcard envelope. A
184                    // full 512-byte payload made encode fail and the client
185                    // saw an empty reply (surfaced by /bin ELF reads).
186                    let n = data.len().min(480);
187                    resp_buf[..n].copy_from_slice(&data[..n]);
188                    api::ipc::VfsResponse::Data(&resp_buf[..n])
189                }
190                None => api::ipc::VfsResponse::Err(VfsError::Handle as u8),
191            }
192        }
193
194        // ── Zero-Copy Grant I/O (Storage 2.0) ──────────────────────────────
195        api::ipc::VfsRequest::ReadGrant {
196            cap,
197            offset,
198            size,
199            grant,
200        } => crate::grant_read::read_grant(vfs, caller, cap, offset, size, grant),
201
202        api::ipc::VfsRequest::WriteGrant {
203            cap,
204            offset,
205            grant,
206            bytes,
207        } => {
208            if !caller.may_mutate() {
209                return api::ipc::VfsResponse::Err(VfsError::Denied as u8);
210            }
211            crate::grant_write::write(vfs, caller, cap, offset, grant, bytes)
212        }
213
214        api::ipc::VfsRequest::ReadFileGrant { path, grant, max } => {
215            crate::grant_read::read_file_grant(vfs, caller, path, grant, max)
216        }
217
218        // ── Directory capabilities ──────────────────────────────────────────
219        // Grouped and delegated: these arms share a resolution step that has no
220        // counterpart above, and interleaving them would put the two addressing
221        // models in one match where a reader has to check which one each arm is
222        // in.
223        api::ipc::VfsRequest::WriteHandleGrant { .. }
224        | api::ipc::VfsRequest::SyncHandle { .. }
225        | api::ipc::VfsRequest::WriteAt { .. }
226        | api::ipc::VfsRequest::UnlinkAt { .. }
227        | api::ipc::VfsRequest::MkdirAt { .. }
228        | api::ipc::VfsRequest::RmdirAt { .. }
229        | api::ipc::VfsRequest::RenameAt { .. }
230            if !caller.may_mutate() =>
231        {
232            api::ipc::VfsResponse::Err(VfsError::Denied as u8)
233        }
234
235        api::ipc::VfsRequest::OpenRootDir { .. }
236        | api::ipc::VfsRequest::OpenDir { .. }
237        | api::ipc::VfsRequest::ReadAt { .. }
238        | api::ipc::VfsRequest::WriteAt { .. }
239        | api::ipc::VfsRequest::StatAt { .. }
240        | api::ipc::VfsRequest::ListAt { .. }
241        | api::ipc::VfsRequest::ListAtPage { .. }
242        | api::ipc::VfsRequest::FileHandleState { .. }
243        | api::ipc::VfsRequest::UnlinkAt { .. }
244        | api::ipc::VfsRequest::CloseDir { .. }
245        | api::ipc::VfsRequest::SealPaths
246        | api::ipc::VfsRequest::OpenFileAt { .. }
247        | api::ipc::VfsRequest::ReadFileHandle { .. }
248        | api::ipc::VfsRequest::CloseFile { .. }
249        | api::ipc::VfsRequest::ReadHandleGrant { .. }
250        | api::ipc::VfsRequest::WriteHandleGrant { .. }
251        | api::ipc::VfsRequest::SyncHandle { .. }
252        | api::ipc::VfsRequest::MkdirAt { .. }
253        | api::ipc::VfsRequest::RmdirAt { .. }
254        | api::ipc::VfsRequest::RenameAt { .. } => {
255            crate::dispatch_dirs::handle(vfs, caller, &req, resp_buf)
256        }
257        _ => api::ipc::VfsResponse::Err(VfsError::Malformed as u8),
258    }
259}
260
261/// Whether `req` is the narrow machine pin exception to the central mutate
262/// gate: a path `Write` or `Mkdir` that `AccessTable::is_machine_pin_writer`
263/// admits. Any other request kind, path or caller still needs the mutate flag.
264fn machine_pin_write(vfs: &VfsManager, caller: Caller, req: &api::ipc::VfsRequest<'_>) -> bool {
265    match *req {
266        api::ipc::VfsRequest::Write { path, .. } | api::ipc::VfsRequest::Mkdir(path) => {
267            vfs.access.is_machine_pin_writer(caller, path)
268        }
269        _ => false,
270    }
271}

273#[cfg(test)]
274mod tests {
275    use api::caller_identity::CALLER_FLAG_NET_CONNECT;
276    use api::cell_session::ObjectRefV1;
277    use api::ipc::{VfsRequest, VfsResponse, IPC_BUF_SIZE};
278    use api::services::proto::vfs::VfsError;
279    use types::CellId;
280
281    use crate::caller::Caller;
282    use crate::manager::VfsManager;
283
284    /// Phase 12 W3: the host-trust store lives in KMS; the tombstones answer Unsupported
285    /// (byte 6) to every caller, including one holding the outbound-connect right.
286    #[test]
287    fn trust_tombstones_answer_unsupported() {
288        let caller = Caller {
289            flags: CALLER_FLAG_NET_CONNECT,
290            ..Caller::principal(CellId(91), 1)
291        };
292        let mut vfs = VfsManager::new();
293        let _ = vfs.dirs.on_contact(caller);
294        vfs.dirs.mark_attested(caller);
295        let session = ObjectRefV1 {
296            id: 41,
297            generation: 3,
298        };
299        for req in [
300            VfsRequest::TrustLookup {
301                host: "h",
302                port: 22,
303                algorithm: "ssh-ed25519",
304                key: b"k",
305            },
306            VfsRequest::TrustChallenge {
307                host: "h",
308                port: 22,
309                algorithm: "ssh-ed25519",
310                key: b"k",
311                session,
312                epoch: 4,
313            },
314            VfsRequest::TrustCommit { challenge: session },
315        ] {
316            let mut request = [0u8; IPC_BUF_SIZE];
317            api::ipc::encode(&req, &mut request).expect("request fits");
318            let mut resp_buf = [0u8; IPC_BUF_SIZE];
319            let resp = super::handle_request(&mut vfs, &request, Some(caller), &mut resp_buf);
320            assert!(
321                matches!(resp, VfsResponse::Err(code) if code == VfsError::Unsupported as u8),
322                "{req:?}: {resp:?}"
323            );
324        }
325        assert_eq!(VfsError::Unsupported as u8, 6);
326    }
