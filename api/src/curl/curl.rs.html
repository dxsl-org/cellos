13/// Maximum accumulated response size (stack-allocated; avoids the 4 MB alloc BSS).
14const RESP_BUF: usize = 4096;
15
16/// Wall-clock ceiling for the send + receive phases of one request.
17///
18/// The loopback mocks answer within microseconds, but a public-internet round
19/// trip (ARP, SYN, handshake, server think time) is orders of magnitude slower
20/// while every poll costs one IPC round-trip with the net service. A fixed
21/// iteration count therefore races the network; this budget does not.
22const EXCHANGE_BUDGET_MS: u64 = 30_000;
23/// Hard stop when the wall clock is unavailable or never advances.
24const EXCHANGE_POLL_CEILING: usize = 200_000;

30fn cell_main() {
31    // ── Parse argv ───────────────────────────────────────────────────────────
32    let argv = ostd::args();
33    if argv.is_empty() {
34        println("Usage: curl http://HOST[:PORT][/path]");
35        return;
36    }
37    let url = argv.first().map(|arg| arg.as_str()).unwrap_or("");
38
39    // ── Parse URL ────────────────────────────────────────────────────────────
40    let (host, port, path) = match parse_url(url) {
41        Some(t) => t,
42        None => {
43            println("curl: invalid URL — expected http://HOST[:PORT][/path]");
44            return;
45        }
46    };
47
48    // ── Resolve net service endpoint ──────────────────────────────────────────
49    let net_ep = match sys_lookup_service(service::NET) {
50        Some(ep) => ep,
51        None => {
52            println("curl: no net service");
53            return;
54        }
55    };
56
57    // ── Resolve the host through the service ─────────────────────────────────
58    // IPv4 literals and the SLIRP aliases are answered service-side, DNS names
59    // by its A-record resolver: the tools carry no resolver of their own.
60    let addr = match resolve_host(host, net_ep) {
61        Some(a) => a,
62        None => {
63            println("curl: cannot resolve host");
64            return;
65        }
66    };
67
68    // ── Guard: HTTP request must fit in the request_data stack buffer ─────────
69    let overhead =
70        b"GET ".len() + b" HTTP/1.0\r\nHost: ".len() + b"\r\nConnection: close\r\n\r\n".len();
71    if overhead + path.len() + host.len() > 500 {
72        println("curl: URL too long");
73        return;
74    }
75
76    // ── TcpConnect (atomic create + connect) ─────────────────────────────────
77    let mut req_buf = [0u8; IPC_BUF_SIZE];
78    let len = api::ipc::encode(&NetRequest::TcpConnect { addr, port }, &mut req_buf)
79        .map(|b| b.len())
80        .unwrap_or(0);
81    sys_send(net_ep, &req_buf[..len]);
82    let mut resp_buf = [0u8; IPC_BUF_SIZE];
83    let cap_id = match sys_recv(0, &mut resp_buf) {
84        SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&resp_buf) {
85            Ok(NetResponse::CapId(c)) => c,
86            _ => {
87                println("curl: connect failed");
88                return;
89            }
90        },
91        _ => {
92            println("curl: TcpConnect syscall failed");
93            return;
94        }
95    };
96
97    // ── Build HTTP GET request into a stack buffer ────────────────────────────
98    let mut request_data = [0u8; 512];
99    let mut pos = 0usize;
100    pos = wb(&mut request_data, pos, b"GET ");
101    pos = wb(&mut request_data, pos, path.as_bytes());
102    pos = wb(&mut request_data, pos, b" HTTP/1.0\r\nHost: ");
103    pos = wb(&mut request_data, pos, host.as_bytes());
104    pos = wb(&mut request_data, pos, b"\r\nConnection: close\r\n\r\n");
105    let request_len = pos;
106
107    // ── Send HTTP request via TcpSend with retry ──────────────────────────────
108    let started_ms = sys_get_time_ms().unwrap_or(0);
109    let mut sent_bytes = 0usize;
110    for _ in 0..EXCHANGE_POLL_CEILING {
111        if sent_bytes >= request_len || exchange_expired(started_ms) {
112            break;
113        }
114        let rem = &request_data[sent_bytes..request_len];
115        let chunk = rem.len().min(480);
116        let mut send_buf = [0u8; IPC_BUF_SIZE];
117        let send_len = api::ipc::encode(
118            &NetRequest::TcpSend {
119                cap_id,
120                data: &rem[..chunk],
121            },
122            &mut send_buf,
123        )
124        .map(|b| b.len())
125        .unwrap_or(0);
126        sys_send(net_ep, &send_buf[..send_len]);
127        let mut cnt_buf = [0u8; IPC_BUF_SIZE];
128        match sys_recv(0, &mut cnt_buf) {
129            SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&cnt_buf) {
130                Ok(NetResponse::Data(b)) if b.len() >= 4 => {
131                    let mut arr = [0u8; 4];
132                    arr.copy_from_slice(&b[0..4]);
133                    let n = u32::from_le_bytes(arr) as usize;
134                    sent_bytes += n;
135                    if n == 0 {
136                        sys_yield();
137                    }
138                }
139                _ => break,
140            },
141            _ => break,
142        }
143    }
144
145    // ── Accumulate HTTP response via TcpRecv ──────────────────────────────────
146    let mut response = [0u8; RESP_BUF];
147    let mut resp_len = 0usize;
148
149    // Pre-encode the TcpRecv request once and reuse it across all iterations.
150    let mut recv_req_buf = [0u8; IPC_BUF_SIZE];
151    let recv_req_len = api::ipc::encode(
152        &NetRequest::TcpRecv {
153            cap_id,
154            buf_len: 500,
155        },
156        &mut recv_req_buf,
157    )
158    .map(|b| b.len())
159    .unwrap_or(0);
160
161    'recv: for _ in 0..EXCHANGE_POLL_CEILING {
162        sys_send(net_ep, &recv_req_buf[..recv_req_len]);
163        let mut data_buf = [0u8; IPC_BUF_SIZE];
164        match sys_recv(0, &mut data_buf) {
165            SyscallResult::Ok(_) => {
166                match api::ipc::decode::<NetResponse>(&data_buf) {
167                    Ok(NetResponse::Data(b)) if !b.is_empty() => {
168                        let n = b.len().min(RESP_BUF - resp_len);
169                        response[resp_len..resp_len + n].copy_from_slice(&b[..n]);
170                        resp_len += n;
171                        if resp_len >= RESP_BUF {
172                            break 'recv;
173                        }
174                    }
175                    Ok(NetResponse::Data(_)) => {
176                        // 0 bytes: check TCP state to distinguish "nothing yet" from FIN.
177                        let st = query_state(cap_id, net_ep);
178                        if st == 0x06 || st == 0x00 {
179                            // CloseWait / Closed — drain one final recv before exiting.
180                            sys_send(net_ep, &recv_req_buf[..recv_req_len]);
181                            let mut fb = [0u8; IPC_BUF_SIZE];
182                            if let SyscallResult::Ok(_) = sys_recv(0, &mut fb) {
183                                if let Ok(NetResponse::Data(b)) =
184                                    api::ipc::decode::<NetResponse>(&fb)
185                                {
186                                    let n = b.len().min(RESP_BUF - resp_len);
187                                    response[resp_len..resp_len + n].copy_from_slice(&b[..n]);
188                                    resp_len += n;
189                                }
190                            }
191                            break 'recv;
192                        }
193                        if exchange_expired(started_ms) {
194                            break 'recv;
195                        }
196                        sys_yield();
197                    }
198                    _ => break,
199                }
200            }
201            _ => break,
202        }
203    }
204
205    // ── Print status line + body ─────────────────────────────────────────────
206    let resp = &response[..resp_len];
207    if let Some(header_end) = resp.windows(4).position(|w| w == b"\r\n\r\n") {
208        let status_end = resp.iter().position(|&b| b == b'\r').unwrap_or(header_end);
209        if let Ok(status) = core::str::from_utf8(&resp[..status_end]) {
210            println(status);
211        }
212        let body = &resp[header_end + 4..];
213        if let Ok(s) = core::str::from_utf8(body) {
214            print(s);
215        }
216    } else if resp_len > 0 {
217        if let Ok(s) = core::str::from_utf8(resp) {
218            print(s);
219        }
220    } else {
221        println("curl: empty response");
222    }
223
224    close_socket(cap_id, net_ep);
225}

227/// Has the request/response exchange exceeded its wall-clock budget?
228///
229/// A missing clock (`GetTime` unavailable) reports "not expired" so the
230/// iteration ceiling stays the only bound rather than cutting the exchange off
231/// at the first poll.
232fn exchange_expired(started_ms: u64) -> bool {
233    sys_get_time_ms().is_some_and(|now| now.saturating_sub(started_ms) > EXCHANGE_BUDGET_MS)
234}

236/// Send `NetRequest::SocketState` and return the 1-byte smoltcp state code.
237fn query_state(cap_id: u32, net_ep: usize) -> u8 {
238    let mut req_buf = [0u8; IPC_BUF_SIZE];
239    let len = api::ipc::encode(&NetRequest::SocketState { cap_id }, &mut req_buf)
240        .map(|b| b.len())
241        .unwrap_or(0);
242    sys_send(net_ep, &req_buf[..len]);
243    let mut resp_buf = [0u8; IPC_BUF_SIZE];
244    match sys_recv(0, &mut resp_buf) {
245        SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&resp_buf) {
246            Ok(NetResponse::State(s)) => s,
247            _ => 0x00,
248        },
249        _ => 0x00,
250    }
251}

253/// Send `NetRequest::TcpClose` and drain the acknowledgement.
254fn close_socket(cap_id: u32, net_ep: usize) {
255    let mut req_buf = [0u8; IPC_BUF_SIZE];
256    let len = api::ipc::encode(&NetRequest::TcpClose { cap_id }, &mut req_buf)
257        .map(|b| b.len())
258        .unwrap_or(0);
259    sys_send(net_ep, &req_buf[..len]);
260    let mut resp_buf = [0u8; IPC_BUF_SIZE];
261    let _ = sys_recv(0, &mut resp_buf);
262}

264/// Append `src` into `buf` at `pos`; returns new pos.
265fn wb(buf: &mut [u8], pos: usize, src: &[u8]) -> usize {
266    buf[pos..pos + src.len()].copy_from_slice(src);
267    pos + src.len()
268}

270fn parse_url(s: &str) -> Option<(&str, u16, &str)> {
271    let rest = s.strip_prefix("http://")?;
272    if rest.is_empty() {
273        return None;
274    }
275    let (host_port, path) = match rest.find('/') {
276        Some(i) => (&rest[..i], &rest[i..]),
277        None => (rest, "/"),
278    };
279    let (host, port) = match host_port.rfind(':') {
280        Some(i) => (&host_port[..i], parse_u16(&host_port[i + 1..])?),
281        None => (host_port, 80u16),
282    };
283    if host.is_empty() {
284        return None;
285    }
286    Some((host, port, path))
287}

289/// Resolve `host` through the net service (`NetRequest::Resolve`).
290///
291/// The service owns the whole resolution order — IPv4 literal, SLIRP alias
292/// (`gateway`, `host`, `dns`, `localhost`), then a UDP A-record query — so the
293/// tools never carry a second, drifting copy of it.
294fn resolve_host(host: &str, net_ep: usize) -> Option<[u8; 4]> {
295    let mut req_buf = [0u8; IPC_BUF_SIZE];
296    let len = api::ipc::encode(&NetRequest::Resolve { hostname: host }, &mut req_buf)
297        .ok()?
298        .len();
299    sys_send(net_ep, &req_buf[..len]);
300    let mut resp_buf = [0u8; IPC_BUF_SIZE];
301    match sys_recv(0, &mut resp_buf) {
302        SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&resp_buf) {
303            Ok(NetResponse::Addr(addr)) => Some(addr),
304            _ => None,
305        },
306        _ => None,
307    }
308}

310fn parse_u16(s: &str) -> Option<u16> {
311    let mut n: u32 = 0;
312    if s.is_empty() {
313        return None;
314    }
315    for ch in s.bytes() {
316        if !ch.is_ascii_digit() {
317            return None;
318        }
319        n = n * 10 + (ch - b'0') as u32;
320        if n > 65535 {
321            return None;
322        }
323    }
324    Some(n as u16)
325}
