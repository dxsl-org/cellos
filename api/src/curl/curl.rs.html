18fn cell_main() {
19    // ── Parse argv ───────────────────────────────────────────────────────────
20    let argv = ostd::args();
21    if argv.is_empty() {
22        println("Usage: curl http://IP[:PORT][/path]");
23        return;
24    }
25    let url = argv.first().map(|arg| arg.as_str()).unwrap_or("");
26
27    // ── Parse URL ────────────────────────────────────────────────────────────
28    let (host, port, path) = match parse_url(url) {
29        Some(t) => t,
30        None => {
31            println("curl: invalid URL — expected http://IP[:PORT][/path]");
32            return;
33        }
34    };
35    let addr = match resolve_host(host) {
36        Some(a) => a,
37        None => {
38            println("curl: invalid host");
39            return;
40        }
41    };
42
43    // ── Guard: HTTP request must fit in the request_data stack buffer ─────────
44    let overhead =
45        b"GET ".len() + b" HTTP/1.0\r\nHost: ".len() + b"\r\nConnection: close\r\n\r\n".len();
46    if overhead + path.len() + host.len() > 500 {
47        println("curl: URL too long");
48        return;
49    }
50
51    // ── Resolve net service endpoint ──────────────────────────────────────────
52    let net_ep = match sys_lookup_service(service::NET) {
53        Some(ep) => ep,
54        None => {
55            println("curl: no net service");
56            return;
57        }
58    };
59
60    // ── TcpConnect (atomic create + connect) ─────────────────────────────────
61    let mut req_buf = [0u8; IPC_BUF_SIZE];
62    let len = api::ipc::encode(&NetRequest::TcpConnect { addr, port }, &mut req_buf)
63        .map(|b| b.len())
64        .unwrap_or(0);
65    sys_send(net_ep, &req_buf[..len]);
66    let mut resp_buf = [0u8; IPC_BUF_SIZE];
67    let cap_id = match sys_recv(0, &mut resp_buf) {
68        SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&resp_buf) {
69            Ok(NetResponse::CapId(c)) => c,
70            _ => {
71                println("curl: connect failed");
72                return;
73            }
74        },
75        _ => {
76            println("curl: TcpConnect syscall failed");
77            return;
78        }
79    };
80
81    // ── Build HTTP GET request into a stack buffer ────────────────────────────
82    let mut request_data = [0u8; 512];
83    let mut pos = 0usize;
84    pos = wb(&mut request_data, pos, b"GET ");
85    pos = wb(&mut request_data, pos, path.as_bytes());
86    pos = wb(&mut request_data, pos, b" HTTP/1.0\r\nHost: ");
87    pos = wb(&mut request_data, pos, host.as_bytes());
88    pos = wb(&mut request_data, pos, b"\r\nConnection: close\r\n\r\n");
89    let request_len = pos;
90
91    // ── Send HTTP request via TcpSend with retry ──────────────────────────────
92    let mut sent_bytes = 0usize;
93    for _ in 0..500 {
94        if sent_bytes >= request_len {
95            break;
96        }
97        let rem = &request_data[sent_bytes..request_len];
98        let chunk = rem.len().min(480);
99        let mut send_buf = [0u8; IPC_BUF_SIZE];
100        let send_len = api::ipc::encode(
101            &NetRequest::TcpSend {
102                cap_id,
103                data: &rem[..chunk],
104            },
105            &mut send_buf,
106        )
107        .map(|b| b.len())
108        .unwrap_or(0);
109        sys_send(net_ep, &send_buf[..send_len]);
110        let mut cnt_buf = [0u8; IPC_BUF_SIZE];
111        match sys_recv(0, &mut cnt_buf) {
112            SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&cnt_buf) {
113                Ok(NetResponse::Data(b)) if b.len() >= 4 => {
114                    let mut arr = [0u8; 4];
115                    arr.copy_from_slice(&b[0..4]);
116                    let n = u32::from_le_bytes(arr) as usize;
117                    sent_bytes += n;
118                    if n == 0 {
119                        sys_yield();
120                    }
121                }
122                _ => break,
123            },
124            _ => break,
125        }
126    }
127
128    // ── Accumulate HTTP response via TcpRecv ──────────────────────────────────
129    let mut response = [0u8; RESP_BUF];
130    let mut resp_len = 0usize;
131
132    // Pre-encode the TcpRecv request once and reuse it across all iterations.
133    let mut recv_req_buf = [0u8; IPC_BUF_SIZE];
134    let recv_req_len = api::ipc::encode(
135        &NetRequest::TcpRecv {
136            cap_id,
137            buf_len: 500,
138        },
139        &mut recv_req_buf,
140    )
141    .map(|b| b.len())
142    .unwrap_or(0);
143
144    'recv: for _ in 0..500 {
145        sys_send(net_ep, &recv_req_buf[..recv_req_len]);
146        let mut data_buf = [0u8; IPC_BUF_SIZE];
147        match sys_recv(0, &mut data_buf) {
148            SyscallResult::Ok(_) => {
149                match api::ipc::decode::<NetResponse>(&data_buf) {
150                    Ok(NetResponse::Data(b)) if !b.is_empty() => {
151                        let n = b.len().min(RESP_BUF - resp_len);
152                        response[resp_len..resp_len + n].copy_from_slice(&b[..n]);
153                        resp_len += n;
154                        if resp_len >= RESP_BUF {
155                            break 'recv;
156                        }
157                    }
158                    Ok(NetResponse::Data(_)) => {
159                        // 0 bytes: check TCP state to distinguish "nothing yet" from FIN.
160                        let st = query_state(cap_id, net_ep);
161                        if st == 0x06 || st == 0x00 {
162                            // CloseWait / Closed — drain one final recv before exiting.
163                            sys_send(net_ep, &recv_req_buf[..recv_req_len]);
164                            let mut fb = [0u8; IPC_BUF_SIZE];
165                            if let SyscallResult::Ok(_) = sys_recv(0, &mut fb) {
166                                if let Ok(NetResponse::Data(b)) =
167                                    api::ipc::decode::<NetResponse>(&fb)
168                                {
169                                    let n = b.len().min(RESP_BUF - resp_len);
170                                    response[resp_len..resp_len + n].copy_from_slice(&b[..n]);
171                                    resp_len += n;
172                                }
173                            }
174                            break 'recv;
175                        }
176                        sys_yield();
177                    }
178                    _ => break,
179                }
180            }
181            _ => break,
182        }
183    }
184
185    // ── Print status line + body ─────────────────────────────────────────────
186    let resp = &response[..resp_len];
187    if let Some(header_end) = resp.windows(4).position(|w| w == b"\r\n\r\n") {
188        let status_end = resp.iter().position(|&b| b == b'\r').unwrap_or(header_end);
189        if let Ok(status) = core::str::from_utf8(&resp[..status_end]) {
190            println(status);
191        }
192        let body = &resp[header_end + 4..];
193        if let Ok(s) = core::str::from_utf8(body) {
194            print(s);
195        }
196    } else if resp_len > 0 {
197        if let Ok(s) = core::str::from_utf8(resp) {
198            print(s);
199        }
200    } else {
201        println("curl: empty response");
202    }
203
204    close_socket(cap_id, net_ep);
205}

207/// Send `NetRequest::SocketState` and return the 1-byte smoltcp state code.
208fn query_state(cap_id: u32, net_ep: usize) -> u8 {
209    let mut req_buf = [0u8; IPC_BUF_SIZE];
210    let len = api::ipc::encode(&NetRequest::SocketState { cap_id }, &mut req_buf)
211        .map(|b| b.len())
212        .unwrap_or(0);
213    sys_send(net_ep, &req_buf[..len]);
214    let mut resp_buf = [0u8; IPC_BUF_SIZE];
215    match sys_recv(0, &mut resp_buf) {
216        SyscallResult::Ok(_) => match api::ipc::decode::<NetResponse>(&resp_buf) {
217            Ok(NetResponse::State(s)) => s,
218            _ => 0x00,
219        },
220        _ => 0x00,
221    }
222}

224/// Send `NetRequest::TcpClose` and drain the acknowledgement.
225fn close_socket(cap_id: u32, net_ep: usize) {
226    let mut req_buf = [0u8; IPC_BUF_SIZE];
227    let len = api::ipc::encode(&NetRequest::TcpClose { cap_id }, &mut req_buf)
228        .map(|b| b.len())
229        .unwrap_or(0);
230    sys_send(net_ep, &req_buf[..len]);
231    let mut resp_buf = [0u8; IPC_BUF_SIZE];
232    let _ = sys_recv(0, &mut resp_buf);
233}

235/// Append `src` into `buf` at `pos`; returns new pos.
236fn wb(buf: &mut [u8], pos: usize, src: &[u8]) -> usize {
237    buf[pos..pos + src.len()].copy_from_slice(src);
238    pos + src.len()
239}

241fn parse_url(s: &str) -> Option<(&str, u16, &str)> {
242    let rest = s.strip_prefix("http://")?;
243    if rest.is_empty() {
244        return None;
245    }
246    let (host_port, path) = match rest.find('/') {
247        Some(i) => (&rest[..i], &rest[i..]),
248        None => (rest, "/"),
249    };
250    let (host, port) = match host_port.rfind(':') {
251        Some(i) => (&host_port[..i], parse_u16(&host_port[i + 1..])?),
252        None => (host_port, 80u16),
253    };
254    if host.is_empty() {
255        return None;
256    }
257    Some((host, port, path))
258}

260fn resolve_host(s: &str) -> Option<[u8; 4]> {
261    match s {
262        "gateway" | "host" => Some([10, 0, 2, 2]),
263        "dns" => Some([10, 0, 2, 3]),
264        "localhost" => Some([127, 0, 0, 1]),
265        _ => parse_ipv4(s),
266    }
267}

269fn parse_ipv4(s: &str) -> Option<[u8; 4]> {
270    let mut it = s.splitn(5, '.');
271    let a = parse_octet(it.next()?)?;
272    let b = parse_octet(it.next()?)?;
273    let c = parse_octet(it.next()?)?;
274    let d = parse_octet(it.next()?)?;
275    if it.next().is_some() {
276        return None;
277    }
278    Some([a, b, c, d])
279}

281fn parse_octet(s: &str) -> Option<u8> {
282    let mut n: u16 = 0;
283    if s.is_empty() {
284        return None;
285    }
286    for ch in s.bytes() {
287        if !ch.is_ascii_digit() {
288            return None;
289        }
290        n = n * 10 + (ch - b'0') as u16;
291        if n > 255 {
292            return None;
293        }
294    }
295    Some(n as u8)
296}

298fn parse_u16(s: &str) -> Option<u16> {
299    let mut n: u32 = 0;
300    if s.is_empty() {
301        return None;
302    }
303    for ch in s.bytes() {
304        if !ch.is_ascii_digit() {
305            return None;
306        }
307        n = n * 10 + (ch - b'0') as u32;
308        if n > 65535 {
309            return None;
310        }
311    }
312    Some(n as u16)
