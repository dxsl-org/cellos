2//! Transport-level feature policy (virtio 1.2 §6).
3
4use crate::error::InitFailure;
5
6/// `VIRTIO_F_VERSION_1`: the device follows the virtio 1.x layout. Without it
7/// the modern common configuration semantics this transport relies on (and
8/// little-endian device config) are not guaranteed.
9pub const VERSION_1: u64 = 1 << 32;
10
11/// `VIRTIO_F_ACCESS_PLATFORM`: the device's DMA goes through the platform's
12/// address translation (IOMMU). A device offering it may refuse to work if
13/// the driver declines it, so it is always accepted when offered.
14pub const ACCESS_PLATFORM: u64 = 1 << 33;
15
16/// Whether the device must offer [`ACCESS_PLATFORM`].
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub enum AccessPlatform {
19    /// Refuse the device unless it offers it: the cell's DMA addresses are
20    /// only meaningful through the platform's translation.
21    Required,
22    /// Accept the device either way; the feature is still negotiated whenever
23    /// it is offered.
24    Accept,
25}
26
27/// The feature word the driver writes, given what the device offers, what
28/// the device-specific driver asked for, and the access-platform policy.
29///
30/// Returns `requested` restricted to `device` (a driver must never accept an
31/// unoffered feature), plus [`VERSION_1`], plus [`ACCESS_PLATFORM`] when the
32/// device offers it.
33///
34/// # Errors
35/// [`InitFailure::Version1NotOffered`] when `device` lacks [`VERSION_1`];
36/// [`InitFailure::AccessPlatformMissing`] when `policy` is
37/// [`AccessPlatform::Required`] and `device` lacks [`ACCESS_PLATFORM`].
38pub fn driver_features(
39    device: u64,
40    requested: u64,
41    policy: AccessPlatform,
42) -> Result<u64, InitFailure> {
43    if device & VERSION_1 == 0 {
44        return Err(InitFailure::Version1NotOffered);
45    }
46    if policy == AccessPlatform::Required && device & ACCESS_PLATFORM == 0 {
47        return Err(InitFailure::AccessPlatformMissing);
48    }
49    Ok((requested & device) | VERSION_1 | (device & ACCESS_PLATFORM))
50}
