2//! `Transport` implementation: feature policy, status tracking, queue setup
3//! and notification.

5use super::{VirtioPciTransport, NOTIFY_CACHE_QUEUES, NOTIFY_UNCACHED};
6use crate::error::InitFailure;
7use crate::features::driver_features;
8use crate::regs::NO_VECTOR;
9use core::mem::{align_of, size_of};
10use core::ptr::NonNull;
11use virtio_drivers::transport::{DeviceStatus, DeviceType, Transport};

14/// ISR status bits: bit 0 queue interrupt, bit 1 configuration change.
15const ISR_ANY: u8 = 0b11;
16
17impl Transport for VirtioPciTransport {
18    fn device_type(&self) -> DeviceType {
19        self.device_type
20    }
21
22    fn read_device_features(&mut self) -> u64 {
23        self.common.device_features()
24    }
25
26    /// Writes `driver_features` amended by [`crate::features::driver_features`]
27    /// (re-reading the device's offer, checked against the construction-time
28    /// [`crate::AccessPlatform`] policy). On [`InitFailure::Version1NotOffered`]
29    /// or [`InitFailure::AccessPlatformMissing`] nothing is written and the
30    /// failure is recorded, which turns the following `FEATURES_OK` write into
31    /// `FAILED`.
32    fn write_driver_features(&mut self, driver_features_requested: u64) {
33        let device = self.common.device_features();
34        match driver_features(device, driver_features_requested, self.policy) {
35            Ok(features) => self.common.set_driver_features(features),
36            Err(failure) => self.fail(failure),
37        }
38    }
39
40    fn max_queue_size(&mut self, queue: u16) -> u32 {
41        self.common.select_queue(queue);
42        self.common.queue_size().into()
43    }
44
45    /// Writes `queue` to the queue's notify address. Nothing is written while
46    /// a failure is recorded, nor for queues whose notify address lies outside
47    /// the notify region (their `queue_set` recorded
48    /// [`InitFailure::NotifyOutOfRange`]).
49    fn notify(&mut self, queue: u16) {
50        if self.failure.is_some() {
51            return;
52        }
53        let cached = self
54            .notify_cache
55            .get(usize::from(queue))
56            .copied()
57            .filter(|&offset| offset != NOTIFY_UNCACHED);
58        let offset = cached.or_else(|| {
59            self.common.select_queue(queue);
60            self.selected_notify_offset()
61        });
62        if let Some(offset) = offset {
63            // SAFETY: offset + 2 <= notify_len (selected_notify_offset), the
64            // notify region lies inside a mapped BAR, and offset is even
65            // (even multiplier) on a 2-aligned base.
66            unsafe {
67                let addr = self.notify.as_ptr().add(offset as usize).cast::<u16>();
68                addr.write_volatile(queue);
69            }
70        }
71    }
72
73    fn get_status(&self) -> DeviceStatus {
74        DeviceStatus::from_bits_truncate(self.common.status().into())
75    }
76
77    /// Writing an empty status resets the device and waits, bounded by
78    /// [`crate::RESET_SPIN_BUDGET`], for the reset to complete (spec: before
79    /// reinitialising, device_status must read 0). A completed reset clears
80    /// any recorded failure and the notify cache; one that never completes
81    /// (or reads the device as gone) records [`InitFailure::ResetFailed`].
82    /// After a failure every other write becomes `FAILED`. A `FEATURES_OK`
83    /// write is read back: a cleared bit records
84    /// [`InitFailure::FeaturesRejected`]; an accepted one routes the
85    /// configuration-change interrupt to [`NO_VECTOR`].
86    fn set_status(&mut self, status: DeviceStatus) {
87        if status.is_empty() {
88            self.features_ok = false;
89            self.notify_cache = [NOTIFY_UNCACHED; NOTIFY_CACHE_QUEUES];
90            self.failure = (!self.common.reset()).then_some(InitFailure::ResetFailed);
91            return;
92        }
93        if let Some(failure) = self.failure {
94            self.fail(failure);
95            return;
96        }
97        self.common.set_status(status.bits() as u8);
98        if status.contains(DeviceStatus::FEATURES_OK) && !self.features_ok {
99            if self.get_status().contains(DeviceStatus::FEATURES_OK) {
100                self.features_ok = true;
101                self.common.set_msix_config(NO_VECTOR);
102            } else {
103                self.fail(InitFailure::FeaturesRejected);
104            }
105        }
106    }
107
108    fn set_guest_page_size(&mut self, _guest_page_size: u32) {}
109
110    fn requires_legacy_layout(&self) -> bool {
111        false
112    }
113
114    /// Programs and enables the queue with interrupts disabled
115    /// ([`NO_VECTOR`]) and caches its notify offset. While a failure is
116    /// recorded the queue is left untouched. A queue whose notify address
117    /// falls outside the notify region is neither programmed nor enabled, and
118    /// [`InitFailure::NotifyOutOfRange`] is recorded.
119    fn queue_set(
120        &mut self,
121        queue: u16,
122        size: u32,
123        descriptors: PhysAddr,
124        driver_area: PhysAddr,
125        device_area: PhysAddr,
126    ) {
127        if self.failure.is_some() {
128            return;
129        }
130        self.common.select_queue(queue);
131        let Some(offset) = self.selected_notify_offset() else {
132            self.fail(InitFailure::NotifyOutOfRange(queue));
133            return;
134        };
135        self.common.set_queue_size(size as u16);
136        self.common
137            .set_queue_areas(descriptors as u64, driver_area as u64, device_area as u64);
138        self.common.set_queue_msix_vector(NO_VECTOR);
139        if let Some(slot) = self.notify_cache.get_mut(usize::from(queue)) {
140            *slot = offset;
141        }
142        self.common.set_queue_enable(1);
143    }
144
145    /// No-op: virtio PCI has no way to disable a single enabled queue short
146    /// of resetting the device.
147    fn queue_unset(&mut self, _queue: u16) {}
148
149    fn queue_used(&mut self, queue: u16) -> bool {
150        self.common.select_queue(queue);
151        self.common.queue_enable() == 1
152    }
153
154    /// Reads (and thereby clears) the ISR status byte; true when a queue or
155    /// configuration-change interrupt was pending.
156    fn ack_interrupt(&mut self) -> bool {
157        // SAFETY: parse_caps proved the ISR region holds at least one byte
158        // inside a BAR the mapper mapped for our lifetime.
159        let isr = unsafe { self.isr.as_ptr().read_volatile() };
160        isr & ISR_ANY != 0
161    }
162
163    /// The device configuration structure as `T`.
164    ///
165    /// # Errors
166    /// `ConfigSpaceMissing` without a device-config capability,
167    /// `ConfigSpaceTooSmall` when `T` is longer than the region, and
168    /// `InvalidParam` when the region is not aligned for `T`.
169    fn config_space<T: 'static>(&self) -> Result<NonNull<T>, Error> {
170        let (ptr, len) = self.device_cfg.ok_or(Error::ConfigSpaceMissing)?;
171        if size_of::<T>() > len as usize {
172            return Err(Error::ConfigSpaceTooSmall);
173        }
174        if !(ptr.as_ptr() as usize).is_multiple_of(align_of::<T>()) {
175            return Err(Error::InvalidParam);
176        }
177        Ok(ptr.cast())
178    }
179}
