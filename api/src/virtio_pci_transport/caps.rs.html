2//! Virtio structure capabilities (virtio 1.2 §4.1.4) decoded from the
3//! capability bytes the kernel copies into the 434 record.
4//!
5//! Each vendor capability (`id` 0x09) starts with the standard header, so
6//! `bytes` holds `cap_vndr`, `cap_next`, `cap_len`, `cfg_type` (3), `bar` (4),
7//! `id`, padding, `offset` (8, u32 LE), `length` (12, u32 LE) and, for the
8//! notify structure only, `notify_off_multiplier` (16, u32 LE). The kernel
9//! bounds `len` by the capability's own `cap_len`, so `len` is the structure
10//! size the device advertised (capped at 20).
11
12use crate::error::TransportError;
13use crate::regs::COMMON_CFG_LEN;
14use abi::pcie_function::{
15    PcieCap, ViPcieFunctionV1, BAR_KIND_MEM32, BAR_KIND_MEM64, PCIE_FUNCTION_BARS,
16    PCIE_FUNCTION_CAPS_MAX,
17};
18
19/// PCI capability ID of a vendor-specific capability.
20pub const PCI_CAP_ID_VNDR: u8 = 0x09;
21
22/// `sizeof(struct virtio_pci_cap)`.
23const VIRTIO_CAP_LEN: u8 = 16;
24/// `sizeof(struct virtio_pci_notify_cap)`.
25const NOTIFY_CAP_LEN: u8 = 20;
26
27/// Virtio structure types this transport uses (`cfg_type` 1–4). Type 5
28/// (PCI configuration access) and vendor types are ignored.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum CfgType {
31    /// `VIRTIO_PCI_CAP_COMMON_CFG`.
32    Common = 1,
33    /// `VIRTIO_PCI_CAP_NOTIFY_CFG`.
34    Notify = 2,
35    /// `VIRTIO_PCI_CAP_ISR_CFG`.
36    Isr = 3,
37    /// `VIRTIO_PCI_CAP_DEVICE_CFG`.
38    Device = 4,
39}
40
41impl CfgType {
42    fn from_raw(raw: u8) -> Option<Self> {
43        match raw {
44            1 => Some(Self::Common),
45            2 => Some(Self::Notify),
46            3 => Some(Self::Isr),
47            4 => Some(Self::Device),
48            _ => None,
49        }
50    }
51
52    /// Bytes the transport accesses inside the structure; a shorter region
53    /// would put register accesses outside what the device decodes.
54    fn min_len(self) -> u32 {
55        match self {
56            Self::Common => COMMON_CFG_LEN as u32,
57            Self::Notify => 2,
58            Self::Isr => 1,
59            Self::Device => 0,
60        }
61    }
62}
63
64/// One virtio structure: `length` bytes at `offset` inside BAR `bar`.
65#[derive(Clone, Copy, Debug, PartialEq, Eq)]
66pub struct CapRegion {
67    pub bar: u8,
68    pub offset: u32,
69    pub length: u32,
70}
71
72/// The structures a modern virtio PCI function exposes, each validated to lie
73/// inside an allocated memory BAR.
74#[derive(Clone, Copy, Debug, PartialEq, Eq)]
75pub struct VirtioCaps {
76    pub common: CapRegion,
77    pub notify: CapRegion,
78    /// Bytes between consecutive `queue_notify_off` units; 0 means every
79    /// queue shares one notify address. Always even.
80    pub notify_off_multiplier: u32,
81    pub isr: CapRegion,
82    /// Optional: some device types (e.g. entropy) have no device config.
83    pub device: Option<CapRegion>,
84}

86fn le32(cap: &PcieCap, at: usize) -> u32 {
87    u32::from_le_bytes([
88        cap.bytes[at],
89        cap.bytes[at + 1],
90        cap.bytes[at + 2],
91        cap.bytes[at + 3],
92    ])
93}

95/// Decodes a capability the transport understands. The spec has drivers skip
96/// structures they cannot use rather than fail: truncated capabilities and
97/// reserved BAR indices (6..=255) are therefore ignored, not errors.
98fn understood(cap: &PcieCap) -> Option<(CfgType, CapRegion, u32)> {
99    if cap.id != PCI_CAP_ID_VNDR || cap.len < VIRTIO_CAP_LEN {
100        return None;
101    }
102    let kind = CfgType::from_raw(cap.bytes[3])?;
103    let bar = cap.bytes[4];
104    if bar as usize >= PCIE_FUNCTION_BARS {
105        return None;
106    }
107    let multiplier = match kind {
108        CfgType::Notify if cap.len < NOTIFY_CAP_LEN => return None,
109        CfgType::Notify => le32(cap, 16),
110        _ => 0,
111    };
112    let region = CapRegion {
113        bar,
114        offset: le32(cap, 8),
115        length: le32(cap, 12),
116    };
117    Some((kind, region, multiplier))
118}
119
120fn validate(
121    record: &ViPcieFunctionV1,
122    kind: CfgType,
123    region: CapRegion,
124) -> Result<CapRegion, TransportError> {
125    let bar = &record.bars[region.bar as usize];
126    if bar.kind != BAR_KIND_MEM32 && bar.kind != BAR_KIND_MEM64 {
127        return Err(TransportError::NotMemoryBar(region.bar));
128    }
129    if bar.base == 0 {
130        return Err(TransportError::BarNotAllocated(region.bar));
131    }
132    if u64::from(region.offset) + u64::from(region.length) > bar.len {
133        return Err(TransportError::RegionOutOfBar(kind));
134    }
135    if region.length < kind.min_len() {
136        return Err(TransportError::RegionTooSmall(kind));
137    }
138    Ok(region)
139}

141/// Finds and validates the common, notify, ISR and (optional) device
142/// structures of `record`.
143///
144/// The first understood capability of each type wins, as the spec directs;
145/// a later duplicate is never consulted, even when the first fails validation.
146///
147/// # Errors
148/// [`TransportError::LegacyOnly`] with no understood virtio capability,
149/// [`TransportError::MissingCap`] when common/notify/ISR is absent,
150/// [`TransportError::InvalidNotifyOffMultiplier`] for an odd multiplier, and
151/// the BAR errors of region validation (`NotMemoryBar`, `BarNotAllocated`,
152/// `RegionOutOfBar`, `RegionTooSmall`).
153pub fn parse_caps(record: &ViPcieFunctionV1) -> Result<VirtioCaps, TransportError> {
154    let count = (record.cap_count as usize).min(PCIE_FUNCTION_CAPS_MAX);
155    let mut found: [Option<(CapRegion, u32)>; 4] = [None; 4];
156    for (kind, region, multiplier) in record.caps[..count].iter().filter_map(understood) {
157        found[kind as usize - 1].get_or_insert((region, multiplier));
158    }
159    if found.iter().all(Option::is_none) {
160        return Err(TransportError::LegacyOnly);
161    }
162    let take = |kind: CfgType| found[kind as usize - 1];
163    let required = |kind: CfgType| -> Result<CapRegion, TransportError> {
164        let (region, _) = take(kind).ok_or(TransportError::MissingCap(kind))?;
165        validate(record, kind, region)
166    };
167    let common = required(CfgType::Common)?;
168    let notify = required(CfgType::Notify)?;
169    let isr = required(CfgType::Isr)?;
170    let notify_off_multiplier = take(CfgType::Notify).map_or(0, |(_, m)| m);
171    if notify_off_multiplier % 2 != 0 {
172        return Err(TransportError::InvalidNotifyOffMultiplier(
173            notify_off_multiplier,
174        ));
175    }
176    let device = match take(CfgType::Device) {
177        Some((region, _)) => Some(validate(record, CfgType::Device, region)?),
178        None => None,
179    };
180    Ok(VirtioCaps {
181        common,
182        notify,
183        notify_off_multiplier,
184        isr,
185        device,
186    })
