2//! PCI vendor/device ID → virtio device type (virtio 1.2 §4.1.2).
3
4use crate::error::TransportError;
5use virtio_drivers::transport::DeviceType;
6
7/// PCI vendor ID of every virtio function.
8pub const VIRTIO_VENDOR_ID: u16 = 0x1af4;
9
10/// Modern device IDs are `0x1040 + virtio device ID`.
11const MODERN_FIRST: u16 = 0x1040;
12const MODERN_LAST: u16 = 0x107f;
13
14/// Virtio device ID of a transitional PCI device ID. Transitional IDs are not
15/// an offset: the spec fixes this table and leaves the rest of
16/// `0x1000..=0x103f` unassigned.
17fn transitional_virtio_id(device_id: u16) -> Option<u8> {
18    match device_id {
19        0x1000 => Some(1), // network card
20        0x1001 => Some(2), // block device
21        0x1002 => Some(5), // memory ballooning (traditional)
22        0x1003 => Some(3), // console
23        0x1004 => Some(8), // SCSI host
24        0x1005 => Some(4), // entropy source
25        0x1009 => Some(9), // 9P transport
26        _ => None,
27    }
28}
29
30/// Maps a PCI function's IDs to the virtio device type its driver serves.
31///
32/// Transitional IDs are accepted because transitional devices also expose the
33/// modern capabilities; whether they do is checked by [`crate::parse_caps`].
34///
35/// # Errors
36/// [`TransportError::InvalidVendor`] unless `vendor_id` is
37/// [`VIRTIO_VENDOR_ID`]; [`TransportError::UnknownDevice`] for IDs outside
38/// both ranges, unassigned transitional IDs, and modern IDs whose virtio type
39/// the driver library does not define.
40pub fn device_type_for(vendor_id: u16, device_id: u16) -> Result<DeviceType, TransportError> {
41    if vendor_id != VIRTIO_VENDOR_ID {
42        return Err(TransportError::InvalidVendor(vendor_id));
43    }
44    let device_type = match device_id {
45        MODERN_FIRST..=MODERN_LAST => DeviceType::from(device_id - MODERN_FIRST),
46        _ => transitional_virtio_id(device_id).map_or(DeviceType::Invalid, DeviceType::from),
47    };
48    if device_type == DeviceType::Invalid {
49        return Err(TransportError::UnknownDevice(device_id));
50    }
51    Ok(device_type)
52}
