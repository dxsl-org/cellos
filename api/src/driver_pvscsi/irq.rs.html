1//! Completion waits: MSI-X/MSI wake-ups (H-S3-5) inside a time budget.
2//!
3//! An interrupt is only a hint to re-read the completion ring: every park is
4//! bounded by [`PARK_TICKS`], every return re-reads the ring, and the whole
5//! wait is bounded by the command's budget on the monotonic clock, so a lost,
6//! masked or revoked interrupt degrades to slow polling, never to a hang.
7//! Before each park the completion causes are acknowledged (`INTR_STATUS` is
8//! write-1-to-clear) and the ring re-read, so the next completion raises a
9//! fresh message. A refusal by 435 or 242 keeps the polled path and says why
10//! (`irq=polled reason=…`).
11//!
12//! Each park that ends by its deadline is classified by the ring re-read that
13//! follows it (phase-09 D6): `ring=ready` means the completion was already
14//! there but no message woke the cell (a late or lost interrupt); `ring=empty`
15//! means the device had not completed yet (a slow host). Both counts and the
16//! time since the wait began are logged.
17
18use crate::regs::{RegisterFile, INTR_CMPL_MASK, REG_INTR_STATUS};
19use abi::completion::source::IRQ;
20use abi::pcie_function::{IRQ_KIND_MSIX, IRQ_MODE_ALLOW_MSI, IRQ_MODE_ALLOW_MSIX};
21use core::sync::atomic::{AtomicBool, AtomicU32, Ordering};
22use ostd::io::print_fmt;
23use ostd::ipc::{wait_any, Wake};
24use ostd::syscall::{sys_alloc_pcie_irq, sys_get_time_ms};
25use types::ViResult;

27/// Bind-line evidence. It outlives the adapter's move into the disk layer,
28/// which owns its executor without exposing it.
29static WAKES: AtomicU32 = AtomicU32::new(0);
30static EXPIRIES: AtomicU32 = AtomicU32::new(0);
31/// Park expiries after which the ring re-read found the completion / none.
32static EXPIRY_READY: AtomicU32 = AtomicU32::new(0);
33static EXPIRY_EMPTY: AtomicU32 = AtomicU32::new(0);
34static WAIT_REFUSED: AtomicBool = AtomicBool::new(false);
35
36/// Parks ended by an interrupt / by their deadline, how the expired parks
37/// classify, and whether a refused wait dropped the adapter to polling.
38#[derive(Clone, Copy, Debug)]
39pub(crate) struct Evidence {
40    pub wakes: u32,
41    pub expiries: u32,
42    pub expiry_ready: u32,
43    pub expiry_empty: u32,
44    pub refused: bool,
45}
46
47/// Interrupt evidence so far.
48pub(crate) fn evidence() -> Evidence {
49    Evidence {
50        wakes: WAKES.load(Ordering::Relaxed),
51        expiries: EXPIRIES.load(Ordering::Relaxed),
52        expiry_ready: EXPIRY_READY.load(Ordering::Relaxed),
53        expiry_empty: EXPIRY_EMPTY.load(Ordering::Relaxed),
54        refused: WAIT_REFUSED.load(Ordering::Relaxed),
55    }
56}

58/// Longest single park, in scheduler ticks (10 ms each).
59const PARK_TICKS: u64 = 10;
60/// Per-command budget, including SYNCHRONIZE CACHE (a host `fsync`).
61pub(crate) const IO_BUDGET_MS: u64 = 30_000;
62/// Ring reads before the first park: a fast completion stays fast.
63const SPIN_BEFORE_PARK: u32 = 64;
64/// Polled path: yield (and read the clock) every this many ring reads.
65const POLL_YIELD_EVERY: u64 = 1024;
66/// The bound when no monotonic clock is available.
67const FALLBACK_POLLS: u64 = 50_000_000;
68/// Park expiries logged one by one per cell lifetime; after that only when
69/// the count reaches a power of two, so the classification stays visible.
70const EXPIRY_LOG_LIMIT: u32 = 3;
71
72/// How one park ended.
73enum Park {
74    Woken,
75    Expired,
76    /// The wait was refused; the adapter is now polled.
77    Refused,
78}
79
80pub(crate) struct CompletionIrq {
81    /// `"msix"` or `"msi"` with the vectors held, or `None` when polled.
82    mode: Option<(&'static str, u8)>,
83    /// Why the adapter is polled (`irq=polled reason=…`).
84    pub reason: &'static str,
85}
86
87impl CompletionIrq {
88    /// Ask 435 for one vector of the owned function `bdf`, MSI-X preferred.
89    pub(crate) fn setup(bdf: u32) -> Self {
90        let polled = |reason| Self { mode: None, reason };
91        // Without a clock the budget counts polls, and a park is not a poll.
92        if sys_get_time_ms().is_none() {
93            return polled("no-clock");
94        }
95        match sys_alloc_pcie_irq(bdf, 1, IRQ_MODE_ALLOW_MSIX | IRQ_MODE_ALLOW_MSI) {
96            Ok(alloc) => Self {
97                mode: Some((
98                    if alloc.kind == IRQ_KIND_MSIX {
99                        "msix"
100                    } else {
101                        "msi"
102                    },
103                    alloc.count,
104                )),
105                ..polled("")
106            },
107            Err(error) => {
108                let _ = print_fmt(format_args!("[pvscsi] interrupt refused ({error:?})\n"));
109                polled("alloc-refused")
110            }
111        }
112    }
113
114    /// `("msix" | "msi", vectors)`, or `None` when polled.
115    pub(crate) fn mode(&self) -> Option<(&'static str, u8)> {
116        self.mode
117    }
118
119    /// Wait until `ready` holds or [`IO_BUDGET_MS`] passes; `false` on timeout.
120    pub(crate) fn wait(
121        &mut self,
122        regs: &impl RegisterFile,
123        mut ready: impl FnMut() -> bool,
124    ) -> ViResult<bool> {
125        let start = sys_get_time_ms();
126        let expired = |polls: u64| match (start, sys_get_time_ms()) {
127            (Some(start), Some(now)) => now.saturating_sub(start) >= IO_BUDGET_MS,
128            _ => polls >= FALLBACK_POLLS,
129        };
130        for _ in 0..SPIN_BEFORE_PARK {
131            if ready() {
132                return Ok(true);
133            }
134            core::hint::spin_loop();
135        }
136        let mut polls = 0u64;
137        loop {
138            if self.mode.is_some() {
139                regs.wr32(REG_INTR_STATUS, INTR_CMPL_MASK)?;
140                if ready() {
141                    return Ok(true);
142                }
143                let park = self.park();
144                if !matches!(park, Park::Refused) {
145                    let ready_now = ready();
146                    if matches!(park, Park::Expired) {
147                        let elapsed_ms = start
148                            .zip(sys_get_time_ms())
149                            .map_or(0, |(start, now)| now.saturating_sub(start));
150                        note_expiry(ready_now, elapsed_ms);
151                    }
152                    if ready_now {
153                        return Ok(true);
154                    }
155                    if expired(polls) {
156                        return Ok(false);
157                    }
158                    continue;
159                }
160            }
161            if ready() {
162                return Ok(true);
163            }
164            polls += 1;
165            if polls.is_multiple_of(POLL_YIELD_EVERY) {
166                ostd::task::yield_now();
167                if expired(polls) {
168                    return Ok(false);
169                }
170            }
171            core::hint::spin_loop();
172        }
173    }
174
175    /// Park on this task's lines for at most [`PARK_TICKS`]. A refusal drops
176    /// the adapter to polling.
177    fn park(&mut self) -> Park {
178        match wait_any(IRQ, PARK_TICKS) {
179            Ok(Wake::Irq(_)) => {
180                WAKES.fetch_add(1, Ordering::Relaxed);
181                Park::Woken
182            }
183            Ok(_) => Park::Expired,
184            Err(error) => {
185                let _ = print_fmt(format_args!(
186                    "[pvscsi] irq=polled reason=wait-refused ({error:?})\n"
187                ));
188                self.mode = None;
189                self.reason = "wait-refused";
190                WAIT_REFUSED.store(true, Ordering::Relaxed);
191                Park::Refused
192            }
193        }
194    }
195}

197/// Count one expired park by what the ring re-read after it found, and log it
198/// with the time since the wait began.
199fn note_expiry(ring_ready: bool, elapsed_ms: u64) {
200    let expiries = EXPIRIES.fetch_add(1, Ordering::Relaxed) + 1;
201    let class = if ring_ready {
202        &EXPIRY_READY
203    } else {
204        &EXPIRY_EMPTY
205    };
206    class.fetch_add(1, Ordering::Relaxed);
207    if should_log_expiry(expiries) {
208        let _ = print_fmt(format_args!(
209            "[pvscsi] irq wait expired after {PARK_TICKS} ticks; ring={} elapsed_ms={elapsed_ms} expiry_ready={} expiry_empty={}\n",
210            if ring_ready { "ready" } else { "empty" },
211            EXPIRY_READY.load(Ordering::Relaxed),
212            EXPIRY_EMPTY.load(Ordering::Relaxed),
213        ));
214    }
215}
216
217/// The first [`EXPIRY_LOG_LIMIT`] expiries, then each power-of-two count.
218fn should_log_expiry(expiries: u32) -> bool {
219    expiries <= EXPIRY_LOG_LIMIT || expiries.is_power_of_two()
220}
