1//! Request encoding, LUN re-encoding and completion decoding (pure;
2//! host-tested). Everything the device writes back is untrusted: decoding
3//! never panics, and a malformed completion is an error.
4
5use crate::error::PvscsiError;
6use crate::rings::{CmpDesc, ReqDesc, CMP_LEN, REQ_LEN, SENSE_LEN};
7use core::mem::offset_of;
8use scsi_block::{status, Completion, Sense};
9
10/// Request `flags`: data direction.
11const FLAG_DIR_NONE: u32 = 1 << 2;
12const FLAG_DIR_TOHOST: u32 = 1 << 3;
13const FLAG_DIR_TODEVICE: u32 = 1 << 4;
14/// SCSI-2 SIMPLE QUEUE TAG message code.
15const TAG_SIMPLE: u8 = 0x20;
16
17/// Host adapter (`host_status`) codes the cell distinguishes.
18const BTSTAT_SUCCESS: u16 = 0x00;
19const BTSTAT_DATA_UNDERRUN: u16 = 0x0C;
20const BTSTAT_SELTIMEO: u16 = 0x11;
21const BTSTAT_DATARUN: u16 = 0x12;
22
23/// The device LUN field for SAM LUN `sam`: single-level peripheral (`00 nn`)
24/// or flat (`01 nn`, LUN < 256) addressing becomes `[0, nn, 0, …]`.
25pub(crate) fn device_lun(sam: [u8; 8]) -> Result<[u8; 8], PvscsiError> {
26    let single_level = sam[2..] == [0; 6];
27    match sam[0] {
28        0x00 | 0x40 if single_level => Ok([0, sam[1], 0, 0, 0, 0, 0, 0]),
29        _ => Err(PvscsiError::UnsupportedLun(sam)),
30    }
31}

33/// One request, as the cell fills it.
34pub(crate) struct Request<'a> {
35    pub context: u64,
36    pub target: u8,
37    pub lun: [u8; 8],
38    pub cdb: &'a [u8],
39    /// Data direction: `Some(true)` device→host, `Some(false)` host→device.
40    pub to_host: Option<bool>,
41    pub data_addr: u64,
42    pub data_len: usize,
43    pub sense_addr: u64,
44}
45
46/// Encode `request` into a 128-byte request descriptor.
47pub(crate) fn encode_request(request: &Request<'_>) -> Result<[u8; REQ_LEN], PvscsiError> {
48    if request.cdb.len() > 16 {
49        return Err(PvscsiError::CdbTooLong(request.cdb.len()));
50    }
51    let lun = device_lun(request.lun)?;
52    let (flags, data_addr, data_len) = match (request.data_len, request.to_host) {
53        (0, _) | (_, None) => (FLAG_DIR_NONE, 0, 0),
54        (len, Some(true)) => (FLAG_DIR_TOHOST, request.data_addr, len as u64),
55        (len, Some(false)) => (FLAG_DIR_TODEVICE, request.data_addr, len as u64),
56    };
57    let mut d = [0u8; REQ_LEN];
58    let mut put = |at: usize, bytes: &[u8]| d[at..at + bytes.len()].copy_from_slice(bytes);
59    put(offset_of!(ReqDesc, context), &request.context.to_le_bytes());
60    put(offset_of!(ReqDesc, data_addr), &data_addr.to_le_bytes());
61    put(offset_of!(ReqDesc, data_len), &data_len.to_le_bytes());
62    put(
63        offset_of!(ReqDesc, sense_addr),
64        &request.sense_addr.to_le_bytes(),
65    );
66    put(
67        offset_of!(ReqDesc, sense_len),
68        &(SENSE_LEN as u32).to_le_bytes(),
69    );
70    put(offset_of!(ReqDesc, flags), &flags.to_le_bytes());
71    put(offset_of!(ReqDesc, cdb), request.cdb);
72    put(offset_of!(ReqDesc, cdb_len), &[request.cdb.len() as u8]);
73    put(offset_of!(ReqDesc, lun), &lun);
74    put(offset_of!(ReqDesc, tag), &[TAG_SIMPLE]);
75    put(offset_of!(ReqDesc, target), &[request.target]);
76    Ok(d)
77}

79/// Decode a 32-byte completion for the request `context` that asked for
80/// `requested` data bytes; `sense` is the sense area as the device left it.
81pub(crate) fn decode_completion(
82    cmp: &[u8; CMP_LEN],
83    context: u64,
84    requested: usize,
85    sense: &[u8],
86) -> Result<Completion, PvscsiError> {
87    let bytes = |at: usize, n: usize| {
88        let mut word = [0u8; 8];
89        word[..n].copy_from_slice(&cmp[at..at + n]);
90        u64::from_le_bytes(word)
91    };
92    if bytes(offset_of!(CmpDesc, context), 8) != context {
93        return Err(PvscsiError::BadCompletion);
94    }
95    let reported = bytes(offset_of!(CmpDesc, data_len), 8);
96    let residual = match bytes(offset_of!(CmpDesc, host_status), 2) as u16 {
97        BTSTAT_SUCCESS => 0,
98        // Short transfer. QEMU's model never fills `data_len`, so 0 means
99        // "amount unknown": the bounce buffer was zeroed before the command,
100        // and the disk layer parses what arrived by its own length fields.
101        BTSTAT_DATARUN | BTSTAT_DATA_UNDERRUN => match reported {
102            0 => 0,
103            n if n <= requested as u64 => requested as u64 - n,
104            _ => return Err(PvscsiError::BadCompletion),
105        },
106        BTSTAT_SELTIMEO => return Err(PvscsiError::NoTarget),
107        other => return Err(PvscsiError::HostStatus(other)),
108    };
109    let Ok(status) = u8::try_from(bytes(offset_of!(CmpDesc, scsi_status), 2)) else {
110        return Err(PvscsiError::BadCompletion);
111    };
112    let sense_len = bytes(offset_of!(CmpDesc, sense_len), 4) as usize;
113    let sense = (status == status::CHECK_CONDITION && sense_len > 0)
114        .then(|| Sense::decode(&sense[..sense_len.min(sense.len())]))
115        .flatten();
116    Ok(Completion {
117        status,
118        sense,
119        residual: residual as u32,
120    })
121}
