1//! Tier 2 positive execution and isolation verification cell.
2//!
3//! Verifies that a Tier 2 domain cell running under private SATP isolation can:
4//! 1. Be admitted to Tier 2 Paged Domain without SAS fallback.
5//! 2. Execute user code safely behind hardware MMU boundaries.
6//! 3. Perform dynamic memory allocations on its private heap.
7//! 4. Invoke system calls (Log, Time, Yield).
8//! 5. Exit cleanly with code 0.
9
10#![no_std]
11#![no_main]
12#![forbid(unsafe_code)]
13
14extern crate alloc;
15
16use alloc::vec::Vec;
17use ostd::io::println;
18use ostd::syscall::{
19    sys_exit, sys_grant_copy_from_slice, sys_grant_copy_to_slice, sys_grant_register,
20    sys_grant_unregister,
21};
22
23api::declare_manifest!(
24    block_io = false,
25    network = false,
26    spawn = false,
27    tier = api::manifest::PROTECTION_CLASS_UNTRUSTED
28);
29
30api::declare_syscalls![
31    Log,
32    Yield,
33    GetTime,
34    Exit,
35    GrantRegister,
36    GrantSlice,
37    GrantUnregister
38];
39
40ostd::cell_main!(cell_main);
41
42fn cell_main() {
43    println("[tier2-smoke] Starting Tier 2 Native Domain Cell execution under private SATP!");
44
45    // 1. Dynamic allocation in private domain heap (.bss arena)
46    let mut numbers = Vec::new();
47    for i in 1..=20 {
48        numbers.push(i * 5);
49    }
50    let sum: i32 = numbers.iter().sum();
51    assert_eq!(sum, 1050);
52    println("[tier2-smoke] Heap allocation verified (Vec len=20, sum=1050)");
53
54    // 2. String allocation and formatting
55    let msg = alloc::format!("Tier 2 message formatted on heap: val={}", sum);
56    assert!(msg.contains("val=1050"));
57    println("[tier2-smoke] String allocation and formatting verified");
58
59    // 3. Syscalls: Yield and Time
60    ostd::task::yield_now();
61    println("[tier2-smoke] Scheduler yield completed in Tier 2 domain");
62
63    // 4. Grant allocation, private SATP mapping, write, read, and unregister
64    let grant_size = 4096;
65    let reg_id = sys_grant_register(grant_size).expect("GrantRegister failed in Tier 2 domain");
66    let test_payload = b"Tier 2 Domain Grant Zero-Copy Buffer Content!";
67    let copied_in =
68        sys_grant_copy_from_slice(reg_id, test_payload).expect("sys_grant_copy_from_slice failed");
69    assert_eq!(copied_in, test_payload.len());
70
71    let mut read_buf = [0u8; 45];
72    let copied_out =
73        sys_grant_copy_to_slice(reg_id, &mut read_buf).expect("sys_grant_copy_to_slice failed");
74    assert_eq!(copied_out, test_payload.len());
75    assert_eq!(&read_buf, test_payload);
76    println(
77        "[tier2-smoke] Grant allocation, private SATP mapping, and RW verified in Tier 2 domain",
78    );
79
80    let unregistered = sys_grant_unregister(reg_id);
81    assert!(unregistered, "GrantUnregister failed in Tier 2 domain");
82    println("[tier2-smoke] Grant unregister and unmapping verified in Tier 2 domain");
83
84    println("[tier2-smoke] PASS: All Tier 2 runtime invariants verified successfully!");
85    sys_exit(0);
86}
