16/// Spawn bound for the sweep.
17///
18/// The refusal must come from the kernel, never from this loop: with the
19/// production ceilings (`MAX_CELLS` 64) it arrives at ~64, while the
20/// `cell-scale-experiment` kernel raises them so memory is what binds. Hitting
21/// this bound prints OOM_NOT_REACHED, so a truncated sweep can never be read as
22/// a capacity result.
23const SPAWN_BOUND: usize = 2100;
24
25/// Report the running count this often: the count *is* the measurement, and a run
26/// cut short by its boot window must still leave its progress in the log.
27const PROGRESS_EVERY: usize = 32;
28
29fn cell_main() {
30    match ostd::syscall::sys_mem_info() {
31        Err(_) => ostd::io::println("[a2a3-probe] MEMINFO_DENIED"),
32        Ok(_) => {
33            ostd::io::println("[a2a3-probe] MEMINFO_UNEXPECTEDLY_ALLOWED");
34            ostd::syscall::sys_exit(2);
35        }
36    }
37
38    // Heavy cells first, then the sweep. The D5 gate measures the light ceiling
39    // *with* M heavy cells resident (`docs/roadmap/beam-parity-backend-roadmap.md`
40    // §2.3), because the per-request server runs beside long-lived data cells.
41    let heavy: usize = ostd::args()
42        .first()
43        .and_then(|arg| arg.as_str().parse().ok())
44        .unwrap_or(0);
45    if heavy > 0 {
46        ostd::io::println(&alloc::format!("[a2a3-probe] heavy_resident={heavy}"));
47        for index in 0..heavy {
48            // A dedicated binary: its 16 MiB heap arena is declared at build time, so
49            // the light children (which are `bench-probe`) keep their small one.
50            match ostd::syscall::sys_spawn_pinned(
51                "/bin/heavy-probe",
52                TaskPriority::Normal as u8,
53                0,
54            ) {
55                SyscallResult::Ok(_) => {}
56                SyscallResult::Err(error) => {
57                    ostd::io::println(&alloc::format!(
58                        "[a2a3-probe] HEAVY_SPAWN_FAILED index={index} error={error:?}"
59                    ));
60                    ostd::syscall::sys_exit(5);
61                }
62            }
63        }
64        // Give them time to commit their grants before the light sweep starts
65        // consuming the same heap; the log shows the ordering either way.
66        for _ in 0..256 {
67            ostd::syscall::sys_yield();
68        }
69    }
70
71    for count in 0..SPAWN_BOUND {
72        if count > 0 && count % PROGRESS_EVERY == 0 {
73            ostd::io::println(&alloc::format!("[a2a3-probe] parked count={count}"));
74        }
75        ostd::syscall::sys_set_spawn_args("resp-echo");
76        match ostd::syscall::sys_spawn_pinned("/bin/bench-probe", TaskPriority::Normal as u8, 0) {
77            SyscallResult::Ok(_) => {}
78            SyscallResult::Err(SyscallError::OutOfMemory) => {
79                ostd::io::println(&alloc::format!(
80                    "[a2a3-probe] OOM_TYPED count={count} heavy={heavy}"
81                ));
82                ostd::syscall::sys_exit(0);
83            }
84            SyscallResult::Err(_) => {
85                ostd::io::println("[a2a3-probe] SPAWN_GENERIC_ERROR");
86                ostd::syscall::sys_exit(3);
87            }
88        }
89    }
90
91    ostd::io::println(&alloc::format!(
92        "[a2a3-probe] OOM_NOT_REACHED bound={SPAWN_BOUND}"
93    ));
94    ostd::syscall::sys_exit(4)
95}
