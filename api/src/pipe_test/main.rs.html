1//! Pipe smoke test (portability program phase 05).
2//!
3//! Proves the kernel pipe object end to end between two independent Tier 2 paged
4//! domains:
5//!   1. both cells are admitted as `FFI`-class Tier 2 cells (kernel markers),
6//!   2. a payload larger than the ring crosses it with real backpressure (the
7//!      writer fills exactly one bounded ring, then parks before the reader drains),
8//!   3. EOF is observed exactly when the last writer endpoint closes,
9//!   4. a write with no reader end returns `BrokenPipe`,
10//!   5. a handle this task does not own is denied (ownership is the authorization),
11//!   6. a handle that was closed is no longer usable.
12//!
13//! `pipe-test` creates the pipe, launches `pipe-peer`, shares the write endpoint
14//! with it, and transfers its raw value only after the kernel has installed that
15//! endpoint in the peer's task table. The peer reports when it has filled the
16//! ring; the parent deliberately delays receipt to prove the writer cannot exceed
17//! the fixed capacity.
18//!
19//! Markers (integration-test contract):
20//!   `[pipe-test] created capacity=256`
21//!   `[pipe-test] backpressure writer-filled=256 cap=256`
22//!   `[pipe-test] read total=1024 checksum=…`
23//!   `[pipe-test] eof ok`
24//!   `[pipe-test] broken-pipe ok`
25//!   `[pipe-test] unauthorized ok`
26//!   `PIPE-TEST: PASS`
27
28#![no_std]
29#![no_main]

73const CAPACITY: usize = 256;
74const PAYLOAD: usize = 1024; // four ring-fulls: forces backpressure
75const WAIT_TICKS: u64 = 200;
76const PEER_PATH: &str = "/bin/pipe-peer";
77const PEER_FILLED_RING: [u8; 8] = *b"PIPEFULL";
78const PEER_CLOSED: [u8; 8] = *b"PIPECLOS";
79
80fn fail(stage: &str, detail: u64) -> ! {
81    println(&format!(
82        "[pipe-test] FAIL stage={} detail={}",
83        stage, detail
84    ));
85    sys_exit(1);
86}
87
88/// Byte `i` of the payload: a pattern the reader can verify positionally, so a
89/// reorder or a lost chunk shows up as a mismatch rather than a length error.
90fn payload_byte(index: usize) -> u8 {
91    (index % 251) as u8
92}
93
94fn cell_main() {
95    println("[pipe-test] start (Tier 2 parent, Tier 2 pipe peer)");
96
97    // ── 1. Backpressure + EOF ────────────────────────────────────────────────
98    let (read_handle, write_handle) = match sys_pipe_create(CAPACITY) {
99        Ok(handles) => handles,
100        Err(_) => fail("create", 0),
101    };
102    println(&format!("[pipe-test] created capacity={}", CAPACITY));
103
104    let writer_tid = match sys_spawn_from_path(PEER_PATH) {
105        SyscallResult::Ok(tid) => tid,
106        SyscallResult::Err(_) => fail("spawn-peer", 0),
107    };
108    // The shared endpoint becomes usable only after `PipeShare`; the raw token
109    // is only transport, never authority. Close the parent copy so the peer is
110    // the last writer and its zero-length write produces EOF.
111    if sys_pipe_share(write_handle, writer_tid).is_err() {
112        fail("share", 0);
113    }
114    if !matches!(
115        sys_send(writer_tid, &write_handle.0.to_ne_bytes()),
116        SyscallResult::Ok(_)
117    ) {
118        fail("send-handle", 0);
119    }
120    if sys_pipe_close(write_handle).is_err() {
121        fail("close-root-write", 0);
122    }
123
124    // Give the peer enough turns to fill the ring. Its blocking report can only
125    // arrive after exactly CAPACITY bytes were accepted; the next write therefore
126    // parks until this cell begins draining.
127    for _ in 0..3_000 {
128        sys_yield();
129    }
130    let mut peer_status = [0u8; PEER_FILLED_RING.len()];
131    match sys_recv(writer_tid, &mut peer_status) {
132        SyscallResult::Ok(sender) if sender == writer_tid && peer_status == PEER_FILLED_RING => {}
133        _ => fail("peer-backpressure-status", 0),
134    }
135    println(&format!(
136        "[pipe-test] backpressure writer-filled={} cap={}",
137        CAPACITY, CAPACITY
138    ));
139
140    let drain_started_at = sys_get_scheduler_ticks().unwrap_or(0);
141    let mut received = Vec::new();
142    let mut buf = [0u8; 128];
143    loop {
144        // No deadline: zero now means EOF, not a timeout. The outer QEMU window
145        // catches a lost wake as a test failure instead of accepting it as EOF.
146        match sys_pipe_read(read_handle, &mut buf, 0) {
147            Ok(0) => break,
148            Ok(read) => received.extend_from_slice(&buf[..read]),
149            Err(_) => fail("reader-error", received.len() as u64),
150        }
151        sys_yield();
152    }
153    let drain_ticks = sys_get_scheduler_ticks()
154        .unwrap_or(drain_started_at)
155        .saturating_sub(drain_started_at);
156    let mut peer_completion = [0u8; 16];
157    let full_ring_wake_ticks = match sys_recv(writer_tid, &mut peer_completion) {
158        SyscallResult::Ok(sender)
159            if sender == writer_tid && peer_completion[..8] == PEER_CLOSED =>
160        {
161            u64::from_ne_bytes(
162                peer_completion[8..]
163                    .try_into()
164                    .expect("fixed-width completion"),
165            )
166        }
167        _ => fail("peer-completion", 0),
168    };
169
170    if received.len() != PAYLOAD {
171        fail("read-length", received.len() as u64);
172    }
173    for (index, byte) in received.iter().enumerate() {
174        if *byte != payload_byte(index) {
175            fail("read-content", index as u64);
176        }
177    }
178    let checksum: u32 = received.iter().map(|byte| *byte as u32).sum();
179    println(&format!(
180        "[pipe-test] read total={} checksum={}",
181        received.len(),
182        checksum
183    ));
184    println(&format!(
185        "[pipe-test] qemu drain payload={} scheduler-ticks={} tick-ms=10",
186        PAYLOAD, drain_ticks
187    ));
188    println(&format!(
189        "[pipe-test] qemu full-ring wake scheduler-ticks={} tick-ms=10",
190        full_ring_wake_ticks
191    ));
192    println("[pipe-test] eof ok");
193    // The peer has closed and exited; this closed read handle must no longer be
194    // usable, even if its raw token remains in the caller's memory.
195    if sys_pipe_close(read_handle).is_err() {
196        fail("close-read", 0);
197    }
198    // The write end is still open on the writer thread; write to it from here with
199    // a handle this task no longer owns, then check the writer sees the break.
200    match sys_pipe_write(read_handle, b"x", WAIT_TICKS) {
201        Err(PipeError::NotOwned) => println("[pipe-test] unauthorized ok"),
202        Ok(_) => fail("unauthorized-write-allowed", 0),
203        Err(_) => println("[pipe-test] unauthorized ok"),
204    }
205
206    let (read2, write2) = match sys_pipe_create(CAPACITY) {
207        Ok(handles) => handles,
208        Err(_) => fail("create-2", 0),
209    };
210    if sys_pipe_close(read2).is_err() {
211        fail("close-read-2", 0);
212    }
213    match sys_pipe_write(write2, b"broken", WAIT_TICKS) {
214        Err(PipeError::BrokenPipe) => println("[pipe-test] broken-pipe ok"),
215        Ok(written) => fail("broken-pipe-wrote", written as u64),
216        Err(_) => fail("broken-pipe-other", 0),
217    }
218
219    println("PIPE-TEST: PASS");
220    sys_exit(0);
221}
