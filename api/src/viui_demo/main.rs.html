35const DISPLAY_W: u32 = 640;
36const DISPLAY_H: u32 = 400;
37
38ostd::cell_main!(cell_main);
39
40fn cell_main() {
41    ostd::io::println("[viui-demo] starting managed Counter surface");
42
43    let compositor_tid = wait_for_compositor();
44    let surface =
45        match ViSurface::create(compositor_tid, DISPLAY_W, DISPLAY_H, PixelFormat::Bgra8888) {
46            Ok(surface) => surface,
47            Err(_) => {
48                ostd::io::println("[viui-demo] ERROR: could not create surface");
49                sys_exit(1);
50            }
51        };
52    if surface.set_title("ViUI Counter").is_err() {
53        ostd::io::println("[viui-demo] ERROR: could not set surface title");
54        sys_exit(1);
55    }
56
57    // Keep generated state alive so its signal subscriptions serve the live tree.
58    let (_counter_state, counter_root) = Counter::build();
59    let mut app = ManagedSurfaceApp::new(Box::new(counter_root), surface);
60    app.set_close_policy(ClosePolicy::Accept);
61
62    loop {
63        let events = collect_input_events(16);
64        if app.tick(&events) == ManagedTick::Closed {
65            ostd::io::println("[viui-demo] close request accepted");
66            app.shutdown();
67            sys_exit(0);
68        }
69        yield_now();
70    }
71}
