1//! Forward-edge CFI probe cell: AArch64 BTI and x86_64 CET-IBT.
2//!
3//! Law 4 carve-out: this cell MUST use `unsafe` to make an indirect call to an
4//! instruction that is not a landing pad — that is the entire test.
5//!
6//! ## What the probe executes
7//!
8//! `global_asm!` places two adjacent labels in this cell's own `.text`, which
9//! the loader maps executable for the cell, 16-byte aligned so both sit on one
10//! page:
11//!
12//! ```text
13//! cfi_test_padded_target:    BTI c (aarch64) | ENDBR64 (x86_64)
14//! cfi_test_unpadded_target:  mov w0/eax, CFI_PROBE_MAGIC
15//!                            ret
16//! ```
17//!
18//! 1. **Control** — an indirect call (`blr` / `call reg`) to
19//!    `cfi_test_padded_target`. It must return [`CFI_PROBE_MAGIC`]. This proves
20//!    the page is executable for the cell and the indirect-call mechanics work,
21//!    so a fault in step 2 cannot be an NX/permission fault.
22//! 2. **Violation** — the same indirect call to `cfi_test_unpadded_target`: the
23//!    very instruction step 1 just executed, so same page and permissions,
24//!    differing only in the missing landing pad.
25//!
26//! (An earlier version jumped into a byte array in `.rodata`; that page is
27//! non-executable, so the fault it produced proved NX, not BTI/IBT.)
28//!
29//! ## Opt-in
30//!
31//! The kernel arms forward-edge CFI only for an image whose signed ELF carries
32//! the GNU property note (`GNU_PROPERTY_AARCH64_FEATURE_1_AND` BTI /
33//! `GNU_PROPERTY_X86_FEATURE_1_AND` IBT; `kernel/src/loader/gnu_property.rs`).
34//! The linker emits it only when every object has landing pads, so this cell
35//! must be built in its own cargo invocation with
36//! `-Z branch-protection=bti` (aarch64) or `-Z cf-protection=branch` (x86_64)
37//! in RUSTFLAGS — `scripts/qemu-cfi-test.sh` does that. `build.rs` reports which
38//! kind of build this is as `cfg(cfi_landing_pads)`.
39//!
40//! ## What each outcome proves
41//!
42//! - **Violation faults** — forward-edge CFI is enforced for this cell. The
43//!   cell dies and the kernel log is the evidence, matched against the printed
44//!   `cfi_test_unpadded_target` address: on aarch64
45//!   `[fault-probe] a64 ... ec=0xd ... pc=<that address>` plus a `[bti] Branch
46//!   Target exception` line; on x86_64 `[cet] #CP (vector 21) at CPL3
47//!   pc=<that address>` and the `[fault] Cell ... cause=0x15` termination.
48//! - **Violation returns** — CFI was not enforced. The verdict depends on
49//!   whether it should have been:
50//!   - image built without landing pads: `cfi-test: SKIP:` (not opted in; the
51//!     kernel runs it exactly like any other cell), exit 0;
52//!   - x86_64, CPU does not enumerate IBT (CPUID.(EAX=7,ECX=0):EDX[20] = 0):
53//!     `cfi-test: SKIP:` with that reason, exit 0;
54//!   - x86_64, CPU enumerates IBT and the image opted in: `cfi-test: FAIL`,
55//!     exit 1 — the kernel should have set IA32_U_CET.ENDBR_EN;
56//!   - aarch64, image opted in: `cfi-test: SKIP: BTI not enforced` with a
57//!     reason, exit 0. EL0 cannot read `ID_AA64PFR1_EL1` (the access traps and
58//!     the kernel does not emulate it), so the cell cannot tell a CPU without
59//!     FEAT_BTI from a kernel that failed to arm it. The QEMU harness
60//!     (`scripts/qemu-cfi-test.sh`) is the authority: it fails the run when the
61//!     boot log shows `CFI: BTI enabled` and this line appears.
62//!   - riscv64: the kernel does not enable Zicfilp landing pads (no `LPE` bit
63//!     in `senvcfg`/`menvcfg`), so the cell skips without probing.
64//! - **Control fails** — the test itself is broken: `cfi-test: FAIL`, exit 1.
65//!
66//! Spawned UNSUPERVISED so init's never-die watchdog does not restart a
67//! deliberately crashing cell.
68

82/// Value both probe targets return: it distinguishes "the target ran and
83/// returned normally" from a stray return through some other path.
84#[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
85const CFI_PROBE_MAGIC: u32 = 0x5A;
86
87// Both labels are global + hidden so the Rust side can name them as functions
88// without a GOT entry, wherever rustc places this block's codegen unit.
89#[cfg(target_arch = "aarch64")]
90core::arch::global_asm!(
91    ".pushsection .text.cfi_test_targets, \"ax\", %progbits",
92    ".balign 16",
93    ".globl cfi_test_padded_target",
94    ".hidden cfi_test_padded_target",
95    ".type cfi_test_padded_target, %function",
96    "cfi_test_padded_target:",
97    // `hint #34` is `BTI c`, spelled as a hint so the assembler needs no BTI
98    // feature flag; it is the landing pad BLR requires on a guarded page.
99    "    hint #34",
100    ".globl cfi_test_unpadded_target",
101    ".hidden cfi_test_unpadded_target",
102    ".type cfi_test_unpadded_target, %function",
103    "cfi_test_unpadded_target:",
104    "    mov w0, #{magic}",
105    "    ret",
106    ".popsection",
107    magic = const CFI_PROBE_MAGIC,
108);
109
110#[cfg(target_arch = "x86_64")]
111core::arch::global_asm!(
112    ".pushsection .text.cfi_test_targets, \"ax\", @progbits",
113    ".balign 16",
114    ".globl cfi_test_padded_target",
115    ".hidden cfi_test_padded_target",
116    ".type cfi_test_padded_target, @function",
117    "cfi_test_padded_target:",
118    // ENDBR64 (F3 0F 1E FA), as raw bytes so the assembler needs no CET flag.
119    "    .byte 0xf3, 0x0f, 0x1e, 0xfa",
120    ".globl cfi_test_unpadded_target",
121    ".hidden cfi_test_unpadded_target",
122    ".type cfi_test_unpadded_target, @function",
123    "cfi_test_unpadded_target:",
124    "    mov eax, {magic}",
125    "    ret",
126    ".popsection",
127    magic = const CFI_PROBE_MAGIC,
128);

130#[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
131extern "C" {
132    /// Landing pad followed by `cfi_test_unpadded_target`; returns
133    /// [`CFI_PROBE_MAGIC`].
134    fn cfi_test_padded_target() -> u32;
135    /// The instruction after the landing pad; returns [`CFI_PROBE_MAGIC`] when
136    /// forward-edge CFI is not enforced.
137    fn cfi_test_unpadded_target() -> u32;

140/// Signature of both probe labels.
141#[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
142type ProbeFn = unsafe extern "C" fn() -> u32;

144fn cfi_handler(_ctx: &mut AppContext, event: AppEvent) {
145    match event {

152/// Indirect call through `BLR` to `target`, returning its `w0`.
153///
154/// # Safety
155/// `target` must be one of the two probe labels above: code in this cell's
156/// executable `.text` that follows the C ABI, takes no arguments, and touches
157/// nothing but `x0`.
158#[cfg(target_arch = "aarch64")]
159unsafe fn indirect_call(target: ProbeFn) -> u32 {
160    let ret: u64;
161    // SAFETY: the caller guarantees `target` is C-ABI code with no arguments;
162    // clobber_abi("C") covers every register such a callee may clobber,
163    // including x30, which BLR itself overwrites.
164    unsafe {
165        core::arch::asm!(
166            "blr {target}",
167            target = in(reg) target,
168            lateout("x0") ret,
169            clobber_abi("C"),
170        );
171    }
172    ret as u32
173}
174
175/// Indirect call through `CALL reg` to `target`, returning its `eax`.
176///
177/// # Safety
178/// `target` must be one of the two probe labels above: code in this cell's
179/// executable `.text` that follows the C ABI, takes no arguments, and touches
180/// nothing but `rax`.
181#[cfg(target_arch = "x86_64")]
182unsafe fn indirect_call(target: ProbeFn) -> u32 {
183    let ret: u64;
184    // SAFETY: the caller guarantees `target` is C-ABI code with no arguments;
185    // clobber_abi("C") covers every register such a callee may clobber. The
186    // asm block may use the stack (no `nostack`), so rustc keeps RSP aligned
187    // for the CALL's return-address push.
188    unsafe {
189        core::arch::asm!(
190            "call {target}",
191            target = in(reg) target,
192            lateout("rax") ret,
193            clobber_abi("C"),
194        );
195    }
196    ret as u32
197}

199/// What the cell can tell, before probing, about the violation's outcome.
200#[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
201enum Expect {
202    /// The violation must fault; returning is a kernel bug (`FAIL`, exit 1).
203    /// The text says why enforcement is expected. Only x86_64 can know this
204    /// from inside the cell (CPUID is readable at CPL3).
205    #[cfg(target_arch = "x86_64")]
206    Fault(&'static str),
207    /// The violation cannot fault, or the cell cannot know whether it must:
208    /// returning prints these lines (`SKIP`, exit 0).
209    Skip(&'static [&'static str]),
210}
211
212/// Runs the control call, then the violating call; returns only by exiting.
213///
214/// `name` labels the mechanism; `expect` decides the verdict when the
215/// violation returns instead of faulting.
216#[cfg(any(target_arch = "aarch64", target_arch = "x86_64"))]
217fn probe(name: &str, expect: Expect) -> ! {
218    let padded: ProbeFn = cfi_test_padded_target;
219    let unpadded: ProbeFn = cfi_test_unpadded_target;
220    print_fmt(format_args!(
221        "cfi-test: {name} probe: landing pad {padded:p}, non-landing-pad target {unpadded:p}\n"
222    ));
223    match &expect {
224        #[cfg(target_arch = "x86_64")]
225        Expect::Fault(why) => {
226            print_fmt(format_args!(
227                "cfi-test: expecting the violation to fault: {why}\n"
228            ));
229        }
230        Expect::Skip(_) => println(
231            "cfi-test: the cell cannot assert enforcement here; a returned violation reports SKIP with the reason",
232        ),
233    }
234
235    // SAFETY: `padded` is a probe label (see `indirect_call`).
236    if unsafe { indirect_call(padded) } != CFI_PROBE_MAGIC {
237        println("cfi-test: FAIL: control indirect call to the landing pad returned a wrong value");
238        sys_exit(1);
239    }
240    println("cfi-test: control OK: indirect call to the landing pad ran from executable .text");
241
242    println("cfi-test: triggering indirect call to the non-landing-pad instruction");
243    // SAFETY: deliberate CFI violation — `unpadded` is a probe label (see
244    // `indirect_call`) that starts without a landing pad. Enforced CFI faults
245    // here and the kernel terminates the cell; otherwise it returns the magic.
246    if unsafe { indirect_call(unpadded) } != CFI_PROBE_MAGIC {
247        println("cfi-test: FAIL: non-landing-pad call returned a wrong value");
248        sys_exit(1);
249    }
250    match expect {
251        #[cfg(target_arch = "x86_64")]
252        Expect::Fault(why) => {
253            print_fmt(format_args!(
254                "cfi-test: FAIL: {name} not enforced: the non-landing-pad call returned although {why}\n"
255            ));
256            sys_exit(1)
257        }
258        Expect::Skip(lines) => {
259            for line in lines {
260                println(line);
261            }
262            sys_exit(0)
263        }
264    }
265}

267/// AArch64: the cell cannot read `ID_AA64PFR1_EL1` at EL0, so even an opted-in
268/// build can only report a returned violation; the harness decides FAIL from
269/// the boot log (see the module docs).
270#[cfg(target_arch = "aarch64")]
271fn expectation() -> Expect {
272    if cfg!(cfi_landing_pads) {
273        Expect::Skip(&[
274            "cfi-test: SKIP: BTI not enforced for EL0: this image opts in (GNU property BTI), yet BLR to a non-BTI instruction in executable .text returned",
275            "cfi-test: reason: the CPU lacks FEAT_BTI or the kernel did not map the image guarded; EL0 cannot read ID_AA64PFR1_EL1 to tell which, so this is a FAIL when the boot log shows 'CFI: BTI enabled' (scripts/qemu-cfi-test.sh)",
276        ])
277    } else {
278        Expect::Skip(&[
279            "cfi-test: SKIP: image not opted in: built without -Z branch-protection=bti, so it carries no GNU_PROPERTY_AARCH64_FEATURE_1_AND BTI note",
280            "cfi-test: reason: the kernel maps no page of a cell without the note guarded (stage-1 GP); build with scripts/qemu-cfi-test.sh to test enforcement",
281        ])
282    }
283}

285/// x86_64: CPUID is readable at CPL3, so the verdict is exact.
286#[cfg(target_arch = "x86_64")]
287fn expectation() -> Expect {
288    // CPUID leaf 7 subleaf 0, EDX bit 20 = CET-IBT (Intel SDM Vol. 2A, CPUID).
289    let ibt = core::arch::x86_64::__cpuid(0).eax >= 7
290        && (core::arch::x86_64::__cpuid_count(7, 0).edx >> 20) & 1 != 0;
291    if !ibt {
292        Expect::Skip(&[
293            "cfi-test: SKIP: CET-IBT unsupported: the CPU does not enumerate IBT (CPUID.(EAX=7,ECX=0):EDX[20] = 0), so CPL3 indirect branches cannot be tracked",
294        ])
295    } else if !cfg!(cfi_landing_pads) {
296        Expect::Skip(&[
297            "cfi-test: SKIP: image not opted in: built without -Z cf-protection=branch, so it carries no GNU_PROPERTY_X86_FEATURE_1_AND IBT note",
298            "cfi-test: reason: the kernel leaves IA32_U_CET = 0 for a cell without the note; build with scripts/qemu-cfi-test.sh to test enforcement",
299        ])
300    } else {
301        Expect::Fault(
302            "the CPU enumerates CET-IBT and this image carries the GNU_PROPERTY_X86_FEATURE_1_AND IBT note, so the kernel must set IA32_U_CET.ENDBR_EN",
303        )
304    }
305}
306
307fn run_test() -> ! {
308    #[cfg(target_arch = "aarch64")]
309    probe("aarch64 BTI", expectation());
310
311    #[cfg(target_arch = "x86_64")]
312    probe("x86_64 CET-IBT", expectation());
313
314    #[cfg(target_arch = "riscv64")]
315    {
316        println("cfi-test: SKIP: riscv64 forward-edge CFI (Zicfilp landing pads) is not enabled by the kernel; nothing to probe");
317        sys_exit(0);
318    }
