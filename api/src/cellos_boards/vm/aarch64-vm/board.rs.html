18/// QEMU virt RAM starts at 0x4000_0000; the kernel loads at +0x8_0000.
19pub const FALLBACK_KERNEL: MemoryRange = MemoryRange {
20    name: "kernel",
21    base: 0x4000_0000,
22    size: 0x0200_0000,
23    kind: MemoryRangeKind::Kernel,
24};
25/// Usable RAM up to 256 MiB when the firmware tree gives none; a tree memory
26/// node that covers the RAM base replaces this ceiling.
27pub const FALLBACK_USABLE: MemoryRange = MemoryRange {
28    name: "usable",
29    base: 0x4200_0000,
30    size: 0x0E00_0000,
31    kind: MemoryRangeKind::Usable,
32};
33const MEMORY: [MemoryRange; 2] = [FALLBACK_KERNEL, FALLBACK_USABLE];

35/// Generic AArch64 **virtual machine** descriptor (Phase 03, ADR-0032 §5).
36///
37/// Bound by the kernel's `board-aarch64-vm` feature (HAL backend
38/// `aarch64_vm`): the QEMU `virt` memory map with a GICv3, under QEMU TCG,
39/// Apple HVF or KVM. GICv3 distributor/redistributors and RAM come from the
40/// firmware tree when one is passed (`-kernel Image`), otherwise from QEMU
41/// virt's fixed map. PSCI secondaries, PL011 console.
42/// `dma_isolation` is `Optional`: the hypervisor owns host-virtual translation.
43pub const AARCH64_VM: BoardDescriptor = BoardDescriptor {
44    slug: "aarch64-vm",
45    vendor: "generic",
46    model: "aarch64-vm",
47    architecture: Architecture::Aarch64,
48    soc: SocId::GenericAArch64Vm,
49    compatibles: &COMPATIBLES,
50    boot: BootContract {
51        firmware: FirmwareInterface::Uefi,
52        boot_protocol: BootProtocol::DeviceTreeWithFallbackMap,
53        // A direct ELF boot passes no tree; the QEMU virt map then applies.
54        requires_firmware_dtb: false,
55        fallback_dts_path: "boards/vm/aarch64-vm/aarch64-vm.dts",
56        kernel_load_base: 0x4008_0000,
57    },
58    fallback_memory: &MEMORY,
59    wiring: WiringLayout {
60        pinmux_groups: &[],
61        phy_links: &["pl011-uart", "pcie-root"],
62    },
63    enabled_drivers: &DRIVERS,
64    dma_isolation: DmaIsolation::Optional,
65};
