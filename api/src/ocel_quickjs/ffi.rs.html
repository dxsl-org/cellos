1//! C ABI of the vendored engine, plus the shim symbols it needs that the
2//! Tier-A C ABI (`api::services::posix`) does not export.

89/// `lrint` — round to nearest, ties to even — which the Tier-A math set does
90/// not export and the engine's `Math` table takes the address of.
91///
92/// The engine only ever calls it on finite values in the integer range
93/// (`js_math_round`-adjacent paths), so the saturating cast below is the right
94/// contract: out-of-range or NaN input cannot be represented and saturates,
95/// exactly like the hardware conversion instruction.
96#[no_mangle]
97pub extern "C" fn lrint(x: f64) -> i64 {
98    if !x.is_finite() {
99        return 0;
100    }
101    let rounded = libm::rint(x);
102    if rounded <= i64::MIN as f64 {
103        i64::MIN
104    } else if rounded >= i64::MAX as f64 {
105        i64::MAX
106    } else {
107        rounded as i64
108    }
109}

111// ── Shim symbols the engine expects from the OS ─────────────────────────────
112//
113// `Date` is the only engine feature that needs wall-clock time and a
114// calendar. The cell has no timezone database, so `localtime_r` reports UTC —
115// documented in docs/guides/ocel-viewer.md as a limitation, not a silent
116// approximation of some other zone.
117
118const USEC_PER_SEC: u64 = 1_000_000;
119
120/// The only timezone this cell can name (see `localtime_r`).
121static UTC_ZONE: &[u8] = b"UTC\0";
122const SECS_PER_DAY: i64 = 86_400;
123
124fn cell_time_usec() -> u64 {
125    // `GetTime` op 0 is the architected counter at 10 MHz on RV64; op 1 is the
126    // scheduler tick. `sys_get_time` returns the monotonic counter, which is
127    // what `performance.now()`/`Date.now()` want at cell resolution.
128    ostd::syscall::sys_get_time()
129}

131/// Convert a Unix timestamp to the civil date/time, UTC.
132///
133/// Howard Hinnant's `civil_from_days` algorithm, which is exact for the whole
134/// `i64` range of the input and needs no lookup tables.
135fn civil_from_days(z: i64) -> (i64, u32, u32) {
136    let z = z + 719_468;
137    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
138    let doe = (z - era * 146_097) as u64;
139    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
140    let y = yoe as i64 + era * 400;
141    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
142    let mp = (5 * doy + 2) / 153;
143    let d = (doy - (153 * mp + 2) / 5 + 1) as u32;
144    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32;
145    (if m <= 2 { y + 1 } else { y }, m, d)
146}

148/// `gettimeofday` over the cell clock (ignores the (unused) timezone argument).
149///
150/// # Safety
151/// `tv` must be null or point to a writable `struct timeval`.
152#[no_mangle]
153pub unsafe extern "C" fn gettimeofday(tv: *mut CellosTimeval, _tz: *mut c_void) -> c_int {
154    if tv.is_null() {
155        return -1;
156    }
157    let now = cell_time_usec();
158    (*tv).tv_sec = (now / USEC_PER_SEC) as i64;
159    (*tv).tv_usec = (now % USEC_PER_SEC) as i64;
160    0
161}

163/// `clock_gettime` over the cell clock; both clocks this cell can be asked for
164/// are the same monotonic counter.
165///
166/// # Safety
167/// `tp` must be null or point to a writable `struct timespec`.
168#[no_mangle]
169pub unsafe extern "C" fn clock_gettime(_clk_id: c_int, tp: *mut CellosTimespec) -> c_int {
170    if tp.is_null() {
171        return -1;
172    }
173    let now = cell_time_usec();
174    (*tp).tv_sec = (now / USEC_PER_SEC) as i64;
175    (*tp).tv_nsec = ((now % USEC_PER_SEC) * 1000) as i64;
176    0
177}

179/// `localtime_r`, UTC only (see the module comment).
180///
181/// # Safety
182/// `timep` and `result` must be non-null and point to writable objects.
183#[no_mangle]
184pub unsafe extern "C" fn localtime_r(timep: *const i64, result: *mut CellosTm) -> *mut CellosTm {
185    if timep.is_null() || result.is_null() {
186        return core::ptr::null_mut();
187    }
188    let secs = *timep;
189    let mut days = secs.div_euclid(SECS_PER_DAY);
190    let mut rem = secs.rem_euclid(SECS_PER_DAY);
191    // 1970-01-01 was a Thursday (wday = 4, Sunday = 0).
192    let mut wday = (days + 4).rem_euclid(7) as i32;
193    if wday < 0 {
194        wday += 7;
195    }
196    let (year, month, day) = civil_from_days(days);
197    let yday = {
198        let jan1 = days - (day_of_year_utc(year, month, day) as i64);
199        days -= jan1;
200        days as i32
201    };
202    let t = &mut *result;
203    t.tm_hour = (rem / 3600) as i32;
204    rem %= 3600;
205    t.tm_min = (rem / 60) as i32;
206    t.tm_sec = (rem % 60) as i32;
207    t.tm_mday = day as i32;
208    t.tm_mon = month as i32 - 1;
209    t.tm_year = year as i32 - 1900;
210    t.tm_wday = wday;
211    t.tm_yday = yday;
212    t.tm_isdst = 0;
213    t.tm_gmtoff = 0;
214    t.tm_zone = UTC_ZONE.as_ptr() as *const c_char;
215    let _ = &mut days;
216    result
217}

219fn day_of_year_utc(year: i64, month: u32, day: u32) -> u32 {
220    const CUM: [u32; 12] = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334];
221    let leap = (year % 4 == 0 && year % 100 != 0) || year % 400 == 0;
222    let extra = u32::from(leap && month > 2);
223    CUM[(month - 1) as usize] + day - 1 + extra
224}
