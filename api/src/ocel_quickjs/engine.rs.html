1//! The QuickJS-backed script engine.
2//!
3//! The engine object is deliberately thin: QuickJS owns JavaScript state, and
4//! the DOM surface Ocel talks to (`document.title`, `node_<id>.textContent`,
5//! `node_<id>.setAttribute`, `node_<id>.addEventListener`, `console.log`,
6//! event dispatch) is defined by a **JavaScript prelude** evaluated once per
7//! context. Rust never builds JS objects through the C API: it evaluates the
8//! prelude, evaluates the document's script, and then reads the mutation queue
9//! the prelude recorded.
10//!
11//! That keeps the C surface to the handful of entry points in [`super::ffi`]
12//! and makes the DOM dialect a reviewed, readable JS file instead of a wall of
13//! `JS_NewCFunction` calls.
14//!
15//! Mutation records use tab-separated percent-encoded UTF-8 fields; creation,
16//! tree edits and character data therefore survive arbitrary string contents.
17

27/// Recursion budget handed to the engine, in bytes of cell stack.
28///
29/// Cells get a 256 KiB stack (`kernel/src/task.rs`, `STACK_PAGES`). The engine
30/// checks this limit before recursing into the interpreter or the parser, so a
31/// pathological script fails as a JS `RangeError` instead of faulting the cell.
32const MAX_STACK_BYTES: usize = 192 * 1024;
33
34// QuickJS polls during interpretation/parsing; additionally cap promise jobs.
35const MAX_INTERRUPT_POLLS: usize = 100_000;
36const MAX_PENDING_JOBS: usize = 10_000;
37
38unsafe extern "C" fn interrupt_budget(_: *mut ffi::JSRuntime, opaque: *mut core::ffi::c_void) -> core::ffi::c_int {
39    let remaining = &mut *opaque.cast::<usize>();
40    if *remaining == 0 { 1 } else { *remaining -= 1; 0 }
41}
42
43struct InterruptGuard<'a> {
44    rt: *mut ffi::JSRuntime,
45    _budget: core::marker::PhantomData<&'a mut usize>,
46}
47impl<'a> InterruptGuard<'a> {
48    unsafe fn new(rt: *mut ffi::JSRuntime, budget: &'a mut usize) -> Self {
49        ffi::JS_SetInterruptHandler(rt, Some(interrupt_budget), (budget as *mut usize).cast());
50        Self { rt, _budget: core::marker::PhantomData }
51    }
52}
53impl Drop for InterruptGuard<'_> {
54    fn drop(&mut self) {
55        unsafe { ffi::JS_SetInterruptHandler(self.rt, None, core::ptr::null_mut()); }
56    }
57}

61const MAX_HEAP_BYTES: usize = 20 * 1024 * 1024;

101impl Drop for QuickJsContext {
102    fn drop(&mut self) {
103        unsafe {
104            if !self.ctx.is_null() {
105                ffi::JS_FreeContext(self.ctx);

108                ffi::JS_FreeRuntime(self.rt);

110        }

114impl QuickJsContext {
115    fn new() -> Self {
116        let mut out = Self {
117            rt: core::ptr::null_mut(),
118            ctx: core::ptr::null_mut(),
119            mutations: Vec::new(),
120            browser_requests: Vec::new(),
121            browser_wakeup: None,
122            native_layout: alloc::boxed::Box::default(),
123            logs: Vec::new(),
124            output_limited: false,
125            usable: true,
126        };
127        unsafe {
128            let rt = ffi::JS_NewRuntime();
129            if rt.is_null() {
130                out.logs.push(String::from(
131                    "[ocel-quickjs] JS_NewRuntime failed (cell heap exhausted?)",
132                ));
133                return out;
134            }
135            out.rt = rt;
136            ffi::JS_SetMaxStackSize(rt, MAX_STACK_BYTES);
137            ffi::JS_SetMemoryLimit(rt, MAX_HEAP_BYTES);
138            ffi::JS_SetGCThreshold(rt, 256 * 1024);
139            let ctx = ffi::JS_NewContext(rt);
140            if ctx.is_null() {
141                out.logs.push(String::from(
142                    "[ocel-quickjs] JS_NewContext failed; engine disabled",
143                ));
144                return out;
145            }
146            out.ctx = ctx;
147        }
148        super::native_layout::install(out.ctx, &mut out.native_layout);
149        if let Err(e) = out.eval_raw(PRELUDE, "<ocel-prelude>") {
150            // A prelude failure is fatal for the DOM dialect: report it through
151            // the log channel so the cell's start-up line can say so.
152            out.logs.push(alloc::format!(
153                "[ocel-quickjs] DOM prelude failed: {}",
154                e.message
155            ));
156            unsafe { ffi::JS_FreeContext(out.ctx); }
157            out.ctx = core::ptr::null_mut();
158        }
159        out
160    }
161
162    fn ready(&self) -> bool {
163        !self.ctx.is_null()
164    }

166    /// Evaluate `src`, collecting anything the prelude recorded.
167    ///
168    /// Returns `Ok(result_repr)` for a normal completion and `Err(JsError)`
169    /// for a JS exception (the message comes from the engine's own exception
170    /// object, so a script bug reports as the script author's error).
171    fn eval_raw(&mut self, src: &str, filename: &str) -> Result<String, JsError> {
172        if !self.ready() {
173            return Err(JsError {
174                message: String::from("engine unavailable"),
175                line: 0,
176            });
177        }
178        let code = alloc::ffi::CString::new(src).map_err(|_| JsError {
179            message: String::from("script contains an interior NUL byte"),
180            line: 0,
181        })?;
182        let name = alloc::ffi::CString::new(filename)
183            .unwrap_or_else(|_| alloc::ffi::CString::new("<script>").expect("literal has no NUL"));
184        unsafe {
185            let mut budget = MAX_INTERRUPT_POLLS;
186            let guard = InterruptGuard::new(self.rt, &mut budget);
187            let value = ffi::JS_Eval(
188                self.ctx,
189                code.as_ptr(),
190                src.len(),
191                name.as_ptr(),
192                ffi::JS_EVAL_TYPE_GLOBAL,
193            );
194            let exception = if ffi::vios_js_is_exception(value) != 0 {
195                Some(self.take_exception_message())
196            } else { None };
197            let repr = if exception.is_none() { self.value_to_string(value) } else { String::new() };
198            let result_limited = core::mem::take(&mut self.output_limited);
199            ffi::vios_js_free(self.ctx, value);
200            let mut job_error = None;
201            for index in 0..=MAX_PENDING_JOBS {
202                let mut job_ctx = core::ptr::null_mut();
203                let status = ffi::JS_ExecutePendingJob(self.rt, &mut job_ctx);
204                if status == 0 { break; }
205                if status < 0 {
206                    job_error = Some(self.take_exception_message());
207                    break;
208                }
209                if index == MAX_PENDING_JOBS {
210                    job_error = Some(String::from("JavaScript pending-job limit exceeded"));
211                    break;
212                }
213            }
214            drop(guard);
215            self.collect_mutations();
216            match exception.or(job_error) {
217                _ if result_limited => Err(JsError { message: String::from("JavaScript result byte limit exceeded"), line: 0 }),
218                Some(message) => Err(JsError { message, line: 0 }),
219                None => Ok(repr),
220            }
221        }
222    }
223
224    /// Read the pending exception's `message`/`stack` and clear it.
225    unsafe fn take_exception_message(&mut self) -> String {
226        let exc = ffi::JS_GetException(self.ctx);
227        let mut message = self.value_to_string(exc);
228        ffi::vios_js_free(self.ctx, exc);
229        if message.is_empty() {
230            message = String::from("uncaught JavaScript exception");
231        }
232        message
233    }

235    /// `String(value)`-style rendering, using the engine's own conversion so
236    /// numbers keep the engine's formatting.
237    unsafe fn value_to_string(&mut self, value: ffi::JSValue) -> String {
238        self.value_to_string_bounded(value, dom_arena::OCEL_JS_MAX_RESULT_BYTES)
239    }

241    unsafe fn value_to_string_bounded(&mut self, value: ffi::JSValue, limit: usize) -> String {
242        if ffi::vios_js_is_undefined(value) != 0 {
243            return String::from("undefined");
244        }
245        if ffi::vios_js_is_null(value) != 0 {
246            return String::from("null");
247        }
248        if ffi::vios_js_is_number(value) != 0 {
249            if ffi::vios_js_is_float(value) != 0 {
250                return format_float(ffi::vios_js_as_float(value));
251            }
252            let mut d = 0.0f64;
253            if ffi::JS_ToFloat64(self.ctx, &mut d, value) == 0 {
254                return format_float(d);
255            }
256        }
257        if ffi::vios_js_is_bool(value) != 0 {
258            return if ffi::JS_ToBool(self.ctx, value) != 0 {
259                String::from("true")
260            } else {
261                String::from("false")
262            };
263        }
264        let mut len = 0usize;
265        let ptr = ffi::vios_js_to_bounded_string(self.ctx, value, limit, &mut len);
266        if ptr.is_null() {
267            if len > limit { self.output_limited = true; }
268            return String::from("JavaScript output string limit exceeded or conversion failed");
269        }
270        // `.cast()` rather than `as *const u8`: `c_char` is `u8` on
271        // riscv64/aarch64 and `i8` on x86_64.
272        let out = core::str::from_utf8(core::slice::from_raw_parts(ptr.cast::<u8>(), len))
273            .unwrap_or("[invalid utf-8]")
274            .to_string();
275        ffi::JS_FreeCString(self.ctx, ptr);
276        out
277    }

279    /// Drain the prelude's queues into Rust-side mutations and logs.
280    fn collect_mutations(&mut self) {
281        let drained = self.eval_queue("__vios_takeMutations()");
282        match super::native_layout::parse_mutations(&drained) {
283            Ok(parsed) => self.mutations.extend(parsed),
284            Err(_) => self.usable = false,
285        }
286        self.mutations.append(&mut self.native_layout.unapplied);
287        let bytes = dom_arena::ipc::serialized_len_bounded(&self.mutations, dom_arena::OCEL_JS_MAX_MUTATION_BYTES);
288        if self.mutations.len().saturating_add(self.native_layout.mutations) > dom_arena::OCEL_JS_MAX_MUTATIONS
289            || bytes.is_err() || bytes.unwrap_or(usize::MAX).saturating_add(self.native_layout.mutation_bytes) > dom_arena::OCEL_JS_MAX_MUTATION_BYTES
290            || self.native_layout.failed {
291            self.usable = false;
292        }
293
294        let logged = self.eval_queue("__vios_takeLogs()");
295        for line in logged.lines() {
296            if !line.is_empty() {
297                if self.logs.len() >= dom_arena::OCEL_JS_MAX_LOG_LINES ||
298                    self.logs.iter().map(String::len).sum::<usize>().saturating_add(line.len()) > dom_arena::OCEL_JS_MAX_LOG_BYTES {
299                    self.usable = false;
300                    break;
301                }
302                self.logs.push(line.to_string());
303            }
304        }
305        self.collect_browser_output();
306    }
307
308    fn collect_browser_output(&mut self) {
309        let drained = self.eval_queue("__vios_takeRequests()");
310        for row in drained.lines() {
311            if self.browser_requests.len() >= dom_arena::OCEL_JS_MAX_BROWSER_REQUESTS { self.usable = false; break; }
312            let Some(fields) = row.split('\t').map(decode_field).collect::<Option<Vec<_>>>() else { self.usable = false; break; };
313            let parsed = (|| -> Option<BrowserRequest> {
314                let id = || fields.get(1)?.parse::<u32>().ok();
315                match fields.first()?.as_str() {
316                    "L" if fields.len() == 5 => Some(BrowserRequest::Load {
317                        id: id()?, node: Some(NodeId(fields[2].parse().ok()?)), url: fields[3].clone(),
318                        kind: match fields[4].as_str() {
319                            "Script" => BrowserResourceKind::Script,
320                            "Stylesheet" => BrowserResourceKind::Stylesheet,
321                            "Image" => BrowserResourceKind::Image, _ => return None,
322                        },
323                    }),
324                    "F" if fields.len() >= 6 && (fields.len() - 6) % 2 == 0 => Some(BrowserRequest::Fetch {
325                        id: id()?, url: fields[2].clone(), method: fields[3].clone(), body: fields[4].clone(),
326                        credentials: fields[5].clone(),
327                        headers: fields[6..].chunks_exact(2).map(|p| (p[0].clone(), p[1].clone())).collect(),
328                    }),
329                    "A" if fields.len() == 2 => Some(BrowserRequest::Abort { id: id()? }),
330                    "K" if fields.len() == 2 => Some(BrowserRequest::SetCookie { value: fields[1].clone() }),
331                    _ => None,
332                }
333            })();
334            let Some(request) = parsed else { self.usable = false; break; };
335            self.browser_requests.push(request);
336        }
337        if dom_arena::ipc::serialized_len_bounded(&self.browser_requests, dom_arena::OCEL_JS_MAX_BROWSER_REQUEST_BYTES).is_err() {
338            self.usable = false;
339        }
340        let wake = self.eval_queue("__vios_nextWakeup()");
341        self.browser_wakeup = if wake.is_empty() { None } else {
342            match wake.parse::<u64>() { Ok(value) if value <= 9_007_199_254_740_991 => Some(value), _ => { self.usable = false; None } }
343        };
344    }

346    /// Evaluate a prelude helper that returns a string, without recursing into
347    /// [`Self::collect_mutations`].
348    fn eval_queue(&mut self, expr: &str) -> String {
349        if !self.ready() {
350            return String::new();
351        }
352        let Ok(code) = alloc::ffi::CString::new(expr) else {
353            return String::new();
354        };
355        let name = c"<ocel-queue>".as_ptr();
356        unsafe {
357            let mut budget = MAX_INTERRUPT_POLLS;
358            let _guard = InterruptGuard::new(self.rt, &mut budget);
359            let value = ffi::JS_Eval(
360                self.ctx,
361                code.as_ptr(),
362                expr.len(),
363                name,
364                ffi::JS_EVAL_TYPE_GLOBAL,
365            );
366            if ffi::vios_js_is_exception(value) != 0 {
367                let _ = self.take_exception_message();
368                self.usable = false;
369                ffi::vios_js_free(self.ctx, value);
370                return String::new();
371            }
372            // The prelude already bounds joined UTF-16 queues; allow three UTF-8
373            // bytes per unit at the C boundary, then independently validate wire.
374            let limit = match expr {
375                "__vios_takeLogs()" => dom_arena::OCEL_JS_MAX_LOG_BYTES,
376                "__vios_takeRequests()" => dom_arena::OCEL_JS_MAX_BROWSER_REQUEST_BYTES,
377                "__vios_nextWakeup()" => 32,
378                _ => dom_arena::OCEL_JS_MAX_MUTATION_BYTES,
379            };
380            self.output_limited = false;
381            let out = self.value_to_string_bounded(value, limit * 3);
382            if core::mem::take(&mut self.output_limited) { self.usable = false; }
383            ffi::vios_js_free(self.ctx, value);
384            out
385        }
386    }

389impl JsContext for QuickJsContext {
390    fn is_usable(&self) -> bool { self.usable }
391    fn sync_document(&mut self, arena: &DocumentArena, title: &str) -> Result<(), JsError> {
392        if !arena.validate() || !self.mutations.is_empty() {
393            return Err(JsError { message: String::from("Invalid snapshot or unapplied DOM mutations"), line: 0 });
394        }
395        let mut source = String::from("__vios_sync([");
396        for (index, node) in arena.nodes.iter().enumerate() {
397            if index != 0 { source.push(','); }
398            let (kind, tag, text) = match &node.data {
399                NodeData::DocumentRoot => (9, "", ""),
400                NodeData::DocumentFragment => (11, "", ""),
401                NodeData::Element { tag, .. } => (1, tag.as_str(), ""),
402                NodeData::Text(text) => (3, "", text.as_str()),
403                NodeData::Comment(text) => (8, "", text.as_str()),
404            };
405            source.push_str(&alloc::format!("[{index},{kind},"));
406            push_js_string(&mut source, tag); source.push(',');
407            push_js_string(&mut source, text); source.push_str(",[");
408            if let NodeData::Element { attributes, .. } = &node.data {
409                for (i, (key, value)) in attributes.iter().enumerate() {
410                    if i != 0 { source.push(','); }
411                    source.push('['); push_js_string(&mut source, key); source.push(',');
412                    push_js_string(&mut source, value); source.push(']');
413                }
414            }
415            source.push(']');
416            for link in [node.parent, node.first_child, node.next_sibling] {
417                source.push(',');
418                if let Some(id) = link { source.push_str(&alloc::format!("{}", id.0)); }
419                else { source.push_str("null"); }
420            }
421            source.push(']');
422        }
423        source.push_str("],"); push_js_string(&mut source, title); source.push(')');
424        let res = self.eval_raw(&source, "<ocel-document-sync>");
425        unsafe { ffi::JS_RunGC(self.rt); }
426        res.map(|_| ())
427    }
428    fn eval_script(&mut self, script: &str, node: Option<NodeId>) -> Result<String, JsError> {
429        let mut source = String::from("__vios_evalScript(");
430        push_js_string(&mut source, script);
431        source.push(',');
432        if let Some(node) = node { source.push_str(&alloc::format!("{}", node.0)); } else { source.push_str("null"); }
433        source.push(')');
434        let res = self.eval_raw(&source, "<document-script>");
435        unsafe { ffi::JS_RunGC(self.rt); }
436        res
437    }
438
439    fn configure_browser(&mut self, environment: &BrowserEnvironment) -> Result<(), JsError> {
440        if environment.now_ms > 9_007_199_254_740_991 {
441            return Err(JsError { message: String::from("Browser clock exceeds exact JavaScript range"), line: 0 });
442        }
443        let mut source = alloc::format!("__vios_configure({{width:{},height:{},now_ms:{},url:",
444            environment.width, environment.height, environment.now_ms);
445        push_js_string(&mut source, &environment.url); source.push_str(",user_agent:");
446        push_js_string(&mut source, &environment.user_agent); source.push_str(",cookie:");
447        push_js_string(&mut source, &environment.cookie); source.push_str("})");
448        self.eval_raw(&source, "<browser-configure>").map(|_| ())
449    }
450
451    fn pump_browser(&mut self, frame: &BrowserFrame) -> Result<(), JsError> {
452        if frame.now_ms > 9_007_199_254_740_991 {
453            return Err(JsError { message: String::from("Browser clock exceeds exact JavaScript range"), line: 0 });
454        }
455        let mut seen = alloc::collections::BTreeSet::new();
456        for item in &frame.layout {
457            if !seen.insert(item.node) || ![item.x, item.y, item.width, item.height].iter().all(|n| n.is_finite())
458                || item.width < 0.0 || item.height < 0.0 {
459                return Err(JsError { message: String::from("Invalid browser layout snapshot"), line: 0 });
460            }
461        }
462        let source = alloc::format!("__vios_pump({})", frame_expression(frame));
463        self.eval_raw(&source, "<browser-pump>").map(|_| ())
464    }
465
466    fn complete_browser(&mut self, reply: &BrowserReply) -> Result<(), JsError> {
467        let mut source = alloc::format!("__vios_complete({{id:{},status:{},url:", reply.id, reply.status);
468        push_js_string(&mut source, &reply.url); source.push_str(",headers:");
469        push_js_pairs(&mut source, &reply.headers); source.push_str(",body:");
470        push_js_string(&mut source, &reply.body); source.push_str(",error:");
471        if let Some(error) = &reply.error { push_js_string(&mut source, error); } else { source.push_str("null"); }
472        source.push_str("})");
473        self.eval_raw(&source, "<browser-complete>").map(|_| ())
474    }
475
476    fn take_browser_requests(&mut self) -> Vec<BrowserRequest> { core::mem::take(&mut self.browser_requests) }
477    fn next_browser_wakeup(&self) -> Option<u64> { self.browser_wakeup }
478    fn eval(&mut self, script: &str) -> Result<String, JsError> {
479        self.eval_raw(script, "<script>")
480    }
481
482    fn dispatch_event(&mut self, event: &DomEvent) -> Result<(), JsError> {
483        let kind = match event.kind {
484            dom_arena::EventKind::Click => "click",
485            dom_arena::EventKind::KeyDown => "keydown",
486            dom_arena::EventKind::Input => "input",
487            dom_arena::EventKind::Submit => "submit",
488            dom_arena::EventKind::Change => "change",
489        };
490        let mut call = alloc::format!("__vios_dispatch({}, \"{}\", {}, {}, ",
491            event.target.0, kind, event.client_x, event.client_y);
492        if let Some(key) = event.key { push_js_string(&mut call, &key.to_string()); }
493        else { call.push_str("null"); }
494        call.push(')');
495        self.eval_raw(&call, "<dispatch>").map(|_| ())
496    }
497
498    fn take_mutations(&mut self) -> Vec<DomMutation> {
499        core::mem::take(&mut self.mutations)
500    }
501
502    fn reset(&mut self) {
503        // Recreating the context is the only reset that cannot leak a global:
504        // `eval('...')`-based resets always leave the prelude's own state behind.
505        *self = Self::new();
506    }
507}

528/// Minimal `Number`-to-text for the value repr: integers print without a
529/// trailing `.0`, everything else uses Rust's shortest round-trip. The engine's
530/// own `js_dtoa` does the same job in C for `String(n)`; this only affects the
531/// protocol's informational `result_repr`.
532fn format_float(d: f64) -> String {
533    let integral = d.is_finite() && libm::trunc(d) == d;
534    if integral && libm::fabs(d) < 9.007_199_254_740_992e15 {
535        alloc::format!("{}", d as i64)
536    } else if d.is_nan() {
537        String::from("NaN")
538    } else if d.is_infinite() {
539        String::from(if d > 0.0 { "Infinity" } else { "-Infinity" })
540    } else {
541        alloc::format!("{d}")
542    }
543}

545fn push_js_string(out: &mut String, value: &str) {
546    out.push('"');
547    for ch in value.chars() {
548        match ch {
549            '"' => out.push_str("\\\""),
550            '\\' => out.push_str("\\\\"),
551            '\n' => out.push_str("\\n"),
552            '\r' => out.push_str("\\r"),
553            '\t' => out.push_str("\\t"),
554            c if c <= '\u{1f}' || c == '\u{2028}' || c == '\u{2029}' =>
555                out.push_str(&alloc::format!("\\u{:04x}", c as u32)),
556            c => out.push(c),
557        }
558    }
559    out.push('"');
560}
561
562fn push_js_pairs(out: &mut String, pairs: &[(String, String)]) {
563    out.push('[');
564    for (index, (key, value)) in pairs.iter().enumerate() {
565        if index != 0 { out.push(','); }
566        out.push('['); push_js_string(out, key); out.push(','); push_js_string(out, value); out.push(']');
567    }
568    out.push(']');
569}

571pub(super) fn frame_expression(frame: &BrowserFrame) -> String {
572    let mut source = alloc::format!("({{now_ms:{},width:{},height:{},scroll_y:{},parsing_complete:{},cookie:",
573        frame.now_ms, frame.width, frame.height, frame.scroll_y, frame.parsing_complete);
574    push_js_string(&mut source, &frame.cookie); source.push_str(",layout:[");
575    for (index, item) in frame.layout.iter().enumerate() {
576        if index != 0 { source.push(','); }
577        source.push_str(&alloc::format!("{{node:{},x:{},y:{},width:{},height:{},styles:",
578            item.node.0, item.x, item.y, item.width, item.height));
579        push_js_pairs(&mut source, &item.styles); source.push('}');
580    }
581    source.push_str("]})");
582    source
583}

588pub(super) fn decode_field(field: &str) -> Option<String> {
589    let bytes = field.as_bytes();
590    let mut out = Vec::with_capacity(bytes.len());
591    let mut index = 0;
592    while index < bytes.len() {
593        if bytes[index] == b'%' {
594            let hi = (*bytes.get(index + 1)? as char).to_digit(16)?;
595            let lo = (*bytes.get(index + 2)? as char).to_digit(16)?;
596            out.push((hi * 16 + lo) as u8);
597            index += 3;
598        } else { out.push(bytes[index]); index += 1; }
599    }
600    String::from_utf8(out).ok()
601}
