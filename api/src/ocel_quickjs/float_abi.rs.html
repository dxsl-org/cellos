1//! The engine's C99 math, behind an integer-only boundary.
2//!
3//! Rule H1 (`docs/guides/tier1b-c-zig.md`, "x86_64 C and the float boundary"):
4//! no `float`/`double` crosses between this cell's Rust and its C by value. On
5//! x86_64 the engine's C is hard-float SSE2 and every Rust Cell target is
6//! soft-float, so the two sides pass an `f64` in different registers — the
7//! engine would read its own argument back as the result. The C side
8//! (`vendor/include/math.h`) therefore hands each `double` over as its IEEE-754
9//! bit pattern and gets the result back the same way; the bit pattern is exact,
10//! so every architecture computes the same `libm` result.
11//!
12//! Contract with `math.h`: one export `cellos_qjs_<name>_bits` per function it
13//! declares, taking and returning `uint64_t` bit patterns, except `lrint`
14//! (`int64_t` result) and `modf` (fractional part only; `math.h` takes the
15//! integral part from `trunc`).
16
17macro_rules! bits1 {
18    ($($c:ident => $f:path),* $(,)?) => {$(
19        #[no_mangle]
20        pub extern "C" fn $c(x: u64) -> u64 {
21            $f(f64::from_bits(x)).to_bits()
22        }
23    )*};
24}
25
26macro_rules! bits2 {
27    ($($c:ident => $f:path),* $(,)?) => {$(
28        #[no_mangle]
29        pub extern "C" fn $c(x: u64, y: u64) -> u64 {
30            $f(f64::from_bits(x), f64::from_bits(y)).to_bits()
31        }
32    )*};
33}

73/// `modf`'s fractional part.
74#[no_mangle]
75pub extern "C" fn cellos_qjs_modf_bits(x: u64) -> u64 {
76    libm::modf(f64::from_bits(x)).0.to_bits()
77}

79/// `lrint`: round to nearest, ties to even.
80///
81/// The engine calls it on finite values in the integer range (the
82/// `Uint8ClampedArray` store clamps to 0..=255 first), so saturating is the
83/// right contract for the rest: NaN and infinities have no `long` value and
84/// give 0, and out-of-range values saturate like the hardware conversion.
85#[no_mangle]
86pub extern "C" fn cellos_qjs_lrint_bits(x: u64) -> i64 {
87    let x = f64::from_bits(x);
88    if !x.is_finite() {
89        return 0;
90    }
91    let rounded = libm::rint(x);
92    if rounded <= i64::MIN as f64 {
93        i64::MIN
94    } else if rounded >= i64::MAX as f64 {
95        i64::MAX
96    } else {
97        rounded as i64
98    }
