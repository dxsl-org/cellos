2//! Synchronous native CSSOM queries keep the QuickJS C stack suspended while
3//! the owner acknowledges/apply mutations and commits a real native layout.

13use core::sync::atomic::{AtomicUsize, Ordering};
14static REPLY_OPERATION: AtomicUsize = AtomicUsize::new(0);
15pub(super) fn set_reply_operation(operation: Option<usize>) {
16    REPLY_OPERATION.store(operation.unwrap_or(0), Ordering::Relaxed);
17}
18pub(super) fn reply_operation() -> Option<usize> {
19    match REPLY_OPERATION.load(Ordering::Relaxed) {
20        0 => None,
21        operation => Some(operation),
22    }
23}
24#[derive(Default)]
25pub(super) struct State {
26    pub caller: usize,
27    pub context: u64,
28    pub operation: Option<usize>,
29    pub queries: u32,
30    pub mutations: usize,
31    pub mutation_bytes: usize,
32    pub failed: bool,
33    pub unapplied: Vec<DomMutation>,
34    #[cfg(test)]
35    pub host: Option<fn(&BrowserLayoutQuery) -> Result<BrowserFrame, String>>,
36}

38pub(super) fn install(ctx: *mut ffi::JSContext, state: &mut Box<State>) {
39    unsafe {
40        ffi::vios_js_install_native_layout(
41            ctx,
42            (&mut **state as *mut State).cast(),
43            Some(callback),
44        );
45    }
46}

48unsafe extern "C" fn callback(
49    ctx: *mut ffi::JSContext,
50    opaque: *mut core::ffi::c_void,
51    node: u32,
52    wire: *const core::ffi::c_char,
53    length: usize,
54) -> ffi::JSValue {
55    let state = &mut *opaque.cast::<State>();
56    let result = (|| -> Result<BrowserFrame, String> {
57        if state.failed {
58            return Err(String::from("Native layout session has failed"));
59        }
60        state.queries += 1;
61        if state.queries > 64 {
62            return Err(String::from(
63                "Synchronous native layout query limit exceeded",
64            ));
65        }
66        let text = core::str::from_utf8(core::slice::from_raw_parts(wire.cast::<u8>(), length))
67            .map_err(|_| String::from("Invalid native layout mutation encoding"))?;
68        let mutations = parse_mutations(text)?;
69        let has_host = state.operation.is_some();
70        #[cfg(test)]
71        let has_host = has_host || state.host.is_some();
72        if !has_host {
73            state.unapplied.extend(mutations);
74            return Err(String::from("Native layout host is unavailable"));
75        }
76        state.mutations = state.mutations.saturating_add(mutations.len());
77        state.mutation_bytes = state.mutation_bytes.saturating_add(
78            dom_arena::ipc::serialized_len_bounded(
79                &mutations,
80                dom_arena::OCEL_JS_MAX_MUTATION_BYTES,
81            )
82            .map_err(|_| String::from("Native layout mutation byte budget exceeded"))?,
83        );
84        if state.mutations > dom_arena::OCEL_JS_MAX_MUTATIONS
85            || state.mutation_bytes > dom_arena::OCEL_JS_MAX_MUTATION_BYTES
86        {
87            return Err(String::from("Native layout mutation budget exceeded"));
88        }
89        let query = BrowserLayoutQuery {
90            node: NodeId(node),
91            mutations,
92        };
93        #[cfg(test)]
94        if let Some(host) = state.host {
95            return host(&query);
96        }
97        exchange(state, query)
98    })();
99    match result {
100        Ok(frame) => {
101            let source = engine::frame_expression(&frame);
102            let code = match alloc::ffi::CString::new(source) {
103                Ok(code) => code,
104                Err(_) => return throw(ctx, "Invalid native frame source"),
105            };
106            ffi::JS_Eval(
107                ctx,
108                code.as_ptr(),
109                code.as_bytes().len(),
110                c"<native-layout>".as_ptr(),
111                ffi::JS_EVAL_TYPE_GLOBAL,
112            )
113        }
114        Err(error) => throw(ctx, &error),
115    }
116}
117unsafe fn throw(ctx: *mut ffi::JSContext, message: &str) -> ffi::JSValue {
118    let text = alloc::ffi::CString::new(message.replace('\0', " ")).unwrap();
119    ffi::vios_js_throw_error(ctx, text.as_ptr())
120}

122pub(super) fn parse_mutations(text: &str) -> Result<Vec<DomMutation>, String> {
123    let mut out = Vec::new();
124    for row in text.lines() {
125        if out.len() >= dom_arena::OCEL_JS_MAX_MUTATIONS {
126            return Err(String::from("Mutation count budget exceeded"));
127        }
128        let fields = row
129            .split('\t')
130            .map(engine::decode_field)
131            .collect::<Option<Vec<_>>>()
132            .ok_or_else(|| String::from("Invalid mutation encoding"))?;
133        let field = |index| {
134            fields
135                .get(index)
136                .map(String::as_str)
137                .ok_or_else(|| String::from("Missing mutation field"))
138        };
139        let id = |index| {
140            field(index)?
141                .parse::<u32>()
142                .map(NodeId)
143                .map_err(|_| String::from("Invalid mutation node"))
144        };
145        let mutation = match field(0)? {
146            "C" if fields.len() == 5 => DomMutation::CreateNode {
147                node: id(1)?,
148                data: match field(2)? {
149                    "1" => NodeData::Element {
150                        tag: field(3)?.to_string(),
151                        attributes: Vec::new(),
152                    },
153                    "3" => NodeData::Text(field(4)?.to_string()),
154                    "8" => NodeData::Comment(field(4)?.to_string()),
155                    "11" => NodeData::DocumentFragment,
156                    _ => return Err(String::from("Invalid created node type")),
157                },
158            },
159            "X" if fields.len() == 3 => DomMutation::SetText {
160                node: id(1)?,
161                text: field(2)?.to_string(),
162            },
163            "A" if fields.len() == 4 => DomMutation::SetAttribute {
164                node: id(1)?,
165                key: field(2)?.to_string(),
166                val: field(3)?.to_string(),
167            },
168            "R" if fields.len() == 3 => DomMutation::RemoveAttribute {
169                node: id(1)?,
170                key: field(2)?.to_string(),
171            },
172            "P" if fields.len() == 3 => DomMutation::AppendChild {
173                parent: id(1)?,
174                child: id(2)?,
175            },
176            "I" if fields.len() == 4 => DomMutation::InsertBefore {
177                parent: id(1)?,
178                child: id(2)?,
179                reference: id(3)?,
180            },
181            "D" if fields.len() == 3 => DomMutation::RemoveChild {
182                parent: id(1)?,
183                child: id(2)?,
184            },
185            "T" if fields.len() == 2 => DomMutation::SetDocumentTitle {
186                title: field(1)?.to_string(),
187            },
188            _ => return Err(String::from("Invalid mutation operation")),
189        };
190        out.push(mutation);
191    }
192    Ok(out)
193}
194fn send(operation: usize, reply: &OcelJsReply) -> Result<(), String> {
195    let mut bytes = [0u8; OCEL_JS_IPC_BUF_SIZE];
196    let encoded = postcard::to_slice(reply, &mut bytes)
197        .map_err(|_| String::from("Native layout frame overflow"))?;
198    ostd::ipc::reply(operation, encoded)
199        .map_err(|_| String::from("Native layout operation was abandoned"))
200}
201fn chunk(state: &State, bytes: &[u8], offset: usize) -> Result<OcelJsReply, String> {
202    if offset >= bytes.len() {
203        return Err(String::from("Native layout offset out of bounds"));
204    }
205    let end = offset.saturating_add(OCEL_JS_CHUNK_SIZE).min(bytes.len());
206    Ok(OcelJsReply::LayoutQueryChunk {
207        context: state.context,
208        query: state.queries,
209        offset,
210        total: bytes.len(),
211        data: bytes[offset..end].to_vec(),
212    })
213}
214fn exchange(state: &mut State, query: BrowserLayoutQuery) -> Result<BrowserFrame, String> {
215    let operation = state
216        .operation
217        .take()
218        .ok_or_else(|| String::from("Native layout host is unavailable"))?;
219    let outgoing = dom_arena::ipc::encode_bounded(&query, OCEL_JS_MAX_TRANSFER)
220        .map_err(|_| String::from("Native layout query exceeds transfer budget"))?;
221    if let Err(error) = send(operation, &chunk(state, &outgoing, 0)?) {
222        state.failed = true;
223        return Err(error);
224    }
225    set_reply_operation(None);
226    let Some(started) = ostd::syscall::sys_get_scheduler_ticks() else {
227        state.failed = true;
228        return Err(String::from("Native layout clock unavailable"));
229    };
230    let mut incoming = Vec::new();
231    let mut expected = None;
232    let mut receive = [0u8; OCEL_JS_IPC_BUF_SIZE];
233    loop {
234        let now = ostd::syscall::sys_get_scheduler_ticks();
235        if now.is_none_or(|now| now.wrapping_sub(started) >= 500) {
236            state.failed = true;
237            return Err(String::from("Native layout owner resume timed out"));
238        }
239        let caller = match ostd::syscall::sys_try_recv(0, &mut receive) {
240            ostd::syscall::SyscallResult::Ok(caller) if caller > 0 => caller,
241            _ => {
242                ostd::syscall::sys_yield();
243                continue;
244            }
245        };
246        let Some(operation) = ostd::ipc::current() else {
247            continue;
248        };
249        let decoded = postcard::from_bytes::<OcelJsRequest>(&receive);
250        let rejected = OcelJsReply::Error {
251            message: String::from("No matching suspended native layout query"),
252        };
253        if caller != state.caller {
254            let _ = send(operation, &rejected);
255            continue;
256        }
257        let result = match decoded {
258            Ok(OcelJsRequest::ReadLayoutQuery {
259                context,
260                query,
261                offset,
262            }) if context == state.context && query == state.queries => {
263                chunk(state, &outgoing, offset)
264            }
265            Ok(OcelJsRequest::LayoutResumeBegin {
266                context,
267                query,
268                total,
269            }) if context == state.context
270                && query == state.queries
271                && total > 0
272                && total <= OCEL_JS_MAX_TRANSFER =>
273            {
274                incoming.clear();
275                expected = Some(total);
276                Ok(OcelJsReply::Ack)
277            }
278            Ok(OcelJsRequest::LayoutResumeChunk {
279                context,
280                query,
281                offset,
282                data,
283            }) if context == state.context
284                && query == state.queries
285                && expected.is_some()
286                && offset == incoming.len()
287                && !data.is_empty()
288                && data.len() <= OCEL_JS_CHUNK_SIZE
289                && offset.saturating_add(data.len()) <= expected.unwrap_or(0) =>
290            {
291                incoming.extend(data);
292                Ok(OcelJsReply::Ack)
293            }
294            Ok(OcelJsRequest::LayoutResumeCommit { context, query })
295                if context == state.context
296                    && query == state.queries
297                    && expected == Some(incoming.len()) =>
298            {
299                state.operation = Some(operation);
300                set_reply_operation(Some(operation));
301                return match postcard::from_bytes::<BrowserLayoutReply>(&incoming) {
302                    Ok(BrowserLayoutReply::Frame(frame)) => {
303                        if frame.layout.iter().any(|row| {
304                            ![row.x, row.y, row.width, row.height]
305                                .iter()
306                                .all(|n| n.is_finite())
307                                || row.width < 0.0
308                                || row.height < 0.0
309                        }) {
310                            state.failed = true;
311                            Err(String::from("Invalid native layout geometry"))
312                        } else {
313                            Ok(frame)
314                        }
315                    }
316                    Ok(BrowserLayoutReply::Error(message)) => Err(message),
317                    Err(_) => {
318                        state.failed = true;
319                        Err(String::from("Malformed native layout resume"))
320                    }
321                };
322            }
323            _ => Ok(rejected),
324        };
325        let response = result.unwrap_or_else(|message| OcelJsReply::Error { message });
326        if send(operation, &response).is_err() {
327            state.failed = true;
328            return Err(String::from("Native layout resume operation was abandoned"));
329        }
330    }
