2//! Synchronous native CSSOM queries keep the QuickJS C stack suspended while
3//! the owner acknowledges/apply mutations and commits a real native layout.
4use alloc::boxed::Box;
5use alloc::string::{String, ToString};
6use alloc::vec::Vec;
7use dom_arena::{BrowserFrame, BrowserLayoutQuery, BrowserLayoutReply, DomMutation, NodeData, NodeId,
8    OcelJsReply, OcelJsRequest, OCEL_JS_CHUNK_SIZE, OCEL_JS_IPC_BUF_SIZE, OCEL_JS_MAX_TRANSFER};
9use super::{engine, ffi};

11use core::sync::atomic::{AtomicUsize, Ordering};
12static REPLY_OPERATION: AtomicUsize = AtomicUsize::new(0);
13pub(super) fn set_reply_operation(operation: Option<usize>) { REPLY_OPERATION.store(operation.unwrap_or(0), Ordering::Relaxed); }
14pub(super) fn reply_operation() -> Option<usize> {
15    match REPLY_OPERATION.load(Ordering::Relaxed) { 0 => None, operation => Some(operation) }
16}
17#[derive(Default)]
18pub(super) struct State {
19    pub caller: usize,
20    pub context: u64,
21    pub operation: Option<usize>,
22    pub queries: u32,
23    pub mutations: usize,
24    pub mutation_bytes: usize,
25    pub failed: bool,
26    pub unapplied: Vec<DomMutation>,
27    #[cfg(test)]
28    pub host: Option<fn(&BrowserLayoutQuery) -> Result<BrowserFrame, String>>,
29}

31pub(super) fn install(ctx: *mut ffi::JSContext, state: &mut Box<State>) {
32    unsafe { ffi::vios_js_install_native_layout(ctx, (&mut **state as *mut State).cast(), Some(callback)); }
33}

35unsafe extern "C" fn callback(ctx: *mut ffi::JSContext, opaque: *mut core::ffi::c_void,
36    node: u32, wire: *const core::ffi::c_char, length: usize) -> ffi::JSValue {
37    let state = &mut *opaque.cast::<State>();
38    let result = (|| -> Result<BrowserFrame, String> {
39        if state.failed { return Err(String::from("Native layout session has failed")); }
40        state.queries += 1;
41        if state.queries > 64 { return Err(String::from("Synchronous native layout query limit exceeded")); }
42        let text = core::str::from_utf8(core::slice::from_raw_parts(wire.cast::<u8>(), length))
43            .map_err(|_| String::from("Invalid native layout mutation encoding"))?;
44        let mutations = parse_mutations(text)?;
45        let has_host = state.operation.is_some();
46        #[cfg(test)]
47        let has_host = has_host || state.host.is_some();
48        if !has_host {
49            state.unapplied.extend(mutations);
50            return Err(String::from("Native layout host is unavailable"));
51        }
52        state.mutations = state.mutations.saturating_add(mutations.len());
53        state.mutation_bytes = state.mutation_bytes.saturating_add(dom_arena::ipc::serialized_len_bounded(&mutations, dom_arena::OCEL_JS_MAX_MUTATION_BYTES)
54            .map_err(|_| String::from("Native layout mutation byte budget exceeded"))?);
55        if state.mutations > dom_arena::OCEL_JS_MAX_MUTATIONS || state.mutation_bytes > dom_arena::OCEL_JS_MAX_MUTATION_BYTES {
56            return Err(String::from("Native layout mutation budget exceeded"));
57        }
58        let query = BrowserLayoutQuery { node: NodeId(node), mutations };
59        #[cfg(test)]
60        if let Some(host) = state.host { return host(&query); }
61        exchange(state, query)
62    })();
63    match result {
64        Ok(frame) => {
65            let source = engine::frame_expression(&frame);
66            let code = match alloc::ffi::CString::new(source) { Ok(code) => code, Err(_) => return throw(ctx, "Invalid native frame source") };
67            ffi::JS_Eval(ctx, code.as_ptr(), code.as_bytes().len(), c"<native-layout>".as_ptr(), ffi::JS_EVAL_TYPE_GLOBAL)
68        }
69        Err(error) => throw(ctx, &error),
70    }
71}
72unsafe fn throw(ctx: *mut ffi::JSContext, message: &str) -> ffi::JSValue {
73    let text = alloc::ffi::CString::new(message.replace('\0', " ")).unwrap();
74    ffi::vios_js_throw_error(ctx, text.as_ptr())
75}

77pub(super) fn parse_mutations(text: &str) -> Result<Vec<DomMutation>, String> {
78    let mut out = Vec::new();
79    for row in text.lines() {
80        if out.len() >= dom_arena::OCEL_JS_MAX_MUTATIONS { return Err(String::from("Mutation count budget exceeded")); }
81        let fields = row.split('\t').map(engine::decode_field).collect::<Option<Vec<_>>>()
82            .ok_or_else(|| String::from("Invalid mutation encoding"))?;
83        let field = |index| fields.get(index).map(String::as_str).ok_or_else(|| String::from("Missing mutation field"));
84        let id = |index| field(index)?.parse::<u32>().map(NodeId).map_err(|_| String::from("Invalid mutation node"));
85        let mutation = match field(0)? {
86            "C" if fields.len() == 5 => DomMutation::CreateNode { node: id(1)?, data: match field(2)? {
87                "1" => NodeData::Element { tag: field(3)?.to_string(), attributes: Vec::new() },
88                "3" => NodeData::Text(field(4)?.to_string()), "8" => NodeData::Comment(field(4)?.to_string()),
89                "11" => NodeData::DocumentFragment, _ => return Err(String::from("Invalid created node type")),
90            } },
91            "X" if fields.len() == 3 => DomMutation::SetText { node: id(1)?, text: field(2)?.to_string() },
92            "A" if fields.len() == 4 => DomMutation::SetAttribute { node: id(1)?, key: field(2)?.to_string(), val: field(3)?.to_string() },
93            "R" if fields.len() == 3 => DomMutation::RemoveAttribute { node: id(1)?, key: field(2)?.to_string() },
94            "P" if fields.len() == 3 => DomMutation::AppendChild { parent: id(1)?, child: id(2)? },
95            "I" if fields.len() == 4 => DomMutation::InsertBefore { parent: id(1)?, child: id(2)?, reference: id(3)? },
96            "D" if fields.len() == 3 => DomMutation::RemoveChild { parent: id(1)?, child: id(2)? },
97            "T" if fields.len() == 2 => DomMutation::SetDocumentTitle { title: field(1)?.to_string() },
98            _ => return Err(String::from("Invalid mutation operation")),
99        };
100        out.push(mutation);
101    }
102    Ok(out)
103}
104fn send(operation: usize, reply: &OcelJsReply) -> Result<(), String> {
105    let mut bytes = [0u8; OCEL_JS_IPC_BUF_SIZE];
106    let encoded = postcard::to_slice(reply, &mut bytes).map_err(|_| String::from("Native layout frame overflow"))?;
107    ostd::ipc::reply(operation, encoded).map_err(|_| String::from("Native layout operation was abandoned"))
108}
109fn chunk(state: &State, bytes: &[u8], offset: usize) -> Result<OcelJsReply, String> {
110    if offset >= bytes.len() { return Err(String::from("Native layout offset out of bounds")); }
111    let end = offset.saturating_add(OCEL_JS_CHUNK_SIZE).min(bytes.len());
112    Ok(OcelJsReply::LayoutQueryChunk { context: state.context, query: state.queries, offset, total: bytes.len(), data: bytes[offset..end].to_vec() })
113}
114fn exchange(state: &mut State, query: BrowserLayoutQuery) -> Result<BrowserFrame, String> {
115    let operation = state.operation.take().ok_or_else(|| String::from("Native layout host is unavailable"))?;
116    let outgoing = dom_arena::ipc::encode_bounded(&query, OCEL_JS_MAX_TRANSFER).map_err(|_| String::from("Native layout query exceeds transfer budget"))?;
117    if let Err(error) = send(operation, &chunk(state, &outgoing, 0)?) { state.failed = true; return Err(error); }
118    set_reply_operation(None);
119    let Some(started) = ostd::syscall::sys_get_scheduler_ticks() else { state.failed = true; return Err(String::from("Native layout clock unavailable")); };
120    let mut incoming = Vec::new(); let mut expected = None;
121    let mut receive = [0u8; OCEL_JS_IPC_BUF_SIZE];
122    loop {
123        let now = ostd::syscall::sys_get_scheduler_ticks();
124        if now.is_none_or(|now| now.wrapping_sub(started) >= 500) { state.failed = true; return Err(String::from("Native layout owner resume timed out")); }
125        let caller = match ostd::syscall::sys_try_recv(0, &mut receive) {
126            ostd::syscall::SyscallResult::Ok(caller) if caller > 0 => caller,
127            _ => { ostd::syscall::sys_yield(); continue; }
128        };
129        let Some(operation) = ostd::ipc::current() else { continue; };
130        let decoded = postcard::from_bytes::<OcelJsRequest>(&receive);
131        let rejected = OcelJsReply::Error { message: String::from("No matching suspended native layout query") };
132        if caller != state.caller { let _ = send(operation, &rejected); continue; }
133        let result = match decoded {
134            Ok(OcelJsRequest::ReadLayoutQuery { context, query, offset }) if context == state.context && query == state.queries => chunk(state, &outgoing, offset),
135            Ok(OcelJsRequest::LayoutResumeBegin { context, query, total }) if context == state.context && query == state.queries && total > 0 && total <= OCEL_JS_MAX_TRANSFER => {
136                incoming.clear(); expected = Some(total); Ok(OcelJsReply::Ack)
137            }
138            Ok(OcelJsRequest::LayoutResumeChunk { context, query, offset, data }) if context == state.context && query == state.queries
139                && expected.is_some() && offset == incoming.len() && !data.is_empty() && data.len() <= OCEL_JS_CHUNK_SIZE
140                && offset.saturating_add(data.len()) <= expected.unwrap_or(0) => { incoming.extend(data); Ok(OcelJsReply::Ack) }
141            Ok(OcelJsRequest::LayoutResumeCommit { context, query }) if context == state.context && query == state.queries && expected == Some(incoming.len()) => {
142                state.operation = Some(operation);
143                set_reply_operation(Some(operation));
144                return match postcard::from_bytes::<BrowserLayoutReply>(&incoming) {
145                    Ok(BrowserLayoutReply::Frame(frame)) => {
146                        if frame.layout.iter().any(|row| ![row.x, row.y, row.width, row.height].iter().all(|n| n.is_finite()) || row.width < 0.0 || row.height < 0.0) {
147                            state.failed = true; Err(String::from("Invalid native layout geometry"))
148                        } else { Ok(frame) }
149                    }
150                    Ok(BrowserLayoutReply::Error(message)) => Err(message),
151                    Err(_) => { state.failed = true; Err(String::from("Malformed native layout resume")) },
152                };
153            }
154            _ => Ok(rejected),
155        };
156        let response = result.unwrap_or_else(|message| OcelJsReply::Error { message });
157        if send(operation, &response).is_err() { state.failed = true; return Err(String::from("Native layout resume operation was abandoned")); }
158    }
