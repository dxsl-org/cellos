1//! Orderly shutdown for a `PowerRequest` (owner decision H-S2-8 (i)).
2//!
3//! Contract: freeze (413) every Cell except this supervisor and the storage
4//! stack (the VFS and block-driver services and every live block-route
5//! provider), give the storage stack a bounded settle window to finish the
6//! requests it already holds, then call `SystemPower` (436). Nothing is
7//! flushed or synced, so the markers report only freeze counts and the window.
8//! A second power-button press ends the window at once. If 436 returns, every
9//! Cell this sweep froze is resumed and the 436 error is returned.
10
11use api::syscall::{service, ProcessInfo};
12use ostd::io::print_fmt;
13use ostd::service::provider_tid;
14use ostd::syscall::{
15    sys_freeze_cell, sys_get_procs, sys_lookup_block_route, sys_recv_timeout, sys_resume_cell,
16    sys_system_power, SyscallResult, SystemPowerAction, SystemPowerError,
17};
18
19use crate::protocol::{OP_POWER, POWER_ACTION_POWEROFF};
20
21/// Who asked: the `/bin/shutdown` worker or the kernel's power-button relay.
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub enum Origin {
24    Operator,
25    PowerButton,
26}
27
28const PROCESS_TABLE_ROWS: usize = 64;
29const STATE_TERMINATED: usize = 3;
30/// Upper bound of the settle window, in 10 ms scheduler ticks.
31const SETTLE_TICKS: u64 = 50;
32const SETTLE_POLL_TICKS: u64 = 5;
33const KEEP_MAX: usize = 2 + api::block_route::MAX_BLOCK_ROUTES;
34/// The kernel relay's bytes: App SDK envelope, `OP_POWER`, poweroff.
35const POWER_BUTTON_MESSAGE: [u8; 4] = [0xAC, 0x00, OP_POWER, POWER_ACTION_POWEROFF];
36
37fn action_name(action: SystemPowerAction) -> &'static str {
38    match action {
39        SystemPowerAction::Reboot => "reboot",
40        SystemPowerAction::PowerOff => "poweroff",
41    }
42}
43
44/// Freeze, settle, call 436; on its return resume and report the error.
45pub fn shut_down(action: SystemPowerAction, origin: Origin) -> SystemPowerError {
46    let mut keep = [0usize; KEEP_MAX];
47    let kept = storage_stack(&mut keep);
48    let own = provider_tid(service::SUPERVISOR);
49    let mut rows = [ProcessInfo::default(); PROCESS_TABLE_ROWS];
50    let count = sys_get_procs(&mut rows)
51        .unwrap_or(0)
52        .min(PROCESS_TABLE_ROWS);
53    let mut frozen = [0usize; PROCESS_TABLE_ROWS];
54    let (mut froze, mut refused) = (0usize, 0usize);
55    for row in &rows[..count] {
56        if row.id == 0
57            || Some(row.id) == own
58            || row.state == STATE_TERMINATED
59            || keep[..kept].contains(&row.id)
60        {
61            continue;
62        }
63        if sys_freeze_cell(row.id).is_ok() {
64            frozen[froze] = row.id;
65            froze += 1;
66        } else {
67            refused += 1;
68        }
69    }
70    print_fmt(format_args!(
71        "[supervisor] power: {} ({:?}) froze={} refused={} kept={} settle<={}ms\n",
72        action_name(action),
73        origin,
74        froze,
75        refused,
76        kept,
77        SETTLE_TICKS * 10
78    ));
79    if settle() {
80        print_fmt(format_args!(
81            "[supervisor] power: second power-button press ended the settle window\n"
82        ));
83    }
84    print_fmt(format_args!(
85        "[supervisor] power: requesting {}\n",
86        action_name(action)
87    ));
88    let error = sys_system_power(action);
89    for &tid in &frozen[..froze] {
90        let _ = sys_resume_cell(tid);
91    }
92    print_fmt(format_args!(
93        "[supervisor] power: {} refused ({:?}); resumed {} cells\n",
94        action_name(action),
95        error,
96        froze
97    ));
98    error
99}

101/// Tids the storage stack runs as: VFS, the block-driver service and every
102/// live block-route provider. Returns how many were written to `keep`.
103fn storage_stack(keep: &mut [usize; KEEP_MAX]) -> usize {
104    let mut kept = 0;
105    let mut add = |tid: usize| {
106        if tid != 0 && !keep[..kept].contains(&tid) && kept < KEEP_MAX {
107            keep[kept] = tid;
108            kept += 1;
109        }
110    };
111    for id in [service::VFS, service::BLOCK_DRIVER] {
112        if let Some(tid) = provider_tid(id) {
113            add(tid);
114        }
115    }
116    for index in 0..api::block_route::MAX_BLOCK_ROUTES as u32 {
117        match sys_lookup_block_route(index) {
118            Ok(Some(route)) => {
119                if route.flags & api::block_route::FLAG_PROVIDER_LIVE != 0 {
120                    add(route.provider_tid as usize);
121                }
122            }
123            _ => break,
124        }
125    }
126    kept
127}

129/// Wait out the settle window; `true` when a second power-button press ended
130/// it. Other messages are dropped: every sender but the storage stack is
131/// frozen, and the window only gets shorter for them.
132fn settle() -> bool {
133    let mut buf = [0u8; 64];
134    let mut waited = 0;
135    while waited < SETTLE_TICKS {
136        buf.fill(0);
137        let result = sys_recv_timeout(0, &mut buf, SETTLE_POLL_TICKS);
138        waited += SETTLE_POLL_TICKS;
139        // The relay's sender id is 0, which a timeout also returns; only the
140        // relay writes these bytes into the cleared buffer.
141        if matches!(result, SyscallResult::Ok(0)) && buf[..4] == POWER_BUTTON_MESSAGE {
142            return true;
143        }
144    }
145    false
146}
