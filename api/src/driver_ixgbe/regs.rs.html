66pub(crate) trait RegisterFile {
67    fn rd32(&self, offset: usize) -> ViResult<u32>;
68    fn wr32(&self, offset: usize, value: u32) -> ViResult<()>;
69}
70
71impl RegisterFile for MmioRegion {
72    fn rd32(&self, offset: usize) -> ViResult<u32> {
73        self.read_u32(offset)
74    }
75
76    fn wr32(&self, offset: usize, value: u32) -> ViResult<()> {
77        self.write_u32(offset, value)
78    }
79}
80
81pub(crate) fn nvm_read(regs: &impl RegisterFile, word: u16) -> ViResult<u16> {
82    let cmd = ((word as u32) << EERD_ADDR_SHIFT) | EERD_START;
83    regs.wr32(EERD, cmd)?;
84    for _ in 0..NVM_POLLS {
85        let value = regs.rd32(EERD)?;
86        if value & EERD_DONE != 0 {
87            return Ok((value >> EERD_DATA_SHIFT) as u16);
88        }
89        core::hint::spin_loop();
90    }
91    Err(ViError::IO)
92}
