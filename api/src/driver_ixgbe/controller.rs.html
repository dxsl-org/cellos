1//! Intel 82599/X540/X550 controller initialization and Tx/Rx data path.
2
3use crate::dma_layout::{
4    for_each_initial_dma_program, try_init_array, with_authorized_dma_layout, DmaIovas, DmaSlot,
5    InitialDmaProgram, RX_SLOTS, TX_SLOTS,
6};
7use crate::identity::IxgbeMac;
8use crate::regs::*;
9use crate::rings::{self, RxDesc};
10use crate::sfp;
11use ostd::dma::DmaBuf;
12use ostd::mmio::MmioRegion;
13use types::{ViError, ViResult};
14
15pub use crate::rings::BUF_SIZE;
16
17const RESET_POLLS: u32 = 1_000_000;
18const LINK_POLLS: u32 = 6_000_000;
19
20pub struct IxgbeController {
21    pub(crate) mmio: MmioRegion,
22    pub(crate) tx_ring: DmaBuf,
23    pub(crate) rx_ring: DmaBuf,
24    pub(crate) tx_bufs: [DmaBuf; TX_SLOTS],
25    pub(crate) rx_bufs: [DmaBuf; RX_SLOTS],
26    pub(crate) dma_iovas: DmaIovas,
27    pub(crate) tx_next: usize,
28    pub(crate) rx_head: usize,
29    pub mac: [u8; 6],
30    pub link_up: bool,
31    #[allow(dead_code)]
32    pub mac_type: IxgbeMac,
33}
34
35unsafe impl Send for IxgbeController {}
36
37impl IxgbeController {
38    pub fn new(mmio: MmioRegion, bdf: u32, mac_type: IxgbeMac) -> ViResult<Self> {
39        // 1. Quiesce & reset controller.
40        mmio.write_u32(CTRL, CTRL_RST)?;
41        let mut polls = 0u32;
42        while mmio.read_u32(CTRL)? & CTRL_RST != 0 {
43            polls += 1;
44            if polls > RESET_POLLS {
45                return Err(ViError::IO);
46            }
47            core::hint::spin_loop();
48        }
49
50        // 2. SFP module check for 82599.
51        if mac_type == IxgbeMac::Mac82599 {
52            let _ = sfp::identify_sfp(&mmio);
53        }
54
55        let mac = match (mmio.read_u32(RAL0)?, mmio.read_u32(RAH0)?) {
56            (ral, rah) if rah & RAH_AV != 0 && ral != 0 => [
57                ral as u8,
58                (ral >> 8) as u8,
59                (ral >> 16) as u8,
60                (ral >> 24) as u8,
61                rah as u8,
62                (rah >> 8) as u8,
63            ],
64            _ => {
65                let (w0, w1, w2) = (
66                    nvm_read(&mmio, 0)?,
67                    nvm_read(&mmio, 1)?,
68                    nvm_read(&mmio, 2)?,
69                );
70                [
71                    w0 as u8,
72                    (w0 >> 8) as u8,
73                    w1 as u8,
74                    (w1 >> 8) as u8,
75                    w2 as u8,
76                    (w2 >> 8) as u8,
77                ]
78            }
79        };
80        if mac == [0u8; 6] {
81            return Err(ViError::IO);
82        }
83
84        mmio.write_u32(RAL0, u32::from_le_bytes([mac[0], mac[1], mac[2], mac[3]]))?;
85        mmio.write_u32(RAH0, RAH_AV | (u16::from_le_bytes([mac[4], mac[5]]) as u32))?;
86
87        let tx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
88        let tx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
89        let rx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
90        let rx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
91
92        with_authorized_dma_layout(
93            (mmio, tx_ring, rx_ring, tx_bufs, rx_bufs),
94            |r, s| {
95                match s {
96                    DmaSlot::TxRing => r.1.authorize(bdf),
97                    DmaSlot::TxBuffer(i) => r.3[i].authorize(bdf),
98                    DmaSlot::RxRing => r.2.authorize(bdf),
99                    DmaSlot::RxBuffer(i) => r.4[i].authorize(bdf),
100                }
101                .map_err(|_| ViError::PermissionDenied)
102            },
103            |(mmio, tx_ring, rx_ring, tx_bufs, rx_bufs), dma_iovas| {
104                rings::clear(&tx_ring);
105                rings::clear(&rx_ring);
106                let mut ctrl = IxgbeController {
107                    mmio,
108                    tx_ring,
109                    rx_ring,
110                    tx_bufs,
111                    rx_bufs,
112                    dma_iovas,
113                    tx_next: 0,
114                    rx_head: 0,
115                    mac,
116                    link_up: false,
117                    mac_type,
118                };
119                for_each_initial_dma_program(&dma_iovas, |p| {
120                    let _ = ctrl.apply(p);
121                });
122                let _ = ctrl.finish_init();
123                Ok(ctrl)
124            },
125        )?
126    }
127
128    fn apply(&mut self, program: InitialDmaProgram) -> ViResult<()> {
129        match program {
130            InitialDmaProgram::TxRingBase(iova) => {
131                self.mmio.write_u32(TDBAL0, iova as u32)?;
132                self.mmio.write_u32(TDBAH0, (iova >> 32) as u32)?;
133                self.mmio.write_u32(TDLEN0, (TX_SLOTS * 16) as u32)?;
134                self.mmio.write_u32(TDH0, 0)?;
135                self.mmio.write_u32(TDT0, 0)?;
136            }
137            InitialDmaProgram::RxRingBase(iova) => {
138                self.mmio.write_u32(RDBAL0, iova as u32)?;
139                self.mmio.write_u32(RDBAH0, (iova >> 32) as u32)?;
140                self.mmio.write_u32(RDLEN0, (RX_SLOTS * 16) as u32)?;
141                self.mmio.write_u32(RDH0, 0)?;
142                self.mmio.write_u32(RDT0, 0)?;
143            }
144            InitialDmaProgram::RxBuffer { index, iova } => rings::write_rx_desc(
145                &self.rx_ring,
146                index,
147                RxDesc {
148                    buffer_addr: iova,
149                    status_error: 0,
150                    length: 0,
151                    vlan: 0,
152                },
153            ),
154        }
155        Ok(())
156    }
157
158    fn finish_init(&mut self) -> ViResult<()> {
159        self.mmio.write_u32(RXDCTL0, RXDCTL_ENABLE)?;
160        self.mmio.write_u32(TXDCTL0, TXDCTL_ENABLE)?;
161        self.mmio.write_u32(DMATXCTL, DMATXCTL_TE)?;
162        self.mmio.write_u32(RDT0, (RX_SLOTS - 1) as u32)?;
163        self.mmio.write_u32(RXCTRL, RXCTRL_RXEN)?;
164        self.mmio
165            .write_u32(FCTRL, FCTRL_BAM | FCTRL_MPE | FCTRL_UPE)?;
166
167        let mut polls = 0u32;
168        while self.mmio.read_u32(STATUS)? & STATUS_LU == 0 {
169            polls += 1;
170            if polls > LINK_POLLS {
171                break;
172            }
173            core::hint::spin_loop();
174        }
175        self.link_up = self.mmio.read_u32(STATUS)? & STATUS_LU != 0;
176        Ok(())
177    }
178}
