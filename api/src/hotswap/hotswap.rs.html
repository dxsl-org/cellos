11const APP_MESSAGE_PREFIX: [u8; 2] = [0xAC, 0x00];
12const OP_HOTSWAP: u8 = 0x01;
13const OP_STATUS: u8 = 0x03;
14const SVC_NAME_LEN: usize = 64;
15const ELF_PATH_LEN: usize = 128;
16const REQUEST_LEN: usize = 1 + SVC_NAME_LEN + ELF_PATH_LEN;
17const STATUS_LEN: usize = 3;
18const SUCCESS_PHASE: u8 = 6;
19const SUCCESS_CODE: u8 = 0;
20const REPLY_TIMEOUT_TICKS: u64 = 1_600;
21const USAGE: &str = "hotswap: usage: hotswap <service-name> <new-elf-path>";
22const EXACT_BIN_TARGET_ERROR: &str = "hotswap: ELF path must be an exact /bin/<name> target";
23const MALFORMED_REPLY: &str = "hotswap: protocol error: malformed supervisor reply";
24
25/// hotswap <service-name> <new-elf-path> - ask the Supervisor Cell to replace a
26/// registered service.
27///
28/// Phase 02 keeps this CLI service-only on purpose: the supervisor routes
29/// hotswap requests by registered `service::*` id, not arbitrary tids.
30/// The reply timeout spans the supervisor's bounded snapshot/freeze/spawn/
31/// restore/publish sequence rather than a single IPC hop.
32fn cell_main() {
33    let argv = ostd::args();
34    let [service_name, new_path] = argv.as_slice() else {
35        print_usage_and_exit();
36    };
37    let service_name = service_name.as_str();
38    let new_path = new_path.as_str();
39
40    let request = match encode_hotswap_request(service_name, new_path) {
41        Ok(request) => request,
42        Err(message) => exit_with_message(message),
43    };
44
45    let Some(supervisor_tid) = sys_lookup_service(service::SUPERVISOR) else {
46        exit_with_message("hotswap: supervisor unavailable");
47    };
48
49    if !send_request(supervisor_tid, &request) {
50        exit_with_message("hotswap: cannot send request to supervisor");
51    }
52
53    match recv_status(supervisor_tid) {
54        Ok(()) => {
55            ostd::io::print("hotswap: success: supervisor swapped service '");
56            ostd::io::print(service_name);
57            ostd::io::print("' to ");
58            ostd::io::println(new_path);
59            syscall::sys_exit(0);
60        }
61        Err(StatusError::Message(message)) => exit_with_message(message),
62        Err(StatusError::UnexpectedStatus { phase, code }) => {
63            ostd::io::print("hotswap: supervisor returned unexpected status phase=");
64            ostd::io::print_usize(phase as usize);
65            ostd::io::print(" result=");
66            ostd::io::print_usize(code as usize);
67            ostd::io::println("");
68            syscall::sys_exit(1);
69        }
70    }
71}

73fn encode_hotswap_request(
74    service_name: &str,
75    new_path: &str,
76) -> Result<[u8; REQUEST_LEN], &'static str> {
77    if service_name.is_empty() || new_path.is_empty() {
78        return Err(USAGE);
79    }
80
81    validate_service_name(service_name)?;
82    validate_elf_path(new_path)?;
83
84    let mut request = [0u8; REQUEST_LEN];
85    request[0] = OP_HOTSWAP;
86    let service_bytes = service_name.as_bytes();
87    request[1..1 + service_bytes.len()].copy_from_slice(service_bytes);
88    let path_bytes = new_path.as_bytes();
89    request[1 + SVC_NAME_LEN..1 + SVC_NAME_LEN + path_bytes.len()].copy_from_slice(path_bytes);
90    Ok(request)
91}
92
93fn send_request(supervisor_tid: usize, request: &[u8; REQUEST_LEN]) -> bool {
94    let mut envelope = [0u8; APP_MESSAGE_PREFIX.len() + REQUEST_LEN];
95    envelope[..APP_MESSAGE_PREFIX.len()].copy_from_slice(&APP_MESSAGE_PREFIX);
96    envelope[APP_MESSAGE_PREFIX.len()..].copy_from_slice(request);
97    matches!(sys_send(supervisor_tid, &envelope), SyscallResult::Ok(_))
98}

100fn recv_status(supervisor_tid: usize) -> Result<(), StatusError> {
101    let mut status = [0u8; STATUS_LEN];
102    match sys_recv_timeout(supervisor_tid, &mut status, REPLY_TIMEOUT_TICKS) {
103        SyscallResult::Ok(0) => Err(StatusError::Message(
104            "hotswap: timed out waiting for supervisor status",
105        )),
106        SyscallResult::Ok(sender) if sender != supervisor_tid => Err(StatusError::Message(
107            "hotswap: protocol error: reply came from an unexpected sender",
108        )),
109        SyscallResult::Ok(_) => parse_status(status),
110        SyscallResult::Err(_) => Err(StatusError::Message(
111            "hotswap: protocol error: recv from supervisor failed",
112        )),
113    }
114}
115
116fn parse_status(status: [u8; STATUS_LEN]) -> Result<(), StatusError> {
117    if status[0] != OP_STATUS {
118        return Err(StatusError::Message(MALFORMED_REPLY));
119    }
120
121    match status {
122        [OP_STATUS, SUCCESS_PHASE, SUCCESS_CODE] => Ok(()),
123        [OP_STATUS, 0, 0xFF] => Err(StatusError::Message(
124            "hotswap: protocol error: supervisor rejected request framing",
125        )),
126        [OP_STATUS, 0, 0xFE] => Err(StatusError::Message(
127            "hotswap: target service is not registered with the supervisor",
128        )),
129        [OP_STATUS, 0, 0xFD] => Err(StatusError::Message(
130            "hotswap: supervisor only accepts requests from the shell-launched hotswap CLI",
131        )),
132        [OP_STATUS, 0xFF, code] => Err(StatusError::Message(supervisor_error_message(code))),
133        [OP_STATUS, phase, code] => Err(StatusError::UnexpectedStatus { phase, code }),
134        _ => Err(StatusError::Message(MALFORMED_REPLY)),
135    }
136}

138fn validate_service_name(service_name: &str) -> Result<(), &'static str> {
139    let bytes = service_name.as_bytes();
140    if bytes.contains(&0) || bytes.len() >= SVC_NAME_LEN {
141        return Err("hotswap: service name must fit in 63 bytes plus trailing NUL");
142    }
143    if !bytes.is_ascii()
144        || !bytes
145            .iter()
146            .all(|byte| byte.is_ascii_alphanumeric() || *byte == b'-')
147    {
148        return Err("hotswap: service name must be ASCII letters, digits, or '-'");
149    }
150    Ok(())
151}
152
153fn validate_elf_path(path: &str) -> Result<(), &'static str> {
154    let bytes = path.as_bytes();
155    if bytes.contains(&0) || bytes.len() >= ELF_PATH_LEN {
156        return Err("hotswap: ELF path must fit in 127 bytes plus trailing NUL");
157    }
158    let Some(suffix) = path.strip_prefix("/bin/") else {
159        return Err(EXACT_BIN_TARGET_ERROR);
160    };
161    if suffix.is_empty() || suffix.contains('/') || suffix.contains("..") || path.contains("//") {
162        return Err(EXACT_BIN_TARGET_ERROR);
163    }
164    if !bytes.is_ascii() {
165        return Err("hotswap: ELF path must be ASCII");
166    }
167    Ok(())
168}

170fn supervisor_error_message(code: u8) -> &'static str {
171    match code {
172        0 => "hotswap: supervisor could not find the target service",
173        1 => "hotswap: supervisor could not freeze the target service",
174        2 => "hotswap: supervisor could not deliver the snapshot request",
175        3 => "hotswap: supervisor timed out waiting for the snapshot",
176        4 => "hotswap: supervisor could not spawn the replacement ELF",
177        5 => "hotswap: supervisor could not deliver the restore request",
178        6 => "hotswap: supervisor timed out waiting for the replacement to become ready",
179        7 => "hotswap: supervisor could not pause the service for cutover",
180        8 => "hotswap: supervisor could not publish the replacement service",
181        _ => "hotswap: supervisor returned an unknown error code",
182    }
183}
