1//! Private Phase03 stdin/stdout filter fixture. Mode is `argv[0]` (the first
2//! word after the program name); bytes move through the real delegated stream
3//! endpoints in the child context (Stdin / Stdout grants), never through a
4//! terminal session (only `tui-abandon` touches one, for its raw-mode lease).
5//!
6//! Modes (exact byte semantics, nothing is interpreted or re-framed):
7//! - `cat`        copy stdin to stdout until end of stream, exit 0
8//! - `upper`      like `cat`, ASCII a-z mapped to A-Z, every other byte unchanged
9//! - `noread`     exit 0 at once without touching any endpoint
10//! - `exit N`     exit with status `N` (any `i32`) without touching any endpoint
11//! - `gen`        write the fixed 6 bytes `00 ff 41 fe 80 7a` (NUL, non-UTF-8, no
12//!                trailing newline) to stdout, exit 0
13//! - `hex`        read stdin to end of stream and print
14//!                `p03-hex len=<total> hex=<first 64 bytes as lowercase hex>\n`
15//! - `endless`    write 512-byte blocks until the consumer closes (exit 141) or
16//!                the operation is cancelled (exit 130)
17//! - `rows`       write the fixed plain JSONL `{"a":1}\n{"a":2}\n` (an external
18//!                head for `from jsonl`), exit 0
19//! - `typed`      write the fixed typed JSONL records `pid` 7 and `pid` 9 in the
20//!                `cellbox` encoding (an external head for `from jsonl --typed`)
21//! - `net-probe`  send each rights-gated `NetRequest` (Resolve, TcpConnect,
22//!                UdpCreate, UdpBind, TcpListen, MulticastJoin, L2Send) once to the
23//!                live net service and print one `NETPROBE <op> <result>` line per
24//!                request, then `NETPROBE DONE`; `<result>` is `denied` for
25//!                `Err(NET_ERR_DENIED)` and `other:<detail>` for anything else.
26//!                This Cell holds no network right (its launch edge has an empty
27//!                ceiling), so every line is expected to read `denied`.
28//! - `config-probe get KEY` / `config-probe set KEY VALUE` /
29//!   `config-probe delete KEY`
30//!                send one `ConfigRequest` to the live config service and print
31//!                `CFGPROBE <op> <KEY> <result>`; `<result>` is `ok`, `denied`
32//!                (`Err(CONFIG_ERR_DENIED)`), `value:<v>`, `notfound` or
33//!                `other:<detail>`. This Cell holds no Q right (its edge ceiling
34//!                lacks `config_write`), so `set`/`delete` are expected to read
35//!                `denied`; `get` is ungated (Phase 01 CP1).
36//! - `tui-abandon` simulate a killed TUI: write the `vitui` enter bytes
37//!                `ESC[?1049h ESC[?25l ESC[?2004h` followed by `p03-tui-abandon\n` to
38//!                stdout and exit 0 WITHOUT restoring anything. As the serial session's
39//!                foreground reader it first takes the raw-mode lease (best effort) and
40//!                keeps it to exit; as a pipeline stage it is no reader (no session
41//!                binding, stdin end of file) and the bytes alone are the observable.
42//!
43//! Exit statuses: 0 ok; 2 missing grant / bad argument / unknown mode; 3 other
44//! stream failure; 141 stdout `BrokenPipe`; 130 `Cancelled`.
45//!
46//! Never part of an ordinary image: reachable only through the exact
47//! `phase03-filter-profile` launch row.

97const OK: i32 = 0;
98const USAGE: i32 = 2;
99const IO: i32 = 3;
100const CANCELLED: i32 = 130;
101const BROKEN_PIPE: i32 = 141;
102/// No deadline: block until data, end of stream, close or cancellation.
103const FOREVER: u64 = u64::MAX;
104const GEN_BYTES: [u8; 6] = [0x00, 0xff, 0x41, 0xfe, 0x80, 0x7a];
105const ROWS_JSONL: &[u8] = b"{\"a\":1}\n{\"a\":2}\n";
106const ROWS_TYPED: &[u8] = b"{\"cellbox\":1,\"value\":[\"record\",[[\"pid\",[\"uint\",\"7\"]]]]}\n\
107{\"cellbox\":1,\"value\":[\"record\",[[\"pid\",[\"uint\",\"9\"]]]]}\n";
108const HEX_PREVIEW: usize = 64;
109/// `vitui` enter sequence (alternate screen, cursor hidden, bracketed paste) and a
110/// line of text; no restore follows.
111const TUI_ABANDON: &[u8] = b"\x1b[?1049h\x1b[?25l\x1b[?2004hp03-tui-abandon\n";

113/// The committed endpoint a parent delegated into `slot`.
114fn granted(context: &ChildContext, slot: GrantSlot) -> Option<Endpoint> {
115    let operation = OperationRef::from_raw(context.operation()).ok()?;
116    let grant = context.grants().find(|g| g.slot == slot && g.kind == 1)?;
117    Some(Endpoint::new(EndpointRef::from_raw(grant.object).ok()?, operation))
118}

120fn read_failure(status: Status) -> i32 {
121    match status {
122        Status::Cancelled | Status::Revoked => CANCELLED,
123        _ => IO,
124    }
125}
126
127fn write_failure(status: Status) -> i32 {
128    match status {
129        Status::BrokenPipe => BROKEN_PIPE,
130        Status::Cancelled | Status::Revoked => CANCELLED,
131        _ => IO,
132    }
133}

135/// Write every byte of `data`; the stream accepts at most 512 per call.
136fn write_all(out: Endpoint, mut data: &[u8]) -> Result<(), i32> {
137    while !data.is_empty() {
138        match out.write(data, FOREVER) {
139            Ok(count) if count > 0 && count <= data.len() => data = &data[count..],
140            Ok(_) => return Err(IO),
141            Err(status) => return Err(write_failure(status)),
142        }
143    }
144    Ok(())
145}

147/// Copy stdin to stdout, mapping each chunk with `map`, until end of stream.
148fn copy(context: &ChildContext, map: fn(&mut [u8])) -> i32 {
149    let (Some(input), Some(output)) = (
150        granted(context, GrantSlot::Stdin),
151        granted(context, GrantSlot::Stdout),
152    ) else {
153        return USAGE;
154    };
155    let mut buffer = [0u8; 512];
156    loop {
157        match input.read(&mut buffer, FOREVER) {
158            Ok(count) => {
159                map(&mut buffer[..count]);
160                if let Err(code) = write_all(output, &buffer[..count]) {
161                    return code;
162                }
163            }
164            Err(Status::Eof) => return OK,
165            Err(status) => return read_failure(status),
166        }
167    }
168}

180/// Write the fixed `bytes` to stdout, exit 0.
181fn emit(context: &ChildContext, bytes: &[u8]) -> i32 {
182    let Some(output) = granted(context, GrantSlot::Stdout) else {
183        return USAGE;
184    };
185    match write_all(output, bytes) {
186        Ok(()) => OK,
187        Err(code) => code,
188    }
189}

191/// A TUI that dies without restoring the terminal. The lease (when this Cell is the
192/// session's foreground reader) is deliberately never released: the input owner's
193/// foreground restoration after exit is what takes it back, as for a killed TUI.
194fn tui_abandon(context: &ChildContext) -> i32 {
195    if let Ok(mut session) = TerminalSession::from_own_context(context) {
196        let _lease = RawInput::enter(&mut session);
197    }
198    emit(context, TUI_ABANDON)
199}

201fn hex(context: &ChildContext) -> i32 {
202    let (Some(input), Some(output)) = (
203        granted(context, GrantSlot::Stdin),
204        granted(context, GrantSlot::Stdout),
205    ) else {
206        return USAGE;
207    };
208    let mut preview = [0u8; HEX_PREVIEW];
209    let mut kept = 0usize;
210    let mut total = 0u64;
211    let mut buffer = [0u8; 512];
212    loop {
213        match input.read(&mut buffer, FOREVER) {
214            Ok(count) => {
215                total += count as u64;
216                let take = count.min(HEX_PREVIEW - kept);
217                preview[kept..kept + take].copy_from_slice(&buffer[..take]);
218                kept += take;
219            }
220            Err(Status::Eof) => break,
221            Err(status) => return read_failure(status),
222        }
223    }
224    // "p03-hex len=" + up to 20 digits + " hex=" + 128 hex digits + "\n"
225    let mut line = [0u8; 12 + 20 + 5 + 2 * HEX_PREVIEW + 1];
226    let mut at = 0usize;
227    let mut push = |bytes: &[u8]| {
228        line[at..at + bytes.len()].copy_from_slice(bytes);
229        at += bytes.len();
230    };
231    push(b"p03-hex len=");
232    let mut digits = [0u8; 20];
233    let mut start = digits.len();
234    let mut value = total;
235    loop {
236        start -= 1;
237        digits[start] = b'0' + (value % 10) as u8;
238        value /= 10;
239        if value == 0 {
240            break;
241        }
242    }
243    push(&digits[start..]);
244    push(b" hex=");
245    for byte in &preview[..kept] {
246        push(&[nibble(byte >> 4), nibble(byte & 15)]);
247    }
248    push(b"\n");
249    match write_all(output, &line[..at]) {
250        Ok(()) => OK,
251        Err(code) => code,
252    }
253}

255fn nibble(value: u8) -> u8 {
256    b"0123456789abcdef"[value as usize]
257}

259fn endless(context: &ChildContext) -> i32 {
260    let Some(output) = granted(context, GrantSlot::Stdout) else {
261        return USAGE;
262    };
263    let block = [b'x'; 512];
264    loop {
265        if let Err(code) = write_all(output, &block) {
266            return code;
267        }
268    }
269}

271/// One 60-byte Ethernet frame between two locally administered unicast MACs with the
272/// IEEE local-experimental EtherType 0x88B5: inert on any segment if it were ever sent.
273const PROBE_FRAME: [u8; 60] = {
274    let mut frame = [0u8; 60];
275    let header = [
276        0x02, 0, 0, 0, 0, 0x01, // destination
277        0x02, 0, 0, 0, 0, 0x02, // source
278        0x88, 0xb5, // EtherType
279    ];
280    let mut i = 0;
281    while i < header.len() {
282        frame[i] = header[i];
283        i += 1;
284    }
285    frame
286};

288/// Send `request` once to the live net service and classify the reply. Nothing is
289/// retried: an admitted request is a probe failure the line reports, not an error to
290/// hide (any socket it created is owner-scoped and reclaimed when this Cell exits).
291fn probe(net: &mut NetRef, request: &NetRequest) -> String {
292    let mut reply = [0u8; IPC_BUF_SIZE];
293    match net.call::<NetRequest, NetResponse>(request, &mut reply) {
294        Ok(NetResponse::Err(NET_ERR_DENIED)) => String::from("denied"),
295        Ok(NetResponse::Err(code)) => format!("other:err-0x{code:02x}"),
296        Ok(NetResponse::CapId(id)) => format!("other:cap-{id}"),
297        Ok(NetResponse::Addr([a, b, c, d])) => format!("other:addr-{a}.{b}.{c}.{d}"),
298        Ok(NetResponse::Ok) => String::from("other:ok"),
299        Ok(_) => String::from("other:unexpected-response"),
300        Err(error) => format!("other:ipc-{error:?}"),
301    }
302}

304fn net_probe(context: &ChildContext) -> i32 {
305    let Some(output) = granted(context, GrantSlot::Stdout) else {
306        return USAGE;
307    };
308    let requests: [(&str, NetRequest); 7] = [
309        ("resolve", NetRequest::Resolve { hostname: "10.0.2.2" }),
310        ("tcp-connect", NetRequest::TcpConnect { addr: [10, 0, 2, 2], port: 9 }),
311        ("udp-create", NetRequest::UdpCreate),
312        ("udp-bind", NetRequest::UdpBind { cap_id: 0, port: 0 }),
313        ("tcp-listen", NetRequest::TcpListen { port: 9 }),
314        ("multicast-join", NetRequest::MulticastJoin { cap_id: 0, group: [239, 255, 0, 1] }),
315        ("l2-send", NetRequest::L2Send { data: &PROBE_FRAME }),
316    ];
317    let mut net = NetRef::new();
318    for (op, request) in &requests {
319        let line = format!("NETPROBE {op} {}\n", probe(&mut net, request));
320        if let Err(code) = write_all(output, line.as_bytes()) {
321            return code;
322        }
323    }
324    match write_all(output, b"NETPROBE DONE\n") {
325        Ok(()) => OK,
326        Err(code) => code,
327    }
328}

330fn config_probe(context: &ChildContext) -> i32 {
331    let Some(output) = granted(context, GrantSlot::Stdout) else {
332        return USAGE;
333    };
334    let mut args = context.argv().skip(1);
335    let (op, key) = match (args.next(), args.next()) {
336        (Some(op), Some(key)) => (op, key),
337        _ => return USAGE,
338    };
339    let value = args.next();
340    let request = match (op, value) {
341        ("get", None) => ConfigRequest::Get(key),
342        ("set", Some(value)) => ConfigRequest::Set { key, value },
343        ("delete", None) => ConfigRequest::Delete(key),
344        _ => return USAGE,
345    };
346    let mut reply = [0u8; IPC_BUF_SIZE];
347    let result = match ConfigRef::new().call::<ConfigRequest, ConfigResponse>(&request, &mut reply) {
348        Ok(ConfigResponse::Ok) => String::from("ok"),
349        Ok(ConfigResponse::Err(CONFIG_ERR_DENIED)) => String::from("denied"),
350        Ok(ConfigResponse::Value(v)) => format!("value:{v}"),
351        Ok(ConfigResponse::NotFound) => String::from("notfound"),
352        Ok(ConfigResponse::Err(code)) => format!("other:err-0x{code:02x}"),
353        Ok(ConfigResponse::Keys(_)) => String::from("other:keys"),
354        Err(error) => format!("other:ipc-{error:?}"),
355    };
356    match write_all(output, format!("CFGPROBE {op} {key} {result}\n").as_bytes()) {
357        Ok(()) => OK,
358        Err(code) => code,
359    }
360}

362fn run() -> i32 {
363    let context = match ChildContext::read_own() {
364        Ok(context) => context,
365        Err(_) => return USAGE,
366    };
367    let mut args = context.argv();
368    match args.next().unwrap_or("") {
369        "cat" => copy(&context, identity),
370        "upper" => copy(&context, upper),
371        "noread" => OK,
372        "exit" => match args.next().and_then(|n| n.parse::<i32>().ok()) {
373            Some(code) => code,
374            None => USAGE,
375        },
376        "gen" => generate(&context),
377        "rows" => emit(&context, ROWS_JSONL),
378        "typed" => emit(&context, ROWS_TYPED),
379        "hex" => hex(&context),
380        "endless" => endless(&context),
381        "net-probe" => net_probe(&context),
382        "config-probe" => config_probe(&context),
383        "tui-abandon" => tui_abandon(&context),
384        _ => USAGE,
385    }
386}
387
388fn cell_main() {
389    sys_exit(run() as u32 as usize);
