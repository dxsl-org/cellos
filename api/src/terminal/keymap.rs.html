2//! Compositor/focus key events -> normalized terminal session input.
3//!
4//! Key codes and modifier bits follow the frozen `cellos-vt1` namespace in
5//! `api::services::session` (named keys at `0x110000+`, characters as their own
6//! code point). The input service turns them into bytes; this module never
7//! produces escape sequences itself.
8
9use api::services::session::{encode_key, key_code, modifiers};
10use ostd::input::{KeyEvent, KeyState, KeySym, Modifiers};
11
12/// One normalized input to submit to the session.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum HostInput {
15    /// Printable committed text (one character; never a control character).
16    Text(char),
17    /// Logical key: `pressed` is 1 for press, 2 for auto-repeat (0 emits no bytes).
18    Key { code: u32, modifiers: u32, pressed: u8 },
19}
20
21/// What a key means while no shell child is running.
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub enum IdleAction {
24    Restart,
25    Close,
26    None,
27}
28
29fn named_code(sym: KeySym) -> Option<u32> {
30    Some(match sym {
31        KeySym::Return => key_code::ENTER,
32        KeySym::Escape => key_code::ESCAPE,
33        KeySym::Backspace => key_code::BACKSPACE,
34        KeySym::Tab => key_code::TAB,
35        KeySym::Delete => key_code::DELETE,
36        KeySym::Insert => key_code::INSERT,
37        KeySym::Up => key_code::UP,
38        KeySym::Down => key_code::DOWN,
39        KeySym::Left => key_code::LEFT,
40        KeySym::Right => key_code::RIGHT,
41        KeySym::Home => key_code::HOME,
42        KeySym::End => key_code::END,
43        KeySym::PageUp => key_code::PAGE_UP,
44        KeySym::PageDown => key_code::PAGE_DOWN,
45        KeySym::F1 => key_code::F1,
46        KeySym::F2 => key_code::F1 + 1,
47        KeySym::F3 => key_code::F1 + 2,
48        KeySym::F4 => key_code::F1 + 3,
49        KeySym::F5 => key_code::F1 + 4,
50        KeySym::F6 => key_code::F1 + 5,
51        KeySym::F7 => key_code::F1 + 6,
52        KeySym::F8 => key_code::F1 + 7,
53        KeySym::F9 => key_code::F1 + 8,
54        KeySym::F10 => key_code::F1 + 9,
55        KeySym::F11 => key_code::F1 + 10,
56        KeySym::F12 => key_code::F12,
57        KeySym::Unknown | KeySym::Printable => return None,
58    })
59}

61/// Map one compositor key event. `None` = nothing to submit (release, Meta chords,
62/// control characters, and combinations the session cannot encode).
63///
64/// Return and Ctrl/Alt chords are one-shot: their auto-repeat is dropped so a held
65/// Ctrl-C cannot keep cancelling and a held Enter cannot flood commands. Text,
66/// Backspace and navigation keys repeat.
67pub fn map_key(ke: &KeyEvent) -> Option<HostInput> {
68    let pressed = match ke.state {
69        KeyState::Released => return None,
70        KeyState::Pressed => 1,
71        KeyState::Repeated => 2,
72    };
73    let m = ke.modifiers;
74    if m.contains(Modifiers::META) {
75        return None;
76    }
77    let (ctrl, alt, shift) = (m.contains(Modifiers::CTRL), m.contains(Modifiers::ALT), m.contains(Modifiers::SHIFT));
78    let chord = ctrl || alt;
79    let mods = u32::from(shift) * modifiers::SHIFT
80        | u32::from(ctrl) * modifiers::CONTROL
81        | u32::from(alt) * modifiers::ALT;
82    let repeated = pressed == 2;
83
84    let input = if let Some(code) = named_code(ke.keysym) {
85        if repeated && (chord || ke.keysym == KeySym::Return) {
86            return None;
87        }
88        HostInput::Key { code, modifiers: mods, pressed }
89    } else {
90        let ch = ke.char()?;
91        if ch.is_control() {
92            return None;
93        }
94        if chord {
95            if repeated {
96                return None;
97            }
98            HostInput::Key { code: ch as u32, modifiers: mods, pressed }
99        } else {
100            HostInput::Text(ch)
101        }
102    };
103    match input {
104        HostInput::Key { code, modifiers, pressed } if encode_key(code, modifiers, pressed, &mut [0; 16]).is_err() => None,
105        other => Some(other),
106    }
107}

109/// Key handling while the child is gone: Enter restarts, Escape closes the app.
110pub fn idle_action(ke: &KeyEvent) -> IdleAction {
111    if ke.state != KeyState::Pressed {
112        return IdleAction::None;
113    }
114    match ke.keysym {
115        KeySym::Return => IdleAction::Restart,
116        KeySym::Escape => IdleAction::Close,
117        _ => IdleAction::None,
118    }
119}

121#[cfg(test)]
122mod tests {

125    fn ev(keysym: KeySym, character: char, mods: u8, state: KeyState) -> KeyEvent {
126        KeyEvent {
127            timestamp_ticks: 0,
128            scancode: 0,
129            keysym,
130            character: if character == '\0' { 0 } else { character as u32 },
131            modifiers: Modifiers(mods),
132            state,
133            _pad: [0; 2],
134        }
135    }
136    const CTRL: u8 = Modifiers::CTRL.0;
137    const ALT: u8 = Modifiers::ALT.0;
138    const SHIFT: u8 = Modifiers::SHIFT.0;
139    const META: u8 = Modifiers::META.0;
140
141    #[test]
142    fn printable_text_is_submitted_as_text_including_repeat() {
143        let a = ev(KeySym::Printable, 'a', 0, KeyState::Pressed);
144        assert_eq!(map_key(&a), Some(HostInput::Text('a')));
145        let shifted = ev(KeySym::Printable, 'A', SHIFT, KeyState::Repeated);
146        assert_eq!(map_key(&shifted), Some(HostInput::Text('A')));
147        let unicode = ev(KeySym::Printable, 'é', 0, KeyState::Pressed);
148        assert_eq!(map_key(&unicode), Some(HostInput::Text('é')));
149        // Lock modifiers do not turn text into a chord.
150        let caps = ev(KeySym::Printable, 'A', Modifiers::CAPS_LOCK.0 | Modifiers::NUM_LOCK.0, KeyState::Pressed);
151        assert_eq!(map_key(&caps), Some(HostInput::Text('A')));
152    }

154    #[test]
155    fn release_meta_and_control_characters_submit_nothing() {
156        assert_eq!(map_key(&ev(KeySym::Printable, 'a', 0, KeyState::Released)), None);
157        assert_eq!(map_key(&ev(KeySym::Return, '\0', 0, KeyState::Released)), None);
158        assert_eq!(map_key(&ev(KeySym::Printable, 'a', META, KeyState::Pressed)), None);
159        assert_eq!(map_key(&ev(KeySym::Printable, '\x03', 0, KeyState::Pressed)), None);
160        assert_eq!(map_key(&ev(KeySym::Unknown, '\0', 0, KeyState::Pressed)), None);
161    }

163    #[test]
164    fn ctrl_c_is_the_exact_logical_key_the_input_service_treats_as_interrupt() {
165        let c = ev(KeySym::Printable, 'c', CTRL, KeyState::Pressed);
166        assert_eq!(map_key(&c), Some(HostInput::Key { code: 0x63, modifiers: 2, pressed: 1 }));
167        // A held Ctrl-C is one-shot.
168        assert_eq!(map_key(&ev(KeySym::Printable, 'c', CTRL, KeyState::Repeated)), None);
169        let d = ev(KeySym::Printable, 'd', CTRL, KeyState::Pressed);
170        assert_eq!(map_key(&d), Some(HostInput::Key { code: 0x64, modifiers: 2, pressed: 1 }));
171    }

173    #[test]
174    fn alt_chords_and_unencodable_ctrl_combinations() {
175        let alt_b = ev(KeySym::Printable, 'b', ALT, KeyState::Pressed);
176        assert_eq!(map_key(&alt_b), Some(HostInput::Key { code: 0x62, modifiers: 4, pressed: 1 }));
177        // Ctrl+1 has no control byte: refused here rather than at the owner.
178        assert_eq!(map_key(&ev(KeySym::Printable, '1', CTRL, KeyState::Pressed)), None);
179        // Ctrl with Enter/Backspace/Escape/Tab is not encodable.
180        assert_eq!(map_key(&ev(KeySym::Return, '\0', CTRL, KeyState::Pressed)), None);
181        assert_eq!(map_key(&ev(KeySym::Backspace, '\0', CTRL, KeyState::Pressed)), None);
182    }

205    #[test]
206    fn modifiers_map_to_session_bits_and_named_repeat_policy() {
207        let up = ev(KeySym::Up, '\0', SHIFT | CTRL | ALT, KeyState::Pressed);
208        assert_eq!(map_key(&up), Some(HostInput::Key { code: 0x110010, modifiers: 7, pressed: 1 }));
209        // Navigation and Backspace repeat; Return and chorded keys do not.
210        assert_eq!(map_key(&ev(KeySym::Backspace, '\0', 0, KeyState::Repeated)), Some(HostInput::Key { code: 0x110003, modifiers: 0, pressed: 2 }));
211        assert_eq!(map_key(&ev(KeySym::Left, '\0', 0, KeyState::Repeated)), Some(HostInput::Key { code: 0x110012, modifiers: 0, pressed: 2 }));
212        assert_eq!(map_key(&ev(KeySym::Return, '\0', 0, KeyState::Repeated)), None);
213        assert_eq!(map_key(&ev(KeySym::Left, '\0', CTRL, KeyState::Repeated)), None);

216    #[test]
217    fn every_mapped_key_is_accepted_by_the_session_validator() {
218        let syms = [
219            KeySym::Return, KeySym::Escape, KeySym::Backspace, KeySym::Tab, KeySym::Delete, KeySym::Insert,
220            KeySym::Up, KeySym::Down, KeySym::Left, KeySym::Right, KeySym::Home, KeySym::End,
221            KeySym::PageUp, KeySym::PageDown, KeySym::F1, KeySym::F2, KeySym::F3, KeySym::F4, KeySym::F5,
222            KeySym::F6, KeySym::F7, KeySym::F8, KeySym::F9, KeySym::F10, KeySym::F11, KeySym::F12,
223        ];
224        for sym in syms {
225            let Some(HostInput::Key { code, modifiers, pressed }) = map_key(&ev(sym, '\0', 0, KeyState::Pressed)) else {
226                panic!("{sym:?} must map");
227            };
228            let mut payload = [0u8; 16];
229            payload[..4].copy_from_slice(&code.to_le_bytes());
230            payload[4..8].copy_from_slice(&modifiers.to_le_bytes());
231            payload[8] = pressed;
232            assert!(api::services::session::validate_event(1, &payload).is_ok(), "{sym:?}");
233        }
234    }
