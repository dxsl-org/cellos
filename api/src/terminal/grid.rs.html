2//! `cellos-vt1` screen model of the GUI Terminal (spec §4 "Terminal profile").
3//!
4//! Cells carry a character (plus at most one combining mark) and the attributes the
5//! renderer draws: foreground/background colour, bold and inverse. The model has
6//! autowrap with a pending-wrap state, tab stops every 8 columns, scroll margins,
7//! primary and alternate screens (`?1049` saves/restores the cursor and leaves the
8//! primary content untouched), cursor visibility and the bracketed-paste flag.
9//!
10//! Width follows `unicode-width` `width()` (ambiguous = 1): combining marks join the
11//! previous cell, wide characters take a lead cell plus a trailing cell, and the cursor
12//! never rests on a trailing half. Every allocation is bounded: rows 1..=256, columns
13//! 1..=512, at most 131,072 cells per screen, scrollback at most 1,024 lines and 1 MiB.
14//!
15//! Line feed also returns the carriage: CellOS sessions have no output line discipline
16//! (no ONLCR), so programs write bare `\n` and expect the next line to start at column 0.
17

129fn checked_dims(cols: usize, rows: usize) -> Result<usize, ResizeError> {
130    if !(1..=MAX_COLS).contains(&cols) || !(1..=MAX_ROWS).contains(&rows) || cols * rows > MAX_CELLS
131    {
132        return Err(ResizeError::OutOfBounds);
133    }
134    Ok(cols * rows)
135}

137fn blank_screen(cells: usize) -> Result<Vec<TerminalCell>, ResizeError> {
138    let mut screen = Vec::new();
139    screen
140        .try_reserve_exact(cells)
141        .map_err(|_| ResizeError::NoMemory)?;
142    screen.resize(cells, TerminalCell::BLANK);
143    Ok(screen)
144}

146/// Copy rows `shift..` of `old` (`old_cols` wide) into a blank `new_cols x new_rows` screen.
147fn relayout(
148    old: &[TerminalCell],
149    old_cols: usize,
150    new_cols: usize,
151    new_rows: usize,
152    shift: usize,
153) -> Result<Vec<TerminalCell>, ResizeError> {
154    let mut screen = blank_screen(new_cols * new_rows)?;
155    let old_rows = old.len() / old_cols;
156    let keep = new_cols.min(old_cols);
157    for (dst, src) in (shift..old_rows).take(new_rows).enumerate() {
158        let row = &mut screen[dst * new_cols..dst * new_cols + keep];
159        row.copy_from_slice(&old[src * old_cols..src * old_cols + keep]);
160        // A wide character whose trailing half was cut off is gone.
161        if keep < old_cols && row[keep - 1].width == Width::Wide {
162            row[keep - 1] = TerminalCell {
163                ch: ' ',
164                mark: None,
165                width: Width::Narrow,
166                ..row[keep - 1]
167            };
168        }
169    }
170    Ok(screen)
171}

173/// Append one screen line to the scrollback (trailing default blanks trimmed), dropping
174/// the oldest lines first so both the line and the byte cap hold.
175fn keep_line(
176    scrollback: &mut VecDeque<Box<[TerminalCell]>>,
177    total: &mut usize,
178    row: &[TerminalCell],
179) {
180    let used = row
181        .iter()
182        .rposition(|cell| *cell != TerminalCell::BLANK)
183        .map_or(0, |last| last + 1);
184    let bytes = used * core::mem::size_of::<TerminalCell>();
185    while !scrollback.is_empty()
186        && (scrollback.len() >= MAX_SCROLLBACK_LINES || *total + bytes > MAX_SCROLLBACK_BYTES)
187    {
188        if let Some(old) = scrollback.pop_front() {
189            *total -= old.len() * core::mem::size_of::<TerminalCell>();
190        }
191    }
192    let mut line = Vec::new();
193    if line.try_reserve_exact(used).is_err() || scrollback.try_reserve(1).is_err() {
194        // Out of memory: the line is not kept; the screen itself is unaffected.
195        return;
196    }
197    line.extend_from_slice(&row[..used]);
198    scrollback.push_back(line.into_boxed_slice());
199    *total += bytes;
200}

300    /// The cursor never rests on the trailing half of a wide character.
301    fn settle(&mut self) {
302        if self.cursor_x > 0
303            && self.cells[self.cursor_y * self.cols + self.cursor_x].width == Width::Trail
304        {
305            self.cursor_x -= 1;
306        }
307    }

309    /// Writing over or erasing either half of a wide character removes the whole character.
310    fn break_wide(&mut self, y: usize, x: usize) {
311        let cols = self.cols;
312        let row = self.row_mut(y);
313        let lead = match row[x].width {
314            Width::Narrow => return,
315            Width::Wide => x,
316            Width::Trail => x.saturating_sub(1),
317        };
318        for cell in row[lead..(lead + 2).min(cols)].iter_mut() {
319            *cell = TerminalCell {
320                ch: ' ',
321                mark: None,
322                width: Width::Narrow,
323                ..*cell
324            };
325        }
326    }

328    /// Fill columns `from..to` of row `y` with erased cells.
329    fn erase(&mut self, y: usize, from: usize, to: usize) {
330        if from >= to {
331            return;
332        }
333        self.break_wide(y, from);
334        self.break_wide(y, to - 1);
335        let blank = self.erase_cell();
336        self.row_mut(y)[from..to].fill(blank);
337    }

339    pub fn print(&mut self, ch: char) {
340        let width = match ch.width() {
341            Some(width) => width,
342            // Control characters never reach the screen.
343            None => return,
344        };
345        if width == 0 {
346            self.combine(ch);
347            return;
348        }
349        if width == 2 && self.cols < 2 {
350            return;
351        }
352        if self.wrap_pending {
353            self.carriage_return();
354            self.index();
355        }
356        if width == 2 && self.cursor_x + 1 >= self.cols {
357            if self.autowrap {
358                self.carriage_return();
359                self.index();
360            } else {
361                self.cursor_x = self.cols - 2;
362            }
363        }
364        let (x, y) = (self.cursor_x, self.cursor_y);
365        self.break_wide(y, x);
366        if width == 2 {
367            self.break_wide(y, x + 1);
368        }
369        let pen = self.pen;
370        let row = self.row_mut(y);
371        if width == 2 {
372            row[x] = TerminalCell {
373                ch,
374                mark: None,
375                pen,
376                width: Width::Wide,
377            };
378            row[x + 1] = TerminalCell {
379                ch: ' ',
380                mark: None,
381                pen,
382                width: Width::Trail,
383            };
384        } else {
385            row[x] = TerminalCell {
386                ch,
387                mark: None,
388                pen,
389                width: Width::Narrow,
390            };
391        }
392        let next = x + width;
393        if next >= self.cols {
394            self.cursor_x = self.cols - 1;
395            self.wrap_pending = self.autowrap;
396        } else {
397            self.cursor_x = next;
398        }
399        self.settle();
400    }

402    /// A zero-width character joins the cell written last; with no such cell on the
403    /// line it is dropped.
404    fn combine(&mut self, mark: char) {
405        let x = if self.wrap_pending {
406            self.cursor_x
407        } else if self.cursor_x > 0 {
408            self.cursor_x - 1
409        } else {
410            return;
411        };
412        let y = self.cursor_y;
413        let row = self.row_mut(y);
414        let x = if row[x].width == Width::Trail {
415            x - 1
416        } else {
417            x
418        };
419        if row[x].mark.is_none() {
420            row[x].mark = Some(mark);
421        }
422    }

435    /// Cursor down one line; at the bottom margin the region scrolls up.
436    fn index(&mut self) {
437        self.wrap_pending = false;
438        if self.cursor_y == self.bottom {
439            self.scroll_up(1);
440        } else if self.cursor_y + 1 < self.rows {
441            self.cursor_y += 1;
442        }
443        self.settle();
444    }

446    pub fn backspace(&mut self) {
447        let snapped_wide = self.wrap_pending
448            && self.cells[self.cursor_y * self.cols + self.cursor_x].width == Width::Wide
449            && self.cursor_x + 2 == self.cols;
450        self.wrap_pending = false;
451        if !snapped_wide && self.cursor_x > 0 {
452            self.cursor_x -= 1;
453        }
454        self.settle();
455    }

522    /// ED 0 (cursor to end), 1 (start to cursor), 2 (all); other modes are ignored.
523    pub fn erase_display(&mut self, mode: u16) {
524        let (x, y) = (self.cursor_x, self.cursor_y);
525        match mode {
526            0 => {
527                self.erase(y, x, self.cols);
528                for row in y + 1..self.rows {
529                    self.erase(row, 0, self.cols);
530                }
531            }
532            1 => {
533                for row in 0..y {
534                    self.erase(row, 0, self.cols);
535                }
536                self.erase(y, 0, x + 1);
537            }
538            2 => {
539                for row in 0..self.rows {
540                    self.erase(row, 0, self.cols);
541                }
542            }
543            _ => return,
544        }
545        self.wrap_pending = false;
546    }

560    /// DECSTBM with 1-based rows (0 = default). Invalid regions are ignored; a valid one
561    /// homes the cursor.
562    pub fn set_scroll_region(&mut self, top: usize, bottom: usize) {
563        let top = top.max(1);
564        let bottom = if bottom == 0 {
565            self.rows
566        } else {
567            bottom.min(self.rows)
568        };
569        if top >= bottom {
570            return;
571        }
572        self.top = top - 1;
573        self.bottom = bottom - 1;
574        self.cursor_x = 0;
575        self.cursor_y = 0;
576        self.moved();
577    }

579    /// SU / LF at the bottom margin: the region moves up `n` lines. Lines leaving the
580    /// top of the full primary screen go to scrollback.
581    pub fn scroll_up(&mut self, n: usize) {
582        let height = self.bottom - self.top + 1;
583        let n = n.min(height);
584        if n == 0 {
585            return;
586        }
587        let cols = self.cols;
588        if self.top == 0 && !self.alternate {
589            for line in 0..n {
590                let start = (self.top + line) * cols;
591                keep_line(
592                    &mut self.scrollback,
593                    &mut self.scrollback_bytes,
594                    &self.cells[start..start + cols],
595                );
596            }
597        }
598        let (start, end) = (self.top * cols, (self.bottom + 1) * cols);
599        self.cells.copy_within(start + n * cols..end, start);
600        let blank = self.erase_cell();
601        self.cells[end - n * cols..end].fill(blank);
602        self.settle();
603    }

605    /// SD: the region moves down `n` lines; blank lines appear at the top margin.
606    pub fn scroll_down(&mut self, n: usize) {
607        let height = self.bottom - self.top + 1;
608        let n = n.min(height);
609        if n == 0 {
610            return;
611        }
612        let cols = self.cols;
613        let (start, end) = (self.top * cols, (self.bottom + 1) * cols);
614        self.cells
615            .copy_within(start..end - n * cols, start + n * cols);
616        let blank = self.erase_cell();
617        self.cells[start..start + n * cols].fill(blank);
618        self.settle();
619    }

625    /// ESC 7 / CSI s: position, rendition and pending wrap of the active screen.
626    pub fn save_cursor(&mut self) {
627        let slot = self.screen_slot();
628        self.saved[slot] = Some(Saved {
629            x: self.cursor_x,
630            y: self.cursor_y,
631            pen: self.pen,
632            wrap_pending: self.wrap_pending,
633        });
634    }

636    /// ESC 8 / CSI u. Without a saved state the cursor homes with the default rendition.
637    pub fn restore_cursor(&mut self) {
638        let saved = self.saved[self.screen_slot()].unwrap_or(Saved {
639            x: 0,
640            y: 0,
641            pen: Pen::DEFAULT,
642            wrap_pending: false,
643        });
644        self.cursor_x = saved.x.min(self.cols - 1);
645        self.cursor_y = saved.y.min(self.rows - 1);
646        self.pen = saved.pen;
647        self.wrap_pending = saved.wrap_pending && self.autowrap && self.cursor_x == self.cols - 1;
648        self.settle();
649    }

651    /// DECSET/DECRST: ?7 autowrap, ?25 cursor visibility, ?1049 alternate screen,
652    /// ?2004 bracketed paste. Other modes are ignored.
653    pub fn set_private_mode(&mut self, mode: u16, on: bool) {
654        match mode {
655            7 => {
656                self.autowrap = on;
657                if !on {
658                    self.wrap_pending = false;
659                }
660            }
661            25 => self.cursor_visible = on,
662            1049 => {
663                if on {
664                    self.enter_alternate();
665                } else {
666                    self.leave_alternate();
667                }
668            }
669            2004 => self.bracketed_paste = on,
670            _ => {}
671        }
672    }

674    /// `?1049h`: save the cursor, switch to a cleared alternate screen. If the alternate
675    /// screen cannot be allocated the primary stays active.
676    fn enter_alternate(&mut self) {
677        if self.alternate {
678            return;
679        }
680        self.save_cursor();
681        let Ok(screen) = blank_screen(self.cols * self.rows) else {
682            return;
683        };
684        self.primary = core::mem::replace(&mut self.cells, screen);
685        self.alternate = true;
686        self.saved[1] = None;
687        self.settle();
688    }

690    /// `?1049l`: back to the untouched primary screen and its saved cursor. A no-op on
691    /// the primary screen, so a soft reset after an ordinary program changes nothing.
692    fn leave_alternate(&mut self) {
693        if !self.alternate {
694            return;
695        }
696        self.cells = core::mem::take(&mut self.primary);
697        self.alternate = false;
698        self.restore_cursor();
699    }

701    /// Back to the profile defaults (a new shell starts, or the shell ended): primary
702    /// screen, visible cursor, autowrap, full margins, default rendition, no paste mode.
703    pub fn reset_modes(&mut self) {
704        self.leave_alternate();
705        self.cursor_visible = true;
706        self.autowrap = true;
707        self.bracketed_paste = false;
708        self.pen = Pen::DEFAULT;
709        self.top = 0;
710        self.bottom = self.rows - 1;
711        self.wrap_pending = false;
712    }
713
714    /// SGR: 0 reset, 1 bold, 22 normal, 7/27 inverse, 30-37/90-97 and 40-47/100-107
715    /// colours, 38/48 `5;n` and `2;r;g;b`, 39/49 defaults. Dim, italic, underline, blink,
716    /// hidden and strike (and their resets) are accepted without effect; unknown codes
717    /// are ignored. A malformed 38/48 ends the sequence.
718    pub fn select_graphic_rendition(&mut self, params: &[u16]) {
719        if params.is_empty() {
720            self.pen = Pen::DEFAULT;
721            return;
722        }
723        let mut i = 0;
724        while i < params.len() {
725            match params[i] {
726                0 => self.pen = Pen::DEFAULT,
727                1 => self.pen.bold = true,
728                22 => self.pen.bold = false,
729                7 => self.pen.inverse = true,
730                27 => self.pen.inverse = false,
731                code @ 30..=37 => self.pen.fg = Color::Indexed((code - 30) as u8),
732                39 => self.pen.fg = Color::Default,
733                code @ 40..=47 => self.pen.bg = Color::Indexed((code - 40) as u8),
734                49 => self.pen.bg = Color::Default,
735                code @ 90..=97 => self.pen.fg = Color::Indexed((code - 90 + 8) as u8),
736                code @ 100..=107 => self.pen.bg = Color::Indexed((code - 100 + 8) as u8),
737                code @ (38 | 48) => {
738                    let Some((color, used)) = extended_color(&params[i + 1..]) else {
739                        return;
740                    };
741                    if code == 38 {
742                        self.pen.fg = color;
743                    } else {
744                        self.pen.bg = color;
745                    }
746                    i += used;
747                }
748                _ => {}
749            }
750            i += 1;
751        }
752    }

754    /// Resize both screens, keeping their content from the top-left. When the cursor
755    /// line would fall off the bottom, the oldest lines leave the top instead (primary
756    /// lines into scrollback). Out-of-bounds sizes and failed allocations change nothing.
757    pub fn resize(&mut self, cols: usize, rows: usize) -> Result<(), ResizeError> {
758        checked_dims(cols, rows)?;
759        if cols == self.cols && rows == self.rows {
760            return Ok(());
761        }
762        let shift = (self.cursor_y + 1).saturating_sub(rows);
763        let cells = relayout(&self.cells, self.cols, cols, rows, shift)?;
764        let (primary, primary_shift) = if self.alternate {
765            let y = self.saved[0].map_or(0, |saved| saved.y);
766            let primary_shift = (y + 1).saturating_sub(rows);
767            (
768                relayout(&self.primary, self.cols, cols, rows, primary_shift)?,
769                primary_shift,
770            )
771        } else {
772            (Vec::new(), shift)
773        };
774        // Committed from here on: no further failure.
775        let old_primary = if self.alternate {
776            &self.primary
777        } else {
778            &self.cells
779        };
780        for line in old_primary.chunks_exact(self.cols).take(primary_shift) {
781            keep_line(&mut self.scrollback, &mut self.scrollback_bytes, line);
782        }
783        self.cells = cells;
784        self.primary = primary;
785        self.cols = cols;
786        self.rows = rows;
787        self.cursor_y -= shift;
788        self.cursor_x = self.cursor_x.min(cols - 1);
789        if let Some(saved) = self.saved[0].as_mut() {
790            saved.y = saved.y.saturating_sub(primary_shift);
791        }
792        self.top = 0;
793        self.bottom = rows - 1;
794        self.wrap_pending = false;
795        self.settle();
796        Ok(())
797    }
798}

800/// `5;n` or `2;r;g;b` after SGR 38/48: the colour and how many parameters it used.
801fn extended_color(rest: &[u16]) -> Option<(Color, usize)> {
802    let byte = |value: u16| u8::try_from(value).ok();
803    match rest {
804        [5, index, ..] => Some((Color::Indexed(byte(*index)?), 2)),
805        [2, r, g, b, ..] => Some((Color::Rgb(byte(*r)?, byte(*g)?, byte(*b)?), 4)),
806        _ => None,
807    }
808}
