65    pub fn new(width: u32, height: u32) -> Self {
66        let cell_w = 9.0;
67        let cell_h = 18.0;
68        let (cols, rows) = grid_dims(width, height, cell_w, cell_h);
69
70        Self {
71            session: TerminalSession::new(cols, rows),
72            width,
73            height,
74            cell_w,
75            cell_h,
76        }
77    }

79    pub fn resize(&mut self, width: u32, height: u32) {
80        self.width = width;
81        self.height = height;
82        let (cols, rows) = grid_dims(width, height, self.cell_w, self.cell_h);
83        self.session.resize(cols, rows);
84    }

86    pub fn render(&self, surf: &mut ViSurface, fonts: &mut TextFonts) {
87        let sw = self.width;
88        let sh = self.height;
89        let surf_w = surf.width();
90        let surf_h = surf.height();
91        let stride = surf.stride();
92        let grid = &self.session.grid;
93        fill_rect(surf, 0, 0, sw, sh, DEFAULT_BG);
94
95        // Pass 1: cell backgrounds that differ from the window background. All fills come
96        // before any glyph so a glyph's 2 px clip overhang is never painted over.
97        for (col_x, row_y, cell, span) in self.visible_cells() {
98            let (_, bg) = cell_colors(cell);
99            if bg != DEFAULT_BG {
100                fill_rect(surf, col_x as i32, row_y as i32, (self.cell_w * span as f32) as u32, self.cell_h as u32, bg);
101            }
102        }
103
104        // Pass 2: glyphs (character plus its combining mark), bold in the bold face.
105        for (col_x, row_y, cell, span) in self.visible_cells() {
106            if cell.ch == ' ' && cell.mark.is_none() {
107                continue;
108            }
109            let mut buf = [0u8; 8];
110            let len = cell.ch.encode_utf8(&mut buf).len();
111            let len = len + cell.mark.map_or(0, |mark| mark.encode_utf8(&mut buf[len..]).len());
112            let text = core::str::from_utf8(&buf[..len]).unwrap_or("\u{fffd}");
113            let (fg, _) = cell_colors(cell);
114            let face = if cell.pen.bold { FontFace::MonoBold } else { FontFace::MonoRegular };
115            let clip = (col_x as i32, row_y as i32, (col_x + self.cell_w * span as f32) as i32 + 2, (row_y + self.cell_h) as i32);
116            fonts.draw_text(
117                surf.pixels_mut(),
118                surf_w,
119                surf_h,
120                stride,
121                col_x,
122                row_y,
123                text,
124                face,
125                FONT_PX,
126                [fg.b(), fg.g(), fg.r(), fg.a()],
127                clip,
128            );
129        }
130
131        // Cursor block, unless the program hid it (?25l).
132        if grid.cursor_visible() {
133            let cur_x = PAD_X + grid.cursor_x as f32 * self.cell_w;
134            let cur_y = PAD_Y + grid.cursor_y as f32 * self.cell_h;
135            if cur_x + self.cell_w <= sw as f32 && cur_y + self.cell_h <= sh as f32 {
136                fill_rect(surf, cur_x as i32, cur_y as i32 + 2, self.cell_w as u32, (self.cell_h - 4.0) as u32, Color::rgb(0, 200, 255).with_alpha(160));
137            }
138        }
139        surf.damage_all();
140    }

142    /// Every cell that fits the window and draws something itself (no wide-character
143    /// trailing halves), with its pixel origin and column span.
144    fn visible_cells(&self) -> impl Iterator<Item = (f32, f32, &TerminalCell, usize)> + '_ {
145        let grid = &self.session.grid;
146        let (sw, sh) = (self.width as f32, self.height as f32);
147        let (cell_w, cell_h) = (self.cell_w, self.cell_h);
148        (0..grid.rows)
149            .map(move |r| (r, PAD_Y + r as f32 * cell_h))
150            .take_while(move |&(_, row_y)| row_y + cell_h <= sh)
151            .flat_map(move |(r, row_y)| {
152                (0..grid.cols)
153                    .map(move |c| (c, PAD_X + c as f32 * cell_w))
154                    .take_while(move |&(_, col_x)| col_x + cell_w <= sw)
155                    .filter_map(move |(c, col_x)| {
156                        let cell = grid.cell_at(c, r)?;
157                        let span = match cell.width {
158                            Width::Trail => return None,
159                            Width::Wide => 2,
160                            Width::Narrow => 1,
161                        };
162                        Some((col_x, row_y, cell, span))
163                    })
164            })
165    }
166}

168/// Default text and window colours.
169const DEFAULT_FG: Color = Color::rgb(220, 225, 235);
170const DEFAULT_BG: Color = Color::rgb(18, 20, 26);
171const PAD_X: f32 = 8.0;
172const PAD_Y: f32 = 6.0;
173const FONT_PX: f32 = 14.0;
174
175/// Grid size for a window: the whole cells that fit right of `PAD_X` and below `PAD_Y`,
176/// where the renderer places cell (0, 0), so every cell of the grid is drawn in full and
177/// the session is told exactly the drawn size. Clamped to what the session accepts; only
178/// a window below the session minimum (40x12 cells) loses its trailing cells.
179pub fn grid_dims(width: u32, height: u32, cell_w: f32, cell_h: f32) -> (usize, usize) {
180    let cols = ((width as f32 - PAD_X).max(0.0) / cell_w) as usize;
181    let rows = ((height as f32 - PAD_Y).max(0.0) / cell_h) as usize;
182    (cols.clamp(MIN_COLS, MAX_COLS), rows.clamp(MIN_ROWS, MAX_ROWS))
183}

185/// The 16 ANSI colours (SGR 30-37 / 90-97 and indexes 0-15).
186const ANSI: [(u8, u8, u8); 16] = [
187    (0, 0, 0),
188    (205, 49, 49),
189    (13, 188, 121),
190    (229, 229, 16),
191    (36, 114, 200),
192    (188, 63, 188),
193    (17, 168, 205),
194    (229, 229, 229),
195    (102, 102, 102),
196    (241, 76, 76),
197    (35, 209, 139),
198    (245, 245, 67),
199    (59, 142, 234),
200    (214, 112, 214),
201    (41, 184, 219),
202    (255, 255, 255),
203];

205/// Pixel colour of a cell colour: ANSI 16, the 6x6x6 cube (16-231), the grey ramp
206/// (232-255) or RGB.
207fn resolve(color: GridColor, default: Color) -> Color {
208    const LEVELS: [u8; 6] = [0, 95, 135, 175, 215, 255];
209    match color {
210        GridColor::Default => default,
211        GridColor::Rgb(r, g, b) => Color::rgb(r, g, b),
212        GridColor::Indexed(n @ 0..=15) => {
213            let (r, g, b) = ANSI[n as usize];
214            Color::rgb(r, g, b)
215        }
216        GridColor::Indexed(n @ 16..=231) => {
217            let i = (n - 16) as usize;
218            Color::rgb(LEVELS[i / 36], LEVELS[i / 6 % 6], LEVELS[i % 6])
219        }
220        GridColor::Indexed(n) => {
221            let v = 8 + 10 * (n - 232);
222            Color::rgb(v, v, v)
223        }
224    }
225}
226
227/// Foreground and background pixels of a cell; inverse swaps the resolved pair.
228fn cell_colors(cell: &TerminalCell) -> (Color, Color) {
229    let fg = resolve(cell.pen.fg, DEFAULT_FG);
230    let bg = resolve(cell.pen.bg, DEFAULT_BG);
231    if cell.pen.inverse {
232        (bg, fg)
233    } else {
234        (fg, bg)
235    }
236}

238/// The Terminal window (first open and the re-create after a close inside a live Cell).
239fn open_window(comp_tid: usize, width: u32, height: u32) -> ViSurface {
240    let surf = ViSurface::create(comp_tid, width, height, PixelFormat::Bgra8888)
241        .expect("create terminal surface");
242    let _ = surf.set_title("Terminal — CellOS Shell");
243    surf.move_to(140, 100);
244    surf
245}

247fn cell_main() {
248    #[cfg(target_os = "none")]
249    init_custom_heap();
250
251    ostd::io::println("[terminal] CellOS GUI Terminal initializing...");
252    let comp_tid = ostd::display::wait_for_compositor();
253
254    let (mut screen_w, mut screen_h) = sys_get_resolution();
255    if screen_w == 0 || screen_h == 0 || screen_w > 4096 || screen_h > 4096 {
256        screen_w = 1280;
257        screen_h = 800;
258    }
259
260    let win_w = 720.min(screen_w);
261    let win_h = 480.min(screen_h - 36);
262
263    let mut surf = open_window(comp_tid, win_w, win_h);
264
265    let mut app = TerminalApp::new(win_w, win_h);
266    let mut fonts = TextFonts::new();
267
268    while !ostd::input::request_focus() {
269        sys_yield();
270    }
271
272    // The shell's own prompt and output are the first thing on screen.
273    app.session.start();
274    app.render(&mut surf, &mut fonts);
275    ostd::io::println("[terminal] Terminal surface ready. Entering event loop.");
276
277    loop {
278        let mut need_render = false;
279        let mut busy = false;
280
281        for ev in ostd::input::poll_events(16) {
282            busy = true;
283            match ev {
284                InputEvent::MouseButton { button: MouseButton::Left, state: KeyState::Pressed } => {
285                    let _ = ostd::input::request_focus();
286                }
287                InputEvent::Key(ke) => {
288                    need_render |= app.session.on_key(&ke);
289                }
290                _ => {}
291            }
292        }
293
294        let mut close_serial = None;
295        for ev in poll_surface_events(8) {
296            match ev {
297                SurfaceEvent::Configure(configure) => {
298                    if configure.cap == surf.cap() && surf.apply_configure(configure).is_ok() {
299                        app.resize(surf.width(), surf.height());
300                        need_render = true;
301                        busy = true;
302                    }
303                }
304                SurfaceEvent::CloseRequest(request) if request.cap == surf.cap() => {
305                    close_serial = Some(request.serial);
306                }
307                _ => {}
308            }
309        }
310
311        if let Some(serial) = close_serial {
312            let _ = surf.respond_close(serial, true);
313            if !app.session.running() {
314                ostd::io::println("[terminal] Window closed without a shell; exiting.");
315                break;
316            }
317            // Window close inside this live Cell: release the frontend (the shell keeps a
318            // Detached session), destroy the window, then re-create it at once and resume
319            // the same shell on a fresh frontend. The re-open is automatic: a user-facing
320            // re-open of a running Terminal (desktop single-instance) does not exist.
321            let released = app.session.detach();
322            surf.destroy();
323            match released {
324                Ok(()) => ostd::io::println("[terminal] window closed; frontend released, shell detached"),
325                Err(status) => ostd::io::println(&alloc::format!("[terminal] window closed; frontend release failed: {:?}", status)),
326            }
327            surf = open_window(comp_tid, win_w, win_h);
328            while !ostd::input::request_focus() {
329                sys_yield();
330            }
331            app.width = surf.width();
332            app.height = surf.height();
333            let (cols, rows) = grid_dims(app.width, app.height, app.cell_w, app.cell_h);
334            match app.session.reattach(cols, rows) {
335                Ok(()) => ostd::io::println("[terminal] window re-created; frontend re-attached to the same shell"),
336                Err(status) => ostd::io::println(&alloc::format!("[terminal] window re-created; re-attach failed: {:?}", status)),
337            }
338            need_render = true;
339            busy = true;
340        }
341
342        if app.session.pump() {
343            need_render = true;
344            busy = true;
345        }
346
347        if app.session.should_close {
348            ostd::io::println("[terminal] Exiting terminal application.");
349            break;
350        }
351
352        if need_render {
353            app.render(&mut surf, &mut fonts);
354        }
355
356        if busy {
357            sys_yield();
358        } else {
359            // Idle: sleep one tick instead of spinning on the session RPCs.
360            let _ = sys_set_timer(1);
361        }
362    }
363
364    app.session.shutdown();
365}

371    #[test]
372    fn terminal_app_sizes_the_grid_from_the_window_and_tracks_resize() {
373        let mut app = TerminalApp::new(720, 480);
374        assert_eq!((app.session.grid.cols, app.session.grid.rows), (79, 26));
375        assert!(!app.session.should_close);
376
377        app.resize(900, 600);
378        assert_eq!((app.session.grid.cols, app.session.grid.rows), (99, 33));
379        assert_eq!((app.width, app.height), (900, 600));
380        assert_eq!(app.visible_cells().count(), 99 * 33);
381
382        // Tiny windows never shrink the grid below the usable minimum.
383        app.resize(10, 10);
384        assert!(app.session.grid.cols >= 40);
385        assert!(app.session.grid.rows >= 12);
386    }

389    fn grid_dims_follow_the_window_minus_padding_and_stay_inside_the_session_limits() {
390        assert_eq!(grid_dims(720, 480, 9.0, 18.0), (79, 26));
391        assert_eq!(grid_dims(100, 100, 9.0, 18.0), (MIN_COLS, MIN_ROWS));
392        assert_eq!(grid_dims(4096, 4096, 9.0, 18.0), (454, 227));
393        assert_eq!(grid_dims(60_000, 60_000, 9.0, 18.0), (MAX_COLS, MAX_ROWS));
394        let (cols, rows) = grid_dims(60_000, 60_000, 9.0, 18.0);
395        assert!(api::services::session::validate_dimensions(rows as u16, cols as u16).is_ok());
396    }

398    #[test]
399    fn every_grid_cell_is_drawn_in_full_and_one_more_would_not_fit() {
400        // Exact fits (728 = 8 + 80 * 9, 474 = 6 + 26 * 18), one pixel short of them, and
401        // ordinary sizes.
402        for (w, h) in [(720, 480), (727, 473), (728, 474), (729, 475), (800, 600), (900, 600), (1280, 764), (4096, 4096)] {
403            let mut app = TerminalApp::new(w, h);
404            let (cols, rows) = (app.session.grid.cols, app.session.grid.rows);
405            let (cell_w, cell_h) = (app.cell_w, app.cell_h);
406            assert!(PAD_X + cols as f32 * cell_w <= w as f32, "{w}x{h}: {cols} columns overflow");
407            assert!(PAD_X + (cols + 1) as f32 * cell_w > w as f32, "{w}x{h}: column {} would fit", cols + 1);
408            assert!(PAD_Y + rows as f32 * cell_h <= h as f32, "{w}x{h}: {rows} rows overflow");
409            assert!(PAD_Y + (rows + 1) as f32 * cell_h > h as f32, "{w}x{h}: row {} would fit", rows + 1);
410            // The renderer draws every cell the grid (and so the session) has, the last
411            // column and row included.
412            assert_eq!(app.visible_cells().count(), cols * rows, "{w}x{h}");
413            let last = (PAD_X + (cols - 1) as f32 * cell_w, PAD_Y + (rows - 1) as f32 * cell_h);
414            assert_eq!(app.visible_cells().last().map(|(x, y, _, _)| (x, y)), Some(last), "{w}x{h}");
415            // The resize path sizes the grid with the same function.
416            app.resize(w + 1, h + 1);
417            app.resize(w, h);
418            assert_eq!((app.session.grid.cols, app.session.grid.rows), (cols, rows), "{w}x{h} after resize");
419        }
420        assert_eq!(grid_dims(727, 473, 9.0, 18.0), (79, 25));
421        assert_eq!(grid_dims(728, 474, 9.0, 18.0), (80, 26));
422    }

440    #[test]
441    fn visible_cells_skip_trailing_halves_and_stop_at_the_window_edge() {
442        let mut app = TerminalApp::new(720, 480);
443        app.session.grid.write_str("a\u{4e16}b");
444        let row0: std::vec::Vec<_> = app.visible_cells().take(3).map(|(x, y, cell, span)| (x, y, cell.ch, span)).collect();
445        assert_eq!(row0, [(8.0, 6.0, 'a', 1), (17.0, 6.0, '\u{4e16}', 2), (35.0, 6.0, 'b', 1)]);
446        // 720 px fit 79 whole columns after the 8 px pad, the grid's own width; 26 rows;
447        // minus one trailing half.
448        assert_eq!(app.visible_cells().count(), 79 * 26 - 1);
