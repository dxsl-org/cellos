2//! PCIe function record (`QueryPcieFunction = 434`) and MSI/MSI-X allocation
3//! result (`AllocPcieIrq = 435`); ADR-0031, CP1 §3–§4.
4//!
5//! [`ViPcieFunctionV1`] is a class K record (`docs/abi-boundary-records.md`):
6//! syscall 434 with an exact out length of [`PCIE_FUNCTION_RECORD_LEN`], version
7//! exactly 1, little-endian in declared field order, every reserved byte and
8//! every unknown flag bit zero. Producer: the kernel. Consumer: the Driver Cell
9//! that owns the function (ownership comes from 418/424).
10//!
11//! Layout (`off`, size): 0 `version` u16 · 2 `cap_count` u16 · 4 `flags` u32 ·
12//! 8 `bdf` u32 · 12 `vendor_id` u16 · 14 `device_id` u16 ·
13//! 16 `subsystem_vendor_id` u16 · 18 `subsystem_id` u16 · 20 `class` ·
14//! 21 `subclass` · 22 `prog_if` · 23 `revision` · 24 reserved [8] ·
15//! 32 `bars[6]` 24 bytes each (`base` u64, `len` u64, `kind` u32, `bar_flags` u32) ·
16//! 176 `msix` (`table_size` u16, `table_bar` u8, `pba_bar` u8, `table_offset` u32,
17//! `pba_offset` u32, reserved u32) · 192 `msi` (`control` u16, reserved u16,
18//! reserved u32) · 200 `caps[16]` 24 bytes each (`id`, `offset`, `len`,
19//! reserved, `bytes` [20]).
20
21/// The only record version 434 writes and a v1 consumer accepts.
22pub const PCIE_FUNCTION_VERSION: u16 = 1;
23/// Exact `out_len` syscall 434 accepts and bytes it writes.
24pub const PCIE_FUNCTION_RECORD_LEN: usize = 584;
25/// BAR slots in a type-0 header.
26pub const PCIE_FUNCTION_BARS: usize = 6;
27/// Most standard capabilities one record carries.
28pub const PCIE_FUNCTION_CAPS_MAX: usize = 16;
29/// Capability bytes copied per entry (covers the 20-byte virtio notify cap).
30pub const PCIE_CAP_BYTES: usize = 20;
31
32/// `flags` bit: the function has an MSI-X capability.
33pub const FLAG_MSIX_PRESENT: u32 = 1 << 0;
34/// `flags` bit: the function has an MSI capability.
35pub const FLAG_MSI_PRESENT: u32 = 1 << 1;
36/// `flags` bit: FADT IAPC_BOOT_ARCH bit 3 says MSI must not be used.
37pub const FLAG_MSI_FORBIDDEN_BY_FIRMWARE: u32 = 1 << 2;
38/// `flags` bit: the kernel's VT-d DMA translation is active. A virtio device
39/// that does not negotiate `VIRTIO_F_ACCESS_PLATFORM` bypasses translation, so
40/// a virtio driver must refuse such a device when this bit is set (CP2-04a
41/// amendment, 2026-11-04). It is not proof that this function is confined: on
42/// a host with several DMA remapping units it can be set for a function under
43/// a unit the kernel has not programmed yet (ADR-0031 phase 12).
44pub const FLAG_DMA_REMAPPED: u32 = 1 << 3;
45/// Every `flags` bit v1 defines.
46pub const FLAGS_KNOWN: u32 =
47    FLAG_MSIX_PRESENT | FLAG_MSI_PRESENT | FLAG_MSI_FORBIDDEN_BY_FIRMWARE | FLAG_DMA_REMAPPED;
48
49/// BAR `kind`: absent, or the upper half of a 64-bit BAR.
50pub const BAR_KIND_ABSENT: u32 = 0;
51/// BAR `kind`: 32-bit memory.
52pub const BAR_KIND_MEM32: u32 = 1;
53/// BAR `kind`: 64-bit memory.
54pub const BAR_KIND_MEM64: u32 = 2;
55/// BAR `kind`: I/O ports.
56pub const BAR_KIND_IO: u32 = 3;
57/// `bar_flags` bit: prefetchable memory.
58pub const BAR_FLAG_PREFETCHABLE: u32 = 1 << 0;
59/// `bar_flags` bit: `RequestMmio` maps this BAR without the MSI-X table/PBA pages.
60pub const BAR_FLAG_MSIX_CARVED: u32 = 1 << 1;
61const BAR_FLAGS_KNOWN: u32 = BAR_FLAG_PREFETCHABLE | BAR_FLAG_MSIX_CARVED;
62
63/// `AllocPcieIrq` mode bit: MSI-X may be used.
64pub const IRQ_MODE_ALLOW_MSIX: usize = 1 << 0;
65/// `AllocPcieIrq` mode bit: MSI may be used.
66pub const IRQ_MODE_ALLOW_MSI: usize = 1 << 1;
67/// Largest `want` 435 accepts.
68pub const IRQ_WANT_MAX: usize = 32;
69/// `AllocPcieIrq` result kind: MSI-X.
70pub const IRQ_KIND_MSIX: u8 = 1;
71/// `AllocPcieIrq` result kind: MSI.
72pub const IRQ_KIND_MSI: u8 = 2;
73
74/// 434/435 error sentinels. Generic failure and `PermissionDenied` are
75/// `usize::MAX`; the rest are typed like 432/433.
76pub const ERR_PERMISSION_DENIED: usize = usize::MAX;
77/// 434: no such function.
78pub const ERR_NOT_FOUND: usize = usize::MAX - 2;
79/// Malformed arguments or buffer.
80pub const ERR_INVALID_INPUT: usize = usize::MAX - 3;
81/// 435: no allowed capability, firmware forbids MSI, or not x86.
82pub const ERR_NOT_SUPPORTED: usize = usize::MAX - 5;
83/// 435: vector space exhausted.
84pub const ERR_NO_VECTORS: usize = usize::MAX - 6;

106/// One standard capability (config offsets 0x40..0xFF), in walk order.
107#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
108pub struct PcieCap {
109    pub id: u8,
110    pub offset: u8,
111    /// Valid bytes in `bytes`; the rest are zero.
112    pub len: u8,
113    pub bytes: [u8; PCIE_CAP_BYTES],
114}

116/// Decoded v1 record.
117#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
118pub struct ViPcieFunctionV1 {
119    pub cap_count: u16,
120    pub flags: u32,
121    pub bdf: u32,
122    pub vendor_id: u16,
123    pub device_id: u16,
124    pub subsystem_vendor_id: u16,
125    pub subsystem_id: u16,
126    pub class: u8,
127    pub subclass: u8,
128    pub prog_if: u8,
129    pub revision: u8,
130    pub bars: [PcieBar; PCIE_FUNCTION_BARS],
131    pub msix: PcieMsix,
132    /// MSI Message Control; zero unless [`FLAG_MSI_PRESENT`].
133    pub msi_control: u16,
134    pub caps: [PcieCap; PCIE_FUNCTION_CAPS_MAX],
135}
136
137const BARS_OFF: usize = 32;
138const BAR_LEN: usize = 24;
139const MSIX_OFF: usize = 176;
140const MSI_OFF: usize = 192;
141const CAPS_OFF: usize = 200;
142const CAP_LEN: usize = 24;
143const _: () = assert!(CAPS_OFF + PCIE_FUNCTION_CAPS_MAX * CAP_LEN == PCIE_FUNCTION_RECORD_LEN);
144const _: () = assert!(BARS_OFF + PCIE_FUNCTION_BARS * BAR_LEN == MSIX_OFF);
145
146fn put(out: &mut [u8], off: usize, bytes: &[u8]) {
147    out[off..off + bytes.len()].copy_from_slice(bytes);
148}
149
150fn u16_at(b: &[u8], off: usize) -> u16 {
151    u16::from_le_bytes([b[off], b[off + 1]])
152}
153
154fn u32_at(b: &[u8], off: usize) -> u32 {
155    u32::from_le_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
156}
157
158fn u64_at(b: &[u8], off: usize) -> u64 {
159    (u32_at(b, off) as u64) | ((u32_at(b, off + 4) as u64) << 32)
160}

162impl ViPcieFunctionV1 {
163    /// Wire bytes. Entries past `cap_count` and cap bytes past `len` are written
164    /// as zero whatever the struct holds.
165    pub fn encode(&self) -> [u8; PCIE_FUNCTION_RECORD_LEN] {
166        let mut out = [0u8; PCIE_FUNCTION_RECORD_LEN];
167        put(&mut out, 0, &PCIE_FUNCTION_VERSION.to_le_bytes());
168        put(&mut out, 2, &self.cap_count.to_le_bytes());
169        put(&mut out, 4, &self.flags.to_le_bytes());
170        put(&mut out, 8, &self.bdf.to_le_bytes());
171        put(&mut out, 12, &self.vendor_id.to_le_bytes());
172        put(&mut out, 14, &self.device_id.to_le_bytes());
173        put(&mut out, 16, &self.subsystem_vendor_id.to_le_bytes());
174        put(&mut out, 18, &self.subsystem_id.to_le_bytes());
175        put(
176            &mut out,
177            20,
178            &[self.class, self.subclass, self.prog_if, self.revision],
179        );
180        for (i, bar) in self.bars.iter().enumerate() {
181            let at = BARS_OFF + i * BAR_LEN;
182            put(&mut out, at, &bar.base.to_le_bytes());
183            put(&mut out, at + 8, &bar.len.to_le_bytes());
184            put(&mut out, at + 16, &bar.kind.to_le_bytes());
185            put(&mut out, at + 20, &bar.bar_flags.to_le_bytes());
186        }
187        let m = &self.msix;
188        put(&mut out, MSIX_OFF, &m.table_size.to_le_bytes());
189        put(&mut out, MSIX_OFF + 2, &[m.table_bar, m.pba_bar]);
190        put(&mut out, MSIX_OFF + 4, &m.table_offset.to_le_bytes());
191        put(&mut out, MSIX_OFF + 8, &m.pba_offset.to_le_bytes());
192        put(&mut out, MSI_OFF, &self.msi_control.to_le_bytes());
193        let count = (self.cap_count as usize).min(PCIE_FUNCTION_CAPS_MAX);
194        for (i, cap) in self.caps[..count].iter().enumerate() {
195            let at = CAPS_OFF + i * CAP_LEN;
196            let len = (cap.len as usize).min(PCIE_CAP_BYTES);
197            put(&mut out, at, &[cap.id, cap.offset, len as u8]);
198            put(&mut out, at + 4, &cap.bytes[..len]);
199        }
200        out
201    }
202
203    /// Strict v1 decode: exact length, version 1, known flag bits, a capability
204    /// count within bounds, and every reserved or unused byte zero.
205    pub fn decode(bytes: &[u8]) -> Option<Self> {
206        if bytes.len() != PCIE_FUNCTION_RECORD_LEN || u16_at(bytes, 0) != PCIE_FUNCTION_VERSION {
207            return None;
208        }
209        let mut rec = Self {
210            cap_count: u16_at(bytes, 2),
211            flags: u32_at(bytes, 4),
212            bdf: u32_at(bytes, 8),
213            vendor_id: u16_at(bytes, 12),
214            device_id: u16_at(bytes, 14),
215            subsystem_vendor_id: u16_at(bytes, 16),
216            subsystem_id: u16_at(bytes, 18),
217            class: bytes[20],
218            subclass: bytes[21],
219            prog_if: bytes[22],
220            revision: bytes[23],
221            msix: PcieMsix {
222                table_size: u16_at(bytes, MSIX_OFF),
223                table_bar: bytes[MSIX_OFF + 2],
224                pba_bar: bytes[MSIX_OFF + 3],
225                table_offset: u32_at(bytes, MSIX_OFF + 4),
226                pba_offset: u32_at(bytes, MSIX_OFF + 8),
227            },
228            msi_control: u16_at(bytes, MSI_OFF),
229            ..Self::default()
230        };
231        if rec.flags & !FLAGS_KNOWN != 0
232            || rec.bdf > 0xFFFF
233            || rec.cap_count as usize > PCIE_FUNCTION_CAPS_MAX
234        {
235            return None;
236        }
237        for (i, bar) in rec.bars.iter_mut().enumerate() {
238            let at = BARS_OFF + i * BAR_LEN;
239            *bar = PcieBar {
240                base: u64_at(bytes, at),
241                len: u64_at(bytes, at + 8),
242                kind: u32_at(bytes, at + 16),
243                bar_flags: u32_at(bytes, at + 20),
244            };
245            if bar.kind > BAR_KIND_IO || bar.bar_flags & !BAR_FLAGS_KNOWN != 0 {
246                return None;
247            }
248        }
249        for (i, cap) in rec.caps.iter_mut().enumerate() {
250            let at = CAPS_OFF + i * CAP_LEN;
251            let entry = &bytes[at..at + CAP_LEN];
252            let len = entry[2] as usize;
253            let in_use = i < rec.cap_count as usize;
254            if entry[3] != 0
255                || len > PCIE_CAP_BYTES
256                || (!in_use && entry.iter().any(|&b| b != 0))
257                || entry[4 + len..].iter().any(|&b| b != 0)
258            {
259                return None;
260            }
261            cap.id = entry[0];
262            cap.offset = entry[1];
263            cap.len = len as u8;
264            cap.bytes[..len].copy_from_slice(&entry[4..4 + len]);
265        }
266        let reserved_zero = bytes[24..BARS_OFF].iter().all(|&b| b == 0)
267            && bytes[MSIX_OFF + 12..MSI_OFF].iter().all(|&b| b == 0)
268            && bytes[MSI_OFF + 2..CAPS_OFF].iter().all(|&b| b == 0);
269        reserved_zero.then_some(rec)
270    }
271
272    /// The standard capability with `id`, first in walk order.
273    pub fn cap(&self, id: u8) -> Option<&PcieCap> {
274        self.caps[..(self.cap_count as usize).min(PCIE_FUNCTION_CAPS_MAX)]
275            .iter()
276            .find(|c| c.id == id)
277    }
278}

280/// Decoded `AllocPcieIrq` success value: `first_irq | count << 8 | kind << 16`.
281#[derive(Clone, Copy, Debug, PartialEq, Eq)]
282pub struct PcieIrqAlloc {
283    /// First logical IRQ; wait with `WaitIrq(first_irq + i, 0)` or
284    /// `WaitCompletion(IRQ)`, which covers every line of the allocation.
285    pub first_irq: u8,
286    pub count: u8,
287    /// [`IRQ_KIND_MSIX`] or [`IRQ_KIND_MSI`].
288    pub kind: u8,
289}
290
291impl PcieIrqAlloc {
292    pub const fn pack(self) -> usize {
293        self.first_irq as usize | (self.count as usize) << 8 | (self.kind as usize) << 16
294    }
295
296    pub const fn unpack(raw: usize) -> Option<Self> {
297        let kind = ((raw >> 16) & 0xFF) as u8;
298        let count = ((raw >> 8) & 0xFF) as u8;
299        if raw >> 24 != 0 || count == 0 || (kind != IRQ_KIND_MSIX && kind != IRQ_KIND_MSI) {
300            return None;
301        }
302        Some(Self {
303            first_irq: raw as u8,
304            count,
305            kind,
306        })
307    }
308}
