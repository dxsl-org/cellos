2//! Cell incarnation retirement verdict ABI (`QueryRetirement = 431`).
3//! The kernel writes one [`ViRetirementV1`] per retired incarnation root, after
4//! quiescence (`NotifyOnExit` still fires earlier, at `exit_task`). Class K record
5//! (`docs/abi-boundary-records.md`): syscall 431 plus exact length, version exactly
6//! 1, reserved fields zero, undefined `state` values and `reasons` bits refused,
7//! little-endian in declared field order. No record (unknown or evicted root) must
8//! be treated as [`RetirementState::Unproven`].
9
10/// Exact byte length of [`ViRetirementV1`].
11pub const RETIREMENT_RECORD_LEN: usize = 96;
12/// The only record version this layout writes and accepts.
13pub const RETIREMENT_RECORD_VERSION: u16 = 1;
14
15/// `Exit` code trusted init uses when boot cannot continue (Phase 13 A5).
16///
17/// Init exits with this code, never a plain return, on every boot-stop path
18/// (platform wait failure, VFS bootstrap failure). The kernel's init-death
19/// policy keys on it: init's root retiring with this code records
20/// `[init-death] … action=halt` and halts, so a boot that cannot succeed does
21/// not reset into the same failure forever. Every other init death (any other
22/// exit code, including `0` from the `ostd` startup return and `1` from the
23/// `ostd` panic handler, a fault or a watchdog kill) records `action=reset` and
24/// resets the machine. Ordinary Cells carry no meaning in this value.
25pub const INIT_BOOT_STOP: usize = 0xB007_5709;
26
27/// Exit reason of an incarnation the supervisor retired after a committed
28/// hot-swap (`KillCell` → `kernel/src/cell/hotswap.rs` `exit_task_internal`).
29///
30/// The kernel records it on the old provider's death; trusted init keys swap
31/// adoption on it (a death with this reason whose registry row names a live,
32/// different provider is a swap, not a crash). A `u32` because `KillCell`
33/// carries a 32-bit exit code.
34pub const HOTSWAP_RETIRED: u32 = 0xAAAA_AAAA;
35
36/// Verdict for one retired incarnation.
37#[repr(u16)]
38#[derive(Clone, Copy, Debug, PartialEq, Eq)]
39pub enum RetirementState {
40    /// Quiescence or deferred DMA is not complete yet.
41    Pending = 1,
42    /// No outstanding obligation remains.
43    Contained = 2,
44    /// Only counted, bounded quarantines remain (pins, VFS leases, deferred DMA).
45    Quarantined = 3,
46    /// Memory reachable by live peers cannot be proven released; see the reasons.
47    Unproven = 4,
48}
49
50/// [`ViRetirementV1::reasons`] bits. Every bit outside [`reason::KNOWN`] is refused.
51pub mod reason {
52    /// The incarnation owned SHM frames: USER mappings in the shared root with no
53    /// borrower record. Forces `Unproven`.
54    pub const SHM_UNTRACKED: u32 = 1 << 0;
55    /// The bounded pin quarantine overflowed and pages were leaked. Forces `Unproven`.
56    pub const QUARANTINE_OVERFLOW: u32 = 1 << 1;
57    /// The incarnation died holding writable grantee references to a live peer's
58    /// grant. Reported only, never classified: the grant owner must treat that
59    /// memory's invariants as unproven and never unlock or reuse a lock inside it.
60    pub const WRITABLE_BORROW: u32 = 1 << 2;
61    /// IOMMU/DMA release was deferred and is retried. Gives `Quarantined`.
62    pub const DMA_DEFERRED: u32 = 1 << 3;
63    /// Pins, VFS leases or quarantined pages remain bounded. Gives `Quarantined`.
64    pub const TRACKED_QUARANTINE: u32 = 1 << 4;
65    /// Every defined bit.
66    pub const KNOWN: u32 = 0x1F;
67}
68
69/// Retirement record, version 1. Field order is the wire order.
70#[repr(C)]
71#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
72pub struct ViRetirementV1 {
73    /// [`RETIREMENT_RECORD_VERSION`].
74    pub version: u16,
75    /// [`RetirementState`] discriminant.
76    pub state: u16,
77    /// [`reason`] bits.
78    pub reasons: u32,
79    /// Tasks of the incarnation reaped by the retirement funnel, root included.
80    pub member_threads: u32,
81    /// Zero.
82    pub reserved0: u32,
83    /// Incarnation Cell id.
84    pub cell_id: u64,
85    /// Incarnation generation.
86    pub generation: u64,
87    /// Root task of the incarnation; the query key.
88    pub root_tid: u64,
89    /// Bit `h` set: retirement awaited a remote switch completion on hart `h`.
90    pub remote_hart_mask: u64,
91    /// Grants owned by the incarnation that were reclaimed.
92    pub grants_reclaimed: u32,
93    /// Grantee references to other owners' grants that were cleared.
94    pub grantee_refs_cleared: u32,
95    /// Cleared grantee references that were writable.
96    pub writable_borrows: u32,
97    /// Pin regions moved to quarantine.
98    pub pins_quarantined: u32,
99    /// VFS leases granted by the incarnation still awaiting holder release.
100    pub vfs_leases_pending: u32,
101    /// SHM frames the incarnation owned that were released.
102    pub shm_frames_released: u32,
103    /// Members whose DMA release was deferred.
104    pub dma_deferred: u32,
105    /// Zero.
106    pub reserved1: u32,
107    /// Quarantined pages attributed to the incarnation's members.
108    pub quarantined_pages: u64,
109    /// Pages leaked during the retirement window. May over-report, never under.
110    pub leaked_pages: u64,
111}
112
113const _: () = assert!(core::mem::size_of::<ViRetirementV1>() == RETIREMENT_RECORD_LEN);

115/// Why bytes are not a valid V1 record.
116#[derive(Clone, Copy, Debug, PartialEq, Eq)]
117pub enum RecordError {
118    /// Length is not exactly [`RETIREMENT_RECORD_LEN`].
119    Length,
120    /// `version` is not [`RETIREMENT_RECORD_VERSION`].
121    UnsupportedVersion,
122    /// A reserved field is non-zero.
123    Reserved,
124    /// `state` is not a defined [`RetirementState`].
125    UnknownState,
126    /// `reasons` has a bit outside [`reason::KNOWN`].
127    UnknownReason,
128}
129
130impl ViRetirementV1 {
131    /// Little-endian wire bytes.
132    pub fn encode(&self) -> [u8; RETIREMENT_RECORD_LEN] {
133        let mut out = [0u8; RETIREMENT_RECORD_LEN];
134        out[0..2].copy_from_slice(&self.version.to_le_bytes());
135        out[2..4].copy_from_slice(&self.state.to_le_bytes());
136        let head = [self.reasons, self.member_threads, self.reserved0];
137        for (i, v) in head.iter().enumerate() {
138            out[4 + i * 4..8 + i * 4].copy_from_slice(&v.to_le_bytes());
139        }
140        let ids = [
141            self.cell_id,
142            self.generation,
143            self.root_tid,
144            self.remote_hart_mask,
145        ];
146        for (i, v) in ids.iter().enumerate() {
147            out[16 + i * 8..24 + i * 8].copy_from_slice(&v.to_le_bytes());
148        }
149        let counts = [
150            self.grants_reclaimed,
151            self.grantee_refs_cleared,
152            self.writable_borrows,
153            self.pins_quarantined,
154            self.vfs_leases_pending,
155            self.shm_frames_released,
156            self.dma_deferred,
157            self.reserved1,
158        ];
159        for (i, v) in counts.iter().enumerate() {
160            out[48 + i * 4..52 + i * 4].copy_from_slice(&v.to_le_bytes());
161        }
162        out[80..88].copy_from_slice(&self.quarantined_pages.to_le_bytes());
163        out[88..96].copy_from_slice(&self.leaked_pages.to_le_bytes());
164        out
165    }
166
167    /// Decode and fully validate wire bytes.
168    pub fn decode(bytes: &[u8]) -> Result<Self, RecordError> {
169        if bytes.len() != RETIREMENT_RECORD_LEN {
170            return Err(RecordError::Length);
171        }
172        let le = |at: usize, n: usize| {
173            bytes[at..at + n]
174                .iter()
175                .rev()
176                .fold(0u64, |acc, b| acc << 8 | u64::from(*b))
177        };
178        let record = Self {
179            version: le(0, 2) as u16,
180            state: le(2, 2) as u16,
181            reasons: le(4, 4) as u32,
182            member_threads: le(8, 4) as u32,
183            reserved0: le(12, 4) as u32,
184            cell_id: le(16, 8),
185            generation: le(24, 8),
186            root_tid: le(32, 8),
187            remote_hart_mask: le(40, 8),
188            grants_reclaimed: le(48, 4) as u32,
189            grantee_refs_cleared: le(52, 4) as u32,
190            writable_borrows: le(56, 4) as u32,
191            pins_quarantined: le(60, 4) as u32,
192            vfs_leases_pending: le(64, 4) as u32,
193            shm_frames_released: le(68, 4) as u32,
194            dma_deferred: le(72, 4) as u32,
195            reserved1: le(76, 4) as u32,
196            quarantined_pages: le(80, 8),
197            leaked_pages: le(88, 8),
198        };
199        if record.version != RETIREMENT_RECORD_VERSION {
200            return Err(RecordError::UnsupportedVersion);
201        }
202        if record.reserved0 != 0 || record.reserved1 != 0 {
203            return Err(RecordError::Reserved);
204        }
205        if record.retirement_state().is_none() {
206            return Err(RecordError::UnknownState);
207        }
208        if record.reasons & !reason::KNOWN != 0 {
209            return Err(RecordError::UnknownReason);
210        }
211        Ok(record)
212    }
213
214    /// The typed state; `None` for an undefined value.
215    pub fn retirement_state(&self) -> Option<RetirementState> {
216        match self.state {
217            1 => Some(RetirementState::Pending),
218            2 => Some(RetirementState::Contained),
219            3 => Some(RetirementState::Quarantined),
220            4 => Some(RetirementState::Unproven),
221            _ => None,
222        }
223    }
224}
