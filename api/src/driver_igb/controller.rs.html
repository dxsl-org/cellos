1//! Intel `igb` (82576-class) NIC controller logic.
2//!
3//! Register model, taken from the 82576 datasheet as implemented by QEMU's
4//! `igb` model (`hw/net/igb_regs.h`, `hw/net/igb_core.c`, QEMU 8.2):
5//!
6//! * queue-0 ring registers live at `0x0C000` (RX) / `0x0E000` (TX), 0x40 apart
7//!   per queue — *not* the 82540EM's `0x2800`/`0x3800`;
8//! * the NVM read register `EERD` (0x14) carries the word address at bits 15:2
9//!   and signals completion with bit 1 — *not* the 82540EM's bits 15:8 / bit 4;
10//! * descriptors are the **advanced** 16-byte forms: an RX completion writes
11//!   `status_error` at offset 8 and `length` at offset 12, and a TX descriptor
12//!   must set `DEXT | DTYP_DATA` or the model treats it as a legacy descriptor.
13//!
14//! Every one of those differences is the reason this is a separate cell rather
15//! than a re-run of the e1000 controller.
16//!
17//! Law 4 exception: `unsafe` is used for DMA descriptor and buffer access. Every
18//! `unsafe` block carries a `// SAFETY:` comment. MMIO goes through the
19//! bounds-checked [`MmioRegion`], so a bad offset returns `Err`, never a stray
20//! access.
21

31// ── Register offsets (BAR0, 128 KiB window) ──────────────────────────────────

33const CTRL: usize = 0x0_0000;
34const STATUS: usize = 0x0_0008;
35const EERD: usize = 0x0_0014;
36const MDIC: usize = 0x0_0020;
37const ICR: usize = 0x0_00C0;
38const IMC: usize = 0x0_00D8;
39const RCTL: usize = 0x0_0100;
40const TCTL: usize = 0x0_0400;
41const TIPG: usize = 0x0_0410;
42const MTA: usize = 0x0_5200;
43const RAL0: usize = 0x0_5400;
44const RAH0: usize = 0x0_5404;

46// Queue 0 (the igb `_n` register bank).
47const RDBAL0: usize = 0x0_C000;
48const RDBAH0: usize = 0x0_C004;
49const RDLEN0: usize = 0x0_C008;
50const SRRCTL0: usize = 0x0_C00C;
51const RDH0: usize = 0x0_C010;
52const RDT0: usize = 0x0_C018;
53const RXDCTL0: usize = 0x0_C028;
54const TDBAL0: usize = 0x0_E000;
55const TDBAH0: usize = 0x0_E004;
56const TDLEN0: usize = 0x0_E008;
57const TDH0: usize = 0x0_E010;
58const TDT0: usize = 0x0_E018;
59const TXDCTL0: usize = 0x0_E028;
60const TDWBAL0: usize = 0x0_E038;
61const TDWBAH0: usize = 0x0_E03C;

63// ── Register bits ────────────────────────────────────────────────────────────
64
65const CTRL_RST: u32 = 1 << 26;
66const CTRL_SLU: u32 = 1 << 6;
67const CTRL_ASDE: u32 = 1 << 5;
68const STATUS_LU: u32 = 1 << 1;
69const EERD_START: u32 = 1 << 0;
70const EERD_DONE: u32 = 1 << 1;
71const EERD_ADDR_SHIFT: u32 = 2;
72const EERD_DATA_SHIFT: u32 = 16;
73const RCTL_EN: u32 = 1 << 1;
74const RCTL_UPE: u32 = 1 << 3;
75const RCTL_MPE: u32 = 1 << 4;
76const RCTL_BAM: u32 = 1 << 15;
77const RCTL_SECRC: u32 = 1 << 26;
78const TCTL_EN: u32 = 1 << 1;
79const TCTL_PSP: u32 = 1 << 3;
80const TCTL_CT: u32 = 0x0F << 4;
81const TCTL_COLD: u32 = 0x40 << 12;
82const RAH_AV: u32 = 1 << 31;
83const RXDCTL_QUEUE_ENABLE: u32 = 1 << 25;
84const TXDCTL_QUEUE_ENABLE: u32 = 1 << 25;
85/// `SRRCTL.DESCTYPE = 001b` (advanced, one buffer) — the only descriptor layout
86/// the igb model implements.
87const SRRCTL_DESCTYPE_ADV_ONEBUF: u32 = 0x0200_0000;
88/// `SRRCTL.BSIZEPKT` counts KiB (`2048 >> 10 == 2`).
89const SRRCTL_BSIZEPKT_2048: u32 = 2;
90
91const ADVTXD_DCMD_DEXT: u32 = 1 << 29;
92const ADVTXD_DTYP_DATA: u32 = 0x0030_0000;
93const TXD_CMD_EOP: u32 = 1 << 24;
94const TXD_CMD_IFCS: u32 = 1 << 25;
95const TXD_CMD_RS: u32 = 1 << 27;
96const TXD_STAT_DD: u32 = 1;
97const RXD_STAT_DD: u32 = 1;

99// Management Data Interface Control (`MDIC`) — the PHY register window. The igb
100// model, like the hardware, only ties `STATUS.LU` to PHY autonegotiation: after
101// a global reset the PHY restarts from its power-on defaults and the link stays
102// down until software restarts autoneg through here.
103const MDIC_DATA_MASK: u32 = 0x0000_FFFF;
104const MDIC_REG_SHIFT: u32 = 16;
105const MDIC_PHY_SHIFT: u32 = 21;
106const MDIC_OP_WRITE: u32 = 0x0400_0000;
107/// PHY address 1 — the model rejects any other (`igb_set_mdic`).
108const MDIC_PHY_ADDR: u32 = 1;
109/// MII register 0 = Basic Mode Control Register.
110const MII_BMCR: u32 = 0;
111/// `BMCR = SPEED1000 | FULLDPLX | AUTOEN | ANRESTART` (hw/net/mii.h).
112const MII_BMCR_SPEED1000: u32 = 1 << 6;
113const MII_BMCR_FD: u32 = 1 << 8;
114const MII_BMCR_ANRESTART: u32 = 1 << 9;
115const MII_BMCR_AUTOEN: u32 = 1 << 12;
116
117// ── Poll budgets ─────────────────────────────────────────────────────────────
118
119/// Global-reset self-clear poll.
120const RESET_POLLS: u32 = 1_000_000;
121/// NVM read completion poll.
122const NVM_POLLS: u32 = 100_000;
123/// TX descriptor completion poll.
124const TX_POLLS: u32 = 1_000_000;
125/// Link-up poll. The igb model brings the link up only when the PHY autoneg
126/// timer expires, ~500 ms after `restart_phy_autoneg`; the budget is several
127/// times that so the wait is bounded but comfortably covers it.
128const LINK_POLLS: u32 = 6_000_000;
129
130const N_TX: usize = TX_SLOTS;
131const N_RX: usize = RX_SLOTS;
132
133/// Largest frame this cell moves in one descriptor (Ethernet MTU + headers).
134pub const BUF_SIZE: usize = 2048;

138/// Advanced transmit descriptor. `status` shares offset 12 with
139/// `olinfo_status`: the model writes `TXD_STAT_DD` there after `RS`.
140#[repr(C)]
141struct TxDesc {
142    buf_addr: u64,
143    cmd_type_len: u32,
144    status: u32,
145}
146
147/// Advanced receive descriptor. Programming uses `buf_addr` (offset 0); a
148/// completion writes `status_error` (offset 8) and `length` (offset 12).
149#[repr(C)]
150struct RxDesc {
151    buf_addr: u64,
152    status_error: u32,
153    length: u16,
154    vlan: u16,
155}
156
157const _: () = assert!(core::mem::size_of::<TxDesc>() == 16);
158const _: () = assert!(core::mem::size_of::<RxDesc>() == 16);

162#[inline]
163fn rd(mmio: &MmioRegion, off: usize) -> ViResult<u32> {
164    mmio.read_u32(off)
165}
166
167#[inline]
168fn wr(mmio: &MmioRegion, off: usize, val: u32) -> ViResult<()> {
169    mmio.write_u32(off, val)
170}

172/// Read one 16-bit NVM word through `EERD` (igb layout: address at bits 15:2,
173/// done at bit 1, data at bits 31:16).
174fn nvm_read(mmio: &MmioRegion, word: u16) -> ViResult<u16> {
175    wr(mmio, EERD, ((word as u32) << EERD_ADDR_SHIFT) | EERD_START)?;
176    for _ in 0..NVM_POLLS {
177        let value = rd(mmio, EERD)?;
178        if value & EERD_DONE != 0 {
179            return Ok((value >> EERD_DATA_SHIFT) as u16);
180        }
181        core::hint::spin_loop();
182    }
183    Err(ViError::IO)
184}

186/// Restart PHY autonegotiation, which is the only path to `STATUS.LU` on this
187/// part. The global reset in `new` leaves the PHY at its power-on default and
188/// clears the link, so a driver that skipped this would enable RX on a link the
189/// controller reports as down and never receive a frame.
190fn restart_phy_autoneg(mmio: &MmioRegion) -> ViResult<()> {
191    let bmcr = MII_BMCR_SPEED1000 | MII_BMCR_FD | MII_BMCR_AUTOEN | MII_BMCR_ANRESTART;
192    let command = (bmcr & MDIC_DATA_MASK)
193        | (MII_BMCR << MDIC_REG_SHIFT)
194        | (MDIC_PHY_ADDR << MDIC_PHY_SHIFT)
195        | MDIC_OP_WRITE;
196    wr(mmio, MDIC, command)?;
197    // The model applies the write synchronously; the link then comes up when the
198    // autoneg timer expires (~500 ms of guest time).
199    let _ = rd(mmio, MDIC)?;
200    Ok(())
201}

205pub struct IgbController {
206    mmio: MmioRegion,
207    tx_ring: DmaBuf,
208    rx_ring: DmaBuf,
209    tx_bufs: [DmaBuf; N_TX],
210    rx_bufs: [DmaBuf; N_RX],
211    dma_iovas: DmaIovas,
212    tx_next: usize,
213    rx_head: usize,
214    /// MAC read from the NVM (words 0..3).
215    pub mac: [u8; 6],
216    /// Whether `STATUS.LU` was observed up before RX was enabled.
217    pub link_up: bool,
218}

220// SAFETY: IgbController is only accessed from the single-threaded Cell event loop.
221unsafe impl Send for IgbController {}

223impl IgbController {
224    fn rd32(&self, off: usize) -> ViResult<u32> {
225        rd(&self.mmio, off)
226    }
227
228    fn wr32(&self, off: usize, val: u32) -> ViResult<()> {
229        wr(&self.mmio, off, val)

232    /// Bring the controller up on `mmio` (BAR0) for the PCIe device at `bdf`.
233    pub fn new(mmio: MmioRegion, bdf: u32) -> ViResult<Self> {
234        // 1. Quiesce inherited state. A restarted driver may otherwise inherit an
235        //    enabled RX/TX pair and stale ring addresses.
236        wr(&mmio, CTRL, CTRL_RST)?;
237        let mut polls = 0u32;
238        loop {
239            if rd(&mmio, CTRL)? & CTRL_RST == 0 {
240                break;
241            }
242            polls += 1;
243            if polls > RESET_POLLS {
244                return Err(ViError::IO);
245            }
246            core::hint::spin_loop();
247        }
248
249        // 2. Restart PHY autonegotiation. The reset left `STATUS.LU` cleared and
250        //    the autoneg timer disarmed, so this is what brings the link back;
251        //    without it RX is permanently refused by the controller.
252        restart_phy_autoneg(&mmio)?;
253
254        // 3. Force link up, mask interrupts, clear any latched cause.
255        wr(&mmio, CTRL, CTRL_SLU | CTRL_ASDE)?;
256        wr(&mmio, IMC, 0xFFFF_FFFF)?;
257        let _ = rd(&mmio, ICR)?;
258
259        // 4. MAC from the NVM, then the unicast filter (RAL0/RAH0), then the
260        //    multicast table. RAL0/RAH0 also make an exact-match (non-promiscuous)
261        //    frame pass, so the filter is real even though RCTL also opens the
262        //    promiscuous paths for the DHCP broadcast.
263        let mac_lo = nvm_read(&mmio, 0)?;
264        let mac_mid = nvm_read(&mmio, 1)?;
265        let mac_hi = nvm_read(&mmio, 2)?;
266        let mac = [
267            mac_lo as u8,
268            (mac_lo >> 8) as u8,
269            mac_mid as u8,
270            (mac_mid >> 8) as u8,
271            mac_hi as u8,
272            (mac_hi >> 8) as u8,
273        ];
274        // A MAC of all zeros means the NVM read never completed; binding such a
275        // device would publish an unusable NIC route, so fail before any DMA.
276        if mac == [0u8; 6] {
277            return Err(ViError::IO);
278        }
279        wr(&mmio, RAL0, ((mac_mid as u32) << 16) | mac_lo as u32)?;
280        wr(&mmio, RAH0, RAH_AV | mac_hi as u32)?;
281        for index in 0..128 {
282            wr(&mmio, MTA + index * 4, 0)?;
283        }
284
285        // 4. Allocate every DMA object before authorization. No device-visible
286        //    address is programmed until every object has an approved IOVA.
287        let tx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?; // 16 × TxDesc
288        let tx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
289        let rx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?; // 16 × RxDesc
290        let rx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
291
292        with_authorized_dma_layout(
293            (mmio, tx_ring, rx_ring, tx_bufs, rx_bufs),
294            |resources, slot| {
295                let result = match slot {
296                    DmaSlot::TxRing => resources.1.authorize(bdf),
297                    DmaSlot::TxBuffer(index) => resources.3[index].authorize(bdf),
298                    DmaSlot::RxRing => resources.2.authorize(bdf),
299                    DmaSlot::RxBuffer(index) => resources.4[index].authorize(bdf),
300                };
301                result.map_err(|_| ViError::PermissionDenied)
302            },
303            |(mmio, tx_ring, rx_ring, tx_bufs, rx_bufs), dma_iovas| {
304                // SAFETY: both ring allocations cover one writable DMA page.
305                unsafe {
306                    core::ptr::write_bytes(tx_ring.virt(), 0, tx_ring.size());
307                    core::ptr::write_bytes(rx_ring.virt(), 0, rx_ring.size());
308                }
309
310                let mut ctrl = IgbController {
311                    mmio,
312                    tx_ring,
313                    rx_ring,
314                    tx_bufs,
315                    rx_bufs,
316                    dma_iovas,
317                    tx_next: 0,
318                    rx_head: 0,
319                    mac,
320                    link_up: false,
321                };
322
323                let layout = ctrl.dma_iovas;
324                for_each_initial_dma_program(&layout, |program| match program {
325                    InitialDmaProgram::TxRingBase(iova) => {
326                        let _ = ctrl.wr32(TDBAL0, iova as u32);
327                        let _ = ctrl.wr32(TDBAH0, (iova >> 32) as u32);
328                        let _ = ctrl.wr32(TDLEN0, (N_TX * 16) as u32);
329                        let _ = ctrl.wr32(TDH0, 0);
330                        let _ = ctrl.wr32(TDT0, 0);
331                        // Descriptor write-back, not head-index write-back: the
332                        // model writes DD into the descriptor only when
333                        // TDWBAL0.0 is clear.
334                        let _ = ctrl.wr32(TDWBAL0, 0);
335                        let _ = ctrl.wr32(TDWBAH0, 0);
336                        let _ = ctrl.wr32(TXDCTL0, TXDCTL_QUEUE_ENABLE);
337                    }
338                    InitialDmaProgram::RxDescriptor { slot, iova } => {
339                        // SAFETY: rx_ring covers N_RX × RxDesc; slot < N_RX.
340                        unsafe {
341                            let desc = (ctrl.rx_ring.virt() as *mut RxDesc).add(slot);
342                            core::ptr::write_volatile(&mut (*desc).buf_addr, iova);
343                            core::ptr::write_volatile(&mut (*desc).status_error, 0);
344                            core::ptr::write_volatile(&mut (*desc).length, 0);
345                            core::ptr::write_volatile(&mut (*desc).vlan, 0);
346                        }
347                    }
348                    InitialDmaProgram::RxRingBase(iova) => {
349                        let _ = ctrl.wr32(SRRCTL0, SRRCTL_DESCTYPE_ADV_ONEBUF | SRRCTL_BSIZEPKT_2048);
350                        let _ = ctrl.wr32(RDBAL0, iova as u32);
351                        let _ = ctrl.wr32(RDBAH0, (iova >> 32) as u32);
352                        let _ = ctrl.wr32(RDLEN0, (N_RX * 16) as u32);
353                        let _ = ctrl.wr32(RDH0, 0);
354                        let _ = ctrl.wr32(RXDCTL0, RXDCTL_QUEUE_ENABLE);
355                        let _ = ctrl.wr32(RDT0, (N_RX - 1) as u32);
356                    }
357                    InitialDmaProgram::Enable => {
358                        let _ = ctrl.wr32(TIPG, 0x0060_200A);
359                        let _ = ctrl.wr32(TCTL, TCTL_EN | TCTL_PSP | TCTL_CT | TCTL_COLD);
360                        // Link must be up before RCTL_EN means anything to the
361                        // model (`e1000x_rx_ready` requires STATUS.LU).
362                        let link = ctrl.wait_link_up().unwrap_or(false);
363                        ctrl.link_up = link;
364                        let _ = ctrl
365                            .wr32(RCTL, RCTL_EN | RCTL_UPE | RCTL_MPE | RCTL_BAM | RCTL_SECRC);
366                    }
367                });
368
369                Ok(ctrl)
370            },
371        )?
372    }

374    /// Poll `STATUS.LU` for a bounded window. Returns whether link is up; a
375    /// timeout is not fatal (the model brings link up on its own schedule) but
376    /// the caller reports it.
377    pub fn wait_link_up(&self) -> ViResult<bool> {
378        for _ in 0..LINK_POLLS {
379            if self.rd32(STATUS)? & STATUS_LU != 0 {
380                return Ok(true);
381            }
382            core::hint::spin_loop();
383        }
384        Ok(false)
385    }

387    /// Transmit `frame` (Ethernet frame without FCS), polled until the
388    /// descriptor reports done.
389    pub fn send_frame(&mut self, frame: &[u8]) -> ViResult<()> {
390        if frame.is_empty() || frame.len() > BUF_SIZE {
391            return Err(ViError::InvalidInput);
392        }
393
394        let slot = self.tx_next;
395        self.tx_next = (slot + 1) % N_TX;
396
397        // SAFETY: tx_bufs[slot].virt() is valid DMA memory of size BUF_SIZE, and
398        // frame.len() <= BUF_SIZE.
399        unsafe {
400            core::ptr::copy_nonoverlapping(frame.as_ptr(), self.tx_bufs[slot].virt(), frame.len());
401        }
402        let iova = self.dma_iovas.tx_descriptor_iova(slot);
403        let cmd = ADVTXD_DCMD_DEXT
404            | ADVTXD_DTYP_DATA
405            | TXD_CMD_EOP
406            | TXD_CMD_IFCS
407            | TXD_CMD_RS
408            | frame.len() as u32;
409
410        // SAFETY: tx_ring covers N_TX × TxDesc; slot < N_TX.
411        unsafe {
412            let desc = (self.tx_ring.virt() as *mut TxDesc).add(slot);
413            core::ptr::write_volatile(&mut (*desc).buf_addr, iova);
414            core::ptr::write_volatile(&mut (*desc).cmd_type_len, cmd);
415            core::ptr::write_volatile(&mut (*desc).status, 0);
416        }
417        compiler_fence(Ordering::Release);
418        self.wr32(TDT0, self.tx_next as u32)?;
419
420        for _ in 0..TX_POLLS {
421            // SAFETY: tx_ring[slot] is valid DMA memory.
422            let status = unsafe {
423                core::ptr::read_volatile(&(*((self.tx_ring.virt() as *const TxDesc).add(slot))).status)
424            };
425            if status & TXD_STAT_DD != 0 {
426                return Ok(());
427            }
428            core::hint::spin_loop();
429        }
430        Err(ViError::IO)
431    }

433    /// Poll for a received frame; copies into `out_buf`. Returns the number of
434    /// bytes written, or 0 when nothing is ready.
435    pub fn recv_frame(&mut self, out_buf: &mut [u8]) -> usize {
436        let head = self.rx_head;
437        // SAFETY: rx_ring[head] is valid DMA memory.
438        let (done, length) = unsafe {
439            let desc = (self.rx_ring.virt() as *const RxDesc).add(head);
440            let status_error = core::ptr::read_volatile(&(*desc).status_error);
441            let length = core::ptr::read_volatile(&(*desc).length);
442            (status_error & RXD_STAT_DD != 0, length as usize)
443        };
444        if !done {
445            return 0;
446        }
447
448        let copy_len = length.min(out_buf.len()).min(BUF_SIZE);
449        // SAFETY: rx_bufs[head].virt() holds the received frame, and copy_len is
450        // bounded by both the frame length and the caller's slice.
451        unsafe {
452            core::ptr::copy_nonoverlapping(
453                self.rx_bufs[head].virt(),
454                out_buf.as_mut_ptr(),
455                copy_len,
456            );
457        }
458
459        // Recycle the descriptor: restore the buffer address, clear the
460        // completion fields, and hand the slot back to the controller.
461        let iova = self.dma_iovas.rx_descriptor_iova(head);
462        // SAFETY: rx_ring[head] and rx_bufs[head] are valid DMA memory.
463        unsafe {
464            let desc = (self.rx_ring.virt() as *mut RxDesc).add(head);
465            core::ptr::write_volatile(&mut (*desc).buf_addr, iova);
466            core::ptr::write_volatile(&mut (*desc).status_error, 0);
467            core::ptr::write_volatile(&mut (*desc).length, 0);
468            core::ptr::write_volatile(&mut (*desc).vlan, 0);
469        }
470        self.rx_head = (head + 1) % N_RX;
471        let _ = self.wr32(RDT0, head as u32);
472        copy_len
473    }
