1//! Intel `igb` PCI identity table — the **datasheet** IDs, kept separate from
2//! the **QEMU model** ID.
3//!
4//! The two lists are deliberately not merged (phase 04a):
5//!
6//! * [`QEMU_MODEL_ID`] — `8086:10c9`. QEMU's `igb` device emulates an 82576
7//!   (`hw/net/igb.c`: `c->device_id = E1000_DEV_ID_82576`); this is the only ID
8//!   any gate in this lane can validate end to end.
9//! * [`I210_IDS`] / [`I211_IDS`] — the real SKU IDs, read from the datasheet:
10//!   Intel Ethernet Controller I210 Datasheet §6.2.5 "Device ID (Word 0x0D)"
11//!   gives the I210 copper-only default as `0x1533` and points at §9.3.2 for the
12//!   other SKUs; the I211 SKU is `0x1539` (§3.4 iNVM and the I211 proxy-flow
13//!   note). The fibre/SerDes/SGMII and flash-less values are that same
14//!   datasheet table as transcribed by the upstream `igb` driver
15//!   (`drivers/net/ethernet/intel/igb/e1000_hw.h`): `I210_COPPER 0x1533`,
16//!   `I210_FIBER 0x1536`, `I210_SERDES 0x1537`, `I210_SGMII 0x1538`,
17//!   `I210_COPPER_FLASHLESS 0x157B`, `I210_SERDES_FLASHLESS 0x157C`,
18//!   `I211_COPPER 0x1539`.
19//!
20//! None of the datasheet IDs is validated by the QEMU model (plan `A-01`: the
21//! model is 82576-class, not bit-exact i210/i211 — PHY and NVM differ), so they
22//! bind with the same register model but without a QEMU gate. Every ID outside
23//! this table is a *refusal* that names vendor:device; there is no wildcard.
24
25/// Intel's PCI vendor ID.
26pub const VENDOR_INTEL: u16 = 0x8086;
27
28/// QEMU's `igb` model: 82576-class, `8086:10c9`. Model-validated.
29pub const QEMU_MODEL_ID: u16 = 0x10C9;
30
31/// i210 SKU device IDs (datasheet §9.3.2 / upstream `igb` ID table).
32///
33/// **Target list, not a claim** — see [`SUPPORTED_DEVICE_IDS`] for what this cell
34/// can actually drive today.
35pub const I210_IDS: [u16; 6] = [0x1533, 0x1536, 0x1537, 0x1538, 0x157B, 0x157C];
36
37/// i211 SKU device ID (datasheet §3.4 / I211 proxy flow). Target only: the i211
38/// reads its MAC from iNVM, which this cell does not implement yet.
39pub const I211_IDS: [u16; 1] = [0x1539];
40
41/// The device IDs this cell actually claims: the ones whose NVM access and media
42/// path it implements.
43///
44/// * `0x10C9` — QEMU's 82576 model, the only ID validated in this lane.
45/// * `0x1533` — i210 copper **with external flash**, so the EERD/Shadow-RAM read
46///   implemented here returns the MAC.
47///
48/// Deliberately absent, with the prerequisite each needs (recorded in the phase
49/// evidence so a future claim has to implement it rather than widen the table):
50/// `0x157B`/`0x157C` (i210 flashless) and `0x1539` (i211) need the **iNVM** read
51/// path; `0x1536`/`0x1537`/`0x1538` (fibre/SerDes/SGMII) need **media-specific
52/// link setup** rather than the copper BMCR restart. Claiming any of them without
53/// that work would register a device this cell cannot drive.
54pub const SUPPORTED_DEVICE_IDS: [u16; 2] = [QEMU_MODEL_ID, 0x1533];
55
56/// Every device ID this cell hands to `sys_find_pcie_device_by_vendor`, in query
57/// order, all under [`VENDOR_INTEL`].
58///
59/// Naming the controller exactly is what removes the sibling race: `/bin/e1000`
60/// can never hold this device, so there is no decline-and-retry window.
61pub const QUERY_DEVICE_IDS: [u16; 2] = SUPPORTED_DEVICE_IDS;
62
63// Compile-time check (this cell sets `test = false`, so a `#[cfg(test)]` module
64// would never be compiled): the query list is exactly the supported set, every
65// entry classifies as a driveable SKU, and the unimplemented SKUs stay out until
66// their prerequisite lands.
67const _: () = {
68    assert!(QUERY_DEVICE_IDS[0] == QEMU_MODEL_ID);
69    assert!(QUERY_DEVICE_IDS[1] == 0x1533);
70    assert!(QUERY_DEVICE_IDS.len() == SUPPORTED_DEVICE_IDS.len());
71    let mut i = 0;
72    while i < QUERY_DEVICE_IDS.len() {
73        assert!(!matches!(
74            classify(VENDOR_INTEL, QUERY_DEVICE_IDS[i]),
75            IgbSku::Unsupported
76        ));
77        i += 1;
78    }
79};

81/// The register model this cell implements, as decided by the PCI ID.
82#[derive(Clone, Copy, Debug, Eq, PartialEq)]
83pub enum IgbSku {
84    /// 82576 — the QEMU model, the only ID validated in this lane.
85    I82576,
86    /// A real i210 SKU; same register model, not model-validated.
87    I210,
88    /// The i211 SKU; same register model, not model-validated.
89    I211,
90    /// Anything else: the caller must fail closed and name vendor:device.
91    Unsupported,
92}
93
94/// Classify a `(vendor, device)` pair.
95///
96/// `const fn` so the same table can be used by tests and by the bind path
97/// without allocating; `while` loops are the only construct available here.
98pub const fn classify(vendor_id: u16, device_id: u16) -> IgbSku {
99    if vendor_id != VENDOR_INTEL {
100        return IgbSku::Unsupported;
101    }
102    if device_id == QEMU_MODEL_ID {
103        return IgbSku::I82576;
104    }
105    let mut i = 0;
106    while i < I210_IDS.len() {
107        if device_id == I210_IDS[i] {
108            return IgbSku::I210;
109        }
110        i += 1;
111    }
112    let mut i = 0;
113    while i < I211_IDS.len() {
114        if device_id == I211_IDS[i] {
115            return IgbSku::I211;
116        }
117        i += 1;
118    }
119    IgbSku::Unsupported
120}
121
122/// Human-readable family name for the bind log line.
123pub const fn sku_name(sku: IgbSku) -> &'static str {
124    match sku {
125        IgbSku::I82576 => "82576 (QEMU model)",
126        IgbSku::I210 => "i210 (datasheet)",
127        IgbSku::I211 => "i211 (datasheet)",
128        IgbSku::Unsupported => "unsupported",
129    }
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    #[test]
137    fn qemu_model_is_recognised() {
138        assert_eq!(classify(VENDOR_INTEL, QEMU_MODEL_ID), IgbSku::I82576);
139    }
140
141    #[test]
142    fn datasheet_skus_are_recognised() {
143        assert_eq!(classify(VENDOR_INTEL, 0x1533), IgbSku::I210);
144        assert_eq!(classify(VENDOR_INTEL, 0x1536), IgbSku::I210);
145        assert_eq!(classify(VENDOR_INTEL, 0x1537), IgbSku::I210);
146        assert_eq!(classify(VENDOR_INTEL, 0x1538), IgbSku::I210);
147        assert_eq!(classify(VENDOR_INTEL, 0x157B), IgbSku::I210);
148        assert_eq!(classify(VENDOR_INTEL, 0x157C), IgbSku::I210);
149        assert_eq!(classify(VENDOR_INTEL, 0x1539), IgbSku::I211);
150    }
151
152    #[test]
153    fn other_families_fail_closed() {
154        // The e1000 the sibling cell owns must never classify as igb.
155        assert_eq!(classify(VENDOR_INTEL, 0x100E), IgbSku::Unsupported);
156        // e1000e/I219 stays a recorded follow-up, not a silent accept.
157        assert_eq!(classify(VENDOR_INTEL, 0x10D3), IgbSku::Unsupported);
158        assert_eq!(classify(VENDOR_INTEL, 0x15A1), IgbSku::Unsupported);
159        assert_eq!(classify(0x1AF4, QEMU_MODEL_ID), IgbSku::Unsupported);
160    }
