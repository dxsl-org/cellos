1//! Directory capabilities, exercised from a cell's own side.
2//!
3//! This cell is the first to migrate, and it migrates completely: it acquires
4//! the directories it needs, works entirely through handles, and then gives up
5//! the ability to name a path at all. The last part is the point. Traversal
6//! assertions on the handle operations prove only that the new interface is
7//! tight; a cell can pass every one of them while still sending absolute paths
8//! through the old interface. The assertion that matters is the one after
9//! sealing, where a plain `Write { path }` comes back refused.
10//!
11//! Runs last, because sealing is one-way and every scenario before it needs the
12//! path-string operations.
13
14use api::dir_handles::ViDirHandle;
15use api::ipc::{VfsRequest, VfsResponse};
16
17use crate::{fail, pass, vfs_req};
18
19/// `types::ViError::PermissionDenied`.
20const DENIED: u8 = 3;
21/// Unknown, stale, or not-this-caller's handle.
22const BAD_HANDLE: u8 = 4;
23/// The service could not decode the message at all.
24const MALFORMED: u8 = 0xFF;
25
26/// Directory this cell builds and then reaches only through a handle.
27const WORKDIR: &str = "/tmp/dircap";
28
29pub fn run() {
30    let Some(root) = acquire("/tmp") else {
31        return;
32    };
33    reject_every_traversal_shape(root);
34
35    // Built through the old interface on purpose: a cell bootstraps the
36    // directories it needs and only then narrows itself to them.
37    let _ = vfs_req(&VfsRequest::Mkdir(WORKDIR));
38
39    let Some(work) = derive(root, "dircap") else {
40        return;
41    };
42    file_operations_through_a_handle(work);
43    revoking_a_parent_revokes_what_came_from_it(root, work);
44
45    // Re-acquire: the revocation above took both handles with it, which is
46    // exactly what it was asserting.
47    let Some(root) = acquire("/tmp") else {
48        return;
49    };
50    let Some(work) = derive(root, "dircap") else {
51        return;
52    };
53    let _ = vfs_req(&VfsRequest::UnlinkAt {
54        dir: work,
55        name: "note.txt",
56    });
57
58    seal_and_prove_paths_are_refused(work);
59}

63fn acquire(path: &str) -> Option<ViDirHandle> {
64    match vfs_req(&VfsRequest::OpenRootDir { path }) {
65        VfsResponse::DirHandle(h) => {
66            pass("dircap: acquiring /tmp yields a directory handle");
67            Some(h)
68        }
69        _ => {
70            fail("dircap: acquiring /tmp yields a directory handle");
71            None
72        }
73    }
74}

76fn derive(parent: ViDirHandle, name: &str) -> Option<ViDirHandle> {
77    match vfs_req(&VfsRequest::OpenDir { dir: parent, name }) {
78        VfsResponse::DirHandle(h) => {
79            pass("dircap: deriving a subdirectory handle");
80            Some(h)
81        }
82        _ => {
83            fail("dircap: deriving a subdirectory handle");
84            None
85        }
86    }
87}

89// ── Nothing outside the handle can be named ──────────────────────────────────
90
91/// Each shape gets its own assertion: they are refused for different reasons,
92/// and a single loop would hide which reason stopped firing.
93fn reject_every_traversal_shape(dir: ViDirHandle) {
94    refused(dir, "..", "dircap: `..` is refused");
95    refused(dir, "../..", "dircap: `../..` is refused");
96    refused(dir, "/abs", "dircap: an absolute name is refused");
97    refused(dir, "a/../../b", "dircap: an embedded traversal is refused");
98    refused(dir, ".", "dircap: `.` is refused");
99    refused(dir, "", "dircap: an empty name is refused");
100    refused(dir, "..\\..\\etc", "dircap: a backslash name is refused");
101    refused(dir, "note\ntxt", "dircap: a control byte is refused");
102    reject_odd_utf8(dir);
103}
104
105fn refused(dir: ViDirHandle, name: &str, msg: &str) {
106    // Asserted on every operation that takes a name, not just one: they resolve
107    // through the same check, and a single call site would stop proving that if
108    // one of them ever stopped using it.
109    let attempts = [
110        VfsRequest::ReadAt { dir, name },
111        VfsRequest::StatAt { dir, name },
112        VfsRequest::UnlinkAt { dir, name },
113        VfsRequest::OpenDir { dir, name },
114        VfsRequest::WriteAt {
115            dir,
116            name,
117            content: b"x",
118        },
119    ];
120    for attempt in &attempts {
121        match vfs_req(attempt) {
122            VfsResponse::Err(DENIED) => {}
123            _ => {
124                fail(msg);
125                return;
126            }
127        }
128    }
129    pass(msg);
130}

132/// A name that is not valid UTF-8 cannot be built in Rust, so it is put on the
133/// wire by hand. This is the one traversal shape the type system cannot stop a
134/// caller from sending, and an overlong encoding of `/` is what a decoder that
135/// repaired invalid input would turn into a separator.
136fn reject_odd_utf8(dir: ViDirHandle) {
137    const READ_AT_VARIANT: u8 = 16;
138    if dir.0 >= 128 {
139        fail("dircap: odd UTF-8 is refused (handle too large to hand-encode)");
140        return;
141    }
142    // ReadAt { dir, name }: variant, handle as a varint, then a 2-byte name.
143    let msg = [READ_AT_VARIANT, dir.0 as u8, 2, 0xC0, 0xAF];
144    match crate::vfs_raw(&msg) {
145        // Refused at decode or refused at resolve; either way the backend never
146        // saw it.
147        VfsResponse::Err(MALFORMED) | VfsResponse::Err(DENIED) => {
148            pass("dircap: odd UTF-8 in a name is refused")
149        }
150        _ => fail("dircap: odd UTF-8 in a name is refused"),
151    }
152}

154// ── Ordinary work, through the handle only ───────────────────────────────────
155
156fn file_operations_through_a_handle(dir: ViDirHandle) {
157    match vfs_req(&VfsRequest::WriteAt {
158        dir,
159        name: "note.txt",
160        content: b"cap",
161    }) {
162        VfsResponse::Ok => pass("dircap: WriteAt creates a file inside the handle"),
163        _ => fail("dircap: WriteAt creates a file inside the handle"),
164    }
165
166    match vfs_req(&VfsRequest::StatAt {
167        dir,
168        name: "note.txt",
169    }) {
170        VfsResponse::Stat {
171            size: 3,
172            is_dir: false,
173        } => pass("dircap: StatAt reports the file it just wrote"),
174        _ => fail("dircap: StatAt reports the file it just wrote"),
175    }
176
177    match vfs_req(&VfsRequest::ReadAt {
178        dir,
179        name: "note.txt",
180    }) {
181        VfsResponse::Data(b"cap") => pass("dircap: ReadAt returns the bytes written"),
182        _ => fail("dircap: ReadAt returns the bytes written"),
183    }
184
185    match vfs_req(&VfsRequest::ListAt { dir }) {
186        VfsResponse::Data(bytes) => {
187            if bytes.windows(10).any(|w| w == b"f:note.txt") {
188                pass("dircap: ListAt lists the handle's own directory");
189            } else {
190                fail("dircap: ListAt lists the handle's own directory");
191            }
192        }
193        _ => fail("dircap: ListAt lists the handle's own directory"),
194    }
195
196    // The handle resolved where it claimed to: the same file is visible at the
197    // absolute path, which is still reachable because this cell has not sealed.
198    match vfs_req(&VfsRequest::Stat("/tmp/dircap/note.txt")) {
199        VfsResponse::Stat { size: 3, .. } => {
200            pass("dircap: the handle resolved inside the directory it names")
201        }
202        _ => fail("dircap: the handle resolved inside the directory it names"),
203    }
204}

206// ── Revocation reaches what was derived ──────────────────────────────────────
207
208fn revoking_a_parent_revokes_what_came_from_it(parent: ViDirHandle, derived: ViDirHandle) {
209    match vfs_req(&VfsRequest::CloseDir { dir: parent }) {
210        VfsResponse::Ok => pass("dircap: a handle can be given up"),
211        _ => fail("dircap: a handle can be given up"),
212    }
213    match vfs_req(&VfsRequest::StatAt {
214        dir: derived,
215        name: "note.txt",
216    }) {
217        VfsResponse::Err(BAD_HANDLE) => {
218            pass("dircap: revoking a handle revokes what was derived from it")
219        }
220        _ => fail("dircap: revoking a handle revokes what was derived from it"),
221    }
222}

224// ── The migration is only real once paths stop working ───────────────────────
225
226fn seal_and_prove_paths_are_refused(work: ViDirHandle) {
227    match vfs_req(&VfsRequest::SealPaths) {
228        VfsResponse::Ok => pass("dircap: the cell gives up naming paths"),
229        _ => {
230            fail("dircap: the cell gives up naming paths");
231            return;
232        }
233    }
234
235    // The criterion this whole phase is measured by.
236    match vfs_req(&VfsRequest::Write {
237        path: "/tmp/after_seal.txt",
238        content: b"should not land",
239    }) {
240        VfsResponse::Err(DENIED) => pass("dircap: Write{path} is refused after sealing"),
241        _ => fail("dircap: Write{path} is refused after sealing"),
242    }
243    // Nothing landed, either — the refusal is before the backend, not a reply
244    // written after the fact.
245    match vfs_req(&VfsRequest::StatAt {
246        dir: work,
247        name: "after_seal.txt",
248    }) {
249        VfsResponse::Err(_) => pass("dircap: the refused write left nothing behind"),
250        _ => fail("dircap: the refused write left nothing behind"),
251    }
252
253    // Every other path-addressed operation, so the refusal is not one arm.
254    for (req, msg) in [
255        (
256            VfsRequest::GetFile("/tmp/volatile.txt"),
257            "dircap: GetFile is refused after sealing",
258        ),
259        (
260            VfsRequest::Stat("/tmp/volatile.txt"),
261            "dircap: Stat is refused after sealing",
262        ),
263        (
264            VfsRequest::ListDir("/tmp"),
265            "dircap: ListDir is refused after sealing",
266        ),
267        (
268            VfsRequest::Unlink("/tmp/volatile.txt"),
269            "dircap: Unlink is refused after sealing",
270        ),
271        (
272            VfsRequest::Mkdir("/tmp/after_seal"),
273            "dircap: Mkdir is refused after sealing",
274        ),
275        (
276            VfsRequest::ReadAsync {
277                path: "/tmp/volatile.txt",
278            },
279            "dircap: ReadAsync is refused after sealing",
280        ),
281        (
282            VfsRequest::OpenRootDir { path: "/" },
283            "dircap: widening by acquiring a new root is refused after sealing",
284        ),
285    ] {
286        match vfs_req(&req) {
287            VfsResponse::Err(DENIED) => pass(msg),
288            _ => fail(msg),
289        }
290    }
291
292    // And the cell still works: sealing removed one way of naming things, not
293    // the cell's access to what it holds.
294    match vfs_req(&VfsRequest::WriteAt {
295        dir: work,
296        name: "sealed.txt",
297        content: b"ok",
298    }) {
299        VfsResponse::Ok => pass("dircap: handle operations still work after sealing"),
300        _ => fail("dircap: handle operations still work after sealing"),
301    }
302    match vfs_req(&VfsRequest::UnlinkAt {
303        dir: work,
304        name: "sealed.txt",
305    }) {
306        VfsResponse::Ok => pass("dircap: UnlinkAt still works after sealing"),
307        _ => fail("dircap: UnlinkAt still works after sealing"),
308    }
