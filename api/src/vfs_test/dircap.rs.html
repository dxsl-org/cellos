1//! Directory capabilities, exercised from a cell's own side.
2//!
3//! This cell is the first to migrate, and it migrates completely: it acquires
4//! the directories it needs, works entirely through handles, and then gives up
5//! the ability to name a path at all. The last part is the point. Traversal
6//! assertions on the handle operations prove only that the new interface is
7//! tight; a cell can pass every one of them while still sending absolute paths
8//! through the old interface. The assertion that matters is the one after
9//! sealing, where a plain `Write { path }` comes back refused.
10//!
11//! Runs last, because sealing is one-way and every scenario before it needs the
12//! path-string operations.
13
14use api::dir_handles::ViDirHandle;
15use api::ipc::{VfsRequest, VfsResponse};
16use api::vfs_file_handles::ViVfsFileHandle;
17
18use crate::grant_io;
19use crate::{fail, pass, vfs_req};
20
21/// `types::ViError::PermissionDenied`.
22const DENIED: u8 = 3;
23/// Unknown, stale, or not-this-caller's handle.
24const BAD_HANDLE: u8 = 4;
25/// The service could not decode the message at all.
26const MALFORMED: u8 = 0xFF;
27
28/// Directory this cell builds and then reaches only through a handle.
29const WORKDIR: &str = "/tmp/dircap";
30const READ_FILE_GRANT_SOURCE: &str = "/tmp/grant-read.txt";
31
32pub fn run() {
33    let Some(root) = acquire("/tmp") else {
34        return;
35    };
36    reject_every_traversal_shape(root);
37
38    // Built through the old interface on purpose: a cell bootstraps the
39    // directories it needs and only then narrows itself to them.
40    let _ = vfs_req(&VfsRequest::Mkdir(WORKDIR));
41
42    let Some(work) = derive(root, "dircap") else {
43        return;
44    };
45    file_operations_through_a_handle(work);
46    revoking_a_parent_revokes_what_came_from_it(root, work);
47
48    // Re-acquire: the revocation above took both handles with it, which is
49    // exactly what it was asserting.
50    let Some(root) = acquire("/tmp") else {
51        return;
52    };
53    let Some(work) = derive(root, "dircap") else {
54        return;
55    };
56
57    seal_and_prove_paths_are_refused(work);
58}

63    match vfs_req(&VfsRequest::OpenRootDir { path }) {
64        VfsResponse::DirHandle(h) => {
65            pass("dircap: acquiring /tmp yields a directory handle");
66            Some(h)
67        }
68        _ => {
69            fail("dircap: acquiring /tmp yields a directory handle");
70            None
71        }
72    }

88// ── Nothing outside the handle can be named ──────────────────────────────────

90/// Each shape gets its own assertion: they are refused for different reasons,
91/// and a single loop would hide which reason stopped firing.
92fn reject_every_traversal_shape(dir: ViDirHandle) {
93    refused(dir, "..", "dircap: `..` is refused");
94    refused(dir, "../..", "dircap: `../..` is refused");
95    refused(dir, "/abs", "dircap: an absolute name is refused");
96    refused(dir, "a/../../b", "dircap: an embedded traversal is refused");
97    refused(dir, ".", "dircap: `.` is refused");
98    refused(dir, "", "dircap: an empty name is refused");
99    refused(dir, "..\\..\\etc", "dircap: a backslash name is refused");
100    refused(dir, "note\ntxt", "dircap: a control byte is refused");
101    reject_odd_utf8(dir);
102}
103
104fn refused(dir: ViDirHandle, name: &str, msg: &str) {
105    // Asserted on every operation that takes a name, not just one: they resolve
106    // through the same check, and a single call site would stop proving that if
107    // one of them ever stopped using it.
108    let attempts = [
109        VfsRequest::ReadAt { dir, name },
110        VfsRequest::StatAt { dir, name },
111        VfsRequest::UnlinkAt { dir, name },
112        VfsRequest::OpenDir { dir, name },
113        VfsRequest::WriteAt {
114            dir,
115            name,
116            content: b"x",
117        },
118    ];
119    for attempt in &attempts {
120        match vfs_req(attempt) {
121            VfsResponse::Err(DENIED) => {}
122            _ => {
123                fail(msg);
124                return;
125            }
126        }
127    }
128    pass(msg);
129}

131/// A name that is not valid UTF-8 cannot be built in Rust, so it is put on the
132/// wire by hand. This is the one traversal shape the type system cannot stop a
133/// caller from sending, and an overlong encoding of `/` is what a decoder that
134/// repaired invalid input would turn into a separator.
135fn reject_odd_utf8(dir: ViDirHandle) {
136    const READ_AT_VARIANT: u8 = 16;
137    if dir.0 >= 128 {
138        fail("dircap: odd UTF-8 is refused (handle too large to hand-encode)");
139        return;
140    }
141    // ReadAt { dir, name }: variant, handle as a varint, then a 2-byte name.
142    let msg = [READ_AT_VARIANT, dir.0 as u8, 2, 0xC0, 0xAF];
143    match crate::vfs_raw(&msg) {
144        // Refused at decode or refused at resolve; either way the backend never
145        // saw it.
146        VfsResponse::Err(MALFORMED) | VfsResponse::Err(DENIED) => {
147            pass("dircap: odd UTF-8 in a name is refused")
148        }
149        _ => fail("dircap: odd UTF-8 in a name is refused"),
150    }
151}

155fn file_operations_through_a_handle(dir: ViDirHandle) {
156    match vfs_req(&VfsRequest::WriteAt {
157        dir,
158        name: "note.txt",
159        content: b"cap",
160    }) {
161        VfsResponse::Ok => pass("dircap: WriteAt creates a file inside the handle"),
162        _ => fail("dircap: WriteAt creates a file inside the handle"),
163    }
164
165    match vfs_req(&VfsRequest::StatAt {
166        dir,
167        name: "note.txt",
168    }) {
169        VfsResponse::Stat {
170            size: 3,
171            is_dir: false,
172        } => pass("dircap: StatAt reports the file it just wrote"),
173        _ => fail("dircap: StatAt reports the file it just wrote"),
174    }
175
176    match vfs_req(&VfsRequest::ReadAt {
177        dir,
178        name: "note.txt",
179    }) {
180        VfsResponse::Data(b"cap") => pass("dircap: ReadAt returns the bytes written"),
181        _ => fail("dircap: ReadAt returns the bytes written"),
182    }
183
184    match vfs_req(&VfsRequest::ListAt { dir }) {
185        VfsResponse::Data(bytes) => {
186            if bytes.windows(10).any(|w| w == b"f:note.txt") {
187                pass("dircap: ListAt lists the handle's own directory");
188            } else {
189                fail("dircap: ListAt lists the handle's own directory");
190            }
191        }
192        _ => fail("dircap: ListAt lists the handle's own directory"),
193    }
194
195    // The handle resolved where it claimed to: the same file is visible at the
196    // absolute path, which is still reachable because this cell has not sealed.
197    match vfs_req(&VfsRequest::Stat("/tmp/dircap/note.txt")) {
198        VfsResponse::Stat { size: 3, .. } => {
199            pass("dircap: the handle resolved inside the directory it names")
200        }
201        _ => fail("dircap: the handle resolved inside the directory it names"),
202    }
203}

207fn revoking_a_parent_revokes_what_came_from_it(parent: ViDirHandle, derived: ViDirHandle) {
208    let file = match vfs_req(&VfsRequest::OpenFileAt {
209        dir: derived,
210        name: "note.txt",
211    }) {
212        VfsResponse::FileHandle(file) => {
213            pass("dircap: OpenFileAt yields a file handle");
214            Some(file)
215        }
216        _ => {
217            fail("dircap: OpenFileAt yields a file handle");
218            None
219        }
220    };
221    match vfs_req(&VfsRequest::CloseDir { dir: parent }) {
222        VfsResponse::Ok => pass("dircap: a handle can be given up"),
223        _ => fail("dircap: a handle can be given up"),
224    }
225    match vfs_req(&VfsRequest::StatAt {
226        dir: derived,
227        name: "note.txt",
228    }) {
229        VfsResponse::Err(BAD_HANDLE) => {
230            pass("dircap: revoking a handle revokes what was derived from it")
231        }
232        _ => fail("dircap: revoking a handle revokes what was derived from it"),
233    }
234    if let Some(file) = file {
235        match vfs_req(&VfsRequest::ReadFileHandle {
236            file,
237            offset: 0,
238            max: 8,
239        }) {
240            VfsResponse::Err(BAD_HANDLE) => {
241                pass("dircap: revoking a parent dir reaps file handles opened below it")
242            }
243            _ => fail("dircap: revoking a parent dir reaps file handles opened below it"),
244        }
245        match vfs_req(&VfsRequest::CloseFile { file }) {
246            VfsResponse::Err(BAD_HANDLE) => {
247                pass("dircap: a revoked file handle cannot be closed into success")
248            }
249            _ => fail("dircap: a revoked file handle cannot be closed into success"),
250        }
251    }
252}

254// ── The migration is only real once paths stop working ───────────────────────
255
256fn seal_and_prove_paths_are_refused(work: ViDirHandle) {
257    let read_file_grant_ready = matches!(
258        vfs_req(&VfsRequest::Write {
259            path: READ_FILE_GRANT_SOURCE,
260            content: b"grant-copy-bytes"
261        }),
262        VfsResponse::Ok
263    );
264
265    match grant_io::read_file_into_short_grant(READ_FILE_GRANT_SOURCE, 5) {
266        Ok((grant, bytes))
267            if bytes == grant.len() && grant_io::grant_prefix_equals(&grant, b"grant") =>
268        {
269            pass("grant: ReadFileGrant clamps to grant length");
270        }
271        _ => fail("grant: ReadFileGrant clamps to grant length"),
272    }
273
274    let read_file_grant = match grant_io::read_file_into_grant(READ_FILE_GRANT_SOURCE) {
275        Ok((grant, bytes))
276            if bytes == grant.len()
277                && bytes > 0
278                && grant_io::grant_prefix_equals(&grant, b"grant-copy-bytes") =>
279        {
280            pass("grant: ReadFileGrant copies nonzero bytes");
281            Some(grant)
282        }
283        _ => {
284            fail("grant: ReadFileGrant copies nonzero bytes");
285            None
286        }
287    };
288
289    match vfs_req(&VfsRequest::GetFile("/tmp/volatile.txt")) {
290        VfsResponse::DataPtr { ptr, len } if ptr != 0 && len == 8 => {
291            pass("dircap: GetFile returns a nonempty pointer before sealing");
292        }
293        _ => fail("dircap: GetFile returns a nonempty pointer before sealing"),
294    }
295
296    let sealed_file = open_file_handle(work, "note.txt");
297
298    match vfs_req(&VfsRequest::SealPaths) {
299        VfsResponse::Ok => pass("dircap: the cell gives up naming paths"),
300        _ => {
301            fail("dircap: the cell gives up naming paths");
302            return;
303        }
304    }
305
306    // The criterion this whole phase is measured by.
307    match vfs_req(&VfsRequest::Write {
308        path: "/tmp/after_seal.txt",
309        content: b"should not land",
310    }) {
311        VfsResponse::Err(DENIED) => pass("dircap: Write{path} is refused after sealing"),
312        _ => fail("dircap: Write{path} is refused after sealing"),
313    }
314    // Nothing landed, either — the refusal is before the backend, not a reply
315    // written after the fact.
316    match vfs_req(&VfsRequest::StatAt {
317        dir: work,
318        name: "after_seal.txt",
319    }) {
320        VfsResponse::Err(_) => pass("dircap: the refused write left nothing behind"),
321        _ => fail("dircap: the refused write left nothing behind"),
322    }
323
324    // Every other path-addressed operation, so the refusal is not one arm.
325    for (req, msg) in [
326        (
327            VfsRequest::GetFile("/tmp/volatile.txt"),
328            "dircap: GetFile is refused after sealing",
329        ),
330        (
331            VfsRequest::Stat("/tmp/volatile.txt"),
332            "dircap: Stat is refused after sealing",
333        ),
334        (
335            VfsRequest::ListDir("/tmp"),
336            "dircap: ListDir is refused after sealing",
337        ),
338        (
339            VfsRequest::Unlink("/tmp/volatile.txt"),
340            "dircap: Unlink is refused after sealing",
341        ),
342        (
343            VfsRequest::Mkdir("/tmp/after_seal"),
344            "dircap: Mkdir is refused after sealing",
345        ),
346        (
347            VfsRequest::ReadAsync {
348                path: "/tmp/volatile.txt",
349            },
350            "dircap: ReadAsync is refused after sealing",
351        ),
352        (
353            VfsRequest::OpenRootDir { path: "/" },
354            "dircap: widening by acquiring a new root is refused after sealing",
355        ),
356    ] {
357        match vfs_req(&req) {
358            VfsResponse::Err(DENIED) => pass(msg),
359            _ => fail(msg),
360        }
361    }
362
363    match (read_file_grant_ready, read_file_grant.as_ref()) {
364        (true, Some(grant)) => match vfs_req(&VfsRequest::ReadFileGrant {
365            path: READ_FILE_GRANT_SOURCE,
366            grant: grant.id(),
367            max: grant.len(),
368        }) {
369            VfsResponse::Err(DENIED) => pass("grant: ReadFileGrant is refused after sealing"),
370            _ => fail("grant: ReadFileGrant is refused after sealing"),
371        },
372        _ => fail("grant: ReadFileGrant is refused after sealing"),
373    }
374
375    // And the cell still works: sealing removed one way of naming things, not
376    // the cell's access to what it holds.
377    if let Some(file) = sealed_file {
378        match vfs_req(&VfsRequest::ReadFileHandle {
379            file,
380            offset: 0,
381            max: 2,
382        }) {
383            VfsResponse::Data(b"ca") => pass("dircap: ReadFileHandle still works after sealing"),
384            _ => fail("dircap: ReadFileHandle still works after sealing"),
385        }
386        match vfs_req(&VfsRequest::ReadFileHandle {
387            file,
388            offset: 99,
389            max: 1,
390        }) {
391            VfsResponse::Data(b"") => {
392                pass("dircap: ReadFileHandle returns EOF for an oversized offset")
393            }
394            _ => fail("dircap: ReadFileHandle returns EOF for an oversized offset"),
395        }
396        match vfs_req(&VfsRequest::ReadFileHandle {
397            file,
398            offset: 0,
399            max: 0,
400        }) {
401            VfsResponse::Data(b"") => {
402                pass("dircap: ReadFileHandle returns EOF for a zero-length request")
403            }
404            _ => fail("dircap: ReadFileHandle returns EOF for a zero-length request"),
405        }
406        match vfs_req(&VfsRequest::CloseFile { file }) {
407            VfsResponse::Ok => pass("dircap: CloseFile closes a live file handle"),
408            _ => fail("dircap: CloseFile closes a live file handle"),
409        }
410        match vfs_req(&VfsRequest::ReadFileHandle {
411            file,
412            offset: 0,
413            max: 1,
414        }) {
415            VfsResponse::Err(BAD_HANDLE) => pass("dircap: a closed file handle is stale"),
416            _ => fail("dircap: a closed file handle is stale"),
417        }
418    }
419    match vfs_req(&VfsRequest::WriteAt {
420        dir: work,
421        name: "sealed.txt",
422        content: b"ok",
423    }) {
424        VfsResponse::Ok => pass("dircap: handle operations still work after sealing"),
425        _ => fail("dircap: handle operations still work after sealing"),
426    }
427    match vfs_req(&VfsRequest::UnlinkAt {
428        dir: work,
429        name: "sealed.txt",
430    }) {
431        VfsResponse::Ok => pass("dircap: UnlinkAt still works after sealing"),
432        _ => fail("dircap: UnlinkAt still works after sealing"),
433    }
434}

436fn open_file_handle(dir: ViDirHandle, name: &str) -> Option<ViVfsFileHandle> {
437    match vfs_req(&VfsRequest::OpenFileAt { dir, name }) {
438        VfsResponse::FileHandle(file) => {
439            pass("dircap: OpenFileAt returns a file handle for a file");
440            Some(file)
441        }
442        _ => {
443            fail("dircap: OpenFileAt returns a file handle for a file");
444            None
445        }
446    }
