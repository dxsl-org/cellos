1//! VFS integration test cell.
2//!
3//! Runs automated test scenarios against the VFS service and prints PASS/FAIL
4//! for each.  Exit code 0 = all pass, 1 = at least one failure.
5//!
6//! Spawn with: `spawn /bin/vfs-test` from the shell.
7//!
8//! All paths use /tmp (RamFS) so the cell runs with or without a block device.
9//! The quota tracker in dispatch.rs charges every write regardless of backend,
10//! so /tmp quota tests are equivalent to /data quota tests.
11

38/// Resolve the live VFS service tid via the service registry.
39/// Spins (yield-looping) until init has registered vfs — safe because init
40/// spawns vfs before vfs-test and vfs registers itself before yielding.
41fn vfs_tid() -> usize {
42    use api::syscall::service;
43    loop {
44        if let Some(tid) = ostd::syscall::sys_lookup_service(service::VFS) {
45            return tid;
46        }
47        ostd::task::yield_now();
48    }
49}

56fn vfs_req(req: &api::ipc::VfsRequest<'_>) -> api::ipc::VfsResponse<'static> {
57    let mut send_buf = [0u8; api::ipc::IPC_BUF_SIZE];
58    let n = api::ipc::encode(req, &mut send_buf)
59        .map(|s| s.len())
60        .unwrap_or(0);
61    vfs_raw(&send_buf[..n])
62}
63
64/// Send bytes to the VFS exactly as given, without encoding them first.
65///
66/// The only way to put a message on the wire that the request type cannot
67/// express — an invalid-UTF-8 name, for instance, which Rust will not let a
68/// `&str` hold but a hostile sender has no trouble producing.
69fn vfs_raw(msg: &[u8]) -> api::ipc::VfsResponse<'static> {
70    ostd::syscall::sys_send(vfs_tid(), msg);
71    // Leak the recv buffer so VfsResponse::Data borrows from it safely.
72    // This is fine in a test cell that runs and exits.
73    let buf: &'static mut [u8; api::ipc::IPC_BUF_SIZE] =
74        alloc::boxed::Box::leak(alloc::boxed::Box::new([0u8; api::ipc::IPC_BUF_SIZE]));
75    match ostd::syscall::sys_recv(0, buf) {
76        ostd::syscall::SyscallResult::Ok(_) => api::ipc::decode::<api::ipc::VfsResponse>(buf)
77            .unwrap_or(api::ipc::VfsResponse::Err(0xFE)),
78        _ => api::ipc::VfsResponse::Err(0xFD),
79    }
80}
81
82fn pass(msg: &str) {
83    PASSED.fetch_add(1, Ordering::Relaxed);
84    ostd::io::print("[PASS] ");
85    ostd::io::println(msg);
86}
87
88fn fail(msg: &str) {
89    FAILED.fetch_add(1, Ordering::Relaxed);
90    ostd::io::print("[FAIL] ");
91    ostd::io::println(msg);
92}
93
94macro_rules! assert_ok {
95    ($req:expr, $msg:literal) => {
96        match vfs_req(&$req) {
97            api::ipc::VfsResponse::Ok => pass($msg),
98            _ => fail($msg),
99        }
100    };
101}
102
103macro_rules! assert_err {
104    ($req:expr, $code:expr, $msg:literal) => {
105        match vfs_req(&$req) {
106            api::ipc::VfsResponse::Err(c) if c == $code => pass($msg),
107            api::ipc::VfsResponse::Err(c) => {
108                ostd::io::print("[FAIL] ");
109                ostd::io::print($msg);
110                ostd::io::print(" — wrong code: ");
111                ostd::io::print_usize(c as usize);
112                ostd::io::println("");
113                FAILED.fetch_add(1, Ordering::Relaxed);
114            }
115            _ => fail($msg),
116        }
117    };
118}

122/// 1. File lifecycle: write → stat → unlink → stat-gone.
123fn test_file_lifecycle() {
124    assert_ok!(
125        api::ipc::VfsRequest::Write {
126            path: "/tmp/test_lifecycle.txt",
127            content: b"hello world"
128        },
129        "write /tmp/test_lifecycle.txt"
130    );
131
132    match vfs_req(&api::ipc::VfsRequest::Stat("/tmp/test_lifecycle.txt")) {
133        api::ipc::VfsResponse::Stat {
134            size: 11,
135            is_dir: false,
136        } => pass("stat size=11 is_file"),
137        _ => fail("stat after write"),
138    }
139
140    assert_ok!(
141        api::ipc::VfsRequest::Unlink("/tmp/test_lifecycle.txt"),
142        "unlink /tmp/test_lifecycle.txt"
143    );
144
145    match vfs_req(&api::ipc::VfsRequest::Stat("/tmp/test_lifecycle.txt")) {
146        api::ipc::VfsResponse::Err(_) => pass("stat after unlink returns Err"),
147        _ => fail("stat after unlink should return Err"),
148    }
149}

152fn test_directory_ops() {
153    assert_ok!(
154        api::ipc::VfsRequest::Mkdir("/tmp/testdir"),
155        "mkdir /tmp/testdir"

177    assert_ok!(
178        api::ipc::VfsRequest::Rmdir("/tmp/testdir"),
179        "rmdir /tmp/testdir after cleanup"

183/// 3. Access control: write to /bin/ → PermissionDenied (Err 3).
184fn test_access_control() {
185    assert_err!(
186        api::ipc::VfsRequest::Write {
187            path: "/bin/evil",
188            content: b"hack"
189        },
190        3,
191        "write /bin/ returns PermissionDenied"
192    );
193
194    assert_ok!(
195        api::ipc::VfsRequest::Write {
196            path: "/tmp/access_ok.txt",
197            content: b"ok"
198        },
199        "write /tmp/ still works"
200    );
201    let _ = vfs_req(&api::ipc::VfsRequest::Unlink("/tmp/access_ok.txt"));
202
203    // ── Destructive-op authorization ─────────────────────────────────────────
204    //
205    // `Unlink`/`Rmdir`/`RmdirRecursive` used to skip `AccessTable` entirely: they
206    // called straight into the backend, so the `/` rule (allow_write_all: false)
207    // was never consulted on any delete path.
208    //
209    // `/readme.txt` is the decisive fixture — a real pre-seeded RamFS file under
210    // the read-only `/` prefix, on a backend that DOES implement `unlink`. Before
211    // the gate it was genuinely deleted (Ok); now it must be refused and survive.
212    // (`/bin/` is a weaker case: BinOverlay refuses every mutating call, so a
213    // delete there already failed at the backend with Err(1) — that arm proves
214    // ordering, not that a deletion was ever possible.)
215    assert_err!(
216        api::ipc::VfsRequest::Unlink("/readme.txt"),
217        3,
218        "unlink /readme.txt (read-only / prefix) → PermissionDenied"
219    );
220    match vfs_req(&api::ipc::VfsRequest::Stat("/readme.txt")) {
221        api::ipc::VfsResponse::Stat { is_dir: false, .. } => {
222            pass("/readme.txt survived the refused unlink")
223        }
224        _ => fail("/readme.txt survived the refused unlink"),
225    }
226
227    // Non-existent paths under `/`: the point is WHICH error comes back. Err(3)
228    // proves authorization ran BEFORE the backend; the pre-gate code reached the
229    // backend and reported Err(1) (not found). A non-existent target keeps this
230    // assertion non-destructive if it is ever run against an ungated build.
231    assert_err!(
232        api::ipc::VfsRequest::Rmdir("/no_such_dir"),
233        3,
234        "rmdir under / → PermissionDenied (authorized before backend)"
235    );
236    // Also proves the check precedes `collect_dir_bytes`, so an unauthorized
237    // caller cannot use this op to probe subtree sizes.
238    assert_err!(
239        api::ipc::VfsRequest::RmdirRecursive("/no_such_dir"),
240        3,
241        "rmdir_recursive under / → PermissionDenied (before subtree walk)"
242    );
243
244    // `WriteGrant` is fail-closed until cap→path routing exists: it cannot resolve
245    // a path, so it cannot authorize, so it refuses. It used to drain the grant,
246    // drop the bytes, and reply GrantDone — success for a write that never landed.
247    // grant id 0 is never valid; pre-change this reached sys_grant_slice and
248    // returned Err(1), so Err(3) proves the refusal happens first.
249    assert_err!(
250        api::ipc::VfsRequest::WriteGrant {
251            cap: 0,
252            offset: 0,
253            grant: 0,
254            bytes: 1
255        },
256        3,
257        "WriteGrant is fail-closed (no cap→path routing yet)"
258    );
259
260    match grant_io::unknown_cap_read_grant_returns_zero(5, 4) {
261        Ok(()) => pass("ReadGrant unknown cap returns zero bytes without faulting"),
262        Err(_) => fail("ReadGrant unknown cap returns zero bytes without faulting"),
263    }
264    match grant_io::unknown_cap_read_grant_returns_zero(0, 0) {
265        Ok(()) => pass("ReadGrant unknown cap returns zero bytes for size=0"),
266        Err(_) => fail("ReadGrant unknown cap returns zero bytes for size=0"),
267    }
268    match grant_io::unknown_cap_read_grant_returns_zero(u64::MAX, 4) {
269        Ok(()) => pass("ReadGrant unknown cap returns zero bytes for oversized offset"),
270        Err(_) => fail("ReadGrant unknown cap returns zero bytes for oversized offset"),
271    }
272}

274/// 4. Async read protocol: ReadAsync → PendingHandle → Poll → Data.
275fn test_async_read() {
276    assert_ok!(
277        api::ipc::VfsRequest::Write {
278            path: "/tmp/async_test.txt",
279            content: b"async content"
280        },
281        "write file for async read"
282    );
283
284    let handle = match vfs_req(&api::ipc::VfsRequest::ReadAsync {
285        path: "/tmp/async_test.txt",
286    }) {
287        api::ipc::VfsResponse::PendingHandle(h) => {
288            pass("ReadAsync returns PendingHandle");
289            h
290        }
291        _ => {
292            fail("ReadAsync did not return PendingHandle");
293            0
294        }
295    };
296
297    if handle != 0 {
298        match vfs_req(&api::ipc::VfsRequest::Poll { handle }) {
299            api::ipc::VfsResponse::Data(bytes) => {
300                if bytes.starts_with(b"async content") {
301                    pass("Poll returns correct data");
302                } else {
303                    fail("Poll returned wrong data");
304                }
305            }
306            _ => fail("Poll did not return Data"),
307        }
308
309        assert_err!(
310            api::ipc::VfsRequest::Poll { handle },
311            4,
312            "Poll stale handle returns Err"
313        );
314    }
315
316    let _ = vfs_req(&api::ipc::VfsRequest::Unlink("/tmp/async_test.txt"));
317}

365/// 8. Quota enforcement (Err 2). Only built with `test-hooks`, where the VFS
366/// uses a 1.1 KiB quota.  All paths use /tmp (RamFS) so no block device is needed.
367/// The QuotaTracker in dispatch.rs charges every successful write regardless of
368/// which backend path is used.
369///
370/// Chunk size is 400 B so the encoded VfsRequest::Write fits inside the 512 B
371/// IPC buffer (≈415 B on the wire: 1+1+11+2+400).
372///
373/// NOTE: test_ramfs writes 8 bytes (/tmp/volatile.txt) without cleanup.
374/// Total before this test: 8 B (from test_ramfs).
375/// 8 + 400 = 408 ≤ 1100  ✓
376/// 408 + 400 = 808 ≤ 1100  ✓
377/// 808 + 400 = 1208 > 1100  → Err(2) ✓
378#[cfg(feature = "test-hooks")]
379fn test_quota_limit() {
380    let chunk = [b'q'; 400];
381    assert_ok!(
382        api::ipc::VfsRequest::Write {
383            path: "/tmp/q1.bin",
384            content: &chunk
385        },
386        "quota write 1 (400B) fits"
387    );
388    assert_ok!(
389        api::ipc::VfsRequest::Write {
390            path: "/tmp/q2.bin",
391            content: &chunk
392        },
393        "quota write 2 (800B total) fits"
394    );
395    // Third write → 1208 total > 1100 → quota exceeded.
396    assert_err!(
397        api::ipc::VfsRequest::Write {
398            path: "/tmp/q3.bin",
399            content: &chunk
400        },
401        2,
402        "quota write 3 exceeds 1.1KiB limit → Err(2)"
403    );
404    // Releasing q1 frees 800B; q3 (800B) now fits.
405    assert_ok!(
406        api::ipc::VfsRequest::Unlink("/tmp/q1.bin"),
407        "unlink q1 releases quota"
408    );
409    assert_ok!(
410        api::ipc::VfsRequest::Write {
411            path: "/tmp/q3.bin",
412            content: &chunk
413        },
414        "quota write after release succeeds"
415    );
416    // Cleanup: net 0 delta (q2+q3=1600B remain, but test_rmdir starts fresh).
417    let _ = vfs_req(&api::ipc::VfsRequest::Unlink("/tmp/q2.bin"));
418    let _ = vfs_req(&api::ipc::VfsRequest::Unlink("/tmp/q3.bin"));
419}

421/// 9. RmdirRecursive releases quota (test-hooks only, 1.1 KiB quota).
422/// After test_quota_limit cleanup: 8 B used (volatile.txt).
423/// 8 + 400 = 408 ≤ 1100  ✓
424/// 408 + 400 = 808 ≤ 1100  ✓
425/// 808 + 300 = 1108 > 1100  → Err(2) ✓   (overflow check before delete)
426/// RmdirRecursive → releases 800B → back to 8B
427/// 8 + 300 = 308 ≤ 1100  ✓   (write succeeds after delete)
428#[cfg(feature = "test-hooks")]
429fn test_rmdir_recursive_quota() {
430    let chunk = [b'r'; 400];
431    assert_ok!(
432        api::ipc::VfsRequest::Mkdir("/tmp/rdir_q"),
433        "rdir-quota: mkdir"
434    );
435    assert_ok!(
436        api::ipc::VfsRequest::Write {
437            path: "/tmp/rdir_q/a.bin",
438            content: &chunk
439        },
440        "rdir-quota: write a.bin (400B)"
441    );
442    assert_ok!(
443        api::ipc::VfsRequest::Write {
444            path: "/tmp/rdir_q/b.bin",
445            content: &chunk
446        },
447        "rdir-quota: write b.bin (800B total)"
448    );
449    let small = [b'x'; 300];
450    assert_err!(
451        api::ipc::VfsRequest::Write {
452            path: "/tmp/rdir_quota_overflow.bin",
453            content: &small
454        },
455        2,
456        "rdir-quota: overflow write correctly blocked before delete"
457    );
458    assert_ok!(
459        api::ipc::VfsRequest::RmdirRecursive("/tmp/rdir_q"),
460        "rdir-quota: RmdirRecursive releases 800B"
461    );
462    assert_ok!(
463        api::ipc::VfsRequest::Write {
464            path: "/tmp/rdir_quota_ok.bin",
465            content: &small
466        },
467        "rdir-quota: write after recursive delete succeeds (quota freed)"
468    );
469    let _ = vfs_req(&api::ipc::VfsRequest::Unlink("/tmp/rdir_quota_ok.bin"));
470}

476fn cell_main() {
477    ostd::io::println("[vfs-test] Starting VFS integration test suite...");
478
479    test_file_lifecycle();
480    test_directory_ops();
481    test_access_control();
482    test_async_read();
483    test_ramfs();
484    test_stat_dir();
485    test_edge_cases();
486    #[cfg(feature = "test-hooks")]
487    test_quota_limit();
488    #[cfg(feature = "test-hooks")]
489    test_rmdir_recursive_quota();
490    // Last: it ends by giving up path strings for good, so nothing above it
491    // could run afterwards.
492    dircap::run();
493
494    let passed = PASSED.load(Ordering::Relaxed);
495    let failed = FAILED.load(Ordering::Relaxed);
496
497    ostd::io::println("");
498    ostd::io::print("[vfs-test] Results: ");
499    ostd::io::print_usize(passed as usize);
500    ostd::io::print(" PASS, ");
501    ostd::io::print_usize(failed as usize);
502    ostd::io::println(" FAIL");
503
504    if failed == 0 {
505        ostd::io::println("[vfs-test] ALL TESTS PASSED");
506        ostd::syscall::sys_exit(0);
507    } else {
508        ostd::io::println("[vfs-test] FAILURES DETECTED");
509        ostd::syscall::sys_exit(1);
510    }
511}
