7// Declares spawn capability; the kernel grants SpawnCap at spawn.
8// gpio/uart: held for DELEGATION only (P2 monotonic downgrade intersects a
9// child's manifest with the spawner's caps) — the shell never opens MMIO
10// itself, but interactively-spawned peripheral demos (periph-demo, robot-demo,
11// sensor-demo, …) would otherwise lose their gpio/uart caps and fail with
12// PermissionDenied. The interactive operator at the shell IS the robot
13// operator, so shell-level peripheral delegation matches the trust model.
14api::declare_manifest!(
15    block_io = false,
16    network = false,
17    spawn = true,
18    gpio = true,
19    uart = true
20);

99#[cfg(not(test))]
100#[no_mangle]
101pub fn main() {
102    #[cfg(feature = "shell_test")]
103    shell_test::run();
104
105    #[cfg(not(feature = "shell_test"))]
106    {
107        let _ = ostd::syscall::sys_log("DEBUG: Shell Started (Async Mode)\n");
108        // Claim keyboard focus so VirtIO keyboard events are routed here via
109        // the input service (fb_console keyboard relay).  Spin-wait for the
110        // input service to come online — it races with shell at boot.
111        // Don't spin indefinitely — attempt up to 50 times (~25 seconds),
112        // then proceed without focus (UART fallback still works).
113        for _ in 0..50 {
114            if ostd::input::request_focus() {
115                break;
116            }
117            ostd::task::yield_now();
118        }
119        let _ = ostd::syscall::sys_log("DEBUG: Shell focus acquired (or timed out)\n");
120        let mut shell = ViShell::new();
121        ostd::executor::block_on(shell.run());
122    }
123}
