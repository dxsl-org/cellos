96#[cfg(not(test))]
97#[no_mangle]
98pub fn main() {
99    #[cfg(feature = "shell_test")]
100    shell_test::run();
101
102    #[cfg(not(feature = "shell_test"))]
103    {
104        let _ = ostd::syscall::sys_log("DEBUG: Shell Started (Async Mode)\n");
105        // Claim keyboard focus so VirtIO keyboard events are routed here via
106        // the input service (fb_console keyboard relay).  Spin-wait for the
107        // input service to come online — it races with shell at boot.
108        // Don't spin indefinitely — attempt up to 50 times (~25 seconds),
109        // then proceed without focus (UART fallback still works).
110        for _ in 0..50 {
111            if ostd::input::request_focus() {
112                break;
113            }
114            ostd::task::yield_now();
115        }
116        let _ = ostd::syscall::sys_log("DEBUG: Shell focus acquired (or timed out)\n");
117        let mut shell = ViShell::new();
118        ostd::executor::block_on(shell.run());
119    }
120}
