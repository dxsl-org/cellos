1//! Process-global shell state: output capture, pipe stdin, variables, functions.
2//!
3//! Everything here used to be a `static mut` or an `UnsafeCell` behind a hand-written
4//! `unsafe impl Sync`, justified by "the shell is a single task". That justification
5//! is true today and unenforceable tomorrow: nothing in the type system stops a
6//! second task from calling `shell_print`. These are `spin::Mutex` and atomics
7//! instead, which give the same `Sync` guarantee as a *checked* fact and let the
8//! crate carry `#![forbid(unsafe_code)]` (F1, Spec 16 §6).
9//!
10//! Capacity and truncation limits are byte-for-byte the ones the fixed-size static
11//! arrays imposed, so scripts that relied on them behave identically.
12
13extern crate alloc;
14
15use alloc::string::String;
16use alloc::vec::Vec;
17use core::sync::atomic::{AtomicBool, Ordering};
18use ostd::sync::Mutex;
19
20/// Variable slots. Setting a 17th variable is silently dropped, as before.
21const MAX_VARS: usize = 16;
22/// Shell function slots. Defining a 9th function is silently dropped, as before.
23const MAX_FUNS: usize = 8;
24const KEY_LIMIT: usize = 31;
25const VALUE_LIMIT: usize = 127;
26const BODY_LIMIT: usize = 479;
27
28/// Stack of in-flight output captures; empty means "write to the console".
29///
30/// A stack rather than a saved-and-restored slot: nested captures (a `$(...)`
31/// inside a pipeline stage) push and pop, and the innermost one wins.
32static SINK: Mutex<Vec<Vec<u8>>> = Mutex::new(Vec::new());
33/// Bytes a pipe made available on stdin for the command currently dispatching.
34static STDIN: Mutex<Vec<u8>> = Mutex::new(Vec::new());
35static VARS: Mutex<Vec<(Vec<u8>, Vec<u8>)>> = Mutex::new(Vec::new());
36static FUNS: Mutex<Vec<(Vec<u8>, Vec<u8>)>> = Mutex::new(Vec::new());
37static EXIT_REQUEST: Mutex<Option<i32>> = Mutex::new(None);
38static LOOP_SIGNAL: Mutex<LoopSignal> = Mutex::new(LoopSignal::None);
39static BG_SPAWN: AtomicBool = AtomicBool::new(false);
40
41/// Truncate to `limit` bytes, then stop at the first NUL.
42///
43/// Both steps mirror the retired fixed-array encoding, which wrote at most
44/// `limit` bytes into the slot and NUL-terminated it — a value containing a NUL
45/// was therefore already cut short on read.
46fn slot_bytes(text: &str, limit: usize) -> Vec<u8> {
47    let bytes = &text.as_bytes()[..text.len().min(limit)];
48    let end = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
49    bytes[..end].to_vec()
50}
51
52/// Decode a slot the way the old reader did: invalid UTF-8 reads back as absent.
53fn slot_str(bytes: &[u8]) -> Option<String> {
54    core::str::from_utf8(bytes).ok().map(String::from)
55}
56
57fn put(store: &mut Vec<(Vec<u8>, Vec<u8>)>, key: Vec<u8>, value: Vec<u8>, capacity: usize) {
58    if let Some(slot) = store.iter_mut().find(|(k, _)| *k == key) {
59        slot.1 = value;
60    } else if store.len() < capacity {
61        store.push((key, value));
62    }
63}

65// ── Output sink ──────────────────────────────────────────────────────────────
66
67/// Route a string to the innermost capture buffer, or to the console if none.
68pub fn write_out(s: &str) {
69    {
70        let mut sink = SINK.lock();
71        if let Some(buffer) = sink.last_mut() {
72            buffer.extend_from_slice(s.as_bytes());
73            return;
74        }
75    } // the console write must not hold the sink lock
76    ostd::io::print(s);
77}
78
79/// RAII capture of everything `write_out` receives until `finish` or drop (Law 8).
80///
81/// Dropping without `finish` discards the captured bytes — that is the panic /
82/// early-return path, and it must still pop the stack or every later write would
83/// vanish into an orphaned buffer.
84pub struct CaptureGuard {
85    finished: bool,
86}
87
88impl CaptureGuard {
89    pub fn new() -> Self {
90        SINK.lock().push(Vec::new());
91        Self { finished: false }
92    }
93
94    /// Pop this capture and return its bytes.
95    pub fn finish(mut self) -> Vec<u8> {
96        self.finished = true;
97        SINK.lock().pop().unwrap_or_default()
98    }
99}
100
101impl Drop for CaptureGuard {
102    fn drop(&mut self) {
103        if !self.finished {
104            SINK.lock().pop();
105        }
106    }
107}

109// ── Pipe stdin ───────────────────────────────────────────────────────────────
110
111/// Publish `data` as the stdin of the command about to be dispatched.
112pub fn set_stdin(data: &[u8]) {
113    let mut stdin = STDIN.lock();
114    stdin.clear();
115    stdin.extend_from_slice(data);
116}
117
118pub fn clear_stdin() {
119    STDIN.lock().clear();
120}
121
122/// A copy of the current pipe stdin, empty when no pipe is active.
123///
124/// Returns owned bytes rather than a borrow: a borrow out of the lock cannot
125/// outlive the guard, and the pointer-laundering that made the old `&'static [u8]`
126/// possible is exactly what F1 forbids. Pipe payloads are shell-sized.
127pub fn stdin_bytes() -> Vec<u8> {
128    STDIN.lock().clone()
129}

131// ── Variables ────────────────────────────────────────────────────────────────
132
133pub fn set_var(key: &str, value: &str) {
134    put(
135        &mut VARS.lock(),
136        slot_bytes(key, KEY_LIMIT),
137        slot_bytes(value, VALUE_LIMIT),
138        MAX_VARS,
139    );
140}
141
142pub fn unset_var(key: &str) {
143    let key = slot_bytes(key, KEY_LIMIT);
144    VARS.lock().retain(|(k, _)| *k != key);
145}
146
147pub fn get_var(key: &str) -> Option<String> {
148    let key = slot_bytes(key, KEY_LIMIT);
149    let vars = VARS.lock();
150    vars.iter()
151        .find(|(k, _)| *k == key)
152        .and_then(|(_, v)| slot_str(v))
153}
154
155// ── Shell functions ──────────────────────────────────────────────────────────
156
157pub fn define_function(name: &str, body: &str) {
158    put(
159        &mut FUNS.lock(),
160        slot_bytes(name, KEY_LIMIT),
161        slot_bytes(body, BODY_LIMIT),
162        MAX_FUNS,
163    );
164}
165
166pub fn get_function(name: &str) -> Option<String> {
167    let name = slot_bytes(name, KEY_LIMIT);
168    let funs = FUNS.lock();
169    funs.iter()
170        .find(|(k, _)| *k == name)
171        .and_then(|(_, b)| slot_str(b))
172}

174// ── Control signals ──────────────────────────────────────────────────────────
175
176#[derive(Clone, Copy, PartialEq, Eq)]
177pub enum LoopSignal {
178    None,
179    Break,
180    Continue,
181}
182
183pub fn set_loop_signal(signal: LoopSignal) {
184    *LOOP_SIGNAL.lock() = signal;
185}
186
187/// Read and clear the pending loop signal.
188pub fn take_loop_signal() -> LoopSignal {
189    let mut signal = LOOP_SIGNAL.lock();
190    core::mem::replace(&mut signal, LoopSignal::None)
191}
192
193pub fn request_exit(code: i32) {
194    *EXIT_REQUEST.lock() = Some(code);
195}
196
197/// Read and clear a pending `exit` request.
198#[cfg(not(feature = "shell_test"))] // reason: only the REPL loop can honour an exit request
199pub fn take_exit_request() -> Option<i32> {
200    EXIT_REQUEST.lock().take()
201}

203/// True while executing the command of an `Ast::Background` (`cmd &`).
204///
205/// A backgrounded EXTERNAL cell (e.g. `httpd 9092 /file &`) must NOT be
206/// `sys_wait`'d — httpd loops forever, so waiting parks the shell in `sys_wait`
207/// indefinitely and no subsequent command runs (the symptom: a second `vwrite`
208/// after `httpd &` never executed, so httpd kept serving stale content). When
209/// set, `spawn_external` returns right after spawn. Built-ins are unaffected —
210/// they run synchronously in the shell task either way.
211pub fn bg_spawn() -> bool {
212    BG_SPAWN.load(Ordering::SeqCst)
213}
214
215pub fn set_bg_spawn(value: bool) {
216    BG_SPAWN.store(value, Ordering::SeqCst);
217}
