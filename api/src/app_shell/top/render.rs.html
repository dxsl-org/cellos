5pub fn render_frame(rows: &[TopRow], sort: SortKey, batch: bool, remaining: usize) {
6    if !batch {
7        ostd::io::print("\x1b[2J\x1b[1;1H");
8    }
9    let uptime = ostd::syscall::sys_get_time() / TIMER_HZ;
10    crate::executor::shell_print(&alloc::format!(
11        "top: uptime {}s  tasks {}  sort {}{}{}\n",
12        uptime,
13        rows.len(),
14        sort_label(sort),
15        if batch { "  batch" } else { "  interactive" },
16        if batch {
17            alloc::format!("  remaining {}", remaining)
18        } else {
19            alloc::string::String::from("  q to quit")
20        }
21    ));
22    crate::executor::shell_println("  PID  STATE      CPU%   HEAP      MEM       NAME");
23    crate::executor::shell_println("  ---  ---------  -----  --------  --------  ----------------");
24    for row in rows.iter().take(MAX_ROWS) {
25        crate::executor::shell_print(&alloc::format!(
26            "  {:<4} {:<10} {:>3}.{:01}  {:>8}  {:>8}  {}\n",
27            row.pid,
28            state_label(row.state),
29            row.cpu_tenths / 10,
30            row.cpu_tenths % 10,
31            format_bytes(row.heap_bytes),
32            format_bytes(row.mem_bytes),
33            row.name
34        ));
35    }
36}
