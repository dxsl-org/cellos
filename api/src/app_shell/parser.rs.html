1//! Shell command parser — tokenizes a line and builds an `Ast`.
2//!
3//! Supported syntax (v1.0):
4//!   - Simple command: `ls /bin`
5//!   - Pipeline: `cat /etc/hosts | grep 127`
6//!   - Output redirect: `echo hello > /tmp/a.txt`
7//!   - Input redirect: `cat < /tmp/a.txt`
8//!   - Append redirect: `echo hi >> /tmp/log.txt`
9//!   - Background: `sleep 10 &`
10//!   - Sequence: `echo a ; echo b`
11//!
12//! Intentionally simple: no subshells, no quoting beyond `"..."`, no globs.
13
14extern crate alloc;
15
16use alloc::string::String;
17use alloc::vec::Vec;
18
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum QuoteStyle {
21    None,
22    Single,
23    Double,
24}
25
26#[derive(Debug, Clone, PartialEq, Eq)]
27pub struct Word {
28    pub text: String,
29    pub quote: QuoteStyle,
30    pub segments: Vec<WordSegment>,
31}
32
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub struct WordSegment {
35    pub text: String,
36    pub quote: QuoteStyle,
37}
38
39impl Word {
40    #[cfg(test)]
41    fn new(text: String, quote: QuoteStyle) -> Self {
42        Self {
43            segments: alloc::vec![WordSegment {
44                text: text.clone(),
45                quote,
46            }],
47            text,
48            quote,
49        }
50    }
51
52    fn from_segments(segments: Vec<WordSegment>) -> Self {
53        let mut text = String::new();
54        for segment in &segments {
55            text.push_str(&segment.text);
56        }
57        let quote = if segments.len() == 1 {
58            segments[0].quote
59        } else {
60            QuoteStyle::None
61        };
62        Self {
63            text,
64            quote,
65            segments,
66        }
67    }
68
69    fn is_unquoted(&self) -> bool {
70        self.segments
71            .iter()
72            .all(|segment| segment.quote == QuoteStyle::None)
73    }
74}

76/// One redirect target.
77#[derive(Debug, Clone, PartialEq, Eq)]
78pub enum Redirect {
79    /// `> path`
80    StdoutTo(Word),
81    /// `>> path`
82    StdoutAppend(Word),
83    /// `< path`
84    StdinFrom(Word),
85    /// `2> path`
86    StderrTo(Word),

91pub struct Cmd {

112/// Top-level abstract syntax tree for one shell line.
113#[derive(Debug, Clone)]
114pub enum Ast {
115    /// Empty input.
116    Empty,
117    /// A single simple command.
118    Simple(Cmd),
119    /// `cmd1 | cmd2 | …` — pipeline of commands.
120    Pipeline(Vec<Cmd>),
121    /// `cmd &` — run in background.
122    Background(Cmd),
123    /// `cmd1 ; cmd2` — sequential execution.
124    Sequence(Vec<Ast>),
125    /// `cmd1 && cmd2` — run cmd2 only if cmd1 exits 0 (success).
126    And(alloc::boxed::Box<Ast>, alloc::boxed::Box<Ast>),
127    /// `cmd1 || cmd2` — run cmd2 only if cmd1 exits non-zero (failure).
128    Or(alloc::boxed::Box<Ast>, alloc::boxed::Box<Ast>),
129    /// `while COND; do BODY; done` — loop while COND exits 0.
130    While {
131        cond: alloc::boxed::Box<Ast>,
132        body: alloc::boxed::Box<Ast>,
133    },
134    /// `name() { body; }` — define a shell function.
135    ///
136    /// The body is stored as a reconstructed string; the executor registers it
137    /// in the function table so later invocations of `name` run the body.
138    FuncDef {
139        name: alloc::string::String,
140        body: alloc::string::String,
141    },
142    /// `case EXPR in pattern) BODY ;; … esac` — pattern-match dispatch.
143    ///
144    /// `expr` is the unexpanded word after `case`; expansion happens at
145    /// execute time.  Patterns are exact strings or `*` (wildcard).
146    /// Note: the whole statement must be on a single line or in a single
147    /// `source`-script line; mixing with `;`-sequences on the same line is
148    /// not supported (the outer `;` split would fragment the `;;` markers).
149    Case {
150        expr: alloc::string::String,
151        arms: alloc::vec::Vec<(alloc::string::String, alloc::boxed::Box<Ast>)>,
152    },
153    /// `for VAR in word1 word2 …; do BODY; done` — iterate over a word list.
154    ///
155    /// Sets `$VAR` to each word in order, runs BODY, then advances. `$VAR`
156    /// expansion in BODY uses the same static var store as `VAR=value`.
157    For {
158        var: alloc::string::String,
159        words: alloc::vec::Vec<alloc::string::String>,
160        body: alloc::boxed::Box<Ast>,
161    },
162    /// `if COND; then BODY; fi` — conditional execution.
163    ///
164    /// `cond` exit-code 0 → run `then_b`; non-zero → run `else_b` if present.
165    If {
166        cond: alloc::boxed::Box<Ast>,
167        then_b: alloc::boxed::Box<Ast>,
168        else_b: Option<alloc::boxed::Box<Ast>>,
169    },
170}

174/// Raw token before AST construction.
175#[derive(Debug, Clone, PartialEq, Eq)]
176enum Tok {
177    Word(Word),
178    Pipe,           // |
179    RedirectOut,    // >
180    RedirectAppend, // >>
181    RedirectIn,     // <
182    RedirectErr,    // 2>
183    Ampersand,      // &   (background marker — single &)
184    And,            // &&  (short-circuit AND)
185    Or,             // ||  (short-circuit OR)
186    Semicolon,      // ;
187    LBrace,         // {  (function body open)
188    RBrace,         // }  (function body close)
189    // ── Conditional keywords ─────────────────────────────────────────────────
190    // These variants are NEVER emitted by the tokenizer — `if`/`then`/`else`/`fi`
191    // always remain as Word tokens.  parse_if_stmt detects them by string
192    // comparison so they never silently disappear from external command arguments
193    // (e.g. `lua -e "if x then ... end"` must reach Lua intact).
194    #[allow(dead_code)] // reserved — kept for exhaustive match arms in parse_cmd
195    If,
196    #[allow(dead_code)] // reserved
197    Then,
198    #[allow(dead_code)] // reserved
199    Else,
200    #[allow(dead_code)] // reserved
201    Fi,
202}

204/// Tokenize a shell input line.
205///
206/// Handles:
207/// - Whitespace separation
208/// - Simple `"..."` double-quoted strings (no escape sequences)
209/// - Single-character operators: `|`, `<`, `>`, `&`, `;`
210/// - Two-character operators: `>>`, `2>`
211fn tokenize(line: &str) -> Vec<Tok> {
212    let mut tokens = Vec::new();
213    let mut chars = line.chars().peekable();
214    let mut current: Vec<WordSegment> = Vec::new();
215
216    fn append_segment(current: &mut Vec<WordSegment>, text: String, quote: QuoteStyle) {
217        if let Some(last) = current.last_mut() {
218            if last.quote == quote {
219                last.text.push_str(&text);
220                return;
221            }
222        }
223        current.push(WordSegment { text, quote });
224    }
225
226    fn push_word(tokens: &mut Vec<Tok>, current: &mut Vec<WordSegment>) {
227        if !current.is_empty() {
228            tokens.push(Tok::Word(Word::from_segments(core::mem::take(current))));
229        }
230    }
231
232    fn read_quoted(chars: &mut core::iter::Peekable<core::str::Chars<'_>>, end: char) -> String {
233        let mut text = String::new();
234        loop {
235            match chars.next() {
236                Some(ch) if ch == end => break,
237                Some(ch) => text.push(ch),
238                None => break,
239            }
240        }
241        text
242    }
243
244    macro_rules! flush {
245        () => {
246            push_word(&mut tokens, &mut current);
247        };
248    }
249
250    while let Some(c) = chars.next() {
251        match c {
252            ' ' | '\t' => {
253                flush!();
254            }
255            '"' | '\'' => {
256                let quote = if c == '"' {
257                    QuoteStyle::Double
258                } else {
259                    QuoteStyle::Single
260                };
261                let quoted = read_quoted(&mut chars, c);
262                append_segment(&mut current, quoted, quote);
263            }
264            '|' => {
265                flush!();
266                if chars.peek() == Some(&'|') {
267                    chars.next();
268                    tokens.push(Tok::Or);
269                } else {
270                    tokens.push(Tok::Pipe);
271                }
272            }
273            '&' => {
274                flush!();
275                if chars.peek() == Some(&'&') {
276                    chars.next();
277                    tokens.push(Tok::And);
278                } else {
279                    tokens.push(Tok::Ampersand);
280                }
281            }
282            ';' => {
283                flush!();
284                tokens.push(Tok::Semicolon);
285            }
286            '{' => {
287                flush!();
288                tokens.push(Tok::LBrace);
289            }
290            '}' => {
291                flush!();
292                tokens.push(Tok::RBrace);
293            }
294            '<' => {
295                flush!();
296                tokens.push(Tok::RedirectIn);
297            }
298            '>' => {
299                flush!();
300                if chars.peek() == Some(&'>') {
301                    chars.next();
302                    tokens.push(Tok::RedirectAppend);
303                } else {
304                    tokens.push(Tok::RedirectOut);
305                }
306            }
307            '2' if chars.peek() == Some(&'>') => {
308                // "2>" — only if current buffer is empty (i.e. not part of a word).
309                if current.is_empty() {
310                    chars.next();
311                    tokens.push(Tok::RedirectErr);
312                } else {
313                    append_segment(&mut current, String::from(c), QuoteStyle::None);
314                }
315            }
316            // `$(...)` — command substitution.  Consume everything up to the
317            // matching `)` into the current word so spaces inside don't split
318            // the token.  Nested `$( $() )` increments depth correctly.
319            '$' if chars.peek() == Some(&'(') => {
320                chars.next(); // consume '('
321                let mut substitution = String::from("$(");
322                let mut depth = 1usize;
323                loop {
324                    match chars.next() {
325                        None => break, // unclosed: pass through
326                        Some('(') => {
327                            depth += 1;
328                            substitution.push('(');
329                        }
330                        Some(')') => {
331                            depth -= 1;
332                            substitution.push(')');
333                            if depth == 0 {
334                                break;
335                            }
336                        }
337                        Some(ch) => {
338                            substitution.push(ch);
339                        }
340                    }
341                }
342                append_segment(&mut current, substitution, QuoteStyle::None);
343            }
344            other => {
345                append_segment(&mut current, String::from(other), QuoteStyle::None);
346            }
347        }
348    }
349    flush!();
350    // All tokens remain as their natural type — no keyword conversion.
351    // The if-statement parser detects `if`/`then`/`else`/`fi` by string
352    // comparison on Word tokens so they are never eaten from command arguments.
353    tokens
354}

358/// Parse a shell line into an `Ast`.
359///
360/// Returns `Ast::Empty` for blank input.
361pub fn parse(line: &str) -> Ast {
362    let tokens = tokenize(line.trim());
363    if tokens.is_empty() {
364        return Ast::Empty;
365    }
366
367    // `if...then...fi` must be parsed BEFORE semicolon splitting, because the
368    // semicolons inside an if-statement are structural (not sequence separators).
369    // Keywords remain as Word tokens (not converted) so they survive in external
370    // command argument strings (e.g. `lua -e "if x then ... end"`).
371    // `name() { body; }` — function definition.
372    // name() is a single Word token ending with "()" (no spaces inside).
373    if let Some(Tok::Word(w)) = tokens.first() {
374        if w.is_unquoted()
375            && w.text.ends_with("()")
376            && tokens.get(1) == Some(&Tok::LBrace)
377            && tokens.last() == Some(&Tok::RBrace)
378        {
379            let name = String::from(&w.text[..w.text.len() - 2]);
380            let body = tokens_to_string(&tokens[2..tokens.len() - 1]);
381            return Ast::FuncDef { name, body };
382        }
383    }
384
385    if is_kw_token(tokens.first(), "if") {
386        return parse_if_stmt(&tokens);
387    }
388    if is_kw_token(tokens.first(), "while") {
389        return parse_while_stmt(&tokens);
390    }
391    if is_kw_token(tokens.first(), "for") {
392        return parse_for_stmt(&tokens);
393    }
394    // `case` must be parsed before semicolon splitting because `;;` arm
395    // separators are also Semicolon tokens and would fragment the statement.
396    if is_kw_token(tokens.first(), "case") {
397        return parse_case_stmt(&tokens);
398    }
399
400    // Split on `;` into sub-sequences first.
401    let segments: Vec<&[Tok]> = split_on(&tokens, |t| t == &Tok::Semicolon);
402    if segments.len() > 1 {
403        let seq: Vec<Ast> = segments.iter().map(|seg| parse_pipeline(seg)).collect();
404        return Ast::Sequence(seq);
405    }
406
407    parse_pipeline(&tokens)
408}

410/// Parse a token sub-slice that may contain `;`-separated commands.
411///
412/// Equivalent to the main `parse()` body but operating on a pre-tokenized
413/// slice — used by `parse_if_stmt` to parse condition and body sections.
414fn parse_tokens(tokens: &[Tok]) -> Ast {
415    // Strip leading/trailing semicolons that linger from the structural split.
416    let start = tokens
417        .iter()
418        .position(|t| t != &Tok::Semicolon)
419        .unwrap_or(tokens.len());
420    let end = tokens
421        .iter()
422        .rposition(|t| t != &Tok::Semicolon)
423        .map(|i| i + 1)
424        .unwrap_or(0);
425    let tokens = &tokens[start..end];
426    if tokens.is_empty() {
427        return Ast::Empty;
428    }
429    let segments: Vec<&[Tok]> = split_on(tokens, |t| t == &Tok::Semicolon);
430    if segments.len() > 1 {
431        let seq: Vec<Ast> = segments.iter().map(|seg| parse_pipeline(seg)).collect();
432        return Ast::Sequence(seq);
433    }
434    parse_pipeline(tokens)
435}

437/// Parse `if COND; then BODY; fi` or `if COND; then BODY; else BODY; fi`.
438///
439/// Handles any number of semicolons around the keywords — the structure is
440/// determined by the `Then`, `Else`, and `Fi` token positions.
441/// Serialize a token slice back to a space-separated shell line.
442///
443/// Used by function definition to store the body as a re-parseable string.
444fn tokens_to_string(tokens: &[Tok]) -> String {
445    let parts: alloc::vec::Vec<&str> = tokens
446        .iter()
447        .map(|t| match t {
448            Tok::Word(w) => w.text.as_str(),
449            Tok::Pipe => "|",
450            Tok::And => "&&",
451            Tok::Or => "||",
452            Tok::Semicolon => ";",
453            Tok::Ampersand => "&",
454            Tok::RedirectOut => ">",
455            Tok::RedirectAppend => ">>",
456            Tok::RedirectIn => "<",
457            Tok::RedirectErr => "2>",
458            Tok::LBrace => "{",
459            Tok::RBrace => "}",
460            _ => "",
461        })
462        .collect();
463    parts.join(" ")
464}

466/// Helper: returns true when a token is the keyword word `w`.
467fn is_kw(tok: &Tok, w: &str) -> bool {
468    matches!(tok, Tok::Word(word) if word.is_unquoted() && word.text == w)
469}
470
471fn is_kw_token(tok: Option<&Tok>, w: &str) -> bool {
472    matches!(tok, Some(tok) if is_kw(tok, w))
473}

475/// Parse `while COND; do BODY; done`.
476///
477/// Keywords stay as `Word` tokens (no Tok variants) so `while`/`do`/`done`
478/// survive intact when used as external command arguments.  Malformed input
479/// (missing `do` or `done`) calls `parse_tokens` — NOT `parse()` — to avoid
480/// re-dispatching on the leading `while` and recursing infinitely.
481/// Parse `for VAR in word1 word2 …; do BODY; done`.
482///
483/// Keywords stay as `Word` tokens (same Phase N/O rule) so `for`/`in`/`do`/`done`
484/// survive as external command arguments.  Malformed input falls back to
485/// `parse_tokens` (not `parse()`) to prevent infinite recursion.
486fn parse_for_stmt(tokens: &[Tok]) -> Ast {
487    // tokens[1] = variable name; tokens[2] should be Word("in").
488    let var = match tokens.get(1) {
489        Some(Tok::Word(w)) if w.is_unquoted() && w.text != "in" => w.text.clone(),
490        _ => return parse_tokens(tokens),
491    };
492    let in_pos = match tokens.iter().position(|t| is_kw(t, "in")) {
493        Some(p) => p,
494        None => return parse_tokens(tokens),
495    };
496    let do_pos = tokens.iter().position(|t| is_kw(t, "do"));
497    let done_pos = tokens.iter().rposition(|t| is_kw(t, "done"));
498    let (dp, np) = match (do_pos, done_pos) {
499        (Some(d), Some(n)) if n > d => (d, n),
500        _ => return parse_tokens(tokens),
501    };
502    // Word list: tokens between `in` and `do`, stripping Semicolons.
503    let words: alloc::vec::Vec<alloc::string::String> = tokens[in_pos + 1..dp]
504        .iter()
505        .filter_map(|t| {
506            if let Tok::Word(w) = t {
507                Some(w.text.clone())
508            } else {
509                None
510            }
511        })
512        .collect();
513    let body = parse_tokens(&tokens[dp + 1..np]);
514    Ast::For {
515        var,
516        words,
517        body: alloc::boxed::Box::new(body),
518    }
519}

521/// Parse `case EXPR in pattern1) BODY ;; pattern2) BODY ;; esac`.
522///
523/// `;;` is two consecutive `Tok::Semicolon` tokens.  Patterns are the first
524/// token of each arm with a trailing `)` stripped.  `*` is a catch-all.
525/// Malformed input (missing `in` or `esac`) falls back to `parse_tokens`.
526fn parse_case_stmt(tokens: &[Tok]) -> Ast {
527    let in_pos = tokens.iter().position(|t| is_kw(t, "in"));
528    let esac_pos = tokens.iter().rposition(|t| is_kw(t, "esac"));
529    let (ip, ep) = match (in_pos, esac_pos) {
530        (Some(i), Some(e)) if e > i => (i, e),
531        _ => return parse_tokens(tokens),
532    };
533
534    // Expression: single word between `case` and `in`.
535    let expr = match tokens.get(1) {
536        Some(Tok::Word(w)) => w.text.clone(),
537        _ => String::new(),
538    };
539
540    // Arms: tokens[ip+1..ep] split on `;;` (two consecutive Semicolons).
541    let arm_tokens = &tokens[ip + 1..ep];
542    let mut arms: alloc::vec::Vec<(String, alloc::boxed::Box<Ast>)> = alloc::vec::Vec::new();
543    let mut start = 0;
544    let mut i = 0;
545    while i < arm_tokens.len() {
546        if i + 1 < arm_tokens.len()
547            && arm_tokens[i] == Tok::Semicolon
548            && arm_tokens[i + 1] == Tok::Semicolon
549        {
550            push_arm(&arm_tokens[start..i], &mut arms);
551            i += 2;
552            start = i;
553        } else {
554            i += 1;
555        }
556    }
557    push_arm(&arm_tokens[start..], &mut arms); // last arm (no trailing ;;)
558
559    Ast::Case { expr, arms }
560}
561
562fn push_arm(tokens: &[Tok], arms: &mut alloc::vec::Vec<(String, alloc::boxed::Box<Ast>)>) {
563    // Strip leading/trailing Semicolons left from the split.
564    let start = tokens
565        .iter()
566        .position(|t| t != &Tok::Semicolon)
567        .unwrap_or(tokens.len());
568    let tokens = &tokens[start..];
569    if let Some(Tok::Word(pat)) = tokens.first() {
570        let pattern = String::from(pat.text.trim_end_matches(')'));
571        let body = parse_tokens(&tokens[1..]);
572        arms.push((pattern, alloc::boxed::Box::new(body)));
573    }
574}

576fn parse_while_stmt(tokens: &[Tok]) -> Ast {
577    let do_pos = tokens.iter().position(|t| is_kw(t, "do"));
578    let done_pos = tokens.iter().rposition(|t| is_kw(t, "done"));
579    let (dp, np) = match (do_pos, done_pos) {
580        (Some(d), Some(n)) if n > d => (d, n),
581        _ => return parse_tokens(tokens), // malformed: fall back without infinite recursion
582    };
583    let cond = parse_tokens(&tokens[1..dp]);
584    let body = parse_tokens(&tokens[dp + 1..np]);
585    Ast::While {
586        cond: alloc::boxed::Box::new(cond),
587        body: alloc::boxed::Box::new(body),
588    }
589}

591fn parse_if_stmt(tokens: &[Tok]) -> Ast {
592    // Locate structural keywords after the leading `if` Word.
593    // Keywords are plain Word tokens — never converted — so they survive intact
594    // in external command argument strings.
595    let then_pos = tokens
596        .iter()
597        .position(|t| is_kw(t, "then"))
598        .unwrap_or(tokens.len());
599    let else_pos = tokens.iter().position(|t| is_kw(t, "else"));
600    let fi_pos = tokens
601        .iter()
602        .rposition(|t| is_kw(t, "fi"))
603        .unwrap_or(tokens.len());
604
605    // Condition: tokens[1..then_pos]   (skip leading `If`)
606    let cond_slice = &tokens[1..then_pos];
607    let cond = parse_tokens(cond_slice);
608
609    // Then body: tokens[then_pos+1..else_or_fi]
610    let then_end = else_pos.unwrap_or(fi_pos);
611    let then_slice = &tokens[then_pos + 1..then_end];
612    let then_b = parse_tokens(then_slice);
613
614    // Else body (optional): tokens[else_pos+1..fi_pos]
615    let else_b = else_pos.map(|ep| {
616        let slice = &tokens[ep + 1..fi_pos];
617        alloc::boxed::Box::new(parse_tokens(slice))
618    });
619
620    Ast::If {
621        cond: alloc::boxed::Box::new(cond),
622        then_b: alloc::boxed::Box::new(then_b),
623        else_b,
624    }
625}

627fn parse_pipeline(tokens: &[Tok]) -> Ast {
628    // `&&` / `||` have lower precedence than pipelines — check first.
629    // Split on the FIRST occurrence; right side is parsed recursively so
630    // `A && B && C` builds `And(A, And(B, C))` with left-to-right evaluation.
631    if let Some(pos) = tokens.iter().position(|t| t == &Tok::And || t == &Tok::Or) {
632        let left = parse_pipeline(&tokens[..pos]);
633        let right = parse_pipeline(&tokens[pos + 1..]);
634        return match &tokens[pos] {
635            Tok::And => Ast::And(alloc::boxed::Box::new(left), alloc::boxed::Box::new(right)),
636            Tok::Or => Ast::Or(alloc::boxed::Box::new(left), alloc::boxed::Box::new(right)),
637            _ => unreachable!(),
638        };
639    }
640
641    let pipe_segs: Vec<&[Tok]> = split_on(tokens, |t| t == &Tok::Pipe);
642
643    let cmds: Vec<Cmd> = pipe_segs
644        .iter()
645        .map(|seg| {
646            // Ignore the per-segment `bg` flag here — the trailing `&` check on
647            // `tokens.last()` below is the authoritative background detector.
648            // Filtering out `bg=true` segments caused single-command background
649            // jobs (`httpd 9091 /path &`) to be parsed as Ast::Empty.
650            let (cmd, _bg) = parse_cmd(seg);
651            cmd
652        })
653        .filter(|c| !c.is_empty())
654        .collect();
655
656    // Check for trailing `&` (background marker).
657    let background = tokens.last() == Some(&Tok::Ampersand);
658
659    if background && cmds.len() == 1 {
660        return Ast::Background(cmds.into_iter().next().unwrap_or_else(Cmd::new));
661    }
662    match cmds.len() {
663        0 => Ast::Empty,
664        1 => Ast::Simple(cmds.into_iter().next().unwrap_or_else(Cmd::new)),
665        _ => Ast::Pipeline(cmds),
666    }
667}
668
669/// Parse one command segment (no `|`, `;`, or `&` except trailing `&`).
670/// Returns (Cmd, is_background).
671fn parse_cmd(tokens: &[Tok]) -> (Cmd, bool) {
672    let mut cmd = Cmd::new();
673    let mut background = false;
674    let mut iter = tokens.iter().peekable();
675
676    while let Some(tok) = iter.next() {
677        match tok {
678            Tok::Word(w) => cmd.argv.push(w.clone()),
679            Tok::Ampersand => background = true,
680            Tok::RedirectOut => {
681                if let Some(Tok::Word(path)) = iter.next() {
682                    cmd.redirects.push(Redirect::StdoutTo(path.clone()));
683                }
684            }
685            Tok::RedirectAppend => {
686                if let Some(Tok::Word(path)) = iter.next() {
687                    cmd.redirects.push(Redirect::StdoutAppend(path.clone()));
688                }
689            }
690            Tok::RedirectIn => {
691                if let Some(Tok::Word(path)) = iter.next() {
692                    cmd.redirects.push(Redirect::StdinFrom(path.clone()));
693                }
694            }
695            Tok::RedirectErr => {
696                if let Some(Tok::Word(path)) = iter.next() {
697                    cmd.redirects.push(Redirect::StderrTo(path.clone()));
698                }
699            }
700            _ => {}
701        }
702    }
703    (cmd, background)
704}
705
706/// Split a token slice on positions where `pred` returns true.
707fn split_on<F>(tokens: &[Tok], pred: F) -> Vec<&[Tok]>
708where
709    F: Fn(&Tok) -> bool,
710{
711    let mut result = Vec::new();
712    let mut start = 0;
713    for (i, tok) in tokens.iter().enumerate() {
714        if pred(tok) {
715            result.push(&tokens[start..i]);
716            start = i + 1;
717        }
718    }
719    result.push(&tokens[start..]);
720    result
721}
