8pub(super) fn read_path_bytes(path: &str) -> Result<Vec<u8>, ReadError> {
9    if let Some((size, is_dir)) = crate::cmd_fs::stat_file_vfs(path) {
10        if is_dir {
11            return Err(ReadError::Io);
12        }
13        if size > MAX_INPUT_BYTES {
14            return Err(ReadError::InputTooLarge);
15        }
16        let mut bytes = Vec::new();
17        bytes
18            .try_reserve_exact(size)
19            .map_err(|_| ReadError::AllocationFailed)?;
20        bytes.resize(size, 0);
21        if size == 0 {
22            return Ok(bytes);
23        }
24        if crate::cmd_fs::read_file_vfs(path, &mut bytes) == size {
25            return Ok(bytes);
26        }
27        return Err(ReadError::Io);
28    }
29
30    let fd = syscall::sys_open(path).map_err(|_| ReadError::Io)?;
31    let mut bytes = Vec::new();
32    let mut chunk = [0u8; 512];
33    loop {
34        match syscall::sys_read(fd, &mut chunk) {
35            Ok(0) => break,
36            Ok(n) => {
37                if let Err(err) = extend_input(&mut bytes, &chunk[..n]) {
38                    syscall::sys_close(fd);
39                    return Err(match err {
40                        InputBufferError::TooLarge => ReadError::InputTooLarge,
41                        InputBufferError::AllocationFailed => ReadError::AllocationFailed,
42                    });
43                }
44            }
45            Err(_) => {
46                syscall::sys_close(fd);
47                return Err(ReadError::Io);
48            }
49        }
50    }
51    syscall::sys_close(fd);
52    Ok(bytes)
53}
