1//! Embedded guest artifact admission before any VM syscall.
2
3use sha2::{Digest, Sha256};
4
5/// Maximum flat image length before the final mailbox page.
6pub use crate::layout::MAX_GUEST_BYTES;
7
8/// Artifact admission failure, always detected before VM creation.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum ArtifactError {
11    /// Development packaging was not selected or emitted no bytes.
12    Empty,
13    /// Flat bytes would overlap the mailbox page.
14    Oversized,
15    /// Embedded bytes do not match build-generated integrity metadata.
16    DigestMismatch,
17}
18
19#[cfg(feature = "development-silo-provider")]
20mod packaged {
21    pub static BYTES: &[u8] = include_bytes!(concat!(env!("OUT_DIR"), "/silo-guest.bin"));
22    include!(concat!(env!("OUT_DIR"), "/silo-guest-digest.rs"));
23}
24
25/// Return the admitted development guest bytes.
26///
27/// Empty, oversized, and digest-mismatched artifacts are rejected before the
28/// caller is permitted to issue `CreateVm`.
29pub fn admitted_guest() -> Result<&'static [u8], ArtifactError> {
30    #[cfg(not(feature = "development-silo-provider"))]
31    let (bytes, expected): (&'static [u8], [u8; 32]) = (&[], [0; 32]);
32    #[cfg(feature = "development-silo-provider")]
33    let (bytes, expected) = (packaged::BYTES, packaged::GUEST_SHA256);
34    validate(bytes, expected)
35}
36
37fn validate(bytes: &[u8], expected: [u8; 32]) -> Result<&[u8], ArtifactError> {
38    if bytes.is_empty() {
39        return Err(ArtifactError::Empty);
40    }
41    if bytes.len() > MAX_GUEST_BYTES {
42        return Err(ArtifactError::Oversized);
43    }
44    let actual: [u8; 32] = Sha256::digest(bytes).into();
45    if actual != expected {
46        return Err(ArtifactError::DigestMismatch);
47    }
48    Ok(bytes)
49}

51#[cfg(test)]
52mod tests {
53    use super::*;
54
55    #[test]
56    fn empty_oversized_and_tampered_artifacts_are_rejected() {
57        static OVERSIZED: [u8; MAX_GUEST_BYTES + 1] = [1; MAX_GUEST_BYTES + 1];
58        assert_eq!(validate(&[], [0; 32]), Err(ArtifactError::Empty));
59        assert_eq!(validate(&OVERSIZED, [0; 32]), Err(ArtifactError::Oversized));
60        assert_eq!(validate(&[1], [0; 32]), Err(ArtifactError::DigestMismatch));
61    }
62
63    #[test]
64    fn matching_integrity_metadata_admits_nonempty_bytes() {
65        let bytes = [1u8, 2, 3];
66        let digest: [u8; 32] = Sha256::digest(bytes).into();
67        assert_eq!(validate(&bytes, digest), Ok(bytes.as_slice()));
68    }
