1//! Watch `init`'s crash-storm budget engage.
2//!
3//! `init` restarts a `Permanent` service and gives up on it after five restarts inside one
4//! window (`MAX_RESTARTS_PER_WINDOW`, Spec 12 §4.3). The window is a scheduler-tick count; when
5//! it was compared against `GetTime` op 0 — the raw architected counter, 10 MHz `mtime` on RV64
6//! — it was ~0.1 ms wide, so the budget rolled on every exit and a crash-looping service was
7//! restarted forever.
8//!
9//! This scenario makes the storm real: it force-exits the service cell six times in a row and
10//! requires that `init` stops bringing it back. It runs in the orchestrator context, which is
11//! the one holding `SpawnCap` (`ForceExit` is gated on that capability).
12//!
13//! Markers: `[init-giveup] PASS` / `[init-giveup] FAIL — <reason>`.
14
15use alloc::format;
16use api::syscall::service;
17use ostd::{
18    io::println,
19    syscall::{
20        sys_exit, sys_force_exit, sys_get_scheduler_ticks, sys_lookup_service, SyscallResult,
21    },
22    task::yield_now,
23};
24
25/// The service `init` supervises that this scenario kills. `/bin/config` is `Permanent`, holds
26/// no block/network capability (so `ForceExit` is allowed), and registers a well-known id — which
27/// is what makes the give-up observable from userspace.
28const TARGET_SERVICE: u16 = service::CONFIG;
29const TARGET_NAME: &str = "/bin/config";
30
31/// Abnormal exits to deliver. `init` checks `restart_count >= MAX_RESTARTS_PER_WINDOW` *before*
32/// incrementing, so five restarts are allowed and the sixth abnormal exit is the one it refuses.
33const KILLS: u32 = 6;
34/// How long one restart may take before the scenario calls it a failure.
35const RESTART_WAIT_TICKS: u64 = 300;
36/// How long the service must stay down after the last kill for the give-up to count.
37const ABSENT_TICKS: u64 = 200;
38/// The window `init` measures its budget in (`RESTART_WINDOW_TICKS`).
39///
40/// A storm has to fit inside one window: five restarts, then the sixth abnormal exit that
41/// exhausts the budget. When the host cannot restart the service that fast — a slow QEMU-TCG
42/// runner needs seconds per respawn — the window legitimately rolls and `init` is *right* to
43/// keep restarting, so the scenario reports `SKIP` with its measurement instead of a verdict
44/// it cannot support.
45const WINDOW_TICKS: u64 = 1000;
46
47fn now() -> u64 {
48    sys_get_scheduler_ticks().unwrap_or(0)
49}
50
51fn fail(why: &str) -> ! {
52    println(&format!("[init-giveup] FAIL — {why}"));
53    sys_exit(1);
54}
55
56pub fn run() -> ! {
57    println(&format!(
58        "[init-giveup] START: {KILLS} forced exits on {TARGET_NAME}"
59    ));
60
61    let mut first_exit_tick = 0u64;
62    let mut last_exit_tick = 0u64;
63    for kill in 1..=KILLS {
64        let Some(tid) = sys_lookup_service(TARGET_SERVICE) else {
65            fail(&format!(
66                "kill {kill}: {TARGET_NAME} is already absent — init gave up before its budget was spent"
67            ));
68        };
69        if !matches!(sys_force_exit(tid), SyscallResult::Ok(_)) {
70            fail(&format!("kill {kill}: ForceExit refused for tid {tid}"));
71        }
72        let killed_at = now();
73        if kill == 1 {
74            first_exit_tick = killed_at;
75        }
76        last_exit_tick = killed_at;
77        println(&format!("[init-giveup] kill {kill}/{KILLS} tid={tid}"));
78
79        let deadline = killed_at + RESTART_WAIT_TICKS;
80        while now() < deadline {
81            match sys_lookup_service(TARGET_SERVICE) {
82                Some(new_tid) if new_tid != tid => {
83                    println(&format!(
84                        "[init-giveup] restart {kill}: tid={new_tid} after {} ticks",
85                        now() - killed_at
86                    ));
87                    break;
88                }
89                _ => yield_now(),
90            }
91        }
92    }
93
94    // If the six exits did not fit inside one window, `init` was right to keep restarting and
95    // this host simply cannot host the experiment. Say so with the measurement rather than
96    // reporting a verdict the run cannot support.
97    let span = last_exit_tick.saturating_sub(first_exit_tick);
98    if span > WINDOW_TICKS {
99        println(&format!(
100            "[init-giveup] SKIP — {KILLS} exits spanned {span} ticks, beyond init's {WINDOW_TICKS}-tick window; this host still restarted {TARGET_NAME} correctly"
101        ));
102        sys_exit(0);
103    }
104
105    // The budget is spent: the service must stay down for a whole window's worth of ticks.
106    let deadline = now() + ABSENT_TICKS;
107    while now() < deadline {
108        if let Some(tid) = sys_lookup_service(TARGET_SERVICE) {
109            fail(&format!(
110                "init restarted {TARGET_NAME} again (tid={tid}) — the crash-storm budget never engaged"
111            ));
112        }
113        yield_now();
114    }
115
116    println(&format!(
117        "[init-giveup] PASS (init gave up on {TARGET_NAME} after {KILLS} abnormal exits and left it down)"
118    ));
119    sys_exit(0);
120}
