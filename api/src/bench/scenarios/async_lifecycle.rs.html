1//! Phase-03 step-1 prototype: bounded **multi-outstanding** local calls over the
2//! shipped exact-operation IPC, measured on the Phase-02 lifecycle image.
3//!
4//! Phase 03 has to decide between a userland scheduler over the primitive that
5//! already exists and a new kernel submission/completion source — and a new public
6//! source needs two Law-1 confirmations. This module is the measurement that decision
7//! rests on, and it changes nothing: no ABI, no kernel path, no production callsite.
8//!
9//! Three legs, one marker per leg:
10//!
11//! | Leg | Question it answers with a number |
12//! |---|---|
13//! | 1 | Can one Cell hold N outstanding bounded calls against a real peer, get **exactly one** correlated completion for each, and at what latency and poll cost? |
14//! | 2 | Is `sys_try_send` a submission mechanism, or is it dropped unless the receiver is parked in `Recv`? |
15//! | 3 | Does a peer that dies mid-flight terminalise **every** outstanding operation, with none lost? |
16//!
17//! The stranding baseline this work exists to fix is measured in
18//! `scenarios::local_service_lifecycle` (synchronous leg: the caller is never woken)
19//! and is not duplicated here.
20

37const PROBE_PATH: &str = "/bin/bench-probe";
38const ECHO_PREFIX: &str = "c2c-echo-reverse:";
39const SLOW_ROLE: &str = "c2c-slow-parked";
40/// The existing role: consume one tagged request, then exit **without** replying.
41const VANISH_ROLE: &str = "c2c-provider";

43/// First byte of every request, and of the echo reply.
44const REQ_TAG: u8 = 0x5A;
45const REPLY_TAG: u8 = 0x5B;
46/// Sequence a caller sends on the ordinary path to release a probe that must outlive
47/// the bounded operations it answered.
48const DONE_SEQ: u8 = 0xFF;
49
50/// Yields allowed for a spawned probe to reach its parked receive.
51const SETTLE_YIELDS: usize = 60;
52/// Outstanding calls in the bounded leg.
53const OUTSTANDING: usize = 8;
54/// Outstanding calls in the mid-flight-death leg.
55const DEATH_OUTSTANDING: usize = 4;
56/// Per-`wait` tick budget (10 ms ticks) and the rounds the collector allows.
57const WAIT_TICKS: u64 = 200;
58const WAIT_ROUNDS: usize = 40;
59/// How long the slow provider stays out of `Recv` (long enough for two try_sends).
60const SLOW_SLEEP_YIELDS: usize = 1200;

62fn fail(reason: &str) -> ! {
63    println(&format!("[async-lifecycle] FAIL — {reason}"));
64    sys_exit(1)
65}

67fn spawn(role: &str) -> usize {
68    if !sys_set_spawn_args(role) {
69        fail("spawn args");
70    }
71    match sys_spawn_pinned(PROBE_PATH, TaskPriority::Normal as u8, 0) {
72        SyscallResult::Ok(tid) if tid != 0 => tid,
73        _ => fail("spawn"),
74    }
75}

77fn settle() {
78    for _ in 0..SETTLE_YIELDS {
79        yield_now();
80    }

85/// Echo provider: consume exactly `n` tagged requests, then answer them **in reverse
86/// order**, so a caller that assumes FIFO correlation cannot pass, and print one line
87/// per reply so the log shows the order they left in.
88#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
89pub fn run_echo_reverse(role: &str) -> ! {
90    let count = role
91        .strip_prefix(ECHO_PREFIX)
92        .and_then(|n| n.parse::<usize>().ok())
93        .unwrap_or(0);
94    println(&format!(
95        "[async-lifecycle] echo-provider-ready count={count}"
96    ));
97    let mut buf = [0u8; IPC_BUF_SIZE];
98    let mut pending: Vec<(usize, u8)> = Vec::new();
99    while pending.len() < count {
100        match sys_recv(0, &mut buf) {
101            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] != DONE_SEQ => {
102                // The kernel records the exact operation this message belongs to; a
103                // bounded reply must name it, not just the sender.
104                match current() {
105                    Some(operation) => pending.push((operation, buf[1])),
106                    None => fail("echo provider: no operation id for a bounded request"),
107                }
108            }
109            SyscallResult::Ok(_) => { /* not ours: keep waiting */ }
110            SyscallResult::Err(_) => fail("echo provider recv"),
111        }
112    }
113    for (operation, seq) in pending.iter().rev() {
114        if reply(*operation, &[REPLY_TAG, *seq]).is_err() {
115            fail("echo provider bounded reply");
116        }
117    }
118    println(&format!(
119        "[async-lifecycle] echo-provider-replied count={count} reversed"
120    ));
121
122    // Stay alive until the caller releases us: exiting here would terminalise the
123    // operations the caller has not taken yet as peer-gone, which is leg 3's shape,
124    // not this leg's.
125    loop {
126        match sys_recv(0, &mut buf) {
127            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] == DONE_SEQ => {
128                println("[async-lifecycle] echo-provider-done");
129                sys_exit(0)
130            }
131            SyscallResult::Ok(_) => continue,
132            SyscallResult::Err(_) => fail("echo provider done-recv"),
133        }
134    }
135}

137/// Parked provider: consume one request, stay busy for a bounded while, then take a
138/// second receive. What that second receive finds is the measurement: a frame the
139/// caller's `sys_try_send` accepted while this cell was *not* in `Recv` is either
140/// queued for us or silently dropped.
141#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
142pub fn run_slow_parked() -> ! {
143    println("[async-lifecycle] slow-provider-ready");
144    let mut buf = [0u8; IPC_BUF_SIZE];
145    let _sender = match sys_recv(0, &mut buf) {
146        SyscallResult::Ok(from) => from,
147        SyscallResult::Err(_) => fail("slow provider recv"),
148    };
149    println("[async-lifecycle] slow-provider-busy");
150    for _ in 0..SLOW_SLEEP_YIELDS {
151        yield_now();
152    }
153    // The caller releases us either way, so drain until that release arrives:
154    // anything else found first is a frame the kernel accepted while we were not
155    // in `Recv`, i.e. queued rather than dropped.
156    let mut queued = false;
157    loop {
158        match sys_recv(0, &mut buf) {
159            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] == DONE_SEQ => break,
160            SyscallResult::Ok(_) => queued = true,
161            SyscallResult::Err(_) => {
162                println("[async-lifecycle] slow-provider-second=error");
163                break;
164            }
165        }
166    }
167    println(&format!(
168        "[async-lifecycle] slow-provider-second queued={queued}"
169    ));
170    println("[async-lifecycle] slow-provider-done");
171    sys_exit(0)
172}

176/// Leg 1: N outstanding bounded calls, exactly one correlated completion each.
177fn bounded_multi_outstanding_leg() {
178    let provider = spawn(&format!("{ECHO_PREFIX}{OUTSTANDING}"));
179    settle();
180
181    let mut calls: Vec<PendingCall> = Vec::new();
182    let mut submitted_at: Vec<u64> = Vec::new();
183    for seq in 0..OUTSTANDING {
184        let request = [REQ_TAG, seq as u8];
185        match PendingCall::submit(provider, &request) {
186            Ok(call) => {
187                calls.push(call);
188                submitted_at.push(sys_get_time());
189            }
190            Err(error) => {
191                println(&format!(
192                    "[async-lifecycle] BOUNDED-SUBMIT-FAILED {error:?}"
193                ));
194                fail("a bounded submit was refused with the peer parked");
195            }
196        }
197    }
198
199    let mut done = alloc::vec![false; calls.len()];
200    let mut completed = 0usize;
201    let mut lost = 0usize;
202    let mut wrong_seq = 0usize;
203    let mut latencies: Vec<u64> = Vec::new();
204    let mut rounds_used = 0usize;
205    let mut reply = [0u8; IPC_BUF_SIZE];
206
207    while completed + lost < calls.len() && rounds_used < WAIT_ROUNDS {
208        rounds_used += 1;
209        if !wait(WAIT_TICKS) {
210            continue;
211        }
212        for index in 0..calls.len() {
213            if done[index] {
214                continue;
215            }
216            match calls[index].try_take(&mut reply) {
217                Ok(None) => {}
218                Ok(Some(completion)) => {
219                    done[index] = true;
220                    if !completion.is_definite() || completion.len < 2 {
221                        println(&format!(
222                            "[async-lifecycle] BOUNDED-COMPLETION {completion:?}"
223                        ));
224                        lost += 1;
225                        continue;
226                    }
227                    // Correlation is the token, not the arrival order: the provider
228                    // answers in reverse, and the payload must still carry this
229                    // operation's own sequence number.
230                    if completion.len < 2 || reply[0] != REPLY_TAG || reply[1] as usize != index {
231                        wrong_seq += 1;
232                    }
233                    latencies.push(sys_get_time().saturating_sub(submitted_at[index]));
234                    completed += 1;
235                }
236                Err(error) => {
237                    println(&format!("[async-lifecycle] BOUNDED-TAKE-ERR {error:?}"));
238                    done[index] = true;
239                    lost += 1;
240                }
241            }
242        }
243    }
244    // Anything still pending after the budget is a completion that never arrived.
245    lost += calls.len() - completed - lost;
246
247    latencies.sort_unstable();
248    let p50 = latencies.get(latencies.len() / 2).copied().unwrap_or(0);
249    let p99 = latencies
250        .get(latencies.len().saturating_sub(1))
251        .copied()
252        .unwrap_or(0);
253    let freq = ostd::syscall::sys_get_timer_freq().unwrap_or(0);
254    println(&format!(
255        "[async-lifecycle] BOUNDED completed={} lost={} wrong_seq={} p50_ticks={} p99_ticks={} freq_hz={} wait_rounds={}",
256        completed,
257        lost,
258        wrong_seq,
259        p50,
260        p99,
261        freq,
262        rounds_used
263    ));
264    let released = matches!(
265        sys_send(provider, &[REQ_TAG, DONE_SEQ]),
266        SyscallResult::Ok(_)
267    );
268    println(&format!("[async-lifecycle] BOUNDED-RELEASED ok={released}"));
269    if completed != OUTSTANDING || lost != 0 || wrong_seq != 0 {
270        fail("bounded multi-outstanding leg did not complete exactly once per request");
271    }
272}

274/// Leg 2: is `sys_try_send` a submission mechanism?
275///
276/// `TrySend` refuses a non-blocking delivery **in its return value** — the kernel
277/// answers `isize::MAX` and the SDK wrapper hands that back as `Ok`, so a caller must
278/// compare the value rather than the `Result`. This leg measures the rule: one send
279/// to a parked receiver, one while the receiver is busy, and the provider's own
280/// second receive saying whether anything the kernel accepted was queued.
281fn try_send_leg() {
282    /// The kernel's "dropped, target not receiving" sentinel, **as the SDK wrapper
283    /// hands it back**: `usize::MAX`. It arrives as a value, not as an `Err`, so a
284    /// caller that checks only the `Result` reads a dropped frame as a delivered one.
285    const REFUSED: usize = usize::MAX;
286    /// How long the parked send may keep retrying before the provider counts as
287    /// never parked (scheduler ticks, about 5 s at the 100 Hz tick).
288    const PARK_WINDOW_TICKS: u64 = 500;
289
290    let provider = spawn(SLOW_ROLE);
291    settle();
292
293    // The provider parks in `Recv` only after its ready line reaches the UART, and
294    // on SMP it runs on another CPU, so no fixed number of yields here guarantees it
295    // is parked. A refusal while it is still running is correct and drops the frame
296    // (nothing is queued), so retry until it is accepted, within a bounded window.
297    let park_deadline = sys_get_scheduler_ticks().unwrap_or(0) + PARK_WINDOW_TICKS;
298    let mut park_attempts = 1usize;
299    let mut parked_ret = value_of(sys_try_send(provider, &[REQ_TAG, 1]));
300    while parked_ret == REFUSED && sys_get_scheduler_ticks().unwrap_or(u64::MAX) < park_deadline {
301        yield_now();
302        park_attempts += 1;
303        parked_ret = value_of(sys_try_send(provider, &[REQ_TAG, 1]));
304    }
305
306    // Give the provider time to consume that frame and leave `Recv`.
307    for _ in 0..200 {
308        yield_now();
309    }
310    let busy_ret = value_of(sys_try_send(provider, &[REQ_TAG, 2]));
311
312    // Release the provider so its verdict is bounded, then let it run.
313    let _ = sys_send(provider, &[REQ_TAG, DONE_SEQ]);
314    for _ in 0..SLOW_SLEEP_YIELDS {
315        yield_now();
316    }
317
318    let parked_refused = parked_ret == REFUSED;
319    let busy_refused = busy_ret == REFUSED;
320    println(&format!(
321        "[async-lifecycle] TRY-SEND parked_refused={parked_refused} busy_refused={busy_refused} parked_ret={parked_ret} busy_ret={busy_ret} park_attempts={park_attempts}"
322    ));
323    if parked_refused {
324        fail("try_send refused every frame to the slow provider within the park window");
325    }
326    if !busy_refused {
327        fail("try_send accepted a frame for a receiver that was not in Recv");
328    }
329}

331/// The syscall's return value, whichever arm it came back on.
332fn value_of(result: SyscallResult) -> usize {
333    match result {
334        SyscallResult::Ok(value) => value,
335        SyscallResult::Err(_) => usize::MAX,
336    }
337}

339/// Leg 3: a peer that dies mid-flight terminalises every outstanding operation.
340fn mid_flight_death_leg() {
341    let provider = spawn(VANISH_ROLE);
342    settle();
343
344    let mut calls: Vec<PendingCall> = Vec::new();
345    let mut submit_refused = 0usize;
346    for seq in 0..DEATH_OUTSTANDING {
347        match PendingCall::submit(provider, &[REQ_TAG, seq as u8]) {
348            Ok(call) => calls.push(call),
349            // A pre-dispatch refusal is also a definite outcome, and must not be
350            // counted as a lost operation.
351            Err(_) => submit_refused += 1,
352        }
353    }
354
355    let mut done = alloc::vec![false; calls.len()];
356    let mut terminals = 0usize;
357    let mut peer_gone = 0usize;
358    let mut lost = 0usize;
359    let mut reply = [0u8; IPC_BUF_SIZE];
360    let mut rounds_used = 0usize;
361    while terminals < calls.len() && rounds_used < WAIT_ROUNDS {
362        rounds_used += 1;
363        if !wait(WAIT_TICKS) {
364            continue;
365        }
366        for index in 0..calls.len() {
367            if done[index] {
368                continue;
369            }
370            match calls[index].try_take(&mut reply) {
371                Ok(None) => {}
372                Ok(Some(completion)) => {
373                    done[index] = true;
374                    terminals += 1;
375                    if completion.terminal == ostd::ipc::IpcTerminal::PeerGone {
376                        peer_gone += 1;
377                    } else {
378                        fail("dead peer produced a terminal other than PeerGone");
379                    }
380                }
381                Err(error) => {
382                    println(&format!("[async-lifecycle] DEATH-TAKE-ERR {error:?}"));
383                    done[index] = true;
384                    terminals += 1;
385                    lost += 1;
386                }
387            }
388        }
389    }
390    let unterminal = calls.len() - terminals;
391    println(&format!(
392        "[async-lifecycle] DEATH outstanding={} submit_refused={} terminal={} peer_gone={} unterminal={} lost={}",
393        calls.len(),
394        submit_refused,
395        terminals,
396        peer_gone,
397        unterminal,
398        lost
399    ));
400    if unterminal != 0 || lost != 0 || terminals == 0 {
401        fail("a dead peer left an outstanding operation without a terminal outcome");
402    }
403}

420// The caller runs in `bench-probe`, whose allowlist holds `WaitCompletion`

424const MSW_CALLER: &str = "c2c-msw-caller";
425const MSW_ECHO: &str = "c2c-msw-echo";

428const MSW_RAW: &str = "c2c-msw-raw";
429/// Control frames between the orchestrator, the caller and the hold provider.

432const RAW_TAG: u8 = 0x5E;

434const M5_ROUNDS: usize = 200;

436const A_OUTSTANDING: usize = 24;

441/// The kernel's per-operation expiry (`async_ipc::OPERATION_TIMEOUT_TICKS`).

443/// Fairness bound stated by CP1 §5.5 for one waiter.
444const MAX_OVERTAKES: usize = 63;

446fn msw_fail(reason: &str) -> ! {
447    println(&format!("[async-lifecycle] FAIL — multi-source: {reason}"));
448    sys_exit(1)

455fn submit_or_fail(peer: usize, request: &[u8]) -> PendingCall {

459fn take_or_fail(call: &PendingCall, reply: &mut [u8]) -> ostd::ipc::Completion {
460    match call.try_take(reply) {
461        Ok(Some(completion)) => completion,
462        Ok(None) => msw_fail("a reported operation had no terminal to take"),

464    }

467/// Fast provider: answer each bounded request at once; `[REQ_TAG, DONE_SEQ]`
468/// (a raw message) exits.
469#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
470pub fn run_msw_echo() -> ! {
471    let mut buf = [0u8; IPC_BUF_SIZE];
472    loop {
473        match sys_recv(0, &mut buf) {
474            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] == DONE_SEQ => sys_exit(0),
475            SyscallResult::Ok(_) => {}
476            SyscallResult::Err(_) => msw_fail("echo recv"),
477        }
478        let Some(operation) = current() else {
479            continue;
480        };
481        if reply(operation, &[REPLY_TAG, buf[1]]).is_err() {
482            msw_fail("echo reply");
483        }
484    }
485}

487/// Raw event source with no RPC context: on each raw `[CTRL_TAG]` from the
488/// caller it sends one raw `[RAW_TAG]` back. A provider serving the caller's
489/// operation cannot do this: a Send to the currently served caller is that
490/// operation's reply (Spec 20 §2.6 immediate-reply bridge).
491#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
492pub fn run_msw_raw() -> ! {
493    let mut buf = [0u8; IPC_BUF_SIZE];
494    loop {
495        match sys_recv(0, &mut buf) {
496            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] == DONE_SEQ => sys_exit(0),
497            SyscallResult::Ok(sender) if buf[0] == CTRL_TAG => {
498                for _ in 0..SETTLE_YIELDS {
499                    yield_now();
500                }
501                let _ = sys_send(sender, &[RAW_TAG]);
502            }
503            SyscallResult::Ok(_) => {}
504            SyscallResult::Err(_) => msw_fail("raw source recv"),
505        }
506    }
507}

509/// Slow provider: answer each bounded request after a fixed busy period, and
510/// stamp the reply with the time it was sent.
511#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
512pub fn run_msw_slow() -> ! {
513    let mut buf = [0u8; IPC_BUF_SIZE];
514    loop {
515        match sys_recv(0, &mut buf) {
516            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] == DONE_SEQ => sys_exit(0),
517            SyscallResult::Ok(_) => {}
518            SyscallResult::Err(_) => msw_fail("slow recv"),
519        }
520        let Some(operation) = current() else {
521            continue;
522        };
523        for _ in 0..SLOW_REPLY_YIELDS {
524            yield_now();
525        }
526        let mut out = [0u8; 10];
527        out[0] = REPLY_TAG;
528        out[1] = buf[1];
529        out[2..10].copy_from_slice(&sys_get_time().to_le_bytes());
530        if reply(operation, &out).is_err() {
531            msw_fail("slow reply");
532        }
533    }
534}

536/// Hold provider: keep the first request dispatched and unanswered until the
537/// caller reports its expiry, then try the late reply (it must be refused),
538/// answer one fresh request with that verdict, and exit on `DONE`.
539#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
540pub fn run_msw_hold() -> ! {
541    let mut buf = [0u8; IPC_BUF_SIZE];
542    let held = loop {
543        match sys_recv(0, &mut buf) {
544            SyscallResult::Ok(_) if buf[0] == REQ_TAG => match current() {
545                Some(operation) => break operation,
546                None => msw_fail("hold token"),
547            },
548            SyscallResult::Ok(_) => {}
549            SyscallResult::Err(_) => msw_fail("hold recv"),
550        }
551    };
552    loop {
553        match sys_recv(0, &mut buf) {
554            SyscallResult::Ok(_) if buf[0] == CTRL_TAG => break,
555            SyscallResult::Ok(_) => {}
556            SyscallResult::Err(_) => msw_fail("hold control"),
557        }
558    }
559    let late_refused = reply(held, &[REPLY_TAG, 0xEE]).is_err();
560    loop {
561        match sys_recv(0, &mut buf) {
562            SyscallResult::Ok(_) if buf[0] == REQ_TAG && buf[1] == DONE_SEQ => sys_exit(0),
563            SyscallResult::Ok(_) if buf[0] == REQ_TAG => {
564                let fresh = current().unwrap_or_else(|| msw_fail("hold fresh token"));
565                if fresh == held || reply(fresh, &[REPLY_TAG, late_refused as u8]).is_err() {
566                    msw_fail("hold fresh reply");
567                }
568            }
569            SyscallResult::Ok(_) => {}
570            SyscallResult::Err(_) => msw_fail("hold drain"),
571        }
572    }
573}

575/// The multi-source caller. Receives the provider tids from the orchestrator,
576/// runs every leg, then reports `[CTRL_TAG, 1]` to the orchestrator.
577#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
578pub fn run_msw_caller() -> ! {
579    let mut buf = [0u8; IPC_BUF_SIZE];
580    let parent = match sys_recv(0, &mut buf) {
581        SyscallResult::Ok(sender) if buf[0] == CTRL_TAG => sender,
582        _ => msw_fail("caller start"),
583    };
584    let tid = |index: usize| {
585        let at = 1 + 8 * index;
586        u64::from_le_bytes(buf[at..at + 8].try_into().unwrap()) as usize
587    };
588    let (echo, slow, hold, vanish, echo_reverse, raw) =
589        (tid(0), tid(1), tid(2), tid(3), tid(4), tid(5));
590    msw_reply_leg(echo_reverse);
591    msw_m5_leg(echo);
592    msw_mixed_leg(echo, raw);
593    msw_peer_death_leg(vanish);
594    msw_starvation_leg(echo, slow);
595    msw_deadline_and_stale_leg(hold);
596    for provider in [echo, slow, raw] {
597        let _ = sys_send(provider, &[REQ_TAG, DONE_SEQ]);
598    }
599    let _ = sys_send(parent, &[CTRL_TAG, 1]);
600    sys_exit(0)
601}

603/// A1: eight outstanding calls, one IPC record each, nothing re-reported while
604/// every result stays retained.
605fn msw_reply_leg(provider: usize) {
606    let calls: Vec<PendingCall> = (0..OUTSTANDING)
607        .map(|seq| submit_or_fail(provider, &[REQ_TAG, seq as u8]))
608        .collect();
609    let mut seen = alloc::vec![false; calls.len()];
610    let (mut records, mut dup, mut foreign, mut timer_returns) = (0usize, 0usize, 0usize, 0usize);
611    let deadline = ticks() + 2_000;
612    while records < calls.len() && ticks() < deadline {
613        match wait_any(IPC | TIMER, 200) {
614            Ok(Wake::Ipc(op)) => match calls.iter().position(|call| call.operation() == op) {
615                Some(index) if seen[index] => dup += 1,
616                Some(index) => {
617                    seen[index] = true;
618                    records += 1;
619                }
620                None => foreign += 1,
621            },
622            Ok(Wake::Timer) => timer_returns += 1,
623            _ => msw_fail("reply leg wait"),
624        }
625    }
626    // Every result is still retained: a one-shot source must stay quiet.
627    let rereport = match wait_any(IPC | TIMER, 5) {
628        Ok(Wake::Timer) => 0,
629        Ok(Wake::Ipc(_)) => 1,
630        _ => msw_fail("reply leg quiet check"),
631    };
632    let mut reply_buf = [0u8; IPC_BUF_SIZE];
633    let mut wrong_seq = 0usize;
634    for (index, call) in calls.iter().enumerate() {
635        let completion = take_or_fail(call, &mut reply_buf);
636        if !completion.is_definite() || reply_buf[..2] != [REPLY_TAG, index as u8] {
637            wrong_seq += 1;
638        }
639    }
640    let _ = sys_send(provider, &[REQ_TAG, DONE_SEQ]);
641    println(&format!(
642        "[async-lifecycle] MSW-REPLY records={records} dup={dup} foreign={foreign} \
643         timer_returns={timer_returns} rereport={rereport} wrong_seq={wrong_seq}"
644    ));
645    if records != OUTSTANDING || dup + foreign + rereport + wrong_seq != 0 {
646        msw_fail("reply leg");
647    }
648}

650fn percentile(samples: &mut [u64], pct: usize) -> u64 {
651    samples.sort_unstable();
652    let index = (samples.len() * pct / 100).min(samples.len().saturating_sub(1));
653    samples.get(index).copied().unwrap_or(0)
654}

656/// M5: terminal→running latency, `WaitCompletion(IPC)` against `IpcWait`,
657/// alternating rounds to the same provider in the same thread.
658fn msw_m5_leg(provider: usize) {
659    let mut completion_wait: Vec<u64> = Vec::with_capacity(M5_ROUNDS);
660    let mut ipc_wait: Vec<u64> = Vec::with_capacity(M5_ROUNDS);
661    let mut reply_buf = [0u8; IPC_BUF_SIZE];
662    for round in 0..2 * M5_ROUNDS {
663        let started = sys_get_time();
664        let call = submit_or_fail(provider, &[REQ_TAG, (round % 200) as u8]);
665        if round % 2 == 0 {
666            match wait_any(IPC | TIMER, 200) {
667                Ok(Wake::Ipc(op)) if op == call.operation() => {}
668                _ => msw_fail("m5 completion wait"),
669            }
670            take_or_fail(&call, &mut reply_buf);
671            completion_wait.push(sys_get_time().saturating_sub(started));
672        } else {
673            loop {
674                if let Ok(Some(_)) = call.try_take(&mut reply_buf) {
675                    break;
676                }
677                if ostd::syscall::sys_ipc_wait(200) != 1 {
678                    msw_fail("m5 IpcWait");
679                }
680            }
681            ipc_wait.push(sys_get_time().saturating_sub(started));
682        }
683    }
684    let tick = ostd::syscall::sys_get_timer_freq().unwrap_or(0) / 100;
685    let (wc50, wc99) = (
686        percentile(&mut completion_wait, 50),
687        percentile(&mut completion_wait, 99),
688    );
689    let (iw50, iw99) = (percentile(&mut ipc_wait, 50), percentile(&mut ipc_wait, 99));
690    let pass = tick != 0 && wc99 <= iw99 + tick;
691    println(&format!(
692        "[async-lifecycle] MSW-M5 rounds={M5_ROUNDS} wc_p50={wc50} wc_p99={wc99} \
693         ipcwait_p50={iw50} ipcwait_p99={iw99} tick={tick} pass={pass}"
694    ));
695    if !pass {
696        msw_fail("M5 latency above IpcWait p99 + 1 tick");
697    }
698}

700/// Mixed NET_RX + IPC + TIMER. x86 has no NET_RX producer (D-8): the NET_RX
701/// bit can only show its raw-mailbox interruption rule here, never a record.
702fn msw_mixed_leg(provider: usize, raw_source: usize) {
703    let all = NET_RX | IPC | TIMER;
704    let mut reply_buf = [0u8; IPC_BUF_SIZE];
705    let mut raw = [0u8; IPC_BUF_SIZE];
706    let raw_from_source = |raw: &mut [u8]| {
707        matches!(sys_recv(raw_source, raw), SyscallResult::Ok(s) if s == raw_source)
708            && raw[0] == RAW_TAG
709    };
710    // Every wait goes through this counter, so `netrx_records` is measured: on
711    // x86 (no NET_RX producer) it must stay zero.
712    let mut netrx_records = 0usize;
713    let mut wait = |sources: u32, ticks: u64| {
714        let wake = wait_any(sources, ticks);
715        if matches!(wake, Ok(Wake::NetRx)) {
716            netrx_records += 1;
717        }
718        wake
719    };
720    let timer = matches!(wait(all, 5), Ok(Wake::Timer));
721
722    // A raw message interrupts a wait whose mask names NET_RX.
723    let _ = sys_send(raw_source, &[CTRL_TAG]);
724    let interrupted = matches!(wait(all, 500), Ok(Wake::Interrupted));
725    let raw_ok = raw_from_source(&mut raw);
726    let call = submit_or_fail(provider, &[REQ_TAG, 1]);
727    let ipc_after_raw = matches!(wait(all, 500), Ok(Wake::Ipc(op)) if op == call.operation());
728    take_or_fail(&call, &mut reply_buf);
729
730    // Without NET_RX in the mask the same raw message does not end the wait,
731    // and a pending IPC terminal is still reported past the queued message.
732    let _ = sys_send(raw_source, &[CTRL_TAG]);
733    let raw_ignored = matches!(wait(IPC | TIMER, 30), Ok(Wake::Timer));
734    let call = submit_or_fail(provider, &[REQ_TAG, 2]);
735    let ipc_second = matches!(wait(IPC | TIMER, 500), Ok(Wake::Ipc(op)) if op == call.operation());
736    take_or_fail(&call, &mut reply_buf);
737    let raw2_ok = raw_from_source(&mut raw);
738    #[cfg(target_arch = "x86_64")]
739    let unallocated_irq_refused = matches!(wait_any(IRQ, 5), Err(ostd::ipc::IpcError::Recv));
740    #[cfg(not(target_arch = "x86_64"))]
741    let unallocated_irq_refused = true;
742    println(&format!(
743        "[async-lifecycle] MSW-MIXED timer={timer} interrupted_by_raw={interrupted} \
744         ipc_after_raw={ipc_after_raw} raw_ignored_without_netrx={raw_ignored} \
745         ipc_second={ipc_second} netrx_records={netrx_records} netrx_producer=absent-x86 \
746         unallocated_irq_refused={unallocated_irq_refused}"
747    ));
748    if !(timer
749        && interrupted
750        && raw_ok
751        && ipc_after_raw
752        && raw_ignored
753        && raw2_ok
754        && ipc_second
755        && netrx_records == 0
756        && unallocated_irq_refused)
757    {
758        msw_fail("mixed leg");
759    }
760}

762/// Peer death: every outstanding call is reported once and takes `PeerGone`.
763fn msw_peer_death_leg(provider: usize) {
764    let calls: Vec<PendingCall> = (0..DEATH_OUTSTANDING)
765        .filter_map(|seq| PendingCall::submit(provider, &[REQ_TAG, seq as u8]).ok())
766        .collect();
767    let mut seen = alloc::vec![false; calls.len()];
768    let (mut records, mut dup) = (0usize, 0usize);
769    let deadline = ticks() + 2_000;
770    while records < calls.len() && ticks() < deadline {
771        match wait_any(IPC | TIMER, 200) {
772            Ok(Wake::Ipc(op)) => match calls.iter().position(|call| call.operation() == op) {
773                Some(index) if seen[index] => dup += 1,
774                Some(index) => {
775                    seen[index] = true;
776                    records += 1;
777                }
778                None => msw_fail("peer death foreign token"),
779            },
780            Ok(Wake::Timer) => {}
781            _ => msw_fail("peer death wait"),
782        }
783    }
784    let mut reply_buf = [0u8; IPC_BUF_SIZE];
785    let peer_gone = calls
786        .iter()
787        .filter(|call| {
788            take_or_fail(call, &mut reply_buf).terminal == ostd::ipc::IpcTerminal::PeerGone
789        })
790        .count();
791    let unterminal = calls.len() - records;
792    println(&format!(
793        "[async-lifecycle] MSW-DEATH outstanding={} records={records} peer_gone={peer_gone} \
794         dup={dup} unterminal={unterminal}",
795        calls.len()
796    ));
797    if calls.is_empty() || unterminal != 0 || dup != 0 || peer_gone != calls.len() {
798        msw_fail("peer death leg");
799    }
800}

802/// Sustained two-producer load: B's results must be reported while A floods,
803/// with at most `MAX_OVERTAKES` other records between B's reply and its record.
804fn msw_starvation_leg(fast: usize, slow: usize) {
805    struct Outstanding {
806        call: PendingCall,
807        slow: bool,
808    }
809    let mut outstanding: Vec<Outstanding> = Vec::new();
810    let mut seq = 0u8;
811    let mut submit = |outstanding: &mut Vec<Outstanding>, slow_peer: bool| {
812        seq = seq.wrapping_add(1) % 0xF0;
813        let peer = if slow_peer { slow } else { fast };
814        outstanding.push(Outstanding {
815            call: submit_or_fail(peer, &[REQ_TAG, seq]),
816            slow: slow_peer,
817        });
818    };
819    for _ in 0..A_OUTSTANDING {
820        submit(&mut outstanding, false);
821    }
822    for _ in 0..B_OUTSTANDING {
823        submit(&mut outstanding, true);
824    }
825    let (mut a_done, mut b_done, mut b_during_flood) = (0usize, 0usize, 0usize);
826    let (mut a_sent, mut b_sent) = (A_OUTSTANDING, B_OUTSTANDING);
827    let (mut dup, mut foreign) = (0usize, 0usize);
828    let mut taken: Vec<usize> = Vec::new();
829    let mut arrivals: Vec<u64> = Vec::new();
830    let mut b_windows: Vec<(u64, u64)> = Vec::new();
831    let mut reply_buf = [0u8; IPC_BUF_SIZE];
832    let deadline = ticks() + 6_000;
833    // A keeps flooding (at least A_TARGET calls) until every B call has been
834    // reported, so each B result competes with a live A backlog.
835    while !outstanding.is_empty() && ticks() < deadline {
836        let op = match wait_any(IPC | TIMER, 200) {
837            Ok(Wake::Ipc(op)) => op,
838            Ok(Wake::Timer) => continue,
839            _ => msw_fail("starvation wait"),
840        };
841        let arrived = sys_get_time();
842        let Some(index) = outstanding.iter().position(|o| o.call.operation() == op) else {
843            // Counted, not fatal: a second record for a taken token is a
844            // duplicate, any other unknown token is foreign.
845            if taken.contains(&op) {
846                dup += 1;
847            } else {
848                foreign += 1;
849            }
850            continue;
851        };
852        arrivals.push(arrived);
853        taken.push(op);
854        let entry = outstanding.swap_remove(index);
855        let completion = take_or_fail(&entry.call, &mut reply_buf);
856        if !completion.is_definite() {
857            msw_fail("starvation terminal");
858        }
859        if entry.slow {
860            b_done += 1;
861            if outstanding.iter().any(|o| !o.slow) {
862                b_during_flood += 1;
863            }
864            let sent = u64::from_le_bytes(reply_buf[2..10].try_into().unwrap());
865            b_windows.push((sent, arrived));
866            if b_sent < B_TARGET {
867                b_sent += 1;
868                submit(&mut outstanding, true);
869            }
870        } else {
871            a_done += 1;
872            if a_sent < A_TARGET || b_done < B_TARGET {
873                a_sent += 1;
874                submit(&mut outstanding, false);
875            }
876        }
877    }
878    // Anything still outstanding when the watchdog ended the loop was never
879    // reported: that is a lost result.
880    let lost = outstanding.len();
881    let max_overtakes = b_windows
882        .iter()
883        .map(|&(sent, arrived)| {
884            arrivals
885                .iter()
886                .filter(|&&t| t > sent && t < arrived)
887                .count()
888        })
889        .max()
890        .unwrap_or(0);
891    println(&format!(
892        "[async-lifecycle] MSW-STARVATION a_done={a_done} b_done={b_done} \
893         b_during_flood={b_during_flood} records={} max_b_overtakes={max_overtakes} \
894         bound={MAX_OVERTAKES} lost={lost} dup={dup} foreign={foreign}",
895        arrivals.len()
896    ));
897    if lost + dup + foreign != 0
898        || a_done < A_TARGET
899        || b_done != B_TARGET
900        || b_during_flood != B_TARGET
901        || max_overtakes > MAX_OVERTAKES
902    {
903        msw_fail("starvation leg");
904    }
905}

907/// Deadlines (wait TIMER and real operation expiry) and a refused late reply.
908fn msw_deadline_and_stale_leg(hold: usize) {
909    let mut reply_buf = [0u8; IPC_BUF_SIZE];
910    let submitted = ticks();
911    let held = submit_or_fail(hold, &[REQ_TAG, 1]);
912    let started = ticks();
913    let timer_ok = matches!(wait_any(IPC | TIMER, 20), Ok(Wake::Timer));
914    let timer_ticks = ticks() - started;
915    let expired = match wait_any(IPC, 0) {
916        Ok(Wake::Ipc(op)) => op == held.operation(),
917        _ => false,
918    };
919    let expiry_ticks = ticks() - submitted;
920    let terminal = take_or_fail(&held, &mut reply_buf).terminal;
921    // Let the provider try its late reply, then prove only a fresh token answers.
922    // Every record until the fresh one, and in the quiet window after it, is
923    // classified: one naming the expired token is a stale record.
924    let _ = sys_send(hold, &[CTRL_TAG, 0]);
925    let fresh = submit_or_fail(hold, &[REQ_TAG, 2]);
926    let mut stale_records = 0usize;
927    let mut other_records = 0usize;
928    let (mut fresh_ok, mut late_refused) = (false, false);
929    let fresh_deadline = ticks() + 500;
930    while !fresh_ok && ticks() < fresh_deadline {
931        match wait_any(IPC | TIMER, 100) {
932            Ok(Wake::Ipc(op)) if op == fresh.operation() => {
933                take_or_fail(&fresh, &mut reply_buf);
934                fresh_ok = true;
935                late_refused = reply_buf[1] == 1;
936            }
937            Ok(Wake::Ipc(op)) if op == held.operation() => stale_records += 1,
938            Ok(Wake::Ipc(_)) => other_records += 1,
939            Ok(_) => {}
940            Err(_) => msw_fail("stale leg wait"),
941        }
942    }
943    let quiet = match wait_any(IPC | TIMER, 5) {
944        Ok(Wake::Timer) => true,
945        Ok(Wake::Ipc(op)) if op == held.operation() => {
946            stale_records += 1;
947            false
948        }
949        _ => false,
950    };
951    let _ = sys_send(hold, &[REQ_TAG, DONE_SEQ]);
952    println(&format!(
953        "[async-lifecycle] MSW-DEADLINE timer={timer_ok} timer_ticks={timer_ticks} \
954         op_expiry={expired} op_expiry_ticks={expiry_ticks} terminal={terminal:?}"
955    ));
956    println(&format!(
957        "[async-lifecycle] MSW-STALE late_reply_refused={late_refused} fresh_reply={fresh_ok} \
958         stale_records={stale_records} other_records={other_records} quiet_after={quiet}"
959    ));
960    let timer_in_window = (20..=22).contains(&timer_ticks);
961    if !(timer_ok
962        && timer_in_window
963        && expired
964        && expiry_ticks >= OPERATION_TIMEOUT_TICKS
965        && terminal == ostd::ipc::IpcTerminal::Indeterminate
966        && fresh_ok
967        && late_refused
968        && quiet
969        && stale_records == 0
970        && other_records == 0)
971    {
972        msw_fail("deadline or stale leg");
973    }
974}

976/// Orchestrator side: spawn the providers and the caller, hand the caller the
977/// provider tids, and wait for its verdict or its exit.
978fn multi_source_leg() {
979    let echo = spawn(MSW_ECHO);
980    let slow = spawn(MSW_SLOW);
981    let hold = spawn(MSW_HOLD);
982    let vanish = spawn(VANISH_ROLE);
983    let echo_reverse = spawn(&format!("{ECHO_PREFIX}{OUTSTANDING}"));
984    let raw = spawn(MSW_RAW);
985    let caller = spawn(MSW_CALLER);
986    settle();
987    if !matches!(sys_notify_on_exit(caller), SyscallResult::Ok(0)) {
988        fail("multi-source caller exit watch");
989    }
990    let mut start = [0u8; 1 + 8 * 6];
991    start[0] = CTRL_TAG;
992    for (index, tid) in [echo, slow, hold, vanish, echo_reverse, raw]
993        .into_iter()
994        .enumerate()
995    {
996        start[1 + 8 * index..9 + 8 * index].copy_from_slice(&(tid as u64).to_le_bytes());
997    }
998    if !matches!(sys_send(caller, &start), SyscallResult::Ok(_)) {
999        fail("multi-source caller start");
1000    }
1001    let mut verdict = [0u8; 16];
1002    match sys_recv(caller, &mut verdict) {
1003        SyscallResult::Ok(sender) if sender == caller && verdict[..2] == [CTRL_TAG, 1] => {
1004            println("[async-lifecycle] MSW PASS");
1005        }
1006        _ => fail("multi-source caller did not report success"),
1007    }
1008}

1010// ── Phase 03e: dispatched cancellation, sync timeout, dropped interest ───────
1011//
1012// The orchestrator (`bench`) drives these through the SDK's `IpcWait` path.

1014const RACE_ROLE: &str = "c2c-race-provider";
1015const SYNC_HOLD_ROLE: &str = "c2c-sync-hold";
1016const DROP_ROLE: &str = "c2c-drop-provider";
1017/// Raw control steps between the orchestrator and the 03e providers.
1018const STEP_START: u8 = 0;
1019const STEP_DONE: u8 = 2;
1020const STEP_REPORT: u8 = 3;
1021const STEP_AWAKE: u8 = 4;
1022const STEP_LATE: u8 = 5;
1023/// Submit-then-cancel iterations of the cancellation race.
1024const RACE_CALLS: usize = 256;
1025/// The sync-hold provider sleeps through the first (queued) bounded call, then
1026/// holds the second (dispatched) one past its budget.
1027const SYNC_QUEUED_HOLD_TICKS: usize = 40;
1028const SYNC_DISPATCHED_HOLD_TICKS: usize = 60;
1029const SYNC_SHORT_BUDGET: u64 = 10;
1030const SYNC_BUDGET: u64 = 20;
1031/// Dropped-interest rounds; each round drops a full owner quota of handles.
1032const DROP_ROUNDS: usize = 3;
1033const DROP_BATCH: usize = 64;
1034/// Yields one admission may spend on `Busy` before the leg calls it a leak.
1035const ADMISSION_YIELDS: usize = 4_000;
1036
1037fn start_provider(role: &str) -> usize {
1038    let provider = spawn(role);
1039    settle();
1040    if !matches!(
1041        sys_send(provider, &[CTRL_TAG, STEP_START]),
1042        SyscallResult::Ok(_)
1043    ) {
1044        fail("03e provider start");
1045    }
1046    provider
1047}

1049/// The orchestrator's tid, from the provider's raw start frame.
1050fn await_start(buf: &mut [u8]) -> usize {
1051    match sys_recv(0, buf) {
1052        SyscallResult::Ok(sender) if buf[..2] == [CTRL_TAG, STEP_START] => sender,
1053        _ => fail("03e provider start frame"),
1054    }
1055}

1057/// Masked provider for CANCEL-RACE: receives only from the orchestrator,
1058/// records every request it was handed and never replies, so a delivered
1059/// request can only end `Indeterminate`.
1060#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
1061pub fn run_race_provider() -> ! {
1062    let mut buf = [0u8; IPC_BUF_SIZE];
1063    let caller = await_start(&mut buf);
1064    let mut seen = [0u8; RACE_CALLS / 8];
1065    let mut twice = 0u8;
1066    loop {
1067        match sys_recv(caller, &mut buf) {
1068            SyscallResult::Ok(_) if buf[..2] == [CTRL_TAG, STEP_DONE] => break,
1069            SyscallResult::Ok(_) if buf[0] == REQ_TAG => {
1070                let seq = usize::from(u16::from_le_bytes([buf[1], buf[2]]));
1071                if seq >= RACE_CALLS {
1072                    fail("race provider sequence");
1073                }
1074                if seen[seq / 8] & (1 << (seq % 8)) != 0 {
1075                    twice = twice.saturating_add(1);
1076                }
1077                seen[seq / 8] |= 1 << (seq % 8);
1078            }
1079            SyscallResult::Ok(_) => {}
1080            SyscallResult::Err(_) => fail("race provider recv"),
1081        }
1082    }
1083    // The DONE frame is raw, so this Send is raw too (no served operation).
1084    let mut report = [0u8; 3 + RACE_CALLS / 8];
1085    report[..3].copy_from_slice(&[CTRL_TAG, STEP_REPORT, twice]);
1086    report[3..].copy_from_slice(&seen);
1087    let _ = sys_send(caller, &report);
1088    sys_exit(0)
1089}

1091/// Dispatched cancellation is never reported as "not executed": a request the
1092/// provider received ends `Indeterminate`; `Cancelled` only for one it never saw.
1093fn cancel_race_leg() {
1094    use ostd::ipc::IpcTerminal;
1095    let provider = start_provider(RACE_ROLE);
1096    let mut outcome: Vec<Option<IpcTerminal>> = alloc::vec![None; RACE_CALLS];
1097    let mut reply_buf = [0u8; IPC_BUF_SIZE];
1098    for (seq, slot) in outcome.iter_mut().enumerate() {
1099        let [lo, hi] = (seq as u16).to_le_bytes();
1100        let call = admit(provider, &[REQ_TAG, lo, hi]);
1101        // Vary how far the provider gets before the cancel lands.
1102        for _ in 0..seq % 4 {
1103            yield_now();
1104        }
1105        if call.cancel().is_err() {
1106            fail("cancel race cancel");
1107        }
1108        match call.try_take(&mut reply_buf) {
1109            Ok(Some(completion)) => *slot = Some(completion.terminal),
1110            _ => fail("cancel race take"),
1111        }
1112    }
1113    if !matches!(
1114        sys_send(provider, &[CTRL_TAG, STEP_DONE]),
1115        SyscallResult::Ok(_)
1116    ) {
1117        fail("cancel race release");
1118    }
1119    let mut report = [0u8; 3 + RACE_CALLS / 8];
1120    if !recv_frame_from(provider, &mut report, STEP_REPORT) {
1121        fail("cancel race report");
1122    }
1123    let received = |seq: usize| report[3 + seq / 8] & (1 << (seq % 8)) != 0;
1124    let (mut cancelled, mut indeterminate, mut other, mut unterminal) = (0, 0, 0, 0);
1125    let (mut cancelled_but_received, mut received_but_definite) = (0, 0);
1126    for (seq, terminal) in outcome.iter().enumerate() {
1127        match terminal {
1128            Some(IpcTerminal::Cancelled) => cancelled += 1,
1129            Some(IpcTerminal::Indeterminate) => indeterminate += 1,
1130            Some(_) => other += 1,
1131            None => unterminal += 1,
1132        }
1133        let definite = matches!(
1134            terminal,
1135            Some(IpcTerminal::Cancelled | IpcTerminal::PreDispatchTimeout | IpcTerminal::Reply)
1136        );
1137        if received(seq) && definite {
1138            received_but_definite += 1;
1139        }
1140        if received(seq) && *terminal == Some(IpcTerminal::Cancelled) {
1141            cancelled_but_received += 1;
1142        }
1143    }
1144    let twice = report[2];
1145    println(&format!(
1146        "[async-lifecycle] CANCEL-RACE total={RACE_CALLS} cancelled={cancelled} \
1147         indeterminate={indeterminate} other={other} cancelled_but_received={cancelled_but_received} \
1148         received_but_definite={received_but_definite} delivered_twice={twice} unterminal={unterminal}"
1149    ));
1150    if cancelled + indeterminate != RACE_CALLS
1151        || cancelled_but_received != 0
1152        || received_but_definite != 0
1153        || twice != 0
1154        || unterminal != 0
1155    {
1156        fail("dispatched cancellation reported a delivered request as not executed");
1157    }
1158}

1160/// Submit, retrying only `Busy` (nothing was accepted) under a yield watchdog.
1161fn admit(provider: usize, request: &[u8]) -> PendingCall {
1162    for _ in 0..ADMISSION_YIELDS {
1163        match PendingCall::submit(provider, request) {
1164            Ok(call) => return call,
1165            Err(ostd::ipc::IpcSubmitError::Busy) => yield_now(),
1166            Err(_) => fail("03e admission"),
1167        }
1168    }
1169    fail("admission stayed Busy: a released slot was never returned")
1170}

1172/// Sync-timeout provider: sleep through the caller's queued call, then hold its
1173/// dispatched call past the budget and try both late replies.
1174#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
1175pub fn run_sync_hold() -> ! {
1176    let mut buf = [0u8; IPC_BUF_SIZE];
1177    let caller = await_start(&mut buf);
1178    let _ = ostd::syscall::sys_set_timer(SYNC_QUEUED_HOLD_TICKS);
1179    let _ = sys_send(caller, &[CTRL_TAG, STEP_AWAKE]);
1180    let first_seq = match sys_recv(caller, &mut buf) {
1181        SyscallResult::Ok(_) if buf[0] == REQ_TAG => buf[1],
1182        _ => fail("sync hold first request"),
1183    };
1184    let held = current().unwrap_or_else(|| fail("sync hold token"));
1185    let _ = ostd::syscall::sys_set_timer(SYNC_DISPATCHED_HOLD_TICKS);
1186    // Late implicit reply: Send to the still-served caller is the operation's
1187    // reply bridge. It must be quarantined, never reach the raw mailbox.
1188    let _ = sys_send(caller, &[REPLY_TAG, first_seq]);
1189    let explicit_refused = reply(held, &[REPLY_TAG, first_seq]).is_err();
1190    let _ = sys_send(
1191        caller,
1192        &[CTRL_TAG, STEP_LATE, first_seq, explicit_refused as u8],
1193    );
1194    loop {
1195        match sys_recv(caller, &mut buf) {
1196            SyscallResult::Ok(_) if buf[..2] == [CTRL_TAG, STEP_DONE] => sys_exit(0),
1197            SyscallResult::Ok(_) if buf[0] == REQ_TAG => {
1198                let operation = current().unwrap_or_else(|| fail("sync hold fresh token"));
1199                if reply(operation, &[REPLY_TAG, buf[1]]).is_err() {
1200                    fail("sync hold fresh reply");
1201                }
1202            }
1203            SyscallResult::Ok(_) => {}
1204            SyscallResult::Err(_) => fail("sync hold recv"),
1205        }
1206    }
1207}

1209/// The synchronous SDK's caller budget: a queued expiry is `PreDispatchTimeout`
1210/// and is never delivered; a dispatched expiry is `Indeterminate`; both late
1211/// replies are refused; the next call returns only its own reply.
1212fn sync_timeout_late_leg() {
1213    use ostd::ipc::{service_call_bounded, IpcError};
1214    let provider = start_provider(SYNC_HOLD_ROLE);
1215    let (mut send, mut recv) = ([0u8; 32], [0u8; 32]);
1216    let queued = service_call_bounded(
1217        provider,
1218        &(REQ_TAG, 0u8),
1219        &mut send,
1220        &mut recv,
1221        SYNC_SHORT_BUDGET,
1222    )
1223    .err();
1224    let mut frame = [0u8; 8];
1225    if !recv_frame_from(provider, &mut frame, STEP_AWAKE) {
1226        fail("sync hold awake");
1227    }
1228    let dispatched =
1229        service_call_bounded(provider, &(REQ_TAG, 1u8), &mut send, &mut recv, SYNC_BUDGET).err();
1230    // The first frame from the provider after expiry must be its report: the
1231    // quarantined bridge reply never lands in the raw mailbox.
1232    let late_send_raw_absent = recv_frame_from(provider, &mut frame, STEP_LATE);
1233    let queued_delivered = late_send_raw_absent && frame[2] != 1;
1234    let late_explicit_refused = late_send_raw_absent && frame[3] == 1;
1235    let next = service_call_bounded(provider, &(REQ_TAG, 2u8), &mut send, &mut recv, 200)
1236        .map(|bytes| bytes == [REPLY_TAG, 2])
1237        .unwrap_or(false);
1238    let _ = sys_send(provider, &[CTRL_TAG, STEP_DONE]);
1239    println(&format!(
1240        "[async-lifecycle] SYNC-TIMEOUT-LATE queued={queued:?} queued_delivered={queued_delivered} \
1241         first={dispatched:?} late_explicit_refused={late_explicit_refused} \
1242         late_send_raw_absent={late_send_raw_absent} next_reply={next}"
1243    ));
1244    if queued != Some(IpcError::PreDispatchTimeout)
1245        || queued_delivered
1246        || dispatched != Some(IpcError::Indeterminate)
1247        || !late_explicit_refused
1248        || !late_send_raw_absent
1249        || !next
1250    {
1251        fail("sync timeout leg");
1252    }
1253}

1255/// Dropped-interest provider: replies at once to even sequences and holds odd
1256/// ones; on DONE tries a late reply to every held token.
1257#[allow(dead_code)] // Shared module: dispatched by the bench-probe binary only.
1258pub fn run_drop_provider() -> ! {
1259    let mut buf = [0u8; IPC_BUF_SIZE];
1260    let caller = await_start(&mut buf);
1261    let mut held: Vec<usize> = Vec::new();
1262    loop {
1263        match sys_recv(caller, &mut buf) {
1264            SyscallResult::Ok(_) if buf[..2] == [CTRL_TAG, STEP_DONE] => break,
1265            SyscallResult::Ok(_) if buf[0] == REQ_TAG => {
1266                let operation = current().unwrap_or_else(|| fail("drop provider token"));
1267                if buf[1] % 2 == 0 {
1268                    // The caller may already have dropped it; a refusal is fine.
1269                    let _ = reply(operation, &[REPLY_TAG, buf[1]]);
1270                } else {
1271                    held.push(operation);
1272                }
1273            }
1274            SyscallResult::Ok(_) => {}
1275            SyscallResult::Err(_) => fail("drop provider recv"),
1276        }
1277    }
1278    let late_accepted = held
1279        .iter()
1280        .filter(|operation| reply(**operation, &[REPLY_TAG, 0xEE]).is_ok())
1281        .count();
1282    let _ = sys_send(
1283        caller,
1284        &[
1285            CTRL_TAG,
1286            STEP_REPORT,
1287            late_accepted.min(255) as u8,
1288            held.len().min(255) as u8,
1289        ],
1290    );
1291    sys_exit(0)
1292}

1294/// Dropping an unsettled `PendingCall` releases its kernel slot: three full
1295/// owner quotas of dropped handles, then a fourth quota is admitted.
1296fn drop_reclaim_leg() {
1297    let provider = start_provider(DROP_ROLE);
1298    let mut dropped = 0usize;
1299    for round in 0..=DROP_ROUNDS {
1300        let calls: Vec<PendingCall> = (0..DROP_BATCH)
1301            .map(|index| admit(provider, &[REQ_TAG, (round * DROP_BATCH + index) as u8]))
1302            .collect();
1303        // Let the provider answer some and hold others before the handles go.
1304        settle();
1305        if round < DROP_ROUNDS {
1306            dropped += calls.len();
1307        }
1308        drop(calls);
1309    }
1310    if !matches!(
1311        sys_send(provider, &[CTRL_TAG, STEP_DONE]),
1312        SyscallResult::Ok(_)
1313    ) {
1314        fail("drop provider release");
1315    }
1316    let mut report = [0u8; 8];
1317    if !recv_frame_from(provider, &mut report, STEP_REPORT) {
1318        fail("drop provider report");
1319    }
1320    let (late_accepted, held) = (report[2], report[3]);
1321    println(&format!(
1322        "[async-lifecycle] DROP-RECLAIM dropped={dropped} fresh_accepted={DROP_BATCH} \
1323         held_by_provider={held} late_accepted={late_accepted}"
1324    ));
1325    if dropped != DROP_ROUNDS * DROP_BATCH || late_accepted != 0 {
1326        fail("dropped interest left a reply slot or accepted a late reply");
1327    }
1328}

1330/// The next frame from `provider` is `[CTRL_TAG, step, ..]`. A masked `Recv`
1331/// may first return an unrelated queued death record (the MSW caller's exit
1332/// watch); those are skipped, anything else from `provider` is a mismatch.
1333fn recv_frame_from(provider: usize, frame: &mut [u8], step: u8) -> bool {
1334    for _ in 0..8 {
1335        match sys_recv(provider, frame) {
1336            SyscallResult::Ok(sender) if sender == provider => {
1337                return frame[..2] == [CTRL_TAG, step];
1338            }
1339            SyscallResult::Ok(_) => {}
1340            SyscallResult::Err(_) => return false,
1341        }
1342    }
1343    false
1344}
