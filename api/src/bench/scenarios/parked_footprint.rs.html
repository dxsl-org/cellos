1//! Finite private P01 fixture. Lifecycle success is not resource-settlement evidence.
2use alloc::format;
3use api::task::TaskPriority;
4use ostd::io::println;
5use ostd::ipc::{self, IpcTerminal, PendingCall};
6use ostd::syscall::{self, SyscallError, SyscallResult};
7
8const BOUND: u64 = 500;
9const TAG: u8 = 0x91;
10const READY: u8 = 1;
11const RELEASE: u8 = 2;
12
13fn fail(reason: &str) -> ! {
14    println(&format!("P01-WORKLOAD {{\"status\":\"FAILED\",\"reason\":\"{reason}\"}}"));
15    println("P01:closing");
16    syscall::sys_exit(1)
17}
18
19fn ticks() -> u64 {
20    syscall::sys_get_scheduler_ticks().unwrap_or_else(|| fail("scheduler-clock-unavailable"))
21}
22
23fn exchange(tid: usize, phase: u8, index: u8, deadline: u64) -> bool {
24    let Ok(call) = PendingCall::submit(tid, &[TAG, phase, index]) else { return false; };
25    let mut bytes = [0u8; 8];
26    while ticks() < deadline {
27        match call.try_take(&mut bytes) {
28            Ok(Some(done)) => return done.terminal == IpcTerminal::Reply
29                && done.len == 3 && bytes[..3] == [TAG, phase, index],
30            Ok(None) => { ipc::wait(1); }
31            Err(_) => return false,
32        }
33    }
34    // Cancellation is not success; drain our retained slot when possible.
35    let _ = ipc::cancel(call.operation());
36    let _ = call.try_take(&mut bytes);
37    false
38}

40fn all_live(tids: &[usize]) -> bool {
41    let mut rows = [api::syscall::ProcessInfo::default(); 64];
42    let Ok(count) = syscall::sys_get_procs(&mut rows) else { return false; };
43    count < rows.len() && tids.iter().all(|tid| rows[..count].iter()
44        .any(|row| row.id == *tid && row.state != 3))
45}

47/// Exactly two bounded requests; release acknowledgment precedes normal exit.
48#[allow(dead_code)]
49pub fn run_probe() -> ! {
50    let deadline = ticks().checked_add(BOUND).unwrap_or_else(|| fail("clock-overflow"));
51    let mut bytes = [0u8; 8];
52    let mut parent = 0;
53    let mut index = 0;
54    for phase in [READY, RELEASE] {
55        let remaining = deadline.saturating_sub(ticks());
56        if remaining == 0 { fail("probe-timeout"); }
57        let sender = match syscall::sys_recv_timeout(0, &mut bytes, remaining) {
58            SyscallResult::Ok(sender) if sender != 0 => sender,
59            _ => fail("probe-receive"),
60        };
61        if phase == READY { parent = sender; index = bytes[2]; }
62        if sender != parent || bytes[..3] != [TAG, phase, index] || index >= 8 {
63            fail("probe-protocol");
64        }
65        let Some(operation) = ipc::current() else { fail("probe-operation-missing"); };
66        if ipc::reply(operation, &[TAG, phase, index]).is_err() { fail("probe-reply"); }
67    }
68    syscall::sys_exit(0)
69}

71pub fn run(role: &str) -> ! {
72    let denied = role == "parked-footprint:denied";
73    let requested = if denied { 1 } else {
74        match role.strip_prefix("parked-footprint:") {
75            Some("0") => 0, Some("1") => 1, Some("4") => 4, Some("8") => 8,
76            _ => fail("count-invalid"),
77        }
78    };
79    println(&format!("P01:pre count={requested}"));
80    let deadline = ticks().checked_add(BOUND).unwrap_or_else(|| fail("clock-overflow"));
81    if denied {
82        if !syscall::sys_set_spawn_args("parked-footprint-probe") { fail("denied-argv-not-run"); }
83        let result = syscall::sys_spawn_pinned("/bin/vfs", TaskPriority::Normal as u8, 0);
84        // SpawnPinned preserves the legacy -1 -> Unknown public mapping.
85        // This observation alone is not denial proof: the host must join it to
86        // the exact raw kernel PermissionDenied outcome and absence of a child.
87        let public_result_preserved = matches!(result, SyscallResult::Err(SyscallError::Unknown));
88        let admitted = usize::from(matches!(result, SyscallResult::Ok(_)));
89        println("P01:cleanup");
90        println(&format!("P01-WORKLOAD {{\"status\":\"{}\",\"requested\":1,\"admitted\":{admitted},\"public_result\":\"{result:?}\"}}",
91            if public_result_preserved { "DENIED_CONTROL" } else { "DENIED_CONTROL_PUBLIC_MISMATCH" }));
92        println("P01:closing");
93        syscall::sys_exit(if public_result_preserved { 0 } else { 1 })
94    }
95    let mut tids = [0usize; 8];
96    let mut admitted = 0;
97    let mut valid = true;
98    for index in 0..requested {
99        if !syscall::sys_set_spawn_args("parked-footprint-probe") {
100            println("P01-NOT-RUN child-argv"); valid = false; break;
101        }
102        match syscall::sys_spawn_pinned("/bin/bench-probe", TaskPriority::Normal as u8, 0) {
103            SyscallResult::Ok(tid) if tid != 0 && !tids[..admitted].contains(&tid) => {
104                tids[index] = tid; admitted += 1;
105                if !matches!(syscall::sys_notify_on_exit(tid), SyscallResult::Ok(0))
106                    || !exchange(tid, READY, index as u8, deadline) {
107                    valid = false; break;
108                }
109            }
110            result => { println(&format!("P01-REFUSAL index={index} result={result:?}")); valid = false; break; }
111        }
112    }
113    valid &= admitted == requested && all_live(&tids[..admitted]);
114    if valid { println("P01:live"); }
115    println("P01:cleanup");
116    // Attempt release for every admitted exact child even after another failed.
117    for (index, &tid) in tids[..admitted].iter().enumerate() {
118        valid &= exchange(tid, RELEASE, index as u8, deadline);
119    }
120    let mut exited = [false; 8];
121    let mut exit_count = 0;
122    while exit_count < admitted && ticks() < deadline {
123        let mut bytes = [0u8; 8];
124        match syscall::sys_recv_timeout(0, &mut bytes, 1) {
125            SyscallResult::Ok(0) => {}
126            SyscallResult::Ok(sender) => {
127                match tids[..admitted].iter().position(|tid| *tid == sender) {
128                    Some(index) if !exited[index] && u64::from_ne_bytes(bytes) == 0 => {
129                        exited[index] = true; exit_count += 1;
130                    }
131                    _ => { valid = false; break; }
132                }
133            }
134            _ => { valid = false; break; }
135        }
136    }
137    valid &= exit_count == admitted;
138    if valid {
139        for _ in 0..3 {
140            if !matches!(syscall::sys_set_timer(1), SyscallResult::Ok(_)) { fail("settle-timer"); }
141            println("P01:settle");
142        }
143    }
144    println(&format!("P01-WORKLOAD {{\"status\":\"{}\",\"requested\":{requested},\"admitted\":{admitted},\"exited\":{exit_count}}}",
145        if valid { "LIFECYCLE_COMPLETE" } else { "COUNT_OR_HANDSHAKE_FAILED" }));
146    println("P01:closing");
147    syscall::sys_exit(if valid { 0 } else { 1 })
148}
