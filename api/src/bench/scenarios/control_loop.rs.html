1//! control_loop_jitter — periodic deadline adherence under load.
2//!
3//! A RealTime cell wakes every `PERIOD_TICKS` (via `recv_timeout`, which blocks
4//! until timeout since nothing sends to it), measures the actual elapsed period,
5//! and records the per-cycle error (|actual − period|) plus a deadline-miss when
6//! the cycle overruns by more than `SLACK_TICKS`. Mirrors a fixed-rate control
7//! loop (PID / software PWM) and proves "control-loop meets deadline" (G1 #3).
8//!
9//! Period is 50 ms (5 scheduler ticks) so jitter is meaningful above the 10 ms
10//! tick quantum. Runs under background load cells spawned by the orchestrator.
11//!
12//! ⚠️ Runtime numbers require the bench cell embedded at `/bin/bench` (phase-05).
13
14extern crate alloc;
15use crate::framework::rt_report::RtReport;
16use alloc::vec::Vec;
17use ostd::syscall::{sys_exit, sys_get_time, sys_recv, sys_recv_timeout, sys_send, SyscallResult};
18
19/// Receive timeout uses 10 ms scheduler ticks; five ticks request a 50 ms period.
20const PERIOD_SCHEDULER_TICKS: u64 = 5;
21/// `sys_get_time` reports the 10 MHz hardware timebase used for measured error.
22const PERIOD_TIME_TICKS: u64 = 50 * 10_000;
23/// Allowed measured overrun before a deadline miss: 5 ms in the hardware timebase.
24const SLACK_TIME_TICKS: u64 = 5 * 10_000;
25/// Number of measured periods.
26const CL_ITERS: u32 = 200;
27
28pub const RESULT_LEN: usize = 57;
29
30fn put_u64(buf: &mut [u8; RESULT_LEN], offset: usize, value: u64) {
31    buf[offset..offset + 8].copy_from_slice(&value.to_le_bytes());
32}
33
34fn get_u64(buf: &[u8; 64], offset: usize) -> u64 {
35    u64::from_le_bytes(buf[offset..offset + 8].try_into().unwrap_or([0; 8]))
36}
37
38pub fn decode_result(buf: &[u8; 64]) -> api::ViResult<RtReport> {
39    if buf[0] != 0 || buf[RESULT_LEN..].iter().any(|&byte| byte != 0xa5) {
40        return Err(api::ViError::InvalidInput);
41    }
42    let min = get_u64(buf, 1);
43    let p50 = get_u64(buf, 9);
44    let p99 = get_u64(buf, 17);
45    let p99_9 = get_u64(buf, 25);
46    let max = get_u64(buf, 33);
47    let jitter = get_u64(buf, 41);
48    if min > p50 || p50 > p99 || p99 > p99_9 || p99_9 > max || jitter != max.saturating_sub(min) {
49        return Err(api::ViError::InvalidInput);
50    }
51    Ok(RtReport {
52        name: "control_loop",
53        n: CL_ITERS,
54        min,
55        p50,
56        p99,
57        p99_9,
58        max,
59        jitter,
60        deadline_miss: u32::try_from(get_u64(buf, 49)).map_err(|_| api::ViError::InvalidInput)?,
61    })
62}
63
64/// RealTime probe role: run the periodic loop and return a private report wire
65/// to the orchestrator, which publishes it only after cleanup succeeds.
66pub fn run_control_loop() -> ! {
67    let mut buf = [0u8; 8];
68    // Block for the orchestrator's start ping so we can reply "done" to it later.
69    let orch = loop {
70        match sys_recv(0, &mut buf) {
71            SyscallResult::Ok(s) if s > 0 => break s,
72            _ => ostd::task::yield_now(),
73        }
74    };
75
76    let mut errors: Vec<u64> = Vec::with_capacity(CL_ITERS as usize);
77    let mut miss = 0u32;
78    let mut valid = true;
79    let mut prev = sys_get_time();
80    for _ in 0..CL_ITERS {
81        match sys_recv_timeout(0, &mut buf, PERIOD_SCHEDULER_TICKS) {
82            SyscallResult::Ok(0) => {}
83            _ => {
84                valid = false;
85                break;
86            }
87        }
88        let now = sys_get_time();
89        let actual = now.saturating_sub(prev);
90        let err = actual.abs_diff(PERIOD_TIME_TICKS);
91        errors.push(err);
92        if actual > PERIOD_TIME_TICKS + SLACK_TIME_TICKS {
93            miss += 1;
94        }
95        prev = now;
96    }
97
98    if valid {
99        let r = RtReport::build("control_loop", &mut errors, miss);
100        let mut result = [0u8; RESULT_LEN];
101        put_u64(&mut result, 1, r.min);
102        put_u64(&mut result, 9, r.p50);
103        put_u64(&mut result, 17, r.p99);
104        put_u64(&mut result, 25, r.p99_9);
105        put_u64(&mut result, 33, r.max);
106        put_u64(&mut result, 41, r.jitter);
107        put_u64(&mut result, 49, u64::from(r.deadline_miss));
108        let _ = sys_send(orch, &result);
109    } else {
110        let _ = sys_send(orch, &[1u8]);
111    }
112    sys_exit(0);
113}
