19pub fn run() -> ! {
20    println("[c2c-broker-oracle] START");
21    let mut broker_tid = calibration::run();
22    let mut role_gate = false;
23    for (stage_idx, n) in SWEEP_LEVELS.into_iter().enumerate() {
24        let mut latencies = [0u64; MAX_CLIENTS];
25        let mut send_latencies = [0u64; MAX_CLIENTS];
26        let mut reply_waits = [0u64; MAX_CLIENTS];
27        let mut worker_latencies = [0u64; MAX_CLIENTS];
28        let mut reply_pump_latencies = [0u64; MAX_CLIENTS];
29        let mut client_wake_latencies = [0u64; MAX_CLIENTS];
30        let (summaries, observed_tid, before, after) =
31            run_stage(n, stage_idx as u64, ClientMode::EchoSync, 1, 0, false);
32        broker_tid = broker_tid.max(observed_tid);
33        let total = aggregate(&summaries[..n]);
34        for (idx, summary) in summaries[..n].iter().enumerate() {
35            latencies[idx] = summary.latency_ns;
36            send_latencies[idx] = summary.send_latency_ns;
37            reply_waits[idx] = summary.reply_wait_ns;
38            worker_latencies[idx] = summary.worker_latency_ns;
39            reply_pump_latencies[idx] = summary.reply_pump_latency_ns;
40            client_wake_latencies[idx] = summary.client_wake_latency_ns;
41        }
42        let delta = snapshot_delta(before, after);
43        let (p50, p99) = percentile_pair(&mut latencies, n);
44        let (_, send_p99) = percentile_pair(&mut send_latencies, n);
45        let (_, reply_wait_p99) = percentile_pair(&mut reply_waits, n);
46        let (_, worker_p99) = percentile_pair(&mut worker_latencies, n);
47        let (_, reply_pump_p99) = percentile_pair(&mut reply_pump_latencies, n);
48        let (_, client_wake_p99) = percentile_pair(&mut client_wake_latencies, n);
49        print_sweep_line(
50            n,
51            total,
52            delta,
53            p50,
54            p99,
55            send_p99,
56            reply_wait_p99,
57            worker_p99,
58            reply_pump_p99,
59            client_wake_p99,
60        );
61        if !role_gate && total.success > 0 && delta.network_progress > 0 {
62            role_gate = true;
63            print_role_gate(true, observed_tid);
64        }
65    }
66    if !role_gate {
67        print_role_gate(false, broker_tid);
68    }
69    run_soak();
70    run_overflow();
71    ostd::syscall::sys_exit((role_gate && lifecycle::run()) as usize ^ 1);
72}

74fn run_soak() {
75    let per_client = SOAK_CALLS / MAX_CLIENTS as u16;
76    let (summaries, _, before, after) =
77        run_stage(MAX_CLIENTS, 100, ClientMode::EchoSync, per_client, 0, false);
78    let total = aggregate(&summaries);
79    print_soak_line(total, snapshot_delta(before, after), after);
80}

82fn run_overflow() {
83    let before = broker_snapshot(9000).unwrap_or_default();
84    let (hold_tid, broker_tid) = spawn_and_ready(ClientConfig {
85        mode: ClientMode::HoldAsync,
86        request_count: 1,
87        base_sequence: 9001,
88        hold_turns: service_net_broker::bench_oracle::MAX_HOLD_TURNS,
89        ack_posts: true,
90        wait_for_start: false,
91        wait_for_drain: true,
92    });
93    recv_posted(hold_tid, 1);
94    for _ in 0..64 {
95        yield_now();
96    }
97    let mut tids = [0usize; 5];
98    for (idx, slot) in tids[..4].iter_mut().enumerate() {
99        let (tid, _) = spawn_and_ready(ClientConfig {
100            mode: ClientMode::EchoAsync,
101            request_count: 4,
102            base_sequence: 9100 + (idx as u64 * 8),
103            hold_turns: 0,
104            ack_posts: true,
105            wait_for_start: false,
106            wait_for_drain: true,
107        });
108        recv_posted(tid, 4);
109        *slot = tid;
110    }
111    let (overflow_tid, _) = spawn_and_ready(ClientConfig {
112        mode: ClientMode::EchoAsync,
113        request_count: 1,
114        base_sequence: 9500,
115        hold_turns: 0,
116        ack_posts: true,
117        wait_for_start: false,
118        wait_for_drain: true,
119    });
120    recv_posted(overflow_tid, 1);
121    let mut drain = [0u8; DRAIN_BYTES];
122    encode_drain(&mut drain);
123    let _ = sys_send(hold_tid, &drain);
124    for &tid in &tids[..4] {
125        let _ = sys_send(tid, &drain);
126    }
127    let _ = sys_send(overflow_tid, &drain);
128    let mut summaries = [ClientSummary::default(); 6];
129    summaries[0] = recv_summary(hold_tid);
130    for (idx, &tid) in tids[..4].iter().enumerate() {
131        summaries[idx + 1] = recv_summary(tid);
132    }
133    summaries[5] = recv_summary(overflow_tid);
134    let after = broker_snapshot(9900).unwrap_or(before);
135    let total = aggregate(&summaries);
136    let overflow_ok = summaries[0].success == 1
137        && summaries[1..5]
138            .iter()
139            .all(|s| s.success == 4 && s.busy == 0 && s.indeterminate == 0)
140        && summaries[5].busy == 1
141        && summaries[5].success == 0
142        && after.peak_request_queue >= 16;
143    print_overflow_line(
144        if overflow_ok { "PASS" } else { "BLOCKED" },
145        total,
146        snapshot_delta(before, after),
147        after.peak_request_queue,
148    );
149    let _ = broker_tid;
150}

152fn run_stage(
153    n: usize,
154    seed: u64,
155    mode: ClientMode,
156    request_count: u16,
157    hold_turns: u16,
158    staged: bool,
159) -> (
160    [ClientSummary; MAX_CLIENTS],
161    usize,
162    OracleSnapshot,
163    OracleSnapshot,
164) {
165    let before = broker_snapshot(seed).unwrap_or_default();
166    let mut tids = [0usize; MAX_CLIENTS];
167    let mut summaries = [ClientSummary::default(); MAX_CLIENTS];
168    let mut broker_tid = 0usize;
169    for (idx, slot) in tids[..n].iter_mut().enumerate() {
170        let (tid, observed_tid) = spawn_and_ready(ClientConfig {
171            mode,
172            request_count,
173            base_sequence: seed * 100 + idx as u64,
174            hold_turns,
175            ack_posts: staged,
176            wait_for_start: true,
177            wait_for_drain: staged,
178        });
179        *slot = tid;
180        broker_tid = observed_tid;
181    }
182    let mut start = [0u8; START_BYTES];
183    encode_start(&mut start);
184    for &tid in &tids[..n] {
185        let _ = sys_send(tid, &start);
186    }
187    if staged {
188        let mut drain = [0u8; DRAIN_BYTES];
189        encode_drain(&mut drain);
190        for &tid in &tids[..n] {
191            let _ = sys_send(tid, &drain);
192        }
193    }
194    for (idx, &tid) in tids[..n].iter().enumerate() {
195        summaries[idx] = recv_summary(tid);
196    }
197    let after = broker_snapshot(seed + 1).unwrap_or(before);
198    (summaries, broker_tid, before, after)
199}
