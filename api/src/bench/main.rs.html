41/// PDR performance targets (nanoseconds).  All checked against p99.
42const TARGET_CTX_SWITCH_NS: u64 = 100_000; //  100 µs
43const HARDWARE_TARGET_IPC_NS: u64 = 50_000; // 50 µs p99 on a qualified hardware profile

45// QEMU TCG target (real hardware target is 10 µs; TCG adds 2-4× overhead).
46const TARGET_SYSCALL_NS: u64 = 40_000; //   40 µs (QEMU TCG; real-HW target: 10 µs)
47const TARGET_FOOTPRINT_BYTES: u64 = 10 * 1024 * 1024; // 10 MB

49/// Path for probe/load child cells.  A separate binary (bench-probe) is used
50/// so both orchestrator and children can coexist in the SAS page table without
51/// VA collision (bench @ 0x18000000, bench-probe @ 0x19000000).
52const SELF_PATH: &str = "/bin/bench-probe";
53/// Number of background load cells for RT-under-contention measurement.
54const LOAD_CELLS: usize = 3;
55
56/// Re-spawn this binary in `role` at the given priority, pinned to core 0.
57/// Returns the new task id, or `None` if spawn failed (e.g. /bin/bench absent).
58fn spawn_role(role: &str, priority: u8) -> Option<usize> {
59    ostd::syscall::sys_set_spawn_args(role);
60    match ostd::syscall::sys_spawn_pinned(SELF_PATH, priority, 0) {
61        ostd::syscall::SyscallResult::Ok(tid) => Some(tid),
62        _ => None,
63    }
64}

66/// Orchestrate the RealTime preempt-latency scenario: spawn load + probe,
67/// measure wake-to-run latency under contention, report, then clean up.
68fn run_rt_preempt() {
69    use api::task::TaskPriority;
70    // Background load (Normal priority).
71    let mut load_tids = [0usize; LOAD_CELLS];
72    for slot in load_tids.iter_mut() {
73        if let Some(tid) = spawn_role("load", TaskPriority::Normal as u8) {
74            *slot = tid;
75        }
76    }
77    // RealTime probe.
78    let Some(probe_tid) = spawn_role("rt-probe", TaskPriority::RealTime as u8) else {
79        println("[rt] preempt_latency SKIP — probe spawn failed (bench not at /bin/bench yet)");
80        return;
81    };
82    // Let cells reach their loops before measuring.
83    for _ in 0..100 {
84        ostd::task::yield_now();
85    }
86
87    let r = scenarios::preempt_latency::measure(probe_tid);
88    r.print();
89    r.print_json();
90    // PDR-ish placeholder target (200 µs p99); first real run calibrates baseline.
91    if r.meets(200_000) {
92        println("[rt] preempt_latency PASS");
93    } else {
94        println("[rt] preempt_latency FAIL (p99 over 200µs or deadline miss)");
95    }
96
97    // Tear down spawned cells.
98    let _ = ostd::syscall::sys_force_exit(probe_tid);
99    for &tid in &load_tids {
100        if tid != 0 {
101            let _ = ostd::syscall::sys_force_exit(tid);
102        }
103    }
104}

106/// Spawn `LOAD_CELLS` Normal-priority load cells; returns their tids (0 = failed).
107fn spawn_load() -> [usize; LOAD_CELLS] {
108    use api::task::TaskPriority;
109    let mut tids = [0usize; LOAD_CELLS];
110    for slot in tids.iter_mut() {
111        if let Some(tid) = spawn_role("load", TaskPriority::Normal as u8) {
112            *slot = tid;
113        }
114    }
115    tids
116}
117
118/// Force-exit every non-zero tid in `tids`.
119fn kill_all(tids: &[usize]) {
120    for &tid in tids {
121        if tid != 0 {
122            let _ = ostd::syscall::sys_force_exit(tid);
123        }
124    }
125}

127/// Control-loop jitter scenario: a RealTime cell measures its own period
128/// adherence under load and prints the report itself; we just orchestrate.
129fn run_rt_control_loop() {
130    use api::task::TaskPriority;
131    let load_tids = spawn_load();
132    let Some(probe_tid) = spawn_role("ctl-loop", TaskPriority::RealTime as u8) else {
133        println("[rt] control_loop SKIP — probe spawn failed (bench not at /bin/bench yet)");
134        kill_all(&load_tids);
135        return;
136    };
137    for _ in 0..100 {
138        ostd::task::yield_now();
139    }
140    let _ = ostd::syscall::sys_send(probe_tid, &[0u8]); // start ping
141    let mut done = [0u8; 8];
142    let _ = ostd::syscall::sys_recv(0, &mut done); // wait for completion
143                                                   // The probe sys_exit's itself; only the load cells need teardown.
144    kill_all(&load_tids);
145}

147/// IPC/syscall latency under load: idle baseline vs with load cells spinning.
148fn run_rt_under_load() {
149    let ipc_idle = runner::run_default(&mut IpcSendRecvBench::new());
150    let sys_idle = runner::run_default(&mut SyscallYieldBench);
151
152    let load_tids = spawn_load();
153    if load_tids.iter().all(|&t| t == 0) {
154        println("[rt] under_load SKIP — load spawn failed (bench not at /bin/bench yet)");
155        return;
156    }
157    for _ in 0..100 {
158        ostd::task::yield_now();
159    }
160
161    let ipc_load = runner::run_default(&mut IpcSendRecvBench::new());
162    let sys_load = runner::run_default(&mut SyscallYieldBench);
163    print_under_load("ipc_send_recv", ipc_idle.p99, ipc_load.p99);
164    print_under_load("syscall_yield", sys_idle.p99, sys_load.p99);
165    kill_all(&load_tids);
166}

168/// Print idle vs under-load p99 plus the integer ratio (×100 → fixed-point x.xx).
169fn print_under_load(name: &str, idle_p99: u64, load_p99: u64) {
170    let ratio = load_p99
171        .saturating_mul(100)
172        .checked_div(idle_p99)
173        .unwrap_or(0);
174    println(&alloc::format!(
175        "[rt] {:14} idle_p99={}ns load_p99={}ns ratio={}.{:02}x",
176        name,
177        idle_p99,
178        load_p99,
179        ratio / 100,
180        ratio % 100
181    ));
182}

186fn cell_main() {
187    // Multi-role dispatch: load/rt-probe cells are re-spawns of this binary with
188    // a role arg; the default (no arg) role is the orchestrator.
189    let argv = ostd::args();
190    let role = argv.first().map(|arg| arg.as_str()).unwrap_or("");
191    if role.starts_with("hotswap-cached-inc:") {
192        scenarios::hotswap_supervisor::run_cached_sender_probe(role);
193    }
194    match role {
195        "load" => scenarios::rt_load::run_load(),
196        "rt-probe" => scenarios::preempt_latency::run_probe(),
197        "ctl-loop" => scenarios::control_loop::run_control_loop(),
198        "ipc-echo" => {
199            // Dedicated IPC echo peer: recv any message, reply to sender, repeat.
200            let mut buf = [0u8; 64];
201            loop {
202                let sender = match ostd::syscall::sys_recv(0, &mut buf) {
203                    ostd::syscall::SyscallResult::Ok(sid) => sid,
204                    _ => continue,
205                };
206                ostd::syscall::sys_send(sender, &[]);
207            }
208        }
209        "resp-echo" => scenarios::vfs_getfile_breakdown::run_resp_echo(),
210        "smp-worker" => scenarios::smp::run_worker(),
211        "peer-death-guard" => scenarios::smp::run_peer_death_guard(),
212        "hotswap-cli-probe" => scenarios::hotswap_cli_probe::run(),
213        "hotswap-supervisor" => scenarios::hotswap_supervisor::run(),
214        "hotswap-unauthorized" => scenarios::hotswap_supervisor::run_unauthorized_probe(),
215        "snapshot-authority" => scenarios::snapshot_authority::run(),
216        _ => {} // orchestrator falls through
217    }
218
219    println("[bench] Cellos Performance Benchmark Suite v0.1");
220    println("[bench] gates: qemu ctx<100µs syscall<40µs; hardware ipc<50µs; mem<10MB");
221    println("");
222
223    let mut passed = 0u32;
224    let mut failed = 0u32;
225    let mut informational = 0u32;
226
227    // ── 1. Context switch ─────────────────────────────────────────────────────
228    {
229        let r = runner::run_default(&mut ContextSwitchBench);
230        report::print_report(&r);
231        report::print_json(&r);
232        if r.meets_target(TARGET_CTX_SWITCH_NS) {
233            passed += 1;
234            println("[bench] context_switch PASS");
235        } else {
236            failed += 1;
237            println("[bench] context_switch FAIL (p99 exceeds 100µs target)");
238        }
239    }
240
241    // ── 2. IPC send/recv ──────────────────────────────────────────────────────
242    {
243        let r = runner::run_default(&mut IpcSendRecvBench::new());
244        report::print_report(&r);
245        report::print_json(&r);
246        informational += 1;
247        if r.meets_target(HARDWARE_TARGET_IPC_NS) {
248            println("[bench] ipc_send_recv HW-TARGET-MET (QEMU evidence only)");
249        } else {
250            println("[bench] ipc_send_recv HW-TARGET-MISS (QEMU evidence only)");
251        }
252    }
253
254    // ── 3. Syscall yield ─────────────────────────────────────────────────────
255    {
256        let r = runner::run_default(&mut SyscallYieldBench);
257        report::print_report(&r);
258        report::print_json(&r);
259        if r.meets_target(TARGET_SYSCALL_NS) {
260            passed += 1;
261            println("[bench] syscall_yield PASS");
262        } else {
263            failed += 1;
264            println(
265                "[bench] syscall_yield FAIL (p99 exceeds 40µs QEMU target; real-HW target: 10µs)",
266            );
267        }
268    }
269
270    // ── 4. Memory footprint ───────────────────────────────────────────────────
271    {
272        let mut mb = MemoryFootprintBench::new();
273        if mb.run_once().is_err() {
274            failed += 1;
275            println("[bench] memory_footprint FAIL (MemInfo unavailable)");
276        } else {
277            let r = mb.footprint_report();
278            println(&alloc::format!(
279                "[bench] allocator_committed_bytes={}",
280                mb.bytes()
281            ));
282            report::print_report(&r);
283            report::print_json(&r);
284            if r.p50 <= TARGET_FOOTPRINT_BYTES {
285                passed += 1;
286                println("[bench] memory_footprint PASS");
287            } else {
288                failed += 1;
289                println("[bench] memory_footprint FAIL (exceeds 10 MB target)");
290            }
291        }
292    }
293
294    // ── 5. RT preempt latency (under load) ────────────────────────────────────
295    println("");
296    println("[rt] Real-time latency suite (under load):");
297    run_rt_preempt();
298    run_rt_control_loop();
299    run_rt_under_load();
300
301    // ── 6. SMP throughput (work-stealing, 2 harts) ────────────────────────────
302    println("");
303    println("[smp] SMP throughput suite (2-hart work-stealing):");
304    let (sp, sf) = scenarios::smp::run_smp_suite();
305    passed += sp;
306    failed += sf;
307
308    // ── Service-call stage breakdown ─────────────────────────────────────────
309    // Not pass/fail: it apportions the cost of a service call between the work a
310    // direct (non-ecall) call would skip and the work both paths pay, which is
311    // the evidence for whether a direct path is worth its constraints. Each
312    // sample is 1000 operations — divide a reported figure by 1000 for per-op.
313    {
314        use scenarios::vfs_getfile_breakdown as vgb;
315        println("");
316        println("[breakdown] service call — per-stage cost (each sample = 1000 ops)");
317        report::print_report(&runner::run(&mut vgb::EncodeRequestBench::new(), 2, 10));
318        report::print_report(&runner::run(&mut vgb::DecodeReplyBench::new(), 2, 10));
319        report::print_report(&runner::run(&mut vgb::TrapRoundTripBench, 2, 10));
320        report::print_report(&runner::run(&mut vgb::RoundTripBench::new(), 2, 10));
321        println("[breakdown] done");
322    }
323
324    // ── Summary ───────────────────────────────────────────────────────────────
325    println("");
326    println(&alloc::format!(
327        "[bench] Results: {}/{} PASS  {}/{} FAIL  {}/{} INFO",
328        passed,
329        passed + failed + informational,
330        failed,
331        passed + failed + informational,
332        informational,
333        passed + failed + informational
334    ));
335
336    // Completion marker — printed unconditionally, BEFORE the threshold verdict.
337    // IPC qualification is hardware-only; scheduled QEMU runs retain their
338    // explicit QEMU gates and sustained historical-regression check.
339    println(&alloc::format!(
340        "[bench] BENCHMARK SUITE COMPLETE ({}/{} QEMU gates within target; {} informational)",
341        passed,
342        passed + failed,
343        informational
344    ));
345
346    if failed == 0 {
347        println("[bench] ALL QEMU BENCHMARK GATES PASS");
348    } else {
349        println("[bench] BENCHMARK FAILURES DETECTED");
350    }
351}
