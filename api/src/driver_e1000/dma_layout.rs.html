4/// Initialize a fixed-size array without heap allocation or panic-on-error.
5pub(crate) fn try_init_array<T, E, const N: usize>(
6    mut initialize: impl FnMut(usize) -> Result<T, E>,
7) -> Result<[T; N], E> {
8    struct InitGuard<'a, T> {
9        storage: &'a mut [MaybeUninit<T>],
10        initialized: usize,
11    }
12
13    impl<T> Drop for InitGuard<'_, T> {
14        fn drop(&mut self) {
15            for value in &mut self.storage[..self.initialized] {
16                // SAFETY: only the prefix counted by `initialized` was written.
17                unsafe { value.assume_init_drop() };
18            }
19        }
20    }
21
22    // SAFETY: an array of MaybeUninit<T> requires no initialization.
23    let mut storage: [MaybeUninit<T>; N] = unsafe { MaybeUninit::uninit().assume_init() };
24    let mut guard = InitGuard {
25        storage: &mut storage,
26        initialized: 0,
27    };
28    for index in 0..N {
29        guard.storage[index].write(initialize(index)?);
30        guard.initialized += 1;
31    }
32
33    // SAFETY: all N elements were initialized above. Reading transfers their
34    // ownership into the returned array; clearing the count prevents double-drop.
35    let initialized = unsafe { (guard.storage.as_ptr() as *const [T; N]).read() };
36    guard.initialized = 0;
37    Ok(initialized)
38}

40pub(crate) const TX_SLOTS: usize = 16;
41pub(crate) const RX_SLOTS: usize = 16;
42
43#[derive(Clone, Copy, Debug, Eq, PartialEq)]
44pub(crate) enum DmaSlot {
45    TxRing,
46    TxBuffer(usize),
47    RxRing,
48    RxBuffer(usize),
49}
50
51#[derive(Clone, Copy, Debug, Eq, PartialEq)]
52pub(crate) enum InitialDmaProgram {
53    TxRingBase(u64),
54    RxDescriptor { slot: usize, iova: u64 },
55    RxRingBase(u64),
56    Enable,
57}
58
59#[derive(Clone, Copy)]
60pub(crate) struct DmaIovas {
61    tx_ring: u64,
62    tx_buffers: [u64; TX_SLOTS],
63    rx_ring: u64,
64    rx_buffers: [u64; RX_SLOTS],
65}
66
67impl DmaIovas {
68    #[inline]
69    pub(crate) fn tx_ring_iova(&self) -> u64 {
70        self.tx_ring
71    }
72
73    #[inline]
74    pub(crate) fn rx_ring_iova(&self) -> u64 {
75        self.rx_ring
76    }
77
78    #[inline]
79    pub(crate) fn tx_descriptor_iova(&self, slot: usize) -> u64 {
80        self.tx_buffers[slot]
81    }
82
83    #[inline]
84    pub(crate) fn rx_descriptor_iova(&self, slot: usize) -> u64 {
85        self.rx_buffers[slot]
86    }
87}

89pub(crate) fn for_each_initial_dma_program(
90    layout: &DmaIovas,
91    mut emit: impl FnMut(InitialDmaProgram),
92) {
93    emit(InitialDmaProgram::TxRingBase(layout.tx_ring_iova()));
94    for slot in 0..RX_SLOTS {
95        emit(InitialDmaProgram::RxDescriptor {
96            slot,
97            iova: layout.rx_descriptor_iova(slot),
98        });
99    }
100    emit(InitialDmaProgram::RxRingBase(layout.rx_ring_iova()));
101    emit(InitialDmaProgram::Enable);
102}

104/// Authorize every e1000 DMA object in deterministic initialization order.
105///
106/// The first rejection is returned immediately, before later slots are exposed
107/// to the device. Returned IOVAs are retained verbatim; callers must not replace
108/// them with CPU physical addresses when programming device-visible fields.
109pub(crate) fn authorize_dma_layout<E>(
110    mut authorize: impl FnMut(DmaSlot) -> Result<u64, E>,
111) -> Result<DmaIovas, E> {
112    let tx_ring = authorize(DmaSlot::TxRing)?;
113    let mut tx_buffers = [0u64; TX_SLOTS];
114    for (index, iova) in tx_buffers.iter_mut().enumerate() {
115        *iova = authorize(DmaSlot::TxBuffer(index))?;
116    }
117
118    let rx_ring = authorize(DmaSlot::RxRing)?;
119    let mut rx_buffers = [0u64; RX_SLOTS];
120    for (index, iova) in rx_buffers.iter_mut().enumerate() {
121        *iova = authorize(DmaSlot::RxBuffer(index))?;
122    }
123
124    Ok(DmaIovas {
125        tx_ring,
126        tx_buffers,
127        rx_ring,
128        rx_buffers,
129    })
130}

132/// Run DMA-address programming and TX/RX enablement only after every DMA
133/// object is authorized.
134///
135/// Authorization rejection skips the side-effecting closure entirely.
136pub(crate) fn with_authorized_dma_layout<R, E, T>(
137    resources: R,
138    mut authorize: impl FnMut(&R, DmaSlot) -> Result<u64, E>,
139    initialize: impl FnOnce(R, DmaIovas) -> T,
140) -> Result<T, E> {
141    let layout = authorize_dma_layout(|slot| authorize(&resources, slot))?;
142    Ok(initialize(resources, layout))
143}
