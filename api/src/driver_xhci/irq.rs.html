1//! MSI-X event-ring wake-ups (x86 phase 04b, CP1 `cp1-04b-waitcompletion-irq.md` §7).
2//!
3//! Contract: an interrupt is only ever a hint to re-read the event ring. Every
4//! park is bounded by [`PARK_TICKS`] and every return re-reads the ring, so a lost,
5//! masked or revoked interrupt degrades to slow polling inside the wait's own
6//! budget and never to a hang. A refusal by `AllocPcieIrq` (435) or by
7//! `WaitCompletion(IRQ)` (242) keeps the polled path and says why.
8
9use abi::completion::source::IRQ;
10use abi::pcie_function::{IRQ_KIND_MSIX, IRQ_MODE_ALLOW_MSI, IRQ_MODE_ALLOW_MSIX};
11use ostd::io::print_fmt;
12use ostd::ipc::{wait_any, Wake};
13use ostd::syscall::sys_alloc_pcie_irq;
14
15/// Longest single park, in scheduler ticks (10 ms each).
16pub(crate) const PARK_TICKS: u64 = 10;
17
18/// How many park expiries (an interrupt that should have arrived and did not) to log.
19const EXPIRY_LOG_LIMIT: u32 = 3;
20
21/// How this driver learns that interrupter 0 posted an event.
22///
23/// `Default` is the polled state before [`EventIrq::setup`] has run.
24#[derive(Default)]
25pub(crate) struct EventIrq {
26    /// The MSI/MSI-X vectors held, or `None` when polling.
27    vectors: Option<u8>,
28    expiries_logged: u32,
29}
30
31impl EventIrq {
32    /// Ask for one MSI-X vector (MSI accepted as a fallback) for the owned function
33    /// `bdf`, and log the outcome as `irq=msix|msi` or `irq=polled reason=…`.
34    pub(crate) fn setup(bdf: u32) -> Self {
35        match sys_alloc_pcie_irq(bdf, 1, IRQ_MODE_ALLOW_MSIX | IRQ_MODE_ALLOW_MSI) {
36            Ok(alloc) => {
37                let _ = print_fmt(format_args!(
38                    "[xhci] irq={} vectors={:#x}+{}\n",
39                    if alloc.kind == IRQ_KIND_MSIX {
40                        "msix"
41                    } else {
42                        "msi"
43                    },
44                    alloc.first_irq,
45                    alloc.count
46                ));
47                Self {
48                    vectors: Some(alloc.count),
49                    expiries_logged: 0,
50                }
51            }
52            Err(error) => {
53                let _ = print_fmt(format_args!(
54                    "[xhci] irq=polled reason=alloc-refused ({:?})\n",
55                    error
56                ));
57                Self::default()
58            }
59        }
60    }
61
62    /// Whether 435 granted a vector, so interrupter 0 should be enabled.
63    pub(crate) fn granted(&self) -> bool {
64        self.vectors.is_some()
65    }
66
67    /// Park until this task's line fires or [`PARK_TICKS`] pass.
68    ///
69    /// `false` means the driver is (now) polled and the caller must use its
70    /// poll/yield cadence; `true` means the caller should re-read the ring.
71    pub(crate) fn park(&mut self) -> bool {
72        if self.vectors.is_none() {
73            return false;
74        }
75        match wait_any(IRQ, PARK_TICKS) {
76            Ok(Wake::Irq(_)) => true,
77            Ok(Wake::Interrupted) => {
78                if self.expiries_logged < EXPIRY_LOG_LIMIT {
79                    self.expiries_logged += 1;
80                    let _ = print_fmt(format_args!(
81                        "[xhci] irq wait expired after {} ticks; re-reading the event ring\n",
82                        PARK_TICKS
83                    ));
84                }
85                true
86            }
87            other => {
88                let _ = print_fmt(format_args!(
89                    "[xhci] irq=polled reason=wait-refused ({:?})\n",
90                    other
91                ));
92                self.vectors = None;
93                false
94            }
95        }
96    }
97}
