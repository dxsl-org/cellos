1//! Input-service producer.
2//!
3//! Decoded HID events travel through the **same** path the BCM DWC2 USB driver
4//! uses: register once as a raw event source with the input service, then send
5//! `EV_DEVICE` frames. No second input stack.
6//!
7//! The input service accepts `USB_HID_HOST` only from the tid the kernel has
8//! published as `service::USB_HID_PRODUCER` (see `cells/services/input/src/main.rs`),
9//! so the cell calls `sys_register_usb_hid_producer()` first to acquire that
10//! kernel-verified producer identity. It deliberately does **not** call
11//! `sys_register_nic_driver()`: that role is the singleton NIC owner and
12//! claiming it would redirect network IPC to this cell.
13
14use api::ipc::{self, input_source, InputRequest};
15use driver_hid::{EvdevEvent, HidDeviceId, DEVICE_EVENT_LEN};
16use ostd::syscall::{sys_send, sys_try_send, SyscallResult};
17
18/// Register this cell as a raw event producer with the input service.
19///
20/// `TrySend` reports a refused non-blocking delivery as `Ok(usize::MAX)`, so
21/// only `Ok(0)` means the service received or queued the registration.
22pub fn register_as_source(tid: usize) -> bool {
23    if tid == 0 {
24        return false;
25    }
26    let mut buf = [0u8; 32];
27    match ipc::encode(
28        &InputRequest::RegisterEventSource {
29            kind: input_source::USB_HID_HOST,
30        },
31        &mut buf,
32    ) {
33        Ok(encoded) => matches!(sys_try_send(tid, encoded), SyscallResult::Ok(0)),
34        Err(_) => false,
35    }
36}
37
38/// Send one device-identified event to the input service.
39pub fn forward_device_event(tid: usize, device: HidDeviceId, ev: &EvdevEvent) -> bool {
40    if tid == 0 {
41        return false;
42    }
43    let mut buf = [0u8; DEVICE_EVENT_LEN];
44    ev.encode_device(device, &mut buf);
45    matches!(sys_send(tid, &buf), SyscallResult::Ok(_))
46}
