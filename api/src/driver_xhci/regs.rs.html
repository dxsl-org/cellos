1//! xHCI register offsets, PORTSC bits, TRB layout, and context field encodings.
2//!
3//! Layouts follow the xHCI specification (Rev 1.1/1.2) as cross-checked against
4//! the iPXE `include/ipxe/xhci.h` definitions. The driver only supports 32-byte
5//! contexts (HCCPARAMS1.CSZ = 0), which is what `qemu-xhci` and every USB2/3
6//! xHCI implementation this milestone targets advertise; CSZ = 1 fails closed.
7

111/// `control` field of a Transfer Event: endpoint ID bits 20:16, slot bits 31:24.
112pub const fn event_ep_id(control: u32) -> u8 {
113    ((control >> 16) & 0x1F) as u8
114}
115pub const fn event_slot(control: u32) -> u8 {
116    (control >> 24) as u8
117}
118pub const fn event_type(control: u32) -> u32 {
119    (control >> 10) & 0x3F
120}
121/// Completion code of a Transfer/Command Completion Event (status bits 31:24).
122pub const fn event_code(status: u32) -> u8 {
123    (status >> 24) as u8
124}
125/// Residual bytes of a Transfer Event (status bits 23:0).
126pub const fn event_residual(status: u32) -> u32 {
127    status & 0x00FF_FFFF
128}

130// ── Contexts (32-byte stride; CSZ = 0) ───────────────────────────────────────
131
132pub const CTX_STRIDE: usize = 32;
133/// Input Control Context: Add flags (dword 1). Drop flags (dword 0) stay zero
134/// because this driver only ever adds contexts.
135pub const ICC_ADD: usize = 4;
136/// Slot Context starts after the 8-dword Input Control Context.
137pub const SLOT_CTX: usize = 32;
138/// Input-context offset of the Endpoint Context for Device Context Index `dci`.
139///
140/// The DCI is `2 * endpoint_number + direction` (IN = 1): DCI 1 = EP0,
141/// DCI 2 = EP1 OUT, DCI 3 = EP1 IN. The Input Control Context occupies the
142/// first context slot, so context `dci` lives at `32 * (dci + 1)`.
143pub const fn ep_ctx(dci: u8) -> usize {
144    32 + (dci as usize) * CTX_STRIDE
145}
146
147/// Input Control Context Add/Drop flag for a context index (0 = slot, 1 = EP0…).
148pub const fn ctx_flag(index: u8) -> u32 {
149    1 << index
150}
151
152/// Slot Context dword 0: Context Entries (31:27), Hub (26), Speed (23:20), Route.
153pub const fn slot_info(context_entries: u32, speed: u32) -> u32 {
154    (context_entries << 27) | (speed << 20)
155}
156
157/// Endpoint Context dword 1: MaxPacketSize (31:16), MaxBurst (15:8),
158/// HID (7), EP Type (5:3), CErr (2:1).
159pub const fn ep_dw1(mps: u16, ep_type: u32) -> u32 {
160    ((mps as u32) << 16) | (3 << 1) | (ep_type << 3)
161}

193/// TRBs per command/transfer ring. The last slot holds the Link TRB, so 63 are
194/// usable — far more than the handful of commands and control TDs enumeration

202/// Bounded poll for register/event transitions that complete in microseconds.
203/// Each iteration reads MMIO (a VM exit under QEMU), so the budget is small
204/// enough that a failing stage reports instead of blowing the boot window.
205pub const POLL_LIMIT: u64 = 2_000_000;
206/// Port reset can take longer than a register transition.
207pub const RESET_POLL_LIMIT: u64 = 500_000;
208/// Yield every this many polls so a long wait does not starve the system.
209pub const POLL_YIELD_EVERY: u64 = 4096;
