1//! USB topology and route string management (xHCI 1.2 §8.9).
2
3#[derive(Clone, Copy, Debug, Eq, PartialEq)]
4pub struct DeviceTopology {
5    /// Root hub port number on the controller (1-based).
6    pub root_port: u8,
7    /// 20-bit xHCI route string.
8    pub route_string: u32,
9    /// Hub tier depth (0 = direct on root port, 1 = behind 1 hub, 2 = behind 2 hubs).
10    pub depth: u8,
11    /// Speed ID (1=full, 2=low, 3=high, 4=super).
12    pub speed: u32,
13    /// Slot ID of the parent Transaction Translator (TT) hub, if applicable.
14    pub tt_slot: u8,
15    /// Port number on the TT hub, if applicable.
16    pub tt_port: u8,
17}
18
19impl DeviceTopology {
20    pub const fn root(port: u8, speed: u32) -> Self {
21        Self {
22            root_port: port,
23            route_string: 0,
24            depth: 0,
25            speed,
26            tt_slot: 0,
27            tt_port: 0,
28        }
29    }
30
31    /// Compute topology for a device connected to `hub_port` of a parent hub.
32    pub fn child_of(&self, hub_slot: u8, hub_port: u8, child_speed: u32) -> Option<Self> {
33        if self.depth >= 2 || hub_port == 0 || hub_port > 15 {
34            return None;
35        }
36        let shift = self.depth * 4;
37        let route = self.route_string | (((hub_port as u32) & 0xF) << shift);
38        let (tt_slot, tt_port) = if child_speed < 3 {
39            // Low or full speed: needs TT if hub is high-speed
40            if self.speed == 3 {
41                (hub_slot, hub_port)
42            } else {
43                (self.tt_slot, self.tt_port)
44            }
45        } else {
46            (0, 0)
47        };
48        Some(Self {
49            root_port: self.root_port,
50            route_string: route,
51            depth: self.depth + 1,
52            speed: child_speed,
53            tt_slot,
54            tt_port,
55        })
56    }
57}

59#[cfg(test)]
60mod tests {
61    use super::*;
62
63    #[test]
64    fn root_topology_has_zero_route_string() {
65        let topo = DeviceTopology::root(1, 3);
66        assert_eq!(topo.root_port, 1);
67        assert_eq!(topo.route_string, 0);
68        assert_eq!(topo.depth, 0);
69    }
70
71    #[test]
72    fn tier1_child_route_string_and_tt() {
73        let root = DeviceTopology::root(1, 3);
74        let child = root.child_of(1, 2, 2).expect("child");
75        assert_eq!(child.root_port, 1);
76        assert_eq!(child.route_string, 2);
77        assert_eq!(child.depth, 1);
78        assert_eq!(child.tt_slot, 1);
79        assert_eq!(child.tt_port, 2);
80    }
81
82    #[test]
83    fn tier2_child_route_string() {
84        let root = DeviceTopology::root(1, 3);
85        let hub1 = root.child_of(1, 2, 3).expect("hub1");
86        let dev = hub1.child_of(2, 4, 1).expect("dev");
87        assert_eq!(dev.root_port, 1);
88        assert_eq!(dev.route_string, 2 | (4 << 4)); // 0x42
89        assert_eq!(dev.depth, 2);
90    }
91
92    #[test]
93    fn depth_greater_than_2_is_refused() {
94        let root = DeviceTopology::root(1, 3);
95        let h1 = root.child_of(1, 1, 3).unwrap();
96        let h2 = h1.child_of(2, 1, 3).unwrap();
97        assert_eq!(h2.child_of(3, 1, 1), None);
98    }
