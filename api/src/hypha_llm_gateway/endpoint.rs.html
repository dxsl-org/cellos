1//! Which transport the network backend uses for an endpoint.
2//!
3//! WHY: the network backend's reply is parsed for `TOOL_CALL:` lines, so whoever
4//! controls the bytes on the wire proposes the agent's next action. Plaintext
5//! HTTP hands that to anyone on path. TLS (through service-net, which verifies
6//! the server chain and SNI) is therefore the default for every endpoint.
7//!
8//! The single exception is opt-in and lab-only: with the `qemu-plaintext-llm`
9//! feature, the QEMU user-network host alias [`QEMU_SLIRP_HOST`] may be reached
10//! in plaintext, because there the "network" is QEMU's in-process NAT to the
11//! developer's own machine (`tools/hypha-mock-llm`). A cell cannot tell QEMU from
12//! a real LAN where 10.0.2.2 is an arbitrary host, which is why this is a build
13//! choice rather than a runtime guess. Tool calls stay operator-confirmed either
14//! way (`core`'s `tool_gate`).
15
16/// QEMU user-mode networking's alias for the host (`-netdev user` default).
17pub const QEMU_SLIRP_HOST: [u8; 4] = [10, 0, 2, 2];
18/// Ports the host-side mock serves (`mock_proxy.py`, `--plain`).
19pub const TLS_PORT: u16 = 8443;
20pub const PLAINTEXT_PORT: u16 = 8080;
21
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub enum Transport {
24    Tls,
25    Plaintext,
26}
27
28impl Transport {
29    pub const fn port(self) -> u16 {
30        match self {
31            Transport::Tls => TLS_PORT,
32            Transport::Plaintext => PLAINTEXT_PORT,
33        }
34    }
35
36    pub const fn is_tls(self) -> bool {
37        matches!(self, Transport::Tls)
38    }
39}
40
41/// TLS unless the build opted into QEMU plaintext *and* `ip` is the QEMU host
42/// alias. Every other endpoint is TLS whatever the build says.
43pub const fn select_transport(ip: [u8; 4], qemu_plaintext: bool) -> Transport {
44    let slirp_host = ip[0] == QEMU_SLIRP_HOST[0]
45        && ip[1] == QEMU_SLIRP_HOST[1]
46        && ip[2] == QEMU_SLIRP_HOST[2]
47        && ip[3] == QEMU_SLIRP_HOST[3];
48    if qemu_plaintext && slirp_host {
49        Transport::Plaintext
50    } else {
51        Transport::Tls
52    }
53}
