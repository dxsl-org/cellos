1//! Node candidate ranking and deterministic placement selection.
2
3use alloc::vec::Vec;
4use core::cmp::Ordering;
5
6#[derive(Clone, Debug, PartialEq, Eq)]
7pub struct CandidateNode {
8    pub node_id: [u8; 32],
9    /// Current number of replicas of this application placed on this node
10    pub app_replicas: usize,
11    /// Normalized available headroom (larger is better)
12    pub normalized_headroom: u64,
13    pub is_eligible: bool,
14}
15
16/// Rank candidate destination nodes by:
17/// 1. Fewer existing application replicas on node
18/// 2. Greater normalized capacity headroom
19/// 3. Lexicographical NodeId tie-break
20pub fn rank_nodes<'a>(candidates: &'a mut [CandidateNode]) -> Vec<&'a CandidateNode> {
21    let mut eligible: Vec<&'a CandidateNode> = candidates.iter().filter(|n| n.is_eligible).collect();
22    eligible.sort_by(|a, b| {
23        match a.app_replicas.cmp(&b.app_replicas) {
24            Ordering::Equal => match b.normalized_headroom.cmp(&a.normalized_headroom) {
25                Ordering::Equal => a.node_id.cmp(&b.node_id),
26                other => other,
27            },
28            other => other,
29        }
30    });
31    eligible
32}
