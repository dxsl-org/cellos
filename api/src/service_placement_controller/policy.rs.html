1//! Homogeneous autoscale domain policy configuration.
2
3#[derive(Clone, Debug, PartialEq, Eq)]
4pub struct AutoscaleDomainPolicy {
5    pub domain_id: u32,
6    /// Target concurrency per replica (T)
7    pub target_t: u32,
8    pub min_replicas: usize,
9    pub max_replicas: usize,
10    pub cooldown_ms: u64,
11    pub max_change_per_window: usize,
12}
13
14impl AutoscaleDomainPolicy {
15    pub fn new(
16        domain_id: u32,
17        target_t: u32,
18        min_replicas: usize,
19        max_replicas: usize,
20        cooldown_ms: u64,
21        max_change_per_window: usize,
22    ) -> Self {
23        Self {
24            domain_id,
25            target_t: if target_t == 0 { 1 } else { target_t },
26            min_replicas,
27            max_replicas: max_replicas.max(min_replicas),
28            cooldown_ms,
29            max_change_per_window: if max_change_per_window == 0 { 1 } else { max_change_per_window },
30        }
31    }
32}
