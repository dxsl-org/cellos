1//! `/bin/vmxnet3` PCI identity table (`reference-nic-cell-contract.md`).
2//!
3//! * `15AD:07B0` — VMware vmxnet3, the paravirtual NIC VMware recommends and
4//!   QEMU's `vmxnet3` model. The only entry, `Driveable`.
5//!
6//! VMware's older `vmxnet` (`15AD:0720`) is a different register model and is
7//! refused like any unknown ID.
8
9/// VMware's PCI vendor ID.
10pub const VENDOR_VMWARE: u16 = 0x15AD;
11
12/// Whether the cell may bind an ID.
13#[derive(Clone, Copy, Debug, Eq, PartialEq)]
14pub enum Admission {
15    /// Implemented and validated end to end; queried at discovery.
16    Driveable,
17}
18
19#[derive(Clone, Copy, Debug, Eq, PartialEq)]
20pub struct Identity {
21    pub device_id: u16,
22    pub admission: Admission,
23    pub name: &'static str,
24}
25
26/// Every ID this cell knows, all under [`VENDOR_VMWARE`].
27pub const IDENTITIES: [Identity; 1] = [Identity {
28    device_id: 0x07B0,
29    admission: Admission::Driveable,
30    name: "vmxnet3",
31}];
32
33/// The `Driveable` device IDs, in discovery order (424 queries).
34pub const QUERY_DEVICE_IDS: [u16; 1] = [0x07B0];
35
36/// The identity of `(vendor_id, device_id)` when this cell may drive it.
37pub fn classify(vendor_id: u16, device_id: u16) -> Option<Identity> {
38    if vendor_id != VENDOR_VMWARE {
39        return None;
40    }
41    IDENTITIES
42        .iter()
43        .find(|identity| identity.device_id == device_id)
44        .filter(|identity| identity.admission == Admission::Driveable)
45        .copied()
46}
47
48#[cfg(test)]
49mod tests {
50    use super::*;
51
52    #[test]
53    fn only_vmxnet3_is_driveable_and_queried() {
54        assert_eq!(classify(0x15AD, 0x07B0).map(|i| i.name), Some("vmxnet3"));
55        // vmxnet v1/v2, PVSCSI, a foreign vendor with the same device ID, and
56        // an Intel NIC are all refused.
57        for (vendor, device) in [
58            (0x15AD, 0x0720),
59            (0x15AD, 0x07C0),
60            (0x8086, 0x07B0),
61            (0x8086, 0x10D3),
62        ] {
63            assert_eq!(classify(vendor, device), None, "{vendor:04x}:{device:04x}");
64        }
65        let driveable: Vec<u16> = IDENTITIES
66            .iter()
67            .filter(|i| i.admission == Admission::Driveable)
68            .map(|i| i.device_id)
69            .collect();
70        assert_eq!(driveable, QUERY_DEVICE_IDS);
71    }
72}
