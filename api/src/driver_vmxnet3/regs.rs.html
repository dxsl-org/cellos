1//! vmxnet3 register map and the command handshake.
2//!
3//! Offsets and command codes follow FreeBSD's `sys/dev/vmware/vmxnet3/
4//! if_vmxreg.h` (ISC licence, attribution in `NOTICE`; H-S3-14). The device
5//! has two register BARs:
6//!
7//! * BAR0 "PT" — per-vector interrupt masks and the TX/RX producer doorbells;
8//! * BAR1 "VD" — revision selection, the driver-shared address, the command
9//!   register, the MAC registers and the event-acknowledge register.
10//!
11//! A command is a 32-bit write of its code to `VD_CMD`; a query command's
12//! result is the following read of `VD_CMD`.
13//!
14//! [`RegisterFile`] (`nic_driver::regs`) is the only path to the device:
15//! `MmioRegion` implements it for the cell, and the host tests implement it
16//! with a fake.
17
18pub(crate) use nic_driver::regs::RegisterFile;
19use types::ViResult;
20
21// ── BAR0 (PT) ────────────────────────────────────────────────────────────────
22
23/// Interrupt mask of vector `vector`: 1 masks, 0 unmasks.
24pub(crate) const fn pt_imask(vector: u8) -> usize {
25    vector as usize * 8
26}
27/// TX producer doorbell of queue 0 (`TXH`).
28pub(crate) const PT_TXPROD: usize = 0x600;
29/// RX ring 0 / ring 1 producer of queue 0 (`RXH1`/`RXH2`).
30pub(crate) const PT_RXPROD: usize = 0x800;
31pub(crate) const PT_RXPROD2: usize = 0xA00;
32/// Smallest BAR0 that holds every PT register this cell writes.
33pub(crate) const PT_MIN: usize = PT_RXPROD2 + 4;
34
35// ── BAR1 (VD) ────────────────────────────────────────────────────────────────
36
37pub(crate) const VD_VRRS: usize = 0x000;
38pub(crate) const VD_UVRS: usize = 0x008;
39pub(crate) const VD_DSL: usize = 0x010;
40pub(crate) const VD_DSH: usize = 0x018;
41pub(crate) const VD_CMD: usize = 0x020;
42pub(crate) const VD_MACL: usize = 0x028;
43pub(crate) const VD_MACH: usize = 0x030;
44/// Event-cause acknowledge: writing bits clears them in `DriverShared.event`.
45pub(crate) const VD_ECR: usize = 0x040;
46/// Smallest BAR1 that holds every VD register this cell touches.
47pub(crate) const VD_MIN: usize = VD_ECR + 4;
48
49/// Revision 1 of the device interface (`VRRS`) and of UPT (`UVRS`).
50pub(crate) const REVISION_1: u32 = 1;
51
52// ── Commands ─────────────────────────────────────────────────────────────────
53
54/// `ENABLE` in FreeBSD's naming: activate the device from `DriverShared`.
55pub(crate) const CMD_ACTIVATE_DEV: u32 = 0xCAFE_0000;
56pub(crate) const CMD_QUIESCE_DEV: u32 = 0xCAFE_0001;
57pub(crate) const CMD_RESET_DEV: u32 = 0xCAFE_0002;
58/// `SET_RXMODE`: apply `DriverShared.rxmode`.
59pub(crate) const CMD_UPDATE_RX_MODE: u32 = 0xCAFE_0003;
60/// `SET_FILTER`: apply the multicast table.
61pub(crate) const CMD_UPDATE_MAC_FILTERS: u32 = 0xCAFE_0004;
62/// `VLAN_FILTER`: apply `DriverShared.vlan_filter`.
63pub(crate) const CMD_UPDATE_VLAN_FILTERS: u32 = 0xCAFE_0005;
64/// Query: bit 0 link up, bits 31:16 speed in Mb/s.
65pub(crate) const CMD_GET_LINK: u32 = 0xF00D_0002;
66/// Queries: the permanent MAC, low four and high two bytes.
67pub(crate) const CMD_GET_PERM_MAC_LO: u32 = 0xF00D_0003;
68pub(crate) const CMD_GET_PERM_MAC_HI: u32 = 0xF00D_0004;
69
70pub(crate) const LINK_UP: u32 = 1 << 0;
71
72/// Event causes in `DriverShared.event` (acknowledged through [`VD_ECR`]).
73pub(crate) const EVENT_RQERROR: u32 = 0x01;
74pub(crate) const EVENT_TQERROR: u32 = 0x02;
75
76/// Why the device cannot be driven.
77#[derive(Clone, Copy, Debug, Eq, PartialEq)]
78pub(crate) enum Refusal {
79    /// `VRRS` does not offer revision 1 (the value read).
80    Revision(u32),
81    /// `UVRS` does not offer UPT version 1 (the value read).
82    Upt(u32),
83}
84
85/// Select device revision 1 and UPT version 1, the only ones this cell speaks.
86/// An all-ones read is a BAR that does not decode, never an offer.
87pub(crate) fn negotiate(vd: &impl RegisterFile) -> ViResult<Result<(), Refusal>> {
88    let revisions = vd.rd32(VD_VRRS)?;
89    if revisions == u32::MAX || revisions & REVISION_1 == 0 {
90        return Ok(Err(Refusal::Revision(revisions)));
91    }
92    vd.wr32(VD_VRRS, REVISION_1)?;
93    let upt = vd.rd32(VD_UVRS)?;
94    if upt == u32::MAX || upt & REVISION_1 == 0 {
95        return Ok(Err(Refusal::Upt(upt)));
96    }
97    vd.wr32(VD_UVRS, REVISION_1)?;
98    Ok(Ok(()))
99}
100
101/// Issue a query command and return its result.
102pub(crate) fn query(vd: &impl RegisterFile, command: u32) -> ViResult<u32> {
103    vd.wr32(VD_CMD, command)?;
104    vd.rd32(VD_CMD)
105}
106
107/// The permanent station address.
108pub(crate) fn permanent_mac(vd: &impl RegisterFile) -> ViResult<[u8; 6]> {
109    let low = query(vd, CMD_GET_PERM_MAC_LO)?.to_le_bytes();
110    let high = query(vd, CMD_GET_PERM_MAC_HI)?.to_le_bytes();
111    Ok([low[0], low[1], low[2], low[3], high[0], high[1]])
112}
113
114/// Program `mac` as the current station address (`MACL`, then `MACH`).
115pub(crate) fn set_mac(vd: &impl RegisterFile, mac: [u8; 6]) -> ViResult<()> {
116    vd.wr32(
117        VD_MACL,
118        u32::from_le_bytes([mac[0], mac[1], mac[2], mac[3]]),
119    )?;
120    vd.wr32(VD_MACH, u32::from_le_bytes([mac[4], mac[5], 0, 0]))
121}
122
123/// Whether `mac` is a usable unicast station address.
124pub(crate) fn mac_usable(mac: [u8; 6]) -> bool {
125    mac != [0; 6] && mac[0] & 1 == 0
126}
