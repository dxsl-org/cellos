1//! `DriverShared` and the queue descriptors: the configuration the driver
2//! hands the device by address (`VD_DSL`/`VD_DSH`, revision 1 layout).
3//!
4//! The `#[repr(C)]` mirrors below follow FreeBSD's `if_vmxreg.h` structs
5//! (`vmxnet3_driver_shared`, `vmxnet3_txq_shared`, `vmxnet3_rxq_shared`; ISC,
6//! see `NOTICE`). They exist only to derive field offsets: every field is
7//! naturally aligned in the packed device layout, so `repr(C)` reproduces it,
8//! and the tests pin each offset and size. Values are written byte-exact
9//! through a [`Sink`] (the DMA area, or a byte vector in tests).
10
11pub(crate) use crate::dma::Sink;
12use crate::irq::IntrPlan;
13use core::mem::{offset_of, size_of};
14
15/// Most interrupt vectors the device interface describes (8 TX + 16 RX + 1).
16const MAX_INTRS: usize = 25;
17/// `DriverShared.magic` of revision 1.
18pub(crate) const REV1_MAGIC: u32 = 0xBABE_FEE1;
19/// Guest OS word: FreeBSD-type, 64-bit (the value OpenBSD's driver also uses).
20const GOS_FREEBSD_64: u32 = 0x10 | 0x02;
21/// `rxmode`: unicast (0x01) | broadcast (0x04) | all-multicast (0x08) |
22/// promiscuous (0x10). The net service frames with its own fixed station
23/// address (`52:54:00:12:34:56`), not the device's, so replies to it arrive
24/// only through the promiscuous path, as on the e1000 family (`RCTL.UPE|MPE`).
25const RXMODE: u32 = 0x01 | 0x04 | 0x08 | 0x10;
26const MTU: u32 = 1500;

140/// Write `DriverShared` at `base`: one TX and one RX queue described at
141/// `queues_iova`, unicast + broadcast + all-multicast, untagged frames
142/// (VLAN ID 0) admitted, no offloads.
143pub(crate) fn write_driver_shared(
144    s: &mut impl Sink,
145    base: usize,
146    queues_iova: u64,
147    plan: IntrPlan,
148) {
149    let at = |field: usize| base + field;
150    s.put32(at(offset_of!(DriverShared, magic)), REV1_MAGIC);
151    s.put32(at(offset_of!(DriverShared, version)), 1);
152    s.put32(at(offset_of!(DriverShared, guest)), GOS_FREEBSD_64);
153    s.put32(at(offset_of!(DriverShared, vmxnet3_revision)), 1);
154    s.put32(at(offset_of!(DriverShared, upt_version)), 1);
155    s.put64(at(offset_of!(DriverShared, queue_shared)), queues_iova);
156    s.put32(
157        at(offset_of!(DriverShared, queue_shared_len)),
158        QUEUES_LEN as u32,
159    );
160    s.put32(at(offset_of!(DriverShared, mtu)), MTU);
161    s.put16(at(offset_of!(DriverShared, nrxsg_max)), 1);
162    s.put8(at(offset_of!(DriverShared, ntxqueue)), 1);
163    s.put8(at(offset_of!(DriverShared, nrxqueue)), 1);
164    s.put8(at(offset_of!(DriverShared, automask)), 0);
165    s.put8(at(offset_of!(DriverShared, nintr)), plan.nintr);
166    s.put8(at(offset_of!(DriverShared, evintr)), plan.event);
167    s.put32(at(offset_of!(DriverShared, ictrl)), plan.ictrl);
168    s.put32(at(offset_of!(DriverShared, rxmode)), RXMODE);
169    s.put32(at(offset_of!(DriverShared, vlan_filter)), 1);
170}

172/// Write the TX then the RX queue descriptor at `base`.
173pub(crate) fn write_queues(s: &mut impl Sink, base: usize, q: &QueueConfig, plan: IntrPlan) {
174    let tx = |field: usize| base + field;
175    s.put64(tx(offset_of!(TxQueueShared, cmd_ring)), q.tx_ring);
176    s.put64(tx(offset_of!(TxQueueShared, data_ring)), q.tx_data);
177    s.put64(tx(offset_of!(TxQueueShared, comp_ring)), q.tx_comp);
178    s.put32(tx(offset_of!(TxQueueShared, cmd_ring_len)), q.tx_len);
179    s.put32(tx(offset_of!(TxQueueShared, data_ring_len)), q.tx_len);
180    s.put32(tx(offset_of!(TxQueueShared, comp_ring_len)), q.tx_len);
181    s.put8(tx(offset_of!(TxQueueShared, intr_idx)), plan.tx);
182    let rx = |field: usize| base + TXQ_LEN + field;
183    for ring in 0..2 {
184        s.put64(
185            rx(offset_of!(RxQueueShared, cmd_ring) + ring * 8),
186            q.rx_ring[ring],
187        );
188        s.put32(
189            rx(offset_of!(RxQueueShared, cmd_ring_len) + ring * 4),
190            q.rx_len[ring],
191        );
192    }
193    s.put64(rx(offset_of!(RxQueueShared, comp_ring)), q.rx_comp);
194    s.put32(rx(offset_of!(RxQueueShared, comp_ring_len)), q.rx_comp_len);
195    s.put8(rx(offset_of!(RxQueueShared, intr_idx)), plan.rx);
196}
