14/// TX completion: up to 20 parks (~2 s) with MSI-X, or this many polls.
15const TX_PARKS: u32 = 20;
16const TX_POLLS: u32 = 1_000_000;
17
18impl Vmxnet3 {
19    /// Transmit `frame` (no FCS) and wait, bounded, for its completion.
20    pub fn send_frame(&mut self, frame: &[u8]) -> ViResult<()> {
21        if frame.is_empty() || frame.len() > BUF_SIZE {
22            return Err(ViError::InvalidInput);
23        }
24        reap_tx(&self.dma, &mut self.tx_comp, &mut self.tx_clean);
25        let slot = self.tx.next;
26        if (slot + 1) % TX_SLOTS == self.tx_clean {
27            return Err(ViError::IO); // ring full: earlier frames never completed
28        }
29        let buffer = TX_BUF_OFF + slot * BUF_SIZE;
30        self.dma.copy_in(buffer, frame);
31        let addr = self.dma.iova(buffer);
32        post_tx(&mut self.dma, slot, addr, frame.len() as u16, self.tx.gen);
33        self.tx.advance();
34        fence(Ordering::SeqCst);
35        self.pt.wr32(PT_TXPROD, self.tx.next as u32)?;
36
37        let (dma, comp, clean, next) = (
38            &self.dma,
39            &mut self.tx_comp,
40            &mut self.tx_clean,
41            self.tx.next,
42        );
43        if self.irq.wait_for(TX_PARKS, TX_POLLS, || {
44            reap_tx(dma, comp, clean);
45            Ok(*clean == next)
46        })? {
47            Ok(())
48        } else {
49            Err(ViError::IO)
50        }
51    }
52
53    /// Take the next received frame into `out`; 0 when none is ready.
54    /// Fragmented, errored or oversized frames are dropped.
55    pub fn recv_frame(&mut self, out: &mut [u8]) -> usize {
56        loop {
57            let dma = &self.dma;
58            let Some(done) = rx_completion(|o| dma.read32(o), &self.rx_comp) else {
59                return 0;
60            };
61            self.rx_comp.advance();
62            if done.ring != 0 || done.index >= RX_SLOTS {
63                continue; // nothing this cell posted
64            }
65            let good = done.sop && done.eop && !done.error && done.len > 0 && done.len <= BUF_SIZE;
66            let len = if good { done.len.min(out.len()) } else { 0 };
67            self.dma
68                .copy_out(RX_BUF_OFF + done.index * BUF_SIZE, &mut out[..len]);
69            self.refill();
70            if good {
71                return len;
72            }
73        }
74    }
75
76    /// Hand the empty ring-0 slot back to the device with its own buffer.
77    fn refill(&mut self) {
78        let slot = self.rx_fill.next;
79        let addr = self.dma.iova(RX_BUF_OFF + slot * BUF_SIZE);
80        post_rx(&mut self.dma, slot, addr, self.rx_fill.gen);
81        self.rx_fill.advance();
82        if self.dma.read8(QUEUES_OFF + RXQ_UPDATE_RXHEAD_OFF) != 0 {
83            fence(Ordering::SeqCst);
84            let _ = self.pt.wr32(PT_RXPROD, self.rx_fill.next as u32);
85        }
86    }
87}
88
89/// Consume every TX completion the device wrote.
90fn reap_tx(dma: &DmaArea, comp: &mut GenRing, clean: &mut usize) {
91    while let Some(index) = tx_completion(|o| dma.read32(o), comp) {
92        comp.advance();
93        if index < TX_SLOTS {
94            *clean = (index + 1) % TX_SLOTS;
95        }
96    }
97}

99/// Acknowledge pending device events; queue errors are logged.
100pub(crate) fn ack_events(vd: &MmioRegion, dma: &DmaArea) -> ViResult<()> {
101    let events = dma.read32(SHARED_OFF + EVENT_OFF);
102    if events != 0 {
103        vd.wr32(VD_ECR, events)?;
104        if events & (EVENT_RQERROR | EVENT_TQERROR) != 0 {
105            let _ = print_fmt(format_args!(
106                "[vmxnet3] device queue error event {events:#x}\n"
107            ));
108        }
109    }
110    Ok(())
111}
