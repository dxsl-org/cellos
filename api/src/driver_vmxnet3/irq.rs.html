1//! MSI-X wake-ups for bounded bring-up, link and TX-completion waits (H-S3-4 b).
2//!
3//! An interrupt is only a hint to re-read the device: every park is bounded by
4//! [`PARK_TICKS`], every return re-reads the condition, and every wait has its
5//! own park budget, so a lost or revoked interrupt degrades to slow polling,
6//! never to a hang. RX is **not** interrupt-driven: the net service asks for
7//! frames (`OP_RX`), so the RX vector stays masked (`rx=request-driven`).
8//!
9//! Vectors (`shared::IntrPlan`): TX queue 0 → 0, RX queue 0 → 1 (masked),
10//! events (link change, queue errors) → 2. `DriverShared.automask` is 0, so
11//! the device sends a message for every cause while its vector is unmasked
12//! and the cell never has to re-arm a vector.
13
14use api::completion::source::IRQ;
15use api::pcie_function::IRQ_MODE_ALLOW_MSIX;
16use ostd::io::print_fmt;
17use ostd::ipc::{wait_any, Wake};
18use ostd::syscall::sys_alloc_pcie_irq;
19use types::ViResult;
20
21/// Longest single park, in scheduler ticks (10 ms each).
22pub(crate) const PARK_TICKS: u64 = 10;
23/// MSI-X entries asked for: TX queue 0, RX queue 0, events.
24pub(crate) const WANT_VECTORS: usize = 3;
25/// Condition polls before the first park (QEMU completes TX synchronously).
26const SPIN_BEFORE_PARK: u32 = 4_096;
27/// Park expiries logged per cell lifetime.
28const EXPIRY_LOG_LIMIT: u32 = 3;
29/// `DriverShared.ictrl`: every interrupt disabled (the polled configuration).
30const ICTRL_DISABLE_ALL: u32 = 0x01;
31
32/// Vector routing (`nintr`, per-queue `intr_idx`, `evintr`) and `ictrl`.
33#[derive(Clone, Copy, Debug, Eq, PartialEq)]
34pub(crate) struct IntrPlan {
35    pub nintr: u8,
36    pub tx: u8,
37    pub rx: u8,
38    pub event: u8,
39    pub ictrl: u32,
40}
41
42impl IntrPlan {
43    /// `granted` MSI-X vectors (`None`: polled). With three: TX → 0, RX → 1,
44    /// events → 2. Fewer fold onto the last granted vector; the RX vector
45    /// stays masked unless it is shared with TX or events (RX is
46    /// request-driven).
47    pub(crate) const fn for_vectors(granted: Option<u8>) -> Self {
48        let (nintr, ictrl) = match granted {
49            None | Some(0) => (1, ICTRL_DISABLE_ALL),
50            Some(n) if n > 3 => (3, 0),
51            Some(n) => (n, 0),
52        };
53        let last = nintr - 1;
54        let rx = if last >= 1 { 1 } else { 0 };
55        Self {
56            nintr,
57            tx: 0,
58            rx,
59            event: last,
60            ictrl,
61        }
62    }
63
64    /// Whether the RX vector is masked: only when it serves RX alone. Masking
65    /// a vector RX shares with TX or events would silence those causes too
66    /// (with two vectors RX and events share vector 1).
67    pub(crate) const fn masks_rx(&self) -> bool {
68        self.rx != self.tx && self.rx != self.event
69    }
70}

72/// How the controller's waits learn that something changed.
73pub(crate) struct NicIrq {
74    /// MSI-X entries held, or `None` when polled.
75    vectors: Option<u8>,
76    /// Why the cell is polled (`irq=polled reason=…`).
77    pub reason: &'static str,
78    /// Parks ended by an interrupt / by their deadline (bind-line evidence).
79    pub wakes: u32,
80    pub expiries: u32,
81}
82
83impl NicIrq {
84    /// Ask 435 for [`WANT_VECTORS`] MSI-X entries of the owned function `bdf`.
85    /// MSI is not accepted: per-queue vectors need MSI-X.
86    pub(crate) fn setup(bdf: u32) -> Self {
87        let (vectors, reason) = match sys_alloc_pcie_irq(bdf, WANT_VECTORS, IRQ_MODE_ALLOW_MSIX) {
88            Ok(alloc) => (Some(alloc.count), ""),
89            Err(error) => {
90                let _ = print_fmt(format_args!(
91                    "[vmxnet3] MSI-X refused ({:?}); polling\n",
92                    error
93                ));
94                (None, "alloc-refused")
95            }
96        };
97        Self {
98            vectors,
99            reason,
100            wakes: 0,
101            expiries: 0,
102        }
103    }
104
105    /// MSI-X entries held (`None` when polled).
106    pub(crate) fn vectors(&self) -> Option<u8> {
107        self.vectors
108    }
109
110    /// Wait (bounded) until `ready` holds: a short spin, then up to `parks`
111    /// interrupt parks with MSI-X, or `polls` spins when polled. The device is
112    /// re-read after every return. `Ok(false)` means the budget ran out.
113    pub(crate) fn wait_for(
114        &mut self,
115        parks: u32,
116        polls: u32,
117        mut ready: impl FnMut() -> ViResult<bool>,
118    ) -> ViResult<bool> {
119        for _ in 0..SPIN_BEFORE_PARK {
120            if ready()? {
121                return Ok(true);
122            }
123            core::hint::spin_loop();
124        }
125        if self.vectors.is_some() {
126            for _ in 0..parks {
127                if !self.park() {
128                    break;
129                }
130                if ready()? {
131                    return Ok(true);
132                }
133            }
134            if self.vectors.is_some() {
135                return Ok(false);
136            }
137        }
138        for _ in 0..polls {
139            if ready()? {
140                return Ok(true);
141            }
142            core::hint::spin_loop();
143        }
144        Ok(false)
145    }
146
147    /// Park on this task's MSI-X lines for at most [`PARK_TICKS`]. `false`
148    /// means the wait was refused and the cell is now polled.
149    fn park(&mut self) -> bool {
150        match wait_any(IRQ, PARK_TICKS) {
151            Ok(Wake::Irq(_)) => {
152                self.wakes += 1;
153                true
154            }
155            Ok(_) => {
156                self.expiries += 1;
157                if self.expiries <= EXPIRY_LOG_LIMIT {
158                    let _ = print_fmt(format_args!(
159                        "[vmxnet3] irq wait expired after {} ticks; re-reading the device\n",
160                        PARK_TICKS
161                    ));
162                }
163                true
164            }
165            Err(error) => {
166                let _ = print_fmt(format_args!(
167                    "[vmxnet3] irq=polled reason=wait-refused ({:?})\n",
168                    error
169                ));
170                self.vectors = None;
171                self.reason = "wait-refused";
172                false
173            }
174        }
175    }
176}

178#[cfg(test)]
179mod tests {
180    use super::*;
181
182    #[test]
183    fn interrupt_plan_folds_onto_granted_vectors_and_polls_without_msix() {
184        let three = IntrPlan::for_vectors(Some(3));
185        assert_eq!(
186            (three.nintr, three.tx, three.rx, three.event, three.ictrl),
187            (3, 0, 1, 2, 0)
188        );
189        assert_eq!(
190            IntrPlan::for_vectors(Some(8)),
191            three,
192            "never more than three"
193        );
194        let two = IntrPlan::for_vectors(Some(2));
195        assert_eq!((two.nintr, two.tx, two.rx, two.event), (2, 0, 1, 1));
196        let one = IntrPlan::for_vectors(Some(1));
197        assert_eq!(
198            (one.nintr, one.tx, one.rx, one.event, one.ictrl),
199            (1, 0, 0, 0, 0)
200        );
201        let polled = IntrPlan::for_vectors(None);
202        assert_eq!((polled.nintr, polled.ictrl), (1, ICTRL_DISABLE_ALL));
203        assert_eq!(IntrPlan::for_vectors(Some(0)), polled);
204        // RX alone on its vector is masked; a vector it shares never is.
205        assert!(three.masks_rx());
206        assert!(!two.masks_rx(), "vector 1 also carries events");
207        assert!(!one.masks_rx(), "vector 0 also carries TX and events");
208    }
