1//! DMA layout, generation-bit rings and descriptor encoding (pure; host-tested).
2//!
3//! Every vmxnet3 ring is owned by whichever side's generation bit a slot
4//! carries: a producer writes a slot with its current generation and flips the
5//! generation each time its index wraps, and a consumer accepts a slot only
6//! while the slot's bit equals its own generation. Both sides start at 1, so a
7//! zeroed ring is empty for the device's command rings and free for its
8//! completion rings. Descriptor bit positions follow FreeBSD's `if_vmxreg.h`
9//! (`vmxnet3_txdesc`, `vmxnet3_rxdesc`, `vmxnet3_txcompdesc`,
10//! `vmxnet3_rxcompdesc`; ISC, see `NOTICE`).
11
12use crate::shared::{Sink, DRIVER_SHARED_LEN, QUEUES_LEN};
13use core::sync::atomic::{fence, Ordering};
14
15pub(crate) const TX_SLOTS: usize = 32;
16pub(crate) const RX_SLOTS: usize = 32;
17pub(crate) const RX_COMP_SLOTS: usize = 2 * RX_SLOTS;
18/// Command and completion descriptors are 16 bytes; TX data-ring entries 128.
19const DESC_LEN: usize = 16;
20const TX_DATA_DESC_LEN: usize = 128;
21/// Frame buffer per slot (one Ethernet frame without FCS fits).
22pub const BUF_SIZE: usize = 2048;
23
24// ── Layout of the single DMA area ────────────────────────────────────────────
25
26pub(crate) const SHARED_OFF: usize = 0;
27pub(crate) const QUEUES_OFF: usize = 1024;
28pub(crate) const TX_RING_OFF: usize = 4096;
29pub(crate) const TX_COMP_OFF: usize = TX_RING_OFF + TX_SLOTS * DESC_LEN;
30pub(crate) const RX_RING_OFF: [usize; 2] = [TX_COMP_OFF + TX_SLOTS * DESC_LEN, RX_RING_OFF_1];
31const RX_RING_OFF_1: usize = TX_COMP_OFF + (TX_SLOTS + RX_SLOTS) * DESC_LEN;
32pub(crate) const RX_COMP_OFF: usize = RX_RING_OFF_1 + RX_SLOTS * DESC_LEN;
33pub(crate) const TX_DATA_OFF: usize = 2 * 4096;
34pub(crate) const TX_BUF_OFF: usize = TX_DATA_OFF + TX_SLOTS * TX_DATA_DESC_LEN;
35pub(crate) const RX_BUF_OFF: usize = TX_BUF_OFF + TX_SLOTS * BUF_SIZE;
36pub(crate) const DMA_PAGES: usize = (RX_BUF_OFF + RX_SLOTS * BUF_SIZE) / 4096;
37
38const _: () = assert!(SHARED_OFF + DRIVER_SHARED_LEN <= QUEUES_OFF);
39const _: () = assert!(QUEUES_OFF.is_multiple_of(128) && QUEUES_OFF + QUEUES_LEN <= TX_RING_OFF);
40const _: () = assert!(RX_COMP_OFF + RX_COMP_SLOTS * DESC_LEN <= TX_DATA_OFF);
41const _: () = assert!(
42    TX_COMP_OFF.is_multiple_of(512)
43        && RX_RING_OFF_1.is_multiple_of(512)
44        && RX_COMP_OFF.is_multiple_of(512)
45);
46const _: () = assert!((RX_BUF_OFF + RX_SLOTS * BUF_SIZE).is_multiple_of(4096));
47
48// ── Descriptor bits ──────────────────────────────────────────────────────────
49
50const LEN_MASK: u32 = 0x3FFF;
51const TXD_GEN: u32 = 1 << 14;
52const TXD_EOP: u32 = 1 << 12;
53const TXD_CQ: u32 = 1 << 13;
54const RXD_GEN: u32 = 1 << 31;
55const COMP_GEN: u32 = 1 << 31;
56const RXCD_EOP: u32 = 1 << 14;
57const RXCD_SOP: u32 = 1 << 15;
58const RXCD_ERR: u32 = 1 << 14;
59const IDX_MASK: u32 = 0xFFF;
60
61/// One side's position in a ring: next slot and current generation.
62#[derive(Clone, Copy, Debug, Eq, PartialEq)]
63pub(crate) struct GenRing {
64    pub next: usize,
65    pub gen: bool,
66    size: usize,
67}
68
69impl GenRing {
70    pub(crate) const fn new(size: usize) -> Self {
71        Self {
72            next: 0,
73            gen: true,
74            size,
75        }
76    }
77
78    /// Step past the current slot, flipping the generation on wrap.
79    pub(crate) fn advance(&mut self) {
80        self.next += 1;
81        if self.next == self.size {
82            self.next = 0;
83            self.gen = !self.gen;
84        }
85    }
86}
87
88/// Write TX descriptor `slot` for one `len`-byte frame at `addr` with
89/// generation `gen`; the generation word goes last, after a release fence.
90pub(crate) fn post_tx(s: &mut impl Sink, slot: usize, addr: u64, len: u16, gen: bool) {
91    let desc = TX_RING_OFF + slot % TX_SLOTS * DESC_LEN;
92    s.put64(desc, addr);
93    s.put32(desc + 12, TXD_EOP | TXD_CQ);
94    fence(Ordering::Release);
95    s.put32(
96        desc + 8,
97        (u32::from(len) & LEN_MASK) | if gen { TXD_GEN } else { 0 },
98    );
99}
100
101/// Write RX ring-0 descriptor `slot` as a head buffer of [`BUF_SIZE`] bytes.
102pub(crate) fn post_rx(s: &mut impl Sink, slot: usize, addr: u64, gen: bool) {
103    let desc = RX_RING_OFF[0] + slot % RX_SLOTS * DESC_LEN;
104    s.put64(desc, addr);
105    s.put32(desc + 12, 0);
106    fence(Ordering::Release);
107    s.put32(
108        desc + 8,
109        BUF_SIZE as u32 & LEN_MASK | if gen { RXD_GEN } else { 0 },
110    );
111}
112
113/// The TX ring index of the completion at `ring.next`, once the device wrote
114/// it with the consumer's generation.
115pub(crate) fn tx_completion(read32: impl Fn(usize) -> u32, ring: &GenRing) -> Option<usize> {
116    let desc = TX_COMP_OFF + ring.next * DESC_LEN;
117    if (read32(desc + 12) & COMP_GEN != 0) != ring.gen {
118        return None;
119    }
120    fence(Ordering::Acquire);
121    Some((read32(desc) & IDX_MASK) as usize)
122}
123
124/// A received-frame completion.
125#[derive(Clone, Copy, Debug, Eq, PartialEq)]
126pub(crate) struct RxCompletion {
127    pub index: usize,
128    /// 0 = ring 0 (head buffers), 1 = ring 1 (body buffers).
129    pub ring: u32,
130    pub sop: bool,
131    pub eop: bool,
132    pub error: bool,
133    pub len: usize,
134}
135
136/// The RX completion at `ring.next`, once written with the consumer's
137/// generation.
138pub(crate) fn rx_completion(read32: impl Fn(usize) -> u32, ring: &GenRing) -> Option<RxCompletion> {
139    let desc = RX_COMP_OFF + ring.next * DESC_LEN;
140    if (read32(desc + 12) & COMP_GEN != 0) != ring.gen {
141        return None;
142    }
143    fence(Ordering::Acquire);
144    let (word0, word2) = (read32(desc), read32(desc + 8));
145    Some(RxCompletion {
146        index: (word0 & IDX_MASK) as usize,
147        ring: (word0 >> 16) & 0x3FF,
148        sop: word0 & RXCD_SOP != 0,
149        eop: word0 & RXCD_EOP != 0,
150        error: word2 & RXCD_ERR != 0,
151        len: (word2 & LEN_MASK) as usize,
152    })
153}
