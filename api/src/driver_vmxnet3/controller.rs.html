1//! vmxnet3 bring-up (the data path is in `datapath.rs`).
2//!
3//! Order: revision negotiation, device reset, permanent MAC, MSI-X (435), one
4//! DMA allocation authorized for the BDF before any of its addresses is
5//! written anywhere, RX ring 0 filled, queue descriptors and `DriverShared`
6//! written, `DriverShared` address programmed, `ACTIVATE_DEV`, RX producers,
7//! receive filters, vector masks, bounded link wait. The device is reset
8//! before `GrantDma` turns bus mastering on; the kernel turns memory decoding
9//! on when the cell claims the function (`FindPcieDevice*`), so the first
10//! register read does not depend on firmware.
11

27/// Link wait: up to 50 parks (~5 s) with MSI-X, or this many polls.
28const LINK_PARKS: u32 = 50;
29const LINK_POLLS: u32 = 2_000_000;
30
31/// Why bring-up stopped. `Debug` (what the skeleton prints) is the `Display`
32/// text.
33#[derive(Clone, Copy)]
34pub enum InitError {
35    Mmio(ViError),
36    Refused(Refusal),
37    /// The permanent MAC is zero or multicast.
38    Mac([u8; 6]),
39    /// DMA allocation or authorization failed.
40    Dma,
41    /// `ACTIVATE_DEV` returned this non-zero status.
42    Activate(u32),
43}
44
45impl From<ViError> for InitError {
46    fn from(error: ViError) -> Self {
47        Self::Mmio(error)
48    }
49}
50
51impl core::fmt::Display for InitError {
52    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
53        match self {
54            Self::Mmio(error) => write!(f, "register access: {error:?}"),
55            Self::Refused(Refusal::Revision(v)) => write!(f, "no revision 1 (VRRS={v:#x})"),
56            Self::Refused(Refusal::Upt(v)) => write!(f, "no UPT version 1 (UVRS={v:#x})"),
57            Self::Mac(mac) => write!(f, "unusable MAC {}", MacFmt(*mac)),
58            Self::Dma => write!(f, "DMA allocation/authorization failed"),
59            Self::Activate(status) => write!(f, "ACTIVATE_DEV status {status:#x}"),
60        }
61    }
62}
63
64impl core::fmt::Debug for InitError {
65    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
66        core::fmt::Display::fmt(self, f)
67    }
68}

86impl NicDevice for Vmxnet3 {
87    const TAG: &'static str = "[vmxnet3]";
88    const VENDOR: u16 = VENDOR_VMWARE;
89    const QUERY_IDS: &'static [u16] = &QUERY_DEVICE_IDS;
90    const ABSENT_LINE: Option<&'static str> =
91        Some("no vmxnet3 controller present; driver cell idle");
92    type Identity = Identity;
93    type Error = InitError;
94
95    fn classify(vendor: u16, device: u16) -> Option<Identity> {
96        identity::classify(vendor, device)
97    }
98
99    fn identity_name(id: &Identity) -> &'static str {
100        id.name
101    }
102
103    /// Claim BAR0 (PT) and BAR1 (VD) from the 434 function record, then bring
104    /// the device up.
105    fn init(probe: &Probe, _id: Identity) -> Result<Self, InitError> {
106        let bdf = probe.bdf();
107        let record = match sys_query_pcie_function(bdf) {
108            Ok(record) => record,
109            Err(error) => probe.fail(format_args!("query {bdf:#x} failed: {error:?}")),
110        };
111        let pt = claim(probe, 0, &record.bars[0], PT_MIN);
112        let vd = claim(probe, 1, &record.bars[1], VD_MIN);
113        Self::new(pt, vd, bdf)
114    }
115
116    /// Revision, link, MAC, and the interrupt evidence: MSI-X entries held (0
117    /// when polled), why the cell is polled, and how many parks so far ended
118    /// by an interrupt / by their deadline.
119    fn bind_detail(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
120        let vectors = self.irq.vectors().unwrap_or(0);
121        let (irq, why) = if vectors > 0 {
122            ("msix", "")
123        } else {
124            ("polled reason=", self.irq.reason())
125        };
126        write!(
127            f,
128            "rev=1 link_up={} speed={}Mb/s mac={} irq={}{} rx=request-driven vectors={} irq_wakes={} park_expiries={}",
129            self.link_up,
130            self.link_mbps,
131            MacFmt(self.mac),
132            irq,
133            why,
134            vectors,
135            self.irq.wakes(),
136            self.irq.expiries()
137        )
138    }
139
140    fn mac(&self) -> [u8; 6] {
141        self.mac
142    }
143
144    fn send(&mut self, frame: &[u8]) -> ViResult<()> {
145        self.send_frame(frame)
146    }
147
148    fn recv(&mut self, out: &mut [u8]) -> usize {
149        self.recv_frame(out)
150    }
151}

153/// Claim memory BAR `index` when it holds at least `min` bytes; exits 1
154/// otherwise.
155fn claim(probe: &Probe, index: usize, bar: &PcieBar, min: usize) -> MmioRegion {
156    let memory = bar.kind == BAR_KIND_MEM32 || bar.kind == BAR_KIND_MEM64;
157    let len = usize::try_from(bar.len).unwrap_or(0);
158    if !memory || bar.base == 0 || len < min {
159        probe.fail(format_args!(
160            "BAR{index} unusable (kind={} base={:#x} len={:#x})",
161            bar.kind, bar.base, bar.len
162        ));
163    }
164    match mmio::request_region(bar.base as usize, len) {
165        Ok(region) => region,
166        Err(error) => probe.fail(format_args!("BAR{index} claim failed: {error:?}")),
167    }
168}

170impl Vmxnet3 {
171    /// Bring the device up on BAR0 `pt` and BAR1 `vd` for the function `bdf`.
172    fn new(pt: MmioRegion, vd: MmioRegion, bdf: u32) -> Result<Self, InitError> {
173        negotiate(&vd)?.map_err(InitError::Refused)?;
174        vd.wr32(VD_CMD, CMD_RESET_DEV)?;
175        let mac = permanent_mac(&vd)?;
176        if !mac_usable(mac) {
177            return Err(InitError::Mac(mac));
178        }
179        let irq = BoundedWait::setup(Self::TAG, bdf, WANT_VECTORS);
180        let plan = IntrPlan::for_vectors(irq.vectors());
181        let dma = DmaArea::new(DMA_PAGES, bdf).ok_or(InitError::Dma)?;
182        // From here on, dropping `ctrl` quiesces and resets the device.
183        let mut ctrl = Self {
184            pt,
185            vd,
186            dma,
187            irq,
188            tx: GenRing::new(TX_SLOTS),
189            tx_comp: GenRing::new(TX_SLOTS),
190            tx_clean: 0,
191            rx_fill: GenRing::new(RX_SLOTS),
192            rx_comp: GenRing::new(RX_COMP_SLOTS),
193            mac,
194            link_up: false,
195            link_mbps: 0,
196        };
197        ctrl.configure(plan)?;
198        ctrl.wait_link()?;
199        Ok(ctrl)
200    }
201
202    fn configure(&mut self, plan: IntrPlan) -> Result<(), InitError> {
203        let dma = &mut self.dma;
204        dma.zero(0, DMA_PAGES * 4096);
205        // One slot stays empty so "producer == consumer" always means empty.
206        for _ in 0..RX_SLOTS - 1 {
207            let slot = self.rx_fill.next;
208            let addr = dma.iova(RX_BUF_OFF + slot * BUF_SIZE);
209            post_rx(dma, slot, addr, self.rx_fill.gen);
210            self.rx_fill.advance();
211        }
212        let queues = QueueConfig {
213            tx_ring: dma.iova(TX_RING_OFF),
214            tx_data: dma.iova(TX_DATA_OFF),
215            tx_comp: dma.iova(TX_COMP_OFF),
216            rx_ring: [dma.iova(RX_RING_OFF[0]), dma.iova(RX_RING_OFF[1])],
217            rx_comp: dma.iova(RX_COMP_OFF),
218            tx_len: TX_SLOTS as u32,
219            rx_len: [RX_SLOTS as u32; 2],
220            rx_comp_len: RX_COMP_SLOTS as u32,
221        };
222        write_queues(dma, QUEUES_OFF, &queues, plan);
223        let queues_iova = dma.iova(QUEUES_OFF);
224        write_driver_shared(dma, SHARED_OFF, queues_iova, plan);
225        fence(Ordering::SeqCst);
226
227        let (pt, vd) = (&self.pt, &self.vd);
228        set_mac(vd, self.mac)?;
229        let shared = self.dma.iova(SHARED_OFF);
230        vd.wr32(VD_DSL, shared as u32)?;
231        vd.wr32(VD_DSH, (shared >> 32) as u32)?;
232        let status = query(vd, CMD_ACTIVATE_DEV)?;
233        if status != 0 {
234            return Err(InitError::Activate(status));
235        }
236        pt.wr32(PT_RXPROD, self.rx_fill.next as u32)?;
237        pt.wr32(PT_RXPROD2, 0)?;
238        for command in [
239            CMD_UPDATE_RX_MODE,
240            CMD_UPDATE_MAC_FILTERS,
241            CMD_UPDATE_VLAN_FILTERS,
242        ] {
243            vd.wr32(VD_CMD, command)?;
244        }
245        // PT writes are ignored before activation, so the masks go last. RX
246        // stays masked unless it shares the TX or the event vector.
247        if self.irq.vectors().is_some() {
248            pt.wr32(pt_imask(plan.tx), 0)?;
249            pt.wr32(pt_imask(plan.event), 0)?;
250            if plan.masks_rx() {
251                pt.wr32(pt_imask(plan.rx), 1)?;
252            }
253        } else {
254            pt.wr32(pt_imask(0), 1)?;
255        }
256        Ok(())
257    }
258
259    fn wait_link(&mut self) -> ViResult<()> {
260        let (vd, dma) = (&self.vd, &self.dma);
261        let mut link = 0;
262        self.link_up = self.irq.wait_for(LINK_PARKS, LINK_POLLS, || {
263            ack_events(vd, dma)?;
264            link = query(vd, CMD_GET_LINK)?;
265            Ok(link & LINK_UP != 0)
266        })?;
267        self.link_mbps = link >> 16;
268        Ok(())
269    }
270}

272impl Drop for Vmxnet3 {
273    /// Stop all DMA before the cell lets go of the device.
274    fn drop(&mut self) {
275        let _ = self.vd.wr32(VD_CMD, CMD_QUIESCE_DEV);
276        let _ = self.vd.wr32(VD_CMD, CMD_RESET_DEV);
277    }
