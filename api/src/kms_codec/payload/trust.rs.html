2//! Fixed wire payloads of the SSH host-trust store (opcodes 25-29).
3//!
4//! A query names a host (up to 253 bytes), an algorithm and a complete SSH key blob
5//! (up to 4 KiB), which do not fit the 112-byte payload, so it is staged like an SSH
6//! import: `BeginTrustQuery` declares the kind, port, field lengths and the SHA-256 of
7//! the staged bytes `host || algorithm || key`; `TrustQueryChunk` delivers them in the
8//! canonical 96-byte chunks ([`crate::ssh_chunk_len`]); `CommitTrustQuery` verifies
9//! count and digest and answers a [`TrustOutcomePayload`]. Chunk, chunk ack, Begin
10//! response, Commit request and Abort request are the SSH transfer payloads
11//! ([`super::SshChunkRequestPayload`], [`super::SshChunkAckPayload`],
12//! [`super::SshTransferBeginResponsePayload`], [`super::SshImportCommitRequestPayload`],
13//! [`super::SshAbortRequestPayload`]).
14//!
15//! The outcome byte is one of `wire::ipc::ssh_trust` (0-7); this crate does not depend on
16//! `wire`, so only its range is checked here.
17
18use super::{put_u16, put_u64, read_32, read_u16, read_u64};
19use crate::KmsTrustQueryKind;
20
21/// Longest host identifier (a DNS name). Frozen.
22pub const KMS_TRUST_HOST_MAX: usize = 253;
23/// Longest algorithm name. Frozen.
24pub const KMS_TRUST_ALGORITHM_MAX: usize = 64;
25/// Longest SSH key blob. Frozen.
26pub const KMS_TRUST_KEY_MAX: usize = 4096;
27/// Longest staged query (`host || algorithm || key`).
28pub const KMS_TRUST_QUERY_MAX: usize =
29    KMS_TRUST_HOST_MAX + KMS_TRUST_ALGORITHM_MAX + KMS_TRUST_KEY_MAX;
30/// Highest `wire::ipc::ssh_trust` outcome (`PENDING`).
31pub const KMS_TRUST_OUTCOME_MAX: u8 = 7;
32/// `wire::ipc::ssh_trust::UNKNOWN`: the outcome that accompanies an issued challenge.
33pub const KMS_TRUST_OUTCOME_UNKNOWN: u8 = 1;
34
35fn is_zero(bytes: &[u8]) -> bool {
36    bytes.iter().all(|byte| *byte == 0)
37}
38
39/// An `ObjectRefV1` (`id`, `generation`) as two little-endian u64 words.
40fn read_ref(bytes: &[u8], at: usize) -> (u64, u64) {
41    (read_u64(bytes, at), read_u64(bytes, at + 8))
42}
43
44fn put_ref(out: &mut [u8], at: usize, value: (u64, u64)) {
45    put_u64(out, at, value.0);
46    put_u64(out, at + 8, value.1);
47}

49/// `BeginTrustQuery` request.
50#[repr(C)]
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub struct TrustQueryBeginRequestPayload {
53    pub kind: KmsTrustQueryKind,
54    /// `1..=64`.
55    pub algorithm_len: u8,
56    /// Nonzero.
57    pub port: u16,
58    /// `1..=253`.
59    pub host_len: u16,
60    /// `1..=4096`.
61    pub key_len: u16,
62    /// Challenge: the caller's terminal-session epoch. Lookup: zero.
63    pub epoch: u64,
64    /// Challenge: the caller's own terminal session (`id`, `generation`), never zero.
65    /// Lookup: zero.
66    pub session: (u64, u64),
67    /// SHA-256 of `host || algorithm || key`; never all zero.
68    pub query_sha256: [u8; 32],
69}
70
71impl TrustQueryBeginRequestPayload {
72    pub const LEN: usize = 64;
73
74    pub fn encode(&self) -> [u8; Self::LEN] {
75        let mut out = [0u8; Self::LEN];
76        out[0] = self.kind as u8;
77        out[1] = self.algorithm_len;
78        put_u16(&mut out, 2, self.port);
79        put_u16(&mut out, 4, self.host_len);
80        put_u16(&mut out, 6, self.key_len);
81        put_u64(&mut out, 8, self.epoch);
82        put_ref(&mut out, 16, self.session);
83        out[32..64].copy_from_slice(&self.query_sha256);
84        out
85    }
86
87    /// Strict decode: every length within its bound, a nonzero port and digest, and a
88    /// session (and epoch) exactly when the kind is a challenge.
89    pub fn decode(bytes: &[u8]) -> Option<Self> {
90        if bytes.len() != Self::LEN {
91            return None;
92        }
93        let kind = KmsTrustQueryKind::try_from(bytes[0]).ok()?;
94        let algorithm_len = bytes[1];
95        let port = read_u16(bytes, 2);
96        let host_len = read_u16(bytes, 4);
97        let key_len = read_u16(bytes, 6);
98        let epoch = read_u64(bytes, 8);
99        let session = read_ref(bytes, 16);
100        let query_sha256 = read_32(bytes, 32);
101        let no_session = session == (0, 0);
102        let session_ok = match kind {
103            KmsTrustQueryKind::Lookup => no_session && epoch == 0,
104            KmsTrustQueryKind::Challenge => !no_session,
105        };
106        if port == 0
107            || !(1..=KMS_TRUST_ALGORITHM_MAX).contains(&usize::from(algorithm_len))
108            || !(1..=KMS_TRUST_HOST_MAX).contains(&usize::from(host_len))
109            || !(1..=KMS_TRUST_KEY_MAX).contains(&usize::from(key_len))
110            || is_zero(&query_sha256)
111            || !session_ok
112        {
113            return None;
114        }
115        Some(Self {
116            kind,
117            algorithm_len,
118            port,
119            host_len,
120            key_len,
121            epoch,
122            session,
123            query_sha256,
124        })
125    }
126
127    /// Length of the staged bytes `host || algorithm || key`.
128    pub fn total_len(&self) -> u32 {
129        u32::from(self.host_len) + u32::from(self.algorithm_len) + u32::from(self.key_len)
130    }
131}

133/// `CommitTrustQuery` and `CommitTrustChallenge` response.
134///
135/// `outcome` is one `wire::ipc::ssh_trust` code. A challenge issued by `CommitTrustQuery`
136/// sets `challenged`, carries the outcome Unknown and both nonzero refs: the input-issued
137/// `confirmation` the caller begins on its terminal, and the KMS-issued `challenge` it
138/// commits. Every other reply has `challenged == 0` and zero refs.
139#[repr(C)]
140#[derive(Debug, Clone, Copy, PartialEq, Eq)]
141pub struct TrustOutcomePayload {
142    pub outcome: u8,
143    pub challenged: u8,
144    pub reserved: [u8; 6],
145    pub confirmation: (u64, u64),
146    pub challenge: (u64, u64),
147}
148
149impl TrustOutcomePayload {
150    pub const LEN: usize = 40;
151
152    /// A reply carrying only `outcome`.
153    pub const fn outcome(outcome: u8) -> Self {
154        Self {
155            outcome,
156            challenged: 0,
157            reserved: [0; 6],
158            confirmation: (0, 0),
159            challenge: (0, 0),
160        }
161    }
162
163    /// An issued challenge.
164    pub const fn challenged(confirmation: (u64, u64), challenge: (u64, u64)) -> Self {
165        Self {
166            outcome: KMS_TRUST_OUTCOME_UNKNOWN,
167            challenged: 1,
168            reserved: [0; 6],
169            confirmation,
170            challenge,
171        }
172    }
173
174    pub fn encode(&self) -> [u8; Self::LEN] {
175        let mut out = [0u8; Self::LEN];
176        out[0] = self.outcome;
177        out[1] = self.challenged;
178        out[2..8].copy_from_slice(&self.reserved);
179        put_ref(&mut out, 8, self.confirmation);
180        put_ref(&mut out, 24, self.challenge);
181        out
182    }
183
184    /// Strict decode: one of the two canonical shapes above.
185    pub fn decode(bytes: &[u8]) -> Option<Self> {
186        if bytes.len() != Self::LEN || !is_zero(&bytes[2..8]) {
187            return None;
188        }
189        let (outcome, confirmation, challenge) =
190            (bytes[0], read_ref(bytes, 8), read_ref(bytes, 24));
191        match bytes[1] {
192            0 if outcome <= KMS_TRUST_OUTCOME_MAX && is_zero(&bytes[8..40]) => {
193                Some(Self::outcome(outcome))
194            }
195            1 if outcome == KMS_TRUST_OUTCOME_UNKNOWN
196                && confirmation != (0, 0)
197                && challenge != (0, 0) =>
198            {
199                Some(Self::challenged(confirmation, challenge))
200            }
201            _ => None,
202        }
203    }
204}

206/// `CommitTrustChallenge` request: the KMS-issued challenge ref.
207#[repr(C)]
208#[derive(Debug, Clone, Copy, PartialEq, Eq)]
209pub struct TrustChallengeCommitRequestPayload {
210    /// Nonzero.
211    pub challenge: (u64, u64),
212}
213
214impl TrustChallengeCommitRequestPayload {
215    pub const LEN: usize = 16;
216
217    pub fn encode(&self) -> [u8; Self::LEN] {
218        let mut out = [0u8; Self::LEN];
219        put_ref(&mut out, 0, self.challenge);
220        out
221    }
222
223    pub fn decode(bytes: &[u8]) -> Option<Self> {
224        if bytes.len() != Self::LEN {
225            return None;
226        }
227        let challenge = read_ref(bytes, 0);
228        (challenge != (0, 0)).then_some(Self { challenge })
229    }
230}
