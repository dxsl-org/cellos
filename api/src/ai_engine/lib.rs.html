1//! Native CPU transformer engine for the Cellos unified AI inference service.
2//!
3//! Normative source: [`docs/specs/24-ai-inference-architecture.md`] §4. This crate is the CPU
4//! engine behind the service: it loads a GGUF checkpoint, tokenizes a prompt with the model's own
5//! tokenizer, runs a Llama-architecture forward pass with a per-session KV cache, samples the next
6//! token, and hands generated ids back to the caller in bounded slices.
7//!
8//! # Scope and invariants
9//!
10//! - **One architecture**: `general.architecture == "llama"` (Llama/Mistral/Qwen-style: RMSNorm,
11//!   RoPE, grouped-query attention, SwiGLU MLP). Any other architecture is refused with
12//!   [`EngineError::UnsupportedArchitecture`] rather than guessed at.
13//! - **Two weight layouts**: `Q8_0` tensors stay quantized and run through
14//!   [`tensor_math::matvec_q8_0`]; `F32`/`F16` tensors are materialized as `f32` at load time.
15//! - **Owned after load**: [`Engine::load`] copies everything it needs out of the caller's model
16//!   buffer, so the caller may drop or reuse that buffer. No borrowing, no leaking.
17//! - **Bounded work per call**: [`Engine::generate`] performs at most the caller's step budget.
18//!   One request can never monopolise the Cell's event loop, which is what makes cancel, fairness
19//!   between sessions, and bounded reply latency possible.
20//! - **No allocation in the token loop**: every scratch buffer is sized at load time.
21//! - **Determinism**: with `temperature_milli == 0` (greedy) or a fixed seed, identical inputs
22//!   produce identical token ids; the golden-oracle test pins this to an independent reference.
23//! - **No `unsafe`**: `#![forbid(unsafe_code)]`.
24//!
25//! # Memory
26//!
27//! [`Engine::resident_bytes`] reports weights + tokenizer tables + KV cache + scratch. [`Engine::load`]
28//! refuses a model above the caller's `memory_limit` instead of failing later mid-request.
29
30#![cfg_attr(not(test), no_std)]
31#![forbid(unsafe_code)]

56/// Failure to load a model. Every variant names the exact contract that was violated.
57#[derive(Debug, Clone, PartialEq, Eq)]
58pub enum EngineError {
59    /// The GGUF container is malformed.
60    Gguf(GgufError),
61    /// The tokenizer metadata is missing or not byte-level BPE.
62    Tokenizer(ai_tokenizer::TokenizerError),
63    /// `general.architecture` is absent or not `"llama"`.
64    UnsupportedArchitecture,
65    /// A required metadata key is missing.
66    MissingMetadata(&'static str),
67    /// A metadata value is outside the supported range.
68    InvalidMetadata(&'static str),
69    /// A required tensor is missing from the file.
70    MissingTensor(String),
71    /// A tensor's shape does not match the architecture metadata.
72    BadTensorShape(String),
73    /// A tensor uses a dtype this build cannot execute.
74    UnsupportedDType(u32),
75    /// The model needs more memory than the caller allowed.
76    TooLarge {
77        /// Bytes the loaded engine would hold.
78        needed: usize,
79        /// Caller-provided ceiling.
80        limit: usize,
81    },
82}
83
84impl fmt::Display for EngineError {
85    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
86        match self {
87            EngineError::Gguf(e) => write!(f, "gguf: {e:?}"),
88            EngineError::Tokenizer(e) => write!(f, "tokenizer: {e:?}"),
89            EngineError::UnsupportedArchitecture => write!(f, "unsupported architecture"),
90            EngineError::MissingMetadata(key) => write!(f, "missing metadata: {key}"),
91            EngineError::InvalidMetadata(key) => write!(f, "invalid metadata: {key}"),
92            EngineError::MissingTensor(name) => write!(f, "missing tensor: {name}"),
93            EngineError::BadTensorShape(name) => write!(f, "bad tensor shape: {name}"),
94            EngineError::UnsupportedDType(dtype) => write!(f, "unsupported dtype: {dtype}"),
95            EngineError::TooLarge { needed, limit } => {
96                write!(f, "model needs {needed} bytes, limit {limit}")
97            }
98        }
99    }
100}

127impl ModelConfig {
128    /// Per-head width (`n_embd / n_head`).
129    pub fn head_dim(&self) -> usize {
130        self.n_embd / self.n_head
131    }
132
133    /// Width of one K/V vector (`n_head_kv * head_dim`).
134    pub fn kv_dim(&self) -> usize {
135        self.n_head_kv * self.head_dim()
136    }
137
138    /// Query heads sharing one K/V head.
139    pub fn group_size(&self) -> usize {
140        self.n_head / self.n_head_kv
141    }

144/// A weight matrix, kept in whichever layout the file stored it in.
145enum Matrix {
146    /// Row-major `rows × cols` f32.
147    F32 { rows: usize, data: Vec<f32> },
148    /// `rows` rows of Q8_0 blocks (`cols` a multiple of 32).
149    Q8_0 {
150        rows: usize,
151        cols: usize,
152        data: Vec<u8>,
153    },
154}
155
156impl Matrix {
157    fn rows(&self) -> usize {
158        match self {
159            Matrix::F32 { rows, .. } | Matrix::Q8_0 { rows, .. } => *rows,
160        }
161    }
162
163    /// Write `W · x` into `out[..rows]`.
164    ///
165    /// The kernel writes straight into the destination: every call site passes an exactly-sized
166    /// slice, so a staging buffer would only add a full copy per projection — and for the tied
167    /// output projection (a `vocab × n_embd` matrix) that copy is what forced a per-row
168    /// dequantise of the whole vocabulary on every generated token.
169    fn project(&self, out: &mut [f32], x: &[f32]) -> Result<(), MathError> {
170        let rows = self.rows();
171        if out.len() < rows {
172            return Err(MathError::ShapeMismatch);
173        }
174        let out = &mut out[..rows];
175        match self {
176            Matrix::F32 { rows, data } => tensor_math::matvec(out, data, x, *rows, x.len()),
177            Matrix::Q8_0 { rows, cols, data } => {
178                tensor_math::matvec_q8_0(out, data, x, *rows, *cols)
179            }
180        }
181    }
182
183    fn bytes(&self) -> usize {
184        match self {
185            Matrix::F32 { data, .. } => data.len() * 4,
186            Matrix::Q8_0 { data, .. } => data.len(),
187        }
188    }
189}

191/// One transformer block's weights.
192struct Layer {
193    attn_norm: Vec<f32>,
194    wq: Matrix,
195    wk: Matrix,
196    wv: Matrix,
197    wo: Matrix,
198    ffn_norm: Vec<f32>,
199    w_gate: Matrix,
200    w_up: Matrix,
201    w_down: Matrix,
202}
203
204/// All weights of one model.
205struct Weights {
206    token_embd: Matrix,
207    output_norm: Vec<f32>,
208    /// `None` when the checkpoint ties output to the token embedding.
209    output: Option<Matrix>,
210    layers: Vec<Layer>,
211}
212
213impl Weights {
214    fn bytes(&self) -> usize {
215        let mut total = self.token_embd.bytes() + self.output_norm.len() * 4;
216        if let Some(output) = &self.output {
217            total += output.bytes();
218        }
219        for layer in &self.layers {
220            total += layer.attn_norm.len() * 4 + layer.ffn_norm.len() * 4;
221            total += layer.wq.bytes()
222                + layer.wk.bytes()
223                + layer.wv.bytes()
224                + layer.wo.bytes()
225                + layer.w_gate.bytes()
226                + layer.w_up.bytes()
227                + layer.w_down.bytes();
228        }
229        total
230    }
231}

233/// Key/value cache for one session: `n_layer × n_ctx × kv_dim` f32, filled left to right.
234struct KvCache {
235    k: Vec<f32>,
236    v: Vec<f32>,
237    len: usize,
238}
239
240impl KvCache {
241    fn new(cfg: &ModelConfig) -> Self {
242        let size = cfg.n_layer * cfg.n_ctx * cfg.kv_dim();
243        Self {
244            k: vec![0.0; size],
245            v: vec![0.0; size],
246            len: 0,
247        }
248    }
249
250    fn bytes(&self) -> usize {
251        (self.k.len() + self.v.len()) * 4
252    }
253
254    /// Cache slot for layer `layer` at position `pos`.
255    fn slot(&self, cfg: &ModelConfig, layer: usize, pos: usize) -> Range<usize> {
256        let stride = cfg.n_ctx * cfg.kv_dim();
257        let base = layer * stride + pos * cfg.kv_dim();
258        base..base + cfg.kv_dim()
259    }
260}

286/// One live generation session.
287struct Session {
288    id: u32,
289    /// Every token in the session: prompt first, then generated.
290    tokens: Vec<u32>,
291    /// Prompt length, fixed at submit time.
292    prompt_len: usize,
293    /// Ids generated since the caller last drained them.
294    pending: Vec<u32>,
295    /// Byte-level decode buffer; holds an incomplete trailing UTF-8 sequence between polls.
296    pending_bytes: Vec<u8>,
297    kv: KvCache,
298    /// Tokens already fed through the model.
299    computed: usize,
300    rng: Rng,
301    params: SamplingParams,
302    finish: Option<FinishReason>,
303}
304
305impl Session {
306    fn generated(&self) -> usize {
307        self.tokens.len().saturating_sub(self.prompt_len)
308    }
309
310    fn bytes(&self) -> usize {
311        self.tokens.len() * 4 + self.pending.len() * 4 + self.pending_bytes.len() + self.kv.bytes()
312    }
313}

385/// The engine: model weights plus a bounded session table.
386pub struct Engine {
387    cfg: ModelConfig,
388    tokenizer: Tokenizer,
389    weights: Weights,
390    sessions: [Option<Session>; MAX_SESSIONS as usize],
391    next_request_id: u32,
392    scratch: Scratch,
393}
394
395impl Engine {
396    /// Load a model from a GGUF image.
397    ///
398    /// Everything the engine needs is copied out of `bytes`, so the caller keeps ownership of the
399    /// buffer and may drop it (freeing the transient copy) once this returns. A model whose resident
400    /// size would exceed `memory_limit` is refused with [`EngineError::TooLarge`].
401    pub fn load(bytes: &[u8], memory_limit: usize) -> Result<Self, EngineError> {
402        let file = GgufFile::parse(bytes).map_err(EngineError::Gguf)?;
403        let tokenizer = Tokenizer::from_gguf(&file).map_err(EngineError::Tokenizer)?;
404        let cfg = read_config(&file, tokenizer.vocab_size())?;
405        let weights = read_weights(&file, &cfg)?;
406        let scratch = Scratch::new(&cfg);
407
408        let engine = Self {
409            cfg,
410            tokenizer,
411            weights,
412            sessions: [const { None }; MAX_SESSIONS as usize],
413            next_request_id: 1,
414            scratch,
415        };
416
417        let needed = engine.resident_bytes();
418        if needed > memory_limit {
419            return Err(EngineError::TooLarge {
420                needed,
421                limit: memory_limit,
422            });
423        }
424        Ok(engine)
425    }

437    /// Bytes held by weights, tokenizer tables, KV caches, and scratch buffers.
438    pub fn resident_bytes(&self) -> usize {
439        let sessions: usize = self.sessions.iter().flatten().map(Session::bytes).sum();
440        self.weights.bytes() + self.tokenizer.bytes() + sessions + self.scratch.bytes()
441    }

443    /// Capability description for [`ai_proto::AiRequest::Describe`].
444    pub fn describe(&self) -> Describe<'_> {
445        let live = self.sessions.iter().flatten().count() as u8;
446        Describe {
447            proto_version: 1,
448            active_backend: backend::CPU,
449            backends: backend::CPU,
450            model: self.cfg.name.as_str(),
451            arch: "llama",

453            context_tokens: self.cfg.n_ctx as u32,
454            vocab_size: self.cfg.vocab_size as u32,
455            live_sessions: live,
456            max_sessions: MAX_SESSIONS,
457            resident_bytes: self.resident_bytes().min(u32::MAX as usize) as u32,
458        }
459    }

469    /// Start a generation session and return its service-assigned id.
470    pub fn submit(&mut self, prompt: &str, params: SamplingParams) -> Result<u32, AiError> {
471        if self.sessions.iter().all(Option::is_some) {
472            return Err(AiError::Busy);
473        }
474        if params.max_tokens == 0 || params.max_tokens > ai_proto::MAX_TOKENS_PER_REQUEST {
475            return Err(AiError::BadRequest(limit::Violation::MaxTokensOutOfRange));
476        }
477
478        let mut tokens =
479            self.tokenizer
480                .encode_with_specials(prompt, self.tokenizer.add_bos(), false);
481        if tokens.is_empty() {
482            return Err(AiError::BadRequest(limit::Violation::PromptTooLong));
483        }
484        // Keep at least one free slot for the first generated token.
485        if tokens.len() >= self.cfg.n_ctx {
486            tokens.truncate(self.cfg.n_ctx.saturating_sub(1));
487        }
488
489        let request_id = self.allocate_id();
490        let session = Session {
491            id: request_id,
492            prompt_len: tokens.len(),
493            tokens,
494            pending: Vec::new(),
495            pending_bytes: Vec::new(),
496            kv: KvCache::new(&self.cfg),
497            computed: 0,
498            rng: Rng::new(u64::from(params.seed) ^ u64::from(request_id)),
499            params,
500            finish: None,
501        };
502        let slot = self
503            .sessions
504            .iter_mut()
505            .find(|slot| slot.is_none())
506            .ok_or(AiError::Busy)?;
507        *slot = Some(session);
508        Ok(request_id)
509    }

511    /// Advance a session by at most `budget` model steps (prefill or generated tokens).
512    ///
513    /// Returns [`AiError::UnknownRequest`] for an id this engine does not own.
514    pub fn generate(&mut self, request_id: u32, budget: usize) -> Result<Progress, AiError> {
515        let Engine {
516            cfg,
517            weights,
518            scratch,
519            sessions,
520            tokenizer,
521            ..
522        } = self;
523
524        let session = sessions
525            .iter_mut()
526            .flatten()
527            .find(|session| session.id == request_id)
528            .ok_or(AiError::UnknownRequest)?;
529
530        if let Some(finish) = session.finish {
531            return Ok(Progress {
532                done: true,
533                finish: Some(finish),
534            });
535        }
536
537        let eos = tokenizer.eos_id();
538        let mut steps = 0usize;
539        while steps < budget {
540            if session.computed < session.tokens.len() {
541                let token = session.tokens[session.computed];
542                let position = session.computed;
543                let kv = &mut session.kv;
544                forward(cfg, weights, scratch, kv, token, position)
545                    .map_err(|_| AiError::Internal)?;
546                session.computed += 1;
547                steps += 1;
548                continue;
549            }
550
551            // The sequence is fully computed: sample the next token.
552            if session.tokens.len() >= cfg.n_ctx {
553                session.finish = Some(FinishReason::ContextFull);
554                break;
555            }
556            let sampled = sample(cfg, weights, scratch, session)?;
557            session.tokens.push(sampled);
558            session.pending.push(sampled);
559            steps += 1;
560
561            if Some(sampled) == eos {
562                session.finish = Some(FinishReason::Stop);
563                break;
564            }
565            if session.generated() >= usize::from(session.params.max_tokens) {
566                session.finish = Some(FinishReason::Length);
567                break;
568            }
569        }
570
571        Ok(Progress {
572            done: session.finish.is_some(),
573            finish: session.finish,
574        })
575    }

577    /// Drain up to `max` generated ids plus their complete text.
578    pub fn drain(&mut self, request_id: u32, max: usize) -> Result<Drained, AiError> {
579        let Engine {
580            tokenizer,
581            sessions,
582            ..
583        } = self;
584        let session = sessions
585            .iter_mut()
586            .flatten()
587            .find(|session| session.id == request_id)
588            .ok_or(AiError::UnknownRequest)?;
589
590        let take = max.min(session.pending.len());
591        let ids: Vec<u32> = session.pending.drain(..take).collect();
592        for id in &ids {
593            if let Some(bytes) = tokenizer.token_bytes(*id) {
594                session.pending_bytes.extend_from_slice(&bytes);
595            }
596        }
597
598        let finished = session.finish.is_some() && session.pending.is_empty();
599        let text = if finished {
600            // Nothing more will arrive: emit any trailing bytes, replacing a stray partial
601            // sequence rather than holding it forever.
602            let bytes = core::mem::take(&mut session.pending_bytes);
603            String::from_utf8_lossy(&bytes).into_owned()
604        } else {
605            take_complete_text(&mut session.pending_bytes)
606        };
607
608        Ok(Drained {
609            ids,
610            text,
611            done: finished,
612            finish: session.finish,
613        })
614    }

645    /// Mean-pooled, L2-normalized hidden state of `text` (Spec 24 §6 `embed`).
646    ///
647    /// Uses a scratch KV cache of its own, so it never disturbs a live session.
648    pub fn embed(&mut self, text: &str) -> Result<Vec<f32>, AiError> {
649        let mut tokens = self
650            .tokenizer
651            .encode_with_specials(text, self.tokenizer.add_bos(), false);
652        if tokens.is_empty() {
653            return Err(AiError::BadRequest(limit::Violation::EmbedInputOutOfRange));
654        }
655        if tokens.len() > self.cfg.n_ctx {
656            tokens.truncate(self.cfg.n_ctx);
657        }
658
659        let Engine {
660            cfg,
661            weights,
662            scratch,
663            ..
664        } = self;
665        let mut kv = KvCache::new(cfg);
666        let mut pooled = vec![0.0f32; cfg.n_embd];
667
668        for (position, token) in tokens.iter().enumerate() {
669            forward(cfg, weights, scratch, &mut kv, *token, position)
670                .map_err(|_| AiError::Internal)?;
671            tensor_math::add_in_place(&mut pooled, &scratch.x).map_err(|_| AiError::Internal)?;
672        }
673
674        let scale = 1.0 / tokens.len() as f32;
675        pooled.iter_mut().for_each(|value| *value *= scale);
676        let norm = libm::sqrtf(tensor_math::dot(&pooled, &pooled).map_err(|_| AiError::Internal)?);
677        if norm > 0.0 {
678            pooled.iter_mut().for_each(|value| *value /= norm);
679        }
680        Ok(pooled)
681    }
682
683    fn session(&self, request_id: u32) -> Option<&Session> {
684        self.sessions
685            .iter()
686            .flatten()
687            .find(|session| session.id == request_id)
688    }
689
690    fn find_slot(&self, request_id: u32) -> Option<usize> {
691        self.sessions
692            .iter()
693            .position(|slot| matches!(slot, Some(session) if session.id == request_id))
694    }
695
696    fn allocate_id(&mut self) -> u32 {
697        let id = self.next_request_id;
698        self.next_request_id = self.next_request_id.wrapping_add(1).max(1);
699        id
700    }
701}

703/// One forward pass for `token` at `position`, updating the KV cache.
704fn forward(
705    cfg: &ModelConfig,
706    weights: &Weights,
707    scratch: &mut Scratch,
708    kv: &mut KvCache,
709    token: u32,
710    position: usize,
711) -> Result<(), MathError> {
712    let token = token as usize;
713    if token >= cfg.vocab_size || position >= cfg.n_ctx {
714        return Err(MathError::ShapeMismatch);
715    }
716
717    row_to_f32(&weights.token_embd, token, &mut scratch.x[..cfg.n_embd])?;
718
719    for (layer_index, layer) in weights.layers.iter().enumerate() {
720        tensor_math::rms_norm(
721            &mut scratch.xb[..cfg.n_embd],
722            &scratch.x[..cfg.n_embd],
723            &layer.attn_norm,
724            cfg.rms_eps,
725        )?;
726
727        layer
728            .wq
729            .project(&mut scratch.q[..cfg.n_embd], &scratch.xb[..cfg.n_embd])?;
730        layer
731            .wk
732            .project(&mut scratch.k[..cfg.kv_dim()], &scratch.xb[..cfg.n_embd])?;
733        layer
734            .wv
735            .project(&mut scratch.v[..cfg.kv_dim()], &scratch.xb[..cfg.n_embd])?;
736
737        rope(&mut scratch.q[..cfg.n_embd], cfg, position)?;
738        rope(&mut scratch.k[..cfg.kv_dim()], cfg, position)?;
739
740        let slot = kv.slot(cfg, layer_index, position);
741        kv.k[slot.clone()].copy_from_slice(&scratch.k[..cfg.kv_dim()]);
742        kv.v[slot].copy_from_slice(&scratch.v[..cfg.kv_dim()]);
743
744        attention(cfg, kv, layer_index, scratch, position)?;
745        layer
746            .wo
747            .project(&mut scratch.down[..cfg.n_embd], &scratch.attn[..cfg.n_embd])?;
748        tensor_math::add_in_place(&mut scratch.x[..cfg.n_embd], &scratch.down[..cfg.n_embd])?;
749
750        tensor_math::rms_norm(
751            &mut scratch.xb[..cfg.n_embd],
752            &scratch.x[..cfg.n_embd],
753            &layer.ffn_norm,
754            cfg.rms_eps,
755        )?;
756        layer
757            .w_gate
758            .project(&mut scratch.gate[..cfg.n_ff], &scratch.xb[..cfg.n_embd])?;
759        layer
760            .w_up
761            .project(&mut scratch.up[..cfg.n_ff], &scratch.xb[..cfg.n_embd])?;
762        tensor_math::swiglu_in_place(&mut scratch.gate[..cfg.n_ff], &scratch.up[..cfg.n_ff])?;
763        layer
764            .w_down
765            .project(&mut scratch.down[..cfg.n_embd], &scratch.gate[..cfg.n_ff])?;
766        tensor_math::add_in_place(&mut scratch.x[..cfg.n_embd], &scratch.down[..cfg.n_embd])?;
767    }
768
769    kv.len = position + 1;
770    Ok(())
771}

773/// Dot-product attention for one layer at `position`, written into `scratch.attn`.
774fn attention(
775    cfg: &ModelConfig,
776    kv: &KvCache,
777    layer: usize,
778    scratch: &mut Scratch,
779    position: usize,
780) -> Result<(), MathError> {
781    let head_dim = cfg.head_dim();
782    let group = cfg.group_size();
783
784    for head in 0..cfg.n_head {
785        let kv_head = head / group;
786        let q = &scratch.q[head * head_dim..(head + 1) * head_dim];
787        for key_position in 0..=position {
788            let slot = kv.slot(cfg, layer, key_position);
789            let k = &kv.k[slot.start + kv_head * head_dim..slot.start + (kv_head + 1) * head_dim];
790            scratch.scores[key_position] = tensor_math::dot(q, k)?;
791        }
792        tensor_math::softmax_in_place(&mut scratch.scores[..=position]);
793
794        let out = &mut scratch.attn[head * head_dim..(head + 1) * head_dim];
795        out.iter_mut().for_each(|value| *value = 0.0);
796        for key_position in 0..=position {
797            let slot = kv.slot(cfg, layer, key_position);
798            let v = &kv.v[slot.start + kv_head * head_dim..slot.start + (kv_head + 1) * head_dim];
799            tensor_math::scaled_add_in_place(out, v, scratch.scores[key_position])?;
800        }
801    }
802    Ok(())
803}

805/// Sample the next token from the logits of the current hidden state.
806fn sample(
807    cfg: &ModelConfig,
808    weights: &Weights,
809    scratch: &mut Scratch,
810    session: &mut Session,
811) -> Result<u32, AiError> {
812    tensor_math::rms_norm(
813        &mut scratch.xb[..cfg.n_embd],
814        &scratch.x[..cfg.n_embd],
815        &weights.output_norm,
816        cfg.rms_eps,
817    )
818    .map_err(|_| AiError::Internal)?;
819
820    match &weights.output {
821        Some(output) => output
822            .project(
823                &mut scratch.logits[..cfg.vocab_size],
824                &scratch.xb[..cfg.n_embd],
825            )
826            .map_err(|_| AiError::Internal)?,
827        // Tied embedding: `logits = token_embd · xb` — the same projection every other matrix
828        // uses, over the packed Q8_0 weights the file already stores.
829        None => weights
830            .token_embd
831            .project(
832                &mut scratch.logits[..cfg.vocab_size],
833                &scratch.xb[..cfg.n_embd],
834            )
835            .map_err(|_| AiError::Internal)?,
836    }
837
838    let temperature = f32::from(session.params.temperature_milli) / 1000.0;
839    let index = tensor_math::sample_top_k(
840        &mut scratch.logits[..cfg.vocab_size],
841        usize::from(session.params.top_k),
842        temperature,
843        &mut session.rng,
844    )
845    .ok_or(AiError::Internal)?;
846    Ok(index as u32)
847}

849/// Apply RoPE to each head of a q/k vector.
850fn rope(vector: &mut [f32], cfg: &ModelConfig, position: usize) -> Result<(), MathError> {
851    let head_dim = cfg.head_dim();
852    for head in vector.chunks_exact_mut(head_dim) {
853        tensor_math::rope_normal(head, position, cfg.rope_freq_base)?;
854    }
855    Ok(())
856}

858/// Copy one matrix row into f32, dequantizing when needed.
859fn row_to_f32(matrix: &Matrix, row: usize, out: &mut [f32]) -> Result<(), MathError> {
860    match matrix {
861        Matrix::F32 { rows, data } => {
862            if row >= *rows || out.is_empty() {
863                return Err(MathError::ShapeMismatch);
864            }
865            let cols = data.len() / rows;
866            if out.len() < cols {
867                return Err(MathError::ShapeMismatch);
868            }
869            out[..cols].copy_from_slice(&data[row * cols..(row + 1) * cols]);
870            Ok(())
871        }
872        Matrix::Q8_0 { rows, cols, data } => {
873            if row >= *rows || out.len() < *cols {
874                return Err(MathError::ShapeMismatch);
875            }
876            let row_bytes = tensor_math::q8_0_row_bytes(*cols).ok_or(MathError::NotDivisible)?;
877            let start = row.checked_mul(row_bytes).ok_or(MathError::ShapeMismatch)?;
878            let slice = data
879                .get(start..start + row_bytes)
880                .ok_or(MathError::ShapeMismatch)?;
881            for (block, out_block) in slice
882                .chunks_exact(quant::Q8_0_BLOCK_BYTES)
883                .zip(out.chunks_exact_mut(quant::Q8_0_BLOCK_WEIGHTS))
884            {
885                let mut decoded = [0f32; quant::Q8_0_BLOCK_WEIGHTS];
886                quant::q8_0_block_to_f32(block, &mut decoded)?;
887                out_block.copy_from_slice(&decoded);
888            }
889            Ok(())
890        }
891    }
892}

894/// Split the longest complete-UTF-8 prefix out of `buffer`, leaving the partial tail behind.
895fn take_complete_text(buffer: &mut Vec<u8>) -> String {
896    let split = complete_utf8_prefix(buffer);
897    let tail = buffer.split_off(split);
898    let head = core::mem::replace(buffer, tail);
899    match String::from_utf8(head) {
900        Ok(text) => text,
901        // The prefix ended on a boundary by construction, so this is unreachable for well-formed
902        // input; a lossy conversion keeps the Cell alive on a corrupt token table.
903        Err(error) => String::from_utf8_lossy(error.as_bytes()).into_owned(),
904    }
905}
906
907/// Length of the longest prefix of `bytes` that ends on a UTF-8 character boundary.
908fn complete_utf8_prefix(bytes: &[u8]) -> usize {
909    let mut index = 0;
910    while index < bytes.len() {
911        let byte = bytes[index];
912        let width = if byte < 0x80 {
913            1
914        } else if byte >> 5 == 0b110 {
915            2
916        } else if byte >> 4 == 0b1110 {
917            3
918        } else if byte >> 3 == 0b11110 {
919            4
920        } else {
921            // Invalid leading byte: consume it so the caller can emit a replacement character.
922            1
923        };
924        if index + width > bytes.len() {
925            break;
926        }
927        index += width;
928    }
929    index
930}

932/// Read `llama.*` metadata into [`ModelConfig`].
933fn read_config(file: &GgufFile<'_>, vocab_size: usize) -> Result<ModelConfig, EngineError> {
934    let arch = file
935        .metadata_str("general.architecture")
936        .ok_or(EngineError::MissingMetadata("general.architecture"))?;
937    if arch != "llama" {
938        return Err(EngineError::UnsupportedArchitecture);
939    }
940
941    let n_embd = meta_usize(file, "llama.embedding_length")?;
942    let n_head = meta_usize(file, "llama.attention.head_count")?;
943    let n_head_kv = file
944        .metadata_u32("llama.attention.head_count_kv")
945        .map(|value| value as usize)
946        .unwrap_or(n_head);
947    let n_layer = meta_usize(file, "llama.block_count")?;
948    let n_ff = meta_usize(file, "llama.feed_forward_length")?;
949    let n_ctx = meta_usize(file, "llama.context_length")?;
950
951    if vocab_size == 0 || vocab_size > MAX_VOCAB {
952        return Err(EngineError::InvalidMetadata("tokenizer.ggml.tokens"));
953    }
954    if n_embd == 0 || n_head == 0 || n_embd % n_head != 0 {
955        return Err(EngineError::InvalidMetadata("llama.embedding_length"));
956    }
957    if n_head_kv == 0 || n_head % n_head_kv != 0 {
958        return Err(EngineError::InvalidMetadata(
959            "llama.attention.head_count_kv",
960        ));
961    }
962    if n_layer == 0 || n_ff == 0 {
963        return Err(EngineError::InvalidMetadata("llama.block_count"));
964    }
965    if n_ctx == 0 || n_ctx > MAX_CONTEXT_TOKENS {
966        return Err(EngineError::InvalidMetadata("llama.context_length"));
967    }
968    if !(n_embd / n_head).is_multiple_of(2) {
969        return Err(EngineError::InvalidMetadata("llama.attention.head_count"));
970    }
971
972    Ok(ModelConfig {
973        name: file
974            .metadata_str("general.name")
975            .unwrap_or("unnamed")
976            .to_owned(),
977        n_ctx,
978        n_embd,
979        n_layer,
980        n_head,
981        n_head_kv,
982        n_ff,
983        rms_eps: file
984            .metadata_f32("llama.attention.layer_norm_rms_epsilon")
985            .unwrap_or(1e-5),
986        rope_freq_base: file
987            .metadata_f32("llama.rope.freq_base")
988            .unwrap_or(10_000.0),
989        vocab_size,
990    })
991}

1000/// Load every weight tensor the Llama forward pass needs.
1001fn read_weights(file: &GgufFile<'_>, cfg: &ModelConfig) -> Result<Weights, EngineError> {
1002    let token_embd = read_matrix(file, "token_embd.weight", cfg.vocab_size, cfg.n_embd)?;
1003    let output_norm = read_vector(file, "output_norm.weight", cfg.n_embd)?;
1004    // Tied-embedding checkpoints omit `output.weight`.
1005    let output = if file.tensor("output.weight").is_some() {
1006        Some(read_matrix(
1007            file,
1008            "output.weight",
1009            cfg.vocab_size,
1010            cfg.n_embd,
1011        )?)
1012    } else {
1013        None
1014    };
1015
1016    let mut layers = Vec::with_capacity(cfg.n_layer);
1017    for index in 0..cfg.n_layer {
1018        let name = |suffix: &str| format!("blk.{index}.{suffix}");
1019        layers.push(Layer {
1020            attn_norm: read_vector(file, &name("attn_norm.weight"), cfg.n_embd)?,
1021            wq: read_matrix(file, &name("attn_q.weight"), cfg.n_embd, cfg.n_embd)?,
1022            wk: read_matrix(file, &name("attn_k.weight"), cfg.kv_dim(), cfg.n_embd)?,
1023            wv: read_matrix(file, &name("attn_v.weight"), cfg.kv_dim(), cfg.n_embd)?,
1024            wo: read_matrix(file, &name("attn_output.weight"), cfg.n_embd, cfg.n_embd)?,
1025            ffn_norm: read_vector(file, &name("ffn_norm.weight"), cfg.n_embd)?,
1026            w_gate: read_matrix(file, &name("ffn_gate.weight"), cfg.n_ff, cfg.n_embd)?,
1027            w_up: read_matrix(file, &name("ffn_up.weight"), cfg.n_ff, cfg.n_embd)?,
1028            w_down: read_matrix(file, &name("ffn_down.weight"), cfg.n_embd, cfg.n_ff)?,
1029        });
1030    }
1031
1032    Ok(Weights {
1033        token_embd,
1034        output_norm,
1035        output,
1036        layers,
1037    })
1038}

1040/// Load one 1-D tensor (norm weights) as f32.
1041fn read_vector(file: &GgufFile<'_>, name: &str, len: usize) -> Result<Vec<f32>, EngineError> {
1042    let info = file
1043        .tensor(name)
1044        .ok_or_else(|| EngineError::MissingTensor(name.to_owned()))?;
1045    if info.dims.len() != 1 || info.dims[0] as usize != len {
1046        return Err(EngineError::BadTensorShape(name.to_owned()));
1047    }
1048    let data = file.tensor_data(info).map_err(EngineError::Gguf)?;
1049    let mut values = vec![0.0f32; len];
1050    file.dequant_row(info.dtype, data, len, &mut values)
1051        .map_err(EngineError::Gguf)?;
1052    Ok(values)
1053}

1055/// Load one 2-D tensor, keeping Q8_0 in its packed form.
1056fn read_matrix(
1057    file: &GgufFile<'_>,
1058    name: &str,
1059    rows: usize,
1060    cols: usize,
1061) -> Result<Matrix, EngineError> {
1062    let info = file
1063        .tensor(name)
1064        .ok_or_else(|| EngineError::MissingTensor(name.to_owned()))?;
1065    if info.dims.len() != 2 || info.dims[0] as usize != cols || info.dims[1] as usize != rows {
1066        return Err(EngineError::BadTensorShape(name.to_owned()));
1067    }
1068    let data = file.tensor_data(info).map_err(EngineError::Gguf)?;
1069    match info.dtype {
1070        GgmlDType::Q8_0 => Ok(Matrix::Q8_0 {
1071            rows,
1072            cols,
1073            data: data.to_vec(),
1074        }),
1075        GgmlDType::F32 | GgmlDType::F16 => {
1076            let mut values = vec![0.0f32; rows * cols];
1077            file.dequant_row(info.dtype, data, rows * cols, &mut values)
1078                .map_err(EngineError::Gguf)?;
1079            Ok(Matrix::F32 { rows, data: values })
1080        }
1081        GgmlDType::Unsupported(dtype) => Err(EngineError::UnsupportedDType(dtype)),
1082    }
1083}

1351    /// Real-weight validation (Spec 24 §4.1, plan Phase 04).
1352    ///
1353    /// Skips itself unless `CELLOS_AI_MODEL` points at a GGUF checkpoint, so the host suite stays
1354    /// hermetic; `scripts/fetch-ai-test-model.sh` fetches the pinned SmolLM-135M-Instruct Q8_0
1355    /// checkpoint this test is written for.
1356    ///
1357    /// What it establishes that the fixture cannot: the engine loads a production-sized file
1358    /// (49,152-token vocabulary, 30 layers, tied output embedding, 138 MB of Q8_0 weights) and
1359    /// generates text from real trained weights rather than from a numerical oracle.
1360    #[test]
1361    fn generates_text_from_a_real_checkpoint() {
1362        let path = match std::env::var("CELLOS_AI_MODEL") {
1363            Ok(path) => path,
1364            Err(_) => {
1365                eprintln!(
1366                    "SKIP generates_text_from_a_real_checkpoint: set CELLOS_AI_MODEL \
1367                     (scripts/fetch-ai-test-model.sh prints the path)"
1368                );
1369                return;
1370            }
1371        };
1372        let bytes = match std::fs::read(&path) {
1373            Ok(bytes) => bytes,
1374            Err(error) => {
1375                eprintln!("SKIP generates_text_from_a_real_checkpoint: {path}: {error}");
1376                return;
1377            }
1378        };
1379
1380        let mut engine = Engine::load(&bytes, 1 << 30).expect("checkpoint loads");
1381        assert!(
1382            engine.config().vocab_size > 40_000,
1383            "expected a real tokenizer"
1384        );
1385        assert!(engine.config().n_layer >= 8, "expected a real transformer");
1386
1387        let prompt = "The capital of France is";
1388        let params = SamplingParams {
1389            max_tokens: 16,
1390            temperature_milli: 0,
1391            top_k: 0,
1392            seed: 1,
1393        };
1394        let request_id = engine.submit(prompt, params).expect("submit");
1395        let started = std::time::Instant::now();
1396        let mut ids = Vec::new();
1397        let mut text = String::new();
1398        let mut finish = None;
1399        for _ in 0..64 {
1400            let progress = engine.generate(request_id, 8).expect("generate");
1401            let drained = engine.drain(request_id, 16).expect("drain");
1402            ids.extend(drained.ids.iter().copied());
1403            text.push_str(&drained.text);
1404            if drained.done {
1405                finish = progress.finish.or(drained.finish);
1406                break;
1407            }
1408        }
1409        let elapsed = started.elapsed();
1410
1411        assert_eq!(ids.len(), 16, "requested 16 tokens: {ids:?}");
1412        assert!(
1413            finish.is_some(),
1414            "a length-capped generation must report a finish reason"
1415        );
1416        let distinct = {
1417            let mut sorted = ids.clone();
1418            sorted.sort_unstable();
1419            sorted.dedup();
1420            sorted.len()
1421        };
1422        assert!(distinct >= 4, "degenerate generation: {ids:?}");
1423        let printable: usize = text
1424            .chars()
1425            .filter(|c| c.is_ascii_alphanumeric() || *c == ' ')
1426            .count();
1427        assert!(
1428            printable * 2 >= text.chars().count(),
1429            "generated text is not text-like: {text:?}"
1430        );
1431
1432        let seconds = elapsed.as_secs_f64();
1433        std::println!(
1434            "[ai-engine] real checkpoint: model={} layers={} vocab={} resident={} MiB \
1435             tokens={} tps={:.2} text={:?}",
1436            engine.config().name,
1437            engine.config().n_layer,
1438            engine.config().vocab_size,
1439            engine.resident_bytes() / (1024 * 1024),
1440            ids.len(),
1441            ids.len() as f64 / seconds.max(f64::MIN_POSITIVE),
1442            text,
1443        );
1444    }
