1//! Unified AI inference service Cell (`/bin/ai`) — Spec 24 §2.
2//!
3//! One event loop serves every caller:
4//!
5//! * `Describe` answers from the resident engine.
6//! * `InferSubmit` seats a session in a bounded table (owner = the sender tid that asked for it).
7//! * `InferStreamPoll` advances *its own* session by a bounded number of model steps and returns
8//!   the tokens produced since the previous poll, so one request can never monopolise the loop and
9//!   several callers make progress concurrently.
10//! * `InferCancel` releases a session; a session abandoned by a dead client is reclaimed after
11//!   [`SESSION_IDLE_TICKS`] of inactivity, because a service without lifecycle authority cannot
12//!   watch another Cell's exit.
13//!
14//! Truthfulness rules:
15//!
16//! * No model, an unsupported architecture, or an over-budget model is reported through
17//!   [`AiError`] — the service never answers with fabricated tokens.
18//! * A session belongs to the Cell that submitted it: another Cell polling or cancelling that id
19//!   gets `UnknownRequest`, the same answer it would get for an id that never existed.
20//! * Non-CPU device targets are refused with `NotSupported` rather than silently downgraded.
21

55// Cell heap. Holds the model bytes, the engine's weights/scratch, and the session KV caches.
56//
57// The engine addresses weights in place inside the model buffer, so the arena needs the file plus
58// the per-session and scratch buffers — not a second copy of the file.
59//
60// Default sizing serves the small checkpoints (llama2.c `stories260K`, the deterministic fixture)
61// with room for four sessions. `large-arena` sizes it for a 25 MB checkpoint, which the Cell's
62// 32 MiB virtual-address slot (`kernel/src/loader/va_alloc.rs`) holds but the default arena does
63// not; a deployment that serves a model that size builds the cell with that feature instead of
64// making every image pay 29 MB of resident RAM.
65#[cfg(feature = "large-arena")]
66ostd::declare_custom_heap!(29 * 1024 * 1024);
67#[cfg(not(feature = "large-arena"))]
68ostd::declare_custom_heap!(16 * 1024 * 1024);
69
70/// Where the service looks for its checkpoint, in order.
71///
72/// `/bin` is the P6 FAT cell-store `gen_disk.ps1` deploys into (and on a netboot image it is the
73/// VIFS1 ramdisk). `/mnt/sd` is the board's own FAT partition: keeping the checkpoint there keeps
74/// the *image* small, which matters where the kernel's memory layout is what breaks first — a
75/// 43.6 MB payload (this checkpoint embedded in VIFS1) panics the Raspberry Pi 3 kernel while the
76/// same cells with the checkpoint on the card boot and serve it. The resident cost inside the Cell
77/// is identical either way, so a deployment should prefer the card.
78const MODEL_PATHS: [&str; 2] = ["/bin/ai-model.gguf", "/mnt/sd/ai-model.gguf"];
79
80/// Ceiling handed to the engine: the arena above minus room for the loader's bookkeeping and the
81/// session table.
82#[cfg(feature = "large-arena")]
83const ENGINE_LIMIT: usize = 29 * 1024 * 1024;
84#[cfg(not(feature = "large-arena"))]
85const ENGINE_LIMIT: usize = 14 * 1024 * 1024;
86
87/// Model steps advanced per poll. Bounds one session's share of the event loop; a longer
88/// generation simply takes more polls.
89const STEPS_PER_POLL: usize = 4;
90
91/// Sessions idle for this many 10 ms ticks are reclaimed.
92///
93/// A service Cell cannot watch another Cell's exit (`NotifyOnExit` needs lifecycle authority), so
94/// an abandoned session is released by deadline instead: 30 s of silence is far longer than any
95/// poll interval a live client produces.
96const SESSION_IDLE_TICKS: u64 = 3_000;
97
98/// One live session and the Cell that owns it.
99struct Session {
100    request_id: u32,
101    owner: usize,
102    last_activity: u64,
103}
104
105/// The service state.
106struct AiService {
107    engine: Option<Engine>,
108    sessions: [Option<Session>; MAX_SESSIONS as usize],
109    /// Why inference is unavailable when `engine` is `None`, so refusals stay specific.
110    unavailable: AiError,
111}

113ostd::cell_main!(cell_main);

115fn cell_main() {
116    init_custom_heap();
117    let mut service = AiService::load();
118
119    let mut buf = [0u8; ai_proto::AI_IPC_BUF_SIZE];
120    let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
121
122    loop {
123        match ostd::syscall::sys_recv(0, &mut buf) {
124            SyscallResult::Ok(sender) if sender > 0 => {
125                service.reap_idle_sessions();
126                if let Some(encoded) = service.handle(sender, &buf, &mut reply) {
127                    let _ = ostd::syscall::sys_send(sender, encoded);
128                }
129            }
130            _ => ostd::task::yield_now(),
131        }
132    }
133}

135impl AiService {
136    /// Load the model (if present) and build the service.
137    fn load() -> Self {
138        println("[ai] inference service starting");
139        let mut service = Self {
140            engine: None,
141            sessions: [const { None }; MAX_SESSIONS as usize],
142            unavailable: AiError::NoModel,
143        };
144
145        // Read timing is part of the service's contract with its operators: a model that is slow to
146        // read on a given board is an operational fact, and without the number the only symptom is
147        // a silent stall on the serial console.
148        let read_started = ostd::syscall::sys_get_time_ms().unwrap_or(0);
149        let mut model_path = MODEL_PATHS[0];
150        let mut found = None;
151        for candidate in MODEL_PATHS {
152            if let Some(bytes) = read_model(candidate, ENGINE_LIMIT) {
153                model_path = candidate;
154                found = Some(bytes);
155                break;
156            }
157        }
158        let bytes = match found {
159            Some(bytes) => bytes,
160            None => {
161                print("[ai] no model at any of ");
162                print_usize(MODEL_PATHS.len());
163                println(" candidate paths — inference will be refused");
164                return service;
165            }
166        };
167        print("[ai] model path: ");
168        println(model_path);
169        let read_ms = ostd::syscall::sys_get_time_ms()
170            .unwrap_or(0)
171            .saturating_sub(read_started);
172
173        print("[ai] model bytes: ");
174        print_usize(bytes.len());
175        print(" read in ");
176        print_usize(read_ms as usize);
177        println(" ms");
178        let load_started = ostd::syscall::sys_get_time_ms().unwrap_or(0);
179        // The engine takes ownership of the model bytes and addresses the weights in place, so this
180        // is the only copy of the file in the cell.
181        match Engine::load(bytes, ENGINE_LIMIT) {
182            Ok(engine) => {
183                let load_ms = ostd::syscall::sys_get_time_ms()
184                    .unwrap_or(0)
185                    .saturating_sub(load_started);
186                print("[ai] engine load: ");
187                print_usize(load_ms as usize);
188                println(" ms");
189                let describe = engine.describe();
190                print("[ai] model ready: ");
191                print_usize(describe.vocab_size as usize);
192                print(" vocab, context ");
193                print_usize(describe.context_tokens as usize);
194                print(", resident bytes ");
195                print_usize(describe.resident_bytes as usize);
196                println("");
197                service.engine = Some(engine);
198            }
199            Err(error) => {
200                service.unavailable = match error {
201                    EngineError::UnsupportedArchitecture
202                    | EngineError::UnsupportedDType(_)
203                    | EngineError::TooLarge { .. } => AiError::NotSupported,
204                    _ => AiError::NoModel,
205                };
206                println("[ai] model rejected — inference will be refused");
207            }
208        }
209        service
210    }

212    /// Release sessions whose owner has gone quiet.
213    ///
214    /// Dead client Cells cannot be watched without lifecycle authority, so silence is the signal:
215    /// a live client polls or cancels well inside [`SESSION_IDLE_TICKS`].
216    fn reap_idle_sessions(&mut self) {
217        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
218        let mut expired = [0u32; MAX_SESSIONS as usize];
219        let mut expired_count = 0usize;
220
221        for slot in self.sessions.iter_mut() {
222            let Some(session) = slot.as_ref() else {
223                continue;
224            };
225            if now.wrapping_sub(session.last_activity) <= SESSION_IDLE_TICKS {
226                continue;
227            }
228            expired[expired_count] = session.request_id;
229            expired_count += 1;
230            *slot = None;
231        }
232
233        if let Some(engine) = self.engine.as_mut() {
234            for request_id in expired.iter().take(expired_count) {
235                let _ = engine.cancel(*request_id);
236            }
237        }
238    }

240    /// Answer one request. `None` when the request must be ignored (nothing can be replied to it).
241    fn handle<'r>(&mut self, sender: usize, buf: &[u8], reply: &'r mut [u8]) -> Option<&'r [u8]> {
242        let request = match ai_proto::decode::<AiRequest<'_>>(buf) {
243            Ok(request) => request,
244            Err(_) => {
245                let error = AiError::BadRequest(limit::Violation::MessageTooLarge);
246                log_refusal("decode", error);
247                return Some(encode_failure(reply, 0, error));
248            }
249        };
250        if let Err(error) = ai_proto::validate_request(&request) {
251            log_refusal("validate", error);
252            return Some(encode_failure(reply, 0, error));
253        }
254
255        match request {
256            AiRequest::Describe => match self.engine.as_ref() {
257                Some(engine) => {
258                    ai_proto::encode(&AiResponse::Description(engine.describe()), reply)
259                        .ok()
260                        .map(|encoded| &*encoded)
261                }
262                None => Some(encode_failure(reply, 0, self.unavailable)),
263            },
264            AiRequest::InferSubmit(submit) => {
265                if !matches!(submit.device, DeviceTarget::Auto | DeviceTarget::Cpu) {
266                    return Some(encode_failure(reply, 0, AiError::NotSupported));
267                }
268                let unavailable = self.unavailable;
269                let Some(engine) = self.engine.as_mut() else {
270                    return Some(encode_failure(reply, 0, unavailable));
271                };
272                let params = SamplingParams {
273                    max_tokens: submit.max_tokens,
274                    temperature_milli: submit.temperature_milli,
275                    top_k: submit.top_k,
276                    seed: submit.seed,
277                };
278                match engine.submit(submit.prompt, params) {
279                    Ok(request_id) => {
280                        self.seat(sender, request_id);
281                        ai_proto::encode(
282                            &AiResponse::Accepted {
283                                request_id,
284                                backend: backend::CPU,
285                            },
286                            reply,
287                        )
288                        .ok()
289                        .map(|encoded| &*encoded)
290                    }
291                    Err(error) => {
292                        log_refusal("submit", error);
293                        Some(encode_failure(reply, 0, error))
294                    }
295                }
296            }
297            AiRequest::InferStreamPoll {
298                request_id,
299                max_tokens,
300            } => self.poll(sender, request_id, max_tokens, reply),
301            AiRequest::InferCancel { request_id } => {
302                if !self.owned_by(sender, request_id) {
303                    return Some(encode_failure(reply, request_id, AiError::UnknownRequest));
304                }
305                if let Some(engine) = self.engine.as_mut() {
306                    let _ = engine.cancel(request_id);
307                }
308                self.release(request_id);
309                ai_proto::encode(
310                    &AiResponse::TokenChunk {
311                        request_id,
312                        tokens: &[],
313                        text: "",
314                        done: true,
315                        finish: FinishReason::Cancelled,
316                    },
317                    reply,
318                )
319                .ok()
320                .map(|encoded| &*encoded)
321            }
322            AiRequest::InferEmbed { text, device } => {
323                if !matches!(device, DeviceTarget::Auto | DeviceTarget::Cpu) {
324                    return Some(encode_failure(reply, 0, AiError::NotSupported));
325                }
326                let unavailable = self.unavailable;
327                let Some(engine) = self.engine.as_mut() else {
328                    return Some(encode_failure(reply, 0, unavailable));
329                };
330                match engine.embed(text) {
331                    Ok(values) => {
332                        let mut value_bytes =
333                            [0u8; MAX_EMBED_DIM as usize * ai_proto::EMBED_VALUE_BYTES];
334                        let encoded_values =
335                            ai_proto::encode_embedding_values(&values, &mut value_bytes)?;
336                        ai_proto::encode(
337                            &AiResponse::Embedding {
338                                dim: values.len() as u16,
339                                values: encoded_values,
340                            },
341                            reply,
342                        )
343                        .ok()
344                        .map(|encoded| &*encoded)
345                    }
346                    Err(error) => {
347                        log_refusal("embed", error);
348                        Some(encode_failure(reply, 0, error))
349                    }
350                }
351            }
352        }
353    }

355    /// Advance a session and report what it produced.
356    fn poll<'r>(
357        &mut self,
358        sender: usize,
359        request_id: u32,
360        max_tokens: u8,
361        reply: &'r mut [u8],
362    ) -> Option<&'r [u8]> {
363        if !self.owned_by(sender, request_id) {
364            return Some(encode_failure(reply, request_id, AiError::UnknownRequest));
365        }
366        let unavailable = self.unavailable;
367        let Some(engine) = self.engine.as_mut() else {
368            return Some(encode_failure(reply, request_id, unavailable));
369        };
370
371        if let Err(error) = engine.generate(request_id, STEPS_PER_POLL) {
372            if error == AiError::UnknownRequest {
373                self.release(request_id);
374            }
375            log_refusal("generate", error);
376            return Some(encode_failure(reply, request_id, error));
377        }
378        let drained = match engine.drain(request_id, max_tokens as usize) {
379            Ok(drained) => drained,
380            Err(error) => {
381                log_refusal("drain", error);
382                return Some(encode_failure(reply, request_id, error));
383            }
384        };
385
386        let mut token_bytes = [0u8; MAX_TOKENS_PER_POLL as usize * ai_proto::TOKEN_ID_BYTES];
387        let encoded_tokens = ai_proto::encode_token_ids(&drained.ids, &mut token_bytes)?;
388        let finish = drained.finish.unwrap_or(FinishReason::Length);
389        let encoded = ai_proto::encode(
390            &AiResponse::TokenChunk {
391                request_id,
392                tokens: encoded_tokens,
393                text: drained.text.as_str(),
394                done: drained.done,
395                finish,
396            },
397            reply,
398        )
399        .ok()
400        .map(|encoded| &*encoded);
401
402        if drained.done {
403            if let Some(engine) = self.engine.as_mut() {
404                let _ = engine.release(request_id);
405            }
406            self.release(request_id);
407        } else {
408            self.touch(request_id);
409        }
410        encoded
411    }

413    /// Take a session slot for `owner`.
414    fn seat(&mut self, owner: usize, request_id: u32) {
415        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
416        if let Some(slot) = self.sessions.iter_mut().find(|slot| slot.is_none()) {
417            *slot = Some(Session {
418                request_id,
419                owner,
420                last_activity: now,
421            });
422        }
423    }
424
425    /// Whether `request_id` is seated and belongs to `owner`.
426    fn owned_by(&self, owner: usize, request_id: u32) -> bool {
427        self.sessions
428            .iter()
429            .flatten()
430            .any(|session| session.request_id == request_id && session.owner == owner)
431    }
432
433    /// Drop a session slot.
434    fn release(&mut self, request_id: u32) {
435        for slot in self.sessions.iter_mut() {
436            if matches!(slot, Some(session) if session.request_id == request_id) {
437                *slot = None;
438            }
439        }
440    }
441
442    /// Mark a session as active so idle reaping leaves it alone.
443    fn touch(&mut self, request_id: u32) {
444        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
445        for session in self.sessions.iter_mut().flatten() {
446            if session.request_id == request_id {
447                session.last_activity = now;
448            }
449        }
450    }
451}

453/// Bytes moved per capability read. `ReadCap` accepts up to the kernel's 64 MiB user-buffer ceiling,
454/// so this trades syscall count against a stack/heap buffer: 256 KiB reads a 25 MB model in ~100
455/// syscalls instead of ~6,700 IPC round trips.
456const MODEL_READ_CHUNK: usize = 256 * 1024;
457
458/// Read the model file, preferring the kernel capability path.
459///
460/// Two paths exist and they cost very different things:
461///
462/// * **Kernel cap path** (`OpenCap`/`SeekCap`/`ReadCap`): a handful of syscalls for the whole file.
463///   It resolves the kernel's VIFS1 ramdisk — where a development image puts the model — and needs
464///   the uppercase path FAT16 uses.
465/// * **VFS service**: the only path for a model deployed in the on-disk cell-store. It moves one
466///   4 KiB IPC message per round trip, so a 25 MB model costs thousands of round trips (measured:
467///   146 s in QEMU TCG against a few seconds for the cap path). Kept as the fallback because a
468///   cell-store deployment is the shape a released image has.
469///
470/// Returns `None` when the file does not exist or exceeds `limit`.
471fn read_model(path: &str, limit: usize) -> Option<Vec<u8>> {
472    let upper: alloc::string::String = path.chars().map(|c| c.to_ascii_uppercase()).collect();
473    if let Some(bytes) = read_model_cap(&upper, limit) {
474        return Some(bytes);
475    }
476    let mut vfs = ostd::clients::VfsClient::new();
477    vfs.read_file_bounded(path, limit).ok()
478}
479
480/// One whole-file read through the kernel capability syscalls.
481fn read_model_cap(path: &str, limit: usize) -> Option<Vec<u8>> {
482    let mut file = ostd::fs::File::open(path).ok()?;
483    let size = usize::try_from(file.size().ok()?).ok()?;
484    if size == 0 || size > limit {
485        return None;
486    }
487    let mut bytes = Vec::with_capacity(size);
488    let mut chunk = alloc::vec![0u8; MODEL_READ_CHUNK.min(size)];
489    while bytes.len() < size {
490        let want = (size - bytes.len()).min(chunk.len());
491        match file.read_at(bytes.len() as u64, &mut chunk[..want]) {
492            Ok(0) => break,
493            Ok(n) => bytes.extend_from_slice(&chunk[..n]),
494            Err(_) => return None,
495        }
496    }
497    let _ = file.close();
498    if bytes.len() == size {
499        Some(bytes)
500    } else {
501        None
502    }
503}

505/// Log a refusal with its typed cause, so a failing deployment is diagnosable from the
506/// serial console instead of only from the caller's side.
507fn log_refusal(stage: &str, error: AiError) {
508    ostd::io::print("[ai] refused ");
509    ostd::io::println(stage);
510    ostd::io::print_fmt(format_args!("[ai] cause: {error:?}\n"));
511}
512
513/// Encode a typed refusal.
514fn encode_failure(reply: &mut [u8], request_id: u32, error: AiError) -> &[u8] {
515    ai_proto::encode(&AiResponse::Failed { request_id, error }, reply)
516        .map(|encoded| &*encoded)
517        .unwrap_or(&[])
