1//! Unified AI inference service Cell (`/bin/ai`) — Spec 24 §2.
2//!
3//! One event loop serves every caller:
4//!
5//! * `Describe` answers from the resident engine.
6//! * `InferSubmit` seats a session in a bounded table (owner = the sender tid that asked for it).
7//! * `InferStreamPoll` advances *its own* session by a bounded number of model steps and returns
8//!   the tokens produced since the previous poll, so one request can never monopolise the loop and
9//!   several callers make progress concurrently.
10//! * `InferCancel` releases a session; a session abandoned by a dead client is reclaimed after
11//!   [`SESSION_IDLE_TICKS`] of inactivity, because a service without lifecycle authority cannot
12//!   watch another Cell's exit.
13//!
14//! Truthfulness rules:
15//!
16//! * No model, an unsupported architecture, or an over-budget model is reported through
17//!   [`AiError`] — the service never answers with fabricated tokens.
18//! * A session belongs to the Cell that submitted it: another Cell polling or cancelling that id
19//!   gets `UnknownRequest`, the same answer it would get for an id that never existed.
20//! * Non-CPU device targets are refused with `NotSupported` rather than silently downgraded.
21

40// Cell heap. Holds the model bytes, the engine's weights/scratch, and the session KV caches.
41ostd::declare_custom_heap!(8 * 1024 * 1024);
42
43/// Model path. The P6 FAT cell-store is mounted at `/bin`, so this is the FAT root entry
44/// `ai-model.gguf` (deployed by `gen_disk.ps1`).
45const MODEL_PATH: &str = "/bin/ai-model.gguf";
46
47/// Ceiling handed to the engine. Must stay below the arena above, leaving room for the read
48/// buffer and the session table.
49const ENGINE_LIMIT: usize = 6 * 1024 * 1024;
50
51/// Model steps advanced per poll. Bounds one session's share of the event loop; a longer
52/// generation simply takes more polls.
53const STEPS_PER_POLL: usize = 4;
54
55/// Sessions idle for this many 10 ms ticks are reclaimed.
56///
57/// A service Cell cannot watch another Cell's exit (`NotifyOnExit` needs lifecycle authority), so
58/// an abandoned session is released by deadline instead: 30 s of silence is far longer than any
59/// poll interval a live client produces.
60const SESSION_IDLE_TICKS: u64 = 3_000;
61
62/// One live session and the Cell that owns it.
63struct Session {
64    request_id: u32,
65    owner: usize,
66    last_activity: u64,
67}
68
69/// The service state.
70struct AiService {
71    engine: Option<Engine>,
72    sessions: [Option<Session>; MAX_SESSIONS as usize],
73    /// Why inference is unavailable when `engine` is `None`, so refusals stay specific.
74    unavailable: AiError,
75}

77ostd::cell_main!(cell_main);

79fn cell_main() {
80    init_custom_heap();
81    let mut service = AiService::load();
82
83    let mut buf = [0u8; ai_proto::AI_IPC_BUF_SIZE];
84    let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
85
86    loop {
87        match ostd::syscall::sys_recv(0, &mut buf) {
88            SyscallResult::Ok(sender) if sender > 0 => {
89                service.reap_idle_sessions();
90                if let Some(encoded) = service.handle(sender, &buf, &mut reply) {
91                    let _ = ostd::syscall::sys_send(sender, encoded);
92                }
93            }
94            _ => ostd::task::yield_now(),
95        }
96    }
97}

99impl AiService {
100    /// Load the model (if present) and build the service.
101    fn load() -> Self {
102        println("[ai] inference service starting");
103        let mut service = Self {
104            engine: None,
105            sessions: [const { None }; MAX_SESSIONS as usize],
106            unavailable: AiError::NoModel,
107        };
108
109        let mut vfs = ostd::clients::VfsClient::new();
110        let bytes = match vfs.read_file_bounded(MODEL_PATH, ENGINE_LIMIT) {
111            Ok(bytes) => bytes,
112            Err(_) => {
113                print("[ai] no model at ");
114        print_usize(MODEL_PATH.len());
115                println(" — inference will be refused");
116                return service;
117            }
118        };
119
120        print("[ai] model bytes: ");
121        print_usize(bytes.len());
122        match Engine::load(&bytes, ENGINE_LIMIT) {
123            Ok(engine) => {
124                let describe = engine.describe();
125                print("[ai] model ready: ");
126        print_usize(describe.vocab_size as usize);
127                print(" vocab, context ");
128        print_usize(describe.context_tokens as usize);
129                print(", resident bytes ");
130        print_usize(describe.resident_bytes as usize);
131                println("");
132                service.engine = Some(engine);
133            }
134            Err(error) => {
135                service.unavailable = match error {
136                    EngineError::UnsupportedArchitecture
137                    | EngineError::UnsupportedDType(_)
138                    | EngineError::TooLarge { .. } => AiError::NotSupported,
139                    _ => AiError::NoModel,
140                };
141                println("[ai] model rejected — inference will be refused");
142            }
143        }
144        service
145    }

147    /// Release sessions whose owner has gone quiet.
148    ///
149    /// Dead client Cells cannot be watched without lifecycle authority, so silence is the signal:
150    /// a live client polls or cancels well inside [`SESSION_IDLE_TICKS`].
151    fn reap_idle_sessions(&mut self) {
152        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
153        let mut expired = [0u32; MAX_SESSIONS as usize];
154        let mut expired_count = 0usize;
155
156        for slot in self.sessions.iter_mut() {
157            let Some(session) = slot.as_ref() else {
158                continue;
159            };
160            if now.wrapping_sub(session.last_activity) <= SESSION_IDLE_TICKS {
161                continue;
162            }
163            expired[expired_count] = session.request_id;
164            expired_count += 1;
165            *slot = None;
166        }
167
168        if let Some(engine) = self.engine.as_mut() {
169            for request_id in expired.iter().take(expired_count) {
170                let _ = engine.cancel(*request_id);
171            }
172        }
173    }

175    /// Answer one request. `None` when the request must be ignored (nothing can be replied to it).
176    fn handle<'r>(
177        &mut self,
178        sender: usize,
179        buf: &[u8],
180        reply: &'r mut [u8],
181    ) -> Option<&'r [u8]> {
182        let request = match ai_proto::decode::<AiRequest<'_>>(buf) {
183            Ok(request) => request,
184            Err(_) => {
185                let error = AiError::BadRequest(limit::Violation::MessageTooLarge);
186                log_refusal("decode", error);
187                return Some(encode_failure(reply, 0, error));
188            }
189        };
190        if let Err(error) = ai_proto::validate_request(&request) {
191            log_refusal("validate", error);
192            return Some(encode_failure(reply, 0, error));
193        }
194
195        match request {
196            AiRequest::Describe => match self.engine.as_ref() {
197                Some(engine) => ai_proto::encode(&AiResponse::Description(engine.describe()), reply)
198                    .ok()
199                    .map(|encoded| &*encoded),
200                None => Some(encode_failure(reply, 0, self.unavailable)),
201            },
202            AiRequest::InferSubmit(submit) => {
203                if !matches!(submit.device, DeviceTarget::Auto | DeviceTarget::Cpu) {
204                    return Some(encode_failure(reply, 0, AiError::NotSupported));
205                }
206                let unavailable = self.unavailable;
207                let Some(engine) = self.engine.as_mut() else {
208                    return Some(encode_failure(reply, 0, unavailable));
209                };
210                let params = SamplingParams {
211                    max_tokens: submit.max_tokens,
212                    temperature_milli: submit.temperature_milli,
213                    top_k: submit.top_k,
214                    seed: submit.seed,
215                };
216                match engine.submit(submit.prompt, params) {
217                    Ok(request_id) => {
218                        self.seat(sender, request_id);
219                        ai_proto::encode(
220                            &AiResponse::Accepted {
221                                request_id,
222                                backend: backend::CPU,
223                            },
224                            reply,
225                        )
226                        .ok()
227                        .map(|encoded| &*encoded)
228                    }
229                    Err(error) => {
230                        log_refusal("submit", error);
231                        Some(encode_failure(reply, 0, error))
232                    }
233                }
234            }
235            AiRequest::InferStreamPoll {
236                request_id,
237                max_tokens,
238            } => self.poll(sender, request_id, max_tokens, reply),
239            AiRequest::InferCancel { request_id } => {
240                if !self.owned_by(sender, request_id) {
241                    return Some(encode_failure(reply, request_id, AiError::UnknownRequest));
242                }
243                if let Some(engine) = self.engine.as_mut() {
244                    let _ = engine.cancel(request_id);
245                }
246                self.release(request_id);
247                ai_proto::encode(
248                    &AiResponse::TokenChunk {
249                        request_id,
250                        tokens: &[],
251                        text: "",
252                        done: true,
253                        finish: FinishReason::Cancelled,
254                    },
255                    reply,
256                )
257                .ok()
258                .map(|encoded| &*encoded)
259            }
260            AiRequest::InferEmbed { text, device } => {
261                if !matches!(device, DeviceTarget::Auto | DeviceTarget::Cpu) {
262                    return Some(encode_failure(reply, 0, AiError::NotSupported));
263                }
264                let unavailable = self.unavailable;
265                let Some(engine) = self.engine.as_mut() else {
266                    return Some(encode_failure(reply, 0, unavailable));
267                };
268                match engine.embed(text) {
269                    Ok(values) => {
270                        let mut value_bytes =
271                            [0u8; MAX_EMBED_DIM as usize * ai_proto::EMBED_VALUE_BYTES];
272                        let encoded_values =
273                            ai_proto::encode_embedding_values(&values, &mut value_bytes)?;
274                        ai_proto::encode(
275                            &AiResponse::Embedding {
276                                dim: values.len() as u16,
277                                values: encoded_values,
278                            },
279                            reply,
280                        )
281                        .ok()
282                        .map(|encoded| &*encoded)
283                    }
284                    Err(error) => {
285                        log_refusal("embed", error);
286                        Some(encode_failure(reply, 0, error))
287                    }
288                }
289            }
290        }
291    }

293    /// Advance a session and report what it produced.
294    fn poll<'r>(
295        &mut self,
296        sender: usize,
297        request_id: u32,
298        max_tokens: u8,
299        reply: &'r mut [u8],
300    ) -> Option<&'r [u8]> {
301        if !self.owned_by(sender, request_id) {
302            return Some(encode_failure(reply, request_id, AiError::UnknownRequest));
303        }
304        let unavailable = self.unavailable;
305        let Some(engine) = self.engine.as_mut() else {
306            return Some(encode_failure(reply, request_id, unavailable));
307        };
308
309        if let Err(error) = engine.generate(request_id, STEPS_PER_POLL) {
310            if error == AiError::UnknownRequest {
311                self.release(request_id);
312            }
313            log_refusal("generate", error);
314            return Some(encode_failure(reply, request_id, error));
315        }
316        let drained = match engine.drain(request_id, max_tokens as usize) {
317            Ok(drained) => drained,
318            Err(error) => {
319                log_refusal("drain", error);
320                return Some(encode_failure(reply, request_id, error));
321            }
322        };
323
324        let mut token_bytes = [0u8; MAX_TOKENS_PER_POLL as usize * ai_proto::TOKEN_ID_BYTES];
325        let encoded_tokens = ai_proto::encode_token_ids(&drained.ids, &mut token_bytes)?;
326        let finish = drained.finish.unwrap_or(FinishReason::Length);
327        let encoded = ai_proto::encode(
328            &AiResponse::TokenChunk {
329                request_id,
330                tokens: encoded_tokens,
331                text: drained.text.as_str(),
332                done: drained.done,
333                finish,
334            },
335            reply,
336        )
337        .ok()
338        .map(|encoded| &*encoded);
339
340        if drained.done {
341            if let Some(engine) = self.engine.as_mut() {
342                let _ = engine.release(request_id);
343            }
344            self.release(request_id);
345        } else {
346            self.touch(request_id);
347        }
348        encoded
349    }

351    /// Take a session slot for `owner`.
352    fn seat(&mut self, owner: usize, request_id: u32) {
353        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
354        if let Some(slot) = self.sessions.iter_mut().find(|slot| slot.is_none()) {
355            *slot = Some(Session {
356                request_id,
357                owner,
358                last_activity: now,
359            });
360        }
361    }
362
363    /// Whether `request_id` is seated and belongs to `owner`.
364    fn owned_by(&self, owner: usize, request_id: u32) -> bool {
365        self.sessions
366            .iter()
367            .flatten()
368            .any(|session| session.request_id == request_id && session.owner == owner)
369    }
370
371    /// Drop a session slot.
372    fn release(&mut self, request_id: u32) {
373        for slot in self.sessions.iter_mut() {
374            if matches!(slot, Some(session) if session.request_id == request_id) {
375                *slot = None;
376            }
377        }
378    }
379
380    /// Mark a session as active so idle reaping leaves it alone.
381    fn touch(&mut self, request_id: u32) {
382        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
383        for session in self.sessions.iter_mut().flatten() {
384            if session.request_id == request_id {
385                session.last_activity = now;
386            }
387        }
388    }
389}

391/// Log a refusal with its typed cause, so a failing deployment is diagnosable from the
392/// serial console instead of only from the caller's side.
393fn log_refusal(stage: &str, error: AiError) {
394    ostd::io::print("[ai] refused ");
395    ostd::io::println(stage);
396    ostd::io::print_fmt(format_args!("[ai] cause: {error:?}\n"));
397}
398
399/// Encode a typed refusal.
400fn encode_failure(reply: &mut [u8], request_id: u32, error: AiError) -> &[u8] {
401    ai_proto::encode(
402        &AiResponse::Failed {
403            request_id,
404            error,
405        },
406        reply,
407    )
408    .map(|encoded| &*encoded)
409    .unwrap_or(&[])
