1//! Unified AI inference service Cell (`/bin/ai`) — Spec 24 §2.
2//!
3//! One event loop serves every caller:
4//!
5//! * `Describe` answers from the resident engine.
6//! * `InferSubmit` seats a session in a bounded table, owned by the kernel-attested Cell
7//!   incarnation that asked for it; an unattested caller is never served.
8//! * `InferStreamPoll` advances *its own* session by a bounded number of model steps and returns
9//!   the tokens produced since the previous poll, so one request can never monopolise the loop and
10//!   several callers make progress concurrently.
11//! * `InferCancel` releases a session; a session abandoned by a dead client is reclaimed after
12//!   [`SESSION_IDLE_TICKS`] of inactivity, because a service without lifecycle authority cannot
13//!   watch another Cell's exit.
14//!
15//! Truthfulness rules:
16//!
17//! * No model, an unsupported architecture, or an over-budget model is reported through
18//!   [`AiError`] — the service never answers with fabricated tokens.
19//! * A session belongs to the Cell that submitted it: another Cell polling or cancelling that id
20//!   gets `UnknownRequest`, the same answer it would get for an id that never existed.
21//! * Non-CPU device targets are refused with `NotSupported` rather than silently downgraded.
22

58// Cell heap. Holds the model bytes, the engine's weights/scratch, and the session KV caches.
59//
60// The engine addresses weights in place inside the model buffer, so the arena needs the file plus
61// the per-session and scratch buffers — not a second copy of the file.
62//
63// Default sizing serves the small checkpoints (llama2.c `stories260K`, the deterministic fixture)
64// with room for four sessions. `large-arena` sizes it for a 25 MB checkpoint, which the Cell's
65// 32 MiB virtual-address slot (`kernel/src/loader/va_alloc.rs`) holds but the default arena does
66// not; a deployment that serves a model that size builds the cell with that feature instead of
67// making every image pay 29 MB of resident RAM.
68#[cfg(feature = "large-arena")]
69ostd::declare_custom_heap!(29 * 1024 * 1024);
70#[cfg(not(feature = "large-arena"))]
71ostd::declare_custom_heap!(16 * 1024 * 1024);
72
73/// Where the service looks for its checkpoint, in order.
74///
75/// `/bin` is the P6 FAT cell-store `gen_disk.ps1` deploys into (and on a netboot image it is the
76/// VIFS1 ramdisk). `/mnt/sd` is the board's own FAT partition: keeping the checkpoint there keeps
77/// the *image* small, which matters where the kernel's memory layout is what breaks first — a
78/// 43.6 MB payload (this checkpoint embedded in VIFS1) panics the Raspberry Pi 3 kernel while the
79/// same cells with the checkpoint on the card boot and serve it. The resident cost inside the Cell
80/// is identical either way, so a deployment should prefer the card.
81const MODEL_PATHS: [&str; 2] = ["/bin/ai-model.gguf", "/mnt/sd/ai-model.gguf"];
82
83/// Ceiling handed to the engine: the arena above minus room for the loader's bookkeeping and the
84/// session table.
85#[cfg(feature = "large-arena")]
86const ENGINE_LIMIT: usize = 29 * 1024 * 1024;
87#[cfg(not(feature = "large-arena"))]
88const ENGINE_LIMIT: usize = 14 * 1024 * 1024;
89
90/// Model steps advanced per poll. Bounds one session's share of the event loop; a longer
91/// generation simply takes more polls.
92const STEPS_PER_POLL: usize = 4;
93
94/// Sessions idle for this many 10 ms ticks are reclaimed.
95///
96/// A service Cell cannot watch another Cell's exit (`NotifyOnExit` needs lifecycle authority), so
97/// an abandoned session is released by deadline instead: 30 s of silence is far longer than any
98/// poll interval a live client produces.
99const SESSION_IDLE_TICKS: u64 = 3_000;
100
101/// One live session and the Cell that owns it.
102struct Session {
103    request_id: u32,
104    owner: Owner,
105    last_activity: u64,
106}

108/// A session's owner: one incarnation of a Cell, as the kernel attests it. Its threads share it.
109#[derive(Clone, Copy, PartialEq, Eq)]
110struct Owner {
111    cell_id: u64,
112    generation: u64,
113}
114
115impl Owner {
116    fn of(caller: &CallerIdentity) -> Self {
117        Self {
118            cell_id: caller.cell_id,
119            generation: caller.generation,
120        }
121    }
122}

124/// The service state.
125struct AiService {
126    engine: Option<Engine>,
127    sessions: [Option<Session>; MAX_SESSIONS as usize],
128    /// Why inference is unavailable when `engine` is `None`, so refusals stay specific.
129    unavailable: AiError,
130}

134fn cell_main() {
135    init_custom_heap();
136    let service = AiService::load();
137
138    let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
139    ostd::service::serve(
140        ostd::service::ServeConfig::with_heartbeat(),
141        &mut Ai {
142            service,
143            reply: &mut reply,
144        },
145    );
146}

148struct Ai<'r> {

150    reply: &'r mut [u8; ai_proto::AI_IPC_BUF_SIZE],
151}

153impl ostd::service::Handler for Ai<'_> {
154    fn on_request(&mut self, request: ostd::service::Request<'_>) {
155        self.service.reap_idle_sessions();
156        let owner = Owner::of(request.caller());
157        if let Some(encoded) = self.service.handle(owner, request.payload(), self.reply) {
158            let _ = request.reply(encoded);
159        }
160    }
161
162    /// Abandoned sessions are reaped on a tick too, not only when a request arrives.
163    fn on_idle(&mut self) -> ostd::service::Wait {
164        self.service.reap_idle_sessions();
165        ostd::service::Wait::Ticks(SESSION_IDLE_TICKS)
166    }
167}

169impl AiService {
170    /// Load the model (if present) and build the service.
171    fn load() -> Self {
172        println("[ai] inference service starting");
173        let mut service = Self {
174            engine: None,
175            sessions: [const { None }; MAX_SESSIONS as usize],
176            unavailable: AiError::NoModel,
177        };
178
179        // Read timing is part of the service's contract with its operators: a model that is slow to
180        // read on a given board is an operational fact, and without the number the only symptom is
181        // a silent stall on the serial console.
182        let read_started = ostd::syscall::sys_get_time_ms().unwrap_or(0);
183        let mut model_path = MODEL_PATHS[0];
184        let mut found = None;
185        for candidate in MODEL_PATHS {
186            if let Some(bytes) = read_model(candidate, ENGINE_LIMIT) {
187                model_path = candidate;
188                found = Some(bytes);
189                break;
190            }
191        }
192        let bytes = match found {
193            Some(bytes) => bytes,
194            None => {
195                print("[ai] no model at any of ");
196                print_usize(MODEL_PATHS.len());
197                println(" candidate paths — inference will be refused");
198                return service;
199            }
200        };
201        print("[ai] model path: ");
202        println(model_path);
203        let read_ms = ostd::syscall::sys_get_time_ms()
204            .unwrap_or(0)
205            .saturating_sub(read_started);
206
207        print("[ai] model bytes: ");
208        print_usize(bytes.len());
209        print(" read in ");
210        print_usize(read_ms as usize);
211        println(" ms");
212        let load_started = ostd::syscall::sys_get_time_ms().unwrap_or(0);
213        // The engine takes ownership of the model bytes and addresses the weights in place, so this
214        // is the only copy of the file in the cell.
215        match Engine::load(bytes, ENGINE_LIMIT) {
216            Ok(engine) => {
217                let load_ms = ostd::syscall::sys_get_time_ms()
218                    .unwrap_or(0)
219                    .saturating_sub(load_started);
220                print("[ai] engine load: ");
221                print_usize(load_ms as usize);
222                println(" ms");
223                let describe = engine.describe();
224                print("[ai] model ready: ");
225                print_usize(describe.vocab_size as usize);
226                print(" vocab, context ");
227                print_usize(describe.context_tokens as usize);
228                print(", resident bytes ");
229                print_usize(describe.resident_bytes as usize);
230                println("");
231                service.engine = Some(engine);
232            }
233            Err(error) => {
234                service.unavailable = match error {
235                    EngineError::UnsupportedArchitecture
236                    | EngineError::UnsupportedDType(_)
237                    | EngineError::TooLarge { .. } => AiError::NotSupported,
238                    _ => AiError::NoModel,
239                };
240                println("[ai] model rejected — inference will be refused");
241            }
242        }
243        service
244    }

246    /// Release sessions whose owner has gone quiet.
247    ///
248    /// Dead client Cells cannot be watched without lifecycle authority, so silence is the signal:
249    /// a live client polls or cancels well inside [`SESSION_IDLE_TICKS`].
250    fn reap_idle_sessions(&mut self) {
251        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
252        let mut expired = [0u32; MAX_SESSIONS as usize];
253        let mut expired_count = 0usize;
254
255        for slot in self.sessions.iter_mut() {
256            let Some(session) = slot.as_ref() else {
257                continue;
258            };
259            if now.wrapping_sub(session.last_activity) <= SESSION_IDLE_TICKS {
260                continue;
261            }
262            expired[expired_count] = session.request_id;
263            expired_count += 1;
264            *slot = None;
265        }
266
267        if let Some(engine) = self.engine.as_mut() {
268            for request_id in expired.iter().take(expired_count) {
269                let _ = engine.cancel(*request_id);
270            }
271        }
272    }

274    /// The engine runs only on the CPU: no GPU inference path exists yet, so an explicit GPU
275    /// target is refused instead of being acknowledged and silently executed on the CPU.
276    fn select_backend(device: DeviceTarget) -> Result<u32, AiError> {
277        match device {
278            DeviceTarget::Auto | DeviceTarget::Cpu => Ok(backend::CPU),
279            _ => Err(AiError::NotSupported),
280        }
281    }

283    /// Answer one request. `None` when the request must be ignored (nothing can be replied to it).
284    fn handle<'r>(&mut self, owner: Owner, buf: &[u8], reply: &'r mut [u8]) -> Option<&'r [u8]> {
285        let request = match ai_proto::decode::<AiRequest<'_>>(buf) {
286            Ok(request) => request,
287            Err(_) => {
288                let error = AiError::BadRequest(limit::Violation::MessageTooLarge);
289                log_refusal("decode", error);
290                return Some(encode_failure(reply, 0, error));
291            }
292        };
293        if let Err(error) = ai_proto::validate_request(&request) {
294            log_refusal("validate", error);
295            return Some(encode_failure(reply, 0, error));
296        }
297
298        match request {
299            AiRequest::Describe => match self.engine.as_ref() {
300                Some(engine) => {
301                    ai_proto::encode(&AiResponse::Description(engine.describe()), reply)
302                        .ok()
303                        .map(|encoded| &*encoded)
304                }
305                None => Some(encode_failure(reply, 0, self.unavailable)),
306            },
307            AiRequest::InferSubmit(submit) => {
308                let chosen_backend = match Self::select_backend(submit.device) {
309                    Ok(b) => b,
310                    Err(error) => {
311                        log_refusal("submit device", error);
312                        return Some(encode_failure(reply, 0, error));
313                    }
314                };
315                let unavailable = self.unavailable;
316                let Some(engine) = self.engine.as_mut() else {
317                    return Some(encode_failure(reply, 0, unavailable));
318                };
319                let params = SamplingParams {
320                    max_tokens: submit.max_tokens,
321                    temperature_milli: submit.temperature_milli,
322                    top_k: submit.top_k,
323                    seed: submit.seed,
324                };
325                match engine.submit(submit.prompt, params) {
326                    Ok(request_id) => {
327                        self.seat(owner, request_id);
328                        ai_proto::encode(
329                            &AiResponse::Accepted {
330                                request_id,
331                                backend: chosen_backend,
332                            },
333                            reply,
334                        )
335                        .ok()
336                        .map(|encoded| &*encoded)
337                    }
338                    Err(error) => {
339                        log_refusal("submit", error);
340                        Some(encode_failure(reply, 0, error))
341                    }
342                }
343            }
344            AiRequest::InferStreamPoll {
345                request_id,
346                max_tokens,
347            } => self.poll(owner, request_id, max_tokens, reply),
348            AiRequest::InferCancel { request_id } => {
349                if !self.owned_by(owner, request_id) {
350                    return Some(encode_failure(reply, request_id, AiError::UnknownRequest));
351                }
352                if let Some(engine) = self.engine.as_mut() {
353                    let _ = engine.cancel(request_id);
354                }
355                self.release(request_id);
356                ai_proto::encode(
357                    &AiResponse::TokenChunk {
358                        request_id,
359                        tokens: &[],
360                        text: "",
361                        done: true,
362                        finish: FinishReason::Cancelled,
363                    },
364                    reply,
365                )
366                .ok()
367                .map(|encoded| &*encoded)
368            }
369            AiRequest::InferEmbed { text, device } => {
370                if let Err(error) = Self::select_backend(device) {
371                    log_refusal("embed device", error);
372                    return Some(encode_failure(reply, 0, error));
373                }
374                let unavailable = self.unavailable;
375                let Some(engine) = self.engine.as_mut() else {
376                    return Some(encode_failure(reply, 0, unavailable));
377                };
378                match engine.embed(text) {
379                    Ok(values) => {
380                        let mut value_bytes =
381                            [0u8; MAX_EMBED_DIM as usize * ai_proto::EMBED_VALUE_BYTES];
382                        let encoded_values =
383                            ai_proto::encode_embedding_values(&values, &mut value_bytes)?;
384                        ai_proto::encode(
385                            &AiResponse::Embedding {
386                                dim: values.len() as u16,
387                                values: encoded_values,
388                            },
389                            reply,
390                        )
391                        .ok()
392                        .map(|encoded| &*encoded)
393                    }
394                    Err(error) => {
395                        log_refusal("embed", error);
396                        Some(encode_failure(reply, 0, error))
397                    }
398                }
399            }
400        }
401    }

403    /// Advance a session and report what it produced.
404    fn poll<'r>(
405        &mut self,
406        owner: Owner,
407        request_id: u32,
408        max_tokens: u8,
409        reply: &'r mut [u8],
410    ) -> Option<&'r [u8]> {
411        if !self.owned_by(owner, request_id) {
412            return Some(encode_failure(reply, request_id, AiError::UnknownRequest));
413        }
414        let unavailable = self.unavailable;
415        let Some(engine) = self.engine.as_mut() else {
416            return Some(encode_failure(reply, request_id, unavailable));
417        };
418
419        if let Err(error) = engine.generate(request_id, STEPS_PER_POLL) {
420            if error == AiError::UnknownRequest {
421                self.release(request_id);
422            }
423            log_refusal("generate", error);
424            return Some(encode_failure(reply, request_id, error));
425        }
426        let drained = match engine.drain(request_id, max_tokens as usize) {
427            Ok(drained) => drained,
428            Err(error) => {
429                log_refusal("drain", error);
430                return Some(encode_failure(reply, request_id, error));
431            }
432        };
433
434        let mut token_bytes = [0u8; MAX_TOKENS_PER_POLL as usize * ai_proto::TOKEN_ID_BYTES];
435        let encoded_tokens = ai_proto::encode_token_ids(&drained.ids, &mut token_bytes)?;
436        let finish = drained.finish.unwrap_or(FinishReason::Length);
437        let encoded = ai_proto::encode(
438            &AiResponse::TokenChunk {
439                request_id,
440                tokens: encoded_tokens,
441                text: drained.text.as_str(),
442                done: drained.done,
443                finish,
444            },
445            reply,
446        )
447        .ok()
448        .map(|encoded| &*encoded);
449
450        if drained.done {
451            if let Some(engine) = self.engine.as_mut() {
452                let _ = engine.release(request_id);
453            }
454            self.release(request_id);
455        } else {
456            self.touch(request_id);
457        }
458        encoded
459    }

461    /// Take a session slot for `owner`.
462    fn seat(&mut self, owner: Owner, request_id: u32) {
463        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
464        if let Some(slot) = self.sessions.iter_mut().find(|slot| slot.is_none()) {
465            *slot = Some(Session {
466                request_id,
467                owner,
468                last_activity: now,
469            });
470        }
471    }
472
473    /// Whether `request_id` is seated and belongs to `owner`.
474    fn owned_by(&self, owner: Owner, request_id: u32) -> bool {
475        self.sessions
476            .iter()
477            .flatten()
478            .any(|session| session.request_id == request_id && session.owner == owner)
479    }
480
481    /// Drop a session slot.
482    fn release(&mut self, request_id: u32) {
483        for slot in self.sessions.iter_mut() {
484            if matches!(slot, Some(session) if session.request_id == request_id) {
485                *slot = None;
486            }
487        }
488    }
489
490    /// Mark a session as active so idle reaping leaves it alone.
491    fn touch(&mut self, request_id: u32) {
492        let now = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
493        for session in self.sessions.iter_mut().flatten() {
494            if session.request_id == request_id {
495                session.last_activity = now;
496            }
497        }
498    }

501/// Bytes moved per capability read. `ReadCap` accepts up to the kernel's 64 MiB user-buffer ceiling,
502/// so this trades syscall count against a stack/heap buffer: 256 KiB reads a 25 MB model in ~100
503/// syscalls instead of ~6,700 IPC round trips.
504const MODEL_READ_CHUNK: usize = 256 * 1024;
505
506/// Read the model file, preferring the kernel capability path.
507///
508/// Two paths exist and they cost very different things:
509///
510/// * **Kernel cap path** (`OpenCap`/`SeekCap`/`ReadCap`): a handful of syscalls for the whole file.
511///   It resolves the kernel's VIFS1 ramdisk — where a development image puts the model — and needs
512///   the uppercase path FAT16 uses.
513/// * **VFS service**: the only path for a model deployed in the on-disk cell-store. It moves one
514///   4 KiB IPC message per round trip, so a 25 MB model costs thousands of round trips (measured:
515///   146 s in QEMU TCG against a few seconds for the cap path). Kept as the fallback because a
516///   cell-store deployment is the shape a released image has.
517///
518/// Returns `None` when the file does not exist or exceeds `limit`.
519fn read_model(path: &str, limit: usize) -> Option<Vec<u8>> {
520    let upper: alloc::string::String = path.chars().map(|c| c.to_ascii_uppercase()).collect();
521    if let Some(bytes) = read_model_cap(&upper, limit) {
522        return Some(bytes);
523    }
524    let mut vfs = ostd::clients::VfsClient::new();
525    vfs.read_file_bounded(path, limit).ok()
526}
527
528/// One whole-file read through the kernel capability syscalls.
529fn read_model_cap(path: &str, limit: usize) -> Option<Vec<u8>> {
530    let mut file = ostd::fs::File::open(path).ok()?;
531    let size = usize::try_from(file.size().ok()?).ok()?;
532    if size == 0 || size > limit {
533        return None;
534    }
535    let mut bytes = Vec::with_capacity(size);
536    let mut chunk = alloc::vec![0u8; MODEL_READ_CHUNK.min(size)];
537    while bytes.len() < size {
538        let want = (size - bytes.len()).min(chunk.len());
539        match file.read_at(bytes.len() as u64, &mut chunk[..want]) {
540            Ok(0) => break,
541            Ok(n) => bytes.extend_from_slice(&chunk[..n]),
542            Err(_) => return None,
543        }
544    }
545    let _ = file.close();
546    if bytes.len() == size {
547        Some(bytes)
548    } else {
549        None
550    }
551}

553/// Log a refusal with its typed cause, so a failing deployment is diagnosable from the
554/// serial console instead of only from the caller's side.
555fn log_refusal(stage: &str, error: AiError) {
556    ostd::io::print("[ai] refused ");
557    ostd::io::println(stage);
558    ostd::io::print_fmt(format_args!("[ai] cause: {error:?}\n"));
559}
560
561/// Encode a typed refusal.
562fn encode_failure(reply: &mut [u8], request_id: u32, error: AiError) -> &[u8] {
563    ai_proto::encode(&AiResponse::Failed { request_id, error }, reply)
564        .map(|encoded| &*encoded)
565        .unwrap_or(&[])
