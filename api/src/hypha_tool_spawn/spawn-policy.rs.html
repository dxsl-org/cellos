1//! What `tool-spawn` may start and stop on behalf of its `core`.
2//!
3//! WHY: `tool-spawn` holds SpawnCap and acts on LLM output that crossed the
4//! network. Unconstrained, `spawn_cell` reaches every `/bin` cell the
5//! tool-spawn launch edge admits (network clients and listeners included) and
6//! `kill_cell` reaches every non-system task. Both are narrowed to what an
7//! assistant needs: start a reviewed, capability-free app, and stop only what it
8//! started itself.
9
10/// Exact paths `spawn_cell` may launch. Matching is byte-exact: no prefixes, no
11/// normalisation, so `..`, doubled slashes and other absolute paths can never
12/// match. Each entry must declare no VFS-mutate, network, device or spawn
13/// authority, and must run without argv (tool-spawn passes none).
14pub const SPAWN_ALLOWLIST: &[&str] = &[
15    // Self-contained game; manifest is block_io/network/spawn = false.
16    "/bin/tetris",
17];
18
19/// Most cells one session may have alive through `spawn_cell` at once. Bounds
20/// the tracking table and how many cells LLM output can keep resident.
21pub const MAX_SPAWNED: usize = 8;
22
23/// Whether `spawn_cell` may launch `path`.
24pub fn spawn_allowed(path: &str) -> bool {
25    SPAWN_ALLOWLIST.contains(&path)
26}
27
28/// Tids this `tool-spawn` instance started and has not yet killed.
29///
30/// Invariant: `kill_cell` acts only on a tid in this set. Tids are never
31/// re-issued within one boot, so an entry can never come to name a cell this
32/// session did not start.
33pub struct SpawnedCells {
34    tids: [usize; MAX_SPAWNED],
35    len: usize,
36}
37
38impl SpawnedCells {
39    pub const fn new() -> Self {
40        Self {
41            tids: [0; MAX_SPAWNED],
42            len: 0,
43        }
44    }
45
46    /// No room for another spawn until one tracked cell is killed.
47    pub fn is_full(&self) -> bool {
48        self.len == MAX_SPAWNED
49    }
50
51    /// Track a freshly spawned `tid`. Returns `false` (and tracks nothing) when
52    /// full or for tid 0, which never names a task.
53    pub fn record(&mut self, tid: usize) -> bool {
54        if tid == 0 || self.is_full() || self.contains(tid) {
55            return false;
56        }
57        self.tids[self.len] = tid;
58        self.len += 1;
59        true
60    }
61
62    pub fn contains(&self, tid: usize) -> bool {
63        tid != 0 && self.tids[..self.len].contains(&tid)
64    }
65
66    /// Stop tracking `tid`; `false` when it was not tracked.
67    pub fn release(&mut self, tid: usize) -> bool {
68        let Some(at) = self.tids[..self.len].iter().position(|t| *t == tid) else {
69            return false;
70        };
71        self.len -= 1;
72        self.tids[at] = self.tids[self.len];
73        self.tids[self.len] = 0;
74        true
75    }
76}
77
78impl Default for SpawnedCells {
79    fn default() -> Self {
80        Self::new()
81    }
82}
