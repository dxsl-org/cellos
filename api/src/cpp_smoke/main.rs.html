1//! `cpp-freestanding` runtime profile smoke cell.
2//!
3//! Proves the profile end to end in a Tier 2 paged domain:
4//!   1. the cell is admitted as an `FFI`-class Tier 2 cell (kernel marker),
5//!   2. C++ static constructors ran (crt0 walks `__init_array`),
6//!   3. virtual dispatch, templates, and `new`/`delete` work,
7//!   4. file I/O works both ways: the VFS service over typed IPC and the shim's
8//!      C `open`/`read`/`close` ABI through the kernel file table.
9//!
10//! Markers (integration-test contract):
11//!   `[cpp-smoke] static-ctor marker=0xC0FFEE11`
12//!   `[cpp-smoke] virtual-dispatch area=37`
13//!   `[cpp-smoke] templates total=64`
14//!   `[cpp-smoke] heap churn checksum=…`
15//!   `[cpp-smoke] vfs client roundtrip bytes=…`
16//!   `[cpp-smoke] c-abi read magic=ELF`
17//!   `CPP-SMOKE: PASS`
18
19#![no_std]
20#![no_main]

53const STATIC_CTOR_SENTINEL: u32 = 0xC0FFEE11;
54const VFS_PATH: &str = "/srv/cpp-smoke.txt";
55/// `/BIN/INIT` is the embedded init ELF: the kernel file table resolves it, and
56/// the shim's C `open`/`read` ABI goes through that table (not the VFS service).
57const BIN_INIT_NUL: &[u8] = b"/BIN/INIT\0";
58const ELF_MAGIC: [u8; 4] = [0x7F, b'E', b'L', b'F'];
59const PAYLOAD: &[u8] = b"cpp-freestanding payload read through the VFS service";
60const HEAP_ITERATIONS: i32 = 64;
61
62fn fail(stage: &str) -> ! {
63    println(&format!("[cpp-smoke] FAIL stage={}", stage));
64    sys_exit(1)
65}
66
67fn expect_eq(stage: &str, actual: i64, expected: i64) {
68    if actual != expected {
69        println(&format!(
70            "[cpp-smoke] mismatch stage={} actual={} expected={}",
71            stage, actual, expected
72        ));
73        fail(stage);
74    }
75}
76
77/// The same checksum the C++ side computes, implemented independently here: a
78/// wrong `operator new`/`delete` (aliasing, truncation, double free) shows up as
79/// a mismatch rather than as a lucky pass.
80fn expected_heap_checksum(iterations: i32) -> i32 {
81    let mut checksum: i32 = 0;
82    for i in 0..iterations {
83        for j in 0..16u32 {
84            checksum += ((i as u32 + j) % 7) as i32;
85        }
86    }
87    checksum
88}

90fn cell_main() {
91    println("[cpp-smoke] start (Tier 2 FFI cell, cpp-freestanding profile)");
92
93    // 1. Static construction: the sentinel is written by a global constructor in
94    //    engine.cpp, so a non-zero value proves crt0 ran `__init_array`.
95    let ctor = cpp::static_ctor_marker();
96    expect_eq("static-ctor", ctor as i64, STATIC_CTOR_SENTINEL as i64);
97    println(&format!("[cpp-smoke] static-ctor marker=0x{:08X}", ctor));
98
99    // Static destructors are a documented non-feature (atexit is a stub): the
100    // cell is reclaimed by the kernel, so nothing runs them.
101    expect_eq("static-dtor", cpp::static_dtor_marker() as i64, 0);
102
103    // 2. Virtual dispatch through a base pointer, then through a virtual
104    //    destructor on a heap object.
105    let area = cpp::virtual_dispatch_total();
106    expect_eq("virtual-dispatch", area as i64, 37);
107    println(&format!("[cpp-smoke] virtual-dispatch area={}", area));
108
109    let deleted_area = cpp::virtual_delete_area();
110    expect_eq("virtual-delete", deleted_area as i64, 36);
111    println(&format!("[cpp-smoke] virtual-delete area={}", deleted_area));
112
113    // 3. Templates: tmax/tmin plus two Stack instantiations and twice<i32>.
114    let total = cpp::template_total();
115    expect_eq("templates", total as i64, 64);
116    println(&format!("[cpp-smoke] templates total={}", total));
117
118    // 4. Allocator: operator new/delete over the shim allocator.
119    let checksum = cpp::heap_churn(HEAP_ITERATIONS);
120    expect_eq(
121        "heap-churn",
122        checksum as i64,
123        expected_heap_checksum(HEAP_ITERATIONS) as i64,
124    );
125    println(&format!(
126        "[cpp-smoke] heap churn checksum={} iterations={}",
127        checksum, HEAP_ITERATIONS
128    ));
129
130    // 5. File I/O, two paths that are genuinely different in Cellos:
131    //    (a) the VFS service over typed IPC (the `/srv` CellosFS volume), and
132    //    (b) the shim's C `open`/`read`/`close` ABI, which resolves through the
133    //        kernel file table (`/BIN/INIT` is the embedded init ELF).
134    let mut vfs = VfsClient::new();
135    if vfs.write_file(VFS_PATH, PAYLOAD).is_err() {
136        fail("vfs-write");
137    }
138    let roundtrip = match vfs.read_file(VFS_PATH) {
139        Ok(bytes) => bytes,
140        Err(_) => fail("vfs-read"),
141    };
142    if roundtrip != PAYLOAD {
143        fail("vfs-content");
144    }
145    println(&format!(
146        "[cpp-smoke] vfs client roundtrip bytes={}",
147        roundtrip.len()
148    ));
149
150    let mut magic = [0u8; 4];
151    let read = cpp::read_file(BIN_INIT_NUL, &mut magic);
152    expect_eq("c-abi-read", read as i64, 4);
153    if magic != ELF_MAGIC {
154        fail("c-abi-content");
155    }
156    println("[cpp-smoke] c-abi read magic=ELF");
157
158    println("CPP-SMOKE: PASS");
159    sys_exit(0);
160}
