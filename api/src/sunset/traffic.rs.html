24/// Number of `DeferredPacket`s to queue.
25///
26/// Each entry takes around 40 bytes.
27const DEFER_COUNT: usize = 10;
28
29// Either a slice or boxed array.
30// Similar to managed::ManagedSlice.
31//
32// Zeroize is slow for fuzzing, so skip it.
33// In normal operation one zeroize per connection is fine.
34#[cfg_attr(not(fuzzing), derive(zeroize::ZeroizeOnDrop))]
35enum SliceOrVec<'a> {
36    Borrowed(&'a mut [u8]),
37
38    /// `'static` variant
39    #[cfg(feature = "alloc")]
40    Owned(Box<[u8; config::SSH_MAX_PACKET]>),
41}
42
43impl Deref for SliceOrVec<'_> {
44    type Target = [u8];
45
46    fn deref(&self) -> &Self::Target {
47        match self {
48            Self::Borrowed(r) => r,
49            #[cfg(feature = "alloc")]
50            Self::Owned(r) => r.as_ref(),
51        }
52    }
53}
54
55impl DerefMut for SliceOrVec<'_> {
56    fn deref_mut(&mut self) -> &mut Self::Target {
57        match self {
58            Self::Borrowed(r) => r,
59            #[cfg(feature = "alloc")]
60            Self::Owned(r) => r.as_mut(),
61        }
62    }
63}

77pub struct TrafIn<'a> {
78    // TODO: decompression will need another buffer
79    /// Accumulated input buffer.
80    ///
81    /// Should be sized to fit the largest packet allowed for input.
82    /// Contains ciphertext or cleartext, decrypted in-place.
83    /// Only contains a single SSH packet at a time.
84    buf: SliceOrVec<'a>,
85    state: RxState,
86}
87
88#[derive(Debug)]
89enum RxState {
90    /// Awaiting read, buffer is unused
91    Idle,
92    /// Reading initial encrypted block for packet length. idx > 0.
93    ReadInitial { idx: usize },
94    /// Reading remainder of encrypted packet
95    Read { idx: usize, expect: usize },
96    /// Whole encrypted packet has been read
97    ReadComplete { len: usize },
98    /// Decrypted complete input payload
99    InPayload { len: usize, seq: u32 },
100    /// Decrypted incoming channel data
101    InChannelData {
102        /// channel number
103        chan: ChanNum,
104        /// Normal or Stderr
105        dt: ChanData,
106        /// read index of channel data. should transition to Idle once `idx==len`
107        idx: usize,
108        /// length of channel data
109        len: usize,
110    },
111}
112
113impl core::fmt::Debug for TrafIn<'_> {
114    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
115        f.debug_struct("TrafIn").field("state", &self.state).finish_non_exhaustive()
116    }
117}

119impl<'a> TrafIn<'a> {
120    pub fn new(buf: &'a mut [u8]) -> Self {
121        Self { buf: SliceOrVec::Borrowed(buf), state: RxState::Idle }
122    }
123
124    pub fn is_input_ready(&self) -> bool {
125        match self.state {
126            RxState::Idle | RxState::ReadInitial { .. } | RxState::Read { .. } => {
127                true
128            }
129            RxState::ReadComplete { .. }
130            | RxState::InPayload { .. }
131            | RxState::InChannelData { .. } => false,
132        }
133    }
134
135    /// Returns the number of bytes consumed.
136    pub fn input(
137        &mut self,
138        keys: &mut KeyState,
139        remote_version: &mut RemoteVersion,
140        buf: &[u8],
141    ) -> Result<usize, Error> {
142        let mut inlen = 0;
143        debug_assert!(self.is_input_ready());
144        if remote_version.version().is_none() && matches!(self.state, RxState::Idle)
145        {
146            // Handle initial version string
147            inlen += remote_version.consume(buf)?;
148        }
149        let buf = &buf[inlen..];
150
151        inlen += self.fill_input(keys, buf)?;
152        Ok(inlen)
153    }
154
155    /// Called when `payload()` is complete.
156    pub(crate) fn done_payload(&mut self) {
157        if let RxState::InPayload { .. } = self.state {
158            self.state = RxState::Idle
159        }
160    }
161
162    /// Called when `payload()` is complete, zeroizes the payload
163    /// Also calls `done_payload()`.
164    pub(crate) fn zeroize_payload(&mut self) {
165        if let RxState::InPayload { len, .. } = self.state {
166            self.buf[SSH_PAYLOAD_START..SSH_PAYLOAD_START + len].zeroize();
167            self.done_payload()
168        }
169    }
170
171    /// Returns a reference to the decrypted payload buffer if ready,
172    /// and the `seq` of that packet.
173    pub(crate) fn payload(&self) -> Option<(&[u8], u32)> {
174        match self.state {
175            RxState::InPayload { len, seq } => {
176                let payload = &self.buf[SSH_PAYLOAD_START..SSH_PAYLOAD_START + len];
177                Some((payload, seq))
178            }
179            _ => None,
180        }
181    }
182
183    fn fill_input(
184        &mut self,
185        keys: &mut KeyState,
186        buf: &[u8],
187    ) -> Result<usize, Error> {
188        let size_block = keys.size_block_dec();
189        // 'r' is the remaining input, a slice that moves along.
190        // Used to calculate the size to return
191        let mut r = buf;
192
193        trace!("fill_input {:?}", self.state);
194
195        // Fill the initial block from either Idle with input,
196        // partial initial block
197        if let Some(idx) = match self.state {
198            RxState::Idle if !r.is_empty() => Some(0),
199            RxState::ReadInitial { idx } => Some(idx),
200            _ => None,
201        } {
202            trace!("fill_input idle idx {idx}");
203            let need = (size_block - idx).clamp(0, r.len());
204            let x;
205            (x, r) = r.split_at(need);
206            let w = &mut self.buf[idx..idx + need];
207            w.copy_from_slice(x);
208            self.state = RxState::ReadInitial { idx: idx + need }
209        }
210
211        // Have enough input now to decrypt the packet length
212        if let RxState::ReadInitial { idx } = self.state {
213            trace!("fill_input readinit {idx}");
214            if idx >= size_block {
215                let w = &mut self.buf[..size_block];
216                let total_len = keys.decrypt_first_block(w)?;
217                if total_len > self.buf.len() {
218                    // TODO: Or just BadDecrypt could make more sense if
219                    // it were packet corruption/decryption failure
220                    return Err(Error::BigPacket { size: total_len });
221                }
222                if total_len < size_block {
223                    return Err(Error::BadDecrypt);
224                }
225                trace!("fill_input set read  {idx} ex {total_len}");
226                self.state = RxState::Read { idx, expect: total_len }
227            }
228        }
229
230        // Know expected length, read until the end of the packet.
231        // We have already validated that expect_len <= buf_size
232        if let RxState::Read { ref mut idx, expect } = self.state {
233            trace!("expect {expect} idx {idx}");
234            let need = (expect - *idx).min(r.len());
235            let x;
236            (x, r) = r.split_at(need);
237            let w = &mut self.buf[*idx..*idx + need];
238            w.copy_from_slice(x);
239            *idx += need;
240            if *idx == expect {
241                self.state = RxState::ReadComplete { len: expect }
242            }
243        }
244
245        if let RxState::ReadComplete { len } = self.state {
246            let w = &mut self.buf[..len];
247            let seq = keys.recv_seq();
248            let payload_len = keys.decrypt(w)?;
249            self.state = RxState::InPayload { len: payload_len, seq }
250        }
251        trace!("out");
252
253        Ok(buf.len() - r.len())
254    }
255
256    /// Returns `(channel, dt, length)`
257    pub fn read_channel_ready(&self) -> Option<(ChanNum, ChanData, usize)> {
258        match self.state {
259            RxState::InChannelData { chan, dt, idx, len } => {
260                debug_assert!(len > idx);
261                let rem = len - idx;
262                Some((chan, dt, rem))
263            }
264            _ => None,
265        }
266    }
267
268    /// Set channel data ready to be read.
269    pub fn set_read_channel_data(
270        &mut self,
271        di: channel::DataIn,
272    ) -> Result<(ChanNum, ChanData)> {
273        match self.state {
274            RxState::InPayload { .. } => {
275                let idx = SSH_PAYLOAD_START + di.dt.packet_offset();
276                self.state = RxState::InChannelData {
277                    chan: di.num,
278                    dt: di.dt,
279                    idx,
280                    len: idx + di.len.get(),
281                };
282                Ok((di.num, di.dt))
283            }
284            _ => Error::bug(),
285        }
286    }
287
288    // Returns the length returned, and an Option<len> indicating whether the whole
289    // data packet has been completed, or None if some is still pending.
290    pub fn read_channel(
291        &mut self,
292        chan: ChanNum,
293        dt: ChanData,
294        buf: &mut [u8],
295    ) -> (usize, Option<usize>) {
296        match self.state {
297            RxState::InChannelData { chan: c, dt: e, ref mut idx, len }
298                if (c, e) == (chan, dt) =>
299            {
300                debug_assert!(len > *idx);
301                let wlen = (len - *idx).min(buf.len());
302                buf[..wlen].copy_from_slice(&self.buf[*idx..*idx + wlen]);
303                *idx += wlen;
304
305                if *idx == len {
306                    // all done.
307                    self.state = RxState::Idle;
308                    (wlen, Some(len))
309                } else {
310                    (wlen, None)
311                }
312            }
313            _ => (0, None),
314        }
315    }
316
317    // Returns (length, complete: Option<len: usize>>, Option(dt))
318    pub fn read_channel_either(
319        &mut self,
320        chan: ChanNum,
321        buf: &mut [u8],
322    ) -> (usize, Option<usize>, ChanData) {
323        match self.state {
324            RxState::InChannelData { chan: c, dt, ref mut idx, len }
325                if c == chan =>
326            {
327                debug_assert!(len > *idx);
328                let wlen = (len - *idx).min(buf.len());
329                buf[..wlen].copy_from_slice(&self.buf[*idx..*idx + wlen]);
330                // info!("idx {} += wlen {} = {}", *idx, wlen, *idx+wlen);
331                *idx += wlen;
332
333                if *idx == len {
334                    // all done.
335                    self.state = RxState::Idle;
336                    (wlen, Some(len), dt)
337                } else {
338                    (wlen, None, dt)
339                }
340            }
341            _ => (0, None, ChanData::Normal),
342        }
343    }
344
345    /// Returns the length of data discarded
346    pub fn discard_read_channel(&mut self, chan: ChanNum) -> usize {
347        match self.state {
348            RxState::InChannelData { chan: c, len, .. } if c == chan => {
349                self.state = RxState::Idle;
350                len
351            }
352            _ => 0,
353        }
354    }
355}

357pub(crate) struct TrafOut<'a> {
358    // TODO: decompression will need another buffer
359    /// Accumulated output buffer.
360    ///
361    /// Should be sized to fit the largest
362    /// sequence of packets to be sent at once.
363    /// Contains ciphertext or cleartext, encrypted in-place.
364    /// Writing may contain multiple SSH packets to write out, encrypted
365    /// in-place as they are written to `buf`.
366    buf: SliceOrVec<'a>,
367    state: TxState,
368
369    drain: bool,
370
371    // Set between sending KexInit and sending NewKeys.
372    sending_kex: bool,
373
374    deferred_packets: Deque<DeferredPacket, DEFER_COUNT>,
375}
376
377/// State machine for writes
378#[derive(Debug)]
379enum TxState {
380    /// Awaiting write, buffer is unused
381    Idle,
382
383    /// Writing to the socket. Buffer is encrypted in-place.
384    /// Should never be left in `idx==len` state,
385    /// instead should transition to Idle
386    Write {
387        /// Cursor position in the buffer
388        idx: usize,
389        /// Buffer available to write
390        len: usize,
391    },
392
393    /// No more output will be produced
394    Closed,
395}
396
397impl core::fmt::Debug for TrafOut<'_> {
398    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
399        f.debug_struct("TrafOut").field("state", &self.state).finish_non_exhaustive()
400    }
401}

412impl<'a> TrafOut<'a> {
413    pub fn new(buf: &'a mut [u8]) -> Self {
414        Self {
415            buf: SliceOrVec::Borrowed(buf),
416            state: TxState::Idle,
417            drain: false,
418            sending_kex: false,
419            deferred_packets: Deque::new(),
420        }
421    }
422
423    /// Serializes and and encrypts a packet to send
424    ///
425    /// If the output buffer is full or a rekey is in progress, the
426    /// packet will be enqueued to be sent later. If the deferred packet queue
427    /// is full, `NoRoom` will be returned.
428    ///
429    /// `BusySend` error is recoverable, others should be treated as fatal.
430    pub(crate) fn send_packet(
431        &mut self,
432        p: packets::Packet,
433        keys: &mut KeyState,
434    ) -> Result<()> {
435        let is_kex = matches!(p.category(), packets::Category::Kex);
436
437        if is_kex || (self.deferred_packets.is_empty() && !self.sending_kex) {
438            // Send the packet normally if it fits.
439            // KEX packets can be sent even if other packets are deferred
440            // (KEX packets don't get deferred themselves).
441            // A KexInit is only sent when there are no deferred packets,
442            // so we don't need to worry about incorrect reordering.
443            match self.send_packet_inner(&p, keys) {
444                Err(Error::NoRoom { .. }) => {
445                    debug_assert!(!is_kex, "KEX packets should have room");
446                    // non-kex packets get deferred
447                }
448                res => return res,
449            }
450        }
451
452        // Either it didn't fit (NoRoom), or the deferred queue
453        // is already in use so we need to enqueue after that.
454        // Attempt to defer the packet.
455
456        let pnum = p.message_num();
457        trace!("Delay packet type {pnum:?}");
458
459        // Convert to a DeferredPacket if possible
460        let Ok(dp) = DeferredPacket::try_from(p) else {
461            // Packet type isn't expected to be deferred.
462            trace!("NoRoom packet type {pnum:?}");
463            return error::BusySend { packet: pnum, unsupported: true }.fail();
464        };
465
466        self.deferred_packets.push_front(dp).map_err(|_| {
467            error!("No space to queue packet");
468            trace!("NoRoom packet type {pnum:?}");
469            error::BusySend { packet: pnum, unsupported: false }.build()
470        })
471    }
472
473    // Check some invariants, and track whether we're sending KEX.
474    fn track_send_packet(
475        &mut self,
476        p: &packets::Packet,
477        keys: &mut KeyState,
478    ) -> Result<()> {
479        // Check that packets are being encrypted
480        // This is checked in release and debug.
481        match p.category() {
482            packets::Category::All | packets::Category::Kex => (),
483            _ => {
484                if keys.is_send_cleartext() {
485                    return Error::bug_msg("send cleartext");
486                }
487            }
488        }
489
490        // KEX send packet catetory checked in debug builds for fuzzing.
491        if self.sending_kex {
492            // strict kex is ignored since we don't have access to
493            // conn.kex.
494            debug_assert!(matches!(
495                p.category(),
496                packets::Category::All | packets::Category::Kex
497            ));
498        }
499
500        // Track KEX sending state
501        match p {
502            Packet::KexInit(_) => {
503                debug_assert!(!self.sending_kex);
504                self.sending_kex = true;
505            }
506            Packet::NewKeys(_) => {
507                debug_assert!(self.sending_kex);
508                self.sending_kex = false;
509            }
510            _ => (),
511        }
512
513        Ok(())
514    }
515
516    /// Serializes and and encrypts a packet to send
517    ///
518    /// The packet will not be enqueued to the deferred queue.
519    /// This function should not usually be called directly.
520    ///
521    /// `NoRoom` error is recoverable, others should be treated as fatal.
522    pub fn send_packet_inner(
523        &mut self,
524        p: &packets::Packet,
525        keys: &mut KeyState,
526    ) -> Result<()> {
527        self.track_send_packet(p, keys)?;
528
529        // Either a fresh buffer or appending to write
530        let (idx, len) = match self.state {
531            TxState::Idle => (0, 0),
532            TxState::Write { idx, len } => (idx, len),
533            TxState::Closed => {
534                trace!("Dropped output after close {p:?}");
535                return Ok(());
536            }
537        };
538
539        // Use the remainder of our buffer to write the packet. Payload starts
540        // after the length and padding bytes which get filled by encrypt()
541        let wbuf = &mut self.buf[len..];
542        if wbuf.len() < SSH_PAYLOAD_START {
543            return error::NoRoom.fail();
544        }
545        let plen = sshwire::write_ssh(&mut wbuf[SSH_PAYLOAD_START..], &p)?;
546        trace!("Sending {p:?}");
547
548        // Encrypt in place
549        let elen = keys.encrypt(plen, wbuf)?;
550        self.state = TxState::Write { idx, len: len + elen };
551        Ok(())
552    }
553
554    pub fn send_deferred_packets(&mut self, keys: &mut KeyState) -> Result<()> {
555        while let Some(d) = self.deferred_packets.back() {
556            let p = Packet::from(d);
557            match self.send_packet_inner(&p, keys) {
558                Ok(()) => {
559                    self.deferred_packets.pop_back();
560                }
561                Err(Error::NoRoom { .. }) => {
562                    // Can't progress, let the caller retry later
563                    break;
564                }
565                Err(e) => return Err(e),
566            }
567        }
568
569        Ok(())
570    }
571
572    pub fn have_deferred_packets(&self) -> bool {
573        !self.deferred_packets.is_empty()
574    }
575
576    pub fn is_output_pending(&self) -> bool {
577        trace!("is_output_pending st {:?}", self.state);
578        matches!(self.state, TxState::Write { .. })
579    }
580
581    /// Returns payload space available to send a packet. Returns 0 if not ready or full
582    pub fn send_allowed(&self, keys: &KeyState) -> usize {
583        if !self.deferred_packets.is_empty() {
584            // Don't allow sending packets when deferred ones are waiting.
585            // Otherwise the deferred queue will run out of room.
586            return 0;
587        }
588
589        // TODO: test for full output buffer
590        match self.state {
591            TxState::Write { len, .. } => keys.max_enc_payload(self.buf.len() - len),
592            TxState::Idle => keys.max_enc_payload(self.buf.len()),
593            // output will just be dropped in closed state.
594            TxState::Closed => self.buf.len(),
595        }
596    }
597
598    /// Move to Closed state. Current output is lost, future output
599    /// is discarded. This is called when the output tcp pipe
600    /// has closed so there's nowhere to send output anyway.
601    pub fn close(&mut self) {
602        self.state = TxState::Closed
603    }
604
605    pub fn closed(&self) -> bool {
606        matches!(self.state, TxState::Closed)
607    }
608
609    pub fn send_version(&mut self) -> Result<(), Error> {
610        if !matches!(self.state, TxState::Idle) {
611            return Error::bug();
612        }
613
614        let len = ident::write_version(&mut self.buf)?;
615        self.state = TxState::Write { idx: 0, len };
616        Ok(())
617    }
618
619    pub fn output_buf(&mut self) -> &[u8] {
620        match self.state {
621            TxState::Write { ref mut idx, len } => {
622                let wlen = len - *idx;
623                &self.buf[*idx..*idx + wlen]
624            }
625            _ => &[],
626        }
627    }
628
629    pub fn consume_output(&mut self, l: usize) {
630        if let TxState::Write { ref mut idx, len } = self.state {
631            let wlen = (len - *idx).min(l);
632            *idx += wlen;
633
634            if *idx == len {
635                // all done, read the next packet
636                self.state = TxState::Idle
637            }
638        }
639    }

641    pub fn sender<'s>(&'s mut self, keys: &'s mut KeyState) -> TrafSend<'s, 'a> {
642        TrafSend::new(self, keys)
643    }

645    /// Return whether output is draining.
646    ///
647    /// Used to determine whether to initiate outbound traffic, such as channel writes.
648    /// Generally immediate responses to incoming messages should still be sent
649    /// even when draining. Otherwise they would need to be put in deferred_packets
650    /// which may run out.
651    pub fn is_draining(&self) -> bool {
652        self.drain
653    }

708    /// For `TrafOut` itself there isn't a problem with multiple
709    /// callers enabling/disabling drain, but it could result in races
710    /// between callers. Only kex should be using it currently, so
711    /// there is a debug_assert! to that effect.
712    pub fn set_drain_output(&mut self, drain: bool) {
713        debug_assert!(drain != self.out.drain, "set_drain_output() dupe");
714        self.out.drain = drain;
715    }
716
717    /// Test if output buffer is empty.
718    ///
719    /// Fails if `set_drain_output(true)` wasn't set (debug panic)
720    /// This isn't inherent, but helps catch misuse (see comment
721    /// for set_drain_output()).
722    pub fn is_drained(&self) -> bool {
723        debug_assert!(self.out.drain);
724        matches!(self.out.state, TxState::Idle)
725            && self.out.deferred_packets.is_empty()
726    }
727}
728
729/// Packet types that may be sent once a currently-NoRoom TrafOut clears.
730///
731/// Rather than storing an entire `Packet<'static>`, keep a queue of smaller
732/// `DeferredPacket`s.
733///
734/// These packet types account for most packets that may be sent as responses
735/// or from other asynchronous events (rekey packet count reached, as an example).
736///
737/// These also queue packets to be sent while a KEX is in progress
738/// (other packet types aren't allowed)
739///
740/// Some packet types aren't included here since they're deferred via other
741/// mechanisms:
742///
743/// - ChannelWindowAdjust. Can be retried later.
744/// - KEX packets. The output buffer is drained at the start of a KEX.
745/// - Userauth - we hope it only occurs early when traffic is
746///   well defined (no channels) and no KEXes are happening.
747#[derive(Debug)]
748pub enum DeferredPacket {
749    ChannelSuccess(packets::ChannelSuccess),
750    ChannelFailure(packets::ChannelFailure),
751    ChannelOpenFailure(packets::ChannelOpenFailure<'static>),
752    ChannelOpenConfirmation(packets::ChannelOpenConfirmation),
753    ChannelEof(packets::ChannelEof),
754    ChannelClose(packets::ChannelClose),
755    Unimplemented(packets::Unimplemented),
756    RequestFailure(packets::RequestFailure),
757    RequestSuccess(packets::RequestSuccess),
758}
759
760impl DeferredPacket {}
761
762impl From<&DeferredPacket> for Packet<'static> {
763    fn from(d: &DeferredPacket) -> Self {
764        match d {
765            DeferredPacket::ChannelSuccess(p) => (p.clone()).into(),
766            DeferredPacket::ChannelFailure(p) => (p.clone()).into(),
767            DeferredPacket::ChannelOpenFailure(p) => (p.clone()).into(),
768            DeferredPacket::ChannelOpenConfirmation(p) => (p.clone()).into(),
769            DeferredPacket::ChannelEof(p) => (p.clone()).into(),
770            DeferredPacket::ChannelClose(p) => (p.clone()).into(),
771            DeferredPacket::Unimplemented(p) => (p.clone()).into(),
772            DeferredPacket::RequestFailure(p) => (p.clone()).into(),
773            DeferredPacket::RequestSuccess(p) => (p.clone()).into(),
774        }
775    }
776}
777
778impl<'a> TryFrom<Packet<'a>> for DeferredPacket {
779    type Error = Error;
780    fn try_from(packet: Packet<'a>) -> Result<Self> {
781        Ok(match packet {
782            Packet::ChannelSuccess(p) => Self::ChannelSuccess(p),
783            Packet::ChannelFailure(p) => Self::ChannelFailure(p),
784            Packet::ChannelOpenConfirmation(p) => Self::ChannelOpenConfirmation(p),
785            Packet::ChannelEof(p) => Self::ChannelEof(p),
786            Packet::ChannelClose(p) => Self::ChannelClose(p),
787            Packet::Unimplemented(p) => Self::Unimplemented(p),
788            Packet::RequestFailure(p) => Self::RequestFailure(p),
789            Packet::RequestSuccess(p) => Self::RequestSuccess(p),
790
791            Packet::ChannelOpenFailure(p) => {
792                Self::ChannelOpenFailure(packets::ChannelOpenFailure {
793                    // empty desc for 'static
794                    desc: TextString::new(),
795                    lang: "",
796                    ..p
797                })
798            }
799
800            // Unhandled types
801            _ => return error::SSHProtoUnsupported.fail(),
802        })
803    }
804}
