21// Runner public methods take a `ChanHandle` which cannot be cloned. This prevents
22// confusion if an application were to continue using a channel after the channel
23// was completed. The `ChanHandle` is consumed by `Runner::channel_done()`.
24// Internally sunset uses `ChanNum`, which is just a newtype around u32.
25
26/// A SSH session instance
27///
28/// An application provides network or channel data to `Runner` method calls,
29/// and provides customisation callbacks via `CliBehaviour` or `ServBehaviour`.
30pub struct Runner<'a, CS: conn::CliServ> {
31    conn: Conn<CS>,
32
33    /// Binary packet handling from the network buffer
34    traf_in: TrafIn<'a>,
35    /// Binary packet handling to the network buffer
36    traf_out: TrafOut<'a>,
37
38    /// Current encryption/integrity keys
39    keys: KeyState,
40
41    /// Waker when output is ready
42    output_waker: Option<Waker>,
43    /// Waker when ready to consume input.
44    input_waker: Option<Waker>,
45
46    closed_input: bool,
47
48    resume_event: DispatchEvent,
49    // Some incoming packets will produce multiple Events from a single packet.
50    // (such as Userauth, where we query application for a pubkey or password).
51    // The Event handler can set extra_resume_event which will cause that
52    // event to be emitted on the next .progress() call.
53    extra_resume_event: DispatchEvent,
54    // Cellos patch 0004: whether `resume_event` was produced from the current input
55    // payload. An event from `conn.progress()` (for example `Authenticated` or
56    // `SessionOpened`) has none; completing "its" payload on the next `progress()`
57    // would discard an unrelated packet that arrived in between.
58    resume_from_payload: bool,
59}

306    /// Drives connection progress, handling received payload and queueing
307    /// packets to send as required.
308    pub fn progress(&mut self) -> Result<Event<'_, 'a>> {
309        // Any previous Event must have been dropped to be able to call progress()
310        // again, since it borrows from Runner. We can check if it was dropped
311        // without a required response, or complete the payload handling otherwise.
312        let prev = self.resume_event.take();
313        if prev.needs_resume() {
314            // Events that need a response would have cleared runner.resume_event in their
315            // resume handler.
316            debug!("No response provided to {:?} event", prev);
317            return error::BadUsage.fail();
318        }
319
320        // Another event may be pending from the same payload, emit it.
321        let ex = self.extra_resume_event.take();
322        if ex.is_some() {
323            // Cellos patch 0004: extra events are set by resume handlers, which have
324            // already completed their payload.
325            self.resume_from_payload = false;
326            self.resume_event = ex.clone();
327            return CS::dispatch_into_event(self, ex);
328        }
329
330        // Previous event payload is complete (Cellos patch 0004: only if the event
331        // came from a payload).
332        if prev.is_event() && self.resume_from_payload {
333            self.traf_in.done_payload();
334        }
335        self.resume_from_payload = false;
336
337        // Try moving packets from the deferred queue to the
338        // normal output. Can't happen during kex since non-kex
339        // packets are disallowed.
340        // KEX packets aren't included in the deferred list.
341        if !self.conn.is_kex_sending() {
342            match self.traf_out.send_deferred_packets(&mut self.keys) {
343                Ok(()) => (),
344                Err(Error::NoRoom { .. }) => {
345                    // try again once there's space
346                    return Ok(Event::None);
347                }
348                Err(e) => return Err(e),
349            }
350        }
351
352        let mut disp = Dispatched::default();
353
354        // Handle incoming packets
355        if let Some((payload, seq)) = self.traf_in.payload() {
356            let mut s = self.traf_out.sender(&mut self.keys);
357            disp = self.conn.handle_payload(payload, seq, &mut s)?;
358
359            match disp.event {
360                DispatchEvent::Data(data_in) => {
361                    // incoming channel data, we haven't finished with payload
362                    let (num, dt) = self.traf_in.set_read_channel_data(data_in)?;
363                    self.channel_wake_read(num, dt);
364                    disp.event = DispatchEvent::None
365                }
366                DispatchEvent::CliEvent(_) | DispatchEvent::ServEvent(_) => {
367                    // will return as an event
368                    self.resume_from_payload = true;
369                }
370                DispatchEvent::None => {
371                    // packets have been completed
372                    self.traf_in.done_payload()
373                }
374                DispatchEvent::KexDone => {
375                    // Wake any channels that were paused during KEX
376                    self.channel_wake_write();
377                    self.traf_in.done_payload();
378                    disp.event = DispatchEvent::None;
379                }
380                // TODO, may get used later?
381                DispatchEvent::Progressed => return Error::bug(),
382            }
383        } else if self.closed_input {
384            // all incoming packets have been consumed, and we're closed for input,
385            if CS::is_client() {
386                return Ok(Event::Cli(CliEvent::Defunct));
387            } else {
388                return Ok(Event::Serv(ServEvent::Defunct));
389            }
390        }
391
392        // If there isn't any pending event for the application, run conn.progress()
393        // (which may return other events).
394        if disp.event.is_none() {
395            let mut s = self.traf_out.sender(&mut self.keys);
396            disp = self.conn.progress(&mut s)?;
397            trace!("prog disp {disp:?}");
398            match disp.event {
399                DispatchEvent::CliEvent(_)
400                | DispatchEvent::ServEvent(_)
401                | DispatchEvent::None
402                | DispatchEvent::Progressed => (),
403                // Don't expect data from conn.progress()
404                DispatchEvent::Data(_) | DispatchEvent::KexDone => {
405                    return Error::bug();
406                }
407            }
408        }
409
410        self.wake();
411
412        // Record the event for later checks
413        debug_assert!(self.resume_event.is_none());
414        self.resume_event = disp.event.clone();
415
416        // Create an Event that borrows from Runner
417        CS::dispatch_into_event(self, disp.event)
418    }

428    // Accept bytes from the wire, returning the size consumed
429    pub fn input(&mut self, buf: &[u8]) -> Result<usize, Error> {
430        if self.closed_input {
431            return error::SessionEOF.fail();
432        }
433        if !self.is_input_ready() {
434            return Ok(0);
435        }
436        self.traf_in.input(&mut self.keys, &mut self.conn.remote_version, buf)
437    }

439    // Whether [`input()`](input) is ready
440    pub fn is_input_ready(&self) -> bool {
441        if self.closed_input {
442            return true;
443        }
444
445        // During KEX, let the output queue drain before accepting packets,
446        // so we can be sure not to hit NoRoom.
447        // kexdhinit/kexdhreply packets can be large.
448        // The deferred queue may still have packets since they
449        // can't be send during a KEX.
450        if self.conn.is_kex_sending() && self.traf_out.is_output_pending() {
451            return false;
452        }
453
454        self.conn.initial_sent() && self.traf_in.is_input_ready()
455    }

491        if !self.traf_out.is_output_pending() {

518    /// Send data from this application out the wire.
519    ///
520    /// Returns `Ok(len)` consumed, `Err(Error::ChannelEof)` on EOF,
521    /// or other errors.
522    pub fn write_channel(
523        &mut self,
524        chan: &ChanHandle,
525        dt: ChanData,
526        buf: &[u8],
527    ) -> Result<usize> {
528        if self.traf_out.closed() {
529            // TODO: unsure if we need this
530            return error::ChannelEOF.fail();
531        }
532
533        if buf.is_empty() {
534            return Ok(0);
535        }
536
537        let len = self.write_channel_ready(chan, dt)?;
538        let len = match len {
539            Some(0) => return Ok(0),
540            Some(l) => l,
541            None => return Err(Error::ChannelEOF),
542        };
543
544        let len = len.min(buf.len());
545
546        let p = self.conn.channels.send_data(chan.0, dt, &buf[..len])?;
547        trace!("send_packet ch {:?} dt {:?} {}", chan.0, dt, len);
548        self.traf_out.send_packet(p, &mut self.keys)?;
549        self.wake();
550        Ok(len)
551    }

553    /// Receive data coming from the wire into this application.
554    ///
555    /// Returns `Ok(len)` received, `Err(Error::ChannelEof)` on EOF,
556    /// or other errors. Ok(0) indicates no data available, ie pending.
557    /// TODO: EOF is unimplemented
558    pub fn read_channel(
559        &mut self,
560        chan: &ChanHandle,
561        dt: ChanData,
562        buf: &mut [u8],
563    ) -> Result<usize> {
564        if self.closed_input {
565            return error::ChannelEOF.fail();
566        }
567
568        dt.validate_receive(CS::is_client())?;
569
570        if self.is_channel_eof(chan) {
571            return error::ChannelEOF.fail();
572        }
573
574        let (len, complete) = self.traf_in.read_channel(chan.0, dt, buf);
575        if let Some(x) = complete {
576            self.finished_read_channel(chan, x)?;
577        }
578        Ok(len)
579    }

581    /// Receives input data, either normal or extended.
582    pub fn read_channel_either(
583        &mut self,
584        chan: &ChanHandle,
585        buf: &mut [u8],
586    ) -> Result<(usize, ChanData)> {
587        let (len, complete, dt) = self.traf_in.read_channel_either(chan.0, buf);
588        if let Some(x) = complete {
589            self.finished_read_channel(chan, x)?;
590        }
591        Ok((len, dt))

648    /// Returns the maximum data that may be sent to a channel
649    ///
650    /// Returns `Ok(None)` on channel closed.
651    ///
652    /// May fail with `BadChannelData` if dt is invalid for this session.
653    pub fn write_channel_ready(
654        &self,
655        chan: &ChanHandle,
656        dt: ChanData,
657    ) -> Result<Option<usize>> {
658        if self.traf_out.closed() {
659            return Ok(None);
660        }
661        // Avoid apps polling forever on a packet type that won't come
662        dt.validate_send(CS::is_client())?;
663
664        // When write_channel_ready() returns Some(0), a subsequent
665        // channel_wake_write() needs to occur.
666
667        // channel_wake_write() after KexDone.
668        if self.conn.is_kex_sending() {
669            // Only KEX messages are allowed during key exchange,
670            // not data.
671            return Ok(Some(0));
672        }
673
674        // Cellos patch 0005: once a rekey is due, application payload waits for it (the
675        // next progress() starts the KEX), so the thresholds hold under a busy writer.
676        if self.keys.is_rekey_needed() {
677            return Ok(Some(0));
678        }
679        // Drain only happens prior to KEX, so can be woken after KexDone.
680        if self.traf_out.is_draining() {
681            // Continual channel data could prevent drain from completing,
682            // so disallow channel writes when draining.
683            return Ok(Some(0));
684        }
685
686        // write_channel_ready() will happen after the deferred packets are
687        // moved to main queued then written.
688        if self.traf_out.have_deferred_packets() {
689            // Let deferred packets get moved to the output queue
690            // before sending more channel data.
691            return Ok(Some(0));
692        }
693
694        // minimum of buffer space and channel window available
695        let payload_space = self.traf_out.send_allowed(&self.keys);
696        // subtract space for packet headers prior to data
697        let payload_space = payload_space.saturating_sub(dt.packet_offset());
698        let r = Ok(self
699            .conn
700            .channels
701            .send_allowed(chan.0)
702            .map(|s| s.min(payload_space)));
703        trace!("ready_channel_send {chan:?} -> {r:?}");
704        r
705    }

715    /// Must be called when an application has finished with a channel.
716    ///
717    /// Channel numbers will not be re-used without calling this, so
718    /// failing to call this may result in running out of channels.
719    pub fn channel_done(&mut self, chan: ChanHandle) -> Result<()> {
720        self.conn.channels.done(chan.0)?;

723        self.wake();
724        Ok(())
725    }

778    // Wake SSH TCP socket input and output as required.
779    // Channel wakes happen elsewhere.
780    fn wake(&mut self) {
781        trace!("wake");
782        if self.is_input_ready() {
783            trace!("wake ready_input, waker {:?}", self.input_waker);
784            if let Some(w) = self.input_waker.take() {
785                trace!("wake input waker");
786                w.wake()
787            }
788        } else {
789            trace!("no input ready");
790        }
791
792        if self.is_output_pending() {
793            if let Some(w) = self.output_waker.take() {
794                trace!("wake output waker");
795                w.wake()
796            } else {
797                trace!("no waker");
798            }
799        } else {
800            trace!("no output pending")
801        }
802    }

804    fn check_resume_inner(&self, expect: &DispatchEvent, compare: &DispatchEvent) {
805        match (expect, compare) {
806            (DispatchEvent::CliEvent(e), DispatchEvent::CliEvent(c)) => {
807                debug_assert_eq!(
808                    discriminant(c),
809                    discriminant(e),
810                    "Expected response to pending {expect:?} event"
811                )
812            }
813            (DispatchEvent::ServEvent(e), DispatchEvent::ServEvent(c)) => {
814                debug_assert_eq!(
815                    discriminant(c),
816                    discriminant(e),
817                    "Expected response to pending {expect:?} event"
818                )
819            }
820            _ => debug_assert!(false),
821        }
822    }

953/// Sets a waker, waking any existing waker
954pub(crate) fn set_waker(store_waker: &mut Option<Waker>, new_waker: &Waker) {
955    if let Some(w) = store_waker {
956        if w.will_wake(new_waker) {
957            // Avoid churn and clone() overhead if they both wake the same task
958            return;
959        }
960    }
961
962    if let Some(w) = store_waker.take() {
963        w.wake()
964    }
965    *store_waker = Some(new_waker.clone())
966}

968/// Represents an open channel, owned by the application.
969///
970/// Must be released by calling [`Runner::channel_done()`]
971
972// Inner contents are crate-private to ensure that arbitrary
973// channel numbers cannot be used after closing/reuse.
974//
975// This must not be `Clone`
976#[derive(PartialEq, Eq, Hash, Ord, PartialOrd)]
977pub struct ChanHandle(pub(crate) ChanNum);
978
979impl ChanHandle {
980    /// Returns the channel number
981    ///
982    /// This can be used by applications as an index.
983    /// Channel numbers satisfy
984    /// `0 <= num < sunset::config::MAX_CHANNELS`.
985    ///
986    /// An index may be reused after a call to [`Runner::channel_done()`],
987    /// applications must take care not to keep using this `num()` index after
988    /// that.
989    pub fn num(&self) -> ChanNum {
990        self.0
991    }
992}
