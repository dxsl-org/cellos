16/// The Sunset error type.
17#[non_exhaustive]
18#[derive(Snafu, Debug)]
19#[snafu(context(suffix(false)))]
20// TODO: maybe split this into a list of public vs private errors?
21#[snafu(visibility(pub))]
22pub enum Error {
23    /// Input buffer ran out
24    RanOut {
25        #[cfg(feature = "backtrace")]
26        backtrace: Backtrace,
27    },
28
29    /// Can't currently send a packet
30    ///
31    /// Either output buffer is full, or a key exchange is in progress.
32    /// `unsupported` is set if it is not expected for Sunset to be
33    /// able to defer that type of packet (it may indicate a bug in Sunset,
34    //// or traffic that is unanticipated or difficult to handle).
35    // Should only be returned from TrafOut::send_packet(),
36    // NoRoom is for other similar circumstances.
37    BusySend { packet: MessageNumber, unsupported: bool },
38
39    /// No room to write
40    NoRoom {
41        #[cfg(feature = "backtrace")]
42        backtrace: Backtrace,
43    },
44
45    /// Not a UTF-8 string
46    BadString,
47
48    /// Not a valid SSH ASCII string
49    BadName,
50
51    /// Key exchange incorrect
52    BadKex,
53
54    /// Packet integrity failed
55    BadDecrypt,
56
57    /// Signature is incorrect
58    BadSig,
59
60    /// Received a badly formatted number
61    BadNumber,
62
63    /// Error in received SSH protocol. Will disconnect.
64    SSHProto {
65        #[cfg(feature = "backtrace")]
66        backtrace: Backtrace,
67    },
68
69    /// Peer sent something we don't handle. Will disconnect.
70    ///
71    /// This differs to `SSHProtoError`. In this case the peer may be
72    /// behaved within the SSH specifications, but Sunset doesn't
73    /// support it.
74    // TODO: 'static disconnect message to return?
75    SSHProtoUnsupported,
76
77    /// Remote peer isn't SSH 2.0
78    NotSSH,
79
80    /// Bad key format
81    BadKey,
82
83    /// Ran out of channels
84    NoChannels,
85
86    #[snafu(display("Bad channel number {num}"))]
87    BadChannel { num: ChanNum },
88
89    /// Bad channel data type
90    ///
91    /// Returned from an API call that would imply ChanData::Stderr
92    /// being sent to a server. This error will not be returned for
93    /// network data in the incorrect direction, instead that data is dropped.
94    BadChannelData,
95
96    /// Bad application usage
97    ///
98    /// Returned from an API call when the API is used incorrectly.
99    /// Examples could include:
100    /// - A `ChanHandle` is used incorrectly, for example being cloned
101    ///   (millions of times) and not released.
102    // TODO: /// #[snafu(display("Failure from application: {msg}"))]
103    BadUsage {
104        #[cfg(feature = "backtrace")]
105        backtrace: snafu::Backtrace,
106        // TODO
107        // msg: &'static str,
108    },
109
110    /// SSH packet contents doesn't match length
111    WrongPacketLength,
112
113    /// Channel EOF
114    ///
115    /// This is an expected error when a SSH channel completes. Can be returned
116    /// by channel read/write functions. Any further calls in the same direction
117    /// will fail similarly.
118    ChannelEOF,
119
120    /// Session EOF
121    ///
122    /// This is an expected error when the SSH session has finished.
123    SessionEOF,
124
125    // Used for unknown key types etc.
126    #[snafu(display("{what} is not available"))]
127    NotAvailable { what: &'static str },
128
129    #[snafu(display("Unknown packet type {number}"))]
130    UnknownPacket { number: u8 },
131
132    /// Received packet at a disallowed time.
133    // TODO: this is kind of a subset of SSHProtoError, maybe not needed
134    PacketWrong {
135        #[cfg(feature = "backtrace")]
136        backtrace: Backtrace,
137    },
138    // #[snafu(display("Program bug {location}"))]
139    // Bug { location: snafu::Location },
140    #[snafu(display("No matching {algo} algorithm"))]
141    AlgoNoMatch { algo: &'static str },
142
143    #[snafu(display("Packet size {size} too large (or bad decrypt)"))]
144    BigPacket { size: usize },
145
146    /// Ran out of authentication methods
147    ///
148    /// This is only for the client.
149    NoAuthMethods,
150
151    /// An unknown SSH name is provided, for a key type, signature type,
152    /// channel name etc.
153    #[snafu(display("Unknown {kind} method"))]
154    UnknownMethod { kind: &'static str },
155
156    #[snafu(display("{msg}"))]
157    // TODO: these could eventually get categorised
158    Custom { msg: &'static str },
159
160    /// std IO error
161    #[cfg(feature = "std")]
162    IoError { source: std::io::Error },
163
164    /// `embedded-io` error
165    #[cfg(feature = "embedded-io")]
166    EmbeddedIoError { kind: embedded_io::ErrorKind },
167
168    // Error::Bug should not be reached, previous logic should have prevented it.
169    // Create this using [`Error::bug()`] or [`.trap()`](TrapBug::trap).
170    // Location is currently disabled due to bloat of full paths.
171    // https://github.com/rust-lang/rust/issues/95529 is having functions.
172    // #[snafu(display("Program bug {location}"))]
173    // Bug { location: snafu::Location },
174    /// Program bug
175    ///
176    /// Hitting this may instead panic on builds with debug-assertions.
177    Bug,
178}

180impl Error {
181    pub fn msg(m: &'static str) -> Error {
182        Error::Custom { msg: m }
183    }
184
185    #[cold]
186    #[track_caller]
187    /// Panics in debug builds, returns [`Result::Err(Error::Bug)`](Error::Bug) in release.
188    pub fn bug<T>() -> Result<T, Error> {
189        Err(Self::build_bug())
190    }
191
192    #[cold]
193    #[track_caller]
194    /// Panics in debug builds, returns [`Error::Bug`] in release.
195    pub fn build_bug() -> Error {
196        // Easier to track the source of errors in development,
197        // but release builds shouldn't panic.
198        if cfg!(debug_assertions) {
199            panic!("Hit a bug");
200        } else {
201            Error::Bug
202        }
203    }
204
205    /// Like [`bug()`](Error::bug) but with a message
206    ///
207    /// The message can be used instead of a code comment, is logged at `trace` level.
208    #[cold]
209    fn bug_fmt(args: Arguments) -> Error {
210        // Easier to track the source of errors in development,
211        // but release builds shouldn't panic.
212        if cfg!(debug_assertions) {
213            panic!("Hit a bug: {args}");
214        } else {
215            trace!("Hit a bug: {args}");
216            // let caller = core::panic::Location::caller();
217            Error::Bug
218            // {
219            //     location: snafu::Location::new(
220            //         caller.file(),
221            //         caller.line(),
222            //         caller.column(),
223            //     ),
224            // }
225        }
226    }
227
228    #[cold]
229    pub fn bug_msg<T>(msg: &str) -> Result<T, Error> {
230        Err(Self::bug_fmt(format_args!("{}", msg)))
231    }
232
233    #[cold]
234    pub fn build_bug_msg(msg: &str) -> Error {
235        Self::bug_fmt(format_args!("{}", msg))
236    }
237}

254pub type Result<T, E = Error> = core::result::Result<T, E>;

256pub trait TrapBug<T> {
257    /// `.trap()` should be used like `.unwrap()`, in situations
258    /// never expected to fail. Instead it calls [`Error::bug()`].
259    /// (or debug builds may panic)
260    fn trap(self) -> Result<T, Error>;
261
262    /// Like `trap()` but with a message, calls [`Error::bug_msg()`]
263    /// The message can be used instead of a comment.
264    fn trap_msg(self, args: Arguments) -> Result<T, Error>;
265}
266
267impl<T, E> TrapBug<T> for Result<T, E> {
268    #[track_caller]
269    fn trap(self) -> Result<T, Error> {
270        // call directly so that Location::caller() works
271        if let Ok(i) = self { Ok(i) } else { Error::bug() }
272    }
273    #[track_caller]
274    fn trap_msg(self, args: Arguments) -> Result<T, Error> {
275        // call directly so that Location::caller() works
276        if let Ok(i) = self { Ok(i) } else { Err(Error::bug_fmt(args)) }
277    }
278}
279
280impl<T> TrapBug<T> for Option<T> {
281    #[track_caller]

290    }
