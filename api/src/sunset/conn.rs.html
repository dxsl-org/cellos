53#[derive(Debug)]
54enum ConnState {
55    /// The initial state
56    SendIdent,
57    /// Prior to SSH binary packet protocol, receiving remote version identification
58    ReceiveIdent,
59    /// Waiting for first Kex to complete
60    FirstKex,
61    /// Binary protocol has started, auth hasn't succeeded
62    PreAuth,
63    /// After auth success
64    Authed,
65    // Cleanup ??
66}

68// must_use so return values can't be forgotten in Conn::dispatch_packet
69#[must_use]
70#[derive(Debug, Clone, Default)]
71pub(crate) enum DispatchEvent {
72    /// Incoming channel data
73    Data(channel::DataIn),
74    CliEvent(CliEventId),
75    ServEvent(ServEventId),
76    /// NewKeys was received, wake any output channels in case they were waiting.
77    KexDone,
78    /// Connection state has changed, should poll again
79    Progressed,
80    /// No event
81    #[default]
82    None,
83}

85impl DispatchEvent {
86    pub fn take(&mut self) -> Self {
87        core::mem::replace(self, DispatchEvent::None)
88    }
89
90    pub fn is_some(&self) -> bool {
91        !self.is_none()
92    }
93
94    pub fn is_none(&self) -> bool {
95        matches!(self, Self::None)
96    }

98    /// Used by Runner to determine whether an event requires a resume call before
99    /// continuing. Informational events don't.
100    /// Some events don't need calling manually, but their Drop impl will
101    /// call the appropriate resume method.
102    pub(crate) fn needs_resume(&self) -> bool {
103        match self {
104            Self::None | Self::Data(_) | Self::KexDone | Self::Progressed => false,
105            Self::CliEvent(x) => x.needs_resume(),
106            Self::ServEvent(x) => x.needs_resume(),
107        }
108    }
109
110    pub(crate) fn is_event(&self) -> bool {
111        matches!(self, Self::CliEvent(_) | Self::ServEvent(_))
112    }
113}

271    /// Updates `ConnState` and sends any packets required to progress the connection state.
272    pub(crate) fn progress(
273        &mut self,
274        s: &mut TrafSend,
275    ) -> Result<Dispatched, Error> {
276        self.kex.progress(&self.algo_conf, s)?;
277
278        if !self.is_kex_sending() {
279            let event = self.channels.progress(s);
280            if !event.is_none() {
281                // TODO better Dispatched constructor
282                return Ok(Dispatched { event, disconnect: false });
283            }
284        }
285
286        let mut disp = Dispatched::default();
287        match self.state {
288            ConnState::SendIdent => {
289                s.send_version()?;
290                // send early to avoid round trip latency
291                // TODO: first_follows would have a second packet here
292                self.kex.start_kexinit(s);
293                disp.event = DispatchEvent::Progressed;
294                self.state = ConnState::ReceiveIdent
295            }
296            ConnState::ReceiveIdent => {
297                if self.remote_version.version().is_some() {
298                    // Ready to start binary packets. We've already send our KexInit with SendIdent.
299                    disp.event = DispatchEvent::Progressed;
300                    self.state = ConnState::FirstKex
301                }
302            }
303            ConnState::FirstKex => {
304                if self.sess_id.is_some() {
305                    disp.event = DispatchEvent::Progressed;
306                    self.state = ConnState::PreAuth
307                }
308            }
309            ConnState::PreAuth => {
310                if let Some(cli) = self.try_mut_client() {
311                    disp.event = cli.auth.progress();
312                }
313                // send userauth request
314            }
315            ConnState::Authed => {
316                // no events needed
317            }
318        }
319        trace!("-> {:?}, {disp:?}", self.state);
320
321        Ok(disp)
322    }

332    /// Consumes an input payload which is a view into [`traffic::Traffic::rxbuf`].
333    /// We queue response packets that can be sent (written into the same buffer)
334    /// after `handle_payload()` runs.
335    pub(crate) fn handle_payload(
336        &mut self,
337        payload: &[u8],
338        seq: u32,
339        s: &mut TrafSend,
340    ) -> Result<Dispatched, Error> {
341        // Parse the packet
342        match self.packet(payload) {
343            Ok(p) => {
344                let num = p.message_num() as u8;
345                let a = self.dispatch_packet(p, s);
346                match a {
347                    Err(Error::SSHProto { .. }) | Err(Error::PacketWrong { .. }) => {
348                        debug!("Error handling {num} packet")
349                    }
350                    _ => (),
351                }
352                a
353            }
354            Err(Error::UnknownPacket { number }) => {
355                trace!("Unimplemented packet type {number}");
356                s.send(packets::Unimplemented { seq })?;
357                Ok(Dispatched::default())
358            }
359            Err(e) => {
360                debug!("Error decoding packet: {e}");
361                trace!("Input:\n{:02x?}", payload);
362                Err(e)
363            }
364        }
365    }

367    /// Check that a packet is received in the correct state
368    fn check_packet(&self, p: &Packet) -> Result<()> {
369        let r = if self.is_first_kex() && self.kex.is_strict() {
370            // Strict Kex doesn't allow even packets like Ignore or Debug
371            match p.category() {
372                packets::Category::Kex => Ok(()),
373                _ => {
374                    debug!("Non-kex packet during strict kex");
375                    error::SSHProto.fail()
376                }
377            }
378        } else if self.kex.is_receiving() {
379            // Normal KEX only allows certain packets
380            match p.category() {
381                packets::Category::All => Ok(()),
382                packets::Category::Kex => Ok(()),
383                _ => {
384                    debug!("Invalid packet during kex");
385                    error::SSHProto.fail()
386                }
387            }
388        } else {
389            // No KEX in progress, check for post-auth packets
390            match p.category() {
391                packets::Category::All => Ok(()),
392                packets::Category::Kex => Ok(()),
393                packets::Category::Auth => match self.state {
394                    ConnState::PreAuth | ConnState::Authed => Ok(()),
395                    _ => error::SSHProto.fail(),
396                },
397                packets::Category::Sess => match self.state {
398                    ConnState::Authed => Ok(()),
399                    _ => error::SSHProto.fail(),
400                },
401            }
402        };
403
404        if r.is_err() {
405            debug!("Received unexpected packet {}", p.message_num() as u8);
406            trace!("state is {:?}", self.state);
407        }
408        r
409    }

430    pub fn dispatch_packet(
431        &mut self,
432        packet: Packet,
433        s: &mut TrafSend,
434    ) -> Result<Dispatched, Error> {
435        // TODO: perhaps could consolidate packet client vs server checks
436        trace!("Incoming {packet:#?}");
437        let mut disp = Dispatched::default();
438
439        self.check_packet(&packet)?;
440
441        match packet {
442            Packet::KexInit(k) => {
443                self.kex.handle_kexinit(
444                    k,
445                    &self.algo_conf,
446                    &self.remote_version,
447                    self.is_first_kex(),
448                    s,
449                )?;
450            }
451            Packet::KexDHInit(_p) => {
452                disp.event = self.kex.handle_kexdhinit()?;
453            }
454            Packet::KexDHReply(_p) => {
455                disp.event = self.kex.handle_kexdhreply()?;
456            }
457            Packet::NewKeys(_) => {
458                self.kex.handle_newkeys(&mut self.sess_id, s)?;
459                disp.event = DispatchEvent::KexDone;
460            }
461            Packet::ExtInfo(p) => {
462                if let Some(cli) = self.try_mut_client() {
463                    cli.auth.handle_ext_info(&p);
464                }
465                // could potentially pass it to other handlers too
466            }
467            Packet::ServiceRequest(p) => {
468                let Some(serv) = self.try_mut_server() else {
469                    debug!("Server sent a service request");
470                    return error::SSHProto.fail();
471                };
472                serv.service_request(&p, s)?;
473            }
474            Packet::ServiceAccept(p) => {
475                // Don't need to do anything, if a request failed the server disconnects
476                trace!("Received service accept {}", p.name);
477            }
478            Packet::Ignore(_) => {
479                // nothing to do
480            }
481            Packet::Unimplemented(_) => {
482                warn!("Received SSH unimplemented message");
483            }
484            Packet::DebugPacket(p) => {
485                let level = match p.always_display {
486                    true => log::Level::Info,
487                    false => log::Level::Debug,
488                };
489                log!(level, "SSH debug message from remote host: {}", p.message);
490            }
491            Packet::Disconnect(_p) => {
492                // We ignore p.reason.
493                // SSH2_DISCONNECT_BY_APPLICATION is normal, sent by openssh client.
494                disp.disconnect = true;
495            }
496            Packet::UserauthRequest(p) => {
497                let Some(serv) = self.cliserv.try_mut_server() else {
498                    debug!("Server sent an auth request");
499                    return error::SSHProto.fail();
500                };
501                let sess_id = self.sess_id.as_ref().trap()?;
502                disp.event = serv.auth.request(sess_id, s, p)?;
503            }
504            Packet::UserauthFailure(p) => {
505                let Some(cli) = self.cliserv.try_mut_client() else {
506                    debug!("Received UserauthFailure as a server");
507                    return error::SSHProto.fail();
508                };
509                disp.event = cli.auth.failure(&p, &mut self.parse_ctx)?;
510            }
511            Packet::UserauthSuccess(_) => {
512                let Some(cli) = self.cliserv.try_mut_client() else {
513                    debug!("Received UserauthSuccess as a server");
514                    return error::SSHProto.fail();
515                };
516                if matches!(self.state, ConnState::PreAuth) {
517                    self.state = ConnState::Authed;
518                    disp.event = cli.auth_success(&mut self.parse_ctx);
519                } else {
520                    debug!("Received UserauthSuccess unrequested")
521                }
522            }
523            Packet::UserauthBanner(_) => {
524                if self.is_server() {
525                    debug!("Received banner as a server");
526                    return error::SSHProto.fail();
527                }
528                disp.event = DispatchEvent::CliEvent(CliEventId::Banner);
529            }
530            Packet::Userauth60(p) => {
531                let Some(cli) = self.cliserv.try_mut_client() else {
532                    debug!("Received userauth60 as a server");
533                    return error::SSHProto.fail();
534                };
535                let sess_id = self.sess_id.as_ref().trap()?;
536                disp.event = cli.auth.auth60(&p, sess_id, &mut self.parse_ctx, s)?;
537            }
538            Packet::ChannelOpen(_)
539            | Packet::ChannelOpenConfirmation(_)
540            | Packet::ChannelOpenFailure(_)
541            | Packet::ChannelWindowAdjust(_)
542            | Packet::ChannelData(_)
543            | Packet::ChannelDataExt(_)
544            | Packet::ChannelEof(_)
545            | Packet::ChannelClose(_)
546            | Packet::ChannelRequest(_)
547            | Packet::ChannelSuccess(_)
548            | Packet::ChannelFailure(_) => {
549                disp.event = self.channels.dispatch(packet, s)?;
550            }
551            Packet::GlobalRequest(p) => {
552                trace!("Got global request {p:?}");
553                if p.want_reply {
554                    s.send(packets::RequestFailure {})?;
555                }
556            }
557            Packet::RequestSuccess(_p) => {
558                trace!("Got global request success")
559            }
560            Packet::RequestFailure(_) => {
561                trace!("Got global request failure")
562            }
563        };
564        Ok(disp)
565    }
566}

594

689    pub(crate) fn resume_servauth(
690        &mut self,
691        allow: bool,
692        s: &mut TrafSend,
693    ) -> Result<DispatchEvent> {
694        let auth = &mut self.mut_server()?.auth;
695        auth.resume_request(allow, s)?;
696        if auth.is_authed() && matches!(self.state, ConnState::PreAuth) {
697            self.state = ConnState::Authed;
698            Ok(DispatchEvent::ServEvent(ServEventId::Authenticated))
699        } else {
700            Ok(DispatchEvent::None)
701        }
702    }
