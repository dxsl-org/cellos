42// Cellos patch 0005: the rekey policy is per direction since the last NEWKEYS, in
43// encrypted payload bytes and packets (command-contracts §4.2), replacing upstream's
44// block-only RFC4344 thresholds. The embedder may lower it (tests) and adds the time
45// trigger through `Runner::request_rekey`, since this core has no clock.
46
47/// Rekey thresholds per direction, counted since the last completed NEWKEYS.
48#[derive(Debug, Clone, Copy, PartialEq, Eq)]
49pub struct RekeyLimits {
50    /// Encrypted payload bytes.
51    pub bytes: u64,
52    /// Packets.
53    pub packets: u32,
54}
55
56impl RekeyLimits {
57    /// 1 GiB or 2^30 packets per direction, whichever comes first.
58    pub const PRODUCTION: Self = Self { bytes: 1 << 30, packets: 1 << 30 };
59
60    /// A direction past twice its threshold plus room for the packets already in flight
61    /// when the threshold was crossed (a rekey that never completed), or a sequence
62    /// number at 2^31: the connection aborts before anything can wrap.
63    fn ceiling(&self, bytes: u64, packets: u32, seq: u32) -> bool {
64        const IN_FLIGHT_BYTES: u64 = 4 * crate::config::SSH_MAX_PACKET as u64;
65        const IN_FLIGHT_PACKETS: u32 = 64;
66        bytes >= self.bytes.saturating_mul(2).saturating_add(IN_FLIGHT_BYTES)
67            || packets >= self.packets.saturating_mul(2).saturating_add(IN_FLIGHT_PACKETS)
68            || seq >= 0x8000_0000
69    }
70}

175    /// Decrypts the first block in the buffer
176    ///
177    /// Returning the length.
178    pub fn decrypt_first_block(&mut self, buf: &mut [u8]) -> Result<usize, Error> {
179        // Cellos patch 0005: a peer past the ceiling (it never completed a rekey) ends
180        // the connection before the receive sequence number could wrap.
181        if self.limits.ceiling(self.bytes_in, self.packets_in, self.seq_decrypt.0) {
182            return Err(Error::msg("rekey ceiling reached"));
183        }
184        self.dec.decrypt_first_block(buf, self.seq_decrypt.0)
185    }
186
187    /// Decrypt and validate the remainder of the buffer.
188    ///
189    /// Decrypt bytes 4 onwards of the buffer and validate AEAD Tag or MAC.
190    /// Ensures that the packet meets minimum length.
191    pub fn decrypt(&mut self, buf: &mut [u8]) -> Result<usize, Error> {
192        let e = self.dec.decrypt(buf, self.seq_decrypt.0);
193        self.seq_decrypt += 1;
194        if let Ok(payload_len) = e {
195            self.bytes_in = self.bytes_in.saturating_add(payload_len as u64);
196            self.packets_in = self.packets_in.saturating_add(1);
197        }
198        e
199    }
200
201    /// Encrypt the output buffer
202    ///
203    /// [`buf`] is the entire output buffer to encrypt in place.
204    /// payload_len is the length of the payload portion
205    ///
206    /// This is stateful, updating the sequence numbers.
207    /// If `NoRoom` is returned buf and state will be left unmodified to allow a retry.
208    pub fn encrypt(
209        &mut self,
210        payload_len: usize,
211        buf: &mut [u8],
212    ) -> Result<usize, Error> {
213        // Cellos patch 0005: refuse to send past the ceiling (never wraps).
214        if self.limits.ceiling(self.bytes_out, self.packets_out, self.seq_encrypt.0) {
215            return Err(Error::msg("rekey ceiling reached"));
216        }
217        let e = self.enc.encrypt(payload_len, buf, self.seq_encrypt.0);
218
219        // NoRoom failure allows retries.
220        if !matches!(e, Err(Error::NoRoom { .. })) {
221            self.seq_encrypt += 1;
222            self.bytes_out = self.bytes_out.saturating_add(payload_len as u64);
223            self.packets_out = self.packets_out.saturating_add(1);
224        }
225        e
226    }
227
228    pub fn size_block_dec(&self) -> usize {
229        self.dec.cipher.size_block()
230    }
231
232    pub fn max_enc_payload(&self, total_avail: usize) -> usize {
233        self.enc.max_enc_payload(total_avail)
234    }

289    /// Padding is required to meet
290    /// - minimum packet length
291    /// - minimum padding size,
292    /// - encrypted length being a multiple of block length
293    fn calc_encrypt_pad(&self, payload_len: usize) -> usize {
294        let size_block = self.cipher.size_block();
295        // aead ciphers don't include the initial length field in encrypted blocks
296        let len = 1
297            + payload_len
298            + if self.cipher.is_aead() { 0 } else { SSH_LENGTH_SIZE };
299
300        // round padding length upwards so that len is a multiple of block size
301        let mut padlen = size_block - len % size_block;
302
303        // need at least 4 bytes padding
304        if padlen < SSH_MIN_PADLEN {
305            padlen += size_block
306        }
307
308        padlen
309    }

311    /// Returns the maximum payload that can fit in an available buffer
312    /// after header, encryption, padding, mac
313    pub fn max_enc_payload(&self, total_avail: usize) -> usize {
314        // mac is independent of the rest
315        let total_avail = total_avail.saturating_sub(self.integ.size_out());
316
317        let overhead = SSH_LENGTH_SIZE + 1 + SSH_MIN_PADLEN;
318        let mut space = total_avail;
319
320        // multiple of block length
321        let enc_len = if self.cipher.is_aead() {
322            total_avail.saturating_sub(SSH_LENGTH_SIZE)
323        } else {
324            total_avail
325        };
326
327        // round down to block size
328        let extra_block = enc_len % self.cipher.size_block();
329        if extra_block != 0 {
330            space = space.saturating_sub(extra_block);
331        }
332
333        space.saturating_sub(overhead)
334    }

336    /// Encrypt a buffer in-place, adding packet size, padding, MAC etc.
337    /// Returns the total length.
338    /// Ensures that the packet meets minimum and other length requirements.
339    ///
340    /// `buf` and internal state will be left unmodified if
341    // `Err(Error::NoRoom)` is returned, allowing a subsequent retry.
342    /// All other returned errors must be treated as fatal to the session.
343    fn encrypt(
344        &mut self,
345        payload_len: usize,
346        buf: &mut [u8],
347        seq: u32,
348    ) -> Result<usize, Error> {
349        let size_block = self.cipher.size_block();
350        let size_integ = self.integ.size_out();
351        let padlen = self.calc_encrypt_pad(payload_len);
352        // len is everything except the MAC
353        let len = SSH_LENGTH_SIZE + 1 + payload_len + padlen;
354
355        if self.cipher.is_aead() {
356            debug_assert_eq!((len - SSH_LENGTH_SIZE) % size_block, 0);
357        } else {
358            debug_assert_eq!(len % size_block, 0);
359        };
360
361        // buf and internal state must not be modified prior to returning NoRoom
362        if len + size_integ > buf.len() {
363            trace!("Output buffer {} is too small for packet", buf.len());
364            return error::NoRoom.fail();
365        }
366
367        // write the length
368        let blen = ((len - SSH_LENGTH_SIZE) as u32).to_be_bytes();
369        buf[..SSH_LENGTH_SIZE].copy_from_slice(&blen);
370        // write random padding
371        buf[SSH_LENGTH_SIZE] = padlen as u8;
372        let pad_start = SSH_LENGTH_SIZE + 1 + payload_len;
373        debug_assert_eq!(pad_start + padlen, len);
374        random::fill_random(&mut buf[pad_start..pad_start + padlen])?;
375
376        let (enc, rest) = buf.split_at_mut(len);
377        let (mac, _) = rest.split_at_mut(size_integ);
378
379        match self.integ {
380            IntegKey::ChaPoly => {}
381            IntegKey::NoInteg => {}
382            IntegKey::HmacSha256(k) => {
383                let mut h = HmacSha256::new_from_slice(&k).trap()?;
384                h.update(&seq.to_be_bytes());
385                h.update(enc);
386                let result = h.finalize();
387                mac.copy_from_slice(&result.into_bytes());
388            }
389        }
390
391        match &mut self.cipher {
392            EncKey::ChaPoly(k) => k.encrypt(seq, enc, mac).trap()?,
393            EncKey::Aes256Ctr(a) => {
394                a.apply_keystream(enc);
395            }
396            EncKey::NoCipher => {}
397        }
398
399        // ETM integ modes would go here.
400
401        Ok(len + size_integ)
402    }
403}

457    /// Decrypts the first block in the buffer
458    ///
459    /// Returns the length of the
460    /// total SSH packet (including length+mac) which is calculated
461    /// from the decrypted first 4 bytes.
462    /// Whether bytes `buf[4..block_size]` are decrypted depends on the cipher, they may be
463    /// handled later by [`decrypt`]. Bytes `buf[0..4]` may be left unmodified.
464    fn decrypt_first_block(
465        &mut self,
466        buf: &mut [u8],
467        seq: u32,
468    ) -> Result<usize, Error> {
469        if buf.len() < self.cipher.size_block() {
470            return Error::bug();
471        }
472
473        #[cfg(fuzzing)]
474        let len = u32::from_be_bytes(buf[..SSH_LENGTH_SIZE].try_into().unwrap());
475
476        #[cfg(not(fuzzing))]
477        let len = match &mut self.cipher {
478            DecKey::ChaPoly(k) => k.packet_length(seq, buf).trap()?,
479            DecKey::Aes256Ctr(a) => {
480                a.apply_keystream(&mut buf[..16]);
481                u32::from_be_bytes(buf[..SSH_LENGTH_SIZE].try_into().unwrap())
482            }
483            DecKey::NoCipher => {
484                u32::from_be_bytes(buf[..SSH_LENGTH_SIZE].try_into().unwrap())
485            }
486        };
487
488        let total_len = len
489            .checked_add((SSH_LENGTH_SIZE + self.integ.size_out()) as u32)
490            .ok_or(Error::BadDecrypt)?;
491
492        Ok(total_len as usize)
493    }
494
495    /// Decrypt the whole packet buffer and validate AEAD Tag or MAC.
496    ///
497    /// Returns the payload length.
498    /// Ensures that the packet meets minimum length.
499    /// The first block_size bytes may have been already decrypted by
500    /// [`decrypt_first_block`] depending on the cipher.
501    fn decrypt(&mut self, buf: &mut [u8], seq: u32) -> Result<usize, Error> {
502        let size_block = self.cipher.size_block();
503        let size_integ = self.integ.size_out();
504
505        if buf.len() < size_block + size_integ {
506            debug!("Bad packet, {} smaller than block size", buf.len());
507            return error::SSHProto.fail();
508        }
509        // "MUST be a multiple of the cipher block size".
510        // encrypted length for aead ciphers doesn't include the length prefix.
511        let sublength = if self.cipher.is_aead() { SSH_LENGTH_SIZE } else { 0 };
512        let len = buf.len() - size_integ - sublength;
513
514        if !len.is_multiple_of(size_block) {
515            debug!("Bad packet, not multiple of block size");
516            return error::SSHProto.fail();
517        }
518
519        let (data, mac) = buf.split_at_mut(buf.len() - size_integ);
520
521        // roundtrip tests are exhaustive over short packet lengths
522        debug_assert!(data.len() >= size_block);
523
524        // ETM modes would check integrity here.
525
526        #[cfg(not(fuzzing))]
527        match &mut self.cipher {
528            DecKey::ChaPoly(k) => {
529                k.decrypt(seq, data, mac).map_err(|_| Error::BadDecrypt)?;
530            }
531            DecKey::Aes256Ctr(a) => {
532                // safe index, checked data.len()
533                a.apply_keystream(&mut data[16..]);
534            }
535            DecKey::NoCipher => {}
536        }
537
538        #[cfg(not(fuzzing))]
539        match self.integ {
540            IntegKey::ChaPoly => {}
541            IntegKey::NoInteg => {}
542            IntegKey::HmacSha256(k) => {
543                let mut h = HmacSha256::new_from_slice(&k).trap()?;
544                h.update(&seq.to_be_bytes());
545                h.update(data);
546                h.verify_slice(mac).map_err(|_| Error::BadDecrypt)?;
547            }
548        }
549
550        let padlen = data[SSH_LENGTH_SIZE] as usize;
551        if padlen < SSH_MIN_PADLEN {
552            debug!("Packet padding too short");
553            return error::SSHProto.fail();
554        }
555
556        let payload_len = buf
557            .len()
558            .checked_sub(SSH_LENGTH_SIZE + 1 + size_integ + padlen)
559            .ok_or_else(|| {
560                debug!("Bad padding length");
561                error::SSHProto.build()
562            })?;
563
564        Ok(payload_len)
565    }
566}

811    // setting `corrupt` tests that incorrect mac is detected
812    fn do_roundtrips(
813        keys_enc: &mut KeyState,
814        keys_dec: &mut KeyState,
815        corrupt: bool,
816    ) {
817        for i in 0usize..80 {
818            let mut v: std::vec::Vec<u8> = (0u8..i as u8 + 60).collect();
819            let orig_payload = v[SSH_PAYLOAD_START..SSH_PAYLOAD_START + i].to_vec();
820
821            let written = keys_enc.encrypt(i, v.as_mut_slice()).unwrap();
822
823            v.truncate(written);
824
825            if corrupt {
826                // flip a bit of the payload
827                v[SSH_PAYLOAD_START] ^= 4;
828            }
829
830            let l = keys_dec.decrypt_first_block(v.as_mut_slice()).unwrap();
831            assert_eq!(l, v.len());
832
833            let dec = keys_dec.decrypt(v.as_mut_slice());
834
835            if corrupt {
836                assert!(matches!(dec, Err(Error::BadDecrypt)));
837                return;
838            }
839            let payload_len = dec.unwrap();
840            assert_eq!(payload_len, i);
841            let dec_payload = v[SSH_PAYLOAD_START..SSH_PAYLOAD_START + i].to_vec();
842            assert_eq!(orig_payload, dec_payload);
843        }
844    }
