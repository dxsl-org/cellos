37    pub fn open<'b>(
38        &mut self,
39        ty: packets::ChannelOpenType<'b>,
40    ) -> Result<(ChanNum, Packet<'b>)> {
41        let num = self.unused_chan()?;
42
43        let chan = Channel::new(num, (&ty).into());
44        let p = packets::ChannelOpen {
45            sender_num: num.0,
46            initial_window: chan.recv.window as u32,
47            max_packet: chan.recv.max_packet as u32,
48            ty,
49        }
50        .into();
51        let ch = &mut self.ch[num.0 as usize];
52        let ch = ch.insert(chan);
53        Ok((ch.num(), p))
54    }

143    fn reserve_chan(&mut self, co: &ChannelOpen) -> Result<&mut Channel> {
144        let num = self.unused_chan()?;
145        let mut chan = Channel::new(num, (&co.ty).into());
146        chan.send = Some(ChanDir {

158    /// Returns the channel data packet to send.
159    ///
160    /// Caller has already checked valid length with send_allowed(), and
161    /// validated `dt`.
162    /// Don't call with zero length data.
163    pub(crate) fn send_data<'b>(
164        &mut self,
165        num: ChanNum,
166        dt: ChanData,
167        data: &'b [u8],
168    ) -> Result<Packet<'b>> {
169        debug_assert!(!data.is_empty());
170
171        let ch = self.get_mut(num)?;
172        let send = ch.send.as_mut().trap()?;
173        if data.len() > send.max_packet || data.len() > send.window {
174            trace!(
175                "data len {}, max {}, window {}",
176                data.len(),
177                send.max_packet,
178                send.window
179            );
180            return Error::bug();
181        }
182        send.window -= data.len();
183        trace!("send_data: new window {}", send.window);
184        if send.window == 0 {
185            debug!("ch {num} send window empty");
186        }
187
188        let data = BinString(data);
189        let p = match dt {
190            ChanData::Normal => packets::ChannelData { num: send.num, data }.into(),
191            ChanData::Stderr => packets::ChannelDataExt {
192                num: send.num,
193                code: sshnames::SSH_EXTENDED_DATA_STDERR,
194                data,
195            }
196            .into(),
197        };
198
199        Ok(p)
200    }

259                    ChanType::Tcp => {
260                        trace!("TODO tcp channel")
261                    }
262                }
263            }
264        }
265        DispatchEvent::None

333    // the caller will send failure messages if required
334    fn dispatch_open_inner(
335        &mut self,
336        p: &ChannelOpen,
337    ) -> Result<DispatchEvent, DispatchOpenError> {
338        // Check validity before reserving a channel
339        match &p.ty {
340            ChannelOpenType::Unknown(u) => {
341                error!("Rejecting unknown channel type '{u}'");
342                return Err(ChanFail::SSH_OPEN_UNKNOWN_CHANNEL_TYPE.into());
343            }
344            ChannelOpenType::Session if self.is_client => {
345                trace!("dispatch not server");
346                return Err(error::SSHProto.build().into());
347            }
348            ChannelOpenType::ForwardedTcpip(_) => {
349                // TODO implement it
350                debug!("Rejecting forwarded tcp");
351                return Err(ChanFail::SSH_OPEN_UNKNOWN_CHANNEL_TYPE.into());
352            }
353            ChannelOpenType::DirectTcpip(_) => {
354                // TODO implement it
355                debug!("Rejecting direct tcp");
356                return Err(ChanFail::SSH_OPEN_UNKNOWN_CHANNEL_TYPE.into());
357            }
358            _ => (),
359        }
360
361        // Reserve a channel
362        let ch = self.reserve_chan(p)?;
363
364        // Beware that a reserved channel must be cleaned up on failure
365
366        match &p.ty {
367            ChannelOpenType::Session => {
368                Ok(DispatchEvent::ServEvent(ServEventId::OpenSession {
369                    num: ch.num(),
370                }))
371            }
372            // ChannelOpenType::ForwardedTcpip(t) => b.open_tcp_forwarded(handle, t),
373            // ChannelOpenType::DirectTcpip(t) => b.open_tcp_direct(handle, t),
374            _ => {
375                // Checked above
376                unreachable!()
377            }
378        }
379    }

381    pub fn resume_open(
382        &mut self,
383        c: ChanNum,
384        failure: Option<ChanFail>,
385        s: &mut TrafSend,
386    ) -> Result<()> {
387        let ch = self.get_any_mut(c)?;
388        if let Some(failure) = failure {
389            let sender_num = ch.send_num()?;
390            self.remove_any(c)?;
391            s.send(packets::ChannelOpenFailure {
392                num: sender_num,
393                reason: failure as u32,
394                desc: "".into(),
395                lang: "",
396            })?;
397            Ok(())
398        } else {
399            // Success
400            s.send(ch.open_done()?)
401        }
402    }

404    // Some returned errors will be caught by caller and returned as SSH messages
405    fn dispatch_inner(
406        &mut self,
407        packet: Packet,
408        s: &mut TrafSend,
409    ) -> Result<DispatchEvent> {
410        let mut ev = DispatchEvent::default();
411        let is_client = self.is_client;
412
413        match packet {
414            Packet::ChannelOpen(p) => {
415                ev = self.dispatch_open(&p, s)?;
416            }
417
418            Packet::ChannelOpenConfirmation(p) => {
419                let ch = self.get_any_mut(ChanNum(p.num))?;
420                match ch.state {
421                    ChanState::Opening => {
422                        debug_assert!(ch.send.is_none());
423
424                        if ch.app_done {
425                            return Ok(DispatchEvent::None);
426                        }
427
428                        ch.send = Some(ChanDir {
429                            num: p.sender_num,
430                            max_packet: p.max_packet as usize,
431                            window: p.initial_window as usize,
432                        });
433
434                        // A future progress() will notify the application.
435                        ch.open_confirmed = true;
436                        ch.state = ChanState::Normal;
437                    }
438                    _ => {
439                        trace!("Bad channel state {:?}", ch.state);
440                        return error::SSHProto.fail();
441                    }
442                }
443            }
444
445            Packet::ChannelOpenFailure(p) => {
446                let ch = self.get_any(ChanNum(p.num))?;
447                if ch.send.is_some() {
448                    // TODO: or just warn?
449                    trace!("open failure late?");
450                    return error::SSHProto.fail();
451                } else {
452                    self.remove(ChanNum(p.num))?;
453                    // TODO event
454                }
455            }
456            Packet::ChannelWindowAdjust(p) => {
457                let chan = self.get_mut(ChanNum(p.num))?;
458                let send = chan.send.as_mut().trap()?;
459                // Cellos patch 0006: the peer's window is a checked u32 (RFC 4254 §5.2);
460                // an adjustment past 2^32-1 is a protocol error, not a saturated credit.
461                let Some(window) = u32::try_from(send.window).ok().and_then(|w| w.checked_add(p.adjust)) else {
462                    return error::SSHProto.fail();
463                };
464                send.window = window as usize;
465                debug!("ch {} new window +{} = {}", p.num, p.adjust, send.window);
466                // Wake any writers that might have been blocked.
467                chan.wake_write(is_client);
468            }
469            Packet::ChannelData(p) => {
470                let ch = self.get(ChanNum(p.num))?;
471                if ch.app_done {
472                    trace!("Ignoring data for done channel");
473                } else if let Some(len) = NonZeroUsize::new(p.data.0.len()) {
474                    // TODO check we are expecting input
475                    let di =
476                        DataIn { num: ChanNum(p.num), dt: ChanData::Normal, len };
477                    ev = DispatchEvent::Data(di);
478                } else {
479                    trace!("Zero length channeldata");
480                }
481            }
482            Packet::ChannelDataExt(p) => {
483                let ch = self.get_mut(ChanNum(p.num))?;
484                if ch.app_done {
485                    trace!("Ignoring data for done channel");
486                } else if !is_client || p.code != sshnames::SSH_EXTENDED_DATA_STDERR
487                {
488                    // Discard the data, sunset can't handle this
489                    debug!("Ignoring unexpected dt data, code {}", p.code);
490                    ch.finished_input(p.data.0.len());
491                } else if let Some(len) = NonZeroUsize::new(p.data.0.len()) {
492                    // TODO check we are expecting input and dt is valid.
493                    let di =
494                        DataIn { num: ChanNum(p.num), dt: ChanData::Stderr, len };
495                    ev = DispatchEvent::Data(di);
496                } else {
497                    trace!("Zero length channeldataext");
498                }
499            }
500            Packet::ChannelEof(p) => {
501                let ch = self.get_mut(ChanNum(p.num))?;
502                ch.handle_eof(s, is_client)?;
503            }
504            Packet::ChannelClose(p) => {
505                let is_client = self.is_client;
506                let ch = self.get_mut(ChanNum(p.num))?;
507                ch.handle_close(s, is_client)?;
508            }
509            Packet::ChannelRequest(p) => {
510                let is_client = self.is_client;
511                match self.get_mut(ChanNum(p.num)) {
512                    Ok(ch) => {
513                        ev = ch.dispatch_request(&p, s, is_client)?;
514                    }
515                    Err(_) => debug!("Ignoring request to unknown channel: {p:#?}"),
516                }
517            }
518            // Cellos patch 0003: only shell/exec/subsystem requests want a reply
519            // (see `Req::send`), so the first reply on a channel answers that request.
520            Packet::ChannelSuccess(p) => {
521                let ch = self.get_mut(ChanNum(p.num))?;
522                ch.request_reply.get_or_insert(true);
523            }
524            Packet::ChannelFailure(p) => {
525                let ch = self.get_mut(ChanNum(p.num))?;
526                ch.request_reply.get_or_insert(false);
527            }
528            _ => Error::bug_msg("unreachable")?,
529        };
530
531        Ok(ev)
532    }

548                Ok(DispatchEvent::default())

556        &self,
557        p: &Packet,
558        success: bool,
559        s: &mut TrafSend,
560    ) -> Result<()> {

562            let ch = self.get(ChanNum(r.num))?;
563            if r.want_reply {
564                let num = ch.send_num()?;
565                if success {
566                    s.send(packets::ChannelSuccess { num })
567                } else {
568                    s.send(packets::ChannelFailure { num })
569                }
570            } else {
571                Ok(())
572            }
573        } else {
574            Error::bug()
575        }

681impl Req<'_> {
682    pub(crate) fn send(self, ch: &Channel, s: &mut TrafSend) -> Result<()> {
683        // Cellos patch 0003: the session's command request asks for a reply, so the
684        // client can tell an accepted command from one that may not have run.
685        let want_reply = matches!(self, Req::Shell | Req::Exec(_) | Req::Subsystem(_));
686        let t;
687        let req = match self {
688            Req::Shell => ChannelReqType::Shell,
689            Req::Pty(pty) => {
690                debug!("TODO implement pty modes");
691                t = pty.term;
692                ChannelReqType::Pty(packets::PtyReq {
693                    term: TextString(t.as_bytes()),
694                    cols: pty.cols,
695                    rows: pty.rows,
696                    width: pty.width,
697                    height: pty.height,
698                    modes: BinString(&[]),
699                })
700            }
701            Req::Exec(cmd) => {
702                ChannelReqType::Exec(packets::Exec { command: cmd.into() })
703            }
704            Req::Subsystem(cmd) => ChannelReqType::Subsystem(packets::Subsystem {
705                subsystem: cmd.into(),
706            }),
707            Req::WinChange(rt) => ChannelReqType::WinChange(rt),
708            Req::Break(rt) => ChannelReqType::Break(rt),
709        };
710
711        let p = ChannelRequest {
712            num: ch.send_num()?,
713            // Cellos patch 0003: see `want_reply` above.
714            want_reply,
715            req,
716        };
717        let p: Packet = p.into();
718        s.send(p)
719    }
720}

739/// Per-direction channel variables
740#[derive(Debug)]
741struct ChanDir {
742    /// `u32` rather than `ChanNum` because it can also be used
743    /// for the sender-side number
744    num: u32,
745    max_packet: usize,
746    window: usize,
747}
748
749#[derive(Debug)]
750enum ChanState {
751    /// An incoming channel open request that has not yet been responded to.
752    ///
753    /// Not to be used for normal channel messages
754    InOpen,
755
756    // TODO: perhaps .get() and .get_mut() should ignore Opening state channels?
757    Opening,
758    Normal,
759    RecvEof,
760    // TODO: recvclose state probably shouldn't be possible, we remove it straight away?
761    RecvClose,
762    /// The channel is unused and ready to close after a call to `done()`
763    PendingDone,
764}

766#[derive(Debug)]
767pub(crate) struct Channel {
768    ty: ChanType,
769    state: ChanState,
770    sent_eof: bool,
771    sent_close: bool,
772    /// Cellos patch 0003: reply to the shell/exec/subsystem request.
773    request_reply: Option<bool>,
774
775    recv: ChanDir,
776    /// populated in all states except `Opening`
777    send: Option<ChanDir>,
778
779    /// Accumulated bytes for the next window adjustment (inbound data direction)
780    pending_adjust: usize,
781
782    full_window: usize,
783
784    /// Set when Open Confirmation is received.
785    ///
786    /// A subsequent `progress()` will emit a `SessionOpened` event
787    /// for the application to handle.
788    open_confirmed: bool,
789
790    /// Set once application has called `done()`. The channel
791    /// will only be removed from the list
792    /// (allowing channel number re-use) if `app_done` is set
793    app_done: bool,
794
795    // Wakers for notifying readyness. Usually used for async.
796    read_waker: Option<Waker>,
797    write_waker: Option<Waker>,
798    /// Will be a stderr read waker for a client, or stderr write waker for
799    /// a server.
800    ext_waker: Option<Waker>,
801}
802
803impl Channel {
804    fn new(num: ChanNum, ty: ChanType) -> Self {
805        Channel {
806            ty,
807            state: ChanState::Opening,
808            sent_close: false,
809            request_reply: None,
810            sent_eof: false,
811            recv: ChanDir {
812                num: num.0,
813                // TODO these should depend on SSH rx buffer size minus overhead
814                max_packet: config::DEFAULT_MAX_PACKET,
815                window: config::DEFAULT_WINDOW,
816            },
817            send: None,
818            pending_adjust: 0,
819            full_window: config::DEFAULT_WINDOW,
820            open_confirmed: false,
821            app_done: false,
822            read_waker: None,
823            write_waker: None,
824            ext_waker: None,
825        }
826    }
827
828    /// Local channel number
829    pub(crate) fn num(&self) -> ChanNum {
830        ChanNum(self.recv.num)
831    }
832
833    /// Remote channel number, fails if channel is in progress opening
834    ///
835    /// Returned as a plain `u32` since it is a different namespace than `ChanNum`.
836    /// This is the channel number included in most sent packets.
837    pub(crate) fn send_num(&self) -> Result<u32> {
838        Ok(self.send.as_ref().trap()?.num)
839    }
840
841    pub fn set_read_waker(&mut self, dt: ChanData, is_client: bool, waker: &Waker) {
842        match dt {
843            ChanData::Normal => {
844                set_waker(&mut self.read_waker, waker);
845            }
846            ChanData::Stderr => {
847                if is_client {
848                    set_waker(&mut self.ext_waker, waker);
849                } else {
850                    debug_assert!(false, "server ext read waker");
851                }
852            }
853        }
854    }
855
856    pub fn set_write_waker(&mut self, dt: ChanData, is_client: bool, waker: &Waker) {
857        match dt {
858            ChanData::Normal => {
859                set_waker(&mut self.write_waker, waker);
860            }
861            ChanData::Stderr => {
862                if !is_client {
863                    set_waker(&mut self.ext_waker, waker);
864                } else {
865                    debug_assert!(false, "client ext write waker");
866                }
867            }
868        }
869    }
870
871    pub fn wake_read(&mut self, dt: ChanData, is_client: bool) {
872        match dt {
873            ChanData::Normal => {
874                if let Some(w) = self.read_waker.take() {
875                    w.wake()
876                }
877            }
878            ChanData::Stderr => {
879                if is_client {
880                    if let Some(w) = self.ext_waker.take() {
881                        w.wake()
882                    }
883                }
884            }
885        }
886    }
887
888    pub fn wake_write(&mut self, is_client: bool) {
889        if let Some(w) = self.write_waker.take() {
890            w.wake()
891        }
892        if !is_client {
893            if let Some(w) = self.ext_waker.take() {
894                w.wake()
895            }
896        }
897    }
898
899    /// Returns an open confirmation reply packet to send.
900    /// Must be called with state of `InOpen`.
901    fn open_done<'p>(&mut self) -> Result<Packet<'p>> {
902        debug_assert!(matches!(self.state, ChanState::InOpen));
903
904        self.state = ChanState::Normal;
905        let p = packets::ChannelOpenConfirmation {
906            num: self.send_num()?,
907            sender_num: self.recv.num,
908            initial_window: self.recv.window as u32,
909            max_packet: self.recv.max_packet as u32,
910        }
911        .into();
912        Ok(p)
913    }
914
915    fn dispatch_request(
916        &mut self,
917        p: &packets::ChannelRequest,
918        s: &mut TrafSend,
919        is_client: bool,
920    ) -> Result<DispatchEvent> {
921        let r = match (is_client, self.app_done) {
922            // Reject requests if the application has closed
923            // the channel. ChannelEOF is arbitrary.
924            (_, true) => Err(Error::ChannelEOF),
925            (true, _) => self.dispatch_client_request(p, s),
926            (false, _) => self.dispatch_server_request(p, s),
927        };
928
929        match r {
930            Ok(_) | Err(Error::Bug) => r,
931            Err(_) => {
932                // Most errors just send an error response, no failure.
933                if p.want_reply {
934                    let num = self.send_num();
935                    debug_assert!(num.is_ok());
936                    if let Ok(num) = num {
937                        let _ = s.send(packets::ChannelFailure { num });
938                    }
939                }
940                Ok(DispatchEvent::None)
941            }
942        }
943    }
944
945    fn dispatch_server_request(
946        &self,
947        p: &packets::ChannelRequest,
948        _s: &mut TrafSend,
949    ) -> Result<DispatchEvent> {
950        if !matches!(self.ty, ChanType::Session) {
951            return Err(Error::SSHProtoUnsupported);
952        }
953
954        let num = self.num();
955        match &p.req {
956            ChannelReqType::Shell => {
957                Ok(DispatchEvent::ServEvent(ServEventId::SessionShell { num }))
958            }
959            ChannelReqType::Exec(_) => {
960                Ok(DispatchEvent::ServEvent(ServEventId::SessionExec { num }))
961            }
962            ChannelReqType::Subsystem(_) => {
963                Ok(DispatchEvent::ServEvent(ServEventId::SessionSubsystem { num }))
964            }
965            ChannelReqType::Pty(_) => {
966                Ok(DispatchEvent::ServEvent(ServEventId::SessionPty { num }))
967            }
968            ChannelReqType::Environment(_) => {
969                Ok(DispatchEvent::ServEvent(ServEventId::Environment { num }))
970            }
971            _ => {
972                if let ChannelReqType::Unknown(u) = &p.req {
973                    warn!("Unknown channel req type \"{}\"", u)
974                } else {
975                    // OK unwrap: tested for Unknown
976                    warn!(
977                        "Unhandled channel req \"{}\"",
978                        p.req.variant_name().unwrap()
979                    )
980                };
981                Err(Error::SSHProtoUnsupported)
982            }
983        }
984    }
985
986    /// Returns Ok(want_reply: bool) on success
987    fn dispatch_client_request(
988        &mut self,
989        p: &packets::ChannelRequest,
990        _s: &mut TrafSend,
991    ) -> Result<DispatchEvent> {
992        if !matches!(self.ty, ChanType::Session) {
993            return Err(Error::SSHProtoUnsupported);
994        }
995
996        match &p.req {
997            ChannelReqType::ExitStatus(_) => {
998                Ok(DispatchEvent::CliEvent(CliEventId::SessionExit))
999            }
1000            ChannelReqType::ExitSignal(_sig) => {
1001                Ok(DispatchEvent::CliEvent(CliEventId::SessionExit))
1002            }
1003            _ => {
1004                if let ChannelReqType::Unknown(u) = &p.req {
1005                    warn!("Unknown channel req type \"{}\"", u)
1006                } else {
1007                    // OK unwrap: tested for Unknown
1008                    warn!(
1009                        "Unhandled channel req \"{}\"",
1010                        p.req.variant_name().unwrap()
1011                    )
1012                };
1013                Err(Error::SSHProtoUnsupported)
1014            }
1015        }
1016    }
1017
1018    fn handle_eof(&mut self, s: &mut TrafSend, is_client: bool) -> Result<()> {
1019        //TODO: check existing state?
1020        if !self.sent_eof {
1021            s.send(packets::ChannelEof { num: self.send_num()? })?;
1022            self.sent_eof = true;
1023        }
1024
1025        // Wake readers on EOF
1026        self.wake_read(ChanData::Normal, is_client);
1027        if is_client {
1028            self.wake_read(ChanData::Stderr, is_client);
1029        }
1030
1031        self.state = ChanState::RecvEof;
1032        // todo!();
1033        Ok(())
1034    }
1035
1036    fn handle_close(&mut self, s: &mut TrafSend, is_client: bool) -> Result<()> {
1037        //TODO: check existing state?
1038        if !self.sent_close {
1039            s.send(packets::ChannelClose { num: self.send_num()? })?;
1040            self.sent_close = true;
1041        }
1042
1043        // Wake readers and writers on EOF
1044        self.wake_read(ChanData::Normal, is_client);
1045        if is_client {
1046            self.wake_read(ChanData::Stderr, is_client);
1047        }
1048        self.wake_write(is_client);
1049
1050        self.state = ChanState::RecvClose;
1051        Ok(())
1052    }
1053
1054    fn finished_input(&mut self, len: usize) {
1055        self.pending_adjust = self.pending_adjust.saturating_add(len)
1056    }
1057
1058    fn have_recv_eof(&self) -> bool {
1059        matches!(self.state, ChanState::RecvEof | ChanState::RecvClose)
1060    }
1061
1062    fn is_closed(&self) -> bool {
1063        matches!(self.state, ChanState::RecvClose)
1064    }
1065
1066    // None on close
1067    fn send_allowed(&self) -> Option<usize> {
1068        let r = self.send.as_ref().map(|s| usize::min(s.window, s.max_packet));
1069        trace!("send_allowed {r:?}");
1070        r
1071    }
1072
1073    pub(crate) fn valid_send(&self, _dt: ChanData) -> bool {
1074        // TODO: later we should only allow non-pty "session" channels
1075        // to have dt, for stderr only.
1076        true
1077    }
1078
1079    /// Send a window adjust packet if required.
1080    fn check_send_window_adjust(&mut self, s: &mut TrafSend) {
1081        if self.pending_adjust > self.full_window / 2 {
1082            let adjust = self.pending_adjust as u32;
1083            let Some(sdir) = self.send.as_mut() else {
1084                return;
1085            };
1086            let num = sdir.num;
1087            let p = packets::ChannelWindowAdjust { num, adjust };
1088            match s.send(p) {
1089                Ok(()) => self.pending_adjust = 0,
1090                Err(Error::BusySend { .. }) => {
1091                    // Do nothing, the adjustment will be sent later.
1092                }
1093                Err(e) => debug_assert!(false, "Window adjust send failed {e:?}"),
1094            }
1095        }
1096    }
1097}
