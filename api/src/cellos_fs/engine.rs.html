34    /// Format a device with CellosFS.
35    pub fn format(mut disk: D, total_blocks: u64) -> Result<Self, FsError> {
36        if total_blocks != disk.total_blocks() {
37            return Err(FsError::PartitionOverflow);
38        }
39        if total_blocks < 16 {
40            return Err(FsError::NoSpace);
41        }
42
43        // Calculate bitmap blocks: 1 block covers 32,768 blocks (128 MiB)
44        let bitmap_blocks = u32::try_from(total_blocks.div_ceil(32_768))
45            .map_err(|_| FsError::NoSpace)?;
46        let mut allocator = BitmapAllocator::new(FIRST_USABLE_BLOCK, bitmap_blocks, total_blocks)?;
47
48        // Allocate root inode block (FIRST_USABLE_BLOCK + bitmap_blocks)
49        let root_inode_block = allocator.allocate_block()?;
50
51        // Write root inode (inode 1)
52        let mut root_inode = InodeRecord::new(1, INODE_MODE_DIR);
53        write_inode(&mut disk, root_inode_block, 0, &mut root_inode)?;
54
55        // Write empty Superblock B (slot 1)
56        let empty_block = [0u8; BLOCK_SIZE];
57        disk.write_block(SUPERBLOCK_B_BLOCK, &empty_block)?;
58
59        // Create and write Superblock A (slot 0) with sequence 1
60        let mut sb = Superblock::new(total_blocks, bitmap_blocks);
61        sb.root_inode_block = root_inode_block;
62        sb.sequence = 1;
63        sb.update_checksum();
64        disk.write_block(SUPERBLOCK_A_BLOCK, &sb.to_bytes())?;
65
66        // Save allocator
67        allocator.save_to_disk(&mut disk)?;
68        disk.flush()?;
69
70        Ok(Self {
71            disk,
72            active_sb: sb,
73            active_slot: 0,
74            allocator,
75            dirty_inodes: BTreeMap::new(),
76        })
77    }

79    /// Open an existing CellosFS volume.
80    /// Evaluates both Superblock A and B, selecting the valid one with highest sequence.
81    pub fn open(mut disk: D) -> Result<Self, FsError> {
82        let mut buf_a = [0u8; BLOCK_SIZE];
83        let mut buf_b = [0u8; BLOCK_SIZE];
84
85        disk.read_block(SUPERBLOCK_A_BLOCK, &mut buf_a)?;
86        disk.read_block(SUPERBLOCK_B_BLOCK, &mut buf_b)?;
87        let sb_a = Superblock::from_bytes(&buf_a);
88        let sb_b = Superblock::from_bytes(&buf_b);
89        for sb in [sb_a, sb_b].into_iter().flatten() {
90            if !sb.validate_geometry(disk.total_blocks()) {
91                return Err(FsError::Corrupted);
92            }
93        }
94
95        let (active_sb, active_slot) = match (sb_a, sb_b) {
96            (Some(a), Some(b)) => {
97                if a.sequence >= b.sequence {
98                    (a, 0)
99                } else {
100                    (b, 1)
101                }
102            }
103            (Some(a), None) => (a, 0),
104            (None, Some(b)) => (b, 1),
105            (None, None) => return Err(FsError::Corrupted),
106        };
107
108        let allocator = BitmapAllocator::load_from_disk(
109            &mut disk,
110            active_sb.free_bitmap_block,
111            active_sb.free_bitmap_blocks,
112            active_sb.total_blocks,
113        )?;
114        // A selected metadata pointer cannot be simultaneously available for allocation.
115        // This checks existing bits only; it does not reconstruct allocation or repair cuts.
116        for &block in &active_sb.inode_blocks {
117            if block != 0 && !allocator.is_allocated(block)? {
118                return Err(FsError::Corrupted);
119            }
120        }
121
122        let mut fs = Self {
123            disk,
124            active_sb,
125            active_slot,
126            allocator,
127            dirty_inodes: BTreeMap::new(),
128        };
129        if !fs.root_inode()?.is_dir() {
130            return Err(FsError::Corrupted);
131        }
132        Ok(fs)
133    }

149    fn get_inode(&mut self, inode_num: u64) -> Result<InodeRecord, FsError> {
150        if inode_num == 0 || inode_num >= self.active_sb.next_inode_num {
151            return Err(FsError::Corrupted);
152        }
153        let record = if let Some(record) = self.dirty_inodes.get(&inode_num) {
154            *record
155        } else {
156            let slot = ((inode_num - 1) / INODES_PER_BLOCK as u64) as usize;
157            let block = *self.active_sb.inode_blocks.get(slot).ok_or(FsError::Corrupted)?;
158            if block == 0 {
159                return Err(FsError::Corrupted);
160            }
161            let idx = ((inode_num - 1) % INODES_PER_BLOCK as u64) as usize;
162            read_inode(&mut self.disk, block, idx)?
163        };
164        if record.inode_num != inode_num {
165            return Err(FsError::Corrupted);
166        }
167        self.validate_inode(&record)?;
168        Ok(record)
169    }

176    fn validate_inode(&self, inode: &InodeRecord) -> Result<(), FsError> {
177        if inode.is_dir() {
178            if !inode.is_inline() || inode.extent_count as usize > INODE_PAYLOAD_SIZE / DirEntryRecord::SIZE {
179                return Err(FsError::Corrupted);
180            }
181            for i in 0..inode.extent_count as usize {
182                let off = i * DirEntryRecord::SIZE;
183                let bytes = inode.payload[off..off + DirEntryRecord::SIZE].try_into()
184                    .map_err(|_| FsError::Corrupted)?;
185                let entry = DirEntryRecord::from_bytes(bytes).ok_or(FsError::Corrupted)?;
186                if entry.inode_num == 0 || entry.inode_num >= self.active_sb.next_inode_num
187                    || !matches!(entry.file_type, 1 | 2)
188                    || entry.name_len == 0
189                    || core::str::from_utf8(&entry.name[..entry.name_len as usize]).is_err()
190                    || entry.name_as_str().contains('/')
191                {
192                    return Err(FsError::Corrupted);
193                }
194                for previous in 0..i {
195                    let off = previous * DirEntryRecord::SIZE;
196                    let bytes = inode.payload[off..off + DirEntryRecord::SIZE].try_into()
197                        .map_err(|_| FsError::Corrupted)?;
198                    let other = DirEntryRecord::from_bytes(bytes).ok_or(FsError::Corrupted)?;
199                    if other.name_as_str() == entry.name_as_str() {
200                        return Err(FsError::Corrupted);
201                    }
202                }
203            }
204        } else if inode.is_file() {
205            if inode.is_inline() {
206                if inode.size > INODE_PAYLOAD_SIZE as u64 || inode.inline_size as u64 != inode.size
207                    || inode.extent_count != 0
208                {
209                    return Err(FsError::Corrupted);
210                }
211            } else {
212                self.validate_extents(inode)?;
213            }
214        } else {
215            return Err(FsError::Corrupted);
216        }
217        Ok(())
218    }

229    /// Lookup an Inode by absolute path (e.g. "/foo.txt" or "/dir/bar.txt").
230    pub fn lookup(&mut self, path: &str) -> Result<InodeRecord, FsError> {
231        let trimmed = path.trim_matches('/');
232        if trimmed.is_empty() {
233            return self.root_inode();
234        }
235
236        let mut current_inode = self.root_inode()?;
237
238        for component in trimmed.split('/') {
239            if component.is_empty() {
240                continue;
241            }
242            if !current_inode.is_dir() {
243                return Err(FsError::NotADirectory);
244            }
245
246            let entry = DirPayload::find(
247                &current_inode.payload,
248                current_inode.extent_count as usize,
249                component,
250            )
251            .ok_or(FsError::NotFound)?;
252            current_inode = self.get_inode(entry.inode_num)?;
253            if (entry.file_type == 2) != current_inode.is_dir() {
254                return Err(FsError::Corrupted);
255            }
256        }
257
258        Ok(current_inode)
259    }

271    fn create_entry(&mut self, path: &str, mode: u16) -> Result<InodeRecord, FsError> {
272        self.ensure_mutable()?;
273        let trimmed = path.trim_matches('/');
274        if trimmed.is_empty() {
275            return Err(FsError::AlreadyExists);
276        }
277
278        let (parent_path, leaf_name) = match trimmed.rfind('/') {
279            Some(pos) => (&trimmed[..pos], &trimmed[pos + 1..]),
280            None => ("", trimmed),
281        };
282
283        let mut parent_inode = self.lookup(parent_path)?;
284        if !parent_inode.is_dir() {
285            return Err(FsError::NotADirectory);
286        }
287
288        if DirPayload::find(
289            &parent_inode.payload,
290            parent_inode.extent_count as usize,
291            leaf_name,
292        )
293        .is_some()
294        {
295            return Err(FsError::AlreadyExists);
296        }
297
298        let new_inode_num = self.active_sb.next_inode_num;
299        let next_inode_num = new_inode_num.checked_add(1).ok_or(FsError::NoSpace)?;
300        if new_inode_num == 0
301            || new_inode_num > self.active_sb.inode_blocks.len() as u64 * INODES_PER_BLOCK as u64
302        {
303            return Err(FsError::NoSpace);
304        }
305        let file_type = if mode == INODE_MODE_DIR { 2 } else { 1 };
306        let dir_entry =
307            DirEntryRecord::new(new_inode_num, file_type, leaf_name).ok_or(FsError::InvalidName)?;
308
309        let mut count = parent_inode.extent_count as usize;
310        DirPayload::insert(&mut parent_inode.payload, &mut count, &dir_entry)?;
311        parent_inode.extent_count = count as u32;
312        self.active_sb.next_inode_num = next_inode_num;
313
314        let new_inode = InodeRecord::new(new_inode_num, mode);
315        self.put_inode(new_inode);
316        self.put_inode(parent_inode);
317        Ok(new_inode)
318    }

320    /// Read file content, returning the complete requested in-file range or an error.
321    pub fn read_file(&mut self, path: &str, offset: u64, buf: &mut [u8]) -> Result<usize, FsError> {
322        let inode = self.lookup(path)?;
323        if !inode.is_file() {
324            return Err(FsError::IsADirectory);
325        }
326        if offset >= inode.size {
327            return Ok(0);
328        }
329        let to_read = buf.len().min(usize::try_from(inode.size - offset).unwrap_or(usize::MAX));
330        if inode.is_inline() {
331            let start = offset as usize;
332            buf[..to_read].copy_from_slice(&inode.payload[start..start + to_read]);
333            return Ok(to_read);
334        }
335        let mut bytes_read = 0;
336        while bytes_read < to_read {
337            let position = offset + bytes_read as u64;
338            let logical = (position / BLOCK_SIZE as u64) as u32;
339            let block_offset = (position % BLOCK_SIZE as u64) as usize;
340            let physical = self.mapped_block(&inode, logical)?;
341            let mut block = [0u8; BLOCK_SIZE];
342            self.disk.read_block(physical, &mut block)?;
343            let count = (BLOCK_SIZE - block_offset).min(to_read - bytes_read);
344            buf[bytes_read..bytes_read + count].copy_from_slice(&block[block_offset..block_offset + count]);
345            bytes_read += count;
346        }
347        Ok(bytes_read)
348    }

378    fn write_content(&mut self, mut inode: InodeRecord, offset: u64, end: u64,
379        data: &[u8], replace: bool) -> Result<usize, FsError>
380    {
381        if !inode.is_file() {
382            return Err(FsError::IsADirectory);
383        }
384        if !replace && data.is_empty() {
385            return Ok(0);
386        }
387        let size = if replace { end } else { inode.size.max(end) };
388        let needed = u32::try_from(size.div_ceil(BLOCK_SIZE as u64))
389            .map_err(|_| FsError::InvalidOffset)?;
390        if needed as u64 > self.active_sb.total_blocks {
391            return Err(FsError::NoSpace);
392        }
393        let was_inline = inode.is_inline();
394        let old_size = if replace { 0 } else { inode.size };
395        let mut extents = if was_inline { Vec::new() } else { self.read_extents(&inode)? };
396        if size <= INODE_PAYLOAD_SIZE as u64 && (replace || was_inline) {
397            for ext in &extents {
398                for b in 0..ext.block_count {
399                    self.allocator.free_block(ext.physical_block + b as u64)?;
400                }
401            }
402            if replace {
403                inode.payload.fill(0);
404            } else if offset > inode.size {
405                inode.payload[inode.size as usize..offset as usize].fill(0);
406            }
407            inode.payload[offset as usize..end as usize].copy_from_slice(data);
408            inode.payload[size as usize..].fill(0);
409            inode.size = size;
410            inode.inline_size = size as u32;
411            inode.extent_count = 0;
412            inode.flags |= INODE_FLAG_INLINE;
413            self.put_inode(inode);
414            return Ok(data.len());
415        }
416        let current = extents.last().map_or(0, |e| e.logical_block + e.block_count);
417        if needed > current && (needed - current) as u64 > self.allocator.free_block_count() {
418            return Err(FsError::NoSpace);
419        }
420        for logical in current..needed {
421            let physical = self.allocator.allocate_block()?;
422            if let Some(last) = extents.last_mut() {
423                if last.physical_block + last.block_count as u64 == physical {
424                    last.block_count = last.block_count.checked_add(1).ok_or(FsError::InvalidOffset)?;
425                    continue;
426                }
427            }
428            if extents.len() == INODE_PAYLOAD_SIZE / Extent::SIZE {
429                return Err(FsError::NoSpace);
430            }
431            extents.push(Extent { logical_block: logical, block_count: 1, physical_block: physical });
432        }
433        // Include gap blocks and block zero during inline-to-extent conversion.
434        let first = if was_inline { 0 } else { offset.min(old_size) / BLOCK_SIZE as u64 };
435        for logical in first..end.div_ceil(BLOCK_SIZE as u64) {
436            let start = logical * BLOCK_SIZE as u64;
437            let stop = start + BLOCK_SIZE as u64;
438            let mut block = [0u8; BLOCK_SIZE];
439            let physical = Self::extent_block(&extents, logical as u32)?;
440            let preserved_end = old_size.min(stop);
441            if was_inline && logical == 0 && old_size != 0 {
442                block[..old_size as usize].copy_from_slice(&inode.payload[..old_size as usize]);
443            } else if !was_inline && start < old_size
444                && (offset > start || end < preserved_end)
445            {
446                self.disk.read_block(physical, &mut block)?;
447            }
448            if end > old_size {
449                let gap_start = old_size.max(start);
450                let gap_end = offset.min(stop);
451                if gap_start < gap_end {
452                    block[(gap_start - start) as usize..(gap_end - start) as usize].fill(0);
453                }
454            }
455            let copy_start = offset.max(start);
456            let copy_end = end.min(stop);
457            if copy_start < copy_end {
458                let source = (copy_start - offset) as usize;
459                let count = (copy_end - copy_start) as usize;
460                block[(copy_start - start) as usize..(copy_end - start) as usize]
461                    .copy_from_slice(&data[source..source + count]);
462            }
463            self.disk.write_block(physical, &block)?;
464        }
465        // Tail retirement is immediate; crash-preserving retirement is out of scope.
466        if replace {
467            for ext in &mut extents {
468                let keep = needed.saturating_sub(ext.logical_block).min(ext.block_count);
469                for b in keep..ext.block_count {
470                    self.allocator.free_block(ext.physical_block + b as u64)?;
471                }
472                ext.block_count = keep;
473            }
474            extents.retain(|e| e.block_count != 0);
475        }
476        inode.flags &= !INODE_FLAG_INLINE;
477        inode.inline_size = 0;
478        inode.size = size;
479        inode.extent_count = extents.len() as u32;
480        inode.payload.fill(0);
481        for (i, ext) in extents.iter().enumerate() {
482            let off = i * Extent::SIZE;
483            inode.payload[off..off + Extent::SIZE].copy_from_slice(&ext.to_bytes());
484        }
485        self.put_inode(inode);
486        Ok(data.len())
487    }

489    fn extent_at(inode: &InodeRecord, index: usize) -> Extent {

493        Extent::from_bytes(&bytes)

496    fn validate_extents(&self, inode: &InodeRecord) -> Result<(), FsError> {
497        if inode.extent_count as usize > INODE_PAYLOAD_SIZE / Extent::SIZE {
498            return Err(FsError::Corrupted);
499        }
500        let mut logical_end = 0u32;
501        for i in 0..inode.extent_count as usize {
502            let ext = Self::extent_at(inode, i);
503            let end = ext.physical_block.checked_add(ext.block_count as u64).ok_or(FsError::Corrupted)?;
504            if ext.block_count == 0 || ext.logical_block != logical_end
505                || ext.physical_block < self.active_sb.free_bitmap_block + self.active_sb.free_bitmap_blocks as u64
506                || end > self.active_sb.total_blocks
507                || self.active_sb.inode_blocks.iter().any(|&b| b != 0 && b >= ext.physical_block && b < end)
508            {
509                return Err(FsError::Corrupted);
510            }
511            for previous in 0..i {
512                let other = Self::extent_at(inode, previous);
513                if ext.physical_block < other.physical_block + other.block_count as u64
514                    && end > other.physical_block
515                {
516                    return Err(FsError::Corrupted);
517                }
518            }
519            logical_end = logical_end.checked_add(ext.block_count).ok_or(FsError::Corrupted)?;
520        }
521        if inode.size.div_ceil(BLOCK_SIZE as u64) != logical_end as u64 {
522            return Err(FsError::Corrupted);
523        }
524        Ok(())
525    }

527    fn read_extents(&self, inode: &InodeRecord) -> Result<Vec<Extent>, FsError> {
528        self.validate_extents(inode)?;

532    fn mapped_block(&self, inode: &InodeRecord, logical: u32) -> Result<u64, FsError> {
533        for i in 0..inode.extent_count as usize {
534            let ext = Self::extent_at(inode, i);
535            if logical >= ext.logical_block && logical - ext.logical_block < ext.block_count {
536                return Ok(ext.physical_block + (logical - ext.logical_block) as u64);
537            }
538        }
539        Err(FsError::Corrupted)
540    }

543        extents.iter().find(|e| logical >= e.logical_block && logical - e.logical_block < e.block_count)
544            .map(|e| e.physical_block + (logical - e.logical_block) as u64)

548    /// List directory contents: returns vector of `(entry_name, is_dir, size)`.
549    pub fn list_dir(&mut self, path: &str) -> Result<Vec<(String, bool, u64)>, FsError> {
550        let inode = self.lookup(path)?;
551        if !inode.is_dir() {
552            return Err(FsError::NotADirectory);
553        }
554
555        let entries = DirPayload::list(&inode.payload, inode.extent_count as usize);
556        let mut res = Vec::new();
557
558        for e in entries {
559            let is_dir = e.file_type == 2;
560            let child_inode = self.get_inode(e.inode_num)?;
561            if is_dir != child_inode.is_dir() {
562                return Err(FsError::Corrupted);
563            }
564            res.push((e.name_as_str().to_string(), is_dir, child_inode.size));
565        }
566
567        Ok(res)
568    }
569
570    /// Unlink (delete) a regular file.
571    pub fn unlink(&mut self, path: &str) -> Result<(), FsError> {
572        self.ensure_mutable()?;
573        let trimmed = path.trim_matches('/');
574        let (parent_path, leaf_name) = match trimmed.rfind('/') {
575            Some(pos) => (&trimmed[..pos], &trimmed[pos + 1..]),
576            None => ("", trimmed),
577        };
578
579        let child_inode = self.lookup(trimmed)?;
580        if child_inode.is_dir() {
581            return Err(FsError::IsADirectory);
582        }
583
584        let mut parent_inode = self.lookup(parent_path)?;
585        if !parent_inode.is_dir() {
586            return Err(FsError::NotADirectory);
587        }
588
589        let mut count = parent_inode.extent_count as usize;
590        let removed_inode_num =
591            DirPayload::remove(&mut parent_inode.payload, &mut count, leaf_name)?;
592        parent_inode.extent_count = count as u32;
593
594        if !child_inode.is_inline() {
595            let extents = self.read_extents(&child_inode)?;
596            for ext in extents {
597                for b in 0..ext.block_count {
598                    self.allocator.free_block(ext.physical_block + b as u64)?;
599                }
600            }
601        }
602
603        self.put_inode(parent_inode);
604        self.dirty_inodes.remove(&removed_inode_num);
605        Ok(())
606    }
607
608    /// Remove an EMPTY directory at `path`.
609    pub fn rmdir(&mut self, path: &str) -> Result<(), FsError> {
610        self.ensure_mutable()?;
611        let trimmed = path.trim_matches('/');
612        let (parent_path, leaf_name) = match trimmed.rfind('/') {
613            Some(pos) => (&trimmed[..pos], &trimmed[pos + 1..]),
614            None => ("", trimmed),
615        };
616
617        let child_inode = self.lookup(trimmed)?;
618        if !child_inode.is_dir() {
619            return Err(FsError::NotADirectory);
620        }
621
622        if child_inode.extent_count > 0 {
623            return Err(FsError::AlreadyExists); // Directory not empty
624        }
625
626        let mut parent_inode = self.lookup(parent_path)?;
627        if !parent_inode.is_dir() {
628            return Err(FsError::NotADirectory);
629        }
630
631        let mut count = parent_inode.extent_count as usize;
632        let removed_inode_num =
633            DirPayload::remove(&mut parent_inode.payload, &mut count, leaf_name)?;
634        parent_inode.extent_count = count as u32;
635
636        self.put_inode(parent_inode);
637        self.dirty_inodes.remove(&removed_inode_num);
638        Ok(())
639    }
640    /// Atomic non-replacing rename of a file from `old_path` to `new_path`.
641    pub fn rename(&mut self, old_path: &str, new_path: &str) -> Result<(), FsError> {
642        self.ensure_mutable()?;
643        let old_trimmed = old_path.trim_matches('/');
644        let new_trimmed = new_path.trim_matches('/');
645
646        if old_trimmed.is_empty() || new_trimmed.is_empty() {
647            return Err(FsError::InvalidName);
648        }
649
650        let old_inode = self.lookup(old_trimmed)?;
651        if old_inode.is_dir() {
652            return Err(FsError::IsADirectory);
653        }
654
655        if old_trimmed == new_trimmed {
656            return Ok(());
657        }
658
659        // Check target does not exist
660        match self.lookup(new_trimmed) {
661            Ok(_) => return Err(FsError::AlreadyExists),
662            Err(FsError::NotFound) => {}
663            Err(error) => return Err(error),
664        }
665        let (old_parent_path, old_leaf) = match old_trimmed.rfind('/') {
666            Some(pos) => (&old_trimmed[..pos], &old_trimmed[pos + 1..]),
667            None => ("", old_trimmed),
668        };
669
670        let (new_parent_path, new_leaf) = match new_trimmed.rfind('/') {
671            Some(pos) => (&new_trimmed[..pos], &new_trimmed[pos + 1..]),
672            None => ("", new_trimmed),
673        };
674
675        let mut old_parent = self.lookup(old_parent_path)?;
676        let mut new_parent = self.lookup(new_parent_path)?;
677        if !old_parent.is_dir() || !new_parent.is_dir() {
678            return Err(FsError::NotADirectory);
679        }
680        // Insertion independently refuses duplicates, including equivalent path spellings.
681        if DirPayload::find(&new_parent.payload, new_parent.extent_count as usize, new_leaf).is_some() {
682            return Err(FsError::AlreadyExists);
683        }
684        let new_entry = DirEntryRecord::new(old_inode.inode_num, 1, new_leaf)
685            .ok_or(FsError::InvalidName)?;
686        let mut old_count = old_parent.extent_count as usize;
687        DirPayload::remove(&mut old_parent.payload, &mut old_count, old_leaf)?;
688        old_parent.extent_count = old_count as u32;
689        if old_parent.inode_num == new_parent.inode_num {
690            let mut count = old_parent.extent_count as usize;
691            DirPayload::insert(&mut old_parent.payload, &mut count, &new_entry)?;
692            old_parent.extent_count = count as u32;
693            self.put_inode(old_parent);
694        } else {
695            let mut new_count = new_parent.extent_count as usize;
696            DirPayload::insert(&mut new_parent.payload, &mut new_count, &new_entry)?;
697            new_parent.extent_count = new_count as u32;
698            self.put_inode(old_parent);
699            self.put_inode(new_parent);
700        }
701
702        Ok(())
703    }
704
705    /// Publish metadata through the two existing barriers. This is not crash-preservation proof.
706    pub fn sync(&mut self) -> Result<(), FsError> {
707        self.ensure_mutable()?;
708        let mut new_sb = self.active_sb;
709        new_sb.sequence = new_sb.sequence.checked_add(1).ok_or(FsError::NoSpace)?;
710
711        if !self.dirty_inodes.is_empty() {
712            // Group dirty inodes by table block slot (each slot covers 8 inodes)
713            let mut slots: BTreeMap<usize, Vec<(usize, InodeRecord)>> = BTreeMap::new();
714            for (&inode_num, record) in &mut self.dirty_inodes {
715                let slot = ((inode_num - 1) / (INODES_PER_BLOCK as u64)) as usize;
716                let idx = ((inode_num - 1) % (INODES_PER_BLOCK as u64)) as usize;
717                record.update_checksum();
718                slots.entry(slot).or_default().push((idx, *record));
719            }
720
721            for (slot, records) in slots {
722                if slot >= new_sb.inode_blocks.len() {
723                    return Err(FsError::NoSpace);
724                }
725                let old_block = new_sb.inode_blocks[slot];
726                let new_block = self.allocator.allocate_block()?;
727
728                let mut block_buf = [0u8; BLOCK_SIZE];
729                if old_block != 0 {
730                    self.disk.read_block(old_block, &mut block_buf)?;
731                }
732
733                for (idx, record) in records {
734                    let offset = idx * INODE_SIZE;
735                    block_buf[offset..offset + INODE_SIZE].copy_from_slice(&record.to_bytes());
736                }
737
738                self.disk.write_block(new_block, &block_buf)?;
739
740                if old_block != 0 {
741                    self.allocator.free_block(old_block)?;
742                }
743                new_sb.inode_blocks[slot] = new_block;
744            }
745
746            self.dirty_inodes.clear();
747        }
748
749        new_sb.root_inode_block = new_sb.inode_blocks[0];
750
751        // Save allocator bitmap changes
752        self.allocator.save_to_disk(&mut self.disk)?;
753        self.disk.flush()?;
754
755        // Prepare new Superblock with incremented sequence
756        new_sb.free_block_count = self.allocator.free_block_count();
757        new_sb.update_checksum();
758
759        // Alternate slot: 0 -> write to slot 1 (B); 1 -> write to slot 0 (A)
760        let target_slot = 1 - self.active_slot;
761        let target_block = if target_slot == 0 {
762            SUPERBLOCK_A_BLOCK
763        } else {
764            SUPERBLOCK_B_BLOCK
765        };
766
767        self.disk.write_block(target_block, &new_sb.to_bytes())?;
768        self.disk.flush()?;
769
770        self.active_sb = new_sb;
771        self.active_slot = target_slot;
772        Ok(())
773    }
774}

776/// The default-off host experiment preserves the existing library interface.
777#[cfg(all(feature = "g2-cow-experiment", not(target_os = "none")))]
778pub struct CellosFs<D: BlockDevice> {
779    inner: crate::cow_experiment::Cow<D>,
780}
781
782#[cfg(all(feature = "g2-cow-experiment", not(target_os = "none")))]
783impl<D: BlockDevice> CellosFs<D> {
784    pub fn format(disk: D, total_blocks: u64) -> Result<Self, FsError> {
785        crate::cow_experiment::Cow::format(disk, total_blocks).map(|inner| Self { inner })
786    }
787    pub fn open(disk: D) -> Result<Self, FsError> {
788        crate::cow_experiment::Cow::open(disk).map(|inner| Self { inner })
789    }
790    pub fn superblock(&self) -> &Superblock { self.inner.superblock() }
791    pub fn free_blocks(&self) -> u64 { self.inner.free_blocks() }
792    pub fn into_disk(self) -> D { self.inner.into_disk() }
793    pub fn lookup(&mut self, path: &str) -> Result<InodeRecord, FsError> {
794        self.inner.lookup(path)
795    }
796    pub fn create_file(&mut self, path: &str) -> Result<InodeRecord, FsError> {
797        self.inner.create(path, INODE_MODE_FILE)
798    }
799    pub fn create_dir(&mut self, path: &str) -> Result<InodeRecord, FsError> {
800        self.inner.create(path, INODE_MODE_DIR)
801    }
802    pub fn read_file(&mut self, path: &str, offset: u64, buf: &mut [u8]) -> Result<usize, FsError> {
803        self.inner.read_file(path, offset, buf)
804    }
805    pub fn validate_write_range(offset: u64, length: usize) -> Result<u64, FsError> {
806        let end = offset.checked_add(u64::try_from(length).map_err(|_| FsError::InvalidOffset)?)
807            .ok_or(FsError::InvalidOffset)?;
808        if end > u32::MAX as u64 * BLOCK_SIZE as u64 { return Err(FsError::InvalidOffset); }
809        Ok(end)
810    }
811    pub fn write_file(&mut self, path: &str, offset: u64, data: &[u8]) -> Result<usize, FsError> {
812        self.inner.write_file(path, offset, data, false)
813    }
814    pub fn replace_file(&mut self, path: &str, data: &[u8]) -> Result<usize, FsError> {
815        self.inner.write_file(path, 0, data, true)
816    }
817    pub fn list_dir(&mut self, path: &str) -> Result<Vec<(String, bool, u64)>, FsError> {
818        self.inner.list_dir(path)
819    }
820    pub fn unlink(&mut self, path: &str) -> Result<(), FsError> { self.inner.remove(path, false) }
821    pub fn rmdir(&mut self, path: &str) -> Result<(), FsError> { self.inner.remove(path, true) }
822    pub fn rename(&mut self, old: &str, new: &str) -> Result<(), FsError> {
823        self.inner.rename(old, new)
824    }
825    pub fn sync(&mut self) -> Result<(), FsError> { self.inner.sync() }
826}
