1//! CellosFS On-Disk Binary Format Specifications.
2//!
3//! All on-disk integers are Little-Endian.
4//! Standard block size is 4096 bytes (8 x 512-byte sectors).
5
6use crate::crc32::crc32c;
7
8/// Standard block size: 4 KiB.
9pub const BLOCK_SIZE: usize = 4096;
10
11/// Magic identifier: b"CELL_OSF" (0x46534F5F4C4C4543 in LE).
12pub const MAGIC: [u8; 8] = *b"CELL_OSF";
13
14/// Format Version 1.
15pub const FORMAT_VERSION: u32 = 1;
16
17/// Block index for Superblock A.
18pub const SUPERBLOCK_A_BLOCK: u64 = 0;
19
20/// Block index for Superblock B.
21pub const SUPERBLOCK_B_BLOCK: u64 = 1;
22
23/// First allocatable data/metadata block.
24pub const FIRST_USABLE_BLOCK: u64 = 2;
25
26/// Inode size: 512 bytes (8 inodes packed per 4 KiB block).
27pub const INODE_SIZE: usize = 512;
28pub const INODES_PER_BLOCK: usize = BLOCK_SIZE / INODE_SIZE;
29
30/// Maximum payload size inside an Inode (for inline file data or extents).
31pub const INODE_PAYLOAD_SIZE: usize = 428;
32
33/// Inode Modes.
34pub const INODE_MODE_UNUSED: u16 = 0;
35pub const INODE_MODE_FILE: u16 = 1;
36pub const INODE_MODE_DIR: u16 = 2;
37
38/// Inode Flags.
39pub const INODE_FLAG_INLINE: u16 = 1 << 0;
40
41/// Maximum length of a filename in a compact DirEntry (56 bytes total).
42pub const MAX_NAME_LEN: usize = 46;
43
44/// Superblock structure (4096 bytes, stored at block 0 and block 1).
45#[repr(C)]
46#[derive(Clone, Copy, Debug)]
47pub struct Superblock {
48    pub magic: [u8; 8],
49    pub version: u32,
50    pub flags: u32,
51    pub sequence: u64,
52    pub block_size: u32,
53    pub total_blocks: u64,
54    pub root_inode_block: u64,
55    pub free_bitmap_block: u64,
56    pub free_bitmap_blocks: u32,
57    pub free_block_count: u64,
58    pub next_inode_num: u64,
59    pub created_at: u64,
60    pub mounted_at: u64,
61    pub inode_blocks: [u64; 32],
62    pub reserved: [u8; 3748],
63    pub checksum: u32,
64}
65
66impl Superblock {
67    pub fn new(total_blocks: u64, free_bitmap_blocks: u32) -> Self {
68        let mut sb = Self {
69            magic: MAGIC,
70            version: FORMAT_VERSION,
71            flags: 0,
72            sequence: 1,
73            block_size: BLOCK_SIZE as u32,
74            total_blocks,
75            root_inode_block: FIRST_USABLE_BLOCK + free_bitmap_blocks as u64,
76            free_bitmap_block: FIRST_USABLE_BLOCK,
77            free_bitmap_blocks,
78            free_block_count: total_blocks
79                .saturating_sub(FIRST_USABLE_BLOCK + free_bitmap_blocks as u64 + 1),
80            next_inode_num: 2, // 1 is root inode
81            created_at: 0,
82            mounted_at: 0,
83            inode_blocks: {
84                let mut arr = [0u64; 32];
85                arr[0] = FIRST_USABLE_BLOCK + free_bitmap_blocks as u64;
86                arr
87            },
88            reserved: [0u8; 3748],
89            checksum: 0,
90        };
91        sb.update_checksum();
92        sb
93    }
94
95    pub fn to_bytes(&self) -> [u8; BLOCK_SIZE] {
96        let mut buf = [0u8; BLOCK_SIZE];
97        buf[0..8].copy_from_slice(&self.magic);
98        buf[8..12].copy_from_slice(&self.version.to_le_bytes());
99        buf[12..16].copy_from_slice(&self.flags.to_le_bytes());
100        buf[16..24].copy_from_slice(&self.sequence.to_le_bytes());
101        buf[24..28].copy_from_slice(&self.block_size.to_le_bytes());
102        buf[28..36].copy_from_slice(&self.total_blocks.to_le_bytes());
103        buf[36..44].copy_from_slice(&self.root_inode_block.to_le_bytes());
104        buf[44..52].copy_from_slice(&self.free_bitmap_block.to_le_bytes());
105        buf[52..56].copy_from_slice(&self.free_bitmap_blocks.to_le_bytes());
106        buf[56..64].copy_from_slice(&self.free_block_count.to_le_bytes());
107        buf[64..72].copy_from_slice(&self.next_inode_num.to_le_bytes());
108        buf[72..80].copy_from_slice(&self.created_at.to_le_bytes());
109        buf[80..88].copy_from_slice(&self.mounted_at.to_le_bytes());
110        for (i, &blk) in self.inode_blocks.iter().enumerate() {
111            buf[88 + i * 8..88 + (i + 1) * 8].copy_from_slice(&blk.to_le_bytes());
112        }
113        buf[344..4092].copy_from_slice(&self.reserved);
114        buf[4092..4096].copy_from_slice(&self.checksum.to_le_bytes());
115        buf
116    }
117
118    pub fn from_bytes(buf: &[u8; BLOCK_SIZE]) -> Option<Self> {
119        let magic: [u8; 8] = buf[0..8].try_into().ok()?;
120        if magic != MAGIC {
121            return None;
122        }
123        let stored_checksum = u32::from_le_bytes(buf[4092..4096].try_into().ok()?);
124        let computed_checksum = crc32c(&buf[0..4092]);
125        if stored_checksum != computed_checksum {
126            return None;
127        }
128
129        let mut inode_blocks = [0u64; 32];
130        for (i, chunk) in buf[88..344].chunks_exact(8).enumerate() {
131            inode_blocks[i] = u64::from_le_bytes(chunk.try_into().ok()?);
132        }
133        let mut reserved = [0u8; 3748];
134        reserved.copy_from_slice(&buf[344..4092]);
135
136        Some(Self {
137            magic,
138            version: u32::from_le_bytes(buf[8..12].try_into().ok()?),
139            flags: u32::from_le_bytes(buf[12..16].try_into().ok()?),
140            sequence: u64::from_le_bytes(buf[16..24].try_into().ok()?),
141            block_size: u32::from_le_bytes(buf[24..28].try_into().ok()?),
142            total_blocks: u64::from_le_bytes(buf[28..36].try_into().ok()?),
143            root_inode_block: u64::from_le_bytes(buf[36..44].try_into().ok()?),
144            free_bitmap_block: u64::from_le_bytes(buf[44..52].try_into().ok()?),
145            free_bitmap_blocks: u32::from_le_bytes(buf[52..56].try_into().ok()?),
146            free_block_count: u64::from_le_bytes(buf[56..64].try_into().ok()?),
147            next_inode_num: u64::from_le_bytes(buf[64..72].try_into().ok()?),
148            created_at: u64::from_le_bytes(buf[72..80].try_into().ok()?),
149            mounted_at: u64::from_le_bytes(buf[80..88].try_into().ok()?),
150            inode_blocks,
151            reserved,
152            checksum: stored_checksum,
153        })
154    }
155
156    pub fn update_checksum(&mut self) {
157        let bytes = self.to_bytes();
158        self.checksum = crc32c(&bytes[0..4092]);
159    }
160}

162/// Extent mapping contiguous logical file blocks to contiguous physical disk blocks.
163#[repr(C)]
164#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
165pub struct Extent {
166    pub logical_block: u32,
167    pub block_count: u32,
168    pub physical_block: u64,
169}
170
171impl Extent {
172    pub const SIZE: usize = 16;
173
174    pub fn to_bytes(&self) -> [u8; Self::SIZE] {
175        let mut b = [0u8; Self::SIZE];
176        b[0..4].copy_from_slice(&self.logical_block.to_le_bytes());
177        b[4..8].copy_from_slice(&self.block_count.to_le_bytes());
178        b[8..16].copy_from_slice(&self.physical_block.to_le_bytes());
179        b
180    }
181
182    pub fn from_bytes(b: &[u8; Self::SIZE]) -> Self {
183        Self {
184            logical_block: u32::from_le_bytes(b[0..4].try_into().unwrap()),
185            block_count: u32::from_le_bytes(b[4..8].try_into().unwrap()),
186            physical_block: u64::from_le_bytes(b[8..16].try_into().unwrap()),
187        }
188    }
189}

191/// 512-byte packed Inode structure.
192#[repr(C)]
193#[derive(Clone, Copy, Debug)]
194pub struct InodeRecord {
195    pub inode_num: u64,
196    pub mode: u16,
197    pub flags: u16,
198    pub nlink: u32,
199    pub uid: u32,
200    pub gid: u32,
201    pub size: u64,
202    pub atime: u64,
203    pub mtime: u64,
204    pub ctime: u64,
205    pub extent_count: u32,
206    pub inline_size: u32,
207    pub payload: [u8; INODE_PAYLOAD_SIZE],
208    pub checksum: u32,
209}
210
211impl InodeRecord {
212    pub fn new(inode_num: u64, mode: u16) -> Self {
213        let mut record = Self {
214            inode_num,
215            mode,
216            flags: INODE_FLAG_INLINE,
217            nlink: 1,
218            uid: 0,
219            gid: 0,
220            size: 0,
221            atime: 0,
222            mtime: 0,
223            ctime: 0,
224            extent_count: 0,
225            inline_size: 0,
226            payload: [0u8; INODE_PAYLOAD_SIZE],
227            checksum: 0,
228        };
229        record.update_checksum();
230        record
231    }
232
233    pub fn to_bytes(&self) -> [u8; INODE_SIZE] {
234        let mut buf = [0u8; INODE_SIZE];
235        buf[0..8].copy_from_slice(&self.inode_num.to_le_bytes());
236        buf[8..10].copy_from_slice(&self.mode.to_le_bytes());
237        buf[10..12].copy_from_slice(&self.flags.to_le_bytes());
238        buf[12..16].copy_from_slice(&self.nlink.to_le_bytes());
239        buf[16..20].copy_from_slice(&self.uid.to_le_bytes());
240        buf[20..24].copy_from_slice(&self.gid.to_le_bytes());
241        buf[24..32].copy_from_slice(&self.size.to_le_bytes());
242        buf[32..40].copy_from_slice(&self.atime.to_le_bytes());
243        buf[40..48].copy_from_slice(&self.mtime.to_le_bytes());
244        buf[48..56].copy_from_slice(&self.ctime.to_le_bytes());
245        buf[56..60].copy_from_slice(&self.extent_count.to_le_bytes());
246        buf[60..64].copy_from_slice(&self.inline_size.to_le_bytes());
247        buf[64..64 + INODE_PAYLOAD_SIZE].copy_from_slice(&self.payload);
248        buf[508..512].copy_from_slice(&self.checksum.to_le_bytes());
249        buf
250    }
251
252    pub fn from_bytes(buf: &[u8; INODE_SIZE]) -> Option<Self> {
253        let stored_checksum = u32::from_le_bytes(buf[508..512].try_into().ok()?);
254        let computed_checksum = crc32c(&buf[0..508]);
255        if stored_checksum != computed_checksum {
256            return None;
257        }
258
259        let mut payload = [0u8; INODE_PAYLOAD_SIZE];
260        payload.copy_from_slice(&buf[64..64 + INODE_PAYLOAD_SIZE]);
261
262        Some(Self {
263            inode_num: u64::from_le_bytes(buf[0..8].try_into().ok()?),
264            mode: u16::from_le_bytes(buf[8..10].try_into().ok()?),
265            flags: u16::from_le_bytes(buf[10..12].try_into().ok()?),
266            nlink: u32::from_le_bytes(buf[12..16].try_into().ok()?),
267            uid: u32::from_le_bytes(buf[16..20].try_into().ok()?),
268            gid: u32::from_le_bytes(buf[20..24].try_into().ok()?),
269            size: u64::from_le_bytes(buf[24..32].try_into().ok()?),
270            atime: u64::from_le_bytes(buf[32..40].try_into().ok()?),
271            mtime: u64::from_le_bytes(buf[40..48].try_into().ok()?),
272            ctime: u64::from_le_bytes(buf[48..56].try_into().ok()?),
273            extent_count: u32::from_le_bytes(buf[56..60].try_into().ok()?),
274            inline_size: u32::from_le_bytes(buf[60..64].try_into().ok()?),
275            payload,
276            checksum: stored_checksum,
277        })
278    }
279
280    pub fn update_checksum(&mut self) {
281        let bytes = self.to_bytes();
282        self.checksum = crc32c(&bytes[0..508]);
283    }
284
285    pub fn is_inline(&self) -> bool {
286        self.flags & INODE_FLAG_INLINE != 0
287    }
288
289    pub fn is_file(&self) -> bool {
290        self.mode == INODE_MODE_FILE
291    }
292
293    pub fn is_dir(&self) -> bool {
294        self.mode == INODE_MODE_DIR
295    }
296}

298/// Directory entry for naming hierarchy.
299#[repr(C)]
300#[derive(Clone, Copy, Debug)]
301pub struct DirEntryRecord {
302    pub inode_num: u64,
303    pub file_type: u8,
304    pub name_len: u8,
305    pub name: [u8; MAX_NAME_LEN],
306}
307
308impl DirEntryRecord {
309    pub const SIZE: usize = 8 + 1 + 1 + MAX_NAME_LEN; // 258 bytes
310
311    pub fn new(inode_num: u64, file_type: u8, name_str: &str) -> Option<Self> {
312        let bytes = name_str.as_bytes();
313        if bytes.is_empty() || bytes.len() > MAX_NAME_LEN {
314            return None;
315        }
316        let mut name = [0u8; MAX_NAME_LEN];
317        name[..bytes.len()].copy_from_slice(bytes);
318        Some(Self {
319            inode_num,
320            file_type,
321            name_len: bytes.len() as u8,
322            name,
323        })
324    }
325
326    pub fn name_as_str(&self) -> &str {
327        core::str::from_utf8(&self.name[..self.name_len as usize]).unwrap_or("")
328    }
329
330    pub fn to_bytes(&self) -> [u8; Self::SIZE] {
331        let mut b = [0u8; Self::SIZE];
332        b[0..8].copy_from_slice(&self.inode_num.to_le_bytes());
333        b[8] = self.file_type;
334        b[9] = self.name_len;
335        b[10..10 + MAX_NAME_LEN].copy_from_slice(&self.name);
336        b
337    }
338
339    pub fn from_bytes(b: &[u8; Self::SIZE]) -> Option<Self> {
340        let name_len = b[9] as usize;
341        if name_len > MAX_NAME_LEN {
342            return None;
343        }
344        let mut name = [0u8; MAX_NAME_LEN];
345        name.copy_from_slice(&b[10..10 + MAX_NAME_LEN]);
346        Some(Self {
347            inode_num: u64::from_le_bytes(b[0..8].try_into().ok()?),
348            file_type: b[8],
349            name_len: name_len as u8,
350            name,
351        })
352    }
