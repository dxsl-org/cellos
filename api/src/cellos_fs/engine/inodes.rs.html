1//! Inode-number reuse and all-or-nothing staging.
2//!
3//! The inode table holds 256 numbers. Without reuse, every create (including
4//! each D03 save stage) consumes one for good and the volume refuses creates
5//! forever once the high-water mark is reached.
6//!
7//! Reuse must not reintroduce ABA for `(inode_num, generation)` observations:
8//! a retired number keeps a tombstone record (mode `INODE_MODE_UNUSED`) whose
9//! generation is past every state the live inode had, and a reused inode
10//! starts after the tombstone. On-disk format is unchanged; older volumes,
11//! whose unlinked inodes kept their last live record, are scanned at open.
12
13use super::CellosFs;
14use crate::allocator::BitmapAllocator;
15use crate::disk::{BlockDevice, FsError};
16use crate::format::{
17    InodeRecord, Superblock, BLOCK_SIZE, INODES_PER_BLOCK, INODE_MODE_UNUSED, INODE_SIZE,
18};
19use crate::inode::DirPayload;
20use alloc::collections::BTreeMap;
21use alloc::vec;
22use alloc::vec::Vec;
23
24/// In-memory state a request may stage before `sync`.
25struct Staged {
26    active_sb: Superblock,
27    allocator: BitmapAllocator,
28    dirty_inodes: BTreeMap<u64, InodeRecord>,
29    free_inodes: BTreeMap<u64, u64>,
30}
31
32impl<D: BlockDevice> CellosFs<D> {
33    /// Run `f`; if it fails, restore every in-memory change it staged.
34    ///
35    /// Before `sync` publishes a superblock, a request only changes memory and
36    /// blocks free in the committed bitmap, so after a refusal (NotFound,
37    /// NoSpace, a full directory, …) the volume is exactly as before and can
38    /// keep serving. A storage fault ([`FsError::is_storage_fault`]) may have
39    /// written in-place blocks; callers must stop using the instance then.
40    pub fn atomically<R>(
41        &mut self,
42        f: impl FnOnce(&mut Self) -> Result<R, FsError>,
43    ) -> Result<R, FsError> {
44        let staged = Staged {
45            active_sb: self.active_sb,
46            allocator: self.allocator.clone(),
47            dirty_inodes: self.dirty_inodes.clone(),
48            free_inodes: self.free_inodes.clone(),
49        };
50        let result = f(self);
51        if result.is_err() {
52            self.active_sb = staged.active_sb;
53            self.allocator = staged.allocator;
54            self.dirty_inodes = staged.dirty_inodes;
55            self.free_inodes = staged.free_inodes;
56        }
57        result
58    }
59
60    /// Stage a tombstone for the unlinked `record` (its latest, possibly
61    /// unsynced state) and make its number reusable.
62    pub(super) fn retire_inode(&mut self, record: InodeRecord) {
63        let mut tombstone = InodeRecord::new(record.inode_num, INODE_MODE_UNUSED);
64        tombstone.generation = record.generation;
65        self.put_inode(tombstone);
66        let generation = self.dirty_inodes[&record.inode_num].generation;
67        self.free_inodes.insert(record.inode_num, generation);
68    }
69
70    /// Numbers below `next_inode_num` that no directory reaches, with the
71    /// generation of their last on-disk record. An unreadable table block or a
72    /// malformed directory yields no candidates, and a number without a valid
73    /// record is skipped: reuse needs proof the number is unreferenced and a
74    /// generation to continue from.
75    pub(super) fn scan_free_inodes(&mut self) -> BTreeMap<u64, u64> {
76        self.try_scan_free_inodes().unwrap_or_default()
77    }
78
79    fn try_scan_free_inodes(&mut self) -> Result<BTreeMap<u64, u64>, FsError> {
80        // `validate_geometry` bounds this by the 256-entry table + 1.
81        let limit =
82            usize::try_from(self.active_sb.next_inode_num).map_err(|_| FsError::Corrupted)?;
83        let mut table: Vec<Option<InodeRecord>> = vec![None; limit];
84        let blocks = self.active_sb.inode_blocks;
85        for (slot, block) in blocks.into_iter().enumerate() {
86            let first = slot * INODES_PER_BLOCK + 1;
87            if block == 0 || first >= limit {
88                continue;
89            }
90            let mut buf = [0u8; BLOCK_SIZE];
91            self.disk.read_block(block, &mut buf)?;
92            for (idx, bytes) in buf.chunks_exact(INODE_SIZE).enumerate() {
93                let num = first + idx;
94                if num >= limit {
95                    break;
96                }
97                let bytes: &[u8; INODE_SIZE] = bytes.try_into().map_err(|_| FsError::Corrupted)?;
98                table[num] = InodeRecord::from_bytes(bytes).filter(|r| r.inode_num == num as u64);
99            }
100        }
101        let mut reachable = vec![false; limit];
102        reachable[1] = true;
103        let mut pending = vec![1usize];
104        while let Some(num) = pending.pop() {
105            let dir = table[num].ok_or(FsError::Corrupted)?;
106            if !dir.is_dir() {
107                return Err(FsError::Corrupted);
108            }
109            self.validate_inode(&dir)?;
110            for entry in DirPayload::list(&dir.payload, dir.extent_count as usize) {
111                let child = usize::try_from(entry.inode_num).map_err(|_| FsError::Corrupted)?;
112                if child >= limit {
113                    return Err(FsError::Corrupted);
114                }
115                if !reachable[child] {
116                    reachable[child] = true;
117                    if entry.file_type == 2 {
118                        pending.push(child);
119                    }
120                }
121            }
122        }
123        let mut free = BTreeMap::new();
124        for num in 2..limit {
125            if let (false, Some(record)) = (reachable[num], table[num]) {
126                free.insert(num as u64, record.generation);
127            }
128        }
129        Ok(free)
130    }
131}
