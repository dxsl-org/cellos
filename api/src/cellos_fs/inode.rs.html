9/// Read an Inode from disk given its block and index within that block.
10pub fn read_inode<D: BlockDevice>(
11    disk: &mut D,
12    block: u64,
13    index: usize,
14) -> Result<InodeRecord, FsError> {
15    if index >= INODES_PER_BLOCK {
16        return Err(FsError::Corrupted);
17    }
18    let mut block_buf = [0u8; BLOCK_SIZE];
19    disk.read_block(block, &mut block_buf)?;
20
21    let offset = index * INODE_SIZE;
22    let mut inode_bytes = [0u8; INODE_SIZE];
23    inode_bytes.copy_from_slice(&block_buf[offset..offset + INODE_SIZE]);
24
25    InodeRecord::from_bytes(&inode_bytes).ok_or(FsError::Corrupted)
26}
27
28/// Write an Inode to disk given its block and index within that block.
29pub fn write_inode<D: BlockDevice>(
30    disk: &mut D,
31    block: u64,
32    index: usize,
33    record: &mut InodeRecord,
34) -> Result<(), FsError> {
35    if index >= INODES_PER_BLOCK {
36        return Err(FsError::Corrupted);
37    }
38    let mut block_buf = [0u8; BLOCK_SIZE];
39    disk.read_block(block, &mut block_buf)?;
40
41    record.update_checksum();
42    let inode_bytes = record.to_bytes();
43
44    let offset = index * INODE_SIZE;
45    block_buf[offset..offset + INODE_SIZE].copy_from_slice(&inode_bytes);
46
47    disk.write_block(block, &block_buf)
48}

50/// Helper for directory payload parsing (inline entries).
51pub struct DirPayload;
52
53impl DirPayload {
54    pub fn list(payload: &[u8; INODE_PAYLOAD_SIZE], count: usize) -> Vec<DirEntryRecord> {
55        let mut list = Vec::new();
56        let entry_size = DirEntryRecord::SIZE;
57        let max_entries = INODE_PAYLOAD_SIZE / entry_size;
58        let limit = count.min(max_entries);
59
60        for i in 0..limit {
61            let offset = i * entry_size;
62            let mut buf = [0u8; DirEntryRecord::SIZE];
63            buf.copy_from_slice(&payload[offset..offset + entry_size]);
64            if let Some(entry) = DirEntryRecord::from_bytes(&buf) {
65                list.push(entry);
66            }
67        }
68        list
69    }
70
71    pub fn find(
72        payload: &[u8; INODE_PAYLOAD_SIZE],
73        count: usize,
74        name: &str,
75    ) -> Option<DirEntryRecord> {
76        let entries = Self::list(payload, count);
77        entries.into_iter().find(|e| e.name_as_str() == name)
78    }
79
80    pub fn insert(
81        payload: &mut [u8; INODE_PAYLOAD_SIZE],
82        count: &mut usize,
83        entry: &DirEntryRecord,
84    ) -> Result<(), FsError> {
85        let entry_size = DirEntryRecord::SIZE;
86        let max_entries = INODE_PAYLOAD_SIZE / entry_size;
87        if *count >= max_entries {
88            return Err(FsError::NoSpace); // Need indirect block if exceeded
89        }
90
91        let offset = (*count) * entry_size;
92        let bytes = entry.to_bytes();
93        payload[offset..offset + entry_size].copy_from_slice(&bytes);
94        *count += 1;
95        Ok(())
96    }
97
98    pub fn remove(
99        payload: &mut [u8; INODE_PAYLOAD_SIZE],
100        count: &mut usize,
101        name: &str,
102    ) -> Result<u64, FsError> {
103        let entry_size = DirEntryRecord::SIZE;
104        let mut entries = Self::list(payload, *count);
105        let pos = entries
106            .iter()
107            .position(|e| e.name_as_str() == name)
108            .ok_or(FsError::NotFound)?;
109
110        let removed_inode = entries[pos].inode_num;
111        entries.swap_remove(pos);
112
113        // Re-serialize
114        *count = entries.len();
115        payload.fill(0);
116        for (i, entry) in entries.iter().enumerate() {
117            let offset = i * entry_size;
118            payload[offset..offset + entry_size].copy_from_slice(&entry.to_bytes());
119        }
120
121        Ok(removed_inode)
122    }
123}
