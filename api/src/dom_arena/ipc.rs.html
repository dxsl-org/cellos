2//! One active, caller-bound JS context and bounded ordered message transport.
3use alloc::string::String;
4use alloc::vec::Vec;
5use crate::{JsContext, OcelJsCommand, OcelJsReply, OcelJsRequest, OcelJsResponse,
6    OCEL_JS_CHUNK_SIZE, OCEL_JS_MAX_TRANSFER};
7
8#[derive(Default)]
9pub struct JsServiceSession {
10    owner: Option<(usize, u64)>,
11    incoming: Vec<u8>,
12    expected: Option<usize>,
13    outgoing: Vec<u8>,
14    document_ready: bool,
15}
16
17impl JsServiceSession {
18    pub fn handle<C: JsContext>(&mut self, caller: usize, request: OcelJsRequest, ctx: &mut C) -> OcelJsReply {
19        if matches!(request, OcelJsRequest::Ping) {
20            return OcelJsReply::Ready;
21        }
22        if let OcelJsRequest::OpenContext { context } = request {
23            ctx.reset();
24            self.owner = Some((caller, context));
25            self.incoming.clear();
26            self.expected = None;
27            self.outgoing.clear();
28            self.document_ready = false;
29            return OcelJsReply::Ack;
30        }
31        let context = match &request {
32            OcelJsRequest::Begin { context, .. } | OcelJsRequest::Chunk { context, .. }
33            | OcelJsRequest::Commit { context } | OcelJsRequest::ReadResponse { context, .. } => *context,
34            OcelJsRequest::OpenContext { .. } | OcelJsRequest::Ping => unreachable!(),
35        };
36        if self.owner != Some((caller, context)) {
37            return Self::error("JS context is not active; explicitly open and synchronize it");
38        }
39        match request {
40            OcelJsRequest::Begin { total, .. } => {
41                if total == 0 || total > OCEL_JS_MAX_TRANSFER { return Self::error("JS request exceeds transfer limit"); }
42                self.incoming.clear();
43                self.expected = Some(total);
44                self.outgoing.clear();
45                OcelJsReply::Ack
46            }
47            OcelJsRequest::Chunk { offset, data, .. } => {
48                if self.expected.is_none() || offset != self.incoming.len() || data.is_empty()
49                    || data.len() > OCEL_JS_CHUNK_SIZE
50                    || offset.saturating_add(data.len()) > self.expected.unwrap_or(0) {
51                    return Self::error("JS upload chunk is out of order or out of bounds");
52                }
53                self.incoming.extend(data);
54                OcelJsReply::Ack
55            }
56            OcelJsRequest::Commit { .. } => {
57                if self.expected != Some(self.incoming.len()) { return Self::error("JS upload is incomplete"); }
58                self.expected = None;
59                let command = postcard::from_bytes::<OcelJsCommand>(&self.incoming);
60                self.incoming.clear();
61                let result = match command {
62                    Ok(OcelJsCommand::SyncDocument { arena, title }) => {
63                        if !arena.validate() { return Self::error("Invalid DOM snapshot topology"); }
64                        let result = ctx.sync_document(&arena, &title).map(|_| String::from("document_synced"));
65                        if result.is_ok() { self.document_ready = true; }
66                        result
67                    }
68                    Ok(OcelJsCommand::Eval { script }) if self.document_ready => ctx.eval(&script),
69                    Ok(OcelJsCommand::DispatchEvent { event }) if self.document_ready =>
70                        ctx.dispatch_event(&event).map(|_| String::from("event_dispatched")),
71                    Ok(_) => return Self::error("Synchronize the document before executing JavaScript"),
72                    Err(_) => return Self::error("Malformed JS command"),
73                };
74                let mutations = ctx.take_mutations();
75                let response = match result {
76                    Ok(result_repr) => OcelJsResponse::Success { mutations, result_repr },
77                    Err(e) => OcelJsResponse::Error { message: e.message, line: e.line, mutations },
78                };
79                self.outgoing = match postcard::to_allocvec(&response) {
80                    Ok(bytes) if bytes.len() <= OCEL_JS_MAX_TRANSFER => bytes,
81                    _ => {
82                        // Never continue with JS/native state diverged after losing mutations.
83                        self.owner = None;
84                        self.document_ready = false;
85                        ctx.reset();
86                        return Self::error("JS response exceeds transfer limit; context invalidated");
87                    }
88                };
89                self.response_chunk(0)
90            }
91            OcelJsRequest::ReadResponse { offset, .. } => self.response_chunk(offset),
92            OcelJsRequest::OpenContext { .. } | OcelJsRequest::Ping => unreachable!(),
93        }
94    }
95
96    fn response_chunk(&self, offset: usize) -> OcelJsReply {
97        if offset >= self.outgoing.len() { return Self::error("JS response offset is out of bounds"); }
98        let end = offset.saturating_add(OCEL_JS_CHUNK_SIZE).min(self.outgoing.len());
99        OcelJsReply::ResponseChunk { offset, total: self.outgoing.len(), data: self.outgoing[offset..end].to_vec() }
100    }
101
102    pub fn error(message: &str) -> OcelJsReply { OcelJsReply::Error { message: String::from(message) } }
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108    use crate::{DocumentArena, DomEvent, DomMutation, JsError};
109    #[derive(Default)]
110    struct Context { resets: usize, evals: usize }
111    impl JsContext for Context {
112        fn sync_document(&mut self, _: &DocumentArena, _: &str) -> Result<(), JsError> { Ok(()) }
113        fn eval(&mut self, script: &str) -> Result<String, JsError> { self.evals += 1; Ok(String::from(script)) }
114        fn dispatch_event(&mut self, _: &DomEvent) -> Result<(), JsError> { Ok(()) }
115        fn take_mutations(&mut self) -> Vec<DomMutation> { Vec::new() }
116        fn reset(&mut self) { self.resets += 1; }
117    }
118    #[test]
119    fn readiness_probe_preserves_another_callers_inflight_upload() {
120        let mut session = JsServiceSession::default();
121        let mut ctx = Context::default();
122        session.handle(1, OcelJsRequest::OpenContext { context: 7 }, &mut ctx);
123        session.handle(1, OcelJsRequest::Begin { context: 7, total: 2 }, &mut ctx);
124        session.handle(1, OcelJsRequest::Chunk { context: 7, offset: 0, data: alloc::vec![0] }, &mut ctx);
125        session.handle(2, OcelJsRequest::Ping, &mut ctx);
126        assert!(matches!(
127            session.handle(1, OcelJsRequest::Chunk { context: 7, offset: 1, data: alloc::vec![1] }, &mut ctx),
128            OcelJsReply::Ack
129        ));
130        assert!(matches!(
131            session.handle(2, OcelJsRequest::Commit { context: 7 }, &mut ctx),
132            OcelJsReply::Error { .. }
133        ));
134        assert_eq!(ctx.resets, 1);
135    }
136    #[test]
137    fn context_is_bound_to_caller_and_explicit_token() {
138        let mut session = JsServiceSession::default();
139        let mut ctx = Context::default();
140        assert!(matches!(session.handle(1, OcelJsRequest::OpenContext { context: 7 }, &mut ctx), OcelJsReply::Ack));
141        for (caller, context) in [(2, 7), (1, 8)] {
142            assert!(matches!(session.handle(caller, OcelJsRequest::Begin { context, total: 1 }, &mut ctx), OcelJsReply::Error { .. }));
143        }
144        session.handle(2, OcelJsRequest::OpenContext { context: 7 }, &mut ctx);
145        assert_eq!(ctx.resets, 2);
146        assert!(matches!(session.handle(1, OcelJsRequest::Commit { context: 7 }, &mut ctx), OcelJsReply::Error { .. }));
147    }
148    #[test]
149    fn upload_rejects_gaps_truncation_and_oversize() {
150        let mut session = JsServiceSession::default();
151        let mut ctx = Context::default();
152        session.handle(1, OcelJsRequest::OpenContext { context: 1 }, &mut ctx);
153        assert!(matches!(session.handle(1, OcelJsRequest::Begin { context: 1, total: OCEL_JS_MAX_TRANSFER + 1 }, &mut ctx), OcelJsReply::Error { .. }));
154        session.handle(1, OcelJsRequest::Begin { context: 1, total: 2 }, &mut ctx);
155        assert!(matches!(session.handle(1, OcelJsRequest::Chunk { context: 1, offset: 1, data: alloc::vec![1] }, &mut ctx), OcelJsReply::Error { .. }));
156        assert!(matches!(session.handle(1, OcelJsRequest::Commit { context: 1 }, &mut ctx), OcelJsReply::Error { .. }));
157        assert_eq!(ctx.evals, 0);
158    }
