9pub const IPC_BYTES: usize = 4096;
10pub const PIXEL_CHUNK_BYTES: usize = 3072;
11pub const MAX_DOCUMENT_BYTES: usize = 2 * 1024 * 1024;
12pub const MAX_PAGE_PIXELS: usize = 1024 * 1024;
13/// Renewable idle lifetime; expired handles never become valid for a new owner.
14pub const DOCUMENT_IDLE_TICKS: u64 = 6000;
15
16/// Shared lifecycle policy independent of the native renderer. No caller may
17/// renew, close or reclaim another nonexpired document.
18pub struct DocumentOwner { caller: usize, handle: u32, activity: u64 }
19impl DocumentOwner {
20    pub fn new(caller: usize, handle: u32, now: u64) -> Self { Self { caller, handle, activity: now } }
21    pub fn matches(&self, caller: usize, handle: u32) -> bool { self.caller == caller && self.handle == handle }
22    pub fn expired(&self, now: u64) -> bool { now.wrapping_sub(self.activity) >= DOCUMENT_IDLE_TICKS }
23    pub fn renew(&mut self, caller: usize, handle: u32, now: u64) -> bool {
24        if !self.matches(caller, handle) || self.expired(now) { return false; }
25        self.activity = now;
26        true
27    }
28}

30#[cfg(test)]
31mod tests {
32    use super::*;
33    #[test]
34    fn foreign_caller_and_stale_handle_cannot_renew_document() {
35        let mut owner = DocumentOwner::new(7, 1, 0);
36        assert!(!owner.renew(8, 1, 50));
37        assert!(!owner.renew(7, 2, 50));
38        assert!(owner.renew(7, 1, 50));
39        assert!(!owner.expired(50 + DOCUMENT_IDLE_TICKS - 1));
40        assert!(owner.expired(50 + DOCUMENT_IDLE_TICKS));
41        assert!(!owner.renew(7, 1, 50 + DOCUMENT_IDLE_TICKS));
42    }
43    #[test]
44    fn replacement_owner_never_accepts_previous_handle() {
45        let owner = DocumentOwner::new(8, 2, 100);
46        assert!(!owner.matches(7, 1));
47        assert!(!owner.matches(8, 1));
48        assert!(owner.matches(8, 2));
49    }

52/// Document handles are scoped to the sender TID. Closing a document also drops
53/// its cached raster. Pixels are straight, opaque BGRA8888, row-major.
54#[derive(Debug, Serialize, Deserialize)]
55pub enum Request {
56    Open { path: String },
57    RenderPage { document: u32, page: u32, max_width: u32, max_height: u32 },
58    ReadPixels { document: u32, offset: u32, length: u16 },
59    Close { document: u32 },
60    /// Readiness probe; does not open a document or allocate a raster.
61    Ping,
62}
63
64#[derive(Debug, Serialize, Deserialize)]
65pub enum Response {
66    Opened { document: u32, pages: u32 },
67    Rendered { width: u32, height: u32, bytes: u32 },
68    Pixels { offset: u32, bytes: Vec<u8> },
69    Closed,
70    Error { message: String },
71    Ready,
72}
