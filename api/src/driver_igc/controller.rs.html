1//! Intel I225/I226 controller initialization and Tx/Rx data path.
2
3use crate::dma_layout::{
4    for_each_initial_dma_program, try_init_array, with_authorized_dma_layout, DmaIovas, DmaSlot,
5    InitialDmaProgram, RX_SLOTS, TX_SLOTS,
6};
7use crate::phy;
8use crate::regs::*;
9use crate::rings::{self, RxDesc, TxDesc};
10use core::sync::atomic::{compiler_fence, Ordering};
11use ostd::dma::DmaBuf;

13use types::{ViError, ViResult};

15pub use crate::rings::BUF_SIZE;
16
17const RESET_POLLS: u32 = 1_000_000;
18const TX_POLLS: u32 = 1_000_000;
19const LINK_POLLS: u32 = 6_000_000;
20
21pub struct IgcController {
22    mmio: MmioRegion,
23    tx_ring: DmaBuf,
24    rx_ring: DmaBuf,
25    tx_bufs: [DmaBuf; TX_SLOTS],
26    rx_bufs: [DmaBuf; RX_SLOTS],
27    dma_iovas: DmaIovas,
28    tx_next: usize,
29    rx_head: usize,
30    pub mac: [u8; 6],
31    pub link_up: bool,
32}
33
34unsafe impl Send for IgcController {}
35
36impl IgcController {
37    pub fn new(mmio: MmioRegion, bdf: u32) -> ViResult<Self> {
38        mmio.write_u32(CTRL, CTRL_RST)?;
39        let mut polls = 0u32;
40        while mmio.read_u32(CTRL)? & CTRL_RST != 0 {
41            polls += 1;
42            if polls > RESET_POLLS {
43                return Err(ViError::IO);
44            }
45            core::hint::spin_loop();
46        }
47
48        mmio.write_u32(EEER, 0)?;
49        mmio.write_u32(EEE_SU, 0)?;
50        phy::restart_autoneg(&mmio)?;
51
52        mmio.write_u32(CTRL, CTRL_SLU | CTRL_ASDE)?;
53        mmio.write_u32(IMC, u32::MAX)?;
54        let _ = mmio.read_u32(ICR)?;
55
56        let mac = match (mmio.read_u32(RAL0)?, mmio.read_u32(RAH0)?) {
57            (ral, rah) if rah & RAH_AV != 0 && ral != 0 => [
58                ral as u8,
59                (ral >> 8) as u8,
60                (ral >> 16) as u8,
61                (ral >> 24) as u8,
62                rah as u8,
63                (rah >> 8) as u8,
64            ],
65            _ => {
66                let (w0, w1, w2) = (
67                    nvm_read(&mmio, 0)?,
68                    nvm_read(&mmio, 1)?,
69                    nvm_read(&mmio, 2)?,
70                );
71                [
72                    w0 as u8,
73                    (w0 >> 8) as u8,
74                    w1 as u8,
75                    (w1 >> 8) as u8,
76                    w2 as u8,
77                    (w2 >> 8) as u8,
78                ]
79            }
80        };
81        if mac == [0u8; 6] {
82            return Err(ViError::IO);
83        }
84
85        let ral_val = u32::from_le_bytes([mac[0], mac[1], mac[2], mac[3]]);
86        let rah_val = RAH_AV | (u16::from_le_bytes([mac[4], mac[5]]) as u32);
87        mmio.write_u32(RAL0, ral_val)?;
88        mmio.write_u32(RAH0, rah_val)?;
89        for index in 0..MTA_ENTRIES {
90            mmio.write_u32(MTA + index * 4, 0)?;
91        }
92
93        let tx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
94        let tx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
95        let rx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
96        let rx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
97
98        with_authorized_dma_layout(
99            (mmio, tx_ring, rx_ring, tx_bufs, rx_bufs),
100            |r, s| {
101                match s {
102                    DmaSlot::TxRing => r.1.authorize(bdf),
103                    DmaSlot::TxBuffer(i) => r.3[i].authorize(bdf),
104                    DmaSlot::RxRing => r.2.authorize(bdf),
105                    DmaSlot::RxBuffer(i) => r.4[i].authorize(bdf),
106                }
107                .map_err(|_| ViError::PermissionDenied)
108            },
109            |(mmio, tx_ring, rx_ring, tx_bufs, rx_bufs), dma_iovas| {
110                rings::clear(&tx_ring);
111                rings::clear(&rx_ring);
112                let mut ctrl = IgcController {
113                    mmio,
114                    tx_ring,
115                    rx_ring,
116                    tx_bufs,
117                    rx_bufs,
118                    dma_iovas,
119                    tx_next: 0,
120                    rx_head: 0,
121                    mac,
122                    link_up: false,
123                };
124                for_each_initial_dma_program(&dma_iovas, |p| {
125                    let _ = ctrl.apply(p);
126                });
127                let _ = ctrl.finish_init();
128                Ok(ctrl)
129            },
130        )?
131    }
132
133    fn apply(&mut self, program: InitialDmaProgram) -> ViResult<()> {
134        match program {
135            InitialDmaProgram::TxRingBase(iova) => {
136                self.mmio.write_u32(TDBAL0, iova as u32)?;
137                self.mmio.write_u32(TDBAH0, (iova >> 32) as u32)?;
138                self.mmio.write_u32(TDLEN0, (TX_SLOTS * 16) as u32)?;
139                self.mmio.write_u32(TDH0, 0)?;
140                self.mmio.write_u32(TDT0, 0)?;
141            }
142            InitialDmaProgram::RxRingBase(iova) => {
143                self.mmio.write_u32(RDBAL0, iova as u32)?;
144                self.mmio.write_u32(RDBAH0, (iova >> 32) as u32)?;
145                self.mmio.write_u32(RDLEN0, (RX_SLOTS * 16) as u32)?;
146                self.mmio.write_u32(RDH0, 0)?;
147                self.mmio.write_u32(RDT0, 0)?;
148            }
149            InitialDmaProgram::RxBuffer { index, iova } => rings::write_rx_desc(
150                &self.rx_ring,
151                index,
152                RxDesc {
153                    buffer_addr: iova,
154                    status_error: 0,
155                    length: 0,
156                    vlan: 0,
157                },
158            ),
159        }
160        Ok(())
161    }
162
163    fn finish_init(&mut self) -> ViResult<()> {
164        self.mmio
165            .write_u32(SRRCTL0, SRRCTL_DESCTYPE_ADV_ONEBUF | 2)?;
166        self.mmio.write_u32(RXDCTL0, RXDCTL_ENABLE)?;
167        self.mmio.write_u32(TXDCTL0, TXDCTL_ENABLE)?;
168        self.mmio.write_u32(TIPG, TIPG_COPPER)?;
169        self.mmio
170            .write_u32(TCTL, TCTL_EN | TCTL_PSP | TCTL_CT | TCTL_COLD)?;
171        self.mmio.write_u32(RDT0, (RX_SLOTS - 1) as u32)?;
172
173        let mut polls = 0u32;
174        while self.mmio.read_u32(STATUS)? & STATUS_LU == 0 {
175            polls += 1;
176            if polls > LINK_POLLS {
177                break;
178            }
179            core::hint::spin_loop();
180        }
181        self.link_up = self.mmio.read_u32(STATUS)? & STATUS_LU != 0;
182        self.mmio
183            .write_u32(RCTL, RCTL_EN | RCTL_BAM | RCTL_UPE | RCTL_MPE | RCTL_SECRC)
184    }

186    pub fn send_frame(&mut self, frame: &[u8]) -> ViResult<()> {
187        if frame.is_empty() || frame.len() > BUF_SIZE {
188            return Err(ViError::InvalidArgument);
189        }
190        let slot = self.tx_next;
191        let buf = &self.tx_bufs[slot];
192        unsafe {
193            core::ptr::copy_nonoverlapping(frame.as_ptr(), buf.virt(), frame.len());
194        }
195        compiler_fence(Ordering::Release);
196
197        let iova = self.dma_iovas.tx_bufs[slot];
198        rings::write_tx_desc(
199            &self.tx_ring,
200            slot,
201            TxDesc {
202                buffer_addr: iova,
203                cmd_type_len: rings::ADVTXD_DCMD_DEXT
204                    | rings::ADVTXD_DTYP_DATA
205                    | rings::TXD_CMD_EOP
206                    | rings::TXD_CMD_IFCS
207                    | rings::TXD_CMD_RS
208                    | (frame.len() as u32),
209                olinfo_status: (frame.len() as u32) << 16,
210            },
211        );
212        compiler_fence(Ordering::Release);
213        self.tx_next = (slot + 1) % TX_SLOTS;
214        self.mmio.write_u32(TDT0, self.tx_next as u32)?;
215
216        for _ in 0..TX_POLLS {
217            compiler_fence(Ordering::Acquire);
218            if rings::read_tx_desc(&self.tx_ring, slot).cmd_type_len & rings::TXD_STAT_DD != 0 {
219                return Ok(());
220            }
221            core::hint::spin_loop();
222        }
223        Err(ViError::IO)
224    }
225
226    pub fn poll_rx(&mut self, target: &mut [u8]) -> ViResult<usize> {
227        let slot = self.rx_head;
228        compiler_fence(Ordering::Acquire);
229        let desc = rings::read_rx_desc(&self.rx_ring, slot);
230        if desc.status_error & rings::RXD_STAT_DD == 0 {
231            return Ok(0);
232        }
233        let len = (desc.length as usize).min(target.len());
234        let buf = &self.rx_bufs[slot];
235        unsafe {
236            core::ptr::copy_nonoverlapping(buf.virt() as *const u8, target.as_mut_ptr(), len);
237        }
238
239        rings::write_rx_desc(
240            &self.rx_ring,
241            slot,
242            RxDesc {
243                buffer_addr: self.dma_iovas.rx_bufs[slot],
244                status_error: 0,
245                length: 0,
246                vlan: 0,
247            },
248        );
249        compiler_fence(Ordering::Release);
250        self.rx_head = (slot + 1) % RX_SLOTS;
251        self.mmio.write_u32(RDT0, slot as u32)?;
252        Ok(len)
253    }
