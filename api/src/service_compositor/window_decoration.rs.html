1//! Compositor-owned window frame geometry, hit testing, and rasterization.

3use api::display::Rect;

5mod raster;

7pub const FRAME: i32 = 4;
8pub const TITLE: i32 = 20;
9pub(super) const CONTROL: i32 = 16;
10pub(super) const INSET: i32 = 2;
11
12#[derive(Clone, Copy, PartialEq, Eq)]
13pub enum Control {
14    Close,
15    Maximize,
16    Minimize,
17}
18
19#[derive(Clone, Copy, PartialEq, Eq)]
20pub enum ResizeEdge {
21    North,
22    South,
23    West,
24    East,
25    NorthWest,
26    NorthEast,
27    SouthWest,
28    SouthEast,
29}
30
31#[derive(Clone, Copy, PartialEq, Eq)]
32pub enum Hit {
33    Control(Control),
34    Resize(ResizeEdge),
35    Title,
36    Content,
37}

39pub fn bounds(content: Rect) -> Rect {
40    let right = end(content.x, content.w).saturating_add(FRAME);
41    let frame_left = content.x.saturating_sub(FRAME);
42    let control_left = right.saturating_sub(INSET + CONTROL * 3);
43    let x = frame_left.min(control_left);
44    Rect {
45        x,
46        y: content.y.saturating_sub(FRAME.saturating_add(TITLE)),
47        w: right.saturating_sub(x) as u32,
48        h: content.h.saturating_add((FRAME * 2 + TITLE) as u32),
49    }
50}
51
52pub fn clip_to_screen(rect: Rect, width: u32, height: u32) -> Option<Rect> {
53    let x = rect.x.max(0);
54    let y = rect.y.max(0);
55    let right = end(rect.x, rect.w).min(width.min(i32::MAX as u32) as i32);
56    let bottom = end(rect.y, rect.h).min(height.min(i32::MAX as u32) as i32);
57    (right > x && bottom > y).then_some(Rect {
58        x,
59        y,
60        w: (right - x) as u32,
61        h: (bottom - y) as u32,
62    })
63}

65pub fn hit_test(content: Rect, x: i32, y: i32) -> Option<Hit> {
66    let outer = bounds(content);
67    if !contains(outer, x, y) {
68        return None;
69    }
70    for control in [Control::Close, Control::Maximize, Control::Minimize] {
71        if contains(control_rect(content, control), x, y) {
72            return Some(Hit::Control(control));
73        }
74    }
75    let left = x < content.x;
76    let right = x >= end(content.x, content.w);
77    let top = y < content.y.saturating_sub(TITLE);
78    let bottom = y >= end(content.y, content.h);
79    match (top, bottom, left, right) {
80        (true, _, true, _) => Some(Hit::Resize(ResizeEdge::NorthWest)),
81        (true, _, _, true) => Some(Hit::Resize(ResizeEdge::NorthEast)),
82        (_, true, true, _) => Some(Hit::Resize(ResizeEdge::SouthWest)),
83        (_, true, _, true) => Some(Hit::Resize(ResizeEdge::SouthEast)),
84        (true, _, _, _) => Some(Hit::Resize(ResizeEdge::North)),
85        (_, true, _, _) => Some(Hit::Resize(ResizeEdge::South)),
86        (_, _, true, _) => Some(Hit::Resize(ResizeEdge::West)),
87        (_, _, _, true) => Some(Hit::Resize(ResizeEdge::East)),
88        _ if y < content.y => Some(Hit::Title),
89        _ => Some(Hit::Content),
90    }
91}

94    raster::paint(pixels, width, height, content, dirty, active);

97pub(super) fn control_rect(content: Rect, control: Control) -> Rect {
98    let right = end(content.x, content.w)
99        .saturating_add(FRAME)
100        .saturating_sub(INSET);
101    let index = match control {
102        Control::Close => 0,
103        Control::Maximize => 1,
104        Control::Minimize => 2,
105    };
106    Rect {
107        x: right.saturating_sub(CONTROL * (index + 1)),
108        y: content.y.saturating_sub(TITLE).saturating_add(INSET),
109        w: CONTROL as u32,
110        h: CONTROL as u32,
111    }
112}
113
114fn contains(rect: Rect, x: i32, y: i32) -> bool {
115    x >= rect.x && x < end(rect.x, rect.w) && y >= rect.y && y < end(rect.y, rect.h)
116}
117
118pub(super) fn end(start: i32, extent: u32) -> i32 {
119    start.saturating_add(extent.min(i32::MAX as u32) as i32)
120}
