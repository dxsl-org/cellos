1//! Cell Frame geometry, hit testing, and rasterization (CellOS Linear).
2//!
3//! A Cell Tab (ADR-0027) sits on the top-left of the Cell Frame and holds only
4//! the app icon and the title, as in Haiku. The icon opens the window menu
5//! (maximize/restore, minimize, close); a double click on the title toggles
6//! maximize. Everything the compositor draws is 1px hairlines on the pixel grid with lightly rounded
7//! outer corners; a soft drop shadow lives in the invisible
8//! [`SHADOW_MARGIN`], and the innermost [`GRIP`] pixels of it resize.
9//!
10//! Vertical layout above the content: `TITLE` rows of tab, the last of which is
11//! the hairline shared by the tab and the frame body.
12

35/// Visible hairline around the content (sides and bottom).
36pub const BORDER: i32 = 1;
37/// Tab height above the content, including the tab's top hairline and the
38/// separator hairline shared with the frame body.
39pub const TITLE: i32 = 28;
40/// Band outside the visible frame that the drop shadow may paint: the shadow's
41/// 10px fade plus its 3px drop (palette.rs), with a pixel to spare.
42pub const SHADOW_MARGIN: i32 = 14;
43/// Invisible band outside the visible frame that resizes the window.
44pub const GRIP: i32 = 6;
45/// Corner radius of the tab, the frame body and floating panels.
46pub const RADIUS: u32 = PANEL_RADIUS;
47/// Square hit area of the app icon at the tab's leading edge; it opens the
48/// window menu. The 16px icon inside is drawn at 1px.
49pub(super) const CONTROL: i32 = 20;
50/// Inset of the icon box from the tab's leading hairline.
51pub(super) const CONTROL_PAD: i32 = 4;
52/// Length of the margin/border stretch that resizes diagonally near a corner.
53const CORNER_GRIP: i32 = 12;

55/// Content plus its hairline border; the top row is the tab separator.
56pub(super) fn body_rect(content: Rect) -> Rect {
57    Rect {
58        x: content.x.saturating_sub(BORDER),
59        y: content.y.saturating_sub(BORDER),
60        w: content.w.saturating_add((BORDER * 2) as u32),
61        h: content.h.saturating_add((BORDER * 2) as u32),
62    }
63}

65/// The two shapes that make up the visible window: tab and body.
66///
67/// The body's top corners are square where the tab continues straight into
68/// them: the left one unless the tab was slid along, the right one when the
69/// tab reaches the frame's right edge.
70pub(crate) fn shapes(content: Rect, tabs: Tabs) -> [RoundRect; 2] {
71    let tab = tab_rect(content, tabs);
72    let body = body_rect(content);
73    // The body's top corner under the tab is square: the tab continues into it.
74    let flush_left = tab.x <= body.x;
75    let flush_right = end(tab.x, tab.w) >= end(body.x, body.w);
76    [
77        RoundRect::new(
78            tab.x,
79            tab.y,
80            tab.w,
81            tab.h,
82            Radii {
83                top_left: RADIUS,
84                top_right: RADIUS,
85                ..Radii::default()
86            },
87        ),
88        RoundRect::new(
89            body.x,
90            body.y,
91            body.w,
92            body.h,
93            Radii {
94                top_left: if flush_left { 0 } else { RADIUS },
95                top_right: if flush_right { 0 } else { RADIUS },
96                bottom_right: RADIUS,
97                bottom_left: RADIUS,
98            },
99        ),
100    ]
101}

150/// Visible frame grown by `margin` on every side.
151fn around_frame(content: Rect, margin: i32) -> Rect {
152    let x = content.x.saturating_sub(BORDER + margin);
153    let right = end(content.x, content.w).saturating_add(BORDER + margin);
154    Rect {
155        x,
156        y: content.y.saturating_sub(TITLE + margin),
157        w: right.saturating_sub(x) as u32,
158        h: content
159            .h
160            .saturating_add((TITLE + BORDER + margin * 2) as u32),
161    }
162}

167/// Content rect of a window maximized on a `screen_w` x `screen_h` screen:
168/// its frame fills the screen above the taskbar.
169pub fn maximized_content(screen_w: u32, screen_h: u32) -> Rect {
170    content_within(Rect {
171        x: 0,
172        y: 0,
173        w: screen_w,
174        h: screen_h.saturating_sub(TASKBAR_H),
175    })
176}

178/// Content rect whose visible frame (tab and hairlines) exactly fills `area`.
179pub fn content_within(area: Rect) -> Rect {
180    Rect {
181        x: area.x.saturating_add(BORDER),
182        y: area.y.saturating_add(TITLE),
183        w: area.w.saturating_sub((BORDER * 2) as u32),
184        h: area.h.saturating_sub((TITLE + BORDER) as u32),
185    }
186}

188/// Columns hidden at the start and end of content row `row` so the content
189/// stays inside the rounded body outline.
190pub fn content_row_insets(content: Rect, tabs: Tabs, row: u32) -> (u32, u32) {
191    let reach = RADIUS.saturating_sub(BORDER as u32);
192    if row >= reach && row.saturating_add(reach) < content.h {
193        return (0, 0);
194    }
195    let body = shapes(content, tabs)[1];
196    let ((_, left_end), (right_start, _)) = body.outline(row.saturating_add(BORDER as u32));
197    let border = BORDER as u32;
198    (
199        left_end.saturating_sub(border),
200        body.w.saturating_sub(right_start).saturating_sub(border),
201    )
202}

204pub fn clip_to_screen(rect: Rect, width: u32, height: u32) -> Option<Rect> {
205    let x = rect.x.max(0);
206    let y = rect.y.max(0);
207    let right = end(rect.x, rect.w).min(width.min(i32::MAX as u32) as i32);
208    let bottom = end(rect.y, rect.h).min(height.min(i32::MAX as u32) as i32);
209    (right > x && bottom > y).then_some(Rect {
210        x,
211        y,
212        w: (right - x) as u32,
213        h: (bottom - y) as u32,
214    })
215}

217pub fn hit_test(content: Rect, tabs: Tabs, x: i32, y: i32) -> Option<Hit> {
218    // The shadow beyond the grip is see-through: clicks reach the window below.
219    if !contains(around_frame(content, GRIP), x, y) {
220        return None;
221    }
222    if contains(menu_button_rect(content, tabs), x, y) {
223        return Some(Hit::Menu);
224    }
225    let tab = tab_rect(content, tabs);
226    if contains(tab, x, y) {
227        let index = slot_at(content, tabs, x);
228        return Some(if index == tabs.selected {
229            Hit::Title
230        } else {
231            Hit::Tab(index)
232        });
233    }
234    let body = body_rect(content);
235    let (tab_end, body_right, body_bottom) =
236        (end(tab.x, tab.w), end(body.x, body.w), end(body.y, body.h));
237    // The strip beside the tab is see-through: clicks reach the window below.
238    if x >= tab_end && y < body.y {
239        return None;
240    }
241    let content_right = end(content.x, content.w);
242    let content_bottom = end(content.y, content.h);
243    if x >= content.x && x < content_right && y >= content.y && y < content_bottom {
244        return Some(Hit::Content);
245    }
246
247    let beside_tab = x >= tab_end;
248    let mut north = y < tab.y || (beside_tab && y < content.y);
249    let mut south = y >= content_bottom;
250    let mut west = x < content.x;
251    let mut east = x >= content_right;
252    if north || south {
253        west |= x < body.x + CORNER_GRIP;
254        east |= (south || beside_tab) && x >= body_right - CORNER_GRIP;
255    } else {
256        north |= (west && y < tab.y + CORNER_GRIP) || (east && y < body.y + CORNER_GRIP);
257        south |= y >= body_bottom - CORNER_GRIP;
258    }
259    let edge = match (north, south, west, east) {
260        (true, _, true, _) => ResizeEdge::NorthWest,
261        (true, _, _, true) => ResizeEdge::NorthEast,
262        (_, true, true, _) => ResizeEdge::SouthWest,
263        (_, true, _, true) => ResizeEdge::SouthEast,
264        (true, ..) => ResizeEdge::North,
265        (_, true, ..) => ResizeEdge::South,
266        (_, _, true, _) => ResizeEdge::West,
267        _ => ResizeEdge::East,
268    };
269    Some(Hit::Resize(edge))
270}

286fn contains(rect: Rect, x: i32, y: i32) -> bool {
287    x >= rect.x && x < end(rect.x, rect.w) && y >= rect.y && y < end(rect.y, rect.h)
288}

290pub(super) fn end(start: i32, extent: u32) -> i32 {
