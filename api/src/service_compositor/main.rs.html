5//! Compositor Service Cell.
6//!
7//! Manages a z-ordered set of cell surfaces, blends them into a screen
8//! framebuffer, and flushes dirty regions to the VirtIO GPU via the
9//! `GpuFlush` kernel syscall.
10//!
11//! Input routing: on startup the compositor registers as the input service's
12//! focus endpoint. All `InputEvent` IPC frames are dispatched to
13//! `input_handler`, which forwards keyboard events to the focused surface
14//! owner and updates the cursor position on mouse move.

41/// IPC opcode prefix byte identifying an input event from the input service.
42/// Must match `input_handler::INPUT_EVENT_OPCODE` (0x10).
43const INPUT_EVENT_OPCODE: u8 = 0x10;
44
45/// Build a 64×64 BGRA8888 sprite for the VirtIO GPU hardware cursor.
46///
47/// Stamps the 16×16 software cursor into the top-left corner; all other pixels
48/// are transparent (0x00_00_00_00). The 64×64 size is fixed by the VirtIO GPU
49/// spec (`CURSOR_RECT` in virtio-drivers `gpu.rs:145-148`).
50fn build_hw_cursor_sprite() -> [u8; 64 * 64 * 4] {
51    let mut buf = [0u8; 64 * 64 * 4];
52    for row in 0..cursor_sprite::CURSOR_H as usize {
53        for col in 0..cursor_sprite::CURSOR_W as usize {
54            if let Some(px) = cursor_sprite::cursor_pixel(row as u32, col as u32) {
55                let off = (row * 64 + col) * 4;
56                buf[off..off + 4].copy_from_slice(&px);
57            }
58        }
59    }
60    buf
61}

63/// The kernel half of [`owner_watch::WatchSys`]: valid only inside the receive
64/// turn of the message being served (the kernel binds the grant to it).
65#[cfg(not(test))]
66#[cfg(target_os = "none")]
67struct KernelWatch;
68
69#[cfg(not(test))]
70#[cfg(target_os = "none")]
71impl owner_watch::WatchSys for KernelWatch {
72    fn watch(&mut self, cell_id: u64, generation: u64) -> Option<(usize, u64)> {
73        ostd::syscall::sys_watch_cell_owner(cell_id, generation)
74            .map(|(owner, token)| (owner.root_tid as usize, token))
75    }
76
77    fn cancel(&mut self, token: u64) {
78        ostd::syscall::sys_cancel_cell_owner_watch(token);
79    }
80}

85#[cfg(not(test))]
86#[cfg(target_os = "none")]
87#[no_mangle]
88pub fn main() {
89    init_custom_heap();
90    println("[compositor] Compositor v0.2: software blending, VirtIO GPU, input routing");
91    let (w, h) = framebuffer::default_screen_size();
92    let mut fb = ScreenFb::new(w, h);
93    let mut table = SurfaceTable::new();
94    let mut frames = crate::surface_table::FrameRegistry::new();
95    let mut z_order = ZOrder::new();
96    let mut input = InputState::new();
97    // Compositor-initiated repaint region: set by cursor moves, surface destroy/raise.
98    // Consumed (taken) on each render_frame call.
99    let mut pending_dirty: Option<Rect> = None;
100
101    // Register as input focus so keyboard + mouse events flow to us.
102    connect_to_input(&mut input);
103
104    // Attempt to upload the hardware cursor sprite (64×64 BGRA8888).
105    // The 16×16 software sprite is placed in the top-left; the rest is transparent.
106    // If the GPU cursor is unavailable (setup_cursor returns error), we fall back to
107    // the Phase 01 software cursor path for the session.
108    {
109        let (hot_x, hot_y) = cursor_sprite::hotspot();
110        let sprite = build_hw_cursor_sprite();
111        let ok = sys_gpu_cursor(0, sprite.as_ptr(), 0, 0, hot_x as u32, hot_y as u32).is_ok();
112        input.hw_cursor = ok;
113        if ok {
114            println("[compositor] hardware cursor active");
115        } else {
116            println("[compositor] hardware cursor unavailable — using software cursor");
117        }
118    }
119
120    // The kernel writes the 32-byte caller identity at the END of this buffer;
121    // the 512-byte request window at its start is never overwritten.
122    let mut buf = [0u8; api::ipc::IPC_BUF_SIZE];
123    let mut watches = owner_watch::OwnerWatches::new();
124    // Resolution hotplug: check every 5 s; reconstruct ScreenFb on change.
125    let mut last_res_check_ms: u64 = 0;
126
127    loop {
128        // The receive syscall only writes the payload; clear its reusable tail so
129        // fixed-frame decoders never inspect bytes from an earlier sender and a
130        // death record (which carries no trailer) reads as unattributed.
131        buf.fill(0);
132
133        match sys_recv_attested(0, &mut buf) {
134            SyscallResult::Ok(sender) if sender > 0 => {
135                match owner_watch::classify(&buf, sender, &mut watches) {
136                    // Unattributed and not a watched cell's death (e.g. a
137                    // zero-trailer leftover of a client that was already reclaimed).
138                    owner_watch::Incoming::Drop => {}
139                    owner_watch::Incoming::OwnerDeath(dead) => {
140                        let removed = owner_watch::reclaim(
141                            &dead,
142                            &mut input,
143                            &mut table,
144                            &mut frames,
145                            &mut z_order,
146                            &mut pending_dirty,
147                        );
148                        if let Some(line) = owner_watch::death_line(&dead, removed) {
149                            println(&line);
150                        }
151                    }
152                    owner_watch::Incoming::Serve(identity) => {
153                        if let Some(request) = buf.first_chunk::<512>() {
154                            if input.input_tid != 0
155                                && sender == input.input_tid
156                                && request[0] == INPUT_EVENT_OPCODE
157                            {
158                                // On MouseMove, update_cursor sets pending_dirty = union(old, new)
159                                // so the frame is repainted at the interval tick.
160                                handle_input_event(
161                                    request,
162                                    &mut input,
163                                    fb.width,
164                                    fb.height,
165                                    Rect {
166                                        x: 0,
167                                        y: 0,
168                                        w: fb.width,
169                                        h: fb.height.saturating_sub(36),
170                                    },
171                                    &mut table,
172                                    &mut z_order,
173                                    &mut pending_dirty,
174                                );
175                            } else {
176                                // Watch the cell before dispatch: the kernel grants the
177                                // watch only while this message is the one being served.
178                                if owner_watch::creates_owned_state(request[0]) {
179                                    let admission =
180                                        watches.admit(&identity, sender, &mut KernelWatch);
181                                    for stale in &admission.stale {
182                                        let removed = owner_watch::reclaim(
183                                            stale,
184                                            &mut input,
185                                            &mut table,
186                                            &mut frames,
187                                            &mut z_order,
188                                            &mut pending_dirty,
189                                        );
190                                        if let Some(line) = owner_watch::death_line(stale, removed)
191                                        {
192                                            println(&line);
193                                        }
194                                    }
195                                    if admission.unavailable {
196                                        println(
197                                            "[compositor] owner death watch unavailable; window reclaim deferred",
198                                        );
199                                    }
200                                }
201                                message_handler::handle_message(
202                                    request,
203                                    sender,
204                                    message_handler::MessageContext::new(
205                                        fb.width,
206                                        fb.height,
207                                        &mut input,
208                                        &mut table,
209                                        &mut frames,
210                                        &mut z_order,
211                                        &mut pending_dirty,
212                                    ),
213                                );
214                            }
215                        }
216                    }
217                }
218            }
219            _ => ostd::task::yield_now(),
220        }
221
222        // Damage-driven: render when a surface reports damage or cursor/compositor dirty.
223        if table.has_damage() || pending_dirty.is_some() {
224            let _ = render_frame(
225                &mut fb,
226                &mut table,
227                &z_order,
228                pending_dirty.take(),
229                input.selected_cap(),
230                input.mouse_x,
231                input.mouse_y,
232            );
233        }
234
235        // Display hotplug: poll GPU resolution every 5 s; if it changed, rebuild ScreenFb
236        // and mark the full screen dirty so all surfaces are re-blended at the new size.
237        let now_ms = sys_get_time();
238        if now_ms.wrapping_sub(last_res_check_ms) >= 5_000 {
239            last_res_check_ms = now_ms;
240            let (new_w, new_h) = sys_get_resolution();
241            if new_w != fb.width || new_h != fb.height {
242                println("[compositor] resolution changed — rebuilding framebuffer");
243                fb = ScreenFb::new(new_w, new_h);
244                // Repaint entire screen so all surfaces are composited at the new size.
245                pending_dirty = Some(Rect {
246                    x: 0,
247                    y: 0,
248                    w: new_w,
249                    h: new_h,
250                });
251            }
252        }
253    }
254}
