5//! Compositor Service Cell.
6//!
7//! Manages a z-ordered set of cell surfaces, blends them into a screen
8//! framebuffer, and flushes dirty regions to the VirtIO GPU via the
9//! `GpuFlush` kernel syscall.
10//!
11//! Input routing: on startup the compositor registers as the input service's
12//! focus endpoint. All `InputEvent` IPC frames are dispatched to
13//! `input_handler`, which forwards keyboard events to the focused surface
14//! owner and updates the cursor position on mouse move.

38/// IPC opcode prefix byte identifying an input event from the input service.
39/// Must match `input_handler::INPUT_EVENT_OPCODE` (0x10).
40const INPUT_EVENT_OPCODE: u8 = 0x10;
41
42/// Build a 64×64 BGRA8888 sprite for the VirtIO GPU hardware cursor.
43///
44/// Stamps the 16×16 software cursor into the top-left corner; all other pixels
45/// are transparent (0x00_00_00_00). The 64×64 size is fixed by the VirtIO GPU
46/// spec (`CURSOR_RECT` in virtio-drivers `gpu.rs:145-148`).
47fn build_hw_cursor_sprite() -> [u8; 64 * 64 * 4] {
48    let mut buf = [0u8; 64 * 64 * 4];
49    for row in 0..cursor_sprite::CURSOR_H as usize {
50        for col in 0..cursor_sprite::CURSOR_W as usize {
51            if let Some(px) = cursor_sprite::cursor_pixel(row as u32, col as u32) {
52                let off = (row * 64 + col) * 4;
53                buf[off..off + 4].copy_from_slice(&px);
54            }
55        }
56    }
57    buf
58}
59ostd::declare_custom_heap!(10 * 1024 * 1024);

61#[cfg(not(test))]
62#[no_mangle]
63pub fn main() {
64    init_custom_heap();
65    println("[compositor] Compositor v0.2: software blending, VirtIO GPU, input routing");
66    let (w, h) = framebuffer::default_screen_size();
67    let mut fb = ScreenFb::new(w, h);
68    let mut table = SurfaceTable::new();
69    let mut z_order = ZOrder::new();
70    let mut input = InputState::new();
71    // Compositor-initiated repaint region: set by cursor moves, surface destroy/raise.
72    // Consumed (taken) on each render_frame call.
73    let mut pending_dirty: Option<Rect> = None;
74
75    // Register as input focus so keyboard + mouse events flow to us.
76    connect_to_input(&mut input);
77
78    // Attempt to upload the hardware cursor sprite (64×64 BGRA8888).
79    // The 16×16 software sprite is placed in the top-left; the rest is transparent.
80    // If the GPU cursor is unavailable (setup_cursor returns error), we fall back to
81    // the Phase 01 software cursor path for the session.
82    {
83        let (hot_x, hot_y) = cursor_sprite::hotspot();
84        let sprite = build_hw_cursor_sprite();
85        let ok = sys_gpu_cursor(0, sprite.as_ptr(), 0, 0, hot_x as u32, hot_y as u32).is_ok();
86        input.hw_cursor = ok;
87        if ok {
88            println("[compositor] hardware cursor active");
89        } else {
90            println("[compositor] hardware cursor unavailable — using software cursor");
91        }
92    }
93
94    let mut buf = [0u8; 512];
95    // Resolution hotplug: check every 5 s; reconstruct ScreenFb on change.
96    let mut last_res_check_ms: u64 = 0;
97
98    loop {
99        // The receive syscall only writes the payload; clear its reusable tail so
100        // fixed-frame decoders never inspect bytes from an earlier sender.
101        buf.fill(0);
102
103        match sys_recv(0, &mut buf) {
104            SyscallResult::Ok(sender) if sender > 0 => {
105                if input.input_tid != 0 && sender == input.input_tid && buf[0] == INPUT_EVENT_OPCODE
106                {
107                    // On MouseMove, update_cursor sets pending_dirty = union(old, new)
108                    // so the frame is repainted at the interval tick.
109                    handle_input_event(
110                        &buf,
111                        &mut input,
112                        Rect {
113                            x: window_decoration::FRAME,
114                            y: window_decoration::FRAME + window_decoration::TITLE,
115                            w: fb
116                                .width
117                                .saturating_sub((window_decoration::FRAME * 2) as u32),
118                            h: fb.height.saturating_sub(
119                                (window_decoration::FRAME * 2 + window_decoration::TITLE) as u32,
120                            ),
121                        },
122                        &mut table,
123                        &mut z_order,
124                        &mut pending_dirty,
125                    );
126                } else {
127                    message_handler::handle_message(
128                        &buf,
129                        sender,
130                        message_handler::MessageContext::new(
131                            fb.width,
132                            fb.height,
133                            &mut input,
134                            &mut table,
135                            &mut z_order,
136                            &mut pending_dirty,
137                        ),
138                    );
139                }
140            }
141            _ => ostd::task::yield_now(),
142        }
143
144        // Damage-driven: render when a surface reports damage or cursor/compositor dirty.
145        if table.has_damage() || pending_dirty.is_some() {
146            let _ = render_frame(
147                &mut fb,
148                &mut table,
149                &z_order,
150                pending_dirty.take(),
151                input.selected_cap(),
152                input.mouse_x,
153                input.mouse_y,
154            );
155        }
156
157        // Display hotplug: poll GPU resolution every 5 s; if it changed, rebuild ScreenFb
158        // and mark the full screen dirty so all surfaces are re-blended at the new size.
159        let now_ms = sys_get_time();
160        if now_ms.wrapping_sub(last_res_check_ms) >= 5_000 {
161            last_res_check_ms = now_ms;
162            let (new_w, new_h) = sys_get_resolution();
163            if new_w != fb.width || new_h != fb.height {
164                println("[compositor] resolution changed — rebuilding framebuffer");
165                fb = ScreenFb::new(new_w, new_h);
166                // Repaint entire screen so all surfaces are composited at the new size.
167                pending_dirty = Some(Rect {
168                    x: 0,
169                    y: 0,
170                    w: new_w,
171                    h: new_h,
172                });
173            }
174        }
175    }
176}
