1//! Tile groups: frames glued edge to edge. The registry keeps only which
2//! frames are glued along which side; where the shared edges are is read off
3//! the windows themselves (`crate::tiling`), so it can never drift from them.
4
5use alloc::vec::Vec;
6
7use super::FrameRegistry;
8
9/// Frames one tile group holds at most.
10pub const MAX_GROUP: usize = 16;
11
12/// Which side of frame `a` frame `b` is glued to.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum Side {
15    /// `b`'s left edge meets `a`'s right edge.
16    Right,
17    /// `b`'s top edge meets `a`'s bottom edge.
18    Below,
19}
20
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub struct Join {
23    pub a: u32,
24    pub b: u32,
25    pub side: Side,
26}
27
28impl FrameRegistry {
29    /// Glue frame `b` to the `side` of frame `a`, joining (and merging) their
30    /// groups. Refused when either frame is gone, they are already glued, or
31    /// the merged group would exceed [`MAX_GROUP`].
32    pub fn glue(&mut self, a: u32, b: u32, side: Side) -> bool {
33        if a == b || !self.frames.contains_key(&a) || !self.frames.contains_key(&b) {
34            return false;
35        }
36        if self
37            .joins
38            .iter()
39            .any(|j| (j.a, j.b) == (a, b) || (j.a, j.b) == (b, a))
40        {
41            return false;
42        }
43        let mut merged = self.tile_group(a);
44        for frame in self.tile_group(b) {
45            if !merged.contains(&frame) {
46                merged.push(frame);
47            }
48        }
49        if merged.len() > MAX_GROUP {
50            return false;
51        }
52        self.joins.push(Join { a, b, side });
53        true
54    }
55
56    /// Take `frame` out of its tile group; the rest may split in two.
57    pub fn untile(&mut self, frame: u32) {
58        self.joins.retain(|j| j.a != frame && j.b != frame);
59        self.group_minimized.retain(|&f| f != frame);
60    }
61
62    /// Joins touching `frame`.
63    pub fn joins_of(&self, frame: u32) -> Vec<Join> {
64        self.joins
65            .iter()
66            .copied()
67            .filter(|j| j.a == frame || j.b == frame)
68            .collect()
69    }
70
71    /// Remember that the user minimized `frames` together (window menu).
72    pub fn mark_group_minimized(&mut self, frames: &[u32]) {
73        for &frame in frames {
74            if !self.group_minimized.contains(&frame) {
75                self.group_minimized.push(frame);
76            }
77        }
78    }
79
80    /// Whether `frame` went down with its group, and forget that it did.
81    pub fn take_group_minimized(&mut self, frame: u32) -> bool {
82        let before = self.group_minimized.len();
83        self.group_minimized.retain(|&f| f != frame);
84        self.group_minimized.len() != before
85    }
86
87    pub fn is_tiled(&self, frame: u32) -> bool {
88        self.joins.iter().any(|j| j.a == frame || j.b == frame)
89    }
90
91    /// Every frame glued to `frame` directly or through others, `frame` first.
92    pub fn tile_group(&self, frame: u32) -> Vec<u32> {
93        let mut group = alloc::vec![frame];
94        let mut index = 0;
95        while index < group.len() {
96            let current = group[index];
97            for join in &self.joins {
98                let other = if join.a == current {
99                    join.b
100                } else if join.b == current {
101                    join.a
102                } else {
103                    continue;
104                };
105                if !group.contains(&other) {
106                    group.push(other);
107                }
108            }
109            index += 1;
110        }
111        group
112    }
113}
