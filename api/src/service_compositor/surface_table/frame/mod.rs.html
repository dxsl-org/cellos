26#[derive(Debug, Clone)]
27pub struct Frame {
28    pub generation: u32,
29    pub x: i32,
30    pub y: i32,
31    pub w: u32,
32    pub h: u32,
33    pub views: Vec<ViewRef>,
34    pub selected_view: Option<ViewRef>,
35    pub min_w: u32,
36    pub min_h: u32,
37    pub placement: FramePlacement,
38    pub saved_floating_rect: Option<Rect>,
39    pub saved_placement: Option<FramePlacement>,
40    /// The one task owning every view in the frame, re-derived on every
41    /// membership change. Only it may add views to the frame or select, snap
42    /// or fullscreen it over IPC. `None` once the user stacked windows of
43    /// several apps together: such a frame belongs to the compositor and no
44    /// client may plant a tab in, re-stack or move another client's window.
45    pub owner_tid: Option<usize>,
46}

48#[derive(Debug, Clone)]
49pub struct View {
50    pub incarnation: u32,
51    pub frame_id: u32,
52    pub surface_cap: u64,
53    pub owner_tid: usize,
54    /// Minimum content size the view asked for; the frame's minimum is the
55    /// largest over its views.
56    pub min_w: u32,
57    pub min_h: u32,
58}

60#[derive(Default)]
61pub struct FrameRegistry {
62    frames: BTreeMap<u32, Frame>,
63    views: BTreeMap<u32, View>,
64    /// Surface cap -> view id, for every surface that is a view.
65    by_cap: BTreeMap<u64, u32>,
66    /// Tabs of a frame still to be asked by "Close all tabs", in order.
67    close_queue: BTreeMap<u32, Vec<u64>>,
68    /// How far a lone tab was slid along its frame's top edge.
69    tab_offsets: BTreeMap<u32, u32>,
70    /// Frames glued edge to edge (tile groups).
71    joins: Vec<tile::Join>,
72    /// Frames the user minimized together with their tile group.
73    group_minimized: Vec<u32>,
74    next_frame_id: u32,
75    next_view_id: u32,
76}

78impl FrameRegistry {
79    pub fn new() -> Self {
80        Self {
81            frames: BTreeMap::new(),
82            views: BTreeMap::new(),
83            by_cap: BTreeMap::new(),
84            close_queue: BTreeMap::new(),
85            tab_offsets: BTreeMap::new(),
86            joins: Vec::new(),
87            group_minimized: Vec::new(),
88            next_frame_id: 1,
89            next_view_id: 1,
90        }
91    }
92
93    /// Look up backing surface cap for a view.
94    pub fn surface_for_view(&self, view_id: u32) -> Option<u64> {
95        self.views.get(&view_id).map(|v| v.surface_cap)
96    }
97
98    #[allow(dead_code)]
99    pub fn frame_for_view(&self, view_id: u32) -> Option<u32> {
100        self.views.get(&view_id).map(|v| v.frame_id)
101    }
102
103    #[allow(dead_code)]
104    pub fn selected_view_for_frame(&self, frame_id: u32) -> Option<ViewRef> {
105        self.frames.get(&frame_id).and_then(|f| f.selected_view)
106    }
107
108    /// Remove every view owned by a dying task. Returns the frames whose
109    /// selected tab changed to another live surface, as for
110    /// [`Self::remove_surface`].
111    pub fn cleanup_owner(&mut self, dead_tid: usize) -> Vec<Reselected> {
112        let dead: Vec<u64> = self
113            .views
114            .values()
115            .filter(|view| view.owner_tid == dead_tid)
116            .map(|view| view.surface_cap)
117            .collect();
118        let mut reselected: Vec<Reselected> = Vec::new();
119        for cap in dead {
120            if let Some(change) = self.remove_surface(cap) {
121                reselected.retain(|prior| prior.frame_id != change.frame_id);
122                if change
123                    .selected
124                    .is_some_and(|selected| self.frame_of(selected).is_some())
125                {
126                    reselected.push(change);
127                }
128            }
129        }
130        reselected.retain(|change| {
131            change
132                .selected
133                .is_some_and(|selected| self.frame_of(selected) == Some(change.frame_id))
134        });
135        reselected
136    }
137
138    /// Forget `frame_id` and everything keyed by it; it leaves its tile group.
139    fn drop_frame(&mut self, frame_id: u32) {
140        self.frames.remove(&frame_id);
141        self.close_queue.remove(&frame_id);
142        self.tab_offsets.remove(&frame_id);
143        self.untile(frame_id);
144    }
145
146    fn allocate_frame_id(&mut self) -> u32 {
147        let fid = self.next_frame_id;
148        self.next_frame_id = self.next_frame_id.wrapping_add(1).max(1);
149        fid
150    }
151
152    /// Re-derive the frame's owner and minimum size from its views after its
153    /// membership changed; drop it once empty.
154    fn refresh(&mut self, frame_id: u32) {
155        let Some(frame) = self.frames.get_mut(&frame_id) else {
156            return;
157        };
158        if frame.views.is_empty() {
159            self.drop_frame(frame_id);
160            return;
161        }
162        if frame.views.len() > 1 {
163            // A stack's tabs span the frame; a lone tab again starts left.
164            self.tab_offsets.remove(&frame_id);
165        }
166        let mut owner = None;
167        let mut shared = false;
168        let (mut min_w, mut min_h) = (0, 0);
169        for view in frame
170            .views
171            .iter()
172            .filter_map(|v| self.views.get(&v.view_id))
173        {
174            shared |= owner.is_some_and(|tid| tid != view.owner_tid);
175            owner = Some(view.owner_tid);
176            min_w = min_w.max(view.min_w);
177            min_h = min_h.max(view.min_h);
178        }
179        frame.owner_tid = if shared { None } else { owner };
180        frame.min_w = min_w;
181        frame.min_h = min_h;
182        if !frame
183            .selected_view
184            .is_some_and(|s| frame.views.contains(&s))
185        {
186            frame.selected_view = frame.views.first().copied();
187        }
188    }
