2use ostd::syscall::{sys_try_send, SyscallResult};

4use crate::surface_table::SurfaceTable;

7const MIN_CONTENT_W: u32 = 64;
8const MIN_CONTENT_H: u32 = 48;
9const MAX_CONTENT: i32 = 4096;
10
11pub(super) fn propose(
12    cap: u64,
13    rect: Rect,
14    edge: ResizeEdge,
15    dx: i32,
16    dy: i32,
17    table: &mut SurfaceTable,
18) {
19    let proposed = rect_for(rect, edge, dx, dy);
20    if let Some(surface) = table.get_mut(cap) {
21        if let Some(event) = surface.begin_configure(cap as u32, ConfigureKind::Resize, proposed) {
22            let delivered = event.encode().is_ok_and(|bytes| {
23                matches!(sys_try_send(surface.owner, &bytes), SyscallResult::Ok(0))
24            });
25            if !delivered {
26                surface.cancel_configure(event.serial);
27            }
28        }
29    }
30}

32fn rect_for(rect: Rect, edge: ResizeEdge, dx: i32, dy: i32) -> Rect {
33    let right = end(rect.x, rect.w);
34    let bottom = end(rect.y, rect.h);
35    let move_west = matches!(
36        edge,
37        ResizeEdge::West | ResizeEdge::NorthWest | ResizeEdge::SouthWest
38    );
39    let move_east = matches!(
40        edge,
41        ResizeEdge::East | ResizeEdge::NorthEast | ResizeEdge::SouthEast
42    );
43    let move_north = matches!(
44        edge,
45        ResizeEdge::North | ResizeEdge::NorthWest | ResizeEdge::NorthEast
46    );
47    let move_south = matches!(
48        edge,
49        ResizeEdge::South | ResizeEdge::SouthWest | ResizeEdge::SouthEast
50    );
51    let x = if move_west {
52        rect.x
53            .saturating_add(dx)
54            .max(right.saturating_sub(MAX_CONTENT))
55            .min(right.saturating_sub(MIN_CONTENT_W as i32))
56    } else {
57        rect.x
58    };
59    let y = if move_north {
60        rect.y
61            .saturating_add(dy)
62            .max(bottom.saturating_sub(MAX_CONTENT))
63            .min(bottom.saturating_sub(MIN_CONTENT_H as i32))
64    } else {
65        rect.y
66    };
67    let new_right = if move_east {
68        right
69            .saturating_add(dx)
70            .max(rect.x.saturating_add(MIN_CONTENT_W as i32))
71            .min(rect.x.saturating_add(MAX_CONTENT))
72    } else {
73        right
74    };
75    let new_bottom = if move_south {
76        bottom
77            .saturating_add(dy)
78            .max(rect.y.saturating_add(MIN_CONTENT_H as i32))
79            .min(rect.y.saturating_add(MAX_CONTENT))
80    } else {
81        bottom
82    };
83    Rect {
84        x,
85        y,
86        w: new_right
87            .saturating_sub(x)
88            .clamp(MIN_CONTENT_W as i32, MAX_CONTENT) as u32,
89        h: new_bottom
90            .saturating_sub(y)
91            .clamp(MIN_CONTENT_H as i32, MAX_CONTENT) as u32,
92    }
93}

95fn end(start: i32, extent: u32) -> i32 {
96    start.saturating_add(extent.min(i32::MAX as u32) as i32)
