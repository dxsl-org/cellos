13pub(super) fn route<F>(
14    router: &mut PointerRouter,
15    event: InputEvent,
16    x: i32,
17    y: i32,
18    context: &mut RouteContext<'_, F>,
19) where
20    F: FnMut(Rect),
21{
22    match event {
23        InputEvent::MouseMove { .. } => move_pointer(router, event, x, y, context),
24        InputEvent::MouseButton { button, state } => {
25            route_button(router, button, state, x, y, context)
26        }
27        InputEvent::MouseScroll { .. } => {
28            if router.menu.is_some() {
29                return;
30            }
31            if let Some(target) = target::at(x, y, context.table, context.frames, context.z_order)
32                .filter(|target| target.hit == Hit::Content)
33            {
34                target::send_position(target, x, y);
35                target::send_local(target, event);
36            }
37        }
38        InputEvent::Key(_) => {}
39    }
40}

42fn move_pointer<F>(
43    router: &mut PointerRouter,
44    event: InputEvent,
45    x: i32,
46    y: i32,
47    context: &mut RouteContext<'_, F>,
48) where
49    F: FnMut(Rect),
50{
51    if let Some(menu) = router.menu {
52        // The open menu owns the pointer: track its hovered item, nothing else.
53        let hover = window_decoration::menu_item_at(menu.rect, &menu.items, x, y);
54        if hover != menu.hover {
55            router.menu = Some(OpenMenu { hover, ..menu });
56            (context.damage)(menu.rect);
57        }
58        return;
59    }
60    match router.capture {
61        Some(Capture::Client(cap)) => {
62            if let Some(target) = target::for_cap(cap, context.table) {
63                target::send_local(target, event);
64            }
65        }
66        Some(Capture::Drag {
67            cap,
68            start_x,
69            start_y,
70            rect,
71            mode: DragMode::Stack { .. },
72        }) => stack_drag::track(router, cap, (start_x, start_y), rect, x, y, context),
73        Some(Capture::Drag {
74            cap,
75            start_x,
76            mode: DragMode::Slide { start_offset },
77            ..
78        }) => stack_drag::slide(cap, start_offset, x.saturating_sub(start_x), context),
79        Some(Capture::Drag {
80            cap,
81            start_x,
82            start_y,
83            rect,
84            mode: DragMode::Move,
85        }) => {
86            if let Some(surface) = context
87                .table
88                .get_mut(cap)
89                .filter(|surface| surface.accepts_input())
90            {
91                let old = window_decoration::bounds(surface.screen_rect());
92                let before = (surface.x, surface.y);
93                surface.move_to(
94                    rect.x.saturating_add(x.saturating_sub(start_x)),
95                    rect.y.saturating_add(y.saturating_sub(start_y)),
96                );
97                let delta = (surface.x - before.0, surface.y - before.1);
98                (context.damage)(old.union(&window_decoration::bounds(surface.screen_rect())));
99                // The other tabs of its stack and the frames of its tile
100                // group travel with it.
101                stacking::sync_frame_of(context.frames, context.table, cap);
102                if let Some(area) = tiling::follow(context.frames, context.table, cap, delta) {
103                    (context.damage)(area);
104                }
105            }
106        }
107        Some(Capture::Resize {
108            cap,
109            start_x,
110            start_y,
111            rect,
112            edge,
113        }) => {
114            let delta = (x.saturating_sub(start_x), y.saturating_sub(start_y));
115            if !tiling::resize(context.frames, context.table, cap, rect, edge, delta) {
116                resize::propose(cap, rect, edge, delta.0, delta.1, context.table);
117            }
118        }
119        Some(Capture::Menu) => {}
120        None => {
121            if let Some(target) = target::at(x, y, context.table, context.frames, context.z_order)
122                .filter(|target| target.hit == Hit::Content)
123            {
124                target::send_local(target, event);
125            }
126        }
127    }
128}

130fn route_button<F>(
131    router: &mut PointerRouter,
132    button: MouseButton,
133    state: KeyState,
134    x: i32,
135    y: i32,
136    context: &mut RouteContext<'_, F>,
137) where
138    F: FnMut(Rect),
139{
140    let event = InputEvent::MouseButton { button, state };
141    if let Some(menu) = router.menu {
142        // Any press outside the menu dismisses it and goes nowhere else.
143        if state == KeyState::Pressed {
144            if contains(menu.rect, x, y) && button == MouseButton::Left {
145                router.capture = Some(Capture::Menu);
146            } else if let Some(area) = router.close_menu() {
147                (context.damage)(area);
148            }
149            return;
150        }
151        if state == KeyState::Released && matches!(router.capture, Some(Capture::Menu)) {
152            actions::release_in_menu(router, menu, x, y, context);
153            return;
154        }
155    }
156    if button != MouseButton::Left {
157        if let Some(target) = target::at(x, y, context.table, context.frames, context.z_order)
158            .filter(|target| target.hit == Hit::Content)
159        {
160            target::send_position(target, x, y);
161            target::send_local(target, event);
162        }
163        return;
164    }
165    match state {
166        KeyState::Pressed => press(router, event, x, y, context),
167        KeyState::Released => match router.capture.take() {
168            Some(Capture::Client(cap)) => {
169                if let Some(target) = target::for_cap(cap, context.table) {
170                    target::send_position(target, x, y);
171                    target::send_local(target, event);
172                }
173            }
174            Some(Capture::Drag {
175                cap,
176                start_x,
177                start_y,
178                mode: DragMode::Stack { target, glue },
179                ..
180            }) => {
181                let delta = (x.saturating_sub(start_x), y.saturating_sub(start_y));
182                stack_drag::drop(router, cap, delta, (target, glue), context);
183            }
184            _ => {}
185        },
186        KeyState::Repeated => {}
187    }
188}

190fn press<F>(
191    router: &mut PointerRouter,
192    event: InputEvent,
193    x: i32,
194    y: i32,
195    context: &mut RouteContext<'_, F>,
196) where
197    F: FnMut(Rect),
198{
199    let Some(target) = target::at(x, y, context.table, context.frames, context.z_order) else {
200        return;
201    };
202    router.focused_owner = target.owner;
203    let old = router
204        .selected_cap
205        .and_then(|cap| context.table.get(cap))
206        .map(|surface| window_decoration::bounds(surface.screen_rect()));
207    router.selected_cap = Some(target.cap);
208    context.z_order.raise(target.cap);
209    tiling::raise(context.frames, context.z_order, target.cap);
210    let Some(rect) = context
211        .table
212        .get(target.cap)
213        .map(|surface| surface.screen_rect())
214    else {
215        return;
216    };
217    let now = window_decoration::bounds(rect);
218    (context.damage)(old.map_or(now, |prior| prior.union(&now)));
219
220    router.capture = match target.hit {
221        Hit::Content => {
222            target::send_position(target, x, y);
223            target::send_local(target, event);
224            Some(Capture::Client(target.cap))
225        }
226        Hit::Title if is_double_click(router, target.cap, x, y, context.now_ms) => {
227            router.last_title_press = None;
228            actions::apply(router, target.cap, Control::Maximize, context);
229            None
230        }
231        Hit::Tab(index) => {
232            let shown = actions::show_tab(router, target.cap, index, context);
233            // Win held: the picked tab is dragged out right away, as in Haiku.
234            let stack = DragMode::Stack {
235                target: None,
236                glue: None,
237            };
238            match shown {
239                Some(cap) if stack_drag::mode_for(router.modifiers, cap, context) == stack => {
240                    Some(Capture::Drag {
241                        cap,
242                        start_x: x,
243                        start_y: y,
244                        rect,
245                        mode: stack,
246                    })
247                }
248                _ => None,
249            }
250        }
251        Hit::Title => {
252            router.last_title_press = Some((target.cap, context.now_ms, x, y));
253            let mode = stack_drag::mode_for(router.modifiers, target.cap, context);
254            Some(Capture::Drag {
255                cap: target.cap,
256                start_x: x,
257                start_y: y,
258                rect,
259                mode,
260            })
261        }
262        Hit::Resize(edge) => Some(Capture::Resize {
263            cap: target.cap,
264            start_x: x,
265            start_y: y,
266            rect,
267            edge,
268        }),
269        Hit::Menu => {
270            let (screen_w, screen_h) = context.screen;
271            let tabs = stacking::tabs_of(context.frames, target.cap);
272            let items =
273                window_menu::items_for(target.cap, context.table, context.frames, context.z_order);
274            let menu_rect = window_decoration::menu_rect(rect, tabs, &items, screen_w, screen_h);
275            router.menu = Some(OpenMenu {
276                cap: target.cap,
277                rect: menu_rect,
278                items,
279                hover: window_decoration::menu_item_at(menu_rect, &items, x, y),
280            });
281            (context.damage)(window_decoration::menu_region(menu_rect));
282            None
283        }
284    };
285}

287pub(super) fn is_double_click(router: &PointerRouter, cap: u64, x: i32, y: i32, now: u64) -> bool {
288    router
289        .last_title_press
290        .is_some_and(|(pressed, at, px, py)| {
291            pressed == cap
292                && now.saturating_sub(at) <= DOUBLE_CLICK_MS
293                && (x - px).abs() <= DOUBLE_CLICK_SLOP
294                && (y - py).abs() <= DOUBLE_CLICK_SLOP
295        })
296}

298fn contains(rect: Rect, x: i32, y: i32) -> bool {
299    x >= rect.x
300        && y >= rect.y
301        && i64::from(x) < i64::from(rect.x) + i64::from(rect.w)
302        && i64::from(y) < i64::from(rect.y) + i64::from(rect.h)
303}
