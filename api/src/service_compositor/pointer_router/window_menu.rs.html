2//! re-checked when an item runs (the screen may change while it is open).

12/// Windows nearer than this (between visible frames) are offered as stack targets.
13const NEAR: i32 = 48;
14/// Stack targets one menu offers at most.
15const MAX_TARGETS: usize = 5;
16
17/// Items for the window menu of the shown window `cap`.
18pub(super) fn items_for(
19    cap: u64,
20    table: &SurfaceTable,
21    frames: &FrameRegistry,
22    z_order: &ZOrder,
23) -> MenuItems {
24    let tabs = frames
25        .frame_of(cap)
26        .map_or(1, |frame| frames.members(frame).len());
27    let mut items = MenuItems::new();
28    items.push(MenuAction::Maximize);
29    items.push(MenuAction::Minimize);
30    items.push(MenuAction::Close);
31    if tabs > 1 {
32        items.push(MenuAction::CloseAll);
33    }
34    let tiled = frames
35        .frame_of(cap)
36        .is_some_and(|frame| frames.is_tiled(frame));
37    if tiled {
38        items.push(MenuAction::CloseGroup);
39    }
40    for target in stack_targets(cap, table, frames, z_order) {
41        items.push(MenuAction::StackOnto(target));
42    }
43    if tabs > 1 {
44        items.push(MenuAction::Detach);
45    }
46    if tiled {
47        items.push(MenuAction::Untile);
48    }
49    items
50}

52/// Shown windows of other frames that overlap `cap`'s frame or lie within
53/// [`NEAR`] of it, nearest first. A window whose tab and body both lie under
54/// one window above it is not offered: the user cannot see what they would
55/// stack onto.
56pub(super) fn stack_targets(
57    cap: u64,
58    table: &SurfaceTable,
59    frames: &FrameRegistry,
60    z_order: &ZOrder,
61) -> Vec<u64> {
62    let Some(own) = table.get(cap).map(|surface| visible(surface.screen_rect())) else {
63        return Vec::new();
64    };
65    let own_frame = frames.frame_of(cap);
66    let order: Vec<u64> = z_order.iter_bottom_to_top().collect();
67    let mut targets: Vec<(i32, u64)> = order
68        .iter()
69        .enumerate()
70        .filter(|&(_, &other)| {
71            frames
72                .frame_of(other)
73                .is_some_and(|frame| Some(frame) != own_frame)
74        })
75        .filter_map(|(depth, &other)| {
76            let surface = table.get(other).filter(|surface| surface.is_shown())?;
77            let parts = window_decoration::visible_parts(
78                surface.screen_rect(),
79                stacking::tabs_of(frames, other),
80            );
81            let covered = order[depth + 1..].iter().any(|&above| {
82                table.get(above).filter(|s| s.is_shown()).is_some_and(|s| {
83                    let cover = window_decoration::visible_parts(
84                        s.screen_rect(),
85                        stacking::tabs_of(frames, above),
86                    );
87                    parts
88                        .iter()
89                        .all(|part| cover.iter().any(|&c| covers(c, *part)))
90                })
91            });
92            let gap = gap(own, visible(surface.screen_rect()));
93            (!covered && gap <= NEAR).then_some((gap, other))
94        })
95        .collect();
96    targets.sort_by_key(|&(gap, _)| gap);
97    targets
98        .into_iter()
99        .take(MAX_TARGETS)
100        .map(|(_, cap)| cap)
101        .collect()
102}

104/// Whether `target` may still take `source`'s stack when the item runs.
105pub(super) fn can_stack(
106    source: u64,
107    target: u64,
108    table: &SurfaceTable,
109    frames: &FrameRegistry,
110) -> bool {
111    let shown = |cap: u64| table.get(cap).is_some_and(|surface| surface.is_shown());
112    shown(source)
113        && shown(target)
114        && frames
115            .frame_of(target)
116            .is_some_and(|frame| Some(frame) != frames.frame_of(source))
117}

119fn visible(content: Rect) -> Rect {
120    window_decoration::frame_rect(content)
121}

123/// Largest of the horizontal and vertical gaps between `a` and `b` (0 when they touch or overlap).
124fn gap(a: Rect, b: Rect) -> i32 {
125    let span = |start: i32, extent: u32| (i64::from(start), i64::from(start) + i64::from(extent));
126    let ((ax0, ax1), (bx0, bx1)) = (span(a.x, a.w), span(b.x, b.w));
127    let ((ay0, ay1), (by0, by1)) = (span(a.y, a.h), span(b.y, b.h));
128    let dx = (bx0 - ax1).max(ax0 - bx1).max(0);
129    let dy = (by0 - ay1).max(ay0 - by1).max(0);
130    dx.max(dy).min(i64::from(i32::MAX)) as i32
131}
132
133fn covers(outer: Rect, inner: Rect) -> bool {
134    let end = |start: i32, extent: u32| i64::from(start) + i64::from(extent);
135    outer.x <= inner.x
136        && outer.y <= inner.y
137        && end(outer.x, outer.w) >= end(inner.x, inner.w)
138        && end(outer.y, outer.h) >= end(inner.y, inner.h)
139}

147    fn window(
148        table: &mut SurfaceTable,
149        frames: &mut FrameRegistry,
150        z: &mut ZOrder,
151        x: i32,
152        y: i32,
153    ) -> u64 {
154        let cap = table
155            .create(x, y, 200, 150, 7, SurfaceRole::Interactive)
156            .unwrap();
157        frames.adopt(cap, 7);
158        z.push(cap);
159        cap
160    }

162    #[test]
163    fn stack_targets_are_nearby_visible_windows_nearest_first() {
164        let (mut table, mut frames, mut z) =
165            (SurfaceTable::new(), FrameRegistry::new(), ZOrder::new());
166        let me = window(&mut table, &mut frames, &mut z, 100, 100);
167        let far = window(&mut table, &mut frames, &mut z, 800, 600);
168        let near = window(&mut table, &mut frames, &mut z, 340, 100);
169        let overlapping = window(&mut table, &mut frames, &mut z, 150, 150);
170        let covered = window(&mut table, &mut frames, &mut z, 160, 160);
171        let cover = table
172            .create(140, 130, 260, 230, 8, SurfaceRole::Interactive)
173            .unwrap();
174        frames.adopt(cover, 8);
175        z.push(cover);
176        z.raise(me);
177        z.raise(overlapping);
178        // `cover` hides `covered` entirely but sits below `overlapping` and `me`.
179        let targets = stack_targets(me, &table, &frames, &z);
180        // Equally near ones come in stacking order.
181        assert_eq!(
182            targets,
183            [cover, overlapping, near],
184            "covered and far ones are left out"
185        );
186        let _ = (far, covered);
187
188        let items = items_for(me, &table, &frames, &z);
189        assert_eq!(
190            items
191                .iter()
192                .filter(|a| matches!(a, MenuAction::StackOnto(_)))
193                .count(),
194            3
195        );
196        assert!(
197            !items
198                .iter()
199                .any(|a| matches!(a, MenuAction::Detach | MenuAction::CloseAll)),
200            "one tab"
201        );
202    }

204    #[test]
205    fn a_window_whose_tab_peeks_out_is_still_offered() {
206        let (mut table, mut frames, mut z) =
207            (SurfaceTable::new(), FrameRegistry::new(), ZOrder::new());
208        let behind = table
209            .create(100, 100, 600, 300, 8, SurfaceRole::Interactive)
210            .unwrap();
211        frames.adopt(behind, 8);
212        z.push(behind);
213        // Same place, narrower: its 160px tab leaves the 240px tab behind visible.
214        let me = window(&mut table, &mut frames, &mut z, 100, 100);
215        table.get_mut(me).unwrap().w = 300;
216        table.get_mut(me).unwrap().h = 300;
217        let tall = table
218            .create(100, 100, 600, 300, 9, SurfaceRole::Interactive)
219            .unwrap();
220        frames.adopt(tall, 9);
221        z.push(tall);
222        z.raise(me);
223        assert_eq!(
224            stack_targets(me, &table, &frames, &z),
225            [tall],
226            "behind is fully under tall"
227        );
228        z.raise(behind);
229        z.raise(me);
230        assert!(stack_targets(me, &table, &frames, &z).contains(&behind));
231    }

233    #[test]
234    fn a_target_that_left_while_the_menu_was_open_is_refused() {
235        let (mut table, mut frames, mut z) =
236            (SurfaceTable::new(), FrameRegistry::new(), ZOrder::new());
237        let me = window(&mut table, &mut frames, &mut z, 100, 100);
238        let other = window(&mut table, &mut frames, &mut z, 150, 150);
239        assert!(can_stack(me, other, &table, &frames));
240        let (from, onto) = (
241            frames.frame_of(other).unwrap(),
242            frames.frame_of(me).unwrap(),
243        );
244        frames.stack(from, onto);
245        assert!(!can_stack(me, other, &table, &frames), "same frame now");
246        frames.remove_surface(other);
247        table.remove(other);
248        assert!(!can_stack(me, other, &table, &frames), "gone");
249    }
