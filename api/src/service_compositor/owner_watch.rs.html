1//! Owner-death reclaim for the compositor: which requests create owned state,
2//! and what the compositor removes when the owning cell dies.
3//!
4//! The death watch itself lives in `ostd::service::owners`: `serve` watches a
5//! caller's cell while serving its first `CREATE_SURFACE` or `CREATE_VIEW` (no
6//! other request creates owned state), and reports its death, or its retirement
7//! by a newer generation of the same CellId, through `Handler::on_owner_death`.
8//!
9//! Surfaces and views are keyed by the creating **thread** tid, while a death
10//! names the cell incarnation. The compositor therefore records the thread tids
11//! that created state on each owner's behalf; the death reclaims every one.
12//!
13//! Syscall-free so the host test target can exercise it (ostd syscalls are real
14//! Linux syscalls on the host).
15

25/// Cells watched at once. Larger than any realistic number of GUI clients; a
26/// cell that does not fit is served without a watch (fail-open).
27pub(crate) const MAX_WATCHED_CELLS: usize = 64;
28/// Thread tids recorded per cell: the surface (32) and view (64) table
29/// capacities, so a cell cannot own more distinct creators than that.
30pub(crate) const MAX_TIDS_PER_CELL: usize = 96;
31
32/// Window state to reclaim for one dead (or superseded) cell generation.
33#[derive(Debug, PartialEq, Eq)]
34pub(crate) struct Released {
35    pub cell_id: u64,
36    pub generation: u64,
37    pub tids: Vec<usize>,
38}
39
40/// Only these requests create state that outlives the call, so only they need
41/// the owner to be watched.
42pub(crate) fn creates_owned_state(opcode: u8) -> bool {
43    matches!(
44        opcode,
45        compositor_ops::CREATE_SURFACE | compositor_ops::CREATE_VIEW
46    )
47}
48
49/// Remove every surface, view, z-order entry, focus and grant frame that any of
50/// `released`'s thread tids created, and queue the freed rectangles for repaint.
51/// Returns the number of surfaces removed.
52pub(crate) fn reclaim(
53    released: &Released,
54    input: &mut InputState,
55    table: &mut SurfaceTable,
56    frames: &mut FrameRegistry,
57    z_order: &mut ZOrder,
58    pending_dirty: &mut Option<Rect>,
59) -> usize {
60    released
61        .tids
62        .iter()
63        .map(|&tid| {
64            crate::message_handler::cleanup_owner(tid, input, table, frames, z_order, pending_dirty)
65        })
66        .sum()
67}
68
69/// The one log line for a reclaimed cell, only when something was removed.
70pub(crate) fn death_line(released: &Released, removed: usize) -> Option<String> {
71    (removed > 0).then(|| {
72        alloc::format!(
73            "[compositor] owner death: cell {} gen {}: removed {} surfaces",
74            released.cell_id,
75            released.generation,
76            removed
77        )
78    })
79}

90    fn only_create_requests_need_a_watch() {
91        assert!(creates_owned_state(compositor_ops::CREATE_SURFACE));
92        assert!(creates_owned_state(compositor_ops::CREATE_VIEW));
93        #[allow(deprecated)]
94        let others = [
95            0u8,
96            0x10,
97            OWNER_EXITED_RELAY,
98            compositor_ops::WRITE_PIXELS,
99            compositor_ops::DAMAGE_SURFACE,
100            compositor_ops::MOVE_SURFACE,
101            compositor_ops::RAISE_SURFACE,
102            compositor_ops::DESTROY_SURFACE,
103            compositor_ops::SET_TITLE,
104            compositor_ops::ATTACH_GRANT,
105            compositor_ops::DESTROY_VIEW,
106            compositor_ops::SELECT_VIEW,
107            compositor_ops::GET_SCREEN_SIZE,
108        ];
109        for opcode in others {
110            assert!(!creates_owned_state(opcode), "opcode {opcode:#x}");
111        }

147    struct World {
148        input: InputState,
149        table: SurfaceTable,
150        frames: FrameRegistry,
151        z_order: ZOrder,
152        dirty: Option<Rect>,
153    }
154
155    impl World {
156        fn new() -> Self {
157            Self {
158                input: InputState::new(),
159                table: SurfaceTable::new(),
160                frames: FrameRegistry::new(),
161                z_order: ZOrder::new(),
162                dirty: None,
163            }
164        }
165
166        /// A surface (owned pixels, never a registered grant: no syscall at
167        /// cleanup) created by thread `tid`, with one view on it.
168        fn window(&mut self, tid: usize, x: i32) -> (u64, u32) {
169            let cap = self
170                .table
171                .create(x, 10, 64, 48, tid, SurfaceRole::Interactive)
172                .expect("surface");
173            self.z_order.push(cap);
174            let view = self
175                .frames
176                .create_view(&view_request(), tid, cap)
177                .expect("view");
178            assert_eq!(view.result, ViewOpResult::Success);
179            (cap, view.view_id)
180        }
181
182        fn reclaim(&mut self, released: &Released) -> usize {
183            reclaim(
184                released,
185                &mut self.input,
186                &mut self.table,
187                &mut self.frames,
188                &mut self.z_order,
189                &mut self.dirty,
190            )
191        }
192    }
193
194    fn z_caps(world: &World) -> Vec<u64> {
195        world.z_order.iter_bottom_to_top().collect()
196    }

198    #[test]
199    fn owner_death_reclaims_exactly_the_dead_cells_windows() {
200        let mut world = World::new();
201        // Cell A (root 1000, worker 1001) and cell B (root 2000).
202        let (a_root, a_root_view) = world.window(1000, 0);
203        let (a_worker, a_worker_view) = world.window(1001, 200);
204        let (b_cap, b_view) = world.window(2000, 400);
205        assert!(world.input.request_focus(1000, a_root));
206        assert_eq!(world.input.selected_cap(), Some(a_root));
207        let a_bounds: Vec<Rect> = [a_root, a_worker]
208            .iter()
209            .map(|&cap| {
210                crate::window_decoration::bounds(world.table.get(cap).unwrap().screen_rect())
211            })
212            .collect();
213
214        // Cell A's death, as recorded by both of its creating threads.
215        let dead = Released {
216            cell_id: 1,
217            generation: 1,
218            tids: alloc::vec![1000, 1001],
219        };
220        assert_eq!(world.reclaim(&dead), 2);
221
222        for cap in [a_root, a_worker] {
223            assert!(world.table.get(cap).is_none());
224        }
225        assert!(world.table.get(b_cap).is_some());
226        assert_eq!(z_caps(&world), [b_cap]);
227        assert_eq!(world.frames.surface_for_view(a_root_view), None);
228        assert_eq!(world.frames.surface_for_view(a_worker_view), None);
229        assert_eq!(world.frames.surface_for_view(b_view), Some(b_cap));
230        assert_eq!(
231            world.input.selected_cap(),
232            None,
233            "focus on A's surface is gone"
234        );
235
236        // The freed rectangles are queued for repaint.
237        let dirty = world.dirty.expect("dirty rect");
238        for freed in a_bounds {
239            assert!(
240                dirty.x <= freed.x && dirty.y <= freed.y,
241                "{dirty:?} vs {freed:?}"
242            );
243            assert!(
244                dirty.x + dirty.w as i32 >= freed.x + freed.w as i32
245                    && dirty.y + dirty.h as i32 >= freed.y + freed.h as i32,
246                "{dirty:?} vs {freed:?}"
247            );
248        }
249
250        // Repeat death: nothing left to remove, B untouched.
251        assert_eq!(world.reclaim(&dead), 0);
252        assert!(world.table.get(b_cap).is_some());
253        assert_eq!(z_caps(&world), [b_cap]);
254    }

256    #[test]
257    fn focus_on_another_cells_surface_survives_a_death() {
258        let mut world = World::new();
259        let (_a, _) = world.window(1000, 0);
260        let (b_cap, _) = world.window(2000, 400);
261        assert!(world.input.request_focus(2000, b_cap));
262        let dead = Released {
263            cell_id: 1,
264            generation: 1,
265            tids: alloc::vec![1000],
266        };
267        assert_eq!(world.reclaim(&dead), 1);
268        assert_eq!(world.input.selected_cap(), Some(b_cap));

271    /// The supervisor `OWNER_EXITED_RELAY` relay still reaches the same routine per tid.
272    #[test]
273    fn relay_path_cleanup_owner_is_unchanged() {
274        let mut world = World::new();
275        let (cap, view) = world.window(3000, 0);
276        let (other, _) = world.window(3001, 100);
277        let removed = crate::message_handler::cleanup_owner(
278            3000,
279            &mut world.input,
280            &mut world.table,
281            &mut world.frames,
282            &mut world.z_order,
283            &mut world.dirty,
284        );
285        assert_eq!(removed, 1);
286        assert!(world.table.get(cap).is_none());
287        assert_eq!(world.frames.surface_for_view(view), None);
288        assert!(world.table.get(other).is_some());
289    }
