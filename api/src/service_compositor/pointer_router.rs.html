3use api::input::{
4    encode_event, InputEvent, KeyState, MouseButton, INPUT_EVENT_IPC_SIZE, INPUT_EVENT_OPCODE,

9use crate::z_order::ZOrder;

11#[derive(Clone, Copy)]
12struct PointerTarget {
13    cap: u64,
14    owner: usize,
15    origin_x: i32,
16    origin_y: i32,
17}
18
19/// Tracks keyboard focus and left-button capture across pointer events.
20pub struct PointerRouter {
21    focused_owner: usize,
22    capture: Option<u64>,
23}
24
25impl PointerRouter {
26    /// Create a router with no focused or captured surface.
27    pub const fn new() -> Self {
28        Self {
29            focused_owner: 0,
30            capture: None,
31        }
32    }
33
34    /// Return the owner selected by the most recent left-button press.
35    pub const fn focused_owner(&self) -> usize {
36        self.focused_owner
37    }
38
39    /// Route one pointer event using the current screen-space cursor position.
40    ///
41    /// Moves follow the captured surface while the left button is held; otherwise
42    /// they target the topmost surface under the cursor. Coordinates sent to the
43    /// owner are translated into that surface's local coordinate system.
44    pub fn route(
45        &mut self,
46        event: InputEvent,
47        screen_x: i32,
48        screen_y: i32,
49        table: &SurfaceTable,
50        z_order: &ZOrder,
51    ) {
52        match event {
53            InputEvent::MouseMove { .. } => {
54                let target = match self.capture {
55                    Some(cap) => target_for_cap(cap, table),
56                    None => hit_test(screen_x, screen_y, table, z_order),
57                };
58                if let Some(target) = target {
59                    send_local(target, event);
60                }
61            }
62            InputEvent::MouseButton { button, state } => {
63                self.route_button(button, state, event, screen_x, screen_y, table, z_order);
64            }
65            InputEvent::MouseScroll { .. } => {
66                if let Some(target) = hit_test(screen_x, screen_y, table, z_order) {
67                    send_position(target, screen_x, screen_y);
68                    send_local(target, event);
69                }
70            }
71            InputEvent::Key(_) => {}
72        }
73    }
74
75    fn route_button(
76        &mut self,
77        button: MouseButton,
78        state: KeyState,
79        event: InputEvent,
80        screen_x: i32,
81        screen_y: i32,
82        table: &SurfaceTable,
83        z_order: &ZOrder,
84    ) {
85        let left = button == MouseButton::Left;
86        let target = if left && state == KeyState::Released {
87            self.capture.and_then(|cap| target_for_cap(cap, table))
88        } else {
89            hit_test(screen_x, screen_y, table, z_order)
90        };
91
92        if left && state == KeyState::Pressed {
93            self.capture = target.map(|target| target.cap);
94            if let Some(target) = target {
95                self.focused_owner = target.owner;
96            }
97        }
98        if let Some(target) = target {
99            send_position(target, screen_x, screen_y);
100            send_local(target, event);
101        }
102        if left && state == KeyState::Released {
103            self.capture = None;
104        }
105    }
106}
107
108fn target_for_cap(cap: u64, table: &SurfaceTable) -> Option<PointerTarget> {
109    table.get(cap).map(|surface| PointerTarget {
110        cap,
111        owner: surface.owner,
112        origin_x: surface.x,
113        origin_y: surface.y,
114    })
115}
116
117fn hit_test(x: i32, y: i32, table: &SurfaceTable, z_order: &ZOrder) -> Option<PointerTarget> {
118    z_order.iter_top_to_bottom().find_map(|cap| {
119        let surface = table.get(cap)?;
120        let rect = surface.screen_rect();
121        (x >= rect.x && x < rect.x + rect.w as i32 && y >= rect.y && y < rect.y + rect.h as i32)
122            .then_some(PointerTarget {
123                cap,
124                owner: surface.owner,
125                origin_x: surface.x,
126                origin_y: surface.y,
127            })
128    })
129}
130
131fn send_position(target: PointerTarget, screen_x: i32, screen_y: i32) {
132    send_local(
133        target,
134        InputEvent::MouseMove {
135            x: screen_x,
136            y: screen_y,
137            dx: 0,
138            dy: 0,
139        },
140    );
141}
142
143fn send_local(target: PointerTarget, event: InputEvent) {
144    let event = match event {
145        InputEvent::MouseMove { x, y, dx, dy } => InputEvent::MouseMove {
146            x: x - target.origin_x,
147            y: y - target.origin_y,
148            dx,
149            dy,
150        },
151        event => event,
152    };
153    let mut payload = [0u8; INPUT_EVENT_IPC_SIZE];
154    encode_event(&event, &mut payload);
155    let mut frame = [0u8; 1 + INPUT_EVENT_IPC_SIZE];
156    frame[0] = INPUT_EVENT_OPCODE;
157    frame[1..].copy_from_slice(&payload);
158    let _ = sys_try_send(target.owner, &frame);
159}
