1//! Cursor position updates and hardware/software repaint policy.
2
3use api::display::Rect;
4use ostd::syscall::sys_gpu_cursor;
5
6use crate::cursor_sprite::{CURSOR_H, CURSOR_W};
7
8/// Update the logical cursor and either move the hardware sprite or damage both
9/// old and new software-sprite locations.
10pub fn update(
11    x: i32,
12    y: i32,
13    hw_cursor: bool,
14    mouse_x: &mut i32,
15    mouse_y: &mut i32,
16    pending_dirty: &mut Option<Rect>,
17) {
18    let old = rect(*mouse_x, *mouse_y);
19    *mouse_x = x;
20    *mouse_y = y;
21    ostd::io::println(&alloc::format!(
22        "[compositor] cursor at {},{}",
23        *mouse_x,
24        *mouse_y
25    ));
26    if hw_cursor {
27        let _ = sys_gpu_cursor(1, core::ptr::null(), *mouse_x as u32, *mouse_y as u32, 0, 0);
28        return;
29    }
30    let dirty = old.union(&rect(*mouse_x, *mouse_y));
31    *pending_dirty = Some(match pending_dirty.take() {
32        Some(previous) => previous.union(&dirty),
33        None => dirty,
34    });
35}
36
37fn rect(x: i32, y: i32) -> Rect {
38    Rect {
39        x,
40        y,
41        w: CURSOR_W,
42        h: CURSOR_H,
43    }
44}
