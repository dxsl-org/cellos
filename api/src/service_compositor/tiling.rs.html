1//! Tile groups on screen: frames glued edge to edge move, raise and hide
2//! together, and the edge two of them share resizes both (Haiku's Stack &
3//! Tile). The registry knows only which frames are glued along which side;
4//! every edge is read off the windows' own geometry, so the shared edges are
5//! the "tabstops" of the group and cannot drift from the windows.
6

16/// A dropped frame glues to a parallel edge at most this far away...
17const SNAP: u32 = 16;
18/// ...when they overlap along it by at least half the shorter one.
19const MIN_OVERLAP_PERCENT: i64 = 50;
20/// Smallest content a tiled frame is resized to (as for any user resize).
21const MIN_W: i32 = 64;
22const MIN_H: i32 = 48;
23
24/// Where a dropped frame would glue: to `side` of frame `a` (or `a` to its
25/// side when `dropped_first`), with the dropped content moved to `content`.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub struct Glue {
28    pub target: u64,
29    pub side: Side,
30    /// The dropped frame is `a` of the join (left of / above the target).
31    pub dropped_first: bool,
32    pub content: Rect,
33    /// The shared edge, for highlighting.
34    pub edge: Rect,
35}
36
37/// The shown window of each frame of `cap`'s tile group (`cap` first).
38pub fn group_windows(frames: &FrameRegistry, cap: u64) -> Vec<u64> {
39    let Some(frame) = frames.frame_of(cap) else {
40        return alloc::vec![cap];
41    };
42    frames
43        .tile_group(frame)
44        .into_iter()
45        .filter_map(|f| frames.selected_cap(f))
46        .collect()
47}

49/// Where the frame of `cap`, dropped with its content at `content`, would
50/// glue: the nearest parallel edge of another shown frame within [`SNAP`].
51pub fn glue_at(
52    frames: &FrameRegistry,
53    table: &SurfaceTable,
54    z_order: &ZOrder,
55    cap: u64,
56    content: Rect,
57) -> Option<Glue> {
58    let own = frames.frame_of(cap)?;
59    let g = window_decoration::frame_rect(content);
60    let mut best: Option<(u32, Glue)> = None;
61    for other in z_order.iter_top_to_bottom() {
62        let Some(frame) = frames
63            .frame_of(other)
64            .filter(|&f| f != own && frames.selected_cap(f) == Some(other))
65        else {
66            continue;
67        };
68        let Some(t) = table
69            .get(other)
70            .filter(|s| s.is_shown())
71            .map(|s| window_decoration::frame_rect(s.screen_rect()))
72        else {
73            continue;
74        };
75        let _ = frame;
76        for (gap, side, dropped_first, shift, edge) in candidates(g, t) {
77            let gap = gap.unsigned_abs();
78            if gap <= SNAP && best.is_none_or(|(b, _)| gap < b) {
79                let content = Rect {
80                    x: content.x.saturating_add(shift.0),
81                    y: content.y.saturating_add(shift.1),
82                    ..content
83                };
84                best = Some((
85                    gap,
86                    Glue {
87                        target: other,
88                        side,
89                        dropped_first,
90                        content,
91                        edge,
92                    },
93                ));
94            }
95        }
96    }
97    best.map(|(_, glue)| glue)
98}

100/// Glue the frame of `cap` (already at its snapped place) as `glue` says.
101pub fn glue(frames: &mut FrameRegistry, cap: u64, glue: Glue) -> bool {
102    let (Some(dropped), Some(target)) = (frames.frame_of(cap), frames.frame_of(glue.target)) else {
103        return false;
104    };
105    let (a, b) = if glue.dropped_first {
106        (dropped, target)
107    } else {
108        (target, dropped)
109    };
110    frames.glue(a, b, glue.side)
111}

113/// Move every other frame of `cap`'s tile group by `delta` (after `cap`'s
114/// frame moved by it). Returns the area to repaint.
115pub fn follow(
116    frames: &FrameRegistry,
117    table: &mut SurfaceTable,
118    cap: u64,
119    delta: (i32, i32),
120) -> Option<Rect> {
121    if delta == (0, 0) {
122        return None;
123    }
124    let mut area: Option<Rect> = None;
125    for other in group_windows(frames, cap)
126        .into_iter()
127        .filter(|&other| other != cap)
128    {
129        let Some(surface) = table.get_mut(other) else {
130            continue;
131        };
132        let before = window_decoration::bounds(surface.screen_rect());
133        surface.move_to(
134            surface.x.saturating_add(delta.0),
135            surface.y.saturating_add(delta.1),
136        );
137        let after = window_decoration::bounds(surface.screen_rect());
138        stacking::sync_frame_of(frames, table, other);
139        let moved = before.union(&after);
140        area = Some(area.map_or(moved, |a| a.union(&moved)));
141    }
142    area
143}

145/// Raise every frame of `cap`'s tile group above all other windows, `cap` on top.
146pub fn raise(frames: &FrameRegistry, z_order: &mut ZOrder, cap: u64) {
147    let group = group_windows(frames, cap);
148    if group.len() < 2 {
149        return;
150    }
151    for &other in group.iter().filter(|&&other| other != cap) {
152        z_order.raise(other);
153    }
154    z_order.raise(cap);
155}

157/// The window `cap` of a tile group came back from minimized: the windows
158/// the user minimized together with it come back too (not those their own
159/// apps minimized). Returns the area to repaint.
160pub fn restore_group(
161    frames: &mut FrameRegistry,
162    table: &mut SurfaceTable,
163    cap: u64,
164) -> Option<Rect> {
165    if !frames
166        .frame_of(cap)
167        .is_some_and(|frame| frames.take_group_minimized(frame))
168    {
169        return None;
170    }
171    let mut area: Option<Rect> = None;
172    for other in group_windows(frames, cap)
173        .into_iter()
174        .filter(|&other| other != cap)
175    {
176        if !frames
177            .frame_of(other)
178            .is_some_and(|frame| frames.take_group_minimized(frame))
179        {
180            continue;
181        }
182        let Some(surface) = table
183            .get_mut(other)
184            .filter(|s| s.state() == WindowState::Minimized)
185        else {
186            continue;
187        };
188        let restore = api::display::compositor_ops::RESTORE;
189        if let Some(StateTransition::StateChanged(state)) =
190            surface.request_state(other as u32, restore, surface.screen_rect())
191        {
192            if !state
193                .encode()
194                .is_ok_and(|event| stacking::deliver(surface.owner, &event))
195            {
196                surface.rollback_state_change(WindowState::Minimized);
197                continue;
198            }
199            surface.damage = Some(Rect {
200                x: 0,
201                y: 0,
202                w: surface.w,
203                h: surface.h,
204            });
205            let shown = window_decoration::bounds(surface.screen_rect());
206            area = Some(area.map_or(shown, |a| a.union(&shown)));
207        }
208    }
209    area
210}

212/// Every way `g` (dropped) can glue to `t`: (signed gap, side, dropped
213/// first, shift moving `g` flush, shared edge). Client windows may sit
214/// anywhere in `i32`, so every difference saturates.
215fn candidates(g: Rect, t: Rect) -> [(i32, Side, bool, (i32, i32), Rect); 4] {
216    let (gr, gb, tr, tb) = (end(g.x, g.w), end(g.y, g.h), end(t.x, t.w), end(t.y, t.h));
217    let v_ok = overlaps(g.y, gb, t.y, tb);
218    let h_ok = overlaps(g.x, gr, t.x, tr);
219    let far = i32::MAX / 2;
220    let pick = |ok: bool, gap: i32| if ok { gap } else { far };
221    let span = |from: i32, to: i32| to.saturating_sub(from).max(1) as u32;
222    let column = |x: i32| Rect {
223        x: x.saturating_sub(1),
224        y: g.y.max(t.y),
225        w: 2,
226        h: span(g.y.max(t.y), gb.min(tb)),
227    };
228    let row = |y: i32| Rect {
229        x: g.x.max(t.x),
230        y: y.saturating_sub(1),
231        w: span(g.x.max(t.x), gr.min(tr)),
232        h: 2,
233    };
234    let (right, left) = (t.x.saturating_sub(gr), g.x.saturating_sub(tr));
235    let (below, above) = (t.y.saturating_sub(gb), g.y.saturating_sub(tb));
236    [
237        (
238            pick(v_ok, right),
239            Side::Right,
240            true,
241            (right, 0),
242            column(t.x),
243        ),
244        (
245            pick(v_ok, left),
246            Side::Right,
247            false,
248            (left.saturating_neg(), 0),
249            column(tr),
250        ),
251        (pick(h_ok, below), Side::Below, true, (0, below), row(t.y)),
252        (
253            pick(h_ok, above),
254            Side::Below,
255            false,
256            (0, above.saturating_neg()),
257            row(tb),
258        ),
259    ]
260}

262/// Whether two spans overlap by at least half the shorter one.
263fn overlaps(a0: i32, a1: i32, b0: i32, b1: i32) -> bool {
264    let overlap = i64::from(a1.min(b1)) - i64::from(a0.max(b0));
265    let shorter = (i64::from(a1) - i64::from(a0)).min(i64::from(b1) - i64::from(b0));
266    overlap > 0 && overlap * 100 >= shorter * MIN_OVERLAP_PERCENT
267}
268
269fn end(start: i32, extent: u32) -> i32 {
270    start.saturating_add(extent.min(i32::MAX as u32) as i32)
271}
