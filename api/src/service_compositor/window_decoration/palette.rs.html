1//! Chrome colours per system palette (CellOS Linear, contracts.md "Appearance").
2//!
3//! The tab and frame copy Haiku's geometry, not its colours: every value here
4//! derives from the CellOS Light / Dark / Paper tokens. Hairlines sit a step
5//! darker (lighter on Dark) than the `border` token so a window separates from
6//! any wallpaper; tab text keeps >= 4.5:1 and control glyphs >= 3:1 (WCAG
7//! non-text contrast) against their background in every state.
8
9use api::display::AppearancePalette;
10use ostd::display::linear::{Bgra, Shadow};
11
12/// Colours for one focus state.
13pub(crate) struct FocusColors {
14    pub(crate) tab: Bgra,
15    pub(crate) outline: Bgra,
16    pub(crate) glyph: Bgra,
17    pub(crate) title: Bgra,
18}
19
20pub(crate) struct ChromePalette {
21    pub(crate) active: FocusColors,
22    pub(crate) inactive: FocusColors,
23    /// App-icon and menu-item background under the pointer, and while open.
24    pub(crate) hover: Bgra,
25    pub(crate) pressed: Bgra,
26    /// Fill of a stacked tab that is not the shown one: a step behind both
27    /// focus states, drawn with the inactive title and glyph colours.
28    pub(crate) stacked: Bgra,
29    /// Outline of a Win+drag's ghost frame and of the tabs it would join.
30    pub(crate) accent: Bgra,
31    pub(crate) shadow: Shadow,
32}
33
34const fn bgra(r: u8, g: u8, b: u8) -> Bgra {
35    [b, g, r, 0xff]
36}
37
38/// Soft shadow from a light above the screen: fades over 10px around the
39/// caster moved 3px down (13px below, inside `SHADOW_MARGIN`). Black, so it
40/// darkens whatever lies below; only its strength varies by palette.
41const fn shadow(opacity: u8) -> Shadow {
42    Shadow {
43        color: bgra(0, 0, 0),
44        opacity,
45        spread: 10,
46        offset_x: 0,
47        offset_y: 3,
48    }
49}

111pub(crate) const fn chrome(palette: AppearancePalette) -> &'static ChromePalette {
112    match palette {
113        AppearancePalette::Light => &LIGHT,
114        AppearancePalette::Dark => &DARK,
115        AppearancePalette::Paper => &PAPER,
116    }
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122
123    fn luminance(c: Bgra) -> f64 {
124        let lin = |v: u8| {
125            let s = f64::from(v) / 255.0;
126            if s <= 0.04045 {
127                s / 12.92
128            } else {
129                ((s + 0.055) / 1.055).powf(2.4)
130            }
131        };
132        0.2126 * lin(c[2]) + 0.7152 * lin(c[1]) + 0.0722 * lin(c[0])
133    }
134
135    fn contrast(a: Bgra, b: Bgra) -> f64 {
136        let (la, lb) = (luminance(a), luminance(b));
137        (la.max(lb) + 0.05) / (la.min(lb) + 0.05)
138    }
139
140    #[test]
141    fn tab_titles_meet_text_contrast_in_every_palette_and_focus_state() {
142        for palette in [
143            AppearancePalette::Light,
144            AppearancePalette::Dark,
145            AppearancePalette::Paper,
146        ] {
147            let chrome = chrome(palette);
148            for colors in [&chrome.active, &chrome.inactive] {
149                let ratio = contrast(colors.title, colors.tab);
150                assert!(ratio >= 4.5, "{palette:?}: title contrast {ratio:.2}");
151                assert!(
152                    contrast(colors.glyph, colors.tab) >= 3.0,
153                    "{palette:?}: glyph contrast"
154                );
155            }
156            // Tabs behind the shown one use the inactive inks on their fill.
157            let ratio = contrast(chrome.inactive.title, chrome.stacked);
158            assert!(
159                ratio >= 4.5,
160                "{palette:?}: stacked title contrast {ratio:.2}"
161            );
162            assert!(
163                contrast(chrome.inactive.glyph, chrome.stacked) >= 3.0,
164                "{palette:?}: stacked glyph contrast"
165            );
166            // The drag outline stands out from the tabs it frames.
167            assert!(
168                contrast(chrome.accent, chrome.active.tab) >= 3.0,
169                "{palette:?}: accent"
170            );
171            // A hovered control draws the active glyph on the hover/pressed fill.
172            for fill in [chrome.hover, chrome.pressed] {
173                assert!(
174                    contrast(chrome.active.glyph, fill) >= 3.0,
175                    "{palette:?}: hovered glyph"
176                );
177            }
178        }
179    }
