9/// Glyph box side. Odd, so the close diagonals cross on a single pixel
10/// instead of a 2x2 blob.
11const GLYPH: i32 = 9;
12/// Space between the app icon box and the title, and after the title.
13const TITLE_GAP: i32 = 6;

15/// One Cell Tab: the window's app icon and title.
16#[derive(Clone, Copy)]
17pub struct TabLabel<'a> {
18    pub icon: LineIcon,
19    pub title: &'a str,
20    /// The window is closing: its tab stays, faded, until it is gone.
21    pub dimmed: bool,
22}

24/// What a frame's chrome shows this frame.
25pub struct Style<'a> {
26    pub active: bool,
27    pub palette: &'static ChromePalette,
28    /// The frame's tabs in order; at least one.
29    pub tabs: &'a [TabLabel<'a>],
30    /// Index into `tabs` of the shown window.
31    pub selected: usize,
32    /// A lone tab's slide along the top edge ([`Tabs::offset`]).
33    pub offset: u32,
34    /// `Some(open)` while the pointer is over the shown tab's app icon;
35    /// `open` while its window menu is showing.
36    pub menu_button: Option<bool>,
37}

39impl Style<'_> {
40    fn strip(&self) -> Tabs {
41        Tabs {
42            count: self.tabs.len().max(1),
43            selected: self.selected,
44            offset: self.offset,
45        }
46    }
47}

49/// Paint shadow, tab fills, hairlines, app icons and titles inside `dirty`.
50/// The shown tab takes the frame's focus colours; the other tabs of a stack
51/// sit back in the palette's stacked-tab fill.
52///
53/// Content pixels are not touched except where the body outline covers them;
54/// callers blit content afterwards with [`super::content_row_insets`].
55pub(super) fn paint(
56    pixels: &mut [u8],
57    width: u32,
58    height: u32,
59    content: Rect,
60    dirty: Rect,
61    style: &Style<'_>,
62    font: &mut TitleFont,
63) {
64    let colors = if style.active {
65        &style.palette.active
66    } else {
67        &style.palette.inactive
68    };
69    let tabs = style.strip();
70    let [strip, body] = shapes(content, tabs);
71    {
72        let mut canvas = Canvas::new(pixels, width, height, width as usize * 4)
73            .clip_to(dirty.x, dirty.y, dirty.w, dirty.h);
74        canvas.drop_shadow(&[strip, body], &style.palette.shadow);
75        canvas.stroke_round(&body, colors.outline);
76        for (index, label) in style.tabs.iter().enumerate() {
77            let shown = index == style.selected;
78            let slot = slot_shape(content, tabs, index);
79            let fill = if shown {
80                colors.tab
81            } else {
82                style.palette.stacked
83            };
84            canvas.fill_round(
85                &RoundRect {
86                    h: slot.h.saturating_sub(1),
87                    ..slot
88                },
89                fill,
90            );
91            canvas.stroke_round(&slot, colors.outline);
92            let button = icon_box(tab_slot(content, tabs, index));
93            let mut ink = if shown {
94                colors.glyph
95            } else {
96                style.palette.inactive.glyph
97            };
98            if let (true, Some(open)) = (shown, style.menu_button) {
99                let fill = if open {
100                    style.palette.pressed
101                } else {
102                    style.palette.hover
103                };
104                let shape = RoundRect::new(
105                    button.x,
106                    button.y,
107                    button.w,
108                    button.h,
109                    Radii::all(CONTROL_RADIUS),
110                );
111                canvas.fill_round(&shape, fill);
112                ink = style.palette.active.glyph;
113            }
114            let inset = (button.w.saturating_sub(ICON_SIZE) / 2) as i32;
115            paint_icon(
116                &mut canvas,
117                label.icon,
118                button.x + inset,
119                button.y + inset,
120                ink,
121            );
122        }
123    }
124    for (index, label) in style.tabs.iter().enumerate() {
125        let slot = tab_slot(content, tabs, index);
126        let button = icon_box(slot);
127        let left = end(button.x, button.w) + TITLE_GAP;
128        let right = end(slot.x, slot.w) - TITLE_GAP;
129        if right <= left {
130            continue;
131        }
132        let area = Rect {
133            x: left,
134            y: slot.y + 1,
135            w: (right - left) as u32,
136            h: (TITLE - 2) as u32,
137        };
138        let ink = if label.dimmed {
139            style.palette.inactive.glyph
140        } else if index == style.selected {
141            colors.title
142        } else {
143            style.palette.inactive.title
144        };
145        font.draw(pixels, width, height, area, dirty, label.title, ink);
146    }
147}

149/// Outline of tab `index`: only the strip's outer top corners are rounded.
150fn slot_shape(content: Rect, tabs: Tabs, index: usize) -> RoundRect {
151    let slot = tab_slot(content, tabs, index);
152    let first = index == 0;
153    let last = index + 1 >= tabs.count;
154    let radii = Radii {
155        top_left: if first { RADIUS } else { 0 },
156        top_right: if last { RADIUS } else { 0 },
157        ..Radii::default()
158    };
159    RoundRect::new(slot.x, slot.y, slot.w, slot.h, radii)
160}

162/// 1px glyph for a window-menu item, centred in the square `cell`.
163pub(super) fn paint_glyph(
164    canvas: &mut Canvas<'_>,
165    cell: Rect,
166    control: Control,
167    restore: bool,
168    color: Bgra,
169) {
170    let x = cell.x + (cell.w as i32 - GLYPH) / 2;
171    let y = cell.y + (cell.h as i32 - GLYPH) / 2;
172    let last = GLYPH - 1;
173    match control {
174        Control::Close => {
175            canvas.line(x, y, x + last, y + last, color);
176            canvas.line(x + last, y, x, y + last, color);
177        }
178        Control::Maximize if restore => {
179            // A front frame with the back frame's top and right edges peeking
180            // out: the classic "restore down" glyph.
181            canvas.hline(x + 2, y, 7, color);
182            canvas.vline(x + 8, y, 7, color);
183            canvas.stroke_rect(x, y + 2, 7, 7, color);
184        }
185        Control::Maximize => {
186            let frame = RoundRect::new(x, y, GLYPH as u32, GLYPH as u32, Radii::all(2));
187            canvas.stroke_round(&frame, color);
188        }
189        Control::Minimize => canvas.hline(x, y + GLYPH / 2, GLYPH as u32, color),
190    }
