1//! System palette for compositor-drawn chrome, set by the shell.
2
3use api::display::{AppearanceChanged, AppearancePalette, SetAppearance, SurfaceRole};
4use ostd::syscall::sys_try_send;
5
6use crate::surface_table::SurfaceTable;
7
8/// Chrome palette plus the authority and revision that set it.
9///
10/// Only the owner of a Panel surface (the system shell; the table lets one
11/// owner hold Panels at a time) may change it. Revisions are scoped to that
12/// owner, so a restarted shell starting again at revision 1 is accepted while
13/// a stale or replayed request from the same shell is refused.
14pub struct Appearance {
15    palette: AppearancePalette,
16    owner: usize,
17    revision: u32,
18    /// Palette changes since boot, from 1; announced to clients.
19    generation: u32,
20}
21
22impl Appearance {
23    pub const fn new() -> Self {
24        Self {
25            palette: AppearancePalette::Light,
26            owner: 0,
27            revision: 0,
28            generation: 1,
29        }
30    }
31
32    pub const fn palette(&self) -> AppearancePalette {
33        self.palette
34    }
35
36    /// The event announcing the current palette.
37    pub const fn event(&self) -> AppearanceChanged {
38        AppearanceChanged {
39            palette: self.palette,
40            generation: self.generation,
41        }
42    }
43
44    /// Tell `owner` the current palette; never blocks (a full queue drops it,
45    /// the next change or window attach repeats it).
46    pub fn announce_to(&self, owner: usize) {
47        let _ = sys_try_send(owner, &self.event().encode());
48    }
49
50    /// Tell every distinct surface owner the current palette.
51    pub fn broadcast(&self, table: &SurfaceTable) {
52        let mut owners = table.owners();
53        owners.sort_unstable();
54        owners.dedup();
55        for owner in owners {
56            self.announce_to(owner);
57        }
58    }
59
60    /// Apply `request` from `sender`; returns whether the visible palette changed.
61    pub fn apply(&mut self, request: &SetAppearance, sender: usize, table: &SurfaceTable) -> bool {
62        let is_shell_panel = table
63            .get(u64::from(request.cap))
64            .is_some_and(|surface| surface.owner == sender && surface.role == SurfaceRole::Panel);
65        if !is_shell_panel || (sender == self.owner && request.revision <= self.revision) {
66            return false;
67        }
68        let changed = self.palette != request.palette;
69        self.palette = request.palette;
70        self.owner = sender;
71        self.revision = request.revision;
72        if changed {
73            self.generation = self.generation.wrapping_add(1).max(1);
74        }
75        changed
76    }
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82
83    #[test]
84    fn only_the_panel_owner_sets_it_with_growing_revisions() {
85        let mut table = SurfaceTable::new();
86        let panel = table.create(0, 0, 8, 8, 7, SurfaceRole::Panel).unwrap();
87        let window = table
88            .create(0, 0, 8, 8, 9, SurfaceRole::Interactive)
89            .unwrap();
90        let mut appearance = Appearance::new();
91        let request =
92            |cap: u64, palette, revision| SetAppearance::new(cap as u32, palette, revision);
93
94        assert!(!appearance.apply(&request(window, AppearancePalette::Dark, 1), 9, &table));
95        assert!(
96            !appearance.apply(&request(panel, AppearancePalette::Dark, 1), 9, &table),
97            "not the owner"
98        );
99        assert!(appearance.apply(&request(panel, AppearancePalette::Dark, 1), 7, &table));
100        assert!(
101            !appearance.apply(&request(panel, AppearancePalette::Paper, 1), 7, &table),
102            "replay"
103        );
104        assert_eq!(appearance.palette(), AppearancePalette::Dark);
105        assert!(appearance.apply(&request(panel, AppearancePalette::Paper, 2), 7, &table));
106
107        // The shell restarts: its new Panel starts over at revision 1.
108        let _ = table.remove(panel);
109        let reborn = table.create(0, 0, 8, 8, 11, SurfaceRole::Panel).unwrap();
110        assert!(appearance.apply(&request(reborn, AppearancePalette::Light, 1), 11, &table));
111        assert_eq!(appearance.palette(), AppearancePalette::Light);
112        // Three visible changes after boot's generation 1.
113        assert_eq!(appearance.event().generation, 4);
114    }
