1//! Clatter DH adapter with an opaque KMS-owned static key.
2//!
3//! Ephemeral X25519 keys remain local to the broker. Static-key operations use
4//! only a KMS handle and binding epoch; the node private scalar is never stored
5//! in this representation.
6
7use clatter::bytearray::{ByteArray, SensitiveByteArray};
8use clatter::crypto::dh::X25519;
9use clatter::error::{DhError, DhResult};
10use clatter::traits::{CryptoComponent, Dh, Rng};
11use clatter::KeyPair;
12use types::kms::{BindingEpoch, NodeIdentityHandle};
13
14const PRIVATE_REPR_LEN: usize = 77;
15const TAG_LOCAL: u8 = 1;
16const TAG_OPAQUE: u8 = 2;
17const LOCAL_SECRET: core::ops::Range<usize> = 1..33;
18const HANDLE: core::ops::Range<usize> = 1..5;
19const BINDING_EPOCH: core::ops::Range<usize> = 5..13;
20const PUBLIC_KEY: core::ops::Range<usize> = 45..77;
21
22type PrivateRepr = SensitiveByteArray<[u8; PRIVATE_REPR_LEN]>;
23type PublicKey = <X25519 as Dh>::PubKey;
24type DhOutput = <X25519 as Dh>::Output;
25
26/// X25519 implementation whose static private-key representation is opaque.
27#[derive(Clone)]
28pub struct KmsBackedX25519;
29
30impl CryptoComponent for KmsBackedX25519 {
31    fn name() -> &'static str {
32        X25519::name()
33    }
34}
35
36impl Dh for KmsBackedX25519 {
37    type PrivateKey = PrivateRepr;
38    type PubKey = PublicKey;
39    type Output = DhOutput;
40
41    fn genkey_rng<R: Rng>(rng: &mut R) -> DhResult<KeyPair<PublicKey, PrivateRepr>> {
42        let local = X25519::genkey_rng(rng)?;
43        let mut encoded = PrivateRepr::new_zero();
44        encoded[0] = TAG_LOCAL;
45        encoded[LOCAL_SECRET].copy_from_slice(local.secret.as_slice());
46        encoded[PUBLIC_KEY].copy_from_slice(&local.public);
47        Ok(KeyPair::new(local.public, encoded))
48    }
49
50    fn pubkey(key: &PrivateRepr) -> PublicKey {
51        if !matches!(key[0], TAG_LOCAL | TAG_OPAQUE) {
52            return [0; 32];
53        }
54        let mut public = [0u8; 32];
55        public.copy_from_slice(&key[PUBLIC_KEY]);
56        public
57    }
58
59    fn dh(key: &PrivateRepr, peer: &PublicKey) -> DhResult<DhOutput> {
60        match key[0] {
61            TAG_LOCAL => {
62                let local = <X25519 as Dh>::PrivateKey::from_slice(&key[LOCAL_SECRET]);
63                X25519::dh(&local, peer).and_then(require_nonzero_output)
64            }
65            TAG_OPAQUE => opaque_static_dh(key, peer),
66            _ => Err(DhError::KeyGeneration),
67        }
68    }
69}

71/// Public identity plus the KMS authority needed for static DH.
72pub struct OpaqueStaticKey {
73    handle: NodeIdentityHandle,
74    binding_epoch: BindingEpoch,
75    public_key: PublicKey,
76}
77
78impl OpaqueStaticKey {
79    /// Construct a non-exportable static key reference.
80    pub fn new(
81        handle: NodeIdentityHandle,
82        binding_epoch: BindingEpoch,
83        public_key: PublicKey,
84    ) -> Option<Self> {
85        if handle.0 == 0 || binding_epoch.0 == 0 || public_key.iter().all(|byte| *byte == 0) {
86            return None;
87        }
88        Some(Self {
89            handle,
90            binding_epoch,
91            public_key,
92        })
93    }
94
95    /// Convert metadata into Clatter's keypair container without private bytes.
96    pub fn into_keypair(self) -> KeyPair<PublicKey, PrivateRepr> {
97        let mut encoded = PrivateRepr::new_zero();
98        encoded[0] = TAG_OPAQUE;
99        encoded[HANDLE].copy_from_slice(&self.handle.0.to_le_bytes());
100        encoded[BINDING_EPOCH].copy_from_slice(&self.binding_epoch.0.to_le_bytes());
101        encoded[PUBLIC_KEY].copy_from_slice(&self.public_key);
102        KeyPair::new(self.public_key, encoded)
103    }
104}
105
106fn opaque_static_dh(key: &PrivateRepr, peer: &PublicKey) -> DhResult<DhOutput> {
107    let handle = NodeIdentityHandle(read_u32(key.as_slice(), HANDLE.start));
108    let binding_epoch = BindingEpoch(read_u64(key.as_slice(), BINDING_EPOCH.start));
109    #[cfg(target_os = "none")]
110    {
111        let client = ostd::clients::KmsClient::connect().map_err(|_| DhError::KeyGeneration)?;
112        let secret = client
113            .noise_static_dh(handle, binding_epoch, peer)
114            .map_err(|_| DhError::KeyGeneration)?;
115        require_nonzero_output(DhOutput::from_slice(&secret))
116    }
117    #[cfg(not(target_os = "none"))]
118    {
119        let _ = (handle, binding_epoch, peer);
120        Err(DhError::KeyGeneration)
121    }
122}
123
124fn require_nonzero_output(output: DhOutput) -> DhResult<DhOutput> {
125    if output.as_slice().iter().all(|byte| *byte == 0) {
126        return Err(DhError::KeyGeneration);
127    }
128    Ok(output)
129}
130
131fn read_u32(bytes: &[u8], at: usize) -> u32 {
132    u32::from_le_bytes(bytes[at..at + 4].try_into().expect("fixed opaque handle"))
133}
134
135fn read_u64(bytes: &[u8], at: usize) -> u64 {
136    u64::from_le_bytes(bytes[at..at + 8].try_into().expect("fixed binding epoch"))
137}
