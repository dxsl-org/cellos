6pub(crate) const EXPORT_REGISTRY_MAX_BYTES: usize = 4 * 1024;
7const EXPORT_REGISTRY_DIR: &str = "/etc/cellos";
8const EXPORT_REGISTRY_BASENAME: &[u8] = b"c2c-exports.cfg";
9const EXPORT_REGISTRY_PATH: &str = "/etc/cellos/c2c-exports.cfg";

11pub(crate) trait RegistrySource {
12    fn list_dir(&mut self, path: &str) -> Result<Vec<u8>, ()>;
13    fn stat(&mut self, path: &str) -> Result<(u64, bool), ()>;
14    fn read_file_bounded(&mut self, path: &str, max_bytes: usize) -> Result<Vec<u8>, ()>;
15}

17pub(crate) fn load_remote_exports<S: RegistrySource>(source: &mut S) -> RemoteExports {
18    let listing = match source.list_dir(EXPORT_REGISTRY_DIR) {
19        Ok(listing) => listing,
20        Err(()) => return RemoteExports::invalid(),
21    };
22    if !listing_has_registry_entry(&listing) {
23        return RemoteExports::absent();
24    }
25    let (size, is_dir) = match source.stat(EXPORT_REGISTRY_PATH) {
26        Ok(meta) => meta,
27        Err(()) => return RemoteExports::invalid(),
28    };
29    if is_dir || size > EXPORT_REGISTRY_MAX_BYTES as u64 {
30        return RemoteExports::invalid();
31    }
32    let data = match source.read_file_bounded(EXPORT_REGISTRY_PATH, EXPORT_REGISTRY_MAX_BYTES) {
33        Ok(data) => data,
34        Err(()) => return RemoteExports::invalid(),
35    };
36    let registry = match ExportRegistry::parse_bytes(&data) {
37        Ok(registry) => registry,
38        Err(_) => return RemoteExports::invalid(),
39    };
40    RemoteExports::from_registry(registry)
41}
42
43fn listing_has_registry_entry(listing: &[u8]) -> bool {
44    listing
45        .split(|&b| b == b'\n')
46        .any(|entry| entry == EXPORT_REGISTRY_BASENAME)
47}
