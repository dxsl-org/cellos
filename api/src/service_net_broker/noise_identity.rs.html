1/// Build the Noise prologue with identities ordered by protocol role.
2///
3/// Both peers must hash `cluster_id || initiator_id || responder_id`; local
4/// endpoint order would reverse the identities on the responder and make the
5/// handshake transcripts diverge.
6///
7/// `cluster_id` selects the cluster transcript. `local_node_id` and
8/// `remote_node_id` describe this endpoint's view; `is_initiator` converts
9/// that view into protocol-role order. The returned 72 bytes contain the
10/// little-endian cluster ID followed by initiator and responder NodeIds.
11pub fn handshake_prologue(
12    cluster_id: u64,
13    local_node_id: &[u8; 32],
14    remote_node_id: &[u8; 32],
15    is_initiator: bool,
16) -> [u8; 72] {
17    let (initiator_id, responder_id) = if is_initiator {
18        (local_node_id, remote_node_id)
19    } else {
20        (remote_node_id, local_node_id)
21    };
22    let mut prologue = [0u8; 72];
23    prologue[..8].copy_from_slice(&cluster_id.to_le_bytes());
24    prologue[8..40].copy_from_slice(initiator_id);
25    prologue[40..72].copy_from_slice(responder_id);
26    prologue
27}
