2//! BrokerIdentity — per-machine network identity and peer address book.
3//!
4//! G1 model: CellNetId = X25519 static public key from the Noise keypair.
5//! No separate Ed25519 key material is generated; signing comes in G2.
6//!
7//! Config: /etc/cellos/cluster.cfg, simple key=value format.
8//! Export registry: /etc/cellos/c2c-exports.cfg, non-secret key=value format.
9//! See doc-comment on `load_config` for the expected layout.
10
11// reason: BrokerIdentity itself is constructed and driven from main.rs, but
12// several accessors (peer_count, get_peer_by_node_id, update_reflexive) exist
13// for callers that aren't wired yet — connection_manager::reflexive_or_direct
14// reads `reflexive_addr` but nothing ever calls `update_reflexive` because
15// stun::query_reflexive_addr is itself unwired from the dispatch loop.
16#![allow(dead_code)]
17
18extern crate alloc;
19
20use crate::export_registry::{load_remote_exports, RegistrySource, RemoteExports};
21use crate::peer_config::{parse_peer_config_bytes, MAX_PEERS};
22use crate::relay_config::{parse_relay_endpoint_bytes, RelayEndpoint};
23use api::cluster::{CellNetId, PeerTicket};
24use ostd::clients::vfs::VfsClient;
25use ostd::io::{print, println};
26
27const CFG_PATH: &str = "/etc/cellos/cluster.cfg";
28const CFG_READ_MAX_BYTES: usize = 4 * 1024;
29
30#[cfg(test)]
31mod tests;
32
33/// Per-machine network identity and peer address book.
34pub struct BrokerIdentity {
35    pub node_id: CellNetId,
36    peers: [Option<PeerTicket>; MAX_PEERS],
37    peers_len: usize,
38    remote_exports: RemoteExports,
39    relay_endpoint: Option<RelayEndpoint>,
40    /// Reflexive public address discovered via STUN. Updated by `stun` module.
41    pub reflexive_addr: Option<([u8; 4], u16)>,
42}
43
44impl BrokerIdentity {
45    /// Construct from the X25519 static public key (G1 identity model).
46    pub fn from_static_pub(static_pub: [u8; 32]) -> Self {
47        Self {
48            node_id: CellNetId::from_bytes(static_pub),
49            peers: [const { None }; MAX_PEERS],
50            peers_len: 0,
51            remote_exports: RemoteExports::absent(),
52            relay_endpoint: None,
53            reflexive_addr: None,
54        }
55    }
56
57    /// Parse /etc/cellos/cluster.cfg into the peer table.
58    ///
59    /// Expected format (flat key=value, blank lines and `#` comments ignored):
60    /// ```text
61    /// peer_count=2
62    /// peer_0_node_id=deadbeef...  # 64 hex chars = 32 bytes
63    /// peer_0_relay_ip=1.2.3.4
64    /// peer_0_relay_port=8765
65    /// peer_0_direct=192.168.1.10:4521   # optional
66    /// relay_ip=10.0.0.5
67    /// relay_port=443
68    /// relay_hostname=relay.example
69    /// peer_1_node_id=...
70    /// peer_1_relay_ip=...
71    /// peer_1_relay_port=...
72    /// ```
73    pub fn load_config(&mut self) {
74        let data = match VfsClient::new().read_file_bounded(CFG_PATH, CFG_READ_MAX_BYTES) {
75            Ok(d) => d,
76            Err(_) => {
77                println("[net-broker] failed to read cluster.cfg — no peers configured");
78                return;
79            }
80        };
81        self.load_config_bytes(&data);
82    }
83
84    fn load_config_bytes(&mut self, data: &[u8]) {
85        let (peers, peers_len) = parse_peer_config_bytes(data);
86        self.peers = peers;
87        self.peers_len = peers_len;
88        self.relay_endpoint = match parse_relay_endpoint_bytes(data) {
89            Ok(endpoint) => endpoint,
90            Err(_) => {
91                println("[net-broker] invalid relay endpoint — relay remains disabled");
92                None
93            }
94        };
95        print("[net-broker] loaded peers from cluster.cfg: count=");
96        ostd::io::print_usize(self.peers_len);
97        println("");
98    }
99
100    /// Load the non-secret C2C export registry and freeze remote state.
101    pub fn load_export_registry(&mut self) {
102        let mut source = VfsRegistrySource::new();
103        self.remote_exports = load_remote_exports(&mut source);
104    }
105
106    pub fn peer_count(&self) -> usize {
107        self.peers_len
108    }
109
110    pub fn get_peer(&self, idx: usize) -> Option<&PeerTicket> {
111        self.peers.get(idx)?.as_ref()
112    }
113
114    pub fn get_peer_by_node_id(&self, node_id: &CellNetId) -> Option<&PeerTicket> {
115        self.peers[..self.peers_len]
116            .iter()
117            .find_map(|p| p.as_ref().filter(|t| &t.node_id == node_id))
118    }
119
120    pub fn remote_exports(&self) -> &RemoteExports {
121        &self.remote_exports
122    }
123
124    /// Return the validated relay endpoint without activating relay traffic.
125    pub fn relay_endpoint(&self) -> Option<&RelayEndpoint> {
126        self.relay_endpoint.as_ref()
127    }
128    pub fn update_reflexive(&mut self, ip: [u8; 4], port: u16) {
129        self.reflexive_addr = Some((ip, port));
130    }
131}

133struct VfsRegistrySource {
134    vfs: VfsClient,
135}
136
137impl VfsRegistrySource {
138    fn new() -> Self {
139        Self {
140            vfs: VfsClient::new(),
141        }
142    }
143}
144
145impl RegistrySource for VfsRegistrySource {
146    fn list_dir(&mut self, path: &str) -> Result<alloc::vec::Vec<u8>, ()> {
147        self.vfs.list_dir(path).map_err(|_| ())
148    }
149
150    fn stat(&mut self, path: &str) -> Result<(u64, bool), ()> {
151        self.vfs.stat(path).map_err(|_| ())
152    }
153
154    fn read_file_bounded(
155        &mut self,
156        path: &str,
157        max_bytes: usize,
158    ) -> Result<alloc::vec::Vec<u8>, ()> {
159        self.vfs.read_file_bounded(path, max_bytes).map_err(|_| ())
160    }
