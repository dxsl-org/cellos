7use crate::c2c_envelope::{RetryClass, MAX_C2C_PAYLOAD};
8use types::{Entry, EntryState, SourceWindow};

10pub use types::{
11    C2cStatus, CachedReply, DedupDecision, DedupError, DedupKey, DEDUP_CAPACITY, DEDUP_TTL_MS,
12    SOURCE_WINDOW_CAPACITY,
13};
14
15/// No-allocation cache. In-flight entries are never evicted.
16pub struct DedupCache {
17    entries: [Option<Entry>; DEDUP_CAPACITY],
18    sources: [Option<SourceWindow>; SOURCE_WINDOW_CAPACITY],
19}
20
21pub const DEDUP_STATIC_BYTES: usize = core::mem::size_of::<DedupCache>();
22
23impl DedupCache {
24    pub const fn new() -> Self {
25        Self {
26            entries: [None; DEDUP_CAPACITY],
27            sources: [None; SOURCE_WINDOW_CAPACITY],
28        }
29    }
30
31    /// Admit a request or classify its duplicate without dispatching it.
32    pub fn begin(&mut self, key: DedupKey, retry_class: RetryClass, now_ms: u64) -> DedupDecision {
33        if !key.is_valid() {
34            return DedupDecision::Indeterminate;
35        }
36        if self.is_stale_boot(key) {
37            return DedupDecision::Indeterminate;
38        }
39        if let Some(index) = self.find(key) {
40            let entry = self.entries[index].as_mut().expect("found");
41            if entry.retry_class != retry_class {
42                return DedupDecision::Indeterminate;
43            }
44            let expired = now_ms.saturating_sub(entry.first_seen_ms) >= DEDUP_TTL_MS;
45            return match (entry.state, expired, retry_class) {
46                (EntryState::Accepted | EntryState::Dispatched, _, _) => DedupDecision::Busy,
47                (EntryState::Completed, false, _) => DedupDecision::Replay(index),
48                (EntryState::Completed, true, RetryClass::Idempotent) => {
49                    self.entries[index] = Some(Entry::accepted(key, retry_class, now_ms));
50                    DedupDecision::Dispatch
51                }
52                (EntryState::Completed, true, _) => {
53                    entry.state = EntryState::ExpiredNonReplayable;
54                    entry.payload_len = 0;
55                    DedupDecision::Indeterminate
56                }
57                (EntryState::ExpiredNonReplayable, _, _) => DedupDecision::Indeterminate,
58            };
59        }
60
61        let source_slot = match self.source_slot(key) {
62            Ok(slot) => slot,
63            Err(decision) => return decision,
64        };
65        let slot = self
66            .entries
67            .iter()
68            .position(Option::is_none)
69            .or_else(|| self.evictable_slot(now_ms));
70        let Some(index) = slot else {
71            return DedupDecision::Busy;
72        };
73        self.entries[index] = Some(Entry::accepted(key, retry_class, now_ms));
74        self.sources[source_slot]
75            .as_mut()
76            .expect("source slot reserved")
77            .high_request_id = key.request_id;
78        DedupDecision::Dispatch
79    }
80
81    /// Mark a newly admitted request as locally dispatched.
82    pub fn mark_dispatched(&mut self, key: DedupKey) -> Result<(), DedupError> {
83        let entry = self.entry_mut(key)?;
84        if entry.state != EntryState::Accepted {
85            return Err(DedupError::InvalidTransition);
86        }
87        entry.state = EntryState::Dispatched;
88        Ok(())
89    }
90
91    /// Cache one completed response for replay within the retention window.
92    pub fn complete(
93        &mut self,
94        key: DedupKey,
95        status: C2cStatus,
96        payload: &[u8],
97    ) -> Result<(), DedupError> {
98        if payload.len() > MAX_C2C_PAYLOAD {
99            return Err(DedupError::PayloadTooLarge);
100        }
101        let entry = self.entry_mut(key)?;
102        if !matches!(entry.state, EntryState::Accepted | EntryState::Dispatched) {
103            return Err(DedupError::InvalidTransition);
104        }
105        entry.payload[..payload.len()].copy_from_slice(payload);
106        entry.payload_len = payload.len();
107        entry.status = status;
108        entry.state = EntryState::Completed;
109        Ok(())
110    }
111
112    /// Borrow a cached response selected by `DedupDecision::Replay`.
113    pub fn replay(&self, slot: usize, key: DedupKey, now_ms: u64) -> Option<CachedReply<'_>> {
114        let entry = self.entries.get(slot)?.as_ref()?;
115        if entry.key != key
116            || entry.state != EntryState::Completed
117            || now_ms.saturating_sub(entry.first_seen_ms) >= DEDUP_TTL_MS
118        {
119            return None;
120        }
121        Some(CachedReply {
122            status: entry.status,
123            payload: &entry.payload[..entry.payload_len],
124        })
125    }
126
127    pub fn len(&self) -> usize {
128        self.entries.iter().flatten().count()
129    }
130
131    pub(crate) fn retire_entries(&mut self) {
132        self.entries = [None; DEDUP_CAPACITY];
133    }
134
135    fn find(&self, key: DedupKey) -> Option<usize> {
136        self.entries
137            .iter()
138            .position(|entry| entry.is_some_and(|entry| entry.key == key))
139    }
140    fn entry_mut(&mut self, key: DedupKey) -> Result<&mut Entry, DedupError> {
141        let index = self.find(key).ok_or(DedupError::Stale)?;
142        Ok(self.entries[index].as_mut().expect("found"))
143    }
144
145    fn evictable_slot(&self, now_ms: u64) -> Option<usize> {
146        self.entries.iter().position(|entry| {
147            entry.is_some_and(|entry| {
148                matches!(
149                    entry.state,
150                    EntryState::Completed | EntryState::ExpiredNonReplayable
151                ) && now_ms.saturating_sub(entry.first_seen_ms) >= DEDUP_TTL_MS
152            })
153        })
154    }
155}
156
157impl Default for DedupCache {
158    fn default() -> Self {
159        Self::new()
160    }
161}
