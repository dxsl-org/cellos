2//! Canonical bounded V1 envelope for authenticated C2C transport records.
3
4use api::services::cluster::{CellNetId, ClusterId};
5pub use types::c2c::{RelativeDeadline, RetryClass, ServerEpoch};
6
7pub const C2C_VERSION: u8 = 1;
8pub const C2C_HEADER_LEN: usize = 112;
9// V1 forbids streaming. The frame must fit both the attested local request body
10// and one conservative net-cell TcpSend after Noise adds its AEAD tag.
11pub const NOISE_TAG_LEN: usize = 16;
12const NOISE_PLAINTEXT_CAP: usize = api::ipc::NET_TCP_INLINE_DATA_MAX - NOISE_TAG_LEN;
13const NOISE_BODY_CAP: usize = NOISE_PLAINTEXT_CAP - C2C_HEADER_LEN;
14pub const MAX_C2C_PAYLOAD: usize = if crate::local_ingress::MAX_REQUEST_BODY < NOISE_BODY_CAP {
15    crate::local_ingress::MAX_REQUEST_BODY
16} else {
17    NOISE_BODY_CAP
18};
19pub const MAX_C2C_FRAME: usize = C2C_HEADER_LEN + MAX_C2C_PAYLOAD;
20
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22#[repr(u8)]
23pub enum EnvelopeKind {
24    Lookup = 1,
25    Request = 2,
26    Response = 3,
27    Busy = 4,
28    Indeterminate = 5,
29    Cancel = 6,
30    Heartbeat = 7,
31}
32
33impl EnvelopeKind {
34    fn decode(value: u8) -> Option<Self> {
35        Some(match value {
36            1 => Self::Lookup,
37            2 => Self::Request,
38            3 => Self::Response,
39            4 => Self::Busy,
40            5 => Self::Indeterminate,
41            6 => Self::Cancel,
42            7 => Self::Heartbeat,
43            _ => return None,
44        })
45    }
46}
47
48#[derive(Clone, Copy, Debug, PartialEq, Eq)]
49pub enum EnvelopeError {
50    ShortBuffer,
51    UnsupportedVersion,
52    UnknownKind,
53    UnknownRetryClass,
54    NonCanonical,
55    InvalidIdentity,
56    PayloadTooLarge,
57    LengthMismatch,
58}
59
60/// Borrowed decoded envelope. Identity is trusted only after Noise authenticates the record.
61#[derive(Clone, Copy, Debug, PartialEq, Eq)]
62pub struct C2cEnvelope<'a> {
63    pub kind: EnvelopeKind,
64    pub retry_class: RetryClass,
65    pub request_id: u64,
66    pub src_node: CellNetId,
67    pub dst_node: CellNetId,
68    pub src_boot_epoch: u64,
69    pub dst_server_epoch: ServerEpoch,
70    pub cluster_id: ClusterId,
71    pub service_id: u16,
72    pub export_id: u16,
73    pub relative_deadline: RelativeDeadline,
74    pub payload: &'a [u8],
75}
76
77impl C2cEnvelope<'_> {
78    /// Encode one canonical frame into `out`.
79    ///
80    /// Returns the encoded length or a deterministic validation error.
81    pub fn encode(&self, out: &mut [u8]) -> Result<usize, EnvelopeError> {
82        validate(self)?;
83        let len = C2C_HEADER_LEN + self.payload.len();
84        if out.len() < len {
85            return Err(EnvelopeError::ShortBuffer);
86        }
87        out[..len].fill(0);
88        out[0] = C2C_VERSION;
89        out[1] = self.kind as u8;
90        out[2] = self.retry_class as u8;
91        put_u16(out, 4, self.payload.len() as u16);
92        put_u64(out, 8, self.request_id);
93        out[16..48].copy_from_slice(&self.src_node.0);
94        out[48..80].copy_from_slice(&self.dst_node.0);
95        put_u64(out, 80, self.src_boot_epoch);
96        put_u64(out, 88, self.dst_server_epoch.get());
97        put_u64(out, 96, self.cluster_id.0);
98        put_u16(out, 104, self.service_id);
99        put_u16(out, 106, self.export_id);
100        out[108..112].copy_from_slice(&self.relative_deadline.milliseconds().to_le_bytes());
101        out[C2C_HEADER_LEN..len].copy_from_slice(self.payload);
102        Ok(len)
103    }
104}
105
106/// Decode one exact canonical V1 frame without allocation.
107pub fn decode(bytes: &[u8]) -> Result<C2cEnvelope<'_>, EnvelopeError> {
108    if bytes.len() < C2C_HEADER_LEN {
109        return Err(EnvelopeError::ShortBuffer);
110    }
111    if bytes[0] != C2C_VERSION {
112        return Err(EnvelopeError::UnsupportedVersion);
113    }
114    let kind = EnvelopeKind::decode(bytes[1]).ok_or(EnvelopeError::UnknownKind)?;
115    let retry_class = RetryClass::from_wire(bytes[2]).ok_or(EnvelopeError::UnknownRetryClass)?;
116    if bytes[3] != 0 || bytes[6..8] != [0; 2] {
117        return Err(EnvelopeError::NonCanonical);
118    }
119    let payload_len = read_u16(bytes, 4) as usize;
120    if payload_len > MAX_C2C_PAYLOAD {
121        return Err(EnvelopeError::PayloadTooLarge);
122    }
123    if bytes.len() != C2C_HEADER_LEN + payload_len {
124        return Err(EnvelopeError::LengthMismatch);
125    }
126    let envelope = C2cEnvelope {
127        kind,
128        retry_class,
129        request_id: read_u64(bytes, 8),
130        src_node: CellNetId(bytes[16..48].try_into().expect("fixed range")),
131        dst_node: CellNetId(bytes[48..80].try_into().expect("fixed range")),
132        src_boot_epoch: read_u64(bytes, 80),
133        dst_server_epoch: ServerEpoch::new(read_u64(bytes, 88))
134            .ok_or(EnvelopeError::InvalidIdentity)?,
135        cluster_id: ClusterId(read_u64(bytes, 96)),
136        service_id: read_u16(bytes, 104),
137        export_id: read_u16(bytes, 106),
138        relative_deadline: RelativeDeadline::new(u32::from_le_bytes(
139            bytes[108..112].try_into().expect("fixed range"),
140        ))
141        .ok_or(EnvelopeError::InvalidIdentity)?,
142        payload: &bytes[C2C_HEADER_LEN..],
143    };
144    validate(&envelope)?;
145    Ok(envelope)
146}
147
148fn validate(envelope: &C2cEnvelope<'_>) -> Result<(), EnvelopeError> {
149    if envelope.payload.len() > MAX_C2C_PAYLOAD {
150        return Err(EnvelopeError::PayloadTooLarge);
151    }
152    if envelope.request_id == 0
153        || envelope.src_node.0.iter().all(|byte| *byte == 0)
154        || envelope.dst_node.0.iter().all(|byte| *byte == 0)
155        || envelope.src_boot_epoch == 0
156        || envelope.cluster_id.0 == 0
157        || envelope.service_id == 0
158        || envelope.export_id == 0
159    {
160        return Err(EnvelopeError::InvalidIdentity);
161    }
162    Ok(())
163}
164
165fn put_u16(out: &mut [u8], offset: usize, value: u16) {
166    out[offset..offset + 2].copy_from_slice(&value.to_le_bytes());
167}
168
169fn put_u64(out: &mut [u8], offset: usize, value: u64) {
170    out[offset..offset + 8].copy_from_slice(&value.to_le_bytes());
171}
172
173fn read_u16(bytes: &[u8], offset: usize) -> u16 {
174    u16::from_le_bytes(bytes[offset..offset + 2].try_into().expect("fixed range"))
175}
176
177fn read_u64(bytes: &[u8], offset: usize) -> u64 {
178    u64::from_le_bytes(bytes[offset..offset + 8].try_into().expect("fixed range"))
179}
