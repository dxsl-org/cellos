1//! Shell command parser — tokenizes a line and builds an `Ast`.
2//!
3//! Supported syntax (v1.0):
4//!   - Simple command: `ls /bin`
5//!   - Pipeline: `cat /etc/hosts | grep 127`
6//!   - Output redirect: `echo hello > /tmp/a.txt`
7//!   - Input redirect: `cat < /tmp/a.txt`
8//!   - Append redirect: `echo hi >> /tmp/log.txt`
9//!   - Background: `sleep 10 &`
10//!   - Sequence: `echo a ; echo b`
11//!
12//! Intentionally simple: no subshells, no quoting beyond `"..."`, no globs.
13
14extern crate alloc;
15
16use alloc::string::String;
17use alloc::vec::Vec;
18
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum QuoteStyle {
21    None,
22    Single,
23    Double,
24}
25
26#[derive(Debug, Clone, PartialEq, Eq)]
27pub struct Word {
28    pub text: String,
29    pub quote: QuoteStyle,
30    pub segments: Vec<WordSegment>,
31}
32
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub struct WordSegment {
35    pub text: String,
36    pub quote: QuoteStyle,
37}
38
39impl Word {
40    #[cfg(test)]
41    fn new(text: String, quote: QuoteStyle) -> Self {
42        Self {
43            segments: alloc::vec![WordSegment {
44                text: text.clone(),
45                quote,
46            }],
47            text,
48            quote,
49        }
50    }
51
52    fn from_segments(segments: Vec<WordSegment>) -> Self {
53        let mut text = String::new();
54        for segment in &segments {
55            text.push_str(&segment.text);
56        }
57        let quote = if segments.len() == 1 {
58            segments[0].quote
59        } else {
60            QuoteStyle::None
61        };
62        Self {
63            text,
64            quote,
65            segments,
66        }
67    }
68
69    fn is_unquoted(&self) -> bool {
70        self.segments
71            .iter()
72            .all(|segment| segment.quote == QuoteStyle::None)
73    }
74}

76/// One redirect target.
77#[derive(Debug, Clone, PartialEq, Eq)]
78pub enum Redirect {
79    /// `> path`
80    StdoutTo(Word),
81    /// `>> path`
82    StdoutAppend(Word),
83    /// `< path`
84    StdinFrom(Word),
85    /// `2> path`
86    StderrTo(Word),
87}

89/// A single command with its arguments and any redirects.
90#[derive(Debug, Clone)]
91pub struct Cmd {
92    /// `argv[0]` and arguments.
93    pub argv: Vec<Word>,
94    /// Redirects attached to this command.
95    pub redirects: Vec<Redirect>,
96    /// Set when the command carries syntax the grammar cannot honour (a redirect
97    /// operator without a target word). The executor rejects it before any effect
98    /// instead of silently dropping the operator.
99    pub malformed: Option<&'static str>,
100}

102impl Cmd {
103    fn new() -> Self {
104        Cmd {
105            argv: Vec::new(),
106            redirects: Vec::new(),
107            malformed: None,
108        }
109    }

111    /// True if the command has no name (empty line or whitespace-only).
112    pub fn is_empty(&self) -> bool {
113        self.argv.is_empty()
114    }

117/// Top-level abstract syntax tree for one shell line.
118#[derive(Debug, Clone)]
119pub enum Ast {
120    /// Empty input.
121    Empty,
122    /// A single simple command.
123    Simple(Cmd),
124    /// `cmd1 | cmd2 | …` — pipeline of commands.
125    Pipeline(Vec<Cmd>),
126    /// `cmd1 | cmd2 &` — a background marker on a multi-stage pipeline. The
127    /// executor refuses it before any effect; it is never run in the foreground.
128    BackgroundPipeline(Vec<Cmd>),
129    /// `cmd &` — run in background.
130    Background(Cmd),
131    /// Syntax the grammar rejects (misplaced `&`, empty pipeline stage, redirect
132    /// without a target). Carries the one-line reason; executes nothing.
133    Invalid(&'static str),
134    /// `cmd1 ; cmd2` — sequential execution.
135    Sequence(Vec<Ast>),
136    /// `cmd1 && cmd2` — run cmd2 only if cmd1 exits 0 (success).
137    And(alloc::boxed::Box<Ast>, alloc::boxed::Box<Ast>),
138    /// `cmd1 || cmd2` — run cmd2 only if cmd1 exits non-zero (failure).
139    Or(alloc::boxed::Box<Ast>, alloc::boxed::Box<Ast>),
140    /// `while COND; do BODY; done` — loop while COND exits 0.
141    While {
142        cond: alloc::boxed::Box<Ast>,
143        body: alloc::boxed::Box<Ast>,
144    },
145    /// `name() { body; }` — define a shell function.
146    ///
147    /// The body is stored as a reconstructed string; the executor registers it
148    /// in the function table so later invocations of `name` run the body.
149    FuncDef {
150        name: alloc::string::String,
151        body: alloc::string::String,
152    },
153    /// `case EXPR in pattern) BODY ;; … esac` — pattern-match dispatch.
154    ///
155    /// `expr` is the unexpanded word after `case`; expansion happens at
156    /// execute time.  Patterns are exact strings or `*` (wildcard).
157    /// Note: the whole statement must be on a single line or in a single
158    /// `source`-script line; mixing with `;`-sequences on the same line is
159    /// not supported (the outer `;` split would fragment the `;;` markers).
160    Case {
161        expr: alloc::string::String,
162        arms: alloc::vec::Vec<(alloc::string::String, alloc::boxed::Box<Ast>)>,
163    },
164    /// `for VAR in word1 word2 …; do BODY; done` — iterate over a word list.
165    ///
166    /// Sets `$VAR` to each word in order, runs BODY, then advances. `$VAR`
167    /// expansion in BODY uses the same static var store as `VAR=value`.
168    For {
169        var: alloc::string::String,
170        words: alloc::vec::Vec<alloc::string::String>,
171        body: alloc::boxed::Box<Ast>,
172    },
173    /// `if COND; then BODY; fi` — conditional execution.
174    ///
175    /// `cond` exit-code 0 → run `then_b`; non-zero → run `else_b` if present.
176    If {
177        cond: alloc::boxed::Box<Ast>,
178        then_b: alloc::boxed::Box<Ast>,
179        else_b: Option<alloc::boxed::Box<Ast>>,
180    },
181}

185/// Raw token before AST construction.
186#[derive(Debug, Clone, PartialEq, Eq)]
187enum Tok {
188    Word(Word),
189    Pipe,           // |
190    RedirectOut,    // >
191    RedirectAppend, // >>
192    RedirectIn,     // <
193    RedirectErr,    // 2>
194    Ampersand,      // &   (background marker — single &)
195    And,            // &&  (short-circuit AND)
196    Or,             // ||  (short-circuit OR)
197    Semicolon,      // ;
198    LBrace,         // {  (function body open)
199    RBrace,         // }  (function body close)
200    // ── Conditional keywords ─────────────────────────────────────────────────
201    // These variants are NEVER emitted by the tokenizer — `if`/`then`/`else`/`fi`
202    // always remain as Word tokens.  parse_if_stmt detects them by string
203    // comparison so they never silently disappear from external command arguments
204    // (e.g. `lua -e "if x then ... end"` must reach Lua intact).
205    #[allow(dead_code)] // reserved — kept for exhaustive match arms in parse_cmd
206    If,
207    #[allow(dead_code)] // reserved
208    Then,
209    #[allow(dead_code)] // reserved
210    Else,
211    #[allow(dead_code)] // reserved
212    Fi,
213}

215/// Tokenize a shell input line.
216///
217/// Handles:
218/// - Whitespace separation
219/// - Simple `"..."` double-quoted strings (no escape sequences)
220/// - Single-character operators: `|`, `<`, `>`, `&`, `;`
221/// - Two-character operators: `>>`, `2>`
222fn tokenize(line: &str) -> Vec<Tok> {
223    let mut tokens = Vec::new();
224    let mut chars = line.chars().peekable();
225    let mut current: Vec<WordSegment> = Vec::new();
226
227    fn append_segment(current: &mut Vec<WordSegment>, text: String, quote: QuoteStyle) {
228        if let Some(last) = current.last_mut() {
229            if last.quote == quote {
230                last.text.push_str(&text);
231                return;
232            }
233        }
234        current.push(WordSegment { text, quote });
235    }
236
237    fn push_word(tokens: &mut Vec<Tok>, current: &mut Vec<WordSegment>) {
238        if !current.is_empty() {
239            tokens.push(Tok::Word(Word::from_segments(core::mem::take(current))));
240        }
241    }
242
243    fn read_quoted(chars: &mut core::iter::Peekable<core::str::Chars<'_>>, end: char) -> String {
244        let mut text = String::new();
245        loop {
246            match chars.next() {
247                Some(ch) if ch == end => break,
248                Some(ch) => text.push(ch),
249                None => break,
250            }
251        }
252        text
253    }
254
255    macro_rules! flush {
256        () => {
257            push_word(&mut tokens, &mut current);
258        };
259    }
260
261    while let Some(c) = chars.next() {
262        match c {
263            ' ' | '\t' => {
264                flush!();
265            }
266            '"' | '\'' => {
267                let quote = if c == '"' {
268                    QuoteStyle::Double
269                } else {
270                    QuoteStyle::Single
271                };
272                let quoted = read_quoted(&mut chars, c);
273                append_segment(&mut current, quoted, quote);
274            }
275            '|' => {
276                flush!();
277                if chars.peek() == Some(&'|') {
278                    chars.next();
279                    tokens.push(Tok::Or);
280                } else {
281                    tokens.push(Tok::Pipe);
282                }
283            }
284            '&' => {
285                flush!();
286                if chars.peek() == Some(&'&') {
287                    chars.next();
288                    tokens.push(Tok::And);
289                } else {
290                    tokens.push(Tok::Ampersand);
291                }
292            }
293            ';' => {
294                flush!();
295                tokens.push(Tok::Semicolon);
296            }
297            // `${NAME}` / `${#NAME}` — braced parameter expansion stays inside the word
298            // (expanded later by the executor); an unclosed `${` is kept literally.
299            '$' if chars.peek() == Some(&'{') => {
300                chars.next(); // consume '{'
301                let mut parameter = String::from("${");
302                for ch in chars.by_ref() {
303                    parameter.push(ch);
304                    if ch == '}' {
305                        break;
306                    }
307                }
308                append_segment(&mut current, parameter, QuoteStyle::None);
309            }
310            '{' => {
311                flush!();
312                tokens.push(Tok::LBrace);
313            }
314            '}' => {
315                flush!();
316                tokens.push(Tok::RBrace);
317            }
318            '<' => {
319                flush!();
320                tokens.push(Tok::RedirectIn);
321            }
322            '>' => {
323                flush!();
324                if chars.peek() == Some(&'>') {
325                    chars.next();
326                    tokens.push(Tok::RedirectAppend);
327                } else {
328                    tokens.push(Tok::RedirectOut);
329                }
330            }
331            '2' if chars.peek() == Some(&'>') => {
332                // "2>" — only if current buffer is empty (i.e. not part of a word).
333                if current.is_empty() {
334                    chars.next();
335                    tokens.push(Tok::RedirectErr);
336                } else {
337                    append_segment(&mut current, String::from(c), QuoteStyle::None);
338                }
339            }
340            // `$(...)` — command substitution.  Consume everything up to the
341            // matching `)` into the current word so spaces inside don't split
342            // the token.  Nested `$( $() )` increments depth correctly.
343            '$' if chars.peek() == Some(&'(') => {
344                chars.next(); // consume '('
345                let mut substitution = String::from("$(");
346                let mut depth = 1usize;
347                loop {
348                    match chars.next() {
349                        None => break, // unclosed: pass through
350                        Some('(') => {
351                            depth += 1;
352                            substitution.push('(');
353                        }
354                        Some(')') => {
355                            depth -= 1;
356                            substitution.push(')');
357                            if depth == 0 {
358                                break;
359                            }
360                        }
361                        Some(ch) => {
362                            substitution.push(ch);
363                        }
364                    }
365                }
366                append_segment(&mut current, substitution, QuoteStyle::None);
367            }
368            other => {
369                append_segment(&mut current, String::from(other), QuoteStyle::None);
370            }
371        }
372    }
373    flush!();
374    // All tokens remain as their natural type — no keyword conversion.
375    // The if-statement parser detects `if`/`then`/`else`/`fi` by string
376    // comparison on Word tokens so they are never eaten from command arguments.
377    tokens
378}

382/// Parse a shell line into an `Ast`.
383///
384/// Returns `Ast::Empty` for blank input.
385pub fn parse(line: &str) -> Ast {
386    let tokens = tokenize(line.trim());
387    if tokens.is_empty() {
388        return Ast::Empty;
389    }
390
391    // `if...then...fi` must be parsed BEFORE semicolon splitting, because the
392    // semicolons inside an if-statement are structural (not sequence separators).
393    // Keywords remain as Word tokens (not converted) so they survive in external
394    // command argument strings (e.g. `lua -e "if x then ... end"`).
395    // `name() { body; }` — function definition.
396    // name() is a single Word token ending with "()" (no spaces inside).
397    if let Some(Tok::Word(w)) = tokens.first() {
398        if w.is_unquoted()
399            && w.text.ends_with("()")
400            && tokens.get(1) == Some(&Tok::LBrace)
401            && tokens.last() == Some(&Tok::RBrace)
402        {
403            let name = String::from(&w.text[..w.text.len() - 2]);
404            let body = tokens_to_string(&tokens[2..tokens.len() - 1]);
405            return Ast::FuncDef { name, body };
406        }
407    }
408
409    if is_kw_token(tokens.first(), "if") {
410        return parse_if_stmt(&tokens);
411    }
412    if is_kw_token(tokens.first(), "while") {
413        return parse_while_stmt(&tokens);
414    }
415    if is_kw_token(tokens.first(), "for") {
416        return parse_for_stmt(&tokens);
417    }
418    // `case` must be parsed before semicolon splitting because `;;` arm
419    // separators are also Semicolon tokens and would fragment the statement.
420    if is_kw_token(tokens.first(), "case") {
421        return parse_case_stmt(&tokens);
422    }
423
424    // Split on `;` into sub-sequences first.
425    let segments: Vec<&[Tok]> = split_on(&tokens, |t| t == &Tok::Semicolon);
426    if segments.len() > 1 {
427        let seq: Vec<Ast> = segments.iter().map(|seg| parse_pipeline(seg)).collect();
428        return Ast::Sequence(seq);
429    }
430
431    parse_pipeline(&tokens)
432}

434/// Parse a token sub-slice that may contain `;`-separated commands.
435///
436/// Equivalent to the main `parse()` body but operating on a pre-tokenized
437/// slice — used by `parse_if_stmt` to parse condition and body sections.
438fn parse_tokens(tokens: &[Tok]) -> Ast {
439    // Strip leading/trailing semicolons that linger from the structural split.
440    let start = tokens
441        .iter()
442        .position(|t| t != &Tok::Semicolon)
443        .unwrap_or(tokens.len());
444    let end = tokens
445        .iter()
446        .rposition(|t| t != &Tok::Semicolon)
447        .map(|i| i + 1)
448        .unwrap_or(0);
449    let tokens = &tokens[start..end];
450    if tokens.is_empty() {
451        return Ast::Empty;
452    }
453    let segments: Vec<&[Tok]> = split_on(tokens, |t| t == &Tok::Semicolon);
454    if segments.len() > 1 {
455        let seq: Vec<Ast> = segments.iter().map(|seg| parse_pipeline(seg)).collect();
456        return Ast::Sequence(seq);
457    }
458    parse_pipeline(tokens)
459}

461/// Parse `if COND; then BODY; fi` or `if COND; then BODY; else BODY; fi`.
462///
463/// Handles any number of semicolons around the keywords — the structure is
464/// determined by the `Then`, `Else`, and `Fi` token positions.
465/// Serialize a token slice back to a space-separated shell line.
466///
467/// Used by function definition to store the body as a re-parseable string.
468fn tokens_to_string(tokens: &[Tok]) -> String {
469    let parts: alloc::vec::Vec<&str> = tokens
470        .iter()
471        .map(|t| match t {
472            Tok::Word(w) => w.text.as_str(),
473            Tok::Pipe => "|",
474            Tok::And => "&&",
475            Tok::Or => "||",
476            Tok::Semicolon => ";",
477            Tok::Ampersand => "&",
478            Tok::RedirectOut => ">",
479            Tok::RedirectAppend => ">>",
480            Tok::RedirectIn => "<",
481            Tok::RedirectErr => "2>",
482            Tok::LBrace => "{",
483            Tok::RBrace => "}",
484            _ => "",
485        })
486        .collect();
487    parts.join(" ")
488}

490/// Helper: returns true when a token is the keyword word `w`.
491fn is_kw(tok: &Tok, w: &str) -> bool {
492    matches!(tok, Tok::Word(word) if word.is_unquoted() && word.text == w)
493}
494
495fn is_kw_token(tok: Option<&Tok>, w: &str) -> bool {
496    matches!(tok, Some(tok) if is_kw(tok, w))
497}

499/// Parse `while COND; do BODY; done`.
500///
501/// Keywords stay as `Word` tokens (no Tok variants) so `while`/`do`/`done`
502/// survive intact when used as external command arguments.  Malformed input
503/// (missing `do` or `done`) calls `parse_tokens` — NOT `parse()` — to avoid
504/// re-dispatching on the leading `while` and recursing infinitely.
505/// Parse `for VAR in word1 word2 …; do BODY; done`.
506///
507/// Keywords stay as `Word` tokens (same Phase N/O rule) so `for`/`in`/`do`/`done`
508/// survive as external command arguments.  Malformed input falls back to
509/// `parse_tokens` (not `parse()`) to prevent infinite recursion.
510fn parse_for_stmt(tokens: &[Tok]) -> Ast {
511    // tokens[1] = variable name; tokens[2] should be Word("in").
512    let var = match tokens.get(1) {
513        Some(Tok::Word(w)) if w.is_unquoted() && w.text != "in" => w.text.clone(),
514        _ => return parse_tokens(tokens),
515    };
516    let in_pos = match tokens.iter().position(|t| is_kw(t, "in")) {
517        Some(p) => p,
518        None => return parse_tokens(tokens),
519    };
520    let do_pos = tokens.iter().position(|t| is_kw(t, "do"));
521    let done_pos = tokens.iter().rposition(|t| is_kw(t, "done"));
522    let (dp, np) = match (do_pos, done_pos) {
523        (Some(d), Some(n)) if n > d => (d, n),
524        _ => return parse_tokens(tokens),
525    };
526    // Word list: tokens between `in` and `do`, stripping Semicolons.
527    let words: alloc::vec::Vec<alloc::string::String> = tokens[in_pos + 1..dp]
528        .iter()
529        .filter_map(|t| {
530            if let Tok::Word(w) = t {
531                Some(w.text.clone())
532            } else {
533                None
534            }
535        })
536        .collect();
537    let body = parse_tokens(&tokens[dp + 1..np]);
538    Ast::For {
539        var,
540        words,
541        body: alloc::boxed::Box::new(body),
542    }
543}

545/// Parse `case EXPR in pattern1) BODY ;; pattern2) BODY ;; esac`.
546///
547/// `;;` is two consecutive `Tok::Semicolon` tokens.  Patterns are the first
548/// token of each arm with a trailing `)` stripped.  `*` is a catch-all.
549/// Malformed input (missing `in` or `esac`) falls back to `parse_tokens`.
550fn parse_case_stmt(tokens: &[Tok]) -> Ast {
551    let in_pos = tokens.iter().position(|t| is_kw(t, "in"));
552    let esac_pos = tokens.iter().rposition(|t| is_kw(t, "esac"));
553    let (ip, ep) = match (in_pos, esac_pos) {
554        (Some(i), Some(e)) if e > i => (i, e),
555        _ => return parse_tokens(tokens),
556    };
557
558    // Expression: single word between `case` and `in`.
559    let expr = match tokens.get(1) {
560        Some(Tok::Word(w)) => w.text.clone(),
561        _ => String::new(),
562    };
563
564    // Arms: tokens[ip+1..ep] split on `;;` (two consecutive Semicolons).
565    let arm_tokens = &tokens[ip + 1..ep];
566    let mut arms: alloc::vec::Vec<(String, alloc::boxed::Box<Ast>)> = alloc::vec::Vec::new();
567    let mut start = 0;
568    let mut i = 0;
569    while i < arm_tokens.len() {
570        if i + 1 < arm_tokens.len()
571            && arm_tokens[i] == Tok::Semicolon
572            && arm_tokens[i + 1] == Tok::Semicolon
573        {
574            push_arm(&arm_tokens[start..i], &mut arms);
575            i += 2;
576            start = i;
577        } else {
578            i += 1;
579        }
580    }
581    push_arm(&arm_tokens[start..], &mut arms); // last arm (no trailing ;;)
582
583    Ast::Case { expr, arms }
584}
585
586fn push_arm(tokens: &[Tok], arms: &mut alloc::vec::Vec<(String, alloc::boxed::Box<Ast>)>) {
587    // Strip leading/trailing Semicolons left from the split.
588    let start = tokens
589        .iter()
590        .position(|t| t != &Tok::Semicolon)
591        .unwrap_or(tokens.len());
592    let tokens = &tokens[start..];
593    if let Some(Tok::Word(pat)) = tokens.first() {
594        let pattern = String::from(pat.text.trim_end_matches(')'));
595        let body = parse_tokens(&tokens[1..]);
596        arms.push((pattern, alloc::boxed::Box::new(body)));
597    }
598}

600fn parse_while_stmt(tokens: &[Tok]) -> Ast {
601    let do_pos = tokens.iter().position(|t| is_kw(t, "do"));
602    let done_pos = tokens.iter().rposition(|t| is_kw(t, "done"));
603    let (dp, np) = match (do_pos, done_pos) {
604        (Some(d), Some(n)) if n > d => (d, n),
605        _ => return parse_tokens(tokens), // malformed: fall back without infinite recursion
606    };
607    let cond = parse_tokens(&tokens[1..dp]);
608    let body = parse_tokens(&tokens[dp + 1..np]);
609    Ast::While {
610        cond: alloc::boxed::Box::new(cond),
611        body: alloc::boxed::Box::new(body),
612    }
613}

615fn parse_if_stmt(tokens: &[Tok]) -> Ast {
616    // Locate structural keywords after the leading `if` Word.
617    // Keywords are plain Word tokens — never converted — so they survive intact
618    // in external command argument strings.
619    let then_pos = tokens
620        .iter()
621        .position(|t| is_kw(t, "then"))
622        .unwrap_or(tokens.len());
623    let else_pos = tokens.iter().position(|t| is_kw(t, "else"));
624    let fi_pos = tokens
625        .iter()
626        .rposition(|t| is_kw(t, "fi"))
627        .unwrap_or(tokens.len());
628
629    // Condition: tokens[1..then_pos]   (skip leading `If`)
630    let cond_slice = &tokens[1..then_pos];
631    let cond = parse_tokens(cond_slice);
632
633    // Then body: tokens[then_pos+1..else_or_fi]
634    let then_end = else_pos.unwrap_or(fi_pos);
635    let then_slice = &tokens[then_pos + 1..then_end];
636    let then_b = parse_tokens(then_slice);
637
638    // Else body (optional): tokens[else_pos+1..fi_pos]
639    let else_b = else_pos.map(|ep| {
640        let slice = &tokens[ep + 1..fi_pos];
641        alloc::boxed::Box::new(parse_tokens(slice))
642    });
643
644    Ast::If {
645        cond: alloc::boxed::Box::new(cond),
646        then_b: alloc::boxed::Box::new(then_b),
647        else_b,
648    }
649}

651fn parse_pipeline(tokens: &[Tok]) -> Ast {
652    // `&&` / `||` have lower precedence than pipelines — check first.
653    // Split on the FIRST occurrence; right side is parsed recursively so
654    // `A && B && C` builds `And(A, And(B, C))` with left-to-right evaluation.
655    if let Some(pos) = tokens.iter().position(|t| t == &Tok::And || t == &Tok::Or) {
656        let left = parse_pipeline(&tokens[..pos]);
657        let right = parse_pipeline(&tokens[pos + 1..]);
658        return match &tokens[pos] {
659            Tok::And => Ast::And(alloc::boxed::Box::new(left), alloc::boxed::Box::new(right)),
660            Tok::Or => Ast::Or(alloc::boxed::Box::new(left), alloc::boxed::Box::new(right)),
661            _ => unreachable!(),
662        };
663    }
664
665    // A background marker is only meaningful as the last token of the command.
666    let background = tokens.last() == Some(&Tok::Ampersand);
667    let body = if background {
668        &tokens[..tokens.len() - 1]
669    } else {
670        tokens
671    };
672    if body.contains(&Tok::Ampersand) {
673        return Ast::Invalid("`&` is only valid as the final token of a command");
674    }
675
676    let pipe_segs: Vec<&[Tok]> = split_on(body, |t| t == &Tok::Pipe);
677    let mut cmds: Vec<Cmd> = pipe_segs.iter().map(|seg| parse_cmd(seg)).collect();
678    if cmds.len() > 1 {
679        // An empty stage (`a |`, `| a`) is a syntax error, never silently dropped.
680        if cmds.iter().any(Cmd::is_empty) {
681            return Ast::Invalid("empty pipeline stage");
682        }
683    } else {
684        cmds.retain(|c| !c.is_empty());
685    }
686    if let Some(reason) = cmds.iter().find_map(|c| c.malformed) {
687        return Ast::Invalid(reason);
688    }
689
690    if background {
691        return match cmds.len() {
692            0 => Ast::Empty,
693            1 => Ast::Background(cmds.into_iter().next().unwrap_or_else(Cmd::new)),
694            _ => Ast::BackgroundPipeline(cmds),
695        };
696    }
697    match cmds.len() {
698        0 => Ast::Empty,
699        1 => Ast::Simple(cmds.into_iter().next().unwrap_or_else(Cmd::new)),
700        _ => Ast::Pipeline(cmds),
701    }
702}
703
704/// Parse one command segment (no `|`, `;` or `&`).
705fn parse_cmd(tokens: &[Tok]) -> Cmd {
706    const MISSING_TARGET: &str = "redirect operator without a target word";
707    let mut cmd = Cmd::new();
708    let mut iter = tokens.iter();
709
710    while let Some(tok) = iter.next() {
711        let redirect: fn(Word) -> Redirect = match tok {
712            Tok::Word(w) => {
713                cmd.argv.push(w.clone());
714                continue;
715            }
716            Tok::RedirectOut => Redirect::StdoutTo,
717            Tok::RedirectAppend => Redirect::StdoutAppend,
718            Tok::RedirectIn => Redirect::StdinFrom,
719            Tok::RedirectErr => Redirect::StderrTo,
720            _ => continue,
721        };
722        match iter.next() {
723            Some(Tok::Word(path)) => cmd.redirects.push(redirect(path.clone())),
724            _ => cmd.malformed = Some(MISSING_TARGET),
725        }
726    }
727    cmd
728}
729
730/// Split a token slice on positions where `pred` returns true.
731fn split_on<F>(tokens: &[Tok], pred: F) -> Vec<&[Tok]>
732where
733    F: Fn(&Tok) -> bool,
734{
735    let mut result = Vec::new();
736    let mut start = 0;
737    for (i, tok) in tokens.iter().enumerate() {
738        if pred(tok) {
739            result.push(&tokens[start..i]);
740            start = i + 1;
741        }
742    }
743    result.push(&tokens[start..]);
744    result
745}
