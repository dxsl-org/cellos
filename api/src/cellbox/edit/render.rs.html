1//! Draw the editor into a Ratatui buffer: text rows, a status line and a
2//! prompt / message / key-hint line. Control characters are shown escaped
3//! (`^X`, `<XX>`), never sent as terminal data; tabs expand to 8-column stops.
4//! Long lines scroll horizontally; no bytes are ever inserted for display.
5

35/// One drawn unit: a base scalar with any zero-width scalars after it.
36struct Cell {
37    text: String,
38    byte: usize,
39    width: usize,
40}
41
42fn cells(text: &str) -> Vec<Cell> {
43    let mut out: Vec<Cell> = Vec::new();
44    let mut col = 0;
45    for (byte, c) in text.char_indices() {
46        let (shown, width) = match c {
47            '\t' => {
48                let next = (col / TAB_STOP + 1) * TAB_STOP;
49                (" ".repeat(next - col), next - col)
50            }
51            '\0'..='\u{1f}' | '\u{7f}' => (format!("^{}", char::from((c as u8) ^ 0x40)), 2),
52            '\u{80}'..='\u{9f}' => (format!("<{:02X}>", c as u32), 4),
53            _ => (String::from(c), cell_width(c)),
54        };
55        match out.last_mut() {
56            Some(last) if width == 0 => last.text.push_str(&shown),
57            _ => out.push(Cell { text: shown, byte, width }),
58        }
59        col += width;
60    }
61    out
62}

64fn syntax_style(theme: &Theme, kind: Kind) -> Style {
65    let rgb = match kind {
66        Kind::Comment => ANSI16[8],
67        Kind::String => ANSI16[2],
68        Kind::Number => ANSI16[5],
69        Kind::Keyword => ANSI16[4],
70    };
71    theme.base().fg(theme.color(rgb))
72}

74/// Scroll so the cursor is visible, draw, and return the cursor cell.
75pub fn draw(editor: &mut Editor, theme: &Theme, area: Rect, out: &mut Buffer) -> Option<Position> {
76    out.set_style(area, theme.base());
77    let width = usize::from(area.width);
78    let rows = usize::from(area.height.saturating_sub(2)).max(1);
79    editor.rows = rows;
80    let cursor = editor.buf.cursor();
81    let cursor_col = editor.buf.column(cursor.line, cursor.byte, &cell_width);
82    if cursor.line < editor.top {
83        editor.top = cursor.line;
84    } else if cursor.line >= editor.top + rows {
85        editor.top = cursor.line + 1 - rows;
86    }
87    if cursor_col < editor.left {
88        editor.left = cursor_col;
89    } else if cursor_col >= editor.left + width.max(1) {
90        editor.left = cursor_col + 1 - width.max(1);
91    }
92
93    let lang = editor.lang();
94    let selection = editor.buf.selection();
95    // Highlight carry from the start of the file to the first visible line.
96    let mut carry = Carry::None;
97    let mut spans: Vec<Span> = Vec::new();
98    if let Some(lang) = lang {
99        for line in &editor.buf.lines()[..editor.top.min(editor.buf.line_count())] {
100            spans.clear();
101            carry = highlight(lang, carry, &line.text, &mut spans);
102        }
103    }
104    if editor.help {
105        for (row, text) in HELP.iter().enumerate().take(rows) {
106            out.set_stringn(area.x, area.y + row as u16, text, width, theme.base());
107        }
108    } else {
109        for row in 0..rows {
110            let index = editor.top + row;
111            let Some(line) = editor.buf.lines().get(index) else { break };
112            spans.clear();
113            if let Some(lang) = lang {
114                carry = highlight(lang, carry, &line.text, &mut spans);
115            }
116            let y = area.y + row as u16;
117            let mut col = 0;
118            for cell in cells(&line.text) {
119                let start = col;
120                col += cell.width;
121                if col <= editor.left || start >= editor.left + width {
122                    continue;
123                }
124                let x = area.x + (start.saturating_sub(editor.left)) as u16;
125                let selected = selection.is_some_and(|(a, b)| {
126                    let here = text_engine::editor::Pos { line: index, byte: cell.byte };
127                    a <= here && here < b
128                });
129                let mut style = spans.iter().find(|s| s.start <= cell.byte && cell.byte < s.end)
130                    .map_or(theme.base(), |s| syntax_style(theme, s.kind));
131                if selected {
132                    style = style.add_modifier(Modifier::REVERSED);
133                }
134                if start < editor.left || col > editor.left + width {
135                    // A wide cell cut by an edge shows as blanks, never half a glyph.
136                    let visible = col.min(editor.left + width) - start.max(editor.left);
137                    out.set_stringn(x, y, " ".repeat(visible), visible, style);
138                } else {
139                    out.set_stringn(x, y, &cell.text, cell.width, style);
140                }
141            }
142        }
143    }
144
145    let name = editor.path().unwrap_or("[unnamed]");
146    let status = format!(
147        " {name}{} — Ln {}, Col {} ",
148        if editor.buf.is_dirty() { " *" } else { "" },
149        cursor.line + 1,
150        cursor_col + 1,
151    );
152    let status_y = area.y + rows as u16;
153    out.set_style(Rect::new(area.x, status_y, area.width, 1), theme.status_line_style());
154    out.set_stringn(area.x, status_y, &status, width, theme.status_line_style());
155
156    let bottom_y = status_y + 1;
157    let (bottom, prompt_cursor) = match &editor.mode {
158        Mode::Prompt(kind, input) => {
159            let label = match kind {
160                PromptKind::SaveAs => "Save as: ",
161                PromptKind::Find => "Find: ",
162                PromptKind::GoTo => "Go to line: ",
163            };
164            let text = format!("{label}{input}");
165            let col = text.chars().map(cell_width).sum::<usize>();
166            (text, Some(col))
167        }
168        Mode::DirtyExit => (String::from("Unsaved changes: (s)ave, (d)iscard, (c)ancel"), None),
169        Mode::Overwrite { path, .. } => (format!("{path} exists: overwrite? (y/n)"), None),
170        Mode::Edit => (
171            editor.message.clone().unwrap_or_else(|| String::from("^S save  ^O save as  ^Q quit  ^H help")),
172            None,
173        ),
174    };
175    if area.height >= 2 {
176        out.set_stringn(area.x, bottom_y, &bottom, width, theme.base());
177    }
178    match prompt_cursor {
179        Some(col) if area.height >= 2 => Some(Position::new(area.x + col.min(width.saturating_sub(1)) as u16, bottom_y)),
180        _ if editor.help || !matches!(editor.mode, Mode::Edit) => None,
181        _ => Some(Position::new(
182            area.x + (cursor_col - editor.left) as u16,
183            area.y + (cursor.line - editor.top) as u16,
184        )),
185    }
186}
