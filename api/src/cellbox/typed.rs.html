1//! Typed (record) pipeline admission, bounded execution and display.
2//!
3//! This module owns no grammar: the parser still produces `Cmd` stages, and the
4//! stage vocabulary (`select`, `where`, `sort-by`, `take`, `to`, `from`) is parsed
5//! by `text_engine::pipeline::parse_stage`. What lives here is
6//!
7//! * **admission** — [`admit`] classifies every expanded stage and either accepts
8//!   the pipeline as legacy byte composition, as a [`TypedPlan`], or refuses it
9//!   with a one-line reason. It is pure: no syscall, no VFS access, no redirect
10//!   open. The executor calls it before ANY stage runs or file is touched;
11//! * **execution** — [`run_rows`] drives a pull chain of `RowSource`s, moving each
12//!   row once (no intermediate `String`, no global stdin/sink), and closes the
13//!   whole chain when the consumer is done (`take`, error, end of input). A sink
14//!   that needs a whole bounded buffer (a redirect, a byte consumer, an enclosing
15//!   capture) collects with [`run_rows`]; the live terminal streams with
16//!   [`stream_rows`] (one row buffered, Ctrl-C polled between rows);
17//! * **display** — the terminal table renderer. Rows that are not explicitly
18//!   serialized reach a terminal as a table and a redirect/non-terminal sink as
19//!   typed JSONL; the choice never changes the data type or authority.
20//!
21//! * **external stages** — while a terminal session owns the shell, spawned cells
22//!   may join a pipeline through shell-owned byte streams: `rows → to → external`,
23//!   `external → from → rows`, `external | external`, `cat F | external` and an
24//!   external last stage. [`admit`] decides which shapes are legal; the executor
25//!   owns the processes and endpoints. [`StreamRows`] turns the bytes of an
26//!   external stage into rows incrementally so `take` can stop an endless producer.
27

115fn is_assignment(argv: &[String]) -> bool {
116    argv.len() == 1
117        && argv[0].find('=').is_some_and(|eq| {
118            eq > 0 && argv[0].as_bytes()[..eq].iter().all(|b| b.is_ascii_alphanumeric() || *b == b'_')
119        })
120}

122/// Classify an expanded command that is not a typed-vocabulary stage.
123pub fn classify_command(argv: &[String]) -> CmdClass {
124    let Some(prog) = argv.first().map(String::as_str) else {
125        return CmdClass::Builtin;
126    };
127    if is_assignment(argv) || STATEFUL.contains(&prog) {
128        return CmdClass::Stateful;
129    }
130    if prog == "top" {
131        let words: Vec<&str> = argv[1..].iter().map(String::as_str).collect();
132        return match text_engine::top::parse_options(&words) {
133            Ok(options) if !options.batch => CmdClass::Session,
134            _ => CmdClass::Builtin,
135        };
136    }
137    if prog == "edit" {
138        // Bad argv is not a session: the built-in prints usage and exits 2.
139        return if crate::edit::valid_args(&argv[1..]) { CmdClass::Session } else { CmdClass::Builtin };
140    }
141    if prog == "ssh" {
142        // Bad argv is not a session: the built-in prints usage and exits 2.
143        return if vissh::args::parse_owned(&argv[1..]).is_ok() { CmdClass::Session } else { CmdClass::Builtin };
144    }
145    if BUILTINS.contains(&prog) {
146        return CmdClass::Builtin;
147    }
148    if crate::shell_state::get_function(prog).is_some() {
149        return CmdClass::Function;
150    }
151    CmdClass::External
152}

154enum Class {
155    /// `ps|ls|top --records`: native row producer (head only).
156    Producer,
157    /// `rg PATTERN [PATH…]`: the search row producer (head only); argv parsed.
158    Search(RgArgs),
159    /// `rg -a PATTERN -`: search rows from the bytes of the head before it.
160    SearchStdin(RgArgs),
161    Row(Stage),
162    To(Stage),
163    From(Stage),
164    /// `cat PATH`: exact-byte producer (head only).
165    Cat(String),
166    Other(CmdClass),
167    /// A typed stage whose arguments `parse_stage` refused.
168    Bad(String),
169}

171fn cat_operand(argv: &[String]) -> Option<&str> {
172    match argv {
173        [_, path] if !path.starts_with('-') => Some(path.as_str()),
174        _ => None,
175    }

178fn classify_stage(argv: &[String]) -> Class {
179    let name = argv[0].as_str();
180    match name {
181        "select" | "where" | "sort-by" | "take" | "to" | "from" => match pipeline::parse_stage(argv) {
182            Ok(stage) => match name {
183                "to" => Class::To(stage),
184                "from" => Class::From(stage),
185                _ => Class::Row(stage),
186            },
187            Err(error) => Class::Bad(format!("{}: {}", name, error)),
188        },
189        "ps" | "ls" | "top" if argv[1..].iter().any(|arg| arg == "--records") => Class::Producer,
190        // `rg` is always typed; its whole argv is checked here, before any effect.
191        "rg" => match crate::rg::args::parse(argv) {
192            Ok(args) if args.stdin => Class::SearchStdin(args),
193            Ok(args) => Class::Search(args),
194            Err(usage) => Class::Bad(format!("rg: {}", usage.0)),
195        },
196        "cat" if cat_operand(argv).is_some() => {
197            Class::Cat(String::from(cat_operand(argv).unwrap_or("")))
198        }
199        _ => Class::Other(classify_command(argv)),
200    }
201}

203/// True when the first word of an UNEXPANDED simple command can only be, or may
204/// become, a typed stage or an interactive session. Plain commands skip admission.
205/// The explicit form `cellbox CMD` / `/bin/cellbox CMD` (S10-0.6) is looked through, so
206/// `cellbox ps --records | select tid` admits exactly as `ps --records | select tid`.
207pub fn might_be_typed(first: &str, args: &[&str]) -> bool {
208    if crate::oneshot::is_prefix(first) {
209        return match args.split_first() {
210            Some((command, rest)) => might_be_typed(command, rest),
211            None => false,
212        };
213    }
214    matches!(first, "select" | "where" | "sort-by" | "take" | "to" | "from" | "top" | "rg")
215        || first.contains('$')
216        || (matches!(first, "ps" | "ls") && args.contains(&"--records"))
217}

221/// Where the bytes or rows of a typed pipeline come from.
222#[derive(Debug, Clone, PartialEq, Eq)]
223pub enum Head {
224    /// Native row producer, constructed from `TypedPlan::producer_argv`.
225    Producer,
226    /// `cat PATH`: exact file bytes, converted by the following `from`.
227    Cat(String),
228    /// Any other built-in; its printed bytes (captured) feed the following `from`.
229    Capture,
230    /// `from … < PATH`: exact file bytes feed the leading `from`.
231    FromFile(String),
232    /// A one-shot's stdin (`/bin/cellbox rg … -`, S10-0.5): the Stdin grant's bytes feed
233    /// the leading `rg -`. Never produced by [`admit`].
234    Stdin,
235    /// The first `count` stages are external commands joined by kernel streams;
236    /// the last one's stdout feeds the `from` stage that follows.
237    External { count: usize },
238}

240#[derive(Debug)]
241pub struct TypedPlan {
242    pub head: Head,
243    /// Producer argv; for `ps|ls|top` the display selector `--text` is removed
244    /// (`rg` parses its own argv, where `--text` may be a pattern or a path).
245    pub producer_argv: Vec<String>,
246    /// `--text` was given: nonterminal/redirect rows use the plain text table (the
247    /// search text renderer for `rg`) instead of typed JSONL.
248    pub text_display: bool,
249    /// What the head yields into the typed segment: Rows, or Frames (`top --records`).
250    pub category: Category,
251    /// Typed segment. For a byte head the first element is the `from` stage, except
252    /// when `rg -` consumes the bytes (`search` with `stdin`).
253    pub stages: Vec<Stage>,
254    /// Number of leading `cmds` covered by the head + typed segment; the rest are
255    /// byte-consuming built-ins fed by a terminal `to` (or, after `rg`, by its
256    /// default sink).
257    pub typed_cmds: usize,
258    /// `(path, append)` of the final stage's stdout redirect when the typed
259    /// segment is the end of the pipeline.
260    pub output: Option<(String, bool)>,
261    /// Number of external stages that close the pipeline (they follow a terminal
262    /// `to`, or `rg`'s rows, and start at `typed_cmds`).
263    pub tail_externals: usize,
264    /// `(path, append)` of the final stage's stdout redirect when that stage is
265    /// one of the `tail_externals`.
266    pub tail_output: Option<(String, bool)>,
267    /// Parsed argv of the `rg` stage that produces the rows (the head, or `rg -`
268    /// after a byte head).
269    pub search: Option<RgArgs>,
270    /// Pipeline index of that `rg` stage.
271    pub search_index: usize,
272    /// Rows reach their default sink as `rg` produced them (only `where`, `take`
273    /// and `sort-by` follow it, no explicit `to`): the search text renderer
274    /// replaces the table (terminal, `--text`) (D76-9).
275    pub search_text: bool,
276    /// The pipeline status is `rg`'s completion status: nothing but an optional
277    /// explicit `to` follows it (D76-8).
278    pub search_status: bool,
279}

324/// Whole-pipeline static admission. `stages` is the complete, expanded pipeline
325/// (one element for a simple command). Nothing is executed or opened.
326///
327/// `externals` is true only while a terminal session owns the shell: spawned
328/// cells can then join a pipeline through shell-owned byte streams. Without a
329/// session every external stage of a typed pipeline is refused and plain byte
330/// pipelines keep their legacy composition.
331pub fn admit(stages: &[StageIn], externals: bool) -> Admission {
332    match admit_inner(stages, externals, false) {
333        Ok(admission) => admission,
334        Err(message) => Admission::Reject(message),
335    }
336}

357/// Child-launch bound on the argument list (the immutable child context carries
358/// 7 bytes of framing plus every argument and its terminator).
359const EXTERNAL_ARGV_LIMIT: usize = 512;
360
361/// True when `args` (the arguments after the program name) fit one child launch.
362pub fn argv_fits(args: &[String]) -> bool {
363    args.iter()
364        .try_fold(7usize, |n, arg| n.checked_add(arg.len() + 1))
365        .is_some_and(|n| n <= EXTERNAL_ARGV_LIMIT)
366}

368/// Launch-limit refusal for one external stage, raised before any stage starts.
369fn check_external(stage: &StageIn) -> Result<(), String> {
370    let name = stage.argv[0].as_str();
371    if stage.argv.iter().any(|arg| arg.contains('\0')) {
372        return reject(format!("`{}`: arguments cannot contain NUL", name));
373    }
374    if !argv_fits(&stage.argv[1..]) {
375        return reject(format!(
376            "`{}`: argument list exceeds the {}-byte child launch limit",
377            name, EXTERNAL_ARGV_LIMIT
378        ));
379    }
380    Ok(())
381}

383fn admit_inner(stages: &[StageIn], externals: bool, stdin_head: bool) -> Result<Admission, String> {
384    let n = stages.len();
385    if n > MAX_STAGES {
386        return reject(format!("pipeline has {} stages; at most {} are allowed", n, MAX_STAGES));
387    }
388    let classes: Vec<Class> = stages.iter().map(|s| classify_stage(&s.argv)).collect();
389    let typed_any = classes.iter().any(|c| {
390        matches!(
391            c,
392            Class::Producer
393                | Class::Search(_)
394                | Class::SearchStdin(_)
395                | Class::Row(_)
396                | Class::To(_)
397                | Class::From(_)
398                | Class::Bad(_)
399        )
400    });
401
402    if !typed_any {
403        if n == 1 {
404            if matches!(classes[0], Class::Other(CmdClass::Session)) && !stages[0].redirects.is_empty() {
405                return reject(session_message(&stages[0].argv[0]));
406            }
407            // An external whose stdout is redirected runs as a one-stage byte pipeline:
408            // its stdout is a stream into the file, never the terminal session.
409            if externals && is_external(&classes[0]) && has_output_redirect(&stages[0]) {
410                return admit_bytes(stages, &classes);
411            }
412            return Ok(Admission::Legacy);
413        }
414        for (stage, class) in stages.iter().zip(&classes) {
415            match class {
416                Class::Other(CmdClass::Stateful) => {
417                    return reject(stateful_message(&stage.argv[0], "inside a pipeline"))
418                }
419                Class::Other(CmdClass::Function) => {
420                    return reject(stateful_message(&stage.argv[0], "inside a pipeline (functions run in the shell context)"))
421                }
422                Class::Other(CmdClass::Session) => return reject(session_message(&stage.argv[0])),
423                _ => {}
424            }
425        }
426        if externals && classes.iter().any(is_external) {
427            return admit_bytes(stages, &classes);
428        }
429        return Ok(Admission::Legacy);
430    }
431
432    // Typed pipeline: every stage must be a typed stage, a native byte stage, a
433    // pure built-in byte consumer or (terminal session only) an external command.
434    for (stage, class) in stages.iter().zip(&classes) {
435        let name = stage.argv[0].as_str();
436        match class {
437            Class::Bad(message) => return reject(message.clone()),
438            Class::Other(CmdClass::Stateful) => {
439                return reject(stateful_message(name, "inside a pipeline"))
440            }
441            Class::Other(CmdClass::Function) => {
442                return reject(stateful_message(name, "inside a pipeline (functions run in the shell context)"))
443            }
444            Class::Other(CmdClass::Session) => return reject(session_message(name)),
445            Class::Other(CmdClass::External) => {
446                if !externals {
447                    return reject(format!(
448                        "external command `{}` cannot join a typed pipeline without a terminal session",
449                        name
450                    ));
451                }
452                check_external(stage)?;
453            }
454            _ => {}
455        }
456    }
457
458    // Head. Leading external commands feed the `from` (or `rg -`) stage that must
459    // follow them.
460    let external_head = classes.iter().take_while(|class| is_external(class)).count();
461    let head = if external_head > 0 {
462        if !matches!(classes.get(external_head), Some(Class::From(_) | Class::SearchStdin(_))) {
463            return reject(format!(
464                "`{}` output is bytes: `from json|jsonl` must directly follow it",
465                stages[external_head - 1].argv[0]
466            ));
467        }
468        Head::External { count: external_head }
469    } else {
470        match &classes[0] {
471            Class::Producer | Class::Search(_) => Head::Producer,
472            Class::Cat(path) => Head::Cat(path.clone()),
473            Class::Other(CmdClass::Builtin) => Head::Capture,
474            Class::From(_) | Class::SearchStdin(_) => {
475                match stages[0].redirects.iter().find(|r| r.kind == RedirKind::In).map(|r| r.path.clone()) {
476                    Some(path) => Head::FromFile(path),
477                    None if stdin_head && matches!(classes[0], Class::SearchStdin(_)) => Head::Stdin,
478                    None => {
479                        return reject(format!(
480                            "`{}` needs byte input: pipe bytes into it or redirect `< FILE`",
481                            stages[0].argv[0]
482                        ))
483                    }
484                }
485            }
486            Class::Row(_) | Class::To(_) => {
487                return reject(format!("`{}` needs record input; start the pipeline with a record producer or `from`", stages[0].argv[0]))
488            }
489            Class::Bad(_) | Class::Other(_) => unreachable_class()?,
490        }
491    };
492
493    // Typed run.
494    let mut typed: Vec<Stage> = Vec::new();
495    let mut end = match head {
496        Head::FromFile(_) | Head::Stdin => 0,
497        Head::External { count } => count,
498        _ => 1,
499    };
500    // The search stage: the head itself, or `rg -` directly consuming the byte head.
501    let mut search: Option<(RgArgs, usize)> = match &classes[0] {
502        Class::Search(args) => Some((args.clone(), 0)),
503        _ => None,
504    };
505    if !matches!(head, Head::Producer) {
506        if let Some(Class::SearchStdin(args)) = classes.get(end) {
507            search = Some((args.clone(), end));
508            end += 1;
509        }
510    }
511    let mut ended_with_to = false;
512    while end < n {
513        match &classes[end] {
514            Class::Row(stage) | Class::From(stage) => typed.push(stage.clone()),
515            Class::To(stage) => {
516                typed.push(stage.clone());
517                end += 1;
518                ended_with_to = true;
519                break;
520            }
521            _ => break,
522        }
523        end += 1;
524    }
525    // `rg` rows may feed byte consumers and external commands through its default
526    // sink (typed JSONL, or the search text with `--text`); other producers need `to`.
527    let feeds_bytes = ended_with_to || search.is_some();
528
529    if matches!(head, Head::Cat(_) | Head::Capture) && typed.is_empty() && search.is_none() {
530        return reject(format!(
531            "`{}` output is bytes: `from json|jsonl` must directly follow it before any typed stage",
532            stages[0].argv[0]
533        ));
534    }
535    // Anything after the typed run must be a byte consumer fed by an explicit `to`:
536    // either built-ins over captured bytes or external commands, never both.
537    let tail_externals = classes[end..].iter().filter(|class| is_external(class)).count();
538    for (i, class) in classes.iter().enumerate().skip(end) {
539        match class {
540            Class::Other(CmdClass::External) => {
541                if matches!(head, Head::External { .. }) {
542                    return reject(format!(
543                        "`{}` cannot follow typed stages after an external head: an external command may start or end a typed pipeline, not both",
544                        stages[i].argv[0]
545                    ));
546                }
547                if !feeds_bytes {
548                    return reject(format!(
549                        "`{}` cannot consume records or bytes here: end the typed stages with `to json|jsonl|text`",
550                        stages[i].argv[0]
551                    ));
552                }
553            }
554            Class::Other(CmdClass::Builtin) | Class::Cat(_) => {
555                if !feeds_bytes {
556                    return reject(format!(
557                        "`{}` cannot consume records or bytes here: end the typed stages with `to json|jsonl|text`",
558                        stages[i].argv[0]
559                    ));
560                }
561                if tail_externals > 0 {
562                    return reject(format!(
563                        "built-in `{}` cannot be mixed with external commands after `to`",
564                        stages[i].argv[0]
565                    ));
566                }
567            }
568            Class::Producer | Class::Search(_) => {
569                return reject(format!("`{}` is a record producer and must start the pipeline", stages[i].argv[0]))
570            }
571            Class::SearchStdin(_) => {
572                return reject(format!(
573                    "`{}` searches bytes: it must directly follow `cat`, a built-in, external commands or take `< FILE`",
574                    stages[i].argv[0]
575                ))
576            }
577            _ => {
578                return reject(format!(
579                    "typed stage `{}` cannot follow a byte stage; `to` is terminal for typed stages",
580                    stages[i].argv[0]
581                ))
582            }
583        }
584    }
585    let category = match head {
586        Head::Producer => crate::native_rows::producer_category(&stages[0].argv),
587        _ => Category::Rows,
588    };
589    let input = match (&head, category) {
590        _ if search.is_some() => Input::Rows,
591        (Head::Producer, Category::Frames) => Input::Frames,
592        (Head::Producer, Category::Rows) => Input::Rows,
593        _ => Input::Bytes,
594    };
595    pipeline::validate_stage_sequence(input, &typed).map_err(|error| error.to_string())?;
596
597    // Redirects.
598    let mut output: Option<(String, bool)> = None;
599    let mut tail_output: Option<(String, bool)> = None;
600    for (i, stage) in stages.iter().enumerate() {
601        for redirect in &stage.redirects {
602            match redirect.kind {
603                RedirKind::Err => {
604                    return reject(
605                        "`2>` is not supported on typed pipelines: diagnostics never enter data streams",
606                    )
607                }
608                RedirKind::In => {
609                    let allowed = i == 0 && matches!(head, Head::FromFile(_) | Head::Capture);
610                    if !allowed {
611                        return reject(format!("`<` is not valid on `{}` in a typed pipeline", stage.argv[0]));
612                    }
613                }
614                RedirKind::Out | RedirKind::Append => {
615                    if i + 1 != n {
616                        return reject(format!("output redirect on `{}` is only allowed on the final stage", stage.argv[0]));
617                    }
618                    let target = (redirect.path.clone(), redirect.kind == RedirKind::Append);
619                    if end == n {
620                        if output.is_some() {
621                            return reject("more than one output redirect on the final stage");
622                        }
623                        output = Some(target);
624                    } else if tail_externals > 0 {
625                        if tail_output.is_some() {
626                            return reject("more than one output redirect on the final stage");
627                        }
628                        tail_output = Some(target);
629                    }
630                }
631            }
632        }
633    }
634
635    let mut producer_argv = stages[0].argv.clone();
636    let mut text_display = false;
637    if let Some((args, _)) = &search {
638        text_display = args.text_display;
639    } else if matches!(head, Head::Producer) {
640        let before = producer_argv.len();
641        producer_argv.retain(|arg| arg != "--text");
642        text_display = producer_argv.len() != before;
643    }
644    let search_text = search.is_some()
645        && typed.iter().all(|stage| matches!(stage, Stage::Where { .. } | Stage::Take { .. } | Stage::SortBy { .. }));
646    let search_status = search.is_some() && end == n && typed.iter().all(|stage| matches!(stage, Stage::To(_)));
647    let (search, search_index) = match search {
648        Some((args, index)) => (Some(args), index),
649        None => (None, 0),
650    };
651
652    Ok(Admission::Typed(TypedPlan {
653        head,
654        producer_argv,
655        text_display,
656        category,
657        stages: typed,
658        typed_cmds: end,
659        output,
660        tail_externals,
661        tail_output,
662        search,
663        search_index,
664        search_text,
665        search_status,
666    }))
667}

669/// Byte pipeline (no typed stage) with external commands: optional built-in
670/// producers, then only external commands. `classes` holds no stateful, function
671/// or session stage (the caller refused those).
672fn admit_bytes(stages: &[StageIn], classes: &[Class]) -> Result<Admission, String> {
673    let n = stages.len();
674    let prefix = classes.iter().position(is_external).unwrap_or(n);
675    for i in prefix..n {
676        if !is_external(&classes[i]) {
677            return reject(format!(
678                "`{}` cannot consume the output of an external command; only external commands may follow one",
679                stages[i].argv[0]
680            ));
681        }
682        check_external(&stages[i])?;
683    }
684    let head = match (prefix, classes.first()) {
685        (0, _) => ByteHead::None,
686        (1, Some(Class::Cat(path))) => ByteHead::Cat(path.clone()),
687        _ => ByteHead::Capture,
688    };
689
690    let mut output: Option<(String, bool)> = None;
691    for (i, stage) in stages.iter().enumerate() {
692        for redirect in &stage.redirects {
693            match redirect.kind {
694                RedirKind::Err => {
695                    return reject(
696                        "`2>` is not supported on pipelines with external commands: diagnostics never enter data streams",
697                    )
698                }
699                RedirKind::In => {
700                    if !(i == 0 && head == ByteHead::Capture) {
701                        return reject(format!(
702                            "`<` is not valid on `{}` in a pipeline with external commands",
703                            stage.argv[0]
704                        ));
705                    }
706                }
707                RedirKind::Out | RedirKind::Append => {
708                    if i + 1 != n {
709                        return reject(format!("output redirect on `{}` is only allowed on the final stage", stage.argv[0]));
710                    }
711                    if output.is_some() {
712                        return reject("more than one output redirect on the final stage");
713                    }
714                    output = Some((redirect.path.clone(), redirect.kind == RedirKind::Append));
715                }
716            }
717        }
718    }
719    Ok(Admission::Bytes(BytePlan { head, prefix, output }))
720}

863/// The item chain for `stages` over `source`: the producer wrapped once with the
864/// order guard for its `category`, every row stage applied and tapped (so the first
865/// error names its stage), plus the explicit trailing `to` format.
866/// Returns `(chain, explicit_format, failure_slot, sink_stage)`; `sink_stage` is the
867/// pipeline index of the serializer (or of the last row stage when there is none).
868fn build_chain(
869    source: Box<dyn RowSource>,
870    category: Category,
871    first_stage: usize,
872    stages: &[Stage],
873) -> Result<(Box<dyn RowSource>, Option<Format>, Slot, usize), RunFailure> {
874    let slot: Slot = Rc::new(RefCell::new(None));
875    let guarded = pipeline::check_items(source, category);
876    let mut chain: Box<dyn RowSource> = Box::new(Tap { inner: guarded, stage: first_stage, slot: slot.clone() });
877    let (row_stages, explicit) = match stages.split_last() {
878        Some((Stage::To(format), rest)) => (rest, Some(format.clone())),
879        _ => (stages, None),
880    };
881    for (k, stage) in row_stages.iter().enumerate() {
882        let index = first_stage + 1 + k;
883        chain = match pipeline::apply_row_stage(chain, stage) {
884            Ok(next) => next,
885            Err(error) => return Err(RunFailure::new(index, error.to_string())),
886        };
887        chain = Box::new(Tap { inner: chain, stage: index, slot: slot.clone() });
888    }
889    let sink_stage = first_stage + row_stages.len() + usize::from(explicit.is_some());
890    Ok((chain, explicit, slot, sink_stage))
891}

942/// Pull `source` (a producer of `category` items) through the row stages in `stages`
943/// (an optional trailing `to` selects an explicit serializer) and collect the sink bytes.
944///
945/// `first_stage` is the pipeline index of `source`; row stage `k` is `first_stage
946/// + 1 + k`. The whole chain is closed on every exit path, so a satisfied `take`,
947/// an error or end of input always releases the producer. On failure NO bytes are
948/// returned and the caller opens no redirect.
949pub fn run_rows(
950    source: Box<dyn RowSource>,
951    category: Category,
952    first_stage: usize,
953    stages: &[Stage],
954    default_sink: DefaultSink,
955) -> Result<Vec<u8>, RunFailure> {
956    run_rows_within(source, category, first_stage, stages, default_sink, COLLECT_BOUND, &mut || false)
957}

959/// [`run_rows`] with an explicit collect bound and an interrupt poll, made before every
960/// pull. On interrupt the chain is closed, no bytes are returned, and the failure carries
961/// [`INTERRUPTED_STATUS`].
962pub fn run_rows_within(
963    source: Box<dyn RowSource>,
964    category: Category,
965    first_stage: usize,
966    stages: &[Stage],
967    default_sink: DefaultSink,
968    bound: CollectBound,
969    interrupted: &mut dyn FnMut() -> bool,
970) -> Result<Vec<u8>, RunFailure> {
971    let (mut chain, explicit, slot, sink_stage) = build_chain(source, category, first_stage, stages)?;
972    let mut gate = Interruptible { chain: &mut *chain, interrupted, hit: false };
973    let result = match &explicit {
974        Some(format) => encode_all(&mut gate, category, format, bound.bytes),
975        None => match default_sink {
976            DefaultSink::Table => match render_table(&mut gate, category) {
977                Ok(bytes) if bytes.len() > bound.bytes => Err(Collect::Overflow),
978                Ok(bytes) => Ok(bytes),
979                Err(message) => Err(Collect::Failed(message)),
980            },
981            DefaultSink::TypedJsonl => encode_all(&mut gate, category, &Format::JsonlTyped, bound.bytes),
982            DefaultSink::SearchText => render_search_text(&mut gate, bound.bytes),
983        },
984    };
985    let result = if gate.hit { Err(Collect::Interrupted) } else { result };
986    chain.close();
987    result.map_err(|stop| {
988        slot.borrow_mut().take().unwrap_or(match stop {
989            Collect::Failed(message) => RunFailure::new(sink_stage, message),
990            Collect::Overflow => RunFailure {
991                stage: sink_stage,
992                message: format!("output exceeds the {} byte bound", bound.bytes),
993                status: bound.status,
994            },
995            Collect::Interrupted => RunFailure {
996                stage: sink_stage,
997                message: String::from("interrupted"),
998                status: INTERRUPTED_STATUS,
999            },
1000        })
1001    })
1002}

1004fn encode_all(chain: &mut dyn RowSource, category: Category, format: &Format, bound: usize) -> Result<Vec<u8>, Collect> {
1005    let failed = |error: PipelineError| Collect::Failed(error.to_string());
1006    let mut out = Vec::new();
1007    let mut first = true;
1008    while let Some(item) = chain.next_item().map_err(failed)? {
1009        pipeline::encode_item(format, category, item, first, &mut out).map_err(failed)?;
1010        first = false;
1011        if out.len() > bound {
1012            return Err(Collect::Overflow);
1013        }
1014    }
1015    pipeline::finish_encoding(format, !first, &mut out);
1016    Ok(out)
1017}

1076/// Length of the prefix of `bytes` that is final under lossy UTF-8 conversion: a
1077/// trailing incomplete (or invalid) sequence waits for the next bytes, so the
1078/// delivered text equals `String::from_utf8_lossy` of the whole output.
1079fn complete_prefix(bytes: &[u8]) -> usize {
1080    match bytes.utf8_chunks().last() {
1081        Some(chunk) => bytes.len() - chunk.invalid().len(),
1082        None => 0,
1083    }
1084}

1086/// Encoded bytes on their way to a sink: one row's worth is held, never the stream.
1087struct Delivery<'a> {
1088    sink: &'a mut dyn ChunkSink,
1089    pending: Vec<u8>,
1090    last: Option<u8>,
1091}
1092
1093impl<'a> Delivery<'a> {
1094    fn new(sink: &'a mut dyn ChunkSink) -> Self {
1095        Self { sink, pending: Vec::new(), last: None }
1096    }
1097
1098    fn poll(&mut self) -> Result<(), Halt> {
1099        if self.sink.interrupted() {
1100            Err(Halt::Interrupted)
1101        } else {
1102            Ok(())
1103        }
1104    }
1105
1106    /// Hand the settled part of `pending` to the sink in bounded slices.
1107    fn flush(&mut self, end_of_output: bool) -> Result<(), Halt> {
1108        let split = if end_of_output { self.pending.len() } else { complete_prefix(&self.pending) };
1109        if split == 0 {
1110            return Ok(());
1111        }
1112        {
1113            let text = String::from_utf8_lossy(&self.pending[..split]);
1114            for piece in text.as_bytes().chunks(SINK_WRITE_MAX) {
1115                self.sink.write(piece).map_err(|stop| match stop {
1116                    SinkStop::Interrupted => Halt::Interrupted,
1117                    SinkStop::Failed(message) => Halt::Failed(message),
1118                })?;
1119            }
1120            self.last = text.as_bytes().last().copied();
1121        }
1122        self.pending.drain(..split);
1123        Ok(())
1124    }
1125
1126    /// Deliver the rest and end the output with a newline when it has none.
1127    fn finish(&mut self) -> Result<(), Halt> {
1128        self.flush(true)?;
1129        if self.last.is_some_and(|byte| byte != b'\n') {
1130            self.pending.push(b'\n');
1131            self.flush(true)?;
1132        }
1133        Ok(())
1134    }
1135}

1137fn stream_encoded(
1138    chain: &mut dyn RowSource,
1139    category: Category,
1140    format: &Format,
1141    out: &mut Delivery,
1142) -> Result<(), Halt> {
1143    let mut first = true;
1144    loop {
1145        out.poll()?;
1146        let Some(item) = chain.next_item().map_err(halt)? else {
1147            break;
1148        };

1150        first = false;
1151        out.flush(false)?;
1152    }
1153    pipeline::finish_encoding(format, !first, &mut out.pending);
1154    out.finish()
1155}

1181/// Like [`run_rows`] for a sink that is the terminal: each encoded item is written
1182/// to `sink` as soon as it is pulled, so memory stays bounded by one item however
1183/// long the source runs, and the sink is polled for an interrupt before every pull.
1184/// The chain is closed on every exit path (a satisfied `take`, Ctrl-C, an error).
1185///
1186/// Explicit serializers (`to jsonl|json|text|...`), the typed JSONL default and the
1187/// search text default stream item by item. The default table is the one
1188/// exception: it buffers rows to align the columns — the whole stream for Rows,
1189/// one frame for Frames — bounded by [`MAX_TABLE_ROWS`] and
1190/// [`MAX_TABLE_HEAP_BYTES`] (refusing with the usual one-line error), then streams
1191/// the rendered lines; a frame is written at its End.
1192///
1193/// Unlike [`run_rows`] a failure can arrive after bytes were delivered: the rows
1194/// before it have been shown, then the failure is reported.
1195pub fn stream_rows(
1196    source: Box<dyn RowSource>,
1197    category: Category,
1198    first_stage: usize,
1199    stages: &[Stage],
1200    default_sink: DefaultSink,
1201    sink: &mut dyn ChunkSink,
1202) -> Result<StreamEnd, RunFailure> {
1203    let (mut chain, explicit, slot, sink_stage) = build_chain(source, category, first_stage, stages)?;
1204    let mut out = Delivery::new(sink);
1205    let outcome = match (&explicit, default_sink) {
1206        (Some(format), _) => stream_encoded(&mut *chain, category, format, &mut out),
1207        (None, DefaultSink::Table) => write_tables(&mut *chain, category, &mut out).and_then(|()| out.finish()),
1208        (None, DefaultSink::TypedJsonl) => stream_encoded(&mut *chain, category, &Format::JsonlTyped, &mut out),
1209        (None, DefaultSink::SearchText) => stream_search_text(&mut *chain, &mut out),
1210    };
1211    chain.close();
1212    match outcome {
1213        Ok(()) => Ok(StreamEnd::Complete),
1214        Err(Halt::Interrupted) => Ok(StreamEnd::Interrupted),
1215        Err(Halt::Failed(message)) => {
1216            Err(slot.borrow_mut().take().unwrap_or(RunFailure::new(sink_stage, message)))
1217        }
1218    }
1219}

1221/// A typed segment serialized incrementally for an external consumer: items are
1222/// pulled only when the consumer's stdin has room, so memory stays bounded by one
1223/// fill (not by the stream) and an endless producer (`top --records`) never runs
1224/// ahead of the stage reading it.
1225pub struct RowFeed {
1226    chain: Box<dyn RowSource>,
1227    category: Category,
1228    /// The serializer; `None` is the search text renderer.
1229    format: Option<Format>,
1230    /// No row encoded yet (a `json` array needs its opening bracket first).
1231    first: bool,
1232    slot: Slot,
1233    sink_stage: usize,
1234    closed: bool,
1235}
1236
1237impl RowFeed {
1238    /// Build the chain over `source` (row stages applied, nothing pulled yet). The
1239    /// serializer is the explicit trailing `to`; without one, the search text for
1240    /// [`DefaultSink::SearchText`] and typed JSONL otherwise (a pipe is never a table).
1241    pub fn new(
1242        source: Box<dyn RowSource>,
1243        category: Category,
1244        first_stage: usize,
1245        stages: &[Stage],
1246        default_sink: DefaultSink,
1247    ) -> Result<RowFeed, RunFailure> {
1248        let (chain, explicit, slot, sink_stage) = build_chain(source, category, first_stage, stages)?;
1249        let format = match (explicit, default_sink) {
1250            (Some(format), _) => Some(format),
1251            (None, DefaultSink::SearchText) => None,
1252            (None, DefaultSink::Table | DefaultSink::TypedJsonl) => Some(Format::JsonlTyped),
1253        };
1254        Ok(RowFeed { chain, category, format, first: true, slot, sink_stage, closed: false })
1255    }
1256
1257    /// Append encoded items to `out` until it holds at least `want` bytes or the items
1258    /// end. `Ok(false)` means the end of the items was reached and the closing bytes of
1259    /// the serializer are in `out`; the chain is then already closed. An error closes
1260    /// the chain and names the stage that failed.
1261    pub fn fill(&mut self, out: &mut Vec<u8>, want: usize) -> Result<bool, RunFailure> {
1262        if self.closed {
1263            return Ok(false);
1264        }
1265        while out.len() < want {
1266            match self.chain.next_item() {
1267                Ok(Some(item)) => {
1268                    let encoded = match &self.format {
1269                        Some(format) => pipeline::encode_item(format, self.category, item, self.first, out),
1270                        None => search_line(item, out),
1271                    };
1272                    if let Err(error) = encoded {
1273                        return Err(self.fail(error.to_string()));
1274                    }
1275                    self.first = false;
1276                }
1277                Ok(None) => {
1278                    if let Some(format) = &self.format {
1279                        pipeline::finish_encoding(format, !self.first, out);
1280                    }
1281                    self.close();
1282                    return Ok(false);
1283                }
1284                Err(error) => return Err(self.fail(error.to_string())),
1285            }
1286        }
1287        Ok(true)
1288    }
1289
1290    fn fail(&mut self, message: String) -> RunFailure {
1291        self.close();
1292        self.slot.borrow_mut().take().unwrap_or(RunFailure::new(self.sink_stage, message))
1293    }
1294
1295    /// Release the producer chain (a closed `top` never samples again). Idempotent.
1296    pub fn close(&mut self) {
1297        if !self.closed {
1298            self.closed = true;
1299            self.chain.close();
1300        }
1301    }
1302}

1305    fn drop(&mut self) {
1306        self.close();
1307    }

1312/// Largest chunk one stream read can return (the endpoint transfer unit).
1313pub const STREAM_CHUNK: usize = 512;

1315/// Why an external byte stage stopped feeding its consumer.
1316#[derive(Debug, Clone, PartialEq, Eq)]
1317pub enum StreamStop {
1318    /// The user interrupted the pipeline (Ctrl-C).
1319    Interrupted,
1320    /// A stream or process failure, one diagnostic line.
1321    Failed(String),
1322}

1324/// Incremental byte producer behind an external stage.
1325pub trait ByteSource {
1326    /// Next chunk into `out` (at most `STREAM_CHUNK` bytes); `Ok(None)` is end of stream.
1327    fn read_chunk(&mut self, out: &mut [u8]) -> Result<Option<usize>, StreamStop>;
1328    /// The consumer is done with the source. `drained` is true when end of stream
1329    /// was already reached; false means the producer must be stopped.
1330    fn close(&mut self, drained: bool);
1331}
1332
1333enum StreamState {
1334    Reading,
1335    /// End of stream seen, buffered rows may remain.
1336    Ended,
1337    Closed,
1338}

1340/// Rows decoded from a byte stream. JSONL is decoded line by line as bytes
1341/// arrive, so a satisfied `take` (or an error) closes the producer without
1342/// waiting for it to finish; a JSON document is read whole (bounded) and decoded
1343/// at end of stream.
1344pub struct StreamRows<S: ByteSource> {
1345    source: S,
1346    format: Format,
1347    pending: Vec<u8>,
1348    rows: VecDeque<Value>,
1349    state: StreamState,
1350    source_closed: bool,
1351}
1352
1353impl<S: ByteSource> StreamRows<S> {
1354    pub fn new(source: S, format: Format) -> Self {
1355        Self {
1356            source,
1357            format,
1358            pending: Vec::new(),
1359            rows: VecDeque::new(),
1360            state: StreamState::Reading,
1361            source_closed: false,
1362        }
1363    }
1364
1365    fn lines(&self) -> bool {
1366        matches!(self.format, Format::Jsonl | Format::JsonlTyped)
1367    }
1368
1369    fn push(&mut self, chunk: &[u8]) -> Result<(), PipelineError> {
1370        self.pending.extend_from_slice(chunk);
1371        if self.lines() {
1372            if let Some(last) = self.pending.iter().rposition(|byte| *byte == b'\n') {
1373                let complete: Vec<u8> = self.pending.drain(..=last).collect();
1374                self.rows.extend(pipeline::decode_rows(&self.format, &complete)?);
1375            }
1376            if self.pending.len() > pipeline::MAX_RECORD_LINE_BYTES {
1377                return Err(PipelineError::RecordTooLarge);
1378            }
1379        } else if self.pending.len() > pipeline::MAX_JSON_DOCUMENT_BYTES {
1380            return Err(PipelineError::Source(format!(
1381                "input exceeds the {} byte bound",
1382                pipeline::MAX_JSON_DOCUMENT_BYTES
1383            )));
1384        }
1385        Ok(())
1386    }
1387
1388    /// End of stream: the final unterminated JSONL line, or the whole JSON document.
1389    fn end(&mut self) -> Result<(), PipelineError> {
1390        if !self.lines() || !self.pending.is_empty() {
1391            let tail = core::mem::take(&mut self.pending);
1392            self.rows.extend(pipeline::decode_rows(&self.format, &tail)?);
1393        }
1394        self.state = StreamState::Ended;
1395        Ok(())
1396    }
1397
1398    /// Stop the producer after a decode or stream error; nothing more is pulled.
1399    fn abort(&mut self) {
1400        self.rows.clear();
1401        self.pending.clear();
1402        self.close_source(false);
1403        self.state = StreamState::Closed;
1404    }
1405
1406    fn close_source(&mut self, drained: bool) {
1407        if !self.source_closed {
1408            self.source_closed = true;
1409            self.source.close(drained);
1410        }
1411    }
1412}
1413
1414impl<S: ByteSource> RowSource for StreamRows<S> {
1415    fn next_item(&mut self) -> Result<Option<Item>, PipelineError> {
1416        loop {
1417            if let Some(row) = self.rows.pop_front() {
1418                return Ok(Some(Item::Row(row)));
1419            }
1420            if !matches!(self.state, StreamState::Reading) {
1421                return Ok(None);
1422            }
1423            let mut chunk = [0u8; STREAM_CHUNK];
1424            let step = match self.source.read_chunk(&mut chunk) {
1425                Ok(Some(count)) if count > 0 && count <= STREAM_CHUNK => self.push(&chunk[..count]),
1426                Ok(Some(_)) => Err(PipelineError::Source(String::from("malformed stream read"))),
1427                Ok(None) => self.end(),
1428                Err(StreamStop::Interrupted) => Err(PipelineError::Source(String::from("interrupted"))),
1429                Err(StreamStop::Failed(message)) => Err(PipelineError::Source(message)),
1430            };
1431            if let Err(error) = step {
1432                self.abort();
1433                return Err(error);
1434            }
1435        }
1436    }
1437
1438    fn close(&mut self) {
1439        let drained = matches!(self.state, StreamState::Ended);
1440        self.rows.clear();
1441        self.pending.clear();
1442        self.close_source(drained);
1443        self.state = StreamState::Closed;
1444    }
1445}

1447/// Status the shell records for a stage it reaped. `exit` is the code `wait`
1448/// returned (`None` when the kernel refused the wait) and `reason` the kernel's
1449/// stop reason (0 when the stage ended by itself). A stage the kernel stopped
1450/// because its consumer closed (reason 3) reports 141; one stopped for any other
1451/// reason, or cancelled by the shell with a nonzero exit, reports 130 (unless it
1452/// had already exited cleanly); a refused wait on a stage nobody cancelled is 125
1453/// (status unavailable, never guessed).
1454pub fn reaped_status(exit: Option<i32>, reason: u32, cancelled: bool) -> i32 {
1455    if reason == CONSUMER_CLOSED_REASON {
1456        return UPSTREAM_NOT_STARTED_STATUS;
1457    }
1458    if reason != 0 {
1459        return 130;
1460    }
1461    match exit {
1462        Some(0) => 0,
1463        Some(code) if !cancelled => code,
1464        Some(_) => 130,
1465        None if cancelled => 130,
1466        None => 125,
1467    }
1468}

1473/// Status of an upstream stage that never ran, or was stopped, because its
1474/// consumer had already finished (128 + SIGPIPE equivalent). It is a pipe
1475/// outcome, not a failure to report as an exit.
1476pub const UPSTREAM_NOT_STARTED_STATUS: i32 = 141;

1478/// Stages start last to first. A `Cancelled` spawn of a stage that still has a
1479/// consumer in the segment (`index + 1 < stages`) is the kernel's final-reader-close
1480/// teardown: the consumer exited before this producer was committed. That is not a
1481/// spawn failure; the producer (and every stage before it) is simply not started and
1482/// the last stage decides the pipeline status. `Cancelled` for the last stage, and
1483/// every other spawn error, remains a failure.
1484pub fn spawn_cancel_means_consumer_finished(status: api::cell_session::Status, index: usize, stages: usize) -> bool {
1485    status == api::cell_session::Status::Cancelled && index + 1 < stages
1486}

1488/// Non-final stage diagnostics: one `NAME exited with status N` line per failed
1489/// stage before the last (a stage that only lost its consumer, 141, is normal pipe
1490/// behaviour and is not listed). Empty after a cancellation the shell caused itself.
1491pub fn earlier_failures(names: &[&str], statuses: &[i32], cancelled: bool) -> Vec<String> {
1492    if cancelled {
1493        return Vec::new();
1494    }
1495    let last = statuses.len().saturating_sub(1);
1496    statuses
1497        .iter()
1498        .zip(names)
1499        .take(last)
1500        .filter(|(status, _)| **status != 0 && **status != UPSTREAM_NOT_STARTED_STATUS)
1501        .map(|(status, name)| format!("{} exited with status {}", name, status))
1502        .collect()
1503}

1507fn escape_text(text: &str, out: &mut String) {
1508    for ch in text.chars() {
1509        match ch {
1510            '\n' => out.push_str("\\n"),
1511            '\r' => out.push_str("\\r"),
1512            '\t' => out.push_str("\\t"),
1513            c if (c as u32) < 0x20 || c as u32 == 0x7f => out.push_str(&format!("\\x{:02x}", c as u32)),
1514            c => out.push(c),
1515        }
1516    }
1517}

1522        Err(_) => {
1523            for &b in bytes {
1524                if b.is_ascii_graphic() || b == b' ' {
1525                    out.push(b as char);
1526                } else {
1527                    out.push_str(&format!("\\x{:02x}", b));
1528                }
1529            }
1530        }
1531    }

1534/// Human display of one value. Machine encodings never use this.
1535fn display_cell(value: &Value) -> String {
1536    let mut out = String::new();
1537    match value {
1538        Value::Null => out.push_str("null"),
1539        Value::Bool(b) => out.push_str(if *b { "true" } else { "false" }),
1540        Value::Int(n) | Value::Timestamp(n) | Value::Duration(n) => out.push_str(&n.to_string()),
1541        Value::UInt(n) => out.push_str(&n.to_string()),
1542        Value::Float(f) => out.push_str(&f.to_string()),
1543        Value::String(s) => escape_text(s, &mut out),
1544        Value::Path(p) => escape_octets(p, &mut out),
1545        Value::Bytes(b) => {
1546            out.push_str("0x");
1547            for byte in b.iter().take(16) {
1548                out.push_str(&format!("{:02x}", byte));

1550            if b.len() > 16 {
1551                out.push_str(&format!("…({} bytes)", b.len()));
1552            }
1553        }
1554        Value::List(_) | Value::Record(_) => {
1555            out = pipeline::value_to_json(value, false)
1556                .or_else(|_| pipeline::value_to_json(value, true))
1557                .unwrap_or_else(|_| String::from("<value>"));
1558        }
1559    }

1583/// Rows of a terminal table held while the column widths are unknown: this is a
1584/// genuine whole-set buffer, so it is bounded by rows ([`MAX_TABLE_ROWS`]), by cell
1585/// text ([`MAX_SINK_BYTES`]) and by its estimated heap ([`MAX_TABLE_HEAP_BYTES`]).
1586struct Table {
1587    columns: Vec<String>,
1588    rows: Vec<Vec<(usize, String)>>,
1589    widths: Vec<usize>,
1590}

1592/// Heap one buffered table row occupies, rounded the way the cell allocator does,
1593/// plus its slot in the outer vector counted three times (a growing vector keeps
1594/// the old and the new block alive while it copies).
1595fn table_row_heap(cells: &Vec<(usize, String)>) -> usize {
1596    fn block(size: usize) -> usize {
1597        size.div_ceil(16) * 16
1598    }
1599    let slots = block(cells.capacity() * core::mem::size_of::<(usize, String)>());
1600    let text: usize = cells.iter().map(|(_, cell)| block(cell.capacity())).sum();
1601    slots + text + 3 * core::mem::size_of::<Vec<(usize, String)>>()
1602}

1604/// Pull rows until the end of the stream (Rows) or of the current frame (Frames, the
1605/// Begin already consumed), polling `interrupted` before every pull. Returns the table
1606/// (`None` when there was no row: no output, not even a header) and the frame End
1607/// that stopped it (`None` at the end of the stream).
1608fn collect_table(
1609    chain: &mut dyn RowSource,
1610    mut interrupted: impl FnMut() -> bool,
1611) -> Result<(Option<Table>, Option<FrameEnd>), Halt> {
1612    let mut columns: Vec<String> = Vec::new();
1613    let mut rows: Vec<Vec<(usize, String)>> = Vec::new();
1614    let mut payload = 0usize;
1615    let mut heap = 0usize;
1616    let stop = loop {
1617        if interrupted() {
1618            return Err(Halt::Interrupted);
1619        }
1620        let row = match chain.next_item().map_err(halt)? {
1621            None => break None,
1622            Some(Item::End(end)) => break Some(end),
1623            Some(Item::Row(row)) => row,
1624            Some(Item::Begin(_)) => return Err(Halt::Failed(String::from("frame begins inside another frame"))),
1625        };
1626        if rows.len() >= MAX_TABLE_ROWS {
1627            return Err(Halt::Failed(format!(
1628                "more than {} rows for the terminal table; end with `take N` or `to jsonl`",
1629                MAX_TABLE_ROWS
1630            )));
1631        }
1632        let mut cells = Vec::with_capacity(match &row {
1633            Value::Record(fields) => fields.len(),
1634            _ => 1,
1635        });
1636        match &row {
1637            Value::Record(fields) => {
1638                for (name, value) in fields {
1639                    let cell = display_cell(value);
1640                    payload += cell.len();
1641                    cells.push((column_index(&mut columns, name), cell));
1642                }
1643            }
1644            other => {
1645                let cell = display_cell(other);
1646                payload += cell.len();
1647                cells.push((column_index(&mut columns, "value"), cell));
1648            }
1649        }
1650        if payload > MAX_SINK_BYTES {
1651            return Err(Halt::Failed(format!("table exceeds the {} byte bound", MAX_SINK_BYTES)));
1652        }
1653        heap += table_row_heap(&cells);
1654        if heap > MAX_TABLE_HEAP_BYTES {
1655            return Err(Halt::Failed(format!(
1656                "table exceeds the {} byte memory bound; end with `take N` or `to jsonl`",
1657                MAX_TABLE_HEAP_BYTES
1658            )));
1659        }
1660        rows.push(cells);
1661    };
1662    if rows.is_empty() {
1663        return Ok((None, stop));
1664    }
1665    let mut widths: Vec<usize> = columns.iter().map(|c| c.chars().count()).collect();
1666    for cells in &rows {
1667        for (col, cell) in cells {
1668            widths[*col] = widths[*col].max(cell.chars().count());
1669        }
1670    }
1671    Ok((Some(Table { columns, rows, widths }), stop))
1672}

1674impl Table {
1675    /// One padded line (trailing blanks trimmed, newline appended) into `line`.
1676    fn fill<'s>(&'s self, line: &mut String, cell: impl Fn(usize) -> Option<&'s str>) {
1677        line.clear();
1678        for (col, width) in self.widths.iter().enumerate() {
1679            let text = cell(col).unwrap_or("");
1680            line.push_str(text);
1681            for _ in text.chars().count()..*width {
1682                line.push(' ');
1683            }
1684            line.push_str("  ");
1685        }
1686        let trimmed = line.trim_end().len();
1687        line.truncate(trimmed);
1688        line.push('\n');
1689    }
1690
1691    fn header(&self, line: &mut String) {
1692        self.fill(line, |col| Some(self.columns[col].as_str()));
1693    }
1694
1695    fn row(&self, index: usize, line: &mut String) {
1696        let cells = &self.rows[index];
1697        self.fill(line, |col| cells.iter().find(|(c, _)| *c == col).map(|(_, text)| text.as_str()));
1698    }
1699}

1739/// Render the default table: one aligned table for a Rows stream; for a Frames stream
1740/// one `-- sample N --` line, that frame's table and its footer, written at the frame's
1741/// End so an endless producer is never buffered whole.
1742fn write_tables(chain: &mut dyn RowSource, category: Category, out: &mut dyn TableOut) -> Result<(), Halt> {
1743    if category == Category::Rows {
1744        let (table, _) = collect_table(chain, || out.interrupted())?;
1745        return match table {
1746            Some(table) => write_table(&table, out),
1747            None => Ok(()),
1748        };
1749    }
1750    loop {
1751        if out.interrupted() {
1752            return Err(Halt::Interrupted);
1753        }
1754        let begin = match chain.next_item().map_err(halt)? {
1755            None => return Ok(()),
1756            Some(Item::Begin(begin)) => begin,
1757            Some(_) => return Err(Halt::Failed(String::from("frame item outside a frame"))),
1758        };
1759        let (table, end) = collect_table(chain, || out.interrupted())?;
1760        let Some(end) = end else {
1761            return Err(Halt::Failed(String::from("frame stream ended inside a frame")));
1762        };
1763        out.emit(&format!("-- sample {} --\n", begin.sample_id))?;
1764        if let Some(table) = table {
1765            write_table(&table, out)?;
1766        }
1767        out.emit(&frame_footer(&end))?;
1768    }

1771/// Bounded whole-output buffer for [`render_table`].
1772struct TableBuffer(Vec<u8>);
1773
1774impl TableOut for TableBuffer {
1775    fn interrupted(&mut self) -> bool {
1776        false
1777    }
1778
1779    fn emit(&mut self, text: &str) -> Result<(), Halt> {
1780        if self.0.len() + text.len() > MAX_SINK_BYTES {
1781            return Err(Halt::Failed(format!("output exceeds the {} byte bound", MAX_SINK_BYTES)));
1782        }
1783        self.0.extend_from_slice(text.as_bytes());
1784        Ok(())
1785    }
1786}

1788/// Whole table text as bytes, for the sinks that need a bounded buffer (a redirect or
1789/// a byte consumer). The terminal streams its tables with [`stream_rows`] instead.
1790fn render_table(chain: &mut dyn RowSource, category: Category) -> Result<Vec<u8>, String> {
1791    let mut out = TableBuffer(Vec::new());
1792    match write_tables(chain, category, &mut out) {
1793        Ok(()) => Ok(out.0),
1794        Err(Halt::Failed(message)) => Err(message),
1795        Err(Halt::Interrupted) => Err(String::from("interrupted")),
1796    }
1797}

2377    /// Endless Frames producer: `rows` rows per frame. Counts items pulled and closes.
2378    fn endless_frames(rows: u64) -> (Box<dyn RowSource>, Rc<Cell<u64>>, Rc<Cell<u32>>) {
2379        struct Frames {
2380            rows: u64,
2381            sample: u64,
2382            pos: u64,
2383            pulled: Rc<Cell<u64>>,
2384            closed: Rc<Cell<u32>>,
2385        }
2386        impl RowSource for Frames {
2387            fn next_item(&mut self) -> Result<Option<Item>, PipelineError> {
2388                if self.closed.get() > 0 {
2389                    return Ok(None);
2390                }
2391                self.pulled.set(self.pulled.get() + 1);
2392                let pos = self.pos;
2393                self.pos = if pos == self.rows + 1 { 0 } else { pos + 1 };
2394                Ok(Some(match pos {
2395                    0 => {
2396                        self.sample += 1;
2397                        let meta = Value::Record(vec![(String::from("schema"), Value::String(String::from("t.v1")))]);
2398                        Item::Begin(pipeline::FrameBegin { sample_id: self.sample, monotonic_ns: self.sample * 7, meta: Some(meta) })
2399                    }
2400                    p if p <= self.rows => Item::Row(record(self.sample * 100 + p, "running")),
2401                    _ => Item::End(FrameEnd { complete: false, unavailable_fields: vec![String::from("cpu_tenths")], dropped: 2 }),
2402                }))
2403            }
2404            fn close(&mut self) {
2405                self.closed.set(self.closed.get() + 1);
2406            }
2407        }
2408        let pulled = Rc::new(Cell::new(0));
2409        let closed = Rc::new(Cell::new(0));
2410        let source = Frames { rows, sample: 0, pos: 0, pulled: pulled.clone(), closed: closed.clone() };
2411        (Box::new(source), pulled, closed)
2412    }

2538    /// Scripted chunks, then end of stream (or an endless repeat of one chunk).
2539    struct Scripted {
2540        chunks: VecDeque<Vec<u8>>,
2541        endless: Option<Vec<u8>>,
2542        reads: Rc<Cell<u32>>,
2543        closed: Rc<RefCell<Vec<bool>>>,
2544        fail_after: Option<StreamStop>,
2545    }
2546
2547    impl ByteSource for Scripted {
2548        fn read_chunk(&mut self, out: &mut [u8]) -> Result<Option<usize>, StreamStop> {
2549            self.reads.set(self.reads.get() + 1);
2550            let chunk = match self.chunks.pop_front() {
2551                Some(chunk) => chunk,
2552                None => match (&self.endless, &self.fail_after) {
2553                    (_, Some(stop)) => return Err(stop.clone()),
2554                    (Some(chunk), None) => chunk.clone(),
2555                    (None, None) => return Ok(None),
2556                },
2557            };
2558            out[..chunk.len()].copy_from_slice(&chunk);
2559            Ok(Some(chunk.len()))
2560        }
2561        fn close(&mut self, drained: bool) {
2562            self.closed.borrow_mut().push(drained);
2563        }
2564    }

2569        let source = Scripted {
2570            chunks: chunks.iter().map(|chunk| chunk.to_vec()).collect(),
2571            endless: None,
2572            reads: reads.clone(),
2573            closed: closed.clone(),
2574            fail_after: None,
