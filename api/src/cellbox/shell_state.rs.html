1//! Process-global shell state: output capture, pipe stdin, variables, functions.
2//!
3//! Everything here used to be a `static mut` or an `UnsafeCell` behind a hand-written
4//! `unsafe impl Sync`, justified by "the shell is a single task". That justification
5//! is true today and unenforceable tomorrow: nothing in the type system stops a
6//! second task from calling `shell_print`. These are `spin::Mutex` and atomics
7//! instead, which give the same `Sync` guarantee as a *checked* fact and let the
8//! crate carry `#![forbid(unsafe_code)]` (F1, Spec 16 §6).
9//!
10//! Capacity and truncation limits are byte-for-byte the ones the fixed-size static
11//! arrays imposed, so scripts that relied on them behave identically.
12
13extern crate alloc;

15use alloc::string::String;
16use alloc::vec::Vec;
17use api::cell_session::Status;
18use core::sync::atomic::{AtomicBool, Ordering};
19use ostd::io::ProcessOutput;
20use ostd::sync::Mutex;

22/// Variable slots. Setting a 17th variable is silently dropped, as before.
23const MAX_VARS: usize = 16;
24/// Shell function slots. Defining a 9th function is silently dropped, as before.
25const MAX_FUNS: usize = 8;
26const KEY_LIMIT: usize = 31;
27const VALUE_LIMIT: usize = 127;
28const BODY_LIMIT: usize = 479;
29
30/// Stack of in-flight output captures; empty means "write to the console".
31///
32/// A stack rather than a saved-and-restored slot: nested captures (a `$(...)`
33/// inside a pipeline stage) push and pop, and the innermost one wins.
34static SINK: Mutex<Vec<Vec<u8>>> = Mutex::new(Vec::new());
35/// Bytes a pipe made available on stdin for the command currently dispatching.
36static STDIN: Mutex<Vec<u8>> = Mutex::new(Vec::new());
37static VARS: Mutex<Vec<(Vec<u8>, Vec<u8>)>> = Mutex::new(Vec::new());
38static FUNS: Mutex<Vec<(Vec<u8>, Vec<u8>)>> = Mutex::new(Vec::new());
39static EXIT_REQUEST: Mutex<Option<i32>> = Mutex::new(None);
40static LOOP_SIGNAL: Mutex<LoopSignal> = Mutex::new(LoopSignal::None);
41static BG_SPAWN: AtomicBool = AtomicBool::new(false);
42
43/// Truncate to `limit` bytes, then stop at the first NUL.
44///
45/// Both steps mirror the retired fixed-array encoding, which wrote at most
46/// `limit` bytes into the slot and NUL-terminated it — a value containing a NUL
47/// was therefore already cut short on read.
48fn slot_bytes(text: &str, limit: usize) -> Vec<u8> {
49    let bytes = &text.as_bytes()[..text.len().min(limit)];
50    let end = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
51    bytes[..end].to_vec()
52}
53
54/// Decode a slot the way the old reader did: invalid UTF-8 reads back as absent.
55fn slot_str(bytes: &[u8]) -> Option<String> {
56    core::str::from_utf8(bytes).ok().map(String::from)
57}
58
59fn put(store: &mut Vec<(Vec<u8>, Vec<u8>)>, key: Vec<u8>, value: Vec<u8>, capacity: usize) {
60    if let Some(slot) = store.iter_mut().find(|(k, _)| *k == key) {
61        slot.1 = value;
62    } else if store.len() < capacity {
63        store.push((key, value));
64    }
65}

67// ── Output sink ──────────────────────────────────────────────────────────────
68
69/// Route a string to the innermost capture buffer, else the one-shot stdout, else the
70/// console.
71pub fn write_out(s: &str) {
72    write_out_bytes(s.as_bytes());
73}
74
75/// Byte-exact twin of [`write_out`]: an in-flight capture and the one-shot stdout keep
76/// every byte (NUL and non-UTF-8 included). Only the legacy console, which prints text,
77/// converts lossily.
78pub fn write_out_bytes(bytes: &[u8]) {
79    {
80        let mut sink = SINK.lock();
81        if let Some(buffer) = sink.last_mut() {
82            buffer.extend_from_slice(bytes);
83            return;
84        }
85    } // the console write must not hold the sink lock
86    if oneshot_active() {
87        // A failure is sticky and reported once by the one-shot entry (DS0-4).
88        let _ = oneshot_stdout(bytes);
89        return;
90    }
91    ostd::io::print(&String::from_utf8_lossy(bytes));
92}

94// ── One-shot output (S10-0.5) ────────────────────────────────────────────────
95
96/// The output grants of a `/bin/cellbox <command>` one-shot, installed before anything is
97/// written. While installed, uncaptured command output and diagnostics go to these
98/// streams, never to the legacy console.
99struct OneShot {
100    output: ProcessOutput,
101    /// First stdout failure; nothing more is written to stdout after it.
102    stdout_failure: Option<Status>,
103    /// Stderr failed once; later diagnostics are dropped.
104    stderr_failed: bool,
105}
106
107static ONESHOT: Mutex<Option<OneShot>> = Mutex::new(None);
108
109/// Finite per-write deadline of one-shot output: the S3-output stock-tool value.
110const ONESHOT_WRITE_NS: u64 = 30_000_000_000;
111
112/// Install the one-shot's output grants (once, before the command runs).
113#[cfg(all(not(test), not(feature = "shell_test")))]
114pub fn install_oneshot_output(output: ProcessOutput) {
115    *ONESHOT.lock() = Some(OneShot { output, stdout_failure: None, stderr_failed: false });
116}
117
118/// True while a one-shot owns this process's output.
119pub fn oneshot_active() -> bool {
120    ONESHOT.lock().is_some()
121}
122
123fn write_deadline() -> Result<u64, Status> {
124    let now = ostd::session::monotonic_ns()?;
125    match now.checked_add(ONESHOT_WRITE_NS) {
126        Some(deadline) if deadline != u64::MAX => Ok(deadline),
127        _ => Err(Status::Malformed),
128    }
129}
130
131/// Write `bytes` to the one-shot stdout with a finite deadline. After the first failure
132/// (closed, revoked, deadline) nothing more is written: a committed prefix is never
133/// replayed. Returns that failure. Not installed: `Err(WrongSession)`.
134pub fn oneshot_stdout(bytes: &[u8]) -> Result<(), Status> {
135    let mut slot = ONESHOT.lock();
136    let Some(oneshot) = slot.as_mut() else {
137        return Err(Status::WrongSession);
138    };
139    if let Some(failure) = oneshot.stdout_failure {
140        return Err(failure);
141    }
142    let written = write_deadline().and_then(|deadline| oneshot.output.stdout().write_all(bytes, deadline));
143    if let Err(failure) = written {
144        oneshot.stdout_failure = Some(failure);
145    }
146    written
147}
148
149/// Best-effort diagnostic to the one-shot stderr; dropped once stderr has failed.
150pub fn oneshot_stderr(bytes: &[u8]) {
151    let mut slot = ONESHOT.lock();
152    let Some(oneshot) = slot.as_mut() else {
153        return;
154    };
155    if oneshot.stderr_failed {
156        return;
157    }
158    let written = write_deadline().and_then(|deadline| oneshot.output.stderr().write_all(bytes, deadline));
159    oneshot.stderr_failed = written.is_err();
160}
161
162/// The first one-shot stdout failure, if any.
163#[cfg(all(not(test), not(feature = "shell_test")))]
164pub fn oneshot_stdout_failure() -> Option<Status> {
165    ONESHOT.lock().as_ref().and_then(|oneshot| oneshot.stdout_failure)
166}

168/// Take everything the innermost capture holds so far, leaving it empty and
169/// active (empty when no capture is in flight).
170pub fn take_capture() -> Vec<u8> {
171    SINK.lock().last_mut().map(core::mem::take).unwrap_or_default()
172}
173
174/// Number of in-flight captures. A serial REPL line always holds exactly one.
175pub fn capture_depth() -> usize {
176    SINK.lock().len()
177}
178
179/// RAII capture of everything `write_out` receives until `finish` or drop (Law 8).
180///
181/// Dropping without `finish` discards the captured bytes — that is the panic /
182/// early-return path, and it must still pop the stack or every later write would
183/// vanish into an orphaned buffer.
184pub struct CaptureGuard {
185    finished: bool,
186}
187
188impl CaptureGuard {
189    pub fn new() -> Self {
190        SINK.lock().push(Vec::new());
191        Self { finished: false }
192    }
193
194    /// Pop this capture and return its bytes.
195    pub fn finish(mut self) -> Vec<u8> {
196        self.finished = true;
197        SINK.lock().pop().unwrap_or_default()
198    }
199}
200
201impl Drop for CaptureGuard {
202    fn drop(&mut self) {
203        if !self.finished {
204            SINK.lock().pop();
205        }
206    }
207}

209// ── Pipe stdin ───────────────────────────────────────────────────────────────
210
211/// Publish `data` as the stdin of the command about to be dispatched.
212pub fn set_stdin(data: &[u8]) {
213    let mut stdin = STDIN.lock();
214    stdin.clear();
215    stdin.extend_from_slice(data);
216}
217
218pub fn clear_stdin() {
219    STDIN.lock().clear();
220}
221
222/// A one-shot's stdin (S10-0.5): installed only for a stdin-consuming command and run
223/// once, on the command's first demand, so a command that got a file operand never
224/// touches the Stdin grant.
225static STDIN_LOADER: Mutex<Option<fn() -> Vec<u8>>> = Mutex::new(None);
226
227/// Defer the one-shot stdin to `loader`, called on the first [`stdin_bytes`] demand.
228#[cfg(all(not(test), not(feature = "shell_test")))]
229pub fn defer_stdin(loader: fn() -> Vec<u8>) {
230    *STDIN_LOADER.lock() = Some(loader);
231}
232
233/// A copy of the current pipe stdin, empty when no pipe is active. A deferred one-shot
234/// stdin is handed out once, like a consumed stream: a second demand reads nothing.
235///
236/// Returns owned bytes rather than a borrow: a borrow out of the lock cannot
237/// outlive the guard, and the pointer-laundering that made the old `&'static [u8]`
238/// possible is exactly what F1 forbids. Pipe payloads are shell-sized.
239pub fn stdin_bytes() -> Vec<u8> {
240    let loader = STDIN_LOADER.lock().take();
241    if let Some(load) = loader {
242        return load();
243    }
244    STDIN.lock().clone()
245}

247// ── Variables ────────────────────────────────────────────────────────────────
248
249pub fn set_var(key: &str, value: &str) {
250    put(
251        &mut VARS.lock(),
252        slot_bytes(key, KEY_LIMIT),
253        slot_bytes(value, VALUE_LIMIT),
254        MAX_VARS,
255    );
256}
257
258pub fn unset_var(key: &str) {
259    let key = slot_bytes(key, KEY_LIMIT);
260    VARS.lock().retain(|(k, _)| *k != key);
261}
262
263pub fn get_var(key: &str) -> Option<String> {
264    let key = slot_bytes(key, KEY_LIMIT);
265    let vars = VARS.lock();
266    vars.iter()
267        .find(|(k, _)| *k == key)
268        .and_then(|(_, v)| slot_str(v))
269}
270
271/// Record command and whole-pipeline completion as `$?` with the exact decimal
272/// value, including statuses outside the single-digit range (124/125/126/130).
273pub fn set_status(code: i32) {
274    set_var("?", &alloc::format!("{}", code));
275}

277// ── Shell functions ──────────────────────────────────────────────────────────
278
279pub fn define_function(name: &str, body: &str) {
280    put(
281        &mut FUNS.lock(),
282        slot_bytes(name, KEY_LIMIT),
283        slot_bytes(body, BODY_LIMIT),
284        MAX_FUNS,
285    );
286}
287
288pub fn get_function(name: &str) -> Option<String> {
289    let name = slot_bytes(name, KEY_LIMIT);
290    let funs = FUNS.lock();
291    funs.iter()
292        .find(|(k, _)| *k == name)
293        .and_then(|(_, b)| slot_str(b))
294}

298#[derive(Clone, Copy, PartialEq, Eq)]
299pub enum LoopSignal {
300    None,
301    Break,
302    Continue,
303}
304
305pub fn set_loop_signal(signal: LoopSignal) {
306    *LOOP_SIGNAL.lock() = signal;
307}
308
309/// Read and clear the pending loop signal.
310pub fn take_loop_signal() -> LoopSignal {
311    let mut signal = LOOP_SIGNAL.lock();
312    core::mem::replace(&mut signal, LoopSignal::None)
313}
314
315pub fn request_exit(code: i32) {
316    *EXIT_REQUEST.lock() = Some(code);
317}
318
319/// Read and clear a pending `exit` request.
320#[cfg(not(feature = "shell_test"))] // reason: only the REPL loop can honour an exit request
321pub fn take_exit_request() -> Option<i32> {
322    EXIT_REQUEST.lock().take()
323}

325/// True while executing the command of an `Ast::Background` (`cmd &`).
326///
327/// A backgrounded EXTERNAL cell (e.g. `httpd 9092 /file &`) must NOT be
328/// `sys_wait`'d — httpd loops forever, so waiting parks the shell in `sys_wait`
329/// indefinitely and no subsequent command runs (the symptom: a second `vwrite`
330/// after `httpd &` never executed, so httpd kept serving stale content). When
331/// set, `spawn_external` returns right after spawn. Built-ins are unaffected —
332/// they run synchronously in the shell task either way.
333pub fn bg_spawn() -> bool {
334    BG_SPAWN.load(Ordering::SeqCst)
335}
336
337pub fn set_bg_spawn(value: bool) {
338    BG_SPAWN.store(value, Ordering::SeqCst);
339}
