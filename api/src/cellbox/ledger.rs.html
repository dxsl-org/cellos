2//! Command disposition ledger (Phase 04 slice S2-lite).
3//!
4//! One final disposition per shell builtin, sys-tools ELF and net-tools ELF. The ledger is
5//! plain data: it changes no behavior and grants no right. Host tests diff it against
6//! [`crate::executor::BUILTINS`], the `[[bin]]` lists of `sys-tools` / `net-tools` and the
7//! catalog table in
8//! `.agents/261020-1750-cellbox-cli/command-catalog.md`, so a command can neither be
9//! silently lost nor silently joined to the suite.
10//!
11//! A command name is identified by `(Surface, name)`: the `cat` shell builtin and the
12//! `/bin/cat` ELF are two different artifacts with different ceilings and therefore carry
13//! their own disposition. Within one surface a name appears exactly once.
14
15/// Where a command lives.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum Surface {
18    /// A shell builtin of `cellbox sh` (`executor::BUILTINS`).
19    Builtin,
20    /// A binary of the `app-sys-tools` crate (`/bin/<name>`).
21    SysTool,
22    /// A binary of the `app-net-tools` crate (`/bin/<name>`).
23    NetTool,
24}
25
26/// Final disposition of a command.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub enum Disposition {
29    /// Runs inside the cellbox ceiling; every syscall it needs is already in the shell ceiling.
30    InSuite,
31    /// A shell builtin whose privileged effect the kernel or launch edge was **observed to
32    /// deny** for this caller on the private image. Only valid with a named runner check
33    /// proving it ([`DENIAL_CHECKS`]).
34    InSuiteEffectDenied,
35    /// A shell builtin that performs **no effect at all**: it prints a refusal, issues no
36    /// syscall and no IPC, and exits non-zero (S2c-min). The effect needs a lifecycle owner
37    /// the suite shell does not hold. Only valid with a named runner check proving the
38    /// refusal and an unchanged process set ([`REFUSAL_CHECKS`]).
39    InSuiteRefused,
40    /// A separate ELF with its own launch edge and ceiling; never part of the suite ELF.
41    RetainedWorker,
42}
43
44/// One ledger row.
45#[derive(Clone, Copy, Debug)]
46pub struct Entry {
47    pub surface: Surface,
48    pub name: &'static str,
49    pub disposition: Disposition,
50}
51
52const fn e(surface: Surface, name: &'static str, disposition: Disposition) -> Entry {
53    Entry {
54        surface,
55        name,
56        disposition,
57    }
58}

146/// Every `InSuiteEffectDenied` builtin and the runner check (`chk("<name>"` in

154/// Every `InSuiteRefused` builtin and the runner check that observes the refusal (no effect,

161/// Disposition of `name` on `surface`, or `None` when the ledger has no such entry.
162pub fn disposition(surface: Surface, name: &str) -> Option<Disposition> {
163    LEDGER
164        .iter()
165        .find(|entry| entry.surface == surface && entry.name == name)
166        .map(|entry| entry.disposition)
167}

180    /// `name = "…"` of every `[[bin]]` table in a Cargo manifest.
181    fn bin_names(manifest: &str) -> Vec<&str> {
182        let mut names = Vec::new();
183        let mut in_bin = false;
184        for line in manifest.lines().map(str::trim) {
185            if line.starts_with('[') {
186                in_bin = line == "[[bin]]";
187            } else if in_bin {
188                if let Some(rest) = line.strip_prefix("name") {
189                    let value = rest.trim_start().strip_prefix('=').expect("name = ").trim();
190                    names.push(value.trim_matches('"'));
191                    in_bin = false;
192                }
193            }
194        }
195        names
196    }

304            Surface::NetTool => "net-tool",

308    fn disposition_label(d: Disposition) -> &'static str {
309        match d {
310            InSuite => "InSuite",

313            RetainedWorker => "RetainedWorker",

317    /// The catalog table, exactly as `command-catalog.md` must carry it.
318    fn catalog_block() -> String {
319        let mut out = String::from("<!-- ledger:begin -->\n| Surface | Name | Disposition |\n|---|---|---|\n");
320        for entry in LEDGER {
321            out.push_str("| ");
322            out.push_str(surface_label(entry.surface));
323            out.push_str(" | `");
324            out.push_str(entry.name);
325            out.push_str("` | ");
326            out.push_str(disposition_label(entry.disposition));
327            out.push_str(" |\n");
328        }
329        out.push_str("<!-- ledger:end -->");
330        out
331    }

333    #[test]
334    fn catalog_table_is_generated_from_the_ledger() {
335        let begin = CATALOG.find("<!-- ledger:begin -->").expect("catalog ledger begin marker");
336        let end = CATALOG.find("<!-- ledger:end -->").expect("catalog ledger end marker") + "<!-- ledger:end -->".len();
337        let expected = catalog_block();
338        assert!(
339            &CATALOG[begin..end] == expected.as_str(),
340            "catalog ledger table is stale; replace the marked block with:\n{expected}"
341        );
342    }
