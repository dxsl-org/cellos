1//! `cellbox ssh` (Phase 09, command-contracts §4).
2//!
3//! The shell builtin validates argv and re-launches this same ELF from its second
4//! install path `/bin/cellbox-ssh` with argv `["ssh", ...]` (D79-9 as revised): only
5//! that kernel-resolved path edge carries the outbound client rights, so the shell
6//! itself never holds network authority. The child entry below runs in that Cell.
7//!
8//! Host trust comes from the VFS store (S9-3). An unknown key is offered for first trust
9//! (S9-4c): VFS registers a challenge with the input owner, which itself prompts on this
10//! session and captures the answer; this client only begins the confirmation and polls
11//! VFS for the stored result, never reading the answer. Nothing beyond KEX is sent before
12//! the key is trusted.

46/// What a `TrustCommit` (or `TrustChallenge` refusal) code means for the session.
47pub fn commit_trust(code: u8) -> vissh::session::Trust {
48    use api::ipc::ssh_trust;
49    use vissh::session::Trust;
50    match code {
51        ssh_trust::KNOWN_EXACT => Trust::KnownExact,
52        ssh_trust::MISMATCH => Trust::Mismatch,
53        ssh_trust::DENIED => Trust::Denied,
54        ssh_trust::REFUSED => Trust::Refused,
55        // Still undecided when the client stopped waiting: the owner's deadline passed.
56        ssh_trust::EXPIRED | ssh_trust::PENDING => Trust::Expired,
57        _ => Trust::Unavailable,
58    }
59}

61/// Shell builtin: refuse bad argv before any launch, then run the client Cell in the
62/// foreground of this session. Every invocation leaves a completion record.
63pub fn cmd_ssh(argv: &[String]) -> i32 {
64    let mode = if argv.iter().any(|a| a == "--command") {
65        sc::Mode::Exec
66    } else {
67        sc::Mode::Pty
68    };
69    let refuse = |status: i32, code: &str| {
70        crate::executor::set_ssh_completion(admission(mode, code));
71        status
72    };
73    if let Err(e) = args::parse_owned(argv) {
74        crate::executor::shell_stderrln(&format!("ssh: {e}"));
75        crate::executor::shell_stderrln(USAGE);
76        return refuse(USAGE_STATUS, "usage");
77    }
78    if !crate::executor::interactive_terminal() {
79        crate::executor::shell_stderrln("ssh: no interactive terminal here");
80        return refuse(1, "no_terminal");
81    }
82    let mut child_argv: Vec<String> = Vec::with_capacity(argv.len() + 1);
83    child_argv.push(String::from(SSH_ENTRY));
84    child_argv.extend(argv.iter().cloned());
85    let (status, record) = crate::executor::spawn_ssh_foreground(SSH_PATH, &child_argv);
86    crate::executor::set_ssh_completion(settle(record, status, mode));
87    status
88}

105/// The shell's record of a finished ssh child: its newest frame, except that a cancelled
106/// child (killed by the kernel, so it wrote no final frame) is `cancelled` at the phase
107/// and dispatch it last reported. A child that wrote nothing never left admission.
108pub fn settle(record: Option<sc::CompletionV1>, status: i32, mode: sc::Mode) -> sc::CompletionV1 {
109    match record {
110        Some(r) if status == CANCELLED => sc::CompletionV1 {
111            outcome: sc::Outcome::Cancelled,
112            error_code: sc::Text::new("cancelled"),
113            ..r
114        },
115        Some(r) => r,
116        None => admission(
117            mode,
118            if status == CANCELLED {
119                "cancelled"
120            } else {
121                "no_record"
122            },
123        ),
124    }
125}

161/// The final record of a session (§4.3) that last reported `phase`.
162pub fn completion(
163    outcome: &Outcome,
164    mode: sc::Mode,
165    phase: sc::Phase,
166    session_lost: bool,
167) -> sc::CompletionV1 {
168    use sc::Outcome as O;
169    use vissh::session::Trust;
170    let sent = outcome.dispatch != Dispatch::NotSent;
171    let uncertain = if sent { O::Unknown } else { O::LocalError };
172    let (result, code, phase) = match &outcome.ending {
173        _ if session_lost => (O::SessionLost, "session_lost", phase),
174        Ending::RemoteExit(_) => (O::RemoteExit, "", sc::Phase::Complete),
175        Ending::RemoteSignal(_) => (O::RemoteSignal, "", sc::Phase::Complete),
176        Ending::TimedOut => (O::Timeout, "timeout", phase),
177        Ending::TransportLost => (uncertain, "transport_lost", phase),
178        Ending::Unknown => (O::Unknown, "no_exit_status", phase),
179        // A publickey refusal is local: nothing was signed for the server.
180        Ending::Credential(credential) => {
181            (O::LocalError, credential_code(*credential), sc::Phase::Auth)
182        }
183        Ending::HostKey(trust, _) => (
184            O::LocalError,
185            match trust {
186                Trust::Mismatch => "host_key_mismatch",
187                Trust::Unknown | Trust::Refused => "host_key_refused",
188                Trust::Expired => "host_key_confirmation_expired",
189                Trust::NoConfirmation => "host_key_no_frontend",
190                Trust::Denied => "trust_denied",
191                Trust::KnownExact | Trust::Unavailable => "trust_unavailable",
192            },
193            sc::Phase::Trust,
194        ),
195        Ending::Auth => (O::LocalError, "auth_failed", sc::Phase::Auth),
196        Ending::RequestRefused => (O::LocalError, "request_refused", phase),
197        Ending::Disconnected => (uncertain, "disconnected", phase),
198        Ending::Error(_) => (uncertain, "protocol", phase),
199    };
200    sc::CompletionV1 {
201        mode,
202        phase,
203        dispatch: record_dispatch(outcome.dispatch),
204        outcome: result,
205        remote_status: match outcome.ending {
206            Ending::RemoteExit(n) => Some(n),
207            _ => None,
208        },
209        remote_signal: match &outcome.ending {
210            Ending::RemoteSignal(s) => sc::Text::new(s),
211            _ => None,
212        },
213        error_code: sc::Text::new(code),
214        stdout_complete: outcome.stdout_complete,
215        stderr_complete: outcome.stderr_complete,
216    }
217}

222/// The `~.` / `~~` local escape at line start (command-contracts §4.2), over raw PTY
223/// input bytes. `feed` returns the bytes to forward and whether `~.` was typed.
224#[derive(Debug)]
225pub struct Escape {
226    line_start: bool,
227    tilde: bool,
228}
229
230impl Default for Escape {
231    fn default() -> Self {
232        Escape {
233            line_start: true,
234            tilde: false,
235        }
236    }
237}
238
239impl Escape {
240    pub fn feed(&mut self, input: &[u8], out: &mut Vec<u8>) -> bool {
241        for &b in input {
242            if self.tilde {
243                self.tilde = false;
244                match b {
245                    b'.' => return true,
246                    b'~' => out.push(b'~'),
247                    other => out.extend_from_slice(&[b'~', other]),
248                }
249            } else if self.line_start && b == b'~' {
250                self.tilde = true;
251                continue;
252            } else {
253                out.push(b);
254            }
255            self.line_start = matches!(b, b'\r' | b'\n');
256        }
257        false
258    }
259
260    /// End of input: a pending `~` is forwarded.
261    pub fn flush(&mut self, out: &mut Vec<u8>) {
262        if core::mem::take(&mut self.tilde) {
263            out.push(b'~');
264        }
265    }
266}

268/// Local status and message for a finished session (§4.3).
269pub fn report(outcome: &Outcome, host: &Host, port: u16) -> (i32, Option<String>) {
270    use vissh::session::Trust;
271    let message = match &outcome.ending {
272        Ending::RemoteExit(n) if *n <= 255 => return (*n as i32, None),
273        Ending::RemoteExit(n) => format!("remote exit status {n}"),
274        Ending::RemoteSignal(sig) => format!("remote command killed by signal {sig}"),
275        Ending::HostKey(Trust::Mismatch, key) => format!(
276            "HOST KEY CHANGED for {host}:{port}: server offered {} {}; refused",
277            key.algorithm,
278            key.fingerprint()
279        ),
280        Ending::HostKey(Trust::Unknown, key) => format!(
281            "host key for {host}:{port} is not trusted: {} {}",
282            key.algorithm,
283            key.fingerprint()
284        ),
285        Ending::HostKey(Trust::Refused, key) => format!(
286            "host key for {host}:{port} was not trusted: {} {} refused",
287            key.algorithm,
288            key.fingerprint()
289        ),
290        Ending::HostKey(Trust::Expired, _) => {
291            format!("host key confirmation for {host}:{port} timed out; not trusted")
292        }
293        Ending::HostKey(Trust::NoConfirmation, _) => String::from(NO_FRONTEND),
294        Ending::HostKey(Trust::Denied, _) => String::from("host trust lookup denied"),
295        Ending::HostKey(_, _) => String::from("host trust store unavailable"),
296        Ending::Credential(credential) => credential_message(*credential, None),
297        Ending::Auth => String::from("authentication failed"),
298        Ending::RequestRefused => String::from("server refused the session request"),
299        Ending::Disconnected => String::from("disconnected"),
300        Ending::Unknown => {
301            String::from("connection ended without a remote exit status; the command may have run")
302        }
303        Ending::TimedOut => return (TIMED_OUT, Some(String::from("timed out"))),
304        Ending::TransportLost if outcome.dispatch == Dispatch::NotSent => {
305            String::from("connection lost before the command was sent")
306        }
307        Ending::TransportLost => String::from("connection lost; the command may have run"),
308        Ending::Error(e) => e.clone(),
309    };
310    (REMOTE_FAILURE, Some(message))
311}

364/// What a failed key-service call means for the login. A wrong passphrase, a corrupt blob
365/// and a key mismatch are one code (`SshUnlockFailed`): there is no oracle to tell apart.
366pub fn credential_of(error: &KmsClientError) -> Credential {
367    match error {
368        KmsClientError::ServiceNotFound
369        | KmsClientError::Service(KmsErrorCode::SshKeyUnavailable) => Credential::Unavailable,
370        KmsClientError::Service(
371            KmsErrorCode::CallerUnattested | KmsErrorCode::PermissionDenied,
372        ) => Credential::Denied,
373        KmsClientError::Service(KmsErrorCode::SshKeyNotFound) => Credential::UnknownIdentity,
374        KmsClientError::Service(KmsErrorCode::SshUnlockFailed) => Credential::UnlockFailed,
375        KmsClientError::Service(KmsErrorCode::SshUnlockLocked) => Credential::UnlockLocked,
376        KmsClientError::Service(KmsErrorCode::InvalidSignature) => Credential::BadSignature,
377        KmsClientError::Timeout => Credential::Timeout,
378        // `SshUnlockRequired`, `InvalidRequest`, `SshOrderInvalid`, `SshHandleInvalid`, any
379        // other code, and every wire or IPC error.
380        _ => Credential::Protocol,
381    }
382}

412/// One signature: for a passphrase-wrapped identity first ask for the passphrase (once,
413/// no re-prompt), stage it and wipe it; then sign. A plain identity skips the unlock. An
414/// empty passphrase is not representable on the wire (`secret_len` is 1..=1,024), so it is
415/// `unlock_failed` locally, without a key-service call. A staged secret is consumed by the
416/// sign (success or failure), so nothing is aborted afterwards.
417pub fn kms_sign<K: KeyService>(
418    kms: &K,
419    public_key: &[u8; 32],
420    custody: KmsSshKeyCustody,
421    request: &SignRequest<'_>,
422    passphrase: impl FnOnce() -> Result<String, Credential>,
423) -> Result<Signed, Credential> {
424    if custody == KmsSshKeyCustody::PassphraseWrapped {
425        let mut secret = passphrase()?;
426        let staged = if secret.is_empty() {
427            Err(Credential::UnlockFailed)
428        } else {
429            kms.stage_secret(public_key, secret.as_bytes())
430                .map_err(|error| credential_of(&error))
431        };
432        ostd::session::zeroize_string(&mut secret);
433        staged?;
434    }
435    let signed = kms
436        .sign(public_key, &request.session_id, request.username)
437        .map_err(|error| credential_of(&error))?;
438    Ok(Signed {
439        signature: signed.signature,
440        message_sha256: signed.message_sha256,
441    })
442}

577                Err(ViError::PermissionDenied) => Err(TransportError::Io),
578                // The service refuses a read while the socket cannot receive yet (the
579                // TCP handshake). The state is read after that refusal, so a socket that
580                // finished its handshake in between reads Established: also "nothing yet".
581                // Only a socket past its receive half is closed.
582                Err(_) => match self.net.tcp_state(self.socket) {
583                    Ok(0x01..=0x03) => Ok(0),
584                    _ => Err(TransportError::Closed),
585                },
586            }
587        }

590            if self.deadline.passed() {
591                return Err(TransportError::TimedOut);
592            }
593            ostd::syscall::sys_yield();
594            Ok(())
595        }

641    /// Ask the key service about the identity before any network traffic (a status query is
642    /// not an authentication attempt, so trust-before-signature is unaffected). A wrapped
643    /// identity in a pipeline stage is refused here: its passphrase needs a terminal.

657            key_id: String::from(key_id),

676    /// Phase budgets (§4.2): trust runs on the connect deadline, then auth, request (channel
677    /// open to the server's reply) and running (none but `--timeout`); every change is
678    /// recorded write-ahead on the shell's completion stream.
679    fn advance(
680        deadline: &Deadline,
681        completion: Option<Endpoint>,
682        mode: sc::Mode,
683        current: &mut sc::Phase,
684        phase: Phase,
685        dispatch: Dispatch,
686    ) {
687        let next = record_phase(phase);
688        if next != *current {
689            let budget = match phase {
690                Phase::Trust => None,
691                Phase::Auth => Some(now_ms().saturating_add(AUTH_MS)),
692                Phase::Request => Some(now_ms().saturating_add(REQUEST_MS)),
693                Phase::Running => Some(u64::MAX),
694            };
695            if let Some(at) = budget {
696                deadline.phase.set(at);
697            }
698            *current = next;
699        }
700        write_all(
701            completion,
702            &progress_record(mode, *current, dispatch).encode(),
703        );
704    }

706    /// Blocking write of all of `data`; false when the stream failed or closed.
707    fn push(out: Option<Endpoint>, mut data: &[u8]) -> bool {
708        let Some(out) = out else { return true };
709        while !data.is_empty() {
710            match out.write(data, FOREVER) {
711                Ok(n) if n > 0 && n <= data.len() => data = &data[n..],
712                _ => return false,
713            }
714        }
715        true
716    }

767        fn ask_secret(&mut self, prompt: &str, secret: &mut String) -> Secret {
768            self.write(true, prompt.as_bytes());
769            let got = match self.session.set_mode(2, 0) {
770                Ok(mut guard) => {
771                    let got = read_secret(guard.session(), secret, self.deadline);
772                    let released = guard.release().is_ok();
773                    Ok(if released { got } else { Secret::Cancelled })
774                }
775                Err(status) => Err(status),
776            };
777            let got = match got {
778                Ok(got) => got,
779                Err(status) => {
780                    self.note(status);
781                    return Secret::Cancelled;
782                }
783            };
784            self.write(true, b"\n");
785            if got != Secret::Given {
786                secret.zeroize();
787            }
788            if got == Secret::TimedOut {
789                self.prompt_timeout = true;
790            }
791            got

844                match event.kind {

855        fn pty_input(&mut self, buf: &mut [u8]) -> Input {
856            self.enter_raw();
857            if self.raw.is_none() {
858                return if self.stdin_ended {
859                    Input::Eof
860                } else {
861                    Input::Idle
862                };
863            }
864            self.poll_resize();
865            if self.stdin_ended {
866                return Input::Eof;
867            }
868            // One escape byte may expand to two forwarded bytes.
869            let mut raw = [0u8; 512];
870            let take = raw.len().min(buf.len().saturating_sub(1));
871            if take == 0 {
872                return Input::Idle;
873            }
874            match self.session().read_input(&mut raw[..take], 0) {
875                Ok(n) if n > 0 => {
876                    let mut out = Vec::with_capacity(n + 1);
877                    let disconnect = self.escape.feed(&raw[..n], &mut out);
878                    raw.zeroize();
879                    let len = out.len();
880                    buf[..len].copy_from_slice(&out);
881                    out.zeroize();
882                    if disconnect {
883                        Input::Disconnect
884                    } else if len == 0 {
885                        Input::Idle
886                    } else {
887                        Input::Data(len)
888                    }
889                }
890                Err(api::cell_session::Status::Eof) => {
891                    self.stdin_ended = true;
892                    let mut out = Vec::new();
893                    self.escape.flush(&mut out);
894                    if out.is_empty() {
895                        Input::Eof
896                    } else {
897                        buf[..out.len()].copy_from_slice(&out);
898                        Input::Data(out.len())
899                    }
900                }
901                Err(status) => {
902                    self.note(status);
903                    Input::Idle
904                }
905                Ok(_) => Input::Idle,
906            }
907        }

909        /// Exec from the terminal: canonical typed text, Enter as LF, Ctrl-D as EOF.
910        fn exec_input(&mut self, buf: &mut [u8]) -> Input {
911            if self.stdin_ended {
912                return Input::Eof;
913            }
914            match self.session().next_event(0) {
915                Ok(event) => match &event.kind {
916                    EventKind::Text(text) if text.len() <= buf.len() => {
917                        buf[..text.len()].copy_from_slice(text.as_bytes());
918                        Input::Data(text.len())
919                    }
920                    EventKind::Key {
921                        code: ENTER,
922                        modifiers: 0,
923                        pressed: 1 | 2,
924                    } if !buf.is_empty() => {
925                        buf[0] = b'\n';
926                        Input::Data(1)
927                    }
928                    // Ctrl-D arrives as a key, not as end of input: it ends the stdin.
929                    EventKind::Key {
930                        code: 0x64 | 0x44,
931                        modifiers: CTRL,
932                        pressed: 1 | 2,
933                    }
934                    | EventKind::Eof => {
935                        self.stdin_ended = true;
936                        Input::Eof
937                    }
938                    _ => Input::Idle,
939                },
940                Err(status) => {
941                    self.note(status);
942                    Input::Idle
943                }
944            }
945        }

947        /// Challenge (VFS), begin (input owner, on this session at its current epoch), then
948        /// poll VFS for the stored outcome while the owner holds the terminal. Returns the
949        /// answer and whether this confirmation stored the key.
950        fn confirm(&mut self, key: &HostKey) -> (Trust, bool) {
951            let (session, epoch) = (self.session.reference().raw(), self.session.epoch());
952            let mut vfs = VfsClient::new();
953            let challenge = match vfs.trust_challenge(
954                &self.host,
955                self.port,
956                key.algorithm,
957                &key.blob(),
958                session,
959                epoch,
960            ) {
961                Ok(Ok((confirmation, challenge))) => {
962                    match self.session.begin_trust_confirmation(confirmation) {
963                        Ok(()) => challenge,
964                        // A Serial or headless terminal has no trusted way to answer.
965                        Err(api::cell_session::Status::Unsupported) => {
966                            return (Trust::NoConfirmation, false)
967                        }
968                        Err(_) => return (Trust::Unavailable, false),
969                    }
970                }
971                // Stored meanwhile by another client (or changed): no prompt.
972                Ok(Err(code)) => return (commit_trust(code), false),
973                Err(_) => return (Trust::Unavailable, false),
974            };
975            let give_up = now_ms().saturating_add(CONFIRM_MS);
976            loop {
977                match vfs.trust_commit(challenge) {
978                    Ok(api::ipc::ssh_trust::PENDING) if now_ms() < give_up => {
979                        let _ = ostd::syscall::sys_set_timer(CONFIRM_POLL_TICKS);
980                    }
981                    Ok(code) => {
982                        return (commit_trust(code), code == api::ipc::ssh_trust::KNOWN_EXACT)
983                    }
984                    Err(_) => return (Trust::Unavailable, false),
985                }
986            }
987        }

1192    /// One line under the echo-off lease: text, Backspace, Enter; Escape cancels. Pasted
1193    /// text is refused (§4.1: no newline/NUL, typed only). The wait is bounded by the
1194    /// remaining phase budget and `--timeout` (K6 O-K6-6), also for the password.
1195    fn read_secret(
1196        session: &mut TerminalSession,
1197        secret: &mut String,
1198        deadline: &Deadline,
1199    ) -> Secret {
1200        let mut paste = false;
1201        loop {
1202            let remaining_ms = deadline.at().saturating_sub(now_ms());
1203            if remaining_ms == 0 {
1204                return Secret::TimedOut;
1205            }
1206            // The monotonic clock bounds each wait; without it the wait is a poll.
1207            let until = ostd::session::monotonic_ns().map_or(0, |now| {
1208                now.saturating_add(remaining_ms.min(PROMPT_SLICE_MS).saturating_mul(1_000_000))
1209            });
1210            let mut event = match session.next_event(until) {
1211                Ok(event) => event,
1212                Err(
1213                    api::cell_session::Status::TimedOut | api::cell_session::Status::WouldBlock,
1214                ) => {
1215                    if until == 0 {
1216                        let _ = ostd::syscall::sys_set_timer(1);
1217                    }
1218                    continue;
1219                }
1220                Err(_) => return Secret::Cancelled,
1221            };
1222            let done = match &event.kind {
1223                EventKind::PasteBegin => {
1224                    paste = true;
1225                    None
1226                }
1227                EventKind::PasteEnd => {
1228                    paste = false;
1229                    None
1230                }
1231                EventKind::Text(text) if !paste => {
1232                    if secret_accepts(secret.len(), text) {
1233                        secret.push_str(text);
1234                        None
1235                    } else {
1236                        Some(Secret::Cancelled)
1237                    }
1238                }
1239                EventKind::Key {
1240                    code,
1241                    modifiers,
1242                    pressed: 1 | 2,
1243                } if !paste => match *code {
1244                    _ if crate::executor::submits_line(*code, *modifiers) => Some(Secret::Given),
1245                    ESCAPE if *modifiers == 0 => Some(Secret::Cancelled),
1246                    BACKSPACE if *modifiers == 0 => {
1247                        secret.pop();
1248                        None
1249                    }
1250                    _ => None,
1251                },
1252                EventKind::Interrupt | EventKind::Eof => Some(Secret::Cancelled),
1253                _ => None,
1254            };
1255            event.scrub();
1256            if let Some(done) = done {
1257                return done;
1258            }
1259        }
1260    }

1262    /// Entry of `/bin/cellbox-ssh ssh ...`: a foreground child with the terminal session, or
1263    /// (publickey exec only) a pipeline stage on the stage's own stdin, stdout and stderr.
1264    pub fn run(context: &ChildContext) -> i32 {
1265        let stderr = granted(context, GrantSlot::Stderr);
1266        let completion = terminal_sdk::completion_writer(context);
1267        let rmode = if context.argv().any(|a| a == "--command") {
1268            sc::Mode::Exec
1269        } else {
1270            sc::Mode::Pty
1271        };
1272        // A final record and the local message for an ending outside the SSH session.
1273        let end =
1274            |status: i32, phase: sc::Phase, outcome: sc::Outcome, code: &str, message: &str| {
1275                let record = sc::CompletionV1 {
1276                    mode: rmode,
1277                    phase,
1278                    dispatch: sc::Dispatch::NotSent,
1279                    outcome,
1280                    remote_status: None,
1281                    remote_signal: None,
1282                    error_code: sc::Text::new(code),
1283                    stdout_complete: false,
1284                    stderr_complete: false,
1285                };
1286                write_all(completion, &record.encode());
1287                write_all(stderr, format!("ssh: {message}\n").as_bytes());
1288                status
1289            };
1290        let refuse = |status: i32, code: &str, message: &str| {
1291            end(
1292                status,
1293                sc::Phase::Admission,
1294                sc::Outcome::LocalError,
1295                code,
1296                message,
1297            )
1298        };
1299        let parsed = match args::parse(context.argv().skip(1)) {
1300            Ok(parsed) => parsed,
1301            Err(e) => return refuse(USAGE_STATUS, "usage", &format!("{e}")),
1302        };
1303        let key_id = match &parsed.auth {
1304            args::Auth::PublicKey { key_id } => Some(key_id.clone()),
1305            args::Auth::Password => None,
1306        };
1307        // A foreground child has a control grant (slot 3); a pipeline stage never does. Only a
1308        // publickey exec may run as a stage: password and PTY sessions need the terminal.
1309        let staged = !context.grants().any(|g| g.slot == GrantSlot::Control);
1310        if staged && !(key_id.is_some() && parsed.command.is_some()) {
1311            return refuse(
1312                REMOTE_FAILURE,
1313                "no_terminal",
1314                "an interactive terminal session is required",
1315            );
1316        }
1317        let mut session = if staged {
1318            None
1319        } else {
1320            match TerminalSession::from_own_context(context) {
1321                Ok(session) => Some(session),
1322                Err(_) => {
1323                    return refuse(
1324                        REMOTE_FAILURE,
1325                        "no_terminal",
1326                        "an interactive terminal session is required",
1327                    )
1328                }
1329            }
1330        };
1331        // Publickey: the key service is asked about the identity before any network traffic,
1332        // so an unknown KEY_ID never touches the network or the server's login-grace slot.
1333        let identity = match &key_id {
1334            Some(key_id) => {
1335                if !validate_ssh_username(parsed.user.as_bytes()) {
1336                    return refuse(
1337                        USAGE_STATUS,
1338                        "usage",
1339                        "publickey authentication needs a user name of 1 to 31 characters from [A-Za-z0-9._-]",
1340                    );
1341                }
1342                match preflight(key_id, staged) {
1343                    Ok(identity) => Some(identity),
1344                    Err(credential) => {
1345                        return refuse(
1346                            REMOTE_FAILURE,
1347                            credential_code(credential),
1348                            &credential_message(credential, Some(key_id)),
1349                        );
1350                    }
1351                }
1352            }
1353            None => None,
1354        };
1355        // Lane-only (`kssh-faults`): the `: kssh-fault MODE;` command prefix selects the one
1356        // mutation of this invocation's sign call.
1357        #[cfg(feature = "kssh-faults")]
1358        let identity = identity.map(|identity| Identity {
1359            fault: super::faults::Mode::of_command(parsed.command.as_deref()),
1360            ..identity
1361        });
1362        let auth = if key_id.is_some() {
1363            AuthKind::PublicKey
1364        } else {
1365            AuthKind::Password
1366        };
1367        if !vissh::install_entropy(kernel_entropy) {
1368            return refuse(
1369                REMOTE_FAILURE,
1370                "entropy",
1371                "entropy source already installed",
1372            );
1373        }
1374        let start = now_ms();
1375        let deadline = Deadline {
1376            phase: Cell::new(start + u64::from(parsed.connect_timeout) * 1000),
1377            overall: parsed
1378                .timeout
1379                .map_or(u64::MAX, |t| start + u64::from(t) * 1000),
1380        };
1381        write_all(
1382            completion,
1383            &progress_record(rmode, sc::Phase::Connect, Dispatch::NotSent).encode(),
1384        );
1385        let policy = rekey_policy();
1386        #[cfg(feature = "kssh-faults")]
1387        let _ = ostd::syscall::sys_log(super::faults::ANNOUNCE);
1388        let connect = |code: &str, message: &str| {
1389            end(
1390                REMOTE_FAILURE,
1391                sc::Phase::Connect,
1392                sc::Outcome::LocalError,
1393                code,
1394                message,
1395            )
1396        };
1397        let timed_out = || {
1398            end(
1399                TIMED_OUT,
1400                sc::Phase::Connect,
1401                sc::Outcome::Timeout,
1402                "timeout",
1403                "timed out",
1404            )
1405        };
1406        let mut net = NetClient::new();
1407        let addr = match &parsed.host {
1408            Host::V4(a) => a.octets(),
1409            Host::V6(_) => {
1410                return connect(
1411                    "ipv6_unsupported",
1412                    "IPv6 is not supported by the network service",
1413                )
1414            }
1415            Host::Name(name) => match net.dns_lookup(name) {
1416                Ok(a) => a,
1417                Err(ViError::PermissionDenied) => {
1418                    return connect("resolve_denied", "name resolution denied")
1419                }
1420                Err(_) if deadline.passed() => return timed_out(),
1421                Err(e) => {
1422                    return connect("resolve_failed", &format!("cannot resolve {name}: {e:?}"))
1423                }
1424            },
1425        };
1426        if deadline.passed() {
1427            return timed_out();
1428        }
1429        let socket = match net.tcp_connect(addr, parsed.port) {
1430            Ok(s) => s,
1431            Err(ViError::PermissionDenied) => {
1432                return connect("connect_denied", "network connection denied")
1433            }
1434            Err(_) if deadline.passed() => return timed_out(),
1435            Err(e) => return connect("connect_failed", &format!("connect failed: {e:?}")),
1436        };
1437        if deadline.passed() {
1438            let _ = net.tcp_close_graceful(socket);
1439            return timed_out();
1440        }
1441        let mut transport = NetTransport {
1442            net,
1443            socket,
1444            deadline: &deadline,
1445        };
1446        let (mut outcome, phase, lost, prompt_timeout) = match session.as_mut() {
1447            Some(session) => {
1448                let size = session
1449                    .capabilities()
1450                    .map(|c| (c.rows, c.cols))
1451                    .ok()
1452                    .filter(|&(r, c)| r > 0 && c > 0)
1453                    .unwrap_or((24, 80));
1454                let mode = match &parsed.command {
1455                    Some(cmd) => Mode::Exec(cmd.clone()),
1456                    None => Mode::Pty {
1457                        term: "cellos-vt1",
1458                        rows: size.0,
1459                        cols: size.1,
1460                    },
1461                };
1462                let mut term = Term {
1463                    session,
1464                    raw: None,
1465                    host: format!("{}", parsed.host),
1466                    port: parsed.port,
1467                    user: parsed.user.clone(),
1468                    pty: parsed.command.is_none(),
1469                    escape: Escape::default(),
1470                    size,
1471                    pending_resize: None,
1472                    deadline: &deadline,
1473                    stdin_ended: false,
1474                    completion,
1475                    mode: rmode,
1476                    phase: sc::Phase::Connect,
1477                    session_lost: false,
1478                    identity,
1479                    prompt_timeout: false,
1480                    #[cfg(feature = "p09-faults")]
1481                    faults: faults::Faults::new(parsed.command.as_deref()),
1482                };
1483                let outcome = session::run(
1484                    &mut transport,
1485                    &mut term,
1486                    &parsed.user,
1487                    auth,
1488                    &mode,
1489                    &policy,
1490                );
1491                // Leave raw mode before any local message.
1492                if let Some(mut raw) = term.raw.take() {
1493                    let _ = raw.leave(term.session);
1494                }
1495                #[cfg(feature = "p09-faults")]
1496                term.faults.finish(&outcome);
1497                (outcome, term.phase, term.session_lost, term.prompt_timeout)
1498            }
1499            None => {
1500                let mode = Mode::Exec(parsed.command.clone().unwrap_or_default());
1501                let mut term = PipeTerm {
1502                    stdin: granted(context, GrantSlot::Stdin),
1503                    stdout: granted(context, GrantSlot::Stdout),
1504                    stderr,
1505                    host: format!("{}", parsed.host),
1506                    port: parsed.port,
1507                    deadline: &deadline,
1508                    identity,
1509                    completion,
1510                    mode: rmode,
1511                    phase: sc::Phase::Connect,
1512                    stdin_ended: false,
1513                    broken: false,
1514                };
1515                let outcome = session::run(
1516                    &mut transport,
1517                    &mut term,
1518                    &parsed.user,
1519                    auth,
1520                    &mode,
1521                    &policy,
1522                );
1523                (outcome, term.phase, false, false)
1524            }
1525        };
1526        // A secret prompt that ran into the phase budget or `--timeout` is a timeout, not a
1527        // refusal (the prompt itself ended the wait).
1528        if prompt_timeout && matches!(outcome.ending, Ending::Auth | Ending::Credential(_)) {
1529            outcome.ending = Ending::TimedOut;
1530        }
1531        // FIN, not a silent discard: the server must see the connection end (a peer left
1532        // half-open waits out its login grace time and penalises this source).
1533        let _ = transport.net.tcp_close_graceful(socket);
1534        write_all(
1535            completion,
1536            &super::completion(&outcome, rmode, phase, lost).encode(),
1537        );
1538        let (status, message) = if lost {
1539            (SESSION_LOST, Some(String::from("terminal session lost")))
1540        } else {
1541            report(&outcome, &parsed.host, parsed.port)
1542        };
1543        match message {
1544            Some(message) => {
1545                write_all(stderr, format!("ssh: {message}\n").as_bytes());
1546                status
1547            }
1548            None => status,
1549        }
1550    }

1552    /// Lane-only fault hooks and measurements (`p09-faults`; command-contracts §4.4 lane).
1553    /// A remote command starting `: p09-fault <point> <action>;` makes the client announce
1554    /// `<point>` on the kernel log and hold there (transport serviced, no packet
1555    /// processed) so the lane runner can act; `: p09-sink;` consumes stdout into a digest
1556    /// at a throttled pace instead of the terminal.
1557    #[cfg(feature = "p09-faults")]
1558    mod faults {
1559        use super::*;
1560        use sha2::{Digest, Sha256};
1561        use vissh::session::Outcome;
1562
1563        /// Lowered rekey triggers so one lane run crosses them (production: 1 h, 1 GiB).
1564        pub const LANE_REKEY: RekeyPolicy = RekeyPolicy {
1565            max_age_ms: 20_000,
1566            limits: vissh::sunset::RekeyLimits {
1567                bytes: 2 << 20,
1568                packets: 4096,
1569            },
1570            kex_ms: 30_000,
1571        };
1572        const HOLD_MS: u64 = 5_000;
1573        const THROTTLE: usize = 4096;
1574
1575        pub struct Faults {
1576            hook: Option<(String, String)>,
1577            sink: Option<(u64, Sha256, usize)>,
1578        }
1579
1580        fn point_name(point: Point) -> &'static str {
1581            match point {
1582                Point::BeforeRequest => "before-request",
1583                Point::RequestSent => "may-have-run",
1584                Point::Accepted => "accepted",
1585                Point::ExitStatus => "exit-before-drain",
1586                Point::RekeyStart => "rekey-start",
1587            }
1588        }
1589
1590        impl Faults {
1591            pub fn new(command: Option<&str>) -> Self {
1592                let command = command.unwrap_or("");
1593                let hook = command.strip_prefix(": p09-fault ").and_then(|rest| {
1594                    let (spec, _) = rest.split_once(';')?;
1595                    let mut words = spec.split_whitespace();
1596                    let (point, action) = (words.next()?, words.next()?);
1597                    words
1598                        .next()
1599                        .is_none()
1600                        .then(|| (String::from(point), String::from(action)))
1601                });
1602                let sink = command
1603                    .starts_with(": p09-sink;")
1604                    .then(|| (0, Sha256::new(), 0));
1605                Faults { hook, sink }
1606            }
1607
1608            /// The hooked point is announced and held, so the runner's action (seen up to
1609            /// half a second later) lands before the session moves past it.
1610            pub fn checkpoint(&self, point: Point) -> u64 {
1611                let name = point_name(point);
1612                let Some((hooked, action)) = &self.hook else {
1613                    return 0;
1614                };
1615                if hooked != name {
1616                    return 0;
1617                }
1618                let _ = ostd::syscall::sys_log(&format!("P04 INFO p09-hook {name} {action}\n"));
1619                HOLD_MS
1620            }
1621
1622            /// Consume stdout into the digest; false when no sink is configured.
1623            pub fn sink(&mut self, data: &[u8]) -> bool {
1624                let Some((count, hash, since)) = &mut self.sink else {
1625                    return false;
1626                };
1627                *count += data.len() as u64;
1628                hash.update(data);
1629                *since += data.len();
1630                // A slow consumer: one tick per 4 KiB, so flow control has to hold.
1631                while *since >= THROTTLE {
1632                    *since -= THROTTLE;
1633                    let _ = ostd::syscall::sys_set_timer(1);
1634                }
1635                true
1636            }
1637
1638            pub fn finish(&mut self, outcome: &Outcome) {
1639                let _ =
1640                    ostd::syscall::sys_log(&format!("P04 INFO ssh-rekeys {}\n", outcome.rekeys));
1641                if let Some((count, hash, _)) = self.sink.take() {
1642                    let digest = hash.finalize();
1643                    let mut hex = String::with_capacity(64);
1644                    for b in digest.iter() {
1645                        hex.push_str(&format!("{b:02x}"));
1646                    }
1647                    let _ = ostd::syscall::sys_log(&format!(
1648                        "P04 INFO ssh-bytes stdout={count} sha256={hex} peak_pending={} peak_chunk={}\n",
1649                        outcome.peaks.pending, outcome.peaks.chunk
1650                    ));
1651                }
1652            }
1653        }
1654    }
