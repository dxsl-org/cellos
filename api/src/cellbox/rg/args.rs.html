1//! `rg` argv (proposal §3.2, `command-contracts.md` §1 and §2.1).
2//!
3//! Words are scanned left to right until `--` or the first operand; a bare `-` is an
4//! operand. Every other `-` word before `--` must be an exact supported spelling: no
5//! combined short options, `--name=value`, abbreviations or upstream-only flags. Value
6//! options consume exactly the next word. Parsing is pure and runs at admission, so every
7//! error here is status 2 before any effect.
8

15/// Encoded argv bound `Σ(len + 1)` over every word including `rg` (contract §1, D76-13).
16pub const MAX_ARGV_BYTES: usize = 512;
17/// At most this many path operands.
18pub const MAX_PATHS: usize = 16;
19/// At most this many patterns (positional or `-e`).
20pub const MAX_PATTERNS: usize = 32;
21/// Aggregate UTF-8 pattern bytes.
22pub const MAX_PATTERN_BYTES: usize = 4096;
23/// At most this many `-g` globs.
24pub const MAX_GLOBS: usize = 32;
25/// Aggregate glob bytes.
26pub const MAX_GLOB_BYTES: usize = 4096;
27/// Largest `-A`/`-B`/`-C` value.
28pub const MAX_CONTEXT: usize = 20;
29

74/// Parsed `rg` argv.
75#[derive(Debug, Clone, Default, PartialEq, Eq)]
76pub struct RgArgs {
77    /// Patterns in argument order; never empty unless `help`.
78    pub patterns: Vec<String>,
79    /// Path operands in argument order; empty means the shell cwd (or stdin when `stdin`).
80    pub paths: Vec<String>,
81    /// The sole path was `-`: search stdin bytes (`search_binary` is then set).
82    pub stdin: bool,
83    pub fixed: bool,
84    pub ignore_case: bool,
85    pub word: bool,
86    pub line: bool,
87    pub quiet: bool,
88    pub before: usize,
89    pub after: usize,
90    /// `-g` globs in argument order.
91    pub globs: Vec<String>,
92    pub hidden: bool,
93    pub no_ignore: bool,
94    /// `-a`: search NUL-containing input as bytes instead of skipping it.
95    pub search_binary: bool,
96    /// `--text`: the plain-text renderer (not upstream's `-a` alias).
97    pub text_display: bool,
98    /// Sole `-h`/`--help`; every other field is the default.
99    pub help: bool,
100}

102/// A usage error: the message without the `rg: ` prefix (status 2).
103#[derive(Debug, Clone, PartialEq, Eq)]
104pub struct Usage(pub String);
105
106#[derive(Debug, Clone, Copy, PartialEq, Eq)]
107enum Opt {
108    Regexp,
109    Fixed,
110    IgnoreCase,
111    Word,
112    Line,
113    LineNumber,
114    Quiet,
115    After,
116    Before,
117    Context,
118    Glob,
119    Hidden,
120    NoIgnore,
121    Binary,
122    Encoding,
123    Text,
124    Help,
125}
126
127impl Opt {
128    fn repeatable(self) -> bool {
129        matches!(self, Opt::Regexp | Opt::Glob)
130    }
131
132    fn bit(self) -> u32 {
133        1 << self as u32
134    }
135}

137/// Every supported spelling; anything else starting with `-` (except a bare `-`) rejects.
138const SPELLINGS: &[(&str, Opt)] = &[
139    ("-e", Opt::Regexp),
140    ("--regexp", Opt::Regexp),
141    ("-F", Opt::Fixed),
142    ("--fixed-strings", Opt::Fixed),
143    ("-i", Opt::IgnoreCase),
144    ("--ignore-case", Opt::IgnoreCase),
145    ("-w", Opt::Word),
146    ("--word-regexp", Opt::Word),
147    ("-x", Opt::Line),
148    ("--line-regexp", Opt::Line),
149    ("-n", Opt::LineNumber),
150    ("--line-number", Opt::LineNumber),
151    ("-q", Opt::Quiet),
152    ("--quiet", Opt::Quiet),
153    ("-A", Opt::After),
154    ("--after-context", Opt::After),
155    ("-B", Opt::Before),
156    ("--before-context", Opt::Before),
157    ("-C", Opt::Context),
158    ("--context", Opt::Context),
159    ("-g", Opt::Glob),
160    ("--glob", Opt::Glob),
161    ("--hidden", Opt::Hidden),
162    ("--no-ignore", Opt::NoIgnore),
163    ("-a", Opt::Binary),
164    ("--encoding", Opt::Encoding),
165    ("--text", Opt::Text),
166    ("-h", Opt::Help),
167    ("--help", Opt::Help),
168];
169
170fn lookup(word: &str) -> Option<Opt> {
171    SPELLINGS.iter().find(|(spelling, _)| *spelling == word).map(|&(_, opt)| opt)
172}

174fn usage<T>(msg: String) -> Result<T, Usage> {
175    Err(Usage(msg))
176}

178/// `word` as safe display text: argv words are untrusted and reach the terminal.
179fn shown(word: &str) -> String {
180    let mut out = Vec::new();
181    escape_line(word.as_bytes(), &mut out);
182    // The escaped form is ASCII plus valid UTF-8 by construction.
183    String::from_utf8(out).unwrap_or_default()
184}

186/// The value of the option at `words[*index - 1]`: exactly the next word, whatever it is.
187fn value<'a>(words: &'a [String], index: &mut usize, option: &str) -> Result<&'a str, Usage> {
188    match words.get(*index) {
189        Some(word) => {
190            *index += 1;
191            Ok(word.as_str())
192        }
193        None => usage(format!("option {} requires a value", option)),
194    }
195}
196
197/// A context line count: decimal digits only, `0..=MAX_CONTEXT`.
198fn context_lines(option: &str, text: &str) -> Result<usize, Usage> {
199    let decimal = !text.is_empty() && text.bytes().all(|b| b.is_ascii_digit());
200    match text.parse::<usize>() {
201        Ok(n) if decimal && n <= MAX_CONTEXT => Ok(n),
202        _ => usage(format!(
203            "option {} needs a decimal line count 0..{}, got '{}'",
204            option,
205            MAX_CONTEXT,
206            shown(text)
207        )),
208    }
209}

211/// Parse `argv` (`argv[0]` is `rg`) into [`RgArgs`]; every failure is a usage error.
212pub fn parse(argv: &[String]) -> Result<RgArgs, Usage> {
213    parse_within(argv, MAX_ARGV_BYTES)
214}
215
216/// [`parse`] with an explicit encoded-argv bound. The 512-byte argv bound shadows the
217/// 4,096-byte pattern and glob bounds, which stay enforced (and testable) independently.
218pub(super) fn parse_within(argv: &[String], max_argv_bytes: usize) -> Result<RgArgs, Usage> {
219    let encoded = argv.iter().fold(0usize, |sum, word| sum.saturating_add(word.len()).saturating_add(1));
220    if encoded > max_argv_bytes {
221        return usage(format!("argv is {} bytes encoded; the limit is {}", encoded, max_argv_bytes));
222    }
223    if argv.iter().any(|word| word.as_bytes().contains(&0)) {
224        return usage(String::from("an argument contains a NUL byte"));
225    }
226    let words = argv.get(1..).unwrap_or(&[]);
227    let mut args = RgArgs::default();
228    let mut seen = 0u32;
229    let mut context = None;
230    let mut ended = false;
231    let mut index = 0;
232
233    // Options, up to `--` or the first operand.
234    while let Some(word) = words.get(index) {
235        let word = word.as_str();
236        if word == "--" {
237            ended = true;
238            index += 1;
239            break;
240        }
241        if word == "-" || !word.starts_with('-') {
242            break;
243        }
244        let Some(opt) = lookup(word) else {
245            return usage(format!("unsupported option: {}", shown(word)));
246        };
247        if !opt.repeatable() && seen & opt.bit() != 0 {
248            return usage(format!("option given more than once: {}", word));
249        }
250        seen |= opt.bit();
251        index += 1;
252        match opt {
253            Opt::Regexp => args.patterns.push(String::from(value(words, &mut index, word)?)),
254            Opt::Fixed => args.fixed = true,
255            Opt::IgnoreCase => args.ignore_case = true,
256            Opt::Word => args.word = true,
257            Opt::Line => args.line = true,
258            // Line numbers are always rendered and never change records.
259            Opt::LineNumber => {}
260            Opt::Quiet => args.quiet = true,
261            Opt::After => args.after = context_lines(word, value(words, &mut index, word)?)?,
262            Opt::Before => args.before = context_lines(word, value(words, &mut index, word)?)?,
263            Opt::Context => context = Some(context_lines(word, value(words, &mut index, word)?)?),
264            Opt::Glob => args.globs.push(String::from(value(words, &mut index, word)?)),
265            Opt::Hidden => args.hidden = true,
266            Opt::NoIgnore => args.no_ignore = true,
267            Opt::Binary => args.search_binary = true,
268            Opt::Encoding => {
269                let encoding = value(words, &mut index, word)?;
270                if encoding != "none" {
271                    return usage(format!("unsupported encoding: {} (only 'none')", shown(encoding)));
272                }
273            }
274            Opt::Text => args.text_display = true,
275            Opt::Help => {
276                if words.len() != 1 {
277                    return usage(format!("{} must be the only argument", word));
278                }
279                return Ok(RgArgs { help: true, ..RgArgs::default() });
280            }
281        }
282    }
283
284    // Operands. Options precede operands: past the first operand, a `-` word still needs
285    // `--` (which may come there) to be an operand.
286    let positional = args.patterns.is_empty();
287    for word in &words[index..] {
288        let word = word.as_str();
289        if !ended {
290            if word == "--" {
291                ended = true;
292                continue;
293            }
294            if word != "-" && word.starts_with('-') {
295                return usage(format!(
296                    "option after an operand: {} (options precede operands; use -- before an \
297                     operand starting with '-')",
298                    shown(word)
299                ));
300            }
301        }
302        if positional && args.patterns.is_empty() {
303            args.patterns.push(String::from(word));
304        } else {
305            args.paths.push(String::from(word));
306        }
307    }
308
309    let has = |opt: Opt| seen & opt.bit() != 0;
310    if has(Opt::Word) && has(Opt::Line) {
311        return usage(String::from("-w and -x cannot be combined"));
312    }
313    if has(Opt::Quiet) && (has(Opt::After) || has(Opt::Before) || has(Opt::Context)) {
314        return usage(String::from("-q cannot be combined with -A, -B or -C"));
315    }
316    if has(Opt::Context) && (has(Opt::After) || has(Opt::Before)) {
317        return usage(String::from("-C cannot be combined with -A or -B"));
318    }
319    if let Some(n) = context {
320        args.before = n;
321        args.after = n;
322    }
323
324    if args.patterns.is_empty() {
325        return usage(String::from("missing PATTERN"));
326    }
327    if args.patterns.len() > MAX_PATTERNS {
328        return usage(format!("at most {} patterns", MAX_PATTERNS));
329    }
330    if args.patterns.iter().map(String::len).sum::<usize>() > MAX_PATTERN_BYTES {
331        return usage(format!("patterns exceed {} bytes in total", MAX_PATTERN_BYTES));
332    }
333    if args.patterns.iter().any(|pattern| pattern.contains('\n')) {
334        return usage(String::from("a pattern contains a line feed (multiline search is unsupported)"));
335    }
336    if args.globs.len() > MAX_GLOBS {
337        return usage(format!("at most {} globs", MAX_GLOBS));
338    }
339    if args.globs.iter().map(String::len).sum::<usize>() > MAX_GLOB_BYTES {
340        return usage(format!("globs exceed {} bytes in total", MAX_GLOB_BYTES));
341    }
342    if args.globs.iter().any(String::is_empty) {
343        return usage(String::from("empty glob"));
344    }
345    if args.paths.len() > MAX_PATHS {
346        return usage(format!("at most {} paths", MAX_PATHS));
347    }
348    if args.paths.iter().any(String::is_empty) {
349        return usage(String::from("empty path"));
350    }
351    if args.paths.iter().any(|path| path == "-") {
352        if args.paths.len() != 1 {
353            return usage(String::from("stdin path - must be the only path"));
354        }
355        if !args.search_binary {
356            return usage(String::from("stdin path - requires -a"));
357        }
358        args.stdin = true;
359        args.paths.clear();
360    }
361    Ok(args)
362}
