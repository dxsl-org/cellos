1//! Streaming plain-text rendering of `cellbox.search.line` rows (proposal §3.6; D74-3,
2//! D76-11, D76-12).
3//!
4//! One row renders as `ESCAPED_PATH (':' | '-') LINE (':' | '-') ESCAPED_LINE '\n'`, `:`
5//! for a match and `-` for context, `<stdin>` for a Null path. The path is the record's
6//! absolute canonical path. Escaping keeps printable valid UTF-8 and writes everything
7//! else as ASCII escapes, so the output is ASCII plus valid UTF-8 and never carries a
8//! terminal control byte (in particular no ESC).
9
10use alloc::format;
11use alloc::vec::Vec;
12use text_engine::pipeline::{PipelineError, Value};
13
14const SCHEMA: &str = "cellbox.search.line";
15const HEX: &[u8; 16] = b"0123456789ABCDEF";
16
17/// Append `path` escaped for display: [`escape_line`] plus `:` → `\x3A` and `-` → `\x2D`,
18/// so the `PATH:LINE:` / `PATH-LINE-` separators stay unambiguous.
19pub fn escape_path(path: &[u8], out: &mut Vec<u8>) {
20    escape(path, true, out);
21}
22
23/// Append `line` escaped for display: `\` → `\\`; LF, CR, tab, NUL → `\n`, `\r`, `\t`,
24/// `\0`; every byte of other controls (C0, DEL, C1), of the bidi/format controls U+061C,
25/// U+200E, U+200F, U+202A–U+202E, U+2066–U+2069 and of invalid UTF-8 → `\xHH`
26/// (uppercase). Printable valid UTF-8 is kept.
27pub fn escape_line(line: &[u8], out: &mut Vec<u8>) {
28    escape(line, false, out);
29}
30
31fn escape(bytes: &[u8], path: bool, out: &mut Vec<u8>) {
32    for chunk in bytes.utf8_chunks() {
33        let text = chunk.valid();
34        // Start of the pending run of characters copied unchanged.
35        let mut run = 0;
36        for (at, c) in text.char_indices() {
37            let kept = match c {
38                '\\' | '\n' | '\r' | '\t' | '\0' => false,
39                ':' | '-' => !path,
40                c => !hidden(c),
41            };
42            if kept {
43                continue;
44            }
45            out.extend_from_slice(&text.as_bytes()[run..at]);
46            run = at + c.len_utf8();
47            match c {
48                '\\' => out.extend_from_slice(b"\\\\"),
49                '\n' => out.extend_from_slice(b"\\n"),
50                '\r' => out.extend_from_slice(b"\\r"),
51                '\t' => out.extend_from_slice(b"\\t"),
52                '\0' => out.extend_from_slice(b"\\0"),
53                _ => text.as_bytes()[at..run].iter().for_each(|&b| hex(b, out)),
54            }
55        }
56        out.extend_from_slice(&text.as_bytes()[run..]);
57        chunk.invalid().iter().for_each(|&b| hex(b, out));
58    }
59}

61/// Characters never shown raw: Cc controls (C0, DEL, C1) and the D76-11 bidi/format
62/// controls, which are not printable and can reorder the displayed text.
63fn hidden(c: char) -> bool {
64    c.is_control()
65        || matches!(c, '\u{061C}' | '\u{200E}' | '\u{200F}' | '\u{202A}'..='\u{202E}' | '\u{2066}'..='\u{2069}')
66}
67
68fn hex(byte: u8, out: &mut Vec<u8>) {
69    out.extend_from_slice(&[b'\\', b'x', HEX[usize::from(byte >> 4)], HEX[usize::from(byte & 0xF)]]);
70}
71
72fn decimal(mut n: u64, out: &mut Vec<u8>) {
73    let mut digits = [0u8; 20];
74    let mut start = digits.len();
75    loop {
76        start -= 1;
77        digits[start] = b'0' + (n % 10) as u8;
78        n /= 10;
79        if n == 0 {
80            break;
81        }
82    }
83    out.extend_from_slice(&digits[start..]);
84}

86fn not_search(field: &str) -> PipelineError {
87    PipelineError::Sequence(format!(
88        "text renderer: row is not a {} record (field '{}' missing or mistyped)",
89        SCHEMA, field
90    ))
91}
92
93fn field<'a>(row: &'a Value, name: &str) -> Result<&'a Value, PipelineError> {
94    row.get_field(name).ok_or_else(|| not_search(name))
95}

97/// Append the text form of one `cellbox.search.line` row to `out`. A row that is not such
98/// a record is an error and leaves `out` unchanged.
99pub fn render_row(row: &Value, out: &mut Vec<u8>) -> Result<(), PipelineError> {
100    match field(row, "schema")? {
101        Value::String(schema) if schema == SCHEMA => {}
102        _ => return Err(not_search("schema")),
103    }
104    if *field(row, "schema_version")? != Value::UInt(1) {
105        return Err(not_search("schema_version"));
106    }
107    let separator = match field(row, "kind")? {
108        Value::String(kind) if kind == "match" => b':',
109        Value::String(kind) if kind == "context" => b'-',
110        _ => return Err(not_search("kind")),
111    };
112    let path = match field(row, "path")? {
113        Value::Path(path) => Some(path),
114        Value::Null => None,
115        _ => return Err(not_search("path")),
116    };
117    let Value::UInt(line_number) = *field(row, "line_number")? else {
118        return Err(not_search("line_number"));
119    };
120    let Value::Bytes(line) = field(row, "line_bytes")? else {
121        return Err(not_search("line_bytes"));
122    };
123
124    match path {
125        Some(path) => escape_path(path, out),
126        None => out.extend_from_slice(b"<stdin>"),
127    }
128    out.push(separator);
129    decimal(line_number, out);
130    out.push(separator);
131    escape_line(line, out);
132    out.push(b'\n');
133    Ok(())
134}
