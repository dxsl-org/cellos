124/// Entry: the launch context is read once and classified (`oneshot::classify`, S10-0):
125/// the ssh client child, the Cellos shell session (`cellbox sh`) or one catalog command
126/// (`/bin/cellbox <command>`).
127#[cfg(not(test))]
128#[no_mangle]
129pub fn main() {
130    #[cfg(target_os = "none")]
131    init_custom_heap();
132    #[cfg(feature = "shell_test")]
133    shell_test::run();
134
135    #[cfg(not(feature = "shell_test"))]
136    match ostd::child_context::ChildContext::read_own() {
137        Ok(context) => {
138            let argv: alloc::vec::Vec<alloc::string::String> =
139                context.argv().map(alloc::string::String::from).collect();
140            match oneshot::classify(context.frontend_grant().is_some(), &argv) {
141                // The ssh client Cell, which the shell itself starts from
142                // `/bin/cellbox-ssh` with argv `["ssh", ...]`.
143                oneshot::Launch::Ssh => ostd::syscall::sys_exit(ssh::run_child(&context) as u32 as usize),
144                oneshot::Launch::Shell => run_frontend_shell(&context),
145                launch => {
146                    let status = oneshot::run(&context, launch, &argv);
147                    ostd::syscall::sys_exit(status as u32 as usize)
148                }
149            }
150        }
151        Err(status) => match context_session::classify_read_failure(status) {
152            context_session::Launch::Legacy => run_legacy_shell(),
153            context_session::Launch::Refused(status) => {
154                let _ = ostd::syscall::sys_log(&alloc::format!(
155                    "shell: launch context refused: {:?}\n", status
156                ));
157                ostd::syscall::sys_exit(125);
158            }
159        },
160    }
161}

163/// A context carrying an input frontend (a GUI Terminal or another `FrontendHost`
164/// launched this shell with `sh`) makes the shell create and control its own session for
165/// that frontend and run the serial controller path, relaying output to the holder's
166/// writers: it never calls `create_serial`, never touches the physical UART and never
167/// reads legacy fd 0. A context that cannot be bound is refused, not downgraded to the
168/// serial path.
169#[cfg(all(not(test), not(feature = "shell_test")))]
170fn run_frontend_shell(context: &ostd::child_context::ChildContext) {
171    let _ = ostd::syscall::sys_log("shell: starting\n");
172    match context_session::open(context) {
173        Ok(launch) => {
174            executor::start_frontend_session(launch.session, launch.output, launch.merged);
175            if let Err(status) = context_session::ensure_default_cwd() {
176                let _ = ostd::syscall::sys_log(&alloc::format!(
177                    "shell: default cwd unavailable: {:?}\n", status
178                ));
179            }
180            if let Err(status) = run_session_loop() {
181                let _ = ostd::syscall::sys_log(&alloc::format!(
182                    "shell: frontend session failed: {:?}\n", status
183                ));
184                ostd::syscall::sys_exit(125);
185            }
186        }
187        Err(status) => {
188            let _ = ostd::syscall::sys_log(&alloc::format!(
189                "shell: launch context refused: {:?}\n", status
190            ));
191            ostd::syscall::sys_exit(125);
192        }
193    }
194}

196/// Without any committed context (the init launch): serial session unless the
197/// compositor is present.
198#[cfg(all(not(test), not(feature = "shell_test")))]
199fn run_legacy_shell() {
200    let _ = ostd::syscall::sys_log("shell: starting\n");
201    // The graphical frontend has not been admitted to SessionV1. Preserve
202    // its existing path; a serial session never falls back to public fd 0.
203    if ostd::syscall::sys_lookup_service(api::syscall::service::COMPOSITOR).is_some() {
204        let mut shell = ViShell::new();
205        ostd::executor::block_on(shell.run());
206    } else if let Err(status) = run_serial_session() {
207        let _ = ostd::syscall::sys_log(&alloc::format!(
208            "shell: authenticated serial session failed: {:?}\n", status
209        ));
210        ostd::syscall::sys_exit(125);
211    }
212}

214/// Serial is an actual owner-backed terminal consumer, not a public-stdin
215/// adapter. The existing dispatcher retains aliases, configuration and grammar.
216#[cfg(all(not(test), not(feature = "shell_test")))]
217fn run_serial_session() -> Result<(), api::cell_session::Status> {
218    executor::start_serial_session()?;
219    run_session_loop()
220}

222/// The line editor and dispatcher over the installed terminal session (serial or
223/// frontend). Returns when the session reports end of input.
224#[cfg(all(not(test), not(feature = "shell_test")))]
225fn run_session_loop() -> Result<(), api::cell_session::Status> {
226    let mut shell = ViShell::new();
227    let mut lines = alloc::vec::Vec::<alloc::string::String>::new();
228    executor::terminal_stdout("\n=== Cellos shell ready — type 'help' for commands ===\n".as_bytes())?;
229    loop {
230        executor::terminal_stdout(b"Cellos > ")?;
231        let Some(mut line) = executor::terminal_read_line(&mut lines)? else {
232            return executor::close_session();
233        };
234        let skip = {
235            let trimmed = line.trim();
236            trimmed.is_empty() || trimmed.starts_with('#')
237        };
238        if skip {
239            ostd::session::zeroize_string(&mut line);
240            continue;
241        }
242        crate::history::record_history(line.trim());
243        let capture = shell_state::CaptureGuard::new();
244        let dispatched = ostd::executor::block_on(shell.dispatch(&line));
245        let mut output = capture.finish();
246        let written = executor::terminal_stdout(&output);
247        // The typed line and captured output may carry secrets (`echo $SECRET`): wipe both. History
248        // keeps its own copy only because the user asked for command history.
249        ostd::session::zeroize_string(&mut line);
250        ostd::session::zeroize_vec(&mut output);
251        written?;
252        if dispatched.is_err() {
253            executor::terminal_stderr(b"shell: command failed\n")?;
254        }
255        if let Some(code) = executor::take_exit_request() {
256            executor::close_session()?;
257            ostd::syscall::sys_exit(code as usize);
258        }
259    }
