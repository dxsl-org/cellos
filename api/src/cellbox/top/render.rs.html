5/// Write one batch frame: a status line (sample time on the monotonic clock, frames
6/// left) and the shared text table, so `top -b` prints the same numbers as
7/// `top --records` and the interactive view.
8pub fn render_frame(frame: &Frame, options: &TopOptions, remaining: usize) {
9    let seconds = frame.meta.capture_start_ns / 1_000_000_000;
10    let mut out = format!("top: t={}s  batch  remaining {}\n", seconds, remaining);
11    text::render_frame(frame, options.fields.as_deref(), &mut out);
12    crate::executor::shell_print(&out);
13}
