1//! Shell AST executor — runs parsed commands, handles pipes and redirects.
2//!
3//! Byte pipelines between built-in commands are implemented via an in-memory
4//! capture stack (`shell_state`): each stage's output is captured into a
5//! `Vec<u8>`, then passed as stdin to the next stage. `CaptureGuard` (RAII,
6//! Law 8) pops the capture on every exit path, including early returns.
7//!
8//! Typed pipelines (`ps --records | where … | take N | to jsonl`) never touch
9//! that stack: `typed::admit` statically admits the WHOLE pipeline (stages,
10//! redirects, background, stateful built-ins) before any stage runs or any file
11//! is opened, and `typed::run_rows` pulls rows through a bounded chain.
12//!
13//! While a terminal session owns the shell, spawned cells join typed and byte
14//! pipelines as external stages: `ExtRun` (end of file) starts them inside one
15//! pipeline Operation, joins neighbouring stages with kernel streams, pumps the
16//! bytes that enter or leave the external segment, drains every child's stderr to
17//! the shell's own stderr sink, keeps servicing the session event queue (Ctrl-C
18//! cancels the whole pipeline) and reaps every stage on every exit path.
19//! Child bytes relayed to the terminal (that pump, and the serial foreground drain)
20//! are followed by `screen_reset`: a child that left the alternate screen, a hidden
21//! cursor or bracketed paste on is followed by one `cellos-vt1` soft reset.
22//!
23//! All shell state lives in `shell_state` behind locks; this module holds no
24//! mutable statics of its own.

145/// One relayed chunk to the frontend device: the physical UART (fd 1/2) on serial, the
146/// holder's writers in frontend mode (the stderr ring goes to the holder's stdout when
147/// the streams are merged).
148fn relay_device(diagnostic: bool, bytes: &[u8]) -> Result<(), Status> {
149    let frontend = *FRONTEND_OUTPUT.lock();
150    if let Some(output) = frontend {
151        let endpoint = match sink_for(diagnostic, streams_merged()) {
152            Sink::Stdout => output.stdout,
153            Sink::Stderr => output.stderr,
154        };
155        return forward_to_frontend(endpoint, bytes);
156    }
157    let mut offset = 0;
158    while offset < bytes.len() {
159        let written = syscall::sys_write(if diagnostic { 2 } else { 1 }, &bytes[offset..])
160            .map_err(|_| Status::Io)?;
161        if written == 0 || written > bytes.len() - offset {
162            return Err(Status::Io);
163        }
164        offset += written;
165    }
166    Ok(())
167}

176/// `render_serial` that also hands every relayed byte to `screen` (the output of a
177/// foreground child, see `screen_reset`). `Ok(true)` when a stream still had bytes after
178/// its bounded turn.
179fn relay_serial(session: &mut TerminalSession, mut screen: Option<&mut RelayScreen>) -> Result<bool, Status> {
180    // Child/command output may echo secrets: the staging buffer is wiped on every exit path.
181    let mut bytes = ostd::session::Scrub::<512>::new();
182    let mut pending = false;
183    for diagnostic in [false, true] {
184        let mut drained = false;
185        // Bound each turn so a busy stdout cannot starve stderr or completion.
186        for _ in 0..8 {
187            let read = if diagnostic {
188                session.drain_stderr(&mut bytes)
189            } else {
190                session.drain_stdout(&mut bytes)
191            };
192            let count = match read {
193                Ok(0) => return Err(Status::Malformed),
194                Ok(count) => count,
195                Err(Status::WouldBlock | Status::Eof) => {
196                    drained = true;
197                    break;
198                }
199                Err(status) => return Err(status),
200            };
201            if let Some(screen) = screen.as_deref_mut() {
202                screen.feed(diagnostic, &bytes[..count]);
203            }
204            relay_device(diagnostic, &bytes[..count])?;
205        }
206        pending |= !drained;
207    }
208    Ok(pending)
209}

221/// Calls made for one owner-notice acknowledgement (yields first, then one-tick sleeps).
222const ACK_ATTEMPTS: u32 = 64;
223
224/// Acknowledge one owner notice: `ack` is retried with `pause` between calls while input
225/// answers transiently (mid-transition, busy, uncertain transport), at most
226/// [`ACK_ATTEMPTS`] calls. Any other refusal, or exhausted retries, is returned.
227fn acknowledge_with(mut ack: impl FnMut() -> Result<(), Status>, mut pause: impl FnMut()) -> Result<(), Status> {
228    let mut attempt = 1;
229    loop {
230        match ack() {
231            Ok(()) => return Ok(()),
232            Err(Status::WouldBlock | Status::TimedOut | Status::Indeterminate) if attempt < ACK_ATTEMPTS => {
233                attempt += 1;
234                pause();
235            }
236            Err(status) => return Err(status),
237        }
238    }
239}

241/// Input-owner notices while a foreground child runs. The input owner captures the answer
242/// itself and routes `TrustPrompt`/`TrustPromptEnd` to this controller's own control ring,
243/// never the child's. Each notice is acknowledged only after it is drawn: input arms its
244/// capture on the prompt's acknowledgement and keeps the session's input from the child
245/// until the end's acknowledgement. While the prompt is drawn or an acknowledgement is
246/// being retried, the child's output is not relayed (it waits in its rings). The prompt
247/// ends on `TrustPromptEnd` or on foreground restoration (the kernel flushes the
248/// controller's ring there).
249#[derive(Default)]
250struct OwnerPrompt {
251    drawn: bool,
252    failed: bool,
253    /// The handshake failed and the foreground child must be cancelled: retried each turn
254    /// with this backoff until the cancellation is in or the foreground is restored.
255    cancel: Option<crate::context_session::Backoff>,
256}
257
258/// What one `cancel_foreground` result means for a cancellation left pending by a failed
259/// prompt handshake.
260#[derive(Debug, PartialEq, Eq)]
261enum CancelStep {
262    /// Cancelled now or already: the operation is cancelled or over (gone or revoked).
263    Done,
264    /// The owner is busy or the outcome is uncertain: try again next turn.
265    Retry,
266    /// Refused for good: the child runs on until it exits, input keeps its input.
267    Stop(Status),
268}
269fn cancel_step(result: Result<(), Status>) -> CancelStep {
270    match result {
271        Ok(()) | Err(Status::Cancelled | Status::Stale | Status::Revoked) => CancelStep::Done,
272        Err(Status::WouldBlock | Status::TimedOut | Status::Indeterminate) => CancelStep::Retry,
273        Err(status) => CancelStep::Stop(status),
274    }
275}
276
277impl OwnerPrompt {
278    /// Draw and acknowledge the pending notices. When the handshake cannot complete (a
279    /// notice unreadable, undrawable or never acknowledged), input still holds this child's
280    /// input, so a cancellation of the foreground child becomes pending (`cancel_turn`):
281    /// its restoration ends the quarantine and the confirmation.
282    fn poll(&mut self, session: &mut TerminalSession) {
283        while !self.failed {
284            let event = match session.next_owner_notice() {
285                Ok(Some(event)) => event,
286                Ok(None) => return,
287                Err(status) => return self.give_up(status),
288            };
289            let drawn = match &event.kind {
290                EventKind::TrustPromptEnd { .. } if self.drawn => relay_device(false, b"\r\n").map(|()| false),
291                kind => match trust_prompt_text(kind) {
292                    // Drawn from the first byte: a failed write may leave part of it on screen.
293                    Some(text) => { self.drawn = true; relay_device(false, text.as_bytes()).map(|()| true) }
294                    None => Ok(false),
295                },
296            };
297            let mut backoff = crate::context_session::Backoff::new();
298            match drawn.and_then(|drawn| acknowledge_with(|| session.acknowledge_owner_notice(&event), || backoff.pause()).map(|()| drawn)) {
299                Ok(drawn) => self.drawn = drawn,
300                Err(status) => return self.give_up(status),
301            }
302        }
303    }
304    /// Stop prompting for this child and make its cancellation pending. A prompt that is
305    /// (even partly) on screen is ended with `\r\n` before any child output is relayed again.
306    fn give_up(&mut self, status: Status) {
307        self.failed = true;
308        self.cancel = Some(crate::context_session::Backoff::new());
309        if core::mem::take(&mut self.drawn) { let _ = relay_device(false, b"\r\n"); }
310        let _ = syscall::sys_log(&alloc::format!("shell: trust prompt handshake failed: {:?}\n", status));
311    }
312    fn relay(&mut self, session: &mut TerminalSession, screen: &mut RelayScreen) -> Result<bool, Status> {
313        if self.drawn { Ok(false) } else { relay_serial(session, Some(screen)) }
314    }
315    fn restored(&mut self) {
316        self.cancel = None;
317        if core::mem::take(&mut self.drawn) { let _ = relay_device(false, b"\r\n"); }
318    }
319    /// One turn of a pending cancellation (nothing when none is pending).
320    fn cancel_turn(&mut self, cancel: impl FnOnce() -> Result<(), Status>) {
321        let Some(backoff) = self.cancel.as_mut() else { return; };
322        match cancel_step(cancel()) {
323            CancelStep::Done => self.cancel = None,
324            CancelStep::Retry => backoff.pause(),
325            CancelStep::Stop(status) => {
326                self.cancel = None;
327                let _ = syscall::sys_log(&alloc::format!("shell: foreground cancel after a failed trust prompt refused: {:?}\n", status));
328            }
329        }
330    }
331}

333/// Retain the unsent suffix across short writes and backpressure. The blocked
334/// callback runs outside the session lock, before any sleep.
335fn write_terminal_bytes<E: From<Status>>(
336    bytes: &[u8],
337    mut write: impl FnMut(&[u8]) -> Result<usize, Status>,
338    mut blocked: impl FnMut() -> Result<(), E>,
339    mut pause: impl FnMut(),
340) -> Result<(), E> {
341    let mut offset = 0;
342    while offset < bytes.len() {
343        match write(&bytes[offset..]) {
344            Ok(count) if count > 0 && count <= bytes.len() - offset => offset += count,
345            Ok(_) => return Err(Status::Malformed.into()),
346            Err(Status::WouldBlock) => {
347                blocked()?;
348                pause();
349            }
350            Err(status) => return Err(status.into()),
351        }
352    }
353    Ok(())
354}

356/// `write_terminal_bytes` against the endpoint `sink_for(diagnostic, merged)` selects.
357/// The routing is decided once per write, so the unsent suffix always retries on the
358/// endpoint that took the first byte.
359fn route_terminal_write<E: From<Status>>(
360    bytes: &[u8],
361    diagnostic: bool,
362    merged: bool,
363    mut write: impl FnMut(Sink, &[u8]) -> Result<usize, Status>,
364    blocked: impl FnMut() -> Result<(), E>,
365    pause: impl FnMut(),
366) -> Result<(), E> {
367    let sink = sink_for(diagnostic, merged);
368    write_terminal_bytes(bytes, |pending| write(sink, pending), blocked, pause)
369}

371fn terminal_write_while<E: From<Status>>(
372    bytes: &[u8],
373    diagnostic: bool,
374    blocked: impl FnMut() -> Result<(), E>,
375) -> Result<(), E> {
376    route_terminal_write(
377        bytes,
378        diagnostic,
379        streams_merged(),
380        |sink, pending| {
381            // Never retain SESSION while polling input or waiting for the host.
382            let mut slot = SERIAL_SESSION.lock();
383            let session = slot.as_mut().ok_or(Status::WrongSession)?;
384            let write = write_sink(session, sink, pending);
385            if matches!(write, Ok(_) | Err(Status::WouldBlock)) {
386                render_serial(session)?;
387            }
388            write
389        },
390        blocked,
391        || {},
392    )
393}

458/// Output of an in-process TUI. The shell is the only drainer of its own session rings,
459/// so a TUI cannot write the stdout ring while it blocks for its terminal (D64-5 S1):
460/// every chunk goes straight to the device the ring relay would reach (`relay_device`:
461/// the UART fd 1 on serial, the holder's stdout writer in frontend mode), in order.
462#[derive(Clone, Copy)]
463pub struct TuiSink<'a> {
464    screen: &'a core::cell::Cell<TuiScreen>,
465}

467impl vitui::backend::ByteSink for TuiSink<'_> {
468    type Error = Status;
469    const MAX_WRITE: usize = 512;
470    fn write(&mut self, bytes: &[u8]) -> Result<usize, Status> {
471        let mut screen = self.screen.get();
472        let written = relay_device(false, bytes);
473        match written {
474            Ok(()) => screen.tracker.feed(bytes),
475            Err(_) => screen.failed = true,
476        }
477        self.screen.set(screen);
478        written.map(|()| bytes.len())
479    }
480}

482/// Run an in-process TUI on this shell's session: take the session guard once, relay
483/// everything still queued in the rings, then hand `run` the session and a [`TuiSink`].
484/// Afterwards, when the device may still be in a TUI screen mode (a restore was lost, or
485/// any write failed), one `SOFT_RESET` follows. Nothing inside `run` may write through
486/// the shell's terminal sinks (`shell_*`, `terminal_*`): the guard is not reentrant, so
487/// diagnostics are written after this returns.
488pub fn with_tui<R>(run: impl FnOnce(&mut TerminalSession, TuiSink<'_>) -> R) -> Result<R, Status> {
489    let mut slot = SERIAL_SESSION.lock();
490    let session = slot.as_mut().ok_or(Status::WrongSession)?;
491    while relay_serial(session, None)? {}
492    let screen = core::cell::Cell::new(TuiScreen { tracker: crate::screen_reset::ModeTracker::new(), failed: false });
493    let result = run(session, TuiSink { screen: &screen });
494    let screen = screen.get();
495    if (screen.failed || screen.tracker.dirty()) && relay_device(false, SOFT_RESET).is_err() {
496        let _ = syscall::sys_log("shell: terminal soft reset delivery failed\n");
497    }
498    Ok(result)
499}

578/// Longest command line the editor accepts, in bytes.
579pub const SHELL_LINE_MAX: usize = 4096;

581/// The submit key of every line reader (command line, `read -s`, prompts, ssh password):
582/// Enter, or Control+J. The serial decoder delivers a lone LF as Control+J (the `spec.md`
583/// input grammar), and Control+J is accept-line, as in readline, so a client that ends lines
584/// with LF alone submits them. Key-level TUI reading (`TerminalKeys`) keeps Control+J a key.
585pub(crate) fn submits_line(code: u32, modifiers: u32) -> bool {
586    use api::services::session::{key_code, modifiers as key_modifiers};
587    (code == key_code::ENTER && modifiers == 0)
588        || (code == u32::from(b'j') && modifiers == key_modifiers::CONTROL)
589}
590
591/// The key a line reader acts on: Enter for any submit key, the code itself for an unmodified
592/// key, and nothing (`0`, which no reader handles) for any other modified key.
593fn line_key(code: u32, modifiers: u32) -> u32 {
594    if submits_line(code, modifiers) {
595        api::services::session::key_code::ENTER
596    } else if modifiers == 0 {
597        code
598    } else {
599        0
600    }
601}

603/// Appends typed text unless the line would exceed `SHELL_LINE_MAX`. Overflowing text marks the
604/// line as overflowed instead of being kept in part, so a truncated command never runs.
605fn push_line_text(line: &mut Wiped, overflow: &mut bool, text: &str) -> bool {
606    if *overflow || line.0.len() + text.len() > SHELL_LINE_MAX {
607        *overflow = true;
608        return false;
609    }
610    line.0.push_str(text);
611    true
612}

614/// Typed canonical input keeps pasted newlines inert until a separate Enter.
615/// No public fd-0 read is reachable from the admitted serial path. A line longer than
616/// `SHELL_LINE_MAX` is refused whole at Enter, never run truncated.
617pub fn terminal_read_line(history: &mut Vec<String>) -> Result<Option<String>, Status> {
618    let mut line = Wiped::with_capacity(SHELL_LINE_MAX);
619    let mut draft = Wiped::with_capacity(SHELL_LINE_MAX);
620    let mut overflow = false;
621    let mut history_index = history.len();
622    let mut paste = false;
623    loop {
624        let event = ScrubbedEvent({
625            let mut slot = SERIAL_SESSION.lock();
626            slot.as_mut().ok_or(Status::WrongSession)?.next_event(u64::MAX)?
627        });
628        match &event.0.kind {
629            EventKind::Text(text) => {
630                if push_line_text(&mut line, &mut overflow, text) {
631                    terminal_stdout(text.as_bytes())?;
632                }
633            }
634            EventKind::PasteBegin => paste = true,
635            EventKind::PasteEnd => paste = false,
636            EventKind::Key { code, modifiers, pressed: 1 | 2 } if !paste => {
637                match line_key(*code, *modifiers) {
638                    0x110001 if overflow => {
639                        line.replace("");
640                        overflow = false;
641                        history_index = history.len();
642                        shell_stderrln(&alloc::format!(
643                            "\nshell: line longer than {} bytes; discarded",
644                            SHELL_LINE_MAX
645                        ));
646                        terminal_stdout(b"Cellos > ")?;
647                    }
648                    0x110001 => {
649                        terminal_stdout(b"\n")?;
650                        if !line.0.trim().is_empty()
651                            && history.last().map(String::as_str) != Some(line.0.as_str())
652                        {
653                            if history.len() == 128 {
654                                history.remove(0);
655                            }
656                            history.push(line.0.clone());
657                        }
658                        return Ok(Some(line.take()));
659                    }
660                    0x110003 => {
661                        if line.0.pop().is_some() {
662                            terminal_stdout(b"\x08 \x08")?;
663                        }
664                    }
665                    0x110010 | 0x110011 => {
666                        if history_index == history.len() {
667                            draft.replace(&line.0);
668                        }
669                        overflow = false;
670                        if *code == 0x110010 {
671                            history_index = history_index.saturating_sub(1);
672                        } else {
673                            history_index = (history_index + 1).min(history.len());
674                        }
675                        line.replace(history.get(history_index).map(String::as_str).unwrap_or(draft.0.as_str()));
676                        terminal_stdout(b"\r\x1b[2KCellos > ")?;
677                        terminal_stdout(line.0.as_bytes())?;
678                    }
679                    _ => {}
680                }
681            }
682            EventKind::Interrupt => {
683                paste = false;
684                terminal_stdout(b"^C\n")?;
685                return Ok(Some(String::new()));
686            }
687            EventKind::Eof => return Ok(None),
688            // The frontend holder detached: the session waits Detached (no EOF) and the
689            // typed line is intact. Logged (fixed text, never input) so a runner can see it.
690            EventKind::Lost(LOST_FRONTEND) => {
691                paste = false;
692                let _ = syscall::sys_log("shell: frontend detached; session waits\n");
693            }
694            // A frontend re-attached (a Resize with its size came first): redraw the
695            // prompt and the line being edited on the new frontend.
696            EventKind::Lost(FRONTEND_ATTACHED) => {
697                let _ = syscall::sys_log("shell: frontend re-attached; redraw\n");
698                terminal_stdout(b"\r\x1b[2KCellos > ")?;
699                terminal_stdout(line.0.as_bytes())?;
700            }
701            EventKind::Lost(reason) => {
702                // Input emits Lost for routine conditions (unmapped key, undecodable
703                // escape, event-queue overflow). Discard the possibly corrupt partial
704                // line and keep the only console shell alive.
705                line.replace("");
706                draft.replace("");
707                history_index = history.len();
708                overflow = false;
709                paste = false;
710                shell_stderrln(&alloc::format!("\nshell: input lost (reason {}); line discarded", reason));
711                terminal_stdout(b"Cellos > ")?;
712            }
713            _ => {}
714        }
715    }
716}

719    Enter,
720    Interrupt,
721    Eof,

724fn read_secret_events(session: &mut TerminalSession, line: &mut Wiped) -> Result<SecretEnd, Status> {
725    let mut paste = false;
726    loop {
727        let event = ScrubbedEvent(session.next_event(u64::MAX)?);
728        match &event.0.kind {
729            // No echo, no cursor edits, no history navigation: only text, Backspace and Enter.
730            EventKind::Text(text) => {
731                if line.0.len() + text.len() <= 128 {
732                    line.0.push_str(text);
733                }
734            }
735            EventKind::PasteBegin => paste = true,
736            EventKind::PasteEnd => paste = false,
737            EventKind::Key { code, modifiers, pressed: 1 | 2 } if !paste => match line_key(*code, *modifiers) {
738                0x110001 => return Ok(SecretEnd::Enter),
739                0x110003 => {
740                    line.0.pop();
741                }
742                _ => {}
743            },
744            EventKind::Interrupt => return Ok(SecretEnd::Interrupt),
745            EventKind::Eof => return Ok(SecretEnd::Eof),
746            // A frontend detach/re-attach drops nothing; any other lost event may have
747            // dropped part of the secret: never accept a corrupt value.
748            EventKind::Lost(reason) if frontend_notice(*reason) => {}
749            EventKind::Lost(_) => return Err(Status::Io),
750            _ => {}
751        }
752    }
753}

755/// Silent canonical read for `read -s`: the typed text is never echoed, never edited through
756/// history and never recorded. The owner's echo-off lease is acquired before the first key is
757/// read; a frontend that cannot suppress echo (the physical UART) answers `Unsupported` and no
758/// secret is collected. The lease is released and the text wiped on every exit path.
759pub fn terminal_read_secret() -> Result<Option<String>, Status> {
760    let mut line = Wiped::with_capacity(128);
761    let end = {
762        let mut slot = SERIAL_SESSION.lock();
763        let session = slot.as_mut().ok_or(Status::WrongSession)?;
764        let mut guard = session.set_mode(2, 0)?;
765        let read = read_secret_events(guard.session(), &mut line);
766        let released = guard.release();
767        let end = read?;
768        released?;
769        end
770    };
771    match end {
772        SecretEnd::Enter => {
773            terminal_stdout(b"\n")?;
774            Ok(Some(line.take()))
775        }
776        SecretEnd::Interrupt => {
777            terminal_stdout(b"^C\n")?;
778            Err(Status::Cancelled)
779        }
780        SecretEnd::Eof => Ok(None),
781    }
782}

797/// Single-key reader over the serial session. It never echoes anything. The paste state lives
798/// here so it survives between non-blocking polls.
799pub struct TerminalKeys {
800    paste: bool,
801}
802
803impl TerminalKeys {
804    pub const fn new() -> Self {
805        Self { paste: false }
806    }
807
808    /// Pure decoder for one event; `None` means the event is not a key press (release, resize,
809    /// lost, unmapped key, anything inside a paste).
810    fn classify(&mut self, kind: &EventKind) -> Option<TerminalKey> {
811        match kind {
812            EventKind::Text(text) if !self.paste => text.chars().next().map(TerminalKey::Char),
813            EventKind::PasteBegin => {
814                self.paste = true;
815                None
816            }
817            EventKind::PasteEnd => {
818                self.paste = false;
819                None
820            }
821            EventKind::Key { code, modifiers: 0, pressed: 1 | 2 } if !self.paste => match *code {
822                0x110001 => Some(TerminalKey::Enter),
823                0x110003 => Some(TerminalKey::Backspace),
824                _ => None,
825            },
826            EventKind::Interrupt => {
827                self.paste = false;
828                Some(TerminalKey::Interrupt)
829            }
830            EventKind::Eof => Some(TerminalKey::Eof),
831            _ => None,
832        }
833    }
834
835    /// Next key press from `session` (the caller holds the session guard, see `with_tui`).
836    /// `deadline_ns` is an absolute monotonic time: `0` polls without blocking and
837    /// `u64::MAX` waits forever. `Ok(None)` means no key arrived before the deadline.
838    pub fn read_from(&mut self, session: &mut TerminalSession, deadline_ns: u64) -> Result<Option<TerminalKey>, Status> {
839        loop {
840            let event = match session.next_event(deadline_ns) {
841                Ok(event) => ScrubbedEvent(event),
842                Err(Status::WouldBlock | Status::TimedOut) => return Ok(None),
843                Err(status) => return Err(status),
844            };
845            if let Some(key) = self.classify(&event.0.kind) {
846                return Ok(Some(key));
847            }
848        }
849    }
850}

862    Idle,
863    /// Text was appended starting at this byte offset of the line.
864    Appended(usize),
865    Erased,
866    Enter,
867    Cancel,

872/// Pure line editing for prompts: text, Backspace, Enter, Ctrl-C only. No history, no cursor
873/// motion. Pasted text stays inert until a separate Enter, exactly as in `terminal_read_line`.
874fn prompt_step(line: &mut Wiped, paste: &mut bool, kind: &EventKind) -> PromptStep {
875    match kind {
876        EventKind::Text(text) => {
877            if line.0.len() + text.len() <= 128 {
878                let start = line.0.len();
879                line.0.push_str(text);
880                if !text.is_empty() {
881                    return PromptStep::Appended(start);
882                }
883            }
884            PromptStep::Idle
885        }
886        EventKind::PasteBegin => {
887            *paste = true;
888            PromptStep::Idle
889        }
890        EventKind::PasteEnd => {
891            *paste = false;
892            PromptStep::Idle
893        }
894        EventKind::Key { code, modifiers, pressed: 1 | 2 } if !*paste => match line_key(*code, *modifiers) {
895            0x110001 => PromptStep::Enter,
896            0x110003 if line.0.pop().is_some() => PromptStep::Erased,
897            _ => PromptStep::Idle,
898        },
899        EventKind::Interrupt => {
900            *paste = false;
901            PromptStep::Cancel
902        }
903        EventKind::Eof => PromptStep::Eof,
904        // A frontend detach/re-attach drops nothing; any other lost event may have dropped
905        // part of the answer: never accept a corrupt one.
906        EventKind::Lost(reason) if frontend_notice(*reason) => PromptStep::Idle,
907        EventKind::Lost(_) => PromptStep::Lost,
908        _ => PromptStep::Idle,
909    }
910}

912/// Canonical single-line answer for builtin prompts (`rm -i`). Echoes typed text, supports
913/// Backspace, never records history, and Ctrl-C cancels. The caller prints the prompt text
914/// itself with `shell_console`. The typed text is wiped on every exit path.
915pub fn terminal_prompt_line() -> Result<PromptLine, Status> {
916    let mut line = Wiped::with_capacity(128);
917    let mut paste = false;
918    loop {
919        let event = ScrubbedEvent({
920            let mut slot = SERIAL_SESSION.lock();
921            slot.as_mut().ok_or(Status::WrongSession)?.next_event(u64::MAX)?
922        });
923        match prompt_step(&mut line, &mut paste, &event.0.kind) {
924            PromptStep::Idle => {}
925            PromptStep::Appended(start) => terminal_stdout(&line.0.as_bytes()[start..])?,
926            PromptStep::Erased => terminal_stdout(b"\x08 \x08")?,
927            PromptStep::Enter => {
928                terminal_stdout(b"\n")?;
929                return Ok(PromptLine::Line(line.take()));
930            }
931            PromptStep::Cancel => {
932                terminal_stdout(b"^C\n")?;
933                return Ok(PromptLine::Cancelled);
934            }
935            PromptStep::Eof => return Ok(PromptLine::Eof),
936            PromptStep::Lost => return Err(Status::Io),
937        }
938    }

1003/// Capture one supported command for `$(...)`.
1004///
1005/// A capture failure aborts the containing command so it cannot substitute
1006/// fabricated output.
1007fn run_capture(inner: &str) -> Result<String, SubstFailed> {
1008    let mut words = inner.split_whitespace();
1009    let cmd = match words.next() {
1010        Some(command) => command,
1011        None => return Ok(String::new()),
1012    };
1013    let args: alloc::vec::Vec<&str> = words.collect();
1014    let output = match cmd {
1015        "echo" => {
1016            let bytes = crate::commands::cmd_echo_to_vec(&args);
1017            String::from(core::str::from_utf8(&bytes).unwrap_or(""))
1018        }
1019        "vcat" | "cat" => {
1020            let Some(path) = args.first() else {
1021                return Ok(String::new());
1022            };
1023            match crate::cmd_fs::read_file_vfs_owned(path, 4096) {
1024                Ok(bytes) => String::from(core::str::from_utf8(&bytes).unwrap_or("")),
1025                Err(_) => {
1026                    shell_stderrln("shell: command substitution read failed");
1027                    String::new()
1028                }
1029            }
1030        }
1031        "pwd" => {
1032            let mut cwd = crate::cmd_cwd::get_shell_cwd().map_err(|_| SubstFailed(1))?;
1033            cwd.push('\n');
1034            cwd
1035        }
1036        "top" => {
1037            let argv: Vec<String> = inner.split_whitespace().map(String::from).collect();
1038            if typed::classify_command(&argv) != typed::CmdClass::Session {
1039                // Batch forms are not substitutable, like every other unlisted command.
1040                return Ok(String::new());
1041            }
1042            // A substitution is a capture, never the terminal: interactive `top` meets its
1043            // runtime gate under this capture and refuses (status 2, a `-b` hint) before
1044            // drawing anything (D64-9).
1045            let capture = CaptureGuard::new();
1046            let status = crate::commands::cmd_top(&argv[1..]);
1047            let bytes = capture.finish();
1048            if status != 0 {
1049                return Err(SubstFailed(status));
1050            }
1051            String::from_utf8_lossy(&bytes).into_owned()
1052        }
1053        _ => String::new(),
1054    };
1055    Ok(output)
1056}

1058/// Expand variable and single-level command substitutions in one token.
1059///
1060/// A failed capture propagates so the containing command ends with its status.
1061fn expand_token(s: &str) -> Result<String, SubstFailed> {
1062    if !s.contains('$') {
1063        return Ok(String::from(s));
1064    }
1065    let mut result = String::new();
1066    let bytes = s.as_bytes();
1067    let mut i = 0;
1068    while i < bytes.len() {
1069        if bytes[i] == b'$' && i + 1 < bytes.len() {
1070            let next = bytes[i + 1];
1071            if next == b'(' {
1072                // Command substitution: $(...). Scan to matching ')'.
1073                // Single-level only — nested $( $() ) passes through as literal.
1074                let inner_start = i + 2;
1075                let mut depth = 1usize;
1076                let mut j = inner_start;
1077                while j < bytes.len() {
1078                    if bytes[j] == b'(' {
1079                        depth += 1;
1080                    } else if bytes[j] == b')' {
1081                        depth -= 1;
1082                        if depth == 0 {
1083                            break;
1084                        }
1085                    }
1086                    j += 1;
1087                }
1088                if depth == 0 {
1089                    // bytes[inner_start..j] is a run of shell token chars, which the
1090                    // parser guarantees are ASCII — the checked decode cannot fail.
1091                    let inner = core::str::from_utf8(&bytes[inner_start..j]).unwrap_or("");
1092                    // Reject nested $(...) — pass $(  literally so the user can see the issue.
1093                    if !inner.contains("$(") {
1094                        let captured = run_capture(inner.trim())?;
1095                        result.push_str(captured.trim_end_matches('\n'));
1096                        i = j + 1;
1097                        continue;
1098                    }
1099                }
1100                // Unmatched paren or nested: emit '$(' literally and continue.
1101                result.push('$');
1102                i += 1;
1103                continue;
1104            }
1105            if next == b'{' {
1106                // `${NAME}` — the value; `${#NAME}` — its length in characters. Anything
1107                // else (unclosed, empty or not a name) stays literal.
1108                let close = bytes[i + 2..].iter().position(|&b| b == b'}');
1109                let inner = close.and_then(|close| core::str::from_utf8(&bytes[i + 2..i + 2 + close]).ok());
1110                if let (Some(close), Some(inner)) = (close, inner) {
1111                    let (length, name) = match inner.strip_prefix('#') {
1112                        Some(name) => (true, name),
1113                        None => (false, inner),
1114                    };
1115                    if is_var_name(name) {
1116                        let mut value = get_var(name).unwrap_or_default();
1117                        if length {
1118                            result.push_str(&alloc::format!("{}", value.chars().count()));
1119                        } else {
1120                            result.push_str(&value);
1121                        }
1122                        // The value may be a `read -s` secret: wipe the copy.
1123                        ostd::session::zeroize_string(&mut value);
1124                        i += close + 3;
1125                        continue;
1126                    }
1127                }
1128                result.push('$');
1129                i += 1;
1130                continue;
1131            }
1132            if next == b'?' {
1133                // $? — exit code of the last command.
1134                if let Some(v) = get_var("?") {
1135                    result.push_str(&v);
1136                }
1137                i += 2;
1138                continue;
1139            }
1140            if next == b'#' {
1141                // $# — positional argument count.
1142                if let Some(v) = get_var("#") {
1143                    result.push_str(&v);
1144                }
1145                i += 2;
1146                continue;
1147            }
1148            if next == b'@' {
1149                // $@ — all positional arguments joined with spaces.
1150                if let Some(v) = get_var("@") {
1151                    result.push_str(&v);
1152                }
1153                i += 2;
1154                continue;
1155            }
1156            if next.is_ascii_digit() && next != b'0' {
1157                // $1..$9 — single-digit positional parameter.
1158                // Guarded by `next.is_ascii_digit()` above.
1159                let key = core::str::from_utf8(&bytes[i + 1..i + 2]).unwrap_or("");
1160                if let Some(v) = get_var(key) {
1161                    result.push_str(&v);
1162                }
1163                i += 2;
1164                continue;
1165            }
1166            if next.is_ascii_alphabetic() || next == b'_' {
1167                let start = i + 1;
1168                let end = bytes[start..]
1169                    .iter()
1170                    .take_while(|&&b| b.is_ascii_alphanumeric() || b == b'_')
1171                    .count()
1172                    + start;
1173                // `take_while` above accepted only ASCII alphanumeric / '_'.
1174                let name = core::str::from_utf8(&bytes[start..end]).unwrap_or("");
1175                if let Some(v) = get_var(name) {
1176                    result.push_str(&v);
1177                }
1178                // Unset variables expand to empty string (POSIX default).
1179                i = end;
1180                continue;
1181            }
1182        }
1183        result.push(bytes[i] as char); // shell tokens are ASCII
1184        i += 1;
1185    }
1186    Ok(result)
1187}

1222/// Execute an `Ast` and return the last command's exit code.
1223///
1224/// `stdin_data` is the bytes available on stdin for the first command in a pipeline.
1225pub fn execute(ast: &Ast, jobs: &mut Jobs) -> i32 {
1226    match ast {
1227        Ast::Empty => 0,
1228        Ast::Simple(cmd) => exec_simple(cmd, jobs),
1229        Ast::Pipeline(cmds) => exec_pipeline(cmds, jobs),
1230        Ast::Invalid(reason) => reject_line(reason),
1231        Ast::BackgroundPipeline(_) => reject_line(
1232            "background `&` is supported only for a single external command; the pipeline was not started",
1233        ),
1234        Ast::Background(cmd) => {
1235            // Refuse typed, stateful and session commands BEFORE any effect: `&` is
1236            // never silently ignored and never mutates the parent context.
1237            if let Ok(stage) = stage_input(cmd) {
1238                let stage = match explicit_stage(stage) {
1239                    Ok(stage) => stage,
1240                    Err(code) => return code,
1241                };
1242                if let Some(reason) = typed::background_refusal(&stage) {
1243                    return reject_line(&reason);
1244                }
1245            }
1246            // Cooperative background: the shell is a single-task executor with no
1247            // async spawn capability for built-ins. `cmd &` runs synchronously and
1248            // is marked Done before control returns. True async background would
1249            // require spawning the command as a separate Cell via SpawnCap — not
1250            // in scope for G1. `fg`/`bg` built-ins report this limitation.
1251            let name = cmd
1252                .argv
1253                .first()
1254                .map(|word| word.text.as_str())
1255                .unwrap_or("?");
1256            let jid = jobs.add(name);
1257            // Background job notification always goes to console, not the sink.
1258            shell_stderrln(&alloc::format!("[{}] running", jid));
1259            // Signal spawn_external to skip sys_wait so a long-running external
1260            // cell (httpd) does not park the shell forever. Built-ins ignore it.
1261            state::set_bg_spawn(true);
1262            exec_cmd(cmd, &[], jobs);
1263            state::set_bg_spawn(false);
1264            jobs.set_state(jid, JobState::Done);
1265            0
1266        }
1267        Ast::Sequence(sub) => {
1268            let mut last = 0;
1269            for s in sub {
1270                last = execute(s, jobs);
1271            }
1272            last
1273        }
1274        Ast::Case { expr, arms } => {
1275            let value = match expand_token(expr) {
1276                Ok(value) => value,
1277                Err(failed) => return failed.0,
1278            };
1279            for (pattern, body) in arms {
1280                if case_matches(pattern, &value) {
1281                    execute(body, jobs);
1282                    break;
1283                }
1284            }
1285            0
1286        }
1287        Ast::FuncDef { name, body } => {
1288            define_function(name, body);
1289            0
1290        }
1291        Ast::And(left, right) => {
1292            let code = execute(left, jobs);
1293            if code == 0 {
1294                execute(right, jobs)
1295            } else {
1296                code
1297            }
1298        }
1299        Ast::Or(left, right) => {
1300            let code = execute(left, jobs);
1301            if code != 0 {
1302                execute(right, jobs)
1303            } else {
1304                code
1305            }
1306        }
1307        Ast::While { cond, body } => {
1308            loop {
1309                if execute(cond, jobs) != 0 {
1310                    break;
1311                }
1312                execute(body, jobs);
1313                match take_loop_signal() {
1314                    LoopSignal::Break => break,
1315                    LoopSignal::Continue => continue,
1316                    LoopSignal::None => {}
1317                }
1318            }
1319            0
1320        }
1321        Ast::For { var, words, body } => {
1322            'for_loop: for word in words {
1323                set_var(var, word);
1324                execute(body, jobs);
1325                match take_loop_signal() {
1326                    LoopSignal::Break => break 'for_loop,
1327                    LoopSignal::Continue => continue 'for_loop,
1328                    LoopSignal::None => {}
1329                }
1330            }
1331            0
1332        }
1333        Ast::If {
1334            cond,
1335            then_b,
1336            else_b,
1337        } => {
1338            let code = execute(cond, jobs);
1339            if code == 0 {
1340                execute(then_b, jobs)
1341            } else if let Some(eb) = else_b {
1342                execute(eb, jobs)
1343            } else {
1344                0
1345            }
1346        }
1347    }
1348}

1358/// Expand a command's words and redirect targets for whole-pipeline admission.
1359/// Expansion has no effect (substitutions are read-only), so it may run before
1360/// admission; a failure aborts the command with the substitution's status like
1361/// `exec_cmd` does.
1362fn stage_input(cmd: &Cmd) -> Result<StageIn, SubstFailed> {
1363    let argv = cmd
1364        .argv
1365        .iter()
1366        .map(expand_word)
1367        .collect::<Result<Vec<_>, _>>()?;
1368    let mut redirects = Vec::with_capacity(cmd.redirects.len());
1369    for redirect in &cmd.redirects {
1370        let (kind, word) = match redirect {
1371            Redirect::StdoutTo(word) => (RedirKind::Out, word),
1372            Redirect::StdoutAppend(word) => (RedirKind::Append, word),
1373            Redirect::StdinFrom(word) => (RedirKind::In, word),
1374            Redirect::StderrTo(word) => (RedirKind::Err, word),
1375        };
1376        redirects.push(Redir { kind, path: expand_word(word)? });
1377    }
1378    Ok(StageIn { argv, redirects })
1379}

1381/// The in-shell explicit form (S10-0.6) of an expanded argv, in process and without a
1382/// spawn: `Ok` with a Run name's `cellbox`/`/bin/cellbox` prefix dropped (any other argv
1383/// unchanged), or `Err` with the one-shot's answer (`text` on stdout or stderr, status).
1384fn explicit_form(mut argv: Vec<String>) -> Result<Vec<String>, (i32, bool, String)> {
1385    match crate::oneshot::prefixed(&argv) {
1386        crate::oneshot::Prefixed::No => Ok(argv),
1387        crate::oneshot::Prefixed::Strip => {
1388            argv.remove(0);
1389            Ok(argv)
1390        }
1391        crate::oneshot::Prefixed::Answer { status, stdout, text } => Err((status, stdout, text)),
1392    }
1393}

1395/// [`explicit_form`] for a command about to be admitted or dispatched: an answer is
1396/// written and recorded as `$?` before any effect, and returned as `Err(status)`.
1397fn explicit_argv(argv: Vec<String>) -> Result<Vec<String>, i32> {
1398    explicit_form(argv).map_err(|(status, stdout, text)| {
1399        if stdout {
1400            shell_print(&text);
1401        } else {
1402            shell_stderr(&text);
1403        }
1404        state::set_status(status);
1405        status
1406    })
1407}

1415/// One simple command. Only commands that can be typed stages or interactive
1416/// sessions go through whole-command admission, plus an external whose stdout is
1417/// redirected (a one-stage byte pipeline: its stdout is never the terminal session);
1418/// everything else is unchanged.
1419fn exec_simple(cmd: &Cmd, jobs: &mut Jobs) -> i32 {
1420    if let Some(first) = cmd.argv.first() {
1421        let args: Vec<&str> = cmd.argv[1..].iter().map(|word| word.text.as_str()).collect();
1422        let redirected_external = serial_active()
1423            && cmd
1424                .redirects
1425                .iter()
1426                .any(|redirect| matches!(redirect, Redirect::StdoutTo(_) | Redirect::StdoutAppend(_)))
1427            && (matches!(
1428                typed::classify_command(core::slice::from_ref(&first.text)),
1429                typed::CmdClass::External
1430            ) || (first.text == "ssh"
1431                // A publickey exec is an external byte stage: with its stdout redirected it is
1432                // a one-stage pipeline (every other ssh keeps the foreground launch).
1433                && typed::is_ssh_stage(&cmd.argv.iter().map(|word| word.text.clone()).collect::<Vec<String>>())));
1434        if typed::might_be_typed(&first.text, &args) || redirected_external {
1435            let stage = match stage_input(cmd) {
1436                Ok(stage) => stage,
1437                Err(failed) => return failed.abort(),
1438            };
1439            let stage = match explicit_stage(stage) {
1440                Ok(stage) => stage,
1441                Err(code) => return code,
1442            };
1443            match typed::admit(core::slice::from_ref(&stage), serial_active()) {
1444                Admission::Legacy => {}
1445                Admission::Reject(reason) => return reject_line(&reason),
1446                Admission::Typed(plan) => {
1447                    return exec_typed(&plan, core::slice::from_ref(cmd), core::slice::from_ref(&stage), jobs)
1448                }
1449                Admission::Bytes(plan) => {
1450                    return exec_bytes(&plan, core::slice::from_ref(cmd), core::slice::from_ref(&stage), jobs)
1451                }
1452            }
1453        }
1454    }
1455    exec_cmd(cmd, &[], jobs)
1456}

1458/// Execute a multi-stage pipeline.
1459///
1460/// The whole pipeline is expanded and admitted first; a refusal (unsupported
1461/// category combination, stateful built-in, session, >16 stages, bad redirect)
1462/// returns status 2 with no stage run and no file opened. Admitted byte-only
1463/// pipelines keep the capture composition; typed pipelines run as one job.
1464fn exec_pipeline(cmds: &[Cmd], jobs: &mut Jobs) -> i32 {
1465    if cmds.is_empty() {
1466        return 0;
1467    }
1468    let mut stages = Vec::with_capacity(cmds.len());
1469    for cmd in cmds {
1470        let stage = match stage_input(cmd) {
1471            Ok(stage) => stage,
1472            Err(failed) => return failed.abort(),
1473        };
1474        match explicit_stage(stage) {
1475            Ok(stage) => stages.push(stage),
1476            Err(code) => return code,
1477        }
1478    }
1479    match typed::admit(&stages, serial_active()) {
1480        Admission::Reject(reason) => reject_line(&reason),
1481        Admission::Legacy => {
1482            let mut statuses = Vec::new();
1483            run_byte_chain(cmds, Vec::new(), jobs, &mut statuses)
1484        }
1485        Admission::Typed(plan) => exec_typed(&plan, cmds, &stages, jobs),
1486        Admission::Bytes(plan) => exec_bytes(&plan, cmds, &stages, jobs),
1487    }
1488}

1490/// Run byte stages in order, piping each stage's output into the next stage's
1491/// stdin. Intermediate stages are captured into `Vec<u8>` buffers; the final
1492/// stage runs directly through the current sink so its exit code is preserved
1493/// and any outer capture (nested pipeline or `$(...)`) captures it correctly.
1494/// `statuses` receives one status per stage run. `cmds` must be non-empty.
1495fn run_byte_chain(cmds: &[Cmd], mut stdin_data: Vec<u8>, jobs: &mut Jobs, statuses: &mut Vec<i32>) -> i32 {
1496    let last_idx = cmds.len() - 1;
1497    for (i, cmd) in cmds.iter().enumerate() {
1498        if i == last_idx {
1499            // Last stage: run directly (no intermediate capture).
1500            // Wire pipe stdin so built-ins without a file path read from it.
1501            state::set_stdin(&stdin_data);
1502            let code = exec_cmd(cmd, &stdin_data, jobs);
1503            state::clear_stdin();
1504            statuses.push(code);
1505            return code;
1506        }
1507        let (captured, code) = capture_cmd(cmd, &stdin_data, jobs);
1508        statuses.push(code);
1509        stdin_data = captured;
1510    }
1511    0
1512}

1541/// Look at the session event queue without blocking: true when Ctrl-C was seen.
1542/// Everything else typed meanwhile is discarded (scrubbed). An input service
1543/// failure clears `enabled`, after which nothing is polled.
1544fn poll_session_interrupt(enabled: &mut bool) -> bool {
1545    if !*enabled {
1546        return false;
1547    }
1548    for _ in 0..16 {
1549        let event = {
1550            let mut slot = SERIAL_SESSION.lock();
1551            let Some(session) = slot.as_mut() else {
1552                *enabled = false;
1553                return false;
1554            };
1555            match session.next_event(0) {
1556                Ok(event) => ScrubbedEvent(event),
1557                Err(Status::WouldBlock | Status::TimedOut) => return false,
1558                Err(_) => {
1559                    *enabled = false;
1560                    return false;
1561                }
1562            }
1563        };
1564        if matches!(event.0.kind, EventKind::Interrupt) {
1565            return true;
1566        }
1567    }
1568    false
1569}

1571/// The interrupt latch of one job (D74-1): every poller of the session queue —
1572/// the streamed terminal sink, each external segment's pump and a working
1573/// producer (`rg`) — reads Ctrl-C through it, so an interrupt one of them
1574/// consumed is never lost to the others.
1575pub(crate) fn job_latch() -> Rc<InterruptLatch> {
1576    let mut enabled = true;
1577    InterruptLatch::new(Box::new(move || poll_session_interrupt(&mut enabled)))
1578}

1596/// The terminal as the sink of a streamed typed segment: bytes reach the session
1597/// stdout as they are produced (`terminal_write_while` retries backpressure with
1598/// the same backoff as every other writer), and a real Ctrl-C is polled (through
1599/// the job latch) between rows and whenever the terminal is not accepting bytes.
1600struct TerminalSink {
1601    latch: Rc<InterruptLatch>,
1602}
1603
1604impl typed::ChunkSink for TerminalSink {
1605    fn write(&mut self, bytes: &[u8]) -> Result<(), typed::SinkStop> {
1606        if self.latch.is_set() {
1607            return Err(typed::SinkStop::Interrupted);
1608        }
1609        terminal_write_while(bytes, false, || {
1610            if self.latch.poll_now() {
1611                Err(TerminalWriteStop::Interrupted)
1612            } else {
1613                Ok(())
1614            }
1615        })
1616        .map_err(|stop| match stop {
1617            TerminalWriteStop::Interrupted => typed::SinkStop::Interrupted,
1618            TerminalWriteStop::Output(status) => {
1619                typed::SinkStop::Failed(alloc::format!("terminal output failed: {:?}", status))
1620            }
1621        })
1622    }
1623
1624    fn interrupted(&mut self) -> bool {
1625        self.latch.poll_now()
1626    }
1627}

1667/// Bytes entering the external stages of a byte pipeline, then the stages
1668/// themselves. The externals are started first, so an unknown program or an
1669/// oversized launch fails before a built-in head runs or a file is read.
1670fn run_bytes(
1671    plan: &BytePlan,
1672    cmds: &[Cmd],
1673    stages: &[StageIn],
1674    jobs: &mut Jobs,
1675    members: &mut [(String, i32)],
1676) -> i32 {
1677    let first = plan.prefix;
1678    let mut run = match ExtRun::start(&stages[first..], first > 0, &job_latch()) {
1679        Ok(run) => run,
1680        Err(failed) => return fail_member(members, first + failed.stage, failed.status),
1681    };
1682    let feed = match &plan.head {
1683        ByteHead::None => Vec::new(),
1684        ByteHead::Cat(path) => match crate::native_rows::read_file_bytes_exact(path) {
1685            Ok(bytes) if bytes.len() <= typed::MAX_SINK_BYTES => bytes,
1686            Ok(_) => {
1687                shell_stderrln("shell: input exceeds the typed pipeline byte bound");
1688                return fail_member(members, 0, 1);
1689            }
1690            Err(code) => return fail_member(members, 0, code),
1691        },
1692        ByteHead::Capture => {
1693            // Built-in producers run over captured bytes; nothing reaches the terminal.
1694            let guard = CaptureGuard::new();
1695            let mut statuses = Vec::new();
1696            run_byte_chain(&cmds[..first], Vec::new(), jobs, &mut statuses);
1697            let bytes = guard.finish();
1698            for (index, status) in statuses.iter().enumerate() {
1699                if let Some(member) = members.get_mut(index) {
1700                    member.1 = *status;
1701                    if *status != 0 {
1702                        shell_stderrln(&alloc::format!("shell: {} exited with status {}", member.0, status));
1703                    }
1704                }
1705            }
1706            if bytes.len() > typed::MAX_SINK_BYTES {
1707                shell_stderrln("shell: input exceeds the typed pipeline byte bound");
1708                return fail_member(members, first.saturating_sub(1), 1);
1709            }
1710            bytes
1711        }
1712    };
1713    let driven = drive_external_sink(&mut run, feed, plan.output.as_ref());
1714    settle_external_segment(run, driven, members, first, plan.output.as_ref())
1715}

1717/// Bytes feeding the leading `from` of a byte-headed typed pipeline, plus the
1718/// index of that `from` stage.
1719fn byte_head(plan: &TypedPlan, cmds: &[Cmd], jobs: &mut Jobs) -> Result<(Vec<u8>, usize), i32> {
1720    match &plan.head {
1721        Head::Cat(path) => crate::native_rows::read_file_bytes_exact(path).map(|bytes| (bytes, 1)),
1722        Head::FromFile(path) => crate::native_rows::read_file_bytes_exact(path).map(|bytes| (bytes, 0)),
1723        Head::Capture => {
1724            let (bytes, code) = capture_cmd(&cmds[0], &[], jobs);
1725            if code != 0 {
1726                shell_stderrln(&alloc::format!("shell: stage 0 exited with status {}", code));
1727                Err(code)
1728            } else {
1729                Ok((bytes, 1))
1730            }
1731        }
1732        // A one-shot's deferred stdin (`/bin/cellbox rg … -`), read on this demand.
1733        Head::Stdin => Ok((state::stdin_bytes(), 0)),
1734        Head::Producer | Head::External { .. } => Err(2),
1735    }
1736}

1743    members: &mut [(String, i32)],
1744) -> i32 {
1745    use text_engine::pipeline::{RowSource, Stage};
1746    // One interrupt latch for the whole job: the terminal sink, every external
1747    // segment and a working producer read Ctrl-C through it.
1748    let latch = job_latch();
1749    // External stages are started before any stage computes, so an unknown program
1750    // fails before the typed segment (or a capturing head) has run. Dropping an
1751    // `ExtRun` on any early return cancels and reaps what was started.
1752    let mut tail = if plan.tail_externals > 0 {
1753        match ExtRun::start(&stages[plan.typed_cmds..], true, &latch) {
1754            Ok(run) => Some(run),
1755            Err(failed) => return fail_member(members, plan.typed_cmds + failed.stage, failed.status),
1756        }
1757    } else {
1758        None
1759    };
1760    let mut head_run: Option<Rc<core::cell::RefCell<ExtRun>>> = None;
1761    // `rg -` consumes the head's bytes itself (no `from`).
1762    let search_stdin = plan.search.as_ref().filter(|args| args.stdin);
1763    // Head: a native row producer, an external segment, or bytes converted by the
1764    // leading `from` (or searched by `rg -`). Construction validates arguments only;
1765    // producers do no work until pulled.
1766    let (source, first_stage, rest): (Box<dyn RowSource>, usize, &[Stage]) = match &plan.head {
1767        Head::Producer => match crate::native_rows::row_producer(&plan.producer_argv, &latch) {
1768            Some(Ok(source)) => (source, 0, &plan.stages[..]),
1769            Some(Err(code)) => return fail_member(members, 0, code),
1770            None => {
1771                shell_stderrln("shell: record producer is not available");
1772                return fail_member(members, 0, 2);
1773            }
1774        },
1775        Head::External { count } => {
1776            let format = match (search_stdin, plan.stages.first()) {
1777                (Some(_), _) => None,
1778                (None, Some(Stage::From(format))) => Some(format.clone()),
1779                (None, _) => {
1780                    shell_stderrln("shell: external output needs `from json|jsonl`");
1781                    return fail_member(members, *count, 2);
1782                }
1783            };
1784            let run = match ExtRun::start(&stages[..*count], false, &latch) {
1785                Ok(run) => run,
1786                Err(failed) => return fail_member(members, failed.stage, failed.status),
1787            };
1788            let shared = Rc::new(core::cell::RefCell::new(run));
1789            head_run = Some(shared.clone());
1790            match (search_stdin, format) {
1791                (Some(args), _) => {
1792                    let input = crate::rg::StdinInput::Stream(Box::new(ChildBytes(shared)));
1793                    match crate::rg::stdin_producer(args, input, &latch) {
1794                        Ok(source) => (source, *count, &plan.stages[..]),
1795                        Err(code) => return fail_member(members, *count, code),
1796                    }
1797                }
1798                (None, format) => {
1799                    let format = format.unwrap_or(text_engine::pipeline::Format::JsonlTyped);
1800                    let rows: Box<dyn RowSource> = Box::new(typed::StreamRows::new(ChildBytes(shared), format));
1801                    (rows, *count, &plan.stages[1..])
1802                }
1803            }
1804        }
1805        Head::Cat(_) | Head::Capture | Head::FromFile(_) | Head::Stdin => {
1806            let (bytes, from_index) = match byte_head(plan, cmds, jobs) {
1807                Ok(head) => head,
1808                Err(code) => return fail_member(members, 0, code),
1809            };
1810            if bytes.len() > typed::MAX_SINK_BYTES {
1811                shell_stderrln("shell: input exceeds the typed pipeline byte bound");
1812                return fail_member(members, from_index, 1);
1813            }
1814            if let Some(args) = search_stdin {
1815                match crate::rg::stdin_producer(args, crate::rg::StdinInput::Bytes(bytes), &latch) {
1816                    Ok(source) => (source, from_index, &plan.stages[..]),
1817                    Err(code) => return fail_member(members, from_index, code),
1818                }
1819            } else {
1820                let Some(Stage::From(format)) = plan.stages.first() else {
1821                    shell_stderrln("shell: byte input needs `from json|jsonl`");
1822                    return fail_member(members, from_index, 2);
1823                };
1824                // Malformed input fails here, before any redirect is opened or written.
1825                match typed::rows_from_bytes(format, &bytes) {
1826                    Ok(source) => (source, from_index, &plan.stages[1..]),
1827                    Err(error) => {
1828                        shell_stderrln(&alloc::format!("shell: {}: {}", members[from_index].0, error));
1829                        return fail_member(members, from_index, 1);
1830                    }
1831                }
1832            }
1833        }
1834    };
1835    // The producer's completion status, published when the chain releases it.
1836    let (source, completion) = typed::observe_completion(source);
1837    let search = plan.search.as_ref().map(|_| SearchMember { index: plan.search_index, completion });
1838
1839    // Rows feeding external commands: serialized into the first stage's stdin as it
1840    // consumes them (bounded memory, consumer-driven), never collected.
1841    if let Some(mut run) = tail.take() {
1842        let feed_sink = if plan.search_text && plan.text_display { DefaultSink::SearchText } else { DefaultSink::TypedJsonl };
1843        let feed = match typed::RowFeed::new(source, plan.category, first_stage, rest, feed_sink) {
1844            Ok(feed) => feed,
1845            Err(failure) => {
1846                let name = members.get(failure.stage).map_or("pipeline", |member| member.0.as_str());
1847                shell_stderrln(&alloc::format!("shell: {}: {}", name, failure.message));
1848                return fail_member(members, failure.stage, failure.status);
1849            }
1850        };
1851        run.set_rows(feed);
1852        let driven = drive_external_sink(&mut run, Vec::new(), plan.tail_output.as_ref());
1853        if let Some(failure) = run.take_row_failure() {
1854            // A row stage failed after some rows were already delivered: stop the
1855            // external stages (the failed feed is not a clean end), then report it.
1856            let statuses = run.conclude();
1857            for (offset, status) in statuses.iter().enumerate() {
1858                if let Some(member) = members.get_mut(plan.typed_cmds + offset) {
1859                    member.1 = *status;
1860                }
1861            }
1862            drop(run);
1863            if let Some(search) = &search {
1864                search.record(members, &latch, false);
1865            }
1866            let name = members.get(failure.stage).map_or("pipeline", |member| member.0.as_str());
1867            shell_stderrln(&alloc::format!("shell: {}: {}", name, failure.message));
1868            return fail_member(members, failure.stage, failure.status);
1869        }
1870        // The feed (and the producer) is released with the run, before its status is read.
1871        let status = settle_external_segment(run, driven, members, plan.typed_cmds, plan.tail_output.as_ref());
1872        if let Some(search) = &search {
1873            search.record(members, &latch, status != INTERRUPTED_STATUS);
1874        }
1875        return status;
1876    }
1877
1878    let terminal = plan.output.is_none() && plan.typed_cmds == cmds.len();
1879    let oneshot = state::oneshot_active();
1880    let default_sink = typed::default_sink(terminal && !oneshot, plan);
1881    // The live terminal is a streaming sink: each encoded row is written as it is
1882    // pulled and Ctrl-C is polled between rows, so an endless producer is neither
1883    // buffered nor uninterruptible. A one-shot's stdout streams the same way (S10-0.5):
1884    // a closed reader stops the producer at once. Only a redirect target, a byte
1885    // consumer or an enclosing capture (`$(...)`) need the whole, bounded buffer.
1886    let outcome = if terminal && oneshot && state::capture_depth() == 0 {
1887        LocalOutcome::Streamed(typed::stream_rows(source, plan.category, first_stage, rest, default_sink, &mut OneShotSink))
1888    } else if terminal && serial_active() && state::capture_depth() <= 1 {
1889        flush_repl_capture();
1890        let mut sink = TerminalSink { latch: latch.clone() };
1891        LocalOutcome::Streamed(typed::stream_rows(source, plan.category, first_stage, rest, default_sink, &mut sink))
1892    } else {
1893        // A Collected run polls Ctrl-C before every pull (S10-3b), so an endless
1894        // producer feeding a byte consumer (`| cat`) stops with 130 and writes nothing.
1895        let bound = if plan.search.is_some() { typed::SEARCH_COLLECT_BOUND } else { typed::COLLECT_BOUND };
1896        LocalOutcome::Collected(typed::run_rows_within(source, plan.category, first_stage, rest, default_sink, bound, &mut || {
1897            latch.poll_now()
1898        }))
1899    };
1900
1901    // An external head is reaped whatever the typed segment did: a satisfied `take`
1902    // or an error already closed its stream, so the stages are cancelled here.
1903    if let Some(shared) = head_run {
1904        let mut run = shared.borrow_mut();
1905        let statuses = run.conclude();
1906        let cancelled = run.was_cancelled();
1907        let not_started = run.not_started();
1908        drop(run);
1909        for (index, status) in statuses.iter().enumerate() {
1910            if let Some(member) = members.get_mut(index) {
1911                member.1 = *status;
1912                // A stage that never started ran nothing and reports no exit.
1913                if *status != 0 && !cancelled && index >= not_started && !latch.is_set() {
1914                    shell_stderrln(&alloc::format!("shell: {} exited with status {}", member.0, status));
1915                }
1916            }
1917        }
1918    }
1919    // The search member's status; reported as an earlier failure unless it is the
1920    // pipeline status itself (D76-8, D76-15).
1921    let search_status = search.as_ref().map(|search| search.record(members, &latch, !plan.search_status));
1922    // Ctrl-C ends the job with 130 whoever saw it; a Collected run writes nothing.
1923    if latch.is_set() {
1924        shell_console("^C\n");
1925        return INTERRUPTED_STATUS;
1926    }
1927
1928    let bytes = match outcome {
1929        LocalOutcome::Streamed(Ok(end)) => {
1930            if end == typed::StreamEnd::Interrupted {
1931                shell_console("^C\n");
1932                return end.status();
1933            }
1934            return match search_status {
1935                Some(status) if plan.search_status => status,
1936                _ => end.status(),
1937            };
1938        }
1939        LocalOutcome::Collected(Ok(bytes)) => bytes,
1940        LocalOutcome::Streamed(Err(failure)) | LocalOutcome::Collected(Err(failure)) => {
1941            let name = members.get(failure.stage).map_or("pipeline", |member| member.0.as_str());
1942            shell_stderrln(&alloc::format!("shell: {}: {}", name, failure.message));
1943            return fail_member(members, failure.stage, failure.status);
1944        }
1945    };
1946    let success = match search_status {
1947        Some(status) if plan.search_status => status,
1948        _ => 0,
1949    };
1950
1951    // `to` (or `rg`'s default sink) fed byte-consuming built-ins: they run over the
1952    // captured bytes.
1953    if plan.typed_cmds < cmds.len() {
1954        let mut statuses = Vec::new();
1955        let code = run_byte_chain(&cmds[plan.typed_cmds..], bytes, jobs, &mut statuses);
1956        for (offset, status) in statuses.iter().enumerate() {
1957            let index = plan.typed_cmds + offset;
1958            if let Some(member) = members.get_mut(index) {
1959                member.1 = *status;
1960                if *status != 0 && offset + 1 < statuses.len() {
1961                    shell_stderrln(&alloc::format!("shell: {} exited with status {}", member.0, status));
1962                }
1963            }
1964        }
1965        return code;
1966    }
1967    if let Some((path, append)) = &plan.output {
1968        if !crate::cmd_fs::vfs_write_chunked(path, &bytes, *append) {
1969            shell_stderr("shell: cannot write '");
1970            shell_stderr(path);
1971            shell_stderrln("'");
1972            return fail_member(members, usize::MAX, 1);
1973        }
1974        return success;
1975    }
1976    // Terminal display of rows/serializer output; redirects above stay byte-exact.
1977    shell_print(&String::from_utf8_lossy(&bytes));
1978    if !bytes.is_empty() && !bytes.ends_with(b"\n") {
1979        shell_print("\n");
1980    }
1981    success
1982}

1984/// The `rg` stage of a typed job: its pipeline index and completion slot.
1985struct SearchMember {
1986    index: usize,
1987    completion: typed::CompletionSlot,
1988}
1989
1990impl SearchMember {
1991    /// Store the search's status in its member (130 after Ctrl-C) and return it.
1992    /// With `report`, a nonzero status that is not the pipeline status is listed
1993    /// like any other failed non-final stage (`NAME exited with status N`).
1994    fn record(&self, members: &mut [(String, i32)], latch: &InterruptLatch, report: bool) -> i32 {
1995        let status = match self.completion.get() {
1996            Some(status) if !latch.is_set() => status,
1997            _ if latch.is_set() => INTERRUPTED_STATUS,
1998            _ => 0,
1999        };
2000        if let Some(member) = members.get_mut(self.index) {
2001            member.1 = status;
2002            if report && status != 0 && !latch.is_set() {
2003                shell_stderrln(&alloc::format!("shell: {} exited with status {}", member.0, status));
2004            }
2005        }
2006        status
2007    }
2008}

2010/// Execute one simple command.
2011///
2012/// Handles redirection, built-in dispatch, and external binary spawn.
2013fn exec_cmd(cmd: &Cmd, _stdin: &[u8], jobs: &mut Jobs) -> i32 {
2014    if cmd.is_empty() {
2015        return 0;
2016    }
2017
2018    // Failed command substitution aborts before dispatch with a nonzero status.
2019    let expanded = match cmd
2020        .argv
2021        .iter()
2022        .map(expand_word)
2023        .collect::<Result<Vec<_>, _>>()
2024    {
2025        Ok(expanded) => expanded,
2026        Err(failed) => return failed.abort(),
2027    };
2028    // `cellbox CMD` / `/bin/cellbox CMD` (S10-0.6): the same table as the one-shot.
2029    let expanded = match explicit_argv(expanded) {
2030        Ok(expanded) => expanded,
2031        Err(code) => return code,
2032    };
2033    let prog: &str = &expanded[0];
2034    let args: Vec<String> = expanded[1..].to_vec();
2035
2036    // Detect `KEY=VALUE` assignment (key is non-empty alphanumeric+underscore).
2037    if args.is_empty() {
2038        if let Some(eq) = prog.find('=') {
2039            let key = &prog[..eq];
2040            if !key.is_empty() && key.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'_') {
2041                set_var(key, &prog[eq + 1..]);
2042                return 0;
2043            }
2044        }
2045    }
2046
2047    // echo with stdout redirect: fast path using cmd_echo_to_vec (no OutputSink needed).
2048    if prog == "echo" {
2049        if let Some(Redirect::StdoutTo(path)) = cmd
2050            .redirects
2051            .iter()
2052            .find(|r| matches!(r, Redirect::StdoutTo(_)))
2053        {
2054            let arg_refs = args.iter().map(String::as_str).collect::<Vec<_>>();
2055            let bytes = crate::commands::cmd_echo_to_vec(&arg_refs);
2056            let path = match expand_word(path) {
2057                Ok(path) => path,
2058                Err(failed) => return failed.abort(),
2059            };
2060            let code = if crate::cmd_fs::write_file(&path, &bytes) {
2061                0
2062            } else {
2063                shell_stderr("echo: cannot write '");
2064                shell_stderr(&path);
2065                shell_stderrln("'");
2066                1
2067            };
2068            state::set_status(code);
2069            return code;
2070        }
2071        if let Some(Redirect::StdoutAppend(path)) = cmd
2072            .redirects
2073            .iter()
2074            .find(|r| matches!(r, Redirect::StdoutAppend(_)))
2075        {
2076            let arg_refs = args.iter().map(String::as_str).collect::<Vec<_>>();
2077            let bytes = crate::commands::cmd_echo_to_vec(&arg_refs);
2078            let path = match expand_word(path) {
2079                Ok(path) => path,
2080                Err(failed) => return failed.abort(),
2081            };
2082            let code = if crate::cmd_fs::append_file(&path, &bytes) {
2083                0
2084            } else {
2085                shell_stderr("echo: cannot append '");
2086                shell_stderr(&path);
2087                shell_stderrln("'");
2088                1
2089            };
2090            state::set_status(code);
2091            return code;
2092        }
2093    }
2094
2095    // StdinFrom redirect: preload the file into a buffer and expose it via
2096    // shell_stdin() so built-ins (grep, wc, …) can read from it.
2097    let stdin_file_buf: Vec<u8>;
2098    let effective_stdin: &[u8] = if let Some(Redirect::StdinFrom(path)) = cmd
2099        .redirects
2100        .iter()
2101        .find(|r| matches!(r, Redirect::StdinFrom(_)))
2102    {
2103        let path = match expand_word(path) {
2104            Ok(path) => path,
2105            Err(failed) => return failed.abort(),
2106        };
2107        stdin_file_buf = match crate::cmd_fs::read_file_vfs_owned(&path, 4096) {
2108            Ok(bytes) => bytes,
2109            Err(_) => {
2110                shell_stderr("shell: cannot open '");
2111                shell_stderr(&path);
2112                shell_stderrln("'");
2113                return 1;
2114            }
2115        };
2116        &stdin_file_buf
2117    } else {
2118        _stdin
2119    };
2120
2121    // Detect stdout/stderr redirect for non-echo commands.
2122    //
2123    // ViCell has one output channel (serial console). `2>file` is therefore
2124    // semantically equivalent to `>file` — both capture `shell_print` output.
2125    // When both `>` and `2>` are present, `>` takes precedence; `2>` is a
2126    // documented no-op in that case (single-channel limitation).
2127    let stdout_redir = cmd
2128        .redirects
2129        .iter()
2130        .find_map(|redirect| match redirect {
2131            Redirect::StdoutTo(path) => Some((path, false)),
2132            Redirect::StdoutAppend(path) => Some((path, true)),
2133            _ => None,
2134        })
2135        .or_else(|| {
2136            cmd.redirects.iter().find_map(|redirect| match redirect {
2137                // Fallback: StderrTo reuses the stdout-capture path (one-channel shell).
2138                Redirect::StderrTo(path) => Some((path, false)),
2139                _ => None,
2140            })
2141        });
2142    let stdout_redir = match stdout_redir {
2143        Some((path, append)) => match expand_word(path) {
2144            Ok(path) => Some((path, append)),
2145            Err(failed) => return failed.abort(),
2146        },
2147        None => None,
2148    };
2149
2150    // Wire the pipe-fed stdin so pipe-aware built-ins can read it.
2151    state::set_stdin(effective_stdin);
2152
2153    let code = if let Some((path, append)) = stdout_redir {
2154        // Capture this command's output into a buffer, then write to VFS.
2155        let captured: Vec<u8>;
2156        let status;
2157        {
2158            let guard = CaptureGuard::new();
2159            status = dispatch_builtin(prog, &args, jobs);
2160            captured = guard.finish();
2161        } // capture popped here, before the VFS write
2162        let write_ok = crate::cmd_fs::vfs_write_chunked(&path, &captured, append);
2163        if status == 0 && !write_ok {
2164            1
2165        } else {
2166            status
2167        }
2168    } else {
2169        dispatch_builtin(prog, &args, jobs)
2170    };
2171
2172    // Clear pipe stdin; leave the capture stack alone (exec_cmd does not own it).
2173    state::clear_stdin();
2174
2175    state::set_status(code);
2176    code
2177}

2473/// `test` / `[` — evaluate a condition and return 0 (true) or 1 (false).
2474///
2475/// Supported forms:
2476/// - `-f path`   : path exists and is a regular file
2477/// - `-z str`    : string is empty
2478/// - `-n str`    : string is non-empty
2479/// - `a = b`     : string equality
2480/// - `a != b`    : string inequality
2481fn cmd_test(args: &[&str]) -> ViResult<()> {
2482    let ok = Ok(());
2483    let fail = Err(ViError::NotFound); // any non-Ok maps to exit code 1
2484    match args {
2485        ["-f", path] => {
2486            if matches!(crate::cmd_fs::stat_file_vfs(path), Some((_, false))) {
2487                ok
2488            } else {
2489                fail
2490            }
2491        }
2492        ["-d", path] => {
2493            if matches!(crate::cmd_fs::stat_file_vfs(path), Some((_, true))) {
2494                ok
2495            } else {
2496                fail
2497            }
2498        }
2499        ["-e", path] => {
2500            if crate::cmd_fs::stat_file_vfs(path).is_some() {
2501                ok
2502            } else {
2503                fail
2504            }
2505        }
2506        [s1, "-z"] | ["-z", s1] => {
2507            if s1.is_empty() {
2508                ok
2509            } else {
2510                fail
2511            }
2512        }
2513        [s1, "-n"] | ["-n", s1] => {
2514            if !s1.is_empty() {
2515                ok
2516            } else {
2517                fail
2518            }
2519        }
2520        _ => {
2521            // String comparison: `a = b` or `a != b`.
2522            // args may be ["a", "=", "b"] or ["a", "!=", "b"].
2523            if args.len() == 3 {
2524                let (a, op, b) = (args[0], args[1], args[2]);
2525                match op {
2526                    "=" | "==" => {
2527                        if a == b {
2528                            ok
2529                        } else {
2530                            fail
2531                        }
2532                    }
2533                    "!=" => {
2534                        if a != b {
2535                            ok
2536                        } else {
2537                            fail
2538                        }
2539                    }
2540                    _ => fail,
2541                }
2542            } else {
2543                fail
2544            }
2545        }
2546    }
2547}

2549/// `read [-s] [VAR]` uses the active terminal lease; legacy callers retain fd 0. `-s` reads
2550/// silently (echo-off lease, no history, text wiped) and is refused where echo cannot be suppressed.
2551fn cmd_read(args: &[&str]) -> ViResult<()> {
2552    let (silent, rest) = match args.first() {
2553        Some(&"-s") => (true, &args[1..]),
2554        _ => (false, args),
2555    };
2556    let var = rest.first().copied().unwrap_or("REPLY");
2557    let serial = SERIAL_SESSION.lock().is_some();
2558    if silent && !serial {
2559        shell_stderrln("read: -s requires a terminal session that can suppress echo");
2560        return Err(ViError::NotSupported);
2561    }
2562    let mut line = if serial {
2563        let read = if silent {
2564            terminal_read_secret()
2565        } else {
2566            // `read` never records history; the scratch history only exists to satisfy the editor.
2567            let mut scratch = Vec::new();
2568            let read = terminal_read_line(&mut scratch);
2569            for entry in scratch.iter_mut() {
2570                ostd::session::zeroize_string(entry);
2571            }
2572            read
2573        };
2574        match read {
2575            Ok(Some(line)) => line,
2576            Err(Status::Unsupported) if silent => {
2577                shell_stderrln("read: -s is unsupported on this console (echo cannot be suppressed)");
2578                return Err(ViError::NotSupported);
2579            }
2580            Ok(None) | Err(_) => return Err(ViError::IO),
2581        }
2582    } else {
2583        let mut line = String::new();
2584        ostd::io::stdin().read_line(&mut line)?;
2585        line
2586    };
2587    let value = line.trim_end_matches(&['\r', '\n'][..]);
2588    set_var(var, value);
2589    ostd::session::zeroize_string(&mut line);
2590    Ok(())
2591}

2623/// Attempt to spawn an external binary from `/bin/<prog>`.
2624///
2625/// Arguments are published via `sys_set_spawn_args` (a reserved state-stash
2626/// slot) for the spawned cell to read on startup — `sys_spawn_from_path` does
2627/// not yet carry argv on the new cell's stack. We always set the slot (empty
2628/// when there are no args) so the cell never reads a previous command's args.
2629fn spawn_external(prog: &str, args: &[String]) -> i32 {
2630    if SERIAL_SESSION.lock().is_some() {
2631        return spawn_serial_foreground(prog, args);
2632    }
2633    if !ostd::set_spawn_argv(args) {
2634        shell_println("shell: external argv exceeds 512-byte transport limit");
2635        return 1;
2636    }
2637
2638    let path = if prog.starts_with('/') {
2639        alloc::string::String::from(prog)
2640    } else {
2641        let mut p = alloc::string::String::from("/bin/");
2642        p.push_str(prog);
2643        p
2644    };
2645    // Try the kernel's own path route first: it is the only route that reaches a
2646    // cell whose reviewed launch edge carries authority (the VFS-loaded route
2647    // refuses a non-empty child ceiling), and for VIFS1-resident cells it is also
2648    // the first one that can succeed. A *rejected* attempt discards the staged
2649    // command line by design, and the fallback below may need a second attempt, so
2650    // the argv is published again in between — `args` is still owned here, which is
2651    // the only place that can do it.
2652    if !ostd::set_spawn_argv(args) {
2653        shell_println("shell: external argv exceeds 512-byte transport limit");
2654        return 1;
2655    }
2656    let mut spawned = syscall::sys_spawn_from_path_raw(&path);
2657    if matches!(spawned, syscall::SyscallResult::Err(_)) {
2658        let _ = ostd::set_spawn_argv(args);
2659        spawned = syscall::sys_spawn_from_path(&path);
2660    }
2661    match spawned {
2662        syscall::SyscallResult::Ok(tid) => {
2663            // Backgrounded (`cmd &`): do NOT sys_wait. A long-running external
2664            // cell (httpd) would otherwise park the shell forever, so no later
2665            // command runs. The child keeps the shell's focus grant; that is
2666            // acceptable for a server that never reads the keyboard. Foreground
2667            // spawns fall through to the focus-handoff + sys_wait below.
2668            if state::bg_spawn() {
2669                return 0;
2670            }
2671            // Drain any pending input-service IPC events before ClearFocus.
2672            // Shell reads UART via sys_read(0), so Enter arrives via BOTH the ring
2673            // buffer (consumed in read_line) AND input-service EV_ASCII IPC. The IPC
2674            // path blocks input service in sys_send(shell, Enter_event) while shell
2675            // is not in sys_recv. drain_pending_input_events() uses sys_try_recv with
2676            // mask=input_tid (the correct G18 wildcard) to drain the queued message,
2677            // unblocking input service so release_focus does not deadlock.
2678            ostd::input::drain_pending_input_events();
2679            // Release keyboard focus before blocking in sys_wait. While the shell
2680            // is in sys_wait it is NOT in sys_recv, so the input service would
2681            // block indefinitely trying to dispatch key events to this cell and
2682            // eventually be killed by its own heartbeat watchdog (G18 deadlock).
2683            ostd::input::release_focus();
2684            // Foreground: block until the child exits so it owns the console
2685            // (stdin/UART). Without this the shell loops back to read the next
2686            // line and races interactive children (e.g. `hypha`) for keystrokes.
2687            // Fast commands return immediately (kernel Wait short-circuits when
2688            // the child is already Terminated). Background (`&`) already runs
2689            // synchronously in G1, so this does not regress it.
2690            let code = match syscall::sys_wait(tid) {
2691                syscall::SyscallResult::Ok(code) => code as i32,
2692                syscall::SyscallResult::Err(_) => 0,
2693            };
2694            // Re-acquire focus for the next interactive prompt.
2695            for _ in 0..10 {
2696                if ostd::input::request_focus() {
2697                    break;
2698                }
2699            }
2700            code
2701        }
2702        syscall::SyscallResult::Err(_) => {
2703            if prog != "vm-runner" && !prog.starts_with('/') {
2704                let app_toml = alloc::format!("/apps/{}/{}.toml", prog, prog);
2705                let app_lua = alloc::format!("/apps/{}/{}.lua", prog, prog);
2706                let has_app = match syscall::sys_open_cap(&app_toml) {
2707                    Ok(cap) => {
2708                        syscall::sys_close_cap(cap);
2709                        true
2710                    }
2711                    Err(_) => match syscall::sys_open_cap(&app_lua) {
2712                        Ok(cap) => {
2713                            syscall::sys_close_cap(cap);
2714                            true
2715                        }
2716                        Err(_) => false,
2717                    },
2718                };
2719
2720                if has_app {
2721                    let mut runner_args = alloc::vec![alloc::string::String::from(prog)];
2722                    runner_args.extend_from_slice(args);
2723                    return spawn_external("vm-runner", &runner_args);
2724                }
2725
2726                // Check for direct native user app binary at /apps/<prog>/<prog>
2727                let direct_app = alloc::format!("/apps/{}/{}", prog, prog);
2728                if let Ok(cap) = syscall::sys_open_cap(&direct_app) {
2729                    syscall::sys_close_cap(cap);
2730                    return spawn_external(&direct_app, args);
2731                }
2732            }
2733            shell_stderr("shell: command not found: ");
2734            shell_stderrln(prog);
2735            127
2736        }
2737    }
2738}

2924fn read_path_bytes(path: &str) -> Result<Vec<u8>, UtilityReadError> {
2925    if let Some((size, is_dir)) = crate::cmd_fs::stat_file_vfs(path) {
2926        if is_dir {
2927            return Err(UtilityReadError::Io);
2928        }
2929        if size > MAX_INPUT_BYTES {
2930            return Err(UtilityReadError::InputTooLarge);
2931        }
2932        let mut bytes = Vec::new();
2933        bytes
2934            .try_reserve_exact(size)
2935            .map_err(|_| UtilityReadError::AllocationFailed)?;
2936        bytes.resize(size, 0);
2937        if size == 0 {
2938            return Ok(bytes);
2939        }
2940        if crate::cmd_fs::read_file_vfs_known_size(path, size, &mut bytes) == Ok(size) {
2941            return Ok(bytes);
2942        }
2943    }
2944
2945    if !crate::cmd_fs::bootfs_fallback_allowed(path) {
2946        return Err(UtilityReadError::Io);
2947    }
2948    let fd = syscall::sys_open(path).map_err(|_| UtilityReadError::Io)?;
2949    let mut bytes = Vec::new();
2950    let mut chunk = [0u8; 512];
2951    loop {
2952        match syscall::sys_read(fd, &mut chunk) {
2953            Ok(0) => break,
2954            Ok(n) => {
2955                if let Err(err) = extend_input(&mut bytes, &chunk[..n]) {
2956                    syscall::sys_close(fd);
2957                    return Err(match err {
2958                        InputBufferError::TooLarge => UtilityReadError::InputTooLarge,
2959                        InputBufferError::AllocationFailed => UtilityReadError::AllocationFailed,
2960                    });
2961                }
2962            }
2963            Err(_) => {
2964                syscall::sys_close(fd);
2965                return Err(UtilityReadError::Io);
2966            }
2967        }
2968    }
2969    syscall::sys_close(fd);
2970    Ok(bytes)
2971}

3703fn run_foreground(prog: &str, args: &[String], completion: bool) -> (i32, Option<SshCompletion>) {
3704    if state::bg_spawn() {
3705        shell_stderrln("shell: SDK background launch requires explicit nonterminal streams");
3706        return (2, None);
3707    }
3708    let path = external_path(prog);
3709    let cwd = match crate::cmd_cwd::get_shell_cwd() {
3710        Ok(cwd) => cwd,
3711        Err(_) => {
3712            shell_stderrln("shell: cannot obtain current directory");
3713            return (1, None);
3714        }
3715    };
3716    let argv: Vec<&str> = args.iter().map(String::as_str).collect();
3717    let mut slot = SERIAL_SESSION.lock();
3718    let session = match slot.as_mut() {
3719        Some(session) => session,
3720        None => return (125, None),
3721    };
3722    let spawned = if completion {
3723        session.spawn_foreground_with_completion(&path, &argv, cwd.as_bytes(), &[], None)
3724    } else {
3725        session.spawn_foreground(&path, &argv, cwd.as_bytes(), &[], None)
3726    };
3727    let mut child = match spawned {
3728        Ok(child) => child,
3729        Err(status) => {
3730            drop(slot);
3731            shell_stderrln(&alloc::format!("shell: foreground preparation failed: {:?}", status));
3732            return (if status == Status::NotFound { 127 } else { 126 }, None);
3733        }
3734    };
3735    let tid = child.child().root_tid as usize;
3736    // Screen modes the child's relayed output switches (`screen_reset`), from its start.
3737    let mut screen = RelayScreen::new();
3738    let activation_failed = if let Some(status) = child.activation_error() {
3739        // Publication already happened: retain the real handle, cancel it and
3740        // complete owner-proven restoration/reaping instead of retrying spawn.
3741        let _ = session.cancel_foreground(&child, 2);
3742        let diagnostic = alloc::format!(
3743            "shell: foreground activation failed: {:?}\n", status
3744        );
3745        let mut offset = 0;
3746        let sink = sink_for(true, streams_merged());
3747        while offset < diagnostic.len() {
3748            match write_sink(session, sink, &diagnostic.as_bytes()[offset..]) {
3749                Ok(count) if count > 0 && count <= diagnostic.len() - offset => {
3750                    offset += count;
3751                }
3752                Err(Status::WouldBlock) => {}
3753                _ => {
3754                    let _ = syscall::sys_log("shell: activation diagnostic delivery failed\n");
3755                    break;
3756                }
3757            }
3758            if relay_serial(session, Some(&mut screen)).is_err() {
3759                break;
3760            }
3761        }
3762        true
3763    } else {
3764        false
3765    };
3766    // Input-owner trust prompt (D79-13): drawn here, the session's only output path.
3767    let mut prompt = OwnerPrompt::default();
3768    loop {
3769        // A failed prompt handshake (`poll` gives up once) leaves input withholding this child's
3770        // input: the child is cancelled, retried every turn until the cancellation is in or
3771        // restoration is observed below, which ends the quarantine and the confirmation.
3772        prompt.poll(session);
3773        prompt.cancel_turn(|| session.cancel_foreground(&child, 2));
3774        if let Err(status) = prompt.relay(session, &mut screen) {
3775            let _ = session.cancel_foreground(&child, 2);
3776            let _ = syscall::sys_wait(tid);
3777            let _ = syscall::sys_log(&alloc::format!(
3778                "shell: foreground output failed: {:?}\n", status
3779            ));
3780            syscall::sys_exit(125);
3781        }
3782        match session.finish_foreground(&mut child) {
3783            Ok(()) => { prompt.restored(); break; }
3784            Err(Status::WouldBlock | Status::TimedOut) => ostd::task::yield_now(),
3785            Err(status) => {
3786                let _ = session.cancel_foreground(&child, 2);
3787                let _ = syscall::sys_wait(tid);
3788                let _ = syscall::sys_log(&alloc::format!(
3789                    "shell: foreground owner restoration failed: {:?}\n", status
3790                ));
3791                // Never resume a prompt without authenticated restoration.
3792                syscall::sys_exit(125);
3793            }
3794        }
3795    }
3796    // The kernel's terminal record of the child (its preparer's QueryChild), not a TID
3797    // wait: the child may already be fully reaped once restoration is proven, and a
3798    // wait on a reaped TID cannot report its status. A child the input owner stopped
3799    // (Ctrl-C cancels its operation) reports 130.
3800    let code = if activation_failed {
3801        125
3802    } else {
3803        match child.wait_status() {
3804            Ok(record) => typed::reaped_status(Some(record.exit_code), record.reason, false),
3805            Err(_) => 125,
3806        }
3807    };
3808    // The child is reaped: relay what it left (bounded) so the tracker sees all of it,
3809    // then the soft reset iff it left a tracked mode on.
3810    for _ in 0..8 {
3811        match relay_serial(session, Some(&mut screen)) {
3812            Ok(true) => {}
3813            Ok(false) => break,
3814            Err(_) => {
3815                syscall::sys_exit(125);
3816            }
3817        }
3818    }
3819    if screen.needs_reset() {
3820        restore_screen_locked(session);
3821    }
3822    // The ssh Cell's lifecycle record: whatever it wrote before it ended (read before the
3823    // operation is closed; a torn trailing frame is ignored).
3824    let record = completion.then(|| {
3825        let mut frames = [0u8; 4096];
3826        let n = child.read_completion(&mut frames).unwrap_or(0);
3827        api::services::ssh_completion::last_frame(&frames[..n])
3828    }).flatten();
3829    // Reaped and restored: release the child operation so the session's 32-child
3830    // quota is reusable. Failure is diagnostic only; the command already ran.
3831    if let Err(status) = session.close_foreground(&child) {
3832        let _ = syscall::sys_log(&alloc::format!("shell: foreground operation close failed: {:?}\n", status));
3833    }
3834    (code, record)
3835}

4008            None => return Err(setup_failure(Status::Revoked)),

4011        self.errors = Some(error_reader);

4013        let (output_reader, output_writer) = self.stream().map_err(setup_failure)?;
4014        self.output = Some(output_reader);
4015        self.held.push(output_writer);
4016        let mut first_stdin = None;
4017        if with_input {

4019            // reading, the kernel stops only that stage.

4040                    Some(pipeline) => pipeline.link(index - 1),

4082                Err(status) if typed::spawn_cancel_means_consumer_finished(status, index, commands.len()) => {

4386    /// Release the run and return one status per stage. A run whose output ended
4387    /// normally is awaited; anything else (consumer closed early, interrupt,
4388    /// failure) is cancelled first so the reap cannot block on a live stage.
4389    fn conclude(&mut self) -> Vec<i32> {

4391            let clean = self.output_eof && !self.interrupted && !self.cancelled;
4392            if clean && self.settle().is_ok() {
4393                self.reap(false);
4394            } else {
4395                self.reap(true);
4396            }
4397        }
4398        self.statuses.clone()
4399    }

4401    /// Close every shell-held endpoint, reap every stage (terminal record of its
4402    /// root task via `QueryChild`) and release the pipeline Operation. Idempotent.
4403    fn reap(&mut self, cancel: bool) {
4404        if self.reaped {
4405            return;
4406        }
4407        self.reaped = true;
4408        if cancel {
4409            self.cancelled = true;
4410            if let Some(pipeline) = &self.pipeline {
4411                let _ = pipeline.cancel(EXT_CANCEL_STOP);
4412            }
4413            // Diagnostics a stage wrote before it was stopped are still delivered.
4414            self.drain_errors();
4415        }
4416        self.close_input();
4417        self.close_output();
4418        self.close_errors();
4419        if let Some(writer) = self.error_writer.take() {
4420            let _ = writer.close();
4421        }
4422        for endpoint in core::mem::take(&mut self.held) {
4423            let _ = endpoint.close();
4424        }
4425        for (index, stage) in self.stages.iter_mut().enumerate() {
4426            let status = match stage.child.take() {
4427                Some(child) => {
4428                    let mut waited = child.wait_status();
4429                    if waited == Err(Status::TimedOut) && !self.cancelled {
4430                        // Never block forever on a stage that closed its streams but
4431                        // keeps running: stop the pipeline, then ask once more.
4432                        self.cancelled = true;
4433                        if let Some(pipeline) = &self.pipeline {
4434                            let _ = pipeline.cancel(EXT_CANCEL_STOP);
4435                        }
4436                        waited = child.wait_status();
4437                    }
4438                    let code = match waited {
4439                        Ok(record) => typed::reaped_status(Some(record.exit_code), record.reason, self.cancelled),
4440                        Err(status) => {
4441                            let _ = syscall::sys_log(&alloc::format!(
4442                                "shell: {}: exit status unavailable: {:?}\n", stage.name, status
4443                            ));
4444                            typed::reaped_status(None, 0, self.cancelled)
4445                        }
4446                    };
4447                    if stage.ssh {
4448                        // The ssh Cell's lifecycle record: whatever it wrote before it ended
4449                        // (read before the pipeline operation is closed; a torn trailing
4450                        // frame is ignored). The newest ssh stage's record is the shell's.
4451                        let mut frames = [0u8; 4096];
4452                        let n = child.read_completion(&mut frames).unwrap_or(0);
4453                        let record = api::services::ssh_completion::last_frame(&frames[..n]);
4454                        set_ssh_completion(crate::ssh::settle(record, code, api::services::ssh_completion::Mode::Exec));
4455                    }
4456                    code
4457                }
4458                None if index < self.not_started => typed::UPSTREAM_NOT_STARTED_STATUS,
4459                None => 0,
4460            };
4461            self.statuses.push(status);
4462        }
4463        if let Some(pipeline) = self.pipeline.take() {
4464            if let Err(status) = pipeline.close() {
4465                let _ = syscall::sys_log(&alloc::format!(
4466                    "shell: pipeline operation close failed: {:?}\n", status
4467                ));
4468            }
4469        }
4470        // Every stage is gone and its last bytes were relayed: the soft reset, once,
4471        // iff the relayed output left a tracked screen mode on.
4472        if core::mem::take(&mut self.screen).needs_reset() {
4473            restore_screen();
4474        }
4475    }

4515/// Feed `bytes` to the segment's first stage and deliver the last stage's stdout.
4516/// With a redirect the bytes are collected (bounded) and returned for the caller to
4517/// write after success; otherwise they reach the terminal as they arrive (byte
4518/// exact), or the current capture when one besides the REPL's own is active.
4519fn drive_external_sink(
4520    run: &mut ExtRun,
4521    bytes: Vec<u8>,
4522    redirect: Option<&(String, bool)>,
4523) -> Result<Vec<u8>, StreamStop> {
4524    run.set_feed(bytes);
4525    let redirected = redirect.is_some();
4526    let live = !redirected && serial_active() && state::capture_depth() <= 1;
4527    let mut collected = Vec::new();
4528    let mut total = 0usize;
4529    let mut chunk = ostd::session::Scrub::<512>::new();
4530    while let Some(count) = run.read_output(&mut chunk)? {
4531        if live {
4532            run.forward_terminal(&chunk[..count], false)?;
4533            continue;
4534        }
4535        total += count;
4536        if total > typed::MAX_SINK_BYTES {
4537            return Err(StreamStop::Failed(alloc::format!(
4538                "output exceeds the {} byte bound",
4539                typed::MAX_SINK_BYTES
4540            )));
4541        }
4542        if redirected {
4543            collected.extend_from_slice(&chunk[..count]);
4544        } else {
4545            state::write_out_bytes(&chunk[..count]);
4546        }
4547    }
4548    Ok(collected)
4549}

4551/// Reap the segment, record member statuses (stage `base + i`) and fold the pump
4552/// outcome into the pipeline status: 130 after Ctrl-C, 1 after a pump failure,
4553/// otherwise the last stage's status. Earlier nonzero statuses go to stderr; a
4554/// redirect target is written only when the last stage succeeded.
4555fn settle_external_segment(
4556    mut run: ExtRun,
4557    driven: Result<Vec<u8>, StreamStop>,
4558    members: &mut [(String, i32)],
4559    base: usize,
4560    redirect: Option<&(String, bool)>,
4561) -> i32 {
4562    let statuses = run.conclude();
4563    for (offset, status) in statuses.iter().enumerate() {
4564        if let Some(member) = members.get_mut(base + offset) {
4565            member.1 = *status;
4566        }
4567    }
4568    let cancelled = run.was_cancelled();
4569    let interrupted = run.interrupted();
4570    let not_started = run.not_started();
4571    drop(run);
4572    let collected = match driven {
4573        Ok(bytes) if !interrupted => bytes,
4574        Ok(_) | Err(StreamStop::Interrupted) => {
4575            shell_console("^C\n");
4576            return 130;
4577        }
4578        Err(StreamStop::Failed(message)) => {
4579            shell_stderrln(&alloc::format!("shell: {}", message));
4580            return fail_member(members, usize::MAX, 1);
4581        }
4582    };
4583    let names: Vec<&str> = members
4584        .iter()
4585        .skip(base)
4586        .take(statuses.len())
4587        .map(|member| member.0.as_str())
4588        .collect();
4589    // Stages that never started ran nothing: no exit to report for them.
4590    let started = not_started.min(statuses.len()).min(names.len());
4591    for line in typed::earlier_failures(&names[started..], &statuses[started..], cancelled) {
4592        shell_stderrln(&alloc::format!("shell: {}", line));
4593    }
4594    let last = statuses.last().copied().unwrap_or(0);
4595    if last == 0 {
4596        if let Some((path, append)) = redirect {
4597            if !crate::cmd_fs::vfs_write_chunked(path, &collected, *append) {
4598                shell_stderr("shell: cannot write '");
4599                shell_stderr(path);
4600                shell_stderrln("'");
4601                return fail_member(members, usize::MAX, 1);
4602            }
4603        }
4604    }
4605    last
4606}
