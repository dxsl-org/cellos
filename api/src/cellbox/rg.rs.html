1//! `rg`: ripgrep-derived search over the Cellos VFS as a typed row producer
2//! (Phase 07 S7-3, `phase-07-s3-proposal.md`).
3//!
4//! * [`args`] parses the frozen argv surface at admission (status 2 before any
5//!   effect);
6//! * [`producer`] / [`stdin_producer`] compile the matcher and globs and return a
7//!   pull-based [`RowSource`] of `cellbox.search.line` records. Nothing touches
8//!   the VFS until the first pull;
9//! * `tree` adapts `ostd::clients::VfsClient` to `vigrep::walk::ViSearchTree`
10//!   (per-directory reopen with one cached parent handle);
11//! * `source` drives `vigrep`'s resumable file search one bounded segment per
12//!   pull (8 rows / 32 KiB queued at most), reports per-entry diagnostics and
13//!   one end summary, and exposes the 0/1/2 completion status;
14//! * [`text`] is the streaming plain-text renderer with terminal-safe escaping.
15//!
16//! Memory: [`RG_PROFILE`] tightens the contract ceilings so a search fits the
17//! cellbox arena next to the shell (DV7-10, D76-3 M2); the bounds are measured by the
18//! host heap test in `tests.rs`.
19

43/// Search resource bounds (DV7-10). Contract numbers are ceilings; these are the
44/// values that fit the cellbox arena (proposal §6, D76-3 M2).
45#[derive(Debug, Clone, Copy)]
46pub struct RgProfile {
47    /// Compiled regex (NFA) bound.
48    pub regex_size_limit: usize,
49    /// Lazy-DFA cache bound.
50    pub dfa_size_limit: usize,
51    /// Searcher read buffer.
52    pub buffer_capacity: usize,
53    /// Searcher buffer growth bound: capacity plus the 65,536-byte line limit.
54    pub heap_limit: usize,
55    /// Row queue and submatch bounds.
56    pub limits: Limits,
57    /// Resume window of a streamed stdin (an external head cannot rewind).
58    pub stdin_window: usize,
59    /// Walker bounds.
60    pub walk: WalkLimits,
61}

63/// The profile every `rg` uses.
64pub const RG_PROFILE: RgProfile = RgProfile {
65    regex_size_limit: 128 * 1024,
66    dfa_size_limit: 64 * 1024,
67    buffer_capacity: 16 * 1024,
68    heap_limit: 16 * 1024 + 65_536,
69    limits: Limits { queue_rows: 8, queue_bytes: 32 * 1024, max_submatches: 4096 },
70    stdin_window: 96 * 1024,
71    walk: WalkLimits {
72        max_depth: 64,
73        max_entries: 65_536,
74        max_dir_entries: 4_096,
75        max_pending_bytes: 96 * 1024,
76        max_pending_entries: 8_192,
77        max_path_len: 1_024,
78        max_ignore_file_bytes: 16 * 1024,
79        max_ignore_rules_per_file: 256,
80        max_ignore_bytes_total: 32 * 1024,
81        max_ignore_rules_total: 65_536,
82        scratch_bytes: 16 * 1024,
83        resume_window_bytes: 32 * 1024,
84    },
85};

107/// Matcher for `args` with the profile's size limits, configured as upstream rg
108/// configures its Rust regex engine (`hiargs.rs` `matcher_rust`): multi-line
109/// syntax so `^`/`$` see line boundaries, LF as the banned line terminator, NUL
110/// banned unless `-a`, patterns joined leftmost-first in argument order.
111pub(crate) fn build_matcher(args: &RgArgs, profile: &RgProfile) -> Result<RegexMatcher, String> {
112    let mut builder = RegexMatcherBuilder::new();
113    builder
114        .multi_line(true)
115        .unicode(true)
116        .octal(false)
117        .dot_matches_new_line(false)
118        .line_terminator(Some(b'\n'))
119        .case_insensitive(args.ignore_case)
120        .fixed_strings(args.fixed)
121        .size_limit(profile.regex_size_limit)
122        .dfa_size_limit(profile.dfa_size_limit);
123    if args.word {
124        builder.word(true);
125    }
126    if args.line {
127        builder.whole_line(true);
128    }
129    if !args.search_binary {
130        builder.ban_byte(Some(0));
131    }
132    builder.build_many(&args.patterns).map_err(|error| format!("regex: {}", error))
133}
134
135pub(crate) fn build_searcher(args: &RgArgs, profile: &RgProfile) -> Searcher {
136    SearcherBuilder::new()
137        .line_number(true)
138        .before_context(args.before)
139        .after_context(args.after)
140        .buffer_capacity(profile.buffer_capacity)
141        .heap_limit(Some(profile.heap_limit))
142        .build()
143}

145/// The walker over `roots` (canonical absolute paths); `cwd` anchors `-g` globs.
146pub(crate) fn build_walk(
147    args: &RgArgs,
148    roots: &[Vec<u8>],
149    cwd: &[u8],
150    profile: &RgProfile,
151) -> Result<WalkBuilder, String> {
152    let mut overrides = OverrideBuilder::new(cwd);
153    for glob in &args.globs {
154        overrides.add(glob).map_err(|error| format!("glob: {}", error))?;
155    }
156    let overrides = overrides.build().map_err(|error| format!("glob: {}", error))?;
157    let mut walk = WalkBuilder::new(roots);
158    walk.hidden(args.hidden)
159        .ignore_files(!args.no_ignore)
160        .overrides(overrides)
161        .binary(if args.search_binary { BinaryMode::Search } else { BinaryMode::Skip })
162        .limits(profile.walk);
163    Ok(walk)
164}

166/// Search roots: every path resolved against the shell cwd, or the cwd itself
167/// with no path. A cwd that cannot be read fails the search; `/` is never used
168/// as a stand-in. Also returns the glob base (the cwd, `/` only when no glob and
169/// no relative path needs it).
170fn roots(args: &RgArgs) -> Result<(Vec<Vec<u8>>, Vec<u8>), String> {
171    let needs_cwd = args.paths.is_empty()
172        || !args.globs.is_empty()
173        || args.paths.iter().any(|path| !path.starts_with('/'));
174    let cwd = if needs_cwd {
175        let cwd = crate::cmd_cwd::get_shell_cwd()
176            .map_err(|_| String::from("cannot read the current directory"))?;
177        crate::cmd_fs::resolve_path_from(&cwd, "")
178    } else {
179        String::from("/")
180    };
181    let roots = if args.paths.is_empty() {
182        alloc::vec![cwd.clone().into_bytes()]
183    } else {
184        args.paths.iter().map(|path| crate::cmd_fs::resolve_path_from(&cwd, path).into_bytes()).collect()
185    };
186    Ok((roots, cwd.into_bytes()))
187}

189/// Tree-mode producer for parsed `args` (`rg PATTERN [PATH…]`). Compile errors
190/// print one `rg: regex: …`/`rg: glob: …` line and return status 2 before any
191/// VFS request. `-h` prints the usage and yields nothing (completion 0).
192pub fn producer(args: &RgArgs, latch: &Rc<InterruptLatch>) -> Result<Box<dyn RowSource>, i32> {
193    if args.help {
194        crate::executor::shell_println(args::USAGE);
195        return Ok(Box::new(source::Empty));
196    }
197    let profile = &RG_PROFILE;
198    let matcher = build_matcher(args, profile).map_err(|message| usage_error(&message))?;
199    let (roots, cwd) = roots(args).map_err(|message| usage_error(&message))?;
200    let walk = build_walk(args, &roots, &cwd, profile).map_err(|message| usage_error(&message))?;
201    let searcher = build_searcher(args, profile);
202    Ok(Box::new(source::TreeSource::new(
203        tree::VfsTree::new(),
204        walk.build(),
205        searcher,
206        matcher,
207        args.quiet,
208        profile.limits,
209        latch.clone(),
210        stderr_report(),
211    )))
212}

214/// Stdin-mode producer (`… | rg -a PATTERN -`): one search over `input`.
215pub fn stdin_producer(
216    args: &RgArgs,
217    input: StdinInput,
218    latch: &Rc<InterruptLatch>,
219) -> Result<Box<dyn RowSource>, i32> {
220    let profile = &RG_PROFILE;
221    let matcher = match build_matcher(args, profile) {
222        Ok(matcher) => matcher,
223        Err(message) => {
224            if let StdinInput::Stream(mut stream) = input {
225                stream.close(false);
226            }
227            return Err(usage_error(&message));
228        }
229    };
230    let searcher = build_searcher(args, profile);
231    Ok(Box::new(source::StdinSource::new(
232        input,
233        searcher,
234        matcher,
235        args.quiet,
236        profile.limits,
237        profile.stdin_window,
238        latch.clone(),
239        stderr_report(),
240    )))
