1//! Test fixture launcher (Phase 13: the fixtures left init).
2//!
3//! Init launches `/bin/fixture-launcher` from the boot plan when, and only when,
4//! a test image packages it (the plan entry is selected by artifact). It starts
5//! the fixtures init's boot code used to start, in the same order, on its own
6//! reviewed `fixture-launcher` edges with the same child ceilings, then exits.
7//! Every spawn tries the kernel path route first (VIFS1 and the P2 table), so a
8//! boot-resident child is named by its signed image exactly as it was when
9//! trusted init launched it; a cell only the VFS can read (the cell store of an
10//! RV64 disk) is then read through the VFS and spawned from those bytes, the
11//! route init used for every fixture.
12//!
13//! Cargo features select the opt-in sets; without any, it still starts the five
14//! boot tests (`/bin/silo-test`, `/bin/ai-test`, `/bin/vfs-test`, `/bin/srv-test`,
15//! `/bin/std-smoke`), each absent one failing with `NotFound` as before.
16

53/// The Phase-02 cross-tier fixture's service id (`tier2-rpc-proto::SERVICE_ID`).
54/// A literal rather than a dependency on the fixture crate; a drift is caught by
55/// the driver's first resolution, which fails loudly.
56#[cfg(feature = "tier2-rpc-entry")]
57const TIER2_RPC_FIXTURE_SERVICE: u16 = 0x7A01;
58
59/// The DMA-export lane's holder service (`edu-export-proto::SERVICE_ID`, admitted
60/// by `test-hooks` kernels as `dma_export::TEST_EXPORT_SERVICE`).
61#[cfg(feature = "edu-export-entry")]
62const EDU_EXPORT_HOLDER_SERVICE: u16 = 0x7AE0;

64/// Spawn `path` with the command line `args` (`None`: none). A refused attempt
65/// consumes the staged command line, so it is republished before the VFS route.
66fn spawn_with(path: &str, args: Option<&str>) -> Option<usize> {
67    let stage = || args.is_none_or(ostd::syscall::sys_set_spawn_args);
68    if !stage() {
69        return None;
70    }
71    if let SyscallResult::Ok(tid) = sys_spawn_from_path_raw(path) {
72        return (tid != 0).then_some(tid);
73    }
74    if !stage() {
75        return None;
76    }
77    match sys_spawn_from_path(path) {
78        SyscallResult::Ok(tid) if tid != 0 => Some(tid),
79        _ => None,
80    }
81}
82
83fn spawn(path: &str) -> Option<usize> {
84    spawn_with(path, None)
85}

87fn cell_main() {
88    // Tier-2 entry fixtures (phase 02), first in the order, so the AArch64
89    // test-hooks image schedules them before its test root ends the boot.
90    #[cfg(feature = "tier2-entry")]
91    for (path, admitted, failed) in [
92        (
93            "/bin/tier2-smoke",
94            "fixture-launcher: tier2-smoke admitted.",
95            "fixture-launcher: tier2-smoke spawn failed.",
96        ),
97        (
98            "/bin/tier2-exploit",
99            "fixture-launcher: tier2-exploit admitted.",
100            "fixture-launcher: tier2-exploit spawn failed.",
101        ),
102        (
103            "/bin/cpp-smoke",
104            "fixture-launcher: cpp-smoke admitted.",
105            "fixture-launcher: cpp-smoke spawn failed.",
106        ),
107    ] {
108        ostd::io::println(if spawn(path).is_some() { admitted } else { failed });
109    }
110
111    // Phase-02 slice B cross-tier exchange. A private-root provider holds no
112    // capability and cannot register itself, so its spawner registers it; the
113    // kernel captures the provider's identity in the same lock hold as the
114    // liveness check, and a private id is this spawner's to publish.
115    #[cfg(feature = "tier2-rpc-entry")]
116    match spawn("/bin/tier2-rpc-provider") {
117        Some(provider) => {
118            ostd::io::println("fixture-launcher: tier2-rpc-provider admitted.");
119            ostd::io::println(
120                match ostd::syscall::sys_register_service(TIER2_RPC_FIXTURE_SERVICE, provider, 0) {
121                    SyscallResult::Ok(_) => "fixture-launcher: tier2-rpc-provider registered.",
122                    SyscallResult::Err(_) => {
123                        "fixture-launcher: tier2-rpc-provider registration failed."
124                    }
125                },
126            );
127            ostd::io::println(match spawn("/bin/tier2-rpc-driver") {
128                Some(_) => "fixture-launcher: tier2-rpc-driver launched.",
129                None => "fixture-launcher: tier2-rpc-driver spawn failed.",
130            });
131        }
132        None => ostd::io::println("fixture-launcher: tier2-rpc-provider spawn failed."),
133    }
134
135    // ADR-0033 CP1 §7 DMA-export lane: the holder is registered before a client
136    // resolves it.
137    #[cfg(feature = "edu-export-entry")]
138    match spawn("/bin/edu-holder") {
139        Some(holder) => {
140            ostd::io::println(
141                match ostd::syscall::sys_register_service(EDU_EXPORT_HOLDER_SERVICE, holder, 0) {
142                    SyscallResult::Ok(_) => "fixture-launcher: edu-holder registered.",
143                    SyscallResult::Err(_) => "fixture-launcher: edu-holder registration failed.",
144                },
145            );
146            for _ in 0..2 {
147                ostd::io::println(match spawn("/bin/export-client") {
148                    Some(_) => "fixture-launcher: export-client launched.",
149                    None => "fixture-launcher: export-client spawn failed.",
150                });
151            }
152        }
153        None => ostd::io::println("fixture-launcher: edu-holder spawn failed."),
154    }
155
156    // Phase-03 step-5 pair, before the boot tests below so a generation's
157    // terminal fault never races the image's own test root.
158    #[cfg(feature = "tier2-grant-pair")]
159    grant_pair::run();
160
161    for path in [
162        "/bin/silo-test",
163        // Spec 24 G2 Level A oracle: exercises the inference service over typed IPC.
164        "/bin/ai-test",
165        "/bin/vfs-test",
166        "/bin/srv-test",
167        "/bin/std-smoke",
168    ] {
169        let _ = spawn(path);
170    }
171}

173/// The phase-03 Tier-2 grant pair, from the boot order.
174///
175/// No lane can type the pair at a guest shell: the shell refuses to background
176/// an external cell inside a serial session, and the AArch64 test image has no
177/// interactive window at all. The AArch64 test-hooks, RV64 native-domain and
178/// x86_64 domain-test images therefore launch it from the boot order. Both
179/// cells run on the reviewed `/bin/silo` and `/bin/net-broker` paths, whose
180/// real services those images do not build. Every marker the pair emits is
181/// namespaced `S22-<arch>-GRANT-PAIR-*`.
182#[cfg(feature = "tier2-grant-pair")]
183mod grant_pair {
184    use super::spawn_with;
185    use ostd::io::println;
186
187    const OWNER_PATH: &str = "/bin/silo";
188    const RECEIVER_PATH: &str = "/bin/net-broker";
189    /// One receiver generation per property the phase must witness, in the order
190    /// the owner serves them: ReadWrite, ReadOnly, same-recipient downgrade,
191    /// GrantUnregister, owner exit.
192    const MODES: [&str; 5] = ["rw", "ro", "downgrade", "unregister", "exit"];
193    /// ~3 s of 10 ms scheduler ticks: two orders of magnitude above the ~200 ms
194    /// one generation costs, bounding a generation that neither faults nor exits.
195    const DEATH_TIMEOUT_TICKS: u64 = 300;
196    /// Backstop for a tick clock that never advances.
197    const DEATH_POLL_LIMIT: usize = 5_000_000;
198
199    fn task_name(path: &str) -> &str {
200        path.rsplit('/').next().unwrap_or(path)
201    }
202
203    fn now_ticks() -> u64 {
204        ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0)
205    }
206
207    /// Has `tid` terminated? A task the listing no longer carries has; a listing
208    /// that cannot be taken is not evidence of death.
209    fn terminated(tid: usize) -> bool {
210        let mut rows = [abi::syscall::ProcessInfo::default(); 64];
211        match ostd::syscall::sys_get_procs(&mut rows) {
212            Ok(count) => rows
213                .iter()
214                .take(count)
215                .all(|row| row.id != tid || row.state == 3),
216            Err(_) => false,
217        }
218    }
219
220    /// The owner starts once with the receiver's task name; each generation
221    /// starts with its mode and the owner's task name and must be gone before
222    /// the next starts (the phases share grant ids, and the fault address is
223    /// what attributes each fault to its phase). Grant ids travel over the
224    /// pair's own IPC, never on a command line.
225    pub(super) fn run() {
226        let owner_name = task_name(OWNER_PATH);
227        let receiver_name = task_name(RECEIVER_PATH);
228        if spawn_with(OWNER_PATH, Some(&alloc::format!("peer={receiver_name}"))).is_none() {
229            println("fixture-launcher: tier2-grant-pair owner spawn failed.");
230            return;
231        }
232        println("fixture-launcher: tier2-grant-pair owner admitted.");
233        for mode in MODES {
234            // Every launch republishes the command line: a spawn attempt consumes it.
235            let args = alloc::format!("{mode} peer={owner_name}");
236            let Some(tid) = spawn_with(RECEIVER_PATH, Some(&args)) else {
237                println(&alloc::format!(
238                    "fixture-launcher: tier2-grant-pair generation {mode} spawn failed."
239                ));
240                return;
241            };
242            let started = now_ticks();
243            let mut polls = 0usize;
244            while !terminated(tid) {
245                if now_ticks().wrapping_sub(started) >= DEATH_TIMEOUT_TICKS
246                    || polls >= DEATH_POLL_LIMIT
247                {
248                    println(&alloc::format!(
249                        "fixture-launcher: tier2-grant-pair generation {mode} did not terminate."
250                    ));
251                    return;
252                }
253                ostd::task::yield_now();
254                polls += 1;
255            }
256        }
257        println("fixture-launcher: tier2-grant-pair complete.");
258    }
259}
