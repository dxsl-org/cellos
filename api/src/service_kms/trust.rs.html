1//! SSH host trust (Phase 09 S9-3 lookup, S9-4c first trust in [`first`]; abi-vfs-telemetry
2//! §4), served by KMS since Phase 12 (CP1 W3-W5).
3//!
4//! The store is one file under the KMS namespace (`/srv/cellos/kms`, which the VFS admits
5//! only to the live KMS), seeded by the provisioner (D79-10). A lookup compares the
6//! caller's configured host identifier, port, algorithm and complete key blob with it and
7//! answers one [`wire::ipc::ssh_trust`] code; the store's contents never leave the
8//! service. Only an attested caller holding the outbound-connect right may ask. A query
9//! is staged in [`query`] (opcodes 25-27, 29) because host and key do not fit one frame.
10//!
11//! Store format v1 (UTF-8 text, LF line ends, no comments or blank lines):
12//!
13//! ```text
14//! cellos-ssh-trust v1
15//! HOST PORT ALGORITHM BASE64-KEY-BLOB PROVENANCE
16//! ```
17//!
18//! `HOST` is the canonical configured identifier (lowercase ASCII name or canonical IP
19//! literal), `PORT` decimal 1..=65535, `ALGORITHM` `ssh-ed25519`, the blob is the full
20//! SSH wire key (standard base64, padded) and `PROVENANCE` is `preprovisioned` or
21//! `interactive`. At most one line per (host, port); at most 1,024 entries, 4 KiB per
22//! blob and 1 MiB in total. Anything else makes the whole store unavailable, never a
23//! partial answer.

70/// The store as KMS reaches it. Every failure is `Err(())`: the caller answers
71/// Unavailable, never a guess.
72pub(crate) trait TrustIo {

79    /// Publish `bytes` as the store only if it still matches `expect`.

122    /// Handle one trust opcode (25-29). Admission first: an attested caller holding the
123    /// outbound-connect right; nothing is decoded before it passes.
124    pub(crate) fn handle(
125        &mut self,
126        opcode: KmsOpcode,
127        payload: &[u8],
128        sender: usize,
129        caller: Option<CallerIdentity>,
130    ) -> Result<SuccessPayload, KmsErrorCode> {
131        let caller = attested_caller(sender, caller)?;
132        if caller.flags & CALLER_FLAG_NET_CONNECT == 0 {
133            return Err(KmsErrorCode::PermissionDenied);
134        }
135        let principal = Principal {
136            cell_id: caller.cell_id,
137            generation: caller.generation,
138        };
139        match opcode {
140            KmsOpcode::BeginTrustQuery => {
141                let request = invalid(TrustQueryBeginRequestPayload::decode(payload))?;
142                let begin = self.queries.begin(principal, &request)?;
143                SuccessPayload::new(&begin.encode())
144            }
145            KmsOpcode::TrustQueryChunk => {
146                let chunk = invalid(SshChunkRequestPayload::decode(payload))?;
147                let ack = self.queries.chunk(principal, &chunk)?;
148                SuccessPayload::new(&ack.encode())
149            }
150            KmsOpcode::CommitTrustQuery => {
151                let request = invalid(SshImportCommitRequestPayload::decode(payload))?;
152                let staged = self.queries.commit(principal, &request)?;
153                let (host, algorithm, key) = staged.fields()?;
154                let outcome = match staged.kind {
155                    KmsTrustQueryKind::Lookup => TrustOutcomePayload::outcome(
156                        first::current(&mut self.io, host, staged.port, algorithm, key)
157                            .map_or(ssh_trust::UNAVAILABLE, |(_, _, code)| code),
158                    ),
159                    KmsTrustQueryKind::Challenge => first::challenge(
160                        self,
161                        principal,
162                        host,
163                        staged.port,
164                        algorithm,
165                        key,
166                        staged.session,
167                        staged.epoch,
168                    )?,
169                };
170                SuccessPayload::new(&outcome.encode())
171            }
172            KmsOpcode::CommitTrustChallenge => {
173                let request = invalid(TrustChallengeCommitRequestPayload::decode(payload))?;
174                let (id, generation) = request.challenge;
175                let code = first::commit(self, principal, Ref { id, generation });
176                SuccessPayload::new(&TrustOutcomePayload::outcome(code).encode())
177            }
178            KmsOpcode::AbortTrustQuery => {
179                let request = invalid(SshAbortRequestPayload::decode(payload))?;
180                self.queries.abort(principal, request.handle)?;
181                SuccessPayload::new(&[])
182            }
183            _ => Err(KmsErrorCode::UnknownOpcode),
184        }
185    }
186}

188/// The answer for (`host`, `port`, `algorithm`, `key`) against store `bytes`.
189pub(crate) fn lookup(bytes: &[u8], host: &str, port: u16, algorithm: &str, key: &[u8]) -> u8 {
190    let Ok(entries) = parse(bytes) else {
191        return ssh_trust::UNAVAILABLE;
192    };
193    match entries.iter().find(|e| e.host == host && e.port == port) {
194        None => ssh_trust::UNKNOWN,
195        Some(e) if e.algorithm == algorithm && e.blob == key => ssh_trust::KNOWN_EXACT,
196        Some(_) => ssh_trust::MISMATCH,
197    }
198}

200fn parse(bytes: &[u8]) -> Result<Vec<Entry<'_>>, ()> {
201    if bytes.len() > MAX_STORE {
202        return Err(());
203    }
204    let text = core::str::from_utf8(bytes).map_err(|_| ())?;
205    let body = text.strip_suffix('\n').ok_or(())?;
206    let mut lines = body.split('\n');
207    if lines.next() != Some(HEADER) {
208        return Err(());
209    }
210    let mut entries: Vec<Entry> = Vec::new();
211    for line in lines {
212        let mut f = line.split(' ');
213        let (Some(host), Some(port), Some(algorithm), Some(blob), Some(provenance), None) =
214            (f.next(), f.next(), f.next(), f.next(), f.next(), f.next())
215        else {
216            return Err(());
217        };
218        let port = parse_port(port)?;
219        if !canonical_host(host)
220            || algorithm != "ssh-ed25519"
221            || !matches!(provenance, "preprovisioned" | "interactive")
222        {
223            return Err(());
224        }
225        let blob = base64_decode(blob)?;
226        if blob.is_empty() || blob.len() > MAX_BLOB || !blob_names(&blob, algorithm) {
227            return Err(());
228        }
229        if entries.len() == MAX_ENTRIES || entries.iter().any(|e| e.host == host && e.port == port)
230        {
231            return Err(());
232        }
233        entries.push(Entry {
234            host,
235            port,
236            algorithm,
237            blob,
238        });
239    }
240    Ok(entries)
241}

243fn parse_port(s: &str) -> Result<u16, ()> {
244    if s.is_empty() || s.len() > 5 || s.starts_with('0') || !s.bytes().all(|b| b.is_ascii_digit()) {
245        return Err(());
246    }
247    match s.parse::<u32>() {
248        Ok(n) if (1..=65535).contains(&n) => Ok(n as u16),
249        _ => Err(()),
250    }
251}

253/// Lowercase ASCII name labels or an IP literal; never empty, a wildcard or a port.
254fn canonical_host(h: &str) -> bool {
255    (1..=MAX_HOST).contains(&h.len())
256        && h.bytes()
257            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b"-.:".contains(&b))
258        && !h.ends_with('.')
259}

261/// The blob's first SSH string must name the line's algorithm.
262fn blob_names(blob: &[u8], algorithm: &str) -> bool {
263    blob.len() >= 4 + algorithm.len()
264        && u32::from_be_bytes([blob[0], blob[1], blob[2], blob[3]]) as usize == algorithm.len()
265        && &blob[4..4 + algorithm.len()] == algorithm.as_bytes()
266}

268fn base64_decode(s: &str) -> Result<Vec<u8>, ()> {
269    let b = s.as_bytes();
270    if b.is_empty() || !b.len().is_multiple_of(4) {
271        return Err(());
272    }
273    let mut out = Vec::with_capacity(b.len() / 4 * 3);
274    for (i, quad) in b.chunks(4).enumerate() {
275        let last = i == b.len() / 4 - 1;
276        let mut n = 0u32;
277        let mut pad = 0;
278        for (j, &c) in quad.iter().enumerate() {
279            let v = match c {
280                b'A'..=b'Z' => c - b'A',
281                b'a'..=b'z' => c - b'a' + 26,
282                b'0'..=b'9' => c - b'0' + 52,
283                b'+' => 62,
284                b'/' => 63,
285                b'=' if last && j >= 2 => {
286                    pad += 1;
287                    0
288                }
289                _ => return Err(()),
290            };
291            if pad > 0 && c != b'=' {
292                return Err(());
293            }
294            n = n << 6 | u32::from(v);
295        }
296        let bytes = [(n >> 16) as u8, (n >> 8) as u8, n as u8];
297        out.extend_from_slice(&bytes[..3 - pad]);
298    }
299    Ok(out)
300}

302/// Standard base64 of `data` appended to `out`, `=`-padded or not.
303fn base64_encode(data: &[u8], padded: bool, out: &mut alloc::string::String) {
304    const T: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
305    for chunk in data.chunks(3) {
306        let n = chunk
307            .iter()
308            .enumerate()
309            .fold(0u32, |n, (i, b)| n | u32::from(*b) << (16 - 8 * i));
310        for i in 0..4 {
311            if i <= chunk.len() {
312                out.push(T[(n >> (18 - 6 * i) & 63) as usize] as char);
313            } else if padded {
314                out.push('=');
315            }
316        }
317    }
