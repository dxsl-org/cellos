34    fn reconcile_pending_cleanup(&mut self) -> Result<(), KmsErrorCode> {
35        let Some(generation) = self
36            .lifecycle
37            .cleanup_pending()
38            .map(|pending| pending.generation)
39        else {
40            return Ok(());
41        };
42        self.provider
43            .destroy_enrollment_key(generation)
44            .map_err(|_| KmsErrorCode::RelayUnavailable)?;
45        self.lifecycle.confirm_cleanup();
46        Ok(())
47    }

49    /// Opcode 9: supervisor-only enrollment open plus canonical CSR publish.
50    pub(super) fn begin_enrollment(
51        &mut self,
52        request: &KmsRequestV1,
53        sender: usize,
54        caller: Option<CallerIdentity>,
55        registry: ServiceRegistrySnapshot,
56    ) -> Result<SuccessPayload, KmsErrorCode> {
57        let supervisor = authorize_enrollment_supervisor(sender, caller, registry.supervisor_tid)?;
58        let payload = RelayEnrollmentBeginRequestPayload::decode(
59            request
60                .payload()
61                .map_err(|_| KmsErrorCode::InvalidRequest)?,
62        )
63        .ok_or(KmsErrorCode::InvalidRequest)?;
64        // A poisoned slot is a cleanup tombstone. A fresh Begin may replace it
65        // only after the provider confirms deletion or prior absence.
66        self.reconcile_pending_cleanup()?;
67        // Deny before touching the provider whenever the slot is unavailable.
68        let (generation, policy_epoch) = self
69            .lifecycle
70            .open_pending(supervisor, request.request_id)?;
71        let proof = match self
72            .provider
73            .begin_enrollment(generation, payload.hostname())
74        {
75            Ok(proof) => proof,
76            Err(crate::storage::RelaySignError::CleanupFailed) => {
77                self.lifecycle.mark_cleanup_required();
78                return Err(KmsErrorCode::RelayUnavailable);
79            }
80            Err(error) => {
81                self.lifecycle.drop_pending();
82                return Err(map_relay_error(error));
83            }
84        };
85        let assembled =
86            match verify_and_assemble_csr(payload.hostname(), &proof.spki_sec1, proof.signature) {
87                Ok(assembled) => assembled,
88                Err(_) => {
89                    self.lifecycle.mark_cleanup_required();
90                    if self.reconcile_pending_cleanup().is_err() {
91                        return Err(KmsErrorCode::RelayUnavailable);
92                    }
93                    return Err(KmsErrorCode::InvalidSignature);
94                }
95            };
96        let csr_sha256 = assembled.sha256();
97        let csr_handle = self.lifecycle.pending().expect("just opened").csr_handle;
98        self.lifecycle.install_csr(&assembled.bytes, assembled.len);
99        let response = RelayEnrollmentBeginResponsePayload {
100            pending_relay_generation: generation,
101            policy_epoch,
102            restart_epoch: self.lifecycle.restart_epoch(),
103            csr_handle,
104            csr_len: assembled.len as u32,
105            reserved: 0,
106            csr_sha256,
107        };
108        SuccessPayload::new(&response.encode())
109    }

111    /// Opcode 10: one-shot ordered CSR chunk read bound to the live supervisor.
112    pub(super) fn read_relay_csr_chunk(
113        &mut self,
114        request: &KmsRequestV1,
115        sender: usize,
116        caller: Option<CallerIdentity>,
117        registry: ServiceRegistrySnapshot,
118    ) -> Result<SuccessPayload, KmsErrorCode> {
119        let supervisor = authorize_enrollment_supervisor(sender, caller, registry.supervisor_tid)?;
120        let payload = RelayCsrChunkRequestPayload::decode(
121            request
122                .payload()
123                .map_err(|_| KmsErrorCode::InvalidRequest)?,
124        )
125        .ok_or(KmsErrorCode::InvalidRequest)?;
126        let result =
127            self.lifecycle
128                .read_chunk(payload.csr_handle, payload.chunk_index, &supervisor);
129        let (chunk, len) = match result {
130            Ok(output) => output,
131            Err(code) => {
132                if self.lifecycle.cleanup_pending().is_some()
133                    && self.reconcile_pending_cleanup().is_err()
134                {
135                    return Err(KmsErrorCode::RelayUnavailable);
136                }
137                return Err(code);
138            }
139        };
140        let response = RelayCsrChunkResponsePayload {
141            chunk_index: payload.chunk_index,
142            chunk_len: len as u16,
143            reserved: 0,
144            chunk,
145        };
146        SuccessPayload::new(&response.encode())
147    }

149    /// Opcode 11: atomic commit; promotes the provider key between lifecycle
150    /// validation and activation so failure leaves no mixed state.
151    pub(super) fn commit_relay_generation(
152        &mut self,
153        request: &KmsRequestV1,
154        sender: usize,
155        caller: Option<CallerIdentity>,
156        registry: ServiceRegistrySnapshot,
157    ) -> Result<SuccessPayload, KmsErrorCode> {
158        let supervisor = authorize_enrollment_supervisor(sender, caller, registry.supervisor_tid)?;
159        let payload = RelayGenerationCommitRequestPayload::decode(
160            request
161                .payload()
162                .map_err(|_| KmsErrorCode::InvalidRequest)?,
163        )
164        .ok_or(KmsErrorCode::InvalidRequest)?;
165        self.lifecycle.prepare_commit(
166            payload.pending_relay_generation,
167            payload.expected_policy_epoch,
168            payload.profile_digest,
169            &supervisor,
170        )?;
171        self.provider
172            .commit_enrollment(payload.pending_relay_generation, payload.profile_digest)
173            .map_err(map_relay_error)?;
174        let committed = self.lifecycle.apply_commit(
175            payload.pending_relay_generation,
176            payload.expected_policy_epoch,
177            payload.profile_digest,
178            &supervisor,
179        );
180        match committed {
181            Ok(committed) => {
182                if let Err(code) = self.persist_protected_lifecycle() {
183                    self.lifecycle.seal();
184                    return Err(code);
185                }
186                SuccessPayload::new(
187                    &RelayGenerationCommitResponsePayload {
188                        active_relay_generation: committed.generation,
189                        policy_epoch: committed.policy_epoch,
190                        active_profile_digest: committed.profile_digest,
191                    }
192                    .encode(),
193                )
194            }
195            Err(code) => {
196                // Promotion without a matching protected transition is never
197                // allowed to serve.
198                self.lifecycle.seal();
199                Err(code)
200            }
201        }
202    }

204    /// Opcode 12: abort destroys the pending slot and its provider key.
205    pub(super) fn abort_relay_enrollment(
206        &mut self,
207        request: &KmsRequestV1,
208        sender: usize,
209        caller: Option<CallerIdentity>,
210        registry: ServiceRegistrySnapshot,
211    ) -> Result<SuccessPayload, KmsErrorCode> {
212        let supervisor = authorize_enrollment_supervisor(sender, caller, registry.supervisor_tid)?;
213        let payload = RelayEnrollmentAbortRequestPayload::decode(
214            request
215                .payload()
216                .map_err(|_| KmsErrorCode::InvalidRequest)?,
217        )
218        .ok_or(KmsErrorCode::InvalidRequest)?;
219        {
220            let pending = self
221                .lifecycle
222                .pending()
223                .ok_or(KmsErrorCode::CsrHandleInvalid)?;
224            if pending.generation != payload.pending_relay_generation {
225                return Err(KmsErrorCode::RelayGenerationMismatch);
226            }
227            if !pending.supervisor.matches(&supervisor) {
228                return Err(KmsErrorCode::PermissionDenied);
229            }
230        }
231        self.provider
232            .destroy_enrollment_key(payload.pending_relay_generation)
233            .map_err(|_| KmsErrorCode::RelayUnavailable)?;
234        self.lifecycle.drop_pending();
235        Ok(SuccessPayload::new(&[])?)
236    }

238    /// Opcode 13: service-net binds its validated profile digest.
239    pub(super) fn stage_relay_profile(
240        &mut self,
241        request: &KmsRequestV1,
242        sender: usize,
243        caller: Option<CallerIdentity>,
244        registry: ServiceRegistrySnapshot,
245    ) -> Result<SuccessPayload, KmsErrorCode> {
246        self.authorize_service_net(sender, caller, registry)?;
247        let payload = RelayStageProfileRequestPayload::decode(
248            request
249                .payload()
250                .map_err(|_| KmsErrorCode::InvalidRequest)?,
251        )
252        .ok_or(KmsErrorCode::InvalidRequest)?;
253        self.lifecycle.stage_pending(
254            payload.pending_relay_generation,
255            payload.expected_policy_epoch,
256            payload.profile_digest,
257        )?;
258        SuccessPayload::new(&[])
259    }

261    /// Opcode 14: public active-generation key facts for service-net only.
262    pub(super) fn get_relay_active_public_key(
263        &mut self,
264        request: &KmsRequestV1,
265        sender: usize,
266        caller: Option<CallerIdentity>,
267        registry: ServiceRegistrySnapshot,
268    ) -> Result<SuccessPayload, KmsErrorCode> {
269        require_empty(request)?;
270        self.authorize_service_net(sender, caller, registry)?;
271        let active = self
272            .lifecycle
273            .serving()
274            .ok_or(KmsErrorCode::RelayUnavailable)?;
275        let status = self.provider.relay_p256_status();
276        let (spki, spki_len) =
277            p256_spki_der(&status.verifying_key_sec1).ok_or(KmsErrorCode::RelayUnavailable)?;
278        SuccessPayload::new(
279            &RelayActivePublicKeyPayload {
280                relay_generation: active.generation,
281                spki_sec1: status.verifying_key_sec1,
282                spki_sha256: Sha256::digest(&spki[..spki_len]).into(),
283            }
284            .encode(),
285        )
286    }
