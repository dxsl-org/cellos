8/// Canonical CSR assembled inside KMS after self-verifying the provider proof.
9pub(crate) struct AssembledCsr {
10    pub bytes: [u8; RELAY_CSR_MAX_LEN],
11    pub len: usize,
12}
13
14impl AssembledCsr {
15    pub fn sha256(&self) -> [u8; 32] {
16        Sha256::digest(&self.bytes[..self.len]).into()
17    }
18}

20/// Verify the provider's raw `r||s` proof against KMS's *own* canonical CRI
21/// reconstruction and assemble the bounded PKCS#10 CSR.
22///
23/// The provider signed its independent reconstruction of the same frozen
24/// profile; verification therefore proves both sides built byte-identical
25/// CRIs over the fresh non-exportable key before anything is published.
26pub(crate) fn verify_and_assemble_csr(
27    hostname: &[u8],
28    spki_sec1: &[u8; 65],
29    raw_signature: [u8; 64],
30) -> Result<AssembledCsr, RelaySignError> {
31    let (cri, cri_len) =
32        canonical_relay_cri(hostname, spki_sec1).ok_or(RelaySignError::InvalidRequest)?;
33    let digest = Sha256::digest(&cri[..cri_len]);
34    let signature = Signature::from_slice(&raw_signature).map_err(|_| RelaySignError::Failure)?;
35    let normalized = signature.normalize_s().unwrap_or(signature);
36    let verifying_key =
37        VerifyingKey::from_sec1_bytes(spki_sec1).map_err(|_| RelaySignError::Failure)?;
38    verifying_key
39        .verify_prehash(&digest, &normalized)
40        .map_err(|_| RelaySignError::Failure)?;
41    let mut raw = [0u8; 64];
42    raw.copy_from_slice(normalized.to_bytes().as_slice());
43    let (sig_der, sig_len) = der_ecdsa_signature(&raw);
44    let (bytes, len) = assemble_relay_csr(&cri, cri_len, &sig_der, sig_len)
45        .ok_or(RelaySignError::InvalidRequest)?;
46    Ok(AssembledCsr { bytes, len })
47}
