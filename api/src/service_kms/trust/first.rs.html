1//! First trust (Phase 09 S9-4c; abi-vfs-telemetry §4; D79-11/12).
2//!
3//! Challenge (`CommitTrustQuery` of a challenge query): for a host and port the store does
4//! not hold, the service records a challenge bound to the caller principal, the exact
5//! key, the store's current observation and a 60-second deadline, and registers its
6//! immutable descriptor with the input owner (`RegisterTrustChallenge`). The input owner,
7//! not the caller, then runs the confirmation on the caller's terminal session.
8//!
9//! `CommitTrustChallenge`: the caller's own challenge only. The owner-attested decision
10//! (`CollectTrustDecision`) must accept and match every binding of the record (challenge,
11//! caller, session and epoch, descriptor SHA-256); then the store is re-observed and,
12//! still Unknown, extended by one `interactive` line and published through the VFS
13//! conditional save against that observation. Another writer's line for the same host and
14//! port wins: the same key answers KnownExact, a different one Mismatch, and nothing is
15//! ever overwritten. A publication reports `Committed` (synced, not crash-proof; S9-4a).
16//! Every final answer ends the challenge.
17

34/// Challenge lifetime: the input owner's ceiling, never extended.
35const LIFETIME_NS: u64 = tr::EXPIRY_MAX_NS;
36/// Open challenges per caller principal, and in total (§4: four per worker).
37const MAX_PER_CALLER: usize = 1;
38const MAX_CHALLENGES: usize = 4;
39
40struct Challenge {
41    challenge: Ref,
42    owner: Principal,
43    host: String,
44    port: u16,
45    algorithm: String,
46    key: Vec<u8>,
47    /// The store as it was when the challenge was issued.
48    observation: Observation,
49    expiry: u64,
50    confirmation: Ref,
51    session: Ref,
52    epoch: u64,
53    /// SHA-256 of the descriptor registered with the input owner.
54    digest: [u8; 32],
55}

80/// `SHA256:` and the unpadded base64 of the blob's digest, exactly as `vissh` prints it.
81fn fingerprint(key: &[u8]) -> String {
82    let mut s = String::from("SHA256:");
83    base64_encode(&Sha256::digest(key), false, &mut s);
84    s
85}

87/// The store's observation, its bytes and its answer for this host, port and key; `None`
88/// when the store cannot be read.
89pub(super) fn current(
90    io: &mut impl TrustIo,
91    host: &str,
92    port: u16,
93    algorithm: &str,
94    key: &[u8],
95) -> Option<(Observation, Vec<u8>, u8)> {
96    match io.observe().ok()? {
97        Observation::Absent => Some((Observation::Absent, Vec::new(), ssh_trust::UNKNOWN)),
98        observation @ Observation::Present {
99            size,
100            regular: true,
101            ..
102        } => {
103            if size > MAX_STORE as u64 {
104                return None;
105            }
106            let bytes = io.read(MAX_STORE).ok()?;
107            if bytes.len() as u64 != size {
108                return None;
109            }
110            let code = lookup(&bytes, host, port, algorithm, key);
111            Some((observation, bytes, code))
112        }
113        Observation::Present { .. } => None,
114    }
115}

117#[allow(clippy::too_many_arguments)]
118pub(super) fn challenge(
119    leaf: &mut TrustLeaf,
120    caller: Principal,
121    host: &str,
122    port: u16,
123    algorithm: &str,
124    key: &[u8],
125    session: Ref,
126    epoch: u64,
127) -> Result<TrustOutcomePayload, KmsErrorCode> {
128    let answer = |code| Ok(TrustOutcomePayload::outcome(code));
129    // Only what a store line can hold is ever offered to the user.
130    if !canonical_host(host)
131        || port == 0
132        || algorithm != "ssh-ed25519"
133        || key.is_empty()
134        || key.len() > MAX_BLOB
135        || !blob_names(key, algorithm)
136        || session.is_zero()
137    {
138        return Err(KmsErrorCode::InvalidRequest);
139    }
140    let Some((observation, _, code)) = current(&mut leaf.io, host, port, algorithm, key) else {
141        return answer(ssh_trust::UNAVAILABLE);
142    };
143    // KnownExact needs no prompt; a changed key never prompts.
144    if code != ssh_trust::UNKNOWN {
145        return answer(code);
146    }
147    let Ok(now) = owner::now_ns() else {
148        return answer(ssh_trust::UNAVAILABLE);
149    };
150    let table = &mut leaf.challenges;
151    // A caller's own expired challenge gives way; any other is released one lifetime later.
152    table.records.retain(|r| {
153        now < r.expiry.saturating_add(LIFETIME_NS) && !(r.owner == caller && now >= r.expiry)
154    });
155    if table.records.iter().filter(|r| r.owner == caller).count() >= MAX_PER_CALLER
156        || table.records.len() >= MAX_CHALLENGES
157    {
158        return Err(KmsErrorCode::Busy);
159    }
160    let challenge = Ref {
161        id: table.next,
162        generation: 1,
163    };
164    let Some(next) = table.next.checked_add(1) else {
165        return Err(KmsErrorCode::Busy);
166    };
167    let expiry = now.saturating_add(LIFETIME_NS);
168    let fingerprint = fingerprint(key);
169    let descriptor = TrustChallengeV1 {
170        challenge,
171        target: principal(caller),
172        expiry_ns: expiry,
173        prompt: TrustPromptV1 {
174            port,
175            host,
176            algorithm,
177            fingerprint: &fingerprint,
178        },
179    };
180    let mut bytes = [0u8; TrustChallengeV1::MAX_LEN];
181    let Ok(len) = descriptor.encode(&mut bytes) else {
182        return Err(KmsErrorCode::InvalidRequest);
183    };
184    let digest: [u8; 32] = Sha256::digest(&bytes[..len]).into();
185    // Spent before registering: input may have recorded the challenge even
186    // when its reply was lost, and it refuses a reused ref.
187    table.next = next;
188    let Ok(confirmation) = owner::register(&descriptor) else {
189        return answer(ssh_trust::UNAVAILABLE);
190    };
191    table.records.push(Challenge {
192        challenge,
193        owner: caller,
194        host: String::from(host),
195        port,
196        algorithm: String::from(algorithm),
197        key: key.to_vec(),
198        observation,
199        expiry,
200        confirmation,
201        session,
202        epoch,
203        digest,
204    });
205    Ok(TrustOutcomePayload::challenged(
206        (confirmation.id, confirmation.generation),
207        (challenge.id, challenge.generation),
208    ))
209}

211/// One `api::ipc::ssh_trust` code for the caller's own `challenge`.
212pub(super) fn commit(leaf: &mut TrustLeaf, caller: Principal, challenge: Ref) -> u8 {
213    // Only the caller's own challenge: a guessed or another caller's ref is refused.
214    let Some(n) = leaf
215        .challenges
216        .records
217        .iter()
218        .position(|r| r.challenge == challenge && r.owner == caller)
219    else {
220        return ssh_trust::DENIED;
221    };
222    let decision = match owner::collect(leaf.challenges.records[n].confirmation) {
223        Ok(decision) => decision,
224        // Uncertain transport: the decision is still the owner's to give.
225        Err(Status::Indeterminate | Status::WouldBlock | Status::TimedOut) => {
226            return ssh_trust::PENDING
227        }
228        Err(_) => {
229            leaf.challenges.records.swap_remove(n);
230            return ssh_trust::UNAVAILABLE;
231        }
232    };
233    if !decision.is_final() {
234        return ssh_trust::PENDING;
235    }
236    let record = leaf.challenges.records.swap_remove(n);
237    match decision.decision {
238        tr::DECISION_ACCEPT => {
239            // The owner's reply must bind exactly this record: challenge, caller, the
240            // session and epoch the caller named, and the descriptor this service registered.
241            let bound = decision.challenge == record.challenge
242                && decision.target == principal(caller)
243                && decision.session == record.session
244                && decision.epoch == record.epoch
245                && decision.descriptor_sha256 == record.digest;
246            match owner::now_ns() {
247                _ if !bound => ssh_trust::REFUSED,
248                Ok(now) if now < record.expiry => persist(&mut leaf.io, &record),
249                _ => ssh_trust::EXPIRED,
250            }
251        }
252        tr::DECISION_EXPIRED => ssh_trust::EXPIRED,
253        _ => ssh_trust::REFUSED,
254    }
255}

257/// Add the record's line to the store: at most two publication attempts, each against
258/// the observation it was built from.
259fn persist(io: &mut impl TrustIo, r: &Challenge) -> u8 {
260    let mut expected = r.observation;
261    for _ in 0..2 {
262        let Some((observation, current, code)) = current(io, &r.host, r.port, &r.algorithm, &r.key)
263        else {
264            return ssh_trust::UNAVAILABLE;
265        };
266        // Changed since the challenge (or the last attempt): another writer may have
267        // stored this host and port meanwhile; its line is never overwritten.
268        if observation != expected && code != ssh_trust::UNKNOWN {
269            return code;
270        }
271        let mut bytes = if observation == Observation::Absent {
272            String::from(HEADER) + "\n"
273        } else {
274            match String::from_utf8(current) {
275                Ok(text) => text,
276                Err(_) => return ssh_trust::UNAVAILABLE,
277            }
278        };
279        bytes.push_str(&r.host);
280        bytes.push_str(&alloc::format!(" {} {} ", r.port, r.algorithm));
281        base64_encode(&r.key, true, &mut bytes);
282        bytes.push_str(" interactive\n");
283        // The store's own bounds (entries, blob, total size) hold for the new revision too.
284        if parse(bytes.as_bytes()).is_err() || io.ensure_dir().is_err() {
285            return ssh_trust::UNAVAILABLE;
286        }
287        let Some(expect) = observation.expect() else {
288            return ssh_trust::UNAVAILABLE;
289        };
290        match io.publish(bytes.as_bytes(), expect) {
291            outcome if outcome.published().is_some() => return ssh_trust::KNOWN_EXACT,
292            SaveOutcome::Conflict => expected = observation,
293            _ => return ssh_trust::UNAVAILABLE,
294        }
295    }
296    ssh_trust::UNAVAILABLE
297}
