5/// Approved pending flow: `Prepared -> CsrIssued -> Staged`.
6///
7/// `Staged` requires the complete ordered CSR consumption plus an
8/// authenticated service-net staging call that matches the pending
9/// generation; commit is valid only from `Staged`.
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub(crate) enum PendingStage {
12    Prepared,
13    CsrIssued,
14    Staged,
15}
16
17/// Pending enrollment created by `BeginRelayEnrollment` and destroyed by
18/// commit, abort, any handle violation, or process restart.
19#[derive(Debug)]
20pub(crate) struct PendingEnrollment {
21    pub generation: u64,
22    pub policy_epoch: u64,
23    pub supervisor: SupervisorContext,
24    pub csr_handle: u64,
25    pub stage: PendingStage,
26    staged_digest: [u8; 32],
27    csr: [u8; RELAY_CSR_MAX_LEN],
28    csr_len: usize,
29    chunks_remaining: u32,
30}
31
32impl PendingEnrollment {
33    pub(crate) fn new(
34        generation: u64,
35        policy_epoch: u64,
36        supervisor: SupervisorContext,
37        csr_handle: u64,
38    ) -> Self {
39        Self {
40            generation,
41            policy_epoch,
42            supervisor,
43            csr_handle,
44            stage: PendingStage::Prepared,
45            staged_digest: [0; 32],
46            csr: [0; RELAY_CSR_MAX_LEN],
47            csr_len: 0,
48            chunks_remaining: 0,
49        }
50    }
51
52    /// Move `Prepared -> CsrIssued` once the verified canonical CSR exists.
53    pub(crate) fn install(&mut self, csr: &[u8; RELAY_CSR_MAX_LEN], csr_len: usize) {
54        debug_assert_eq!(self.stage, PendingStage::Prepared);
55        self.csr[..csr_len].copy_from_slice(&csr[..csr_len]);
56        self.csr_len = csr_len;
57        self.chunks_remaining = self.total_chunks();
58        self.stage = PendingStage::CsrIssued;
59    }
60
61    fn total_chunks(&self) -> u32 {
62        self.csr_len.div_ceil(RELAY_CSR_CHUNK_LEN) as u32
63    }
64
65    /// Return the exactly-ordered chunk; only valid from `CsrIssued`.
66    pub(crate) fn chunk(
67        &mut self,
68        chunk_index: u32,
69    ) -> Result<([u8; RELAY_CSR_CHUNK_LEN], usize), KmsErrorCode> {
70        if self.stage != PendingStage::CsrIssued {
71            return Err(KmsErrorCode::CsrHandleInvalid);
72        }
73        if self.chunks_remaining == 0 {
74            // The one-shot handle is exhausted; it no longer names anything.
75            return Err(KmsErrorCode::CsrHandleInvalid);
76        }
77        if chunk_index != self.next_chunk_index() || chunk_index >= self.total_chunks() {
78            return Err(KmsErrorCode::CsrOrderInvalid);
79        }
80        let start = chunk_index as usize * RELAY_CSR_CHUNK_LEN;
81        let end = (start + RELAY_CSR_CHUNK_LEN).min(self.csr_len);
82        let mut chunk = [0u8; RELAY_CSR_CHUNK_LEN];
83        chunk[..end - start].copy_from_slice(&self.csr[start..end]);
84        self.chunks_remaining -= 1;
85        Ok((chunk, end - start))
86    }
87
88    fn next_chunk_index(&self) -> u32 {
89        self.total_chunks() - self.chunks_remaining
90    }
91
92    /// Move `CsrIssued -> Staged`: every chunk is gone and the live
93    /// service-net binding attests a validated profile for this generation.
94    pub(crate) fn mark_staged(&mut self, profile_digest: [u8; 32]) -> Result<(), KmsErrorCode> {
95        if self.stage != PendingStage::CsrIssued {
96            return Err(KmsErrorCode::InvalidRequest);
97        }
98        if self.chunks_remaining != 0 || profile_digest.iter().all(|byte| *byte == 0) {
99            return Err(KmsErrorCode::InvalidRequest);
100        }
101        self.staged_digest = profile_digest;
102        self.stage = PendingStage::Staged;
103        Ok(())
104    }
105
106    pub(crate) fn staged_digest_matches(&self, profile_digest: [u8; 32]) -> bool {
107        self.stage == PendingStage::Staged && self.staged_digest == profile_digest
108    }
109}

111/// Opaque one-shot handle mixed from every binding fact. Not a secret; the
112/// binding checks in the lifecycle reader are what enforce access.
113pub(crate) fn derive_csr_handle(
114    generation: u64,
115    policy_epoch: u64,
116    request_id: u64,
117    begin_counter: u64,
118    restart_epoch: u64,
119    supervisor: &SupervisorContext,
120) -> u64 {
121    use blake2::{Blake2s256, Digest};
122    let mut hash = Blake2s256::new();
123    hash.update(b"cellos-relay-csr-handle-v1");
124    hash.update(generation.to_le_bytes());
125    hash.update(policy_epoch.to_le_bytes());
126    hash.update(request_id.to_le_bytes());
127    hash.update(begin_counter.to_le_bytes());
128    hash.update(restart_epoch.to_le_bytes());
129    hash.update(supervisor.cell_id.to_le_bytes());
130    hash.update(supervisor.generation.to_le_bytes());
131    hash.update((supervisor.tid as u64).to_le_bytes());
132    let digest = hash.finalize();
133    let raw = u64::from_le_bytes(digest.as_slice()[..8].try_into().expect("fixed digest"));
134    // Never zero; uniqueness comes from the bound facts, not entropy.
135    raw | 0x8000_0000_0000_0001
136}
