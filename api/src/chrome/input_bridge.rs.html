1//! The window's input as Linux evdev events for the guest's virtio-input tablet.
2//!
3//! Key events carry the evdev code in `KeyEvent::scancode` (the host's own
4//! virtio keyboard codes). Modifier keys are not delivered to apps as separate
5//! events, only as `KeyEvent::modifiers`, so the guest's modifier keys are
6//! pressed and released to match before each key.
7
8use alloc::vec::Vec;
9
10use api::services::vmm_session::{
11    abs_position, ABS_X, ABS_Y, BTN_EXTRA, BTN_LEFT, BTN_MIDDLE, BTN_RIGHT, BTN_SIDE, EV_ABS,
12    EV_KEY, EV_REL, REL_WHEEL,
13};
14use ostd::input::{InputEvent, KeyState, Modifiers, MouseButton};
15
16/// One evdev event: (type, code, value).
17pub type GuestEvent = (u16, u16, i32);
18
19/// Linux keycodes of the modifiers kept in step with `KeyEvent::modifiers`.
20const MODIFIER_KEYS: [(Modifiers, u16); 4] = [
21    (Modifiers::SHIFT, 42), // KEY_LEFTSHIFT
22    (Modifiers::CTRL, 29),  // KEY_LEFTCTRL
23    (Modifiers::ALT, 56),   // KEY_LEFTALT
24    (Modifiers::META, 125), // KEY_LEFTMETA
25];
26/// Highest evdev key code the guest tablet accepts (KEY_MAX).
27const KEY_MAX: u32 = 0x2ff;
28
29pub struct InputBridge {
30    /// Modifiers currently held down in the guest.
31    held: Modifiers,
32}
33
34impl InputBridge {
35    pub const fn new() -> Self {
36        Self {
37            held: Modifiers(0),
38        }
39    }
40
41    /// Guest events for one window event; `width`/`height` are the window's
42    /// content size, which the guest display matches.
43    pub fn translate(&mut self, event: &InputEvent, width: u32, height: u32) -> Vec<GuestEvent> {
44        let mut out = Vec::new();
45        match *event {
46            InputEvent::MouseMove { x, y, .. } => {
47                out.push((EV_ABS, ABS_X, abs_position(x, width)));
48                out.push((EV_ABS, ABS_Y, abs_position(y, height)));
49            }
50            InputEvent::MouseButton { button, state } => {
51                out.push((EV_KEY, button_code(button), i32::from(state != KeyState::Released)));
52            }
53            InputEvent::MouseScroll { dy, .. } if dy != 0 => {
54                // Host scroll is positive downwards; evdev REL_WHEEL is positive upwards.
55                out.push((EV_REL, REL_WHEEL, -dy));
56            }
57            InputEvent::MouseScroll { .. } => {}
58            InputEvent::Key(key) => {
59                if key.scancode == 0 || key.scancode > KEY_MAX {
60                    return out;
61                }
62                self.sync_modifiers(key.modifiers, &mut out);
63                let code = key.scancode as u16;
64                match key.state {
65                    KeyState::Released => out.push((EV_KEY, code, 0)),
66                    KeyState::Pressed => out.push((EV_KEY, code, 1)),
67                    // The host input service owns repeat and the guest's is off
68                    // (scripts/chrome-guest/no-key-repeat.c). libinput drops
69                    // evdev value 2, so a repeat is a fresh press.
70                    KeyState::Repeated => out.extend([(EV_KEY, code, 0), (EV_KEY, code, 1)]),
71                }
72            }
73        }
74        out
75    }
76
77    fn sync_modifiers(&mut self, wanted: Modifiers, out: &mut Vec<GuestEvent>) {
78        for (modifier, code) in MODIFIER_KEYS {
79            let want = wanted.contains(modifier);
80            if want != self.held.contains(modifier) {
81                out.push((EV_KEY, code, i32::from(want)));
82                self.held = Modifiers(self.held.0 ^ modifier.0);
83            }
84        }
85    }
86}
87
88fn button_code(button: MouseButton) -> u16 {
89    match button {
90        MouseButton::Left => BTN_LEFT,
91        MouseButton::Right => BTN_RIGHT,
92        MouseButton::Middle => BTN_MIDDLE,
93        MouseButton::Back => BTN_SIDE,
94        MouseButton::Forward => BTN_EXTRA,
95    }
96}

98#[cfg(test)]
99mod tests {
100    use super::*;
101    use api::input::KeyEvent;
102    use api::services::vmm_session::ABS_MAX;
103
104    fn key(scancode: u32, modifiers: Modifiers, state: KeyState) -> InputEvent {
105        InputEvent::Key(KeyEvent {
106            timestamp_ticks: 0,
107            scancode,
108            keysym: api::input::KeySym::Printable,
109            character: 0,
110            modifiers,
111            state,
112            _pad: [0; 2],
113        })
114    }
115
116    #[test]
117    fn shifted_key_presses_shift_first_and_releases_it_with_the_next_plain_key() {
118        let mut bridge = InputBridge::new();
119        let shifted = bridge.translate(&key(30, Modifiers::SHIFT, KeyState::Pressed), 800, 600);
120        assert_eq!(shifted, [(EV_KEY, 42, 1), (EV_KEY, 30, 1)]);
121        let repeat = bridge.translate(&key(30, Modifiers::SHIFT, KeyState::Repeated), 800, 600);
122        assert_eq!(repeat, [(EV_KEY, 30, 0), (EV_KEY, 30, 1)]);
123        let plain = bridge.translate(&key(48, Modifiers(0), KeyState::Pressed), 800, 600);
124        assert_eq!(plain, [(EV_KEY, 42, 0), (EV_KEY, 48, 1)]);
125    }
126
127    #[test]
128    fn pointer_maps_window_corners_to_the_axis_ends() {
129        let mut bridge = InputBridge::new();
130        let corner = InputEvent::MouseMove {
131            x: 799,
132            y: 0,
133            dx: 0,
134            dy: 0,
135        };
136        assert_eq!(
137            bridge.translate(&corner, 800, 600),
138            [(EV_ABS, ABS_X, ABS_MAX), (EV_ABS, ABS_Y, 0)]
139        );
140        let scroll = InputEvent::MouseScroll { dx: 0, dy: 3 };
141        assert_eq!(bridge.translate(&scroll, 800, 600), [(EV_REL, REL_WHEEL, -3)]);
142    }
143
144    #[test]
145    fn keys_without_an_evdev_code_are_not_forwarded() {
146        let mut bridge = InputBridge::new();
147        assert!(bridge
148            .translate(&key(0, Modifiers::SHIFT, KeyState::Pressed), 800, 600)
149            .is_empty());
150    }
