14const DEFAULT_WIDTH: u32 = 1024;
15const DEFAULT_HEIGHT: u32 = 680;

17/// What the window shows when no guest frame is on screen.
18fn paint_status(surface: &mut ViSurface, fonts: &mut TextFonts, text: &str) {
19    let palette = theme::system();
20    let (width, height, stride) = (surface.width(), surface.height(), surface.stride());
21    let bg = palette.bg();
22    let fill = [bg.b(), bg.g(), bg.r(), bg.a()];
23    for pixel in surface.pixels_mut().chunks_exact_mut(4) {
24        pixel.copy_from_slice(&fill);
25    }
26    let fg = palette.text_secondary();
27    fonts.draw_text(
28        surface.pixels_mut(),
29        width,
30        height,
31        stride,
32        24.0,
33        (height / 2) as f32 - 8.0,
34        text,
35        FontFace::UiRegular,
36        14.0,
37        [fg.b(), fg.g(), fg.r(), fg.a()],
38        (0, 0, width as i32, height as i32),
39    );
40    surface.damage_all();
41}

43pub fn cell_main() {
44    #[cfg(target_os = "none")]
45    crate::init_custom_heap();
46
47    let comp_tid = ostd::display::wait_for_compositor();
48    let (screen_w, screen_h) = ostd::syscall::sys_get_resolution();
49    let width = DEFAULT_WIDTH.min(screen_w.max(320));
50    let height = DEFAULT_HEIGHT.min(screen_h.saturating_sub(80).max(240));
51    let Ok(mut surface) = ViSurface::create(comp_tid, width, height, PixelFormat::Bgra8888) else {
52        ostd::io::println("[chrome] cannot create the window; exiting");
53        ostd::syscall::sys_exit(1);
54    };
55    let _ = surface.set_title("Chrome");
56    let _ = surface.set_icon(ostd::display::linear::icons::LineIcon::Globe);
57    surface.move_to(
58        (screen_w.saturating_sub(width) / 2) as i32,
59        (screen_h.saturating_sub(height) / 2) as i32,
60    );
61    let mut fonts = TextFonts::new();
62    paint_status(&mut surface, &mut fonts, "Starting Chromium in a Linux guest...");
63
64    let guest = match Guest::start(width, height) {
65        Ok(guest) => {
66            ostd::io::println("[chrome] guest session bound");
67            Some(guest)
68        }
69        Err(reason) => {
70            ostd::io::println(&alloc::format!("[chrome] guest not started: {reason}"));
71            paint_status(&mut surface, &mut fonts, reason);
72            None
73        }
74    };
75    while !ostd::input::request_focus() {
76        ostd::syscall::sys_yield();
77    }
78
79    let mut window = WindowLifecycle::new(ClosePolicy::Accept);
80    let mut frame: Option<GuestFrame> = None;
81    let mut bridge = InputBridge::new();
82    let mut outbox = InputOutbox::new();
83    loop {
84        let mut closed = false;
85        let mut resized = false;
86        window.poll(&mut surface, |event| match event {
87            WindowEvent::Closed => closed = true,
88            WindowEvent::Resized { .. } | WindowEvent::Restored => resized = true,
89            WindowEvent::Minimized | WindowEvent::CloseRequested { .. } => {}
90        });
91        if closed {
92            if let Some(guest) = &guest {
93                guest.terminate();
94            }
95            break;
96        }
97        let Some(guest) = &guest else {
98            if resized {
99                paint_status(&mut surface, &mut fonts, "The Linux guest could not be started.");
100            }
101            let _ = ostd::syscall::sys_set_timer(1);
102            continue;
103        };
104        if resized {
105            // The old frame no longer matches; the guest re-announces at the new size.
106            frame = None;
107            paint_status(&mut surface, &mut fonts, "Resizing...");
108            guest.resize(surface.width(), surface.height());
109        }
110
111        let mut busy = false;
112        for event in ostd::input::poll_events(32) {
113            busy = true;
114            if let ostd::input::InputEvent::MouseButton { .. } = event {
115                let _ = ostd::input::request_focus();
116            }
117            for guest_event in bridge.translate(&event, surface.width(), surface.height()) {
118                outbox.push(guest_event);
119            }
120        }
121        busy |= outbox.flush(guest);
122
123        while let Some(event) = guest.next_event() {
124            busy = true;
125            match event {
126                VmmEvent::Scanout { grant, width, height } => {
127                    frame = Some(GuestFrame::new(grant, guest.hypervisor_tid(), width, height));
128                }
129                VmmEvent::Damage { x, y, width, height } => {
130                    let Some(frame) = &frame else { continue };
131                    let rect = Rect { x: x as i32, y: y as i32, w: width, h: height };
132                    let (w, h, stride) = (surface.width(), surface.height(), surface.stride());
133                    if let Some(copied) = frame.copy_to(rect, surface.pixels_mut(), stride, w, h) {
134                        surface.damage(copied);
135                    }
136                }
137                VmmEvent::Exited { .. } => {
138                    frame = None;
139                    paint_status(&mut surface, &mut fonts, "The Linux guest has stopped.");
140                }
141            }
142        }
143        if busy {
144            ostd::syscall::sys_yield();
145        } else {
146            let _ = ostd::syscall::sys_set_timer(1);
147        }
148    }
149    ostd::syscall::sys_exit(0);
150}
