1//! `GlyphAtlas` — on-demand outline rasterization for ViUI.
2//!
3//! Uses ab_glyph with its no_std/libm backend. Bundled font data is borrowed;
4//! only requested glyph bitmaps remain resident, not all font outlines.
5//!
6//! The bounded bitmap cache is keyed by `(glyph_id, px_bits)`; unsupported
7//! scalars share one missing-glyph entry. Cached paints do not allocate.
8
9use alloc::collections::BTreeMap;
10use alloc::vec::Vec;
11use ab_glyph::{point, Font, FontArc, FontRef, FontVec};
12use crate::typography::normalized_chars;
13

16/// Per-glyph layout and rasterization metrics.
17///
18/// Y axis is **math convention** (up = positive):
19/// - `ymin` = bottom of bounding box relative to baseline (≤ 0 for descenders)
20/// - top of bounding box = `ymin + height`
21///
22/// In screen coords (y-down): glyph_top_screen = `baseline_y - (ymin + height)`.
23#[derive(Copy, Clone, Debug)]
24pub struct GlyphMetrics {
25    /// Horizontal offset from pen position to glyph left edge (pixels).
26    pub xmin: i32,
27    /// Distance from baseline to glyph bottom (math y-up; negative for descenders).
28    pub ymin: i32,
29    /// Rasterized bitmap width in pixels.
30    pub width: usize,
31    /// Rasterized bitmap height in pixels.
32    pub height: usize,
33    /// Advance width — how far to move the pen after this glyph.
34    pub advance_width: f32,
35}
36
37// ─── GlyphAtlas ──────────────────────────────────────────────────────────────
38
39struct CachedGlyph {
40    metrics: GlyphMetrics,
41    /// 1 byte per pixel, linear coverage (0 = transparent, 255 = solid).
42    bitmap: Vec<u8>,
43}
44
45// A long-lived console or document must not retain every scalar/size ever seen.
46const MAX_CACHED_GLYPHS: usize = 512;
47const MAX_CACHE_BYTES: usize = 256 * 1024;
48pub const MAX_FONT_PX: f32 = 128.0;
49
50pub fn supported_size(px: f32) -> bool {
51    px.is_finite() && px > 0.0 && px <= MAX_FONT_PX
52}
53
54fn require_supported_size(px: f32) {
55    assert!(supported_size(px), "font size must be finite and in (0, 128] pixels");
56}
57
58/// Scalable outline rasterizer with an application-owned glyph bitmap cache.
59///
60/// Rasterized bitmaps are cached by `(codepoint, size_bits)` — repeated draws
61/// for the same character at the same size are essentially free.
62///
63/// # Usage
64/// ```no_run
65/// use ostd::font_atlas::GlyphAtlas;
66///
67/// # fn rasterize(font_bytes: &[u8]) {
68/// let mut atlas = GlyphAtlas::new(font_bytes).expect("valid font");
69/// let (metrics, bitmap) = atlas.rasterize('A', 16.0);
70/// # let _ = (metrics, bitmap);
71/// # }
72/// ```
73pub struct GlyphAtlas {
74    font: FontArc,
75    units_per_em: f32,
76    cache: BTreeMap<(u32, u32), CachedGlyph>,
77    cache_bytes: usize,
78}
79
80impl GlyphAtlas {
81    /// Load an owned copy of arbitrary TrueType/OpenType data.
82    /// Bundled static assets should use `from_static` to avoid copying.
83    pub fn new(font_bytes: &[u8]) -> Option<Self> {
84        FontVec::try_from_vec(font_bytes.to_vec()).ok()
85            .and_then(|font| Self::from_font(FontArc::new(font)))
86    }
87
88    /// Borrow embedded static data; no duplicate TTF buffer or eager outlines.
89    pub fn from_static(font_bytes: &'static [u8]) -> Option<Self> {
90        FontRef::try_from_slice(font_bytes).ok()
91            .and_then(|font| Self::from_font(FontArc::new(font)))
92    }
93
94    fn from_font(font: FontArc) -> Option<Self> {
95        let units_per_em = font.units_per_em()?;
96        Some(Self { font, units_per_em, cache: BTreeMap::new(), cache_bytes: 0 })
97    }
98
99    fn scale(&self, px: f32) -> f32 {
100        require_supported_size(px);
101        // API sizes are pixels per em, whereas ab_glyph uses ascent-descent.
102        px * self.font.height_unscaled() / self.units_per_em
103    }
104
105    /// Rasterize `c` at `px` pixels tall.
106    ///
107    /// Returns `(GlyphMetrics, coverage_bitmap)`.  The bitmap is 1 byte per pixel
108    /// stored row-major, top-left origin, width × height bytes total.
109    /// Repeated calls for the same `(c, px)` return the cached result.
110    pub fn rasterize(&mut self, c: char, px: f32) -> (GlyphMetrics, &[u8]) {
111        // Unsupported scalars share the font's missing-glyph cache entry.
112        let key = (self.font.glyph_id(c).0 as u32, px.to_bits());
113        if !self.cache.contains_key(&key)
114            && (self.cache.len() >= MAX_CACHED_GLYPHS || self.cache_bytes >= MAX_CACHE_BYTES)
115        {
116            self.cache.clear();
117            self.cache_bytes = 0;
118        }
119        let font = &self.font;
120        let scale = self.scale(px);
121        let units_per_em = self.units_per_em;
122        let cache_bytes = &mut self.cache_bytes;
123        let g = self.cache.entry(key).or_insert_with(|| {
124            let id = font.glyph_id(c);
125            let advance_width = font.h_advance_unscaled(id) * px / units_per_em;
126            let mut metrics = GlyphMetrics {
127                xmin: 0, ymin: 0, width: 0, height: 0, advance_width,
128            };
129            let mut bitmap = Vec::new();
130            if px > 0.0 {
131                if let Some(outlined) = font.outline_glyph(id.with_scale_and_position(scale, point(0.0, 0.0))) {
132                    let bounds = outlined.px_bounds();
133                    metrics.xmin = bounds.min.x as i32;
134                    metrics.ymin = -(bounds.max.y as i32);
135                    metrics.width = bounds.width() as usize;
136                    metrics.height = bounds.height() as usize;
137                    let pixel_count = metrics.width.checked_mul(metrics.height)
138                        .filter(|&count| count <= 64 * 1024)
139                        .expect("glyph exceeds the 64KiB rasterization budget");
140                    bitmap.resize(pixel_count, 0);
141                    outlined.draw(|x, y, coverage| {
142                        bitmap[y as usize * metrics.width + x as usize] =
143                            (coverage * 255.0 + 0.5) as u8;
144                    });
145                }
146            }
147            *cache_bytes += bitmap.len() + core::mem::size_of::<CachedGlyph>();
148            CachedGlyph { metrics, bitmap }
149        });
150        (g.metrics, &g.bitmap)
151    }

154    /// Distance from baseline to the ascender line in screen pixels (positive, y-down).
155    ///
156    /// Use as: `baseline_screen_y = origin_y + atlas.ascender(px)`.
157    pub fn ascender(&self, px: f32) -> f32 {
158        require_supported_size(px);
159        self.font.ascent_unscaled() * px / self.units_per_em
160    }
161
162    /// Total line height (ascent − descent + line_gap) in pixels.
163    pub fn line_height(&self, px: f32) -> f32 {
164        require_supported_size(px);
165        (self.font.height_unscaled() + self.font.line_gap_unscaled()) * px / self.units_per_em
166    }
167
168    /// Horizontal advance without outlining or allocating.
169    pub fn advance(&self, c: char, px: f32) -> f32 {
170        require_supported_size(px);
171        self.font.h_advance_unscaled(self.font.glyph_id(c)) * px / self.units_per_em
172    }
173
174    /// Pair adjustment in pixels; identical in measuring and painting.
175    pub fn kerning(&self, left: char, right: char, px: f32) -> f32 {
176        require_supported_size(px);
177        self.font.kern_unscaled(self.font.glyph_id(left), self.font.glyph_id(right)) * px / self.units_per_em
178    }
179
180    /// Advance of one NFC-normalized line, including pair kerning.
181    pub fn measure(&self, text: &str, px: f32) -> f32 {
182        let mut width = 0.0;
183        let mut previous = None;
184        for c in normalized_chars(text) {
185            if let Some(left) = previous {
186                width += self.kerning(left, c, px);
187            }
188            width += self.advance(c, px);
189            previous = Some(c);
190        }
191        width
192    }
193
194    /// Paint a normalized line using cached grayscale coverage and BGRA
195    /// source-over compositing. `y` is the line top, not the glyph top.
196    /// Returns the same pen advance as `measure`.
197    #[allow(clippy::too_many_arguments)]
198    pub fn draw_text(
199        &mut self,
200        pixels: &mut [u8],
201        width: u32,
202        height: u32,
203        stride: usize,
204        x: f32,
205        y: f32,
206        text: &str,
207        px: f32,
208        color: [u8; 4],
209        clip: (i32, i32, i32, i32),
210    ) -> f32 {
211        if stride == 0 || color[3] == 0 {
212            return self.measure(text, px);
213        }
214        let surface_w = (width as usize).min(stride / 4).min(i32::MAX as usize) as i32;
215        let surface_h = (height as usize).min(pixels.len() / stride).min(i32::MAX as usize) as i32;
216        let clip = (
217            clip.0.max(0), clip.1.max(0),
218            clip.2.min(surface_w), clip.3.min(surface_h),
219        );
220        if clip.0 >= clip.2 || clip.1 >= clip.3 {
221            return self.measure(text, px);
222        }
223        let baseline = y + self.ascender(px);
224        let mut pen = x;
225        let mut previous = None;
226        for c in normalized_chars(text) {
227            if let Some(left) = previous {
228                pen += self.kerning(left, c, px);
229            }
230            let (metrics, bitmap) = self.rasterize(c, px);
231            let gx = round_pixel(pen + metrics.xmin as f32);
232            let gy = round_pixel(baseline - (metrics.ymin as f32 + metrics.height as f32));
233            let x0 = gx.max(clip.0);
234            let y0 = gy.max(clip.1);
235            let x1 = gx.saturating_add(metrics.width.min(i32::MAX as usize) as i32).min(clip.2);
236            let y1 = gy.saturating_add(metrics.height.min(i32::MAX as usize) as i32).min(clip.3);
237            for screen_y in y0..y1 {
238                let source_row = (screen_y - gy) as usize * metrics.width;
239                let destination_row = screen_y as usize * stride;
240                for screen_x in x0..x1 {
241                    let coverage = bitmap[source_row + (screen_x - gx) as usize];
242                    let alpha = (color[3] as u32 * coverage as u32 + 127) / 255;
243                    if alpha == 0 {
244                        continue;
245                    }
246                    let offset = destination_row + screen_x as usize * 4;
247                    blend_coverage(&mut pixels[offset..offset + 4], color, alpha);
248                }
249            }
250            pen += metrics.advance_width;
251            previous = Some(c);
252        }
253        pen - x
254    }
255}
256
257fn round_pixel(value: f32) -> i32 {
258    if value >= 0.0 { (value + 0.5) as i32 } else { (value - 0.5) as i32 }
259}
260
261fn blend_coverage(destination: &mut [u8], color: [u8; 4], alpha: u32) {
262    if alpha == 255 {
263        destination.copy_from_slice(&[color[0], color[1], color[2], 255]);
264        return;
265    }
266    let inverse = 255 - alpha;
267    let destination_alpha = destination[3] as u32;
268    if destination_alpha == 255 {
269        for channel in 0..3 {
270            destination[channel] = ((color[channel] as u32 * alpha
271                + destination[channel] as u32 * inverse + 127) / 255) as u8;
272        }
273        return;
274    }
275    let alpha_numerator = alpha * 255 + destination_alpha * inverse;
276    for channel in 0..3 {
277        destination[channel] = ((color[channel] as u32 * alpha * 255
278            + destination[channel] as u32 * destination_alpha * inverse
279            + alpha_numerator / 2) / alpha_numerator) as u8;
280    }
281    destination[3] = ((alpha_numerator + 127) / 255) as u8;
282}

295    fn bundled_faces_cover_vietnamese_and_keep_terminal_fixed_pitch() {
296        for face in [
297            FontFace::UiRegular, FontFace::UiSemibold, FontFace::UiItalic,
298            FontFace::UiSemiboldItalic, FontFace::MonoRegular, FontFace::MonoBold,

315    #[test]
316    fn decomposed_vietnamese_measures_and_paints_identically() {
317        let mut fonts = TextFonts::new();
318        let text = "Tiếng Việt: ắ ậ ễ ỡ ự Đ";
319        let decomposed: String = text.nfd().collect();
320        let face = FontFace::UiRegular;
321        assert_eq!(fonts.measure(face, text, 18.0), fonts.measure(face, &decomposed, 18.0));
322        let mut composed_pixels = vec![255; 420 * 48 * 4];
323        let mut decomposed_pixels = composed_pixels.clone();
324        let measured = fonts.measure(face, text, 18.0);
325        for (pixels, input) in [
326            (&mut composed_pixels, text),
327            (&mut decomposed_pixels, decomposed.as_str()),
328        ] {
329            let advance = fonts.draw_text(
330                pixels, 420, 48, 420 * 4, 4.0, 4.0, input, face, 18.0,
331                [30, 60, 90, 255], (0, 0, 420, 48),
332            );
333            assert!((advance - measured).abs() < 0.001);
334        }
335        assert_eq!(composed_pixels, decomposed_pixels);
336        assert!(composed_pixels.chunks_exact(4).any(|p| p[0] > 30 && p[0] < 255));
337        assert!(composed_pixels.chunks_exact(4).all(|p| p[3] == 255));
338    }

340    #[test]
341    fn clipping_preserves_outside_pixels_and_stride_padding() {
342        let mut atlas = GlyphAtlas::from_static(FontFace::UiItalic.bytes()).unwrap();
343        let width = 48;
344        let height = 32;
345        let stride = width * 4 + 12;
346        let mut pixels = vec![37; stride * height];
347        let before = pixels.clone();
348        atlas.draw_text(
349            &mut pixels, width as u32, height as u32, stride, -9.0, -3.0,
350            "Tiếng Việt", 24.0, [20, 100, 220, 180], (5, 6, 35, 25),
351        );
352        for y in 0..height {
353            for x in 0..width {
354                if !(5..35).contains(&x) || !(6..25).contains(&y) {
355                    let offset = y * stride + x * 4;
356                    assert_eq!(&pixels[offset..offset + 4], &before[offset..offset + 4]);
357                }
358            }
359            assert_eq!(&pixels[y * stride + width * 4..(y + 1) * stride],
360                &before[y * stride + width * 4..(y + 1) * stride]);
361        }
362        assert!(pixels[6 * stride + 5 * 4..25 * stride].iter()
363            .zip(&before[6 * stride + 5 * 4..25 * stride]).any(|(a, b)| a != b));
364    }

385    #[test]
386    fn long_unicode_stream_reclaims_cache_and_shares_missing_glyph() {
387        let mut atlas = GlyphAtlas::from_static(INTER_REGULAR).unwrap();
388        for value in 0x20..0x2000 {
389            if let Some(c) = char::from_u32(value) {
390                atlas.rasterize(c, 16.0);
391                assert!(atlas.cache.len() <= MAX_CACHED_GLYPHS);
392                assert!(atlas.cache_bytes < MAX_CACHE_BYTES + 4096);
393            }
394        }
395        atlas.cache.clear();
396        atlas.cache_bytes = 0;
397        for value in 0xf0000..0xf1000 {
398            atlas.rasterize(char::from_u32(value).unwrap(), 16.0);
399        }
400        assert_eq!(atlas.cache.len(), 1);
401        let (metrics, bitmap) = atlas.rasterize('ế', 16.0);
402        assert!(metrics.width > 0 && metrics.height > 0);
403        assert!(bitmap.iter().any(|&coverage| coverage > 0 && coverage < 255));
404    }

413    fn invalid_sizes_are_rejected_before_raster_allocation() {
414        for px in [0.0, -1.0, f32::NAN, f32::INFINITY, 100_000.0] {
