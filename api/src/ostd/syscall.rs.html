82/// Submit an owner-authenticated V1 request. Transport success is exactly zero;
83/// operation status remains in the result record, including typed errors.
84pub fn sys_cell_session_v1(request: &[u8], result: &mut [u8]) -> SyscallResult {
85    // SAFETY: both slices remain live for the call; the exclusive result borrow
86    // prevents overlap with the request. The kernel validates both full ranges.
87    let ret = unsafe {
88        syscall(
89            ViSyscall::CellSessionV1,
90            request.as_ptr() as usize,
91            request.len(),
92            result.as_mut_ptr() as usize,
93            result.len(),
94        )
95    };
96    if ret == 0 {
97        SyscallResult::Ok(0)
98    } else {
99        SyscallResult::Err(SyscallError::Unknown)
100    }
101}

103/// Issue syscall `id` with four register arguments and return the raw result.
104///
105/// # Safety
106/// Every argument `id` interprets as a cell address must name memory valid for
107/// the kernel's access (read for a source, write for a destination) over the
108/// length that syscall uses, and no live Rust reference other than the
109/// caller's own exclusive borrow may observe bytes the kernel writes, for the
110/// whole call. Register-only arguments carry no obligation.
111#[inline(always)]
112pub(crate) unsafe fn syscall(id: ViSyscall, a0: usize, a1: usize, a2: usize, a3: usize) -> isize {
113    let mut ret: isize;
114    #[cfg(target_arch = "riscv64")]
115    asm!(
116        "ecall",
117        inlateout("a0") a0 => ret,
118        in("a1") a1,
119        in("a2") a2,
120        in("a3") a3,
121        in("a7") (id as usize),
122        options(nostack, preserves_flags)
123    );
124    // ViCell ARM64 ABI: x0=syscall_nr, x1=a0, x2=a1, x3=a2, x4=a3; ret in x0.
125    #[cfg(target_arch = "aarch64")]
126    asm!(
127        "svc #0",
128        inlateout("x0") id as usize => ret,
129        in("x1") a0,
130        in("x2") a1,
131        in("x3") a2,
132        in("x4") a3,
133        options(nostack, preserves_flags)
134    );
135    // ViCell x86_64 ABI: RAX=syscall_nr (a7 slot), RDI=a0, RSI=a1, RDX=a2, R10=a3
136    // (R10, not RCX — SYSCALL clobbers RCX to save user RIP).  Return value in RAX.
137    // SAFETY: SYSCALL transitions to kernel via LSTAR; the kernel validates all
138    // pointer arguments before use.  RCX and R11 are clobbered by the hardware
139    // (RCX ← user RIP, R11 ← user RFLAGS) and must be declared as outputs.
140    #[cfg(target_arch = "x86_64")]
141    asm!(
142        "syscall",
143        inlateout("rax") id as usize => ret,
144        in("rdi") a0,
145        in("rsi") a1,
146        in("rdx") a2,
147        in("r10") a3,
148        lateout("rcx") _,   // clobbered by SYSCALL (saves user RIP)
149        lateout("r11") _,   // clobbered by SYSCALL (saves user RFLAGS)
150        options(nostack)
151    );
152    ret
153}

155/// Invoke a syscall by raw numeric id (bypasses the `ViSyscall` enum).
156///
157/// Used for block I/O (ids 500/501) which intentionally have no `ViSyscall`
158/// entry — keeping them out of `libs/api` avoids the Interface-is-Sacred
159/// 2x-confirmation gate. The kernel dispatches them via the numeric fallback
160/// in `ViCell_syscall_dispatch`.
161#[inline(always)]
162unsafe fn syscall_raw(id: usize, a0: usize, a1: usize, a2: usize, a3: usize) -> isize {
163    let mut ret: isize;
164    #[cfg(target_arch = "riscv64")]
165    asm!(
166        "ecall",
167        inlateout("a0") a0 => ret,
168        in("a1") a1,
169        in("a2") a2,
170        in("a3") a3,
171        in("a7") id,
172        options(nostack, preserves_flags)
173    );
174    #[cfg(target_arch = "aarch64")]
175    asm!(
176        "svc #0",
177        inlateout("x0") id => ret,
178        in("x1") a0,
179        in("x2") a1,
180        in("x3") a2,
181        in("x4") a3,
182        options(nostack, preserves_flags)
183    );
184    // SAFETY: same contract as the typed `syscall` variant above.
185    // `id` is a raw numeric syscall number (e.g. 500/501/502/503 for block I/O).
186    #[cfg(target_arch = "x86_64")]
187    asm!(
188        "syscall",
189        inlateout("rax") id => ret,
190        in("rdi") a0,
191        in("rsi") a1,
192        in("rdx") a2,
193        in("r10") a3,
194        lateout("rcx") _,
195        lateout("r11") _,
196        options(nostack)
197    );
198    ret
199}

202///

204/// `buf` is filled only when this returns `true`.

223/// Request a clean system shutdown via the kernel's SBI SRST path. Never returns when the
224/// kernel accepts it.
225///
226/// Raw syscall 502 — intentionally absent from `ViSyscall`/`libs/api` to avoid the
227/// ABI 2x-confirm gate (same pattern as `sys_blk_read`/`sys_blk_write` above). The kernel

229pub fn sys_shutdown() -> ! {
230    sys_shutdown_denied();
231    // Unreachable when accepted: the kernel never returns from shutdown. Spin to satisfy `-> !`.
232    loop {
233        sys_yield();
234    }

354/// Outcome of a futex wait.
355#[derive(Debug, Copy, Clone, PartialEq, Eq)]
356pub enum FutexWaitOutcome {
357    /// Another task woke this wait.
358    Woken,
359    /// The word did not hold the expected value, so the caller never parked (or

363    TimedOut,

366/// Park the calling task until another task wakes the same word, the word no
367/// longer holds `expected`, or `timeout_ticks` (10 ms ticks) elapse. A zero
368/// timeout blocks indefinitely.
369///
370/// Waiters are matched by `(address space, generation, address)`: two Tier 2
371/// domains that use the same virtual address never wake each other.
372pub fn sys_futex_wait(
373    addr: usize,
374    expected: u32,
375    timeout_ticks: u64,
376) -> Result<FutexWaitOutcome, SyscallError> {
377    unsafe {
378        let ret = syscall(
379            ViSyscall::FutexWait,
380            addr,
381            expected as usize,
382            timeout_ticks as usize,
383            0,
384        );
385        match ret {
386            0 => Ok(FutexWaitOutcome::Woken),
387            1 => Ok(FutexWaitOutcome::ValueMismatch),
388            2 => Ok(FutexWaitOutcome::TimedOut),
389            SYSCALL_DENIED => Err(SyscallError::PermissionDenied),
390            _ => Err(SyscallError::Unknown),
391        }
392    }
393}

569/// Spawn a cell by loading its ELF from a VFS path (e.g. `/bin/cellbox`).
570///
571/// The kernel reads the ELF from disk or the bootstrap table, parses it,
572/// and spawns a new task.  Returns the new cell's task ID on success.
573///
574/// # Errors
575/// Returns `SyscallError::Unknown` if the path is not found or the ELF is invalid.
576pub fn sys_spawn_from_path(path: &str) -> SyscallResult {
577    // Post-boot (VFS registered): read the cell ELF from VFS into a Grant (reaching
578    // the /bin cell-store overlay + VIFS1 bootstrap cells) and spawn from the bytes,
579    // so the kernel loader needs no disk access for the spawn. Falls back to the raw
580    // kernel bootstrap reader when VFS is not yet up (early boot), the VFS read fails,
581    // or the VFS-loaded ELF is rejected. OOM is returned immediately: retrying the same
582    // allocation through bootstrap would add pressure and hide the typed failure.
583    if let Some(vfs_tid) = crate::service::provider_tid(crate::service::service::VFS) {
584        if let Ok((grant_id, len)) = crate::fs::read_full_via_grant(path, vfs_tid) {
585            let r = sys_spawn_from_elf(grant_id, len, path);
586            sys_grant_free(grant_id);
587            match r {
588                SyscallResult::Ok(_) | SyscallResult::Err(SyscallError::OutOfMemory) => return r,
589                SyscallResult::Err(_) => {}
590            }
591            // VFS read OK but spawn failed → fall through to bootstrap (belt-and-suspenders).
592        }
593    }
594    // Bootstrap path: raw SpawnFromPath syscall (VIFS1 / P2 table, VFS-independent).
595    sys_spawn_from_path_raw(path)
596}

598/// The kernel's own path route, with no VFS read in front of it.
599///
600/// This resolves VIFS1 and the P2 table without any block driver, which makes it the
601/// only route that can launch a cell whose reviewed launch edge carries authority —
602/// the VFS-loaded (`SpawnFromElf`) route refuses a non-empty child ceiling on
603/// caller-supplied bytes. A caller that must fall back between the two routes has to
604/// re-publish its command line between attempts: a rejected attempt discards the
605/// staged argv by design (`StagedSpawnArgvCleanup`).
606pub fn sys_spawn_from_path_raw(path: &str) -> SyscallResult {
607    // SAFETY: path is a valid UTF-8 str; kernel copies it out before returning.
608    unsafe {
609        let ret = syscall(
610            ViSyscall::SpawnFromPath,
611            path.as_ptr() as usize,
612            path.len(),
613            0,
614            0,
615        );
616        decode_spawn_result(ret)
617    }
618}

620/// Spawn a cell from ELF bytes in a caller-owned Grant region (G2 loader redesign).
621///
622/// `grant_id` must be a `sys_grant_alloc` region the caller owns, holding the full
623/// ELF image (`len` bytes). `path_hint` drives the kernel's `/bin/` privilege check,
624/// operator-policy lookup, and measurement label — advisory only (lying about it can
625/// only lose privilege). The kernel applies the IDENTICAL spawn trust gate as
626/// `sys_spawn_from_path`, so the source of the bytes is irrelevant to trust.

1707///

1709/// Returns the bytes written into `out` — decode them with
1710/// [`api::telemetry::TelemetrySnapshot::parse`]. Requires a `GetProcs2` or

1712/// `observe_tasks` right.
1713///
1714/// # Errors
1715/// - `BufferTooSmall` / `InvalidCommand`: refused locally, before the call, for
1716///   a buffer shorter than a header or an undefined flag bit.
1717/// - `PermissionDenied`: the kernel refused a well-formed call, which is the
1718///   dispatch gate (no `observe_tasks`, or no bit 55).
1719/// - `OutOfMemory`: the kernel could not allocate the transient capture.
1720pub fn sys_telemetry_snapshot(out: &mut [u64], flags: usize) -> Result<usize, SyscallError> {
1721    let len = core::mem::size_of_val(out);
1722    if len < api::telemetry::HEADER_LEN {
1723        return Err(SyscallError::BufferTooSmall);
1724    }
1725    if flags & !api::telemetry::request::ALL != 0 {
1726        return Err(SyscallError::InvalidCommand);
1727    }
1728    // SAFETY: `out` is a live, aligned, writable slice of `len` bytes and the
1729    // kernel writes at most `len` bytes into it.
1730    let ret = unsafe {
1731        syscall(
1732            ViSyscall::TelemetrySnapshot,
1733            out.as_mut_ptr() as usize,
1734            len,
1735            flags,
1736            0,
1737        )
1738    };
1739    match ret {
1740        written if written >= 0 && written as usize <= len => Ok(written as usize),
1741        SYSCALL_DENIED => Err(SyscallError::PermissionDenied),
1742        SYSCALL_OOM => Err(SyscallError::OutOfMemory),
1743        _ => Err(SyscallError::Unknown),
1744    }
1745}

1844/// # Errors
1845/// Returns `Err` when the caller is not the registered compositor, the GPU
1846/// cursor hardware is unavailable, or the sprite is not readable caller memory.
1847pub fn sys_gpu_cursor(
1848    op: usize,
1849    sprite: *const u8,
1850    x: u32,
1851    y: u32,
1852    hot_x: u32,
1853    hot_y: u32,
1854) -> Result<(), SyscallError> {
1855    let xy = (((x as usize) & 0xFFFF) << 16) | ((y as usize) & 0xFFFF);
1856    let hot = (((hot_x as usize) & 0xFFFF) << 16) | ((hot_y as usize) & 0xFFFF);
1857    // SAFETY: for op=0 sprite must point to a 64*64*4-byte BGRA buffer owned by the
1858    // caller; the kernel validates the length and reads exactly that many bytes.
1859    // For op=1 sprite is ignored; passing null or a valid pointer are both safe.
1860    let ret = unsafe { syscall(ViSyscall::GpuCursor, op, sprite as usize, xy, hot) };
1861    if ret >= 0 {
1862        Ok(())
1863    } else {
1864        Err(SyscallError::Unknown)
1865    }
1866}

2202/// One rectangle copy out of a Grant: `rows` rows of `row_len` bytes, the
2203/// first starting at byte `src_offset` of the Grant and `dst_offset` of the
2204/// destination, rows `src_stride` / `dst_stride` bytes apart.
2205#[derive(Clone, Copy, Debug)]
2206pub struct GrantRows {
2207    pub src_offset: usize,
2208    pub src_stride: usize,
2209    pub dst_offset: usize,
2210    pub dst_stride: usize,
2211    pub row_len: usize,
2212    pub rows: usize,
2213}
2214
2215impl GrantRows {
2216    /// Whether every row lies inside a `src_len`-byte Grant and `dst_len`-byte destination.
2217    fn fits(&self, src_len: usize, dst_len: usize) -> Option<()> {
2218        if self.rows == 0 {
2219            return Some(());
2220        }
2221        let last = self.rows - 1;
2222        let src_end = self
2223            .src_offset
2224            .checked_add(last.checked_mul(self.src_stride)?)?
2225            .checked_add(self.row_len)?;
2226        let dst_end = self
2227            .dst_offset
2228            .checked_add(last.checked_mul(self.dst_stride)?)?
2229            .checked_add(self.row_len)?;
2230        (src_end <= src_len && dst_end <= dst_len).then_some(())
2231    }
2232}

2400/// one with [`sys_request_mmio`]. The VirtIO window is kernel-only until a claim
2401/// grants it, and a claim is exclusive for the Cell's lifetime, so probing by
2402/// claiming would lock sibling drivers out of their own slots.
2403///
2404/// `base` must be exactly one of the board's VirtIO MMIO slot bases.
2405///
2406/// # Errors
2407/// - `FileNotFound`: no VirtIO transport answers at `base`.
2408/// - `InvalidCommand`: `base` is not a VirtIO MMIO slot base.
2409/// - `PermissionDenied`: the caller holds no `PcieDriverCap`, did not declare
2410///   `RequestMmio` (allowlist bit 40), or the architecture has no VirtIO MMIO
2411///   (x86_64).
2412pub fn sys_probe_virtio_mmio(
2413    base: usize,
2414) -> Result<api::syscall::virtio_probe::SlotIdentity, SyscallError> {
2415    // SAFETY: no pointer crosses the boundary; the kernel validates `base`
2416    // against its own slot list and the caller's capability at dispatch.
2417    let ret = unsafe { syscall(ViSyscall::ProbeVirtioMmio, base, 0, 0, 0) };
2418    match ret {
2419        // Typed sentinels: `usize::MAX - 2` no transport, `usize::MAX - 3` not a slot.
2420        -3 => Err(SyscallError::FileNotFound),
2421        -4 => Err(SyscallError::InvalidCommand),
2422        ret => {
2423            api::syscall::virtio_probe::decode(ret as usize).ok_or(SyscallError::PermissionDenied)
2424        }
2425    }
2426}

2573/// Submit a wait against a set of kernel completion sources and block until one
2574/// of them reports.
2575///
2576/// The call reserves a slot on this cell's completion queue before it parks, so
2577/// a source always has somewhere to put the result. Deadline behavior depends
2578/// on the sources: without `TIMER`, zero waits indefinitely and an expired
2579/// finite timeout returns [`WaitCompletionResult::NoRecord`]; with `TIMER`, a
2580/// nonzero duration is required and expiry returns a `TIMER` completion with
2581/// `result = 0`. An `IPC` completion's `result` is the token of one owned
2582/// operation that reached its terminal; each terminal is reported once. An
2583/// `IRQ` completion (x86_64) reports one fired MSI/MSI-X line this task holds
2584/// from [`sys_alloc_pcie_irq`]; `result` is the line.
2585///
2586/// `mask` is a non-empty set of `api::completion::source` bits; the record
2587/// names exactly one of them.
2588///
2589/// # Errors
2590/// [`WaitCompletionResult::ErrorOrInvalid`] preserves a rejected syscall return
2591/// and also reports a claimed record that does not decode as a completion.
2592///
2593/// Requires `WaitForEvent` or `WaitCompletion` in the cell's `declare_syscalls!`
2594/// list; both name the same authority.
2595pub fn sys_wait_completion_detailed(mask: u32, timeout_ticks: u64) -> WaitCompletionResult {
2596    let mut record = [0u8; COMPLETION_LEN];
2597    // SAFETY: `record` is a live, exclusively borrowed stack buffer of exactly
2598    // COMPLETION_LEN bytes, which is the length this syscall writes; the kernel
2599    // writes nothing when it returns 0.
2600    let ret = unsafe {
2601        syscall(
2602            ViSyscall::WaitCompletion,
2603            mask as usize,
2604            timeout_ticks as usize,
2605            (timeout_ticks >> 32) as usize,
2606            record.as_mut_ptr() as usize,
2607        )
2608    };
2609    decode_wait_completion_result(ret, &record)
2610}
2611
2612fn decode_wait_completion_result(
2613    ret: isize,
2614    record: &[u8; COMPLETION_LEN],
2615) -> WaitCompletionResult {
2616    match ret {
2617        0 => WaitCompletionResult::NoRecord,
2618        1 => ViCompletion::from_bytes(record).map_or(
2619            WaitCompletionResult::ErrorOrInvalid(ret),
2620            WaitCompletionResult::Completion,
2621        ),
2622        _ => WaitCompletionResult::ErrorOrInvalid(ret),
2623    }
2624}

2626/// Backward-compatible completion wait.
2627///
2628/// Returns a completion only for a valid written record. As before, raw zero,
2629/// rejected syscalls, unexpected returns, and malformed records all map to

2631pub fn sys_wait_completion(mask: u32, timeout_ticks: u64) -> Option<ViCompletion> {

2909/// PCIe device descriptor written by `sys_find_pcie_device`.
2910///
2911/// Layout is `#[repr(C)]` to match the kernel's byte-wise write sequence:
2912/// `bdf(u32) + found(u32) + bar0_base(u64) + bar0_len(u64) + bar_mem_base(u64)
2913/// + bar_mem_len(u64) + vendor_id(u16) + device_id(u16)` = 44 bytes padded to
2914/// 48 by the trailing `u64` alignment.
2915///
2916/// `bar_mem_*` were **appended** so the first-BAR fields keep their exact
2917/// meaning: a device whose BAR0 is I/O (ICH9 AHCI, whose ABAR is BAR5) reports
2918/// zero `bar0_base` and its MMIO window in `bar_mem_base`. Every cell that
2919/// reads this record is rebuilt by the same packaging pass, so no stale 24-byte
2920/// reader survives a rebuild.
2921///
2922/// `vendor_id`/`device_id` were appended the same way (phase 04a of the x86 PC
2923/// lane) so a NIC cell can confirm the device the kernel matched by Ethernet
2924/// class triple is actually its own family before it programs a register model.
2925/// A device with no identity reads `0:0`, which matches no family.
2926#[repr(C)]
2927pub struct PcieDeviceInfo {
2928    /// PCIe Requester ID: `bus<<8 | dev<<3 | fn`. Zero means not populated.
2929    pub bdf: u32,
2930    /// 1 if the device was found, 0 otherwise.
2931    pub found: u32,
2932    /// BAR0 physical base address (= virtual address in SAS); 0 when BAR0 is I/O.
2933    pub bar0_base: u64,
2934    /// BAR0 size in bytes.  At least 0x4000 (16 KiB) for NVMe controllers.
2935    pub bar0_len: u64,
2936    /// First *memory* BAR physical base address; equals `bar0_base` when BAR0 is
2937    /// memory. ICH9 AHCI exposes its ABAR at BAR5, so this is the only field a
2938    /// SATA driver can use for MMIO.
2939    pub bar_mem_base: u64,
2940    /// First memory BAR size in bytes (0x4000 fallback when the size is unknown).
2941    pub bar_mem_len: u64,
2942    /// PCI vendor ID (0 when the kernel could not read the identity).
2943    pub vendor_id: u16,
2944    /// PCI device ID (0 when the kernel could not read the identity).
2945    pub device_id: u16,
2946}
