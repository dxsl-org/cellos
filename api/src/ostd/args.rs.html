3//! Command-line argument helpers for Cell applications.
4//!
5//! The shell (or any spawner) calls [`sys_set_spawn_args`][crate::syscall::sys_set_spawn_args]
6//! before spawning a cell; the kernel moves the bytes into a per-task private
7//! slot so back-to-back spawns cannot race.  [`args()`] reads and parses that
8//! slot.  The slot is consumed on first read — subsequent calls return an empty
9//! `Vec`.
10
11extern crate alloc;
12
13use alloc::string::String;
14use alloc::vec::Vec;
15
16const STRUCTURED_ARGV_PREFIX: &str = "\0argv1\0";
17pub const MAX_SPAWN_ARGV_BYTES: usize = 512;
18
19/// Publish structured arguments for the next spawned cell.
20///
21/// Each argument is preserved byte-for-byte, including spaces and empty
22/// strings. Returns `false` without publishing when an argument contains NUL,
23/// the encoded payload exceeds [`MAX_SPAWN_ARGV_BYTES`], or allocation fails.
24/// The caller must invoke this immediately before spawning the target cell.
25pub fn set_spawn_argv(argv: &[String]) -> bool {
26    let mut required = STRUCTURED_ARGV_PREFIX.len();
27    for arg in argv {
28        if arg.as_bytes().contains(&0) {
29            return false;
30        }
31        required = match required.checked_add(arg.len() + 1) {
32            Some(required) if required <= MAX_SPAWN_ARGV_BYTES => required,
33            _ => return false,
34        };
35    }
36    let mut encoded = String::new();
37    if encoded.try_reserve_exact(required).is_err() {
38        return false;
39    }
40    encoded.push_str(STRUCTURED_ARGV_PREFIX);
41    for arg in argv {
42        encoded.push_str(arg);
43        encoded.push('\0');
44    }
45    crate::syscall::sys_set_spawn_args(&encoded);
46    true
47}

49/// Return the command-line arguments passed to this cell by its spawner.
50///
51/// Arguments are space-separated UTF-8 tokens set by the spawner via
52/// [`sys_set_spawn_args`][crate::syscall::sys_set_spawn_args].  Returns an empty
53/// `Vec` when no args were set or after the stash has already been consumed.
54///
55/// The returned `Vec` contains only the arguments — no `argv[0]` program-name
56/// entry is prepended.
57///
58/// # Example
59/// ```no_run
60/// let args = ostd::args();
61/// match args.as_slice() {
62///     [] => ostd::io::println("no args"),
63///     [path, rest @ ..] => { /* process path … */ }
64/// }
65/// ```
66pub fn args() -> Vec<String> {
67    let mut buf = [0u8; MAX_SPAWN_ARGV_BYTES];
68    let n = crate::syscall::sys_spawn_args(&mut buf);
69    if n == 0 {
70        return Vec::new();
71    }
72    decode_spawn_args(&buf[..n])
73}
74
75fn decode_spawn_args(bytes: &[u8]) -> Vec<String> {
76    if let Some(payload) = bytes.strip_prefix(STRUCTURED_ARGV_PREFIX.as_bytes()) {
77        if payload.is_empty() {
78            return Vec::new();
79        }
80        let Some(body) = payload.strip_suffix(&[0]) else {
81            return Vec::new();
82        };
83        let mut argv = Vec::new();
84        for raw in body.split(|byte| *byte == 0) {
85            let Ok(arg) = core::str::from_utf8(raw) else {
86                return Vec::new();
87            };
88            argv.push(String::from(arg));
89        }
90        return argv;
91    }
92    let text = match core::str::from_utf8(bytes) {
93        Ok(s) => s,
94        Err(_) => return Vec::new(),
95    };
96    text.split_ascii_whitespace().map(String::from).collect()
97}

99/// Print `usage` and exit if the cell was spawned with `-h` or `--help`.
100///
101/// Reads the spawn-args stash once.  Call at the start of [`AppEvent::Init`] or
102/// via [`CellRuntime::help`] to get automatic `--help` handling with zero
103/// boilerplate in the event handler.
104pub fn check_help(usage: &str) {
105    let argv = args();
106    if argv.iter().any(|a| a == "-h" || a == "--help") {
107        crate::io::println(usage);
108        crate::syscall::sys_exit(0);
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::{decode_spawn_args, STRUCTURED_ARGV_PREFIX};
115
116    #[test]
117    fn structured_argv_preserves_spaces_and_empty_items() {
118        let mut raw = alloc::string::String::from(STRUCTURED_ARGV_PREFIX);
119        raw.push_str("two words");
120        raw.push('\0');
121        raw.push('\0');
122        assert_eq!(decode_spawn_args(raw.as_bytes()), ["two words", ""]);
123    }
124
125    #[test]
126    fn legacy_argv_remains_whitespace_separated() {
127        assert_eq!(decode_spawn_args(b"one two"), ["one", "two"]);
128    }
129}
