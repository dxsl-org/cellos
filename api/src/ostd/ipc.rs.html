1//! Typed request/reply IPC helpers — the compliant path for talking to a
2//! service cell (Spec 17 — Cell IPC Wire Contract).
3//!
4//! Prefer [`service_call`] / [`service_call_typed`] over a hand-rolled
5//! `sys_send` + `sys_recv(0)`: they recv **masked to the service tid** (Spec 17
6//! §2), so a queued input key event can never be mistaken for the reply, and
7//! they surface every failure as a typed [`IpcError`] instead of a silent empty
8//! result (Spec 17 §7).
9

23/// Why a [`service_call`] did not complete. Never silently swallowed.
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum IpcError {
26    /// The request did not fit `send_buf` (postcard encode failed).
27    Encode,
28    /// `sys_send` to the service failed (service gone / bad tid).
29    Send,
30    /// `sys_recv` returned an error.
31    Recv,
32    /// A message arrived, but from a cell other than the service — never
33    /// treated as the reply (Spec 17 §7 silent-wrong-sender guard).
34    WrongSender,
35    /// The reply bytes did not decode into the expected type.
36    Decode,
37    /// Operation token is stale, foreign, cancelled, or otherwise invalid.
38    InvalidOperation,
39    /// Reply output buffer is too short; the terminal result was not consumed.
40    BufferTooSmall,
41}

68/// Copies `request` into kernel storage, without waiting for the peer to
69/// receive or reply. Busy means no work was accepted; do not wait on this id.
70pub fn submit(peer_tid: usize, request: &[u8]) -> Result<IpcOpId, IpcSubmitError> {
71    match crate::syscall::sys_ipc_submit(peer_tid, request) {
72        token if token > 0 => Ok(token as usize),
73        -2 => Err(IpcSubmitError::Busy),
74        -3 => Err(IpcSubmitError::PeerGone),
75        _ => Err(IpcSubmitError::InvalidRequest),
76    }
77}

79/// Atomically take the exact operation's retained terminal result. The reply
80/// bytes are owned by the kernel until this succeeds; an undersized `reply`
81/// never consumes the result. A terminal result releases the operation slot.
82pub fn take(op: IpcOpId, reply: &mut [u8]) -> Result<IpcTakeResult, IpcError> {
83    let mut status = [0u8; api::syscall::IPC_STATUS_LEN];
84    match crate::syscall::sys_ipc_take(op, reply, &mut status) {
85        0 => Ok(IpcTakeResult::Pending),
86        1 => {
87            let version = u32::from_le_bytes(status[..4].try_into().unwrap());
88            let kind = u32::from_le_bytes(status[4..8].try_into().unwrap());
89            let len = u32::from_le_bytes(status[8..12].try_into().unwrap()) as usize;
90            let reserved = u32::from_le_bytes(status[12..16].try_into().unwrap());
91            if version != api::syscall::IPC_STATUS_VERSION || reserved != 0 || len > reply.len() {
92                return Err(IpcError::Decode);
93            }
94            let status = match kind {
95                api::syscall::ipc_status::REPLY => IpcTerminal::Reply,
96                api::syscall::ipc_status::PEER_GONE => IpcTerminal::PeerGone,
97                api::syscall::ipc_status::PRE_DISPATCH_TIMEOUT => IpcTerminal::PreDispatchTimeout,
98                api::syscall::ipc_status::INDETERMINATE => IpcTerminal::Indeterminate,
99                api::syscall::ipc_status::CANCELLED => IpcTerminal::Cancelled,
100                _ => return Err(IpcError::Decode),
101            };
102            Ok(IpcTakeResult::Terminal { status, len })
103        }
104        -5 => Err(IpcError::BufferTooSmall),
105        _ => Err(IpcError::InvalidOperation),
106    }
107}

109/// Sleep until any result is terminal, or a finite timeout expires. Zero
110/// waits indefinitely; the kernel rechecks results atomically with parking.
111pub fn wait(timeout_ticks: u64) -> bool {
112    crate::syscall::sys_ipc_wait(timeout_ticks) == 1
113}
114
115/// Abandon one exact operation. A pre-dispatch request becomes Cancelled;
116/// an already-dispatched request becomes Indeterminate (not rolled back).
117/// Call `take` afterwards to release the retained terminal slot.
118pub fn cancel(op: IpcOpId) -> Result<(), IpcError> {
119    match crate::syscall::sys_ipc_cancel(op) {
120        0 => Ok(()),
121        _ => Err(IpcError::InvalidOperation),
122    }
123}

125/// Token of the current async service request, if the last receive delivered
126/// one. Save it before receiving another request for a deferred reply.
127pub fn current() -> Option<IpcOpId> {
128    match crate::syscall::sys_ipc_current() {
129        token if token > 0 => Some(token as usize),
130        _ => None,
131    }
132}

134/// Reply to a deferred request bound to this exact provider incarnation.
135/// Cancelled, duplicate, dead-client and foreign tokens are rejected.
136pub fn reply(op: IpcOpId, bytes: &[u8]) -> Result<(), IpcError> {
137    match crate::syscall::sys_ipc_reply(op, bytes) {
138        0 => Ok(()),
139        _ => Err(IpcError::InvalidOperation),
140    }
141}

143/// A masked receive may still return an unrelated NotifyOnExit record.
144#[inline]
145fn reply_sender(peer_tid: usize, result: SyscallResult) -> Result<(), IpcError> {
146    match result {
147        SyscallResult::Ok(sender) if sender == peer_tid => Ok(()),
148        SyscallResult::Ok(_) => Err(IpcError::WrongSender),
149        SyscallResult::Err(_) => Err(IpcError::Recv),
150    }
151}

153/// Receive a reply from one peer. A masked `Recv` can still deliver a queued
154/// NotifyOnExit record for another task, returning that task's id and writing
155/// its exit reason into the reply buffer. Never decode it as the peer's reply.
156pub fn recv_from<'r>(peer_tid: usize, recv_buf: &'r mut [u8]) -> Result<&'r [u8], IpcError> {
157    reply_sender(peer_tid, sys_recv(peer_tid, recv_buf))?;
158    Ok(recv_buf)
159}

161/// One request/reply exchange with `service_tid`, recv **masked** to it.
162///
163/// `send_buf` encodes the request; `recv_buf` receives the reply and backs the
164/// returned slice (caller-owned so the borrow outlives the call). The reply is
165/// accepted only if it came from `service_tid` — a message from any other
166/// sender (e.g. a queued input event, Spec 17 §2) is an [`IpcError::WrongSender`],
167/// not a decode of the wrong bytes.
168pub fn service_call<'r, Req: Serialize>(
169    service_tid: usize,
170    req: &Req,
171    send_buf: &mut [u8],
172    recv_buf: &'r mut [u8],
173) -> Result<&'r [u8], IpcError> {
174    let encoded = api::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
175    if let SyscallResult::Err(_) = sys_send(service_tid, encoded) {
176        return Err(IpcError::Send);
177    }
178    // MASKED recv — Spec 17 §2. A queued death for another task can still
179    // bypass the mask; recv_from checks the returned sender before decoding.
180    recv_from(service_tid, recv_buf)
181}

217/// One bounded request/reply exchange with `service_tid`.
218///
219/// `send_buf` is used to encode `req`; an encoding failure returns
220/// [`IpcError::Encode`]. `recv_buf` receives the reply and backs the returned
221/// slice. `timeout_ticks` is the end-to-end scheduler-tick budget shared by
222/// request admission and the reply wait.
223///
224/// The encoded request is offered with nonblocking [`sys_try_send`] until the
225/// deadline. A rejected admission yields before retrying, so a service waiting
226/// on a nested dependency can return to wildcard `Recv`. Once accepted, the
227/// request is never resent and only the remaining deadline is available to the
228/// sender-masked receive. Admission expiry returns [`IpcError::Send`]; receive
229/// expiry or syscall failure returns [`IpcError::Recv`]. After a receive error,
230/// callers must poison that service generation because a late reply may arrive.
231pub fn service_call_bounded<'r, Req: Serialize>(
232    service_tid: usize,
233    req: &Req,
234    send_buf: &mut [u8],
235    recv_buf: &'r mut [u8],
236    timeout_ticks: u64,
237) -> Result<&'r [u8], IpcError> {
238    let encoded = api::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
239    let result = deadline::exchange_until_deadline(
240        service_tid,
241        timeout_ticks,
242        || matches!(sys_try_send(service_tid, encoded), SyscallResult::Ok(0)),
243        sys_get_scheduler_ticks,
244        sys_yield,
245        |remaining| match sys_recv_timeout(service_tid, recv_buf, remaining) {
246            SyscallResult::Ok(sender) => Ok(sender),
247            SyscallResult::Err(_) => Err(()),
248        },
249    );
250    match result {
251        Ok(()) => {
252            let len = recv_buf.len();
253            Ok(&recv_buf[..len])
254        }
255        Err(deadline::ExchangeError::Send) => Err(IpcError::Send),
256        Err(deadline::ExchangeError::Recv) => Err(IpcError::Recv),
257        Err(deadline::ExchangeError::WrongSender) => Err(IpcError::WrongSender),
258    }
259}

261/// [`service_call_bounded`] that decodes the reply into `Resp`.
262///
263/// Send admission yields until the shared deadline, but the request is never
264/// resent after delivery. Admission expiry returns [`IpcError::Send`]. The
265/// sender-masked receive gets only the remaining budget; timeout and receive
266/// errors return [`IpcError::Recv`], while malformed bytes return
267/// [`IpcError::Decode`].
268///
269/// `send_buf` holds the encoded `req`. `recv_buf` receives the reply and is
270/// borrowed by `Resp` when it contains `&str` or `&[u8]` fields, so consume the
271/// response before reusing the buffer.
272pub fn service_call_typed_bounded<'r, Req, Resp>(
273    service_tid: usize,
274    req: &Req,
275    send_buf: &mut [u8],
276    recv_buf: &'r mut [u8],
277    timeout_ticks: u64,
278) -> Result<Resp, IpcError>
279where
280    Req: Serialize,
281    Resp: Deserialize<'r>,
282{
283    let raw = service_call_bounded(service_tid, req, send_buf, recv_buf, timeout_ticks)?;
284    api::ipc::decode::<Resp>(raw).map_err(|_| IpcError::Decode)
285}

287/// [`service_call_bounded`] whose request admission **queues** instead of
288/// rendezvousing.
289///
290/// [`service_call_bounded`] offers the request with nonblocking [`sys_try_send`],
291/// which the kernel admits only while the target sits in `Recv` with a mask that
292/// matches the caller. A service that idles in a completion wait — the Net Cell
293/// sleeps on the `NET_RX` completion source — or that is simply running its own
294/// loop is never in that state, so the offer is refused for the whole deadline
295/// and the caller sees [`IpcError::Send`] against a perfectly healthy service.
296///
297/// This variant sends with [`sys_send`], which queues into the receiver's
298/// mailbox and wakes a `Recv` or a `NET_RX` completion waiter, then waits for
299/// the reply with the same bounded, sender-masked receive. Admission is
300/// therefore unbounded: a service that never receives leaves the caller blocked
301/// in the kernel's `Sending` state. Use it for a service that is known to drain
302/// its mailbox on every turn (the Net Cell does, before it parks).
303///
304/// `timeout_ticks` bounds only the reply wait. Errors are as in
305/// [`service_call_bounded`].
306pub fn service_call_bounded_queued<'r, Req: Serialize>(
307    service_tid: usize,
308    req: &Req,
309    send_buf: &mut [u8],
310    recv_buf: &'r mut [u8],
311    timeout_ticks: u64,
312) -> Result<&'r [u8], IpcError> {
313    /// Most foreign messages tolerated before the call gives up. The time base
314    /// below bounds the wait; this bounds the loop for a kernel that does not
315    /// provide one, so "keep waiting" can never become an unbounded spin.
316    const MAX_FOREIGN_MESSAGES: u32 = 4;
317
318    let encoded = api::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
319    if let SyscallResult::Err(_) = sys_send(service_tid, encoded) {
320        return Err(IpcError::Send);
321    }
322    // A sender-masked receive can still return a *death notification*: the kernel
323    // serves those regardless of the mask and documents that the caller must tell
324    // them apart by the returned sender. Failing the whole call on the first one
325    // is what cost the board's guest its traffic — the hypervisor's `L2Send` came
326    // back `wrong sender`, so `transmit` returned false, the guest's TX descriptor
327    // was never completed (`[hv-virtio-host] net-tx-complete` never printed) and
328    // everything the guest sent died in its own TX queue. The foreign message is
329    // consumed either way, so the wait names it once and continues under the
330    // *same* deadline; `sys_get_scheduler_ticks` is the time base
331    // `sys_recv_timeout` itself uses, and a kernel without it keeps the old
332    // single-wait behaviour.
333    let deadline = sys_get_scheduler_ticks().map(|now| now.saturating_add(timeout_ticks));
334    let mut foreign = 0u32;
335    loop {
336        let remaining = match deadline {
337            Some(deadline) => {
338                let now = sys_get_scheduler_ticks().unwrap_or(0);
339                if now >= deadline {
340                    return Err(IpcError::Recv);
341                }
342                deadline - now
343            }
344            None => timeout_ticks,
345        };
346        match sys_recv_timeout(service_tid, recv_buf, remaining) {
347            SyscallResult::Ok(sender) if sender == service_tid => {
348                let len = recv_buf.len();
349                return Ok(&recv_buf[..len]);
350            }
351            SyscallResult::Ok(sender) => {
352                use core::sync::atomic::{AtomicBool, Ordering};
353                static FIRST_FOREIGN: AtomicBool = AtomicBool::new(false);
354                if !FIRST_FOREIGN.swap(true, Ordering::Relaxed) {
355                    crate::io::print("[ipc] message from tid ");
356                    crate::io::print_usize(sender);
357                    crate::io::print(", expected ");
358                    crate::io::print_usize(service_tid);
359                    crate::io::println(" — consumed, still waiting for the reply");
360                }
361                foreign += 1;
362                if deadline.is_none() || foreign > MAX_FOREIGN_MESSAGES {
363                    return Err(IpcError::WrongSender);
364                }
365            }
366            SyscallResult::Err(_) => return Err(IpcError::Recv),
367        }
368    }
369}

387/// Fastpath zero-trap RPC service call over an established SPSC ring channel endpoint.
388///
389/// Encodes `req` into `send_buf` via postcard, transmits it directly into the SPSC
390/// ring buffer without trapping to kernel mode, spins briefly and yields if needed,
391/// and receives the reply directly from the return ring buffer.
392pub fn fastpath_call<'r, Req: Serialize>(
393    endpoint: &crate::ring_channel::FastpathEndpoint<'_>,
394    req: &Req,
395    send_buf: &mut [u8],
396    recv_buf: &'r mut [u8],
397) -> Result<&'r [u8], IpcError> {
398    let encoded = api::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
399    let len = endpoint
400        .call(encoded, recv_buf)
401        .map_err(|_| IpcError::Send)?;
402    Ok(&recv_buf[..len])
403}

422/// Future that waits for a message to arrive. Returns the sender id.
423pub struct AsyncRecv<'a> {
424    pub mask: usize,
425    pub buf: &'a mut [u8],
426}
427
428impl<'a> Future for AsyncRecv<'a> {
429    type Output = SyscallResult;
430
431    fn poll(mut self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<Self::Output> {
432        match sys_try_recv(self.mask, self.buf) {
433            // No message yet — the naive executor yields and polls again.
434            SyscallResult::Ok(0) => Poll::Pending,
435            SyscallResult::Ok(id) => Poll::Ready(SyscallResult::Ok(id)),
436            err => Poll::Ready(err),
437        }
438    }
439}
