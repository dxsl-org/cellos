1//! Typed request/reply IPC helpers — the compliant path for talking to a
2//! service cell (Spec 17 — Cell IPC Wire Contract).
3//!
4//! Prefer [`service_call`] / [`service_call_typed`] over a hand-rolled
5//! `sys_send` + `sys_recv`: RPC uses a kernel-owned bounded operation and takes
6//! only that operation's reply, including its actual byte length. Raw one-way
7//! messages and [`recv_from`] retain the mailbox interface.

21/// Why a [`service_call`] did not complete. Never silently swallowed.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum IpcError {
24    /// The request did not fit `send_buf` (postcard encode failed).
25    Encode,
26    /// Request submission was rejected, or a fastpath transport failed.
27    Send,
28    /// A raw receive or operation wait syscall returned an error.
29    Recv,
30    /// A message arrived, but from a cell other than the service — never
31    /// treated as the reply (Spec 17 §7 silent-wrong-sender guard).
32    WrongSender,
33    /// The reply bytes did not decode into the expected type.
34    Decode,
35    /// Operation token is stale, foreign, cancelled, or otherwise invalid.
36    InvalidOperation,
37    /// Reply output buffer is too short. Low-level `take` retains the terminal;
38    /// synchronous service calls drain their owned slot before returning this error.
39    BufferTooSmall,
40    /// Admission was busy; no request was accepted.
41    Busy,
42    /// The provider is gone. An accepted request may already have had side effects.
43    /// This is not permission to retry automatically.
44    PeerGone,
45    /// The deadline elapsed before dispatch; this operation cannot execute later.
46    PreDispatchTimeout,
47    /// The request was dispatched but no reply settled it before expiry/cancellation.
48    /// It may still execute; reconcile application state rather than retrying.
49    Indeterminate,
50    /// The operation was cancelled before dispatch.
51    Cancelled,
52}

79/// Copies `request` into kernel storage, without waiting for the peer to
80/// receive or reply. Busy means no work was accepted; do not wait on this id.
81pub fn submit(peer_tid: usize, request: &[u8]) -> Result<IpcOpId, IpcSubmitError> {
82    match crate::syscall::sys_ipc_submit(peer_tid, request) {
83        token if token > 0 => Ok(token as usize),
84        -2 => Err(IpcSubmitError::Busy),
85        -3 => Err(IpcSubmitError::PeerGone),
86        _ => Err(IpcSubmitError::InvalidRequest),
87    }
88}

90/// Atomically take the exact operation's retained terminal result. The reply
91/// bytes are owned by the kernel until this succeeds; an undersized `reply`
92/// never consumes the result. A terminal result releases the operation slot.
93pub fn take(op: IpcOpId, reply: &mut [u8]) -> Result<IpcTakeResult, IpcError> {
94    let mut status = [0u8; abi::syscall::IPC_STATUS_LEN];
95    match crate::syscall::sys_ipc_take(op, reply, &mut status) {
96        0 => Ok(IpcTakeResult::Pending),
97        1 => {
98            let version = u32::from_le_bytes(status[..4].try_into().unwrap());
99            let kind = u32::from_le_bytes(status[4..8].try_into().unwrap());
100            let len = u32::from_le_bytes(status[8..12].try_into().unwrap()) as usize;
101            let reserved = u32::from_le_bytes(status[12..16].try_into().unwrap());
102            if version != abi::syscall::IPC_STATUS_VERSION || reserved != 0 || len > reply.len() {
103                return Err(IpcError::Decode);
104            }
105            let status = match kind {
106                abi::syscall::ipc_status::REPLY => IpcTerminal::Reply,
107                abi::syscall::ipc_status::PEER_GONE => IpcTerminal::PeerGone,
108                abi::syscall::ipc_status::PRE_DISPATCH_TIMEOUT => IpcTerminal::PreDispatchTimeout,
109                abi::syscall::ipc_status::INDETERMINATE => IpcTerminal::Indeterminate,
110                abi::syscall::ipc_status::CANCELLED => IpcTerminal::Cancelled,
111                _ => return Err(IpcError::Decode),
112            };
113            Ok(IpcTakeResult::Terminal { status, len })
114        }
115        -5 => Err(IpcError::BufferTooSmall),
116        _ => Err(IpcError::InvalidOperation),
117    }
118}

120/// Sleep until an owned operation reports a terminal, or a finite
121/// scheduler-tick timeout expires (one tick is 10 ms). Zero waits without a
122/// caller deadline; accepted operations still have their separate
123/// kernel-enforced expiry. The kernel rechecks results atomically with parking.
124///
125/// Where the cell holds `WaitCompletion` authority this parks on its `IPC`
126/// source, which reports each terminal **once**: a result that was already
127/// reported and is still retained does not wake the caller again, so `take`
128/// what you were told about before waiting. Cells without that authority use
129/// `IpcWait`, which wakes while any terminal is retained. Returns `false` on
130/// timeout and on a wait error; use [`wait_any`] to tell them apart.
131pub fn wait(timeout_ticks: u64) -> bool {
132    matches!(
133        park_for_terminal(timeout_ticks),
134        Ok(Park::Notified(_) | Park::AnyTerminal)
135    )
136}

138/// What ended one [`wait_any`].
139#[derive(Debug, Clone, Copy, PartialEq, Eq)]
140pub enum Wake {
141    /// A NIC receive frame is ready to drain (the net cell's source).
142    NetRx,
143    /// This exact owned operation reached its terminal; [`take`] it.
144    Ipc(IpcOpId),
145    /// The finite deadline elapsed (`TIMER` was requested).
146    Timer,
147    /// This MSI/MSI-X line, one the task holds from `AllocPcieIrq`, fired
148    /// (`IRQ`, x86_64). Other fired lines stay pending for the next wait.
149    Irq(u8),
150    /// Nothing to report: a deadline elapsed without `TIMER` requested, a raw
151    /// message interrupted a wait naming `NET_RX`, or a NET_RX reservation was
152    /// displaced by a newer waiter.
153    Interrupted,
154}
155
156/// Park on several completion sources at once and report the one that fired.
157///
158/// `sources` is a non-empty set of `abi::completion::source::{NET_RX, TIMER,
159/// IPC, IRQ}`; `TIMER` needs a nonzero `timeout_ticks`, and zero otherwise
160/// waits without a deadline. Ready sources rotate, so none starves another;
161/// each IPC terminal is reported once, and its outcome stays behind [`take`].
162/// `IRQ` covers every line this task holds from `AllocPcieIrq`; a driver
163/// passes a deadline and re-reads its device after every return, since a
164/// lost or revoked interrupt is only ever noticed at the deadline.
165///
166/// # Errors
167/// [`IpcError::Recv`] when the kernel refuses the wait (no `WaitCompletion`
168/// authority, an invalid source set, `IRQ` without a held line or off x86_64,
169/// or a full completion queue), and [`IpcError::Decode`] for a record whose
170/// source this SDK does not know.
171pub fn wait_any(sources: u32, timeout_ticks: u64) -> Result<Wake, IpcError> {
172    decode_wake(sys_wait_completion_detailed(sources, timeout_ticks))
173}
174
175fn decode_wake(result: WaitCompletionResult) -> Result<Wake, IpcError> {
176    match result {
177        WaitCompletionResult::NoRecord => Ok(Wake::Interrupted),
178        WaitCompletionResult::Completion(record) => match record.source {
179            source::IPC if record.result > 0 => Ok(Wake::Ipc(record.result as usize)),
180            source::NET_RX if record.result >= 0 => Ok(Wake::NetRx),
181            source::NET_RX => Ok(Wake::Interrupted),
182            source::TIMER => Ok(Wake::Timer),
183            source::IRQ => u8::try_from(record.result)
184                .map(Wake::Irq)
185                .map_err(|_| IpcError::Decode),
186            _ => Err(IpcError::Decode),
187        },
188        WaitCompletionResult::ErrorOrInvalid(_) => Err(IpcError::Recv),
189    }
190}

192/// One park of the SDK's operation waits.
193#[derive(Debug, Clone, Copy, PartialEq, Eq)]
194enum Park {
195    /// `WaitCompletion(IPC)` reported this operation's terminal, once.
196    Notified(IpcOpId),

200    Timeout,

203/// This Cell's compiled syscall allowlist: the exact value the loader reads
204/// from `__ViCell_syscalls` at spawn, which every thread of the Cell inherits.
205///
206/// `declare_syscalls!` and `app_entry!` export their
207/// allowlist static under this symbol; that strong definition overrides this
208/// weak default. A Cell that declares no allowlist is permit-all, which the
209/// default (`u64::MAX`) mirrors exactly as the loader does.
210#[linkage = "weak"]
211#[no_mangle]
212#[allow(non_upper_case_globals)]
213pub static __vicell_syscall_allowlist: u64 = u64::MAX;
214
215/// Whether this Cell may call `WaitCompletion`, decided from its own compiled
216/// allowlist so the SDK never issues a syscall it would be denied.
217fn completion_wait_permitted() -> bool {
218    // SAFETY: the symbol is a plain `u64` defined either here (weak default) or
219    // by the Cell's allowlist macro; it is never written after load. A
220    // volatile read keeps the weak value from being folded at compile time.
221    let allowlist =
222        unsafe { core::ptr::read_volatile(core::ptr::addr_of!(__vicell_syscall_allowlist)) };
223    allowlist_permits_completion_wait(allowlist)
224}
225
226const fn allowlist_permits_completion_wait(allowlist: u64) -> bool {
227    match abi::syscall::ViSyscall::WaitCompletion.allowlist_bit() {
228        Some(bit) => allowlist & (1u64 << bit) != 0,
229        None => true,
230    }
231}

233/// Park for an owned operation's terminal on the syscall this Cell may use:
234/// the `WaitCompletion` IPC source with bit-42 authority, `IpcWait` without
235/// it. Every failure of the chosen syscall is returned to the caller.
236fn park_for_terminal(timeout_ticks: u64) -> Result<Park, IpcError> {
237    if !completion_wait_permitted() {
238        return match crate::syscall::sys_ipc_wait(timeout_ticks) {
239            1 => Ok(Park::AnyTerminal),
240            0 => Ok(Park::Timeout),
241            _ => Err(IpcError::Recv),
242        };
243    }
244    let sources = if timeout_ticks == 0 {
245        source::IPC
246    } else {
247        source::IPC | source::TIMER
248    };
249    match decode_wake(sys_wait_completion_detailed(sources, timeout_ticks))? {
250        Wake::Ipc(operation) => Ok(Park::Notified(operation)),
251        Wake::Timer => Ok(Park::Timeout),
252        // None of these can end an `IPC`/`IPC|TIMER` wait.
253        Wake::NetRx | Wake::Irq(_) | Wake::Interrupted => Err(IpcError::Recv),
254    }
255}

257/// Abandon one exact operation. A pre-dispatch request becomes Cancelled;
258/// an already-dispatched request becomes Indeterminate (not rolled back).
259/// Call `take` afterwards to release the retained terminal slot; dropping an
260/// unsettled [`PendingCall`] does both.
261pub fn cancel(op: IpcOpId) -> Result<(), IpcError> {
262    match crate::syscall::sys_ipc_cancel(op) {
263        0 => Ok(()),
264        _ => Err(IpcError::InvalidOperation),
265    }
266}

291/// One submitted bounded call: the kernel owns the request bytes and the reply slot,
292/// so the caller may reuse its request buffer immediately after submission.
293/// A terminal result retains its slot until taken.
294///
295/// Immediate servers may reply with plain `Send` to the currently served caller.
296/// Deferred or multiple-outstanding servers must capture [`current`] before their
297/// next receive and answer with [`reply`] using that exact token.
298///
299/// Several calls may be outstanding: submit each, then drain with [`wait`] and
300/// [`try_take`][PendingCall::try_take]. Synchronous [`service_call`] uses this same
301/// lifecycle. No accepted request is retried automatically.
302///
303/// Dropping a handle whose result was not taken stops waiting and releases its
304/// kernel slot: it cancels the operation and discards the retained terminal. A
305/// queued request is then never delivered; a dispatched one may still execute
306/// (post-dispatch cancellation is indeterminate). To keep the slot for a later
307/// [`take`] by id, read [`operation`][PendingCall::operation] and
308/// `core::mem::forget` the handle.
309pub struct PendingCall {
310    operation: IpcOpId,
311    /// Set once a `take` released the kernel slot, or found none to release.
312    settled: core::cell::Cell<bool>,
313}
314
315impl Drop for PendingCall {
316    fn drop(&mut self) {
317        if !self.settled.get() {
318            discard_call(self);
319        }
320    }
321}

323/// Whether a `take` outcome means the kernel holds no slot for the operation
324/// any more: a terminal was consumed (even one whose status failed to decode)
325/// or the token is unknown. `Pending` and an undersized buffer keep it.
326const fn take_settles(result: &Result<IpcTakeResult, IpcError>) -> bool {
327    matches!(
328        result,
329        Ok(IpcTakeResult::Terminal { .. }) | Err(IpcError::Decode | IpcError::InvalidOperation)
330    )
331}

333/// A settled operation: what happened, and how many reply bytes the kernel wrote into
334/// the caller's buffer.
335#[derive(Debug, Clone, Copy, PartialEq, Eq)]
336pub struct Completion {
337    /// The terminal kind, mapped to caller-visible outcomes by Spec 20 §2.4/§2.6.
338    pub terminal: IpcTerminal,
339    /// Reply bytes written into the caller's buffer; `0` for anything that is not
340    /// [`IpcTerminal::Reply`].
341    pub len: usize,
342}
343
344impl Completion {
345    /// The response is known, or dispatch was prevented. A known reply can still
346    /// contain application failure; this is not a blanket permission to retry.
347    pub fn is_definite(&self) -> bool {
348        matches!(
349            self.terminal,
350            IpcTerminal::Reply | IpcTerminal::PreDispatchTimeout | IpcTerminal::Cancelled
351        )
352    }
353
354    /// Execution effects are unresolved. PeerGone may follow dispatch, including
355    /// work delegated to a sibling service task. Never retry blindly; reconcile
356    /// by request identity and application policy.
357    pub fn is_uncertain(&self) -> bool {
358        matches!(
359            self.terminal,
360            IpcTerminal::Indeterminate | IpcTerminal::PeerGone
361        )
362    }
363}

365impl PendingCall {
366    /// Submit `request` without waiting for the peer to receive it.
367    ///
368    /// # Errors
369    /// [`IpcSubmitError::Busy`] when nothing was accepted (no work queued — do not wait
370    /// on it), [`IpcSubmitError::PeerGone`] when the peer is already gone, and
371    /// [`IpcSubmitError::InvalidRequest`] for a malformed or oversized request.
372    pub fn submit(peer_tid: usize, request: &[u8]) -> Result<Self, IpcSubmitError> {
373        submit(peer_tid, request).map(|operation| Self {
374            operation,
375            settled: core::cell::Cell::new(false),
376        })
377    }
378
379    /// The exact operation id, for correlating a terminal with the request it belongs to.
380    pub const fn operation(&self) -> IpcOpId {
381        self.operation
382    }
383
384    /// Take the terminal result if it has arrived; `Ok(None)` while it is still
385    /// outstanding. An undersized `reply` never consumes the result.
386    pub fn try_take(&self, reply: &mut [u8]) -> Result<Option<Completion>, IpcError> {
387        let result = take(self.operation, reply);
388        if take_settles(&result) {
389            self.settled.set(true);
390        }
391        match result? {
392            IpcTakeResult::Pending => Ok(None),
393            IpcTakeResult::Terminal { status, len } => Ok(Some(Completion {
394                terminal: status,
395                len,
396            })),
397        }
398    }
399
400    /// Wait, timer-bounded, for this call's terminal.
401    ///
402    /// `Ok(None)` means the budget ran out with the operation still outstanding — it is
403    /// neither lost nor failed, so do not read it as an outcome; keep waiting, or decide
404    /// per the contract that the caller's own deadline has passed.
405    /// Each timed round is measured in scheduler ticks. A terminal reported for
406    /// another call is left retained for that call's own `take`; with
407    /// `WaitCompletion` authority it is reported once and never re-wakes this
408    /// wait, and under `IpcWait` the caller yields rather than spinning on it.
409    pub fn wait_and_take(
410        &self,
411        reply: &mut [u8],
412        ticks_per_round: u64,
413        rounds: usize,
414    ) -> Result<Option<Completion>, IpcError> {
415        for _ in 0..rounds.max(1) {
416            let started = sys_get_scheduler_ticks().ok_or(IpcError::InvalidOperation)?;
417            loop {
418                if let Some(completion) = self.try_take(reply)? {
419                    return Ok(Some(completion));
420                }
421                let ticks = if ticks_per_round == 0 {
422                    0
423                } else {
424                    let ticks = remaining(started, ticks_per_round)?;
425                    if ticks == 0 {
426                        break;
427                    }
428                    ticks
429                };
430                match park_for_terminal(ticks)? {
431                    Park::Timeout => break,
432                    // This call's own terminal, or another one: retry `take`.
433                    Park::Notified(_) => {}
434                    // IpcWait sees every owned terminal, including results the
435                    // caller deliberately retains. Do not spend a timed round
436                    // on an unrelated terminal or starve this call's provider.
437                    Park::AnyTerminal => {
438                        if let Some(completion) = self.try_take(reply)? {
439                            return Ok(Some(completion));
440                        }
441                        sys_yield();
442                    }
443                }
444            }
445        }
446        self.try_take(reply)
447    }
448
449    /// Abandon this operation. Pre-dispatch it becomes
450    /// [`IpcTerminal::Cancelled`]; afterwards [`IpcTerminal::Indeterminate`], because a
451    /// dispatched request is never rolled back. `take` the terminal afterwards to
452    /// release the slot.
453    pub fn cancel(&self) -> Result<(), IpcError> {
454        cancel(self.operation)
455    }
456}

458/// Remaining caller budget in scheduler ticks, not the kernel operation's expiry.
459fn remaining(started: u64, budget: u64) -> Result<u64, IpcError> {
460    let now = sys_get_scheduler_ticks().ok_or(IpcError::InvalidOperation)?;
461    Ok(budget.saturating_sub(now.wrapping_sub(started)))
462}

464fn completion_len(completion: Completion) -> Result<usize, IpcError> {
465    match completion.terminal {
466        IpcTerminal::Reply => Ok(completion.len),
467        IpcTerminal::PeerGone => Err(IpcError::PeerGone),
468        IpcTerminal::PreDispatchTimeout => Err(IpcError::PreDispatchTimeout),
469        IpcTerminal::Indeterminate => Err(IpcError::Indeterminate),
470        IpcTerminal::Cancelled => Err(IpcError::Cancelled),
471    }
472}

474/// Cancel pending execution and drain even a retained oversized reply. This scratch
475/// is needed only on an error path: cancellation does not overwrite terminal Reply.
476fn discard_call(call: &PendingCall) {
477    let _ = call.cancel();
478    let mut scratch = [0u8; wire::ipc::IPC_BUF_SIZE];
479    let _ = call.try_take(&mut scratch);
480}
481
482fn take_call(call: &PendingCall, reply: &mut [u8]) -> Result<Option<Completion>, IpcError> {
483    match call.try_take(reply) {
484        Ok(completion) => Ok(completion),
485        Err(error) => {
486            discard_call(call);
487            Err(error)
488        }
489    }
490}

492fn exchange(
493    service_tid: usize,
494    request: &[u8],
495    reply: &mut [u8],
496    timeout_ticks: Option<u64>,
497) -> Result<usize, IpcError> {
498    let started = match timeout_ticks {
499        Some(_) => sys_get_scheduler_ticks().ok_or(IpcError::InvalidOperation)?,
500        None => 0,
501    };
502    let call = loop {
503        if let Some(budget) = timeout_ticks {
504            if remaining(started, budget)? == 0 {
505                return Err(IpcError::PreDispatchTimeout);
506            }
507        }
508        match PendingCall::submit(service_tid, request) {
509            Ok(call) => break call,
510            Err(IpcSubmitError::Busy) if timeout_ticks.is_some() => sys_yield(),
511            Err(IpcSubmitError::Busy) => return Err(IpcError::Busy),
512            Err(IpcSubmitError::PeerGone) => return Err(IpcError::PeerGone),
513            Err(IpcSubmitError::InvalidRequest) => return Err(IpcError::Send),
514        }
515    };
516    let mut other_terminal = false;
517    loop {
518        if let Some(completion) = take_call(&call, reply)? {
519            return completion_len(completion);
520        }
521        let ticks = match timeout_ticks {
522            Some(budget) => match remaining(started, budget) {
523                Ok(0) => {
524                    if let Err(error) = call.cancel() {
525                        discard_call(&call);
526                        return Err(error);
527                    }
528                    // A concurrent reply wins over cancellation and is still taken.
529                    let completion = take_call(&call, reply)?.ok_or(IpcError::InvalidOperation);
530                    return match completion {
531                        Ok(Completion {
532                            terminal: IpcTerminal::Cancelled,
533                            ..
534                        }) => Err(IpcError::PreDispatchTimeout),
535                        Ok(completion) => completion_len(completion),
536                        Err(error) => {
537                            discard_call(&call);
538                            Err(error)
539                        }
540                    };
541                }
542                Ok(ticks) => ticks,
543                Err(error) => {
544                    discard_call(&call);
545                    return Err(error);
546                }
547            },
548            None => 0,
549        };
550        if other_terminal {
551            // IpcWait observes any terminal. Do not spin on a different owned slot
552            // while starving the provider of this exact operation.
553            sys_yield();
554            other_terminal = false;
555            continue;
556        }
557        match park_for_terminal(ticks) {
558            Ok(Park::Timeout | Park::Notified(_)) => {}
559            Ok(Park::AnyTerminal) => other_terminal = true,
560            Err(error) => {
561                discard_call(&call);
562                return Err(error);
563            }
564        }
565    }
566}

568/// A masked receive may still return an unrelated NotifyOnExit record.
569#[inline]
570fn reply_sender(peer_tid: usize, result: SyscallResult) -> Result<(), IpcError> {
571    match result {
572        SyscallResult::Ok(sender) if sender == peer_tid => Ok(()),
573        SyscallResult::Ok(_) => Err(IpcError::WrongSender),
574        SyscallResult::Err(_) => Err(IpcError::Recv),
575    }

579/// a queued NotifyOnExit record for another task; never decode it as the peer's

586/// One kernel-bounded request/reply exchange with `service_tid`.
587///
588/// Submission never blocks in `Send`; Busy means no work was accepted. Once
589/// accepted, wait for and take only this operation's terminal result. Provider death
590/// and kernel expiry return errors even if the provider never replies. No accepted
591/// request is retried. The returned slice covers exactly the reply bytes.
592pub fn service_call<'r, Req: Serialize>(
593    service_tid: usize,
594    req: &Req,
595    send_buf: &mut [u8],
596    recv_buf: &'r mut [u8],
597) -> Result<&'r [u8], IpcError> {

601}

752/// [`service_call`] that decodes the reply into `Resp`.
753///
754/// `Resp` borrows `recv_buf` for types with `&str`/`&[u8]` fields — consume it
755/// before reusing the buffer.
756pub fn service_call_typed<'r, Req, Resp>(
757    service_tid: usize,
758    req: &Req,
759    send_buf: &mut [u8],
760    recv_buf: &'r mut [u8],
761) -> Result<Resp, IpcError>
762where
763    Req: Serialize,
764    Resp: Deserialize<'r>,
765{
766    let raw = service_call(service_tid, req, send_buf, recv_buf)?;
767    wire::ipc::decode::<Resp>(raw).map_err(|_| IpcError::Decode)
768}

770/// One bounded request/reply exchange with `service_tid`.
771///
772/// `send_buf` is used to encode `req`; an encoding failure returns
773/// [`IpcError::Encode`]. `recv_buf` receives the reply and backs the returned
774/// slice. `timeout_ticks` is the end-to-end scheduler-tick budget shared by
775/// request admission and the reply wait.
776///
777/// Busy admission is retried only before acceptance. Once accepted, only this exact
778/// operation is awaited; expiry cancels and takes its slot. Pre-dispatch expiry
779/// returns [`IpcError::PreDispatchTimeout`]; after dispatch it returns
780/// [`IpcError::Indeterminate`]. Late replies cannot be consumed by a subsequent call.
781/// The caller's scheduler-tick budget is independent of the kernel's operation expiry.
782pub fn service_call_bounded<'r, Req: Serialize>(
783    service_tid: usize,
784    req: &Req,
785    send_buf: &mut [u8],
786    recv_buf: &'r mut [u8],
787    timeout_ticks: u64,
788) -> Result<&'r [u8], IpcError> {
789    let encoded = wire::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
790    let len = exchange(service_tid, encoded, recv_buf, Some(timeout_ticks))?;
791    Ok(&recv_buf[..len])
792}
793
794/// [`service_call_bounded`] for a service's own byte protocol: `request` is
795/// sent as is and the reply bytes are returned, with the same exact-operation,
796/// scheduler-tick budget and cleanup contract.
797///
798/// This is the request path for raw-frame protocols (TLS, KMS, compositor,
799/// net-tools opcodes). A provider that answers with several messages delivers
800/// only the first through the operation; it must post the others.
801pub fn call_bytes<'r>(
802    service_tid: usize,
803    request: &[u8],
804    reply: &'r mut [u8],
805    timeout_ticks: u64,
806) -> Result<&'r [u8], IpcError> {
807    let len = exchange(service_tid, request, reply, Some(timeout_ticks))?;
808    Ok(&reply[..len])
809}

811/// [`service_call_bounded`] that decodes the reply into `Resp`.
812///
813/// Uses the same end-to-end scheduler-tick budget and exact-operation cleanup as
814/// [`service_call_bounded`]. A malformed reply returns [`IpcError::Decode`] after
815/// its terminal slot has already been released.
816///
817/// `send_buf` holds the encoded `req`. `recv_buf` receives the reply and is
818/// borrowed by `Resp` when it contains `&str` or `&[u8]` fields, so consume the
819/// response before reusing the buffer.
820pub fn service_call_typed_bounded<'r, Req, Resp>(
821    service_tid: usize,
822    req: &Req,
823    send_buf: &mut [u8],
824    recv_buf: &'r mut [u8],
825    timeout_ticks: u64,
826) -> Result<Resp, IpcError>
827where
828    Req: Serialize,
829    Resp: Deserialize<'r>,
830{
831    let raw = service_call_bounded(service_tid, req, send_buf, recv_buf, timeout_ticks)?;
832    wire::ipc::decode::<Resp>(raw).map_err(|_| IpcError::Decode)
833}

835/// Fastpath zero-trap RPC service call over an established SPSC ring channel endpoint.
836///
837/// Encodes `req` into `send_buf` via postcard, transmits it directly into the SPSC
838/// ring buffer without trapping to kernel mode, spins briefly and yields if needed,
839/// and receives the reply directly from the return ring buffer.
840pub fn fastpath_call<'r, Req: Serialize>(
841    endpoint: &crate::ring_channel::FastpathEndpoint<'_>,
842    req: &Req,
843    send_buf: &mut [u8],
844    recv_buf: &'r mut [u8],
845) -> Result<&'r [u8], IpcError> {
846    let encoded = wire::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
847    let len = endpoint
848        .call(encoded, recv_buf)
849        .map_err(|_| IpcError::Send)?;
850    Ok(&recv_buf[..len])
851}
852
853/// Typed version of `fastpath_call` that deserializes the reply into `Resp`.
854pub fn fastpath_call_typed<'r, Req, Resp>(
855    endpoint: &crate::ring_channel::FastpathEndpoint<'_>,
856    req: &Req,
857    send_buf: &mut [u8],
858    recv_buf: &'r mut [u8],
859) -> Result<Resp, IpcError>
860where
861    Req: Serialize,
862    Resp: Deserialize<'r>,
863{
864    let raw = fastpath_call(endpoint, req, send_buf, recv_buf)?;
865    wire::ipc::decode::<Resp>(raw).map_err(|_| IpcError::Decode)
866}

870/// Future that waits for a message to arrive. Returns the sender id.
871pub struct AsyncRecv<'a> {
872    pub mask: usize,
873    pub buf: &'a mut [u8],
874}
875
876impl<'a> Future for AsyncRecv<'a> {
877    type Output = SyscallResult;
878
879    fn poll(mut self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<Self::Output> {
880        match sys_try_recv(self.mask, self.buf) {
881            // No message yet — the naive executor yields and polls again.
882            SyscallResult::Ok(0) => Poll::Pending,
883            SyscallResult::Ok(id) => Poll::Ready(SyscallResult::Ok(id)),
884            err => Poll::Ready(err),
885        }
886    }
887}

889/// Await a raw mailbox message on `mask` (0 = wildcard). For request/reply use
890/// [`service_call`] instead, so provider death and late replies have exact outcomes.
891pub fn recv_async(mask: usize, buf: &mut [u8]) -> AsyncRecv<'_> {
892    AsyncRecv { mask, buf }
