1//! Fast-IPC handler table for direct Cell-to-service calls.
2//!
3//! Retained scaffolding for the accepted Tier-1 direct-dispatch rewrite. Current
4//! separately linked Cells cannot share this table and always use the message
5//! fallback; no latency claim attaches to this inactive path.
6//!
7//! ## Safety invariant
8//! The handler function pointer is written once at VFS startup (before any
9//! Cell requests it) and never changed thereafter.  On single-hart QEMU,
10//! there is no concurrent modification risk.  `Ordering::Release` on write
11//! and `Ordering::Acquire` on read ensure the handler body is visible to
12//! the caller after the pointer is stored.
13//!
14//! ## Preemption
15//! `call_vfs` disables S-mode interrupts for the duration of the handler call.
16//! The VFS FAT16 driver holds a spinlock; if the timer ISR preempted the
17//! handler mid-spinlock and switched to another Cell that also called VFS,
18//! the spinlock would deadlock.  Interrupt-disable makes the fast-path call
19//! behave as an atomic critical section w.r.t. the scheduler.
20
21use api::caller_identity::CallerIdentity;
22use api::fast_ipc::{TrustedHandle, VfsCell};
23use api::ipc::{VfsRequest, IPC_BUF_SIZE};
24use core::sync::atomic::{AtomicPtr, AtomicUsize, Ordering};
25
26/// Signature of a registered VFS fast-IPC handler.
27///
28/// Reads the request on behalf of `caller` and writes the response into `out`.
29/// Returns the number of bytes written into `out`.
30///
31/// Must stay identical to `kernel::fast_ipc::VfsFastHandler` if the future
32/// loader bridge makes the kernel table reachable; a mismatch would be a
33/// silently wrong call through a function pointer.
34///
35/// `caller` is `None` when the call could not be attributed to a live cell; the
36/// handler must refuse rather than serve an unattributed read.
37pub type VfsFastHandler = unsafe fn(
38    caller: Option<CallerIdentity>,
39    req: &VfsRequest<'_>,
40    out: &mut [u8; IPC_BUF_SIZE],
41) -> usize;
42
43static VFS_HANDLER_PTR: AtomicPtr<()> = AtomicPtr::new(core::ptr::null_mut());
44/// Raw CellId of the cell that registered the VFS handler; 0 = unregistered.
45/// Used to clear the handler pointer when the VFS cell crashes.
46static VFS_HANDLER_CELL: AtomicUsize = AtomicUsize::new(0);
47
48/// Register the VFS fast-IPC handler.
49///
50/// Called once by the VFS service at startup.  The pointer is published with
51/// `Release` ordering so subsequent `Acquire` loads in `call_vfs` observe the
52/// complete handler function.
53pub fn register_vfs(handler: VfsFastHandler) {
54    VFS_HANDLER_PTR.store(
55        // fn pointer → *mut () for atomic storage; recovered with the same
56        // type in call_vfs.
57        handler as *mut (),
58        Ordering::Release,
59    );
60}
61
62/// Record which cell (by CellId raw value) owns the registered VFS handler.
63///
64/// Call this immediately after `register_vfs` so the kernel can clear the
65/// handler pointer if the owning cell crashes.
66pub fn set_vfs_handler_cell(cell_id_raw: usize) {
67    VFS_HANDLER_CELL.store(cell_id_raw, Ordering::Relaxed);
68}
69
70/// Clear the VFS handler pointer if `cell_id_raw` is the registered owner.
71///
72/// Called by the kernel fault-isolation path when a cell is terminated, so
73/// that a future `call_vfs` does not jump into stale/replaced code.
74pub fn clear_vfs_if_cell(cell_id_raw: usize) {
75    if VFS_HANDLER_CELL.load(Ordering::Relaxed) == cell_id_raw && cell_id_raw != 0 {
76        VFS_HANDLER_PTR.store(core::ptr::null_mut(), Ordering::Release);
77        VFS_HANDLER_CELL.store(0, Ordering::Relaxed);
78    }
79}

81/// `extern "Rust"` shim so the kernel (which does not depend on ostd) can
82/// call `set_vfs_handler_cell` via link-time symbol resolution.
83#[no_mangle]
84pub extern "Rust" fn vi_set_fast_ipc_vfs_cell(cell_id: usize) {
85    set_vfs_handler_cell(cell_id);
86}

88/// `extern "Rust"` shim so the kernel can call `clear_vfs_if_cell`.
89#[no_mangle]
90pub extern "Rust" fn vi_clear_fast_ipc_vfs_cell(cell_id: usize) {
91    clear_vfs_if_cell(cell_id);

94/// RAII guard: disables S-mode interrupt enable (SIE) on construction, restores on drop.
95///
96/// Ensures SIE is always restored even if the fast-IPC handler panics (drop glue
97/// runs before the panic handler, so the restore is guaranteed).
98struct SieGuard(bool);
99
100impl SieGuard {
101    /// Disable SIE and return a guard.  Noop on non-riscv64.
102    ///
103    /// # Safety
104    /// Must be called from S-mode.
105    #[inline]
106    unsafe fn disable() -> Self {
107        #[cfg(target_arch = "riscv64")]
108        {
109            let v: usize;
110            // SAFETY: csrrci reads-and-clears sstatus.SIE (bit 1) atomically.
111            core::arch::asm!("csrrci {}, sstatus, 0x2", out(reg) v);
112            Self(v & 0x2 != 0)
113        }
114        #[cfg(not(target_arch = "riscv64"))]
115        Self(false)
116    }
117}
118
119impl Drop for SieGuard {
120    fn drop(&mut self) {
121        if self.0 {
122            // SAFETY: restoring SIE to the state saved in disable(); S-mode only.
123            #[cfg(target_arch = "riscv64")]
124            unsafe {
125                core::arch::asm!("csrsi sstatus, 0x2");
126            }
127        }
128    }
129}

131/// Call the registered VFS handler directly, bypassing the ecall trap.
132///
133/// Returns the number of bytes written into `out`, or 0 if no handler is
134/// registered (caller should fall back to the `sys_send` / `sys_recv` path).
135///
136/// # Note (PIE limitation)
137/// For non-PIE cells (current default), each cell ELF has its own copy of
138/// `VFS_HANDLER_PTR` (statically linked from `libs/ostd`).  VFS writes to its
139/// copy; client cells read from theirs (null) → always fallback to ecall.
140/// The fast path activates once cells are compiled as PIE and the kernel loader
141/// gains a reviewed import-resolution mechanism for the Tier-1 rewrite.
142///
143/// # Caller identity
144/// This copy of the dispatch table lives inside a cell, where there is no
145/// attested answer to "who is calling". The future bridge must use the kernel's
146/// `kernel::fast_ipc::call_vfs`, which derives identity from scheduler state; no
147/// loader bridge reaches it today. If control reaches this local copy, it passes
148/// `None`, so the handler denies the request instead of serving a read on an
149/// identity nobody vouched for.
150///
151/// # Safety
152/// The caller must own `out` exclusively for the duration of this call.
153/// `_handle: TrustedHandle<VfsCell>` documents that the caller has been
154/// granted fast-path access; it does not enforce this at runtime.
155pub unsafe fn call_vfs(
156    _handle: TrustedHandle<VfsCell>,
157    req: &VfsRequest<'_>,
158    out: &mut [u8; IPC_BUF_SIZE],
159) -> usize {
160    let ptr = VFS_HANDLER_PTR.load(Ordering::Acquire);
161    if ptr.is_null() {
162        return 0; // VFS not yet registered — caller falls back to ecall path
163    }
164    // SAFETY: ptr was stored by register_vfs from a valid VfsFastHandler.
165    let handler: VfsFastHandler = core::mem::transmute(ptr);
166
167    // Disable S-mode interrupts for the handler's duration. VFS's FAT32 driver
168    // holds a spinlock; timer-preemption mid-handler to another VFS caller would
169    // deadlock on it. SieGuard restores on drop — safe even on handler panic.
170    // SAFETY: called from S-mode; SieGuard::disable is safe here.
171    let _sie = SieGuard::disable();
172
173    handler(None, req, out)
174}
