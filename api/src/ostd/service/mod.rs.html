3//! Service discovery helpers for ViCell cells.
4//!
5//! Two levels of abstraction:
6//! - Free functions [`lookup`] / [`register`] — thin wrappers for one-shot calls.
7//! - [`ServiceRef<const ID>`] — a caching handle with a typed [`ServiceRef::call`] method
8//!   for cells that talk to the same service repeatedly.
9//! - [`serve`] — the server side: the one receive loop every service runs.
10//!
11//! [`ServiceRef`] resolves the provider's **binding** (`{tid, cell_id, generation}`) from
12//! the kernel's registry rather than a bare tid. A call returns a [`CallOutcome`]: a reply,
13//! a refusal (the request never ran) or an uncertain result (it may have run). A failed
14//! exchange against a provider that has since been replaced is reported as
15//! [`Uncertainty::ProviderReplaced`] instead of quietly re-targeting the call.
16

31/// Resolve the live provider **binding** of a well-known service
32/// (`LookupServiceBound`): which Cell incarnation serves it, not only where to
33/// send.
34///
35/// Returns `None` when nothing is registered, during the death→respawn window
36/// (the caller should retry with a small delay) and while the provider is
37/// hidden behind a hot-swap pause.
38///
39/// # Example
40/// ```no_run
41/// let vfs = ostd::service::lookup(ostd::service::service::VFS).expect("VFS not ready");
42/// ```
43pub fn lookup(service_id: u16) -> Option<ServiceBinding> {
44    let mut record = [0u8; SERVICE_BINDING_LEN];
45    syscall::sys_lookup_service_bound(service_id, &mut record)
46}
47
48/// Where to send to `service_id` right now: the tid of its live bound provider
49/// ([`lookup`]). A tid is an address, never an identity: to decide whether a
50/// received message came from the provider, use [`sender_is_live_provider`].
51pub fn provider_tid(service_id: u16) -> Option<usize> {
52    lookup(service_id).map(|binding| binding.tid as usize)
53}

55/// The [`api::syscall::service_flags`] the namespace owner (init, and the
56/// supervisor when it rolls a hot swap back) registers `service_id` with.
57///
58/// VFS keeps its outer caller across nested IPC and watches the owners it
59/// serves; NET, the compositor and the Intel GPU driver watch the owners whose
60/// resources they reclaim. Each entry is an owner decision with its own reclaim
61/// consumer.
62pub const fn namespace_flags(service_id: u16) -> usize {
63    use api::syscall::service_flags::{ALL, OWNER_WATCH};
64    match service_id {
65        service::VFS => ALL,
66        service::NET | service::COMPOSITOR | service::XE_GPU => OWNER_WATCH,
67        _ => 0,
68    }
69}

71/// Register the calling cell as the live provider of `service_id`.
72///
73/// Well-known ids require trusted init or `SupervisorCap`; a `SpawnCap`-only cell may
74/// publish only a private id it owns. Kernel-defined, operation-specific self-registration
75/// capabilities are separate exceptions: the development/test-hooks Silo route is one and can
76/// publish only `service::SILO` with `tid=0`; it cannot register any other endpoint.
77///
78/// # Errors
79/// Returns `ViError::PermissionDenied` when the caller lacks matching authority, or
80/// `ViError::Unknown` if the registry is full.
81pub fn register(service_id: u16, tid: usize) -> ViResult<()> {
82    match syscall::sys_register_service(service_id, tid, 0) {
83        syscall::SyscallResult::Ok(_) => Ok(()),
84        syscall::SyscallResult::Err(_) => Err(ViError::PermissionDenied),
85    }
86}

90/// A caching handle to a well-known service.
91///
92/// Resolves the live provider's **binding** — `{tid, cell_id, generation}`, from the
93/// kernel's own registry — on first use and caches it. When the service restarts (new
94/// TID, or a new Cell incarnation), call [`invalidate`][Self::invalidate] so the next
95/// [`call`][Self::call] re-resolves, or ask [`is_live`][Self::is_live] whether the
96/// descriptor this handle holds is still the live one.
97///
98/// # Usage
99/// ```no_run
100/// use ostd::service::{service, ServiceRef};
101/// use ostd::services::ipc::{VfsRequest, VfsResponse, IPC_BUF_SIZE};
102/// use ostd::ViResult;
103///
104/// # fn stat() -> ViResult<()> {
105/// let mut vfs: ServiceRef<{ service::VFS }> = ServiceRef::new();
106/// let mut response_buffer = [0u8; IPC_BUF_SIZE];
107/// let response: VfsResponse = vfs
108///     .call(&VfsRequest::Stat("/tmp"), &mut response_buffer)
109///     .into_result()?;
110/// # let _ = response;
111/// # Ok(())
112/// # }
113/// ```
114pub struct ServiceRef<const ID: u16> {
115    /// The resolved provider binding, not just its tid: a tid says where to send,
116    /// never which provider incarnation is there.
117    cached: Option<ServiceBinding>,
118}
119
120impl<const ID: u16> ServiceRef<ID> {
121    /// Create a new, unresolved handle. Resolution happens lazily on the first [`call`][Self::call].
122    pub const fn new() -> Self {
123        Self { cached: None }
124    }
125
126    /// Resolve the live provider TID, retrying up to 8 times with a scheduler yield between
127    /// attempts (mirrors the [`ConfigClient::endpoint`] pattern).
128    ///
129    /// Returns `None` during the brief death→respawn window.
130    pub fn resolve(&mut self) -> Option<usize> {
131        self.binding().map(|binding| binding.tid as usize)
132    }
133
134    /// Resolve the live provider's binding, retrying like [`resolve`][Self::resolve].
135    ///
136    /// Like [`lookup`], it answers *which provider incarnation* is there, out of
137    /// the kernel's own service registry (`LookupServiceBound`). The handle caches
138    /// it: the record is a descriptor, it
139    /// grants nothing, and the kernel re-verifies the live binding when the tid is
140    /// actually used.
141    ///
142    /// Returns `None` when nothing is registered, during the death→respawn window, and
143    /// while the provider is hidden behind a hot-swap pause — every one of those is
144    /// "no binding", and the only correct response is to refuse the call.
145    pub fn binding(&mut self) -> Option<ServiceBinding> {
146        if let Some(binding) = self.cached {
147            return Some(binding);
148        }
149        for _ in 0..8 {
150            if let Some(binding) = lookup(ID) {
151                self.cached = Some(binding);
152                return Some(binding);
153            }
154            crate::task::yield_now();
155        }
156        None
157    }
158
159    /// Whether the binding this handle holds is still the live provider binding.
160    ///
161    /// Compares the whole descriptor, so a provider that re-registered under a new tid —
162    /// or a Cell that was replaced by a new incarnation — makes this `false`, and the
163    /// stale descriptor is dropped. A handle that has not resolved yet is not live.
164    ///
165    /// This costs one registry lookup, which is why [`call`][Self::call] does not pay for
166    /// it on the happy path: use it before work whose retry safety the caller has to
167    /// decide itself, as the contract requires, rather than to re-check every message.
168    pub fn is_live(&mut self) -> bool {
169        let Some(held) = self.cached else {
170            return false;
171        };
172        let mut record = [0u8; SERVICE_BINDING_LEN];
173        match syscall::sys_lookup_service_bound(ID, &mut record) {
174            Some(live) if live == held => true,
175            _ => {
176                self.cached = None;
177                false
178            }
179        }
180    }
181
182    /// Clear the cached binding. Call this after a send error to force re-resolution on the
183    /// next [`call`][Self::call] (the service may have restarted with a new TID).
184    pub fn invalidate(&mut self) {
185        self.cached = None;
186    }
187
188    /// Send a postcard-encoded `req` to the service and decode the response into `resp_buf`.
189    ///
190    /// The decoded `Resp` may borrow bytes from `resp_buf` (e.g. `VfsResponse::Data(&[u8])`).
191    /// Keep `resp_buf` alive as long as you use the returned value.
192    ///
193    /// The outcome says whether the request may have run: [`Refused`][CallOutcome::Refused]
194    /// never executed, [`Uncertain`][CallOutcome::Uncertain] might have. Lifecycle failures
195    /// are classified against the registry, so a provider that was replaced is reported as
196    /// such, and the cached descriptor is dropped on every lifecycle failure. This method
197    /// never retries a request after acceptance. Use [`CallOutcome::into_result`] when only
198    /// an error code is needed.
199    pub fn call<'b, Req, Resp>(
200        &mut self,
201        req: &Req,
202        resp_buf: &'b mut [u8; IPC_BUF_SIZE],
203    ) -> CallOutcome<Resp>
204    where
205        Req: serde::Serialize,
206        Resp: serde::Deserialize<'b>,
207    {
208        let Some(held) = self.binding() else {
209            return CallOutcome::Refused(Refusal::NoBinding);
210        };
211        let mut send_buf = [0u8; IPC_BUF_SIZE];
212        match crate::ipc::service_call_typed::<Req, Resp>(
213            held.tid as usize,
214            req,
215            &mut send_buf,
216            resp_buf,
217        ) {
218            Ok(resp) => CallOutcome::Reply(resp),
219            Err(error) if call_outcome::needs_binding_check(error) => {
220                let mut record = [0u8; SERVICE_BINDING_LEN];
221                let live = syscall::sys_lookup_service_bound(ID, &mut record);
222                let stale = classify_call_failure(held, live) == CallFailure::StaleBinding;
223                self.invalidate();
224                call_outcome::classify(error, stale).into()
225            }
226            Err(error) => call_outcome::classify(error, false).into(),
227        }
228    }
229}

231/// What a failed typed exchange means, given the binding it used and the binding the
232/// kernel reports for that service now.
233#[derive(Clone, Copy, Debug, PartialEq, Eq)]
234pub enum CallFailure {
235    /// The named provider is no longer the live endpoint. This is a binding verdict,
236    /// not a claim that the failed operation was refused or had no side effects.
237    StaleBinding,
238    /// The same live endpoint failed the exchange or the operation expired.
239    ProviderError,
240}
241
242/// Decide what a failed exchange against `held` means, now that `live` is what the
243/// kernel reports for that service.
244///
245/// The **whole** descriptor must match, not just the Cell identity: a provider Cell can
246/// re-register under a new tid without its generation changing, and the tid is what a
247/// call actually addresses. Absent, changed or non-live all mean the descriptor the
248/// caller held is stale.
249///
250/// Pure by construction: the kernel is the only source of bindings, so this rule is
251/// testable without one.
252pub fn classify_call_failure(held: ServiceBinding, live: Option<ServiceBinding>) -> CallFailure {
253    match live {
254        Some(live) if live == held => CallFailure::ProviderError,
255        _ => CallFailure::StaleBinding,
256    }
257}

259/// Whether `sender_tid` — the tid the kernel stamped on a received message — is the
260/// live provider of `service_id` right now.
261///
262/// For a cell that serves exactly one peer service (a NIC driver serves only the net
263/// service): it asks the kernel's registry on every call instead of trusting a cached
264/// tid, so a restarted or hot-swapped provider is followed and a provider that is
265/// not yet registered (init registers a service after spawning it) is refused until
266/// it is. Never pass a tid taken from a message payload.
267pub fn sender_is_live_provider(service_id: u16, sender_tid: usize) -> bool {
268    binding_names_sender(lookup(service_id), sender_tid)
269}
270
271/// Pure rule behind [`sender_is_live_provider`]: the kernel's live binding must exist,
272/// name a live provider, and address exactly `sender_tid`. Tids are never re-issued
273/// within one boot, so a matching tid is that provider's task and nobody else's.
274pub fn binding_names_sender(live: Option<ServiceBinding>, sender_tid: usize) -> bool {
275    matches!(live, Some(binding) if binding.is_live() && binding.tid == sender_tid as u64)
276}

300    /// Tripwire: which services the namespace owner flags is an owner decision.
301    /// Adding one is a reviewed change with its own reclaim consumer and kernel
302    /// grant coverage, never a silent widening.
303    #[test]
304    fn namespace_flags_are_an_explicit_table() {
305        let flagged: alloc::vec::Vec<(u16, usize)> = (0..=u16::MAX)
306            .map(|id| (id, namespace_flags(id)))
307            .filter(|&(_, flags)| flags != 0)
308            .collect();
309        assert_eq!(
310            flagged,
311            [
312                (service::VFS, ALL),
313                (service::NET, OWNER_WATCH),
314                (service::COMPOSITOR, OWNER_WATCH),
315                (service::XE_GPU, OWNER_WATCH),
316            ]
317        );
318    }
