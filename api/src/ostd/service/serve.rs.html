3//! The one service loop: attested receive, owner-death watches, heartbeat and
4//! idle work, so a service writes only what it does with a request.
5//!
6//! [`serve`] owns the receive. Every request arrives with the sender's
7//! kernel-attested [`CallerIdentity`]; a message without one goes to
8//! [`Handler::on_unattributed`] (a kernel push, a raw producer) or, when its
9//! sender is the root of a watched owner, to [`Handler::on_owner_death`]. It is
10//! never handled as a request on a guessed identity. The [`Incoming`] of a
11//! request is captured before the handler runs, so nested receives inside the
12//! handler (a bounded call to another service) cannot replace the operation
13//! the reply completes.
14//!
15//! With [`WatchPolicy`] set, `serve` subscribes to the caller's owner death
16//! before the handler runs (the only turn in which the kernel grants it), and
17//! reports retired generations and deaths through [`Handler::on_owner_death`].
18//!
19//! With a heartbeat armed, the loop never blocks longer than half the interval,
20//! and it re-arms after every receive. A handler that runs longer than that must
21//! call [`keepalive`]; otherwise the kernel's heartbeat sweep kills the service,
22//! which is the point: a service that stops turning is restarted by init.
23//!
24//! A hot-swap envelope (`[MAGIC, KIND_SNAPSHOT | KIND_RESTORE, …]`) from the
25//! live `SUPERVISOR` provider is serve's own: it runs [`Handler::on_snapshot`]
26//! or [`Handler::on_restore`] and never replies, since the supervisor posts it
27//! without an operation to answer. The same bytes from anyone else are an
28//! ordinary request.

43/// Which callers' owner deaths [`serve`] watches.
44#[derive(Clone, Copy)]
45pub enum WatchPolicy {
46    /// None.
47    Off,
48    /// Callers whose request satisfies the predicate (it creates state that
49    /// outlives the call). A refused watch still serves the request: its
50    /// state is then not reclaimed on death.
51    OnRequest(fn(&[u8]) -> bool),
52    /// Every caller, before any request. A refused watch refuses the request
53    /// ([`Handler::on_unwatched`]): no state may outlive an unwatched owner.
54    Every,
55}

57/// How [`serve`] runs.
58#[derive(Clone, Copy)]
59pub struct ServeConfig {
60    /// Heartbeat interval in scheduler ticks; 0 arms none.
61    pub heartbeat_ticks: u64,
62    pub watch: WatchPolicy,
63    /// Owners watched at once; callers beyond it are unwatched.
64    pub watch_capacity: usize,
65}
66
67impl ServeConfig {
68    /// 60 s: longer than any bounded call a handler makes (the longest SDK
69    /// budget is 50 s), so only a service that stopped turning misses it.
70    pub const DEFAULT_HEARTBEAT_TICKS: u64 = 6_000;
71
72    /// A heartbeat of [`Self::DEFAULT_HEARTBEAT_TICKS`], no owner watches.
73    pub const fn with_heartbeat() -> Self {
74        Self {
75            heartbeat_ticks: Self::DEFAULT_HEARTBEAT_TICKS,
76            watch: WatchPolicy::Off,
77            watch_capacity: 0,
78        }
79    }
80
81    /// Also watch owners per `watch`, at most `capacity` at once.
82    pub const fn watching(self, watch: WatchPolicy, capacity: usize) -> Self {
83        Self {
84            watch,
85            watch_capacity: capacity,
86            ..self
87        }
88    }
89}

91/// How long the next receive may wait.
92#[derive(Debug, Clone, Copy, PartialEq, Eq)]
93pub enum Wait {
94    /// Until a message arrives (or half the heartbeat interval).
95    Block,
96    /// At most this many ticks, then [`Handler::on_idle`] runs again.
97    Ticks(u64),
98    /// Do not wait: take a queued message if there is one, else run
99    /// [`Handler::on_empty_poll`].
100    Poll,
101}

103/// One attested request.
104pub struct Request<'a> {
105    incoming: Incoming,
106    caller: CallerIdentity,
107    payload: &'a [u8],
108    watched: bool,
109}
110
111impl<'a> Request<'a> {
112    /// The task that sent the request: an address for replies, never an identity.
113    pub const fn sender(&self) -> usize {
114        self.incoming.sender()
115    }
116
117    /// The kernel-attested sender.
118    pub const fn caller(&self) -> &CallerIdentity {
119        &self.caller
120    }
121
122    /// The caller's Cell incarnation, the key of state kept for it.
123    pub const fn owner(&self) -> Owner {
124        Owner::of(&self.caller)
125    }
126
127    /// Whether the caller's owner death is watched, so state kept for it is
128    /// reclaimed through [`Handler::on_owner_death`].
129    pub const fn owner_watched(&self) -> bool {
130        self.watched
131    }
132
133    /// The receive buffer up to the identity trailer. Raw IPC carries no byte
134    /// count, so this is the whole zero-padded payload window.
135    pub const fn payload(&self) -> &'a [u8] {
136        self.payload
137    }
138
139    /// The request's reply route, for a handler that answers later.
140    pub const fn incoming(&self) -> Incoming {
141        self.incoming
142    }
143
144    /// Answer without parking ([`Incoming::reply`]).
145    ///
146    /// # Errors
147    /// As [`Incoming::reply`].
148    pub fn reply(&self, bytes: &[u8]) -> Result<(), PostError> {
149        self.incoming.reply(bytes)
150    }
151}

153/// What a service does; [`serve`] does the rest.
154pub trait Handler {
155    /// Handle one attested request. Answer it with [`Request::reply`] or keep
156    /// [`Request::incoming`] to answer later.
157    fn on_request(&mut self, request: Request<'_>);
158
159    /// [`WatchPolicy::Every`] could not watch this caller: refuse the request.
160    /// Unanswered by default.
161    fn on_unwatched(&mut self, _request: Request<'_>) {}
162
163    /// `owner` is gone: its root died, or a newer generation of its CellId
164    /// was watched. Reclaim what was kept for it.
165    fn on_owner_death(&mut self, _owner: Owner) {}
166
167    /// A message without a valid identity trailer that is no watched owner's
168    /// death: sender 0 is the kernel, other senders are raw producers.
169    /// Dropped by default.
170    fn on_unattributed(&mut self, _sender: usize, _payload: &[u8]) {}
171
172    /// Runs after every receive attempt, message or not (render, pump).
173    fn on_turn(&mut self) {}
174
175    /// A [`Wait::Poll`] found nothing queued: park the way the service needs
176    /// (a completion wait on its device), bounded below half the heartbeat.
177    /// Yields by default.
178    fn on_empty_poll(&mut self) {
179        crate::task::yield_now();
180    }
181
182    /// Runs before every receive; periodic work goes here. Its answer bounds the
183    /// receive that follows.
184    fn on_idle(&mut self) -> Wait {
185        Wait::Block
186    }
187
188    /// The supervisor is swapping this service out: the state its replacement
189    /// restores, stashed by `serve` under the swap's key. `None` carries
190    /// nothing, so the replacement never signals Ready and the swap rolls back.
191    /// Off by default.
192    fn on_snapshot(&mut self) -> Option<Vec<u8>> {
193        None
194    }
195
196    /// This incarnation is the replacement: apply `state`, the bytes the
197    /// previous incarnation's [`Handler::on_snapshot`] returned. `true` makes
198    /// `serve` release the stash and signal Ready (`sys_hotswap_ready`);
199    /// `false` leaves the supervisor to roll back. Off by default.
200    fn on_restore(&mut self, _state: &[u8]) -> bool {
201        false
202    }
203}

205/// Re-arm the heartbeat from inside a long handler.
206pub fn keepalive(config: &ServeConfig) {
207    if config.heartbeat_ticks > 0 {
208        syscall::sys_heartbeat(config.heartbeat_ticks);
209    }
210}

212/// Serve requests forever.
213pub fn serve<H: Handler>(config: ServeConfig, handler: &mut H) -> ! {
214    let mut buf = [0u8; IPC_BUF_SIZE];
215    let mut watches = OwnerWatches::new(config.watch_capacity);
216    loop {
217        keepalive(&config);
218        let wait = handler.on_idle();
219        // A stale trailer from the previous message must never attest this one.
220        buf.fill(0);
221        let received = match wait {
222            Wait::Poll => syscall::sys_try_recv_attested(0, &mut buf),
223            wait => match receive_ticks(wait, config.heartbeat_ticks) {
224                0 => syscall::sys_recv_attested(0, &mut buf),
225                ticks => syscall::sys_recv_attested_timeout(0, &mut buf, ticks),
226            },
227        };
228        match received {
229            SyscallResult::Ok(sender) if sender != 0 || wait != Wait::Poll => {
230                dispatch(&config, &mut watches, handler, sender, &buf);
231            }
232            // An empty poll; expiry and refused receives also land here.
233            _ if wait == Wait::Poll => handler.on_empty_poll(),
234            _ => {}
235        }
236        handler.on_turn();
237    }
238}
239
240fn dispatch<H: Handler>(
241    config: &ServeConfig,
242    watches: &mut OwnerWatches,
243    handler: &mut H,
244    sender: usize,
245    buf: &[u8; IPC_BUF_SIZE],
246) {
247    let payload = &buf[..IPC_BUF_SIZE - CALLER_IDENTITY_LEN];
248    // The trailer must name this sender; a death record carries none.
249    let caller = CallerIdentity::from_recv_buf(buf)
250        .filter(|caller| sender != 0 && caller.sender_tid == sender as u64);
251    let Some(caller) = caller else {
252        match watches.take_death(sender) {
253            Some(owner) => handler.on_owner_death(owner),
254            None => handler.on_unattributed(sender, payload),
255        }
256        return;
257    };
258    // A swap envelope only from the live supervisor, and never answered.
259    if let Some(swap) = swap_envelope(payload) {
260        if super::sender_is_live_provider(super::service::SUPERVISOR, sender) {
261            run_swap(handler, swap);
262            return;
263        }
264    }
265    // Captured before any hook runs: a nested receive would replace it.
266    let incoming = Incoming::current(sender);
267    let required = match config.watch {
268        WatchPolicy::Off => None,
269        WatchPolicy::OnRequest(creates) => creates(payload).then_some(false),
270        WatchPolicy::Every => Some(true),
271    };
272    let mut watched = false;
273    if let Some(fail_closed) = required {
274        let admission = watches.admit(&caller, &mut KernelWatch);
275        for owner in admission.stale {
276            handler.on_owner_death(owner);
277        }
278        watched = admission.watched;
279        if fail_closed && !watched {
280            handler.on_unwatched(Request {
281                incoming,
282                caller,
283                payload,
284                watched,
285            });
286            return;
287        }
288    }
289    handler.on_request(Request {
290        incoming,
291        caller,
292        payload,
293        watched,
294    });
295}

297/// A supervisor hot-swap envelope, by its `[MAGIC, kind]` prefix.
298#[derive(Debug, PartialEq, Eq)]
299enum Swap {
300    Snapshot(u64),
301    Restore(u64),
302    /// The prefix of one whose swap id does not parse: dropped.
303    Malformed,
304}
305
306fn swap_envelope(payload: &[u8]) -> Option<Swap> {
307    let swap = match payload {
308        [MAGIC, KIND_SNAPSHOT, ..] => payload
309            .get(ENVELOPE_HEADER_LEN..SNAPSHOT_LEN)
310            .and_then(|id| <[u8; 8]>::try_from(id).ok())
311            .map(|id| Swap::Snapshot(u64::from_le_bytes(id))),
312        [MAGIC, KIND_RESTORE, ..] => payload
313            .get(ENVELOPE_HEADER_LEN..RESTORE_LEN)
314            .and_then(hotswap::swap_id_from_restore_key)
315            .map(Swap::Restore),
316        _ => return None,
317    };
318    Some(swap.unwrap_or(Swap::Malformed))
319}
320
321/// Snapshot: stash what the handler returns under the swap's key. Restore:
322/// hand the stashed bytes to the handler, and on success release them and
323/// signal Ready. Any failure leaves the replacement un-Ready, which is the
324/// supervisor's cue to roll back.
325fn run_swap<H: Handler>(handler: &mut H, swap: Swap) {
326    match swap {
327        Swap::Snapshot(swap_id) => {
328            if let Some(state) = handler.on_snapshot() {
329                let _ = hotswap::stash(hotswap::hotswap_key(swap_id), &state);
330            }
331        }
332        Swap::Restore(swap_id) => {
333            let key = hotswap::hotswap_key(swap_id);
334            if let Ok(state) = hotswap::restore(key) {
335                if handler.on_restore(&state) {
336                    hotswap::clear(key);
337                    syscall::sys_hotswap_ready();
338                }
339            }
340        }
341        Swap::Malformed => {}
342    }
343}

345/// Ticks the next receive waits; 0 = untimed. A heartbeat caps every wait at
346/// half its interval so the loop re-arms it in time.
347fn receive_ticks(wait: Wait, heartbeat_ticks: u64) -> u64 {
348    let cap = heartbeat_ticks / 2;
349    // 0 would mean "untimed": an explicit tick wait never becomes one.
350    let ticks = match wait {
351        Wait::Block => return cap,
352        Wait::Ticks(ticks) => ticks.max(1),
353        Wait::Poll => 1,
354    };
355    if cap == 0 {
356        ticks
357    } else {
358        ticks.min(cap)
359    }
360}

362#[cfg(test)]
363mod tests {
364    use super::{receive_ticks, swap_envelope, Swap, Wait};
365    use wire::proto::app_envelope::{
366        KIND_MESSAGE, KIND_RESTORE, KIND_SNAPSHOT, MAGIC, SNAPSHOT_LEN,
367    };
368
369    fn window(prefix: &[u8]) -> [u8; 128] {
370        let mut payload = [0u8; 128];
371        payload[..prefix.len()].copy_from_slice(prefix);
372        payload
373    }
374
375    #[test]
376    fn swap_envelopes_are_recognised_by_their_prefix() {
377        let mut snapshot = window(&[MAGIC, KIND_SNAPSHOT]);
378        snapshot[2..SNAPSHOT_LEN].copy_from_slice(&0x0102_0304_0506_0708u64.to_le_bytes());
379        assert_eq!(
380            swap_envelope(&snapshot),
381            Some(Swap::Snapshot(0x0102_0304_0506_0708))
382        );
383        let restore = window(&[MAGIC, KIND_RESTORE, b'4', b'2']);
384        assert_eq!(swap_envelope(&restore), Some(Swap::Restore(42)));
385    }
386
387    #[test]
388    fn a_swap_prefix_with_an_unreadable_id_is_still_a_swap_envelope() {
389        // Never handed to `on_request`, so never answered.
390        assert_eq!(
391            swap_envelope(&window(&[MAGIC, KIND_RESTORE])),
392            Some(Swap::Malformed)
393        );
394        assert_eq!(
395            swap_envelope(&window(&[MAGIC, KIND_RESTORE, b'x'])),
396            Some(Swap::Malformed)
397        );
398        assert_eq!(
399            swap_envelope(&[MAGIC, KIND_SNAPSHOT, 1]),
400            Some(Swap::Malformed)
401        );
402        assert_eq!(
403            swap_envelope(&[MAGIC, KIND_RESTORE][..]),
404            Some(Swap::Malformed)
405        );
406    }
407
408    #[test]
409    fn other_payloads_are_requests() {
410        for payload in [
411            &window(&[MAGIC, KIND_MESSAGE])[..],
412            &window(&[0, KIND_SNAPSHOT])[..],
413            &window(&[MAGIC])[..1],
414            &[][..],
415            &window(&[1, 2, 3])[..],
416        ] {
417            assert_eq!(swap_envelope(payload), None);
418        }
419    }
420
421    #[test]
422    fn a_heartbeat_bounds_every_wait_by_half_its_interval() {
423        assert_eq!(receive_ticks(Wait::Block, 0), 0);
424        assert_eq!(receive_ticks(Wait::Block, 6_000), 3_000);
425        assert_eq!(receive_ticks(Wait::Ticks(10_000), 6_000), 3_000);
426        assert_eq!(receive_ticks(Wait::Ticks(100), 6_000), 100);
427        assert_eq!(receive_ticks(Wait::Ticks(0), 0), 1);
428        assert_eq!(receive_ticks(Wait::Ticks(0), 6_000), 1);
429    }
