1//! Task spawning and yielding.
2//!
3//! A "thread" in Cellos is another task of the *same* Cell: the kernel gives it
4//! its own stacks and a new TID, and inherits the cell's identity on every axis it
5//! gates (CellId, capability set, syscall allowlist, PKU domain). Threads therefore
6//! share the cell's heap, quota, and address space — the isolation boundary stays
7//! the cell, not the thread.
8//!
9//! Per-thread state is the cell's own business, carried in the user thread pointer
10//! (`SetTlsBase`): a thread that wants its own TLS block allocates one and claims it
11//! as its first action. A thread inherits its creator's base until it does.
12//!
13//! A thread that returns from its closure exits on its own; only the cell's *root*
14//! task exit retires the whole cell generation.
15
16#![allow(unsafe_code)]
17
18use crate::syscall::{sys_exit, sys_spawn, SyscallResult};
19use alloc::boxed::Box;
20
21extern "C" fn thread_entry(arg: usize) {
22    // SAFETY: `spawn` boxes exactly this shape and hands over the raw pointer; the
23    // thread owns it from here and drops it when the closure is moved out.
24    let outer: Box<Box<dyn FnOnce() + Send + 'static>> = unsafe { Box::from_raw(arg as *mut _) };
25    let inner: Box<dyn FnOnce() + Send + 'static> = *outer;
26    inner();
27
28    // A worker exit terminates only this thread and wakes any `Wait` joiner with
29    // the exit code; it does not retire the cell (the root task's exit does that).
30    sys_exit(0);
31}
32
33/// Spawns a new thread with a closure.
34pub fn spawn<F>(f: F) -> SyscallResult
35where
36    F: FnOnce() + Send + 'static,
37{
38    // 1. Box the closure (Fat Pointer to Closure)
39    let inner: Box<dyn FnOnce() + Send + 'static> = Box::new(f);
40
41    // 2. Box the Fat Pointer (Thin Pointer to Fat Pointer)
42    let outer = Box::new(inner);
43    // 3. Get raw pointer
44    let ptr = Box::into_raw(outer) as usize;
45
46    // 4. Call syscall with static entry point
47    sys_spawn(thread_entry as *const () as usize, ptr)
48}

50/// Yield the current task, letting the scheduler run another one.
51pub fn yield_now() {
52    crate::syscall::sys_yield();
53}
