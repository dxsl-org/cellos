2//! Typed local and remote Cell endpoint descriptors.
3//!
4//! Local calls use direct sender-masked IPC. Remote endpoints carry only
5//! authenticated route metadata; Phase 04 deliberately exposes no transmit
6//! method while remote dispatch remains disabled.
7
8use api::ipc::IPC_BUF_SIZE;
9use api::services::cluster::{CellNetId, ClusterId};
10use core::marker::PhantomData;
11use serde::{Deserialize, Serialize};
12use types::c2c::{RelativeDeadline, RetryClass, ServerEpoch};
13
14use crate::{ipc, ViError, ViResult};

16/// Typed request/response contract shared by local and remote endpoints.
17pub trait CellMethod {
18    /// Serialized request type for this method.
19    type Request: Serialize;
20    /// Response type, optionally borrowing from the caller's receive buffer.
21    type Response<'a>: Deserialize<'a>;
22
23    /// Stable remote service identifier.
24    const SERVICE_ID: u16;
25    /// Stable exported-method identifier within the service.
26    const EXPORT_ID: u16;
27    /// Retry safety applied to transport loss or indeterminate completion.
28    const RETRY_CLASS: RetryClass;
29}
30
31/// Invalid endpoint metadata that cannot identify a routable target.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub enum EndpointError {
34    InvalidLocalTid,
35    InvalidRemoteIdentity,
36}
37
38/// Observable remote-call outcomes; never collapse these into local IPC errors.
39#[derive(Clone, Copy, Debug, PartialEq, Eq)]
40pub enum RemoteCallError {
41    NoService,
42    Unreachable,
43    Timeout,
44    Busy,
45    Indeterminate,
46    AuthFailed,
47    ProtocolError,
48    NotSupported,
49}

51/// Direct local endpoint. Calls never resolve or contact the net-broker.
52pub struct LocalEndpoint<M: CellMethod> {
53    tid: usize,
54    marker: PhantomData<fn() -> M>,
55}
56
57impl<M: CellMethod> LocalEndpoint<M> {
58    /// Bind a typed endpoint to `tid`, rejecting the reserved zero TID.
59    pub const fn new(tid: usize) -> Result<Self, EndpointError> {
60        if tid == 0 {
61            return Err(EndpointError::InvalidLocalTid);
62        }
63        Ok(Self {
64            tid,
65            marker: PhantomData,
66        })
67    }
68
69    /// Return the direct local service TID.
70    pub const fn tid(&self) -> usize {
71        self.tid
72    }
73
74    /// Execute one typed request/reply exchange directly with the local TID.
75    ///
76    /// The returned value may borrow from `response_buffer`.
77    ///
78    /// # Errors
79    /// Returns `InvalidArgument` for an oversized request and `IO` for send,
80    /// receive, wrong-sender, or decode failures.
81    pub fn call<'a>(
82        &self,
83        request: &M::Request,
84        response_buffer: &'a mut [u8; IPC_BUF_SIZE],
85    ) -> ViResult<M::Response<'a>> {
86        let mut send_buffer = [0u8; IPC_BUF_SIZE];
87        match ipc::service_call_typed(self.tid, request, &mut send_buffer, response_buffer) {
88            Ok(response) => Ok(response),
89            Err(ipc::IpcError::Encode) => Err(ViError::InvalidArgument),
90            Err(ipc::IpcError::Send)
91            | Err(ipc::IpcError::Recv)
92            | Err(ipc::IpcError::WrongSender)
93            | Err(ipc::IpcError::Decode) => Err(ViError::IO),
94        }
95    }
96}

98/// Authenticated metadata for one remote exported-server incarnation.
99pub struct RemoteEndpoint<M: CellMethod> {
100    destination: CellNetId,
101    cluster: ClusterId,
102    server_epoch: ServerEpoch,
103    marker: PhantomData<fn() -> M>,
104}
105
106impl<M: CellMethod> RemoteEndpoint<M> {
107    /// Construct a remote descriptor learned from authenticated discovery.
108    ///
109    /// # Errors
110    /// Rejects zero node, cluster, service, or export identities.
111    pub const fn new(
112        destination: CellNetId,
113        cluster: ClusterId,
114        server_epoch: ServerEpoch,
115    ) -> Result<Self, EndpointError> {
116        if all_zero(&destination.0) || cluster.0 == 0 || M::SERVICE_ID == 0 || M::EXPORT_ID == 0 {
117            return Err(EndpointError::InvalidRemoteIdentity);
118        }
119        Ok(Self {
120            destination,
121            cluster,
122            server_epoch,
123            marker: PhantomData,
124        })
125    }
126
127    /// Refuse remote transmission while the protected-provider and dispatch
128    /// gates remain closed.
129    ///
130    /// The validated relative deadline is mandatory, but no argument is encoded
131    /// or sent. This boundary preserves the final typed return shape without
132    /// contacting the net-broker.
133    ///
134    /// # Errors
135    /// Always returns `RemoteCallError::NotSupported` in Phase 04.
136    pub fn call<'a>(
137        &self,
138        _request: &M::Request,
139        _relative_deadline: RelativeDeadline,
140        _response_buffer: &'a mut [u8; IPC_BUF_SIZE],
141    ) -> Result<M::Response<'a>, RemoteCallError> {
142        Err(RemoteCallError::NotSupported)
143    }
144
145    /// Return the authenticated destination node.
146    pub const fn destination(&self) -> CellNetId {
147        self.destination
148    }
149
150    /// Return the destination cluster used for routing.
151    pub const fn cluster(&self) -> ClusterId {
152        self.cluster
153    }
154
155    /// Return the observed live server incarnation.
156    pub const fn server_epoch(&self) -> ServerEpoch {
157        self.server_epoch
158    }
159
160    /// Return the method's stable service identifier.
161    pub const fn service_id(&self) -> u16 {
162        M::SERVICE_ID
163    }
164
165    /// Return the method's stable export identifier.
166    pub const fn export_id(&self) -> u16 {
167        M::EXPORT_ID
168    }
169
170    /// Return the method's declared retry safety.
171    pub const fn retry_class(&self) -> RetryClass {
172        M::RETRY_CLASS
173    }
174}

176/// Deliberate locality union. Callers must match before invoking local IPC or
177/// constructing a future remote envelope.
178pub enum CellEndpoint<M: CellMethod> {
179    Local(LocalEndpoint<M>),
180    Remote(RemoteEndpoint<M>),
181}

183const fn all_zero(bytes: &[u8; 32]) -> bool {
184    let mut index = 0;
185    while index < bytes.len() {
186        if bytes[index] != 0 {
187            return false;
188        }
189        index += 1;
190    }
191    true
192}
