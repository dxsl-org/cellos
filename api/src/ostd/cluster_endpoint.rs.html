2//! Typed local and remote Cell endpoint descriptors.
3//!
4//! Local calls use direct sender-masked IPC. Remote endpoints carry only
5//! authenticated route metadata; Phase 04 deliberately exposes no transmit
6//! method while remote dispatch remains disabled.

17/// Typed request/response contract shared by local and remote endpoints.
18pub trait CellMethod {
19    /// Serialized request type for this method.
20    type Request: Serialize;
21    /// Response type, optionally borrowing from the caller's receive buffer.
22    type Response<'a>: Deserialize<'a>;
23
24    /// Stable remote service identifier.
25    const SERVICE_ID: u16;
26    /// Stable exported-method identifier within the service.
27    const EXPORT_ID: u16;
28    /// Retry safety applied to transport loss or indeterminate completion.
29    const RETRY_CLASS: RetryClass;
30}
31
32/// Invalid endpoint metadata that cannot identify a routable target.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub enum EndpointError {
35    InvalidLocalTid,
36    InvalidRemoteIdentity,
37    /// No live provider binding exists for the method's service.
38    NoLiveBinding,
39}
40
41/// Observable remote-call outcomes; never collapse these into local IPC errors.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub enum RemoteCallError {
44    NoService,
45    Unreachable,
46    Timeout,
47    Busy,
48    Indeterminate,
49    AuthFailed,
50    ProtocolError,
51    NotSupported,
52}

54/// Direct local endpoint. Calls never resolve or contact the net-broker.
55pub struct LocalEndpoint<M: CellMethod> {
56    tid: usize,
57    /// The provider binding this endpoint resolved, when it was created by
58    /// [`LocalEndpoint::bind`]. A tid-only endpoint has none, and then this type makes
59    /// no claim about the provider behind the tid.
60    binding: Option<ServiceBinding>,
61    marker: PhantomData<fn() -> M>,
62}
63
64impl<M: CellMethod> LocalEndpoint<M> {
65    /// Bind a typed endpoint to `tid`, rejecting the reserved zero TID.
66    ///
67    /// This trusts a caller-supplied tid: it resolves nothing, so the endpoint carries no
68    /// provider identity and a failed exchange is reported as `IO` rather than classified
69    /// against the registry. Use [`bind`][Self::bind] when the service identity matters.
70    pub const fn new(tid: usize) -> Result<Self, EndpointError> {
71        if tid == 0 {
72            return Err(EndpointError::InvalidLocalTid);
73        }
74        Ok(Self {
75            tid,
76            binding: None,
77            marker: PhantomData,
78        })
79    }
80
81    /// Resolve the method's service to its live provider binding and bind to it.
82    ///
83    /// Where [`new`][Self::new] trusts a caller-supplied tid, this asks the kernel which
84    /// provider incarnation is live for `M::SERVICE_ID` and refuses when there is none —
85    /// no binding means the call is refused, never sent to a tid that may belong to a
86    /// dead incarnation. The endpoint then carries that identity, so a later exchange
87    /// against a provider that has since been replaced is reported as `NotFound` instead
88    /// of a generic I/O error.
89    ///
90    /// # Errors
91    /// - `EndpointError::NoLiveBinding` — nothing is registered for `M::SERVICE_ID`, or
92    ///   the provider is paused behind a hot-swap: absent bindings are refusals, not
93    ///   retries.
94    pub fn bind() -> Result<Self, EndpointError> {
95        let mut record = [0u8; SERVICE_BINDING_LEN];
96        match crate::syscall::sys_lookup_service_bound(M::SERVICE_ID, &mut record) {
97            Some(binding) => Ok(Self {
98                tid: binding.tid as usize,
99                binding: Some(binding),
100                marker: PhantomData,
101            }),
102            None => Err(EndpointError::NoLiveBinding),
103        }
104    }
105
106    /// Return the direct local service TID.
107    pub const fn tid(&self) -> usize {
108        self.tid
109    }
110
111    /// The provider binding this endpoint resolved, or `None` for a tid-only endpoint.
112    pub const fn binding(&self) -> Option<ServiceBinding> {
113        self.binding
114    }
115
116    /// Execute one typed request/reply exchange directly with the local TID.
117    ///
118    /// The returned value may borrow from `response_buffer`.
119    ///
120    /// # Errors
121    /// Returns `InvalidArgument` for an oversized request and `IO` for send,
122    /// receive, wrong-sender, or decode failures. An endpoint bound with
123    /// [`bind`][Self::bind] additionally reports `NotFound` when the provider it resolved
124    /// is no longer the live one: the exchange was refused, not delivered under a stale
125    /// descriptor.
126    pub fn call<'a>(
127        &self,
128        request: &M::Request,
129        response_buffer: &'a mut [u8; IPC_BUF_SIZE],
130    ) -> ViResult<M::Response<'a>> {
131        let mut send_buffer = [0u8; IPC_BUF_SIZE];
132        match ipc::service_call_typed(self.tid, request, &mut send_buffer, response_buffer) {
133            Ok(response) => Ok(response),
134            Err(ipc::IpcError::Encode) => Err(ViError::InvalidArgument),
135            Err(ipc::IpcError::Decode)
136            | Err(ipc::IpcError::InvalidOperation)
137            | Err(ipc::IpcError::BufferTooSmall) => Err(ViError::IO),
138            Err(ipc::IpcError::Busy)
139            | Err(ipc::IpcError::PreDispatchTimeout)
140            | Err(ipc::IpcError::Indeterminate)
141            | Err(ipc::IpcError::Cancelled) => Err(ViError::IO),
142            Err(ipc::IpcError::Send)
143            | Err(ipc::IpcError::Recv)
144            | Err(ipc::IpcError::WrongSender)
145            | Err(ipc::IpcError::PeerGone) => match self.binding {
146                // A tid-only endpoint knows no provider identity, so it cannot tell a
147                // replaced provider from a failed one and must not guess.
148                None => Err(ViError::IO),
149                Some(held) => {
150                    let mut record = [0u8; SERVICE_BINDING_LEN];
151                    let live = crate::syscall::sys_lookup_service_bound(M::SERVICE_ID, &mut record);
152                    match crate::service::classify_call_failure(held, live) {
153                        crate::service::CallFailure::StaleBinding => Err(ViError::NotFound),
154                        crate::service::CallFailure::ProviderError => Err(ViError::IO),
155                    }
156                }
157            },
158        }
159    }
160}

162/// Authenticated metadata for one remote exported-server incarnation.
163pub struct RemoteEndpoint<M: CellMethod> {
164    destination: CellNetId,
165    cluster: ClusterId,
166    server_epoch: ServerEpoch,
167    marker: PhantomData<fn() -> M>,
168}
169
170impl<M: CellMethod> RemoteEndpoint<M> {
171    /// Construct a remote descriptor learned from authenticated discovery.
172    ///
173    /// # Errors
174    /// Rejects zero node, cluster, service, or export identities.
175    pub const fn new(
176        destination: CellNetId,
177        cluster: ClusterId,
178        server_epoch: ServerEpoch,
179    ) -> Result<Self, EndpointError> {
180        if all_zero(&destination.0) || cluster.0 == 0 || M::SERVICE_ID == 0 || M::EXPORT_ID == 0 {
181            return Err(EndpointError::InvalidRemoteIdentity);
182        }
183        Ok(Self {
184            destination,
185            cluster,
186            server_epoch,
187            marker: PhantomData,
188        })
189    }
190
191    /// Refuse remote transmission while the protected-provider and dispatch
192    /// gates remain closed.
193    ///
194    /// The validated relative deadline is mandatory, but no argument is encoded
195    /// or sent. This boundary preserves the final typed return shape without
196    /// contacting the net-broker.
197    ///
198    /// # Errors
199    /// Always returns `RemoteCallError::NotSupported` in Phase 04.
200    pub fn call<'a>(
201        &self,
202        _request: &M::Request,
203        _relative_deadline: RelativeDeadline,
204        _response_buffer: &'a mut [u8; IPC_BUF_SIZE],
205    ) -> Result<M::Response<'a>, RemoteCallError> {
206        Err(RemoteCallError::NotSupported)
207    }
208
209    /// Return the authenticated destination node.
210    pub const fn destination(&self) -> CellNetId {
211        self.destination
212    }
213
214    /// Return the destination cluster used for routing.
215    pub const fn cluster(&self) -> ClusterId {
216        self.cluster
217    }
218
219    /// Return the observed live server incarnation.
220    pub const fn server_epoch(&self) -> ServerEpoch {
221        self.server_epoch
222    }
223
224    /// Return the method's stable service identifier.
225    pub const fn service_id(&self) -> u16 {
226        M::SERVICE_ID
227    }
228
229    /// Return the method's stable export identifier.
230    pub const fn export_id(&self) -> u16 {
231        M::EXPORT_ID
232    }
233
234    /// Return the method's declared retry safety.
235    pub const fn retry_class(&self) -> RetryClass {
236        M::RETRY_CLASS
237    }
238}

240/// Deliberate locality union. Callers must match before invoking local IPC or
241/// constructing a future remote envelope.
242pub enum CellEndpoint<M: CellMethod> {
243    Local(LocalEndpoint<M>),
244    Remote(RemoteEndpoint<M>),
245}

247const fn all_zero(bytes: &[u8; 32]) -> bool {
248    let mut index = 0;
249    while index < bytes.len() {
250        if bytes[index] != 0 {
251            return false;
252        }
253        index += 1;
254    }
255    true
256}
