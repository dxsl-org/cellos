12/// A compositor surface backed by a Grant buffer the app cell owns directly.
13///
14/// The app writes pixels into [`Self::pixels_mut`] and calls [`Self::damage`] or
15/// [`Self::damage_all`] to request re-blending. `ViSurface` is `!Send`: its
16/// Grant pointer must remain on the cell's task.
17pub struct ViSurface {
18    pub(super) comp_tid: usize,
19    pub(super) cap: u32,
20    pub(super) reg_id: usize,
21    pub(super) retired_reg_id: Option<usize>,
22    pub(super) staged_reg_id: Option<usize>,
23    pub(super) ptr: *mut u8,
24    pub(super) width: u32,
25    pub(super) height: u32,
26    pub(super) fmt: PixelFormat,
27    _not_send: core::marker::PhantomData<*mut ()>,
28}
29
30impl ViSurface {
31    /// Create an interactive `width` × `height` BGRA surface owned by `comp_tid`.
32    ///
33    /// Returns a Grant-backed surface whose pixels are writable by this task.
34    ///
35    /// # Errors
36    /// Returns `OutOfMemory` when Grant registration fails and `IO` when Grant
37    /// mapping or compositor surface setup fails.
38    pub fn create(comp_tid: usize, width: u32, height: u32, fmt: PixelFormat) -> ViResult<Self> {
39        Self::create_with_role(comp_tid, width, height, fmt, SurfaceRole::Interactive)
40    }
41
42    /// Create a non-focusable `width` × `height` background surface for `comp_tid`.
43    ///
44    /// # Errors
45    /// Returns `OutOfMemory` when Grant registration fails and `IO` when Grant
46    /// mapping or compositor surface setup fails.
47    pub fn create_background(
48        comp_tid: usize,
49        width: u32,
50        height: u32,
51        fmt: PixelFormat,
52    ) -> ViResult<Self> {
53        Self::create_with_role(comp_tid, width, height, fmt, SurfaceRole::Background)
54    }
55
56    fn create_with_role(
57        comp_tid: usize,
58        width: u32,
59        height: u32,
60        fmt: PixelFormat,
61        role: SurfaceRole,
62    ) -> ViResult<Self> {
63        let size = (width * height * fmt.bpp()) as usize;
64        let reg_id = sys_grant_register(size).ok_or(ViError::OutOfMemory)?;
65        sys_grant_share(reg_id, comp_tid, 0 /* ReadOnly */);
66        let ptr = sys_grant_slice(reg_id).ok_or_else(|| {
67            sys_grant_unregister(reg_id);
68            ViError::IO
69        })?;
70        let cap = ipc::create_surface(comp_tid, width, height, role).inspect_err(|_| {
71            sys_grant_unregister(reg_id);
72        })?;
73        ipc::attach_grant(comp_tid, cap, reg_id, width, height, fmt).inspect_err(|_| {
74            let _ = ipc::destroy_surface(comp_tid, cap);
75            sys_grant_unregister(reg_id);
76        })?;
77        Ok(Self {
78            comp_tid,
79            cap,
80            reg_id,
81            retired_reg_id: None,
82            staged_reg_id: None,
83            ptr,
84            width,
85            height,
86            fmt,
87            _not_send: core::marker::PhantomData,
88        })
89    }
90
91    /// Return mutable pixels for the current Grant-backed surface dimensions.
92    pub fn pixels_mut(&mut self) -> &mut [u8] {
93        let len = (self.width * self.height * self.fmt.bpp()) as usize;
94        // SAFETY: this is our registered Grant; `&mut self` prevents concurrent use.
95        unsafe { core::slice::from_raw_parts_mut(self.ptr, len) }
96    }
97
98    /// Return the current row stride in bytes.
99    pub fn stride(&self) -> usize {
100        self.width as usize * self.fmt.bpp() as usize
101    }
102
103    /// Return the current surface width in pixels.
104    pub fn width(&self) -> u32 {
105        self.width
106    }
107
108    /// Return the current surface height in pixels.
109    pub fn height(&self) -> u32 {
110        self.height
111    }
112
113    /// Return the capability that identifies this surface in lifecycle events.
114    pub fn cap(&self) -> u32 {
115        self.cap
116    }
117
118    /// Mark the surface-local `rect` dirty for a later compositor blend.
119    pub fn damage(&self, rect: Rect) {
120        let msg = DamageNotify {
121            opcode: compositor_ops::DAMAGE_NOTIFY,
122            _pad: [0; 3],
123            cap: self.cap,
124            rect,
125        };
126        sys_send(self.comp_tid, &msg.encode());
127    }
128
129    /// Mark every current surface pixel dirty for a later compositor blend.
130    pub fn damage_all(&self) {
131        self.damage(Rect {
132            x: 0,
133            y: 0,
134            w: self.width,
135            h: self.height,
136        });
137    }
138
139    /// Move the surface's content origin to screen coordinates (`x`, `y`).
140    pub fn move_to(&self, x: i32, y: i32) {
141        let mut buf = [0u8; 13];
142        buf[0] = compositor_ops::MOVE_SURFACE;
143        buf[1..9].copy_from_slice(&(self.cap as u64).to_le_bytes());
144        buf[9..13].copy_from_slice(&x.to_le_bytes());
145        let mut buf17 = [0u8; 17];
146        buf17[..13].copy_from_slice(&buf);
147        buf17[13..17].copy_from_slice(&y.to_le_bytes());
148        sys_send(self.comp_tid, &buf17);
149    }
150
151    /// Raise this surface above every other compositor surface.
152    pub fn raise(&self) {
153        let mut buf = [0u8; 9];
154        buf[0] = compositor_ops::RAISE_SURFACE;
155        buf[1..9].copy_from_slice(&(self.cap as u64).to_le_bytes());
156        sys_send(self.comp_tid, &buf);
157    }
158}
