11/// Poll up to `max` retained compositor lifecycle events.
12///
13/// The fixed-capacity result does not allocate. Only frames received from the
14/// registered compositor are routed; forwarded input frames remain available to
15/// [`crate::input::poll_events`].
16pub fn poll_surface_events(max: usize) -> heapless::Vec<SurfaceEvent, MAX_SURFACE_EVENTS> {
17    let limit = max.min(MAX_SURFACE_EVENTS);
18    let mut events = heapless::Vec::new();
19    while events.len() < limit {
20        let Some(event) = dispatcher::take_surface() else {
21            break;
22        };
23        let _ = events.push(event);
24    }
25    let Some(compositor_tid) = sys_lookup_service(service::COMPOSITOR) else {
26        return events;
27    };
28    for _ in 0..limit.saturating_sub(events.len()) {
29        if !drain_compositor_once(compositor_tid) {
30            break;
31        }
32        while events.len() < limit {
33            let Some(event) = dispatcher::take_surface() else {
34                break;
35            };
36            let _ = events.push(event);
37        }
38    }
39    events
40}
41
42/// Receive and route one frame from `compositor_tid`.
43pub(crate) fn drain_compositor_once(compositor_tid: usize) -> bool {
44    let mut buffer = [0u8; 72];
45    match crate::syscall::sys_try_recv(compositor_tid, &mut buffer) {
46        SyscallResult::Ok(sender) if sender == compositor_tid => {
47            route_compositor_frame(&buffer);
48            true
49        }
50        _ => false,
51    }
52}
