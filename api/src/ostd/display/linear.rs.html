1//! Linear-style raster primitives shared by the compositor and ViUI.
2//!
3//! Contract: every stroke is exactly one device pixel wide and every edge lands
4//! on the pixel grid (no anti-aliasing), so window chrome, boxes and icons stay
5//! sharp at any position, scale step and background. A rounded shape derives its
6//! fill and its outline from the same per-row inset function, so the outline is
7//! exactly the 4-connected boundary of the fill.
8//!
9//! Buffers are 32-bit BGRA rows with a byte stride. Opaque colors are copied;
10//! translucent colors blend with the destination, as does the drop shadow
11//! ([`Shadow`]).

32/// Per-corner radii in pixels. Each radius is clamped to half the shape's
33/// shorter side when used.
34#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
35pub struct Radii {
36    pub top_left: u32,
37    pub top_right: u32,
38    pub bottom_right: u32,
39    pub bottom_left: u32,
40}
41
42impl Radii {
43    pub const fn all(r: u32) -> Self {
44        Self { top_left: r, top_right: r, bottom_right: r, bottom_left: r }
45    }
46}

48/// Axis-aligned rectangle with independently rounded corners.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub struct RoundRect {
51    pub x: i32,
52    pub y: i32,
53    pub w: u32,
54    pub h: u32,
55    pub radii: Radii,
56}
57
58impl RoundRect {
59    pub const fn new(x: i32, y: i32, w: u32, h: u32, radii: Radii) -> Self {
60        Self { x, y, w, h, radii }
61    }
62
63    fn clamp(&self, r: u32) -> u32 {
64        r.min(self.w / 2).min(self.h / 2)
65    }
66
67    /// Pixels left uncovered at the start and at the end of `row` (0 = top).
68    pub fn insets(&self, row: u32) -> (u32, u32) {
69        if row >= self.h {
70            return (0, 0);
71        }
72        let down = self.h - 1 - row;
73        let left = corner_inset(self.clamp(self.radii.top_left), row)
74            .max(corner_inset(self.clamp(self.radii.bottom_left), down));
75        let right = corner_inset(self.clamp(self.radii.top_right), row)
76            .max(corner_inset(self.clamp(self.radii.bottom_right), down));
77        (left, right)
78    }
79
80    /// Outline pixels of `row`, relative to `x`: the half-open runs
81    /// `[left.0, left.1)` and `[right.0, right.1)`. Top and bottom rows are one run.
82    pub fn outline(&self, row: u32) -> ((u32, u32), (u32, u32)) {
83        let (li, ri) = self.insets(row);
84        let right_edge = self.w.saturating_sub(ri).max(li);
85        if row == 0 || row + 1 >= self.h {
86            return ((li, right_edge), (right_edge, right_edge));
87        }
88        // The row nearer the closest horizontal edge has the wider inset; the run
89        // bridges to it so diagonal steps stay 4-connected.
90        let toward = if row < self.h / 2 { row - 1 } else { row + 1 };
91        let (lo, ro) = self.insets(toward);
92        let left_end = lo.max(li + 1).min(right_edge);
93        let right_start = self.w.saturating_sub(ro).min(right_edge.saturating_sub(1)).max(li);
94        ((li, left_end), (right_start, right_edge))
95    }
96}

98/// Horizontal inset of row `row` (0 = the edge row) of a quarter circle of
99/// radius `r` sampled at pixel centers. Rows at or beyond `r` have no inset.
100pub const fn corner_inset(r: u32, row: u32) -> u32 {
101    if row >= r {
102        return 0;
103    }
104    // Doubled coordinates keep the half-pixel center offset integral:
105    // inset = r - round(sqrt(r^2 - (r - row - 0.5)^2)).
106    let d = 2 * (r - row) - 1;
107    r - (isqrt(4 * r * r - d * d) + 1) / 2
108}

110pub(crate) const fn isqrt(n: u32) -> u32 {
111    if n < 2 {
112        return n;
113    }
114    let mut x = n;
115    let mut y = n / 2 + (n & 1);
116    while y < x {
117        x = y;
118        y = (x + n / x) / 2;
119    }
120    x
121}
