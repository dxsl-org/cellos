5/// Soft drop shadow, alpha-blended over what is already painted.
6///
7/// The caller must have painted everything below the caster inside the clip
8/// first (the compositor repaints its dirty region bottom-to-top), so each
9/// pixel is blended exactly once per frame. Opacity falls off as `(1 - t)^2`
10/// from `opacity / 255` at the caster's outline to zero at `spread` pixels;
11/// a per-call table indexed by distance (1/16 px) holds it, so a ring pixel
12/// costs one integer square root and one blend. The shadow shape is the
13/// caster moved by `(offset_x, offset_y)`; pixels of the caster itself are
14/// never written.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub struct Shadow {
17    pub color: Bgra,
18    pub opacity: u8,
19    /// Fade distance in pixels; values above [`MAX_SPREAD`] are clamped.
20    pub spread: u32,
21    pub offset_x: i32,
22    pub offset_y: i32,
23}
24
25pub const MAX_SPREAD: u32 = 32;
26/// Callers cast one window (tab + body) at a time; a fixed bound keeps the
27/// per-row interior spans on the stack.
28const MAX_CASTERS: usize = 4;
29const STEPS: usize = (MAX_SPREAD * 16) as usize + 1;
30
31impl Canvas<'_> {
32    /// Blend `shadow` cast by the union of `casters` (at most four shapes).
33    pub fn drop_shadow(&mut self, casters: &[RoundRect], shadow: &Shadow) {
34        let casters = &casters[..casters.len().min(MAX_CASTERS)];
35        let spread = shadow.spread.min(MAX_SPREAD);
36        if casters.is_empty() || shadow.opacity == 0 || spread == 0 {
37            return;
38        }
39        let ramp = ramp(shadow.opacity, spread);
40        let reach = i64::from(spread);
41        let offset = (i64::from(shadow.offset_x), i64::from(shadow.offset_y));
42        let (mut rx0, mut ry0, mut rx1, mut ry1) = (i64::MAX, i64::MAX, i64::MIN, i64::MIN);
43        for c in casters {
44            rx0 = rx0.min(i64::from(c.x) + offset.0 - reach);
45            ry0 = ry0.min(i64::from(c.y) + offset.1 - reach);
46            rx1 = rx1.max(i64::from(c.x) + offset.0 + i64::from(c.w) + reach);
47            ry1 = ry1.max(i64::from(c.y) + offset.1 + i64::from(c.h) + reach);
48        }
49        let (rx0, ry0) = (rx0.max(self.x0), ry0.max(self.y0));
50        let (rx1, ry1) = (rx1.min(self.x1), ry1.min(self.y1));
51
52        for y in ry0..ry1 {
53            let mut interiors = [(0i64, 0i64); MAX_CASTERS];
54            let mut count = 0;
55            for c in casters {
56                let row = y - i64::from(c.y);
57                if row < 0 || row >= i64::from(c.h) {
58                    continue;
59                }
60                let (li, ri) = c.insets(row as u32);
61                let x = i64::from(c.x);
62                interiors[count] = (x + i64::from(li), x + i64::from(c.w) - i64::from(ri));
63                count += 1;
64            }
65            let interiors = &mut interiors[..count];
66            interiors.sort_unstable();
67            let mut x = rx0;
68            for &(start, end) in interiors.iter() {
69                if start > x {
70                    self.shade_run(y, x, start.min(rx1), casters, offset, &ramp, shadow.color);
71                }
72                x = x.max(end);
73            }
74            if x < rx1 {
75                self.shade_run(y, x, rx1, casters, offset, &ramp, shadow.color);
76            }
77        }
78    }
79
80    #[allow(clippy::too_many_arguments)]
81    fn shade_run(
82        &mut self,
83        y: i64,
84        x0: i64,
85        x1: i64,
86        casters: &[RoundRect],
87        offset: (i64, i64),
88        ramp: &[u8; STEPS],
89        color: Bgra,
90    ) {
91        let last = (ramp.len() - 1) as i64;
92        for x in x0..x1 {
93            let d16 = casters
94                .iter()
95                .map(|c| distance16(c, offset, x, y))
96                .min()
97                .unwrap_or(i64::MAX);
98            let alpha = ramp[d16.min(last) as usize];
99            if alpha > 0 {
100                self.pixel(x, y, [color[0], color[1], color[2], alpha]);
101            }
102        }
103    }
104}

106/// Opacity per distance step (1/16 px): `opacity * (1 - d / spread)^2`, zero
107/// from `spread` on (the table tail stays zero).
108fn ramp(opacity: u8, spread: u32) -> [u8; STEPS] {
109    let mut out = [0u8; STEPS];
110    let span = i64::from(spread) * 16;
111    for (d, alpha) in out.iter_mut().enumerate().take(span as usize) {
112        let left = span - d as i64;
113        *alpha = (i64::from(opacity) * left * left / (span * span)) as u8;
114    }
115    out
116}
117
118/// Distance (1/16 px, 0 inside) from pixel `(x, y)` to the caster moved by
119/// `offset`, measured from its rounded outline.
120fn distance16(c: &RoundRect, offset: (i64, i64), x: i64, y: i64) -> i64 {
121    let (left, top) = (i64::from(c.x) + offset.0, i64::from(c.y) + offset.1);
122    let (w, h) = (i64::from(c.w), i64::from(c.h));
123    let west = 2 * x < 2 * left + w - 1;
124    let north = 2 * y < 2 * top + h - 1;
125    let radius = match (north, west) {
126        (true, true) => c.radii.top_left,
127        (true, false) => c.radii.top_right,
128        (false, false) => c.radii.bottom_right,
129        (false, true) => c.radii.bottom_left,
130    };
131    let r = i64::from(radius.min(c.w / 2).min(c.h / 2));
132    let reach = i64::from(MAX_SPREAD) + r;
133    let dx = (left + r - x).max(x - (left + w - 1 - r)).clamp(0, reach);
134    let dy = (top + r - y).max(y - (top + h - 1 - r)).clamp(0, reach);
135    let length16 = i64::from(isqrt((256 * (dx * dx + dy * dy)) as u32));
136    (length16 - 16 * r).max(0)
137}
