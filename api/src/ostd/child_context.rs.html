2//! Canonical immutable invocation context; no shared spawn stash or secret logging.
3use alloc::vec::Vec;
4use api::cell_session::{KernelOp,ObjectRefV1,Reader,Writer,Status,HEADER_LEN,CONTEXT_MAX,CONTEXT_HEADER_LEN,CONTEXT_MAX_GRANTS,GRANT_LEN,GRANT_KIND_INPUT_FRONTEND};

6/// Context grant slots. Kind-1 endpoint slots: Stdin/Control readers, Stdout/Stderr and
7/// Completion writers. Completion carries the child's completion-record stream; it is
8/// outside the stdio projection input inspects. Slots 6..=7 are reserved.
9#[derive(Clone,Copy,Debug,PartialEq,Eq)]
10#[repr(u16)]
11pub enum GrantSlot { Stdin=0, Stdout=1, Stderr=2, Control=3, Cwd=4, Completion=5, Frontend=8 }
12#[derive(Clone,Copy,Debug,PartialEq,Eq)]
13pub struct Grant { pub kind:u16, pub rights:u16, pub slot:GrantSlot, pub owner_service:u64, pub object:ObjectRefV1 }
14impl Grant {
15    pub fn endpoint(slot:GrantSlot,object:ObjectRefV1,rights:u16)->Result<Self,Status> {
16        let g=Self{kind:1,rights,slot,owner_service:0,object}; g.validate()?; Ok(g)
17    }
18    pub fn cwd(object:ObjectRefV1)->Result<Self,Status> {
19        let g=Self{kind:2,rights:0,slot:GrantSlot::Cwd,owner_service:api::syscall::service::VFS as u64,object}; g.validate()?; Ok(g)
20    }
21    /// Kind-6 InputFrontend grant: the input-issued frontend Ref, admitted by the
22    /// kernel only with the input service's bound receipt for this exact child.
23    pub fn frontend(object:ObjectRefV1)->Result<Self,Status> {
24        let g=Self{kind:GRANT_KIND_INPUT_FRONTEND,rights:0,slot:GrantSlot::Frontend,owner_service:api::syscall::service::INPUT as u64,object}; g.validate()?; Ok(g)
25    }
26    fn validate(self)->Result<(),Status> {
27        if self.object.id==0 || self.object.generation==0 {return Err(Status::Malformed)}
28        match self.kind {
29            1=> {
30                let dir=match self.slot {GrantSlot::Stdin|GrantSlot::Control=>1,GrantSlot::Stdout|GrantSlot::Stderr|GrantSlot::Completion=>2,_=>return Err(Status::Malformed)};
31                if self.owner_service!=0 || self.rights & !31 !=0 || self.rights & 3 !=dir {return Err(Status::Malformed)}
32            }
33            2=>if self.slot!=GrantSlot::Cwd || self.rights!=0 || self.owner_service!=api::syscall::service::VFS as u64 {return Err(Status::Malformed)},
34            3..=5=>return Err(Status::Unsupported),
35            GRANT_KIND_INPUT_FRONTEND=>if self.slot!=GrantSlot::Frontend || self.rights!=0 || self.owner_service!=api::syscall::service::INPUT as u64 {return Err(Status::Malformed)},
36            _=>return Err(Status::Malformed),
37        } Ok(())
38    }
39    fn encode(self,out:&mut[u8])->Result<(),Status> {
40        self.validate()?; let mut w=Writer::new(out); w.u16(self.kind)?;w.u16(self.rights)?;w.u16(self.slot as u16)?;w.u16(0)?;w.u64(self.owner_service)?;w.reference(self.object)?;w.u64(0)
41    }
42    fn decode(raw:&[u8])->Result<Self,Status> {
43        let mut r=Reader::new(raw); let kind=r.u16()?;let rights=r.u16()?;
44        let slot=match r.u16()? {0=>GrantSlot::Stdin,1=>GrantSlot::Stdout,2=>GrantSlot::Stderr,3=>GrantSlot::Control,4=>GrantSlot::Cwd,5=>GrantSlot::Completion,6..=7=>return Err(Status::Unsupported),8=>GrantSlot::Frontend,_=>return Err(Status::Malformed)};
45        if r.u16()?!=0{return Err(Status::Malformed)} let owner_service=r.u64()?;let object=r.reference()?;if r.u64()?!=0{return Err(Status::Malformed)}r.finish()?;
46        let g=Self{kind,rights,slot,owner_service,object};g.validate()?;Ok(g)
47    }
48}
49/// Holds only validated canonical bytes. Deliberately has no Debug implementation.
50pub struct ChildContext { bytes:Vec<u8> }
51impl ChildContext {
52    pub fn new(operation:ObjectRefV1,argv:&[&str],cwd:&[u8],environment:&[(&str,&str)],grants:&[Grant])->Result<Self,Status> {
53        if operation.id==0 || operation.generation==0 || cwd.len()>4096 || cwd.contains(&0) || environment.len()>64 || grants.len()>CONTEXT_MAX_GRANTS {return Err(Status::Malformed)}
54        let mut argv_len=7usize;
55        for a in argv {if a.as_bytes().contains(&0){return Err(Status::Malformed)}argv_len=argv_len.checked_add(a.len()+1).ok_or(Status::Malformed)?;}
56        if argv_len>512{return Err(Status::QuotaExceeded)}
57        let mut env_len=0usize; let mut previous:Option<&str>=None;
58        for &(key,value) in environment {
59            if key.is_empty() || key.as_bytes().iter().any(|b|*b==0||*b==b'=') || value.as_bytes().contains(&0) || previous.is_some_and(|p|p>=key) || key.len()>u16::MAX as usize || value.len()>u16::MAX as usize {return Err(Status::Malformed)}
60            env_len=env_len.checked_add(4+key.len()+value.len()).ok_or(Status::Malformed)?; previous=Some(key);
61        }
62        if env_len>4096{return Err(Status::QuotaExceeded)}
63        let mut slots=0u16;for &g in grants {g.validate()?;let bit=1u16<<(g.slot as u16);if slots&bit!=0{return Err(Status::Malformed)}slots|=bit;}
64        if !cwd.is_empty() && slots&(1<<4)==0{return Err(Status::Denied)}
65        let len=CONTEXT_HEADER_LEN+argv_len+cwd.len()+env_len+grants.len()*GRANT_LEN;
66        if len>CONTEXT_MAX{return Err(Status::QuotaExceeded)}
67        let mut bytes=Vec::new();bytes.try_reserve_exact(len).map_err(|_|Status::QuotaExceeded)?;bytes.resize(len,0);
68        let mut w=Writer::new(&mut bytes);w.u32(1)?;w.u32(len as u32)?;w.u32(0)?;w.u32(argv_len as u32)?;w.u16(environment.len() as u16)?;w.u16(grants.len() as u16)?;w.u32(cwd.len() as u32)?;w.u32(env_len as u32)?;w.u32(0)?;w.reference(operation)?;
69        w.bytes(b"\0argv1\0")?;for a in argv {w.bytes(a.as_bytes())?;w.u8(0)?;}w.bytes(cwd)?;
70        for &(key,value) in environment {w.u16(key.len() as u16)?;w.u16(value.len() as u16)?;w.bytes(key.as_bytes())?;w.bytes(value.as_bytes())?;}
71        let mut offset=w.position();for &g in grants {g.encode(&mut bytes[offset..offset+GRANT_LEN])?;offset+=GRANT_LEN;}
72        Ok(Self{bytes})
73    }
74    pub fn from_bytes(bytes:Vec<u8>)->Result<Self,Status> {validate(&bytes)?;Ok(Self{bytes})}
75    pub fn as_bytes(&self)->&[u8]{&self.bytes}
76    pub fn operation(&self)->ObjectRefV1 {ObjectRefV1::from_bytes(&self.bytes[32..48]).expect("validated context")}
77    fn lengths(&self)->(usize,usize,usize,usize) {
78        let u32at=|n|u32::from_le_bytes(self.bytes[n..n+4].try_into().expect("validated context")) as usize;
79        (u32at(12),u32at(20),u32at(24),u16::from_le_bytes(self.bytes[18..20].try_into().expect("validated context")) as usize)
80    }
81    pub fn argv(&self)->impl Iterator<Item=&str> {let(a,_,_,_)=self.lengths();core::str::from_utf8(&self.bytes[55..48+a]).expect("validated argv").split_terminator('\0')}
82    pub fn cwd(&self)->&[u8]{let(a,c,_,_)=self.lengths();&self.bytes[48+a..48+a+c]}
83    pub fn environment(&self)->Environment<'_>{let(a,c,e,_)=self.lengths();Environment{bytes:&self.bytes[48+a+c..48+a+c+e]}}
84    pub fn grants(&self)->impl Iterator<Item=Grant>+'_ {let(a,c,e,_)=self.lengths();self.bytes[48+a+c+e..].chunks_exact(40).map(|raw|Grant::decode(raw).expect("validated grant"))}
85    pub fn cwd_grant(&self)->Option<ObjectRefV1>{self.grants().find(|g|g.slot==GrantSlot::Cwd).map(|g|g.object)}
86    /// The input-issued frontend this child was launched for, if any.
87    pub fn frontend_grant(&self)->Option<ObjectRefV1>{self.grants().find(|g|g.slot==GrantSlot::Frontend).map(|g|g.object)}
88    /// The kind-1 completion-record writer delegated into slot 5, if any.
89    pub fn completion_grant(&self)->Option<ObjectRefV1>{self.grants().find(|g|g.slot==GrantSlot::Completion&&g.kind==1).map(|g|g.object)}
90    pub fn read_own()->Result<Self,Status> {
91        let mut bytes=Vec::new();let mut offset=0usize;let mut total=None;
92        loop {
93            let mut body=[0u8;8];body[..4].copy_from_slice(&(offset as u32).to_le_bytes());body[4..].copy_from_slice(&512u32.to_le_bytes());
94            let mut reply=[0u8;544];let h=crate::session::kernel_call(KernelOp::ReadOwnContext,ObjectRefV1::ZERO,false,&body,&mut reply)?;
95            if h.status!=Status::Ok{return Err(h.status)}let mut r=Reader::new(&reply[HEADER_LEN..h.total_len as usize]);let n=r.u32()? as usize;let declared=r.u32()? as usize;
96            if declared<48 || declared>CONTEXT_MAX || n==0 || n>512 || offset+n>declared || total.is_some_and(|t|t!=declared){return Err(Status::Malformed)}
97            if total.is_none(){bytes.try_reserve_exact(declared).map_err(|_|Status::QuotaExceeded)?;total=Some(declared);}
98            bytes.extend_from_slice(r.take(n)?);r.finish()?;offset+=n;if offset==declared{break}
99        } Self::from_bytes(bytes)
100    }
101}
102pub struct Environment<'a>{bytes:&'a[u8]}
103impl<'a> Iterator for Environment<'a>{type Item=(&'a str,&'a str);fn next(&mut self)->Option<Self::Item>{if self.bytes.is_empty(){return None}let mut r=Reader::new(self.bytes);let k=r.u16().ok()? as usize;let v=r.u16().ok()? as usize;let key=core::str::from_utf8(r.take(k).ok()?).ok()?;let value=core::str::from_utf8(r.take(v).ok()?).ok()?;self.bytes=&self.bytes[4+k+v..];Some((key,value))}}
104fn validate(bytes:&[u8])->Result<(),Status>{
105    if bytes.len()<48 || bytes.len()>CONTEXT_MAX{return Err(Status::Malformed)}let mut r=Reader::new(bytes);if r.u32()?!=1{return Err(Status::UnsupportedVersion)}if r.u32()? as usize!=bytes.len()||r.u32()?!=0{return Err(Status::Malformed)}let a=r.u32()? as usize;let ec=r.u16()? as usize;let gc=r.u16()? as usize;let c=r.u32()? as usize;let e=r.u32()? as usize;if r.u32()?!=0{return Err(Status::Malformed)}let op=r.reference()?;
106    if op.id==0||op.generation==0||a>512||c>4096||e>4096||ec>64||gc>CONTEXT_MAX_GRANTS||48+a+c+e+gc*40!=bytes.len(){return Err(Status::Malformed)}
107    let argv=r.take(a)?;let tail=argv.strip_prefix(b"\0argv1\0").ok_or(Status::Malformed)?;if !tail.is_empty(){if tail.last()!=Some(&0){return Err(Status::Malformed)}for arg in tail[..tail.len()-1].split(|b|*b==0){core::str::from_utf8(arg).map_err(|_|Status::Malformed)?;}}
108    if r.take(c)?.contains(&0){return Err(Status::Malformed)}let env=r.take(e)?;let mut er=Reader::new(env);let mut previous:Option<&str>=None;
109    for _ in 0..ec {let k=er.u16()? as usize;let v=er.u16()? as usize;let key=core::str::from_utf8(er.take(k)?).map_err(|_|Status::Malformed)?;let value=core::str::from_utf8(er.take(v)?).map_err(|_|Status::Malformed)?;if key.is_empty()||key.as_bytes().iter().any(|b|*b==0||*b==b'=')||value.as_bytes().contains(&0)||previous.is_some_and(|p|p>=key){return Err(Status::Malformed)}previous=Some(key);}er.finish()?;
110    let mut slots=0u16;for _ in 0..gc{let g=Grant::decode(r.take(40)?)?;let bit=1u16<<(g.slot as u16);if slots&bit!=0{return Err(Status::Malformed)}slots|=bit;}
111    if c!=0&&slots&(1<<4)==0{return Err(Status::Denied)}r.finish()
112}

115mod tests {
116    use super::*;
117    const OPERATION:ObjectRefV1=ObjectRefV1{id:3,generation:1};
118    const FRONTEND:ObjectRefV1=ObjectRefV1{id:11,generation:2};
119    const ENDPOINT:ObjectRefV1=ObjectRefV1{id:12,generation:1};
120
121    #[test]
122    fn frontend_grant_round_trips_in_slot_8() {
123        let grant=Grant::frontend(FRONTEND).unwrap();
124        assert_eq!((grant.kind,grant.rights,grant.slot as u16,grant.owner_service),(6,0,8,api::syscall::service::INPUT as u64));
125        let grants=[Grant::endpoint(GrantSlot::Stdin,ENDPOINT,1).unwrap(),Grant::endpoint(GrantSlot::Stdout,ENDPOINT,2).unwrap(),
126            Grant::endpoint(GrantSlot::Stderr,ENDPOINT,2).unwrap(),Grant::endpoint(GrantSlot::Control,ENDPOINT,1).unwrap(),
127            Grant::cwd(ENDPOINT).unwrap(),grant];
128        let context=ChildContext::new(OPERATION,&["cellbox"],b"/",&[],&grants).unwrap();
129        assert_eq!(context.frontend_grant(),Some(FRONTEND));
130        let decoded=ChildContext::from_bytes(context.as_bytes().to_vec()).unwrap();
131        assert_eq!(decoded.grants().collect::<Vec<_>>(),grants);
132        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&[]).unwrap().frontend_grant(),None);
133    }

135    #[test]
136    fn frontend_kind_and_slot_are_exclusive() {
137        let mut forged=Grant::frontend(FRONTEND).unwrap();
138        forged.slot=GrantSlot::Cwd;
139        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&[forged]).err(),Some(Status::Malformed));
140        forged=Grant::frontend(FRONTEND).unwrap();forged.rights=1;
141        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&[forged]).err(),Some(Status::Malformed));
142        forged=Grant::frontend(FRONTEND).unwrap();forged.owner_service=api::syscall::service::VFS as u64;
143        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&[forged]).err(),Some(Status::Malformed));
144        assert_eq!(Grant::endpoint(GrantSlot::Frontend,ENDPOINT,1).err(),Some(Status::Malformed));
145        assert_eq!(Grant::frontend(ObjectRefV1::ZERO).err(),Some(Status::Malformed));
146        let twice=[Grant::frontend(FRONTEND).unwrap();2];
147        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&twice).err(),Some(Status::Malformed));
148        // Reserved slots 6..=7 stay Unsupported on decode; past 8 is Malformed; slot 5
149        // decodes but admits only a kind-1 writer, so kind 6 there is Malformed.
150        let mut raw=[0u8;40];raw[0..2].copy_from_slice(&6u16.to_le_bytes());raw[8..16].copy_from_slice(&(api::syscall::service::INPUT as u64).to_le_bytes());raw[16..32].copy_from_slice(&FRONTEND.to_bytes());
151        for (slot,status) in [(5u16,Status::Malformed),(6,Status::Unsupported),(7,Status::Unsupported),(9,Status::Malformed)] {
152            raw[4..6].copy_from_slice(&slot.to_le_bytes());assert_eq!(Grant::decode(&raw).err(),Some(status));
153        }
154        raw[4..6].copy_from_slice(&8u16.to_le_bytes());assert_eq!(Grant::decode(&raw),Ok(Grant::frontend(FRONTEND).unwrap()));
155    }

157    #[test]
158    fn completion_grant_is_a_write_only_endpoint_in_slot_5() {
159        let stdio=[Grant::endpoint(GrantSlot::Stdin,ENDPOINT,17).unwrap(),Grant::endpoint(GrantSlot::Stdout,ENDPOINT,30).unwrap(),
160            Grant::endpoint(GrantSlot::Stderr,ENDPOINT,30).unwrap(),Grant::endpoint(GrantSlot::Control,ENDPOINT,17).unwrap()];
161        for rights in [2u16,18,30] {
162            let completion=Grant::endpoint(GrantSlot::Completion,FRONTEND,rights).unwrap();
163            assert_eq!((completion.kind,completion.slot as u16,completion.owner_service),(1,5,0));
164            let mut grants=stdio.to_vec();grants.push(completion);
165            let context=ChildContext::new(OPERATION,&["ssh"],b"",&[],&grants).unwrap();
166            assert_eq!(context.as_bytes().len(),48+7+4+5*40);
167            let decoded=ChildContext::from_bytes(context.as_bytes().to_vec()).unwrap();
168            assert_eq!(decoded.grants().collect::<Vec<_>>(),grants);
169            assert_eq!(decoded.completion_grant(),Some(FRONTEND));
170        }
171        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&stdio).unwrap().completion_grant(),None);
172        // Read, bidirectional and out-of-mask rights are refused; so is a second writer.
173        for rights in [0u16,1,3,17,29,34] {
174            assert_eq!(Grant::endpoint(GrantSlot::Completion,FRONTEND,rights).err(),Some(Status::Malformed),"rights {rights}");
175        }
176        let one=Grant::endpoint(GrantSlot::Completion,FRONTEND,18).unwrap();
177        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&[one,one]).err(),Some(Status::Malformed));
178        // A forged non-kernel owner or a non-endpoint kind in slot 5 is refused on encode and decode.
179        let mut forged=one;forged.owner_service=api::syscall::service::VFS as u64;
180        assert_eq!(ChildContext::new(OPERATION,&[],b"",&[],&[forged]).err(),Some(Status::Malformed));
181        let mut raw=[0u8;40];raw[0..2].copy_from_slice(&1u16.to_le_bytes());raw[2..4].copy_from_slice(&1u16.to_le_bytes());raw[4..6].copy_from_slice(&5u16.to_le_bytes());raw[16..32].copy_from_slice(&FRONTEND.to_bytes());
182        assert_eq!(Grant::decode(&raw).err(),Some(Status::Malformed));
183        raw[2..4].copy_from_slice(&18u16.to_le_bytes());assert_eq!(Grant::decode(&raw),Ok(one));
184    }
