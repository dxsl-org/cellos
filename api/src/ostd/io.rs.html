14/// Opaque I/O error wrapping a [`ViError`] for [`embedded_io`] trait impls.
15///
16/// `ViError` lives in `libs/types`; implementing a foreign trait on a foreign type
17/// violates the orphan rule. This newtype lives in ostd and bridges the two.
18#[derive(Debug)]
19pub struct OstdError(pub ViError);
20
21impl core::fmt::Display for OstdError {
22    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
23        write!(f, "{:?}", self.0)
24    }
25}
26
27impl core::error::Error for OstdError {}

29impl embedded_io::Error for OstdError {
30    fn kind(&self) -> embedded_io::ErrorKind {
31        match self.0 {
32            ViError::NotFound => embedded_io::ErrorKind::NotFound,
33            ViError::PermissionDenied => embedded_io::ErrorKind::PermissionDenied,
34            ViError::OutOfMemory => embedded_io::ErrorKind::OutOfMemory,
35            ViError::WouldBlock => embedded_io::ErrorKind::Other,
36            _ => embedded_io::ErrorKind::Other,
37        }
38    }
39}

41struct FixedLogLine {
42    bytes: [u8; 192],
43    len: usize,
44}
45
46impl FixedLogLine {
47    const fn new() -> Self {
48        Self {
49            bytes: [0; 192],
50            len: 0,
51        }
52    }
53
54    fn as_str(&self) -> &str {
55        // Every appended fragment originated as valid UTF-8 and is copied whole.
56        core::str::from_utf8(&self.bytes[..self.len]).expect("fixed log line is UTF-8")
57    }
58}
59
60impl core::fmt::Write for FixedLogLine {
61    fn write_str(&mut self, value: &str) -> core::fmt::Result {
62        let end = self.len.checked_add(value.len()).ok_or(core::fmt::Error)?;
63        if end > self.bytes.len() {
64            return Err(core::fmt::Error);
65        }
66        self.bytes[self.len..end].copy_from_slice(value.as_bytes());
67        self.len = end;
68        Ok(())
69    }
70}

72/// Format into a bounded stack buffer and publish it through one log syscall.
73///
74/// Returns `false` without publishing if the formatted record exceeds 192 bytes.
75pub fn print_fmt(args: core::fmt::Arguments<'_>) -> bool {
76    use core::fmt::Write;
77
78    let mut line = FixedLogLine::new();
79    if line.write_fmt(args).is_err() {
80        return false;
81    }
82    print(line.as_str());
83    true
84}

97pub fn print_usize(n: usize) {
98    let mut buf = [0u8; 20];
99    let mut i = 0;
100    let mut num = n;
101
102    if num == 0 {
103        print("0");
104        return;
105    }
106
107    while num > 0 {
108        buf[i] = (num % 10) as u8 + b'0';
109        num /= 10;
110        i += 1;
111    }
112
113    // Reverse
114    let mut start = 0;
115    let mut end = i - 1;
116    while start < end {
117        buf.swap(start, end);
118        start += 1;
119        end -= 1;
120    }
121
122    if let Ok(s) = core::str::from_utf8(&buf[..i]) {
123        print(s);
124    }
125}

127/// Output grants captured once from this process's immutable invocation context.
128///
129/// Only a context-free (`Denied`) launch uses the legacy synchronous logger.
130/// A committed context must supply both stdout and stderr; errors never fall
131/// back to logging. This object retains no parsed context or service identity.
132pub struct ProcessOutput {
133    stdout: OutputStream,
134    stderr: OutputStream,
135}
136
137/// One process output target, deliberately without a public raw-handle accessor.
138pub struct OutputStream {
139    target: OutputTarget,
140}
141
142enum OutputTarget {
143    Legacy,
144    Context { operation: ObjectRefV1, endpoint: ObjectRefV1 },
145}
146
147impl ProcessOutput {
148    /// Read own context once and capture its independently routed output grants.
149    pub fn read_own() -> Result<Self, Status> {
150        Self::from_context_result(ChildContext::read_own())
151    }
152
153    fn from_context_result(context: Result<ChildContext, Status>) -> Result<Self, Status> {
154        let context = match context {
155            Ok(context) => context,
156            Err(Status::Denied) => return Ok(Self {
157                stdout: OutputStream { target: OutputTarget::Legacy },
158                stderr: OutputStream { target: OutputTarget::Legacy },
159            }),
160            Err(error) => return Err(error),
161        };
162        let mut stdout = None;
163        let mut stderr = None;
164        for grant in context.grants() {
165            if grant.kind != 1 {
166                continue;
167            }
168            match grant.slot {
169                GrantSlot::Stdout => stdout = Some(grant.object),
170                GrantSlot::Stderr => stderr = Some(grant.object),
171                _ => {},
172            }
173        }
174        let operation = context.operation();
175        Ok(Self {
176            stdout: OutputStream { target: OutputTarget::Context {
177                operation, endpoint: stdout.ok_or(Status::Denied)?,
178            } },
179            stderr: OutputStream { target: OutputTarget::Context {
180                operation, endpoint: stderr.ok_or(Status::Denied)?,
181            } },
182        })
183    }
184
185    pub fn stdout(&self) -> &OutputStream {
186        &self.stdout
187    }
188
189    pub fn stderr(&self) -> &OutputStream {
190        &self.stderr
191    }
192}

194impl OutputStream {
195    /// Write bytes using one finite, nonzero absolute monotonic deadline.
196    ///
197    /// Context output preserves arbitrary bytes and waits inside the kernel for
198    /// backpressure. An error can follow a committed prefix: terminate rather
199    /// than replaying the original slice. Legacy output accepts only UTF-8.
200    pub fn write_all(&self, bytes: &[u8], deadline_ns: u64) -> Result<(), Status> {
201        if deadline_ns == 0 || deadline_ns == u64::MAX {
202            return Err(Status::Malformed);
203        }
204        if bytes.is_empty() {
205            return Ok(());
206        }
207        match &self.target {
208            OutputTarget::Legacy => {
209                let text = core::str::from_utf8(bytes).map_err(|_| Status::UnsupportedInput)?;
210                let _ = syscall::sys_log(text);
211                Ok(())
212            },
213            OutputTarget::Context { operation, endpoint } => {
214                write_all_chunks(bytes, deadline_ns, |chunk, deadline| {
215                    crate::session::write_stream(*operation, *endpoint, chunk, deadline)
216                })
217            },
218        }
219    }
220}
221
222fn write_all_chunks(
223    mut bytes: &[u8],
224    deadline_ns: u64,
225    mut write: impl FnMut(&[u8], u64) -> Result<usize, Status>,
226) -> Result<(), Status> {
227    while !bytes.is_empty() {
228        let chunk = &bytes[..bytes.len().min(TRANSFER_MAX)];
229        let actual = write(chunk, deadline_ns)?;
230        if actual == 0 || actual > chunk.len() {
231            return Err(Status::Malformed);
232        }
233        bytes = &bytes[actual..];
234    }
235    Ok(())
236}

240impl Stdin {
241    /// Read one line from stdin, echoing characters as they arrive.
242    ///
243    /// When the input service is online (registered via `sys_lookup_service`),
244    /// reads via the input-service IPC path so that the caller participates in
245    /// the focus model. Callers must have called `ostd::input::request_focus()`
246    /// at least once before this returns meaningful data; without focus the input
247    /// service will not dispatch key events to this cell.
248    ///
249    /// Falls back to the kernel `sys_read(fd=0)` path when the input service is
250    /// not yet registered (early boot, or input cell not present).
251    pub fn read_line(&self, buf: &mut String) -> ViResult<usize> {
252        if sys_lookup_service(service::INPUT).is_some() {
253            // Input-service path: receive InputEvent frames dispatched to this cell.
254            // Uses a per-iteration timeout only to detect input-service restarts —
255            // we re-request focus only when the TID changes, NOT on every timeout.
256            // Re-requesting focus on every timeout created a ~10ms dead zone where
257            // shell was in sys_send(SetFocus) and sys_try_send drops from dispatcher
258            // would cause burst keystrokes to be silently lost (G18 fix regression).
259            const TIMEOUT_TICKS: u64 = 20; // ~200ms between TID-change checks
260            let mut bytes_read: usize = 0;
261            let mut focused_on_tid: usize = 0; // 0 = no focus yet
262                                               // Re-check the input service TID on each iteration — it may have
263                                               // restarted. If it disappears entirely, fall through to sys_read.
264            while let Some(input_tid) = sys_lookup_service(service::INPUT) {
265                // Request focus only on startup or when input service restarted (TID changed).
266                if focused_on_tid != input_tid {
267                    if !crate::input::request_focus() {
268                        // Input service up but SetFocus failed — service may be
269                        // initialising.  Try again next iteration.
270                        continue;
271                    }
272                    focused_on_tid = input_tid;
273                }
274
275                let mut frame = [0u8; IPC_BUF_SIZE];
276                match sys_recv_timeout(0, &mut frame, TIMEOUT_TICKS) {
277                    SyscallResult::Ok(0) => {
278                        // Timeout — loop back to check if TID changed (restart detection).
279                        // Do NOT reset focused_on_tid here — that would trigger a SetFocus
280                        // sys_send dead zone on every 200ms timeout, causing dropped chars.
281                    }
282                    SyscallResult::Ok(sender) if sender == input_tid => {
283                        if frame[0] != INPUT_EVENT_OPCODE {
284                            continue;
285                        }
286                        let Some(ev) = decode_event(&frame[1..]) else {
287                            continue;
288                        };
289                        if let InputEvent::Key(k) = ev {
290                            if k.state != KeyState::Pressed {
291                                continue;
292                            }
293                            match k.keysym {
294                                KeySym::Return => {
295                                    print("\n");
296                                    buf.push('\n');
297                                    return Ok(bytes_read + 1);
298                                }
299                                KeySym::Backspace if !buf.is_empty() => {
300                                    print("\x08 \x08");
301                                    buf.pop();
302                                    bytes_read = bytes_read.saturating_sub(1);
303                                }
304                                KeySym::Printable if k.character > 0 => {
305                                    if let Some(ch) = char::from_u32(k.character) {
306                                        let mut tmp = [0u8; 4];
307                                        let s = ch.encode_utf8(&mut tmp);
308                                        print(s);
309                                        buf.push(ch);
310                                        bytes_read += 1;
311                                    }
312                                }
313                                _ => {}
314                            }
315                        }
316                    }
317                    SyscallResult::Ok(_other) => {
318                        // Unexpected sender — discard and keep waiting.
319                    }
320                    _ => return Err(ViError::IO),
321                }
322            }
323        }
324        {
325            // Fallback: input service not online; use direct kernel UART buffer.
326            let mut bytes_read: usize = 0;
327            loop {
328                let mut c = [0u8; 1];
329                if let Ok(n) = syscall::sys_read(0, &mut c) {
330                    if n > 0 {
331                        let ch = c[0] as char;
332                        if ch == '\r' || ch == '\n' {
333                            print("\n");
334                            buf.push('\n');
335                            return Ok(bytes_read + 1);
336                        }
337                        if c[0] == 8 || c[0] == 127 {
338                            if !buf.is_empty() {
339                                print("\x08 \x08");
340                                buf.pop();
341                                bytes_read = bytes_read.saturating_sub(1);
342                            }
343                            continue;
344                        }
345                        let mut tmp = [0u8; 4];
346                        let s = ch.encode_utf8(&mut tmp);
347                        print(s);
348                        buf.push(ch);
349                        bytes_read += 1;
350                    }
351                } else {
352                    return Err(ViError::IO);
353                }
354            }
355        }
356    }
357}

375/// Handle to the standard output stream.
376pub struct Stdout;
377
378pub fn stdout() -> Stdout {
379    Stdout
380}
381
382impl embedded_io::ErrorType for Stdout {
383    type Error = OstdError;
384}
385
386impl embedded_io::Write for Stdout {
387    fn write(&mut self, buf: &[u8]) -> Result<usize, OstdError> {
388        match core::str::from_utf8(buf) {
389            Ok(s) => {
390                syscall::sys_log(s); // always succeeds; return value is ignored
391                Ok(buf.len())
392            }
393            Err(_) => Err(OstdError(ViError::InvalidInput)),
394        }
395    }
396
397    fn flush(&mut self) -> Result<(), OstdError> {
398        Ok(()) // sys_log is synchronous; no buffering to flush
399    }

407    const OPERATION: ObjectRefV1 = ObjectRefV1 { id: 1, generation: 7 };
408    const STDOUT: ObjectRefV1 = ObjectRefV1 { id: 2, generation: 8 };
409    const STDERR: ObjectRefV1 = ObjectRefV1 { id: 3, generation: 9 };
410
411    fn context(stdout: bool, stderr: bool) -> ChildContext {
412        let mut grants = alloc::vec::Vec::new();
413        if stdout {
414            grants.push(Grant::endpoint(GrantSlot::Stdout, STDOUT, 2).unwrap());
415        }
416        if stderr {
417            grants.push(Grant::endpoint(GrantSlot::Stderr, STDERR, 2).unwrap());
418        }
419        ChildContext::new(OPERATION, &["echo"], b"", &[], &grants).unwrap()
420    }

422    #[test]
423    fn only_denied_context_read_selects_legacy_output() {
424        let output = ProcessOutput::from_context_result(Err(Status::Denied)).unwrap();
425        assert!(matches!(&output.stdout().target, OutputTarget::Legacy));
426        assert!(matches!(&output.stderr().target, OutputTarget::Legacy));
427        for error in [Status::Malformed, Status::TimedOut, Status::Cancelled,
428            Status::Revoked, Status::Stale, Status::Io, Status::Indeterminate]
429        {
430            assert_eq!(ProcessOutput::from_context_result(Err(error)).err(), Some(error));
431        }
432    }

434    #[test]
435    fn committed_context_requires_both_output_grants() {
436        for (stdout, stderr) in [(false, false), (true, false), (false, true)] {
437            assert_eq!(
438                ProcessOutput::from_context_result(Ok(context(stdout, stderr))).err(),
439                Some(Status::Denied),
440            );
441        }
442        let output = ProcessOutput::from_context_result(Ok(context(true, true))).unwrap();
443        for (stream, expected) in [(output.stdout(), STDOUT), (output.stderr(), STDERR)] {
444            match &stream.target {
445                OutputTarget::Context { operation, endpoint } => {
446                    assert_eq!(*operation, OPERATION);
447                    assert_eq!(*endpoint, expected);
448                },
449                OutputTarget::Legacy => panic!("committed context selected logger"),
450            }
451        }
452    }

454    #[test]

456        let output = ProcessOutput::from_context_result(Ok(context(true, true))).unwrap();
457        let legacy = ProcessOutput::from_context_result(Err(Status::Denied)).unwrap();
458        for stream in [output.stdout(), output.stderr(), legacy.stdout(), legacy.stderr()] {
459            assert_eq!(stream.write_all(b"", 1), Ok(()));

462                assert_eq!(stream.write_all(b"not published", deadline), Err(Status::Malformed));
463            }

474    #[test]
475    fn partial_writes_preserve_bytes_and_one_absolute_deadline() {
476        let bytes: alloc::vec::Vec<u8> = (0..1300).map(|i| (i % 256) as u8).collect();
477        let mut received = alloc::vec::Vec::new();
478        let mut calls = 0;
479        assert_eq!(write_all_chunks(&bytes, 12345, |chunk, deadline| {
480            calls += 1;
481            assert_eq!(deadline, 12345);
482            assert!(!chunk.is_empty() && chunk.len() <= TRANSFER_MAX);
483            let actual = chunk.len().min(73);
484            received.extend_from_slice(&chunk[..actual]);
485            Ok(actual)
486        }), Ok(()));
487        assert!(calls > 3);
488        assert_eq!(received, bytes);
489    }
