1//! DMA buffer helpers for Tier-1 Driver Cells.
2//!
3//! Driver Cells allocate physically contiguous memory for DMA rings via `sys_grant_alloc`
4//! (which uses `allocate_contiguous` internally), then authorise the IOMMU via
5//! `sys_grant_dma`. This module wraps both into a safe interface.
6//!
7//! # Usage
8//! ```ignore
9//! let buf = DmaBuf::alloc(4).expect("OOM");      // 4 pages (16 KiB)
10//! buf.authorize(0x0300).expect("IOMMU deny");     // BDF 0:3:0
11//! let phys = buf.phys();                         // program into NVMe/e1000 regs
12//! ```
13
14use crate::syscall::{
15    sys_grant_alloc, sys_grant_cache_sync_begin, sys_grant_cache_sync_complete, sys_grant_dma,
16    sys_grant_free, SyscallError,
17};
18
19/// A physically-contiguous, page-aligned DMA buffer backed by a Grant region.
20///
21/// In SAS the virtual address equals the physical address (identity mapping),
22/// so [`phys`](DmaBuf::phys) and [`virt`](DmaBuf::virt) return the same value.
23///
24/// Drop does NOT free the buffer — call [`DmaBuf::free`] explicitly.
25/// Driver Cells typically hold DMA buffers for their entire lifetime.
26pub struct DmaBuf {
27    grant_id: usize, // physical base == virtual base in SAS
28    n_pages: usize,
29}
30
31impl DmaBuf {
32    /// Allocate `n_pages` contiguous 4-KiB pages suitable for DMA.
33    pub fn alloc(n_pages: usize) -> Option<Self> {
34        let grant_id = sys_grant_alloc(n_pages * 4096)?;
35        Some(Self { grant_id, n_pages })
36    }
37
38    /// Physical base address — program this into DMA descriptor registers.
39    #[inline]
40    pub fn phys(&self) -> usize {
41        self.grant_id
42    }
43
44    /// Virtual address — identical to `phys()` in SAS.
45    #[inline]
46    pub fn virt(&self) -> *mut u8 {
47        self.grant_id as *mut u8
48    }
49
50    /// Total size in bytes.
51    #[inline]
52    pub fn size(&self) -> usize {
53        self.n_pages * 4096
54    }
55
56    /// Authorise DMA for the PCIe device at `bdf` covering this buffer.
57    ///
58    /// Must be called before issuing any DMA operation.  The IOMMU maps
59    /// `[phys, phys + size)` for this BDF in the calling cell's domain.
60    /// Returns `Ok(iova)` on success (`iova == phys` in SAS).
61    pub fn authorize(&self, bdf: u32) -> Result<u64, SyscallError> {
62        sys_grant_dma(bdf, self.grant_id as u64, self.size())
63    }
64
65    /// Begin a cache-coherent device operation over a bounded subrange.
66    pub fn begin_cache_sync(&self, offset: usize, len: usize) -> Option<usize> {
67        sys_grant_cache_sync_begin(self.grant_id, offset, len)
68    }
69
70    /// Complete a previously started cache synchronization operation.
71    pub fn complete_cache_sync(&self, token: usize) -> bool {
72        sys_grant_cache_sync_complete(token)
73    }
74
75    /// Release the Grant region and return frames to the kernel.
76    pub fn free(self) {
77        sys_grant_free(self.grant_id);
78    }
79}
