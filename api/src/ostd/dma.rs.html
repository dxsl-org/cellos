1//! DMA buffer helpers for Tier-1 Driver Cells.
2//!
3//! Driver Cells allocate physically contiguous memory for DMA rings via `sys_grant_alloc`
4//! (which uses `allocate_contiguous` internally), then authorise the IOMMU via
5//! `sys_grant_dma`. This module wraps both into a safe interface.
6//!
7//! # Usage
8//! ```ignore
9//! let buf = DmaBuf::alloc(4).expect("OOM");       // 4 pages (16 KiB)
10//! let iova = buf.authorize(0x0300).expect("IOMMU deny");
11//! // Program `iova` into device registers/descriptors; use `virt()` for CPU access.
12//! # let _ = iova;
13
14use crate::syscall::{
15    sys_export_dma_range, sys_grant_alloc, sys_grant_cache_sync_begin,
16    sys_grant_cache_sync_complete, sys_grant_dma, sys_grant_free, DmaExportError, SyscallError,
17};
18
19/// A physically-contiguous, page-aligned DMA buffer backed by a Grant region.
20///
21/// In SAS the CPU virtual and physical addresses are currently identical.
22/// Device-visible addresses are established by [`DmaBuf::authorize`] and may
23/// differ from both; callers must program the returned IOVA into hardware.
24///
25/// Drop does NOT free the buffer — call [`DmaBuf::free`] explicitly.
26/// Driver Cells typically hold DMA buffers for their entire lifetime.
27pub struct DmaBuf {
28    grant_id: usize, // physical base == virtual base in SAS
29    n_pages: usize,
30}
31
32impl DmaBuf {
33    /// Allocate `n_pages` contiguous 4-KiB pages suitable for DMA. The pages
34    /// arrive zero-filled: the kernel zeroes every new grant before mapping it.
35    pub fn alloc(n_pages: usize) -> Option<Self> {
36        let grant_id = sys_grant_alloc(n_pages * 4096)?;
37        Some(Self { grant_id, n_pages })
38    }
39
40    /// CPU physical base address; do not substitute this for an authorized IOVA.
41    #[inline]
42    pub fn phys(&self) -> usize {
43        self.grant_id
44    }
45
46    /// Virtual address — identical to `phys()` in SAS.
47    #[inline]
48    pub fn virt(&self) -> *mut u8 {
49        self.grant_id as *mut u8
50    }
51
52    /// Total size in bytes.
53    #[inline]
54    pub fn size(&self) -> usize {
55        self.n_pages * 4096
56    }
57
58    /// Authorise DMA for the PCIe device at `bdf` covering this buffer.
59    ///
60    /// Must be called before issuing any DMA operation.  The IOMMU maps
61    /// `[phys, phys + size)` for this BDF in the calling cell's domain.
62    /// Returns `Ok(iova)` on success (`iova == phys` in SAS).
63    pub fn authorize(&self, bdf: u32) -> Result<u64, SyscallError> {
64        sys_grant_dma(bdf, self.grant_id as u64, self.size())
65    }
66
67    /// Begin a cache-coherent device operation over a bounded subrange.
68    pub fn begin_cache_sync(&self, offset: usize, len: usize) -> Option<usize> {
69        sys_grant_cache_sync_begin(self.grant_id, offset, len)
70    }
71
72    /// Complete a previously started cache synchronization operation.
73    pub fn complete_cache_sync(&self, token: usize) -> bool {
74        sys_grant_cache_sync_complete(token)
75    }
76
77    /// Export `[offset, offset + len)` of this buffer to the live provider of
78    /// `holder_service` (`ExportDmaRange = 437`, ADR-0033). The holder maps it
79    /// with [`map_export`] and its device reaches it through a kernel-chosen
80    /// IOVA window; `device_write` false makes the window device-read-only.
81    /// The buffer cannot be freed until the export is revoked ([`revoke_export`]).
82    pub fn export(
83        &self,
84        offset: usize,
85        len: usize,
86        holder_service: u16,
87        device_write: bool,
88    ) -> Result<u64, DmaExportError> {
89        if offset.checked_add(len).is_none_or(|end| end > self.size()) {
90            return Err(DmaExportError::InvalidInput);
91        }
92        let flags = if device_write {
93            abi::dma_export::EXPORT_DEVICE_WRITE
94        } else {
95            0
96        };
97        sys_export_dma_range(self.grant_id + offset, len, holder_service, flags)
98    }
99
100    /// Copy `src` into the buffer at `offset`. Returns `false` (and copies
101    /// nothing) when the span does not fit.
102    pub fn write_bytes(&self, offset: usize, src: &[u8]) -> bool {
103        if offset
104            .checked_add(src.len())
105            .is_none_or(|end| end > self.size())
106        {
107            return false;
108        }
109        for (i, byte) in src.iter().enumerate() {
110            // SAFETY: the span lies inside this live, caller-owned Grant, which
111            // is mapped USER RW at `grant_id` for the buffer's whole size.
112            // Volatile: a device may read it through an authorized window.
113            unsafe { self.virt().add(offset + i).write_volatile(*byte) };
114        }
115        true
116    }
117
118    /// Copy `dst.len()` bytes at `offset` out of the buffer. Returns `false`
119    /// (and copies nothing) when the span does not fit.
120    pub fn read_bytes(&self, offset: usize, dst: &mut [u8]) -> bool {
121        if offset
122            .checked_add(dst.len())
123            .is_none_or(|end| end > self.size())
124        {
125            return false;
126        }
127        for (i, byte) in dst.iter_mut().enumerate() {
128            // SAFETY: as in `write_bytes`; volatile because a device may have
129            // written the memory behind the compiler's back.
130            *byte = unsafe { self.virt().add(offset + i).read_volatile() };
131        }
132        true
133    }
134
135    /// Release the Grant region and return frames to the kernel.
136    pub fn free(self) {
137        sys_grant_free(self.grant_id);
138    }
139}
