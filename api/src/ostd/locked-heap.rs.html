1//! Preemption-safe lock around the cell's free-list heap.
2//!
3//! `task::spawn` gives a cell several preemptible tasks that share one heap, so
4//! two of them can be inside `allocate_first_fit`/`deallocate` at once. The free
5//! list is a chain of in-place hole headers; an unsynchronised interleaving links
6//! one hole into two lists and hands the same bytes to two owners. Every heap
7//! operation therefore runs under the lock below.
8//!
9//! The lock is a test-and-set flag whose waiters give their hart away. A plain
10//! spinlock deadlocks a single-hart cell: the holder can be preempted inside the
11//! critical section and the spinner would burn every slice the holder needs to
12//! release it. A waiter spins briefly (the holder may be running on another hart
13//! and leave within a few hundred cycles), then calls `relax` — the yield syscall
14//! in a cell — so the scheduler runs the holder. Uncontended, the cost is one
15//! atomic swap and one release store per allocation.
16
17use core::alloc::Layout;
18use core::cell::UnsafeCell;
19use core::ptr::{null_mut, NonNull};
20use core::sync::atomic::{AtomicBool, Ordering};
21use linked_list_allocator::Heap;
22
23/// Spin iterations before a waiter yields: long enough to ride out a holder on
24/// another hart, short enough that a preempted holder costs little.
25const SPINS_BEFORE_RELAX: u32 = 64;
26
27/// A free-list heap whose every access is serialised by a yielding lock.
28///
29/// Holds no pointers in its initial state (`Heap::empty()` is all zero), so a
30/// static of this type is zero-initialised `.bss`.
31pub(crate) struct CellHeap {
32    locked: AtomicBool,
33    heap: UnsafeCell<Heap>,
34}
35
36// SAFETY: `heap` is only reached through `lock`, which grants one holder at a
37// time; the Acquire/Release pair on `locked` orders the holders' accesses.
38unsafe impl Sync for CellHeap {}
39// SAFETY: the raw pointers inside `Heap` name the arena the heap owns, not
40// thread-local data, so the heap may be handed to another thread.
41unsafe impl Send for CellHeap {}
42
43/// Exclusive access to the heap; releases the lock when dropped.
44struct HeapGuard<'a> {
45    owner: &'a CellHeap,
46}
47
48impl HeapGuard<'_> {
49    fn heap(&mut self) -> &mut Heap {
50        // SAFETY: a live guard is the sole holder of `owner.locked`, so no other
51        // reference to the heap exists until it drops.
52        unsafe { &mut *self.owner.heap.get() }
53    }
54}
55
56impl Drop for HeapGuard<'_> {
57    fn drop(&mut self) {
58        self.owner.locked.store(false, Ordering::Release);
59    }
60}
61
62impl CellHeap {
63    pub(crate) const fn empty() -> Self {
64        Self {
65            locked: AtomicBool::new(false),
66            heap: UnsafeCell::new(Heap::empty()),
67        }
68    }
69
70    /// Acquire the lock. `relax` runs whenever the holder outlasts a short spin;
71    /// it must let the holder make progress (yield the hart) and must not
72    /// allocate.
73    fn lock(&self, relax: fn()) -> HeapGuard<'_> {
74        loop {
75            if !self.locked.swap(true, Ordering::Acquire) {
76                return HeapGuard { owner: self };
77            }
78            let mut spins = 0;
79            while self.locked.load(Ordering::Relaxed) {
80                if spins < SPINS_BEFORE_RELAX {
81                    spins += 1;
82                    core::hint::spin_loop();
83                } else {
84                    relax();
85                }
86            }
87        }
88    }
89
90    /// Hand the heap the arena `[ptr, ptr + len)`.
91    ///
92    /// # Safety
93    /// The arena must stay mapped and owned solely by this heap for the rest of
94    /// the program, and no block from an earlier arena may still be live.
95    pub(crate) unsafe fn init(&self, ptr: *mut u8, len: usize, relax: fn()) {
96        self.lock(relax).heap().init(ptr, len);
97    }
98
99    /// Allocate `layout`, first giving the heap `default_arena()` if nothing has
100    /// initialised it yet. Null on exhaustion.
101    ///
102    /// # Safety
103    /// The arena `default_arena` returns must satisfy [`Self::init`]'s contract.
104    pub(crate) unsafe fn alloc(
105        &self,
106        layout: Layout,
107        relax: fn(),
108        default_arena: impl FnOnce() -> (*mut u8, usize),
109    ) -> *mut u8 {
110        let mut guard = self.lock(relax);
111        let heap = guard.heap();
112        // Under the lock, so exactly one task performs the lazy init.
113        if heap.size() == 0 {
114            let (ptr, len) = default_arena();
115            heap.init(ptr, len);
116        }
117        heap.allocate_first_fit(layout)
118            .map_or(null_mut(), |p| p.as_ptr())
119    }
120
121    /// Return a block to the heap; null is ignored.
122    ///
123    /// # Safety
124    /// A non-null `ptr` must come from [`Self::alloc`] on this heap with the same
125    /// `layout` and must not be freed twice.
126    pub(crate) unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout, relax: fn()) {
127        if let Some(p) = NonNull::new(ptr) {
128            self.lock(relax).heap().deallocate(p, layout);
129        }
130    }
131
132    /// Bytes currently handed out (test observation of leaks and double frees).
133    #[cfg(test)]
134    fn used(&self, relax: fn()) -> usize {
135        self.lock(relax).heap().used()
136    }
137}
