150/// Convenience entry-point macro for App SDK cells.
151///
152/// Wraps a handler function in a `main()` body that creates an [`app::AppContext`]
153/// and calls `run()`. Eliminates boilerplate for cells that only need the typed
154/// event loop.
155///
156/// # Usage
157/// ```no_run
158/// use ostd::app::{AppContext, AppEvent};
159///
160/// fn my_handler(_ctx: &mut AppContext, ev: AppEvent) {
161///     if let AppEvent::RawMessage { .. } = ev { /* ... */ }
162/// }
163///
164/// ostd::run_app!(my_handler);
165/// ```
166#[macro_export]
167macro_rules! run_app {
168    ($handler:expr) => {
169        #[no_mangle]
170        pub fn main() {
171            // run_with_lifecycle (not run) so AppEvent::Init fires once before the
172            // first sys_recv — Driver Cells (virtio-net/gpu, nvme, e1000) do all
173            // device probing + service registration in their Init arm. Matches the
174            // app_entry! contract and the documented run_app! behavior.
175            $crate::app::AppContext::new().run_with_lifecycle($handler);
176        }
177    };
178}
