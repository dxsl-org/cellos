2//! Non-parking sends: [`post`] for one-way messages, [`Incoming::reply`] for
3//! answering a request a service received (Phase 11 CP1 §3).
4//!
5//! A blocking `Send` parks the sender until the target receives, so a client
6//! that never receives can stall a service. Nothing here ever parks.
7
8use super::IpcOpId;
9use crate::syscall::{sys_post, SyscallResult};
10
11/// Why [`post`] or [`Incoming::reply`] did not deliver.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub enum PostError {
14    /// The target is gone, or its bounded mailbox is full.
15    Undeliverable,
16    /// The request's exact operation no longer accepts a reply: the caller
17    /// cancelled it, it expired, its caller exited, or it was already answered.
18    ReplyRefused,
19}
20
21/// Queue one message in `target`'s bounded mailbox without parking.
22///
23/// The target's next receive admitting this sender takes it, so the target need
24/// not be receiving now. Uses `SendGather` (202); a Cell with an explicit
25/// syscall allowlist must list it.
26///
27/// # Errors
28/// [`PostError::Undeliverable`] when the target is gone or its mailbox is full.
29/// The message is then dropped; nothing is retried.
30pub fn post(target: usize, msg: &[u8]) -> Result<(), PostError> {
31    match sys_post(target, msg) {
32        SyscallResult::Ok(0) => Ok(()),
33        _ => Err(PostError::Undeliverable),
34    }
35}
36
37/// Send one message and park until `target` takes it.
38///
39/// The one sanctioned blocking send, for producers that need backpressure
40/// from a receiver that polls its mailbox and never answers: input-source
41/// registration and HID events. Every other message uses [`post`] or a reply.
42///
43/// # Errors
44/// [`PostError::Undeliverable`] when the target is gone or refuses senders.
45pub fn post_blocking(target: usize, msg: &[u8]) -> Result<(), PostError> {
46    match crate::syscall::sys_send(target, msg) {
47        SyscallResult::Ok(_) => Ok(()),
48        SyscallResult::Err(_) => Err(PostError::Undeliverable),
49    }
50}

52/// A request a service is answering: its sender and, when it arrived as an
53/// exact operation (`IpcSubmit`), that operation.
54#[derive(Debug, Clone, Copy, PartialEq, Eq)]
55pub struct Incoming {
56    sender: usize,
57    op: Option<IpcOpId>,
58}
59
60impl Incoming {
61    /// Capture the request this task received last, from `sender`.
62    ///
63    /// Call it before any other receive: the kernel records only the latest
64    /// request, so a later receive would replace the operation this captures.
65    pub fn current(sender: usize) -> Self {
66        Self {
67            sender,
68            op: super::current(),
69        }
70    }
71
72    /// The task that sent this request.
73    pub const fn sender(&self) -> usize {
74        self.sender
75    }
76
77    /// Answer the request without parking.
78    ///
79    /// An exact operation is answered with `IpcReply`, which completes it
80    /// whether or not the caller is receiving. A request that arrived as a raw
81    /// `Send` has no operation, so its reply is [`post`]ed to the sender's
82    /// mailbox for the sender's next receive.
83    ///
84    /// # Errors
85    /// [`PostError::ReplyRefused`] when the operation no longer accepts a reply,
86    /// and [`PostError::Undeliverable`] when a posted reply cannot be queued.
87    pub fn reply(&self, reply: &[u8]) -> Result<(), PostError> {
88        match self.op {
89            Some(op) => super::reply(op, reply).map_err(|_| PostError::ReplyRefused),
90            None => post(self.sender, reply),
91        }
92    }
93}
94
95/// Answer `sender`'s request from a reply wrapper, without parking when the
96/// request is the one this task received last.
97///
98/// Every receive replaces the kernel's record of the current request (VFS
99/// keeps its outer request across masked receives), so a wrapper cannot trust
100/// a token it did not capture. Without a current operation the reply is
101/// [`post`]ed. With one, it goes through the kernel's reply bridge, which
102/// completes the operation only when `sender` is the caller it belongs to; for
103/// any other sender the bridge falls back to a raw send, which can park. A
104/// handler that receives again before replying should capture [`Incoming`].
105///
106/// # Errors
107/// [`PostError::Undeliverable`] when the bridge refuses the reply or a posted
108/// reply cannot be queued.
109pub fn reply_to(sender: usize, reply: &[u8]) -> Result<(), PostError> {
110    if super::current().is_none() {
111        return post(sender, reply);
112    }
113    match crate::syscall::sys_send(sender, reply) {
114        SyscallResult::Ok(_) => Ok(()),
115        SyscallResult::Err(_) => Err(PostError::Undeliverable),
116    }
117}
