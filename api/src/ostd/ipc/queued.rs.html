1//! Queued service admission with a bounded, sender-masked reply wait.
2
3use super::IpcError;
4use crate::syscall::{sys_recv_timeout, sys_send, SyscallResult};
5use serde::{Deserialize, Serialize};
6
7/// Queue one request with blocking `Send`, then bound only the reply wait.
8///
9/// This transport is for permanent services that may temporarily receive from a
10/// nested dependency and therefore cannot admit `TrySend`. A gone target fails
11/// the send; a service that accepts but does not reply before `timeout_ticks`
12/// returns [`IpcError::Recv`]. After that receive error, callers must poison the
13/// service generation because an uncorrelated late reply can still arrive.
14pub fn service_call_queued_bounded<'r, Req: Serialize>(
15    service_tid: usize,
16    req: &Req,
17    send_buf: &mut [u8],
18    recv_buf: &'r mut [u8],
19    timeout_ticks: u64,
20) -> Result<&'r [u8], IpcError> {
21    let encoded = api::ipc::encode(req, send_buf).map_err(|_| IpcError::Encode)?;
22    if let SyscallResult::Err(_) = sys_send(service_tid, encoded) {
23        return Err(IpcError::Send);
24    }
25    match sys_recv_timeout(service_tid, recv_buf, timeout_ticks) {
26        SyscallResult::Ok(0) => Err(IpcError::Recv),
27        SyscallResult::Ok(sender) if sender == service_tid => Ok(recv_buf),
28        SyscallResult::Ok(_) => Err(IpcError::WrongSender),
29        SyscallResult::Err(_) => Err(IpcError::Recv),
30    }
31}
32
33/// Decode the reply from [`service_call_queued_bounded`] as `Resp`.
34///
35/// `Resp` may borrow `recv_buf`; consume it before reusing that buffer. Errors
36/// preserve encode, send, receive-timeout, sender, and decode distinctions.
37pub fn service_call_typed_queued_bounded<'r, Req, Resp>(
38    service_tid: usize,
39    req: &Req,
40    send_buf: &mut [u8],
41    recv_buf: &'r mut [u8],
42    timeout_ticks: u64,
43) -> Result<Resp, IpcError>
44where
45    Req: Serialize,
46    Resp: Deserialize<'r>,
47{
48    let raw = service_call_queued_bounded(service_tid, req, send_buf, recv_buf, timeout_ticks)?;
49    api::ipc::decode::<Resp>(raw).map_err(|_| IpcError::Decode)
50}
