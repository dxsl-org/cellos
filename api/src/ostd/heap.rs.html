3//! Cell heap allocator — a real freeing allocator (linked-list free list).
4//!
5//! Replaces the former bump allocator whose `dealloc` was a no-op: that leaked
6//! EVERY allocation, so any long-running cell (shell, services) inexorably
7//! exhausted its 4 MiB arena and then store-faulted on the first failed
8//! allocation — a guaranteed slow death, the opposite of "never-die". This wraps
9//! `linked_list_allocator` (the same crate the kernel heap uses) so transient
10//! `String`/`Vec` allocations are actually reclaimed and a cell can run forever.
11//!
12//! ## Why `static mut Heap` and not `LockedHeap`
13//! Every ViCell cell is single-task / single-hart, so allocator entries never
14//! overlap and no lock is needed. Crucially, `LockedHeap` embeds a spinlock whose
15//! atomic write-back FAULTS in a cell: the const-initialised allocator static lands
16//! in the ELF's RELRO (read-only) segment, and an `amoor.w` to a read-only page
17//! traps (scause=0xf). A zero-sized `CellAllocator` plus mutable state in `.bss`
18//! (`static mut`) sidesteps that entirely — there is no writable allocator static
19//! to be placed read-only.
20
21use core::alloc::{GlobalAlloc, Layout};
22use core::ptr::{addr_of_mut, null_mut, NonNull};
23use linked_list_allocator::Heap;
24
25/// Default per-cell heap size: 512 KiB gives ample headroom for normal cells
26/// while keeping footprint minimal. Cells needing larger heaps (e.g. VFS, compositor)
27/// can initialize their own static arena via `declare_custom_heap!()`.
28const HEAP_SIZE: usize = 512 * 1024;
29/// Backing arena for the cell heap. 16-byte aligned so the allocator can satisfy
30/// the largest natural alignment without wasting the first bytes.
31#[repr(align(16))]
32#[allow(dead_code)] // reason: backing bytes are accessed via addr_of_mut!(HEAP_MEM) raw-pointer cast, not field syntax
33struct HeapArena([u8; HEAP_SIZE]);
34static mut HEAP_MEM: HeapArena = HeapArena([0; HEAP_SIZE]);
35
36/// Free-list allocator state. `static mut` (writable `.bss`), serialized solely by
37/// the single-task execution model — see the module note on why no lock is used.
38static mut HEAP: Heap = Heap::empty();
39static mut HEAP_INIT: bool = false;
40
41/// Initialize the cell heap with a custom static arena before any allocation occurs.
42/// If called before the first allocation, the default `HEAP_MEM` is never touched.
43pub unsafe fn init_custom_heap_raw(ptr: *mut u8, len: usize) {
44    let heap = &mut *addr_of_mut!(HEAP);
45    heap.init(ptr, len);
46    HEAP_INIT = true;
47}
48
49#[macro_export]
50macro_rules! declare_custom_heap {
51    ($bytes:expr) => {
52        #[repr(align(16))]
53        struct __CustomHeapArena([u8; $bytes]);
54        static mut __CUSTOM_HEAP_ARENA: __CustomHeapArena = __CustomHeapArena([0; $bytes]);
55
56        #[inline(always)]
57        pub fn init_custom_heap() {
58            unsafe {
59                $crate::heap::init_custom_heap_raw(
60                    core::ptr::addr_of_mut!(__CUSTOM_HEAP_ARENA.0) as *mut u8,
61                    $bytes,
62                );
63            }
64        }
65    };
66}

68/// Zero-sized handle; all real state lives in the `static mut`s above so this
69/// `#[global_allocator]` static carries no writable data that could land read-only.
70struct CellAllocator;
71// SAFETY: single-task cells never enter the allocator concurrently, so the
72// unsynchronised access to the `static mut` heap state is race-free. Allocation
73// correctness is delegated to the well-tested `linked_list_allocator::Heap`.
74unsafe impl GlobalAlloc for CellAllocator {
75    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
76        let heap = &mut *addr_of_mut!(HEAP);
77        if !HEAP_INIT {
78            // SAFETY: HEAP_MEM is a 'static, 16-byte-aligned arena owned solely by the
79            // heap; init runs exactly once (single-task → no race on HEAP_INIT).
80            heap.init(addr_of_mut!(HEAP_MEM) as *mut u8, HEAP_SIZE);
81            HEAP_INIT = true;
82        }
83        heap.allocate_first_fit(layout)
84            .map_or(null_mut(), |p| p.as_ptr())
85    }
86
87    unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
88        if let Some(p) = NonNull::new(ptr) {
89            let heap = &mut *addr_of_mut!(HEAP);
90            heap.deallocate(p, layout);
91        }
92    }
93}

95#[cfg(target_os = "none")]
96#[global_allocator]
97static ALLOCATOR: CellAllocator = CellAllocator;

99/// Allocation-failure handler. We are genuinely out of heap (4 MiB of LIVE
100/// objects), so in-process recovery is hopeless. Rather than hang — which would
101/// leave a paralyzed-but-alive cell the supervisor cannot detect — log via the
102/// raw syscall (no allocation) and exit abnormally so the supervisor restarts the
103/// cell with a fresh heap. That is the "never-die" response to OOM.
104#[cfg(all(not(test), target_os = "none"))]
105#[alloc_error_handler]
106fn alloc_error(layout: Layout) -> ! {
107    let _ = crate::syscall::sys_log("OOM: cell heap exhausted — size=");
108    let mut buf = [0u8; 32];
109    let mut n = layout.size();
110    let mut i = 0;
111    if n == 0 {
112        buf[0] = b'0';
113        i = 1;
114    } else {
115        let mut tmp = [0u8; 32];
116        let mut j = 0;
117        while n > 0 {
118            tmp[j] = b'0' + (n % 10) as u8;
119            n /= 10;
120            j += 1;
121        }
122        while j > 0 {
123            j -= 1;
124            buf[i] = tmp[j];
125            i += 1;
126        }
127    }
128    buf[i] = b'\n';
129    if let Ok(s) = core::str::from_utf8(&buf[..=i]) {
130        let _ = crate::syscall::sys_log(s);
131    }
132    crate::syscall::sys_exit(0xEE); // non-zero = abnormal → supervisor restarts it
133}
