28pub(super) struct ReadSession<'a, T> {
29    ops: &'a mut T,
30    dirs: Vec<ViDirHandle>,
31    file: Option<ViVfsFileHandle>,
32}
33
34impl<'a, T: VfsReadOps> ReadSession<'a, T> {
35    pub(super) fn new(ops: &'a mut T) -> Self {
36        Self {
37            ops,
38            dirs: Vec::new(),
39            file: None,
40        }
41    }
42
43    pub(super) fn read(&mut self, plan: &FileReadPlan<'_>, max_bytes: usize) -> ViResult<Vec<u8>> {
44        let root = self.open_root("/")?;
45        let dir = plan
46            .parents
47            .iter()
48            .try_fold(root, |current, name| self.open_dir(current, name))?;
49        self.file = Some(self.open_file(dir, plan.file_name)?);
50        self.read_chunks(max_bytes)
51    }
52
53    pub(super) fn cleanup(&mut self) -> ViResult<()> {
54        let file_result = self.close_file();
55        let dir_result = self.close_dirs();
56        file_result.and(dir_result)
57    }
58
59    fn open_root(&mut self, path: &str) -> ViResult<ViDirHandle> {
60        let mut resp_buf = [0u8; IPC_BUF_SIZE];
61        match self
62            .ops
63            .call(&VfsRequest::OpenRootDir { path }, &mut resp_buf)?
64        {
65            VfsResponse::DirHandle(handle) => {
66                self.dirs.push(handle);
67                Ok(handle)
68            }
69            VfsResponse::Err(code) => Err(vfs_err_from_code(code)),
70            _ => Err(ViError::IO),
71        }
72    }
73
74    fn open_dir(&mut self, dir: ViDirHandle, name: &str) -> ViResult<ViDirHandle> {
75        let mut resp_buf = [0u8; IPC_BUF_SIZE];
76        match self
77            .ops
78            .call(&VfsRequest::OpenDir { dir, name }, &mut resp_buf)?
79        {
80            VfsResponse::DirHandle(handle) => {
81                self.dirs.push(handle);
82                Ok(handle)
83            }
84            VfsResponse::Err(code) => Err(vfs_err_from_code(code)),
85            _ => Err(ViError::IO),
86        }
87    }
88
89    fn open_file(&mut self, dir: ViDirHandle, name: &str) -> ViResult<ViVfsFileHandle> {
90        let mut resp_buf = [0u8; IPC_BUF_SIZE];
91        match self
92            .ops
93            .call(&VfsRequest::OpenFileAt { dir, name }, &mut resp_buf)?
94        {
95            VfsResponse::FileHandle(handle) => Ok(handle),
96            VfsResponse::Err(code) => Err(vfs_err_from_code(code)),
97            _ => Err(ViError::IO),
98        }
99    }
100
101    fn read_chunks(&mut self, max_bytes: usize) -> ViResult<Vec<u8>> {
102        let file = self.file.ok_or(ViError::IO)?;
103        let mut bytes = Vec::with_capacity(max_bytes.min(MAX_READ_CHUNK as usize));
104        let mut offset = 0u64;
105        loop {
106            let remaining = max_bytes.saturating_sub(bytes.len());
107            let request_max = remaining.min(MAX_READ_CHUNK as usize).max(1) as u32;
108            let mut resp_buf = [0u8; IPC_BUF_SIZE];
109            match self.ops.call(
110                &VfsRequest::ReadFileHandle {
111                    file,
112                    offset,
113                    max: request_max,
114                },
115                &mut resp_buf,
116            )? {
117                VfsResponse::Data(chunk) => {
118                    if remaining == 0 {
119                        return if chunk.is_empty() {
120                            Ok(bytes)
121                        } else {
122                            Err(ViError::OutOfMemory)
123                        };
124                    }
125                    if chunk.len() > remaining {
126                        return Err(ViError::OutOfMemory);
127                    }
128                    bytes.extend_from_slice(chunk);
129                    if chunk.len() < remaining.min(MAX_READ_CHUNK as usize) {
130                        return Ok(bytes);
131                    }
132                    offset = offset
133                        .checked_add(chunk.len() as u64)
134                        .ok_or(ViError::OutOfMemory)?;
135                }
136                VfsResponse::Err(code) => return Err(vfs_err_from_code(code)),
137                _ => return Err(ViError::IO),
138            }
139        }
140    }
141
142    fn close_file(&mut self) -> ViResult<()> {
143        let Some(file) = self.file.take() else {
144            return Ok(());
145        };
146        let mut resp_buf = [0u8; IPC_BUF_SIZE];
147        match self
148            .ops
149            .call(&VfsRequest::CloseFile { file }, &mut resp_buf)?
150        {
151            VfsResponse::Ok => Ok(()),
152            VfsResponse::Err(code) => Err(vfs_err_from_code(code)),
153            _ => Err(ViError::IO),
154        }
155    }
156
157    fn close_dirs(&mut self) -> ViResult<()> {
158        let mut first_err = None;
159        while let Some(dir) = self.dirs.pop() {
160            let mut resp_buf = [0u8; IPC_BUF_SIZE];
161            let close = match self.ops.call(&VfsRequest::CloseDir { dir }, &mut resp_buf) {
162                Ok(VfsResponse::Ok) => Ok(()),
163                Ok(VfsResponse::Err(code)) => Err(vfs_err_from_code(code)),
164                Ok(_) => Err(ViError::IO),
165                Err(err) => Err(err),
166            };
167            if first_err.is_none() {
168                first_err = close.err();
169            }
170        }
171        first_err.map_or(Ok(()), Err)
172    }
173}
