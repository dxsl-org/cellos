18/// Failure returned by a KMS IPC round trip.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum KmsClientError {
21    ServiceNotFound,
22    Ipc,
23    WrongSender,
24    MismatchedResponse,
25    InvalidPayload,
26    Wire(KmsWireError),
27    Service(KmsErrorCode),
28}
29
30/// Synchronous client for opaque node-identity operations.
31pub struct KmsClient {
32    tid: usize,
33    next_request_id: AtomicU32,
34}
35
36impl KmsClient {
37    /// Resolve the live KMS provider with bounded scheduler retries.
38    pub fn connect() -> Result<Self, KmsClientError> {
39        for _ in 0..8 {
40            if let Some(tid) = syscall::sys_lookup_service(api::syscall::service::KMS) {
41                return Ok(Self {
42                    tid,
43                    next_request_id: AtomicU32::new(FIRST_REQUEST_ID),
44                });
45            }
46            task::yield_now();
47        }
48        Err(KmsClientError::ServiceNotFound)
49    }

83    /// Ask KMS to perform X25519 static DH for an opaque C2C identity handle.
84    ///
85    /// The response contains the ephemeral shared secret, never the node's
86    /// private scalar. A stale handle or binding epoch fails closed in KMS.
87    pub fn noise_static_dh(
88        &self,
89        handle: NodeIdentityHandle,
90        binding_epoch: BindingEpoch,
91        peer_public_key: &[u8; 32],
92    ) -> Result<[u8; 32], KmsClientError> {
93        if handle.0 == 0 || binding_epoch.0 == 0 || peer_public_key.iter().all(|byte| *byte == 0) {
94            return Err(KmsClientError::InvalidPayload);
95        }
96        let payload = NoiseStaticDhRequestPayload {
97            handle,
98            key_id: KMS_NODE_KEY_ID_C2C,
99            reserved: 0,
100            binding_epoch,
101            peer_public_key: *peer_public_key,
102        };
103        let response = self.call_opcode(KmsOpcode::NoiseStaticDh, &payload.encode())?;
104        let payload = self.decode_payload(&response, NoiseStaticDhResponsePayload::decode)?;
105        if payload.handle != handle || payload.binding_epoch != binding_epoch {
106            return Err(KmsClientError::MismatchedResponse);
107        }
108        let shared_secret = payload.shared_secret;
109        if shared_secret.iter().all(|byte| *byte == 0) {
110            return Err(KmsClientError::InvalidPayload);
111        }
112        Ok(shared_secret)
113    }

115    /// Rotate the node identity under supervisor authority.
116    ///
117    /// `expected_blob_revision` is an exact nonzero compare-and-swap guard.
118    /// Callers must first read status; zero is never a wildcard recovery mode.
119    pub fn rotate_node_identity(
120        &self,
121        reason: RotateNodeIdentityReason,
122        expected_blob_revision: u64,
123    ) -> Result<RotateNodeIdentityResponsePayload, KmsClientError> {
124        if expected_blob_revision == 0 {
125            return Err(KmsClientError::InvalidPayload);
126        }
127        let payload = RotateNodeIdentityRequestPayload {
128            reason,
129            reserved0: 0,
130            flags: 0,
131            expected_blob_revision,
132        };
133        let response = self.call_opcode(KmsOpcode::RotateNodeIdentity, &payload.encode())?;
134        self.decode_payload(&response, RotateNodeIdentityResponsePayload::decode)
135    }

137    fn call_opcode(
138        &self,
139        opcode: KmsOpcode,
140        payload: &[u8],
141    ) -> Result<KmsResponseV1, KmsClientError> {
142        let request_id = self
143            .next_request_id
144            .fetch_update(Ordering::Relaxed, Ordering::Relaxed, |current| {
145                current.checked_add(1).filter(|next| *next != 0)
146            })
147            .map_err(|_| KmsClientError::InvalidPayload)?;
148        let request =
149            KmsRequestV1::new(opcode, request_id, payload).map_err(KmsClientError::Wire)?;
150        self.call(&request)
151    }
152
153    fn call(&self, request: &KmsRequestV1) -> Result<KmsResponseV1, KmsClientError> {
154        match syscall::sys_send(self.tid, &request.to_bytes()) {
155            syscall::SyscallResult::Ok(_) => {}
156            syscall::SyscallResult::Err(_) => return Err(KmsClientError::Ipc),
157        }
158
159        let mut bytes = [0u8; KMS_MESSAGE_LEN];
160        // KMS replies must be correlated to the live service tid so queued
161        // traffic from other senders cannot be consumed as a false reply.
162        match syscall::sys_recv(self.tid, &mut bytes) {
163            syscall::SyscallResult::Ok(sender) if sender == self.tid => {}
164            syscall::SyscallResult::Ok(_) => return Err(KmsClientError::WrongSender),
165            syscall::SyscallResult::Err(_) => return Err(KmsClientError::Ipc),
166        }
167
168        let response = KmsResponseV1::from_bytes(&bytes).map_err(KmsClientError::Wire)?;
169        if response.opcode().map_err(KmsClientError::Wire)?
170            != request.opcode().map_err(KmsClientError::Wire)?
171            || response.request_id != request.request_id
172        {
173            return Err(KmsClientError::MismatchedResponse);
174        }
175        if let Some(code) = response.error_code().map_err(KmsClientError::Wire)? {
176            return Err(KmsClientError::Service(code));
177        }
178        Ok(response)
179    }
180
181    fn decode_payload<T>(
182        &self,
183        response: &KmsResponseV1,
184        decode: fn(&[u8]) -> Option<T>,
185    ) -> Result<T, KmsClientError> {
186        let body = response.payload().map_err(KmsClientError::Wire)?;
187        decode(body).ok_or(KmsClientError::InvalidPayload)
188    }
189
190    /// Require an empty success payload (opcodes 12 and 13).
191    fn expect_empty(&self, response: &KmsResponseV1) -> Result<(), KmsClientError> {
192        let body = response.payload().map_err(KmsClientError::Wire)?;
193        if body.is_empty() {
194            Ok(())
195        } else {
196            Err(KmsClientError::InvalidPayload)
197        }
198    }
