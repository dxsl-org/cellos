3//! `TlsStream` — an [`embedded_io`] adapter over the raw `ostd::tls` IPC.
4//!
5//! Mirrors [`TcpStream`][super::net::TcpStream]: it turns the net cell's
6//! request/reply TLS opcodes ([`tls_connect`]/[`tls_write`]/[`tls_read`]/
7//! [`tls_close`]) into a blocking `Read + Write` stream so a generic
8//! [`HttpClient`][crate::http::HttpClient] can drive HTTPS exactly the same way
9//! it drives plaintext HTTP over `TcpStream`.
10//!
11//! # Blocking / IPC contract
12//! - [`connect`][TlsStream::connect] resolves the NET service TID, then performs
13//!   the TCP + TLS handshake synchronously inside the net cell. It returns only
14//!   after the handshake completes (or the net cell reports failure).
15//! - [`Write::write`] loops over the 503-byte-per-call cap of [`tls_write`]
16//!   until every byte of `buf` is accepted (write-all semantics), because a
17//!   single TLS record write may be partially accepted by the net cell.
18//! - [`Read::read`] returns zero only for EOF (or an empty destination). TLS
19//!   service errors propagate immediately instead of being retried as no-data.
20//!   The service retains reads while transport data is unavailable.
21//!
22//! # Security posture
23//! `TlsStream` inherits service-net's selected TLS build profile. The default
24//! `tls-roots-embedded` profile validates the server certificate chain,
25//! validity period, and SNI hostname against its build-selected CA anchor.
26//! `tls-insecure` remains an explicit development-only profile with no server
27//! authentication. Neither profile supplies a client certificate or signer, so
28//! this API is server-authenticated TLS only and cannot connect to the
29//! mutual-TLS relay.
30
31extern crate alloc;
32
33use crate::service;
34use crate::tls::{tls_close, tls_connect, tls_read, tls_write};
35use crate::{ViError, ViResult};
36
37/// Max bytes accepted by a single [`tls_write`] IPC call (mirrors the cap in
38/// `tls.rs`: 512-byte IPC buffer minus the 11-byte `[op][cap][len]` header).
39const MAX_TLS_WRITE: usize = 501;
40
41/// Bound write-side zero-progress retries; receive never retries EOF or errors.
42const WRITE_RETRY_BUDGET: u32 = 100_000;
43
44/// A TLS 1.3 connection handle implementing [`embedded_io::Read`] +
45/// [`embedded_io::Write`].
46///
47/// The underlying connection (and its TCP socket) is closed when the handle is
48/// dropped (RAII — Law 8).
49pub struct TlsStream {
50    net_tid: usize,
51    cap_id: u64,
52}
53
54impl TlsStream {
55    /// Open a TLS 1.3 connection to `addr:port` with SNI `hostname`.
56    ///
57    /// Resolves the NET service via [`service::lookup`] and performs the
58    /// handshake synchronously. Returns `ViError::NotFound` if the net service
59    /// is unavailable, or `ViError::IO` if the handshake fails.
60    pub fn connect(addr: [u8; 4], port: u16, hostname: &str) -> ViResult<Self> {
61        let net_tid = service::lookup(service::service::NET).ok_or(ViError::NotFound)?;
62        let cap_id = tls_connect(net_tid, addr, port, hostname);
63        if cap_id == 0 {
64            return Err(ViError::IO);
65        }
66        Ok(Self { net_tid, cap_id })
67    }
68}
69
70impl Drop for TlsStream {
71    fn drop(&mut self) {
72        tls_close(self.net_tid, self.cap_id);
73    }
74}
75
76impl embedded_io::ErrorType for TlsStream {
77    type Error = crate::io::OstdError;
78}
79
80impl embedded_io::Read for TlsStream {
81    /// Block until data, EOF or a service error is available.
82    fn read(&mut self, buf: &mut [u8]) -> Result<usize, crate::io::OstdError> {
83        if buf.is_empty() {
84            return Ok(0);
85        }
86        tls_read(self.net_tid, self.cap_id, buf).map_err(crate::io::OstdError)
87    }
88}
89
90impl embedded_io::Write for TlsStream {
91    /// Write-all: loop over the 503-byte [`tls_write`] cap until every byte of
92    /// `buf` has been accepted by the net cell.
93    ///
94    /// Returns `Ok(buf.len())` on success. A stalled send (the net cell accepts
95    /// 0 bytes) surfaces as `ViError::WouldBlock` after the retry budget,
96    /// rather than looping forever.
97    fn write(&mut self, buf: &[u8]) -> Result<usize, crate::io::OstdError> {
98        if buf.is_empty() {
99            return Ok(0);
100        }
101        let mut sent = 0usize;
102        let mut stalls: u32 = 0;
103        while sent < buf.len() {
104            let chunk_end = (sent + MAX_TLS_WRITE).min(buf.len());
105            let n = tls_write(self.net_tid, self.cap_id, &buf[sent..chunk_end]);
106            if n == 0 {
107                stalls += 1;
108                if stalls >= WRITE_RETRY_BUDGET {
109                    return Err(crate::io::OstdError(ViError::WouldBlock));
110                }
111                crate::task::yield_now();
112                continue;
113            }
114            stalls = 0;
115            sent += n;
116        }
117        Ok(sent)
118    }
119
120    fn flush(&mut self) -> Result<(), crate::io::OstdError> {
121        Ok(()) // tls_write commits synchronously through the net cell
122    }
123}
