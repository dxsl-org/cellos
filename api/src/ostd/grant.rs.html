1//! Typed linear handles for kernel Grant regions.
2//!
3//! A [`GrantHandle<T>`] wraps a raw grant ID (physical base address in SAS).
4//! It is `!Copy + !Clone + !Send`: Rust affine ownership permits one wrapper,
5//! while the kernel permits only the allocating **task** to free the grant.
6//! Dropping the handle on that task calls `sys_grant_free`.
7//!
8//! `GrantShare` grants another task bounded access; it never transfers owner
9//! authority. Cross-task users carry the raw ID only for access syscalls and
10//! leave deallocation to the allocating task.
11//!
12//! # Typical flow
13//!
14//! ```rust,no_run
15//! use ostd::grant::GrantHandle;
16//!
17//! // Allocate and use a 4-KiB region on one task.
18//! let mut handle = GrantHandle::<u8>::alloc(4096).expect("OOM");
19//! // ... fill/share/use the grant, while retaining this owner handle ...
20//! drop(handle); // owner task frees it
21//! ```
22
23use core::marker::PhantomData;
24
25use crate::syscall::{sys_grant_alloc, sys_grant_free, sys_grant_slice};
26
27/// A typed, linear handle to a kernel-managed grant region.
28///
29/// `T` is a logical element type — the kernel manages raw byte pages and has
30/// no knowledge of `T`. The region holds `len / size_of::<T>()` elements.
31///
32/// # Linear task-local invariant
33/// `!Copy + !Clone + !Send`. At most one `GrantHandle<T>` exists per `grant_id`,
34/// and it remains on the task that allocated the grant. Rust move semantics
35/// prevent duplicate wrappers; kernel owner-TID checks enforce deallocation.
36///
37/// # Drop
38/// `Drop` calls [`sys_grant_free`] from the allocating task, releasing the
39/// frames back to the kernel. [`into_raw`](GrantHandle::into_raw) only detaches
40/// that task-local Rust wrapper; it does not transfer kernel ownership.
41pub struct GrantHandle<T> {
42    id: usize,
43    len: usize,
44    _type: PhantomData<*mut T>,
45}
46
47impl<T> GrantHandle<T> {
48    /// Allocate a new grant region holding `count` elements of type `T`.
49    ///
50    /// The kernel allocates page-aligned contiguous frames; the actual region
51    /// may be slightly larger than `count * size_of::<T>()` due to alignment.
52    ///
53    /// Returns `None` on OOM or if `count * size_of::<T>()` overflows.
54    pub fn alloc(count: usize) -> Option<Self> {
55        let byte_len = count.checked_mul(core::mem::size_of::<T>())?;
56        let byte_len = byte_len.max(1); // zero-size grants are not useful
57        let id = sys_grant_alloc(byte_len)?;
58        Some(Self {
59            id,
60            len: byte_len,
61            _type: PhantomData,
62        })
63    }
64
65    /// Reconstruct a raw grant ID previously detached on this same task.
66    ///
67    /// # Safety
68    /// `id` must name a live grant allocated by the current task, and no other
69    /// owner wrapper may exist. `len` must equal the byte length associated with
70    /// the original `sys_grant_alloc`. `GrantShare`, IPC receipt, CellId
71    /// equality, or access to the SAS address does not establish owner authority.
72    /// Calling this twice with the same `id` creates duplicate owner wrappers.
73    pub unsafe fn from_raw(id: usize, len: usize) -> Self {
74        Self {
75            id,
76            len,
77            _type: PhantomData,
78        }
79    }
80
81    /// Consume the handle, returning `(grant_id, byte_len)` **without** freeing.
82    ///
83    /// This supports a same-task wrapper handoff to code that later reconstructs
84    /// it with [`GrantHandle::from_raw`]. It does not change the kernel owner TID.
85    #[inline]
86    pub fn into_raw(self) -> (usize, usize) {
87        let id = self.id;
88        let len = self.len;
89        // Prevent Drop from calling sys_grant_free — ownership is transferred.
90        core::mem::forget(self);
91        (id, len)
92    }
93
94    /// Raw kernel grant ID (physical base address in SAS == virtual address).
95    #[inline]
96    pub fn id(&self) -> usize {
97        self.id
98    }
99
100    /// Byte length of the grant region.
101    #[inline]
102    pub fn len(&self) -> usize {
103        self.len
104    }
105
106    /// True when the grant region has zero length (never the case for a
107    /// successful `sys_grant_alloc`, but clippy requires `is_empty` beside `len`).
108    #[inline]
109    pub fn is_empty(&self) -> bool {
110        self.len == 0
111    }
112
113    /// Get an exclusive byte slice over the entire grant region.
114    ///
115    /// # Safety
116    /// No other live reference to this grant must exist. The handle must still
117    /// be valid (not freed). The caller must ensure the data at `[0, len)` is
118    /// initialised before reading.
119    pub unsafe fn as_bytes_mut(&mut self) -> &mut [u8] {
120        let ptr = sys_grant_slice(self.id)
121            .expect("GrantHandle::as_bytes_mut: grant not found or permission denied");
122        core::slice::from_raw_parts_mut(ptr, self.len)
123    }
124}
125
126impl GrantHandle<u8> {
127    /// Allocate a byte grant and initialize it from `data` before the handle escapes.
128    ///
129    /// `data` is copied into a freshly allocated grant owned exclusively by the
130    /// returned handle. The copy happens before the handle is exposed to callers,
131    /// preserving the linear-ownership invariant: no grant-sharing alias can
132    /// observe partially initialized bytes.
133    ///
134    /// Returns `Some(handle)` when the kernel allocates a grant large enough to
135    /// hold `data.len()` bytes; returns `None` on allocation failure or if the
136    /// kernel refuses to map the new grant for initialization.
137    pub fn alloc_copy_from_slice(data: &[u8]) -> Option<Self> {
138        let handle = Self::alloc(data.len())?;
139        let ptr = sys_grant_slice(handle.id)?;
140        // SAFETY: `handle` uniquely owns a fresh grant that has not escaped this
141        // function. `sys_grant_slice` returns the writable mapping for exactly
142        // `handle.len` bytes, and `data.len() <= handle.len` by construction.
143        unsafe {
144            core::ptr::copy_nonoverlapping(data.as_ptr(), ptr, data.len());
145        }
146        Some(handle)
147    }
148}
149
150impl<T: Copy> GrantHandle<T> {
151    /// Get an exclusive typed slice over the grant region.
152    ///
153    /// # Safety
154    /// No other live reference must exist. All bytes must be initialised as
155    /// valid `T` values (bitwise validity).
156    pub unsafe fn as_slice_mut(&mut self) -> &mut [T] {
157        let ptr = sys_grant_slice(self.id)
158            .expect("GrantHandle::as_slice_mut: grant not found or permission denied");
159        let count = self.len / core::mem::size_of::<T>();
160        core::slice::from_raw_parts_mut(ptr as *mut T, count)
161    }
162}
163
164impl<T> Drop for GrantHandle<T> {
165    fn drop(&mut self) {
166        sys_grant_free(self.id);
167    }
