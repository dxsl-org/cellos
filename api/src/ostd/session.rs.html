9/// Execute exactly one bounded kernel request. Errors remain in `header.status`.
10pub fn kernel_request(
11    op: KernelOp,
12    request_id: u64,
13    body: &[u8],
14    reply: &mut [u8],
15) -> Result<ResultHeaderV1, Status> {
16    let len = HEADER_LEN
17        .checked_add(body.len())
18        .ok_or(Status::Malformed)?;
19    if len > MAX_REQUEST_LEN || reply.len() < HEADER_LEN || request_id == 0 {
20        return Err(Status::Malformed);
21    }
22    let mut request = Scrub::<MAX_REQUEST_LEN>::new();
23    RequestHeaderV1 {
24        selector: op as u16,
25        total_len: len as u32,
26        request_id,
27    }
28    .encode(&mut request[..HEADER_LEN])?;
29    request[HEADER_LEN..len].copy_from_slice(body);
30    match crate::syscall::sys_cell_session_v1(&request[..len], reply) {
31        crate::syscall::SyscallResult::Ok(0) => {}
32        _ => return Err(Status::Indeterminate),
33    }
34    let required =
35        u32::from_le_bytes(reply[8..12].try_into().map_err(|_| Status::Malformed)?) as usize;
36    let actual = if required > reply.len() {
37        HEADER_LEN
38    } else {
39        required
40    };
41    if actual < HEADER_LEN {
42        return Err(Status::Malformed);
43    }
44    let header = ResultHeaderV1::decode(&reply[..actual])?;
45    if op != KernelOp::QueryRequest
46        && (header.selector != op as u16 || header.request_id != request_id)
47    {
48        return Err(Status::Malformed);
49    }
50    Ok(header)
51}

53/// An accepted-or-not mutation whose outcome could not yet be observed. The next call
54/// on the lane first resolves it by query (never by replay); an identical call returns
55/// the original retained outcome.
56#[derive(Clone, Copy)]
57struct Pending {
58    selector: u16,
59    digest: u64,
60}
61#[derive(Clone, Copy)]
62struct Lane {
63    issuer: u16,
64    incarnation: Option<ServiceBinding>,
65    operation: ObjectRefV1,
66    next: u64,
67    blocked: bool,
68    pending: Option<Pending>,
69}
70// This is only correlation bookkeeping, never a resource/authority registry. Shared
71// across handles/threads so constructing another SDK client cannot reuse ID 1.
72static LANES: spin::Mutex<alloc::vec::Vec<Lane>> = spin::Mutex::new(alloc::vec::Vec::new());
73
74/// Drop every correlation lane for `operation`. Call once the operation is closed,
75/// acknowledged or otherwise retired so the bounded registry cannot be exhausted.
76pub fn retire_operation(operation: ObjectRefV1) {
77    LANES.lock().retain(|lane| lane.operation != operation);
78}

80/// Retained-outcome polling budget for a still-pending original: 20 queries, 50 ms
81/// apart (about 5 scheduler ticks at the 10 ms x86 LAPIC tick).
82const PENDING_QUERIES: u32 = 20;
83const PENDING_WAIT_TICKS: usize = 5;
84
85fn body_digest(body: &[u8]) -> u64 {
86    let mut hash = 0xcbf2_9ce4_8422_2325u64;
87    for byte in body {
88        hash ^= *byte as u64;
89        hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
90    }
91    hash
92}

94enum Answer {
95    Outcome(ResultHeaderV1),
96    Unknown,
97    Pending,
98    Failed(Status),
99}

101/// Ask the owner for the retained outcome of `id`. A `WouldBlock` answer carrying the
102/// query's own selector means the original is still pending: re-query with a bounded
103/// delay rather than treating it as corruption.
104fn query_outcome<F>(
105    call: &mut F,
106    query_selector: u16,
107    selector: u16,
108    operation: ObjectRefV1,
109    id: u64,
110    reply: &mut [u8],
111) -> Answer
112where
113    F: FnMut(u16, u64, &[u8], &mut [u8]) -> Result<ResultHeaderV1, Status>,
114{
115    let mut query = [0u8; 24];
116    query[..16].copy_from_slice(&operation.to_bytes());
117    query[16..].copy_from_slice(&id.to_le_bytes());
118    for attempt in 0..=PENDING_QUERIES {
119        if attempt != 0 {
120            let _ = crate::syscall::sys_set_timer(PENDING_WAIT_TICKS);
121        }
122        match call(query_selector, id, &query, reply) {
123            Ok(h) if h.selector == selector && h.request_id == id => return Answer::Outcome(h),
124            Ok(h) if h.status == Status::UnknownRequest => return Answer::Unknown,
125            Ok(h) if h.status == Status::WouldBlock && h.selector == query_selector => {}
126            Ok(_) => return Answer::Failed(Status::Indeterminate),
127            // Queries are read-only: a transient transport failure is retried too.
128            Err(Status::WouldBlock | Status::TimedOut | Status::Indeterminate) => {}
129            Err(e) => return Answer::Failed(e),
130        }
131    }
132    Answer::Pending
133}

135/// One correlated request: the lane it runs on (issuer, incarnation,
136/// operation) and the selectors it, its outcome query and its ACK use.
137struct Correlated<'a> {
138    issuer: u16,
139    incarnation: Option<ServiceBinding>,
140    operation: ObjectRefV1,
141    selector: u16,
142    mutating: bool,
143    query_selector: u16,
144    ack_selector: u16,
145    body: &'a [u8],
146}
147
148fn correlated<F>(
149    request: Correlated<'_>,
150    reply: &mut [u8],
151    mut call: F,
152) -> Result<ResultHeaderV1, Status>
153where
154    F: FnMut(u16, u64, &[u8], &mut [u8]) -> Result<ResultHeaderV1, Status>,
155{
156    let Correlated {
157        issuer,
158        incarnation,
159        operation,
160        selector,
161        mutating,
162        query_selector,
163        ack_selector,
164        body,
165    } = request;
166    let mut lanes = LANES.lock();
167    let index = match lanes.iter().position(|l| {
168        l.issuer == issuer && l.incarnation == incarnation && l.operation == operation
169    }) {
170        Some(i) => i,
171        None => {
172            if lanes.len() == 192 {
173                return Err(Status::QuotaExceeded);
174            }
175            lanes.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
176            lanes.push(Lane {
177                issuer,
178                incarnation,
179                operation,
180                next: 1,
181                blocked: false,
182                pending: None,
183            });
184            lanes.len() - 1
185        }
186    };
187    let lane = &mut lanes[index];
188    if lane.blocked {
189        return Err(Status::Indeterminate);
190    }
191    let digest = body_digest(body);
192    let mut resumed = None;
193    if let Some(p) = lane.pending {
194        // Resolve the earlier uncertain mutation by query before anything else.
195        match query_outcome(
196            &mut call,
197            query_selector,
198            p.selector,
199            operation,
200            lane.next,
201            reply,
202        ) {
203            Answer::Outcome(h) => {
204                lane.pending = None;
205                if mutating && p.selector == selector && p.digest == digest {
206                    // The caller is retrying the identical operation: hand back the
207                    // original retained outcome; the request is never re-sent.
208                    resumed = Some(h);
209                } else {
210                    // A different operation: retire the observed one, then proceed.
211                    let mut ack = [0u8; 24];
212                    ack[..16].copy_from_slice(&operation.to_bytes());
213                    ack[16..].copy_from_slice(&lane.next.to_le_bytes());
214                    lane.next += 1;
215                    let _ = call(ack_selector, lane.next - 1, &ack, &mut [0u8; HEADER_LEN]);
216                }
217            }
218            // Still pending (the lane keeps the in-flight record and is not blocked).
219            Answer::Pending => return Err(Status::Indeterminate),
220            Answer::Unknown => {
221                lane.blocked = true;
222                return Err(Status::Indeterminate);
223            }
224            Answer::Failed(e) => {
225                lane.blocked = true;
226                return Err(e);
227            }
228        }
229    }
230    let id = lane.next;
231    if id == u64::MAX {
232        return Err(Status::QuotaExceeded);
233    }
234    let header = if let Some(h) = resumed {
235        h
236    } else {
237        let result = call(selector, id, body, reply);
238        match result {
239            Ok(h) => h,
240            Err(Status::Indeterminate) if mutating => {
241                match query_outcome(&mut call, query_selector, selector, operation, id, reply) {
242                    Answer::Outcome(h) => h,
243                    Answer::Pending => {
244                        lane.pending = Some(Pending { selector, digest });
245                        return Err(Status::Indeterminate);
246                    }
247                    Answer::Unknown => {
248                        lane.blocked = true;
249                        return Err(Status::Indeterminate);
250                    }
251                    Answer::Failed(e) => {
252                        lane.blocked = true;
253                        return Err(e);
254                    }
255                }
256            }
257            Err(e) => {
258                // Definite pre-dispatch rejection has no retained outcome or fence.
259                if mutating
260                    && !matches!(
261                        e,
262                        Status::WouldBlock
263                            | Status::TimedOut
264                            | Status::Malformed
265                            | Status::Stale
266                            | Status::NotFound
267                    )
268                {
269                    lane.blocked = true;
270                }
271                return Err(e);
272            }
273        }
274    };
275    if header.selector != selector || header.request_id != id {
276        lane.blocked = true;
277        return Err(Status::Malformed);
278    }
279    if mutating && header.status != Status::BufferTooSmall {
280        if header.status != Status::Ok {
281            // Admission errors can precede outcome reservation. Ask the owner
282            // whether this ID crossed its fence rather than guessing from an
283            // error code (QuotaExceeded can occur on either side).
284            let mut retained = [0u8; HEADER_LEN];
285            match query_outcome(
286                &mut call,
287                query_selector,
288                selector,
289                operation,
290                id,
291                &mut retained,
292            ) {
293                Answer::Outcome(_) => {}
294                Answer::Unknown => return Ok(header),
295                Answer::Pending => {
296                    lane.pending = Some(Pending { selector, digest });
297                    return Err(Status::Indeterminate);
298                }
299                Answer::Failed(_) => {
300                    lane.blocked = true;
301                    return Err(Status::Indeterminate);
302                }
303            }
304        }
305        lane.next += 1;
306        // A complete observed outcome can now be retired. ACK failure does not
307        // turn the already observed operation into an uncertain one.
308        let mut ack = [0u8; 24];
309        ack[..16].copy_from_slice(&operation.to_bytes());
310        ack[16..].copy_from_slice(&id.to_le_bytes());
311        let mut ack_reply = [0u8; HEADER_LEN];
312        let _ = call(ack_selector, id, &ack, &mut ack_reply);
313    }
314    Ok(header)
315}

317/// Operation-scoped kernel request. No accepted mutation is automatically replayed.
318pub fn kernel_call(
319    op: KernelOp,
320    operation: ObjectRefV1,
321    mutating: bool,
322    body: &[u8],
323    reply: &mut [u8],
324) -> Result<ResultHeaderV1, Status> {
325    let request = Correlated {
326        issuer: 0,
327        incarnation: None,
328        operation,
329        selector: op as u16,
330        mutating,
331        query_selector: KernelOp::QueryRequest as u16,
332        ack_selector: KernelOp::AcknowledgeRequests as u16,
333        body,
334    };
335    let outcome = correlated(request, reply, |selector, id, body, reply| {
336        let opcode = KernelOp::try_from(selector).map_err(|_| Status::UnknownOperation)?;
337        if opcode == KernelOp::QueryRequest {
338            // Query returns the original header, not a nested query result.
339            return kernel_query(id, body, reply);
340        }
341        kernel_request(opcode, id, body, reply)
342    });
343    if op == KernelOp::CloseOperation && outcome.as_ref().is_ok_and(|h| h.status == Status::Ok) {
344        retire_operation(operation);
345    }
346    outcome
347}
348fn kernel_query(id: u64, body: &[u8], reply: &mut [u8]) -> Result<ResultHeaderV1, Status> {
349    // The raw helper enforces echo for normal requests; query's successful record
350    // intentionally echoes the *original* selector instead.
351    let mut request = [0u8; 48];
352    RequestHeaderV1 {
353        selector: KernelOp::QueryRequest as u16,
354        total_len: 48,
355        request_id: id,
356    }
357    .encode(&mut request[..24])?;
358    if body.len() != 24 {
359        return Err(Status::Malformed);
360    }
361    request[24..].copy_from_slice(body);
362    // The kernel reserves its maximum retained-record size before executing a
363    // query, so the scratch must be at least that large (kernel RESULT_MAX).
364    let mut scratch = Scrub::<768>::new();
365    if !matches!(
366        crate::syscall::sys_cell_session_v1(&request, &mut scratch),
367        crate::syscall::SyscallResult::Ok(0)
368    ) {
369        return Err(Status::Indeterminate);
370    }
371    let n = u32::from_le_bytes(scratch[8..12].try_into().map_err(|_| Status::Malformed)?) as usize;
372    if n > scratch.len() || n < HEADER_LEN {
373        return Err(Status::Malformed);
374    }
375    let header = ResultHeaderV1::decode(&scratch[..n])?;
376    if n <= reply.len() {
377        reply[..n].copy_from_slice(&scratch[..n]);
378        return Ok(header);
379    }
380    // A header-only probe (the post-status check) needs only the outcome header.
381    if reply.len() == HEADER_LEN {
382        reply.copy_from_slice(&scratch[..HEADER_LEN]);
383        return Ok(header);
384    }
385    Err(Status::BufferTooSmall)
386}

388/// One correlated endpoint mutation; callers may advance only by acknowledged
389/// positive progress. The deadline is absolute and reused across partial writes.
390pub(crate) fn write_stream(
391    operation: ObjectRefV1,
392    endpoint: ObjectRefV1,
393    bytes: &[u8],
394    deadline_ns: u64,
395) -> Result<usize, Status> {
396    write_stream_with(operation, endpoint, bytes, deadline_ns, kernel_call)
397}

399fn write_stream_with(
400    operation: ObjectRefV1,
401    endpoint: ObjectRefV1,
402    bytes: &[u8],
403    deadline_ns: u64,
404    call: impl FnOnce(KernelOp, ObjectRefV1, bool, &[u8], &mut [u8]) -> Result<ResultHeaderV1, Status>,
405) -> Result<usize, Status> {
406    use api::cell_session::{Reader, Writer, TRANSFER_MAX};
407
408    if bytes.is_empty() || bytes.len() > TRANSFER_MAX || deadline_ns == 0 || deadline_ns == u64::MAX
409    {
410        return Err(Status::Malformed);
411    }
412    let mut body = Scrub::<{ 32 + TRANSFER_MAX }>::new();
413    let mut writer = Writer::new(&mut body);
414    writer.reference(endpoint)?;
415    writer.u32(bytes.len() as u32)?;
416    writer.u32(0)?;
417    writer.u64(deadline_ns)?;
418    writer.bytes(bytes)?;
419    let length = writer.position();
420    let mut reply = [0u8; HEADER_LEN + 8];
421    let header = call(
422        KernelOp::WriteStream,
423        operation,
424        true,
425        &body[..length],
426        &mut reply,
427    )?;
428    if header.status != Status::Ok {
429        return Err(header.status);
430    }
431    if header.total_len as usize != reply.len() {
432        return Err(Status::Malformed);
433    }
434    let mut reader = Reader::new(&reply[HEADER_LEN..]);
435    let actual = reader.u32()? as usize;
436    if reader.u32()? != 0 || actual == 0 || actual > bytes.len() {
437        return Err(Status::Malformed);
438    }
439    reader.finish()?;
440    Ok(actual)
441}

443/// Resolve the current full issuer binding; a TID alone is never sufficient.
444pub fn service_binding(service: u16) -> Result<ServiceBinding, Status> {
445    let mut record = [0u8; 24];
446    crate::syscall::sys_lookup_service_bound(service, &mut record).ok_or(Status::NotFound)
447}

449/// Borrowed, exact-length authenticated SessionV1 / PreparedChildV1 RPC.
450/// `service` is restricted to the two admitted issuers. Each IPC wait is <=100 ticks.
451pub fn service_call(
452    binding: ServiceBinding,
453    service: u16,
454    operation: ObjectRefV1,
455    selector: u16,
456    mutating: bool,
457    body: &[u8],
458    reply: &mut [u8],
459) -> Result<ResultHeaderV1, Status> {
460    let (query, ack) = if service == api::syscall::service::INPUT {
461        (19, 18)
462    } else if service == api::syscall::service::VFS {
463        (5, 4)
464    } else {
465        return Err(Status::Denied);
466    };
467    let request = Correlated {
468        issuer: service,
469        incarnation: Some(binding),
470        operation,
471        selector,
472        mutating,
473        query_selector: query,
474        ack_selector: ack,
475        body,
476    };
477    correlated(request, reply, |selector, id, body, reply| {
478        if service_binding(service)? != binding {
479            return Err(Status::Stale);
480        }
481        let n = HEADER_LEN
482            .checked_add(body.len())
483            .ok_or(Status::Malformed)?;
484        if n > MAX_SERVICE_RECORD_LEN {
485            return Err(Status::Malformed);
486        }
487        let mut record = Scrub::<MAX_SERVICE_RECORD_LEN>::new();
488        RequestHeaderV1 {
489            selector,
490            total_len: n as u32,
491            request_id: id,
492        }
493        .encode(&mut record[..HEADER_LEN])?;
494        record[HEADER_LEN..n].copy_from_slice(body);
495        let mut send = Scrub::<{ api::ipc::IPC_BUF_SIZE }>::new();
496        let mut recv = Scrub::<{ api::ipc::IPC_BUF_SIZE }>::new();
497        let inner = if service == api::syscall::service::INPUT {
498            let req = api::ipc::InputRequest::SessionV1(&record[..n]);
499            let response =
500                crate::ipc::service_call_typed_bounded::<_, api::ipc::InputResponse<'_>>(
501                    binding.tid as usize,
502                    &req,
503                    &mut send,
504                    &mut recv,
505                    100,
506                )
507                .map_err(ipc_status)?;
508            match response {
509                api::ipc::InputResponse::SessionV1(bytes) => bytes,
510                _ => return Err(Status::Malformed),
511            }
512        } else {
513            let req = api::ipc::VfsRequest::PreparedChildV1(&record[..n]);
514            let response = crate::ipc::service_call_typed_bounded::<_, api::ipc::VfsResponse<'_>>(
515                binding.tid as usize,
516                &req,
517                &mut send,
518                &mut recv,
519                100,
520            )
521            .map_err(ipc_status)?;
522            match response {
523                api::ipc::VfsResponse::PreparedChildV1(bytes) => bytes,
524                _ => return Err(Status::Malformed),
525            }
526        };
527        if service_binding(service)? != binding {
528            return Err(Status::Stale);
529        }
530        if inner.len() > reply.len() {
531            return Err(Status::BufferTooSmall);
532        }
533        let h = ResultHeaderV1::decode(inner)?;
534        if selector != query && (h.selector != selector || h.request_id != id) {
535            return Err(Status::Malformed);
536        }
537        reply[..inner.len()].copy_from_slice(inner);
538        Ok(h)
539    })
540}

542/// KMS → input: register one first-trust descriptor (`InputOp::RegisterTrustChallenge`);
543/// returns the input-issued confirmation Ref the confirming client presents to
544/// `BeginTrustConfirmation`. Input admits only the live registered KMS task. The request
545/// lane is the single-use challenge Ref; it is kept only while the outcome is
546/// indeterminate (retry the identical descriptor to resolve it).
547pub fn register_trust_challenge(
548    descriptor: &api::cell_session::trust::TrustChallengeV1<'_>,
549) -> Result<ObjectRefV1, Status> {
550    use api::cell_session::trust::TrustChallengeV1;
551    let owner = service_binding(api::syscall::service::INPUT)?;
552    let mut body = [0u8; TrustChallengeV1::MAX_LEN];
553    let len = descriptor.encode(&mut body)?;
554    let mut reply = [0u8; HEADER_LEN + 16];
555    let result = service_call(
556        owner,
557        api::syscall::service::INPUT,
558        descriptor.challenge,
559        api::cell_session::InputOp::RegisterTrustChallenge as u16,
560        true,
561        &body[..len],
562        &mut reply,
563    );
564    if result != Err(Status::Indeterminate) {
565        retire_operation(descriptor.challenge);
566    }
567    let h = result?;
568    if h.status != Status::Ok {
569        return Err(h.status);
570    }
571    ObjectRefV1::from_bytes(&reply[HEADER_LEN..h.total_len as usize])
572}

574/// KMS → input: the decision of `confirmation` (`InputOp::CollectTrustDecision`), answered
575/// only to the registering principal. `DECISION_PENDING` is not consumed; a final decision
576/// is returned once (a later collect is `Stale`), and its lane (the confirmation Ref) is
577/// then retired. Compare every binding, including `descriptor_sha256` against
578/// [`trust_descriptor_sha256`] of the own record, before acting on it.
579pub fn collect_trust_decision(
580    confirmation: ObjectRefV1,
581) -> Result<api::cell_session::trust::TrustDecisionV1, Status> {
582    use api::cell_session::trust::TrustDecisionV1;
583    let owner = service_binding(api::syscall::service::INPUT)?;
584    let mut reply = [0u8; HEADER_LEN + TrustDecisionV1::LEN];
585    let result = service_call(
586        owner,
587        api::syscall::service::INPUT,
588        confirmation,
589        api::cell_session::InputOp::CollectTrustDecision as u16,
590        true,
591        &confirmation.to_bytes(),
592        &mut reply,
593    );
594    let decision = match result {
595        Ok(h) if h.status == Status::Ok => {
596            TrustDecisionV1::decode(&reply[HEADER_LEN..h.total_len as usize])
597        }
598        Ok(h) => Err(h.status),
599        Err(error) => Err(error),
600    };
601    if decision
602        .as_ref()
603        .map_or_else(|e| *e != Status::Indeterminate, TrustDecisionV1::is_final)
604    {
605        retire_operation(confirmation);
606    }
607    decision
608}

610/// SHA-256 of a descriptor's canonical bytes, as `TrustDecisionV1::descriptor_sha256` binds it.
611pub fn trust_descriptor_sha256(
612    descriptor: &api::cell_session::trust::TrustChallengeV1<'_>,
613) -> Result<[u8; 32], Status> {
614    let mut body = [0u8; api::cell_session::trust::TrustChallengeV1::MAX_LEN];
615    let len = descriptor.encode(&mut body)?;
616    Ok(sha256(&body[..len]))
617}
618fn ipc_status(error: crate::ipc::IpcError) -> Status {
619    use crate::ipc::IpcError;
620    match error {
621        IpcError::Busy => Status::WouldBlock,
622        IpcError::PreDispatchTimeout => Status::TimedOut,
623        IpcError::Encode => Status::Malformed,
624        IpcError::Decode | IpcError::BufferTooSmall => Status::Indeterminate,
625        IpcError::Cancelled => Status::Cancelled,
626        _ => Status::Indeterminate,
627    }
628}
629/// Receiver-monotonic nanoseconds, checked rather than architecture-guessed.
630pub fn monotonic_ns() -> Result<u64, Status> {
631    let frequency = crate::syscall::sys_get_timer_freq().ok_or(Status::Unsupported)?;
632    let ticks = crate::syscall::sys_get_time();
633    let ns = (ticks as u128)
634        .checked_mul(1_000_000_000)
635        .ok_or(Status::Malformed)?
636        / (frequency as u128);
637    u64::try_from(ns).map_err(|_| Status::Malformed)
638}

640/// Overwrite `bytes` with zeros using volatile stores, so the optimizer cannot treat the
641/// write as a dead store before the memory is released or reused. Secret-bearing buffers
642/// (typed text, raw stdin bytes, paste staging, retained request bodies) are wiped with
643/// this, never with a plain `fill(0)`/`clear()`.
644pub fn zeroize(bytes: &mut [u8]) {
645    zeroize_raw(bytes.as_mut_ptr(), bytes.len());
646}

648fn zeroize_raw(pointer: *mut u8, length: usize) {
649    for offset in 0..length {
650        // SAFETY: callers pass a pointer valid for `length` exclusive byte writes.
651        unsafe { core::ptr::write_volatile(pointer.add(offset), 0) };
652    }
653    core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::SeqCst);
654}
655
656/// Zero the vector's whole allocation (including spare capacity that earlier
657/// contents may have occupied), then truncate it. The allocation is retained.
658pub fn zeroize_vec(bytes: &mut alloc::vec::Vec<u8>) {
659    zeroize_raw(bytes.as_mut_ptr(), bytes.capacity());
660    bytes.clear();
661}
662
663/// Zero a string's whole allocation, then truncate it (still valid UTF-8: empty).
664pub fn zeroize_string(text: &mut alloc::string::String) {
665    // SAFETY: the vector is cleared before the borrow ends, so the string stays valid UTF-8.
666    zeroize_vec(unsafe { text.as_mut_vec() });
667}

669/// Zero every byte of a ring (both halves), then empty it.
670pub fn zeroize_deque(ring: &mut alloc::collections::VecDeque<u8>) {
671    let (first, second) = ring.as_mut_slices();
672    zeroize(first);
673    zeroize(second);
674    ring.clear();
675}
676
677/// Zero and remove the oldest `count` bytes of a ring, so a consumed prefix never
678/// lingers in a vacated slot.
679pub fn zeroize_deque_front(ring: &mut alloc::collections::VecDeque<u8>, count: usize) {
680    let count = count.min(ring.len());
681    let (first, second) = ring.as_mut_slices();
682    let head = count.min(first.len());
683    zeroize(&mut first[..head]);
684    zeroize(&mut second[..count - head]);
685    ring.drain(..count);
686}

688/// Fixed-size stack buffer that is wiped (volatile) when it goes out of scope, on every
689/// exit path including `?` returns. Dereferences to a byte slice.
690pub struct Scrub<const N: usize>([u8; N]);
691impl<const N: usize> Scrub<N> {
692    pub const fn new() -> Self {
693        Self([0; N])
694    }
695}
696impl<const N: usize> Default for Scrub<N> {
697    fn default() -> Self {
698        Self::new()
699    }
700}
701impl<const N: usize> core::ops::Deref for Scrub<N> {
702    type Target = [u8];
703    fn deref(&self) -> &[u8] {
704        &self.0
705    }
706}
707impl<const N: usize> core::ops::DerefMut for Scrub<N> {
708    fn deref_mut(&mut self) -> &mut [u8] {
709        &mut self.0
710    }
711}
712impl<const N: usize> Drop for Scrub<N> {
713    fn drop(&mut self) {
714        zeroize(&mut self.0);
715    }
716}

718/// SHA-256 (FIPS 180-4). Used only to bind a retained request identity without keeping the
719/// request's plaintext (see the input service's sealed outcomes).
720pub fn sha256(data: &[u8]) -> [u8; 32] {
721    const K: [u32; 64] = [
722        0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4,
723        0xab1c5ed5, 0xd807aa98, 0x12835b01, 0x243185be, 0x550c7dc3, 0x72be5d74, 0x80deb1fe,
724        0x9bdc06a7, 0xc19bf174, 0xe49b69c1, 0xefbe4786, 0x0fc19dc6, 0x240ca1cc, 0x2de92c6f,
725        0x4a7484aa, 0x5cb0a9dc, 0x76f988da, 0x983e5152, 0xa831c66d, 0xb00327c8, 0xbf597fc7,
726        0xc6e00bf3, 0xd5a79147, 0x06ca6351, 0x14292967, 0x27b70a85, 0x2e1b2138, 0x4d2c6dfc,
727        0x53380d13, 0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85, 0xa2bfe8a1, 0xa81a664b,
728        0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070, 0x19a4c116,
729        0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
730        0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7,
731        0xc67178f2,
732    ];
733    let mut state: [u32; 8] = [
734        0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab,
735        0x5be0cd19,
736    ];
737    fn compress(state: &mut [u32; 8], block: &[u8]) {
738        let mut w = [0u32; 64];
739        for (i, word) in block.chunks_exact(4).enumerate() {
740            w[i] = u32::from_be_bytes([word[0], word[1], word[2], word[3]]);
741        }
742        for i in 16..64 {
743            let s0 = w[i - 15].rotate_right(7) ^ w[i - 15].rotate_right(18) ^ (w[i - 15] >> 3);
744            let s1 = w[i - 2].rotate_right(17) ^ w[i - 2].rotate_right(19) ^ (w[i - 2] >> 10);
745            w[i] = w[i - 16]
746                .wrapping_add(s0)
747                .wrapping_add(w[i - 7])
748                .wrapping_add(s1);
749        }
750        let [mut a, mut b, mut c, mut d, mut e, mut f, mut g, mut h] = *state;
751        for i in 0..64 {
752            let s1 = e.rotate_right(6) ^ e.rotate_right(11) ^ e.rotate_right(25);
753            let ch = (e & f) ^ (!e & g);
754            let t1 = h
755                .wrapping_add(s1)
756                .wrapping_add(ch)
757                .wrapping_add(K[i])
758                .wrapping_add(w[i]);
759            let s0 = a.rotate_right(2) ^ a.rotate_right(13) ^ a.rotate_right(22);
760            let maj = (a & b) ^ (a & c) ^ (b & c);
761            let t2 = s0.wrapping_add(maj);
762            h = g;
763            g = f;
764            f = e;
765            e = d.wrapping_add(t1);
766            d = c;
767            c = b;
768            b = a;
769            a = t1.wrapping_add(t2);
770        }
771        for (slot, value) in state.iter_mut().zip([a, b, c, d, e, f, g, h]) {
772            *slot = slot.wrapping_add(value);
773        }
774    }
775    let mut blocks = data.chunks_exact(64);
776    for block in &mut blocks {
777        compress(&mut state, block);
778    }
779    // Final one or two blocks hold the remainder, 0x80, zero padding and the bit length.
780    let remainder = blocks.remainder();
781    let mut tail = Scrub::<128>::new();
782    tail[..remainder.len()].copy_from_slice(remainder);
783    tail[remainder.len()] = 0x80;
784    let total = if remainder.len() < 56 { 64 } else { 128 };
785    tail[total - 8..total].copy_from_slice(&((data.len() as u64).wrapping_mul(8)).to_be_bytes());
786    for block in tail[..total].chunks_exact(64) {
787        compress(&mut state, block);
788    }
789    let mut digest = [0u8; 32];
790    for (chunk, word) in digest.chunks_exact_mut(4).zip(state) {
791        chunk.copy_from_slice(&word.to_be_bytes());
792    }
793    digest
794}
