6/// Execute exactly one bounded kernel request. Errors remain in `header.status`.
7pub fn kernel_request(op: KernelOp, request_id: u64, body: &[u8], reply: &mut [u8]) -> Result<ResultHeaderV1, Status> {
8    let len = HEADER_LEN.checked_add(body.len()).ok_or(Status::Malformed)?;
9    if len > MAX_REQUEST_LEN || reply.len() < HEADER_LEN || request_id == 0 { return Err(Status::Malformed); }
10    let mut request = Scrub::<MAX_REQUEST_LEN>::new();
11    RequestHeaderV1 { selector: op as u16, total_len: len as u32, request_id }.encode(&mut request[..HEADER_LEN])?;
12    request[HEADER_LEN..len].copy_from_slice(body);
13    match crate::syscall::sys_cell_session_v1(&request[..len], reply) {
14        crate::syscall::SyscallResult::Ok(0) => {},
15        _ => return Err(Status::Indeterminate),
16    }
17    let required = u32::from_le_bytes(reply[8..12].try_into().map_err(|_| Status::Malformed)?) as usize;
18    let actual = if required > reply.len() { HEADER_LEN } else { required };
19    if actual < HEADER_LEN { return Err(Status::Malformed); }
20    let header = ResultHeaderV1::decode(&reply[..actual])?;
21    if op != KernelOp::QueryRequest && (header.selector != op as u16 || header.request_id != request_id) { return Err(Status::Malformed); }
22    Ok(header)
23}

25/// An accepted-or-not mutation whose outcome could not yet be observed. The next call
26/// on the lane first resolves it by query (never by replay); an identical call returns
27/// the original retained outcome.
28#[derive(Clone, Copy)]
29struct Pending { selector: u16, digest: u64 }
30#[derive(Clone, Copy)]
31struct Lane { issuer: u16, incarnation: Option<ServiceBinding>, operation: ObjectRefV1, next: u64, blocked: bool, pending: Option<Pending> }
32// This is only correlation bookkeeping, never a resource/authority registry. Shared
33// across handles/threads so constructing another SDK client cannot reuse ID 1.
34static LANES: spin::Mutex<alloc::vec::Vec<Lane>> = spin::Mutex::new(alloc::vec::Vec::new());
35
36/// Drop every correlation lane for `operation`. Call once the operation is closed,
37/// acknowledged or otherwise retired so the bounded registry cannot be exhausted.
38pub fn retire_operation(operation: ObjectRefV1) {
39    LANES.lock().retain(|lane| lane.operation != operation);
40}

42/// Retained-outcome polling budget for a still-pending original: 20 queries, 50 ms
43/// apart (about 5 scheduler ticks at the 10 ms x86 LAPIC tick).
44const PENDING_QUERIES: u32 = 20;
45const PENDING_WAIT_TICKS: usize = 5;
46
47fn body_digest(body: &[u8]) -> u64 {
48    let mut hash = 0xcbf2_9ce4_8422_2325u64;
49    for byte in body { hash ^= *byte as u64; hash = hash.wrapping_mul(0x0000_0100_0000_01b3); }
50    hash
51}

53enum Answer { Outcome(ResultHeaderV1), Unknown, Pending, Failed(Status) }
54
55/// Ask the owner for the retained outcome of `id`. A `WouldBlock` answer carrying the
56/// query's own selector means the original is still pending: re-query with a bounded
57/// delay rather than treating it as corruption.
58fn query_outcome<F>(call: &mut F, query_selector: u16, selector: u16, operation: ObjectRefV1, id: u64, reply: &mut [u8]) -> Answer
59where F: FnMut(u16,u64,&[u8],&mut[u8])->Result<ResultHeaderV1,Status> {
60    let mut query = [0u8;24]; query[..16].copy_from_slice(&operation.to_bytes()); query[16..].copy_from_slice(&id.to_le_bytes());
61    for attempt in 0..=PENDING_QUERIES {
62        if attempt != 0 { let _ = crate::syscall::sys_set_timer(PENDING_WAIT_TICKS); }
63        match call(query_selector, id, &query, reply) {
64            Ok(h) if h.selector == selector && h.request_id == id => return Answer::Outcome(h),
65            Ok(h) if h.status == Status::UnknownRequest => return Answer::Unknown,
66            Ok(h) if h.status == Status::WouldBlock && h.selector == query_selector => {}
67            Ok(_) => return Answer::Failed(Status::Indeterminate),
68            // Queries are read-only: a transient transport failure is retried too.
69            Err(Status::WouldBlock | Status::TimedOut | Status::Indeterminate) => {}
70            Err(e) => return Answer::Failed(e),
71        }
72    }
73    Answer::Pending
74}

76fn correlated<F>(issuer: u16, incarnation: Option<ServiceBinding>, operation: ObjectRefV1, selector: u16, mutating: bool, query_selector: u16, ack_selector: u16, body: &[u8], reply: &mut [u8], mut call: F) -> Result<ResultHeaderV1, Status>
77where F: FnMut(u16,u64,&[u8],&mut[u8])->Result<ResultHeaderV1,Status> {
78    let mut lanes = LANES.lock();
79    let index = match lanes.iter().position(|l| l.issuer == issuer && l.incarnation == incarnation && l.operation == operation) {
80        Some(i) => i,
81        None => {
82            if lanes.len() == 192 { return Err(Status::QuotaExceeded); }
83            lanes.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
84            lanes.push(Lane { issuer, incarnation, operation, next: 1, blocked: false, pending: None });
85            lanes.len()-1
86        }
87    };
88    let lane = &mut lanes[index];
89    if lane.blocked { return Err(Status::Indeterminate); }
90    let digest = body_digest(body);
91    let mut resumed = None;
92    if let Some(p) = lane.pending {
93        // Resolve the earlier uncertain mutation by query before anything else.
94        match query_outcome(&mut call, query_selector, p.selector, operation, lane.next, reply) {
95            Answer::Outcome(h) => {
96                lane.pending = None;
97                if mutating && p.selector == selector && p.digest == digest {
98                    // The caller is retrying the identical operation: hand back the
99                    // original retained outcome; the request is never re-sent.
100                    resumed = Some(h);
101                } else {
102                    // A different operation: retire the observed one, then proceed.
103                    let mut ack = [0u8;24]; ack[..16].copy_from_slice(&operation.to_bytes()); ack[16..].copy_from_slice(&lane.next.to_le_bytes());
104                    lane.next += 1;
105                    let _ = call(ack_selector, lane.next - 1, &ack, &mut [0u8;HEADER_LEN]);
106                }
107            }
108            // Still pending (the lane keeps the in-flight record and is not blocked).
109            Answer::Pending => return Err(Status::Indeterminate),
110            Answer::Unknown => { lane.blocked = true; return Err(Status::Indeterminate); }
111            Answer::Failed(e) => { lane.blocked = true; return Err(e); }
112        }
113    }
114    let id = lane.next;
115    if id == u64::MAX { return Err(Status::QuotaExceeded); }
116    let header = if let Some(h) = resumed { h } else {
117        let result = call(selector,id,body,reply);
118        match result {
119            Ok(h) => h,
120            Err(Status::Indeterminate) if mutating => {
121                match query_outcome(&mut call, query_selector, selector, operation, id, reply) {
122                    Answer::Outcome(h) => h,
123                    Answer::Pending => {
124                        lane.pending = Some(Pending { selector, digest });
125                        return Err(Status::Indeterminate);
126                    }
127                    Answer::Unknown => { lane.blocked = true; return Err(Status::Indeterminate); }
128                    Answer::Failed(e) => { lane.blocked = true; return Err(e); }
129                }
130            }
131            Err(e) => {
132                // Definite pre-dispatch rejection has no retained outcome or fence.
133                if mutating && !matches!(e,Status::WouldBlock|Status::TimedOut|Status::Malformed|Status::Stale|Status::NotFound) {lane.blocked=true;}
134                return Err(e);
135            }
136        }
137    };
138    if header.selector != selector || header.request_id != id { lane.blocked = true; return Err(Status::Malformed); }
139    if mutating && header.status != Status::BufferTooSmall {
140        if header.status!=Status::Ok{
141            // Admission errors can precede outcome reservation. Ask the owner
142            // whether this ID crossed its fence rather than guessing from an
143            // error code (QuotaExceeded can occur on either side).
144            let mut retained=[0u8;HEADER_LEN];
145            match query_outcome(&mut call, query_selector, selector, operation, id, &mut retained) {
146                Answer::Outcome(_) => {}
147                Answer::Unknown => return Ok(header),
148                Answer::Pending => {
149                    lane.pending = Some(Pending { selector, digest });
150                    return Err(Status::Indeterminate);
151                }
152                Answer::Failed(_) => { lane.blocked = true; return Err(Status::Indeterminate); }
153            }
154        }
155        lane.next += 1;
156        // A complete observed outcome can now be retired. ACK failure does not
157        // turn the already observed operation into an uncertain one.
158        let mut ack = [0u8;24]; ack[..16].copy_from_slice(&operation.to_bytes()); ack[16..].copy_from_slice(&id.to_le_bytes());
159        let mut ack_reply = [0u8;HEADER_LEN];
160        let _ = call(ack_selector,id,&ack,&mut ack_reply);
161    }
162    Ok(header)
163}

165/// Operation-scoped kernel request. No accepted mutation is automatically replayed.
166pub fn kernel_call(op: KernelOp, operation: ObjectRefV1, mutating: bool, body: &[u8], reply: &mut [u8]) -> Result<ResultHeaderV1,Status> {
167    let outcome=correlated(0,None,operation,op as u16,mutating,KernelOp::QueryRequest as u16,KernelOp::AcknowledgeRequests as u16,body,reply, |selector,id,body,reply| {
168        let opcode = KernelOp::try_from(selector).map_err(|_| Status::UnknownOperation)?;
169        if opcode == KernelOp::QueryRequest {
170            // Query returns the original header, not a nested query result.
171            return kernel_query(id,body,reply);
172        }
173        kernel_request(opcode,id,body,reply)
174    });
175    if op==KernelOp::CloseOperation&&outcome.as_ref().is_ok_and(|h|h.status==Status::Ok){retire_operation(operation);}
176    outcome
177}
178fn kernel_query(id:u64, body:&[u8], reply:&mut[u8])->Result<ResultHeaderV1,Status> {
179    // The raw helper enforces echo for normal requests; query's successful record
180    // intentionally echoes the *original* selector instead.
181    let mut request=[0u8;48];
182    RequestHeaderV1{selector:KernelOp::QueryRequest as u16,total_len:48,request_id:id}.encode(&mut request[..24])?;
183    if body.len()!=24 {return Err(Status::Malformed)} request[24..].copy_from_slice(body);
184    // The kernel reserves its maximum retained-record size before executing a
185    // query, so the scratch must be at least that large (kernel RESULT_MAX).
186    let mut scratch=Scrub::<768>::new();
187    if !matches!(crate::syscall::sys_cell_session_v1(&request,&mut scratch),crate::syscall::SyscallResult::Ok(0)) { return Err(Status::Indeterminate); }
188    let n=u32::from_le_bytes(scratch[8..12].try_into().map_err(|_|Status::Malformed)?) as usize;
189    if n>scratch.len()||n<HEADER_LEN{return Err(Status::Malformed)}
190    let header=ResultHeaderV1::decode(&scratch[..n])?;
191    if n<=reply.len(){reply[..n].copy_from_slice(&scratch[..n]);return Ok(header)}
192    // A header-only probe (the post-status check) needs only the outcome header.
193    if reply.len()==HEADER_LEN{reply.copy_from_slice(&scratch[..HEADER_LEN]);return Ok(header)}
194    Err(Status::BufferTooSmall)
195}

197/// One correlated endpoint mutation; callers may advance only by acknowledged
198/// positive progress. The deadline is absolute and reused across partial writes.
199pub(crate) fn write_stream(operation: ObjectRefV1, endpoint: ObjectRefV1, bytes: &[u8], deadline_ns: u64) -> Result<usize, Status> {
200    write_stream_with(operation, endpoint, bytes, deadline_ns, kernel_call)
201}
202
203fn write_stream_with(
204    operation: ObjectRefV1,
205    endpoint: ObjectRefV1,
206    bytes: &[u8],
207    deadline_ns: u64,
208    call: impl FnOnce(KernelOp, ObjectRefV1, bool, &[u8], &mut [u8]) -> Result<ResultHeaderV1, Status>,
209) -> Result<usize, Status> {
210    use api::cell_session::{Reader, Writer, TRANSFER_MAX};
211
212    if bytes.is_empty() || bytes.len() > TRANSFER_MAX || deadline_ns == 0 || deadline_ns == u64::MAX {
213        return Err(Status::Malformed);
214    }
215    let mut body = Scrub::<{ 32 + TRANSFER_MAX }>::new();
216    let mut writer = Writer::new(&mut body);
217    writer.reference(endpoint)?;
218    writer.u32(bytes.len() as u32)?;
219    writer.u32(0)?;
220    writer.u64(deadline_ns)?;
221    writer.bytes(bytes)?;
222    let length = writer.position();
223    let mut reply = [0u8; HEADER_LEN + 8];
224    let header = call(KernelOp::WriteStream, operation, true, &body[..length], &mut reply)?;
225    if header.status != Status::Ok {
226        return Err(header.status);
227    }
228    if header.total_len as usize != reply.len() {
229        return Err(Status::Malformed);
230    }
231    let mut reader = Reader::new(&reply[HEADER_LEN..]);
232    let actual = reader.u32()? as usize;
233    if reader.u32()? != 0 || actual == 0 || actual > bytes.len() {
234        return Err(Status::Malformed);
235    }
236    reader.finish()?;
237    Ok(actual)
238}

245/// Borrowed, exact-length authenticated SessionV1 / PreparedChildV1 RPC.
246/// `service` is restricted to the two admitted issuers. Each IPC wait is <=100 ticks.
247pub fn service_call(binding:ServiceBinding, service:u16, operation:ObjectRefV1, selector:u16, mutating:bool, body:&[u8], reply:&mut[u8])->Result<ResultHeaderV1,Status> {
248    let (query,ack)=if service==api::syscall::service::INPUT {(19,18)} else if service==api::syscall::service::VFS {(5,4)} else {return Err(Status::Denied)};
249    correlated(service,Some(binding),operation,selector,mutating,query,ack,body,reply,|selector,id,body,reply| {
250        if service_binding(service)? != binding {return Err(Status::Stale)}
251        let n=HEADER_LEN.checked_add(body.len()).ok_or(Status::Malformed)?;
252        if n>MAX_SERVICE_RECORD_LEN {return Err(Status::Malformed)}
253        let mut record=Scrub::<MAX_SERVICE_RECORD_LEN>::new();
254        RequestHeaderV1{selector,total_len:n as u32,request_id:id}.encode(&mut record[..HEADER_LEN])?;
255        record[HEADER_LEN..n].copy_from_slice(body);
256        let mut send=Scrub::<{api::ipc::IPC_BUF_SIZE}>::new(); let mut recv=Scrub::<{api::ipc::IPC_BUF_SIZE}>::new();
257        let inner = if service==api::syscall::service::INPUT {
258            let req=api::ipc::InputRequest::SessionV1(&record[..n]);
259            let response=crate::ipc::service_call_typed_bounded::<_,api::ipc::InputResponse<'_>>(binding.tid as usize,&req,&mut send,&mut recv,100).map_err(ipc_status)?;
260            match response { api::ipc::InputResponse::SessionV1(bytes)=>bytes,_=>return Err(Status::Malformed) }
261        } else {
262            let req=api::ipc::VfsRequest::PreparedChildV1(&record[..n]);
263            let response=crate::ipc::service_call_typed_bounded::<_,api::ipc::VfsResponse<'_>>(binding.tid as usize,&req,&mut send,&mut recv,100).map_err(ipc_status)?;
264            match response { api::ipc::VfsResponse::PreparedChildV1(bytes)=>bytes,_=>return Err(Status::Malformed) }
265        };
266        if service_binding(service)? != binding {return Err(Status::Stale)}
267        if inner.len()>reply.len(){return Err(Status::BufferTooSmall)}
268        let h=ResultHeaderV1::decode(inner)?;
269        if selector!=query && (h.selector!=selector || h.request_id!=id) {return Err(Status::Malformed)}
270        reply[..inner.len()].copy_from_slice(inner); Ok(h)
271    })
272}

274/// VFS → input: register one first-trust descriptor (`InputOp::RegisterTrustChallenge`);
275/// returns the input-issued confirmation Ref the confirming client presents to
276/// `BeginTrustConfirmation`. Input admits only the live registered VFS task. The request
277/// lane is the single-use challenge Ref; it is kept only while the outcome is
278/// indeterminate (retry the identical descriptor to resolve it).
279pub fn register_trust_challenge(descriptor:&api::cell_session::trust::TrustChallengeV1<'_>)->Result<ObjectRefV1,Status> {
280    use api::cell_session::trust::TrustChallengeV1;
281    let owner=service_binding(api::syscall::service::INPUT)?;
282    let mut body=[0u8;TrustChallengeV1::MAX_LEN]; let len=descriptor.encode(&mut body)?;
283    let mut reply=[0u8;HEADER_LEN+16];
284    let result=service_call(owner,api::syscall::service::INPUT,descriptor.challenge,api::cell_session::InputOp::RegisterTrustChallenge as u16,true,&body[..len],&mut reply);
285    if result!=Err(Status::Indeterminate) { retire_operation(descriptor.challenge); }
286    let h=result?;
287    if h.status!=Status::Ok { return Err(h.status); }
288    ObjectRefV1::from_bytes(&reply[HEADER_LEN..h.total_len as usize])
289}

291/// VFS → input: the decision of `confirmation` (`InputOp::CollectTrustDecision`), answered
292/// only to the registering principal. `DECISION_PENDING` is not consumed; a final decision
293/// is returned once (a later collect is `Stale`), and its lane (the confirmation Ref) is
294/// then retired. Compare every binding, including `descriptor_sha256` against
295/// [`trust_descriptor_sha256`] of the own record, before acting on it.
296pub fn collect_trust_decision(confirmation:ObjectRefV1)->Result<api::cell_session::trust::TrustDecisionV1,Status> {
297    use api::cell_session::trust::TrustDecisionV1;
298    let owner=service_binding(api::syscall::service::INPUT)?;
299    let mut reply=[0u8;HEADER_LEN+TrustDecisionV1::LEN];
300    let result=service_call(owner,api::syscall::service::INPUT,confirmation,api::cell_session::InputOp::CollectTrustDecision as u16,true,&confirmation.to_bytes(),&mut reply);
301    let decision=match result {
302        Ok(h) if h.status==Status::Ok => TrustDecisionV1::decode(&reply[HEADER_LEN..h.total_len as usize]),
303        Ok(h) => Err(h.status),
304        Err(error) => Err(error),
305    };
306    if decision.as_ref().map_or_else(|e| *e!=Status::Indeterminate, TrustDecisionV1::is_final) { retire_operation(confirmation); }
307    decision
308}

310/// SHA-256 of a descriptor's canonical bytes, as `TrustDecisionV1::descriptor_sha256` binds it.
311pub fn trust_descriptor_sha256(descriptor:&api::cell_session::trust::TrustChallengeV1<'_>)->Result<[u8;32],Status> {
312    let mut body=[0u8;api::cell_session::trust::TrustChallengeV1::MAX_LEN]; let len=descriptor.encode(&mut body)?;

315fn ipc_status(error:crate::ipc::IpcError)->Status {
316    use crate::ipc::IpcError;
317    match error {
318        IpcError::Busy=>Status::WouldBlock, IpcError::PreDispatchTimeout=>Status::TimedOut,
319        IpcError::Encode=>Status::Malformed,
320        IpcError::Decode|IpcError::BufferTooSmall=>Status::Indeterminate,
321        IpcError::Cancelled=>Status::Cancelled,
322        _=>Status::Indeterminate,
323    }
324}
325/// Receiver-monotonic nanoseconds, checked rather than architecture-guessed.
326pub fn monotonic_ns()->Result<u64,Status> {
327    let frequency=crate::syscall::sys_get_timer_freq().ok_or(Status::Unsupported)?;
328    let ticks=crate::syscall::sys_get_time();
329    let ns=(ticks as u128).checked_mul(1_000_000_000).ok_or(Status::Malformed)?/(frequency as u128);
330    u64::try_from(ns).map_err(|_|Status::Malformed)
331}

333/// Overwrite `bytes` with zeros using volatile stores, so the optimizer cannot treat the
334/// write as a dead store before the memory is released or reused. Secret-bearing buffers
335/// (typed text, raw stdin bytes, paste staging, retained request bodies) are wiped with
336/// this, never with a plain `fill(0)`/`clear()`.
337pub fn zeroize(bytes:&mut [u8]) { zeroize_raw(bytes.as_mut_ptr(),bytes.len()); }
338
339fn zeroize_raw(pointer:*mut u8, length:usize) {
340    for offset in 0..length {
341        // SAFETY: callers pass a pointer valid for `length` exclusive byte writes.
342        unsafe { core::ptr::write_volatile(pointer.add(offset),0) };
343    }
344    core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::SeqCst);
345}

347/// Zero the vector's whole allocation (including spare capacity that earlier
348/// contents may have occupied), then truncate it. The allocation is retained.
349pub fn zeroize_vec(bytes:&mut alloc::vec::Vec<u8>) {
350    zeroize_raw(bytes.as_mut_ptr(),bytes.capacity());
351    bytes.clear();
352}

354/// Zero a string's whole allocation, then truncate it (still valid UTF-8: empty).
355pub fn zeroize_string(text:&mut alloc::string::String) {
356    // SAFETY: the vector is cleared before the borrow ends, so the string stays valid UTF-8.
357    zeroize_vec(unsafe { text.as_mut_vec() });
358}

360/// Zero every byte of a ring (both halves), then empty it.
361pub fn zeroize_deque(ring:&mut alloc::collections::VecDeque<u8>) {
362    let (first,second)=ring.as_mut_slices();
363    zeroize(first); zeroize(second);
364    ring.clear();
365}

367/// Zero and remove the oldest `count` bytes of a ring, so a consumed prefix never
368/// lingers in a vacated slot.
369pub fn zeroize_deque_front(ring:&mut alloc::collections::VecDeque<u8>, count:usize) {
370    let count=count.min(ring.len());
371    let (first,second)=ring.as_mut_slices();
372    let head=count.min(first.len());
373    zeroize(&mut first[..head]); zeroize(&mut second[..count-head]);
374    ring.drain(..count);
375}

377/// Fixed-size stack buffer that is wiped (volatile) when it goes out of scope, on every
378/// exit path including `?` returns. Dereferences to a byte slice.
379pub struct Scrub<const N:usize>([u8;N]);
380impl<const N:usize> Scrub<N> { pub const fn new()->Self { Self([0;N]) } }
381impl<const N:usize> Default for Scrub<N> { fn default()->Self { Self::new() } }
382impl<const N:usize> core::ops::Deref for Scrub<N> { type Target=[u8]; fn deref(&self)->&[u8] { &self.0 } }
383impl<const N:usize> core::ops::DerefMut for Scrub<N> { fn deref_mut(&mut self)->&mut [u8] { &mut self.0 } }
384impl<const N:usize> Drop for Scrub<N> { fn drop(&mut self) { zeroize(&mut self.0); } }

399    let mut state:[u32;8]=[0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19];
400    fn compress(state:&mut [u32;8], block:&[u8])  {
401        let mut w=[0u32;64];
402        for (i,word) in block.chunks_exact(4).enumerate() { w[i]=u32::from_be_bytes([word[0],word[1],word[2],word[3]]); }
403        for i in 16..64 {
404            let s0=w[i-15].rotate_right(7)^w[i-15].rotate_right(18)^(w[i-15]>>3);
405            let s1=w[i-2].rotate_right(17)^w[i-2].rotate_right(19)^(w[i-2]>>10);
406            w[i]=w[i-16].wrapping_add(s0).wrapping_add(w[i-7]).wrapping_add(s1);
407        }
408        let [mut a,mut b,mut c,mut d,mut e,mut f,mut g,mut h]=*state;
409        for i in 0..64 {
410            let s1=e.rotate_right(6)^e.rotate_right(11)^e.rotate_right(25);
411            let ch=(e&f)^(!e&g);
412            let t1=h.wrapping_add(s1).wrapping_add(ch).wrapping_add(K[i]).wrapping_add(w[i]);
413            let s0=a.rotate_right(2)^a.rotate_right(13)^a.rotate_right(22);
414            let maj=(a&b)^(a&c)^(b&c);
415            let t2=s0.wrapping_add(maj);
416            h=g; g=f; f=e; e=d.wrapping_add(t1); d=c; c=b; b=a; a=t1.wrapping_add(t2);
417        }
418        for (slot,value) in state.iter_mut().zip([a,b,c,d,e,f,g,h]) { *slot=slot.wrapping_add(value); }
419    }
420    let mut blocks=data.chunks_exact(64);
421    for block in &mut blocks { compress(&mut state,block); }
422    // Final one or two blocks hold the remainder, 0x80, zero padding and the bit length.
423    let remainder=blocks.remainder();
424    let mut tail=Scrub::<128>::new();
425    tail[..remainder.len()].copy_from_slice(remainder);
426    tail[remainder.len()]=0x80;
427    let total=if remainder.len()<56 {64} else {128};
428    tail[total-8..total].copy_from_slice(&((data.len() as u64).wrapping_mul(8)).to_be_bytes());
429    for block in tail[..total].chunks_exact(64) { compress(&mut state,block); }
430    let mut digest=[0u8;32];
431    for (chunk,word) in digest.chunks_exact_mut(4).zip(state) { chunk.copy_from_slice(&word.to_be_bytes()); }
432    digest
433}
