1//! VMM syscall wrappers (220-227) for cells holding `HypervisorCap`.
2//!
3//! One safe function per kernel VMM call, shared by the silo and hypervisor
4//! services and the hypervisor ABI smoke test. Every function returns the raw
5//! kernel result register: an id, a byte count or 0 on success, and a value
6//! the caller compares against its documented error sentinels (0 and/or
7//! `usize::MAX`) on failure. The kernel gates every call on `HypervisorCap`
8//! and returns an error to a cell without it.
9//!
10//! Guest-memory invariant behind every call: guest RAM is a run of frames the
11//! kernel carves from its own frame allocator at `CreateVm`
12//! (`Stage2Table::carve_guest_ram`), and stage 2 maps only that carve. No byte of
13//! the calling cell's memory is guest-visible, and the cell never holds a
14//! pointer into guest RAM — it moves bytes only through kernel copies that
15//! bounds-check the cell side (`validate_user_buf` + the caller copy view) and
16//! the guest side (`check_guest_range`). The only cell memory the kernel writes
17//! is the out-parameter of the call in progress, which the wrapper borrows
18//! exclusively.
19#![allow(unsafe_code)]
20

26/// Scheduler budget the VMM services pass to each [`run_vcpu`] call (~10 ms).
27pub const SCHED_TICK_BUDGET_NS: u64 = 10_000_000;

29/// Issue VMM syscall `id`, returning the raw result register unchanged.
30///
31/// # Safety
32/// Same contract as [`crate::syscall::syscall`]: every argument `id` reads as a
33/// cell address must be valid for the kernel's access over the length that
34/// call uses, and only the caller's exclusive borrow may observe bytes the
35/// kernel writes.
36#[inline(always)]
37unsafe fn vmm_call(id: ViSyscall, a0: usize, a1: usize, a2: usize, a3: usize) -> usize {
38    // The kernel result register is a machine word; the VMM callers compare
39    // its bit pattern against `usize` sentinels, so reinterpret, not convert.
40    // SAFETY: forwarded verbatim from this function's `# Safety` contract.
41    unsafe { syscall(id, a0, a1, a2, a3) as usize }
42}

44/// Allocate guest RAM + Stage-2 table; returns vm_id, or 0 / `usize::MAX` on error.
45pub fn create_vm(guest_pages: usize) -> usize {
46    // SAFETY: register-only. The kernel carves the guest RAM from its own frame
47    // allocator; it reads and writes no memory of this cell.
48    unsafe { vmm_call(ViSyscall::CreateVm, guest_pages, 0, 0, 0) }
49}
50
51/// Create a vCPU with initial PC `entry_pc` in `vm_id`; returns vcpu_id, or 0 /
52/// `usize::MAX` on error.
53pub fn create_vcpu(vm_id: usize, entry_pc: u64) -> usize {
54    // SAFETY: register-only. `entry_pc` is a guest IPA, meaningful only through
55    // the VM's stage-2 table; no cell memory is accessed.
56    unsafe { vmm_call(ViSyscall::CreateVcpu, vm_id, entry_pc as usize, 0, 0) }
57}
58
59/// Map guest IPA range in `vm_id`; returns 0 on success.
60pub fn map_guest_memory(vm_id: usize, ipa: u64, size: usize, writable: bool) -> usize {
61    // SAFETY: register-only. The kernel rejects `ipa + size` overflow and any
62    // `size` larger than the VM's carve, and backs the range with carve frames
63    // only, so stage 2 can never expose this cell's memory to the guest.
64    unsafe {
65        vmm_call(
66            ViSyscall::MapGuestMemory,
67            vm_id,
68            ipa as usize,
69            size,
70            writable as usize,
71        )
72    }
73}

75/// Copy `src` bytes into guest RAM at `gpa`; returns bytes written or an error.
76pub fn write_guest_memory(vm_id: usize, gpa: u64, src: &[u8]) -> usize {
77    // SAFETY: `src` is a live shared borrow of exactly `src.len()` bytes for the
78    // whole call. The kernel only reads it (through the caller copy view, after
79    // `validate_user_buf`) and writes nothing in this cell; the guest side is
80    // bounded to the carve by `check_guest_range`.
81    unsafe {
82        vmm_call(
83            ViSyscall::WriteGuestMemory,
84            vm_id,
85            gpa as usize,
86            src.as_ptr() as usize,
87            src.len(),
88        )
89    }
90}

92/// Copy `dst.len()` bytes from guest RAM at `gpa` into `dst`; returns bytes read
93/// or an error.
94pub fn read_guest_memory(vm_id: usize, gpa: u64, dst: &mut [u8]) -> usize {
95    // SAFETY: `dst` is exclusively borrowed for the whole call, so no other
96    // reference observes the kernel's writes; the kernel writes at most
97    // `dst.len()` bytes through the caller copy view after `validate_user_buf`
98    // and reads only carve bytes `check_guest_range` admitted. Every byte value
99    // is a valid `u8`.
100    unsafe {
101        vmm_call(
102            ViSyscall::ReadGuestMemory,
103            vm_id,
104            gpa as usize,
105            dst.as_mut_ptr() as usize,
106            dst.len(),
107        )
108    }
109}

111/// World-switch into `vcpu_id` for at most `budget_ns` (0 = no budget); writes
112/// the exit reason to `*exit`. Returns 0 or an error.
113pub fn run_vcpu(vm_id: usize, vcpu_id: usize, budget_ns: u64, exit: &mut ViVmExit) -> usize {
114    // SAFETY: `exit` is exclusively borrowed for the whole call. The kernel
115    // writes exactly `size_of::<ViVmExit>()` bytes (80, pinned by const asserts
116    // in `abi::hypervisor` and the kernel's `vmm_wire`): first all zeros — tag 0
117    // with zero payload, a valid `MmioRead` — then `encode_exit`'s image, whose
118    // tag is a declared discriminant, whose `bool`s are 0/1 and whose padding is
119    // zero. The guest runs on stage-2 mappings of its carve only, so the world
120    // switch cannot reach this cell's memory.
121    unsafe {
122        vmm_call(
123            ViSyscall::RunVcpu,
124            vm_id,
125            vcpu_id,
126            budget_ns as usize,
127            exit as *mut ViVmExit as usize,
128        )
129    }
130}

132/// Read (`write == false`, mode 0) or write (`write == true`, mode 1) the vCPU
133/// GP registers: 32 × u64 = x0..x30 plus the guest PC.
134pub fn vcpu_regs(vm_id: usize, vcpu_id: usize, regs: &mut [u64; 32], write: bool) -> usize {
135    // SAFETY: `regs` is exclusively borrowed and is exactly the kernel's
136    // `REGS_LEN` (32 × u64 = 256 bytes). Every mode reads that buffer through
137    // the caller copy view after `validate_user_buf`; mode 0 also writes it
138    // back. Every bit pattern is a valid `u64`.
139    unsafe {
140        vmm_call(
141            ViSyscall::VcpuRegs,
142            vm_id,
143            vcpu_id,
144            regs.as_mut_ptr() as usize,
145            write as usize,
146        )
147    }
148}

150/// VcpuRegs mode 2: read the guest's saved CNTV_CTL/CNTV_CVAL into words 0 and
151/// 1, zeroing the remaining words. Only AArch64 kernels built for the Pi
152/// monitor accept mode 2; every other kernel rejects it as invalid input.
153#[cfg(target_arch = "aarch64")]
154pub fn guest_timer_regs(vm_id: usize, vcpu_id: usize, regs: &mut [u64; 32]) -> usize {
155    // SAFETY: as `vcpu_regs` mode 0 — the exclusively borrowed 256-byte `regs`
156    // is read and then written back whole through the caller copy view.
157    unsafe {
158        vmm_call(
159            ViSyscall::VcpuRegs,
160            vm_id,
161            vcpu_id,
162            regs.as_mut_ptr() as usize,
163            2,
164        )
165    }
166}

168/// Raise interrupt `intid` on `vcpu_id` of `vm_id` through the kernel's
169/// virtual interrupt controller; returns 0 or an error.
170pub fn inject_irq(vm_id: usize, vcpu_id: usize, intid: u32) -> usize {
171    // SAFETY: register-only; the kernel bounds `intid` and touches no cell
172    // memory.
173    unsafe { vmm_call(ViSyscall::InjectIrq, vm_id, vcpu_id, intid as usize, 0) }
174}

176/// The host's time-stamp counter, which an x86 guest reads unchanged: guest
177/// RDTSC does not exit and both VMM backends run with a TSC offset of 0.
178#[cfg(target_arch = "x86_64")]
179pub fn read_tsc() -> u64 {
180    // SAFETY: RDTSC reads a counter register only; it touches no memory and
181    // the kernel leaves CR4.TSD clear, so it does not fault at CPL 3.
182    unsafe { core::arch::x86_64::_rdtsc() }
