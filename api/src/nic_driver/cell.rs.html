13/// Bounded wait for the Platform Cell / kernel ECAM scan to publish the device
14/// table. The cell names its controller exactly, so this is not a sibling
15/// race: a registration delay fails closed as "not present".
16pub const FIND_ATTEMPTS: usize = 200;
17
18/// Largest frame any NIC cell moves (one 2 KiB receive buffer).
19pub const FRAME_MAX: usize = 2048;
20
21/// Reply buffer: 2-byte length header plus the largest frame.
22const REPLY_CAP: usize = 2 + FRAME_MAX;
23
24/// The cell's device, published once bring-up succeeds.
25pub struct State<D>(Mutex<Option<D>>);
26
27impl<D> State<D> {
28    pub const fn new() -> Self {
29        Self(Mutex::new(None))
30    }
31}
32
33impl<D> Default for State<D> {

35        Self::new()
36    }
37}

39/// The NIC cell's event handler (expanded by [`crate::cell!`]).
40pub fn on_event<D: NicDevice>(state: &State<D>, event: AppEvent) {
41    match event {
42        AppEvent::Init => bring_up(state),
43        // The net service speaks the raw NIC wire (no 0xAC App-SDK envelope),
44        // so requests arrive as RawMessage; Message carries the same payload.
45        AppEvent::Message { sender_tid, data } | AppEvent::RawMessage { sender_tid, data } => {
46            serve(state, sender_tid, &data)
47        }
48        AppEvent::Shutdown | AppEvent::ShutdownWith { .. } => sys_exit(0),
49        _ => {}
50    }
51}

53fn find<D: NicDevice>() -> Option<PcieDeviceInfo> {
54    let mut info = PcieDeviceInfo::zeroed();
55    for _ in 0..FIND_ATTEMPTS {
56        for &device_id in D::QUERY_IDS {
57            // `Ok(false)` is "absent", not found.
58            if let Ok(true) = sys_find_pcie_device_by_vendor(D::VENDOR, device_id, &mut info) {
59                return Some(info);
60            }
61        }
62        ostd::task::yield_now();
63    }
64    None
65}

67fn bring_up<D: NicDevice>(state: &State<D>) {
68    let Some(info) = find::<D>() else {
69        if let Some(line) = D::ABSENT_LINE {
70            let _ = print_fmt(format_args!("{} {}\n", D::TAG, line));
71        }
72        sys_exit(0);
73    };
74
75    // Defense in depth behind the kernel's family gate: the register model
76    // is never programmed on an unclassified device.
77    let (vendor, device) = (info.vendor_id, info.device_id);
78    let Some(identity) = D::classify(vendor, device) else {
79        let _ = print_fmt(format_args!(
80            "{} unsupported Ethernet {:04x}:{:04x}; driver gate closed\n",
81            D::TAG,
82            vendor,
83            device
84        ));
85        sys_exit(0);
86    };
87
88    let probe = Probe::new(D::TAG, info);
89    let dev = match D::init(&probe, identity) {
90        Ok(dev) => dev,
91        Err(error) => {
92            let _ = print_fmt(format_args!(
93                "{} controller initialization failed: {:?}\n",
94                D::TAG,
95                error
96            ));
97            sys_exit(1)
98        }
99    };
100
101    // Publish before registration: the kernel may route requests as soon as
102    // the syscall succeeds.
103    let _ = print_fmt(format_args!(
104        "{} controller bound {:04x}:{:04x} {} {}\n",
105        D::TAG,
106        vendor,
107        device,
108        D::identity_name(&identity),
109        BindDetail(&dev)
110    ));
111    #[cfg(feature = "fault-inject")]
112    crate::fault_inject::bound(probe.bdf());
113    *state.0.lock() = Some(dev);
114    if let Err(error) = sys_register_nic_driver() {
115        // Dropping the device lets a chip quiesce it on the way out.
116        let dev = state.0.lock().take();
117        drop(dev);
118        let _ = print_fmt(format_args!(
119            "{} NIC driver registration failed: {:?}\n",
120            D::TAG,
121            error
122        ));
123        sys_exit(1);
124    }
125}

135/// Serve one request from `sender_tid`, replying on the operation (or with a
136/// `try_send` for a sender that did not use one).
137///
138/// Only the live net service may drive the NIC; any other cell could inject or
139/// read raw frames. `sender_tid` is kernel-stamped and the registry is asked
140/// per request, so a restarted net service is followed without a cached tid.
141/// Refusal is the ordinary failure byte the net service retries on.
142fn serve<D: NicDevice>(state: &State<D>, sender_tid: usize, data: &[u8]) {
143    let operation = ostd::ipc::current();
144    let reply = |bytes: &[u8]| {
145        if let Some(operation) = operation {
146            let _ = ostd::ipc::reply(operation, bytes);
147        } else {
148            let _ = sys_try_send(sender_tid, bytes);
149        }
150    };
151    if !sender_is_live_provider(service::NET, sender_tid) {
152        reply(&[NicStatus::Failed as u8]);
153        return;
154    }
155    let mut out = [0u8; REPLY_CAP];
156    match state.0.lock().as_mut() {
157        Some(dev) => {
158            let len = handle(dev, data, &mut out);
159            reply(&out[..len]);
160        }
161        None => reply(&[NicStatus::Failed as u8]),
162    }
163}
164
165/// Dispatch one NIC wire request; the reply is `out[..returned]`.
166///
167/// `OP_TX` is `[op, len u16 LE, frame]`, the length bounding the frame inside
168/// the (padded) IPC buffer, and replies one status byte. `OP_RX` replies
169/// `[len u16 LE, frame]` (`len == 0`: nothing ready). `OP_GETMAC` replies the
170/// six address bytes. Anything else is `Failed`.
171fn handle<D: NicDevice>(dev: &mut D, data: &[u8], out: &mut [u8; REPLY_CAP]) -> usize {
172    const { assert!(D::MAX_TX <= FRAME_MAX) };
173    let status = |out: &mut [u8; REPLY_CAP], status: NicStatus| {
174        out[0] = status as u8;
175        1
176    };
177    match data.first() {
178        Some(&OP_TX) => {
179            if data.len() < 3 {
180                return status(out, NicStatus::Failed);
181            }
182            let len = u16::from_le_bytes([data[1], data[2]]) as usize;
183            if len == 0 || len > D::MAX_TX || 3 + len > data.len() {
184                return status(out, NicStatus::Failed);
185            }
186            let frame = &data[3..3 + len];
187            #[cfg(feature = "fault-inject")]
188            crate::fault_inject::before_tx(D::TAG, frame);
189            match dev.send(frame) {
190                Ok(()) => status(out, NicStatus::Ok),
191                Err(_) => status(out, NicStatus::Failed),
192            }
193        }
194        Some(&OP_RX) => {
195            let n = dev.recv(&mut out[2..2 + D::MAX_TX]).min(D::MAX_TX);
196            out[..2].copy_from_slice(&(n as u16).to_le_bytes());
197            2 + n
198        }
199        Some(&OP_GETMAC) => {
200            out[..6].copy_from_slice(&dev.mac());
201            6
202        }
203        _ => status(out, NicStatus::Failed),
204    }
205}

212    /// A NIC that records transmitted frames and returns queued receives.
213    struct FakeNic {
214        sent: Vec<Vec<u8>>,
215        rx: Vec<u8>,
216        tx_ok: bool,
217    }
218
219    impl NicDevice for FakeNic {
220        const TAG: &'static str = "[fake]";
221        const VENDOR: u16 = 0;
222        const QUERY_IDS: &'static [u16] = &[];
223        const ABSENT_LINE: Option<&'static str> = None;
224        type Identity = ();
225        type Error = ();
226
227        fn classify(_: u16, _: u16) -> Option<()> {
228            None
229        }
230        fn identity_name(_: &()) -> &'static str {
231            "fake"
232        }
233        fn init(_: &Probe, _: ()) -> Result<Self, ()> {
234            Err(())
235        }
236        fn bind_detail(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
237            f.write_str("detail")
238        }
239        fn mac(&self) -> [u8; 6] {
240            [2, 0, 0, 0, 0, 1]
241        }
242        fn send(&mut self, frame: &[u8]) -> ViResult<()> {
243            self.sent.push(frame.to_vec());
244            if self.tx_ok {
245                Ok(())
246            } else {
247                Err(ViError::IO)
248            }
249        }
250        fn recv(&mut self, out: &mut [u8]) -> usize {
251            let n = self.rx.len().min(out.len());
252            out[..n].copy_from_slice(&self.rx[..n]);
253            n
254        }
255    }
256
257    fn nic() -> FakeNic {
258        FakeNic {
259            sent: Vec::new(),
260            rx: Vec::new(),
261            tx_ok: true,
262        }
263    }
264
265    fn run(dev: &mut FakeNic, data: &[u8]) -> Vec<u8> {
266        let mut out = [0u8; REPLY_CAP];
267        let len = handle(dev, data, &mut out);
268        out[..len].to_vec()
269    }
270
271    const OK: u8 = NicStatus::Ok as u8;
272    const FAILED: u8 = NicStatus::Failed as u8;
273
274    #[test]
275    fn tx_sends_exactly_the_length_prefixed_frame() {
276        let mut dev = nic();
277        // Padded IPC buffer: bytes past the length header are not sent.
278        assert_eq!(
279            run(&mut dev, &[OP_TX, 3, 0, 0xA, 0xB, 0xC, 0xFF, 0xFF]),
280            [OK]
281        );
282        assert_eq!(dev.sent, [vec![0xA, 0xB, 0xC]]);
283        dev.tx_ok = false;
284        assert_eq!(run(&mut dev, &[OP_TX, 1, 0, 7]), [FAILED]);
285    }

287    #[test]
288    fn tx_refuses_empty_short_oversized_and_truncated_frames() {
289        let mut dev = nic();
290        assert_eq!(run(&mut dev, &[OP_TX]), [FAILED], "no length");
291        assert_eq!(run(&mut dev, &[OP_TX, 0, 0]), [FAILED], "zero length");
292        assert_eq!(run(&mut dev, &[OP_TX, 4, 0, 1, 2]), [FAILED], "truncated");
293        let big = (FRAME_MAX as u16 + 1).to_le_bytes();
294        let mut data = vec![OP_TX, big[0], big[1]];
295        data.resize(3 + FRAME_MAX + 1, 0);
296        assert_eq!(run(&mut dev, &data), [FAILED], "over MAX_TX");
297        assert!(dev.sent.is_empty());
298
299        let max = (FRAME_MAX as u16).to_le_bytes();
300        let mut data = vec![OP_TX, max[0], max[1]];
301        data.resize(3 + FRAME_MAX, 0x5A);
302        assert_eq!(run(&mut dev, &data), [OK], "exactly MAX_TX");
303    }

305    #[test]
306    fn rx_replies_length_prefixed_frame_or_zero() {
307        let mut dev = nic();

309        dev.rx = (0..300u16).map(|i| i as u8).collect();

311        assert_eq!(&reply[..2], &300u16.to_le_bytes());
312        assert_eq!(&reply[2..], &dev.rx[..]);
