1//! The RFC 4252 section 7 input an `ssh-ed25519` publickey signature covers.
2//!
3//! This is the client's own construction of the bytes KMS reconstructs from
4//! `(session id, username, stored key)` (`k2-cp1.md` 1.4): `string session_id | byte 50 |
5//! string user | string "ssh-connection" | string "publickey" | boolean TRUE |
6//! string "ssh-ed25519" | string pubkey_blob`, 143 + n bytes for an n-byte user name.
7//! Sunset serialises the same message itself ([`crate::sunset::AuthSigMsg`]); the session
8//! compares the two byte for byte before anything is signed, so a disagreement between
9//! the three implementations (Sunset, this one, KMS) stops the login rather than sending
10//! a signature the server would refuse.
11use heapless::Vec;
12use sha2::{Digest, Sha256};
13
14use crate::sunset::ed25519_dalek::{Signature, VerifyingKey};
15
16/// Longest user name KMS signs for (`KMS_SSH_USERNAME_MAX`).
17pub const MAX_USERNAME: usize = 31;
18/// Longest signing input: 143 + [`MAX_USERNAME`].
19pub const MAX_INPUT: usize = 143 + MAX_USERNAME;
20
21const SERVICE: &[u8] = b"ssh-connection";
22const METHOD: &[u8] = b"publickey";
23const ALGORITHM: &[u8] = b"ssh-ed25519";
24/// `SSH_MSG_USERAUTH_REQUEST`.
25const USERAUTH_REQUEST: u8 = 50;
26
27fn string(out: &mut Vec<u8, MAX_INPUT>, bytes: &[u8]) -> Option<()> {
28    out.extend_from_slice(&(bytes.len() as u32).to_be_bytes())
29        .ok()?;
30    out.extend_from_slice(bytes).ok()
31}
32
33/// The signing input for `user` over `session_id` with the ed25519 key `public_key`, or
34/// `None` for a user name of 0 or more than [`MAX_USERNAME`] bytes.
35pub fn signing_input(
36    session_id: &[u8; 32],
37    user: &str,
38    public_key: &[u8; 32],
39) -> Option<Vec<u8, MAX_INPUT>> {
40    if user.is_empty() || user.len() > MAX_USERNAME {
41        return None;
42    }
43    let mut out = Vec::new();
44    string(&mut out, session_id)?;
45    out.push(USERAUTH_REQUEST).ok()?;
46    string(&mut out, user.as_bytes())?;
47    string(&mut out, SERVICE)?;
48    string(&mut out, METHOD)?;
49    out.push(1).ok()?;
50    string(&mut out, ALGORITHM)?;
51    // The 51-byte public key blob: string "ssh-ed25519" | string key.
52    out.extend_from_slice(&(4 + ALGORITHM.len() as u32 + 4 + 32).to_be_bytes())
53        .ok()?;
54    string(&mut out, ALGORITHM)?;
55    string(&mut out, public_key)?;
56    Some(out)
57}
58
59/// SHA-256 of a signing input (the digest KMS returns beside the signature).
60pub fn digest(message: &[u8]) -> [u8; 32] {
61    Sha256::digest(message).into()
62}
63
64/// Whether `signature` (`R || S`) is a valid ed25519 signature of `message` under
65/// `public_key`. Strict: a malleable or small-order form is refused.
66pub fn verify(public_key: &[u8; 32], message: &[u8], signature: &[u8; 64]) -> bool {
67    match VerifyingKey::from_bytes(public_key) {
68        Ok(key) => key
69            .verify_strict(message, &Signature::from_bytes(signature))
70            .is_ok(),
71        Err(_) => false,
72    }
73}
