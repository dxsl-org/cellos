1//! One connection, one session channel (command-contracts §4.2/§4.3): host trust before
2//! any credential, password or publickey authentication (exclusive: no fallback from one
3//! to the other), then a PTY shell or one exec command.
4//!
5//! S9-6: the rekey policy is the embedder's ([`RekeyPolicy`]); every KEX (the first and
6//! each rekey, from either side) must finish within its deadline; a rekey's host key must
7//! be exactly the one accepted at the first KEX and is never offered to the trust store
8//! again; the session's lifecycle is reported to [`Local::progress`] write-ahead, so a
9//! request frame is never handed to the transport before `MayHaveRun` is recorded.

11use alloc::vec::Vec;

21/// D03 trust lookup result (abi-vfs-telemetry §4), and the end of a first-trust
22/// confirmation.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub enum Trust {
25    KnownExact,
26    Unknown,
27    Mismatch,
28    Denied,
29    Unavailable,
30    /// The user refused the unknown key (or the confirmation was cancelled).
31    Refused,
32    /// The confirmation's time ran out.
33    Expired,
34    /// This terminal cannot run a trusted confirmation (no interactive frontend).
35    NoConfirmation,
36}

38/// What the session runs.
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub enum Mode {
41    /// PTY shell with `TERM` and the terminal size at start.
42    Pty { term: &'static str, rows: u16, cols: u16 },
43    /// One non-PTY command string, sent byte-for-byte.
44    Exec(String),
45}

47/// How the session authenticates. Exactly one method is tried, once.
48#[derive(Debug, Clone, Copy, PartialEq, Eq)]
49pub enum AuthKind {
50    /// Ask [`Local::password`] once.
51    Password,
52    /// Offer [`Local::identity`] and sign through [`Local::sign`]. The password method is
53    /// never offered, not even after the server refuses the key.
54    PublicKey,
55}

57/// Why a publickey login ended before the server decided. Every one is a local refusal:
58/// nothing was signed for the server (or the signature was not sent), so the login is
59/// `NotSent` and in the `Auth` phase.
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61pub enum Credential {
62    /// The key service is not available.
63    Unavailable,
64    /// The key service refuses this caller.
65    Denied,
66    /// The key service does not hold the named identity.
67    UnknownIdentity,
68    /// The user cancelled the passphrase prompt.
69    UnlockCancelled,
70    /// The identity is passphrase-wrapped and there is no terminal to ask on.
71    UnlockNeedsTerminal,
72    /// The key service could not unlock the identity (wrong passphrase, corrupt key, ...).
73    UnlockFailed,
74    /// Too many failed unlocks: the identity is locked.
75    UnlockLocked,
76    /// The key service did not answer in time.
77    Timeout,
78    /// The key service answered something this client does not accept.
79    Protocol,
80    /// The key service signed a different input than the one Sunset asked for.
81    InputMismatch,
82    /// The key service returned a signature that does not verify.
83    BadSignature,
84}

86/// What the key service is asked to sign: the typed state of one userauth attempt. The
87/// session already compared Sunset's own serialisation with the RFC 4252 construction
88/// this client builds ([`userauth::signing_input`]) and `message_sha256` is the digest of
89/// those exact bytes.
90#[derive(Debug, Clone, PartialEq, Eq)]
91pub struct SignRequest<'a> {
92    pub session_id: [u8; 32],
93    pub username: &'a str,
94    pub message_sha256: [u8; 32],
95}

97/// The key service's answer: the raw `R || S` signature and the digest of the input it
98/// signed (which must equal [`SignRequest::message_sha256`]).
99#[derive(Debug, Clone, PartialEq, Eq)]
100pub struct Signed {
101    pub signature: [u8; 64],
102    pub message_sha256: [u8; 32],
103}

118/// Session phase (`phase` of the completion record, §4.3). `Connect` is the embedder's
119/// own phase until the host key is offered.
120#[derive(Debug, Clone, Copy, PartialEq, Eq)]
121pub enum Phase {
122    Trust,

124    Request,
125    Running,

128/// Named points of the session an embedder may observe (and, in a fault-injection
129/// build, hold): see [`Local::checkpoint`].
130#[derive(Debug, Clone, Copy, PartialEq, Eq)]
131pub enum Point {
132    /// Authenticated; the channel open and the request are not sent yet.
133    BeforeRequest,
134    /// The request frame exists (`MayHaveRun` is recorded) and is next to be written to
135    /// the transport. A hold here writes nothing: an interruption during it leaves the
136    /// request unsent while the record already says it may have run.
137    RequestSent,
138    /// The server accepted the request.
139    Accepted,
140    /// The remote exit status (or signal) arrived; channel output may still be pending.
141    ExitStatus,
142    /// A rekey (a KEX after the first) started, from either side; the peer's KEX packets
143    /// are not processed during a hold.
144    RekeyStart,
145}

147/// The rekey and KEX-deadline policy. [`RekeyPolicy::PRODUCTION`] is the contract
148/// (§4.2): rekey at the first of one hour since NEWKEYS, 1 GiB or 2^30 packets per
149/// direction; every KEX finishes within 30 seconds.
150#[derive(Debug, Clone, Copy, PartialEq, Eq)]
151pub struct RekeyPolicy {
152    pub max_age_ms: u64,
153    pub limits: RekeyLimits,
154    pub kex_ms: u64,
155}
156
157impl RekeyPolicy {
158    pub const PRODUCTION: Self = Self { max_age_ms: 3_600_000, limits: RekeyLimits::PRODUCTION, kex_ms: 30_000 };
159}

161/// The local side of the session.
162pub trait Local {
163    /// Decide on the server's host key. Only `KnownExact` lets the session continue.
164    fn trust(&mut self, key: &HostKey) -> Trust;
165    /// Called only when `trust` answered `Unknown`, before anything else is sent: run the
166    /// first-trust confirmation. `KnownExact` (the key is now stored) lets the session
167    /// continue; any other answer ends it with that answer.
168    fn confirm_unknown(&mut self, key: &HostKey) -> Trust;
169    /// Ask once for the password, after the host key was trusted. Write it into
170    /// `secret` and return true, or return false to cancel. `secret` is wiped after use.
171    fn password(&mut self, secret: &mut String) -> bool;
172    /// Publickey only, called once after the host key was trusted and the server listed
173    /// publickey: the 32-byte ed25519 public key to offer. A refusal ends the login with
174    /// that [`Credential`]. Never called in password mode.
175    fn identity(&mut self) -> Result<[u8; 32], Credential>;
176    /// Publickey only, called once after the server accepted the key (so an identity the
177    /// server does not know never costs a passphrase): produce the signature for `r`.
178    /// Never called in password mode.
179    fn sign(&mut self, r: &SignRequest<'_>) -> Result<Signed, Credential>;
180    /// Fill `buf` with local input.
181    fn input(&mut self, buf: &mut [u8]) -> Input;
182    /// Remote standard output (or, in PTY mode, the whole terminal output).
183    fn stdout(&mut self, data: &[u8]);
184    /// Remote extended data type 1 (exec mode only).
185    fn stderr(&mut self, data: &[u8]);
186    /// A new terminal size since the last call, if any (PTY mode).
187    fn resized(&mut self) -> Option<(u16, u16)>;
188    /// The session entered `phase` with `dispatch` (write-ahead: `MayHaveRun` is reported
189    /// before the request frame exists).
190    fn progress(&mut self, phase: Phase, dispatch: Dispatch);
191    /// The session reached `point`. The answer is how long to hold there, in ms, while
192    /// only the transport is serviced (no packet is processed; at
193    /// [`Point::RequestSent`] not even the request is written). Production answers 0.
194    fn checkpoint(&mut self, _point: Point) -> u64 {
195        0
196    }
197}

261/// Largest channel data read at once (§4.2: chunks of at most 16 KiB).
262pub const CHUNK: usize = 16 * 1024;

264pub fn run<T: Transport, L: Local>(t: &mut T, local: &mut L, user: &str, auth: AuthKind, mode: &Mode, policy: &RekeyPolicy) -> Outcome {
265    let mut runner: Runner<'static, Client> = Runner::new_client_owned();
266    runner.set_rekey_limits(policy.limits);
267    let mut state = State {
268        dispatch: Dispatch::NotSent, chan: None, exit: None, password_asked: false, key_offered: false, signed: false,
269        identity: None, eof_sent: false, host: None, clock: RekeyClock::default(), peaks: Peaks::default(),
270    };
271    let ending = drive(&mut runner, t, local, user, auth, mode, policy, &mut state).unwrap_or_else(|e| match e {
272        ClientError::Transport(TransportError::TimedOut) => Ending::TimedOut,
273        ClientError::Transport(_) | ClientError::Ended => Ending::TransportLost,
274        ClientError::Protocol(sunset::Error::NoAuthMethods) => Ending::Auth,
275        other => Ending::Error(alloc::format!("{other}")),
276    });
277    // An exit status that arrived before the transport ended stands; the incomplete
278    // output is reported by `stdout_complete`.
279    let ending = match (ending, state.exit.take()) {
280        (Ending::Unknown | Ending::TransportLost, Some(exit)) => exit,
281        (ending, _) => ending,
282    };
283    let complete = state.chan.as_ref().is_some_and(|c| runner.is_channel_eof(c));
284    Outcome {
285        dispatch: state.dispatch, ending, stdout_complete: complete, stderr_complete: complete,
286        rekeys: runner.newkeys_count().saturating_sub(1), peaks: state.peaks,
287    }
288}

308#[derive(Default)]
309struct RekeyClock {
310    /// When the KEX in progress was first seen.
311    kex_started: Option<u64>,
312    /// Completed key exchanges seen so far, and when the last one completed.
313    newkeys: u32,
314    newkeys_at: u64,
315}

317/// Service only the transport for `ms` (a fault-injection hold): runner output goes
318/// out and input is queued, but no packet is processed. A closed or failed transport
319/// ends the hold with its error.
320fn hold<T: Transport>(runner: &mut Runner<'static, Client>, t: &mut T, pending: &mut Vec<u8>, ms: u64) -> Result<(), ClientError> {
321    if ms == 0 {
322        return Ok(());
323    }
324    let end = t.now_ms().saturating_add(ms);
325    while t.now_ms() < end {
326        pump(runner, t, pending)?;
327        t.idle()?;
328    }
329    Ok(())
330}

332/// Rekey policy and KEX deadline, once per turn. Returns how long to hold at a rekey's
333/// start ([`Point::RekeyStart`]).
334fn rekey_turn<T: Transport, L: Local>(
335    runner: &mut Runner<'static, Client>,
336    t: &mut T,
337    local: &mut L,
338    policy: &RekeyPolicy,
339    st: &mut RekeyClock,
340) -> Result<u64, ClientError> {
341    let mut hold_ms = 0;
342    let now = t.now_ms();
343    let newkeys = runner.newkeys_count();
344    if newkeys != st.newkeys {
345        st.newkeys = newkeys;
346        st.newkeys_at = now;
347    }
348    if runner.kex_active() {
349        let started = *st.kex_started.get_or_insert_with(|| {
350            if newkeys > 0 {
351                hold_ms = local.checkpoint(Point::RekeyStart);
352            }
353            now
354        });
355        if now.saturating_sub(started) >= policy.kex_ms {
356            return Err(ClientError::Transport(TransportError::TimedOut));
357        }
358    } else {
359        st.kex_started = None;
360        if newkeys > 0 && now.saturating_sub(st.newkeys_at) >= policy.max_age_ms {
361            runner.request_rekey();
362        }
363    }
364    Ok(hold_ms)
365}

367#[allow(clippy::too_many_arguments)]
368fn drive<T: Transport, L: Local>(
369    runner: &mut Runner<'static, Client>,
370    t: &mut T,
371    local: &mut L,
372    user: &str,
373    auth: AuthKind,
374    mode: &Mode,
375    policy: &RekeyPolicy,
376    st: &mut State,
377) -> Result<Ending, ClientError> {
378    let mut pending = Vec::with_capacity(RECV_CHUNK);
379    let mut buf = alloc::vec![0u8; CHUNK];
380    // Local input not yet taken by the channel: one chunk, wiped as it is sent and on drop.
381    let mut out = zeroize::Zeroizing::new(alloc::vec![0u8; CHUNK]);
382    let mut unsent = (0usize, 0usize);
383    let mut open_session = false;
384    loop {
385        let hold_ms = rekey_turn(runner, t, local, policy, &mut st.clock)?;
386        hold(runner, t, &mut pending, hold_ms)?;
387        let mut progressed = false;
388        let mut stop = None;
389        let mut point = None;
390        match runner.progress() {
391            // Every offered method was refused (or the password was not given).
392            Err(sunset::Error::NoAuthMethods) => return Ok(Ending::Auth),
393            Err(e) => return Err(e.into()),
394            Ok(Event::Cli(CliEvent::Hostkey(h))) => {
395                let key = match h.hostkey()? {
396                    PubKey::Ed25519(k) => HostKey { algorithm: "ssh-ed25519", key: k.key.0 },
397                    #[allow(unreachable_patterns)]
398                    _ => return Ok(Ending::Error(String::from("unsupported host key type"))),
399                };
400                let decision = match &st.host {
401                    // A rekey: exactly the accepted key, never a new trust decision.
402                    Some(accepted) if *accepted == key => Trust::KnownExact,
403                    Some(_) => Trust::Mismatch,
404                    None => {
405                        local.progress(Phase::Trust, st.dispatch);
406                        match local.trust(&key) {
407                            Trust::Unknown => local.confirm_unknown(&key),
408                            other => other,
409                        }
410                    }
411                };
412                match decision {
413                    Trust::KnownExact => {
414                        h.accept()?;
415                        if st.host.is_none() {
416                            st.host = Some(key);
417                            local.progress(Phase::Auth, st.dispatch);
418                        }
419                    }
420                    other => {
421                        // Upstream reports the refusal as `BadUsage`; it is intended here.
422                        let _ = h.reject();
423                        stop = Some(Ending::HostKey(other, key));
424                    }
425                }
426            }
427            Ok(Event::Cli(CliEvent::Username(u))) => u.username(user)?,
428            Ok(Event::Cli(CliEvent::Password(p))) => {
429                if auth == AuthKind::PublicKey {
430                    // Exclusive auth: once the key is skipped or refused Sunset falls back to
431                    // the password method; that fallback is closed here.
432                    p.skip()?;
433                    stop = Some(Ending::Auth);
434                } else if st.password_asked {
435                    // Exclusive auth, no retry: a second request means the first failed.
436                    p.skip()?;
437                    stop = Some(Ending::Auth);
438                } else {
439                    st.password_asked = true;
440                    let mut secret = String::new();
441                    let given = local.password(&mut secret);
442                    let r = if given { p.password(&secret) } else { p.skip() };
443                    secret.zeroize();
444                    r?;
445                    if !given {
446                        stop = Some(Ending::Auth);
447                    }
448                }
449            }
450            Ok(Event::Cli(CliEvent::Pubkey(p))) => match auth {
451                AuthKind::Password => p.skip()?,
452                // The server refused the key or its signature: nothing else is offered.
453                AuthKind::PublicKey if st.key_offered => {
454                    let _ = p.skip();
455                    stop = Some(Ending::Auth);
456                }
457                AuthKind::PublicKey => {
458                    let offered = local.identity().and_then(|key| {
459                        VerifyingKey::from_bytes(&key).map(|verifying| (key, verifying)).map_err(|_| Credential::Protocol)
460                    });
461                    match offered {
462                        Ok((key, verifying)) => {
463                            st.key_offered = true;
464                            st.identity = Some(key);
465                            p.pubkey(SignKey::AgentEd25519(verifying))?;
466                        }
467                        Err(credential) => {
468                            let _ = p.skip();
469                            stop = Some(Ending::Credential(credential));
470                        }
471                    }
472                }
473            },
474            Ok(Event::Cli(CliEvent::AgentSign(s))) => match (auth, st.identity, st.signed) {
475                (AuthKind::PublicKey, Some(key), false) => match answer_sign(s, local, user, &key)? {
476                    None => st.signed = true,
477                    Some(credential) => stop = Some(Ending::Credential(credential)),
478                },
479                // Password mode cannot raise this; a second request means the first
480                // signature was refused.
481                _ => {
482                    let _ = s.skip();
483                    stop = Some(Ending::Auth);
484                }
485            },
486            Ok(Event::Cli(CliEvent::Authenticated)) => {
487                local.progress(Phase::Request, st.dispatch);
488                point = Some(Point::BeforeRequest);
489                open_session = true;
490            }
491            Ok(Event::Cli(CliEvent::SessionOpened(mut o))) => {
492                // Write-ahead: the request may run remotely once its frame exists.
493                st.dispatch = Dispatch::MayHaveRun;
494                local.progress(Phase::Request, st.dispatch);
495                match mode {
496                    Mode::Pty { term, rows, cols } => {
497                        let mut pty = Pty {
498                            term: heapless::String::new(),
499                            cols: u32::from(*cols),
500                            rows: u32::from(*rows),
501                            width: 0,
502                            height: 0,
503                            modes: heapless::Vec::new(),
504                        };
505                        pty.term.push_str(term).map_err(|_| ClientError::Protocol(sunset::Error::msg("TERM")))?;
506                        o.pty(pty)?;
507                        o.shell()?;
508                    }
509                    Mode::Exec(cmd) => o.exec(cmd)?,
510                }
511                point = Some(Point::RequestSent);
512            }
513            Ok(Event::Cli(CliEvent::SessionExit(x))) => {
514                if st.exit.is_some() {
515                    // Duplicate or conflicting outcomes: uncertainty, not last-wins.
516                    st.exit = None;
517                    stop = Some(Ending::Error(String::from("duplicate remote exit outcome")));
518                } else {
519                    st.exit = Some(match x {
520                        CliSessionExit::Status(n) => Ending::RemoteExit(n),
521                        CliSessionExit::Signal(s) => Ending::RemoteSignal(String::from(s.signal)),
522                    });
523                    point = Some(Point::ExitStatus);
524                }
525            }
526            Ok(Event::Cli(CliEvent::Banner(_))) => {}
527            Ok(Event::Cli(CliEvent::Defunct)) => {
528                return Ok(st.exit.take().unwrap_or(match st.dispatch {
529                    Dispatch::NotSent if !(st.password_asked || st.signed) => Ending::Error(String::from("server closed the connection")),
530                    Dispatch::NotSent => Ending::Auth,
531                    _ => Ending::Unknown,
532                }));
533            }
534            Ok(Event::None) => {}
535            Ok(_) => progressed = true,
536        }
537        if let Some(ending) = stop {
538            let _ = pump(runner, t, &mut pending);
539            return Ok(ending);
540        }
541        if let Some(point) = point {
542            if point == Point::RequestSent {
543                // The request frame is queued, not yet written: a hold here writes nothing.
544                let ms = local.checkpoint(point);
545                let end = t.now_ms().saturating_add(ms);
546                while t.now_ms() < end {
547                    t.idle()?;
548                }
549                pump(runner, t, &mut pending)?;
550            } else {
551                // Anything queued (the channel open, a reply) reaches the transport first.
552                pump(runner, t, &mut pending)?;
553                let ms = local.checkpoint(point);
554                hold(runner, t, &mut pending, ms)?;
555            }
556            progressed = true;
557        }
558        st.peaks.pending = st.peaks.pending.max(pending.capacity());
559        if open_session && st.chan.is_none() {
560            st.chan = Some(runner.open_client_session()?);
561            progressed = true;
562        }
563        if let Some(chan) = st.chan.as_ref() {
564            if st.dispatch == Dispatch::MayHaveRun {
565                match runner.channel_request_reply(chan) {
566                    Some(true) => {
567                        st.dispatch = Dispatch::Accepted;
568                        local.progress(Phase::Running, st.dispatch);
569                        pump(runner, t, &mut pending)?;
570                        let ms = local.checkpoint(Point::Accepted);
571                        hold(runner, t, &mut pending, ms)?;
572                    }
573                    Some(false) => return Ok(Ending::RequestRefused),
574                    None => {}
575                }
576            }
577            // Remote output first, so local input cannot starve it. In PTY mode extended
578            // data is part of the one terminal output stream.
579            for dt in [ChanData::Normal, ChanData::Stderr] {
580                match runner.read_channel(chan, dt, &mut buf) {
581                    Ok(0) | Err(_) => {}
582                    Ok(n) => {
583                        progressed = true;
584                        st.peaks.chunk = st.peaks.chunk.max(n);
585                        match (mode, dt) {
586                            (Mode::Exec(_), ChanData::Stderr) => local.stderr(&buf[..n]),
587                            _ => local.stdout(&buf[..n]),
588                        }
589                    }
590                }
591            }
592            if runner.is_channel_closed(chan) {
593                let _ = pump(runner, t, &mut pending);
594                return Ok(st.exit.take().unwrap_or(Ending::Unknown));
595            }
596            if st.eof_sent {
597                // Local input has ended; only a local disconnect can still come.
598                if local.input(&mut []) == Input::Disconnect {
599                    let _ = pump(runner, t, &mut pending);
600                    return Ok(Ending::Disconnected);
601                }
602            } else if st.dispatch != Dispatch::NotSent {
603                if let Some((rows, cols)) = local.resized() {
604                    if matches!(mode, Mode::Pty { .. }) {
605                        let w = WinChange { cols: u32::from(cols), rows: u32::from(rows), width: 0, height: 0 };
606                        runner.term_window_change(chan, &w)?;
607                        progressed = true;
608                    }
609                }
610                // At most one unsent chunk waits (KEX pause, window or buffer full); the
611                // next turn retries it after `progress()` had its chance to move the KEX.
612                if unsent.1 > unsent.0 {
613                    let w = runner.write_channel(chan, ChanData::Normal, &out[unsent.0..unsent.1])?;
614                    out[unsent.0..unsent.0 + w].zeroize();
615                    unsent.0 += w;
616                    progressed |= w > 0;
617                } else {
618                    let room = runner.write_channel_ready(chan, ChanData::Normal)?.unwrap_or(0).min(CHUNK);
619                    if room > 0 {
620                        match local.input(&mut out[..room]) {
621                            Input::Data(n) => {
622                                let w = runner.write_channel(chan, ChanData::Normal, &out[..n])?;
623                                out[..w].zeroize();
624                                unsent = (w, n);
625                                progressed = true;
626                            }
627                            Input::Eof => {
628                                runner.send_channel_eof(chan)?;
629                                st.eof_sent = true;
630                                progressed = true;
631                            }
632                            Input::Disconnect => {
633                                let _ = pump(runner, t, &mut pending);
634                                return Ok(Ending::Disconnected);
635                            }
636                            Input::Idle => {}
637                        }
638                    }
639                }
640            }
641        }
642        if !pump(runner, t, &mut pending)? && !progressed {
643            t.idle()?;
644        }
645    }
646}

648/// Answer the server's accepted key with a signature from the local side, or say why not.
649/// Sunset's own serialisation of the signed message must equal the RFC 4252 construction
650/// this crate builds from the same three values the key service signs for (session id,
651/// user name, key); the key service's reply must carry the digest of exactly those bytes
652/// and a signature that verifies. Anything else skips the key without sending a signature.
653fn answer_sign<L: Local>(
654    s: sunset::event::RequestSign<'_, '_>,
655    local: &mut L,
656    user: &str,
657    identity: &[u8; 32],
658) -> Result<Option<Credential>, ClientError> {
659    let mut msg = [0u8; 256];
660    let written = s.message().and_then(|m| sunset::sshwire::write_ssh(&mut msg, &m));
661    let mut session_id = [0u8; 32];
662    let expected = match written {
663        Ok(n) if n >= 36 && msg[..4] == [0, 0, 0, 32] => {
664            session_id.copy_from_slice(&msg[4..36]);
665            userauth::signing_input(&session_id, user, identity).filter(|e| e.as_slice() == &msg[..n]).map(|_| n)
666        }
667        _ => None,
668    };
669    let Some(n) = expected else {
670        let _ = s.skip();
671        return Ok(Some(Credential::InputMismatch));
672    };
673    let request = SignRequest { session_id, username: user, message_sha256: userauth::digest(&msg[..n]) };
674    let signed = match local.sign(&request) {
675        Ok(signed) => signed,
676        Err(credential) => {
677            let _ = s.skip();
678            return Ok(Some(credential));
679        }
680    };
681    if signed.message_sha256 != request.message_sha256 {
682        let _ = s.skip();
683        return Ok(Some(Credential::InputMismatch));
684    }
685    if !userauth::verify(identity, &msg[..n], &signed.signature) {
686        let _ = s.skip();
687        return Ok(Some(Credential::BadSignature));
688    }
689    s.signed(&OwnedSig::Ed25519(signed.signature))?;
690    Ok(None)
691}
