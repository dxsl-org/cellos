1//! One connection, one session channel (command-contracts §4.2/§4.3): host trust before
2//! any credential, password authentication, then a PTY shell or one exec command.
3//!
4//! S9-6: the rekey policy is the embedder's ([`RekeyPolicy`]); every KEX (the first and
5//! each rekey, from either side) must finish within its deadline; a rekey's host key must
6//! be exactly the one accepted at the first KEX and is never offered to the trust store
7//! again; the session's lifecycle is reported to [`Local::progress`] write-ahead, so a
8//! request frame is never handed to the transport before `MayHaveRun` is recorded.
9use alloc::string::String;
10use alloc::vec::Vec;
11
12use sunset::packets::WinChange;
13use sunset::{ChanData, ChanHandle, CliEvent, CliSessionExit, Client, Event, PubKey, Pty, RekeyLimits, Runner};
14use zeroize::Zeroize;

16use crate::client::{pump, ClientError, HostKey, Transport, TransportError, RECV_CHUNK};

18/// D03 trust lookup result (abi-vfs-telemetry §4), and the end of a first-trust
19/// confirmation.
20#[derive(Debug, Clone, Copy, PartialEq, Eq)]
21pub enum Trust {
22    KnownExact,
23    Unknown,
24    Mismatch,
25    Denied,
26    Unavailable,
27    /// The user refused the unknown key (or the confirmation was cancelled).
28    Refused,
29    /// The confirmation's time ran out.
30    Expired,
31    /// This terminal cannot run a trusted confirmation (no interactive frontend).
32    NoConfirmation,
33}
34
35/// What the session runs.
36#[derive(Debug, Clone, PartialEq, Eq)]
37pub enum Mode {
38    /// PTY shell with `TERM` and the terminal size at start.
39    Pty { term: &'static str, rows: u16, cols: u16 },
40    /// One non-PTY command string, sent byte-for-byte.
41    Exec(String),
42}
43
44/// One unit of local input for the channel.
45#[derive(Debug, PartialEq, Eq)]
46pub enum Input {
47    /// `n` bytes were written to the buffer.
48    Data(usize),
49    /// Nothing available now.
50    Idle,
51    /// End of local input: send channel EOF once.
52    Eof,
53    /// The local user asked to disconnect (`~.`).
54    Disconnect,
55}

57/// Session phase (`phase` of the completion record, §4.3). `Connect` is the embedder's
58/// own phase until the host key is offered.
59#[derive(Debug, Clone, Copy, PartialEq, Eq)]
60pub enum Phase {
61    Trust,
62    Auth,
63    Request,
64    Running,
65}

67/// Named points of the session an embedder may observe (and, in a fault-injection
68/// build, hold): see [`Local::checkpoint`].
69#[derive(Debug, Clone, Copy, PartialEq, Eq)]
70pub enum Point {
71    /// Authenticated; the channel open and the request are not sent yet.
72    BeforeRequest,
73    /// The request frame exists (`MayHaveRun` is recorded) and is next to be written to
74    /// the transport. A hold here writes nothing: an interruption during it leaves the
75    /// request unsent while the record already says it may have run.
76    RequestSent,
77    /// The server accepted the request.
78    Accepted,
79    /// The remote exit status (or signal) arrived; channel output may still be pending.
80    ExitStatus,
81    /// A rekey (a KEX after the first) started, from either side; the peer's KEX packets
82    /// are not processed during a hold.
83    RekeyStart,
84}
85
86/// The rekey and KEX-deadline policy. [`RekeyPolicy::PRODUCTION`] is the contract
87/// (§4.2): rekey at the first of one hour since NEWKEYS, 1 GiB or 2^30 packets per
88/// direction; every KEX finishes within 30 seconds.
89#[derive(Debug, Clone, Copy, PartialEq, Eq)]
90pub struct RekeyPolicy {
91    pub max_age_ms: u64,
92    pub limits: RekeyLimits,
93    pub kex_ms: u64,
94}
95
96impl RekeyPolicy {
97    pub const PRODUCTION: Self = Self { max_age_ms: 3_600_000, limits: RekeyLimits::PRODUCTION, kex_ms: 30_000 };
98}

100/// The local side of the session.
101pub trait Local {
102    /// Decide on the server's host key. Only `KnownExact` lets the session continue.
103    fn trust(&mut self, key: &HostKey) -> Trust;
104    /// Called only when `trust` answered `Unknown`, before anything else is sent: run the
105    /// first-trust confirmation. `KnownExact` (the key is now stored) lets the session
106    /// continue; any other answer ends it with that answer.
107    fn confirm_unknown(&mut self, key: &HostKey) -> Trust;
108    /// Ask once for the password, after the host key was trusted. Write it into
109    /// `secret` and return true, or return false to cancel. `secret` is wiped after use.
110    fn password(&mut self, secret: &mut String) -> bool;
111    /// Fill `buf` with local input.
112    fn input(&mut self, buf: &mut [u8]) -> Input;
113    /// Remote standard output (or, in PTY mode, the whole terminal output).
114    fn stdout(&mut self, data: &[u8]);
115    /// Remote extended data type 1 (exec mode only).
116    fn stderr(&mut self, data: &[u8]);
117    /// A new terminal size since the last call, if any (PTY mode).
118    fn resized(&mut self) -> Option<(u16, u16)>;
119    /// The session entered `phase` with `dispatch` (write-ahead: `MayHaveRun` is reported
120    /// before the request frame exists).
121    fn progress(&mut self, phase: Phase, dispatch: Dispatch);
122    /// The session reached `point`. The answer is how long to hold there, in ms, while
123    /// only the transport is serviced (no packet is processed; at
124    /// [`Point::RequestSent`] not even the request is written). Production answers 0.
125    fn checkpoint(&mut self, _point: Point) -> u64 {
126        0
127    }
128}

130/// `dispatch` of the completion record (§4.3).
131#[derive(Debug, Clone, Copy, PartialEq, Eq)]
132pub enum Dispatch {
133    NotSent,
134    MayHaveRun,
135    Accepted,
136}

138#[derive(Debug, Clone, PartialEq, Eq)]
139pub enum Ending {
140    RemoteExit(u32),
141    RemoteSignal(String),
142    /// The host key was not accepted (`Trust` other than `KnownExact`), or a rekey
143    /// offered a key other than the one accepted first (`Mismatch`).
144    HostKey(Trust, HostKey),
145    /// Authentication was refused or cancelled.
146    Auth,
147    /// The server refused the shell/exec request.
148    RequestRefused,
149    /// The local user disconnected.
150    Disconnected,
151    /// The connection ended without an exit status or signal.
152    Unknown,
153    /// A phase, KEX or overall deadline passed.
154    TimedOut,
155    /// The transport ended (closed or failed) before the session completed.
156    TransportLost,
157    Error(String),
158}

162pub struct Peaks {

169#[derive(Debug, Clone, PartialEq, Eq)]
170pub struct Outcome {
171    pub dispatch: Dispatch,
172    pub ending: Ending,
173    pub stdout_complete: bool,
174    pub stderr_complete: bool,
175    /// Completed rekeys (key exchanges after the first).
176    pub rekeys: u32,
177    pub peaks: Peaks,
178}
179
180impl Outcome {
181    /// Local shell status (§4.3): the remote status when 0..=255, otherwise 255.
182    pub fn status(&self) -> i32 {
183        match self.ending {
184            Ending::RemoteExit(n) if n <= 255 => n as i32,
185            _ => 255,
186        }
187    }
188}

190/// Largest channel data read at once (§4.2: chunks of at most 16 KiB).
191pub const CHUNK: usize = 16 * 1024;
192
193pub fn run<T: Transport, L: Local>(t: &mut T, local: &mut L, user: &str, mode: &Mode, policy: &RekeyPolicy) -> Outcome {
194    let mut runner: Runner<'static, Client> = Runner::new_client_owned();
195    runner.set_rekey_limits(policy.limits);
196    let mut state = State {
197        dispatch: Dispatch::NotSent, chan: None, exit: None, password_asked: false, eof_sent: false,
198        host: None, clock: RekeyClock::default(), peaks: Peaks::default(),
199    };
200    let ending = drive(&mut runner, t, local, user, mode, policy, &mut state).unwrap_or_else(|e| match e {
201        ClientError::Transport(TransportError::TimedOut) => Ending::TimedOut,
202        ClientError::Transport(_) | ClientError::Ended => Ending::TransportLost,
203        ClientError::Protocol(sunset::Error::NoAuthMethods) => Ending::Auth,
204        other => Ending::Error(alloc::format!("{other}")),
205    });
206    // An exit status that arrived before the transport ended stands; the incomplete
207    // output is reported by `stdout_complete`.
208    let ending = match (ending, state.exit.take()) {
209        (Ending::Unknown | Ending::TransportLost, Some(exit)) => exit,
210        (ending, _) => ending,
211    };
212    let complete = state.chan.as_ref().is_some_and(|c| runner.is_channel_eof(c));
213    Outcome {
214        dispatch: state.dispatch, ending, stdout_complete: complete, stderr_complete: complete,
215        rekeys: runner.newkeys_count().saturating_sub(1), peaks: state.peaks,
216    }
217}

231#[derive(Default)]
232struct RekeyClock {
233    /// When the KEX in progress was first seen.
234    kex_started: Option<u64>,
235    /// Completed key exchanges seen so far, and when the last one completed.
236    newkeys: u32,
237    newkeys_at: u64,
238}

240/// Service only the transport for `ms` (a fault-injection hold): runner output goes
241/// out and input is queued, but no packet is processed. A closed or failed transport
242/// ends the hold with its error.
243fn hold<T: Transport>(runner: &mut Runner<'static, Client>, t: &mut T, pending: &mut Vec<u8>, ms: u64) -> Result<(), ClientError> {
244    if ms == 0 {
245        return Ok(());
246    }
247    let end = t.now_ms().saturating_add(ms);
248    while t.now_ms() < end {
249        pump(runner, t, pending)?;
250        t.idle()?;
251    }
252    Ok(())
253}

255/// Rekey policy and KEX deadline, once per turn. Returns how long to hold at a rekey's
256/// start ([`Point::RekeyStart`]).
257fn rekey_turn<T: Transport, L: Local>(
258    runner: &mut Runner<'static, Client>,
259    t: &mut T,
260    local: &mut L,
261    policy: &RekeyPolicy,
262    st: &mut RekeyClock,
263) -> Result<u64, ClientError> {
264    let mut hold_ms = 0;
265    let now = t.now_ms();
266    let newkeys = runner.newkeys_count();
267    if newkeys != st.newkeys {
268        st.newkeys = newkeys;
269        st.newkeys_at = now;
270    }
271    if runner.kex_active() {
272        let started = *st.kex_started.get_or_insert_with(|| {
273            if newkeys > 0 {
274                hold_ms = local.checkpoint(Point::RekeyStart);
275            }
276            now
277        });
278        if now.saturating_sub(started) >= policy.kex_ms {
279            return Err(ClientError::Transport(TransportError::TimedOut));
280        }
281    } else {
282        st.kex_started = None;
283        if newkeys > 0 && now.saturating_sub(st.newkeys_at) >= policy.max_age_ms {
284            runner.request_rekey();
285        }
286    }
287    Ok(hold_ms)
288}

290#[allow(clippy::too_many_arguments)]
291fn drive<T: Transport, L: Local>(
292    runner: &mut Runner<'static, Client>,
293    t: &mut T,
294    local: &mut L,
295    user: &str,
296    mode: &Mode,
297    policy: &RekeyPolicy,
298    st: &mut State,
299) -> Result<Ending, ClientError> {
300    let mut pending = Vec::with_capacity(RECV_CHUNK);
301    let mut buf = alloc::vec![0u8; CHUNK];
302    // Local input not yet taken by the channel: one chunk, wiped as it is sent and on drop.
303    let mut out = zeroize::Zeroizing::new(alloc::vec![0u8; CHUNK]);
304    let mut unsent = (0usize, 0usize);
305    let mut open_session = false;
306    loop {
307        let hold_ms = rekey_turn(runner, t, local, policy, &mut st.clock)?;
308        hold(runner, t, &mut pending, hold_ms)?;
309        let mut progressed = false;
310        let mut stop = None;
311        let mut point = None;
312        match runner.progress() {
313            // Every offered method was refused (or the password was not given).
314            Err(sunset::Error::NoAuthMethods) => return Ok(Ending::Auth),
315            Err(e) => return Err(e.into()),
316            Ok(Event::Cli(CliEvent::Hostkey(h))) => {
317                let key = match h.hostkey()? {
318                    PubKey::Ed25519(k) => HostKey { algorithm: "ssh-ed25519", key: k.key.0 },
319                    #[allow(unreachable_patterns)]
320                    _ => return Ok(Ending::Error(String::from("unsupported host key type"))),
321                };
322                let decision = match &st.host {
323                    // A rekey: exactly the accepted key, never a new trust decision.
324                    Some(accepted) if *accepted == key => Trust::KnownExact,
325                    Some(_) => Trust::Mismatch,
326                    None => {
327                        local.progress(Phase::Trust, st.dispatch);
328                        match local.trust(&key) {
329                            Trust::Unknown => local.confirm_unknown(&key),
330                            other => other,
331                        }
332                    }
333                };
334                match decision {
335                    Trust::KnownExact => {
336                        h.accept()?;
337                        if st.host.is_none() {
338                            st.host = Some(key);
339                            local.progress(Phase::Auth, st.dispatch);
340                        }
341                    }
342                    other => {
343                        // Upstream reports the refusal as `BadUsage`; it is intended here.
344                        let _ = h.reject();
345                        stop = Some(Ending::HostKey(other, key));
346                    }
347                }
348            }
349            Ok(Event::Cli(CliEvent::Username(u))) => u.username(user)?,
350            Ok(Event::Cli(CliEvent::Password(p))) => {
351                if st.password_asked {
352                    // Exclusive auth, no retry: a second request means the first failed.
353                    p.skip()?;
354                    stop = Some(Ending::Auth);
355                } else {
356                    st.password_asked = true;
357                    let mut secret = String::new();
358                    let given = local.password(&mut secret);
359                    let r = if given { p.password(&secret) } else { p.skip() };
360                    secret.zeroize();
361                    r?;
362                    if !given {
363                        stop = Some(Ending::Auth);
364                    }
365                }
366            }
367            Ok(Event::Cli(CliEvent::Pubkey(p))) => p.skip()?,
368            Ok(Event::Cli(CliEvent::AgentSign(s))) => s.skip()?,
369            Ok(Event::Cli(CliEvent::Authenticated)) => {
370                local.progress(Phase::Request, st.dispatch);
371                point = Some(Point::BeforeRequest);
372                open_session = true;
373            }
374            Ok(Event::Cli(CliEvent::SessionOpened(mut o))) => {
375                // Write-ahead: the request may run remotely once its frame exists.
376                st.dispatch = Dispatch::MayHaveRun;
377                local.progress(Phase::Request, st.dispatch);
378                match mode {
379                    Mode::Pty { term, rows, cols } => {
380                        let mut pty = Pty {
381                            term: heapless::String::new(),
382                            cols: u32::from(*cols),
383                            rows: u32::from(*rows),
384                            width: 0,
385                            height: 0,
386                            modes: heapless::Vec::new(),
387                        };
388                        pty.term.push_str(term).map_err(|_| ClientError::Protocol(sunset::Error::msg("TERM")))?;
389                        o.pty(pty)?;
390                        o.shell()?;
391                    }
392                    Mode::Exec(cmd) => o.exec(cmd)?,
393                }
394                point = Some(Point::RequestSent);
395            }
396            Ok(Event::Cli(CliEvent::SessionExit(x))) => {
397                if st.exit.is_some() {
398                    // Duplicate or conflicting outcomes: uncertainty, not last-wins.
399                    st.exit = None;
400                    stop = Some(Ending::Error(String::from("duplicate remote exit outcome")));
401                } else {
402                    st.exit = Some(match x {
403                        CliSessionExit::Status(n) => Ending::RemoteExit(n),
404                        CliSessionExit::Signal(s) => Ending::RemoteSignal(String::from(s.signal)),
405                    });
406                    point = Some(Point::ExitStatus);
407                }
408            }
409            Ok(Event::Cli(CliEvent::Banner(_))) => {}
410            Ok(Event::Cli(CliEvent::Defunct)) => {
411                return Ok(st.exit.take().unwrap_or(match st.dispatch {
412                    Dispatch::NotSent if !st.password_asked => Ending::Error(String::from("server closed the connection")),
413                    Dispatch::NotSent => Ending::Auth,
414                    _ => Ending::Unknown,
415                }));
416            }
417            Ok(Event::None) => {}
418            Ok(_) => progressed = true,
419        }
420        if let Some(ending) = stop {
421            let _ = pump(runner, t, &mut pending);
422            return Ok(ending);
423        }
424        if let Some(point) = point {
425            if point == Point::RequestSent {
426                // The request frame is queued, not yet written: a hold here writes nothing.
427                let ms = local.checkpoint(point);
428                let end = t.now_ms().saturating_add(ms);
429                while t.now_ms() < end {
430                    t.idle()?;
431                }
432                pump(runner, t, &mut pending)?;
433            } else {
434                // Anything queued (the channel open, a reply) reaches the transport first.
435                pump(runner, t, &mut pending)?;
436                let ms = local.checkpoint(point);
437                hold(runner, t, &mut pending, ms)?;
438            }
439            progressed = true;
440        }
441        st.peaks.pending = st.peaks.pending.max(pending.capacity());
442        if open_session && st.chan.is_none() {
443            st.chan = Some(runner.open_client_session()?);
444            progressed = true;
445        }
446        if let Some(chan) = st.chan.as_ref() {
447            if st.dispatch == Dispatch::MayHaveRun {
448                match runner.channel_request_reply(chan) {
449                    Some(true) => {
450                        st.dispatch = Dispatch::Accepted;
451                        local.progress(Phase::Running, st.dispatch);
452                        pump(runner, t, &mut pending)?;
453                        let ms = local.checkpoint(Point::Accepted);
454                        hold(runner, t, &mut pending, ms)?;
455                    }
456                    Some(false) => return Ok(Ending::RequestRefused),
457                    None => {}
458                }
459            }
460            // Remote output first, so local input cannot starve it. In PTY mode extended
461            // data is part of the one terminal output stream.
462            for dt in [ChanData::Normal, ChanData::Stderr] {
463                match runner.read_channel(chan, dt, &mut buf) {
464                    Ok(0) | Err(_) => {}
465                    Ok(n) => {
466                        progressed = true;
467                        st.peaks.chunk = st.peaks.chunk.max(n);
468                        match (mode, dt) {
469                            (Mode::Exec(_), ChanData::Stderr) => local.stderr(&buf[..n]),
470                            _ => local.stdout(&buf[..n]),
471                        }
472                    }
473                }
474            }
475            if runner.is_channel_closed(chan) {
476                let _ = pump(runner, t, &mut pending);
477                return Ok(st.exit.take().unwrap_or(Ending::Unknown));
478            }
479            if st.eof_sent {
480                // Local input has ended; only a local disconnect can still come.
481                if local.input(&mut []) == Input::Disconnect {
482                    let _ = pump(runner, t, &mut pending);
483                    return Ok(Ending::Disconnected);
484                }
485            } else if st.dispatch != Dispatch::NotSent {
486                if let Some((rows, cols)) = local.resized() {
487                    if matches!(mode, Mode::Pty { .. }) {
488                        let w = WinChange { cols: u32::from(cols), rows: u32::from(rows), width: 0, height: 0 };
489                        runner.term_window_change(chan, &w)?;
490                        progressed = true;
491                    }
492                }
493                // At most one unsent chunk waits (KEX pause, window or buffer full); the
494                // next turn retries it after `progress()` had its chance to move the KEX.
495                if unsent.1 > unsent.0 {
496                    let w = runner.write_channel(chan, ChanData::Normal, &out[unsent.0..unsent.1])?;
497                    out[unsent.0..unsent.0 + w].zeroize();
498                    unsent.0 += w;
499                    progressed |= w > 0;
500                } else {
501                    let room = runner.write_channel_ready(chan, ChanData::Normal)?.unwrap_or(0).min(CHUNK);
502                    if room > 0 {
503                        match local.input(&mut out[..room]) {
504                            Input::Data(n) => {
505                                let w = runner.write_channel(chan, ChanData::Normal, &out[..n])?;
506                                out[..w].zeroize();
507                                unsent = (w, n);
508                                progressed = true;
509                            }
510                            Input::Eof => {
511                                runner.send_channel_eof(chan)?;
512                                st.eof_sent = true;
513                                progressed = true;
514                            }
515                            Input::Disconnect => {
516                                let _ = pump(runner, t, &mut pending);
517                                return Ok(Ending::Disconnected);
518                            }
519                            Input::Idle => {}
520                        }
521                    }
522                }
523            }
524        }
525        if !pump(runner, t, &mut pending)? && !progressed {
526            t.idle()?;
527        }
528    }
