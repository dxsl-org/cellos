1//! One connection, one session channel (command-contracts §4.2/§4.3): host trust before
2//! any credential, password or publickey authentication (exclusive: no fallback from one
3//! to the other), then a PTY shell or one exec command.
4//!
5//! S9-6: the rekey policy is the embedder's ([`RekeyPolicy`]); every KEX (the first and
6//! each rekey, from either side) must finish within its deadline; a rekey's host key must
7//! be exactly the one accepted at the first KEX and is never offered to the trust store
8//! again; the session's lifecycle is reported to [`Local::progress`] write-ahead, so a
9//! request frame is never handed to the transport before `MayHaveRun` is recorded.

24/// D03 trust lookup result (abi-vfs-telemetry §4), and the end of a first-trust
25/// confirmation.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum Trust {
28    KnownExact,
29    Unknown,
30    Mismatch,
31    Denied,
32    Unavailable,
33    /// The user refused the unknown key (or the confirmation was cancelled).
34    Refused,
35    /// The confirmation's time ran out.
36    Expired,
37    /// This terminal cannot run a trusted confirmation (no interactive frontend).
38    NoConfirmation,
39}

54/// How the session authenticates. Exactly one method is tried, once.
55#[derive(Debug, Clone, Copy, PartialEq, Eq)]
56pub enum AuthKind {
57    /// Ask [`Local::password`] once.
58    Password,
59    /// Offer [`Local::identity`] and sign through [`Local::sign`]. The password method is
60    /// never offered, not even after the server refuses the key.
61    PublicKey,
62}

64/// Why a publickey login ended before the server decided. Every one is a local refusal:
65/// nothing was signed for the server (or the signature was not sent), so the login is
66/// `NotSent` and in the `Auth` phase.
67#[derive(Debug, Clone, Copy, PartialEq, Eq)]
68pub enum Credential {
69    /// The key service is not available.
70    Unavailable,
71    /// The key service refuses this caller.
72    Denied,
73    /// The key service does not hold the named identity.
74    UnknownIdentity,
75    /// The user cancelled the passphrase prompt.
76    UnlockCancelled,
77    /// The identity is passphrase-wrapped and there is no terminal to ask on.
78    UnlockNeedsTerminal,
79    /// The key service could not unlock the identity (wrong passphrase, corrupt key, ...).
80    UnlockFailed,
81    /// Too many failed unlocks: the identity is locked.
82    UnlockLocked,
83    /// The key service did not answer in time.
84    Timeout,
85    /// The key service answered something this client does not accept.
86    Protocol,
87    /// The key service signed a different input than the one Sunset asked for.
88    InputMismatch,
89    /// The key service returned a signature that does not verify.
90    BadSignature,
91}

93/// What the key service is asked to sign: the typed state of one userauth attempt. The
94/// session already compared Sunset's own serialisation with the RFC 4252 construction
95/// this client builds ([`userauth::signing_input`]) and `message_sha256` is the digest of
96/// those exact bytes.
97#[derive(Debug, Clone, PartialEq, Eq)]
98pub struct SignRequest<'a> {
99    pub session_id: [u8; 32],
100    pub username: &'a str,
101    pub message_sha256: [u8; 32],
102}

104/// The key service's answer: the raw `R || S` signature and the digest of the input it

106#[derive(Debug, Clone, PartialEq, Eq)]
107pub struct Signed {
108    pub signature: [u8; 64],
109    pub message_sha256: [u8; 32],

128pub enum Phase {

135/// Named points of the session an embedder may observe (and, in a fault-injection
136/// build, hold): see [`Local::checkpoint`].
137#[derive(Debug, Clone, Copy, PartialEq, Eq)]
138pub enum Point {
139    /// Authenticated; the channel open and the request are not sent yet.
140    BeforeRequest,
141    /// The request frame exists (`MayHaveRun` is recorded) and is next to be written to
142    /// the transport. A hold here writes nothing: an interruption during it leaves the
143    /// request unsent while the record already says it may have run.
144    RequestSent,
145    /// The server accepted the request.
146    Accepted,
147    /// The remote exit status (or signal) arrived; channel output may still be pending.
148    ExitStatus,
149    /// A rekey (a KEX after the first) started, from either side; the peer's KEX packets
150    /// are not processed during a hold.
151    RekeyStart,
152}
153
154/// The rekey and KEX-deadline policy. [`RekeyPolicy::PRODUCTION`] is the contract
155/// (§4.2): rekey at the first of one hour since NEWKEYS, 1 GiB or 2^30 packets per
156/// direction; every KEX finishes within 30 seconds.
157#[derive(Debug, Clone, Copy, PartialEq, Eq)]
158pub struct RekeyPolicy {
159    pub max_age_ms: u64,
160    pub limits: RekeyLimits,
161    pub kex_ms: u64,
162}
163
164impl RekeyPolicy {
165    pub const PRODUCTION: Self = Self {
166        max_age_ms: 3_600_000,
167        limits: RekeyLimits::PRODUCTION,
168        kex_ms: 30_000,
169    };
170}

172/// The local side of the session.
173pub trait Local {
174    /// Decide on the server's host key. Only `KnownExact` lets the session continue.
175    fn trust(&mut self, key: &HostKey) -> Trust;
176    /// Called only when `trust` answered `Unknown`, before anything else is sent: run the
177    /// first-trust confirmation. `KnownExact` (the key is now stored) lets the session
178    /// continue; any other answer ends it with that answer.
179    fn confirm_unknown(&mut self, key: &HostKey) -> Trust;
180    /// Ask once for the password, after the host key was trusted. Write it into
181    /// `secret` and return true, or return false to cancel. `secret` is wiped after use.
182    fn password(&mut self, secret: &mut String) -> bool;
183    /// Publickey only, called once after the host key was trusted and the server listed
184    /// publickey: the 32-byte ed25519 public key to offer. A refusal ends the login with
185    /// that [`Credential`]. Never called in password mode.
186    fn identity(&mut self) -> Result<[u8; 32], Credential>;
187    /// Publickey only, called once after the server accepted the key (so an identity the
188    /// server does not know never costs a passphrase): produce the signature for `r`.
189    /// Never called in password mode.
190    fn sign(&mut self, r: &SignRequest<'_>) -> Result<Signed, Credential>;
191    /// Fill `buf` with local input.
192    fn input(&mut self, buf: &mut [u8]) -> Input;
193    /// Remote standard output (or, in PTY mode, the whole terminal output).
194    fn stdout(&mut self, data: &[u8]);
195    /// Remote extended data type 1 (exec mode only).
196    fn stderr(&mut self, data: &[u8]);
197    /// A new terminal size since the last call, if any (PTY mode).
198    fn resized(&mut self) -> Option<(u16, u16)>;
199    /// The session entered `phase` with `dispatch` (write-ahead: `MayHaveRun` is reported
200    /// before the request frame exists).
201    fn progress(&mut self, phase: Phase, dispatch: Dispatch);
202    /// The session reached `point`. The answer is how long to hold there, in ms, while
203    /// only the transport is serviced (no packet is processed; at
204    /// [`Point::RequestSent`] not even the request is written). Production answers 0.
205    fn checkpoint(&mut self, _point: Point) -> u64 {
206        0
207    }
208}

219pub enum Ending {

272/// Largest channel data read at once (§4.2: chunks of at most 16 KiB).

275pub fn run<T: Transport, L: Local>(
276    t: &mut T,
277    local: &mut L,
278    user: &str,
279    auth: AuthKind,
280    mode: &Mode,
281    policy: &RekeyPolicy,
282) -> Outcome {
283    let mut runner: Runner<'static, Client> = Runner::new_client_owned();
284    runner.set_rekey_limits(policy.limits);
285    let mut state = State {
286        dispatch: Dispatch::NotSent,
287        chan: None,
288        exit: None,
289        password_asked: false,
290        key_offered: false,
291        signed: false,
292        identity: None,
293        eof_sent: false,
294        host: None,
295        clock: RekeyClock::default(),
296        peaks: Peaks::default(),
297    };
298    let ending =
299        drive(&mut runner, t, local, user, auth, mode, policy, &mut state).unwrap_or_else(|e| {
300            match e {
301                ClientError::Transport(TransportError::TimedOut) => Ending::TimedOut,
302                ClientError::Transport(_) | ClientError::Ended => Ending::TransportLost,
303                ClientError::Protocol(sunset::Error::NoAuthMethods) => Ending::Auth,
304                other => Ending::Error(alloc::format!("{other}")),
305            }
306        });
307    // An exit status that arrived before the transport ended stands; the incomplete
308    // output is reported by `stdout_complete`.
309    let ending = match (ending, state.exit.take()) {
310        (Ending::Unknown | Ending::TransportLost, Some(exit)) => exit,
311        (ending, _) => ending,
312    };
313    let complete = state
314        .chan
315        .as_ref()
316        .is_some_and(|c| runner.is_channel_eof(c));
317    Outcome {
318        dispatch: state.dispatch,
319        ending,
320        stdout_complete: complete,
321        stderr_complete: complete,
322        rekeys: runner.newkeys_count().saturating_sub(1),
323        peaks: state.peaks,
324    }
325}

345#[derive(Default)]
346struct RekeyClock {
347    /// When the KEX in progress was first seen.
348    kex_started: Option<u64>,
349    /// Completed key exchanges seen so far, and when the last one completed.
350    newkeys: u32,
351    newkeys_at: u64,
352}

354/// Service only the transport for `ms` (a fault-injection hold): runner output goes
355/// out and input is queued, but no packet is processed. A closed or failed transport
356/// ends the hold with its error.
357fn hold<T: Transport>(
358    runner: &mut Runner<'static, Client>,
359    t: &mut T,
360    pending: &mut Vec<u8>,
361    ms: u64,
362) -> Result<(), ClientError> {
363    if ms == 0 {
364        return Ok(());
365    }
366    let end = t.now_ms().saturating_add(ms);
367    while t.now_ms() < end {
368        pump(runner, t, pending)?;
369        t.idle()?;
370    }
371    Ok(())
372}

374/// Rekey policy and KEX deadline, once per turn. Returns how long to hold at a rekey's
375/// start ([`Point::RekeyStart`]).
376fn rekey_turn<T: Transport, L: Local>(
377    runner: &mut Runner<'static, Client>,
378    t: &mut T,
379    local: &mut L,
380    policy: &RekeyPolicy,
381    st: &mut RekeyClock,
382) -> Result<u64, ClientError> {
383    let mut hold_ms = 0;
384    let now = t.now_ms();
385    let newkeys = runner.newkeys_count();
386    if newkeys != st.newkeys {
387        st.newkeys = newkeys;
388        st.newkeys_at = now;
389    }
390    if runner.kex_active() {
391        let started = *st.kex_started.get_or_insert_with(|| {
392            if newkeys > 0 {
393                hold_ms = local.checkpoint(Point::RekeyStart);
394            }
395            now
396        });
397        if now.saturating_sub(started) >= policy.kex_ms {
398            return Err(ClientError::Transport(TransportError::TimedOut));
399        }
400    } else {
401        st.kex_started = None;
402        if newkeys > 0 && now.saturating_sub(st.newkeys_at) >= policy.max_age_ms {
403            runner.request_rekey();
404        }
405    }
406    Ok(hold_ms)
407}

409#[allow(clippy::too_many_arguments)]
410fn drive<T: Transport, L: Local>(
411    runner: &mut Runner<'static, Client>,
412    t: &mut T,
413    local: &mut L,
414    user: &str,
415    auth: AuthKind,
416    mode: &Mode,
417    policy: &RekeyPolicy,
418    st: &mut State,
419) -> Result<Ending, ClientError> {
420    let mut pending = Vec::with_capacity(RECV_CHUNK);
421    let mut buf = alloc::vec![0u8; CHUNK];
422    // Local input not yet taken by the channel: one chunk, wiped as it is sent and on drop.
423    let mut out = zeroize::Zeroizing::new(alloc::vec![0u8; CHUNK]);
424    let mut unsent = (0usize, 0usize);
425    let mut open_session = false;
426    loop {
427        let hold_ms = rekey_turn(runner, t, local, policy, &mut st.clock)?;
428        hold(runner, t, &mut pending, hold_ms)?;
429        let mut progressed = false;
430        let mut stop = None;
431        let mut point = None;
432        match runner.progress() {
433            // Every offered method was refused (or the password was not given).
434            Err(sunset::Error::NoAuthMethods) => return Ok(Ending::Auth),
435            Err(e) => return Err(e.into()),
436            Ok(Event::Cli(CliEvent::Hostkey(h))) => {
437                let key = match h.hostkey()? {
438                    PubKey::Ed25519(k) => HostKey {
439                        algorithm: "ssh-ed25519",
440                        key: k.key.0,
441                    },
442                    #[allow(unreachable_patterns)]
443                    _ => return Ok(Ending::Error(String::from("unsupported host key type"))),
444                };
445                let decision = match &st.host {
446                    // A rekey: exactly the accepted key, never a new trust decision.
447                    Some(accepted) if *accepted == key => Trust::KnownExact,
448                    Some(_) => Trust::Mismatch,
449                    None => {
450                        local.progress(Phase::Trust, st.dispatch);
451                        match local.trust(&key) {
452                            Trust::Unknown => local.confirm_unknown(&key),
453                            other => other,
454                        }
455                    }
456                };
457                match decision {
458                    Trust::KnownExact => {
459                        h.accept()?;
460                        if st.host.is_none() {
461                            st.host = Some(key);
462                            local.progress(Phase::Auth, st.dispatch);
463                        }
464                    }
465                    other => {
466                        // Upstream reports the refusal as `BadUsage`; it is intended here.
467                        let _ = h.reject();
468                        stop = Some(Ending::HostKey(other, key));
469                    }
470                }
471            }
472            Ok(Event::Cli(CliEvent::Username(u))) => u.username(user)?,
473            Ok(Event::Cli(CliEvent::Password(p))) => {
474                if auth == AuthKind::PublicKey {
475                    // Exclusive auth: once the key is skipped or refused Sunset falls back to
476                    // the password method; that fallback is closed here.
477                    p.skip()?;
478                    stop = Some(Ending::Auth);
479                } else if st.password_asked {
480                    // Exclusive auth, no retry: a second request means the first failed.
481                    p.skip()?;
482                    stop = Some(Ending::Auth);
483                } else {
484                    st.password_asked = true;
485                    let mut secret = String::new();
486                    let given = local.password(&mut secret);
487                    let r = if given { p.password(&secret) } else { p.skip() };
488                    secret.zeroize();
489                    r?;
490                    if !given {
491                        stop = Some(Ending::Auth);
492                    }
493                }
494            }
495            Ok(Event::Cli(CliEvent::Pubkey(p))) => match auth {
496                AuthKind::Password => p.skip()?,
497                // The server refused the key or its signature: nothing else is offered.
498                AuthKind::PublicKey if st.key_offered => {
499                    let _ = p.skip();
500                    stop = Some(Ending::Auth);
501                }
502                AuthKind::PublicKey => {
503                    let offered = local.identity().and_then(|key| {
504                        VerifyingKey::from_bytes(&key)
505                            .map(|verifying| (key, verifying))
506                            .map_err(|_| Credential::Protocol)
507                    });
508                    match offered {
509                        Ok((key, verifying)) => {
510                            st.key_offered = true;
511                            st.identity = Some(key);
512                            p.pubkey(SignKey::AgentEd25519(verifying))?;
513                        }
514                        Err(credential) => {
515                            let _ = p.skip();
516                            stop = Some(Ending::Credential(credential));
517                        }
518                    }
519                }
520            },
521            Ok(Event::Cli(CliEvent::AgentSign(s))) => match (auth, st.identity, st.signed) {
522                (AuthKind::PublicKey, Some(key), false) => match answer_sign(s, local, user, &key)?
523                {
524                    None => st.signed = true,
525                    Some(credential) => stop = Some(Ending::Credential(credential)),
526                },
527                // Password mode cannot raise this; a second request means the first
528                // signature was refused.
529                _ => {
530                    let _ = s.skip();
531                    stop = Some(Ending::Auth);
532                }
533            },
534            Ok(Event::Cli(CliEvent::Authenticated)) => {
535                local.progress(Phase::Request, st.dispatch);
536                point = Some(Point::BeforeRequest);
537                open_session = true;
538            }
539            Ok(Event::Cli(CliEvent::SessionOpened(mut o))) => {
540                // Write-ahead: the request may run remotely once its frame exists.
541                st.dispatch = Dispatch::MayHaveRun;
542                local.progress(Phase::Request, st.dispatch);
543                match mode {
544                    Mode::Pty { term, rows, cols } => {
545                        let mut pty = Pty {
546                            term: heapless::String::new(),
547                            cols: u32::from(*cols),
548                            rows: u32::from(*rows),
549                            width: 0,
550                            height: 0,
551                            modes: heapless::Vec::new(),
552                        };
553                        pty.term
554                            .push_str(term)
555                            .map_err(|_| ClientError::Protocol(sunset::Error::msg("TERM")))?;
556                        o.pty(pty)?;
557                        o.shell()?;
558                    }
559                    Mode::Exec(cmd) => o.exec(cmd)?,
560                }
561                point = Some(Point::RequestSent);
562            }
563            Ok(Event::Cli(CliEvent::SessionExit(x))) => {
564                if st.exit.is_some() {
565                    // Duplicate or conflicting outcomes: uncertainty, not last-wins.
566                    st.exit = None;
567                    stop = Some(Ending::Error(String::from("duplicate remote exit outcome")));
568                } else {
569                    st.exit = Some(match x {
570                        CliSessionExit::Status(n) => Ending::RemoteExit(n),
571                        CliSessionExit::Signal(s) => Ending::RemoteSignal(String::from(s.signal)),
572                    });
573                    point = Some(Point::ExitStatus);
574                }
575            }
576            Ok(Event::Cli(CliEvent::Banner(_))) => {}
577            Ok(Event::Cli(CliEvent::Defunct)) => {
578                return Ok(st.exit.take().unwrap_or(match st.dispatch {
579                    Dispatch::NotSent if !(st.password_asked || st.signed) => {
580                        Ending::Error(String::from("server closed the connection"))
581                    }
582                    Dispatch::NotSent => Ending::Auth,
583                    _ => Ending::Unknown,
584                }));
585            }
586            Ok(Event::None) => {}
587            Ok(_) => progressed = true,
588        }
589        if let Some(ending) = stop {
590            let _ = pump(runner, t, &mut pending);
591            return Ok(ending);
592        }
593        if let Some(point) = point {
594            if point == Point::RequestSent {
595                // The request frame is queued, not yet written: a hold here writes nothing.
596                let ms = local.checkpoint(point);
597                let end = t.now_ms().saturating_add(ms);
598                while t.now_ms() < end {
599                    t.idle()?;
600                }
601                pump(runner, t, &mut pending)?;
602            } else {
603                // Anything queued (the channel open, a reply) reaches the transport first.
604                pump(runner, t, &mut pending)?;
605                let ms = local.checkpoint(point);
606                hold(runner, t, &mut pending, ms)?;
607            }
608            progressed = true;
609        }
610        st.peaks.pending = st.peaks.pending.max(pending.capacity());
611        if open_session && st.chan.is_none() {
612            st.chan = Some(runner.open_client_session()?);
613            progressed = true;
614        }
615        if let Some(chan) = st.chan.as_ref() {
616            if st.dispatch == Dispatch::MayHaveRun {
617                match runner.channel_request_reply(chan) {
618                    Some(true) => {
619                        st.dispatch = Dispatch::Accepted;
620                        local.progress(Phase::Running, st.dispatch);
621                        pump(runner, t, &mut pending)?;
622                        let ms = local.checkpoint(Point::Accepted);
623                        hold(runner, t, &mut pending, ms)?;
624                    }
625                    Some(false) => return Ok(Ending::RequestRefused),
626                    None => {}
627                }
628            }
629            // Remote output first, so local input cannot starve it. In PTY mode extended
630            // data is part of the one terminal output stream.
631            for dt in [ChanData::Normal, ChanData::Stderr] {
632                match runner.read_channel(chan, dt, &mut buf) {
633                    Ok(0) | Err(_) => {}
634                    Ok(n) => {
635                        progressed = true;
636                        st.peaks.chunk = st.peaks.chunk.max(n);
637                        match (mode, dt) {
638                            (Mode::Exec(_), ChanData::Stderr) => local.stderr(&buf[..n]),
639                            _ => local.stdout(&buf[..n]),
640                        }
641                    }
642                }
643            }
644            if runner.is_channel_closed(chan) {
645                let _ = pump(runner, t, &mut pending);
646                return Ok(st.exit.take().unwrap_or(Ending::Unknown));
647            }
648            if st.eof_sent {
649                // Local input has ended; only a local disconnect can still come.
650                if local.input(&mut []) == Input::Disconnect {
651                    let _ = pump(runner, t, &mut pending);
652                    return Ok(Ending::Disconnected);
653                }
654            } else if st.dispatch != Dispatch::NotSent {
655                if let Some((rows, cols)) = local.resized()
656                    && matches!(mode, Mode::Pty { .. })
657                {
658                    let w = WinChange {
659                        cols: u32::from(cols),
660                        rows: u32::from(rows),
661                        width: 0,
662                        height: 0,
663                    };
664                    runner.term_window_change(chan, &w)?;
665                    progressed = true;
666                }
667                // At most one unsent chunk waits (KEX pause, window or buffer full); the
668                // next turn retries it after `progress()` had its chance to move the KEX.
669                if unsent.1 > unsent.0 {
670                    let w =
671                        runner.write_channel(chan, ChanData::Normal, &out[unsent.0..unsent.1])?;
672                    out[unsent.0..unsent.0 + w].zeroize();
673                    unsent.0 += w;
674                    progressed |= w > 0;
675                } else {
676                    let room = runner
677                        .write_channel_ready(chan, ChanData::Normal)?
678                        .unwrap_or(0)
679                        .min(CHUNK);
680                    if room > 0 {
681                        match local.input(&mut out[..room]) {
682                            Input::Data(n) => {
683                                let w = runner.write_channel(chan, ChanData::Normal, &out[..n])?;
684                                out[..w].zeroize();
685                                unsent = (w, n);
686                                progressed = true;
687                            }
688                            Input::Eof => {
689                                runner.send_channel_eof(chan)?;
690                                st.eof_sent = true;
691                                progressed = true;
692                            }
693                            Input::Disconnect => {
694                                let _ = pump(runner, t, &mut pending);
695                                return Ok(Ending::Disconnected);
696                            }
697                            Input::Idle => {}
698                        }
699                    }
700                }
701            }
702        }
703        if !pump(runner, t, &mut pending)? && !progressed {
704            t.idle()?;
705        }
706    }
707}

709/// Answer the server's accepted key with a signature from the local side, or say why not.
710/// Sunset's own serialisation of the signed message must equal the RFC 4252 construction
711/// this crate builds from the same three values the key service signs for (session id,
712/// user name, key); the key service's reply must carry the digest of exactly those bytes
713/// and a signature that verifies. Anything else skips the key without sending a signature.
714fn answer_sign<L: Local>(
715    s: sunset::event::RequestSign<'_, '_>,
716    local: &mut L,
717    user: &str,
718    identity: &[u8; 32],
719) -> Result<Option<Credential>, ClientError> {
720    let mut msg = [0u8; 256];
721    let written = s
722        .message()
723        .and_then(|m| sunset::sshwire::write_ssh(&mut msg, &m));
724    let mut session_id = [0u8; 32];
725    let expected = match written {
726        Ok(n) if n >= 36 && msg[..4] == [0, 0, 0, 32] => {
727            session_id.copy_from_slice(&msg[4..36]);
728            userauth::signing_input(&session_id, user, identity)
729                .filter(|e| e.as_slice() == &msg[..n])
730                .map(|_| n)
731        }
732        _ => None,
733    };
734    let Some(n) = expected else {
735        let _ = s.skip();
736        return Ok(Some(Credential::InputMismatch));
737    };
738    let request = SignRequest {
739        session_id,
740        username: user,
741        message_sha256: userauth::digest(&msg[..n]),
742    };
743    let signed = match local.sign(&request) {
744        Ok(signed) => signed,
745        Err(credential) => {
746            let _ = s.skip();
747            return Ok(Some(credential));
748        }
749    };
750    if signed.message_sha256 != request.message_sha256 {
751        let _ = s.skip();
752        return Ok(Some(Credential::InputMismatch));
753    }
754    if !userauth::verify(identity, &msg[..n], &signed.signature) {
755        let _ = s.skip();
756        return Ok(Some(Credential::BadSignature));
757    }
758    s.signed(&OwnedSig::Ed25519(signed.signature))?;
759    Ok(None)
760}
