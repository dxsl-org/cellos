2//! Frozen bounds, validators, and canonical chunk arithmetic for the single
3//! KMS-custodied OpenSSH ed25519 identity (opcodes 15-24).
4//!
5//! These definitions are shared by KMS, its clients, and tests so that one
6//! implementation of every bound exists. Nothing here selects a key or an
7//! algorithm: a KMS store holds exactly one SSH identity and the signing
8//! request names only typed protocol state.
9
10/// Longest admin-chosen KEY_ID name (`[A-Za-z0-9_-]`, 1..=64). Frozen.
11pub const KMS_SSH_KEY_NAME_MAX: usize = 64;
12/// Longest SSH userauth username (`[A-Za-z0-9._-]`, 1..=31). Frozen.
13pub const KMS_SSH_USERNAME_MAX: usize = 31;
14/// SSH session identifier length (curve25519-sha256 exchange hash). Frozen.
15pub const KMS_SSH_SESSION_ID_LEN: usize = 32;
16/// Raw ed25519 public key length. Frozen.
17pub const KMS_SSH_PUBLIC_KEY_LEN: usize = 32;
18/// Raw ed25519 signature length (`R || S`). Frozen.
19pub const KMS_SSH_SIGNATURE_LEN: usize = 64;
20/// Largest OpenSSH key blob (binary, not armored) one import may carry. Frozen.
21pub const KMS_SSH_IMPORT_MAX_LEN: usize = 64 * 1024;
22/// Data bytes per transfer chunk: the 112-byte payload minus a 16-byte header.
23/// Frozen.
24pub const KMS_SSH_CHUNK_LEN: usize = 96;
25/// Longest passphrase staged for one unlock. Frozen.
26pub const KMS_SSH_SECRET_MAX: usize = 1024;
27
28/// Validate an admin-chosen key name: 1..=64 bytes of `[A-Za-z0-9_-]`.
29pub fn validate_ssh_key_name(name: &[u8]) -> bool {
30    !name.is_empty()
31        && name.len() <= KMS_SSH_KEY_NAME_MAX
32        && name
33            .iter()
34            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'_' | b'-'))
35}
36
37/// Validate an SSH userauth username: 1..=31 bytes of `[A-Za-z0-9._-]`.
38pub fn validate_ssh_username(username: &[u8]) -> bool {
39    !username.is_empty()
40        && username.len() <= KMS_SSH_USERNAME_MAX
41        && username
42            .iter()
43            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-'))
44}
45
46/// Canonical number of chunks for a transfer of `total_len` bytes:
47/// `ceil(total_len / 96)`. Zero bytes need zero chunks (never a valid transfer).
48pub const fn ssh_chunk_count(total_len: u32) -> u32 {
49    total_len.div_ceil(KMS_SSH_CHUNK_LEN as u32)
50}
51
52/// Canonical length of chunk `chunk_index` of a `total_len`-byte transfer:
53/// 96 for every chunk but the last, the remainder for the last, and `None` for
54/// an index past the end. One transfer has exactly one valid chunk sequence.
55pub const fn ssh_chunk_len(total_len: u32, chunk_index: u32) -> Option<u16> {
56    let count = ssh_chunk_count(total_len);
57    if chunk_index >= count {
58        return None;
59    }
60    if chunk_index + 1 < count {
61        Some(KMS_SSH_CHUNK_LEN as u16)
62    } else {
63        Some((total_len - KMS_SSH_CHUNK_LEN as u32 * chunk_index) as u16)
64    }
65}
