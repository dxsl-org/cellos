24fn read_32(bytes: &[u8], at: usize) -> [u8; 32] {
25    let mut out = [0u8; 32];
26    out.copy_from_slice(&bytes[at..at + 32]);
27    out
28}

30fn read_u16(bytes: &[u8], at: usize) -> u16 {
31    u16::from_le_bytes([bytes[at], bytes[at + 1]])
32}

34fn read_u32(bytes: &[u8], at: usize) -> u32 {
35    u32::from_le_bytes([bytes[at], bytes[at + 1], bytes[at + 2], bytes[at + 3]])
36}

38fn read_u64(bytes: &[u8], at: usize) -> u64 {
39    u64::from_le_bytes(bytes[at..at + 8].try_into().expect("fixed KMS u64"))
40}
