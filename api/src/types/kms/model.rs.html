1macro_rules! wire_enum {
2    ($(#[$meta:meta])* $vis:vis enum $name:ident : $repr:ty {
3        $($(#[$variant_meta:meta])* $variant:ident = $value:expr),+ $(,)?
4    }) => {
5        $(#[$meta])*
6        #[repr($repr)]
7        #[derive(Debug, Clone, Copy, PartialEq, Eq)]
8        $vis enum $name {
9            $($(#[$variant_meta])* $variant = $value),+
10        }
11
12        impl TryFrom<$repr> for $name {
13            type Error = $repr;
14
15            fn try_from(value: $repr) -> Result<Self, $repr> {
16                match value {
17                    $(x if x == Self::$variant as $repr => Ok(Self::$variant),)+
18                    _ => Err(value),
19                }
20            }
21        }
22    };
23}
