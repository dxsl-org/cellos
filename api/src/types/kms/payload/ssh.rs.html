2//! Fixed wire payloads for the single KMS-custodied OpenSSH ed25519 identity.
3//!
4//! Opcodes 15-24 are purpose-specific: there is no generic upload, generic
5//! sign, algorithm, key-handle, or raw-message field. The signing request
6//! carries only typed userauth state (session id, username) plus the expected
7//! public key; KMS reconstructs the RFC 4252 section 7 input itself.
8//!
9//! Every payload has a fixed `LEN`. Decoders are strict: exact length, zero
10//! reserved bytes, a length byte followed by zero padding for variable fields,
11//! validated charsets, and the value bounds frozen in [`crate::kms`]'s SSH
12//! constants. Stateful rules (floors, handle ownership, order) belong to the
13//! service, not to these codecs. A zero transfer handle in a request is
14//! structurally valid and is the service's `SshHandleInvalid`.
15
16use super::{put_u16, put_u32, put_u64, read_32, read_u16, read_u32, read_u64};
17use crate::kms::ssh::{
18    ssh_chunk_count, validate_ssh_key_name, validate_ssh_username, KMS_SSH_CHUNK_LEN,
19    KMS_SSH_IMPORT_MAX_LEN, KMS_SSH_KEY_NAME_MAX, KMS_SSH_PUBLIC_KEY_LEN, KMS_SSH_SECRET_MAX,
20    KMS_SSH_SESSION_ID_LEN, KMS_SSH_SIGNATURE_LEN, KMS_SSH_USERNAME_MAX,
21};
22use crate::kms::{
23    KmsCapabilityReadiness, KmsCustodyPosture, KmsKeyAlgorithm, KmsProviderKind, KmsSshKeyCustody,
24};

26fn is_zero(bytes: &[u8]) -> bool {
27    bytes.iter().all(|byte| *byte == 0)
28}

30/// Decode a `len` byte plus zero-padded name field. `allow_empty` admits
31/// `len == 0` (the leaf status query); otherwise the name must validate.
32fn decode_name(len: u8, field: &[u8], allow_empty: bool) -> Option<[u8; KMS_SSH_KEY_NAME_MAX]> {
33    let len = len as usize;
34    if field.len() != KMS_SSH_KEY_NAME_MAX || len > KMS_SSH_KEY_NAME_MAX {
35        return None;
36    }
37    if !is_zero(&field[len..])
38        || !((len == 0 && allow_empty) || validate_ssh_key_name(&field[..len]))
39    {
40        return None;
41    }
42    let mut name = [0u8; KMS_SSH_KEY_NAME_MAX];
43    name[..len].copy_from_slice(&field[..len]);
44    Some(name)
45}
46
47fn decode_username(len: u8, field: &[u8]) -> Option<[u8; KMS_SSH_USERNAME_MAX]> {
48    let len = len as usize;
49    if field.len() != KMS_SSH_USERNAME_MAX || len > KMS_SSH_USERNAME_MAX {
50        return None;
51    }
52    if !is_zero(&field[len..]) || !validate_ssh_username(&field[..len]) {
53        return None;
54    }
55    let mut username = [0u8; KMS_SSH_USERNAME_MAX];
56    username[..len].copy_from_slice(&field[..len]);
57    Some(username)
58}

60/// `BeginSshKeyImport` request: admin-chosen name, blob length, blob SHA-256.
61#[repr(C)]
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub struct SshImportBeginRequestPayload {
64    /// Nonzero and strictly above the caller's floor (enforced by the service).
65    pub request_id: u64,
66    /// Complete OpenSSH blob length, `1..=KMS_SSH_IMPORT_MAX_LEN`.
67    pub total_len: u32,
68    pub name_len: u8,
69    pub reserved: [u8; 3],
70    /// SHA-256 of the complete binary OpenSSH blob; never all zero.
71    pub blob_sha256: [u8; 32],
72    pub name: [u8; KMS_SSH_KEY_NAME_MAX],
73}
74
75impl SshImportBeginRequestPayload {
76    pub const LEN: usize = 112;
77
78    pub fn encode(&self) -> [u8; Self::LEN] {
79        let mut out = [0u8; Self::LEN];
80        put_u64(&mut out, 0, self.request_id);
81        put_u32(&mut out, 8, self.total_len);
82        out[12] = self.name_len;
83        out[13..16].copy_from_slice(&self.reserved);
84        out[16..48].copy_from_slice(&self.blob_sha256);
85        out[48..112].copy_from_slice(&self.name);
86        out
87    }
88
89    pub fn decode(bytes: &[u8]) -> Option<Self> {
90        if bytes.len() != Self::LEN || !is_zero(&bytes[13..16]) {
91            return None;
92        }
93        let request_id = read_u64(bytes, 0);
94        let total_len = read_u32(bytes, 8);
95        let blob_sha256 = read_32(bytes, 16);
96        if request_id == 0
97            || total_len == 0
98            || total_len as usize > KMS_SSH_IMPORT_MAX_LEN
99            || is_zero(&blob_sha256)
100        {
101            return None;
102        }
103        Some(Self {
104            request_id,
105            total_len,
106            name_len: bytes[12],
107            reserved: [0; 3],
108            blob_sha256,
109            name: decode_name(bytes[12], &bytes[48..112], false)?,
110        })
111    }
112
113    pub fn name(&self) -> &[u8] {
114        &self.name[..self.name_len as usize]
115    }
116}

118/// `BeginSshKeyImport`, `BeginSshUnlock` and `BeginTrustQuery` response: the
119/// one-shot transfer handle and the canonical chunk plan.
120#[repr(C)]
121#[derive(Debug, Clone, Copy, PartialEq, Eq)]
122pub struct SshTransferBeginResponsePayload {
123    /// Nonzero, bound to the opening principal; not a secret.
124    pub handle: u64,
125    /// Import: blob length. Unlock: secret length. Trust: query length.
126    pub total_len: u32,
127    /// `ceil(total_len / 96)`.
128    pub chunk_count: u32,
129    /// Always [`KMS_SSH_CHUNK_LEN`].
130    pub chunk_capacity: u16,
131    pub reserved: [u8; 6],
132}
133
134impl SshTransferBeginResponsePayload {
135    pub const LEN: usize = 24;
136
137    pub fn encode(&self) -> [u8; Self::LEN] {
138        let mut out = [0u8; Self::LEN];
139        put_u64(&mut out, 0, self.handle);
140        put_u32(&mut out, 8, self.total_len);
141        put_u32(&mut out, 12, self.chunk_count);
142        put_u16(&mut out, 16, self.chunk_capacity);
143        out[18..24].copy_from_slice(&self.reserved);
144        out
145    }
146
147    /// Strict decode: nonzero handle, total within the import bound, the
148    /// canonical chunk count, the frozen capacity, and zero reserved bytes.
149    pub fn decode(bytes: &[u8]) -> Option<Self> {
150        if bytes.len() != Self::LEN || !is_zero(&bytes[18..24]) {
151            return None;
152        }
153        let handle = read_u64(bytes, 0);
154        let total_len = read_u32(bytes, 8);
155        if handle == 0
156            || total_len == 0
157            || total_len as usize > KMS_SSH_IMPORT_MAX_LEN
158            || read_u32(bytes, 12) != ssh_chunk_count(total_len)
159            || read_u16(bytes, 16) as usize != KMS_SSH_CHUNK_LEN
160        {
161            return None;
162        }
163        Some(Self {
164            handle,
165            total_len,
166            chunk_count: read_u32(bytes, 12),
167            chunk_capacity: read_u16(bytes, 16),
168            reserved: [0; 6],
169        })
170    }
171}

173/// `ImportSshKeyChunk`, `SshUnlockChunk` and `TrustQueryChunk` request: one
174/// ordered chunk.
175///
176/// Canonical chunking: every chunk but the last carries exactly 96 bytes, so
177/// one blob has exactly one valid chunk sequence. That rule is checked by the
178/// service against the recorded total; the codec checks the per-chunk bound.
179#[repr(C)]
180#[derive(Debug, Clone, Copy, PartialEq, Eq)]
181pub struct SshChunkRequestPayload {
182    pub handle: u64,
183    pub chunk_index: u32,
184    /// `1..=96`.
185    pub chunk_len: u16,
186    pub reserved: u16,
187    /// Data, zero after `chunk_len`.
188    pub chunk: [u8; KMS_SSH_CHUNK_LEN],
189}
190
191impl SshChunkRequestPayload {
192    pub const LEN: usize = 112;
193
194    pub fn encode(&self) -> [u8; Self::LEN] {
195        let mut out = [0u8; Self::LEN];
196        put_u64(&mut out, 0, self.handle);
197        put_u32(&mut out, 8, self.chunk_index);
198        put_u16(&mut out, 12, self.chunk_len);
199        put_u16(&mut out, 14, self.reserved);
200        out[16..112].copy_from_slice(&self.chunk);
201        out
202    }
203
204    pub fn decode(bytes: &[u8]) -> Option<Self> {
205        if bytes.len() != Self::LEN || read_u16(bytes, 14) != 0 {
206            return None;
207        }
208        let chunk_len = read_u16(bytes, 12) as usize;
209        if chunk_len == 0 || chunk_len > KMS_SSH_CHUNK_LEN || !is_zero(&bytes[16 + chunk_len..]) {
210            return None;
211        }
212        Some(Self {
213            handle: read_u64(bytes, 0),
214            chunk_index: read_u32(bytes, 8),
215            chunk_len: read_u16(bytes, 12),
216            reserved: 0,
217            chunk: bytes[16..112].try_into().expect("fixed chunk body"),
218        })
219    }
220
221    pub fn data(&self) -> &[u8] {
222        &self.chunk[..self.chunk_len as usize]
223    }
224}

226/// `ImportSshKeyChunk`, `SshUnlockChunk` and `TrustQueryChunk` response: running
227/// transfer state.
228#[repr(C)]
229#[derive(Debug, Clone, Copy, PartialEq, Eq)]
230pub struct SshChunkAckPayload {
231    pub handle: u64,
232    pub next_chunk_index: u32,
233    /// Cumulative bytes received, `1..=KMS_SSH_IMPORT_MAX_LEN`.
234    pub received_len: u32,
235}
236
237impl SshChunkAckPayload {
238    pub const LEN: usize = 16;
239
240    pub fn encode(&self) -> [u8; Self::LEN] {
241        let mut out = [0u8; Self::LEN];
242        put_u64(&mut out, 0, self.handle);
243        put_u32(&mut out, 8, self.next_chunk_index);
244        put_u32(&mut out, 12, self.received_len);
245        out
246    }
247
248    /// Strict decode: nonzero handle, cumulative bytes within bounds, and the
249    /// next index that canonical chunking implies (`ceil(received / 96)`).
250    pub fn decode(bytes: &[u8]) -> Option<Self> {
251        if bytes.len() != Self::LEN {
252            return None;
253        }
254        let handle = read_u64(bytes, 0);
255        let next_chunk_index = read_u32(bytes, 8);
256        let received_len = read_u32(bytes, 12);
257        if handle == 0
258            || received_len == 0
259            || received_len as usize > KMS_SSH_IMPORT_MAX_LEN
260            || next_chunk_index != ssh_chunk_count(received_len)
261        {
262            return None;
263        }
264        Some(Self {
265            handle,
266            next_chunk_index,
267            received_len,
268        })
269    }
270}

337/// `CommitSshKeyImport` and `GetSshKeyStatus` response: public SSH leaf facts.
338///
339/// Independent of C2C X25519 and Relay P-256 readiness. Carries no private,
340/// wrapped, or passphrase material.
341#[repr(C)]
342#[derive(Debug, Clone, Copy, PartialEq, Eq)]
343pub struct SshKeyStatusPayload {
344    pub algorithm: KmsKeyAlgorithm,
345    pub readiness: KmsCapabilityReadiness,
346    pub provider: KmsProviderKind,
347    pub posture: KmsCustodyPosture,
348    /// 1 only when the named key is stored; 0 for the leaf query.
349    pub key_present: u8,
350    pub custody: KmsSshKeyCustody,
351    /// Stored keys; at most one (exactly one identity per store).
352    pub key_count: u16,
353    /// Bumps on every import and delete; resets at boot at T0.
354    pub store_revision: u64,
355    /// All zero when `key_present == 0`.
356    pub public_key: [u8; KMS_SSH_PUBLIC_KEY_LEN],
357    /// SHA-256 of the 51-byte SSH wire public key blob; all zero when
358    /// `key_present == 0`.
359    pub fingerprint_sha256: [u8; 32],
360}
361
362impl SshKeyStatusPayload {
363    pub const LEN: usize = 80;
364
365    pub fn encode(&self) -> [u8; Self::LEN] {
366        let mut out = [0u8; Self::LEN];
367        out[0] = self.algorithm as u8;
368        out[1] = self.readiness as u8;
369        out[2] = self.provider as u8;
370        out[3] = self.posture as u8;
371        out[4] = self.key_present;
372        out[5] = self.custody as u8;
373        put_u16(&mut out, 6, self.key_count);
374        put_u64(&mut out, 8, self.store_revision);
375        out[16..48].copy_from_slice(&self.public_key);
376        out[48..80].copy_from_slice(&self.fingerprint_sha256);
377        out
378    }
379
380    /// Strict decode: the SSH algorithm, known enum values, and the presence
381    /// invariants (no key means no custody, public key, or fingerprint; a
382    /// present key means a custody mode and exactly one stored key).
383    pub fn decode(bytes: &[u8]) -> Option<Self> {
384        if bytes.len() != Self::LEN {
385            return None;
386        }
387        let algorithm = KmsKeyAlgorithm::try_from(bytes[0]).ok()?;
388        let custody = KmsSshKeyCustody::try_from(bytes[5]).ok()?;
389        let key_count = read_u16(bytes, 6);
390        let public_key = read_32(bytes, 16);
391        let fingerprint_sha256 = read_32(bytes, 48);
392        if algorithm != KmsKeyAlgorithm::SshEd25519 || key_count > 1 {
393            return None;
394        }
395        match bytes[4] {
396            0 if custody == KmsSshKeyCustody::None
397                && is_zero(&public_key)
398                && is_zero(&fingerprint_sha256) => {}
399            1 if custody != KmsSshKeyCustody::None && key_count == 1 => {}
400            _ => return None,
401        }
402        Some(Self {
403            algorithm,
404            readiness: KmsCapabilityReadiness::try_from(bytes[1]).ok()?,
405            provider: KmsProviderKind::try_from(bytes[2]).ok()?,
406            posture: KmsCustodyPosture::try_from(bytes[3]).ok()?,
407            key_present: bytes[4],
408            custody,
409            key_count,
410            store_revision: read_u64(bytes, 8),
411            public_key,
412            fingerprint_sha256,
413        })
414    }
415}

612        let mut out = [0u8; Self::LEN];
613        put_u64(&mut out, 0, self.request_id);
614        out[8..72].copy_from_slice(&self.signature);
615        out[72..104].copy_from_slice(&self.message_sha256);
616        out
617    }
618
619    pub fn decode(bytes: &[u8]) -> Option<Self> {
620        if bytes.len() != Self::LEN {
621            return None;
622        }
623        let request_id = read_u64(bytes, 0);
624        if request_id == 0 {
625            return None;
626        }
627        let mut signature = [0u8; KMS_SSH_SIGNATURE_LEN];
628        signature.copy_from_slice(&bytes[8..72]);
629        Some(Self {
630            request_id,
631            signature,
632            message_sha256: read_32(bytes, 72),
633        })
634    }
635}
