4/// Supervisor-only rotation request with optimistic revision protection.
5#[repr(C)]
6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7pub struct RotateNodeIdentityRequestPayload {
8    pub reason: RotateNodeIdentityReason,
9    pub reserved0: u8,
10    pub flags: u16,
11    pub expected_blob_revision: u64,
12}
13
14impl RotateNodeIdentityRequestPayload {
15    pub const LEN: usize = 16;
16
17    pub fn encode(&self) -> [u8; Self::LEN] {
18        let mut out = [0u8; Self::LEN];
19        out[0] = self.reason as u8;
20        out[1] = self.reserved0;
21        put_u16(&mut out, 2, self.flags);
22        put_u64(&mut out, 8, self.expected_blob_revision);
23        out
24    }
25
26    pub fn decode(bytes: &[u8]) -> Option<Self> {
27        if bytes.len() != Self::LEN || bytes[1] != 0 || read_u32(bytes, 4) != 0 {
28            return None;
29        }
30        Some(Self {
31            reason: RotateNodeIdentityReason::try_from(bytes[0]).ok()?,
32            reserved0: 0,
33            flags: read_u16(bytes, 2),
34            expected_blob_revision: read_u64(bytes, 8),
35        })
36    }
37}

39/// Rotation result. A new public identity always requires peer re-enrollment.
40#[repr(C)]
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub struct RotateNodeIdentityResponsePayload {
43    pub new_public_key: [u8; 32],
44    pub blob_revision: u64,
45    pub re_enroll_required: u8,
46    pub reserved: [u8; 7],
47}
48
49impl RotateNodeIdentityResponsePayload {
50    pub const LEN: usize = 48;
51
52    pub fn encode(&self) -> [u8; Self::LEN] {
53        let mut out = [0u8; Self::LEN];
54        out[..32].copy_from_slice(&self.new_public_key);
55        put_u64(&mut out, 32, self.blob_revision);
56        out[40] = self.re_enroll_required;
57        out[41..48].copy_from_slice(&self.reserved);
58        out
59    }
60
61    pub fn decode(bytes: &[u8]) -> Option<Self> {
62        if bytes.len() != Self::LEN || !matches!(bytes[40], 0 | 1) || bytes[41..48] != [0; 7] {
63            return None;
64        }
65        Some(Self {
66            new_public_key: read_32(bytes, 0),
67            blob_revision: read_u64(bytes, 32),
68            re_enroll_required: bytes[40],
69            reserved: [0; 7],
70        })
71    }
72}
