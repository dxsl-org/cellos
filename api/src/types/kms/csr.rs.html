2//! Frozen bounded profile and canonical RFC 2986 assembly for the relay CSR.
3//!
4//! KMS and the relay provider independently build byte-identical
5//! CertificationRequestInfo values from these definitions alone. The profile
6//! is deliberately minimal: exactly one subject attribute (CN, UTF8String),
7//! no requested extensions (the managed CA adds `clientAuth` EKU and the
8//! NodeId extension), a P-256 SPKI, and `ecdsa-with-SHA256`. Nothing here can
9//! express an arbitrary subject, extension, scheme, or digest.
10
11/// Maximum relay hostname length (also the CN bound). Frozen.
12pub const RELAY_HOSTNAME_MAX: usize = 64;
13/// Maximum canonical CSR length. Frozen; larger CSRs never leave KMS.
14pub const RELAY_CSR_MAX_LEN: usize = 1024;
15/// CSR chunk capacity inside one fixed 112-byte KMS payload. Frozen.
16pub const RELAY_CSR_CHUNK_LEN: usize = 104;
17/// Maximum certificates in a mounted client chain. Frozen.
18pub const RELAY_CHAIN_MAX_CERTS: usize = 3;
19/// Maximum total bytes of a mounted client chain. Frozen.
20pub const RELAY_CHAIN_MAX_LEN: usize = 12 * 1024;
21
22/// DER content bytes of the exact NodeId extension OID
23/// `1.3.6.1.4.1.55555.1.1` (`06 09 ...`).
24pub const NODE_ID_OID_CONTENT: [u8; 10] =
25    [0x2B, 0x06, 0x01, 0x04, 0x01, 0x83, 0xB2, 0x03, 0x01, 0x01];
26/// DER content bytes of `id-ecPublicKey` (`1.2.840.10045.2.1`).
27const EC_PUBLIC_KEY_OID: [u8; 7] = [0x2A, 0x86, 0x48, 0xCE, 0x3D, 0x02, 0x01];
28/// DER content bytes of `prime256v1` (`1.2.840.10045.3.1.7`).
29const PRIME256V1_OID: [u8; 8] = [0x2A, 0x86, 0x48, 0xCE, 0x3D, 0x03, 0x01, 0x07];
30/// DER content bytes of `ecdsa-with-SHA256` (`1.2.840.10045.4.3.2`).
31const ECDSA_WITH_SHA256_OID: [u8; 8] = [0x2A, 0x86, 0x48, 0xCE, 0x3D, 0x04, 0x03, 0x02];
32/// DER content bytes of the commonName attribute type (`2.5.4.3`).
33const COMMON_NAME_OID: [u8; 3] = [0x55, 0x04, 0x03];
34
35const TAG_INTEGER: u8 = 0x02;
36const TAG_BIT_STRING: u8 = 0x03;
37const TAG_UTF8_STRING: u8 = 0x0C;
38const TAG_SEQUENCE: u8 = 0x30;
39const TAG_SET: u8 = 0x31;
40const TAG_OID: u8 = 0x06;

42/// Upper bound on the canonical CRI for this frozen profile.
43pub const CRI_MAX_LEN: usize = 256;
44
45/// Validate a hostname against the frozen DNS profile.
46///
47/// Lowercase letters, digits, hyphen, and dot only; non-empty labels of at
48/// most 63 characters that neither start nor end with a hyphen; single dots
49/// between labels; no leading or trailing dot; total length within
50/// [`RELAY_HOSTNAME_MAX`].
51pub fn validate_hostname(hostname: &[u8]) -> bool {
52    if hostname.is_empty() || hostname.len() > RELAY_HOSTNAME_MAX {
53        return false;
54    }
55    let mut label_len = 0usize;
56    let mut ends_with_hyphen = false;
57    for (index, &byte) in hostname.iter().enumerate() {
58        match byte {
59            b'.' => {
60                if label_len == 0 || label_len > 63 || ends_with_hyphen {
61                    return false; // empty, oversized, or hyphen-terminated label
62                }
63                if index + 1 == hostname.len() {
64                    return false; // trailing dot
65                }
66                label_len = 0;
67                ends_with_hyphen = false;
68            }
69            b'a'..=b'z' | b'0'..=b'9' => {
70                label_len += 1;
71                ends_with_hyphen = false;
72            }
73            b'-' => {
74                if label_len == 0 {
75                    return false; // hyphen may not open a label
76                }
77                label_len += 1;
78                ends_with_hyphen = true;
79            }
80            _ => return false,
81        }
82    }
83    label_len != 0 && label_len <= 63 && !ends_with_hyphen
84}

86/// Build the canonical 91-byte P-256 SubjectPublicKeyInfo DER from an
87/// uncompressed SEC1 point (65 bytes, `04 || X || Y`).
88///
89/// Returns `(buffer, length)`; the buffer tail beyond `length` is zero.
90pub fn p256_spki_der(sec1_point: &[u8; 65]) -> Option<([u8; 96], usize)> {
91    if sec1_point[0] != 0x04 {
92        return None;
93    }
94    let mut alg_oids = [0u8; 32];
95    let mut oids_len = put_tlv(&mut alg_oids, 0, TAG_OID, &EC_PUBLIC_KEY_OID);
96    oids_len = put_tlv(&mut alg_oids, oids_len, TAG_OID, &PRIME256V1_OID);
97    let mut alg = [0u8; 32];
98    let mut alg_len =
99        put_head(&mut alg, 0, TAG_SEQUENCE, oids_len).expect("fixed algorithm scratch");
100    alg_len += copy_into(&mut alg, alg_len, &alg_oids[..oids_len]);
101    let mut spki = [0u8; 96];
102    let mut spki_len = put_head(&mut spki, 0, TAG_SEQUENCE, alg_len + 68)?;
103    spki_len += copy_into(&mut spki, spki_len, &alg[..alg_len]);
104    spki_len = put_head(&mut spki, spki_len, TAG_BIT_STRING, 66)?;
105    spki[spki_len] = 0x00;
106    spki_len += 1;
107    spki_len += copy_into(&mut spki, spki_len, sec1_point);
108    debug_assert_eq!(spki_len, 91);
109    Some((spki, spki_len))
110}

112/// Canonical RFC 2986 CertificationRequestInfo for the frozen relay profile.
113///
114/// `SEQUENCE { INTEGER 0, SEQUENCE { SET { SEQ { CN, UTF8String(host) } } },
115/// SPKI, [0] IMPLICIT SET OF Attribute (empty) }`.
116///
117/// The subject is a full `Name ::= SEQUENCE OF RelativeDistinguishedName`,
118/// and the `[0]` attributes element is mandatory even when empty (`A0 00`);
119/// it is never omitted.
120/// Returns `(buffer, length)`; the buffer tail beyond `length` is zero.
121pub fn canonical_relay_cri(
122    hostname: &[u8],
123    sec1_point: &[u8; 65],
124) -> Option<([u8; CRI_MAX_LEN], usize)> {
125    if !validate_hostname(hostname) {
126        return None;
127    }
128    let (spki, spki_len) = p256_spki_der(sec1_point)?;
129    let mut body = [0u8; CRI_MAX_LEN];
130    let mut len = put_tlv(&mut body, 0, TAG_INTEGER, &[0x00]);
131    // Name: SEQUENCE { SET { SEQUENCE { OID cn, UTF8String hostname } } }.
132    let attr_len = 2 + COMMON_NAME_OID.len() + 2 + hostname.len();
133    let rdn_content_len = 2 + attr_len;
134    let name_content_len = 2 + rdn_content_len;
135    len = put_head(&mut body, len, TAG_SEQUENCE, name_content_len)?;
136    len = put_head(&mut body, len, TAG_SET, rdn_content_len)?;
137    len = put_head(&mut body, len, TAG_SEQUENCE, attr_len)?;
138    len = put_tlv(&mut body, len, TAG_OID, &COMMON_NAME_OID);
139    len = put_tlv(&mut body, len, TAG_UTF8_STRING, hostname);
140    len += copy_into(&mut body, len, &spki[..spki_len]);
141    // Mandatory empty [0] IMPLICIT Attributes element.
142    len += copy_into(&mut body, len, &[0xA0, 0x00]);
143    let mut cri = [0u8; CRI_MAX_LEN];
144    let cri_len = put_head(&mut cri, 0, TAG_SEQUENCE, len)?;
145    cri_len_checked(cri, cri_len, &body[..len])
146}

148fn cri_len_checked(
149    mut cri: [u8; CRI_MAX_LEN],
150    head_len: usize,
151    body: &[u8],
152) -> Option<([u8; CRI_MAX_LEN], usize)> {
153    if head_len + body.len() > CRI_MAX_LEN {
154        return None;
155    }
156    let total = head_len + copy_into(&mut cri, head_len, body);
157    Some((cri, total))
158}

160/// DER-encode an ECDSA-Sig-Value from raw big-endian `r||s` scalars,
161/// stripping leading zero octets as required by DER INTEGER rules.
162///
163/// Returns `(buffer, length)` holding the complete `SEQUENCE { r, s }`.
164pub fn der_ecdsa_signature(raw: &[u8; 64]) -> ([u8; 72], usize) {
165    let mut body = [0u8; 70];
166    let mut len = put_integer(&mut body, 0, &raw[..32]);
167    len = put_integer(&mut body, len, &raw[32..]);
168    let mut out = [0u8; 72];
169    let head = der_seq_head(len as u8);
170    out[..head.len()].copy_from_slice(&head);
171    let total = head.len() + copy_into(&mut out, head.len(), &body[..len]);
172    (out, total)
173}

175/// Assemble the full canonical PKCS#10 CSR:
176/// `SEQUENCE { CRI, SEQ { OID ecdsa-with-SHA256 }, BIT STRING sig }`.
177///
178/// Returns `(buffer, length)`; the buffer tail beyond `length` is zero.
179pub fn assemble_relay_csr(
180    cri: &[u8],
181    cri_len: usize,
182    sig_der: &[u8],
183    sig_der_len: usize,
184) -> Option<([u8; RELAY_CSR_MAX_LEN], usize)> {
185    if cri_len == 0 || cri_len > cri.len() || sig_der_len == 0 || sig_der_len > sig_der.len() {
186        return None;
187    }
188    // AlgorithmIdentifier: SEQUENCE head plus the full OID TLV inside it.
189    let alg_len = 4 + ECDSA_WITH_SHA256_OID.len();
190    let bit_len = 3 + sig_der_len; // tag+len header, unused-bits byte, scalars
191    let body_len = cri_len + alg_len + bit_len;
192    if body_len > RELAY_CSR_MAX_LEN {
193        return None;
194    }
195    let mut out = [0u8; RELAY_CSR_MAX_LEN];
196    let mut len = put_head(&mut out, 0, TAG_SEQUENCE, body_len)?;
197    len += copy_into(&mut out, len, &cri[..cri_len]);
198    len = put_head(&mut out, len, TAG_SEQUENCE, 2 + ECDSA_WITH_SHA256_OID.len())?;
199    len = put_tlv(&mut out, len, TAG_OID, &ECDSA_WITH_SHA256_OID);
200    len = put_head(&mut out, len, TAG_BIT_STRING, sig_der_len + 1)?;
201    out[len] = 0x00;
202    len += 1;
203    len += copy_into(&mut out, len, &sig_der[..sig_der_len]);
204    let seq_head_len = if body_len < 0x80 {
205        2
206    } else if body_len <= 0xFF {
207        3
208    } else {
209        4
210    };
211    debug_assert_eq!(len, seq_head_len + body_len);
212    Some((out, len))
213}

215fn put_integer(out: &mut [u8], at: usize, value: &[u8]) -> usize {
216    let mut first = 0usize;
217    while first + 1 < value.len() && value[first] == 0 {
218        first += 1;
219    }
220    // A high bit on the leading octet needs a zero pad to stay positive.
221    let pad = (value[first] & 0x80 != 0) as usize;
222    let content_len = value.len() - first + pad;
223    let mut tmp = [0u8; 34];
224    tmp[..pad].fill(0x00);
225    tmp[content_len - (value.len() - first)..content_len].copy_from_slice(&value[first..]);
226    put_tlv(out, at, TAG_INTEGER, &tmp[..content_len])
227}

229fn put_tlv(out: &mut [u8], at: usize, tag: u8, content: &[u8]) -> usize {
230    let written = put_head(out, at, tag, content.len()).expect("fixed TLV scratch");
231    written + copy_into(out, written, content)
232}
233
234fn put_head(out: &mut [u8], at: usize, tag: u8, content_len: usize) -> Option<usize> {
235    let (head, head_len) = der_head(tag, content_len)?;
236    if at + head_len > out.len() {
237        return None;
238    }
239    out[at..at + head_len].copy_from_slice(&head[..head_len]);
240    Some(at + head_len)
241}

243fn der_head(tag: u8, content_len: usize) -> Option<([u8; 5], usize)> {
244    let mut head = [0u8; 5];
245    head[0] = tag;
246    if content_len < 0x80 {
247        head[1] = content_len as u8;
248        Some((head, 2))
249    } else if content_len <= 0xFF {
250        head[1] = 0x81;
251        head[2] = content_len as u8;
252        Some((head, 3))
253    } else {
254        head[1] = 0x82;
255        head[2] = (content_len >> 8) as u8;
256        head[3] = content_len as u8;
257        Some((head, 4))
258    }
259}
260
261fn der_seq_head(content_len: u8) -> [u8; 2] {
262    [TAG_SEQUENCE, content_len]
263}
264
265fn copy_into(out: &mut [u8], at: usize, src: &[u8]) -> usize {
266    out[at..at + src.len()].copy_from_slice(src);
267    src.len()
