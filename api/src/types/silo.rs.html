2//! Purpose-specific wire contract for the development AArch64 Silo provider.
3//!
4//! This is not a general signing interface. The accepted operations are the
5//! Phase 1 TLS 1.3 client CertificateVerify operation, public status, and the
6//! Phase 3 purpose-bound enrollment triple (create/sign-CRI/destroy) over one
7//! pending generation key that never leaves the guest.
8
9/// Fixed IPC frame length for KMS-to-Silo messages.
10pub const DEVELOPMENT_SILO_FRAME_LEN: usize = 192;
11/// Development relay generation exposed by the reference provider.
12pub const DEVELOPMENT_RELAY_GENERATION: u64 = 1;
13/// Development profile bound into every reference-provider request.
14pub const DEVELOPMENT_PROFILE_DIGEST: [u8; 32] = [0x53; 32];
15
16const MAGIC: [u8; 4] = *b"DSLO";
17const VERSION: u8 = 1;
18const PAYLOAD: usize = 24;
19
20/// Typed, canonical request accepted by the development Silo service.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum DevelopmentSiloRequest {
23    /// Return the relay P-256 public key.
24    RelayStatus { request_seq: u64 },
25    /// Sign exactly one TLS 1.3 client CertificateVerify transcript.
26    SignTls13ClientCertificateVerify {
27        request_seq: u64,
28        transcript_hash: [u8; 32],
29        relay_generation: u64,
30        active_profile_digest: [u8; 32],
31        request_id: u64,
32    },
33    /// Create the fresh non-exportable key for one pending generation.
34    ///
35    /// The nonce is fresh admitted entropy per call so a destroyed key can
36    /// never be regenerated from generation facts alone.
37    CreateEnrollmentKey {
38        request_seq: u64,
39        pending_generation: u64,
40        nonce: [u8; 32],
41    },
42    /// Reconstruct the canonical CRI independently and sign it raw.
43    SignEnrollmentCri {
44        request_seq: u64,
45        pending_generation: u64,
46        hostname_len: u8,
47        hostname: [u8; 64],
48    },
49    /// Explicitly destroy the pending generation key.
50    DestroyEnrollmentKey {
51        request_seq: u64,
52        pending_generation: u64,
53    },
54    /// Atomically promote the pending key to the active TLS signer and
55    /// retire the previous active key inside the guest. The new serving
56    /// (generation, policy/profile digest) tuple is bound into promotion so
57    /// status and TLS authorization follow the promoted key dynamically.
58    PromoteEnrollmentKey {
59        request_seq: u64,
60        pending_generation: u64,
61        active_profile_digest: [u8; 32],
62    },
63}

65/// Bounded failures returned by the development Silo boundary.
66#[repr(u8)]
67#[derive(Debug, Clone, Copy, PartialEq, Eq)]
68pub enum DevelopmentSiloError {
69    /// Kernel-attested caller is not the live KMS instance.
70    Unauthorized = 1,
71    /// Sequence is zero, stale, or replayed.
72    Sequence = 2,
73    /// Frame or mailbox bytes are non-canonical.
74    Malformed = 3,
75    /// Guest artifact or VM is unavailable.
76    Unavailable = 4,
77    /// Guest reset or execution faulted; the service is fail-closed.
78    GuestFault = 5,
79    /// Relay generation does not match the development lane.
80    GenerationMismatch = 6,
81    /// Active relay profile does not match the development lane.
82    ProfileMismatch = 7,
83    /// No pending enrollment key exists for the requested generation.
84    NoEnrollmentKey = 8,
85}
86
87impl DevelopmentSiloError {
88    fn from_byte(value: u8) -> Option<Self> {
89        Some(match value {
90            1 => Self::Unauthorized,
91            2 => Self::Sequence,
92            3 => Self::Malformed,
93            4 => Self::Unavailable,
94            5 => Self::GuestFault,
95            6 => Self::GenerationMismatch,
96            7 => Self::ProfileMismatch,
97            8 => Self::NoEnrollmentKey,
98            _ => return None,
99        })
100    }
101}

103/// Typed response emitted by the development Silo service.
104#[derive(Debug, Clone, Copy, PartialEq, Eq)]
105pub enum DevelopmentSiloResponse {
106    /// Relay public status; private or seed material is never returned.
107    RelayStatus {
108        request_seq: u64,
109        response_seq: u64,
110        verifying_key_sec1: [u8; 65],
111    },
112    /// Fixed-width normalized P-256 signature (`r || s`).
113    Tls13ClientCertificateVerify {
114        request_seq: u64,
115        response_seq: u64,
116        signature: [u8; 64],
117    },
118    /// Public point of the freshly created enrollment key.
119    EnrollmentKeyCreated {
120        request_seq: u64,
121        response_seq: u64,
122        verifying_key_sec1: [u8; 65],
123    },
124    /// Signature over the independently reconstructed canonical CRI.
125    EnrollmentCriSigned {
126        request_seq: u64,
127        response_seq: u64,
128        signature: [u8; 64],
129    },
130    /// Destruction acknowledged; no key material or handle remains.
131    EnrollmentKeyDestroyed { request_seq: u64, response_seq: u64 },
132    /// Promotion acknowledged; the new active signer's public point follows.
133    EnrollmentKeyPromoted {
134        request_seq: u64,
135        response_seq: u64,
136        verifying_key_sec1: [u8; 65],
137    },
138    /// A typed bounded failure.
139    Error {
140        request_seq: u64,
141        response_seq: u64,
142        error: DevelopmentSiloError,
143    },
144}

146fn header(request_seq: u64, response_seq: u64) -> [u8; DEVELOPMENT_SILO_FRAME_LEN] {
147    let mut out = [0; DEVELOPMENT_SILO_FRAME_LEN];
148    out[..4].copy_from_slice(&MAGIC);
149    out[4] = VERSION;
150    out[8..16].copy_from_slice(&request_seq.to_le_bytes());
151    out[16..24].copy_from_slice(&response_seq.to_le_bytes());
152    out
153}
154
155fn parse_header(bytes: &[u8]) -> Option<(u8, u8, u64, u64)> {
156    (bytes.len() == DEVELOPMENT_SILO_FRAME_LEN
157        && bytes[..4] == MAGIC
158        && bytes[4] == VERSION
159        && bytes[7] == 0)
160        .then(|| (bytes[5], bytes[6], word(bytes, 8), word(bytes, 16)))
161}
162fn word(bytes: &[u8], at: usize) -> u64 {
163    u64::from_le_bytes(array(bytes, at))
164}
165fn array<const N: usize>(bytes: &[u8], at: usize) -> [u8; N] {
166    bytes[at..at + N].try_into().ok().unwrap()
167}
168fn zero(bytes: &[u8]) -> bool {
169    bytes.iter().all(|byte| *byte == 0)
170}
