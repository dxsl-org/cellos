15const I40E_BAR0_LEN: usize = 0x20_0000;
16const I40E_MMIO_MIN: usize = 0x0013_0000;
17const RESET_POLLS: u32 = 1_000_000;
18
19/// Extra DMA slots, authorized before the rings: admin send queue, admin
20/// receive queue, admin command buffer.
21const ATQ: usize = 0;
22const ARQ: usize = 1;
23const AQ_BUFFER: usize = 2;
24const LAYOUT: Layout<3> = Layout::new([Position::Before; 3]);
25
26pub struct I40eController {
27    mmio: MmioRegion,
28    atq: AdminQueue,
29    /// Programmed into `PF_ARQ*`; the cell does not consume events yet.
30    #[allow(dead_code)]
31    arq: DmaBuf,
32    aq_buffer: DmaBuf,
33    tx_ring: DescRing<TxDesc, TX_SLOTS>,
34    tx_bufs: FrameBufs<TX_SLOTS>,
35    rx_ring: DescRing<RxDesc, RX_SLOTS>,
36    rx_bufs: FrameBufs<RX_SLOTS>,
37    iovas: Iovas<TX_SLOTS, RX_SLOTS, 3>,
38    tx_next: usize,
39    rx_head: usize,
40    atq_head: usize,
41    mac: [u8; 6],
42    link_up: bool,
43}
44
45impl NicDevice for I40eController {
46    const TAG: &'static str = "[i40e]";
47    const VENDOR: u16 = VENDOR_INTEL;
48    const QUERY_IDS: &'static [u16] = &QUERY_DEVICE_IDS;
49    const ABSENT_LINE: Option<&'static str> =
50        Some("no X710/XL710 controller present; driver cell idle");
51    type Identity = Identity;
52    type Error = ViError;
53
54    fn classify(vendor: u16, device: u16) -> Option<Identity> {
55        identity::classify(vendor, device)
56    }
57
58    fn identity_name(id: &Identity) -> &'static str {
59        id.name
60    }
61
62    fn init(probe: &Probe, _id: Identity) -> ViResult<Self> {
63        let mmio = probe.claim_bar0(I40E_BAR0_LEN, I40E_MMIO_MIN);
64        Self::new(mmio, probe.bdf())
65    }
66
67    fn bind_detail(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
68        write!(f, "link_up={} mac={}", self.link_up, MacFmt(self.mac))
69    }
70
71    fn mac(&self) -> [u8; 6] {
72        self.mac
73    }
74
75    fn send(&mut self, frame: &[u8]) -> ViResult<()> {
76        self.send_frame(frame)
77    }
78
79    fn recv(&mut self, out: &mut [u8]) -> usize {
80        self.poll_rx(out).unwrap_or_default()
81    }
82}

84impl I40eController {
85    fn new(mmio: MmioRegion, bdf: u32) -> ViResult<Self> {
86        // 1. Quiesce & reset controller.
87        mmio.write_u32(GLGEN_RST, GLGEN_RST_DEV_ARM)?;
88        for _ in 0..RESET_POLLS {
89            if mmio.read_u32(GLGEN_RST)? & GLGEN_RST_DEV_ARM == 0 {
90                break;
91            }
92            core::hint::spin_loop();
93        }
94
95        // 2. Allocate and authorize every DMA object (admin queues first);
96        //    nothing is programmed before this returns.
97        let DmaSet {
98            tx_ring,
99            tx_bufs,
100            rx_ring,
101            rx_bufs,
102            extra: [atq, arq, aq_buffer],
103            iovas,
104        } = DmaSet::alloc_authorized(&LAYOUT, bdf)?;
105        let mut ctrl = I40eController {
106            mmio,
107            atq: DescRing::new(atq)?,
108            arq,
109            aq_buffer,
110            tx_ring,
111            tx_bufs,
112            rx_ring,
113            rx_bufs,
114            iovas,
115            tx_next: 0,
116            rx_head: 0,
117            atq_head: 0,
118            mac: [0; 6],
119            link_up: false,
120        };
121        ctrl.program_rings()?;
122        ctrl.init_admin_queue()?;
123        ctrl.query_mac_and_link()?;
124        Ok(ctrl)
125    }
126
127    /// Initial DMA programming: admin send queue, admin receive queue, then
128    /// every RX descriptor, all from authorized IOVAs.
129    fn program_rings(&mut self) -> ViResult<()> {
130        let iovas = self.iovas;
131        let m = &self.mmio;
132        let atq = iovas.extra[ATQ];
133        m.write_u32(PF_ATQBAL, atq as u32)?;
134        m.write_u32(PF_ATQBAH, (atq >> 32) as u32)?;
135        m.write_u32(PF_ATQLEN, (AQ_SLOTS as u32) | PF_ATQLEN_ENABLE)?;
136        m.write_u32(PF_ATQH, 0)?;
137        m.write_u32(PF_ATQT, 0)?;
138        let arq = iovas.extra[ARQ];
139        m.write_u32(PF_ARQBAL, arq as u32)?;
140        m.write_u32(PF_ARQBAH, (arq >> 32) as u32)?;
141        m.write_u32(PF_ARQLEN, (AQ_SLOTS as u32) | PF_ARQLEN_ENABLE)?;
142        m.write_u32(PF_ARQH, 0)?;
143        m.write_u32(PF_ARQT, 0)?;
144        for (index, &iova) in iovas.rx_bufs.iter().enumerate() {
145            self.rx_ring.write(
146                index,
147                RxDesc {
148                    pkt_addr: iova,
149                    hdr_addr: 0,
150                },
151            );
152        }
153        Ok(())
154    }
155
156    fn init_admin_queue(&mut self) -> ViResult<()> {
157        let cmd = aq::format_get_version();
158        let resp = post_aq_command(&self.mmio, &self.atq, self.atq_head, cmd)?;
159        self.atq_head = (self.atq_head + 1) % AQ_SLOTS;
160        // Verify API major version is 1.
161        let api_major = (resp.addr_high >> 16) as u16;
162        if api_major != 1 && api_major != 0 {
163            return Err(ViError::NotSupported);
164        }
165        Ok(())
166    }
167
168    fn query_mac_and_link(&mut self) -> ViResult<()> {
169        let iova = self.iovas.extra[AQ_BUFFER];
170        let cmd = aq::format_mac_read(iova, 24);
171        let _ = post_aq_command(&self.mmio, &self.atq, self.atq_head, cmd)?;
172        self.atq_head = (self.atq_head + 1) % AQ_SLOTS;
173
174        let mut mac = [0u8; 6];
175        self.aq_buffer.read_bytes(0, &mut mac);
176        if mac == [0u8; 6] {
177            return Err(ViError::IO);
178        }
179        self.mac = mac;
180
181        let cmd = aq::format_get_link_status();
182        let resp = post_aq_command(&self.mmio, &self.atq, self.atq_head, cmd)?;
183        self.atq_head = (self.atq_head + 1) % AQ_SLOTS;
184        self.link_up = resp.param0 & 1 != 0;
185        Ok(())
186    }
187
188    fn send_frame(&mut self, frame: &[u8]) -> ViResult<()> {
189        if frame.is_empty() || frame.len() > BUF_SIZE {
190            return Err(ViError::InvalidArgument);
191        }
192        let slot = self.tx_next;
193        self.tx_bufs.copy_in(slot, frame)?;
194        compiler_fence(Ordering::Release);
195
196        let iova = self.iovas.tx_bufs[slot];
197        self.tx_ring.write(
198            slot,
199            TxDesc {
200                buffer_addr: iova,
201                cmd_type_offset_bsz: (rings::I40E_TXD_QW1_CMD_EOP | rings::I40E_TXD_QW1_CMD_RS)
202                    | ((frame.len() as u64) << 34),
203            },
204        );
205        compiler_fence(Ordering::Release);
206        self.tx_next = (slot + 1) % TX_SLOTS;
207        self.mmio.write_u32(QTX_TAIL, self.tx_next as u32)?;
208        Ok(())
209    }
210
211    fn poll_rx(&mut self, target: &mut [u8]) -> ViResult<usize> {
212        let slot = self.rx_head;
213        let len = self.rx_bufs.copy_out(slot, target, BUF_SIZE);
214        self.rx_head = (slot + 1) % RX_SLOTS;
215        self.mmio.write_u32(QRX_TAIL, self.rx_head as u32)?;
216        Ok(len)
217    }
218}
