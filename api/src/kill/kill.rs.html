11/// kill <tid> — send cooperative shutdown signal or force-exit a task.
12///
13/// Cooperative (state=Waiting): sends 0xFF via sys_send; the task's recv loop
14/// checks for this sentinel and exits cleanly.
15/// Force (state=Ready/Running): calls sys_force_exit; the kernel terminates the
16/// task immediately.  System Cells with SpawnCap may reject force-exit.
17fn cell_main() {
18    let output = io::ProcessOutput::read_own().unwrap_or_else(|_| syscall::sys_exit(1));
19    let argv = ostd::args();
20    let tid_str = argv.first().map(|arg| arg.as_str()).unwrap_or("");
21
22    if tid_str.is_empty() {
23        emit(output.stderr(), &[b"usage: kill <tid>\n"]);
24        syscall::sys_exit(1);
25    }
26
27    let tid: usize = match tid_str.parse() {
28        Ok(n) => n,
29        Err(_) => {
30            emit(output.stderr(), &[b"kill: invalid tid: ", tid_str.as_bytes(), b"\n"]);
31            syscall::sys_exit(1);
32        }
33    };
34    let mut tid_buf = [0u8; 20];
35    let tid_bytes = decimal(tid as u64, &mut tid_buf);
36
37    // Read process table to determine the target task's state.
38    let mut buf = [ProcessInfo::default(); 32];
39    let count = syscall::sys_get_procs(&mut buf).unwrap_or(0);
40
41    let state = (0..count).find(|&i| buf[i].id == tid).map(|i| buf[i].state);
42
43    match state {
44        Some(2) => {
45            // Task is blocked in a recv call — send cooperative shutdown sentinel.
46            let _ = syscall::sys_send(tid, &[0xFF_u8]);
47            emit(output.stdout(), &[b"kill: sent shutdown signal to ", tid_bytes, b"\n"]);
48        }
49        Some(0) | Some(1) => {
50            // Task is Ready or Running — force terminate.
51            let _ = syscall::sys_force_exit(tid);
52            emit(output.stdout(), &[b"kill: force-exited ", tid_bytes, b"\n"]);
53        }
54        Some(3) => {
55            emit(output.stdout(), &[b"kill: task ", tid_bytes, b" is already terminated\n"]);
56        }
57        _ => {
58            emit(output.stderr(), &[b"kill: task not found: ", tid_bytes, b"\n"]);
59            syscall::sys_exit(1);
60        }
61    }
62
63    syscall::sys_exit(0);
64}

66fn decimal(mut value: u64, buffer: &mut [u8; 20]) -> &[u8] {
67    let mut cursor = buffer.len();
68    loop {
69        cursor -= 1;
70        buffer[cursor] = b'0' + (value % 10) as u8;
71        value /= 10;
72        if value == 0 {
73            break;
74        }
75    }
76    &buffer[cursor..]
77}

79fn emit(stream: &io::OutputStream, parts: &[&[u8]]) {
80    let deadline = match ostd::session::monotonic_ns()
81        .ok()
82        .and_then(|now| now.checked_add(30_000_000_000))
83    {
84        Some(deadline) if deadline != u64::MAX => deadline,
85        _ => syscall::sys_exit(1),
86    };
87    for bytes in parts {
88        if stream.write_all(bytes, deadline).is_err() {
89            syscall::sys_exit(1);
90        }
91    }
