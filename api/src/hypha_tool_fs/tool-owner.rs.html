1//! Owner binding shared by the Hypha tool cells and the `core` that spawns them.
2//!
3//! WHY: `tool-fs` (VfsMutate) and `tool-spawn` (SpawnCap) act for whoever sends
4//! them `AgentToolRequest::Invoke`. Every task's tid is public (`GetProcs`), so an
5//! unbound tool is a confused deputy that lends its authority to any cell.
6//!
7//! Binding: `core` publishes `--owner=<its tid>` as the tool's spawn argv, and the
8//! tool serves only messages whose kernel-stamped sender equals that tid. This is
9//! sound because (1) only the spawner can publish a child's argv — the kernel
10//! moves the stash into the child's private slot at spawn, (2) the receive path
11//! stamps the real sender, never a payload claim, and (3) tids are never
12//! re-issued within one boot, so the binding cannot outlive `core` by being
13//! inherited by a later task.
14//!
15//! `core` cannot read its own tid from the kernel; it learns it from the
16//! capability-free `tool-sys`, which echoes the kernel-stamped sender of a
17//! `caller_tid` request as `{"tid":N}` ([`parse_caller_tid`]).
18
19use alloc::string::String;
20
21/// The single argv entry a bound tool is spawned with.
22const OWNER_FLAG: &str = "--owner=";
23/// `tool-sys`'s reply to `caller_tid`, around the decimal tid.
24const CALLER_TID_PREFIX: &str = "{\"tid\":";
25const CALLER_TID_SUFFIX: &str = "}";
26
27/// The argv entry that binds a tool to `tid`.
28pub fn owner_arg(tid: usize) -> String {
29    alloc::format!("{OWNER_FLAG}{tid}")
30}
31
32/// The owner a tool was spawned with, or `None` when the argv is anything other
33/// than exactly one well-formed `--owner=<nonzero decimal>` entry. `None` means
34/// "unbound": the tool must refuse to serve at all.
35pub fn parse_owner(argv: &[String]) -> Option<usize> {
36    let [only] = argv else {
37        return None;
38    };
39    nonzero_decimal(only.strip_prefix(OWNER_FLAG)?)
40}
41
42/// Whether a request whose kernel-stamped sender is `sender` may be served by a
43/// tool bound to `owner`. Never pass a tid taken from a message payload.
44pub fn admits(owner: usize, sender: usize) -> bool {
45    sender == owner
46}
47
48/// Parse `tool-sys`'s `caller_tid` reply. Strict: anything but `{"tid":N}` with a
49/// nonzero decimal `N` is `None`.
50pub fn parse_caller_tid(result_json: &str) -> Option<usize> {
51    let digits = result_json
52        .strip_prefix(CALLER_TID_PREFIX)?
53        .strip_suffix(CALLER_TID_SUFFIX)?;
54    nonzero_decimal(digits)
55}
56
57fn nonzero_decimal(digits: &str) -> Option<usize> {
58    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) {
59        return None;
60    }
61    digits.parse::<usize>().ok().filter(|tid| *tid != 0)
62}
