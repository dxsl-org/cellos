1//! `/srv/sshd/authorized_keys`: the only source of login authority.
2//!
3//! Format, one key per line: `ssh-ed25519 <base64 blob> [comment]`. Blank lines
4//! and lines starting with `#` are skipped; a trailing `\r` is tolerated. Any
5//! other key type, option prefix (`from=…`, `command=…`) or malformed blob makes
6//! that line authorize nothing — sshd does not implement OpenSSH key options, so
7//! honouring the key while dropping its restriction would widen what the
8//! administrator wrote.
9
10extern crate alloc;
11use alloc::vec::Vec;
12
13use crate::base64;
14use crate::ssh_packet::PacketReader;
15
16/// Path of the store. `/srv/<service>/` is the per-service convention on the
17/// CellosFS service volume.
18#[cfg(target_os = "none")]
19const PATH: &str = "/srv/sshd/authorized_keys";
20/// Largest file read; a longer file is refused as a whole (no key authorizes).
21#[cfg(target_os = "none")]
22const MAX_FILE: usize = 16 * 1024;
23
24pub const KEY_TYPE: &[u8] = b"ssh-ed25519";
25
26/// The 32-byte key inside an `ssh-ed25519` public key blob
27/// (`string "ssh-ed25519" || string key`), with no trailing bytes.
28pub fn ed25519_from_blob(blob: &[u8]) -> Option<[u8; 32]> {
29    let mut r = PacketReader::new(blob);
30    if r.read_string()? != KEY_TYPE {
31        return None;
32    }
33    let key: [u8; 32] = r.read_string()?.try_into().ok()?;
34    r.is_at_end().then_some(key)
35}
36
37/// One line's key, or `None` when the line authorizes nothing.
38fn parse_line(line: &[u8]) -> Option<[u8; 32]> {
39    let line = line.strip_suffix(b"\r").unwrap_or(line);
40    let mut fields = line
41        .split(|&b| b == b' ' || b == b'\t')
42        .filter(|f| !f.is_empty());
43    if fields.next()? != KEY_TYPE {
44        return None;
45    }
46    ed25519_from_blob(&base64::decode(fields.next()?)?)
47}
48
49/// Every authorized Ed25519 key in `text` (comments and invalid lines skipped).
50pub fn parse(text: &[u8]) -> Vec<[u8; 32]> {
51    text.split(|&b| b == b'\n')
52        .filter(|line| {
53            let first = line.iter().find(|&&b| b != b' ' && b != b'\t');
54            !matches!(first, None | Some(b'#') | Some(b'\r'))
55        })
56        .filter_map(parse_line)
57        .collect()
58}
59
60/// Read the store. Missing, unreadable or oversized → no key (fail closed:
61/// nobody logs in rather than everybody).
62#[cfg(target_os = "none")]
63pub fn load() -> Vec<[u8; 32]> {
64    let mut vfs = ostd::clients::vfs::VfsClient::new();
65    match vfs.read_file_bounded(PATH, MAX_FILE) {
66        Ok(text) => {
67            let keys = parse(&text);
68            if keys.is_empty() {
69                crate::log("[sshd] authorized_keys holds no ssh-ed25519 key: logins refused");
70            }
71            keys
72        }
73        Err(_) => {
74            crate::log("[sshd] /srv/sshd/authorized_keys unreadable: logins refused");
75            Vec::new()
76        }
77    }
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83    use alloc::string::String;
84
85    /// `ssh-ed25519` blob of the key `0x01 0x02 … 0x20`, base64.
86    const B64: &str = "AAAAC3NzaC1lZDI1NTE5AAAAIAECAwQFBgcICQoLDA0ODxAREhMUFRYXGBkaGxwdHh8g";
87
88    fn key() -> [u8; 32] {
89        core::array::from_fn(|i| i as u8 + 1)
90    }
91
92    #[test]
93    fn accepts_valid_lines_with_comments_and_crlf() {
94        let text = alloc::format!(
95            "# admin keys\r\n\r\nssh-ed25519 {B64}\nssh-ed25519 {B64} admin@laptop\r\n  \n"
96        );
97        assert_eq!(parse(text.as_bytes()), [key(), key()]);
98    }
99
100    #[test]
101    fn rejects_bad_base64_and_other_key_types() {
102        let wrong_type_blob = "AAAAB3NzaC1yc2EAAAAgAQIDBAUGBwgJCgsMDQ4PEBESExQVFhcYGRobHB0eHyA=";
103        for line in [
104            alloc::format!("ssh-ed25519 {}!", &B64[..B64.len() - 1]),
105            alloc::format!("ssh-ed25519 {}", &B64[..B64.len() - 2]),
106            alloc::format!("ssh-rsa {B64}"),
107            alloc::format!("ssh-ed25519 {wrong_type_blob}"),
108            alloc::format!("from=\"10.0.0.1\" ssh-ed25519 {B64}"),
109            String::from("ssh-ed25519"),
110        ] {
111            assert!(parse(line.as_bytes()).is_empty(), "{line}");
112        }
113    }
