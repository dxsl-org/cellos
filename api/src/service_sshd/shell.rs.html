1//! The built-in remote shell: a bounded line editor and canned commands.

3extern crate alloc;
4use alloc::format;
5use alloc::string::String;

8/// Longest command line kept; further printable bytes are dropped (not echoed)
9/// so a newline-less stream cannot grow the cell heap.
10pub const MAX_LINE: usize = 256;
11pub const PROMPT: &[u8] = b"Cellos > ";
12pub const BANNER: &[u8] = b"\r\n\x1b[1;36m=== Cellos Remote SSH Server v0.2.1 (x86_64) ===\x1b[0m\r\nType 'help' for built-in commands.\r\n\r\nCellos > ";
13
14/// Feed terminal input into `line`; return everything to echo back (characters,
15/// command output and prompts) as one stream, so one inbound packet yields one
16/// outbound burst rather than a packet per byte.
17pub fn feed(line: &mut Vec<u8>, data: &[u8]) -> Vec<u8> {
18    let mut out = Vec::new();
19    for &b in data {
20        match b {
21            b'\r' | b'\n' => {
22                out.extend_from_slice(b"\r\n");
23                let cmd = core::str::from_utf8(line).unwrap_or("");
24                out.extend_from_slice(execute(cmd.trim()).as_bytes());
25                out.extend_from_slice(PROMPT);
26                line.clear();
27            }
28            0x7F | 0x08 if line.pop().is_some() => out.extend_from_slice(b"\x08 \x08"),
29            0x20..=0x7E if line.len() < MAX_LINE => {
30                line.push(b);
31                out.push(b);
32            }
33            _ => {}
34        }
35    }
36    out
37}

39pub fn execute(cmd: &str) -> String {
40    let Some(name) = cmd.split_whitespace().next() else {
41        return String::new();
42    };
43    match name {
44        "help" => String::from(
45            "Cellos Shell v0.2.1 (SSH Remote Session) — built-in commands:\r\n  System:  uname  ps  uptime  free  clear  help  exit\r\n  Files:   ls /bin\r\n",
46        ),
47        "uname" => String::from("Cellos cellos-kernel 0.2.1 x86_64\r\n"),
48        "uptime" => {
49            let ticks = ostd::syscall::sys_get_scheduler_ticks().unwrap_or(0);
50            format!("uptime: {} seconds ({} ticks)\r\n", ticks / 100, ticks)
51        }
52        "clear" => String::from("\x1b[2J\x1b[H"),
53        "ps" => {
54            let mut rows = [abi::syscall::ProcessInfo::default(); 32];
55            let mut out =
56                String::from("  PID  STATE      NAME\r\n  ---  ---------  ----------------\r\n");
57            if let Ok(count) = ostd::syscall::sys_get_procs(&mut rows) {
58                for row in rows.iter().take(count) {
59                    let state = match row.state {
60                        1 => "Ready",
61                        2 => "Waiting",
62                        3 => "Exited",
63                        4 => "Running",
64                        _ => "Unknown",
65                    };
66                    let name = core::str::from_utf8(&row.name)
67                        .unwrap_or("unknown")
68                        .trim_matches('\0');
69                    out.push_str(&format!("  {:<4} {:<10} {}\r\n", row.id, state, name));
70                }
71            }
72            out
73        }
74        "ls" => String::from(
75            "/bin:\r\n  ahci       cat        compositor config     e1000      echo\r\n  fb-console igb        input      kill       ls         net\r\n  nvme       platform   ps         serial     shell      sshd\r\n  xhci\r\n",
76        ),
77        "free" => String::from("Heap: 4096 KB total, single-address-space isolation\r\n"),
78        "exit" => String::from("Connection closed by foreign host.\r\n"),
79        other => format!("shell: command not found: {}\r\n", other),
80    }
81}

83#[cfg(test)]
84mod tests {
85    use super::*;

87    #[test]
88    fn line_is_capped_and_overflow_is_not_echoed() {
89        let mut line = Vec::new();
90        let echo = feed(&mut line, &[b'a'; MAX_LINE + 50]);
91        assert_eq!(line.len(), MAX_LINE);
92        assert_eq!(echo.len(), MAX_LINE);
93    }
