1//! SSH-2.0 Key Exchange (RFC 4253, RFC 8731) and Host Key Signing.
2//!
3//! Every secret here is fresh per connection: the KEXINIT cookie and the x25519
4//! scalar come from the injected [`Entropy`] source (the kernel's in the cell),
5//! and a failed draw aborts the exchange — there is no fallback value.
6
7#![forbid(unsafe_code)]
8
9extern crate alloc;
10use alloc::vec::Vec;
11use ed25519_compact::KeyPair;
12use sha2::{Digest, Sha256};
13use x25519_dalek::{PublicKey, StaticSecret};

15use crate::ssh_packet::{PacketWriter, SSH_MSG_KEXINIT, SSH_MSG_KEX_ECDH_REPLY, SSH_MSG_NEWKEYS};

17pub const SERVER_VERSION: &[u8] = b"SSH-2.0-Cellos_1.0\r\n";
18pub const SERVER_VERSION_RAW: &[u8] = b"SSH-2.0-Cellos_1.0";
19
20/// Fill the whole buffer with cryptographic random bytes, or return `false`.
21pub type Entropy = fn(&mut [u8]) -> bool;
22
23/// Kernel entropy: every 64-byte chunk (the syscall's per-call cap) must be
24/// filled completely, otherwise the draw failed.
25#[cfg(target_os = "none")]
26pub fn kernel_entropy(buf: &mut [u8]) -> bool {
27    buf.chunks_mut(64)
28        .all(|c| ostd::syscall::sys_get_random(c) == c.len())
29}
30
31/// Why a key exchange was refused.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum KexError {
34    /// The entropy source could not supply a fresh scalar.
35    Entropy,
36    /// `Q_C` is not a 32-byte x25519 point.
37    BadPublicKey,
38    /// `Q_C` is a small-order point: the shared secret would be all-zero and
39    /// known to anyone (RFC 7748 §6.1, RFC 8731 §3).
40    NonContributory,
41}
42
43impl KexError {
44    /// The `SSH_MSG_DISCONNECT` description.
45    pub fn reason(self) -> &'static str {
46        match self {
47            Self::Entropy => "kernel entropy unavailable",
48            Self::BadPublicKey => "bad x25519 public key",
49            Self::NonContributory => "non-contributory x25519 key",
50        }
51    }
52}

54pub struct KexState {
55    pub client_version: Vec<u8>,
56    pub client_kexinit: Vec<u8>,
57    server_kexinit: Vec<u8>,
58    host_keypair: KeyPair,
59    entropy: Entropy,
60    /// H of the first (only) exchange; RFC 4253 §7.2 keeps it for the session.
61    pub session_id: [u8; 32],
62    shared_secret_k: [u8; 32],
63}

65impl KexState {
66    pub fn new(host_keypair: KeyPair, entropy: Entropy) -> Self {
67        Self {
68            client_version: Vec::new(),
69            client_kexinit: Vec::new(),
70            server_kexinit: Vec::new(),
71            host_keypair,
72            entropy,
73            session_id: [0; 32],
74            shared_secret_k: [0; 32],
75        }
76    }

78    /// Server KEXINIT with a fresh random cookie; `None` if entropy failed.
79    pub fn make_server_kexinit(&mut self) -> Option<Vec<u8>> {
80        let mut cookie = [0u8; 16];
81        if !(self.entropy)(&mut cookie) {
82            return None;
83        }
84        let mut w = PacketWriter::new();
85        w.write_byte(SSH_MSG_KEXINIT);
86        w.write_raw(&cookie);
87        w.write_name_list(&["curve25519-sha256", "curve25519-sha256@libssh.org"]);
88        w.write_name_list(&["ssh-ed25519"]);
89        // The only cipher implemented; `none` is never offered.
90        w.write_name_list(&["chacha20-poly1305@openssh.com"]);
91        w.write_name_list(&["chacha20-poly1305@openssh.com"]);
92        // MAC is implicit in chacha20-poly1305; compression is not supported.
93        w.write_name_list(&["none"]);
94        w.write_name_list(&["none"]);
95        w.write_name_list(&["none"]);
96        w.write_name_list(&["none"]);
97        w.write_string(b""); // languages client to server
98        w.write_string(b""); // languages server to client
99        w.write_bool(false); // first_kex_packet_follows
100        w.write_u32(0); // reserved
101
102        let payload = w.as_slice().to_vec();
103        self.server_kexinit = payload.clone();
104        Some(payload)
105    }

107    /// Handle client ECDH init: fresh x25519 scalar, shared secret K, exchange
108    /// hash H signed by the host key. Returns (ECDH_REPLY, NEWKEYS) payloads.
109    pub fn handle_ecdh_init(
110        &mut self,
111        client_pubkey_bytes: &[u8],
112    ) -> Result<(Vec<u8>, Vec<u8>), KexError> {
113        let client_arr: [u8; 32] = client_pubkey_bytes
114            .try_into()
115            .map_err(|_| KexError::BadPublicKey)?;
116        let client_pub = PublicKey::from(client_arr);
117
118        let mut scalar = [0u8; 32];
119        if !(self.entropy)(&mut scalar) {
120            return Err(KexError::Entropy);
121        }
122        let server_secret = StaticSecret::from(scalar);
123        let server_pub = PublicKey::from(&server_secret);
124        let shared_secret = server_secret.diffie_hellman(&client_pub);
125        if !shared_secret.was_contributory() {
126            return Err(KexError::NonContributory);
127        }
128        let k_bytes = shared_secret.as_bytes();
129        let host_key_blob = host_key_blob(&self.host_keypair);
130
131        // H = SHA256(V_C || V_S || I_C || I_S || K_S || Q_C || Q_S || K)
132        let mut hasher = Sha256::new();
133        for string in [
134            &self.client_version[..],
135            SERVER_VERSION_RAW,
136            &self.client_kexinit[..],
137            &self.server_kexinit[..],
138            &host_key_blob[..],
139            client_pubkey_bytes,
140            &server_pub.as_bytes()[..],
141        ] {
142            hasher.update((string.len() as u32).to_be_bytes());
143            hasher.update(string);
144        }
145        let mut mpint_w = PacketWriter::new();
146        mpint_w.write_mpint(k_bytes);
147        hasher.update(mpint_w.as_slice());
148        let h: [u8; 32] = hasher.finalize().into();
149
150        let sig = self.host_keypair.sk.sign(h, None);
151        let mut sig_w = PacketWriter::new();
152        sig_w.write_string(b"ssh-ed25519");
153        sig_w.write_string(sig.as_ref());
154
155        let mut reply = PacketWriter::new();
156        reply.write_byte(SSH_MSG_KEX_ECDH_REPLY);
157        reply.write_string(&host_key_blob);
158        reply.write_string(server_pub.as_bytes());
159        reply.write_string(sig_w.as_slice());
160
161        // Re-keying is not supported, so this is always the first exchange.
162        self.session_id = h;
163        self.shared_secret_k = *k_bytes;
164        Ok((reply.as_slice().to_vec(), [SSH_MSG_NEWKEYS].to_vec()))
165    }
166
167    /// Derive key letter X ('A'..'F') per RFC 4253 §7.2
168    pub fn derive_key(&self, letter: u8, key_len: usize) -> Vec<u8> {
169        let mut out = Vec::with_capacity(key_len);
170        let mut mpint_w = PacketWriter::new();
171        mpint_w.write_mpint(&self.shared_secret_k);
172
173        // K1 = HASH(K || H || X || session_id); H == session_id (single KEX).
174        let mut hasher = Sha256::new();
175        hasher.update(mpint_w.as_slice());
176        hasher.update(self.session_id);
177        hasher.update([letter]);
178        hasher.update(self.session_id);
179        out.extend_from_slice(&hasher.finalize());
180
181        // Kn = HASH(K || H || K1 || … || Kn-1)
182        while out.len() < key_len {
183            let mut hasher2 = Sha256::new();
184            hasher2.update(mpint_w.as_slice());
185            hasher2.update(self.session_id);
186            hasher2.update(&out);
187            out.extend_from_slice(&hasher2.finalize());
188        }
189        out.truncate(key_len);
190        out
191    }
192}

194/// `string "ssh-ed25519" || string pk`.
195pub fn host_key_blob(keypair: &KeyPair) -> Vec<u8> {
196    let mut w = PacketWriter::new();
197    w.write_string(b"ssh-ed25519");
198    w.write_string(keypair.pk.as_ref());
199    w.as_slice().to_vec()
200}
