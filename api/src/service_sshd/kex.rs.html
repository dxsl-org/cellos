13pub const SERVER_VERSION: &[u8] = b"SSH-2.0-Cellos_1.0\r\n";
14pub const SERVER_VERSION_RAW: &[u8] = b"SSH-2.0-Cellos_1.0";

29impl KexState {
30    pub fn new() -> Self {
31        // Deterministic or seeded host key for Cellos
32        let seed = Seed::from_slice(&[
33            0x43, 0x65, 0x6c, 0x6c, 0x6f, 0x73, 0x53, 0x53,
34            0x48, 0x44, 0x65, 0x76, 0x48, 0x6f, 0x73, 0x74,
35            0x4b, 0x65, 0x79, 0x53, 0x65, 0x65, 0x64, 0x32,
36            0x30, 0x32, 0x36, 0x31, 0x30, 0x31, 0x37, 0x01,
37        ]).unwrap();
38        let host_keypair = KeyPair::from_seed(seed);
39
40        Self {
41            client_version: Vec::new(),
42            server_version: SERVER_VERSION_RAW.to_vec(),
43            client_kexinit: Vec::new(),
44            server_kexinit: Vec::new(),
45            host_keypair,
46            server_secret: None,
47            server_public: None,
48            session_id: Vec::new(),
49            shared_secret_k: Vec::new(),
50            exchange_hash_h: Vec::new(),
51        }
52    }

54    /// Build Server Host Key blob: string "ssh-ed25519" || string pk
55    pub fn host_key_blob(&self) -> Vec<u8> {
56        let mut w = PacketWriter::new();
57        w.write_string(b"ssh-ed25519");
58        w.write_string(self.host_keypair.pk.as_ref());
59        w.as_slice().to_vec()
60    }
61
62    /// Create Server KEXINIT payload
63    pub fn make_server_kexinit(&mut self) -> Vec<u8> {
64        let mut w = PacketWriter::new();
65        w.write_byte(SSH_MSG_KEXINIT);
66        // 16 bytes cookie
67        w.write_raw(&[
68            0x10, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77, 0x88,
69            0x99, 0xaa, 0xbb, 0xcc, 0xdd, 0xee, 0xff, 0x01,
70        ]);
71        w.write_name_list(&["curve25519-sha256", "curve25519-sha256@libssh.org"]);
72        w.write_name_list(&["ssh-ed25519"]);
73        w.write_name_list(&["chacha20-poly1305@openssh.com", "none"]);
74        w.write_name_list(&["chacha20-poly1305@openssh.com", "none"]);
75        w.write_name_list(&["none"]);
76        w.write_name_list(&["none"]);
77        w.write_name_list(&["none"]);
78        w.write_name_list(&["none"]);
79        w.write_string(b""); // languages client to server
80        w.write_string(b""); // languages server to client
81        w.write_bool(false); // first_kex_packet_follows
82        w.write_u32(0);      // reserved
83
84        let payload = w.as_slice().to_vec();
85        self.server_kexinit = payload.clone();
86        payload
87    }

89    /// Handle client ECDH init, calculate shared secret K, exchange hash H, and reply
90    pub fn handle_ecdh_init(&mut self, client_pubkey_bytes: &[u8]) -> Option<(Vec<u8>, Vec<u8>)> {
91        if client_pubkey_bytes.len() != 32 {
92            return None;
93        }
94        let mut client_arr = [0u8; 32];
95        client_arr.copy_from_slice(client_pubkey_bytes);
96        let client_pub = PublicKey::from(client_arr);
97
98        // Server ephemeral keys
99        let server_secret = StaticSecret::from([
100            0x71, 0x82, 0x93, 0xa4, 0xb5, 0xc6, 0xd7, 0xe8,
101            0x09, 0x1a, 0x2b, 0x3c, 0x4d, 0x5e, 0x6f, 0x70,
102            0x81, 0x92, 0xa3, 0xb4, 0xc5, 0xd6, 0xe7, 0xf8,
103            0x19, 0x2a, 0x3b, 0x4c, 0x5d, 0x6e, 0x7f, 0x80,
104        ]);
105        let server_pub = PublicKey::from(&server_secret);
106        let shared_secret = server_secret.diffie_hellman(&client_pub);
107        let k_bytes = shared_secret.as_bytes();
108
109        let host_key_blob = self.host_key_blob();
110
111        // Calculate Exchange Hash H = SHA256(V_C || V_S || I_C || I_S || K_S || Q_C || Q_S || K)
112        let mut hasher = Sha256::new();
113        // V_C (string)
114        hasher.update(&(self.client_version.len() as u32).to_be_bytes());
115        hasher.update(&self.client_version);
116        // V_S (string)
117        hasher.update(&(self.server_version.len() as u32).to_be_bytes());
118        hasher.update(&self.server_version);
119        // I_C (string)
120        hasher.update(&(self.client_kexinit.len() as u32).to_be_bytes());
121        hasher.update(&self.client_kexinit);
122        // I_S (string)
123        hasher.update(&(self.server_kexinit.len() as u32).to_be_bytes());
124        hasher.update(&self.server_kexinit);
125        // K_S (string)
126        hasher.update(&(host_key_blob.len() as u32).to_be_bytes());
127        hasher.update(&host_key_blob);
128        // Q_C (string)
129        hasher.update(&(client_pubkey_bytes.len() as u32).to_be_bytes());
130        hasher.update(client_pubkey_bytes);
131        // Q_S (string)
132        hasher.update(&(server_pub.as_bytes().len() as u32).to_be_bytes());
133        hasher.update(server_pub.as_bytes());
134        // K (mpint)
135        let mut mpint_w = PacketWriter::new();
136        mpint_w.write_mpint(k_bytes);
137        hasher.update(mpint_w.as_slice());
138
139        let h_hash = hasher.finalize();
140        let h_bytes = h_hash.as_slice();
141
142        // Sign H with host key Ed25519
143        let sig = self.host_keypair.sk.sign(h_bytes, None);
144        let mut sig_w = PacketWriter::new();
145        sig_w.write_string(b"ssh-ed25519");
146        sig_w.write_string(sig.as_ref());
147        let sig_blob = sig_w.as_slice().to_vec();
148
149        // Build SSH_MSG_KEX_ECDH_REPLY
150        let mut reply = PacketWriter::new();
151        reply.write_byte(SSH_MSG_KEX_ECDH_REPLY);
152        reply.write_string(&host_key_blob);
153        reply.write_string(server_pub.as_bytes());
154        reply.write_string(&sig_blob);
155
156        if self.session_id.is_empty() {
157            self.session_id = h_bytes.to_vec();
158        }
159        self.shared_secret_k = k_bytes.to_vec();
160        self.exchange_hash_h = h_bytes.to_vec();
161
162        // Also build SSH_MSG_NEWKEYS
163        let mut newkeys = PacketWriter::new();
164        newkeys.write_byte(SSH_MSG_NEWKEYS);
165
166        Some((reply.as_slice().to_vec(), newkeys.as_slice().to_vec()))
167    }

169    /// Derive key letter X ('A'..'F') per RFC 4253 §7.2
170    pub fn derive_key(&self, letter: u8, key_len: usize) -> Vec<u8> {
171        let mut out = Vec::with_capacity(key_len);
172        let mut mpint_w = PacketWriter::new();
173        mpint_w.write_mpint(&self.shared_secret_k);
174
175        // K1 = HASH(K || H || X || session_id)
176        let mut hasher = Sha256::new();
177        hasher.update(mpint_w.as_slice());
178        hasher.update(&self.exchange_hash_h);
179        hasher.update(&[letter]);
180        hasher.update(&self.session_id);
181        let h1 = hasher.finalize();
182        out.extend_from_slice(&h1);
183
184        // If key_len > 32, extend with K2 = HASH(K || H || K1)
185        while out.len() < key_len {
186            let mut hasher2 = Sha256::new();
187            hasher2.update(mpint_w.as_slice());
188            hasher2.update(&self.exchange_hash_h);
189            hasher2.update(&out[out.len() - 32..]);
190            let h2 = hasher2.finalize();
191            out.extend_from_slice(&h2);
192        }
193        out.truncate(key_len);
194        out
195    }
