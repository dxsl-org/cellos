1//! Standard-alphabet base64 (RFC 4648 §4): strict decoding for `authorized_keys`
2//! blobs and unpadded encoding for the OpenSSH `SHA256:` fingerprint.
3
4extern crate alloc;
5use alloc::string::String;
6use alloc::vec::Vec;
7
8const ALPHABET: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
9
10fn value(c: u8) -> Option<u32> {
11    match c {
12        b'A'..=b'Z' => Some((c - b'A') as u32),
13        b'a'..=b'z' => Some((c - b'a') as u32 + 26),
14        b'0'..=b'9' => Some((c - b'0') as u32 + 52),
15        b'+' => Some(62),
16        b'/' => Some(63),
17        _ => None,
18    }
19}
20
21/// Decode padded base64. `None` for any character outside the alphabet, a
22/// length that is not a multiple of 4, misplaced `=`, or non-zero trailing
23/// bits: a key line that is not exactly one canonical encoding is refused
24/// rather than guessed at.
25pub fn decode(text: &[u8]) -> Option<Vec<u8>> {
26    if !text.len().is_multiple_of(4) {
27        return None;
28    }
29    let mut out = Vec::with_capacity(text.len() / 4 * 3);
30    let quads = text.len() / 4;
31    for (i, quad) in text.chunks_exact(4).enumerate() {
32        let pad = quad.iter().rev().take_while(|&&c| c == b'=').count();
33        if pad > 2 || (pad > 0 && i + 1 != quads) {
34            return None;
35        }
36        let mut acc = 0u32;
37        for &c in &quad[..4 - pad] {
38            acc = (acc << 6) | value(c)?;
39        }
40        acc <<= 6 * pad as u32;
41        let bytes = acc.to_be_bytes();
42        let keep = 3 - pad;
43        if bytes[1 + keep..].iter().any(|&b| b != 0) {
44            return None;
45        }
46        out.extend_from_slice(&bytes[1..1 + keep]);
47    }
48    Some(out)
49}
50
51/// Encode without `=` padding (the form OpenSSH prints after `SHA256:`).
52pub fn encode_unpadded(bytes: &[u8], out: &mut String) {
53    for chunk in bytes.chunks(3) {
54        let mut acc = [0u8; 3];
55        acc[..chunk.len()].copy_from_slice(chunk);
56        let n = u32::from_be_bytes([0, acc[0], acc[1], acc[2]]);
57        for i in 0..=chunk.len() {
58            out.push(ALPHABET[(n >> (18 - 6 * i)) as usize & 63] as char);
59        }
60    }
61}

63#[cfg(test)]
64mod tests {
65    use super::*;
66
67    #[test]
68    fn round_trips_and_rejects_non_canonical_text() {
69        assert_eq!(decode(b"Zm9vYmFy").unwrap(), b"foobar");
70        assert_eq!(decode(b"Zm9vYg==").unwrap(), b"foob");
71        assert_eq!(decode(b"Zm9vYmE=").unwrap(), b"fooba");
72        for bad in [
73            &b"Zm9vYg="[..],
74            b"Zm9v=mFy",
75            b"Zm9vYh==",
76            b"Zm9v*mFy",
77            b"Zg==Zg==",
78        ] {
79            assert_eq!(decode(bad), None, "{bad:?}");
80        }
81        let mut s = String::new();
82        encode_unpadded(b"foob", &mut s);
83        assert_eq!(s, "Zm9vYg");
84    }
