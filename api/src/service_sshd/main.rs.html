1//! SSH-2.0 server daemon for Cellos.
2//!
3//! Listens on TCP port 22 and serves one connection at a time:
4//! curve25519-sha256 key exchange with a fresh per-connection scalar, the
5//! persisted Ed25519 host key (`host_key`), chacha20-poly1305, and `publickey`
6//! (`ssh-ed25519`) login against `/srv/sshd/authorized_keys` (`auth`).
7//! A connection that has not logged in within [`LOGIN_GRACE_MS`] is dropped so
8//! one idle peer cannot hold the single slot.
9

44/// Time from accept to `USERAUTH_SUCCESS` before the connection is dropped.
45pub const LOGIN_GRACE_MS: u64 = 60_000;

47/// Serial log line. Host unit tests have no kernel to log to.
48pub fn log(msg: &str) {
49    #[cfg(target_os = "none")]
50    ostd::io::println(msg);
51    #[cfg(not(target_os = "none"))]
52    let _ = msg;
53}
54
55pub fn log2(prefix: &str, msg: &str) {
56    #[cfg(target_os = "none")]
57    {
58        ostd::io::print(prefix);
59        ostd::io::println(msg);
60    }
61    #[cfg(not(target_os = "none"))]
62    let _ = (prefix, msg);
63}

68#[cfg(target_os = "none")]
69fn cell_main() {
70    init_custom_heap();
71    log("[sshd] Cellos SSH-2.0 daemon starting...");
72
73    while ostd::service::provider_tid(api::syscall::service::VFS).is_none() {
74        ostd::syscall::sys_yield();
75    }
76    let host_key = match host_key::load_or_create(kex::kernel_entropy) {
77        Ok(keypair) => keypair,
78        Err(why) => {
79            log2("[sshd] refusing to start: ", why);
80            return;
81        }
82    };
83    log2(
84        "[sshd] host key ssh-ed25519 ",
85        &host_key::fingerprint(&host_key),
86    );
87    reactor::run(host_key);
88}
