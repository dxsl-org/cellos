9use crate::session::{SessionPhase, SshSession, DISCONNECT_KEY_EXCHANGE_FAILED, MAX_VERSION_LINE};
10use crate::ssh_packet::PacketReader;

12/// Packets each side sends before the cipher switch (KEXINIT, ECDH, NEWKEYS).
13/// The session admits nothing else before NEWKEYS, so this is exact.
14const KEX_PACKETS: u32 = 3;
15
16impl SshSession {
17    /// Consume the client identification line. `false` while more bytes are needed.
18    pub(crate) fn version_exchange(&mut self, out: &mut Vec<u8>) -> bool {
19        let Some(pos) = self.rx_buf.iter().position(|&b| b == b'\n') else {
20            if self.rx_buf.len() >= MAX_VERSION_LINE {
21                self.close("version line too long");
22            }
23            return false;
24        };
25        let line = &self.rx_buf[..pos];
26        let line = line.strip_suffix(b"\r").unwrap_or(line);
27        if pos + 1 > MAX_VERSION_LINE || !line.starts_with(b"SSH-2.0-") {
28            self.close("bad version line");
29            return false;
30        }
31        self.kex.client_version = line.to_vec();
32        self.rx_buf.drain(..=pos);
33        match self.kex.make_server_kexinit() {
34            Some(kexinit) => {
35                out.extend_from_slice(&self.cipher.pack_outbound(&kexinit));
36                self.phase = SessionPhase::KexInit;
37            }
38            None => self.close("kernel entropy unavailable"),
39        }
40        true
41    }
42
43    /// `KEX_ECDH_INIT`: reply with ECDH_REPLY and our NEWKEYS, or disconnect.
44    pub(crate) fn ecdh_init(&mut self, body: &[u8]) -> Vec<Vec<u8>> {
45        let mut r = PacketReader::new(body);
46        let Some(q_c) = r.read_string().filter(|_| r.is_at_end()) else {
47            return self.disconnect(DISCONNECT_PROTOCOL_ERROR, "malformed ECDH init");
48        };
49        match self.kex.handle_ecdh_init(q_c) {
50            Ok((reply, newkeys)) => {
51                self.phase = SessionPhase::WaitNewKeys;
52                vec![reply, newkeys]
53            }
54            Err(e) => self.disconnect(DISCONNECT_KEY_EXCHANGE_FAILED, e.reason()),
55        }
56    }
57
58    /// The client's NEWKEYS: both directions switch to chacha20-poly1305.
59    pub(crate) fn new_keys(&mut self) -> Vec<Vec<u8>> {
60        let c2s_key = self.kex.derive_key(b'C', 64);
61        let s2c_key = self.kex.derive_key(b'D', 64);
62        self.cipher = CipherMode::new_chacha20(&c2s_key, &s2c_key, KEX_PACKETS, KEX_PACKETS);
63        self.phase = SessionPhase::ServiceRequest;
64        Vec::new()
65    }
66}
