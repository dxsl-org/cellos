1//! RFC 4252 user authentication: `publickey` with `ssh-ed25519` only.
2//!
3//! - Query form (no signature): `USERAUTH_PK_OK` if the key is authorized,
4//!   otherwise a failure.
5//! - Signed form: the signature must verify over the RFC 4252 §7 input built by
6//!   `vissh::userauth::signing_input` (the same bytes the Cellos ssh client and
7//!   KMS sign) against a key listed in `/srv/sshd/authorized_keys`.
8//! - `none` and every other method fail; there is no password method.
9//!
10//! Every failure answers `USERAUTH_FAILURE` naming `publickey` alone with
11//! `partial success = false`. The [`MAX_FAILURES`]th failure disconnects
12//! instead. Like OpenSSH, the client's first `none` probe is not counted.
13//!
14//! The user name is checked (non-empty UTF-8, at most [`MAX_USER`] bytes) and
15//! signed over, but it is not mapped to an account: this is a single-user
16//! device, and any authorized key logs in as that user whatever name it gives.
17
18extern crate alloc;
19use alloc::vec::Vec;
20
21use crate::authorized_keys::{ed25519_from_blob, KEY_TYPE};
22use crate::ssh_packet::{
23    PacketReader, PacketWriter, SSH_MSG_USERAUTH_FAILURE, SSH_MSG_USERAUTH_PK_OK,
24};
25
26/// Failures before the connection is dropped.
27pub const MAX_FAILURES: u32 = 5;
28/// Longest user name: the bound of the shared signing-input builder (and of
29/// KMS, which signs for the client), so all three agree on what can be signed.
30pub const MAX_USER: usize = vissh::userauth::MAX_USERNAME;
31/// The only service a successful login opens.
32pub const SERVICE: &str = "ssh-connection";
33
34/// RFC 4253 §11.1 reason codes used by authentication.
35pub const DISCONNECT_PROTOCOL_ERROR: u32 = 2;
36pub const DISCONNECT_SERVICE_NOT_AVAILABLE: u32 = 7;
37pub const DISCONNECT_NO_MORE_AUTH_METHODS: u32 = 14;
38
39#[derive(Debug, PartialEq, Eq)]
40pub enum AuthReply {
41    /// Send this payload and keep waiting for `USERAUTH_REQUEST`.
42    Continue(Vec<u8>),
43    /// Authenticated: send `USERAUTH_SUCCESS` and open the connection layer.
44    Success,
45    /// Send `SSH_MSG_DISCONNECT` with this reason and close.
46    Disconnect(u32, &'static str),
47}
48
49pub struct Auth {
50    keys: Vec<[u8; 32]>,
51    failures: u32,
52    free_none_used: bool,
53}
54
55impl Auth {
56    pub fn new(keys: Vec<[u8; 32]>) -> Self {
57        Self {
58            keys,
59            failures: 0,
60            free_none_used: false,
61        }
62    }
63
64    /// Handle one `USERAUTH_REQUEST` body (everything after the message byte).
65    pub fn handle(&mut self, session_id: &[u8; 32], body: &[u8]) -> AuthReply {
66        let mut r = PacketReader::new(body);
67        let (Some(user), Some(service), Some(method)) =
68            (r.read_utf8_string(), r.read_utf8_string(), r.read_string())
69        else {
70            return AuthReply::Disconnect(DISCONNECT_PROTOCOL_ERROR, "malformed userauth request");
71        };
72        if service != SERVICE {
73            return AuthReply::Disconnect(DISCONNECT_SERVICE_NOT_AVAILABLE, "unknown service");
74        }
75        if method == b"none" && !self.free_none_used {
76            self.free_none_used = true;
77            return AuthReply::Continue(failure_payload());
78        }
79        if user.is_empty() || user.len() > MAX_USER || method != b"publickey" {
80            return self.fail();
81        }
82        let (Some(signed), Some(algorithm), Some(blob)) =
83            (r.read_bool(), r.read_string(), r.read_string())
84        else {
85            return AuthReply::Disconnect(DISCONNECT_PROTOCOL_ERROR, "malformed publickey request");
86        };
87        let key = match ed25519_from_blob(blob) {
88            Some(key) if algorithm == KEY_TYPE && self.keys.contains(&key) => key,
89            _ => return self.fail(),
90        };
91        if !signed {
92            if !r.is_at_end() {
93                return AuthReply::Disconnect(DISCONNECT_PROTOCOL_ERROR, "trailing data");
94            }
95            let mut w = PacketWriter::new();
96            w.write_byte(SSH_MSG_USERAUTH_PK_OK);
97            w.write_string(algorithm);
98            w.write_string(blob);
99            return AuthReply::Continue(w.as_slice().to_vec());
100        }
101        let Some(signature) = r.read_string().filter(|_| r.is_at_end()) else {
102            return AuthReply::Disconnect(DISCONNECT_PROTOCOL_ERROR, "malformed signature");
103        };
104        match (
105            signature_bytes(signature),
106            vissh::userauth::signing_input(session_id, user, &key),
107        ) {
108            (Some(sig), Some(input)) if vissh::userauth::verify(&key, &input, &sig) => {
109                AuthReply::Success
110            }
111            _ => self.fail(),
112        }
113    }
114
115    fn fail(&mut self) -> AuthReply {
116        self.failures += 1;
117        if self.failures >= MAX_FAILURES {
118            AuthReply::Disconnect(
119                DISCONNECT_NO_MORE_AUTH_METHODS,
120                "too many authentication failures",
121            )
122        } else {
123            AuthReply::Continue(failure_payload())
124        }
125    }
126}
127
128/// The 64 signature bytes of `string "ssh-ed25519" || string sig`.
129fn signature_bytes(blob: &[u8]) -> Option<[u8; 64]> {
130    let mut r = PacketReader::new(blob);
131    if r.read_string()? != KEY_TYPE {
132        return None;
133    }
134    let sig: [u8; 64] = r.read_string()?.try_into().ok()?;
135    r.is_at_end().then_some(sig)
136}
137
138/// `USERAUTH_FAILURE`: `publickey` is the only method that can continue;
139/// partial success is never claimed.
140pub fn failure_payload() -> Vec<u8> {
141    let mut w = PacketWriter::new();
142    w.write_byte(SSH_MSG_USERAUTH_FAILURE);
143    w.write_name_list(&["publickey"]);
144    w.write_bool(false);
145    w.as_slice().to_vec()
146}
