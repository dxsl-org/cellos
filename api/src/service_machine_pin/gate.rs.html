1//! Machine pin gate (ADR-0029 §3, Phase 13 amendment).
2//!
3//! Once trusted init has registered this Cell as `MACHINE_PIN`, the gate
4//! compares the kernel's machine facts (syscall 433, answered only to the live
5//! provider) with the record stored at `/srv/.cellos-machine/pin`:
6//! - no record on a healthy store: first boot, record and continue;
7//! - the same bytes: continue;
8//! - anything else: print the field-level difference and decide safe mode.
9//!
10//! The pin carries no authority. It never changes a profile, driver gate or
11//! DMA decision; safe mode only withholds services, and init does the
12//! withholding after asking this Cell (`MachinePinRequest::Gate`).
13

25/// The catalog's only profile that does not require DMA isolation. The kernel
26/// binds it only when the command line names it and CPUID reports a
27/// hypervisor; without a pin store it warns and continues (CP1 Q5).
28const STORE_OPTIONAL_PROFILE: &[u8] = b"qemu-q35-x86_64";
29/// Scheduler ticks to wait for init's `MACHINE_PIN` registration of this Cell,
30/// which 433 requires; init registers it right after the spawn.
31const REGISTRATION_TICKS: u64 = 500;
32/// Poll backstop for a scheduler-tick clock that never advances.
33const REGISTRATION_POLLS: usize = 5_000_000;
34
35/// What the gate decided for this boot.
36pub(crate) struct Gate {
37    pub(crate) state: PinState,
38    pub(crate) current: Option<MachineFactsRecord>,
39    pub(crate) pinned: Option<[u8; 32]>,
40    /// Console lines, without the `machine-pin: ` prefix.
41    pub(crate) lines: Vec<String>,
42}
43
44pub(crate) fn short(digest: &[u8; 32]) -> String {
45    diff::hex(&digest[..6])
46}
47
48/// The decision table, with the store abstracted. Writes only when the store
49/// positively reports absence.
50pub(crate) fn evaluate(current: Option<MachineFactsRecord>, store: &mut impl PinStore) -> Gate {
51    let mut gate = Gate {
52        state: PinState::FactsUnavailable,
53        current,
54        pinned: None,
55        lines: Vec::new(),
56    };
57    let Some(current) = gate.current.as_ref() else {
58        gate.lines.push(String::from("facts unavailable"));
59        return gate;
60    };
61    let digest = current.digest();
62    match store.read() {
63        PinRead::Absent => {
64            if store.write(current.bytes()) {
65                gate.state = PinState::FirstBootRecorded;
66                gate.pinned = Some(digest);
67                gate.lines
68                    .push(format!("first boot — recorded {}", short(&digest)));
69            } else {
70                gate.state = PinState::StoreUnavailable;
71                gate.lines.push(String::from(
72                    "store unavailable (first-boot write or read-back failed)",
73                ));
74            }
75        }
76        PinRead::Present(bytes) if bytes == current.bytes() => {
77            gate.state = PinState::Match;
78            gate.pinned = Some(digest);
79            gate.lines.push(format!("match {}", short(&digest)));
80        }
81        PinRead::Present(bytes) => match MachineFactsRecord::decode(&bytes) {
82            Ok(pinned) => {
83                gate.state = PinState::Mismatch;
84                gate.pinned = Some(pinned.digest());
85                gate.lines.push(format!(
86                    "mismatch — pinned {}, current {}",
87                    short(&pinned.digest()),
88                    short(&digest)
89                ));
90                gate.lines
91                    .extend(diff::lines(pinned.facts(), current.facts()));
92            }
93            Err(error) => {
94                gate.state = PinState::RecordUnreadable;
95                gate.lines.push(format!("record unreadable ({error:?})"));
96            }
97        },
98        PinRead::Oversized => {
99            gate.state = PinState::RecordUnreadable;
100            gate.lines
101                .push(String::from("record unreadable (larger than any record)"));
102        }
103        PinRead::Unavailable(reason) => {
104            gate.state = PinState::StoreUnavailable;
105            gate.lines.push(format!("store unavailable ({reason})"));
106        }
107    }
108    gate
109}

111impl Gate {
112    /// Whether this outcome withholds non-boot services (CP1 Q4, Q5).
113    pub(crate) fn safe_mode(&self) -> bool {
114        match self.state {
115            PinState::Match | PinState::FirstBootRecorded => false,
116            PinState::StoreUnavailable => !self.store_optional(),
117            PinState::Mismatch | PinState::RecordUnreadable | PinState::FactsUnavailable => true,
118        }
119    }
120
121    /// The q35 profile, bound from the command line under a hypervisor.
122    fn store_optional(&self) -> bool {
123        self.current.as_ref().is_some_and(|record| {
124            let facts = record.facts();
125            facts.slug.as_bytes() == STORE_OPTIONAL_PROFILE
126                && facts.profile_source == PROFILE_SOURCE_CMDLINE
127                && facts.flags & FLAG_HYPERVISOR != 0
128        })
129    }
130
131    fn report(&self) {
132        for line in &self.lines {
133            println(&format!("machine-pin: {line}"));
134        }
135        if self.state == PinState::StoreUnavailable && !self.safe_mode() {
136            println("machine-pin: disabled — no store; continuing under qemu-q35-x86_64.");
137        }
138    }
139}

141/// This boot's gate and whether it withholds services, printing the outcome.
142/// Off x86 there are no machine facts (433 answers "no facts"), and nothing is
143/// withheld; init launches this Cell only on x86.
144pub(crate) fn run(store: &mut impl PinStore) -> (Gate, bool) {
145    if !cfg!(target_arch = "x86_64") {
146        println("machine-pin: not applicable on this architecture.");
147        let gate = Gate {
148            state: PinState::FactsUnavailable,
149            current: None,
150            pinned: None,
151            lines: Vec::new(),
152        };
153        return (gate, false);
154    }
155    let current = match registered_facts() {
156        Ok(record) => Some(record),
157        Err(SyscallError::PermissionDenied) => {
158            println("machine-pin: endpoint registration not observed.");
159            None
160        }
161        Err(_) => None,
162    };
163    let gate = evaluate(current, store);
164    gate.report();
165    let safe_mode = gate.safe_mode();
166    (gate, safe_mode)
167}

169/// The facts, once init's registration makes this Cell the live provider:
170/// 433 refuses every other caller with `PermissionDenied`, which is retried
171/// until the bound. Any other answer is final.
172fn registered_facts() -> Result<MachineFactsRecord, SyscallError> {
173    let started = crate::now_ticks();
174    let mut polls = 0usize;
175    loop {
176        let answer = ostd::syscall::sys_query_machine_facts();
177        if !matches!(answer, Err(SyscallError::PermissionDenied))
178            || crate::now_ticks().wrapping_sub(started) >= REGISTRATION_TICKS
179            || polls >= REGISTRATION_POLLS
180        {
181            return answer;
182        }
183        polls += 1;
184        ostd::task::yield_now();
185    }
186}

188/// Confirm rules in order: safe mode, the console shell, the current digest.
189pub(crate) fn authorize_confirm(
190    safe_mode: bool,
191    from_shell: bool,
192    requested: [u8; 32],
193    current: Option<[u8; 32]>,
194) -> Result<(), Refusal> {
195    if !safe_mode {
196        return Err(Refusal::NotSafeMode);
197    }
198    if !from_shell {
199        return Err(Refusal::NotConsoleShell);
200    }
201    if current != Some(requested) {
202        return Err(Refusal::DigestStale);
203    }
204    Ok(())
