1//! Wire of the ADR-0033 CP1 §7 QEMU `edu` export lanes between `/bin/export-client`
2//! (export owner) and `/bin/edu-holder` (export holder driving QEMU's `edu` DMA).
3//!
4//! One fixed 32-byte little-endian message in both directions: `op` (u32), four
5//! reserved bytes, then three u64 arguments. Clients submit with `IpcSubmit`; the
6//! holder answers each request with `IpcReply` carrying [`OP_STATUS`].
7
8#![no_std]
9#![forbid(unsafe_code)]
10
11/// Holder service id in `test-hooks` images (`kernel::task::dma_export::TEST_EXPORT_SERVICE`).
12pub const SERVICE_ID: u16 = 0x7AE0;
13/// Encoded message length.
14pub const MSG_LEN: usize = 32;
15/// QEMU `edu` PCI identity.
16pub const EDU_VENDOR: u16 = 0x1234;
17/// QEMU `edu` PCI device id.
18pub const EDU_DEVICE: u16 = 0x11e8;
19
20/// Client hello; the reply's `a` is the client's role (1 or 2).
21pub const OP_HELLO: u32 = 1;
22/// L1: `a` = export id of two pages; the holder maps it and copies page 0 to page 1
23/// through the device. Reply `a` = 0 on success, `b` = window IOVA.
24pub const OP_COPY: u32 = 2;
25/// `a` = export id; the holder maps it. Reply `a` = 0 or an error code, `b` = IOVA.
26pub const OP_MAP: u32 = 3;
27/// L2: `a` = IOVA, `b` = bytes; the holder makes the device write 0xEE there.
28pub const OP_DMA_TO: u32 = 4;
29/// L3: `a` = export id, `b` = IOVA, `c` = owner VA; the client dies next. The
30/// holder replies at once, then waits for the lease to vanish and DMAs to `b`.
31pub const OP_STALE_AFTER_DEATH: u32 = 5;
32/// L4: the holder replies, then exits with its lease still mapped.
33pub const OP_DIE: u32 = 6;
34/// Holder reply.
35pub const OP_STATUS: u32 = 0x80;
36
37/// Reply status: done.
38pub const STATUS_OK: u64 = 0;
39/// Reply status: the kernel refused the holder's 438.
40pub const STATUS_MAP_REFUSED: u64 = 1;
41/// Reply status: the device DMA did not complete.
42pub const STATUS_DMA_FAILED: u64 = 2;
43/// Reply status: unknown request.
44pub const STATUS_BAD_REQUEST: u64 = 3;
45
46/// One lane message.
47#[derive(Clone, Copy, Debug, PartialEq, Eq)]
48pub struct Msg {
49    pub op: u32,
50    pub a: u64,
51    pub b: u64,
52    pub c: u64,
53}
54
55impl Msg {
56    /// A message with `op` and its three arguments.
57    pub const fn new(op: u32, a: u64, b: u64, c: u64) -> Self {
58        Self { op, a, b, c }
59    }
60
61    /// Little-endian encoding, reserved bytes zero.
62    pub fn encode(&self) -> [u8; MSG_LEN] {
63        let mut out = [0u8; MSG_LEN];
64        out[0..4].copy_from_slice(&self.op.to_le_bytes());
65        out[8..16].copy_from_slice(&self.a.to_le_bytes());
66        out[16..24].copy_from_slice(&self.b.to_le_bytes());
67        out[24..32].copy_from_slice(&self.c.to_le_bytes());
68        out
69    }
70
71    /// Decode exactly [`MSG_LEN`] leading bytes; `None` when short or reserved bytes are set.
72    pub fn decode(buf: &[u8]) -> Option<Self> {
73        let buf = buf.get(..MSG_LEN)?;
74        if buf[4..8] != [0; 4] {
75            return None;
76        }
77        let word = |at: usize| {
78            let mut w = [0u8; 8];
79            w.copy_from_slice(&buf[at..at + 8]);
80            u64::from_le_bytes(w)
81        };
82        Some(Self {
83            op: u32::from_le_bytes([buf[0], buf[1], buf[2], buf[3]]),
84            a: word(8),
85            b: word(16),
86            c: word(24),
87        })
88    }
89}

91#[cfg(test)]
92mod tests {
93    use super::*;
94
95    #[test]
96    fn round_trip_and_reserved_refusal() {
97        let msg = Msg::new(OP_STALE_AFTER_DEATH, 7, 1 << 38, 0x1234_5000);
98        let mut bytes = msg.encode();
99        assert_eq!(Msg::decode(&bytes), Some(msg));
100        assert_eq!(Msg::decode(&bytes[..31]), None);
101        bytes[5] = 1;
102        assert_eq!(Msg::decode(&bytes), None);
103    }
