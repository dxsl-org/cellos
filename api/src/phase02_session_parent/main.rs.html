60fn sleep_ms(ms: u64) {
61    let end = now() + ms * 1_000_000;
62    while now() < end {
63        let _ = sys_set_timer(1);
64    }

67fn check(name: &str, ok: bool, detail: &str) {
68    CHECKS.fetch_add(1, Ordering::Relaxed);
69    if ok {
70        println(&format!("P02 PASS {name}"));
71    } else {
72        FAILURES.fetch_add(1, Ordering::Relaxed);
73        println(&format!("P02 FAIL {name} {detail}"));
74    }
75}

91fn probe_grants(grants: &[(GrantSlot, EndpointRights)]) -> Result<Vec<Grant>, Status> {
92    let mut out = Vec::new();
93    for (slot, rights) in grants {
94        out.push(Grant::endpoint(
95            *slot,
96            ObjectRefV1 {
97                id: 1,
98                generation: 1,
99            },
100            rights.bits() as u16,
101        )?);
102    }
103    Ok(out)
104}

106/// Commit one child under `operation` with exactly the delegated endpoints.
107fn spawn(
108    operation: &Operation,
109    argv: &[&str],
110    grants: &[(GrantSlot, Endpoint, EndpointRights)],
111) -> Result<ChildRefV1, Status> {
112    let shape: Vec<(GrantSlot, EndpointRights)> = grants.iter().map(|(s, _, r)| (*s, *r)).collect();
113    let probe = ChildContext::new(
114        operation.reference().raw(),
115        argv,
116        b"",
117        &[],
118        &probe_grants(&shape)?,
119    )?;
120    let mut prepared = operation.prepare_child(CHILD, probe.as_bytes().len())?;
121    let mut real = Vec::new();
122    for (slot, endpoint, rights) in grants {
123        let delegated = endpoint.delegate(&prepared, *rights)?;
124        real.push(Grant::endpoint(
125            *slot,
126            delegated.reference().raw(),
127            rights.bits() as u16,
128        )?);
129    }
130    let context = ChildContext::new(operation.reference().raw(), argv, b"", &[], &real)?;
131    prepared.write_context(&context)?;
132    prepared.commit()
133}

135fn settle(session: &mut TerminalSession, child: &mut ForegroundChild) -> Result<(), Status> {
136    let deadline = now() + 10_000_000_000;
137    loop {
138        match session.finish_foreground(child) {
139            Ok(()) => return Ok(()),
140            Err(Status::WouldBlock) => {}
141            Err(other) => return Err(other),
142        }
143        if now() >= deadline {
144            return Err(Status::TimedOut);
145        }
146        let _ = sys_set_timer(1);
147    }
148}

150fn drain_until(session: &mut TerminalSession, stderr: bool, needle: &str, wait_ms: u64) -> String {
151    let end = now() + wait_ms * 1_000_000;
152    let mut all = Vec::new();
153    loop {
154        let mut buffer = [0u8; 512];
155        let read = if stderr {
156            session.drain_stderr(&mut buffer)
157        } else {
158            session.drain_stdout(&mut buffer)
159        };
160        if let Ok(n) = read {
161            all.extend_from_slice(&buffer[..n]);
162        }
163        let text = String::from_utf8_lossy(&all).into_owned();
164        if text.contains(needle) || now() >= end {
165            return text;
166        }
167        let _ = sys_set_timer(1);
168    }
169}

171fn cell_main() {
172    println("P02 START");
173    let mut a = match TerminalSession::create_serial(24, 80) {
174        Ok(session) => session,
175        Err(status) => {
176            println(&format!("P02 FAIL create-a {status:?}"));
177            sys_exit(2);
178        }
179    };
180    let mut b = match TerminalSession::create_serial(25, 81) {
181        Ok(session) => session,
182        Err(status) => {
183            println(&format!("P02 FAIL create-b {status:?}"));
184            sys_exit(2);
185        }
186    };
187    two_sessions(&mut a, &mut b);
188    context_and_foreground(&mut a);
189    fault_and_cancel(&mut a, &mut b);
190    stream_semantics();
191    descendants();
192    consumer_close_and_cancel(&mut b);
193    forged_and_prepare();
194    let _ = a.close();
195    let _ = b.close();
196    let failures = FAILURES.load(Ordering::Relaxed);
197    println(
198        "P02 NOT-EXERCISED issuer-death commit-race-wall-clock cross-process-frontend gui-bridge",
199    );
200    if failures == 0 {
201        println(&format!(
202            "P02-SESSION: PASS checks={}",
203            CHECKS.load(Ordering::Relaxed)
204        ));
205    } else {
206        println(&format!("P02-SESSION: FAIL failures={failures}"));
207    }
208    orphan();
209    sys_exit(if failures == 0 { 0 } else { 1 });
210}

212fn two_sessions(a: &mut TerminalSession, b: &mut TerminalSession) {
213    let _ = a.submit_text("alpha");
214    let _ = b.submit_text("beta");
215    let deadline = now() + 3_000_000_000;
216    let ea = a.next_event(deadline);
217    let eb = b.next_event(deadline);
218    check(
219        "two-session-routing",
220        ea.clone().ok().and_then(text_of).as_deref() == Some("alpha")
221            && eb.clone().ok().and_then(text_of).as_deref() == Some("beta"),
222        &format!("a={} b={}", status(&ea), status(&eb)),
223    );
224    check(
225        "two-session-no-leak",
226        a.next_event(0) == Err(Status::WouldBlock) && b.next_event(0) == Err(Status::WouldBlock),
227        "event crossed or lingered",
228    );
229    let resize = a.resize(30, 100);
230    check("resize-owner", resize.is_ok(), &status(&resize));
231    let ctl = a.next_event(now() + 2_000_000_000);
232    check(
233        "resize-control-frame",
234        matches!(
235            ctl.as_ref().map(|e| &e.kind),
236            Ok(EventKind::Resize {
237                rows: 30,
238                cols: 100
239            })
240        ),
241        &status(&ctl),
242    );
243    check(
244        "resize-isolated",
245        b.next_event(0) == Err(Status::WouldBlock),
246        "resize leaked to b",
247    );
248}

250fn context_and_foreground(a: &mut TerminalSession) {
251    let argv = ["ctx", "", "two words", "h\u{e9}llo"];
252    let env = [("A", "1"), ("EMPTY", ""), ("Z", "last")];
253    // A cell that inherited a directory grant is path-sealed: its child's cwd
254    // must derive from that grant, never from an ambient path lookup.
255    let cwd_source = ChildContext::read_own()
256        .ok()
257        .and_then(|own| own.cwd_grant());
258    let non_utf8 = a.spawn_foreground(CHILD, &["ctx"], b"/\xff", &[], cwd_source);
259    check(
260        "non-utf8-cwd-refused-before-launch",
261        matches!(non_utf8, Err(Status::UnsupportedInput)),
262        &status(&non_utf8),
263    );
264    let big = "x".repeat(600);
265    let over = a.spawn_foreground(CHILD, &[big.as_str()], b"", &[], None);
266    check(
267        "argv-limit-refused-before-launch",
268        matches!(over, Err(Status::QuotaExceeded)),
269        &status(&over),
270    );
271
272    let spawned = a.spawn_foreground(CHILD, &argv, b"/", &env, cwd_source);
273    check("foreground-spawn", spawned.is_ok(), &status(&spawned));
274    let Ok(mut child) = spawned else { return };
275    check(
276        "foreground-activation",
277        child.activation_error().is_none(),
278        "reconciliation error",
279    );
280    let expected = format!(
281        "ctx|argc=4|argv={:?}|cwd=/|cwdgrant=true|env=A=1;EMPTY=;Z=last;",
282        argv.iter().map(|s| String::from(*s)).collect::<Vec<_>>()
283    );
284    let out = drain_until(a, false, &expected, 4000);
285    check(
286        "context-exact-at-first-instruction",
287        out.contains(&expected),
288        &out,
289    );
290    let err = drain_until(a, true, "diag|ctx", 2000);
291    check(
292        "stderr-separate-from-stdout",
293        err.contains("diag|ctx") && !out.contains("diag|"),
294        &err,
295    );
296    let _ = a.submit_text("typed-1");
297    let got = drain_until(a, false, "got|typed-1", 3000);
298    check(
299        "typed-input-to-foreground-only",
300        got.contains("got|typed-1"),
301        &got,
302    );
303    let ended = a.end_input();
304    check("end-input-half-close", ended.is_ok(), &status(&ended));
305    let eof = drain_until(a, false, "eof|seen", 3000);
306    check(
307        "eof-reaches-child-outputs-preserved",
308        eof.contains("eof|seen"),
309        &eof,
310    );
311    let settled = settle(a, &mut child);
312    check(
313        "foreground-restored-after-exit",
314        settled.is_ok(),
315        &status(&settled),
316    );
317    let _ = a.submit_text("after-exit");
318    let back = a.next_event(now() + 3_000_000_000);
319    check(
320        "baseline-input-after-restore",
321        back.ok().and_then(text_of).as_deref() == Some("after-exit"),
322        "baseline did not regain input",
323    );
324}

326fn fault_and_cancel(a: &mut TerminalSession, b: &mut TerminalSession) {
327    for (label, mode) in [
328        ("fault", "panic"),
329        ("raw-mode-fault", "raw-panic"),
330        ("timeout", "timeout"),
331    ] {
332        let spawned = a.spawn_foreground(CHILD, &[mode], b"", &[], None);
333        let Ok(mut child) = spawned else {
334            check(label, false, &status(&spawned));
335            continue;
336        };
337        let settled = settle(a, &mut child);
338        let _ = a.submit_text("typed");
339        let typed = a.next_event(now() + 3_000_000_000);
340        check(
341            &format!("{label}-restores-foreground-and-mode"),
342            settled.is_ok() && typed.ok().and_then(text_of).as_deref() == Some("typed"),
343            &status(&settled),
344        );
345    }
346    for label in ["cancel", "ctrl-c"] {
347        let spawned = a.spawn_foreground(CHILD, &["hang"], b"", &[], None);
348        let Ok(mut child) = spawned else {
349            check(label, false, &status(&spawned));
350            continue;
351        };
352        sleep_ms(600);
353        if label == "cancel" {
354            let _ = a.cancel_foreground(&child, 1);
355        } else {
356            let _ = a.submit_key(0x63, 2, 1);
357        }
358        let settled = settle(a, &mut child);
359        check(
360            &format!("{label}-restores-foreground"),
361            settled.is_ok(),
362            &status(&settled),
363        );
364        let _ = b.submit_text("independent");
365        let other = b.next_event(now() + 3_000_000_000);
366        check(
367            &format!("{label}-other-session-unaffected"),
368            other.ok().and_then(text_of).as_deref() == Some("independent"),
369            "unrelated session disturbed",
370        );
371    }
372}

374fn stream_semantics() {
375    let Ok(op) = Operation::create(None) else {
376        check("stream-operation", false, "create");
377        return;
378    };
379    let Ok((r, w)) = op.create_stream(64) else {
380        check("stream-create", false, "create_stream");
381        return;
382    };
383    // Raw bytes (invalid UTF-8, NUL, DEL, ESC) cross a stream unchanged.
384    {
385        let raw = [0xffu8, 0x00, 0xfe, 0x80, 0x7f, 0x1b, 0xc3, 0x28, 0xa0, 0xa1];
386        let sent = w.write(&raw, 0);
387        let mut got = [0u8; 16];
388        let read = r.read(&mut got, now() + 2_000_000_000);
389        check(
390            "non-utf8-nul-bytes-preserved",
391            matches!((&sent, &read), (Ok(10), Ok(10))) && got[..10] == raw,
392            &format!("sent={} read={}", status(&sent), status(&read)),
393        );
394    }
395    let big = [7u8; 200];
396    let first = w.write(&big, 0);
397    check(
398        "partial-write-bounded",
399        matches!(first, Ok(n) if (1..=64).contains(&n)),
400        &status(&first),
401    );
402    let mut filled = first.unwrap_or(0);
403    loop {
404        match w.write(&big, 0) {
405            Ok(n) => filled += n,
406            Err(status) => {
407                check(
408                    "full-ring-would-block",
409                    status == Status::WouldBlock && filled == 64,
410                    &format!("{status:?} filled={filled}"),
411                );
412                break;
413            }
414        }
415        if filled > 64 {
416            check("full-ring-would-block", false, "ring exceeded capacity");
417            break;
418        }
419    }
420    let mut buffer = [0u8; 128];
421    let mut drained = 0usize;
422    while let Ok(n) = r.read(&mut buffer, 0) {
423        drained += n;
424    }
425    check(
426        "bytes-preserved-count",
427        drained == filled,
428        &format!("drained={drained} filled={filled}"),
429    );
430    check(
431        "empty-nonblocking-is-would-block",
432        r.read(&mut buffer, 0) == Err(Status::WouldBlock),
433        "",
434    );
435    let timed = r.read(&mut buffer, now() + 30_000_000);
436    check(
437        "timeout-distinct-from-eof",
438        timed == Err(Status::TimedOut),
439        &status(&timed),
440    );
441    let duplicate = w.duplicate(EndpointRights::WRITE_ONLY);
442    let _ = w.close();
443    check(
444        "retained-duplicate-prevents-eof",
445        duplicate.is_ok() && r.read(&mut buffer, 0) == Err(Status::WouldBlock),
446        "EOF despite retained writer",
447    );
448    if let Ok(duplicate) = duplicate {
449        let _ = duplicate.close();
450    }
451    let eof = r.read(&mut buffer, 0);
452    check("last-writer-eof", eof == Err(Status::Eof), &status(&eof));
453    check(
454        "closed-ref-stale",
455        r.close().is_ok() && r.read(&mut buffer, 0).is_err(),
456        "closed ref still usable",
457    );
458    if let Ok((r2, w2)) = op.create_stream(64) {
459        let _ = r2.close();
460        let broken = w2.write(b"x", 0);
461        check(
462            "no-reader-write-refused",
463            matches!(
464                broken,
465                Err(Status::BrokenPipe | Status::Revoked | Status::Stale)
466            ),
467            &status(&broken),
468        );
469    }
470    // The consumer close above retired `op`; a forged reference needs a live one.
471    match Operation::create(None)
472        .and_then(|fresh| fresh.create_stream(64).map(|ends| (fresh, ends)))
473    {
474        Ok((fresh, (r3, w3))) => {
475            let raw = w3.reference().raw();
476            let forged = EndpointRef::from_raw(ObjectRefV1 {
477                id: raw.id + 7777,
478                generation: raw.generation,
479            })
480            .map(|reference| Endpoint::new(reference, fresh.reference()));
481            let attempt = forged.map(|e| e.write(b"x", 0));
482            check(
483                "forged-endpoint-denied",
484                matches!(attempt, Ok(Err(_))),
485                &format!("{attempt:?}"),
486            );
487            let _ = (r3.close(), w3.close());
488            let _ = fresh.close();
489        }
490        Err(status) => check(
491            "forged-endpoint-denied",
492            false,
493            &format!("setup {status:?}"),
494        ),
495    }
496    let _ = op.close();
497}

499fn descendants() {
500    let Ok(op) = Operation::create(None) else {
501        return;
502    };
503    let Ok((r, w)) = op.create_stream(64) else {
504        return;
505    };
506    // The child nests its descendant under the operation it was delegated into.
507    let spawned = spawn(
508        &op,
509        &["descend", "400"],
510        &[(GrantSlot::Stdout, w, EndpointRights::WRITE_DELEGATE)],
511    );
512    check("descendant-spawn", spawned.is_ok(), &status(&spawned));
513    let duplicate = w.duplicate(EndpointRights::WRITE_ONLY);
514    let _ = w.close();
515    let mut buffer = [0u8; 16];
516    check(
517        "descendant-and-duplicate-hold-writer",
518        r.read(&mut buffer, 0) == Err(Status::WouldBlock),
519        "EOF while descendant/duplicate writer retained",
520    );
521    if let Ok(duplicate) = duplicate {
522        let _ = duplicate.close();
523    }
524    let data = r.read(&mut buffer, now() + 4_000_000_000);
525    check(
526        "descendant-bytes-arrive",
527        matches!(data, Ok(2)) && &buffer[..2] == b"gc",
528        &status(&data),
529    );
530    let eof = r.read(&mut buffer, now() + 4_000_000_000);
531    check(
532        "eof-after-descendant-exit",
533        eof == Err(Status::Eof),
534        &status(&eof),
535    );
536    let _ = op.close();
537
538    // Holder death: the child faults while holding the delegated writer.
539    let Ok(op) = Operation::create(None) else {
540        return;
541    };
542    let Ok((r, w)) = op.create_stream(64) else {
543        return;
544    };
545    let spawned = spawn(
546        &op,
547        &["panic"],
548        &[(GrantSlot::Stdout, w, EndpointRights::WRITE_DELEGATE)],
549    );
550    let _ = w.close();
551    let eof = r.read(&mut buffer, now() + 5_000_000_000);
552    check(
553        "holder-death-closes-writer",
554        spawned.is_ok() && eof == Err(Status::Eof),
555        &status(&eof),
556    );
557    let _ = op.close();
558}

560/// Poll the owner until the operation's delegated subtree has drained.
561fn subtree_drained(op: &Operation, want_cancelled: bool) -> (bool, String) {
562    let end = now() + 5_000_000_000;
563    loop {
564        let last = match op.query() {
565            Ok(state) => {
566                let last = format!(
567                    "state={} holders={} children={}",
568                    state.state, state.holders, state.children
569                );
570                if state.children == 0 && (!want_cancelled || state.state == 1) {
571                    return (true, last);
572                }
573                last
574            }
575            Err(status) => format!("{status:?}"),
576        };
577        if now() >= end {
578            return (false, last);
579        }
580        sleep_ms(50);
581    }
582}

584fn consumer_close_and_cancel(b: &mut TerminalSession) {
585    // Final consumer close must wake/reap a producer blocked in a child subtree.
586    if let Ok(op) = Operation::create(None) {
587        if let Ok((r, w)) = op.create_stream(64) {
588            let spawned = spawn(
589                &op,
590                &["blocked-writer"],
591                &[(GrantSlot::Stdout, w, EndpointRights::WRITE_DELEGATE)],
592            );
593            let _ = w.close();
594            sleep_ms(700);
595            let before = op.query().map(|s| s.children).unwrap_or(99);
596            let closed = r.close();
597            check(
598                "consumer-close-issued",
599                spawned.is_ok() && closed.is_ok() && before >= 1,
600                &format!("{} children_before={before}", status(&closed)),
601            );
602            let (drained, detail) = subtree_drained(&op, false);
603            check(
604                "consumer-close-reaps-blocked-producer-subtree",
605                drained,
606                &detail,
607            );
608        }
609    }
610    // Root cancellation revokes the subtree; an unrelated session continues.
611    if let Ok(op) = Operation::create(None) {
612        if let Ok((r, w)) = op.create_stream(64) {
613            let spawned = spawn(
614                &op,
615                &["blocked-reader"],
616                &[(GrantSlot::Stdin, r, EndpointRights::READ_DELEGATE)],
617            );
618            sleep_ms(700);
619            let before = op.query().map(|s| s.children).unwrap_or(99);
620            let cancelled = op.cancel(2);
621            check(
622                "root-cancel-issued",
623                spawned.is_ok() && cancelled.is_ok() && before >= 1,
624                &format!("{} children_before={before}", status(&cancelled)),
625            );
626            let (drained, detail) = subtree_drained(&op, true);
627            check(
628                "root-cancel-reaps-blocked-consumer-subtree",
629                drained,
630                &detail,
631            );
632            let mut buffer = [0u8; 4];
633            let after = w.write(b"x", 0);
634            check(
635                "cancelled-subtree-endpoints-revoked",
636                after.is_err() || r.read(&mut buffer, 0).is_err(),
637                "cancelled endpoints live",
638            );
639        }
640    }
641    let _ = b.submit_text("still-alive");
642    let alive = b.next_event(now() + 3_000_000_000);
643    check(
644        "unrelated-session-survives-cancel",
645        alive.ok().and_then(text_of).as_deref() == Some("still-alive"),
646        "",
647    );
648}

650fn forged_and_prepare() {
651    let (Ok(private), Ok(foreign)) = (Operation::create(None), Operation::create(None)) else {
652        return;
653    };
654    if let Ok((pr, pw)) = private.create_stream(64) {
655        let _ = pw.write(b"keep", 0);
656        let op = private.reference().raw();
657        let ep = pr.reference().raw();
658        let (id, gen, eid, egen) = (
659            format!("{}", op.id),
660            format!("{}", op.generation),
661            format!("{}", ep.id),
662            format!("{}", ep.generation),
663        );
664        let spawned = spawn(&foreign, &["forged", &id, &gen, &eid, &egen], &[]);
665        sleep_ms(800);
666        let mut buffer = [0u8; 8];
667        let intact = pr.read(&mut buffer, 0);
668        check(
669            "foreign-child-cannot-touch-private-endpoint",
670            spawned.is_ok()
671                && matches!(intact, Ok(4))
672                && private.query().map(|s| s.state == 0).unwrap_or(false),
673            &status(&intact),
674        );
675    }
676    // Aborted and expired preparations never become runnable children.
677    if let Ok(op) = Operation::create(None) {
678        let probe = ChildContext::new(op.reference().raw(), &["unexpected-run"], b"", &[], &[]);
679        if let Ok(probe) = probe {
680            if let Ok(mut prepared) = op.prepare_child(CHILD, probe.as_bytes().len()) {
681                let wrote = prepared.write_context(&probe);
682                let aborted = prepared.abort();
683                let late = prepared.commit();
684                check(
685                    "abort-then-commit-refused",
686                    wrote.is_ok() && aborted.is_ok() && late.is_err(),
687                    &status(&late),
688                );
689            }
690            if let Ok(mut prepared) = op.prepare_child(CHILD, probe.as_bytes().len()) {
691                let _ = prepared.write_context(&probe);
692                sleep_ms(5500);
693                let late = prepared.commit();
694                check(
695                    "expired-prepare-refuses-commit",
696                    late.is_err(),
697                    &status(&late),
698                );
699            }
700        }
701    }
702}

704/// Last act: exit while a child still blocks on a stream whose only writer this
705/// cell holds. Its observation of the parent's death is read from the log.
706fn orphan() {
707    if let Ok(op) = Operation::create(None) {
708        if let Ok((r, _w)) = op.create_stream(64) {
709            let _ = spawn(
710                &op,
711                &["orphan"],
712                &[(GrantSlot::Stdin, r, EndpointRights::READ_DELEGATE)],
713            );
714            sleep_ms(500);
715        }
716    }
717}
