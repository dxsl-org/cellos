61fn sleep_ms(ms: u64) {
62    let end = now() + ms * 1_000_000;
63    while now() < end {
64        let _ = sys_set_timer(1);
65    }
66}

68fn check(name: &str, ok: bool, detail: &str) {
69    CHECKS.fetch_add(1, Ordering::Relaxed);
70    if ok {
71        println(&format!("P02 PASS {name}"));
72    } else {
73        FAILURES.fetch_add(1, Ordering::Relaxed);
74        println(&format!("P02 FAIL {name} {detail}"));
75    }
76}

92fn probe_grants(grants: &[(GrantSlot, EndpointRights)]) -> Result<Vec<Grant>, Status> {
93    let mut out = Vec::new();
94    for (slot, rights) in grants {
95        out.push(Grant::endpoint(*slot, ObjectRefV1 { id: 1, generation: 1 }, rights.bits() as u16)?);
96    }
97    Ok(out)
98}

100/// Commit one child under `operation` with exactly the delegated endpoints.
101fn spawn(
102    operation: &Operation,
103    argv: &[&str],
104    grants: &[(GrantSlot, Endpoint, EndpointRights)],
105) -> Result<ChildRefV1, Status> {
106    let shape: Vec<(GrantSlot, EndpointRights)> = grants.iter().map(|(s, _, r)| (*s, *r)).collect();
107    let probe = ChildContext::new(operation.reference().raw(), argv, b"", &[], &probe_grants(&shape)?)?;
108    let mut prepared = operation.prepare_child(CHILD, probe.as_bytes().len())?;
109    let mut real = Vec::new();
110    for (slot, endpoint, rights) in grants {
111        let delegated = endpoint.delegate(&prepared, *rights)?;
112        real.push(Grant::endpoint(*slot, delegated.reference().raw(), rights.bits() as u16)?);
113    }
114    let context = ChildContext::new(operation.reference().raw(), argv, b"", &[], &real)?;
115    prepared.write_context(&context)?;
116    prepared.commit()
117}

119fn settle(session: &mut TerminalSession, child: &mut ForegroundChild) -> Result<(), Status> {
120    let deadline = now() + 10_000_000_000;
121    loop {
122        match session.finish_foreground(child) {
123            Ok(()) => return Ok(()),
124            Err(Status::WouldBlock) => {}
125            Err(other) => return Err(other),
126        }
127        if now() >= deadline {
128            return Err(Status::TimedOut);
129        }
130        let _ = sys_set_timer(1);
131    }
132}

134fn drain_until(session: &mut TerminalSession, stderr: bool, needle: &str, wait_ms: u64) -> String {
135    let end = now() + wait_ms * 1_000_000;
136    let mut all = Vec::new();
137    loop {
138        let mut buffer = [0u8; 512];
139        let read = if stderr {
140            session.drain_stderr(&mut buffer)
141        } else {
142            session.drain_stdout(&mut buffer)
143        };
144        if let Ok(n) = read {
145            all.extend_from_slice(&buffer[..n]);
146        }
147        let text = String::from_utf8_lossy(&all).into_owned();
148        if text.contains(needle) || now() >= end {
149            return text;
150        }
151        let _ = sys_set_timer(1);
152    }
153}

155fn cell_main() {
156    println("P02 START");
157    let mut a = match TerminalSession::create_serial(24, 80) {
158        Ok(session) => session,
159        Err(status) => {
160            println(&format!("P02 FAIL create-a {status:?}"));
161            sys_exit(2);
162        }
163    };
164    let mut b = match TerminalSession::create_serial(25, 81) {
165        Ok(session) => session,
166        Err(status) => {
167            println(&format!("P02 FAIL create-b {status:?}"));
168            sys_exit(2);
169        }
170    };
171    two_sessions(&mut a, &mut b);
172    context_and_foreground(&mut a);
173    fault_and_cancel(&mut a, &mut b);
174    stream_semantics();
175    descendants();
176    consumer_close_and_cancel(&mut b);
177    forged_and_prepare();
178    let _ = a.close();
179    let _ = b.close();
180    let failures = FAILURES.load(Ordering::Relaxed);
181    println(&format!(
182        "P02 NOT-EXERCISED issuer-death commit-race-wall-clock cross-process-frontend gui-bridge"
183    ));
184    if failures == 0 {
185        println(&format!("P02-SESSION: PASS checks={}", CHECKS.load(Ordering::Relaxed)));
186    } else {
187        println(&format!("P02-SESSION: FAIL failures={failures}"));
188    }
189    orphan();
190    sys_exit(if failures == 0 { 0 } else { 1 });
191}

193fn two_sessions(a: &mut TerminalSession, b: &mut TerminalSession) {
194    let _ = a.submit_text("alpha");
195    let _ = b.submit_text("beta");
196    let deadline = now() + 3_000_000_000;
197    let ea = a.next_event(deadline);
198    let eb = b.next_event(deadline);
199    check(
200        "two-session-routing",
201        ea.clone().ok().and_then(text_of).as_deref() == Some("alpha")
202            && eb.clone().ok().and_then(text_of).as_deref() == Some("beta"),
203        &format!("a={} b={}", status(&ea), status(&eb)),
204    );
205    check(
206        "two-session-no-leak",
207        a.next_event(0) == Err(Status::WouldBlock) && b.next_event(0) == Err(Status::WouldBlock),
208        "event crossed or lingered",
209    );
210    let resize = a.resize(30, 100);
211    check("resize-owner", resize.is_ok(), &status(&resize));
212    let ctl = a.next_event(now() + 2_000_000_000);
213    check(
214        "resize-control-frame",
215        matches!(ctl.as_ref().map(|e| &e.kind), Ok(EventKind::Resize { rows: 30, cols: 100 })),
216        &status(&ctl),
217    );
218    check("resize-isolated", b.next_event(0) == Err(Status::WouldBlock), "resize leaked to b");
219}

221fn context_and_foreground(a: &mut TerminalSession) {
222    let argv = ["ctx", "", "two words", "h\u{e9}llo"];
223    let env = [("A", "1"), ("EMPTY", ""), ("Z", "last")];
224    // A cell that inherited a directory grant is path-sealed: its child's cwd
225    // must derive from that grant, never from an ambient path lookup.
226    let cwd_source = ChildContext::read_own().ok().and_then(|own| own.cwd_grant());
227    let non_utf8 = a.spawn_foreground(CHILD, &["ctx"], b"/\xff", &[], cwd_source);
228    check("non-utf8-cwd-refused-before-launch", matches!(non_utf8, Err(Status::UnsupportedInput)), &status(&non_utf8));
229    let big = "x".repeat(600);
230    let over = a.spawn_foreground(CHILD, &[big.as_str()], b"", &[], None);
231    check("argv-limit-refused-before-launch", matches!(over, Err(Status::QuotaExceeded)), &status(&over));
232
233    let spawned = a.spawn_foreground(CHILD, &argv, b"/", &env, cwd_source);
234    check("foreground-spawn", spawned.is_ok(), &status(&spawned));
235    let Ok(mut child) = spawned else { return };
236    check("foreground-activation", child.activation_error().is_none(), "reconciliation error");
237    let expected = format!(
238        "ctx|argc=4|argv={:?}|cwd=/|cwdgrant=true|env=A=1;EMPTY=;Z=last;",
239        argv.iter().map(|s| String::from(*s)).collect::<Vec<_>>()
240    );
241    let out = drain_until(a, false, &expected, 4000);
242    check("context-exact-at-first-instruction", out.contains(&expected), &out);
243    let err = drain_until(a, true, "diag|ctx", 2000);
244    check("stderr-separate-from-stdout", err.contains("diag|ctx") && !out.contains("diag|"), &err);
245    let _ = a.submit_text("typed-1");
246    let got = drain_until(a, false, "got|typed-1", 3000);
247    check("typed-input-to-foreground-only", got.contains("got|typed-1"), &got);
248    let ended = a.end_input();
249    check("end-input-half-close", ended.is_ok(), &status(&ended));
250    let eof = drain_until(a, false, "eof|seen", 3000);
251    check("eof-reaches-child-outputs-preserved", eof.contains("eof|seen"), &eof);
252    let settled = settle(a, &mut child);
253    check("foreground-restored-after-exit", settled.is_ok(), &status(&settled));
254    let _ = a.submit_text("after-exit");
255    let back = a.next_event(now() + 3_000_000_000);
256    check(
257        "baseline-input-after-restore",
258        back.ok().and_then(text_of).as_deref() == Some("after-exit"),
259        "baseline did not regain input",
260    );
261}

263fn fault_and_cancel(a: &mut TerminalSession, b: &mut TerminalSession) {
264    for (label, mode) in [("fault", "panic"), ("raw-mode-fault", "raw-panic"), ("timeout", "timeout")] {
265        let spawned = a.spawn_foreground(CHILD, &[mode], b"", &[], None);
266        let Ok(mut child) = spawned else {
267            check(label, false, &status(&spawned));
268            continue;
269        };
270        let settled = settle(a, &mut child);
271        let _ = a.submit_text("typed");
272        let typed = a.next_event(now() + 3_000_000_000);
273        check(
274            &format!("{label}-restores-foreground-and-mode"),
275            settled.is_ok() && typed.ok().and_then(text_of).as_deref() == Some("typed"),
276            &status(&settled),
277        );
278    }
279    for label in ["cancel", "ctrl-c"] {
280        let spawned = a.spawn_foreground(CHILD, &["hang"], b"", &[], None);
281        let Ok(mut child) = spawned else {
282            check(label, false, &status(&spawned));
283            continue;
284        };
285        sleep_ms(600);
286        if label == "cancel" {
287            let _ = a.cancel_foreground(&child, 1);
288        } else {
289            let _ = a.submit_key(0x63, 2, 1);
290        }
291        let settled = settle(a, &mut child);
292        check(&format!("{label}-restores-foreground"), settled.is_ok(), &status(&settled));
293        let _ = b.submit_text("independent");
294        let other = b.next_event(now() + 3_000_000_000);
295        check(
296            &format!("{label}-other-session-unaffected"),
297            other.ok().and_then(text_of).as_deref() == Some("independent"),
298            "unrelated session disturbed",
299        );
300    }
301}

303fn stream_semantics() {
304    let Ok(op) = Operation::create(None) else {
305        check("stream-operation", false, "create");
306        return;
307    };
308    let Ok((r, w)) = op.create_stream(64) else {
309        check("stream-create", false, "create_stream");
310        return;
311    };
312    // Raw bytes (invalid UTF-8, NUL, DEL, ESC) cross a stream unchanged.
313    {
314        let raw = [0xffu8, 0x00, 0xfe, 0x80, 0x7f, 0x1b, 0xc3, 0x28, 0xa0, 0xa1];
315        let sent = w.write(&raw, 0);
316        let mut got = [0u8; 16];
317        let read = r.read(&mut got, now() + 2_000_000_000);
318        check(
319            "non-utf8-nul-bytes-preserved",
320            matches!((&sent, &read), (Ok(10), Ok(10))) && got[..10] == raw,
321            &format!("sent={} read={}", status(&sent), status(&read)),
322        );
323    }
324    let big = [7u8; 200];
325    let first = w.write(&big, 0);
326    check("partial-write-bounded", matches!(first, Ok(n) if n >= 1 && n <= 64), &status(&first));
327    let mut filled = first.unwrap_or(0);
328    loop {
329        match w.write(&big, 0) {
330            Ok(n) => filled += n,
331            Err(status) => {
332                check("full-ring-would-block", status == Status::WouldBlock && filled == 64, &format!("{status:?} filled={filled}"));
333                break;
334            }
335        }
336        if filled > 64 {
337            check("full-ring-would-block", false, "ring exceeded capacity");
338            break;
339        }
340    }
341    let mut buffer = [0u8; 128];
342    let mut drained = 0usize;
343    while let Ok(n) = r.read(&mut buffer, 0) {
344        drained += n;
345    }
346    check("bytes-preserved-count", drained == filled, &format!("drained={drained} filled={filled}"));
347    check("empty-nonblocking-is-would-block", r.read(&mut buffer, 0) == Err(Status::WouldBlock), "");
348    let timed = r.read(&mut buffer, now() + 30_000_000);
349    check("timeout-distinct-from-eof", timed == Err(Status::TimedOut), &status(&timed));
350    let duplicate = w.duplicate(EndpointRights::WRITE_ONLY);
351    let _ = w.close();
352    check(
353        "retained-duplicate-prevents-eof",
354        duplicate.is_ok() && r.read(&mut buffer, 0) == Err(Status::WouldBlock),
355        "EOF despite retained writer",
356    );
357    if let Ok(duplicate) = duplicate {
358        let _ = duplicate.close();
359    }
360    let eof = r.read(&mut buffer, 0);
361    check("last-writer-eof", eof == Err(Status::Eof), &status(&eof));
362    check("closed-ref-stale", r.close().is_ok() && r.read(&mut buffer, 0).is_err(), "closed ref still usable");
363    if let Ok((r2, w2)) = op.create_stream(64) {
364        let _ = r2.close();
365        let broken = w2.write(b"x", 0);
366        check("no-reader-write-refused", matches!(broken, Err(Status::BrokenPipe | Status::Revoked | Status::Stale)), &status(&broken));
367    }
368    // The consumer close above retired `op`; a forged reference needs a live one.
369    match Operation::create(None).and_then(|fresh| fresh.create_stream(64).map(|ends| (fresh, ends))) {
370        Ok((fresh, (r3, w3))) => {
371            let raw = w3.reference().raw();
372            let forged = EndpointRef::from_raw(ObjectRefV1 { id: raw.id + 7777, generation: raw.generation })
373                .map(|reference| Endpoint::new(reference, fresh.reference()));
374            let attempt = forged.map(|e| e.write(b"x", 0));
375            check("forged-endpoint-denied", matches!(attempt, Ok(Err(_))), &format!("{attempt:?}"));
376            let _ = (r3.close(), w3.close());
377            let _ = fresh.close();
378        }
379        Err(status) => check("forged-endpoint-denied", false, &format!("setup {status:?}")),
380    }
381    let _ = op.close();
382}

384fn descendants() {
385    let Ok(op) = Operation::create(None) else { return };
386    let Ok((r, w)) = op.create_stream(64) else { return };
387    // The child nests its descendant under the operation it was delegated into.
388    let spawned = spawn(&op, &["descend", "400"], &[(GrantSlot::Stdout, w, EndpointRights::WRITE_DELEGATE)]);
389    check("descendant-spawn", spawned.is_ok(), &status(&spawned));
390    let duplicate = w.duplicate(EndpointRights::WRITE_ONLY);
391    let _ = w.close();
392    let mut buffer = [0u8; 16];
393    check(
394        "descendant-and-duplicate-hold-writer",
395        r.read(&mut buffer, 0) == Err(Status::WouldBlock),
396        "EOF while descendant/duplicate writer retained",
397    );
398    if let Ok(duplicate) = duplicate {
399        let _ = duplicate.close();
400    }
401    let data = r.read(&mut buffer, now() + 4_000_000_000);
402    check("descendant-bytes-arrive", matches!(data, Ok(2)) && &buffer[..2] == b"gc", &status(&data));
403    let eof = r.read(&mut buffer, now() + 4_000_000_000);
404    check("eof-after-descendant-exit", eof == Err(Status::Eof), &status(&eof));
405    let _ = op.close();
406
407    // Holder death: the child faults while holding the delegated writer.
408    let Ok(op) = Operation::create(None) else { return };
409    let Ok((r, w)) = op.create_stream(64) else { return };
410    let spawned = spawn(&op, &["panic"], &[(GrantSlot::Stdout, w, EndpointRights::WRITE_DELEGATE)]);
411    let _ = w.close();
412    let eof = r.read(&mut buffer, now() + 5_000_000_000);
413    check("holder-death-closes-writer", spawned.is_ok() && eof == Err(Status::Eof), &status(&eof));
414    let _ = op.close();
415}

417/// Poll the owner until the operation's delegated subtree has drained.
418fn subtree_drained(op: &Operation, want_cancelled: bool) -> (bool, String) {
419    let end = now() + 5_000_000_000;
420    let mut last = String::from("no query");
421    loop {
422        match op.query() {
423            Ok(state) => {
424                last = format!("state={} holders={} children={}", state.state, state.holders, state.children);
425                if state.children == 0 && (!want_cancelled || state.state == 1) {
426                    return (true, last);
427                }
428            }
429            Err(status) => last = format!("{status:?}"),
430        }
431        if now() >= end {
432            return (false, last);
433        }
434        sleep_ms(50);
435    }
436}

438fn consumer_close_and_cancel(b: &mut TerminalSession) {
439    // Final consumer close must wake/reap a producer blocked in a child subtree.
440    if let Ok(op) = Operation::create(None) {
441        if let Ok((r, w)) = op.create_stream(64) {
442            let spawned = spawn(&op, &["blocked-writer"], &[(GrantSlot::Stdout, w, EndpointRights::WRITE_DELEGATE)]);
443            let _ = w.close();
444            sleep_ms(700);
445            let before = op.query().map(|s| s.children).unwrap_or(99);
446            let closed = r.close();
447            check("consumer-close-issued", spawned.is_ok() && closed.is_ok() && before >= 1, &format!("{} children_before={before}", status(&closed)));
448            let (drained, detail) = subtree_drained(&op, false);
449            check("consumer-close-reaps-blocked-producer-subtree", drained, &detail);
450        }
451    }
452    // Root cancellation revokes the subtree; an unrelated session continues.
453    if let Ok(op) = Operation::create(None) {
454        if let Ok((r, w)) = op.create_stream(64) {
455            let spawned = spawn(&op, &["blocked-reader"], &[(GrantSlot::Stdin, r, EndpointRights::READ_DELEGATE)]);
456            sleep_ms(700);
457            let before = op.query().map(|s| s.children).unwrap_or(99);
458            let cancelled = op.cancel(2);
459            check("root-cancel-issued", spawned.is_ok() && cancelled.is_ok() && before >= 1, &format!("{} children_before={before}", status(&cancelled)));
460            let (drained, detail) = subtree_drained(&op, true);
461            check("root-cancel-reaps-blocked-consumer-subtree", drained, &detail);
462            let mut buffer = [0u8; 4];
463            let after = w.write(b"x", 0);
464            check("cancelled-subtree-endpoints-revoked", after.is_err() || r.read(&mut buffer, 0).is_err(), "cancelled endpoints live");
465        }
466    }
467    let _ = b.submit_text("still-alive");
468    let alive = b.next_event(now() + 3_000_000_000);
469    check("unrelated-session-survives-cancel", alive.ok().and_then(text_of).as_deref() == Some("still-alive"), "");
470}

472fn forged_and_prepare() {
473    let (Ok(private), Ok(foreign)) = (Operation::create(None), Operation::create(None)) else { return };
474    if let Ok((pr, pw)) = private.create_stream(64) {
475        let _ = pw.write(b"keep", 0);
476        let op = private.reference().raw();
477        let ep = pr.reference().raw();
478        let (id, gen, eid, egen) = (format!("{}", op.id), format!("{}", op.generation), format!("{}", ep.id), format!("{}", ep.generation));
479        let spawned = spawn(&foreign, &["forged", &id, &gen, &eid, &egen], &[]);
480        sleep_ms(800);
481        let mut buffer = [0u8; 8];
482        let intact = pr.read(&mut buffer, 0);
483        check(
484            "foreign-child-cannot-touch-private-endpoint",
485            spawned.is_ok() && matches!(intact, Ok(4)) && private.query().map(|s| s.state == 0).unwrap_or(false),
486            &status(&intact),
487        );
488    }
489    // Aborted and expired preparations never become runnable children.
490    if let Ok(op) = Operation::create(None) {
491        let probe = ChildContext::new(op.reference().raw(), &["unexpected-run"], b"", &[], &[]);
492        if let Ok(probe) = probe {
493            if let Ok(mut prepared) = op.prepare_child(CHILD, probe.as_bytes().len()) {
494                let wrote = prepared.write_context(&probe);
495                let aborted = prepared.abort();
496                let late = prepared.commit();
497                check("abort-then-commit-refused", wrote.is_ok() && aborted.is_ok() && late.is_err(), &status(&late));
498            }
499            if let Ok(mut prepared) = op.prepare_child(CHILD, probe.as_bytes().len()) {
500                let _ = prepared.write_context(&probe);
501                sleep_ms(5500);
502                let late = prepared.commit();
503                check("expired-prepare-refuses-commit", late.is_err(), &status(&late));
504            }
505        }
506    }
507}

509/// Last act: exit while a child still blocks on a stream whose only writer this
510/// cell holds. Its observation of the parent's death is read from the log.
511fn orphan() {
512    if let Ok(op) = Operation::create(None) {
513        if let Ok((r, _w)) = op.create_stream(64) {
514            let _ = spawn(&op, &["orphan"], &[(GrantSlot::Stdin, r, EndpointRights::READ_DELEGATE)]);
515            sleep_ms(500);
516        }
517    }
