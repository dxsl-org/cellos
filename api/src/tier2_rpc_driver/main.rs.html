1//! Tier-1 driver of the Phase-02 cross-tier fixture.
2//!
3//! `init` spawns the Tier-2 provider and **registers** it under the fixture's service
4//! id: a private-root Cell cannot register itself (`RegisterService` is `SpawnCap`-gated
5//! and its ceiling is empty), but its spawner can, which is the same pattern init uses
6//! for the hypervisor. The driver therefore resolves the **named** service, not a tid it
7//! was handed, and every leg goes through the shipped SDK.
8//!
9//! | Leg | What it witnesses |
10//! |---|---|
11//! | 1 | The named service resolves to a live binding with the provider's real `(cell_id, generation)`, and the SDK agrees with the raw opcode |
12//! | 2 | Tier-1 → Tier-2: a typed, nontrivial copied request is delivered and answered |
13//! | 3 | Tier-2 → Tier-1: the private-root provider called the named VFS service and said so in its reply |
14//! | 4 | An oversize frame is refused by the kernel **before** delivery |
15//! | 5 | A method the provider does not authorize is refused, with no side effect |
16//! | 6 | Once the provider is gone, the descriptor the handle holds is **refused**, not re-targeted |
17//!
18//! Leg 4b: a wrong **user buffer** on the syscall copy path. Fabricates an
19//! unmapped raw pointer under a reviewed unsafe-allowlist exemption, proving
20//! the kernel's TaskCopyView refuses the buffer before delivery.
21
22#![no_std]
23#![no_main]
24#![allow(unsafe_code)]

49/// Payload size for the echo leg: big enough that a truncated or substituted body
50/// changes the checksum, small enough to leave the reply inside one frame.
51const PAYLOAD_BYTES: usize = 512;
52
53/// Yields allowed for the provider to exit before the stale-descriptor leg gives up.
54const STALE_POLLS: usize = 2000;
55
56fn fail(reason: &str) -> ! {
57    println(&format!("[tier2-rpc] FAIL — {reason}"));
58    sys_exit(1)
59}
60
61/// The kernel's current binding for the fixture's service, or `None` when nothing live
62/// is registered under it. Reads the registry directly, without touching the SDK's cache.
63fn live_binding() -> Option<ServiceBinding> {
64    let mut record = [0u8; SERVICE_BINDING_LEN];
65    sys_lookup_service_bound(SERVICE_ID, &mut record)
66}
67
68fn cell_main() {
69    // ── Leg 1: the named service resolves, with the provider's real identity ──
70    let mut provider: ServiceRef<{ SERVICE_ID }> = ServiceRef::new();
71    let binding = match provider.binding() {
72        Some(binding) => binding,
73        None => fail("the fixture's service is not resolvable by name"),
74    };
75    println(&format!(
76        "[tier2-rpc] PROVIDER-BINDING tid={} cell={} gen={}",
77        binding.tid, binding.cell_id, binding.generation
78    ));
79    if binding.cell_id == 0 || binding.generation == 0 {
80        fail("the registered binding carries no live Cell identity");
81    }
82    if live_binding() != Some(binding) {
83        fail("the SDK's named resolution disagreed with the raw bound lookup");
84    }
85    println("[tier2-rpc] PROVIDER-NAME-MATCHES-RAW=true");
86
87    let mut response_buffer = [0u8; IPC_BUF_SIZE];
88
89    // ── Leg 2 + 3: one typed exchange each way, through the SDK ──────────────
90    let nonce = 0x5A17_0002u64;
91    let payload = vec![0xA5u8; PAYLOAD_BYTES];
92    let request = EchoRequest {
93        method: METHOD_ECHO,
94        nonce,
95        payload: &payload,
96    };
97    let response = match provider.call::<EchoRequest, EchoResponse>(&request, &mut response_buffer)
98    {
99        Ok(response) => response,
100        Err(error) => {
101            println(&format!("[tier2-rpc] EXCHANGE-ERROR {error:?}"));
102            fail("typed cross-tier call did not complete")
103        }
104    };
105    if response.nonce != nonce
106        || response.payload_len as usize != PAYLOAD_BYTES
107        || response.payload_checksum != checksum(&payload)
108    {
109        println(&format!("[tier2-rpc] ECHO-MISMATCH {response:?}"));
110        fail("the Tier-2 provider echoed a different body");
111    }
112    println("[tier2-rpc] TIER1-TO-TIER2=OK");
113    if response.vfs_stat_ok {
114        println(&format!(
115            "[tier2-rpc] TIER2-TO-TIER1=OK root_is_dir={}",
116            response.vfs_root_is_dir
117        ));
118    } else {
119        fail("the private-root provider could not call the named VFS service");
120    }
121
122    // ── Leg 4: an oversize frame, refused by the kernel before delivery ───────
123    // The provider is still waiting for its second request here, so a frame that were
124    // delivered would be observed as one.
125    let oversize = vec![0u8; IPC_BUF_SIZE + 1];
126    match sys_send(binding.tid as usize, &oversize) {
127        SyscallResult::Err(_) => println("[tier2-rpc] OVERSIZE=REFUSED"),
128        SyscallResult::Ok(bytes) => {
129            println(&format!("[tier2-rpc] OVERSIZE=ACCEPTED bytes={bytes}"));
130            fail("the kernel delivered an oversize frame across tiers")
131        }
132    }
133    // ── Leg 4b: an invalid user buffer on the syscall copy path ───────────────
134    // Passes an unmapped raw pointer to sys_send. The kernel's TaskCopyView must
135    // refuse it before delivery; the provider must not receive it.
136    let invalid_slice = unsafe {
137        core::slice::from_raw_parts(0xDEAD_0000usize as *const u8, 64)
138    };
139    match sys_send(binding.tid as usize, invalid_slice) {
140        SyscallResult::Err(_) => println("[tier2-rpc] INVALID-BUFFER=REFUSED"),
141        SyscallResult::Ok(bytes) => {
142            println(&format!("[tier2-rpc] INVALID-BUFFER=ACCEPTED bytes={bytes}"));
143            fail("the kernel accepted an invalid user buffer across tiers")
144        }
145    }
146
147
148    // ── Leg 5: a method the provider does not authorize ──────────────────────
149    let denied = EchoRequest {
150        method: METHOD_UNAUTHORIZED,
151        nonce,
152        payload: &[],
153    };
154    match provider.call::<EchoRequest, EchoResponse>(&denied, &mut response_buffer) {
155        Ok(response) if response.error == Some(UNAUTHORIZED) => {
156            println("[tier2-rpc] UNAUTHORIZED-METHOD=REFUSED")
157        }
158        Ok(other) => {
159            println(&format!(
160                "[tier2-rpc] UNAUTHORIZED-METHOD=ACCEPTED {other:?}"
161            ));
162            fail("an unauthorized method was served")
163        }
164        Err(error) => {
165            println(&format!("[tier2-rpc] UNAUTHORIZED-METHOD=ERROR {error:?}"));
166            fail("the unauthorized-method leg did not get a typed refusal")
167        }
168    }
169
170    // ── Leg 6: the descriptor the handle holds is refused, not re-targeted ───
171    // The provider served its budget and exits. Watch the registry through the raw
172    // opcode — deliberately not through the SDK, whose cache this leg is about — until
173    // no live binding remains, then call. The call must fail: the SDK must not send to
174    // the retired tid, and must not silently resolve onto some other incarnation.
175    let mut gone = false;
176    for _ in 0..STALE_POLLS {
177        if live_binding().is_none() {
178            gone = true;
179            break;
180        }
181        yield_now();
182    }
183    if !gone {
184        fail("the fixture's registry entry stayed live after the provider exited");
185    }
186    println("[tier2-rpc] PROVIDER-GONE=none");
187    match provider.call::<EchoRequest, EchoResponse>(&denied, &mut response_buffer) {
188        Err(ViError::NotFound) => println("[tier2-rpc] STALE-BINDING=REFUSED"),
189        Ok(response) => {
190            println(&format!("[tier2-rpc] STALE-BINDING=SERVED {response:?}"));
191            fail("a call under a stale descriptor was served")
192        }
193        Err(other) => {
194            println(&format!("[tier2-rpc] STALE-BINDING=ERROR {other:?}"));
195            fail("a stale descriptor did not report the refusal the contract names")
196        }
197    }
198    if provider.is_live() {
199        fail("the handle still calls its descriptor live after the refusal");
200    }
201    println("[tier2-rpc] STALE-BINDING-CLEARED=true");
202
203    println("[tier2-rpc] DRIVER-DONE");
204    sys_exit(0)
205}
