1//! Real target transport: typed IPC to the registered AI inference service.
2//!
3//! Compiled only for bare-metal targets with the `ostd-transport` feature, so host tests never link
4//! `ostd`'s global allocator. The transport reuses `ostd`'s sender-masked service call (Spec 17
5//! §7): a reply from any cell other than the resolved service is a failure, never decoded.
6
7use ai_proto::{AiRequest, AiResponse, AI_IPC_BUF_SIZE};
8use ostd::ipc::{service_call, IpcError};
9use ostd::service::{service, ServiceRef};
10
11use crate::transport::AiTransport;
12use crate::AiClientError;
13
14/// Transport backed by the live `service::AI` provider.
15pub struct OstdTransport {
16    svc: ServiceRef<{ service::AI }>,
17}
18
19impl OstdTransport {
20    /// Create an unresolved transport; resolution happens on the first call and is cached.
21    pub const fn new() -> Self {
22        Self {
23            svc: ServiceRef::new(),
24        }
25    }
26
27    /// Drop the cached provider tid (for example after the supervisor respawned the service).
28    pub fn invalidate(&mut self) {
29        self.svc.invalidate();
30    }
31}
32
33impl Default for OstdTransport {
34    fn default() -> Self {
35        Self::new()
36    }
37}
38
39impl AiTransport for OstdTransport {
40    fn round_trip<'r>(
41        &mut self,
42        request: &AiRequest<'_>,
43        reply: &'r mut [u8],
44    ) -> Result<AiResponse<'r>, AiClientError> {
45        let tid = self.svc.resolve().ok_or(AiClientError::NoService)?;
46        let mut send = [0u8; AI_IPC_BUF_SIZE];
47        match service_call(tid, request, &mut send, reply) {
48            // The receive buffer arrives without a length (the syscall reports the sender), so the
49            // reply is decoded tolerantly — the same convention every other typed client uses.
50            Ok(raw) => ai_proto::decode(raw).map_err(|_| AiClientError::Protocol),
51            Err(IpcError::Encode) | Err(IpcError::Decode) => Err(AiClientError::Protocol),
52            Err(IpcError::Send) | Err(IpcError::Recv) | Err(IpcError::WrongSender) => {
53                // The provider may have died or restarted under a new tid.
54                self.svc.invalidate();
55                Err(AiClientError::Transport)
56            }
57        }
58    }
59}
