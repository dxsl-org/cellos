1//! `AiClient` — the Spec 24 §6 public programming surface for the Cellos AI inference service.
2//!
3//! Normative source: [`docs/specs/24-ai-inference-architecture.md`] §6. The wire records live in
4//! `ai-proto`; this crate is what a Cell imports to talk to `/bin/ai`.
5//!
6//! # Shape
7//!
8//! [`AiTransport`] is the only target-specific piece: on a Cell it is a typed-IPC round trip
9//! through `ostd` ([`ostd_transport::OstdTransport`], compiled only for bare-metal targets so host
10//! tests link without `ostd`'s global allocator). Everything else — request ids, reply matching,
11//! streaming, bounds — is target-independent and host-tested.
12//!
13//! # Deliberate deviations from the ratified signature
14//!
15//! * `prompt` keeps the async shape but the returned [`TokenStream`] is a synchronous iterator: the
16//!   async reactor (Spec 20 §5) is not landed, and pretending otherwise would hide a blocking IPC
17//!   round trip behind `await`. Streaming itself is real — the service holds sessions and the
18//!   caller polls them.
19//! * Errors are [`AiClientError`], not `ViError`, so a caller can distinguish "transport failed"
20//!   from "the service refused with `Busy`". `From<AiClientError> for ViError` is provided for
21//!   callers that want the collapsed form.
22//!
23//! # Capacity
24//!
25//! One call is one 4 KiB IPC message both ways: prompts up to [`ai_proto::MAX_PROMPT_BYTES`],
26//! replies up to [`ai_proto::MAX_TOKENS_PER_POLL`] token ids / [`ai_proto::MAX_REPLY_TEXT_BYTES`]
27//! text bytes. Larger context/messages belong on the VFS/grant path (Spec 24 §6).
28

46/// Failure from the service boundary.
47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub enum AiClientError {
49    /// The transport itself failed (IPC send/receive, no registered service).
50    Transport,
51    /// The service is not registered yet; the caller may retry after the supervisor spawns it.
52    NoService,
53    /// The reply did not decode, or did not match the request that was sent.
54    Protocol,
55    /// The service answered with a typed refusal.
56    Service(AiError),
57    /// A synchronous drain exceeded the caller's poll budget.
58    PollLimitExceeded,
59}
60
61/// Result alias for this crate's boundaries.
62pub type AiResult<T> = Result<T, AiClientError>;
63
64impl From<AiClientError> for crate::types::ViError {
65    fn from(error: AiClientError) -> Self {
66        match error {
67            AiClientError::Transport | AiClientError::PollLimitExceeded => crate::types::ViError::IO,
68            AiClientError::NoService => crate::types::ViError::NotFound,
69            AiClientError::Protocol => crate::types::ViError::InvalidInput,
70            AiClientError::Service(error) => match error {
71                AiError::BadRequest(_) => crate::types::ViError::InvalidArgument,
72                AiError::Busy => crate::types::ViError::WouldBlock,
73                AiError::NoModel => crate::types::ViError::NotFound,
74                AiError::NotSupported => crate::types::ViError::NotSupported,
75                AiError::UnknownRequest | AiError::NotRunning => crate::types::ViError::InvalidArgument,
76                AiError::Internal => crate::types::ViError::IO,
77            },
78        }
79    }
80}

194/// Client for one AI inference service provider.
195pub struct AiClient<T: AiTransport> {
196    transport: T,
197    device: ai_proto::DeviceTarget,
198}
199
200impl<T: AiTransport> AiClient<T> {
201    /// Connect through `transport`, letting the service choose the backend.
202    pub fn new(transport: T) -> Self {
203        Self {
204            transport,
205            device: ai_proto::DeviceTarget::Auto,
206        }
207    }
208
209    /// Connect through `transport` with an explicit device preference.
210    pub fn with_device(transport: T, device: ai_proto::DeviceTarget) -> Self {
211        Self { transport, device }
212    }
213
214    /// Current device preference.
215    pub fn device(&self) -> ai_proto::DeviceTarget {
216        self.device
217    }
218
219    /// Capability probe.
220    pub fn describe(&mut self) -> AiResult<ServiceInfo> {
221        let request = AiRequest::Describe;
222        let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
223        let response = self.exchange(&request, &mut reply)?;
224        match response {
225            AiResponse::Description(describe) => Ok(describe.into()),
226            AiResponse::Failed { error, .. } => Err(AiClientError::Service(error)),
227            _ => Err(AiClientError::Protocol),
228        }
229    }
230
231    /// Start a session; returns the service-assigned id.
232    pub fn submit(&mut self, params: &InferParams) -> AiResult<u32> {
233        if params.prompt.len() > ai_proto::MAX_PROMPT_BYTES {
234            return Err(AiClientError::Service(AiError::BadRequest(
235                ai_proto::limit::Violation::PromptTooLong,
236            )));
237        }
238        let request = AiRequest::InferSubmit(ai_proto::InferSubmit {
239            prompt: params.prompt.as_str(),
240            max_tokens: params.max_tokens,
241            temperature_milli: params.temperature_milli,
242            top_k: params.top_k,
243            seed: params.seed,
244            device: self.device,
245        });
246        let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
247        match self.exchange(&request, &mut reply)? {
248            AiResponse::Accepted { request_id, .. } => Ok(request_id),
249            AiResponse::Failed { error, .. } => Err(AiClientError::Service(error)),
250            _ => Err(AiClientError::Protocol),
251        }
252    }
253
254    /// Take up to `max_tokens` ready tokens from a session.
255    pub fn poll(&mut self, request_id: u32, max_tokens: u8) -> AiResult<Chunk> {
256        if max_tokens == 0 || max_tokens > ai_proto::MAX_TOKENS_PER_POLL {
257            return Err(AiClientError::Service(AiError::BadRequest(
258                ai_proto::limit::Violation::PollTooLarge,
259            )));
260        }
261        let request = AiRequest::InferStreamPoll {
262            request_id,
263            max_tokens,
264        };
265        let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
266        match self.exchange(&request, &mut reply)? {
267            AiResponse::TokenChunk {
268                tokens,
269                text,
270                done,
271                finish,
272                ..
273            } => Ok(Chunk {
274                ids: ai_proto::token_ids(tokens).collect(),
275                text: String::from(text),
276                done,
277                finish: if done { Some(finish) } else { None },
278            }),
279            AiResponse::Failed { error, .. } => Err(AiClientError::Service(error)),
280            _ => Err(AiClientError::Protocol),
281        }
282    }
283
284    /// Abandon a session.
285    pub fn cancel(&mut self, request_id: u32) -> AiResult<()> {
286        let request = AiRequest::InferCancel { request_id };
287        let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
288        match self.exchange(&request, &mut reply)? {
289            // The service answers a cancel with the final chunk state, an empty chunk, or a
290            // refusal; treating "finished" as success keeps cancel idempotent for callers.
291            AiResponse::TokenChunk { .. } => Ok(()),
292            AiResponse::Failed { error: AiError::UnknownRequest, .. } => Ok(()),
293            AiResponse::Failed { error, .. } => Err(AiClientError::Service(error)),
294            _ => Err(AiClientError::Protocol),
295        }
296    }
297
298    /// Embedding of `text`.
299    pub fn embed(&mut self, text: &str) -> AiResult<Vec<f32>> {
300        let request = AiRequest::InferEmbed {
301            text,
302            device: self.device,
303        };
304        let mut reply = [0u8; ai_proto::AI_IPC_BUF_SIZE];
305        match self.exchange(&request, &mut reply)? {
306            AiResponse::Embedding { dim, values } => {
307                let collected: Vec<f32> = ai_proto::embedding_values(values).collect();
308                if collected.len() != usize::from(dim) {
309                    return Err(AiClientError::Protocol);
310                }
311                Ok(collected)
312            }
313            AiResponse::Failed { error, .. } => Err(AiClientError::Service(error)),
314            _ => Err(AiClientError::Protocol),
315        }
316    }
317
318    /// Submit and drain a generation bounded by `max_polls` round trips.
319    pub fn generate(&mut self, params: &InferParams, max_polls: usize) -> AiResult<Generation> {
320        let request_id = self.submit(params)?;
321        let mut ids = Vec::new();
322        let mut text = String::new();
323        let mut polls = 0usize;
324
325        while polls < max_polls {
326            let chunk = match self.poll(request_id, ai_proto::MAX_TOKENS_PER_POLL) {
327                Ok(chunk) => chunk,
328                Err(error) => {
329                    // Do not leave a half-consumed session behind on a late failure.
330                    let _ = self.cancel(request_id);
331                    return Err(error);
332                }
333            };
334            polls += 1;
335            ids.extend(chunk.ids.iter().copied());
336            text.push_str(&chunk.text);
337            if chunk.done {
338                return Ok(Generation {
339                    ids,
340                    text,
341                    finish: chunk.finish.unwrap_or(FinishReason::Length),
342                    polls,
343                });
344            }
345        }
346
347        let _ = self.cancel(request_id);
348        Err(AiClientError::PollLimitExceeded)
349    }

351    /// Spec 24 §6 `prompt`: submit and drive the session as a stream of tokens.
352    ///
353    /// The future resolves as soon as the service accepts the submission; the stream then performs
354    /// one blocking poll per `next()`. It is `async` to keep the ratified call shape.
355    pub async fn prompt(&mut self, params: &InferParams) -> AiResult<TokenStream<'_, T>> {
356        let request_id = self.submit(params)?;
357        Ok(TokenStream {
358            client: self,
359            request_id,
360            finished: false,
361            pending: Vec::new(),
362            text: String::new(),
363        })
364    }
365
366    /// Send `request` and decode its reply from `reply`.
367    ///
368    /// The reply buffer is caller-owned so a decoded [`AiResponse`] can borrow from it without
369    /// copying: every public method keeps its buffer on the stack and consumes the response in the
370    /// same scope.
371    fn exchange<'r>(
372        &mut self,
373        request: &AiRequest<'_>,
374        reply: &'r mut [u8],
375    ) -> AiResult<AiResponse<'r>> {
376        let response = self.transport.round_trip(request, reply)?;
377        if !response.matches(request) {
378            return Err(AiClientError::Protocol);
379        }
380        Ok(response)
381    }
382}

384/// Token stream returned by [`AiClient::prompt`].
385///
386/// Yields each generated token; `None` when the session finishes. Dropping the stream without
387/// cancelling leaves the session live on the service until the client's process exits or the
388/// service reclaims it, matching the service's bounded session table (call [`TokenStream::cancel`]
389/// to release early).
390pub struct TokenStream<'a, T: AiTransport> {
391    client: &'a mut AiClient<T>,
392    request_id: u32,
393    finished: bool,
394    pending: Vec<u32>,
395    text: String,
396}
397
398impl<T: AiTransport> TokenStream<'_, T> {
399    /// Session id assigned by the service.
400    pub fn request_id(&self) -> u32 {
401        self.request_id
402    }
403
404    /// Text decoded so far.
405    pub fn text(&self) -> &str {
406        &self.text
407    }
408
409    /// Release the session early.
410    pub fn cancel(&mut self) -> AiResult<()> {
411        self.finished = true;
412        self.client.cancel(self.request_id)
413    }
414}
415
416impl<T: AiTransport> Iterator for TokenStream<'_, T> {
417    type Item = AiResult<Token>;
418
419    fn next(&mut self) -> Option<Self::Item> {
420        if let Some(id) = self.pending.pop() {
421            return Some(Ok(Token {
422                id,
423                text: String::new(),
424            }));
425        }
426        if self.finished {
427            return None;
428        }
429        match self.client.poll(self.request_id, ai_proto::MAX_TOKENS_PER_POLL) {
430            Ok(chunk) => {
431                self.text.push_str(&chunk.text);
432                if chunk.done {
433                    self.finished = true;
434                }
435                let mut ids = chunk.ids;
436                ids.reverse();
437                self.pending = ids;
438                // Text for the drained ids was already appended; yield ids first, then the text is
439                // available through `text()`. Each token's own text needs its own decode, so the
440                // iterator surfaces ids and the caller reads `text()` for the decoded run.
441                if let Some(id) = self.pending.pop() {
442                    return Some(Ok(Token {
443                        id,
444                        text: String::new(),
445                    }));
446                }
447                if self.finished {
448                    None
449                } else {
450                    self.next()
451                }
452            }
453            Err(error) => {
454                self.finished = true;
455                Some(Err(error))
456            }
457        }
458    }
459}

461/// True when the service reports at least one live session slot for this client.
462///
463/// A caller-side hint only: the service is the authority on admission.
464pub fn has_free_session(info: &ServiceInfo) -> bool {
465    info.live_sessions < info.max_sessions.min(MAX_SESSIONS)
466}

640    /// Minimal executor: every future here is immediately ready, so a single poll suffices.
641    fn block_on<F: core::future::Future>(future: F) -> F::Output {
642        let mut future = core::pin::pin!(future);
643        let waker = core::task::Waker::noop();
644        let mut context = core::task::Context::from_waker(waker);
645        match future.as_mut().poll(&mut context) {
646            core::task::Poll::Ready(value) => value,
647            core::task::Poll::Pending => panic!("AI client futures never park"),
648        }
649    }
