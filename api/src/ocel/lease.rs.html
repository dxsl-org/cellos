4use alloc::format;
5use alloc::string::String;
6use ocel_service_proto::{Engine, Failure, Request, Response, FRAME_BYTES};
7use ostd::ipc::{self, IpcSubmitError, IpcTakeResult, IpcTerminal};
8use ostd::syscall::{sys_get_scheduler_ticks, sys_lookup_service, sys_yield};
9
10const CALL_TICKS: u64 = 500;
11const RELEASE_TICKS: u64 = 100;
12
13pub(crate) trait Transport {
14    fn lookup(&mut self, service: u16) -> Option<usize>;
15    fn call(&mut self, peer: usize, request: &[u8], reply: &mut [u8], ticks: u64) -> Result<usize, String>;
16}
17
18#[derive(Default)]
19pub(crate) struct KernelTransport;
20impl Transport for KernelTransport {
21    fn lookup(&mut self, service: u16) -> Option<usize> { sys_lookup_service(service) }
22    fn call(&mut self, peer: usize, request: &[u8], reply: &mut [u8], ticks: u64) -> Result<usize, String> {
23        bounded_call(&mut KernelOperations, peer, request, reply, ticks)
24    }
25}
26
27trait Operations {
28    fn now(&mut self) -> Option<u64>;
29    fn submit(&mut self, peer: usize, request: &[u8]) -> Result<usize, IpcSubmitError>;
30    fn take(&mut self, op: usize, reply: &mut [u8]) -> Result<IpcTakeResult, ipc::IpcError>;
31    fn wait(&mut self, ticks: u64);
32    fn yield_once(&mut self);
33    fn cancel(&mut self, op: usize);
34}
35struct KernelOperations;
36impl Operations for KernelOperations {
37    fn now(&mut self) -> Option<u64> { sys_get_scheduler_ticks() }
38    fn submit(&mut self, peer: usize, request: &[u8]) -> Result<usize, IpcSubmitError> { ipc::submit(peer, request) }
39    fn take(&mut self, op: usize, reply: &mut [u8]) -> Result<IpcTakeResult, ipc::IpcError> { ipc::take(op, reply) }
40    fn wait(&mut self, ticks: u64) { ipc::wait(ticks); }
41    fn yield_once(&mut self) { sys_yield(); }
42    fn cancel(&mut self, op: usize) { let _ = ipc::cancel(op); }
43}
44
45fn bounded_call(ops: &mut impl Operations, peer: usize, request: &[u8], reply: &mut [u8], ticks: u64) -> Result<usize, String> {
46    let started = ops.now().ok_or_else(|| String::from("Engine IPC deadline clock is unavailable."))?;
47    let remaining = |now: Option<u64>| now.map(|now| ticks.saturating_sub(now.wrapping_sub(started))).unwrap_or(0);
48    let op = loop {
49        if remaining(ops.now()) == 0 { return Err(String::from("Engine IPC admission timed out.")); }
50        match ops.submit(peer, request) {
51            Ok(op) => break op,
52            Err(IpcSubmitError::Busy) => ops.yield_once(),
53            Err(error) => return Err(format!("Engine IPC admission failed: {:?}.", error)),
54        }
55    };
56    let result = loop {
57        match ops.take(op, reply) {
58            Ok(IpcTakeResult::Terminal { status: IpcTerminal::Reply, len }) => return Ok(len),
59            Ok(IpcTakeResult::Terminal { status, .. }) => return Err(format!("Engine IPC failed: {:?}.", status)),
60            Err(error) => break Err(format!("Engine IPC response failed: {:?}.", error)),
61            Ok(IpcTakeResult::Pending) => {}
62        }
63        let left = remaining(ops.now());
64        if left == 0 { break Err(String::from("Engine IPC response timed out.")); }
65        ops.wait(left);
66    };
67    // Cancellation isolates late replies from subsequent calls. Take the retained
68    // cancellation terminal even on decode/buffer errors to free the operation slot.
69    ops.cancel(op);
70    let mut discard = [0u8; 4096];
71    let _ = ops.take(op, &mut discard);
72    result
73}

75fn service(engine: Engine) -> u16 {
76    match engine { Engine::JavaScript => api::syscall::service::OCEL_JS, Engine::Pdf => api::syscall::service::OCEL_PDF }
77}
78fn name(engine: Engine) -> &'static str {
79    match engine { Engine::JavaScript => "JavaScript", Engine::Pdf => "PDF" }
80}

82pub(crate) struct Lease<T: Transport = KernelTransport> {
83    transport: T,
84    engine: Engine,
85    broker: usize,
86    token: Option<u64>,
87    provider: usize,
88}
89impl<T: Transport> Lease<T> {
90    pub(crate) fn acquire_with(engine: Engine, mut transport: T) -> Result<Self, String> {
91        let broker = transport.lookup(api::syscall::service::OCEL_ACTIVATOR)
92            .ok_or_else(|| format!("{} engine is unavailable: activation service is not registered. No fallback engine is available.", name(engine)))?;
93        let mut frame = [0u8; FRAME_BYTES];
94        // Exact operation tokens authenticate the provider incarnation: unlike
95        // sender-masked legacy receive, only this broker can complete the call.
96        let len = transport.call(broker, &Request::Acquire { engine }.encode(), &mut frame, CALL_TICKS)?;
97        let response = if len == FRAME_BYTES { Response::decode(&frame) } else { None };
98        let (received, token, provider) = match response {
99            Some(Response::Ready { engine: received, lease, tid }) => (received, lease, usize::try_from(tid).ok()),
100            Some(Response::Error { engine: received, failure }) if received == engine => {
101                let reason = match failure {
102                    Failure::Unavailable => "optional engine is not installed or available",
103                    Failure::Denied => "activation was denied",
104                    Failure::Busy => "activation lease limit is reached",
105                    Failure::Failed => "engine initialization failed",
106                };
107                return Err(format!("{} engine is unavailable: {}. No fallback engine is available.", name(engine), reason));
108            }
109            _ => return Err(String::from("Invalid engine activation reply (engine or frame mismatch).")),
110        };
111        let mut lease = Self { transport, engine: received, broker, token: Some(token), provider: provider.unwrap_or(0) };
112        // Own every valid Ready token before validating it. Even an unexpected
113        // engine must not leave forgotten demand behind when the reply is rejected.
114        if received != engine { return Err(String::from("Invalid engine activation reply (engine mismatch).")); }
115        lease.validate()?;
116        Ok(lease)
117    }
118
119    fn validate(&mut self) -> Result<(), String> {
120        if self.token.is_none()
121            || self.transport.lookup(api::syscall::service::OCEL_ACTIVATOR) != Some(self.broker)
122            || self.transport.lookup(service(self.engine)) != Some(self.provider)
123            || self.provider == 0
124        {
125            return Err(format!("{} engine provider changed or stopped; reload the document to activate it again.", name(self.engine)));
126        }
127        Ok(())
128    }
129
130    pub(crate) fn exchange(&mut self, request: &[u8], reply: &mut [u8]) -> Result<usize, String> {
131        self.validate()?;
132        self.transport.call(self.provider, request, reply, CALL_TICKS)
133    }
134
135    pub(crate) fn release(&mut self) -> Result<(), String> {
136        let Some(token) = self.token else { return Ok(()); };
137        if self.transport.lookup(api::syscall::service::OCEL_ACTIVATOR) != Some(self.broker) {
138            self.token = None; // The old broker incarnation cannot own live leases.
139            return Err(String::from("Engine activation provider stopped before release."));
140        }
141        let mut frame = [0u8; FRAME_BYTES];
142        let len = self.transport.call(self.broker, &Request::Release { engine: self.engine, lease: token }.encode(), &mut frame, RELEASE_TICKS)?;
143        if len != FRAME_BYTES || Response::decode(&frame) != Some(Response::Released { engine: self.engine }) {
144            return Err(String::from("Engine lease release was not acknowledged."));
145        }
146        self.token = None;
147        Ok(())
148    }
149}
150impl<T: Transport> Drop for Lease<T> {
151    fn drop(&mut self) {
152        if let Err(error) = self.release() { ostd::io::println(&format!("[ocel] {}", error)); }
153    }
154}
