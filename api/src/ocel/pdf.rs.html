2//! Copied, sender-filtered IPC client for the isolated native PDF service.
3//! No PDF parser or engine runs inside Ocel. Each load owns one document handle,

18pub fn is_pdf_path(url: &str) -> bool {
19    let path = url.split('#').next().unwrap_or(url);
20    path.rsplit_once('.').is_some_and(|(_, ext)| ext.eq_ignore_ascii_case("pdf"))
21}

23/// A missing fragment selects page one; malformed or zero pages are errors,
24/// never a request to feed binary PDF source into the text reader.
25pub fn page_number(url: &str) -> Result<u32, String> {
26    let Some((_, fragment)) = url.split_once('#') else {
27        return Ok(1);
28    };
29    let invalid = || String::from("Invalid PDF page fragment. Use #page=N with a positive, one-based page number.");
30    let value = fragment.strip_prefix("page=").ok_or_else(invalid)?;
31    if value.is_empty() || !value.bytes().all(|byte| byte.is_ascii_digit()) {
32        return Err(invalid());
33    }
34    value.parse::<u32>().ok().filter(|page| *page != 0).ok_or_else(invalid)
35}

37fn page_url(url: &str, page: u32) -> String {
38    let base = url.split('#').next().unwrap_or(url);
39    format!("{}#page={}", base, page)
40}

42pub struct PdfPage {
43    page: u32,
44    pages: u32,
45    width: u32,
46    height: u32,
47    pixels: Vec<u8>,
48}
49
50impl PdfPage {
51    pub fn into_nodes(self, original_url: &str) -> Vec<DocNode> {
52        let mut spans = alloc::vec![StyledSpan::plain(&format!("Page {} of {}", self.page, self.pages))];
53        if self.page > 1 {
54            spans.push(StyledSpan::plain("   "));
55            let mut previous = StyledSpan::plain("Previous page");
56            previous.link = Some(page_url(original_url, self.page - 1));
57            spans.push(previous);
58        }
59        if self.page < self.pages {
60            spans.push(StyledSpan::plain("   "));
61            let mut next = StyledSpan::plain("Next page");
62            next.link = Some(page_url(original_url, self.page + 1));
63            spans.push(next);
64        }
65        alloc::vec![
66            DocNode::Paragraph { spans },
67            DocNode::Image {
68                width: self.width,
69                height: self.height,
70                pixels: Rc::new(self.pixels),
71            },
72        ]
73    }
74}

76pub fn load_page(path: &str, page: u32) -> Result<PdfPage, String> {
77    load_page_with(path, page, KernelTransport)
78}

80fn load_page_with(path: &str, page: u32, transport: impl Transport) -> Result<PdfPage, String> {
81    let mut lease = Lease::acquire_with(Engine::Pdf, transport)?;
82    let raster = load_with(path, page, |request| exchange(&mut lease, request));
83    let released = lease.release();
84    let raster = raster?;
85    released?;
86    Ok(raster)
87}
88
89fn exchange(lease: &mut Lease<impl Transport>, request: &Request) -> Result<Response, String> {
90    let mut send_buf = [0u8; IPC_BYTES];
91    let encoded = postcard::to_slice(request, &mut send_buf)
92        .map_err(|_| String::from("PDF request exceeds the 4096-byte IPC limit."))?;
93    let mut recv_buf = [0u8; IPC_BYTES];
94    let len = lease.exchange(encoded, &mut recv_buf)?;
95    postcard::from_bytes(&recv_buf[..len])
96        .map_err(|_| String::from("The native PDF service returned a malformed IPC reply."))
97}

99fn load_with(
100    path: &str,
101    page: u32,
102    mut exchange: impl FnMut(&Request) -> Result<Response, String>,
103) -> Result<PdfPage, String> {
104    let (document, pages) = match exchange(&Request::Open { path: String::from(path) })? {
105        Response::Opened { document, pages } => (document, pages),
106        Response::Error { message } => return Err(format!("Could not open PDF: {}", message)),
107        _ => return Err(String::from("The PDF service returned an unexpected Open reply.")),
108    };
109    let rendered = read_page(document, pages, page, &mut exchange);
110    // Every path after Opened attempts Close, including bad metadata, invalid
111    // page numbers, render failures, and malformed or interrupted pixel chunks.
112    let closed = match exchange(&Request::Close { document }) {
113        Ok(Response::Closed) => Ok(()),
114        Ok(Response::Error { message }) => Err(format!("Could not close PDF: {}", message)),
115        Ok(_) => Err(String::from("The PDF service returned an unexpected Close reply.")),
116        Err(error) => Err(error),
117    };
118    let raster = rendered?;
119    closed?;
120    Ok(raster)
121}

123fn read_page(
124    document: u32,
125    pages: u32,
126    page: u32,
127    exchange: &mut impl FnMut(&Request) -> Result<Response, String>,
128) -> Result<PdfPage, String> {
129    if pages == 0 {
130        return Err(String::from("The PDF service reported a document with no pages."));
131    }
132    if page == 0 || page > pages {
133        return Err(format!("PDF page {} is out of range. This document has {} pages.", page, pages));
134    }
135    let (width, height, total) = match exchange(&Request::RenderPage {
136        document,
137        page: page - 1,
138        max_width: PAGE_BOUND,
139        max_height: PAGE_BOUND,
140    })? {
141        Response::Rendered { width, height, bytes } => {
142            (width, height, validated_size(width, height, bytes)?)
143        }
144        Response::Error { message } => return Err(format!("Could not render PDF page: {}", message)),
145        _ => return Err(String::from("The PDF service returned an unexpected RenderPage reply.")),
146    };
147    let mut pixels = Vec::with_capacity(total);
148    while pixels.len() < total {
149        let offset = pixels.len();
150        let length = (total - offset).min(PIXEL_CHUNK_BYTES);
151        match exchange(&Request::ReadPixels {
152            document,
153            offset: offset as u32,
154            length: length as u16,
155        })? {
156            Response::Pixels { offset: received_offset, bytes } => {
157                if received_offset as usize != offset || bytes.len() != length {
158                    return Err(String::from("The PDF service returned a non-contiguous or incorrectly sized pixel chunk."));
159                }
160                // Chunk size and all requested offsets are multiples of four.
161                if bytes.chunks_exact(4).any(|pixel| pixel[3] != 255) {
162                    return Err(String::from("The PDF service returned non-opaque pixels instead of opaque BGRA8888."));
163                }
164                pixels.extend_from_slice(&bytes);
165            }
166            Response::Error { message } => return Err(format!("Could not read PDF pixels: {}", message)),
167            _ => return Err(String::from("The PDF service returned an unexpected ReadPixels reply.")),
168        }
169    }
170    Ok(PdfPage { page, pages, width, height, pixels })
171}

173fn validated_size(width: u32, height: u32, bytes: u32) -> Result<usize, String> {
174    let pixel_count = (width as usize).checked_mul(height as usize);
175    let expected = pixel_count.and_then(|count| count.checked_mul(4));
176    if width == 0 || height == 0 || width > PAGE_BOUND || height > PAGE_BOUND
177        || !matches!(pixel_count, Some(count) if count <= MAX_PAGE_PIXELS)
178        || expected != Some(bytes as usize)
179    {
180        return Err(String::from("The PDF service returned invalid raster dimensions or byte count (limit: 1024 x 1024, 1,048,576 pixels)."));
181    }
182    Ok(bytes as usize)
183}
