11pub struct LoadedDocument {
12    pub url: String,
13    pub title: String,
14    pub content: String,
15    pub direct_nodes: Option<Vec<DocNode>>,
16}
17
18pub fn load_document(url: &str) -> LoadedDocument {
19    let clean_url = url.trim();
20
21    // Strip "file://" prefix if provided
22    let path = if let Some(stripped) = clean_url.strip_prefix("file://") {
23        stripped
24    } else {
25        clean_url
26    };
27
28    // 1. Check for built-in virtual documents
29    if path == "welcome" || path == "/welcome" || path == "/welcome.md" || path == "welcome.md" {
30        return LoadedDocument {
31            url: String::from("file:///welcome.md"),
32            title: String::from("Ocel — Welcome Guide"),
33            content: get_welcome_content(),
34            direct_nodes: None,
35        };
36    }
37
38    if path == "help" || path == "/help" || path == "/help.md" || path == "help.md" {
39        return LoadedDocument {
40            url: String::from("file:///help.md"),
41            title: String::from("Ocel — Keyboard & Usage Help"),
42            content: get_help_content(),
43            direct_nodes: None,
44        };
45    }
46
47    // 2. Check for HTTP network URL
48    if clean_url.starts_with("http://") {
49        return match crate::net::fetch_http(clean_url) {
50            Ok(content) => {
51                let display_host = clean_url
52                    .trim_start_matches("http://")
53                    .split('/')
54                    .next()
55                    .unwrap_or(clean_url);
56                LoadedDocument {
57                    url: String::from(clean_url),
58                    title: format!("Ocel — {}", display_host),
59                    content,
60                    direct_nodes: None,
61                }
62            }
63            Err(err) => LoadedDocument {
64                url: String::from(clean_url),
65                title: String::from("Ocel — Network Error"),
66                content: format!(
67                    "# Network Error\n\nCould not fetch `{}`.\n\n**Reason:** {}\n\n### Suggestions:\n- Verify host address and port.\n- Ensure CellOS `service-net` is active.\n- Try connecting to gateway: `http://10.0.2.2:8080/`",
68                    clean_url, err
69                ),
70                direct_nodes: None,
71            },
72        };
73    }
74
75    // 3. Check for HTTPS encrypted network URL
76    if clean_url.starts_with("https://") {
77        return match crate::net::fetch_https(clean_url) {
78            Ok(content) => {
79                let display_host = clean_url
80                    .trim_start_matches("https://")
81                    .split('/')
82                    .next()
83                    .unwrap_or(clean_url);
84                LoadedDocument {
85                    url: String::from(clean_url),
86                    title: format!("Ocel — {} [HTTPS]", display_host),
87                    content,
88                    direct_nodes: None,
89                }
90            }
91            Err(err) => LoadedDocument {
92                url: String::from(clean_url),
93                title: String::from("Ocel — TLS Error"),
94                content: format!(
95                    "# HTTPS / TLS Connection Error\n\nCould not securely fetch `{}`.\n\n**Reason:** {}\n\n### Suggestions:\n- Check network connection.\n- Ensure CellOS `service-net` TLS broker is active.\n- Verify host and port 443.",
96                    clean_url, err
97                ),
98                direct_nodes: None,
99            },
100        };
101    }
102
103    // PDFs are binary and are owned/decoded only by the isolated PDF service.
104    // Keep the original URL for history and links; fragments never reach VFS.
105    if crate::pdf::is_pdf_path(path) {
106        let pdf_path = path.split('#').next().unwrap_or(path);
107        let result = crate::pdf::page_number(clean_url)
108            .and_then(|page| crate::pdf::load_page(pdf_path, page));
109        return match result {
110            Ok(page) => LoadedDocument {
111                url: String::from(clean_url),
112                title: format!("Ocel — PDF ({})", pdf_path.rsplit('/').next().unwrap_or(pdf_path)),
113                content: String::new(),
114                direct_nodes: Some(page.into_nodes(clean_url)),
115            },
116            Err(error) => LoadedDocument {
117                url: String::from(clean_url),
118                title: String::from("Ocel — PDF Error"),
119                content: String::new(),
120                direct_nodes: Some(alloc::vec![
121                    DocNode::Heading { level: 1, text: String::from("PDF Viewing Error") },
122                    DocNode::Paragraph {
123                        spans: alloc::vec![crate::doc::StyledSpan::plain(&format!("Could not view {}: {}", pdf_path, error))],
124                    },
125                ]),
126            },
127        };
128    }
129
130    // Binary image files must bypass the UTF-8 document reader.
131    if crate::image::is_image_path(path) {
132        match ostd::fs::File::open(path) {
133            Ok(mut file) => {
134                if let Some(bytes) = read_image_bytes(&mut file) {
135                    if let Some(img) = crate::image::decode(&bytes) {
136                        let filename = path.rsplit('/').next().unwrap_or(path);
137                        return LoadedDocument {
138                            url: String::from(clean_url),
139                            title: format!("Ocel — Image ({})", filename),
140                            content: String::new(),
141                            direct_nodes: Some(alloc::vec![DocNode::Image {
142                                width: img.width,
143                                height: img.height,
144                                pixels: alloc::rc::Rc::new(img.pixels),
145                            }]),
146                        };
147                    }
148                }
149                return LoadedDocument {
150                    url: String::from(clean_url),
151                    title: String::from("Ocel — Image Decode Error"),
152                    content: format!("# Error Decoding Image\n\nCould not decode `{}`. Supported formats: PNG, JPEG and BMP. Limits: 2 MiB encoded, 4096 pixels per dimension and 1,048,576 pixels total.", path),
153                    direct_nodes: None,
154                };
155            }
156            Err(_) => {
157                return LoadedDocument {
158                    url: String::from(clean_url),
159                    title: String::from("Ocel — File Not Found"),
160                    content: format!("# 404: Image Not Found\n\nCould not find `{}`.", path),
161                    direct_nodes: None,
162                };
163            }
164        }
165    }
166
167    // 3. Attempt to open text-based file from VFS
168    match ostd::fs::File::open(path) {
169        Ok(mut file) => match file.read_to_string() {
170            Ok(content) => {
171                let filename = path.rsplit('/').next().unwrap_or(path);
172                LoadedDocument {
173                    url: String::from(clean_url),
174                    title: format!("Ocel — {}", filename),
175                    content,
176                    direct_nodes: None,
177                }
178            }
179            Err(_) => LoadedDocument {
180                url: String::from(clean_url),
181                title: String::from("Ocel — Read Error"),
182                content: format!(
183                    "# Error Reading File\n\nCould not read data from `{}`.\n\nThe file might be non-UTF8 or corrupted.",
184                    path
185                ),
186                direct_nodes: None,
187            },
188        },
189        Err(_) => LoadedDocument {
190            url: String::from(clean_url),
191            title: String::from("Ocel — File Not Found"),
192            content: format!(
193                "# 404: Document Not Found\n\nCould not open `{}` from CellOS VFS.\n\n### Suggestions:\n- Verify the file path in `/data/` or `/srv/`.\n- Type `file:///welcome.md` to return to the welcome page.\n- Type `file:///help.md` for keyboard shortcuts.",
194                path
195            ),
196            direct_nodes: None,
197        },
198    }
199}

201fn read_image_bytes(file: &mut ostd::fs::File) -> Option<Vec<u8>> {
202    let mut bytes = Vec::new();
203    let mut chunk = [0u8; 4096];
204    loop {
205        let count = file.read(&mut chunk).ok()?;
206        if count == 0 {
207            return Some(bytes);
208        }
209        if bytes.len() + count > 2 * 1024 * 1024 {
210            return None;
211        }
212        bytes.extend_from_slice(&chunk[..count]);
213    }
214}

263fn get_help_content() -> String {
264    String::from(
265        r#"# Ocel Keyboard & Navigation Help
266
267### Keyboard Shortcuts
268- **Up Arrow / Down Arrow:** Scroll viewport by 1 line (24 px).
269- **PageUp / PageDown:** Scroll viewport by one screen height.
270- **Home / End:** Jump to top or bottom of document.
271- **Enter:** Navigate to URL typed in the address bar.
272- **Backspace:** Edit or erase address bar characters.
273
274---
275
276### Supported URL Schemes
277- `file:///path/to/file.md` — Open Markdown file from VFS.
278- `file:///path/to/file.txt` — Open plain text or source code.
279- `file:///path/to/file.html` — Open HTML document.
280- `file:///path/to/file.pdf#page=2` — View a local PDF page (one-based; no fragment opens page 1).
281- **PDF navigation:** Click Previous page / Next page above the raster. PageUp / PageDown still scroll; tab shortcuts are unchanged.
282- **PDF availability:** Requires the registered native `ocel-pdf` service. One opaque page at a time, bounded to 1024 × 1024 pixels; the service accepts PDFs up to 2 MiB. No fallback renderer or PDF-source text view.
283- `file:///welcome.md` — Built-in Welcome guide.
284- `file:///help.md` — This help page.
285"#,
286    )
