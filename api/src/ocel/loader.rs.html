4extern crate alloc;
5use alloc::format;
6use alloc::string::String;
7use alloc::vec::Vec;
8
9use crate::doc::DocNode;
10
11pub struct LoadedDocument {
12    pub url: String,
13    pub title: String,
14    pub content: String,
15    pub direct_nodes: Option<Vec<DocNode>>,
16}
17
18pub fn load_document(url: &str) -> LoadedDocument {
19    let clean_url = url.trim();
20
21    // Strip "file://" prefix if provided
22    let path = if let Some(stripped) = clean_url.strip_prefix("file://") {
23        stripped
24    } else {
25        clean_url
26    };
27
28    // 1. Check for built-in virtual documents
29    if path == "welcome" || path == "/welcome" || path == "/welcome.md" || path == "welcome.md" {
30        return LoadedDocument {
31            url: String::from("file:///welcome.md"),
32            title: String::from("Ocel — Welcome Guide"),
33            content: get_welcome_content(),
34            direct_nodes: None,
35        };
36    }
37
38    if path == "help" || path == "/help" || path == "/help.md" || path == "help.md" {
39        return LoadedDocument {
40            url: String::from("file:///help.md"),
41            title: String::from("Ocel — Keyboard & Usage Help"),
42            content: get_help_content(),
43            direct_nodes: None,
44        };
45    }
46
47    // 2. Check for HTTP network URL
48    if clean_url.starts_with("http://") {
49        return match crate::net::fetch_http(clean_url) {
50            Ok(content) => {
51                let display_host = clean_url
52                    .trim_start_matches("http://")
53                    .split('/')
54                    .next()
55                    .unwrap_or(clean_url);
56                LoadedDocument {
57                    url: String::from(clean_url),
58                    title: format!("Ocel — {}", display_host),
59                    content,
60                    direct_nodes: None,
61                }
62            }
63            Err(err) => LoadedDocument {
64                url: String::from(clean_url),
65                title: String::from("Ocel — Network Error"),
66                content: format!(
67                    "# Network Error\n\nCould not fetch `{}`.\n\n**Reason:** {}\n\n### Suggestions:\n- Verify host address and port.\n- Ensure CellOS `service-net` is active.\n- Try connecting to gateway: `http://10.0.2.2:8080/`",
68                    clean_url, err
69                ),
70                direct_nodes: None,
71            },
72        };
73    }
74
75    // 3. Check for HTTPS encrypted network URL
76    if clean_url.starts_with("https://") {
77        return match crate::net::fetch_https(clean_url) {
78            Ok(content) => {
79                let display_host = clean_url
80                    .trim_start_matches("https://")
81                    .split('/')
82                    .next()
83                    .unwrap_or(clean_url);
84                LoadedDocument {
85                    url: String::from(clean_url),
86                    title: format!("Ocel — {} [HTTPS]", display_host),
87                    content,
88                    direct_nodes: None,
89                }
90            }
91            Err(err) => LoadedDocument {
92                url: String::from(clean_url),
93                title: String::from("Ocel — TLS Error"),
94                content: format!(
95                    "# HTTPS / TLS Connection Error\n\nCould not securely fetch `{}`.\n\n**Reason:** {}\n\n### Suggestions:\n- Check network connection.\n- Ensure CellOS `service-net` TLS broker is active.\n- Verify host and port 443.",
96                    clean_url, err
97                ),
98                direct_nodes: None,
99            },
100        };
101    }
102
103    // 2. Check if binary image (.bmp)
104    if path.to_ascii_lowercase().ends_with(".bmp") {
105        match ostd::fs::File::open(path) {
106            Ok(mut file) => {
107                let mut bytes = Vec::new();
108                if file.read_to_end(&mut bytes).is_ok() {
109                    if let Some(img) = crate::image::decode_bmp(&bytes) {
110                        let filename = path.rsplit('/').next().unwrap_or(path);
111                        return LoadedDocument {
112                            url: String::from(clean_url),
113                            title: format!("Ocel — Image ({})", filename),
114                            content: String::new(),
115                            direct_nodes: Some(alloc::vec![DocNode::Image {
116                                width: img.width,
117                                height: img.height,
118                                pixels: img.pixels,
119                            }]),
120                        };
121                    }
122                }
123                return LoadedDocument {
124                    url: String::from(clean_url),
125                    title: String::from("Ocel — Image Decode Error"),
126                    content: format!("# Error Decoding BMP\n\nCould not decode `{}`.", path),
127                    direct_nodes: None,
128                };
129            }
130            Err(_) => {
131                return LoadedDocument {
132                    url: String::from(clean_url),
133                    title: String::from("Ocel — File Not Found"),
134                    content: format!("# 404: Image Not Found\n\nCould not find `{}`.", path),
135                    direct_nodes: None,
136                };
137            }
138        }
139    }
140
141    // 3. Attempt to open text-based file from VFS
142    match ostd::fs::File::open(path) {
143        Ok(mut file) => match file.read_to_string() {
144            Ok(content) => {
145                let filename = path.rsplit('/').next().unwrap_or(path);
146                LoadedDocument {
147                    url: String::from(clean_url),
148                    title: format!("Ocel — {}", filename),
149                    content,
150                    direct_nodes: None,
151                }
152            }
153            Err(_) => LoadedDocument {
154                url: String::from(clean_url),
155                title: String::from("Ocel — Read Error"),
156                content: format!(
157                    "# Error Reading File\n\nCould not read data from `{}`.\n\nThe file might be non-UTF8 or corrupted.",
158                    path
159                ),
160                direct_nodes: None,
161            },
162        },
163        Err(_) => LoadedDocument {
164            url: String::from(clean_url),
165            title: String::from("Ocel — File Not Found"),
166            content: format!(
167                "# 404: Document Not Found\n\nCould not open `{}` from CellOS VFS.\n\n### Suggestions:\n- Verify the file path in `/data/` or `/srv/`.\n- Type `file:///welcome.md` to return to the welcome page.\n- Type `file:///help.md` for keyboard shortcuts.",
168                path
169            ),
170            direct_nodes: None,
171        },
172    }
173}

178
179**Ocel** (derived from *Ocellus* — the simple eye) is the native document and web content viewer for **CellOS**.

184- **Lightweight & Fast:** Starts up in <100 ms with sub-10 MB memory footprint.

223fn get_help_content() -> String {
224    String::from(
225        r#"# Ocel Keyboard & Navigation Help
226
227### Keyboard Shortcuts
228- **Up Arrow / Down Arrow:** Scroll viewport by 1 line (24 px).
229- **PageUp / PageDown:** Scroll viewport by one screen height.
230- **Home / End:** Jump to top or bottom of document.
231- **Enter:** Navigate to URL typed in the address bar.
232- **Backspace:** Edit or erase address bar characters.

236### Supported URL Schemes
237- `file:///path/to/file.md` — Open Markdown file from VFS.
238- `file:///path/to/file.txt` — Open plain text or source code.
239- `file:///path/to/file.html` — Open HTML document.
240- `file:///welcome.md` — Built-in Welcome guide.
241- `file:///help.md` — This help page.
