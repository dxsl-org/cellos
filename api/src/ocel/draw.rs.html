26static ACTIVE_SYSTEM_PALETTE: AtomicU8 = AtomicU8::new(viui::theme::SystemPalette::Light as u8);
27
28pub fn set_active_system_palette(palette: viui::theme::SystemPalette) {
29    ACTIVE_SYSTEM_PALETTE.store(palette as u8, Ordering::Relaxed);
30}
31
32pub fn active_system_palette() -> viui::theme::SystemPalette {
33    viui::theme::SystemPalette::from_u8(ACTIVE_SYSTEM_PALETTE.load(Ordering::Relaxed))
34}
35#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
36pub enum ReaderThemeMode {
37    #[default]
38    FollowSystem,
39    Light,
40    Dark,
41    Paper,
42}
43
44impl ReaderThemeMode {
45    pub const fn next(self) -> Self {
46        match self {
47            Self::FollowSystem => Self::Light,
48            Self::Light => Self::Dark,
49            Self::Dark => Self::Paper,
50            Self::Paper => Self::FollowSystem,
51        }
52    }
53
54    /// Resolve effective palette, mapping FollowSystem to the active system palette.
55    pub fn resolved_palette(self, system: viui::theme::SystemPalette) -> viui::theme::SystemPalette {
56        match self {
57            Self::FollowSystem => system,
58            Self::Light => viui::theme::SystemPalette::Light,
59            Self::Dark => viui::theme::SystemPalette::Dark,
60            Self::Paper => viui::theme::SystemPalette::Paper,
61        }
62    }
63
64    /// Obtain the shared ViTheme reference for the resolved palette.
65    pub fn resolved_theme(self, system: viui::theme::SystemPalette) -> &'static dyn viui::theme::ViTheme {
66        match self.resolved_palette(system) {
67            viui::theme::SystemPalette::Light => &viui::theme::LIGHT_THEME,
68            viui::theme::SystemPalette::Dark => &viui::theme::DARK_THEME,
69            viui::theme::SystemPalette::Paper => &viui::theme::PAPER_THEME,
70        }
71    }
72
73    pub fn bg_color(self) -> Color {
74        self.resolved_bg(active_system_palette())
75    }
76
77    pub fn resolved_bg(self, system: viui::theme::SystemPalette) -> Color {
78        self.resolved_theme(system).reader_bg()
79    }
80
81    pub fn text_color(self) -> Color {
82        self.resolved_text(active_system_palette())
83    }
84
85    pub fn resolved_text(self, system: viui::theme::SystemPalette) -> Color {
86        self.resolved_theme(system).reader_text()
87    }
88
89    pub fn muted_color(self) -> Color {
90        self.resolved_muted(active_system_palette())
91    }
92
93    pub fn resolved_muted(self, system: viui::theme::SystemPalette) -> Color {
94        self.resolved_theme(system).reader_subtext()
95    }
96
97    pub fn accent_color(self) -> Color {
98        self.resolved_accent(active_system_palette())
99    }
100
101    pub fn resolved_accent(self, system: viui::theme::SystemPalette) -> Color {
102        self.resolved_theme(system).accent()
103    }
104
105    pub fn link_color(self) -> Color {
106        self.resolved_link(active_system_palette())
107    }
108
109    pub fn resolved_link(self, system: viui::theme::SystemPalette) -> Color {
110        self.resolved_theme(system).reader_link()
111    }
112
113    pub const fn name(self) -> &'static str {
114        match self {
115            Self::FollowSystem => "Follow System",
116            Self::Light => "Light",
117            Self::Dark => "Dark",
118            Self::Paper => "Paper",
119        }
120    }
121
122    /// Remap legacy dark-theme span colors to the corresponding semantic color of this reader mode.
123    pub fn remap_span_color(self, c: Color) -> Color {
124        if c == theme::ACCENT_BLUE {
125            self.accent_color()
126        } else if c == theme::ACCENT_CYAN {
127            self.text_color()
128        } else if c == theme::TEXT_MUTED {
129            self.muted_color()
130        } else if c == theme::TEXT_PRIMARY {
131            self.text_color()
132        } else {
133            c
134        }
135    }
136}

138pub fn draw_book_icon(surf: &mut ViSurface, cx: i32, cy: i32, color: Color) {
139    // Left page outline
140    fill_rect(surf, cx - 7, cy - 5, 6, 1, color);
141    fill_rect(surf, cx - 7, cy + 5, 6, 1, color);
142    fill_rect(surf, cx - 7, cy - 5, 1, 11, color);
143    fill_rect(surf, cx - 1, cy - 5, 1, 11, color);
144    // Left page lines

146    fill_rect(surf, cx - 5, cy + 1, 3, 1, color);
147
148    // Right page outline
149    fill_rect(surf, cx + 1, cy - 5, 6, 1, color);
150    fill_rect(surf, cx + 1, cy + 5, 6, 1, color);
151    fill_rect(surf, cx + 1, cy - 5, 1, 11, color);
152    fill_rect(surf, cx + 7, cy - 5, 1, 11, color);
153    // Right page lines
154    fill_rect(surf, cx + 3, cy - 2, 3, 1, color);
155    fill_rect(surf, cx + 3, cy + 1, 3, 1, color);
156
157    // Center spine
158    fill_rect(surf, cx, cy - 5, 1, 12, color);
159}

188/// Paint measured, grayscale-antialiased text within a caller's viewport.
189pub fn text(
190    fonts: &mut TextFonts, surf: &mut ViSurface, x: f32, y: f32,
191    value: &str, face: FontFace, px: f32, color: Color,
192    clip: (i32, i32, i32, i32),
193) -> f32 {
194    let width = surf.width();
195    let height = surf.height();
196    let stride = surf.stride();
197    fonts.draw_text(surf.pixels_mut(), width, height, stride, x, y, value,
198        face, px, color.as_bgra_bytes(), clip)
199}

201pub fn clipped_rect(
202    surf: &mut ViSurface, x: i32, y: i32, w: u32, h: u32,
203    color: Color, clip: (i32, i32, i32, i32),
204) {
205    let x0 = x.max(clip.0);
206    let y0 = y.max(clip.1);
207    let x1 = x.saturating_add(w as i32).min(clip.2);
208    let y1 = y.saturating_add(h as i32).min(clip.3);
209    if x1 > x0 && y1 > y0 {
210        fill_rect(surf, x0, y0, (x1 - x0) as u32, (y1 - y0) as u32, color);
211    }
212}

214pub fn draw_image_clipped(
215    surf: &mut ViSurface,
216    x: i32,
217    y: i32,
218    w: u32,
219    h: u32,
220    pixels: &[u8],
221    src_stride: usize,
222    source_height: u32,
223    clip: (i32, i32, i32, i32),
224) {
225    let sw = surf.width() as i32;
226    let sh = surf.height() as i32;
227    let dst_stride = surf.stride();
228    let dst_pixels = surf.pixels_mut();
229
230    let x0 = x.max(0).max(clip.0);
231    let y0 = y.max(0).max(clip.1);
232    let x1 = (x + w as i32).min(sw).min(clip.2);
233    let y1 = (y + h as i32).min(sh).min(clip.3);
234    if x0 >= x1 || y0 >= y1 {
235        return;
236    }
237    let source_width = src_stride / 4;
238
239    for py in y0..y1 {
240        let src_y = ((py - y) as u64 * source_height as u64 / h as u64) as usize;
241        let src_row = src_y * src_stride;
242        let dst_row = py as usize * dst_stride;
243
244        for px in x0..x1 {
245            let src_x = ((px - x) as u64 * source_width as u64 / w as u64) as usize;
246            let src_offset = src_row + src_x * 4;
247            let dst_offset = dst_row + px as usize * 4;
248
249            if src_offset + 3 < pixels.len() && dst_offset + 3 < dst_pixels.len() {
250                let sa = pixels[src_offset + 3] as u32;
251                if sa == 255 {
252                    dst_pixels[dst_offset..dst_offset + 4]
253                        .copy_from_slice(&pixels[src_offset..src_offset + 4]);
254                } else if sa > 0 {
255                    let inv = 255 - sa;
256                    let sb = pixels[src_offset] as u32;
257                    let sg = pixels[src_offset + 1] as u32;
258                    let sr = pixels[src_offset + 2] as u32;
259
260                    let db = dst_pixels[dst_offset] as u32;
261                    let dg = dst_pixels[dst_offset + 1] as u32;
262                    let dr = dst_pixels[dst_offset + 2] as u32;
263
264                    dst_pixels[dst_offset] = ((sb * sa + db * inv) / 255) as u8;
265                    dst_pixels[dst_offset + 1] = ((sg * sa + dg * inv) / 255) as u8;
266                    dst_pixels[dst_offset + 2] = ((sr * sa + dr * inv) / 255) as u8;
267                    dst_pixels[dst_offset + 3] = 255;
268                }
269            }
270        }
271    }
