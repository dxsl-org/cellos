16/// Semantic text roles the document parsers stamp on spans. The values are
17/// tags, not paint: [`ink::resolve`] maps them to the painting theme, so a
18/// parsed document re-tints whenever the reader or system palette changes.
19pub mod ink {
20    use super::Color;
21    use viui::theme::ViTheme;
22
23    /// Secondary text: captions, metadata, image placeholders, quotes.
24    pub const MUTED: Color = Color::rgb(166, 173, 200);
25    /// Emphasised text: top-level headings, prices, pagination and menu links.
26    pub const ACCENT: Color = Color::rgb(137, 180, 250);
27
28    /// Colour that paints a span tagged `c` under `theme`; other colours pass through.
29    pub fn resolve(theme: &dyn ViTheme, c: Color) -> Color {
30        if c == ACCENT {
31            theme.accent()
32        } else if c == MUTED {
33            theme.reader_subtext()
34        } else {
35            c
36        }
37    }
38}

124#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
125pub enum ReaderThemeMode {
126    #[default]
127    FollowSystem,
128    Light,
129    Dark,
130    Paper,
131}
132
133impl ReaderThemeMode {
134    pub const fn next(self) -> Self {
135        match self {
136            Self::FollowSystem => Self::Light,
137            Self::Light => Self::Dark,
138            Self::Dark => Self::Paper,
139            Self::Paper => Self::FollowSystem,
140        }
141    }
142
143    /// Effective palette, with FollowSystem mapped to `system`.
144    pub fn resolved_palette(self, system: SystemPalette) -> SystemPalette {
145        match self {
146            Self::FollowSystem => system,
147            Self::Light => SystemPalette::Light,
148            Self::Dark => SystemPalette::Dark,
149            Self::Paper => SystemPalette::Paper,
150        }
151    }
152
153    pub const fn name(self) -> &'static str {
154        match self {
155            Self::FollowSystem => "Follow System",
156            Self::Light => "Light",
157            Self::Dark => "Dark",
158            Self::Paper => "Paper",
159        }
160    }
161}

163/// Rounded box clipped to the caller's viewport: `fill`, then a 1px `stroke`
164/// covering the same outline.
165pub fn clipped_round_rect(
166    surf: &mut ViSurface,
167    (x, y, w, h): (i32, i32, u32, u32),
168    radius: u32,
169    fill: Option<Color>,
170    stroke: Option<Color>,
171    clip: (i32, i32, i32, i32),
172) {
173    if clip.2 <= clip.0 || clip.3 <= clip.1 {
174        return;
175    }
176    let (width, height, stride) = (surf.width(), surf.height(), surf.stride());
177    let mut canvas = Canvas::new(surf.pixels_mut(), width, height, stride).clip_to(
178        clip.0,
179        clip.1,
180        (clip.2 - clip.0) as u32,
181        (clip.3 - clip.1) as u32,
182    );
183    let shape = RoundRect::new(x, y, w, h, Radii::all(radius));
184    if let Some(color) = fill {
185        canvas.fill_round(&shape, color.as_bgra_bytes());
186    }
187    if let Some(color) = stroke {
188        canvas.stroke_round(&shape, color.as_bgra_bytes());
189    }
190}

192/// Paint measured, grayscale-antialiased text within a caller's viewport.
193// reason: thin wrapper over `TextFonts::draw_text` with many drawing call sites; bundling the
194// position/style/clip args would only churn callers.
195#[allow(clippy::too_many_arguments)]
196pub fn text(
197    fonts: &mut TextFonts,
198    surf: &mut ViSurface,
199    x: f32,
200    y: f32,
201    value: &str,
202    face: FontFace,
203    px: f32,
204    color: Color,
205    clip: (i32, i32, i32, i32),
206) -> f32 {
207    let width = surf.width();
208    let height = surf.height();
209    let stride = surf.stride();
210    fonts.draw_text(
211        surf.pixels_mut(),
212        width,
213        height,
214        stride,
215        x,
216        y,
217        value,
218        face,
219        px,
220        color.as_bgra_bytes(),
221        clip,
222    )
223}

225pub fn clipped_rect(
226    surf: &mut ViSurface,
227    x: i32,
228    y: i32,
229    w: u32,
230    h: u32,
231    color: Color,
232    clip: (i32, i32, i32, i32),
233) {
234    let x0 = x.max(clip.0);
235    let y0 = y.max(clip.1);
236    let x1 = x.saturating_add(w as i32).min(clip.2);
237    let y1 = y.saturating_add(h as i32).min(clip.3);
238    if x1 > x0 && y1 > y0 {
239        fill_rect(surf, x0, y0, (x1 - x0) as u32, (y1 - y0) as u32, color);
240    }
241}

243// reason: low-level blit helper with multiple drawing call sites; bundling geometry and source
244// args would only churn callers.
245#[allow(clippy::too_many_arguments)]
246pub fn draw_image_clipped(
247    surf: &mut ViSurface,
248    x: i32,
249    y: i32,
250    w: u32,
251    h: u32,
252    pixels: &[u8],
253    src_stride: usize,
254    source_height: u32,
255    clip: (i32, i32, i32, i32),
256) {
257    let sw = surf.width() as i32;
258    let sh = surf.height() as i32;
259    let dst_stride = surf.stride();
260    let dst_pixels = surf.pixels_mut();
261
262    let x0 = x.max(0).max(clip.0);
263    let y0 = y.max(0).max(clip.1);
264    let x1 = (x + w as i32).min(sw).min(clip.2);
265    let y1 = (y + h as i32).min(sh).min(clip.3);
266    if x0 >= x1 || y0 >= y1 {
267        return;
268    }
269    let source_width = src_stride / 4;
270
271    for py in y0..y1 {
272        let src_y = ((py - y) as u64 * source_height as u64 / h as u64) as usize;
273        let src_row = src_y * src_stride;
274        let dst_row = py as usize * dst_stride;
275
276        for px in x0..x1 {
277            let src_x = ((px - x) as u64 * source_width as u64 / w as u64) as usize;
278            let src_offset = src_row + src_x * 4;
279            let dst_offset = dst_row + px as usize * 4;
280
281            if src_offset + 3 < pixels.len() && dst_offset + 3 < dst_pixels.len() {
282                let sa = pixels[src_offset + 3] as u32;
283                if sa == 255 {
284                    dst_pixels[dst_offset..dst_offset + 4]
285                        .copy_from_slice(&pixels[src_offset..src_offset + 4]);
286                } else if sa > 0 {
287                    let inv = 255 - sa;
288                    let sb = pixels[src_offset] as u32;
289                    let sg = pixels[src_offset + 1] as u32;
290                    let sr = pixels[src_offset + 2] as u32;
291
292                    let db = dst_pixels[dst_offset] as u32;
293                    let dg = dst_pixels[dst_offset + 1] as u32;
294                    let dr = dst_pixels[dst_offset + 2] as u32;
295
296                    dst_pixels[dst_offset] = ((sb * sa + db * inv) / 255) as u8;
297                    dst_pixels[dst_offset + 1] = ((sg * sa + dg * inv) / 255) as u8;
298                    dst_pixels[dst_offset + 2] = ((sr * sa + dr * inv) / 255) as u8;
299                    dst_pixels[dst_offset + 3] = 255;
300                }
301            }
302        }
303    }
304}

313    /// WCAG 2.1 relative luminance of an sRGB colour.
314    fn luminance(c: Color) -> f64 {
315        let channel = |v: u8| {
316            let s = f64::from(v) / 255.0;
317            if s <= 0.04045 {
318                s / 12.92
319            } else {
320                ((s + 0.055) / 1.055).powf(2.4)
321            }
322        };
323        0.2126 * channel(c.r()) + 0.7152 * channel(c.g()) + 0.0722 * channel(c.b())
324    }
325
326    /// WCAG 2.1 contrast ratio between two colours.
327    fn contrast(a: Color, b: Color) -> f64 {
328        let (la, lb) = (luminance(a), luminance(b));
329        (la.max(lb) + 0.05) / (la.min(lb) + 0.05)
330    }

332    #[test]
333    fn every_code_token_keeps_wcag_aa_contrast_on_its_surface() {
334        let palettes: [(&str, CodePalette); 3] =
335            [("Dark", DARK), ("Light", LIGHT), ("Paper", PAPER)];
336        for (name, p) in palettes {
337            for (role, ink) in [
338                ("plain", p.plain),
339                ("comment", p.comment),
340                ("keyword", p.keyword),
341                ("string", p.string),
342                ("number", p.number),
343                ("inline", p.inline),
344            ] {
345                let ratio = contrast(ink, p.surface);
346                assert!(
347                    ratio >= 4.5,
348                    "{name} {role} contrast {ratio:.2}:1 is below 4.5:1"
349                );
350            }
351        }
