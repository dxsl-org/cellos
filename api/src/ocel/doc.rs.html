48    pub fn face(&self) -> FontFace {
49        if self.code {
50            if self.bold { FontFace::MonoBold } else { FontFace::MonoRegular }
51        } else {
52            match (self.bold, self.italic) {
53                (false, false) => FontFace::UiRegular,
54                (true, false) => FontFace::UiSemibold,
55                (false, true) => FontFace::UiItalic,
56                (true, true) => FontFace::UiSemiboldItalic,
57            }
58        }
59    }

211fn validate_arena(arena:&dom_arena::DocumentArena,external:&[(dom_arena::NodeId,String)])->Result<(),String> {
212    use dom_arena::NodeData;
213    if arena.nodes.len()>crate::budget::MAX_DOM_NODES {return Err(String::from("DOM exceeds node budget"));}
214    let mut bytes=0usize;
215    for node in &arena.nodes {
216        let mut current=Some(node.id);let mut depth=0usize;
217        while let Some(id)=current {
218            depth+=1;if depth>crate::budget::MAX_DOM_DEPTH+1 {return Err(String::from("DOM exceeds depth budget or contains a cycle"));}
219            current=arena.get(id).and_then(|node|node.parent);
220        }
221        match &node.data {
222            NodeData::Text(text)|NodeData::Comment(text)=>bytes=bytes.saturating_add(text.len()),
223            NodeData::Element{tag,attributes}=>{bytes=bytes.saturating_add(tag.len());for (name,value) in attributes {bytes=bytes.saturating_add(name.len()+value.len());}},
224            _=>{}
225        }
226        if bytes>crate::budget::MAX_TEXT_BYTES {return Err(String::from("DOM exceeds text byte budget"));}
227    }
228    let mut css=String::new();
229    for node in &arena.nodes {
230        if let Some(style)=node.get_attribute("style") {crate::budget::validate_css(&format!("x{{{}}}",style))?;}
231        if node.tag()==Some("style") {
232            let text=arena.get_text_content(node.id);
233            if css.len().saturating_add(text.len())>crate::budget::MAX_CSS_BYTES{return Err(String::from("CSS exceeds byte budget"));}
234            css.push_str(&text);
235        }
236    }
237    for (_,text) in external {
238        if css.len().saturating_add(text.len())>crate::budget::MAX_CSS_BYTES{return Err(String::from("CSS exceeds byte budget"));}
239        css.push_str(text);
240    }
241    crate::budget::validate_css(&css)
242}

252pub fn apply_bionic_reading(spans: &[StyledSpan]) -> Vec<StyledSpan> {
253    let mut out = Vec::new();
254    for span in spans {
255        if span.code || span.text.is_empty() {
256            out.push(span.clone());
257            continue;
258        }
259        let text = &span.text;
260        let mut word_start = None;
261        let bytes = text.as_bytes();
262        let mut i = 0;
263
264        while i < bytes.len() {
265            let ch = text[i..].chars().next().unwrap();
266            let ch_len = ch.len_utf8();
267            let is_word_char = ch.is_alphanumeric();
268
269            if is_word_char {
270                if word_start.is_none() {
271                    word_start = Some(i);
272                }
273            } else {
274                if let Some(start) = word_start.take() {
275                    let word = &text[start..i];
276                    push_bionic_word(word, span, &mut out);
277                }
278                let non_word = &text[i..i + ch_len];
279                let mut non_span = span.clone();
280                non_span.text = String::from(non_word);
281                out.push(non_span);
282            }
283            i += ch_len;
284        }
285        if let Some(start) = word_start.take() {
286            let word = &text[start..];
287            push_bionic_word(word, span, &mut out);
288        }
289    }
290    out
291}

293fn push_bionic_word(word: &str, base_span: &StyledSpan, out: &mut Vec<StyledSpan>) {
294    let count = word.chars().count();
295    if count <= 1 {
296        let mut s = base_span.clone();
297        s.text = String::from(word);
298        out.push(s);
299        return;
300    }
301    let bold_count = (count + 1) / 2;
302    let mut bold_part = String::new();
303    let mut rest_part = String::new();
304    for (idx, ch) in word.chars().enumerate() {
305        if idx < bold_count {
306            bold_part.push(ch);
307        } else {
308            rest_part.push(ch);
309        }
310    }
311    if !bold_part.is_empty() {
312        let mut s = base_span.clone();
313        s.bold = true;
314        s.text = bold_part;
315        out.push(s);
316    }
317    if !rest_part.is_empty() {
318        let mut s = base_span.clone();
319        s.text = rest_part;
320        out.push(s);
321    }
322}

349        if let Some(web)=&self.web_layout {return web.search(query);}
350        let query=fold_text(query.trim());
351        if query.is_empty(){return Vec::new();}
352        let mut matches=Vec::new();let mut text=String::new();
353        for b in &self.layout_boxes {
354            let mut found=false;
355            for line in &b.lines {
356                text.clear();for run in &line.runs {text.push_str(&run.span.text);}

359            if let Some(rows)=&b.table_rows {for row in rows {for cell in &row.cells {for line in &cell.lines {
360                text.clear();for run in &line.runs {text.push_str(&run.span.text);}
361                found|=fold_text(&text).contains(&query);
362            }}}}
363            if found {matches.push(b.y_offset);}
364        }

384    pub fn compute_layout(&mut self,fonts:&mut TextFonts,content_width:u32) {
385        if self.arena.is_some() && !self.is_reader_mode(){self.relayout_from_arena(fonts,content_width,600);return;}
386        self.last_error=None;
387        if let Err(error)=self.reserve_layout(content_width){self.last_error=Some(error);return;}
388        self.web_layout=None;self.layout_boxes.clear();
389        let scale=self.zoom as f32/100.0;
390        let content_width=if self.is_reader_mode(){content_width.min(self.reader_column)}else{content_width};
391        let logical_width=(content_width as f32/scale).max(1.0) as u32;
392        let mut y=16;
393        let width=logical_width.saturating_sub(40).max(1) as f32;
394        let nodes=self.reader_nodes.as_ref().unwrap_or(&self.nodes);
395        for node in nodes {
396            let mut b=LayoutBox::new(y);
397            let mut gap=0;
398            match node {
399                DocNode::Heading{level,text}=>{
400                    let px=match level {1=>32.0,2=>26.0,3=>22.0,_=>18.0};
401                    let mut span=StyledSpan::plain(text);span.bold=true;
402                    if *level==1 {span.color=Some(theme::ACCENT_BLUE);}
403                    b.lines=lines_for(fonts,&[span],px,width,false);gap=8;
404                }
405                DocNode::Paragraph{spans}=>{
406                    let effective_spans;
407                    let p_spans: &[StyledSpan] = if self.bionic_reading && self.is_reader_mode() {
408                        effective_spans = apply_bionic_reading(spans);
409                        &effective_spans
410                    } else {
411                        spans
412                    };
413                    let mono_spans;
414                    let final_spans: &[StyledSpan] = if self.reader_mono && self.is_reader_mode() {
415                        mono_spans = p_spans.iter().map(|s| { let mut m = s.clone(); m.code = true; m }).collect::<Vec<_>>();
416                        &mono_spans
417                    } else {
418                        p_spans
419                    };
420                    b.lines=lines_for(fonts,final_spans,16.0,width,false);gap=6;
421                }
422                DocNode::CodeBlock{lang,lines}=>{
423                    let mut highlighter=crate::parser::syntax::Highlighter::new(lang);
424                    for text in lines {
425                        let mut spans=highlighter.line(text);
426                        for span in &mut spans {span.code=true;}
427                        b.lines.extend(lines_for(fonts,&spans,14.0,(width-8.0).max(1.0),true));
428                    }
429                    b.bg_color=Some(theme::BG_INPUT);b.border_color=Some(theme::BORDER);
430                    b.left_margin=8;gap=8;
431                }
432                DocNode::ListItem{bullet,indent,spans}=>{
433                    b.left_margin=*indent as i32*20+12;
434                    let mut list=alloc::vec![StyledSpan::plain(&if *bullet=='\0'{String::new()}else{format!("{} ",bullet)})];
435                    list[0].bold=true;list[0].color=Some(theme::ACCENT_CYAN);
436                    let effective_spans;
437                    let li_spans: &[StyledSpan] = if self.bionic_reading && self.is_reader_mode() {
438                        effective_spans = apply_bionic_reading(spans);
439                        &effective_spans
440                    } else {
441                        spans
442                    };
443                    let mono_spans;
444                    let final_spans: &[StyledSpan] = if self.reader_mono && self.is_reader_mode() {
445                        mono_spans = li_spans.iter().map(|s| { let mut m = s.clone(); m.code = true; m }).collect::<Vec<_>>();
446                        &mono_spans
447                    } else {
448                        li_spans
449                    };
450                    list.extend_from_slice(final_spans);
451                    b.lines=lines_for(fonts,&list,16.0,(width-b.left_margin as f32).max(1.0),false);
452                }
453                DocNode::Blockquote{lines}=>{
454                    b.left_margin=16;b.border_color=Some(theme::ACCENT_BLUE);
455                    for text in lines {
456                        let mut span=StyledSpan::plain(text);span.italic=true;span.color=Some(theme::TEXT_MUTED);
457                        b.lines.extend(lines_for(fonts,&[span],16.0,(width-16.0).max(1.0),false));
458                    }
459                    gap=4;
460                }
461                DocNode::RawLines{lines}=>{
462                    b.left_margin=4;
463                    for text in lines {
464                        b.lines.extend(lines_for(fonts,&[StyledSpan::plain(text)],16.0,(width-4.0).max(1.0),true));
465                    }
466                }
467                DocNode::Rule=>{b.is_rule=true;b.height=16;}
468                DocNode::Image{width:iw,height:ih,pixels} | DocNode::DomImage{width:iw,height:ih,pixels,..}=>{
469                    let display_w=(*iw).min(logical_width.saturating_sub(32).max(1));
470                    let display_h=((*ih as u64*display_w as u64)/(*iw).max(1) as u64).max(1) as u32;
471                    b.height=display_h as i32+16;b.left_margin=16;
472                    b.table_width=display_w;
473                    b.image=Some((*iw,*ih,pixels.clone()));
474                }
475                DocNode::LinkedImage{width:iw,height:ih,pixels,link}=>{
476                    let display_w=(*iw).min(logical_width.saturating_sub(32).max(1));
477                    let display_h=((*ih as u64*display_w as u64)/(*iw).max(1) as u64).max(1) as u32;
478                    b.height=display_h as i32+16;b.left_margin=16;
479                    b.table_width=display_w;
480                    b.image=Some((*iw,*ih,pixels.clone()));
481                    b.image_link=Some(link.clone());
482                }
483                DocNode::Table{headers,rows}=>{
484                    let count=headers.len().max(rows.iter().map(|r|r.len()).max().unwrap_or(0));
485                    if count==0{b.height=8;y+=8;self.layout_boxes.push(b);continue;}
486                    let mut desired=alloc::vec![12.0f32;count];
487                    for (i,text) in headers.iter().enumerate() {
488                        desired[i]=desired[i].max(fonts.measure(FontFace::UiSemibold,text,16.0)+12.0);
489                    }
490                    for row in rows {for (i,text) in row.iter().enumerate() {
491                        desired[i]=desired[i].max(fonts.measure(FontFace::UiRegular,text,16.0)+12.0);
492                    }}
493                    let sum: f32=desired.iter().sum();
494                    let available=logical_width.saturating_sub(40).max(1);
495                    let mut widths=Vec::new();let mut used=0;
496                    for (i,w) in desired.iter().enumerate() {
497                        let w=if i+1==count {available.saturating_sub(used)}
498                            else {(*w/sum*available as f32) as u32};
499                        widths.push(w);used+=w;
500                    }
501                    let mut table_rows=Vec::new();let mut row_y=0;
502                    for (header,row) in core::iter::once((true,headers)).filter(|(_,r)|!r.is_empty())
503                        .chain(rows.iter().map(|r|(false,r))) {
504                        let mut cells=Vec::new();let mut x=0;let mut row_h=0;
505                        for (i,&w) in widths.iter().enumerate() {
506                            let mut span=StyledSpan::plain(row.get(i).map_or("",String::as_str));
507                            span.bold=header;span.color=Some(if header{theme::ACCENT_CYAN}else{theme::TEXT_PRIMARY});
508                            let lines=lines_for(fonts,&[span],16.0,w.saturating_sub(12).max(1) as f32,false);
509                            row_h=row_h.max(lines_height(&lines) as u32+12);
510                            cells.push(TableCellLayout{lines,x,width:w});x+=w as i32;
511                        }
512                        table_rows.push(TableRowLayout{y_rel:row_y,height:row_h,cells,is_header:header});
513                        row_y+=row_h as i32;
514                    }
515                    b.table_rows=Some(table_rows);b.table_width=used;
516                    b.left_margin=16;b.height=row_y+12;gap=8;
517                }
518                DocNode::Button{text,node_id}=>{
519                    let mut span=StyledSpan::plain(text.trim());span.bold=true;
520                    b.lines=lines_for(fonts,&[span],16.0,width,false);
521                    b.table_width=b.lines.iter().map(|l|l.width as u32+16).max().unwrap_or(16);
522                    b.left_margin=8;b.bg_color=Some(theme::ACCENT_BLUE);
523                    b.border_color=Some(theme::BORDER);b.node_id=*node_id;gap=8;
524                }
525                DocNode::Input { input_type, name, value, placeholder, checked, node_id } => {
526                    let inp_w = match input_type.as_str() {
527                        "checkbox" => 20,
528                        "submit" | "button" => {
529                            let label = if value.is_empty() { "Submit" } else { value.as_str() };

858    pub fn form_url(&self,submit:dom_arena::NodeId,base:&str)->Result<String,String> {
859        let arena=self.arena.as_ref().ok_or_else(||String::from("Document has no HTML form"))?;
860        let owner=form_owner(arena,submit).ok_or_else(||String::from("Control does not belong to a form"))?;
861        let form=arena.get(owner).ok_or_else(||String::from("Form no longer exists"))?;
862        let submit_node=arena.get(submit).ok_or_else(||String::from("Control no longer exists"))?;
863        if submit_node.get_attribute("disabled").is_some(){return Err(String::from("Disabled control cannot submit a form"));}
864        if submit_node.tag()==Some("button")&&!submit_node.get_attribute("type").unwrap_or("submit").eq_ignore_ascii_case("submit"){return Err(String::from("Button is not a submit control"));}
865        let method=submit_node.get_attribute("formmethod").or_else(||form.get_attribute("method")).unwrap_or("get");
866        if !method.eq_ignore_ascii_case("get"){return Err(format!("Unsupported form method: {}",method));}
867        let enctype=submit_node.get_attribute("formenctype").or_else(||form.get_attribute("enctype")).unwrap_or("application/x-www-form-urlencoded");
868        if !enctype.eq_ignore_ascii_case("application/x-www-form-urlencoded"){return Err(format!("Unsupported form encoding: {}",enctype));}
869        let action=submit_node.get_attribute("formaction").or_else(||form.get_attribute("action")).unwrap_or("");
870        let action=if action.is_empty(){String::from(base)}else{crate::resources::resolve(base,action)?};
871        let (target,fragment)=action.split_once('#').map_or((action.as_str(),None),|(path,fragment)|(path,Some(fragment)));
872        let target=target.split('?').next().unwrap_or(target);
873        let mut query=String::new();
874        for node in &arena.nodes {
875            if !matches!(node.tag(),Some("input"|"textarea"|"button"|"select"))||form_owner(arena,node.id)!=Some(owner)||node.get_attribute("disabled").is_some(){continue;}
876            if node.tag()==Some("select"){return Err(String::from("Unsupported form control type: select"));}
877            let name=node.get_attribute("name").unwrap_or("");if name.is_empty(){continue;}
878            let kind=if node.tag()==Some("button"){node.get_attribute("type").unwrap_or("submit")}else{node.get_attribute("type").unwrap_or("text")}.to_ascii_lowercase();
879            let kind=kind.as_str();
880            if matches!(kind,"reset"|"button")||(kind=="submit"&&node.id!=submit){continue;}
881            if !matches!(kind,"text"|"textarea"|"password"|"email"|"search"|"url"|"tel"|"hidden"|"checkbox"|"submit"){return Err(format!("Unsupported form control type: {}",kind));}
882            let input=self.input(node.id);
883            let checked=input.map_or(node.get_attribute("checked").is_some(),|input|input.checked);
884            if kind=="checkbox"&&!checked{continue;}
885            let text;
886            let value=if node.tag()==Some("button"){node.get_attribute("value").unwrap_or("")}
887                else if let Some(input)=input{input.value.as_str()}
888                else if node.tag()==Some("textarea"){text=arena.get_text_content(node.id);text.as_str()}
889                else{node.get_attribute("value").unwrap_or(if kind=="checkbox"{"on"}else{""})};
890            let value=if kind=="checkbox"&&value.is_empty()&&node.get_attribute("value").is_none(){"on"}else{value};
891            if !query.is_empty(){query.push('&');}encode_form(name,&mut query);query.push('=');encode_form(value,&mut query);
892        }
893        let mut result=format!("{}?{}",target,query);
894        if let Some(fragment)=fragment{result.push('#');result.push_str(fragment);}

961        bytes+=(self.layout_boxes.capacity()-self.layout_boxes.len())*core::mem::size_of::<LayoutBox>();

965    }

983fn encode_form(text:&str,out:&mut String) {
984    const HEX:&[u8]=b"0123456789ABCDEF";
985    for byte in text.bytes() {match byte {b' '=>out.push('+'),b'A'..=b'Z'|b'a'..=b'z'|b'0'..=b'9'|b'*'|b'-'|b'.'|b'_'=>out.push(byte as char),_=>{out.push('%');out.push(HEX[(byte>>4) as usize] as char);out.push(HEX[(byte&15) as usize] as char);}}}
986}
987fn decode_fragment(text:&str)->Option<String> {
988    let mut bytes=Vec::new();let mut source=text.bytes();
989    while let Some(byte)=source.next(){if byte==b'%' {let hi=(source.next()? as char).to_digit(16)?;let lo=(source.next()? as char).to_digit(16)?;bytes.push((hi*16+lo) as u8);}else{bytes.push(byte);}}
990    String::from_utf8(bytes).ok()
991}

1017    /// Stage a bounded batch so an invalid mutation never partially changes a live DOM.
1018    pub fn apply_mutations_checked(&mut self,mutations:&[dom_arena::DomMutation])->Result<(),String> {
1019        use dom_arena::{DomMutation,NodeData};
1020        if mutations.len()>dom_arena::OCEL_JS_MAX_MUTATIONS{return Err(String::from("Mutation count budget exceeded"));}
1021        let arena=self.arena.as_ref().ok_or_else(||String::from("Document has no DOM"))?;
1022        validate_arena(arena,&self.external_stylesheets)?;
1023        let mut growth=0usize;let mut nodes=arena.nodes.len();
1024        for mutation in mutations {
1025            let size=match mutation {
1026                DomMutation::CreateNode{data,..}=>{nodes=nodes.saturating_add(1);match data {
1027                    NodeData::Text(text)|NodeData::Comment(text)=>text.len(),
1028                    NodeData::Element{tag,attributes}=>tag.len()+attributes.iter().map(|(name,value)|name.len()+value.len()).sum::<usize>(),_=>0,
1029                }},
1030                DomMutation::SetText{text,..}=>{nodes=nodes.saturating_add(1);text.len()},
1031                DomMutation::SetAttribute{key,val,..}=>key.len()+val.len(),
1032                DomMutation::RemoveAttribute{key,..}=>key.len(),
1033                DomMutation::SetDocumentTitle{title}=>title.len(),
1034                _=>0,
1035            };
1036            growth=growth.saturating_add(size);
1037            if growth>dom_arena::OCEL_JS_MAX_MUTATION_BYTES||nodes>crate::budget::MAX_DOM_NODES {
1038                return Err(String::from("Mutation allocation budget exceeded"));
1039            }
1040        }
1041        let current_bytes:usize=arena.nodes.iter().map(|node|match &node.data {
1042            NodeData::Text(text)|NodeData::Comment(text)=>text.len(),
1043            NodeData::Element{tag,attributes}=>tag.len()+attributes.iter().map(|(name,value)|name.len()+value.len()).sum::<usize>(),_=>0,
1044        }).sum();
1045        if current_bytes.saturating_add(growth)>crate::budget::MAX_TEXT_BYTES{return Err(String::from("Mutation exceeds document text budget"));}
1046        if mutations.iter().all(|mutation|matches!(mutation,DomMutation::SetDocumentTitle{..})){return Ok(());}
1047        let clone_bytes=arena.nodes.iter().map(|node|core::mem::size_of::<dom_arena::DomNode>()+match &node.data {
1048            NodeData::Text(text)|NodeData::Comment(text)=>text.len(),
1049            NodeData::Element{tag,attributes}=>tag.len()+attributes.len()*core::mem::size_of::<(String,String)>()+attributes.iter().map(|(name,value)|name.len()+value.len()).sum::<usize>(),_=>0,
1050        }).sum::<usize>();
1051        if self.resident_bytes().saturating_add(clone_bytes).saturating_add(growth.saturating_mul(4))>self.memory_budget{return Err(String::from("Mutation staging exceeds resident memory budget"));}
1052        let mutation_layout_reserve = growth.saturating_mul(64).saturating_add(mutations.len().saturating_mul(1024)).saturating_add(64 * 1024);
1053        if self.resident_bytes().saturating_add(clone_bytes).saturating_add(mutation_layout_reserve) > self.memory_budget {
1054            return Err(String::from("Mutation layout exceeds resident memory budget"));
1055        }
1056        let mut staged=arena.clone();
1057        for mutation in mutations {
1058            if !staged.apply_mutation(mutation){return Err(String::from("Invalid or divergent DOM mutation"));}
1059            validate_arena(&staged,&self.external_stylesheets)?;
1060        }
1061        if !staged.validate(){return Err(String::from("Mutation creates invalid DOM links"));}
1062        self.arena=Some(staged);Ok(())
1063    }
1064}
