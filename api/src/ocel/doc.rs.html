15#[derive(Clone, Debug, PartialEq, Eq)]
16pub struct StyledSpan {
17    pub text: String,
18    pub bold: bool,
19    pub italic: bool,
20    pub code: bool,
21    pub link: Option<String>,
22    pub color: Option<Color>,
23}
24
25impl StyledSpan {
26    pub fn plain(text: &str) -> Self {
27        Self {
28            text: String::from(text),
29            bold: false,
30            italic: false,
31            code: false,
32            link: None,
33            color: None,
34        }
35    }
36
37    pub fn code(text: &str) -> Self {
38        Self {
39            text: String::from(text),
40            bold: false,
41            italic: false,
42            code: true,
43            link: None,
44            color: Some(theme::ACCENT_CYAN),
45        }
46    }
47
48    pub fn face(&self) -> FontFace {
49        if self.code {
50            if self.bold { FontFace::MonoBold } else { FontFace::MonoRegular }
51        } else {
52            match (self.bold, self.italic) {
53                (false, false) => FontFace::UiRegular,
54                (true, false) => FontFace::UiSemibold,
55                (false, true) => FontFace::UiItalic,
56                (true, true) => FontFace::UiSemiboldItalic,
57            }
58        }
59    }
60}

147fn lines_for(fonts:&mut TextFonts,spans:&[StyledSpan],px:f32,width:f32,pre:bool)->Vec<TextLine> {
148    let mut runs=Vec::new();
149    for span in spans {crate::parser::web::append_run(&mut runs,TextRun::new(span.clone(),px,None,0.0));}
150    let height=crate::parser::web::ceil_pixel(fonts.line_height(
151        spans.first().map_or(FontFace::UiRegular,StyledSpan::face),px));
152    wrap_runs(fonts,&runs,width,height,pre,true,0)
153}
154fn lines_height(lines:&[TextLine])->i32 { lines.iter().map(|l|l.height as i32).sum() }

157    pub fn new() -> Self {

161    pub fn clear(&mut self) {
162        self.nodes.clear();self.layout_boxes.clear();self.total_height=0;
163        self.arena=None;self.external_stylesheets.clear();self.web_layout=None;
164    }
165    pub fn search(&self,query:&str)->Vec<i32> {
166        if let Some(web)=&self.web_layout {return web.search(query);}
167        let query:String=ostd::typography::normalized_chars(query.trim()).map(|c|c.to_ascii_lowercase()).collect();
168        if query.is_empty(){return Vec::new();}
169        let mut matches=Vec::new();let mut text=String::new();
170        for b in &self.layout_boxes {
171            let mut found=false;
172            for line in &b.lines {
173                text.clear();for run in &line.runs {text.push_str(&run.span.text);}
174                text.make_ascii_lowercase();found|=text.contains(&query);
175            }
176            if let Some(rows)=&b.table_rows {for row in rows {for cell in &row.cells {for line in &cell.lines {
177                text.clear();for run in &line.runs {text.push_str(&run.span.text);}
178                text.make_ascii_lowercase();found|=text.contains(&query);
179            }}}}
180            if found {matches.push(b.y_offset);}
181        }
182        matches
183    }
184    pub fn relayout_from_arena(&mut self,fonts:&mut TextFonts,width:u32,height:u32)->bool {
185        let Some(arena)=&self.arena else{return false;};
186        match crate::parser::web::layout(fonts,arena,&self.external_stylesheets,width,height) {
187            Ok(layout)=>{
188                self.nodes=crate::parser::html::arena_to_doc_nodes(arena);
189                self.layout_boxes.clear();self.total_height=layout.height;
190                self.web_layout=Some(layout);true
191            }
192            Err(_)=>{self.layout_boxes.clear();self.web_layout=None;self.total_height=0;false}
193        }
194    }
195    pub fn compute_layout(&mut self,fonts:&mut TextFonts,content_width:u32) {
196        if self.arena.is_some(){self.relayout_from_arena(fonts,content_width,600);return;}
197        self.web_layout=None;self.layout_boxes.clear();
198        let mut y=16;
199        let width=content_width.saturating_sub(40).max(1) as f32;
200        for node in &self.nodes {
201            let mut b=LayoutBox::new(y);
202            let mut gap=0;
203            match node {
204                DocNode::Heading{level,text}=>{
205                    let px=match level {1=>32.0,2=>26.0,3=>22.0,_=>18.0};
206                    let mut span=StyledSpan::plain(text);span.bold=true;
207                    if *level==1 {span.color=Some(theme::ACCENT_BLUE);}
208                    b.lines=lines_for(fonts,&[span],px,width,false);gap=8;
209                }
210                DocNode::Paragraph{spans}=>{
211                    b.lines=lines_for(fonts,spans,16.0,width,false);gap=6;
212                }
213                DocNode::CodeBlock{lang,lines}=>{
214                    let mut highlighter=crate::parser::syntax::Highlighter::new(lang);
215                    for text in lines {
216                        let mut spans=highlighter.line(text);
217                        for span in &mut spans {span.code=true;}
218                        b.lines.extend(lines_for(fonts,&spans,14.0,(width-8.0).max(1.0),true));
219                    }
220                    b.bg_color=Some(theme::BG_INPUT);b.border_color=Some(theme::BORDER);
221                    b.left_margin=8;gap=8;
222                }
223                DocNode::ListItem{bullet,indent,spans}=>{
224                    b.left_margin=*indent as i32*20+12;
225                    let mut list=alloc::vec![StyledSpan::plain(&format!("{} ",bullet))];
226                    list[0].bold=true;list[0].color=Some(theme::ACCENT_CYAN);
227                    list.extend_from_slice(spans);
228                    b.lines=lines_for(fonts,&list,16.0,(width-b.left_margin as f32).max(1.0),false);
229                }
230                DocNode::Blockquote{lines}=>{
231                    b.left_margin=16;b.border_color=Some(theme::ACCENT_BLUE);
232                    for text in lines {
233                        let mut span=StyledSpan::plain(text);span.italic=true;span.color=Some(theme::TEXT_MUTED);
234                        b.lines.extend(lines_for(fonts,&[span],16.0,(width-16.0).max(1.0),false));
235                    }
236                    gap=4;
237                }
238                DocNode::RawLines{lines}=>{
239                    b.left_margin=4;
240                    for text in lines {
241                        b.lines.extend(lines_for(fonts,&[StyledSpan::plain(text)],16.0,(width-4.0).max(1.0),true));
242                    }
243                }
244                DocNode::Rule=>{b.is_rule=true;b.height=16;}
245                DocNode::Image{width:iw,height:ih,pixels}=>{
246                    let display_w=(*iw).min(content_width.saturating_sub(32).max(1));
247                    let display_h=((*ih as u64*display_w as u64)/(*iw).max(1) as u64).max(1) as u32;
248                    b.height=display_h as i32+16;b.left_margin=16;
249                    b.image=Some((*iw,*ih,pixels.clone()));
250                }
251                DocNode::Table{headers,rows}=>{
252                    let count=headers.len().max(rows.iter().map(|r|r.len()).max().unwrap_or(0));
253                    if count==0{continue;}
254                    let mut desired=alloc::vec![12.0f32;count];
255                    for (i,text) in headers.iter().enumerate() {
256                        desired[i]=desired[i].max(fonts.measure(FontFace::UiSemibold,text,16.0)+12.0);
257                    }
258                    for row in rows {for (i,text) in row.iter().enumerate() {
259                        desired[i]=desired[i].max(fonts.measure(FontFace::UiRegular,text,16.0)+12.0);
260                    }}
261                    let sum: f32=desired.iter().sum();
262                    let available=content_width.saturating_sub(40).max(1);
263                    let mut widths=Vec::new();let mut used=0;
264                    for (i,w) in desired.iter().enumerate() {
265                        let w=if i+1==count {available.saturating_sub(used)}
266                            else {(*w/sum*available as f32) as u32};
267                        widths.push(w);used+=w;
268                    }
269                    let mut table_rows=Vec::new();let mut row_y=0;
270                    for (header,row) in core::iter::once((true,headers)).filter(|(_,r)|!r.is_empty())
271                        .chain(rows.iter().map(|r|(false,r))) {
272                        let mut cells=Vec::new();let mut x=0;let mut row_h=0;
273                        for (i,&w) in widths.iter().enumerate() {
274                            let mut span=StyledSpan::plain(row.get(i).map_or("",String::as_str));
275                            span.bold=header;span.color=Some(if header{theme::ACCENT_CYAN}else{theme::TEXT_PRIMARY});
276                            let lines=lines_for(fonts,&[span],16.0,w.saturating_sub(12).max(1) as f32,false);
277                            row_h=row_h.max(lines_height(&lines) as u32+12);
278                            cells.push(TableCellLayout{lines,x,width:w});x+=w as i32;
279                        }
280                        table_rows.push(TableRowLayout{y_rel:row_y,height:row_h,cells,is_header:header});
281                        row_y+=row_h as i32;
282                    }
283                    b.table_rows=Some(table_rows);b.table_width=used;
284                    b.left_margin=16;b.height=row_y+12;gap=8;
285                }
286                DocNode::Button{text,node_id}=>{
287                    let mut span=StyledSpan::plain(text.trim());span.bold=true;
288                    b.lines=lines_for(fonts,&[span],16.0,width,false);
289                    b.table_width=b.lines.iter().map(|l|l.width as u32+16).max().unwrap_or(16);
290                    b.left_margin=8;b.bg_color=Some(theme::ACCENT_BLUE);
291                    b.border_color=Some(theme::BORDER);b.node_id=*node_id;gap=8;
292                }
293            }
294            if b.height==0 {b.height=lines_height(&b.lines)+8;}
295            y+=b.height+gap;self.layout_boxes.push(b);
296        }
297        self.total_height=y+32;
298    }

300    pub fn render_viewport(&self,fonts:&mut TextFonts,surf:&mut ViSurface,
301        view_x:i32,view_y:i32,view_w:u32,view_h:u32,scroll_y:i32) {
302        if let Some(web)=&self.web_layout {web.render(fonts,surf,(view_x,view_y,view_w,view_h),scroll_y);return;}
303        let clip=(view_x,view_y,view_x+view_w as i32,view_y+view_h as i32);
304        for b in &self.layout_boxes {
305            let y=view_y+b.y_offset-scroll_y;
306            if y+b.height<=clip.1 || y>=clip.3 {continue;}
307            let text_x=view_x+16+b.left_margin;
308            if let Some(bg)=b.bg_color {
309                let w=if b.node_id.is_some(){b.table_width}else{view_w.saturating_sub(b.left_margin as u32+20)};
310                draw::clipped_rect(surf,text_x-8,y,w,b.height as u32,bg,clip);
311                if let Some(border)=b.border_color {
312                    draw::clipped_rect(surf,text_x-8,y,w,1,border,clip);
313                    draw::clipped_rect(surf,text_x-8,y+b.height-1,w,1,border,clip);
314                }
315            } else if let Some(border)=b.border_color {
316                draw::clipped_rect(surf,text_x-8,y,2,b.height as u32,border,clip);
317            }
318            if b.is_rule {
319                draw::clipped_rect(surf,view_x+10,y+4,view_w.saturating_sub(30),1,theme::BORDER,clip);continue;
320            }
321            if let Some((iw,ih,pixels))=&b.image {
322                draw::draw_image_clipped(surf,view_x+b.left_margin,y+8,
323                    (*iw).min(view_w.saturating_sub(32).max(1)),(b.height-16) as u32,
324                    pixels,(*iw*4) as usize,*ih,clip);continue;
325            }
326            if let Some(rows)=&b.table_rows {
327                let x=view_x+b.left_margin;
328                for row in rows {
329                    let ry=y+row.y_rel;
330                    if row.is_header {draw::clipped_rect(surf,x,ry,b.table_width,row.height,theme::BG_INPUT,clip);}
331                    draw::clipped_rect(surf,x,ry+row.height as i32-1,b.table_width,1,theme::BORDER,clip);
332                    for cell in &row.cells {
333                        draw::clipped_rect(surf,x+cell.x+cell.width as i32-1,ry,1,row.height,theme::BORDER,clip);
334                        let cell_clip=((x+cell.x).max(clip.0),ry.max(clip.1),
335                            (x+cell.x+cell.width as i32).min(clip.2),(ry+row.height as i32).min(clip.3));
336                        paint_lines(fonts,surf,&cell.lines,(x+cell.x+6) as f32,(ry+6) as f32,theme::TEXT_PRIMARY,cell_clip);
337                    }
338                }
339                continue;
340            }
341            if b.bg_color.is_none() {
342                let mut line_y=y+4;
343                for line in &b.lines {
344                    for run in &line.runs {
345                        if run.span.code {
346                            draw::clipped_rect(surf,text_x+run.x as i32-2,line_y,
347                                crate::parser::web::ceil_pixel(run.width)+4,line.height,
348                                theme::BG_INPUT,clip);
349                        }
350                    }
351                    line_y+=line.height as i32;
352                }
353            }
354            paint_lines(fonts,surf,&b.lines,text_x as f32,(y+4) as f32,theme::TEXT_PRIMARY,clip);
355        }
356    }

358    pub fn hit_test_link(&self,click_x:i32,click_y:i32,view_x:i32,view_y:i32,scroll_y:i32)->Option<String> {
359        if let Some(web)=&self.web_layout {return web.hit_link(click_x-view_x,click_y-view_y+scroll_y);}
360        for b in &self.layout_boxes {
361            let mut y=view_y+b.y_offset-scroll_y+4;
362            for line in &b.lines {
363                if click_y>=y && click_y<y+line.height as i32 {
364                    for run in &line.runs {
365                        let x=(view_x+16+b.left_margin) as f32+run.x;
366                        if (click_x as f32)>=x && (click_x as f32)<x+run.width {
367                            if let Some(link)=&run.span.link{return Some(link.clone());}
368                        }
369                    }
370                }
371                y+=line.height as i32;
372            }
373        }
374        None
375    }
376    pub fn hit_test_node(&self,click_x:i32,click_y:i32,view_x:i32,view_y:i32,scroll_y:i32)->Option<dom_arena::NodeId> {
377        if let Some(web)=&self.web_layout {return web.hit_node(click_x-view_x,click_y-view_y+scroll_y);}
378        for b in &self.layout_boxes {
379            let y=view_y+b.y_offset-scroll_y;
380            let x=view_x+8+b.left_margin;
381            if click_y>=y && click_y<y+b.height && click_x>=x && click_x<x+b.table_width as i32 {
382                if let Some(id)=b.node_id{return Some(id);}
383            }
384        }
385        None
386    }
