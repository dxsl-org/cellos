2//! Allocation limits shared by acquisition, parsers, and tab residency.
3use alloc::string::String;

5pub const MAX_TEXT_BYTES: usize = 1024 * 1024;
6pub const MAX_RESOURCE_BYTES: usize = 4 * 1024 * 1024;
7pub const MAX_DOM_NODES: usize = 16_384;
8pub const MAX_DOM_DEPTH: usize = 64;
9pub const MAX_CSS_RULES: usize = 1024;
10pub const MAX_CSS_SELECTORS: usize = 4096;
11pub const MAX_SELECTOR_BYTES: usize = 512;
12pub const MAX_IMAGES: usize = 16;
13pub const MAX_IMAGE_PIXELS: usize = 1024 * 1024;
14pub const MAX_IMAGE_TOTAL_BYTES: usize = 4 * 1024 * 1024;
15pub const MAX_TABS: usize = 4;
16pub const MAX_RESIDENT_BYTES: usize = 8 * 1024 * 1024;
17pub const MAX_TABLE_ROWS: usize = 1024;
18pub const MAX_TABLE_CELLS: usize = 4096;

20pub fn validate_text(text: &str) -> Result<(), String> {
21    if text.len() > MAX_TEXT_BYTES { Err(String::from("Document exceeds text byte budget")) } else { Ok(()) }
22}

24pub fn validate_css(css: &str) -> Result<(), String> {
25    validate_text(css)?;
26    let mut rules=0usize;let mut selectors=0usize;let mut depth=0usize;
27    let mut start=0usize;let mut quote=None;let mut escaped=false;let mut comment=false;
28    let bytes=css.as_bytes();let mut i=0;
29    while i<bytes.len() {
30        let c=bytes[i];
31        if comment {if c==b'*'&&bytes.get(i+1)==Some(&b'/'){comment=false;i+=2;}else{i+=1;}continue;}
32        if escaped{escaped=false;i+=1;continue;}
33        if c==b'\\'{escaped=true;i+=1;continue;}
34        if let Some(q)=quote{if c==q{quote=None;}i+=1;continue;}
35        if c==b'/'&&bytes.get(i+1)==Some(&b'*'){comment=true;i+=2;continue;}
36        if c==b'\''||c==b'"'{quote=Some(c);i+=1;continue;}
37        if c==b'{' {
38            rules+=1;depth+=1;
39            for selector in css[start..i].split(',') {
40                selectors+=1;
41                if selector.trim().len()>MAX_SELECTOR_BYTES{return Err(String::from("CSS selector exceeds byte budget"));}
42            }
43            if rules>MAX_CSS_RULES||selectors>MAX_CSS_SELECTORS{return Err(String::from("CSS exceeds rule/selector budget"));}
44            if depth>MAX_DOM_DEPTH{return Err(String::from("CSS exceeds nesting depth budget"));}
45            start=i+1;
46        }else if c==b'}'{depth=depth.saturating_sub(1);start=i+1;}else if c==b';'&&depth==0{start=i+1;}
47        i+=1;
48    }
49    Ok(())
50}

52/// Match the tokenizer's stack without allocating tokens or attacker-sized stacks.
53pub fn validate_html(text: &str) -> Result<(), String> {
54    validate_text(text)?;
55    let mut stack=["";MAX_DOM_DEPTH];let mut depth=0usize;let mut nodes=1usize;
56    let mut rest=text;let mut images=0usize;
57    while let Some(open)=rest.find('<') {
58        if open>0{nodes+=1;}
59        rest=&rest[open+1..];
60        if let Some(comment)=rest.strip_prefix("!--") {
61            rest=comment.find("-->").map_or("",|end|&comment[end+3..]);continue;
62        }
63        let mut quote=None;let mut close=None;
64        for (index,c) in rest.char_indices() {
65            if quote==Some(c){quote=None;}else if quote.is_none()&&(c=='\''||c=='"'){quote=Some(c);}
66            else if c=='>'&&quote.is_none(){close=Some(index);break;}
67        }
68        let end=close.unwrap_or(rest.len());let tag=rest[..end].trim();
69        rest=if end<rest.len(){&rest[end+1..]}else{""};
70        if let Some(name)=tag.strip_prefix('/') {
71            if let Some(position)=stack[..depth].iter().rposition(|open|open.eq_ignore_ascii_case(name.trim())){depth=position;}
72        }else{
73            let name=tag.split_whitespace().next().unwrap_or("").trim_end_matches('/');
74            let self_closing=tag.ends_with('/');
75            if name.eq_ignore_ascii_case("script")&&!self_closing {
76                let end=rawtext_end(rest,name);
77                rest=end.map_or("",|(_,end)|&rest[end..]);continue;
78            }
79            nodes+=1;
80            if name.eq_ignore_ascii_case("img"){images+=1;if images>MAX_IMAGES{return Err(String::from("Document exceeds image count budget"));}}
81            let void=["hr","br","img","input","meta","link"].iter().any(|void|name.eq_ignore_ascii_case(void));
82            if !self_closing&&!void {
83                if depth==MAX_DOM_DEPTH{return Err(String::from("HTML exceeds DOM depth budget"));}
84                stack[depth]=name;depth+=1;
85            }
86            if !self_closing&&["style","textarea","title","xmp","iframe","noembed","noframes"].iter().any(|raw|name.eq_ignore_ascii_case(raw)) {
87                let end=rawtext_end(rest,name);
88                let body=end.map_or(rest,|(start,_)|&rest[..start]);
89                if name.eq_ignore_ascii_case("style"){validate_css(body)?;}nodes+=1;
90                rest=end.map_or("",|(_,end)|&rest[end..]);depth=depth.saturating_sub(1);
91            }
92        }
93        if nodes>MAX_DOM_NODES{return Err(String::from("HTML exceeds DOM node budget"));}
94    }
95    if nodes+usize::from(!rest.is_empty())>MAX_DOM_NODES{return Err(String::from("HTML exceeds DOM node budget"));}
96    Ok(())
97}

99/// Header-only inspection: reserve the aggregate decoded bytes before decoding.
100pub fn image_bytes(data: &[u8]) -> Result<usize, String> {
101    use zune_core::{bytestream::ZCursor, options::DecoderOptions};
102    let dimensions = if data.starts_with(b"BM") && data.len() >= 26 {
103        let w = i32::from_le_bytes(data[18..22].try_into().unwrap());
104        let h = i32::from_le_bytes(data[22..26].try_into().unwrap());
105        if w <= 0 || h == i32::MIN { None } else { Some((w as usize, h.unsigned_abs() as usize)) }
106    } else if data.starts_with(b"\x89PNG\r\n\x1a\n") {
107        let mut decoder = zune_png::PngDecoder::new_with_options(ZCursor::new(data), DecoderOptions::new_safe());
108        decoder.decode_headers().ok().and_then(|_| decoder.dimensions())
109    } else if data.starts_with(b"\xff\xd8") {
110        let mut decoder = zune_jpeg::JpegDecoder::new_with_options(ZCursor::new(data), DecoderOptions::new_safe());
111        decoder.decode_headers().ok().and_then(|_| decoder.dimensions())
112    } else { None };
113    let (w,h) = dimensions.ok_or_else(|| String::from("Unsupported or invalid image"))?;
114    let pixels = w.checked_mul(h).filter(|p| *p > 0 && *p <= MAX_IMAGE_PIXELS).ok_or_else(|| String::from("Image exceeds pixel budget"))?;
115    Ok(pixels * 4)
116}

118/// Number of characters after an already-consumed '<' in a raw-text end tag.
119/// HTML whitespace is permitted after the case-insensitive tag name.
120pub fn rawtext_close_chars(mut chars:impl Iterator<Item=char>,name:&str)->Option<usize> {
121    if chars.next()!=Some('/'){return None;}
122    let mut count=1;
123    for expected in name.chars(){let actual=chars.next()?;count+=1;if !actual.eq_ignore_ascii_case(&expected){return None;}}
124    loop {let c=chars.next()?;count+=1;
125        if c=='>'{return Some(count);}
126        if !matches!(c,' '|'\t'|'\n'|'\r'|'\u{c}'){return None;}
127    }
128}
129
130pub fn rawtext_end(text:&str,name:&str)->Option<(usize,usize)> {
131    for (index,c) in text.char_indices(){if c=='<' {
132        if let Some(count)=rawtext_close_chars(text[index+1..].chars(),name){return Some((index,index+1+count));}
133    }}
134    None
135}

137/// Count nested table allocation units before allocating Markdown nodes/cells.
138pub fn validate_markdown(text:&str)->Result<(),String> {
139    validate_text(text)?;
140    let mut rows=0usize;let mut cells=0usize;let mut fenced=false;
141    for line in text.lines() {
142        let line=line.trim();
143        if line.starts_with("```"){fenced=!fenced;continue;}
144        if fenced{continue;}
145        if line.starts_with('|')&&line.ends_with('|')&&line.len()>1 {
146            rows+=1;cells=cells.saturating_add(line.trim_start_matches('|').trim_end_matches('|').split('|').count());
147            if rows>MAX_TABLE_ROWS||cells>MAX_TABLE_CELLS{return Err(String::from("Markdown exceeds table row/cell budget"));}
148        }
149    }
150    Ok(())
151}
