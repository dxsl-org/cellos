49pub fn validate_text(text: &str) -> Result<(), String> {
50    if text.len() > MAX_TEXT_BYTES { Err(String::from("Document exceeds text byte budget")) } else { Ok(()) }
51}

53pub fn validate_css(css: &str) -> Result<(), String> {
54    if css.len() > MAX_CSS_BYTES { return Err(String::from("CSS exceeds byte budget")); }
55    let mut rules=0usize;let mut selectors=0usize;let mut depth=0usize;
56    let mut start=0usize;let mut quote=None;let mut escaped=false;let mut comment=false;
57    let bytes=css.as_bytes();let mut i=0;
58    while i<bytes.len() {
59        let c=bytes[i];
60        if comment {if c==b'*'&&bytes.get(i+1)==Some(&b'/'){comment=false;i+=2;}else{i+=1;}continue;}
61        if escaped{escaped=false;i+=1;continue;}
62        if c==b'\\'{escaped=true;i+=1;continue;}
63        if let Some(q)=quote{if c==q{quote=None;}i+=1;continue;}
64        if c==b'/'&&bytes.get(i+1)==Some(&b'*'){comment=true;i+=2;continue;}
65        if c==b'\''||c==b'"'{quote=Some(c);i+=1;continue;}
66        if c==b'{' {
67            rules+=1;depth+=1;
68            for selector in css[start..i].split(',') {
69                selectors+=1;
70                if selector.trim().len()>MAX_SELECTOR_BYTES{return Err(String::from("CSS selector exceeds byte budget"));}
71            }
72            if rules>MAX_CSS_RULES||selectors>MAX_CSS_SELECTORS{return Err(String::from("CSS exceeds rule/selector budget"));}
73            if depth>MAX_DOM_DEPTH{return Err(String::from("CSS exceeds nesting depth budget"));}
74            start=i+1;
75        }else if c==b'}'{depth=depth.saturating_sub(1);start=i+1;}else if c==b';'&&depth==0{start=i+1;}
76        i+=1;
77    }
78    Ok(())
79}

81/// Match the tokenizer's stack without allocating tokens or attacker-sized stacks.
82pub fn validate_html(text: &str) -> Result<(), String> {
83    validate_text(text)?;
84    let mut stack=["";MAX_DOM_DEPTH];let mut depth=0usize;let mut nodes=1usize;
85    let mut rest=text;let mut images=0usize;
86    while let Some(open)=rest.find('<') {
87        if open>0{nodes+=1;}
88        rest=&rest[open+1..];
89        if let Some(comment)=rest.strip_prefix("!--") {
90            rest=comment.find("-->").map_or("",|end|&comment[end+3..]);continue;
91        }
92        let mut quote=None;let mut close=None;
93        for (index,c) in rest.char_indices() {
94            if quote==Some(c){quote=None;}else if quote.is_none()&&(c=='\''||c=='"'){quote=Some(c);}
95            else if c=='>'&&quote.is_none(){close=Some(index);break;}
96        }
97        let end=close.unwrap_or(rest.len());let tag=rest[..end].trim();
98        rest=if end<rest.len(){&rest[end+1..]}else{""};
99        if let Some(name)=tag.strip_prefix('/') {
100            if let Some(position)=stack[..depth].iter().rposition(|open|open.eq_ignore_ascii_case(name.trim())){depth=position;}
101        }else{
102            let name=tag.split_whitespace().next().unwrap_or("").trim_end_matches('/');
103            let self_closing=tag.ends_with('/');
104            if name.eq_ignore_ascii_case("script")&&!self_closing {
105                let end=rawtext_end(rest,name);
106                rest=end.map_or("",|(_,end)|&rest[end..]);continue;
107            }
108            nodes+=1;
109            if name.eq_ignore_ascii_case("img"){images+=1;if images>MAX_IMAGES{return Err(String::from("Document exceeds image count budget"));}}
110            let void=["hr","br","img","input","meta","link"].iter().any(|void|name.eq_ignore_ascii_case(void));
111            if !self_closing&&!void {
112                if depth==MAX_DOM_DEPTH{return Err(String::from("HTML exceeds DOM depth budget"));}
113                stack[depth]=name;depth+=1;
114            }
115            if !self_closing&&["style","textarea","title","xmp","iframe","noembed","noframes"].iter().any(|raw|name.eq_ignore_ascii_case(raw)) {
116                let end=rawtext_end(rest,name);
117                let body=end.map_or(rest,|(start,_)|&rest[..start]);
118                if name.eq_ignore_ascii_case("style"){validate_css(body)?;}nodes+=1;
119                rest=end.map_or("",|(_,end)|&rest[end..]);depth=depth.saturating_sub(1);
120            }
121        }
122        if nodes>MAX_DOM_NODES{return Err(String::from("HTML exceeds DOM node budget"));}
123    }
124    if nodes+usize::from(!rest.is_empty())>MAX_DOM_NODES{return Err(String::from("HTML exceeds DOM node budget"));}
125    Ok(())
126}

128/// Header-only inspection: reserve the aggregate decoded bytes before decoding.
129pub fn image_bytes(data: &[u8]) -> Result<usize, String> {
130    use zune_core::{bytestream::ZCursor, options::DecoderOptions};
131    let dimensions = if data.starts_with(b"BM") && data.len() >= 26 {
132        let w = i32::from_le_bytes(data[18..22].try_into().unwrap());
133        let h = i32::from_le_bytes(data[22..26].try_into().unwrap());
134        if w <= 0 || h == i32::MIN { None } else { Some((w as usize, h.unsigned_abs() as usize)) }
135    } else if data.starts_with(b"\x89PNG\r\n\x1a\n") {
136        let mut decoder = zune_png::PngDecoder::new_with_options(ZCursor::new(data), DecoderOptions::new_safe());
137        decoder.decode_headers().ok().and_then(|_| decoder.dimensions())
138    } else if data.starts_with(b"\xff\xd8") {
139        let mut decoder = zune_jpeg::JpegDecoder::new_with_options(ZCursor::new(data), DecoderOptions::new_safe());
140        decoder.decode_headers().ok().and_then(|_| decoder.dimensions())
141    } else {
142        #[cfg(feature = "desktop")]
143        {
144            if data.starts_with(b"<svg")
145                || data.starts_with(b"<?xml")
146                || (data.len() > 10 && data[..data.len().min(256)].windows(4).any(|w| w == b"<svg"))
147            {
148                crate::image::svg::svg_dimensions(data)
149            } else {
150                None
151            }
152        }
153        #[cfg(not(feature = "desktop"))]
154        None
155    };
156    let (w,h) = dimensions.ok_or_else(|| String::from("Unsupported or invalid image"))?;
157    let pixels = w.checked_mul(h).filter(|p| *p > 0 && *p <= MAX_IMAGE_PIXELS).ok_or_else(|| String::from("Image exceeds pixel budget"))?;
158    Ok(pixels * 4)
159}

161/// Number of characters after an already-consumed '<' in a raw-text end tag.
162/// HTML whitespace is permitted after the case-insensitive tag name.
163pub fn rawtext_close_chars(mut chars:impl Iterator<Item=char>,name:&str)->Option<usize> {
164    if chars.next()!=Some('/'){return None;}
165    let mut count=1;
166    for expected in name.chars(){let actual=chars.next()?;count+=1;if !actual.eq_ignore_ascii_case(&expected){return None;}}
167    loop {let c=chars.next()?;count+=1;
168        if c=='>'{return Some(count);}
169        if !matches!(c,' '|'\t'|'\n'|'\r'|'\u{c}'){return None;}
170    }
171}
172
173pub fn rawtext_end(text:&str,name:&str)->Option<(usize,usize)> {
174    for (index,c) in text.char_indices(){if c=='<' {
175        if let Some(count)=rawtext_close_chars(text[index+1..].chars(),name){return Some((index,index+1+count));}
176    }}
177    None
178}

180/// Count nested table allocation units before allocating Markdown nodes/cells.
181pub fn validate_markdown(text:&str)->Result<(),String> {
182    validate_text(text)?;
183    let mut rows=0usize;let mut cells=0usize;let mut fenced=false;
184    for line in text.lines() {
185        let line=line.trim();
186        if line.starts_with("```"){fenced=!fenced;continue;}
187        if fenced{continue;}
188        if line.starts_with('|')&&line.ends_with('|')&&line.len()>1 {
189            rows+=1;cells=cells.saturating_add(line.trim_start_matches('|').trim_end_matches('|').split('|').count());
190            if rows>MAX_TABLE_ROWS||cells>MAX_TABLE_CELLS{return Err(String::from("Markdown exceeds table row/cell budget"));}
191        }
192    }
193    Ok(())
194}

200    #[test]
201    fn ordinary_site_admission_depends_on_build_profile() {
202        let css = "p{color:black}\n".repeat(1_211);
203        assert_eq!(validate_css(&css).is_ok(), cfg!(feature = "desktop"));
204        let news = alloc::format!("<html><body>{}</body></html>", "<img src='/thumb.png'>".repeat(100));
205        assert_eq!(validate_html(&news).is_ok(), cfg!(feature = "desktop"));
206    }
207
208    #[test]
209    fn css_and_image_count_boundaries_still_reject_excess() {
210        assert!(validate_css(&"p{x:0}".repeat(MAX_CSS_RULES)).is_ok());
211        assert!(validate_css(&"p{x:0}".repeat(MAX_CSS_RULES + 1)).is_err());
212        assert!(validate_html(&"<img src='x'>".repeat(MAX_IMAGES)).is_ok());
213        assert!(validate_html(&"<img src='x'>".repeat(MAX_IMAGES + 1)).is_err());
214        let selectors = alloc::format!("{}{{x:0}}", alloc::vec!["p"; MAX_CSS_SELECTORS].join(","));
215        assert!(validate_css(&selectors).is_ok());
216        let selectors = alloc::format!("{}{{x:0}}", alloc::vec!["p"; MAX_CSS_SELECTORS + 1].join(","));
217        assert!(validate_css(&selectors).is_err());
218    }
219
220    #[test]
221    fn text_css_and_image_bytes_accept_boundary_and_reject_excess() {
222        assert!(validate_text(&" ".repeat(MAX_TEXT_BYTES)).is_ok());
223        assert!(validate_text(&" ".repeat(MAX_TEXT_BYTES + 1)).is_err());
224        assert!(validate_css(&" ".repeat(MAX_CSS_BYTES)).is_ok());
225        assert!(validate_css(&" ".repeat(MAX_CSS_BYTES + 1)).is_err());
226        let mut header = alloc::vec![0; 26];
227        header[..2].copy_from_slice(b"BM");
228        header[18..22].copy_from_slice(&(MAX_IMAGE_PIXELS as i32).to_le_bytes());
229        header[22..26].copy_from_slice(&1i32.to_le_bytes());
230        assert_eq!(image_bytes(&header).unwrap(), MAX_IMAGE_PIXELS * 4);
231        header[18..22].copy_from_slice(&((MAX_IMAGE_PIXELS + 1) as i32).to_le_bytes());
232        assert!(image_bytes(&header).is_err());
233    }
