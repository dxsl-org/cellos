16static NEXT_CONTEXT: AtomicUsize = AtomicUsize::new(1);

18pub struct Tier2JsBridge<T: Transport = KernelTransport> {
19    lease: Option<Lease<T>>,
20    context: u64,
21    opened: bool,
22    document_ready: bool,
23    mutations: Vec<DomMutation>,
24    backend_named: bool,
25    last_activity: Option<u64>,
26    browser_requests: Vec<BrowserRequest>,
27    browser_wakeup: Option<u64>,
28}

30impl<T: Transport + Default> Default for Tier2JsBridge<T> { fn default() -> Self { Self::new() } }

32impl<T: Transport + Default> Tier2JsBridge<T> {
33    pub fn new() -> Self {
34        Self { lease: None, context: NEXT_CONTEXT.fetch_add(1, Ordering::Relaxed) as u64,
35            opened: false, document_ready: false, mutations: Vec::new(), backend_named: false, last_activity: None,
36            browser_requests: Vec::new(), browser_wakeup: None }
37    }
38
39    /// Bind this token to the current document/tab. A reset invalidates old tokens.
40    /// Never sync a different document into an existing token: reset it first.
41    pub fn context_id(&self) -> u64 { self.context }
42    /// False after navigation, scripts-off, or any terminal service error.
43    pub fn is_active(&self) -> bool {
44        self.opened && self.document_ready && !self.last_activity.is_some_and(|last| {
45            self.lease.as_ref().and_then(Lease::now).is_some_and(|now| now.wrapping_sub(last) >= dom_arena::OCEL_JS_IDLE_TICKS)
46        })
47    }
48
49    /// Call before scripts, and after native edits only once JS mutations have
50    /// been drained AND applied. Same-document wrapper identity/listeners survive.
51    pub fn sync_document(&mut self, arena: &DocumentArena, title: &str) -> Result<(), JsError> {
52        let result = self.sync_active_document(arena, title);
53        if result.is_err() { self.invalidate(); }
54        result
55    }
56
57    /// The native load path calls this gate for every format. An HTML arena by
58    /// itself is not JavaScript demand: only supported executable scripts are.
59    pub fn sync_script_document(&mut self, arena: Option<&DocumentArena>, has_scripts: bool, title: &str) -> Result<bool, JsError> {
60        self.reset();
61        if !has_scripts { return Ok(false); }
62        let Some(arena) = arena else { return Ok(false); };
63        self.sync_document(arena, title).map(|_| true)
64    }
65
66    fn sync_active_document(&mut self, arena: &DocumentArena, title: &str) -> Result<(), JsError> {
67        if !arena.validate() || !self.mutations.is_empty() {
68            return Err(Self::error("Invalid DOM snapshot or undrained JavaScript mutations"));
69        }
70        if self.opened && !self.is_active() { self.close_context(); self.document_ready = false; }
71        if self.lease.is_none() {
72            self.lease = Some(Lease::acquire_with(Engine::JavaScript, T::default())
73                .map_err(|message| JsError { message, line: 0 })?);
74        }
75        if !self.opened {
76            self.expect_ack(&OcelJsRequest::OpenContext { context: self.context })?;
77            self.opened = true;
78        }
79        self.execute(OcelJsCommand::SyncDocument { arena: arena.clone(), title: String::from(title) })?;
80        self.document_ready = true;
81        Ok(())
82    }

85        self.require_active()?;

134    fn require_active(&mut self) -> Result<(), JsError> {
135        if !self.is_active() {
136            self.invalidate();
137            return Err(Self::error("No active synchronized JavaScript document context"));
138        }
139        Ok(())
140    }
141
142    fn invalidate(&mut self) {
143        self.close_context();
144        self.opened = false;
145        self.document_ready = false;
146        self.last_activity = None;
147        self.lease = None;
148        self.browser_requests.clear();
149        self.browser_wakeup = None;
150    }
151
152    fn error(message: &str) -> JsError { JsError { message: String::from(message), line: 0 } }
153
154    fn active_lease(&mut self) -> Result<&mut Lease<T>, JsError> {
155        self.lease.as_mut().ok_or_else(|| Self::error("No active JavaScript engine lease; synchronize a script-bearing document first"))
156    }

158    fn exchange(&mut self, request: &OcelJsRequest) -> Result<OcelJsReply, JsError> {
159        if self.lease.is_none() { return Err(Self::error("No active JavaScript engine lease")); }
160        let mut send_buf = [0u8; OCEL_JS_IPC_BUF_SIZE];
161        let mut recv_buf = [0u8; OCEL_JS_IPC_BUF_SIZE];
162        let encoded = postcard::to_slice(request, &mut send_buf)
163            .map_err(|_| Self::error("JavaScript IPC frame exceeds 4 KiB"))?;
164        let activity = self.lease.as_ref().and_then(Lease::now);
165        let reply = self.active_lease()?.exchange(encoded, &mut recv_buf)
166            .map_err(|message| JsError { message, line: 0 })
167            .and_then(|len| postcard::from_bytes::<OcelJsReply>(&recv_buf[..len])
168                .map_err(|_| Self::error("Malformed JavaScript service response")));
169        match reply {
170            Ok(OcelJsReply::Busy) => {
171                self.invalidate();
172                Err(Self::error("JavaScript engine busy: another live document owns the context"))
173            }
174            Ok(OcelJsReply::Error { message }) => {
175                self.invalidate();
176                Err(JsError { message, line: 0 })
177            }
178            Err(error) => {
179                self.invalidate();
180                Err(error)
181            }
182            Ok(reply) => {
183                self.last_activity = activity;
184                Ok(reply)
185            }
186        }
187    }
188
189    fn expect_ack(&mut self, request: &OcelJsRequest) -> Result<(), JsError> {
190        match self.exchange(request)? {
191            OcelJsReply::Ack => Ok(()),
192            _ => {
193                self.invalidate();
194                Err(Self::error("Unexpected JavaScript IPC acknowledgement"))
195            }
196        }
197    }
198
199    fn execute(&mut self, command: OcelJsCommand) -> Result<String, JsError> {
200        self.execute_with_layout(command, None)
201    }

203    fn execute_with_layout(&mut self, command: OcelJsCommand,
204        mut host: Option<&mut dyn FnMut(&[DomMutation], NodeId) -> Result<BrowserFrame, JsError>>) -> Result<String, JsError> {
205        if !self.mutations.is_empty() {
206            return Err(Self::error("Apply pending JavaScript mutations before executing another command"));
207        }
208        let payload = dom_arena::ipc::encode_bounded(&command, OCEL_JS_MAX_TRANSFER)
209            .map_err(|_| Self::error("JavaScript request exceeds 512 KiB transfer limit"))?;
210        self.expect_ack(&OcelJsRequest::Begin { context: self.context, total: payload.len() })?;
211        for (index, chunk) in payload.chunks(OCEL_JS_CHUNK_SIZE).enumerate() {
212            self.expect_ack(&OcelJsRequest::Chunk { context: self.context,
213                offset: index * OCEL_JS_CHUNK_SIZE, data: chunk.to_vec() })?;
214        }
215        let mut reply = self.exchange(&OcelJsRequest::Commit { context: self.context })?;
216        let mut queries = 0;
217        let mut query_mutations = 0usize;
218        let mut query_bytes = 0usize;
219        let mut bytes = Vec::new();
220        let mut expected_total = None;
221        loop {
222            if matches!(reply, OcelJsReply::LayoutQueryChunk { .. }) {
223                if !bytes.is_empty() { self.invalidate(); return Err(Self::error("Layout query interrupted a final response")); }
224                queries += 1;
225                if queries > 64 { self.invalidate(); return Err(Self::error("Native layout query limit exceeded")); }
226                let (query_id, query) = self.receive_layout_query(reply)?;
227                if query_id != queries { self.invalidate(); return Err(Self::error("Native layout query sequence mismatch")); }
228                query_mutations = query_mutations.saturating_add(query.mutations.len());
229                query_bytes = query_bytes.saturating_add(dom_arena::ipc::serialized_len_bounded(&query.mutations, dom_arena::OCEL_JS_MAX_MUTATION_BYTES)
230                    .unwrap_or(usize::MAX));
231                if query_mutations > dom_arena::OCEL_JS_MAX_MUTATIONS || query_bytes > dom_arena::OCEL_JS_MAX_MUTATION_BYTES {
232                    self.invalidate(); return Err(Self::error("Native layout mutation budget exceeded"));
233                }
234                let result = match host.as_mut() {
235                    Some(host) => host(&query.mutations, query.node),
236                    None => {
237                        self.mutations.extend(query.mutations.iter().cloned());
238                        Err(Self::error("Synchronous native layout host is unavailable"))
239                    }
240                };
241                let resume = match result {
242                    Ok(frame) => dom_arena::BrowserLayoutReply::Frame(frame),
243                    Err(error) => dom_arena::BrowserLayoutReply::Error(error.message),
244                };
245                let payload = dom_arena::ipc::encode_bounded(&resume, OCEL_JS_MAX_TRANSFER)
246                    .map_err(|_| { self.invalidate(); Self::error("Native layout response exceeds transfer budget") })?;
247                self.expect_ack(&OcelJsRequest::LayoutResumeBegin { context: self.context, query: query_id, total: payload.len() })?;
248                for (index, chunk) in payload.chunks(OCEL_JS_CHUNK_SIZE).enumerate() {
249                    self.expect_ack(&OcelJsRequest::LayoutResumeChunk { context: self.context, query: query_id,
250                        offset: index * OCEL_JS_CHUNK_SIZE, data: chunk.to_vec() })?;
251                }
252                reply = self.exchange(&OcelJsRequest::LayoutResumeCommit { context: self.context, query: query_id })?;
253                continue;
254            }
255            let OcelJsReply::ResponseChunk { offset, total, data } = reply else {
256                self.invalidate();
257                return Err(Self::error("Unexpected JavaScript IPC response"));
258            };
259            if total == 0 || total > OCEL_JS_MAX_TRANSFER || offset != bytes.len()
260                || data.is_empty() || data.len() > OCEL_JS_CHUNK_SIZE
261                || offset.saturating_add(data.len()) > total
262                || expected_total.is_some_and(|expected| expected != total) {
263                self.invalidate();
264                return Err(Self::error("Invalid JavaScript response chunk"));
265            }
266            expected_total = Some(total);
267            bytes.extend(data);
268            if bytes.len() == total { break; }
269            reply = self.exchange(&OcelJsRequest::ReadResponse { context: self.context, offset: bytes.len() })?;
270        }
271        let response = postcard::from_bytes::<OcelJsResponse>(&bytes)
272            .map_err(|_| { self.invalidate(); Self::error("Malformed JavaScript mutation response") })?;
273        let (mutations, text, requests, wake_at) = match &response {
274            OcelJsResponse::Success { mutations, result_repr, requests, wake_at } => (mutations, result_repr, requests, wake_at),
275            OcelJsResponse::Error { mutations, message, requests, wake_at, .. } => (mutations, message, requests, wake_at),
276        };
277        let request_bytes = dom_arena::ipc::serialized_len_bounded(requests, dom_arena::OCEL_JS_MAX_BROWSER_REQUEST_BYTES)
278            .and_then(|incoming| dom_arena::ipc::serialized_len_bounded(&self.browser_requests, dom_arena::OCEL_JS_MAX_BROWSER_REQUEST_BYTES)
279                .and_then(|queued| if incoming.saturating_add(queued) <= dom_arena::OCEL_JS_MAX_BROWSER_REQUEST_BYTES {
280                    Ok(incoming.saturating_add(queued))
281                } else { Err(postcard::Error::SerializeBufferFull) }));
282        if mutations.len().saturating_add(query_mutations) > dom_arena::OCEL_JS_MAX_MUTATIONS || text.len() > dom_arena::OCEL_JS_MAX_RESULT_BYTES
283            || dom_arena::ipc::serialized_len_bounded(mutations, dom_arena::OCEL_JS_MAX_MUTATION_BYTES)
284                .map_or(true, |bytes| bytes.saturating_add(query_bytes) > dom_arena::OCEL_JS_MAX_MUTATION_BYTES)
285            || requests.len().saturating_add(self.browser_requests.len()) > dom_arena::OCEL_JS_MAX_BROWSER_REQUESTS
286            || request_bytes.is_err() {
287            self.invalidate();
288            return Err(Self::error("JavaScript service exceeded output budgets; context closed"));
289        }
290        self.browser_wakeup = *wake_at;
291        if !self.backend_named {
292            ostd::io::println("[ocel] js backend: Tier 2 domain service (document-backed DOM required)");
293            self.backend_named = true;
294        }
295        match response {
296            OcelJsResponse::Success { mutations, result_repr, requests, .. } => {
297                self.mutations.extend(mutations); self.browser_requests.extend(requests); Ok(result_repr)
298            }
299            OcelJsResponse::Error { message, line, mutations, requests, .. } => {
300                self.mutations.extend(mutations); self.browser_requests.extend(requests); Err(JsError { message, line })
301            }
302        }
303    }

305    fn receive_layout_query(&mut self, mut reply: OcelJsReply) -> Result<(u32, dom_arena::BrowserLayoutQuery), JsError> {
306        let mut bytes = Vec::new();
307        let mut expected = None;
308        let mut expected_query = None;
309        loop {
310            let OcelJsReply::LayoutQueryChunk { context, query, offset, total, data } = reply else {
311                self.invalidate(); return Err(Self::error("Malformed native layout query chunk"));
312            };
313            if context != self.context || query == 0 || total == 0 || total > OCEL_JS_MAX_TRANSFER || offset != bytes.len()
314                || data.is_empty() || data.len() > OCEL_JS_CHUNK_SIZE || offset.saturating_add(data.len()) > total
315                || expected.is_some_and(|value| value != total) || expected_query.is_some_and(|value| value != query) {
316                self.invalidate(); return Err(Self::error("Invalid native layout query sequence"));
317            }
318            expected = Some(total); expected_query = Some(query); bytes.extend(data);
319            if bytes.len() == total {
320                let query = postcard::from_bytes(&bytes).map_err(|_| { self.invalidate(); Self::error("Malformed native layout query") })?;
321                return Ok((expected_query.unwrap(), query));
322            }
323            reply = self.exchange(&OcelJsRequest::ReadLayoutQuery { context: self.context, query, offset: bytes.len() })?;
324        }
325    }

328impl<T: Transport + Default> JsEngine for Tier2JsBridge<T> {
329    type Context = Tier2JsBridge<T>;
330    fn create_context(&mut self) -> Self::Context { Self::new() }
331}

333impl<T: Transport + Default> JsContext for Tier2JsBridge<T> {
334    fn sync_document(&mut self, arena: &DocumentArena, title: &str) -> Result<(), JsError> {
335        Tier2JsBridge::sync_document(self, arena, title)
336    }
337    fn eval_script(&mut self, script: &str, node: Option<NodeId>) -> Result<String, JsError> { Tier2JsBridge::eval_script(self, script, node) }
338    fn configure_browser(&mut self, environment: &BrowserEnvironment) -> Result<(), JsError> { Tier2JsBridge::configure_browser(self, environment) }
339    fn pump_browser(&mut self, frame: &BrowserFrame) -> Result<(), JsError> { Tier2JsBridge::pump_browser(self, frame) }
340    fn complete_browser(&mut self, reply: &BrowserReply) -> Result<(), JsError> { Tier2JsBridge::complete_browser(self, reply) }
341    fn take_browser_requests(&mut self) -> Vec<BrowserRequest> { Tier2JsBridge::take_browser_requests(self) }
342    fn next_browser_wakeup(&self) -> Option<u64> { self.browser_wakeup }
343    fn eval(&mut self, script: &str) -> Result<String, JsError> {
344        if !self.document_ready { return Err(Self::error("Synchronize the active document before executing JavaScript")); }
345        self.execute(OcelJsCommand::Eval { script: String::from(script) })
346    }
347    fn dispatch_event(&mut self, event: &DomEvent) -> Result<(), JsError> {
348        if !self.document_ready { return Err(Self::error("No synchronized JavaScript document context")); }
349        self.execute(OcelJsCommand::DispatchEvent { event: event.clone() }).map(|_| ())
350    }
351    fn take_mutations(&mut self) -> Vec<DomMutation> { core::mem::take(&mut self.mutations) }
352    fn reset(&mut self) {
353        self.invalidate();
354        self.context = NEXT_CONTEXT.fetch_add(1, Ordering::Relaxed) as u64;
355        self.mutations.clear();
356    }
357}

359impl<T: Transport> Tier2JsBridge<T> {
360    fn close_context(&mut self) {
361        if self.opened {
362            self.opened = false;
363            if let Some(lease) = self.lease.as_mut() {
364                let mut send = [0u8; OCEL_JS_IPC_BUF_SIZE];
365                let mut receive = [0u8; OCEL_JS_IPC_BUF_SIZE];
366                if let Ok(bytes) = postcard::to_slice(&OcelJsRequest::CloseContext { context: self.context }, &mut send) {
367                    let _ = lease.exchange(bytes, &mut receive);
368                }
369            }
370        }
371    }
372}
373
374impl<T: Transport> Drop for Tier2JsBridge<T> {
375    fn drop(&mut self) {
376        // Close before Lease::drop releases activation demand.
377        self.close_context();
378    }
