17fn now_ms() -> u64 { ostd::syscall::sys_get_time_ms().unwrap_or(u64::MAX) }
18fn wall_secs() -> u64 { ostd::syscall::sys_get_wall_secs() }
19fn domain_matches(host: &str, domain: &str) -> bool {
20    host == domain || host.strip_suffix(domain).is_some_and(|prefix| prefix.ends_with('.'))
21}
22fn path_matches(path: &str, cookie: &str) -> bool {
23    path == cookie || path.strip_prefix(cookie).is_some_and(|tail| cookie.ends_with('/') || tail.starts_with('/'))
24}

26#[derive(Clone)]
27struct Cookie { name: String, value: String, domain: String, path: String, host_only: bool, secure: bool, http_only: bool, expires: Option<u64> }
28impl Cookie { fn bytes(&self) -> usize { self.name.capacity() + self.value.capacity() + self.domain.capacity() + self.path.capacity() } }
29/// Bounded, in-memory cookies; transferable between successful navigations in one tab.
30#[derive(Clone, Default)]
31pub struct CookieJar { entries: Vec<Cookie> }
32impl CookieJar {
33    pub fn resident_bytes(&self) -> usize { self.entries.capacity() * core::mem::size_of::<Cookie>() + self.entries.iter().map(Cookie::bytes).sum::<usize>() }
34    fn header(&self, url: &Url, script: bool, now: u64) -> String {
35        let path = url.path.split('?').next().unwrap_or("/");
36        let mut matches: Vec<&Cookie> = self.entries.iter().filter(|c| (!script || !c.http_only) && (!c.secure || url.scheme == "https")
37            && (if c.host_only { url.host == c.domain } else { domain_matches(&url.host, &c.domain) })
38            && path_matches(path, &c.path) && c.expires.is_none_or(|expiry| expiry > now)).collect();
39        matches.sort_by(|a,b| b.path.len().cmp(&a.path.len()));
40        let mut value = String::new();
41        for cookie in matches { if !value.is_empty() { value.push_str("; "); } value.push_str(&cookie.name); value.push('='); value.push_str(&cookie.value); }
42        value
43    }
44    fn store(&mut self, url: &Url, value: &str, script: bool, now: u64, wall: u64) -> Result<(), String> {
45        if value.len() > 4096 || value.bytes().any(|b| b < 32 || b == 127) { return Err("Invalid/overlong cookie".into()); }
46        let mut fields = value.split(';');
47        let (name, value) = fields.next().unwrap_or("").split_once('=').ok_or("Cookie lacks value")?;
48        let name = name.trim(); let value = value.trim();
49        if name.is_empty() || !name.bytes().all(|b| b.is_ascii_alphanumeric() || b"!#$%&'*+-.^_`|~".contains(&b))
50            || !value.bytes().all(|b| b == 0x21 || (0x23..=0x2b).contains(&b) || (0x2d..=0x3a).contains(&b) || (0x3c..=0x5b).contains(&b) || (0x5d..=0x7e).contains(&b)) {
51            return Err("Invalid cookie name/value".into());
52        }
53        let request_path = url.path.split('?').next().unwrap_or("/");
54        let default_path = request_path.rsplit_once('/').map(|(prefix,_)| if prefix.is_empty() { "/" } else { prefix }).unwrap_or("/");
55        let mut cookie = Cookie { name: name.into(), value: value.into(), domain: url.host.clone(), path: default_path.into(), host_only: true, secure: false, http_only: false, expires: None };
56        let mut max_age = None; let mut expires = None;
57        for field in fields {
58            let (key, val) = field.trim().split_once('=').unwrap_or((field.trim(), "")); let val = val.trim();
59            if key.eq_ignore_ascii_case("domain") {
60                let domain = val.trim_start_matches('.').to_ascii_lowercase();
61                if domain.is_empty() || domain.ends_with('.') || !domain_matches(&url.host, &domain)
62                    || resources::registrable_domain(&domain).is_none()
63                    || !resources::is_same_site(&url.host,&domain) { return Err("Cookie Domain is not permitted".into()); }
64                cookie.domain = domain; cookie.host_only = false;
65            } else if key.eq_ignore_ascii_case("path") && val.starts_with('/') { cookie.path = val.into(); }
66            else if key.eq_ignore_ascii_case("secure") { cookie.secure = true; }
67            else if key.eq_ignore_ascii_case("httponly") { if script { return Err("Script cannot set HttpOnly cookie".into()); } cookie.http_only = true; }
68            else if key.eq_ignore_ascii_case("max-age") { max_age = val.parse::<i64>().ok(); }
69            else if key.eq_ignore_ascii_case("expires") { expires = cookie_date(val); }
70        }
71        if cookie.secure && url.scheme != "https" || name.starts_with("__Secure-") && !cookie.secure
72            || name.starts_with("__Host-") && (!cookie.secure || !cookie.host_only || cookie.path != "/") { return Err("Cookie secure policy rejected".into()); }
73        if let Some(age) = max_age { cookie.expires = Some(if age <= 0 { now } else { now.saturating_add((age as u64).saturating_mul(1000)) }); }
74        else if let Some(expiry) = expires { if wall == 0 { return Err("Cookie expiry requires wall clock".into()); } cookie.expires = Some(now.saturating_add(expiry.saturating_sub(wall).saturating_mul(1000))); }
75        self.entries.retain(|c| c.expires.is_none_or(|expiry| expiry > now));
76        if self.entries.iter().any(|c| c.name == name && (script && c.http_only || url.scheme != "https" && c.secure)
77            && domain_matches(&url.host, &c.domain) && path_matches(&cookie.path, &c.path)) { return Err("Protected cookie cannot be overwritten".into()); }
78        let previous = self.entries.iter().position(|c| c.name == cookie.name && c.domain == cookie.domain && c.path == cookie.path);
79        if cookie.expires.is_some_and(|expiry| expiry <= now) { if let Some(i) = previous { self.entries.remove(i); } return Ok(()); }
80        let old = previous.map(|i| self.entries[i].bytes()).unwrap_or(0);
81        let bytes = self.entries.iter().map(Cookie::bytes).sum::<usize>();
82        if bytes.saturating_sub(old).saturating_add(cookie.bytes()) > MAX_COOKIE_BYTES || previous.is_none() && self.entries.len() >= MAX_COOKIES { return Err("Cookie jar budget exceeded".into()); }
83        if let Some(i) = previous { self.entries[i] = cookie; } else { self.entries.push(cookie); }
84        Ok(())
85    }
86}
87// RFC 6265 date tokens, including legacy cookie separators and two-digit years.
88fn cookie_date(value: &str) -> Option<u64> {
89    let mut time = None; let mut day = None; let mut month = None; let mut year = None;
90    for token in value.split(|c: char| c.is_ascii_whitespace() || matches!(c, ',' | '-' | '/')) {
91        if token.is_empty() { continue; }
92        if time.is_none() && token.contains(':') {
93            let parts: Vec<&str> = token.split(':').collect();
94            if parts.len() == 3 { time = Some((parts[0].parse::<u64>().ok()?, parts[1].parse::<u64>().ok()?, parts[2].parse::<u64>().ok()?)); }
95        } else if month.is_none() && token.len() >= 3 {
96            month = ["jan","feb","mar","apr","may","jun","jul","aug","sep","oct","nov","dec"].iter().position(|m| token.get(..3).is_some_and(|prefix| prefix.eq_ignore_ascii_case(m))).map(|m| m as u64 + 1);
97            if month.is_some() { continue; }
98            if token.bytes().all(|b| b.is_ascii_digit()) { if day.is_none() && token.len() <= 2 { day = token.parse::<u64>().ok(); } else { year = token.parse::<u64>().ok(); } }
99        } else if token.bytes().all(|b| b.is_ascii_digit()) { if day.is_none() && token.len() <= 2 { day = token.parse::<u64>().ok(); } else { year = token.parse::<u64>().ok(); } }
100    }
101    let (h,m,s) = time?; let day = day?; let month = month?; let mut year = year?;
102    if year < 70 { year += 2000; } else if year < 100 { year += 1900; }
103    if year < 1601 || year > 9999 || day == 0 || h > 23 || m > 59 || s > 59 { return None; }
104    fn leap(y:u64)->bool { y % 4 == 0 && (y % 100 != 0 || y % 400 == 0) }
105    let days_in = [31, if leap(year) {29} else {28},31,30,31,30,31,31,30,31,30,31];
106    if day > days_in[(month - 1) as usize] { return None; }
107    if year < 1970 { return Some(0); }
108    let mut days = 0; for y in 1970..year { days += if leap(y) {366} else {365}; }
109    days += days_in[..(month - 1) as usize].iter().sum::<u64>() + day - 1;
110    Some(((days * 24 + h) * 60 + m) * 60 + s)
111}

113pub struct BrowserCompletion { pub request: BrowserRequest, pub reply: BrowserReply, pub bytes: Vec<u8> }
114struct Pending { request: BrowserRequest, url: String, visited: Vec<String>, get: bool, job: Option<FetchJob> }
115impl Pending {
116    fn id(&self) -> u32 { match &self.request { BrowserRequest::Load{id,..} | BrowserRequest::Fetch{id,..} => *id, _ => 0 } }
117    fn bytes(&self) -> usize { request_bytes(&self.request) + self.url.capacity() + self.visited.capacity() * core::mem::size_of::<String>() + self.visited.iter().map(String::capacity).sum::<usize>() + self.job.as_ref().map_or(0, FetchJob::resident_bytes) }
118}
119fn request_bytes(request: &BrowserRequest) -> usize {
120    match request {
121        BrowserRequest::Load {url,..} => url.capacity(),
122        BrowserRequest::Fetch {url,method,headers,body,credentials,..} => url.capacity() + method.capacity() + body.capacity() + credentials.capacity() + headers.capacity() * core::mem::size_of::<(String,String)>() + headers.iter().map(|(n,v)| n.capacity()+v.capacity()).sum::<usize>(),
123        BrowserRequest::SetCookie{value} => value.capacity(), BrowserRequest::Abort{..} => 0,
124    }
125}
126fn failure(request: BrowserRequest, url: String, error: String) -> BrowserCompletion {
127    let id = match &request { BrowserRequest::Load{id,..} | BrowserRequest::Fetch{id,..} => *id, _ => 0 };
128    BrowserCompletion { request, reply: BrowserReply {id,url,status:0,headers:Vec::new(),body:String::new(),error:Some(error)}, bytes:Vec::new() }
129}

131pub struct BrowserSession {
132    document_url: String, cookies: CookieJar, queue: VecDeque<Pending>, active: Vec<Pending>,
133    memory_budget: usize, accepted: usize, cursor: usize,
134}
135impl BrowserSession {
136    pub fn new(document_url: &str) -> Self { Self {document_url:document_url.into(),cookies:CookieJar::default(),queue:VecDeque::new(),active:Vec::with_capacity(MAX_ACTIVE),memory_budget:budget::MAX_RESOURCE_BYTES,accepted:0,cursor:0} }
137    pub fn document_url(&self) -> &str { &self.document_url }
138    pub fn cookies(&self) -> CookieJar { self.cookies.clone() }
139    pub fn set_cookies(&mut self, cookies: CookieJar) { self.cookies = cookies; }
140    pub fn accept_response_cookies(&mut self, url:&str, headers:&[(String,String)]) {
141        if let Ok(url) = Url::parse(url) { for (name,value) in headers { if name.eq_ignore_ascii_case("set-cookie") { let _ = self.cookies.store(&url,value,false,now_ms(),wall_secs()); } } }
142    }
143    pub fn document_cookie(&self) -> String { Url::parse(&self.document_url).map(|url| self.cookies.header(&url,true,now_ms())).unwrap_or_default() }
144    pub fn set_memory_budget(&mut self, bytes:usize) { self.memory_budget = bytes.min(budget::MAX_RESOURCE_BYTES); }
145    pub fn resident_bytes(&self) -> usize {
146        self.document_url.capacity() + self.cookies.resident_bytes() + self.queue.capacity() * core::mem::size_of::<Pending>()
147            + self.active.capacity() * core::mem::size_of::<Pending>() + self.queue.iter().chain(self.active.iter()).map(Pending::bytes).sum::<usize>()
148    }
149    pub fn is_pending(&self) -> bool { !self.queue.is_empty() || !self.active.is_empty() }
150    pub fn cancel_all(&mut self) { self.queue.clear(); self.active.clear(); }
151    pub fn enqueue(&mut self, request: BrowserRequest) -> Result<(),String> {
152        match &request {
153            BrowserRequest::Abort{id} => {
154                self.queue.retain(|p| p.id() != *id); self.active.retain(|p| p.id() != *id); return Ok(());
155            }
156            BrowserRequest::SetCookie{value} => {
157                if self.memory_budget.saturating_sub(self.resident_bytes()) < 8192 { return Err("Cookie memory budget exceeded".into()); }
158                let url = Url::parse(&self.document_url)?;
159                return self.cookies.store(&url,value,true,now_ms(),wall_secs());
160            }
161            _ => {}
162        }
163        let (id,reference) = match &request { BrowserRequest::Load{id,url,..} | BrowserRequest::Fetch{id,url,..} => (*id,url.as_str()), _ => unreachable!() };
164        if self.queue.iter().chain(self.active.iter()).any(|p| p.id() == id) { return Err("Duplicate pending browser request ID".into()); }
165        if self.queue.len() + self.active.len() >= MAX_PENDING || self.accepted >= budget::MAX_REFERENCES { return Err("Browser request count budget exceeded".into()); }
166        if request_bytes(&request) > MAX_REQUEST_BYTES { return Err("Browser request byte budget exceeded".into()); }
167        let url = resources::resolve(&self.document_url, reference)?;
168        validate_target(&self.document_url,&url)?;
169        if url.starts_with("file://") && matches!(request,BrowserRequest::Fetch{..}) { return Err("Fetch requires an HTTP(S) document".into()); }
170        if let BrowserRequest::Fetch{method,headers,body,credentials,..} = &request { validate_fetch(method,headers,body,credentials)?; }
171        let capacity = if self.queue.len() == self.queue.capacity() { (self.queue.capacity().max(1) * 2).min(MAX_PENDING) } else { self.queue.capacity() };
172        let growth = capacity.saturating_sub(self.queue.capacity()) * core::mem::size_of::<Pending>();
173        let reserve = request_bytes(&request).saturating_add(url.capacity() * 2).saturating_add(core::mem::size_of::<String>()).saturating_add(growth);
174        if reserve > self.memory_budget.saturating_sub(self.resident_bytes()) { return Err("Browser request exceeds remaining memory budget".into()); }
175        if growth != 0 { self.queue.reserve_exact(capacity - self.queue.len()); }
176        let visited = alloc::vec![url.clone()];
177        self.queue.push_back(Pending {request,url,visited,get:false,job:None}); self.accepted += 1; Ok(())
178    }
179    /// At most four exact transport operations are advanced per call; one completion is returned.
180    pub fn poll(&mut self) -> Option<BrowserCompletion> {
181        while self.active.len() < MAX_ACTIVE { match self.queue.pop_front() { Some(pending) => self.active.push(pending), None => break } }
182        let count = self.active.len();
183        for _ in 0..count {
184            if self.active.is_empty() { break; }
185            let i = self.cursor % self.active.len(); self.cursor = i + 1;
186            if self.active[i].job.is_none() {
187                if self.active[i].url.starts_with("file://") {
188                    let pending = self.active.remove(i);
189                    let limit = response_limit(&pending.request);
190                    let result = resources::read_local(&pending.url, limit.min(self.memory_budget.saturating_sub(self.resident_bytes()) / 2));
191                    return Some(match result { Ok(body) => complete(&self.document_url,pending, FetchResult {url:String::new(),body,media_type:String::new(),gemini:false,status:200,headers:Vec::new()}), Err(error) => failure(pending.request,pending.url,error) });
192                }
193                let available = self.memory_budget.saturating_sub(self.resident_bytes());
194                if available < 32 * 1024 {
195                    let pending = self.active.remove(i); return Some(failure(pending.request,pending.url,"Browser transport memory budget exceeded".into()));
196                }
197                let result = self.start(&self.active[i], available);
198                match result { Ok(job) => self.active[i].job = Some(job), Err(error) => { let pending = self.active.remove(i); return Some(failure(pending.request,pending.url,error)); } }
199            }
200            let other_bytes = self.resident_bytes().saturating_sub(self.active[i].job.as_ref().map_or(0,FetchJob::resident_bytes));
201            let available = self.memory_budget.saturating_sub(other_bytes);
202            let job = self.active[i].job.as_mut().unwrap(); job.set_memory_budget(available / 2);
203            match job.poll() {
204                Poll::Pending => {},
205                Poll::Ready(result) => {
206                    let mut pending = self.active.remove(i); pending.job = None;
207                    match result {
208                        Err(error) => return Some(failure(pending.request,pending.url,error)),
209                        Ok(result) => {
210                            let target = match Url::parse(&pending.url) { Ok(url) => url, Err(error) => return Some(failure(pending.request,pending.url,error)) };
211                            let credentialed = credentials_allowed(&pending.request,&self.document_url,&target);
212                            if credentialed { self.accept_response_cookies(&pending.url,&result.headers); }
213                            if let BrowserRequest::Fetch{credentials,..} = &pending.request {
214                                if let Err(error) = validate_cors(&self.document_url,&target,&result.headers,credentials == "include") { return Some(failure(pending.request,pending.url,error)); }
215                            }
216                            if matches!(result.status,301|302|303|307|308) {
217                                match redirect_pending(&self.document_url,&mut pending,&result) {
218                                    Ok(()) => { self.active.push(pending); },
219                                    Err(error) => return Some(failure(pending.request,pending.url,error)),
220                                }
221                            } else { return Some(complete(&self.document_url,pending,result)); }
222                        }
223                    }
224                }
225            }
226        }
227        None
228    }
229    fn start(&self, pending:&Pending, available:usize) -> Result<FetchJob,String> {
230        let target = Url::parse(&pending.url)?;
231        let mut headers: Vec<(&str,&str)> = Vec::new(); let mut method = "GET"; let mut body:&[u8] = &[];
232        let origin = if matches!(pending.request,BrowserRequest::Fetch{..}) { Url::parse(&self.document_url)?.origin() } else { String::new() };
233        if let BrowserRequest::Fetch{method:m,headers:h,body:b,..} = &pending.request {
234            if !pending.get { method = m; body = b.as_bytes(); }
235            for (name,value) in h { if redirect_keeps_header(pending.get,name) { headers.push((name.as_str(),value.as_str())); } }
236            if origin != target.origin() || method == "POST" { headers.push(("Origin",origin.as_str())); }
237        }
238        let cookie = if credentials_allowed(&pending.request,&self.document_url,&target) { self.cookies.header(&target,false,now_ms()) } else { String::new() };
239        if !cookie.is_empty() { headers.push(("Cookie",cookie.as_str())); }
240        let mut job = FetchJob::browser(&pending.url,response_limit(&pending.request).min(available / 2),method,&headers,body)?;
241        job.set_memory_budget(available / 2); Ok(job)
242    }
243}
244fn redirect_keeps_header(converted_to_get:bool,name:&str) -> bool {
245    !converted_to_get || !["content-type","content-language"].iter().any(|header| name.eq_ignore_ascii_case(header))
246}
247impl Drop for BrowserSession { fn drop(&mut self) { self.cancel_all(); } }
248fn validate_target(document:&str,target:&str) -> Result<(),String> {
249    if document.starts_with("file://") { if target.starts_with("file://") { resources::local_path(target)?; return Ok(()); } return Err("Local document cannot load network resources".into()); }
250    let base = Url::parse(document)?; let target = Url::parse(target)?;
251    if !matches!(base.scheme.as_str(),"http"|"https") || !matches!(target.scheme.as_str(),"http"|"https") { return Err("Unsupported browser resource scheme".into()); }
252    if base.scheme == "https" && target.scheme != "https" { return Err("Insecure mixed content blocked".into()); }
253    if !resources::is_same_site(&base.host,&target.host) { return Err("Cross-site browser request blocked".into()); }
254    Ok(())
255}
256fn validate_fetch(method:&str,headers:&[(String,String)],body:&str,credentials:&str) -> Result<(),String> {
257    if !matches!(method,"GET"|"HEAD"|"POST") || method != "POST" && !body.is_empty() { return Err("Unsupported fetch method/body".into()); }
258    if !matches!(credentials,"omit"|"same-origin"|"include") { return Err("Invalid fetch credentials mode".into()); }
259    if body.len() > 64 * 1024 || headers.len() > 24 { return Err("Fetch request exceeds budget".into()); }
260    let mut total = 0;
261    for (name,value) in headers {
262        total += name.len() + value.len();
263        if total > 8192 || value.len() > 128 || value.bytes().any(|b| b < 32 || b == 127 || b"\"():<>?@[\\]{}".contains(&b)) { return Err("Unsafe/non-simple fetch header".into()); }
264        match name.to_ascii_lowercase().as_str() {
265            "accept" => {},
266            "accept-language" | "content-language" if value.bytes().all(|b| b.is_ascii_alphanumeric() || b" *,-.;=".contains(&b)) => {},
267            "content-type" if matches!(value.split(';').next().unwrap_or("").trim().to_ascii_lowercase().as_str(),"application/x-www-form-urlencoded"|"multipart/form-data"|"text/plain") => {},
268            _ => return Err("Fetch header requires unavailable CORS preflight or is forbidden".into()),
269        }
270    }
271    Ok(())
272}
273fn credentials_allowed(request:&BrowserRequest,document:&str,target:&Url) -> bool {
274    match request {
275        BrowserRequest::Load{..} => true,
276        BrowserRequest::Fetch{credentials,..} => credentials == "include" || credentials == "same-origin" && Url::parse(document).is_ok_and(|base| base.origin() == target.origin()),
277        _ => false,
278    }
279}
280fn one_header<'a>(headers:&'a [(String,String)],name:&str) -> Result<Option<&'a str>,String> {
281    let mut found = None; for (key,value) in headers { if key.eq_ignore_ascii_case(name) { if found.is_some() { return Err("Duplicate CORS/redirect header".into()); } found = Some(value.as_str()); } } Ok(found)
282}
283fn validate_cors(document:&str,target:&Url,headers:&[(String,String)],credentials:bool) -> Result<(),String> {
284    let origin = Url::parse(document)?.origin(); if origin == target.origin() { return Ok(()); }
285    let allow = one_header(headers,"access-control-allow-origin")?.ok_or("Cross-origin response lacks CORS permission")?;
286    if allow != origin && !(allow == "*" && !credentials) { return Err("Cross-origin response origin rejected".into()); }
287    if credentials && one_header(headers,"access-control-allow-credentials")? != Some("true") { return Err("Cross-origin credentials rejected".into()); }
288    Ok(())
289}
290fn exposed_headers(request:&BrowserRequest,document:&str,target:&Url,headers:Vec<(String,String)>) -> Vec<(String,String)> {
291    let cross = matches!(request,BrowserRequest::Fetch{..}) && Url::parse(document).is_ok_and(|base| base.origin() != target.origin());
292    let include = matches!(request,BrowserRequest::Fetch{credentials,..} if credentials == "include");
293    let mut exposed = Vec::new();
294    if cross { for (name,value) in &headers { if name.eq_ignore_ascii_case("access-control-expose-headers") { for token in value.split(',') { exposed.push(token.trim().to_ascii_lowercase()); } } } }
295    headers.into_iter().filter(|(name,_)| !name.eq_ignore_ascii_case("set-cookie") && !name.eq_ignore_ascii_case("set-cookie2")
296        && (!cross || ["cache-control","content-language","content-length","content-type","expires","last-modified","pragma"].contains(&name.as_str()) || exposed.iter().any(|item| item == name || item == "*" && !include))).collect()
297}
298fn redirect_pending(document:&str,pending:&mut Pending,result:&FetchResult) -> Result<(),String> {
299    if pending.visited.len() > MAX_REDIRECTS { return Err("Too many browser redirects".into()); }
300    let location = one_header(&result.headers,"location")?.ok_or("Redirect lacks Location")?;
301    let next = Url::parse(&pending.url)?.redirect(location)?.full;
302    validate_target(document,&next)?;
303    if pending.visited.contains(&next) { return Err("Browser redirect loop".into()); }
304    if let BrowserRequest::Fetch{method,..} = &pending.request { if result.status == 303 && method != "HEAD" || matches!(result.status,301|302) && method == "POST" { pending.get = true; } }
305    pending.visited.push(next.clone()); pending.url = next; Ok(())
306}
307fn response_limit(request:&BrowserRequest) -> usize { match request { BrowserRequest::Load{kind:BrowserResourceKind::Image,..} => budget::MAX_ENCODED_IMAGE_BYTES, _ => MAX_BODY } }
308fn lossy_length(mut bytes:&[u8]) -> usize {
309    let mut size = 0usize;
310    loop {
311        match core::str::from_utf8(bytes) {
312            Ok(text) => return size.saturating_add(text.len()),
313            Err(error) => {
314                size = size.saturating_add(error.valid_up_to()).saturating_add(3);
315                match error.error_len() { Some(length) => bytes = &bytes[error.valid_up_to()+length..], None => return size }
316            }
317        }
318    }
319}
320fn complete(document:&str,pending:Pending,mut result:FetchResult) -> BrowserCompletion {
321    if result.url.is_empty() { result.url = pending.url.clone(); }
322    let image = matches!(pending.request,BrowserRequest::Load{kind:BrowserResourceKind::Image,..});
323    if matches!(pending.request,BrowserRequest::Load{..}) && !(200..300).contains(&result.status) { return failure(pending.request,result.url,format!("HTTP status {}",result.status)); }
324    let strict = matches!(pending.request,BrowserRequest::Load{..});
325    let headers = match Url::parse(&result.url) {
326        Ok(target) => exposed_headers(&pending.request,document,&target,result.headers),
327        Err(_) => Vec::new(),
328    };
329    let length = if image { 0 } else if strict { result.body.len() } else { lossy_length(&result.body) };
330    let size = length + result.url.len() + headers.iter().map(|(n,v)|n.len()+v.len()+12).sum::<usize>() + 128;
331    if length > MAX_BODY || size > dom_arena::OCEL_JS_MAX_TRANSFER { return failure(pending.request,result.url,"Browser reply exceeds wire budget".into()); }
332    let body = if image { String::new() } else if strict {
333        match String::from_utf8(core::mem::take(&mut result.body)) { Ok(body) => body, Err(_) => return failure(pending.request,result.url,"Browser resource is not UTF-8".into()) }
334    } else {
335        match String::from_utf8(core::mem::take(&mut result.body)) { Ok(body) => body, Err(error) => String::from_utf8_lossy(error.as_bytes()).into_owned() }
336    };
337    BrowserCompletion { reply:BrowserReply{id:pending.id(),url:result.url,status:result.status,headers,body,error:None},request:pending.request,bytes:if image {result.body} else {Vec::new()} }
338}
