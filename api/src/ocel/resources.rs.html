2//! Same-origin classic script and stylesheet loading. No cross-origin requests.
3extern crate alloc;
4use alloc::{format, string::String, vec::Vec};
5const MAX_RESOURCE_BYTES: usize = 256 * 1024;
6
7pub fn resolve(base: &str, reference: &str) -> Result<String, String> {
8    let reference = reference.trim();
9    let base = base.split('#').next().unwrap_or(base);
10    let (origin, base_path) = if let Some((scheme, rest)) = base.split_once("://") {
11        if !matches!(scheme, "http" | "https" | "file") { return Err(String::from("Unsupported resource scheme")); }
12        let split = rest.find('/').unwrap_or(rest.len());
13        (format!("{}://{}", scheme, &rest[..split]), &rest[split..])
14    } else { (String::new(), base) };
15    let reference = if reference.contains("://") {
16        let suffix = reference.strip_prefix(&origin).ok_or_else(|| String::from("Cross-origin resources are not supported"))?;
17        if origin.is_empty() || !suffix.starts_with('/') { return Err(String::from("Cross-origin resources are not supported")); }
18        suffix
19    } else {
20        if reference.starts_with("//") || reference.split('/').next().unwrap_or("").contains(':') { return Err(String::from("Unsupported resource URL")); }
21        reference
22    };
23    let base_path = base_path.split('?').next().unwrap_or(base_path);
24    let reference = reference.split('#').next().unwrap_or(reference);
25    let joined = if reference.is_empty() { String::from(base_path) }
26        else if reference.starts_with('/') { String::from(reference) }
27        else if reference.starts_with('?') { format!("{}{}", base_path, reference) }
28        else { format!("{}/{}", base_path.rsplit_once('/').map(|(dir, _)| dir).unwrap_or(""), reference) };
29    let (path, query) = joined.split_once('?').map(|(p,q)| (p,Some(q))).unwrap_or((&joined,None));
30    let mut parts: Vec<&str> = Vec::new();
31    for part in path.split('/') { match part { "" | "." => {}, ".." => { parts.pop(); }, _ => parts.push(part) } }
32    let mut result = format!("{}/{}", origin, parts.join("/"));
33    if let Some(query) = query { result.push('?'); result.push_str(query); }
34    Ok(result)
35}

37pub fn load_text(base: &str, reference: &str) -> Result<String, String> {
38    let url = resolve(base, reference)?;
39    let content = if url.starts_with("https://") { crate::net::fetch_https(&url)? }
40        else if url.starts_with("http://") { crate::net::fetch_http(&url)? }
41        else {
42            use ostd::fs::File;
43            let mut file = File::open(url.strip_prefix("file://").unwrap_or(&url)).map_err(|_| format!("Cannot open {}", url))?;
44            let mut bytes = Vec::new();
45            let mut chunk = [0u8; 4096];
46            loop {
47                let count = file.read(&mut chunk).map_err(|_| format!("Cannot read {}", url))?;
48                if count == 0 { break; }
49                if bytes.len() + count > MAX_RESOURCE_BYTES { return Err(String::from("Resource exceeds 256 KiB")); }
50                bytes.extend_from_slice(&chunk[..count]);
51            }
52            String::from_utf8(bytes).map_err(|_| String::from("Resource is not UTF-8"))?
53        };
54    if content.len() > MAX_RESOURCE_BYTES { return Err(String::from("Resource exceeds 256 KiB")); }
55    Ok(content)
56}

58#[cfg(test)]
59mod tests {
60    use super::*;
61    #[test]
62    fn relative_resource_resolution_and_origin_boundary() {
63        assert_eq!(resolve("https://example.com/a/b.html?q=1#x", "../js/app.js?v=2").unwrap(), "https://example.com/js/app.js?v=2");
64        assert_eq!(resolve("file:///data/web/index.html", "./style.css").unwrap(), "file:///data/web/style.css");
65        assert_eq!(resolve("https://example.com/a.html", "/css/site.css").unwrap(), "https://example.com/css/site.css");
66        for url in ["https://example.com.evil/x", "https://evil/x", "//evil/x", "javascript:alert(1)", "data:text/plain,x"] {
67            assert!(resolve("https://example.com/a.html", url).is_err());
68        }
69    }
