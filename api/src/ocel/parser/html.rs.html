19enum HtmlToken {
20    StartTag {
21        name: String,
22        attributes: Vec<(String, String)>,
23        self_closing: bool,
24    },
25    EndTag {
26        name: String,
27    },
28    Text(String),
29}

31/// Tokenize an HTML string into tags and text chunks.
32fn tokenize_html(input: &str) -> (Vec<HtmlToken>, Vec<String>) {
33    let mut tokens = Vec::new();
34    let mut scripts = Vec::new();
35    let mut chars = input.chars().peekable();
36
37    while let Some(c) = chars.next() {
38        if c == '<' {
39            // Check comment
40            if chars.clone().take(3).collect::<String>() == "!--" {
41                // Consume <!--
42                for _ in 0..3 {
43                    chars.next();
44                }
45                // Skip until -->
46                while let Some(ch) = chars.next() {
47                    if ch == '-' && chars.clone().take(2).collect::<String>() == "->" {
48                        chars.next();
49                        chars.next();
50                        break;
51                    }
52                }
53                continue;
54            }
55
56            // Read tag content
57            let mut tag_content = String::new();
58            while let Some(&ch) = chars.peek() {
59                chars.next();
60                if ch == '>' {
61                    break;
62                }
63                tag_content.push(ch);
64            }
65
66            let trimmed_tag = tag_content.trim();
67            if let Some(rest) = trimmed_tag.strip_prefix('/') {
68                let name = rest.trim().to_ascii_lowercase();
69                tokens.push(HtmlToken::EndTag { name });
70            } else {
71                let self_closing = trimmed_tag.ends_with('/');
72                let tag_body = if self_closing {
73                    trimmed_tag.trim_end_matches('/').trim()
74                } else {
75                    trimmed_tag
76                };
77
78                let mut parts = tag_body.split_whitespace();
79                let name = parts.next().unwrap_or("").to_ascii_lowercase();
80                let mut attributes = Vec::new();
81
82                for part in parts {
83                    if let Some((k, v)) = part.split_once('=') {
84                        let clean_v = v.trim_matches('"').trim_matches('\'');
85                        attributes.push((k.to_ascii_lowercase(), String::from(clean_v)));
86                    } else if !part.is_empty() {
87                        attributes.push((part.to_ascii_lowercase(), String::new()));
88                    }
89                }
90
91                // If this is a <script> tag, capture the script body directly
92                if name == "script" && !self_closing {
93                    let mut script_body = String::new();
94                    while let Some(sc) = chars.next() {
95                        if sc == '<'
96                            && chars
97                                .clone()
98                                .take(8)
99                                .collect::<String>()
100                                .eq_ignore_ascii_case("/script>")
101                        {
102                            for _ in 0..8 {
103                                chars.next();
104                            }
105                            break;
106                        }
107                        script_body.push(sc);
108                    }
109                    scripts.push(script_body);
110                    continue;
111                }
112
113                tokens.push(HtmlToken::StartTag {
114                    name,
115                    attributes,
116                    self_closing,
117                });
118            }
119        } else {
120            // Text chunk
121            let mut text = String::new();
122            text.push(c);
123            while let Some(&ch) = chars.peek() {
124                if ch == '<' {
125                    break;
126                }
127                text.push(ch);
128                chars.next();
129            }
130            let decoded = decode_entities(&text);
131            if !decoded.trim().is_empty() {
132                tokens.push(HtmlToken::Text(decoded));
133            }
134        }
135    }
136
137    (tokens, scripts)
138}

140/// Parse HTML text into a `DocumentArena` and renderable `DocNode`s.
141pub fn parse_html(input: &str) -> HtmlOutput {
142    let (tokens, scripts) = tokenize_html(input);
143    let mut arena = DocumentArena::new();
144    let mut tag_stack: Vec<(String, NodeId)> = alloc::vec![(String::from("root"), arena.root)];
145
146    for tok in tokens {
147        match tok {
148            HtmlToken::StartTag {
149                name,
150                attributes,
151                self_closing,
152            } => {
153                let parent_id = tag_stack.last().map(|(_, id)| *id).unwrap_or(arena.root);
154                let node_id = arena.alloc_node(NodeData::Element {
155                    tag: name.clone(),
156                    attributes,
157                });
158                arena.append_child(parent_id, node_id);
159
160                let is_void_tag = self_closing
161                    || matches!(
162                        name.as_str(),
163                        "hr" | "br" | "img" | "input" | "meta" | "link"
164                    );
165
166                if !is_void_tag {
167                    tag_stack.push((name, node_id));
168                }
169            }
170            HtmlToken::EndTag { name } => {
171                if let Some(pos) = tag_stack.iter().rposition(|(t, _)| t == &name) {
172                    tag_stack.truncate(pos);
173                }
174            }
175            HtmlToken::Text(text) => {
176                let parent_id = tag_stack.last().map(|(_, id)| *id).unwrap_or(arena.root);
177                let text_id = arena.alloc_node(NodeData::Text(text));
178                arena.append_child(parent_id, text_id);
179            }
180        }
181    }
182
183    let nodes = arena_to_doc_nodes(&arena);
184    HtmlOutput {
185        nodes,
186        scripts,
187        arena,
188    }
189}

191/// Converts a `DocumentArena` DOM tree into renderable `DocNode`s for the layout engine.
192pub fn arena_to_doc_nodes(arena: &DocumentArena) -> Vec<DocNode> {
193    let mut doc_nodes = Vec::new();
194    render_children(arena, arena.root, &mut doc_nodes);
195    doc_nodes
196}

198fn render_children(arena: &DocumentArena, parent_id: NodeId, out: &mut Vec<DocNode>) {
199    let Some(parent) = arena.get(parent_id) else {
200        return;
201    };
202    let mut cur = parent.first_child;
203
204    while let Some(child_id) = cur {
205        if let Some(child) = arena.get(child_id) {
206            match &child.data {
207                NodeData::Element { tag, attributes: _ } => {
208                    match tag.as_str() {
209                        "h1" | "h2" | "h3" | "h4" | "h5" | "h6" => {
210                            let level = tag[1..].parse::<u8>().unwrap_or(1);
211                            let text = arena.get_text_content(child_id);
212                            out.push(DocNode::Heading {
213                                level: level.clamp(1, 4),
214                                text: text.trim().into(),
215                            });
216                        }
217                        "p" | "div" | "section" | "article" | "header" | "footer" | "main" => {
218                            let mut spans = Vec::new();
219                            collect_inline_spans(
220                                arena, child_id, &mut spans, false, false, false, None,
221                            );
222                            if !spans.is_empty() {
223                                out.push(DocNode::Paragraph { spans });
224                            }
225                        }
226                        "pre" => {
227                            let raw_text = arena.get_text_content(child_id);
228                            let lines: Vec<String> = raw_text.lines().map(String::from).collect();
229                            out.push(DocNode::CodeBlock {
230                                lang: String::new(),
231                                lines,
232                            });
233                        }
234                        "ul" | "ol" => {
235                            let mut li_cur = child.first_child;
236                            while let Some(li_id) = li_cur {
237                                if let Some(li_node) = arena.get(li_id) {
238                                    if li_node.tag() == Some("li") {
239                                        let mut spans = Vec::new();
240                                        collect_inline_spans(
241                                            arena, li_id, &mut spans, false, false, false, None,
242                                        );
243                                        out.push(DocNode::ListItem {
244                                            bullet: '•',
245                                            indent: 0,
246                                            spans,
247                                        });
248                                    }
249                                }
250                                li_cur = arena.get(li_id).and_then(|n| n.next_sibling);
251                            }
252                        }
253                        "table" => {
254                            let (headers, rows) = parse_table_node(arena, child_id);
255                            if !headers.is_empty() || !rows.is_empty() {
256                                out.push(DocNode::Table { headers, rows });
257                            }
258                        }
259                        "hr" => {
260                            out.push(DocNode::Rule);
261                        }
262                        "button" => {
263                            let btn_text = arena.get_text_content(child_id);
264                            out.push(DocNode::Button {
265                                text: btn_text.trim().into(),
266                                node_id: Some(child_id),
267                            });
268                        }
269                        _ => {
270                            // Recursively render other elements
271                            render_children(arena, child_id, out);
272                        }
273                    }
274                }
275                NodeData::Text(text) => {
276                    let trimmed = text.trim();
277                    if !trimmed.is_empty() {
278                        out.push(DocNode::Paragraph {
279                            spans: alloc::vec![StyledSpan::plain(trimmed)],
280                        });
281                    }
282                }
283                _ => {}
284            }
285        }
286        cur = arena.get(child_id).and_then(|n| n.next_sibling);
287    }
288}

290/// Recursively collects styled spans with inheritance of bold, italic, code, and link properties.
291fn collect_inline_spans(
292    arena: &DocumentArena,
293    node_id: NodeId,
294    out: &mut Vec<StyledSpan>,
295    bold: bool,
296    italic: bool,
297    code: bool,
298    link: Option<String>,
299) {
300    let Some(node) = arena.get(node_id) else {
301        return;
302    };
303    let mut cur = node.first_child;
304
305    while let Some(child_id) = cur {
306        if let Some(child) = arena.get(child_id) {
307            match &child.data {
308                NodeData::Element { tag, attributes } => {
309                    let next_bold = bold || tag == "b" || tag == "strong";
310                    let next_italic = italic || tag == "i" || tag == "em";
311                    let next_code = code || tag == "code";
312                    let next_link = if tag == "a" {
313                        attributes
314                            .iter()
315                            .find(|(k, _)| k == "href")
316                            .map(|(_, v)| v.clone())
317                            .or(link.clone())
318                    } else {
319                        link.clone()
320                    };
321
322                    collect_inline_spans(
323                        arena,
324                        child_id,
325                        out,
326                        next_bold,
327                        next_italic,
328                        next_code,
329                        next_link,
330                    );
331                }
332                NodeData::Text(text) if !text.is_empty() => {
333                    let color = if link.is_some() {
334                        Some(theme::ACCENT_BLUE)
335                    } else if code {
336                        Some(theme::ACCENT_CYAN)
337                    } else {
338                        None
339                    };
340
341                    out.push(StyledSpan {
342                        text: text.clone(),
343                        bold,
344                        italic,
345                        code,
346                        link: link.clone(),
347                        color,
348                    });
349                }
350                _ => {}
351            }
352        }
353        cur = arena.get(child_id).and_then(|n| n.next_sibling);
354    }
355}

357/// Extracts table headers and rows from a `<table>` node.
358fn parse_table_node(arena: &DocumentArena, table_id: NodeId) -> (Vec<String>, Vec<Vec<String>>) {
359    let mut headers = Vec::new();
360    let mut rows = Vec::new();
361
362    let Some(table) = arena.get(table_id) else {
363        return (headers, rows);
364    };
365
366    let mut cur = table.first_child;
367    while let Some(tr_id) = cur {
368        if let Some(tr_node) = arena.get(tr_id) {
369            let tag = tr_node.tag().unwrap_or("");
370            if tag == "tr" {
371                let mut cell_cur = tr_node.first_child;
372                let mut row_cells = Vec::new();
373                let mut is_header_row = false;
374
375                while let Some(cell_id) = cell_cur {
376                    if let Some(cell_node) = arena.get(cell_id) {
377                        let cell_tag = cell_node.tag().unwrap_or("");
378                        if cell_tag == "th" {
379                            is_header_row = true;
380                            row_cells.push(arena.get_text_content(cell_id).trim().into());
381                        } else if cell_tag == "td" {
382                            row_cells.push(arena.get_text_content(cell_id).trim().into());
383                        }
384                    }
385                    cell_cur = arena.get(cell_id).and_then(|n| n.next_sibling);
386                }
387
388                if is_header_row && headers.is_empty() {
389                    headers = row_cells;
390                } else if !row_cells.is_empty() {
391                    rows.push(row_cells);
392                }
393            } else if tag == "tbody" || tag == "thead" {
394                // Nested tbody/thead
395                let (sub_h, sub_r) = parse_table_node(arena, tr_id);
396                if headers.is_empty() && !sub_h.is_empty() {
397                    headers = sub_h;
398                }
399                rows.extend(sub_r);
400            }
401        }
402        cur = arena.get(tr_id).and_then(|n| n.next_sibling);
403    }
404
405    (headers, rows)
406}

408/// Decode basic HTML entities into UTF-8 characters.
409pub fn decode_entities(input: &str) -> String {
410    input
411        .replace("&nbsp;", " ")
412        .replace("&amp;", "&")
413        .replace("&lt;", "<")
414        .replace("&gt;", ">")
415        .replace("&quot;", "\"")
416        .replace("&#39;", "'")
417        .replace("&apos;", "'")
418}

420/// Strip all `<...>` tags and decode basic HTML entities (utility helper).
421#[allow(dead_code)]
422pub fn strip_tags(input: &str) -> String {
423    let mut out = String::new();
424    let mut in_tag = false;
425
426    for c in input.chars() {
427        if c == '<' {
428            in_tag = true;
429        } else if c == '>' {
430            in_tag = false;
431        } else if !in_tag {
432            out.push(c);
433        }
434    }
435
436    decode_entities(&out)
