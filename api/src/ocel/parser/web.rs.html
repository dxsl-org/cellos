47fn collect_styles(arena:&DocumentArena,id:NodeId,external:&[(NodeId,String)],sheet:&mut Stylesheet,viewport:(u32,u32)) {
48    let Some(node)=arena.get(id) else{return;};
49    if node.tag()==Some("style") {sheet.append(&arena.get_text_content(id),viewport);}
50    if node.tag()==Some("link") {if let Some((_,css))=external.iter().find(|(n,_)|*n==id) {sheet.append(css,viewport);}}
51    let mut child=node.first_child;while let Some(id)=child{collect_styles(arena,id,external,sheet,viewport);child=arena.get(id).and_then(|n|n.next_sibling);}
52}
53impl<'a> Builder<'a> {
54    fn text_leaf(&mut self,runs:&mut Vec<TextRun>,style:&ComputedStyle,parent:usize)->Result<taffy::NodeId,taffy::TaffyError> {
55        let context=Rc::new(TextContext {runs:core::mem::take(runs),line_height:pixel(style.line_height.max(1.0)),pre:style.pre,wrap_pre:style.wrap_pre,align:style.text_align});
56        let leaf_style=Style{display:Display::Block,..Style::default()};
57        let id=self.tree.new_leaf_with_context(leaf_style,context.clone())?;
58        self.entries.push(Entry{taffy:id,parent:Some(parent),node_id:None,hidden:false,background:None,border_color:style.border_color,text:Some(context)});Ok(id)
59    }
60    fn element(&mut self,id:NodeId,parent:Option<usize>,style:ComputedStyle)->Result<taffy::NodeId,taffy::TaffyError> {
61        let tid=self.tree.new_leaf(Style::default())?;let index=self.entries.len();
62        self.entries.push(Entry{taffy:tid,parent,node_id:Some(id),hidden:style.layout.display==Display::None,background:style.background,border_color:style.border_color,text:None});
63        if style.layout.display==Display::None{self.tree.set_style(tid,style.layout)?;return Ok(tid);}
64        let mut children=Vec::new();let mut runs=Vec::new();
65        let arena=self.arena;
66        let mut child=arena.get(id).and_then(|n|n.first_child);
67        while let Some(cid)=child {
68            let Some(node)=arena.get(cid) else{break;};
69            match &node.data {
70                NodeData::Text(text)=>append_text(&mut runs,text,&style,None,id),
71                NodeData::Element{..}=>{
72                    let child_style=self.sheet.compute(self.arena,cid,Some(&style),self.viewport);
73                    if child_style.layout.display!=Display::None {
74                        if child_style.inline {self.inline(cid,&child_style,None,&mut runs);}
75                        else {if !runs.is_empty(){children.push(self.text_leaf(&mut runs,&style,index)?);}children.push(self.element(cid,Some(index),child_style)?);}
76                    }
77                },_=>{}
78            }
79            child=node.next_sibling;
80        }
81        if !runs.is_empty(){children.push(self.text_leaf(&mut runs,&style,index)?);}
82        self.tree.set_children(tid,&children)?;self.tree.set_style(tid,style.layout)?;Ok(tid)
83    }
84    fn inline(&self,id:NodeId,style:&ComputedStyle,link:Option<&str>,runs:&mut Vec<TextRun>) {
85        let Some(node)=self.arena.get(id) else{return;};
86        let link=node.get_attribute("href").or(link);
87        if node.tag()==Some("br") {append_text(runs,"\u{2028}",style,link,id);return;}
88        let mut child=node.first_child;while let Some(cid)=child {let Some(n)=self.arena.get(cid) else{break;};match &n.data{
89            NodeData::Text(text)=>append_text(runs,text,style,link,id),
90            NodeData::Element{..}=>{let s=self.sheet.compute(self.arena,cid,Some(style),self.viewport);if s.layout.display!=Display::None{self.inline(cid,&s,link,runs);}},_=>{}
91        }child=n.next_sibling;}
92    }
93}
94fn append_text(runs:&mut Vec<TextRun>,text:&str,style:&ComputedStyle,link:Option<&str>,node_id:NodeId) {
95    let mut span=StyledSpan::plain(text);
96    span.color=Some(style.color);span.bold=style.bold;span.italic=style.italic;
97    span.code=style.mono;span.link=link.map(String::from);
98    append_run(runs,TextRun::new(span,style.font_size,Some(node_id),style.line_height));
99}

104fn same_run(a:&TextRun,b:&TextRun)->bool {
105    a.node_id==b.node_id && a.px==b.px && a.line_height==b.line_height
106        && a.span.bold==b.span.bold && a.span.italic==b.span.italic
107        && a.span.code==b.span.code && a.span.color==b.span.color && a.span.link==b.span.link
108}
109pub fn append_run(runs:&mut Vec<TextRun>,run:TextRun) {
110    if let Some(last)=runs.last_mut().filter(|last|same_run(last,&run)) {
111        last.span.text.push_str(&run.span.text);
112    } else {runs.push(run);}
113}
114fn next_advance(fonts:&mut TextFonts,line:&TextLine,source:&TextRun,c:char)->f32 {
115    let kern=line.runs.last().filter(|r|same_run(r,source))
116        .and_then(|r|r.span.text.chars().next_back())
117        .map_or(0.0,|left|fonts.kerning(source.face(),left,c,source.px));
118    fonts.advance(source.face(),c,source.px)+kern
119}
120fn push_char(fonts:&mut TextFonts,line:&mut TextLine,source:&TextRun,c:char) {
121    let advance=next_advance(fonts,line,source,c);
122    line.baseline=line.baseline.max(fonts.ascender(source.face(),source.px));
123    line.height=line.height.max(ceil_pixel(source.line_height.max(fonts.line_height(source.face(),source.px))));
124    if let Some(last)=line.runs.last_mut().filter(|r|same_run(r,source)) {
125        last.span.text.push(c);last.width+=advance;
126    } else {
127        let mut span=StyledSpan::plain("");
128        span.bold=source.span.bold;span.italic=source.span.italic;span.code=source.span.code;
129        span.link=source.span.link.clone();span.color=source.span.color;span.text.push(c);
130        let mut run=TextRun::new(span,source.px,source.node_id,source.line_height);
131        run.x=line.width;run.width=advance;line.runs.push(run);
132    }
133    line.width+=advance;
134}
135fn finish_line(fonts:&mut TextFonts,line:&mut TextLine,lines:&mut Vec<TextLine>,height:u32) {
136    // The shared baseline also covers mixed-size ascenders and descenders.
137    let descent=line.runs.iter().map(|r|
138        fonts.line_height(r.face(),r.px)-fonts.ascender(r.face(),r.px))
139        .fold(0.0f32,f32::max);
140    line.height=line.height.max(ceil_pixel(line.baseline+descent));
141    lines.push(core::mem::replace(line,empty_line(height)));
142}
143fn flush_word<'a>(fonts:&mut TextFonts,word:&mut Vec<(&'a TextRun,char)>,line:&mut TextLine,
144    lines:&mut Vec<TextLine>,pending:&mut Option<&'a TextRun>,limit:f32,height:u32) {
145    if word.is_empty(){return;}
146    let mut width=0.0;let mut previous:Option<(&TextRun,char)>=None;
147    for &(r,c) in word.iter() {
148        width+=fonts.advance(r.face(),c,r.px);
149        if let Some((left,l))=previous.filter(|(left,_)|same_run(left,r)) {
150            width+=fonts.kerning(left.face(),l,c,left.px);
151        }
152        previous=Some((r,c));
153    }
154    let source=word[0].0;
155    let space_source=pending.filter(|_|line.width>0.0);
156    let space=space_source.map_or(0.0,|r|next_advance(fonts,line,r,' '));
157    let boundary=if let Some(r)=space_source {
158        if same_run(r,source) {fonts.kerning(source.face(),' ',word[0].1,source.px)}else{0.0}
159    } else if line.width>0.0 {
160        next_advance(fonts,line,source,word[0].1)-fonts.advance(source.face(),word[0].1,source.px)
161    } else {0.0};
162    if line.width>0.0 && line.width+space+width+boundary>limit {
163        finish_line(fonts,line,lines,height);
164    } else if let Some(r)=space_source {push_char(fonts,line,r,' ');}
165    for (source,c) in word.drain(..) {
166        if line.width>0.0 && line.width+next_advance(fonts,line,source,c)>limit {
167            finish_line(fonts,line,lines,height);
168        }
169        push_char(fonts,line,source,c);
170    }
171    *pending=None;
172}

174/// Shared measured wrapping for HTML, Markdown, plain text, and table cells.
175pub fn wrap_runs(fonts:&mut TextFonts,runs:&[TextRun],width:f32,height:u32,
176    pre:bool,wrap_pre:bool,align:u8)->Vec<TextLine> {
177    let limit=width.max(1.0);let mut lines=Vec::new();let mut line=empty_line(height);
178    let mut pending=None;let mut word=Vec::new();
179    for run in runs {for c in normalized_chars(&run.span.text) {
180        if c=='\u{2028}' || (pre && c=='\n') {
181            flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
182            finish_line(fonts,&mut line,&mut lines,height);pending=None;
183        } else if pre {
184            // Expand tabs into measured spaces, rather than attempting a tab glyph.
185            let count=if c=='\t' {4}else{1};let c=if c=='\t' {' '}else{c};
186            for _ in 0..count {
187                if wrap_pre && line.width>0.0 && line.width+next_advance(fonts,&line,run,c)>limit {
188                    finish_line(fonts,&mut line,&mut lines,height);
189                }
190                push_char(fonts,&mut line,run,c);
191            }
192        } else if c.is_whitespace() {
193            flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
194            pending.get_or_insert(run);
195        } else {word.push((run,c));}
196    }}
197    flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
198    if !line.runs.is_empty() || (pre && lines.is_empty()) {finish_line(fonts,&mut line,&mut lines,height);}
199    for line in &mut lines {
200        let offset=match align {1=>(width-line.width).max(0.0)*0.5,2=>(width-line.width).max(0.0),_=>0.0};
201        for run in &mut line.runs {run.x+=offset;}
202    }
203    lines
204}
205fn wrap(fonts:&mut TextFonts,context:&TextContext,width:f32)->Vec<TextLine> {
206    wrap_runs(fonts,&context.runs,width,context.line_height,context.pre,context.wrap_pre,context.align)
207}
208fn intrinsic_widths(fonts:&mut TextFonts,context:&TextContext)->(f32,f32) {
209    let mut maximum=0.0f32;let mut longest=0.0f32;
210    let mut width=0.0f32;let mut word=0.0f32;
211    let mut previous:Option<(&TextRun,char)>=None;let mut pending:Option<&TextRun>=None;
212    for run in &context.runs {for c in normalized_chars(&run.span.text) {
213        if c=='\u{2028}' || (context.pre && c=='\n') {
214            maximum=maximum.max(width);longest=longest.max(word);
215            width=0.0;word=0.0;previous=None;pending=None;continue;
216        }
217        if !context.pre && c.is_whitespace() {
218            longest=longest.max(word);word=0.0;pending.get_or_insert(run);continue;
219        }
220        if let Some(space_source)=pending.filter(|_|width>0.0) {
221            width+=fonts.advance(space_source.face(),' ',space_source.px);
222            if let Some((left,l))=previous.filter(|(left,_)|same_run(left,space_source)) {
223                width+=fonts.kerning(left.face(),l,' ',left.px);
224            }
225            previous=Some((space_source,' '));
226        }
227        pending=None;
228        let count=if context.pre && c=='\t' {4}else{1};
229        let c=if context.pre && c=='\t' {' '}else{c};
230        for _ in 0..count {
231            let glyph=fonts.advance(run.face(),c,run.px);
232            let kern=previous.filter(|(left,_)|same_run(left,run))
233                .map_or(0.0,|(left,l)|fonts.kerning(left.face(),l,c,left.px));
234            width+=glyph+kern;
235            if c.is_whitespace() {longest=longest.max(word);word=0.0;}
236            else {
237                word+=glyph+if previous.is_some_and(|(_,l)|!l.is_whitespace()) {kern}else{0.0};
238            }
239            previous=Some((run,c));
240        }
241    }}
242    maximum=maximum.max(width);longest=longest.max(word);
243    (maximum,if context.pre && !context.wrap_pre {maximum}else{longest})
244}
245fn measure(fonts:&mut TextFonts,context:&TextContext,known:Size<Option<f32>>,available:Size<AvailableSpace>)->Size<f32> {
246    let (max_width,longest)=intrinsic_widths(fonts,context);
247    let width=known.width.unwrap_or_else(||match available.width {
248        AvailableSpace::Definite(w)=>w.max(0.0),AvailableSpace::MinContent=>longest,AvailableSpace::MaxContent=>max_width,
249    });
250    let lines=wrap(fonts,context,width);
251    let actual=lines.iter().map(|l|l.width).fold(0.0f32,f32::max);
252    Size{width:known.width.unwrap_or(actual.min(width)),height:known.height.unwrap_or(lines.iter().map(|l|l.height).sum::<u32>() as f32)}
253}

255pub fn paint_lines(fonts:&mut TextFonts,surf:&mut ostd::display::ViSurface,lines:&[TextLine],
256    x:f32,y:f32,color:Color,clip:(i32,i32,i32,i32)) {
257    let mut top=y;
258    for line in lines {
259        for run in &line.runs {
260            let run_top=top+line.baseline-fonts.ascender(run.face(),run.px);
261            crate::draw::text(fonts,surf,x+run.x,run_top,&run.span.text,run.face(),run.px,
262                run.span.color.unwrap_or(color),clip);
263        }
264        top+=line.height as f32;
265    }
266}

268fn pixel(value:f32)->u32 { (value.max(0.0) as i32) as u32 }
269pub fn ceil_pixel(value:f32)->u32 {
270    let whole=value.max(0.0) as u32;
271    whole+u32::from(value>whole as f32)
272}

274pub fn layout(fonts:&mut TextFonts,arena:&DocumentArena,external:&[(NodeId,String)],width:u32,height:u32)->Result<WebLayout,taffy::TaffyError> {
275    let viewport=(width.max(1),height.max(1));let mut sheet=Stylesheet::default();collect_styles(arena,arena.root,external,&mut sheet,viewport);
276    let mut builder=Builder{arena,sheet,viewport,tree:TaffyTree::new(),entries:Vec::new()};
277    let mut root_style=builder.sheet.compute(arena,arena.root,None,viewport);
278    root_style.layout.size.width=length(viewport.0 as f32);
279    let root=builder.element(arena.root,None,root_style)?;
280    builder.tree.compute_layout_with_measure(root,Size{width:AvailableSpace::Definite(viewport.0 as f32),height:AvailableSpace::MaxContent},|inputs,_,context,style| {
281        taffy::compute_leaf_layout(inputs,style,|_,_|0.0,|known,available|context.as_ref().map_or(Size::ZERO,|ctx|measure(fonts,ctx,known,available)))
282    })?;
283    let mut result=WebLayout::default();let mut origins:Vec<(i32,i32)>=Vec::new();
284    for entry in &builder.entries {
285        let l=builder.tree.layout(entry.taffy)?;let (px,py)=entry.parent.map_or((0,0),|p|origins[p]);let x=px.saturating_add(l.location.x as i32);let y=py.saturating_add(l.location.y as i32);origins.push((x,y));
286        if entry.hidden {continue;}
287        let border=[pixel(l.border.left),pixel(l.border.right),pixel(l.border.top),pixel(l.border.bottom)];
288        let (bx,by,bw,bh,lines)=if let Some(text)=&entry.text {(x,y,pixel(l.size.width),pixel(l.size.height),wrap(fonts,text,l.size.width))}else{(x,y,pixel(l.size.width),pixel(l.size.height),Vec::new())};
289        result.height=result.height.max(by.saturating_add(bh as i32));
290        result.boxes.push(WebBox{node_id:entry.node_id,x:bx,y:by,width:bw,height:bh,background:entry.background,border_color:entry.border_color,border,lines});
291    }
292    // White default canvas is a web default, independent of the viewer chrome theme.
293    if let Some(root)=result.boxes.first_mut(){if root.background.is_none(){root.background=Some(Color::rgb(255,255,255));}root.height=root.height.max(height);}
294    Ok(result)
295}

298    pub fn render(&self,fonts:&mut TextFonts,surf:&mut ostd::display::ViSurface,view:(i32,i32,u32,u32),scroll:i32) {
299        let (vx,vy,vw,vh)=view;let clip=(vx,vy,vx.saturating_add(vw as i32),vy.saturating_add(vh as i32));
300        fn rect(surf:&mut ostd::display::ViSurface,x:i32,y:i32,w:u32,h:u32,color:Color,clip:(i32,i32,i32,i32)) {
301            let x0=x.max(clip.0);let y0=y.max(clip.1);let x1=x.saturating_add(w as i32).min(clip.2);let y1=y.saturating_add(h as i32).min(clip.3);
302            if x1>x0&&y1>y0 {crate::draw::fill_rect(surf,x0,y0,(x1-x0) as u32,(y1-y0) as u32,color);}
303        }
304        for b in &self.boxes {
305            let x=vx.saturating_add(b.x);let y=vy.saturating_add(b.y).saturating_sub(scroll);
306            if y.saturating_add(b.height as i32)<=clip.1||y>=clip.3 {continue;}
307            if let Some(bg)=b.background {rect(surf,x,y,b.width,b.height,bg,clip);}
308            let [left,right,top,bottom]=b.border;
309            rect(surf,x,y,left,b.height,b.border_color,clip);
310            rect(surf,x.saturating_add(b.width.saturating_sub(right) as i32),y,right,b.height,b.border_color,clip);
311            rect(surf,x,y,b.width,top,b.border_color,clip);
312            rect(surf,x,y.saturating_add(b.height.saturating_sub(bottom) as i32),b.width,bottom,b.border_color,clip);
313            paint_lines(fonts,surf,&b.lines,x as f32,y as f32,Color::rgb(32,32,32),clip);
314        }
315    }
316    pub fn search(&self,query:&str)->Vec<i32> {
317        let query:String=normalized_chars(query.trim()).map(|c|c.to_ascii_lowercase()).collect();
318        if query.is_empty(){return Vec::new();}
319        let mut result=Vec::new();let mut text=String::new();
320        for b in &self.boxes {for line in &b.lines {text.clear();for run in &line.runs {text.push_str(&run.span.text);}text.make_ascii_lowercase();if text.contains(&query){result.push(b.y);break;}}}result
321    }
322    pub fn hit_link(&self,x:i32,y:i32)->Option<String> {
323        for b in self.boxes.iter().rev() {
324            let mut line_y=b.y;
325            for line in &b.lines {
326                if y>=line_y && y<line_y+line.height as i32 {
327                    for run in &line.runs {
328                        let left=b.x as f32+run.x;
329                        if (x as f32)>=left && (x as f32)<left+run.width {
330                            if let Some(link)=&run.span.link {return Some(link.clone());}
331                        }
332                    }
333                }
334                line_y+=line.height as i32;
335            }
336        }
337        None
338    }
339    pub fn hit_node(&self,x:i32,y:i32)->Option<NodeId> {
340        for b in self.boxes.iter().rev() {
341            let mut line_y=b.y;
342            for line in &b.lines {
343                if y>=line_y&&y<line_y.saturating_add(line.height as i32) {
344                    for run in &line.runs {
345                        let left=b.x as f32+run.x;
346                        if (x as f32)>=left && (x as f32)<left+run.width {return run.node_id;}
347                    }
348                }
349                line_y=line_y.saturating_add(line.height as i32);
350            }
351            if x>=b.x&&x<b.x.saturating_add(b.width as i32)&&y>=b.y&&y<b.y.saturating_add(b.height as i32) {if let Some(id)=b.node_id{return Some(id);}}
352        }
353        None
354    }
