82        self.x.map_or(true,|(a,b)|x>=a&&x<b)&&self.y.map_or(true,|(a,b)|y>=a&&y<b)

85        let x=self.x.map_or((view.0,view.2),|(a,b)|(view.0.saturating_add(a).max(view.0),view.0.saturating_add(b).min(view.2)));
86        let y=self.y.map_or((view.1,view.3),|(a,b)|(view.1.saturating_add(a).saturating_sub(scroll).max(view.1),view.1.saturating_add(b).saturating_sub(scroll).min(view.3)));
87        (x.0,y.0,x.1,y.1)
88    }
89    fn scale(&mut self,scale:f32){
90        for axis in [&mut self.x,&mut self.y]{if let Some((a,b))=axis{*a=(*a as f32*scale) as i32;*b=(*b as f32*scale) as i32;}}

134impl LayoutView<'_> {
135    fn inline_layout(&mut self,id:taffy::NodeId,inputs:LayoutInput)->LayoutOutput {
136        use taffy::util::{MaybeResolve,ResolveOrZero};
137        let style=self.tree.nodes[usize::from(id)].style.clone();
138        let calcs=self.tree.calcs.clone();
139        let strut=self.tree.nodes[usize::from(id)].intrinsic;
140        let mut first_baseline=None;
141        let mut output=taffy::compute_leaf_layout(inputs,&style,|h,b|super::css::resolve_calc(&calcs,h,b),|known,available|{
142            let limit=known.width.or(available.width.into_option()).unwrap_or(match available.width{AvailableSpace::MinContent=>0.0,_=>f32::MAX});
143            let basis=Size{width:known.width.or(inputs.parent_size.width),height:inputs.parent_size.height};
144            let mut x=0.0f32;let mut y=0.0f32;let mut width=0.0f32;
145            let baseline=self.fonts.ascender(FontFace::UiRegular,strut.width)+(strut.height-self.fonts.line_height(FontFace::UiRegular,strut.width)).max(0.0)*0.5;
146            let mut ascent=baseline;let mut descent=(strut.height-baseline).max(0.0);
147            let mut row:Vec<(taffy::NodeId,Layout,f32)>=Vec::new();
148            let count=self.tree.nodes[usize::from(id)].children.len();
149            for index in 0..count {
150                let child=self.tree.nodes[usize::from(id)].children[index];
151                let node=&self.tree.nodes[usize::from(child)];
152                let child_style=node.style.clone();let gap=node.gap;let force_break=node.break_before;
153                let margin=child_style.margin.resolve_or_zero(basis.width,|h,b|super::css::resolve_calc(&calcs,h,b));
154                let child_inputs=LayoutInput{
155                    parent_size:basis,known_dimensions:Size::NONE,known_dimensions_are_definite:Size{width:true,height:true},
156                    available_space:Size{width:if node.context.is_some(){AvailableSpace::MaxContent}else{AvailableSpace::Definite(limit)},height:AvailableSpace::MaxContent},
157                    vertical_margins_are_collapsible:Line::FALSE,..inputs
158                };
159                let child_output=self.child_layout(child,child_inputs,None);
160                let size=child_output.size;
161                let gap=if x>0.0{gap.map_or(0.0,|(face,px)|self.fonts.advance(face,' ',px))}else{0.0};
162                if !row.is_empty()&&(force_break||x+gap+margin.left+size.width+margin.right>limit) {
163                    first_baseline.get_or_insert(y+ascent);
164                    for (child,mut layout,base) in row.drain(..){layout.location.y+=y+ascent-base;self.set_unrounded_layout(child,&layout);}
165                    width=width.max(x);y+=ascent+descent;x=0.0;ascent=baseline;descent=(strut.height-baseline).max(0.0);
166                }
167                let gap=if x>0.0{gap}else{0.0};
168                let node=&self.tree.nodes[usize::from(child)];
169                let child_baseline=if let Some(context)=&node.context {
170                    context.runs.iter().map(|run|self.fonts.ascender(run.face(),run.px)+(run.line_height-self.fonts.line_height(run.face(),run.px)).max(0.0)*0.5).fold(0.0,f32::max)
171                }else{child_output.baselines.first.unwrap_or(size.height)};
172                let inset=Rect{
173                    left:child_style.inset.left.maybe_resolve(basis.width,|h,b|super::css::resolve_calc(&calcs,h,b)),
174                    right:child_style.inset.right.maybe_resolve(basis.width,|h,b|super::css::resolve_calc(&calcs,h,b)),
175                    top:child_style.inset.top.maybe_resolve(basis.height,|h,b|super::css::resolve_calc(&calcs,h,b)),
176                    bottom:child_style.inset.bottom.maybe_resolve(basis.height,|h,b|super::css::resolve_calc(&calcs,h,b)),
177                };
178                let offset=Point{x:inset.left.or(inset.right.map(|v|-v)).unwrap_or(0.0),y:inset.top.or(inset.bottom.map(|v|-v)).unwrap_or(0.0)};
179                let padding=child_style.padding.resolve_or_zero(basis.width,|h,b|super::css::resolve_calc(&calcs,h,b));
180                let border=child_style.border.resolve_or_zero(basis.width,|h,b|super::css::resolve_calc(&calcs,h,b));
181                let layout=Layout{size,location:Point{x:x+gap+margin.left+offset.x,y:margin.top+offset.y},padding,border,margin,..Layout::default()};
182                ascent=ascent.max(margin.top+child_baseline);
183                descent=descent.max(margin.bottom+size.height-child_baseline);
184                row.push((child,layout,margin.top+child_baseline));
185                x+=gap+margin.left+size.width+margin.right;
186            }
187            if !row.is_empty() {
188                first_baseline.get_or_insert(y+ascent);
189                for (child,mut layout,base) in row.drain(..){layout.location.y+=y+ascent-base;self.set_unrounded_layout(child,&layout);}
190                width=width.max(x);y+=ascent+descent;
191            }
192            Size{width,height:y}
193        });
194        output.baselines=taffy::tree::Baselines{first:first_baseline,last:first_baseline};output
195    }
196    fn child_layout(&mut self,id:taffy::NodeId,inputs:LayoutInput,block:Option<&mut taffy::BlockContext<'_>>)->LayoutOutput {
197        if inputs.run_mode==RunMode::PerformHiddenLayout{return taffy::compute_hidden_layout(self,id);}
198        taffy::compute_cached_layout(self,id,inputs,|tree,id,inputs|{
199            if tree.tree.nodes[usize::from(id)].inline_flow{return tree.inline_layout(id,inputs);}
200            let node=&tree.tree.nodes[usize::from(id)];
201            match (node.style.display,!node.children.is_empty()) {
202                (Display::None,_)=>taffy::compute_hidden_layout(tree,id),
203                (Display::Block,true)=>taffy::compute_block_layout(tree,id,inputs,block),
204                (Display::FlowRoot,true)=>taffy::compute_block_layout(tree,id,inputs,None),
205                (Display::Flex,true)=>taffy::compute_flexbox_layout(tree,id,inputs),
206                (Display::Grid,true)=>taffy::compute_grid_layout(tree,id,inputs),
207                _=>taffy::compute_leaf_layout(inputs,&node.style,|handle,basis|super::css::resolve_calc(&tree.tree.calcs,handle,basis),|known,available|{
208                    node.context.as_ref().map_or(node.intrinsic,|ctx|measure(tree.fonts,ctx,known,available))
209                }),
210            }
211        })
212    }

252impl<'a> Builder<'a> {
253    fn inline_groups(&mut self,children:Vec<taffy::NodeId>)->Result<Vec<taffy::NodeId>,taffy::TaffyError>{
254        let mut result=Vec::new();let mut group=Vec::new();let mut atomic=false;
255        for child in children {
256            let entry=&self.entries[self.tree.nodes[usize::from(child)].entry.unwrap()];
257            let inline=entry.inline&&!entry.position.out_of_flow();
258            if !inline {
259                self.finish_inline_group(&mut result,&mut group,atomic)?;atomic=false;
260                result.push(child);
261            } else {atomic|=entry.atomic_inline||entry.text.is_none();group.push(child);}
262        }
263        self.finish_inline_group(&mut result,&mut group,atomic)?;Ok(result)
264    }
265    fn finish_inline_group(&mut self,result:&mut Vec<taffy::NodeId>,group:&mut Vec<taffy::NodeId>,atomic:bool)->Result<(),taffy::TaffyError>{
266        if !atomic{result.append(group);return Ok(());}
267        let tid=self.tree.new_leaf(Style{display:Display::Block,..Style::default()})?;
268        self.tree.nodes[usize::from(tid)].inline_flow=true;
269        let mut children=Vec::new();let mut pending=None;let mut force_break=false;
270        for child in group.drain(..) {
271            let index=self.tree.nodes[usize::from(child)].entry.unwrap();
272            let original=self.entries[index].clone();
273            let Some(context)=original.text.clone() else {
274                self.tree.nodes[usize::from(child)].gap=pending.take();
275                self.tree.nodes[usize::from(child)].break_before=core::mem::take(&mut force_break);
276                children.push(child);continue;
277            };
278            self.entries[index].hidden=true;
279            let mut words:Vec<(Vec<TextRun>,Option<(FontFace,f32)>,bool)>=Vec::new();
280            let mut word=Vec::new();
281            for run in &context.runs {
282                let mut segment=String::new();
283                for c in normalized_chars(&run.span.text) {
284                    if c.is_whitespace()||c=='\u{2028}' {
285                        if !segment.is_empty(){
286                            let span=StyledSpan{text:core::mem::take(&mut segment),bold:run.span.bold,italic:run.span.italic,code:run.span.code,link:run.span.link.clone(),color:run.span.color};
287                            append_run(&mut word,TextRun::new(span,run.px,run.node_id,run.line_height));
288                        }
289                        if !word.is_empty(){words.push((core::mem::take(&mut word),pending.take(),core::mem::take(&mut force_break)));}
290                        if c=='\u{2028}'||(context.pre&&c=='\n'){force_break=true;pending=None;}
291                        else{pending.get_or_insert((run.face(),run.px));}
292                    }else{segment.push(c);}
293                }
294                if !segment.is_empty(){
295                    let span=StyledSpan{text:segment,bold:run.span.bold,italic:run.span.italic,code:run.span.code,link:run.span.link.clone(),color:run.span.color};
296                    append_run(&mut word,TextRun::new(span,run.px,run.node_id,run.line_height));
297                }
298            }
299            if !word.is_empty(){words.push((word,pending.take(),core::mem::take(&mut force_break)));}
300            for (runs,gap,break_before) in words {
301                let context=Rc::new(TextContext{runs,line_height:context.line_height,pre:true,wrap_pre:false,align:0});
302                let child=self.tree.new_leaf_with_context(Style{display:Display::Block,..Style::default()},context.clone())?;
303                let node=&mut self.tree.nodes[usize::from(child)];node.gap=gap;node.break_before=break_before;node.entry=Some(self.entries.len());
304                let mut entry=original.clone();entry.taffy=child;entry.text=Some(context);self.entries.push(entry);
305                children.push(child);
306            }
307            self.tree.nodes[usize::from(tid)].intrinsic=Size{width:original.font_size,height:original.line_height};
308        }
309        if self.tree.nodes[usize::from(tid)].intrinsic.height==0.0 {
310            if let Some(first)=children.first().and_then(|id|self.tree.nodes[usize::from(*id)].entry).map(|index|&self.entries[index]){
311                self.tree.nodes[usize::from(tid)].intrinsic=Size{width:first.font_size,height:first.line_height};
312            }
313        }
314        self.tree.set_children(tid,&children)?;result.push(tid);Ok(())
315    }

323    fn element(&mut self,id:NodeId,parent:Option<usize>,mut style:ComputedStyle,link:Option<&str>)->Result<taffy::NodeId,taffy::TaffyError> {
324        let link=self.arena.get(id).and_then(|n|n.get_attribute("href")).or(link);

429pub fn append_run(runs:&mut Vec<TextRun>,run:TextRun) {
430    if let Some(last)=runs.last_mut().filter(|last|same_run(last,&run)) {

434fn next_advance(fonts:&mut TextFonts,line:&TextLine,source:&TextRun,c:char)->f32 {
435    let kern=line.runs.last().filter(|r|same_run(r,source))
436        .and_then(|r|r.span.text.chars().next_back())
437        .map_or(0.0,|left|fonts.kerning(source.face(),left,c,source.px));
438    fonts.advance(source.face(),c,source.px)+kern
439}
440fn push_char(fonts:&mut TextFonts,line:&mut TextLine,source:&TextRun,c:char) {
441    let advance=next_advance(fonts,line,source,c);
442    line.baseline=line.baseline.max(fonts.ascender(source.face(),source.px));
443    line.height=line.height.max(ceil_pixel(source.line_height.max(fonts.line_height(source.face(),source.px))));
444    if let Some(last)=line.runs.last_mut().filter(|r|same_run(r,source)) {
445        last.span.text.push(c);last.width+=advance;
446    } else {
447        let mut span=StyledSpan::plain("");
448        span.bold=source.span.bold;span.italic=source.span.italic;span.code=source.span.code;
449        span.link=source.span.link.clone();span.color=source.span.color;span.text.push(c);
450        let mut run=TextRun::with_visible(span,source.px,source.node_id,source.line_height,source.visible);
451        run.x=line.width;run.width=advance;line.runs.push(run);
452    }
453    line.width+=advance;
454}
455fn finish_line(fonts:&mut TextFonts,line:&mut TextLine,lines:&mut Vec<TextLine>,height:u32) {
456    // The shared baseline also covers mixed-size ascenders and descenders.
457    let descent=line.runs.iter().map(|r|
458        fonts.line_height(r.face(),r.px)-fonts.ascender(r.face(),r.px))
459        .fold(0.0f32,f32::max);
460    line.height=line.height.max(ceil_pixel(line.baseline+descent));
461    lines.push(core::mem::replace(line,empty_line(height)));
462}
463fn flush_word<'a>(fonts:&mut TextFonts,word:&mut Vec<(&'a TextRun,char)>,line:&mut TextLine,
464    lines:&mut Vec<TextLine>,pending:&mut Option<&'a TextRun>,limit:f32,height:u32) {
465    if word.is_empty(){return;}
466    let mut width=0.0;let mut previous:Option<(&TextRun,char)>=None;
467    for &(r,c) in word.iter() {
468        width+=fonts.advance(r.face(),c,r.px);
469        if let Some((left,l))=previous.filter(|(left,_)|same_run(left,r)) {
470            width+=fonts.kerning(left.face(),l,c,left.px);
471        }
472        previous=Some((r,c));
473    }
474    let source=word[0].0;
475    let space_source=pending.filter(|_|line.width>0.0);
476    let space=space_source.map_or(0.0,|r|next_advance(fonts,line,r,' '));
477    let boundary=if let Some(r)=space_source {
478        if same_run(r,source) {fonts.kerning(source.face(),' ',word[0].1,source.px)}else{0.0}
479    } else if line.width>0.0 {
480        next_advance(fonts,line,source,word[0].1)-fonts.advance(source.face(),word[0].1,source.px)
481    } else {0.0};
482    if line.width>0.0 && line.width+space+width+boundary>limit {
483        finish_line(fonts,line,lines,height);
484    } else if let Some(r)=space_source {push_char(fonts,line,r,' ');}
485    for (source,c) in word.drain(..) {
486        if line.width>0.0 && line.width+next_advance(fonts,line,source,c)>limit {
487            finish_line(fonts,line,lines,height);
488        }
489        push_char(fonts,line,source,c);
490    }
491    *pending=None;
492}

494/// Shared measured wrapping for HTML, Markdown, plain text, and table cells.
495pub fn wrap_runs(fonts:&mut TextFonts,runs:&[TextRun],width:f32,height:u32,
496    pre:bool,wrap_pre:bool,align:u8)->Vec<TextLine> {
497    let limit=width.max(1.0);let mut lines=Vec::new();let mut line=empty_line(height);
498    let mut pending=None;let mut word=Vec::new();
499    for run in runs {for c in normalized_chars(&run.span.text) {
500        if c=='\u{2028}' || (pre && c=='\n') {
501            flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
502            finish_line(fonts,&mut line,&mut lines,height);pending=None;
503        } else if pre {
504            // Expand tabs into measured spaces, rather than attempting a tab glyph.
505            let count=if c=='\t' {4}else{1};let c=if c=='\t' {' '}else{c};
506            for _ in 0..count {
507                if wrap_pre && line.width>0.0 && line.width+next_advance(fonts,&line,run,c)>limit {
508                    finish_line(fonts,&mut line,&mut lines,height);
509                }
510                push_char(fonts,&mut line,run,c);
511            }
512        } else if c.is_whitespace() {
513            flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
514            pending.get_or_insert(run);
515        } else {word.push((run,c));}
516    }}
517    flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
518    if !line.runs.is_empty() || (pre && lines.is_empty()) {finish_line(fonts,&mut line,&mut lines,height);}
519    for line in &mut lines {
520        let offset=match align {1=>(width-line.width).max(0.0)*0.5,2=>(width-line.width).max(0.0),_=>0.0};
521        for run in &mut line.runs {run.x+=offset;}
522    }
523    lines
524}
525fn wrap(fonts:&mut TextFonts,context:&TextContext,width:f32)->Vec<TextLine> {
526    wrap_runs(fonts,&context.runs,width,context.line_height,context.pre,context.wrap_pre,context.align)
527}
528fn intrinsic_widths(fonts:&mut TextFonts,context:&TextContext)->(f32,f32) {
529    let mut maximum=0.0f32;let mut longest=0.0f32;
530    let mut width=0.0f32;let mut word=0.0f32;
531    let mut previous:Option<(&TextRun,char)>=None;let mut pending:Option<&TextRun>=None;
532    for run in &context.runs {for c in normalized_chars(&run.span.text) {
533        if c=='\u{2028}' || (context.pre && c=='\n') {
534            maximum=maximum.max(width);longest=longest.max(word);
535            width=0.0;word=0.0;previous=None;pending=None;continue;
536        }
537        if !context.pre && c.is_whitespace() {
538            longest=longest.max(word);word=0.0;pending.get_or_insert(run);continue;
539        }
540        if let Some(space_source)=pending.filter(|_|width>0.0) {
541            width+=fonts.advance(space_source.face(),' ',space_source.px);
542            if let Some((left,l))=previous.filter(|(left,_)|same_run(left,space_source)) {
543                width+=fonts.kerning(left.face(),l,' ',left.px);
544            }
545            previous=Some((space_source,' '));
546        }
547        pending=None;
548        let count=if context.pre && c=='\t' {4}else{1};
549        let c=if context.pre && c=='\t' {' '}else{c};
550        for _ in 0..count {
551            let glyph=fonts.advance(run.face(),c,run.px);
552            let kern=previous.filter(|(left,_)|same_run(left,run))
553                .map_or(0.0,|(left,l)|fonts.kerning(left.face(),l,c,left.px));
554            width+=glyph+kern;
555            if c.is_whitespace() {longest=longest.max(word);word=0.0;}
556            else {
557                word+=glyph+if previous.is_some_and(|(_,l)|!l.is_whitespace()) {kern}else{0.0};
558            }
559            previous=Some((run,c));
560        }
561    }}
562    maximum=maximum.max(width);longest=longest.max(word);
563    (maximum,if context.pre && !context.wrap_pre {maximum}else{longest})
564}
565fn measure(fonts:&mut TextFonts,context:&TextContext,known:Size<Option<f32>>,available:Size<AvailableSpace>)->Size<f32> {
566    let (max_width,longest)=intrinsic_widths(fonts,context);
567    let width=known.width.unwrap_or_else(||match available.width {
568        AvailableSpace::Definite(w)=>w.max(0.0),AvailableSpace::MinContent=>longest,AvailableSpace::MaxContent=>max_width,
569    });
570    let lines=wrap(fonts,context,width);
571    let actual=lines.iter().map(|l|l.width).fold(0.0f32,f32::max);
572    Size{width:known.width.unwrap_or(actual.min(width)),height:known.height.unwrap_or(lines.iter().map(|l|l.height).sum::<u32>() as f32)}
573}

698            }

711        }
712        descendant_clips.push(children_clip);
713        if entry.hidden {boxes.push(None);continue;}
714        let border=[pixel(l.border.left),pixel(l.border.right),pixel(l.border.top),pixel(l.border.bottom)];
715        let bw=pixel(l.size.width);let bh=pixel(l.size.height);
716        let lines=entry.text.as_ref().map_or_else(Vec::new,|text|wrap(fonts,text,l.size.width));
717        if !fixed {result.height=result.height.max(clip.y.map_or(y.saturating_add(bh as i32),|(_,end)|end.min(y.saturating_add(bh as i32))));}
718        let node_style=&builder.tree.nodes[usize::from(entry.taffy)].style;
719        boxes.push(Some(WebBox{node_id:entry.node_id,x,y,width:bw,height:bh,background:entry.background,border_color:entry.border_color,border,lines,input_data:entry.input_data.clone(),image:entry.image.clone(),highlights:Vec::new(),control_px:14.0,fixed,clip,font_size:entry.font_size,line_height:entry.line_height,color:entry.color,display:node_style.display,overflow,padding:[pixel(l.padding.left),pixel(l.padding.right),pixel(l.padding.top),pixel(l.padding.bottom)],box_sizing:node_style.box_sizing,position:entry.position,inline:entry.inline,atomic_inline:entry.atomic_inline,visible:entry.visible}));
720    }
721    for index in stacking_order(&builder.entries,&builder.tree) {
722        if let Some(b)=boxes[index].take(){result.boxes.push(b);}
723    }
724    // White default canvas is a web default, independent of the viewer chrome theme.
725    for b in &mut result.boxes {
726        if b.node_id == Some(arena.root) {
727            if b.background.is_none() { b.background = Some(Color::rgb(255,255,255)); }
728            b.height = b.height.max(height);
729            break;
730        }
731    }
732    Ok(result)
733}

735/// Sort only the children of each stacking context, then emit each context as
736/// an indivisible unit. A descendant's high z-index cannot escape its context.
737fn stacking_order(entries:&[Entry],tree:&WebTree)->Vec<usize> {
738    let mut contexts=Vec::with_capacity(entries.len());let mut owners=Vec::with_capacity(entries.len());
739    let mut positioned=Vec::with_capacity(entries.len());
740    let mut children:Vec<Vec<(i8,i32,usize,usize)>>=(0..entries.len()).map(|_|Vec::new()).collect();
741    for (index,entry) in entries.iter().enumerate() {
742        let item=entry.parent.is_some_and(|p|matches!(tree.nodes[usize::from(entries[p].taffy)].style.display,Display::Flex|Display::Grid));
743        let context=index==0||entry.position==CssPosition::Fixed||(entry.z_index.is_some()&&(entry.position!=CssPosition::Static||item));
744        contexts.push(context);
745        let owner=entry.parent.map_or(0,|p|if contexts[p]{p}else{owners[p]});owners.push(owner);
746        let raised=entry.position!=CssPosition::Static||entry.parent.is_some_and(|p|!contexts[p]&&positioned[p]);positioned.push(raised);
747        if index==0{continue;}
748        let z=if context{entry.z_index.unwrap_or(0)}else{0};
749        let phase=if context&&z<0{-1}else if context&&z>0{2}else if context||raised{1}else{0};
750        children[owner].push((phase,z,entry.paint_order,index));
751    }
752    for group in &mut children{group.sort_unstable();}
753    fn emit(index:usize,contexts:&[bool],children:&[Vec<(i8,i32,usize,usize)>],out:&mut Vec<usize>){
754        out.push(index);
755        for &(_,_,_,child) in &children[index] {
756            if contexts[child]{emit(child,contexts,children,out);}else{out.push(child);}
757        }
758    }
759    let mut out=Vec::with_capacity(entries.len());if !entries.is_empty(){emit(0,&contexts,&children,&mut out);}out
760}

896    /// A committed native layout snapshot. Coordinates are viewport-relative,
897    /// including fixed positioning; inline boxes use their measured run union.
898    pub fn browser_layout(&self, arena: &DocumentArena, scroll: i32) -> Vec<dom_arena::BrowserNodeLayout> {
899        use alloc::{collections::BTreeMap, format};
900        use dom_arena::BrowserNodeLayout;
901        fn styles(px: f32, line_height: f32, color: Color, display: &str) -> Vec<(String, String)> {
902            alloc::vec![
903                ("font-size".into(), format!("{px}px")),
904                ("line-height".into(), format!("{line_height}px")),
905                ("color".into(), format!("rgba({},{},{},{})", color.r(), color.g(), color.b(), color.a() as f32 / 255.0)),
906                ("display".into(), display.into()),
907            ]
908        }
909        fn overflow(value: taffy::style::Overflow) -> &'static str {
910            match value {
911                taffy::style::Overflow::Visible => "visible",
912                taffy::style::Overflow::Hidden => "hidden",
913                taffy::style::Overflow::Scroll => "scroll",
914                taffy::style::Overflow::Clip => "clip",
915            }
916        }
917        let mut layout = BTreeMap::<NodeId, BrowserNodeLayout>::new();
918        for b in &self.boxes {
919            if let Some(node) = b.node_id {
920                let display = if b.atomic_inline { "inline-block" } else if b.inline { "inline" } else { match b.display {
921                    Display::Block => "block", Display::FlowRoot => "flow-root", Display::Flex => "flex",
922                    Display::Grid => "grid", Display::None => "none",
923                }};
924                let mut computed = styles(b.font_size, b.line_height, b.color, display);
925                computed.push(("overflow-x".into(), overflow(b.overflow.x).into()));
926                computed.push(("overflow-y".into(), overflow(b.overflow.y).into()));
927                computed.push(("position".into(), match b.position { CssPosition::Static=>"static",CssPosition::Relative=>"relative",CssPosition::Absolute=>"absolute",CssPosition::Fixed=>"fixed" }.into()));
928                computed.push(("box-sizing".into(), if b.box_sizing == BoxSizing::BorderBox {"border-box"} else {"content-box"}.into()));
929                let content_width = b.width.saturating_sub(b.border[0]+b.border[1]+b.padding[0]+b.padding[1]);
930                let content_height = b.height.saturating_sub(b.border[2]+b.border[3]+b.padding[2]+b.padding[3]);
931                computed.push(("width".into(), if b.inline && !b.atomic_inline { "auto".into() } else { format!("{}px", if b.box_sizing == BoxSizing::BorderBox {b.width} else {content_width}) }));
932                computed.push(("height".into(), if b.inline && !b.atomic_inline { "auto".into() } else { format!("{}px", if b.box_sizing == BoxSizing::BorderBox {b.height} else {content_height}) }));
933                for (index,side) in ["left","right","top","bottom"].into_iter().enumerate() {
934                    computed.push((format!("padding-{side}"),format!("{}px",b.padding[index])));
935                    computed.push((format!("border-{side}-width"),format!("{}px",b.border[index])));
936                }
937                layout.insert(node, BrowserNodeLayout {
938                    node, x: b.x as f32, y: b.y.saturating_sub(if b.fixed { 0 } else { scroll }) as f32,
939                    width: b.width as f32, height: b.height as f32, styles: computed,
940                });
941            }
942        }
943        let elements: alloc::collections::BTreeSet<NodeId> = layout.keys().copied().collect();
944        for b in &self.boxes {
945            let mut y = b.y.saturating_sub(if b.fixed { 0 } else { scroll }) as f32;
946            for line in &b.lines {
947                for run in &line.runs {
948                    let Some(node) = run.node_id else { continue; };
949                    if elements.contains(&node) { continue; }
950                    let x = b.x as f32 + run.x;
951                    let right = x + run.width;
952                    let bottom = y + line.height as f32;
953                    let mut current = Some(node);
954                    while let Some(id) = current {
955                        if elements.contains(&id) { break; }
956                        layout.entry(id).and_modify(|old| {
957                            let right = (old.x + old.width).max(right);
958                            let bottom = (old.y + old.height).max(bottom);
959                            old.x = old.x.min(x); old.y = old.y.min(y);
960                            old.width = right - old.x; old.height = bottom - old.y;
961                        }).or_insert_with(|| BrowserNodeLayout {
962                            node: id, x, y, width: run.width, height: line.height as f32,
963                            styles: if id == node {
964                                let mut computed = styles(run.px, run.line_height, run.span.color.unwrap_or(b.color), "inline");
965                                computed.push(("width".into(),"auto".into())); computed.push(("height".into(),"auto".into()));
966                                computed
967                            } else { Vec::new() },
968                        });
969                        current = arena.get(id).and_then(|node| node.parent);
970                    }
971                }
972                y += line.height as f32;
973            }
974        }
975        layout.into_values().collect()
976    }

979    /// or disconnection means the queried element has no generated box.
980    pub fn browser_node_layout(&self, arena: &DocumentArena, external: &[(NodeId,String)], node: NodeId,
981        width: u32, height: u32, scroll: i32) -> Result<dom_arena::BrowserNodeLayout,String> {
982        use alloc::format;
983        if !arena.get(node).is_some_and(|node| node.is_element()) { return Err("CSSOM requires an element".into()); }
984        let mut sheet = Stylesheet::default();
985        collect_styles(arena, arena.root, external, &mut sheet, (width,height));
986        let mut ancestors = Vec::new();
987        let mut cursor = Some(node);
988        while let Some(id) = cursor {
989            if ancestors.len() > crate::budget::MAX_DOM_DEPTH { return Err("CSSOM ancestry exceeds depth budget".into()); }
990            ancestors.push(id);
991            cursor = arena.get(id).and_then(|node| node.parent);
992        }
993        let mut parent = None;
994        for id in ancestors.into_iter().rev() {
995            parent = Some(sheet.compute(arena,id,parent.as_ref(),(width,height)));
996        }
997        let style = parent.ok_or("CSSOM style unavailable")?;
998        let mut result = self.browser_layout(arena,scroll).into_iter().find(|entry| entry.node == node)
999            .unwrap_or(dom_arena::BrowserNodeLayout { node, x:0.0,y:0.0,width:0.0,height:0.0,styles:Vec::new() });
1000        let display = if style.layout.display == Display::None { "none" } else if style.atomic_inline { "inline-block" } else if style.inline { "inline" } else {
1001            match style.layout.display { Display::Block => "block", Display::FlowRoot => "flow-root", Display::Flex => "flex", Display::Grid => "grid", Display::None => "none" }
1002        };
1003        let mut computed = alloc::vec![
1004            ("display".into(),display.into()),
1005            ("font-size".into(),format!("{}px",style.font_size)),
1006            ("line-height".into(),format!("{}px",style.line_height)),
1007            ("color".into(),format!("rgba({},{},{},{})",style.color.r(),style.color.g(),style.color.b(),style.color.a() as f32/255.0)),
1008            ("box-sizing".into(),if style.layout.box_sizing == BoxSizing::BorderBox { "border-box".into() } else { "content-box".into() }),
1009            ("position".into(),match style.position { CssPosition::Static=>"static",CssPosition::Relative=>"relative",CssPosition::Absolute=>"absolute",CssPosition::Fixed=>"fixed" }.into()),
1010        ];
1011        for (key,value) in core::mem::take(&mut result.styles) {
1012            if !computed.iter().any(|(name,_)| name == &key) { computed.push((key,value)); }
1013        }
1014        result.styles = computed;
1015        Ok(result)
1016    }
