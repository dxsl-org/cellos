139impl BoxClip {
140    fn intersect_axis(old: Option<(i32, i32)>, new: (i32, i32)) -> Option<(i32, i32)> {
141        Some(old.map_or(new, |(a, b)| (a.max(new.0), b.min(new.1))))
142    }
143    fn contains(self, x: i32, y: i32) -> bool {
144        self.x.is_none_or(|(a, b)| x >= a && x < b) && self.y.is_none_or(|(a, b)| y >= a && y < b)
145    }
146    fn screen(self, view: (i32, i32, i32, i32), scroll: i32) -> (i32, i32, i32, i32) {
147        let x = self.x.map_or((view.0, view.2), |(a, b)| {
148            (
149                view.0.saturating_add(a).max(view.0),
150                view.0.saturating_add(b).min(view.2),
151            )
152        });
153        let y = self.y.map_or((view.1, view.3), |(a, b)| {
154            (
155                view.1.saturating_add(a).saturating_sub(scroll).max(view.1),
156                view.1.saturating_add(b).saturating_sub(scroll).min(view.3),
157            )
158        });
159        (x.0, y.0, x.1, y.1)
160    }
161    fn scale(&mut self, scale: f32) {
162        for (a, b) in [&mut self.x, &mut self.y].into_iter().flatten() {
163            *a = (*a as f32 * scale) as i32;
164            *b = (*b as f32 * scale) as i32;
165        }
166    }

238        self.nodes[usize::from(id)].style = style;

242        &mut self,

244        children: &[taffy::NodeId],
245    ) -> Result<(), taffy::TaffyError> {
246        for child in children {
247            self.nodes[usize::from(*child)].parent = Some(id);
248        }

250        Ok(())
251    }

253        Ok(&self.nodes[usize::from(id)].rounded)

287impl LayoutView<'_> {
288    fn inline_layout(&mut self, id: taffy::NodeId, inputs: LayoutInput) -> LayoutOutput {
289        use taffy::util::{MaybeResolve, ResolveOrZero};
290        let style = self.tree.nodes[usize::from(id)].style.clone();
291        let calcs = self.tree.calcs.clone();
292        let strut = self.tree.nodes[usize::from(id)].intrinsic;
293        let mut first_baseline = None;
294        let mut output = taffy::compute_leaf_layout(
295            inputs,
296            &style,
297            |h, b| super::css::resolve_calc(&calcs, h, b),
298            |known, available| {
299                let limit = known.width.or(available.width.into_option()).unwrap_or(
300                    match available.width {
301                        AvailableSpace::MinContent => 0.0,
302                        _ => f32::MAX,
303                    },
304                );
305                let basis = Size {
306                    width: known.width.or(inputs.parent_size.width),
307                    height: inputs.parent_size.height,
308                };
309                let mut x = 0.0f32;
310                let mut y = 0.0f32;
311                let mut width = 0.0f32;
312                let baseline = self.fonts.ascender(FontFace::UiRegular, strut.width)
313                    + (strut.height - self.fonts.line_height(FontFace::UiRegular, strut.width))
314                        .max(0.0)
315                        * 0.5;
316                let mut ascent = baseline;
317                let mut descent = (strut.height - baseline).max(0.0);
318                let mut row: Vec<(taffy::NodeId, Layout, f32)> = Vec::new();
319                let count = self.tree.nodes[usize::from(id)].children.len();
320                for index in 0..count {
321                    let child = self.tree.nodes[usize::from(id)].children[index];
322                    let node = &self.tree.nodes[usize::from(child)];
323                    let child_style = node.style.clone();
324                    let gap = node.gap;
325                    let force_break = node.break_before;
326                    let margin = child_style.margin.resolve_or_zero(basis.width, |h, b| {
327                        super::css::resolve_calc(&calcs, h, b)
328                    });
329                    let child_inputs = LayoutInput {
330                        parent_size: basis,
331                        known_dimensions: Size::NONE,
332                        known_dimensions_are_definite: Size {
333                            width: true,
334                            height: true,
335                        },
336                        available_space: Size {
337                            width: if node.context.is_some() {
338                                AvailableSpace::MaxContent
339                            } else {
340                                AvailableSpace::Definite(limit)
341                            },
342                            height: AvailableSpace::MaxContent,
343                        },
344                        vertical_margins_are_collapsible: Line::FALSE,
345                        ..inputs
346                    };
347                    let child_output = self.child_layout(child, child_inputs, None);
348                    let size = child_output.size;
349                    let gap = if x > 0.0 {
350                        gap.map_or(0.0, |(face, px)| self.fonts.advance(face, ' ', px))
351                    } else {
352                        0.0
353                    };
354                    if !row.is_empty()
355                        && (force_break
356                            || x + gap + margin.left + size.width + margin.right > limit)
357                    {
358                        first_baseline.get_or_insert(y + ascent);
359                        for (child, mut layout, base) in row.drain(..) {
360                            layout.location.y += y + ascent - base;
361                            self.set_unrounded_layout(child, &layout);
362                        }
363                        width = width.max(x);
364                        y += ascent + descent;
365                        x = 0.0;
366                        ascent = baseline;
367                        descent = (strut.height - baseline).max(0.0);
368                    }
369                    let gap = if x > 0.0 { gap } else { 0.0 };
370                    let node = &self.tree.nodes[usize::from(child)];
371                    let child_baseline = if let Some(context) = &node.context {
372                        context
373                            .runs
374                            .iter()
375                            .map(|run| {
376                                self.fonts.ascender(run.face(), run.px)
377                                    + (run.line_height - self.fonts.line_height(run.face(), run.px))
378                                        .max(0.0)
379                                        * 0.5
380                            })
381                            .fold(0.0, f32::max)
382                    } else {
383                        child_output.baselines.first.unwrap_or(size.height)
384                    };
385                    let inset = Rect {
386                        left: child_style.inset.left.maybe_resolve(basis.width, |h, b| {
387                            super::css::resolve_calc(&calcs, h, b)
388                        }),
389                        right: child_style.inset.right.maybe_resolve(basis.width, |h, b| {
390                            super::css::resolve_calc(&calcs, h, b)
391                        }),
392                        top: child_style.inset.top.maybe_resolve(basis.height, |h, b| {
393                            super::css::resolve_calc(&calcs, h, b)
394                        }),
395                        bottom: child_style
396                            .inset
397                            .bottom
398                            .maybe_resolve(basis.height, |h, b| {
399                                super::css::resolve_calc(&calcs, h, b)
400                            }),
401                    };
402                    let offset = Point {
403                        x: inset.left.or(inset.right.map(|v| -v)).unwrap_or(0.0),
404                        y: inset.top.or(inset.bottom.map(|v| -v)).unwrap_or(0.0),
405                    };
406                    let padding = child_style.padding.resolve_or_zero(basis.width, |h, b| {
407                        super::css::resolve_calc(&calcs, h, b)
408                    });
409                    let border = child_style.border.resolve_or_zero(basis.width, |h, b| {
410                        super::css::resolve_calc(&calcs, h, b)
411                    });
412                    let layout = Layout {
413                        size,
414                        location: Point {
415                            x: x + gap + margin.left + offset.x,
416                            y: margin.top + offset.y,
417                        },
418                        padding,
419                        border,
420                        margin,
421                        ..Layout::default()
422                    };
423                    ascent = ascent.max(margin.top + child_baseline);
424                    descent = descent.max(margin.bottom + size.height - child_baseline);
425                    row.push((child, layout, margin.top + child_baseline));
426                    x += gap + margin.left + size.width + margin.right;
427                }
428                if !row.is_empty() {
429                    first_baseline.get_or_insert(y + ascent);
430                    for (child, mut layout, base) in row.drain(..) {
431                        layout.location.y += y + ascent - base;
432                        self.set_unrounded_layout(child, &layout);
433                    }
434                    width = width.max(x);
435                    y += ascent + descent;
436                }
437                Size { width, height: y }
438            },
439        );
440        output.baselines = taffy::tree::Baselines {
441            first: first_baseline,
442            last: first_baseline,
443        };
444        output
445    }
446    fn child_layout(
447        &mut self,
448        id: taffy::NodeId,
449        inputs: LayoutInput,
450        block: Option<&mut taffy::BlockContext<'_>>,
451    ) -> LayoutOutput {
452        if inputs.run_mode == RunMode::PerformHiddenLayout {
453            return taffy::compute_hidden_layout(self, id);
454        }
455        taffy::compute_cached_layout(self, id, inputs, |tree, id, inputs| {
456            if tree.tree.nodes[usize::from(id)].inline_flow {
457                return tree.inline_layout(id, inputs);
458            }
459            let node = &tree.tree.nodes[usize::from(id)];
460            match (node.style.display, !node.children.is_empty()) {
461                (Display::None, _) => taffy::compute_hidden_layout(tree, id),
462                (Display::Block, true) => taffy::compute_block_layout(tree, id, inputs, block),
463                (Display::FlowRoot, true) => taffy::compute_block_layout(tree, id, inputs, None),
464                (Display::Flex, true) => taffy::compute_flexbox_layout(tree, id, inputs),
465                (Display::Grid, true) => taffy::compute_grid_layout(tree, id, inputs),
466                _ => taffy::compute_leaf_layout(
467                    inputs,
468                    &node.style,
469                    |handle, basis| super::css::resolve_calc(&tree.tree.calcs, handle, basis),
470                    |known, available| {
471                        node.context.as_ref().map_or(node.intrinsic, |ctx| {
472                            measure(tree.fonts, ctx, known, available)
473                        })
474                    },
475                ),
476            }
477        })
478    }

564fn collect_styles(
565    arena: &DocumentArena,
566    id: NodeId,
567    external: &[(NodeId, String)],
568    sheet: &mut Stylesheet,
569    viewport: (u32, u32),
570) {
571    let Some(node) = arena.get(id) else {
572        return;
573    };
574    if node.tag() == Some("style") {
575        sheet.append(&arena.get_text_content(id), viewport);
576    }
577    if node.tag() == Some("link") {
578        if let Some((_, css)) = external.iter().find(|(n, _)| *n == id) {
579            sheet.append(css, viewport);
580        }
581    }
582    let mut child = node.first_child;
583    while let Some(id) = child {
584        collect_styles(arena, id, external, sheet, viewport);
585        child = arena.get(id).and_then(|n| n.next_sibling);
586    }
587}
588impl<'a> Builder<'a> {
589    fn inline_groups(
590        &mut self,
591        children: Vec<taffy::NodeId>,
592    ) -> Result<Vec<taffy::NodeId>, taffy::TaffyError> {
593        let mut result = Vec::new();
594        let mut group = Vec::new();
595        let mut atomic = false;
596        for child in children {
597            let entry = &self.entries[self.tree.nodes[usize::from(child)].entry.unwrap()];
598            let inline = entry.inline && !entry.position.out_of_flow();
599            if !inline {
600                self.finish_inline_group(&mut result, &mut group, atomic)?;
601                atomic = false;
602                result.push(child);
603            } else {
604                atomic |= entry.atomic_inline || entry.text.is_none();
605                group.push(child);
606            }
607        }
608        self.finish_inline_group(&mut result, &mut group, atomic)?;
609        Ok(result)
610    }
611    fn finish_inline_group(
612        &mut self,
613        result: &mut Vec<taffy::NodeId>,
614        group: &mut Vec<taffy::NodeId>,
615        atomic: bool,
616    ) -> Result<(), taffy::TaffyError> {
617        if !atomic {
618            result.append(group);
619            return Ok(());
620        }
621        let tid = self.tree.new_leaf(Style {
622            display: Display::Block,
623            ..Style::default()
624        })?;
625        self.tree.nodes[usize::from(tid)].inline_flow = true;
626        let mut children = Vec::new();
627        let mut pending = None;
628        let mut force_break = false;
629        for child in group.drain(..) {
630            let index = self.tree.nodes[usize::from(child)].entry.unwrap();
631            let original = self.entries[index].clone();
632            let Some(context) = original.text.clone() else {
633                self.tree.nodes[usize::from(child)].gap = pending.take();
634                self.tree.nodes[usize::from(child)].break_before =
635                    core::mem::take(&mut force_break);
636                children.push(child);
637                continue;
638            };
639            self.entries[index].hidden = true;
640            let mut words: Vec<SplitWord> = Vec::new();
641            let mut word = Vec::new();
642            for run in &context.runs {
643                let mut segment = String::new();
644                for c in normalized_chars(&run.span.text) {
645                    if c.is_whitespace() || c == '\u{2028}' {
646                        if !segment.is_empty() {
647                            let span = StyledSpan {
648                                text: core::mem::take(&mut segment),
649                                bold: run.span.bold,
650                                italic: run.span.italic,
651                                code: run.span.code,
652                                mono: run.span.mono,
653                                link: run.span.link.clone(),
654                                color: run.span.color,
655                            };
656                            append_run(
657                                &mut word,
658                                TextRun::new(span, run.px, run.node_id, run.line_height),
659                            );
660                        }
661                        if !word.is_empty() {
662                            words.push((
663                                core::mem::take(&mut word),
664                                pending.take(),
665                                core::mem::take(&mut force_break),
666                            ));
667                        }
668                        if c == '\u{2028}' || (context.pre && c == '\n') {
669                            force_break = true;
670                            pending = None;
671                        } else {
672                            pending.get_or_insert((run.face(), run.px));
673                        }
674                    } else {
675                        segment.push(c);
676                    }
677                }
678                if !segment.is_empty() {
679                    let span = StyledSpan {
680                        text: segment,
681                        bold: run.span.bold,
682                        italic: run.span.italic,
683                        code: run.span.code,
684                        mono: run.span.mono,
685                        link: run.span.link.clone(),
686                        color: run.span.color,
687                    };
688                    append_run(
689                        &mut word,
690                        TextRun::new(span, run.px, run.node_id, run.line_height),
691                    );
692                }
693            }
694            if !word.is_empty() {
695                words.push((word, pending.take(), core::mem::take(&mut force_break)));
696            }
697            for (runs, gap, break_before) in words {
698                let context = Rc::new(TextContext {
699                    runs,
700                    line_height: context.line_height,
701                    pre: true,
702                    wrap_pre: false,
703                    align: 0,
704                });
705                let child = self.tree.new_leaf_with_context(
706                    Style {
707                        display: Display::Block,
708                        ..Style::default()
709                    },
710                    context.clone(),
711                )?;
712                let node = &mut self.tree.nodes[usize::from(child)];
713                node.gap = gap;
714                node.break_before = break_before;
715                node.entry = Some(self.entries.len());
716                let mut entry = original.clone();
717                entry.taffy = child;
718                entry.text = Some(context);
719                self.entries.push(entry);
720                children.push(child);
721            }
722            self.tree.nodes[usize::from(tid)].intrinsic = Size {
723                width: original.font_size,
724                height: original.line_height,
725            };
726        }
727        if self.tree.nodes[usize::from(tid)].intrinsic.height == 0.0 {
728            if let Some(first) = children
729                .first()
730                .and_then(|id| self.tree.nodes[usize::from(*id)].entry)
731                .map(|index| &self.entries[index])
732            {
733                self.tree.nodes[usize::from(tid)].intrinsic = Size {
734                    width: first.font_size,
735                    height: first.line_height,
736                };
737            }
738        }
739        self.tree.set_children(tid, &children)?;
740        result.push(tid);
741        Ok(())
742    }

788    fn element(
789        &mut self,
790        id: NodeId,
791        parent: Option<usize>,
792        mut style: ComputedStyle,
793        link: Option<&str>,
794    ) -> Result<taffy::NodeId, taffy::TaffyError> {
795        let link = self
796            .arena
797            .get(id)
798            .and_then(|n| n.get_attribute("href"))
799            .or(link);
800        style.font_size = style.font_size.min(self.max_font);
801        style.line_height = style.line_height.clamp(1.0, self.max_font * 4.0);
802        let tid = self.tree.new_leaf(Style::default())?;
803        let index = self.entries.len();
804        self.tree.nodes[usize::from(tid)].entry = Some(index);
805        self.entries.push(Entry {
806            taffy: tid,
807            parent,
808            node_id: Some(id),
809            hidden: style.layout.display == Display::None,
810            background: style.background,
811            border_color: style.border_color,
812            text: None,
813            input_data: None,
814            image: None,
815            position: style.position,
816            z_index: style.z_index,
817            containing: None,
818            placeholder: None,
819            font_size: style.font_size,
820            line_height: style.line_height,
821            color: style.color,
822            inline: style.inline,
823            atomic_inline: style.atomic_inline,
824            paint_order: index,
825            visible: style.visible,
826        });
827        if style.layout.display == Display::None {
828            self.tree.set_style(tid, style.layout)?;
829            return Ok(tid);
830        }
831        if let Some(node) = self.arena.get(id) {
832            if matches!(node.tag(), Some("input" | "textarea" | "button" | "select")) {
833                let kind = if node.tag() == Some("input") {
834                    node.get_attribute("type")
835                        .unwrap_or("text")
836                        .to_ascii_lowercase()
837                } else if node.tag() == Some("button") {
838                    node.get_attribute("type")
839                        .unwrap_or("submit")
840                        .to_ascii_lowercase()
841                } else if node.tag() == Some("select") {
842                    String::from("select")
843                } else {
844                    String::from("textarea")
845                };
846                if kind == "hidden" {
847                    style.layout.display = Display::None;
848                    self.entries[index].hidden = true;
849                }
850                let value = if node.tag() == Some("input") {
851                    String::from(node.get_attribute("value").unwrap_or(""))
852                } else {
853                    node.get_attribute("value")
854                        .map(String::from)
855                        .unwrap_or_else(|| self.arena.get_text_content(id))
856                };
857                let checkbox = kind == "checkbox";
858                if style.layout.size.width.is_auto() {
859                    style.layout.size.width = length(if checkbox {
860                        20.0
861                    } else if kind == "submit" || kind == "button" {
862                        160.0
863                    } else {
864                        300.0f32.min(self.viewport.0 as f32)
865                    });
866                }
867                if style.layout.size.height.is_auto() {
868                    style.layout.size.height = length(if checkbox {
869                        20.0
870                    } else if kind == "textarea" {
871                        80.0
872                    } else {
873                        30.0
874                    });
875                }
876                self.entries[index].input_data = Some(crate::doc::FormInputLayout {
877                    input_type: kind,
878                    name: String::from(node.get_attribute("name").unwrap_or("")),
879                    value,
880                    placeholder: String::from(node.get_attribute("placeholder").unwrap_or("")),
881                    checked: node.get_attribute("checked").is_some(),
882                    is_focused: false,
883                });
884                self.tree.set_style(tid, style.layout)?;
885                return Ok(tid);
886            }
887            if node.tag() == Some("img") {
888                if let Some((w, h, pixels)) = self.assets.iter().find_map(|asset| match asset {
889                    crate::doc::DocNode::DomImage {
890                        node_id,
891                        width,
892                        height,
893                        pixels,
894                    } if *node_id == id => Some((*width, *height, pixels.clone())),
895                    _ => None,
896                }) {
897                    let attr_w = node
898                        .get_attribute("width")
899                        .and_then(|v| v.parse::<f32>().ok())
900                        .filter(|v| *v > 0.0);
901                    let attr_h = node
902                        .get_attribute("height")
903                        .and_then(|v| v.parse::<f32>().ok())
904                        .filter(|v| *v > 0.0);
905                    if style.layout.size.width.is_auto() {
906                        if let Some(width) = attr_w {
907                            style.layout.size.width = length(width);
908                        }
909                    }
910                    if style.layout.size.height.is_auto() {
911                        if let Some(height) = attr_h {
912                            style.layout.size.height = length(height);
913                        }
914                    }
915                    if style.layout.size.width.is_auto() || style.layout.size.height.is_auto() {
916                        style.layout.aspect_ratio = Some(w.max(1) as f32 / h.max(1) as f32);
917                    }
918                    self.tree.nodes[usize::from(tid)].intrinsic = Size {
919                        width: w as f32,
920                        height: h as f32,
921                    };
922                    self.entries[index].image = Some((w, h, pixels));
923                } else {
924                    let alt = node.get_attribute("alt").unwrap_or("[Image unavailable]");
925                    let mut runs = Vec::new();
926                    append_text(&mut runs, alt, &style, link, id);
927                    let context = Rc::new(TextContext {
928                        runs,
929                        line_height: pixel(style.line_height),
930                        pre: false,
931                        wrap_pre: true,
932                        align: 0,
933                    });
934                    self.tree.set_node_context(tid, Some(context.clone()))?;
935                    self.entries[index].text = Some(context);
936                }
937                self.tree.set_style(tid, style.layout)?;
938                return Ok(tid);
939            }
940        }
941        let mut children = Vec::new();
942        let mut runs = Vec::new();
943        let arena = self.arena;
944        let mut child = arena.get(id).and_then(|n| n.first_child);
945        while let Some(cid) = child {
946            let Some(node) = arena.get(cid) else {
947                break;
948            };
949            match &node.data {
950                NodeData::Text(text) => append_text(&mut runs, text, &style, link, id),
951                NodeData::Element { .. } => {
952                    let mut child_style =
953                        self.sheet
954                            .compute(self.arena, cid, Some(&style), self.viewport);
955                    child_style.font_size = child_style.font_size.min(self.max_font);
956                    child_style.line_height =
957                        child_style.line_height.clamp(1.0, self.max_font * 4.0);
958                    if child_style.layout.display != Display::None {
959                        if child_style.inline
960                            && !child_style.atomic_inline
961                            && child_style.position == CssPosition::Static
962                            && !is_replaced(arena, cid)
963                        {
964                            self.inline(
965                                cid,
966                                &child_style,
967                                link,
968                                &mut InlineFlow {
969                                    runs: &mut runs,
970                                    children: &mut children,
971                                    parent: index,
972                                    block_style: &style,
973                                },
974                            )?;
975                        } else {
976                            if !runs.is_empty() {
977                                children.push(self.text_leaf(&mut runs, &style, index)?);
978                            }
979                            children.push(self.element(cid, Some(index), child_style, link)?);
980                        }
981                    }
982                }
983                _ => {}
984            }
985            child = node.next_sibling;
986        }
987        if !runs.is_empty() {
988            children.push(self.text_leaf(&mut runs, &style, index)?);
989        }
990        let children = if matches!(style.layout.display, Display::Block | Display::FlowRoot) {
991            self.inline_groups(children)?
992        } else {
993            children
994        };
995        self.tree.set_children(tid, &children)?;
996        self.tree.set_style(tid, style.layout)?;
997        Ok(tid)
998    }
999    fn inline(
1000        &mut self,
1001        id: NodeId,
1002        style: &ComputedStyle,
1003        link: Option<&str>,
1004        flow: &mut InlineFlow<'_>,
1005    ) -> Result<(), taffy::TaffyError> {
1006        let arena = self.arena;
1007        let Some(node) = arena.get(id) else {
1008            return Ok(());
1009        };
1010        let link = node.get_attribute("href").or(link);
1011        if node.tag() == Some("br") {
1012            append_text(flow.runs, "\u{2028}", style, link, id);
1013            return Ok(());
1014        }
1015        let mut child = node.first_child;
1016        while let Some(cid) = child {
1017            let Some(n) = arena.get(cid) else {
1018                break;
1019            };
1020            match &n.data {
1021                NodeData::Text(text) => append_text(flow.runs, text, style, link, id),
1022                NodeData::Element { .. } => {
1023                    let mut s = self.sheet.compute(arena, cid, Some(style), self.viewport);
1024                    s.font_size = s.font_size.min(self.max_font);
1025                    s.line_height = s.line_height.clamp(1.0, self.max_font * 4.0);
1026                    if s.layout.display != Display::None {
1027                        if s.inline
1028                            && !s.atomic_inline
1029                            && s.position == CssPosition::Static
1030                            && !is_replaced(arena, cid)
1031                        {
1032                            self.inline(cid, &s, link, flow)?;
1033                        } else {
1034                            if !flow.runs.is_empty() {
1035                                flow.children.push(self.text_leaf(
1036                                    flow.runs,
1037                                    flow.block_style,
1038                                    flow.parent,
1039                                )?);
1040                            }
1041                            flow.children
1042                                .push(self.element(cid, Some(flow.parent), s, link)?);
1043                        }
1044                    }
1045                }
1046                _ => {}
1047            }
1048            child = n.next_sibling;
1049        }
1050        Ok(())
1051    }
1052}

1093fn same_run(a: &TextRun, b: &TextRun) -> bool {
1094    a.node_id == b.node_id
1095        && a.px == b.px
1096        && a.line_height == b.line_height
1097        && a.span.bold == b.span.bold
1098        && a.span.italic == b.span.italic
1099        && a.span.code == b.span.code
1100        && a.span.mono == b.span.mono
1101        && a.span.color == b.span.color
1102        && a.span.link == b.span.link
1103        && a.visible == b.visible
1104}
1105pub fn append_run(runs: &mut Vec<TextRun>, run: TextRun) {
1106    if let Some(last) = runs.last_mut().filter(|last| same_run(last, &run)) {
1107        last.span.text.push_str(&run.span.text);
1108    } else {
1109        runs.push(run);
1110    }
1111}
1112fn next_advance(fonts: &mut TextFonts, line: &TextLine, source: &TextRun, c: char) -> f32 {
1113    let kern = line
1114        .runs
1115        .last()
1116        .filter(|r| same_run(r, source))
1117        .and_then(|r| r.span.text.chars().next_back())
1118        .map_or(0.0, |left| fonts.kerning(source.face(), left, c, source.px));
1119    fonts.advance(source.face(), c, source.px) + kern
1120}
1121fn push_char(fonts: &mut TextFonts, line: &mut TextLine, source: &TextRun, c: char) {
1122    let advance = next_advance(fonts, line, source, c);
1123    line.baseline = line.baseline.max(fonts.ascender(source.face(), source.px));
1124    line.height = line.height.max(ceil_pixel(
1125        source
1126            .line_height
1127            .max(fonts.line_height(source.face(), source.px)),
1128    ));
1129    if let Some(last) = line.runs.last_mut().filter(|r| same_run(r, source)) {
1130        last.span.text.push(c);
1131        last.width += advance;
1132    } else {
1133        let mut span = StyledSpan::plain("");
1134        span.bold = source.span.bold;
1135        span.italic = source.span.italic;
1136        span.code = source.span.code;
1137        span.mono = source.span.mono;
1138        span.link = source.span.link.clone();
1139        span.color = source.span.color;
1140        span.text.push(c);
1141        let mut run = TextRun::with_visible(
1142            span,
1143            source.px,
1144            source.node_id,
1145            source.line_height,
1146            source.visible,
1147        );
1148        run.x = line.width;
1149        run.width = advance;
1150        line.runs.push(run);
1151    }
1152    line.width += advance;
1153}
1154fn finish_line(fonts: &mut TextFonts, line: &mut TextLine, lines: &mut Vec<TextLine>, height: u32) {
1155    // The shared baseline also covers mixed-size ascenders and descenders.
1156    let descent = line
1157        .runs
1158        .iter()
1159        .map(|r| fonts.line_height(r.face(), r.px) - fonts.ascender(r.face(), r.px))
1160        .fold(0.0f32, f32::max);
1161    line.height = line.height.max(ceil_pixel(line.baseline + descent));
1162    lines.push(core::mem::replace(line, empty_line(height)));
1163}
1164fn flush_word<'a>(
1165    fonts: &mut TextFonts,
1166    word: &mut Vec<(&'a TextRun, char)>,
1167    line: &mut TextLine,
1168    lines: &mut Vec<TextLine>,
1169    pending: &mut Option<&'a TextRun>,
1170    limit: f32,
1171    height: u32,
1172) {
1173    if word.is_empty() {
1174        return;
1175    }
1176    let mut width = 0.0;
1177    let mut previous: Option<(&TextRun, char)> = None;
1178    for &(r, c) in word.iter() {
1179        width += fonts.advance(r.face(), c, r.px);
1180        if let Some((left, l)) = previous.filter(|(left, _)| same_run(left, r)) {
1181            width += fonts.kerning(left.face(), l, c, left.px);
1182        }
1183        previous = Some((r, c));
1184    }
1185    let source = word[0].0;
1186    let space_source = pending.filter(|_| line.width > 0.0);
1187    let space = space_source.map_or(0.0, |r| next_advance(fonts, line, r, ' '));
1188    let boundary = if let Some(r) = space_source {
1189        if same_run(r, source) {
1190            fonts.kerning(source.face(), ' ', word[0].1, source.px)
1191        } else {
1192            0.0
1193        }
1194    } else if line.width > 0.0 {
1195        next_advance(fonts, line, source, word[0].1)
1196            - fonts.advance(source.face(), word[0].1, source.px)
1197    } else {
1198        0.0
1199    };
1200    if line.width > 0.0 && line.width + space + width + boundary > limit {
1201        finish_line(fonts, line, lines, height);
1202    } else if let Some(r) = space_source {
1203        push_char(fonts, line, r, ' ');
1204    }
1205    for (source, c) in word.drain(..) {
1206        if line.width > 0.0 && line.width + next_advance(fonts, line, source, c) > limit {
1207            finish_line(fonts, line, lines, height);
1208        }
1209        push_char(fonts, line, source, c);
1210    }
1211    *pending = None;
1212}

1214/// Shared measured wrapping for HTML, Markdown, plain text, and table cells.
1215pub fn wrap_runs(
1216    fonts: &mut TextFonts,
1217    runs: &[TextRun],
1218    width: f32,
1219    height: u32,
1220    pre: bool,
1221    wrap_pre: bool,
1222    align: u8,
1223) -> Vec<TextLine> {
1224    let limit = width.max(1.0);
1225    let mut lines = Vec::new();
1226    let mut line = empty_line(height);
1227    let mut pending = None;
1228    let mut word = Vec::new();
1229    for run in runs {
1230        for c in normalized_chars(&run.span.text) {
1231            if c == '\u{2028}' || (pre && c == '\n') {
1232                flush_word(
1233                    fonts,
1234                    &mut word,
1235                    &mut line,
1236                    &mut lines,
1237                    &mut pending,
1238                    limit,
1239                    height,
1240                );
1241                finish_line(fonts, &mut line, &mut lines, height);
1242                pending = None;
1243            } else if pre {
1244                // Expand tabs into measured spaces, rather than attempting a tab glyph.
1245                let count = if c == '\t' { 4 } else { 1 };
1246                let c = if c == '\t' { ' ' } else { c };
1247                for _ in 0..count {
1248                    if wrap_pre
1249                        && line.width > 0.0
1250                        && line.width + next_advance(fonts, &line, run, c) > limit
1251                    {
1252                        finish_line(fonts, &mut line, &mut lines, height);
1253                    }
1254                    push_char(fonts, &mut line, run, c);
1255                }
1256            } else if c.is_whitespace() {
1257                flush_word(
1258                    fonts,
1259                    &mut word,
1260                    &mut line,
1261                    &mut lines,
1262                    &mut pending,
1263                    limit,
1264                    height,
1265                );
1266                pending.get_or_insert(run);
1267            } else {
1268                word.push((run, c));
1269            }
1270        }
1271    }
1272    flush_word(
1273        fonts,
1274        &mut word,
1275        &mut line,
1276        &mut lines,
1277        &mut pending,
1278        limit,
1279        height,
1280    );
1281    if !line.runs.is_empty() || (pre && lines.is_empty()) {
1282        finish_line(fonts, &mut line, &mut lines, height);
1283    }
1284    for line in &mut lines {
1285        let offset = match align {
1286            1 => (width - line.width).max(0.0) * 0.5,
1287            2 => (width - line.width).max(0.0),
1288            _ => 0.0,
1289        };
1290        for run in &mut line.runs {
1291            run.x += offset;
1292        }
1293    }
1294    lines
1295}
1296fn wrap(fonts: &mut TextFonts, context: &TextContext, width: f32) -> Vec<TextLine> {
1297    wrap_runs(
1298        fonts,
1299        &context.runs,
1300        width,
1301        context.line_height,
1302        context.pre,
1303        context.wrap_pre,
1304        context.align,
1305    )
1306}
1307fn intrinsic_widths(fonts: &mut TextFonts, context: &TextContext) -> (f32, f32) {
1308    let mut maximum = 0.0f32;
1309    let mut longest = 0.0f32;
1310    let mut width = 0.0f32;
1311    let mut word = 0.0f32;
1312    let mut previous: Option<(&TextRun, char)> = None;
1313    let mut pending: Option<&TextRun> = None;
1314    for run in &context.runs {
1315        for c in normalized_chars(&run.span.text) {
1316            if c == '\u{2028}' || (context.pre && c == '\n') {
1317                maximum = maximum.max(width);
1318                longest = longest.max(word);
1319                width = 0.0;
1320                word = 0.0;
1321                previous = None;
1322                pending = None;
1323                continue;
1324            }
1325            if !context.pre && c.is_whitespace() {
1326                longest = longest.max(word);
1327                word = 0.0;
1328                pending.get_or_insert(run);
1329                continue;
1330            }
1331            if let Some(space_source) = pending.filter(|_| width > 0.0) {
1332                width += fonts.advance(space_source.face(), ' ', space_source.px);
1333                if let Some((left, l)) = previous.filter(|(left, _)| same_run(left, space_source)) {
1334                    width += fonts.kerning(left.face(), l, ' ', left.px);
1335                }
1336                previous = Some((space_source, ' '));
1337            }
1338            pending = None;
1339            let count = if context.pre && c == '\t' { 4 } else { 1 };
1340            let c = if context.pre && c == '\t' { ' ' } else { c };
1341            for _ in 0..count {
1342                let glyph = fonts.advance(run.face(), c, run.px);
1343                let kern = previous
1344                    .filter(|(left, _)| same_run(left, run))
1345                    .map_or(0.0, |(left, l)| fonts.kerning(left.face(), l, c, left.px));
1346                width += glyph + kern;
1347                if c.is_whitespace() {
1348                    longest = longest.max(word);
1349                    word = 0.0;
1350                } else {
1351                    word += glyph
1352                        + if previous.is_some_and(|(_, l)| !l.is_whitespace()) {
1353                            kern
1354                        } else {
1355                            0.0
1356                        };
1357                }
1358                previous = Some((run, c));
1359            }
1360        }
1361    }
1362    maximum = maximum.max(width);
1363    longest = longest.max(word);
1364    (
1365        maximum,
1366        if context.pre && !context.wrap_pre {
1367            maximum
1368        } else {
1369            longest
1370        },
1371    )
1372}
1373fn measure(
1374    fonts: &mut TextFonts,
1375    context: &TextContext,
1376    known: Size<Option<f32>>,
1377    available: Size<AvailableSpace>,
1378) -> Size<f32> {
1379    let (max_width, longest) = intrinsic_widths(fonts, context);
1380    let width = known.width.unwrap_or_else(|| match available.width {
1381        AvailableSpace::Definite(w) => w.max(0.0),
1382        AvailableSpace::MinContent => longest,
1383        AvailableSpace::MaxContent => max_width,
1384    });
1385    let lines = wrap(fonts, context, width);
1386    let actual = lines.iter().map(|l| l.width).fold(0.0f32, f32::max);
1387    Size {
1388        width: known.width.unwrap_or(actual.min(width)),
1389        height: known
1390            .height
1391            .unwrap_or(lines.iter().map(|l| l.height).sum::<u32>() as f32),
1392    }
1393}

1395/// Colour `run` paints with: under a document palette, links, role tags and

1405        (None, _) => base,
1406    }

1477pub fn layout_with_nodes_zoom(
1478    fonts: &mut TextFonts,
1479    arena: &DocumentArena,
1480    external: &[(NodeId, String)],
1481    width: u32,
1482    height: u32,
1483    assets: &[crate::doc::DocNode],
1484    zoom: f32,
1485) -> Result<WebLayout, taffy::TaffyError> {
1486    let viewport = (width.max(1), height.max(1));
1487    let mut sheet = Stylesheet::default();
1488    collect_styles(arena, arena.root, external, &mut sheet, viewport);
1489    let mut builder = Builder {
1490        arena,
1491        sheet,
1492        viewport,
1493        tree: WebTree::default(),
1494        entries: Vec::new(),
1495        assets,
1496        max_font: super::css::MAX_FONT_SIZE / zoom.max(1.0),
1497    };
1498    let mut root_style = builder.sheet.compute(arena, arena.root, None, viewport);
1499    root_style.layout.size.width = length(viewport.0 as f32);
1500    // The document node represents the initial containing block, not an
1501    // auto-height HTML element. Normal auto-height descendants remain indefinite.
1502    root_style.layout.size.height = length(viewport.1 as f32);
1503    builder.tree.calcs = root_style.calcs.clone();
1504    // An out-of-flow placeholder records the hypothetical static position but
1505    // contributes no size. Positioned subtrees are then laid out once, after
1506    // their actual containing block is known, in ancestor-before-child order.
1507    let root = builder.element(arena.root, None, root_style, None)?;
1508    for index in 0..builder.entries.len() {
1509        if !builder.entries[index].position.out_of_flow() {
1510            continue;
1511        }
1512        let mut containing = builder.entries[index].parent;
1513        if builder.entries[index].position == CssPosition::Fixed {
1514            containing = None;
1515        } else {
1516            while let Some(p) = containing {
1517                if builder.entries[p].position != CssPosition::Static {
1518                    break;
1519                }
1520                containing = builder.entries[p].parent;
1521            }
1522        }
1523        builder.entries[index].containing = containing;
1524        if let Some(parent) = builder.entries[index].parent {
1525            let placeholder = builder.tree.new_leaf(Style {
1526                position: Position::Absolute,
1527                size: Size {
1528                    width: length(0.0),
1529                    height: length(0.0),
1530                },
1531                ..Style::default()
1532            })?;
1533            let tid = builder.entries[index].taffy;
1534            let children =
1535                &mut builder.tree.nodes[usize::from(builder.entries[parent].taffy)].children;
1536            if let Some(child) = children.iter_mut().find(|child| **child == tid) {
1537                *child = placeholder;
1538            }
1539            builder.entries[index].placeholder = Some(placeholder);
1540        }
1541    }
1542    let mut view = LayoutView {
1543        tree: &mut builder.tree,
1544        fonts,
1545    };
1546    taffy::compute_root_layout(
1547        &mut view,
1548        root,
1549        Size {
1550            width: AvailableSpace::Definite(viewport.0 as f32),
1551            height: AvailableSpace::Definite(viewport.1 as f32),
1552        },
1553    );
1554    taffy::round_layout(&mut view, root);
1555    let mut result = WebLayout::default();
1556    let mut origins: Vec<(i32, i32)> = Vec::new();
1557    let mut fixed_flags = Vec::new();
1558    let mut descendant_clips: Vec<BoxClip> = Vec::new();
1559    let mut boxes: Vec<Option<WebBox>> = Vec::with_capacity(builder.entries.len());
1560    for index in 0..builder.entries.len() {
1561        let entry = &builder.entries[index];
1562        let mut base = entry.parent.map_or((0, 0), |p| origins[p]);
1563        if entry.position.out_of_flow() {
1564            let mut area = (viewport.0 as f32, viewport.1 as f32);
1565            base = (0, 0);
1566            if let Some(cb) = entry.containing {
1567                let l = builder.tree.layout(builder.entries[cb].taffy)?;
1568                let origin = origins[cb];
1569                base = (
1570                    origin.0.saturating_add(l.border.left as i32),
1571                    origin.1.saturating_add(l.border.top as i32),
1572                );
1573                area = (
1574                    l.size.width - l.border.left - l.border.right,
1575                    l.size.height - l.border.top - l.border.bottom,
1576                );
1577            }
1578            let mut fallback = entry.parent.map_or((0, 0), |p| origins[p]);
1579            if let Some(placeholder) = entry.placeholder {
1580                let l = builder.tree.layout(placeholder)?;
1581                fallback.0 = fallback.0.saturating_add(l.location.x as i32);
1582                fallback.1 = fallback.1.saturating_add(l.location.y as i32);
1583            }
1584            let node = &mut builder.tree.nodes[usize::from(entry.taffy)];
1585            if node.style.inset.left.is_auto() && node.style.inset.right.is_auto() {
1586                node.style.inset.left = length((fallback.0 - base.0) as f32);
1587            }
1588            if node.style.inset.top.is_auto() && node.style.inset.bottom.is_auto() {
1589                node.style.inset.top = length((fallback.1 - base.1) as f32);
1590            }
1591            let containing = builder.tree.new_leaf(Style {
1592                display: Display::Block,
1593                size: Size {
1594                    width: length(area.0.max(0.0)),
1595                    height: length(area.1.max(0.0)),
1596                },
1597                ..Style::default()
1598            })?;
1599            builder.tree.set_children(containing, &[entry.taffy])?;
1600            let mut view = LayoutView {
1601                tree: &mut builder.tree,
1602                fonts,
1603            };
1604            taffy::compute_root_layout(
1605                &mut view,
1606                containing,
1607                Size {
1608                    width: AvailableSpace::Definite(area.0),
1609                    height: AvailableSpace::Definite(area.1),
1610                },
1611            );
1612            taffy::round_layout(&mut view, containing);
1613        }
1614        let l = builder.tree.layout(entry.taffy)?;
1615        if !entry.position.out_of_flow() {
1616            let stop = entry.parent.map(|p| builder.entries[p].taffy);
1617            let mut parent = builder.tree.nodes[usize::from(entry.taffy)].parent;
1618            while let Some(id) = parent {
1619                if Some(id) == stop {
1620                    break;
1621                }
1622                let node = &builder.tree.nodes[usize::from(id)];
1623                base.0 = base.0.saturating_add(node.rounded.location.x as i32);
1624                base.1 = base.1.saturating_add(node.rounded.location.y as i32);
1625                parent = node.parent;
1626            }
1627        }
1628        let x = base.0.saturating_add(l.location.x as i32);
1629        let y = base.1.saturating_add(l.location.y as i32);
1630        origins.push((x, y));
1631        let inheritance = if entry.position.out_of_flow() {
1632            entry.containing
1633        } else {
1634            entry.parent
1635        };
1636        let fixed =
1637            entry.position == CssPosition::Fixed || inheritance.is_some_and(|p| fixed_flags[p]);
1638        fixed_flags.push(fixed);
1639        let clip = inheritance.map_or(BoxClip::default(), |p| descendant_clips[p]);
1640        let mut children_clip = clip;
1641        let overflow = builder.tree.nodes[usize::from(entry.taffy)].style.overflow;
1642        if overflow.x != taffy::style::Overflow::Visible {
1643            children_clip.x = BoxClip::intersect_axis(
1644                children_clip.x,
1645                (
1646                    x.saturating_add(l.border.left as i32),
1647                    x.saturating_add(l.size.width as i32)
1648                        .saturating_sub(l.border.right as i32),
1649                ),
1650            );
1651        }
1652        if overflow.y != taffy::style::Overflow::Visible {
1653            children_clip.y = BoxClip::intersect_axis(
1654                children_clip.y,
1655                (
1656                    y.saturating_add(l.border.top as i32),
1657                    y.saturating_add(l.size.height as i32)
1658                        .saturating_sub(l.border.bottom as i32),
1659                ),
1660            );
1661        }
1662        descendant_clips.push(children_clip);
1663        if entry.hidden {
1664            boxes.push(None);
1665            continue;
1666        }
1667        let border = [
1668            pixel(l.border.left),
1669            pixel(l.border.right),
1670            pixel(l.border.top),
1671            pixel(l.border.bottom),
1672        ];
1673        let bw = pixel(l.size.width);
1674        let bh = pixel(l.size.height);
1675        let lines = entry
1676            .text
1677            .as_ref()
1678            .map_or_else(Vec::new, |text| wrap(fonts, text, l.size.width));
1679        if !fixed {
1680            result.height =
1681                result
1682                    .height
1683                    .max(clip.y.map_or(y.saturating_add(bh as i32), |(_, end)| {
1684                        end.min(y.saturating_add(bh as i32))
1685                    }));
1686        }
1687        let node_style = &builder.tree.nodes[usize::from(entry.taffy)].style;
1688        boxes.push(Some(WebBox {
1689            node_id: entry.node_id,
1690            x,
1691            y,
1692            width: bw,
1693            height: bh,
1694            background: entry.background,
1695            border_color: entry.border_color,
1696            border,
1697            lines,
1698            input_data: entry.input_data.clone(),
1699            image: entry.image.clone(),
1700            highlights: Vec::new(),
1701            control_px: 14.0,
1702            fixed,
1703            clip,
1704            font_size: entry.font_size,
1705            line_height: entry.line_height,
1706            color: entry.color,
1707            display: node_style.display,
1708            overflow,
1709            padding: [
1710                pixel(l.padding.left),
1711                pixel(l.padding.right),
1712                pixel(l.padding.top),
1713                pixel(l.padding.bottom),
1714            ],
1715            box_sizing: node_style.box_sizing,
1716            position: entry.position,
1717            inline: entry.inline,
1718            atomic_inline: entry.atomic_inline,
1719            visible: entry.visible,
1720        }));
1721    }
1722    for index in stacking_order(&builder.entries, &builder.tree) {
1723        if let Some(b) = boxes[index].take() {
1724            result.boxes.push(b);
1725        }
1726    }
1727    // White default canvas is a web default, independent of the viewer chrome theme.
1728    for b in &mut result.boxes {
1729        if b.node_id == Some(arena.root) {
1730            if b.background.is_none() {
1731                b.background = Some(Color::rgb(255, 255, 255));
1732            }
1733            b.height = b.height.max(height);
1734            break;
1735        }
1736    }
1737    Ok(result)
1738}

1740/// Sort only the children of each stacking context, then emit each context as
1741/// an indivisible unit. A descendant's high z-index cannot escape its context.
1742fn stacking_order(entries: &[Entry], tree: &WebTree) -> Vec<usize> {
1743    let mut contexts = Vec::with_capacity(entries.len());
1744    let mut owners = Vec::with_capacity(entries.len());
1745    let mut positioned = Vec::with_capacity(entries.len());
1746    let mut children: Vec<Vec<(i8, i32, usize, usize)>> =
1747        (0..entries.len()).map(|_| Vec::new()).collect();
1748    for (index, entry) in entries.iter().enumerate() {
1749        let item = entry.parent.is_some_and(|p| {
1750            matches!(
1751                tree.nodes[usize::from(entries[p].taffy)].style.display,
1752                Display::Flex | Display::Grid
1753            )
1754        });
1755        let context = index == 0
1756            || entry.position == CssPosition::Fixed
1757            || (entry.z_index.is_some() && (entry.position != CssPosition::Static || item));
1758        contexts.push(context);
1759        let owner = entry
1760            .parent
1761            .map_or(0, |p| if contexts[p] { p } else { owners[p] });
1762        owners.push(owner);
1763        let raised = entry.position != CssPosition::Static
1764            || entry.parent.is_some_and(|p| !contexts[p] && positioned[p]);
1765        positioned.push(raised);
1766        if index == 0 {
1767            continue;
1768        }
1769        let z = if context {
1770            entry.z_index.unwrap_or(0)
1771        } else {
1772            0
1773        };
1774        let phase = if context && z < 0 {
1775            -1
1776        } else if context && z > 0 {
1777            2
1778        } else if context || raised {
1779            1
1780        } else {
1781            0
1782        };
1783        children[owner].push((phase, z, entry.paint_order, index));
1784    }
1785    for group in &mut children {
1786        group.sort_unstable();
1787    }
1788    fn emit(
1789        index: usize,
1790        contexts: &[bool],
1791        children: &[Vec<(i8, i32, usize, usize)>],
1792        out: &mut Vec<usize>,
1793    ) {
1794        out.push(index);
1795        for &(_, _, _, child) in &children[index] {
1796            if contexts[child] {
1797                emit(child, contexts, children, out);
1798            } else {
1799                out.push(child);
1800            }
1801        }
1802    }
1803    let mut out = Vec::with_capacity(entries.len());
1804    if !entries.is_empty() {
1805        emit(0, &contexts, &children, &mut out);
1806    }
1807    out
1808}

1810impl WebLayout {
1811    pub fn scale(&mut self, scale: f32) {
1812        self.height = (self.height as f32 * scale) as i32;
1813        for b in &mut self.boxes {
1814            b.x = (b.x as f32 * scale) as i32;
1815            b.y = (b.y as f32 * scale) as i32;
1816            b.width = (b.width as f32 * scale) as u32;
1817            b.height = (b.height as f32 * scale) as u32;
1818            for width in b.border.iter_mut().chain(b.padding.iter_mut()) {
1819                *width = (*width as f32 * scale) as u32;
1820            }
1821            crate::doc::scale_lines(&mut b.lines, scale);
1822        }
1823        for b in &mut self.boxes {
1824            b.control_px *= scale;
1825            b.font_size *= scale;
1826            b.line_height *= scale;
1827            b.clip.scale(scale);
1828        }
1829    }
1830    fn paint_boxes(
1831        &self,
1832        view: (i32, i32, u32, u32),
1833        scroll: i32,
1834    ) -> impl Iterator<Item = (&WebBox, i32, i32, (i32, i32, i32, i32))> {
1835        let (vx, vy, vw, vh) = view;
1836        let view_clip = (
1837            vx,
1838            vy,
1839            vx.saturating_add(vw as i32),
1840            vy.saturating_add(vh as i32),
1841        );
1842        self.boxes.iter().filter_map(move |b| {
1843            if !b.visible {
1844                return None;
1845            }
1846            let scroll = if b.fixed { 0 } else { scroll };
