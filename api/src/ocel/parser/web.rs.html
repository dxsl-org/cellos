51fn collect_styles(arena:&DocumentArena,id:NodeId,external:&[(NodeId,String)],sheet:&mut Stylesheet,viewport:(u32,u32)) {
52    let Some(node)=arena.get(id) else{return;};
53    if node.tag()==Some("style") {sheet.append(&arena.get_text_content(id),viewport);}
54    if node.tag()==Some("link") {if let Some((_,css))=external.iter().find(|(n,_)|*n==id) {sheet.append(css,viewport);}}
55    let mut child=node.first_child;while let Some(id)=child{collect_styles(arena,id,external,sheet,viewport);child=arena.get(id).and_then(|n|n.next_sibling);}
56}
57impl<'a> Builder<'a> {
58    fn text_leaf(&mut self,runs:&mut Vec<TextRun>,style:&ComputedStyle,parent:usize)->Result<taffy::NodeId,taffy::TaffyError> {
59        let context=Rc::new(TextContext {runs:core::mem::take(runs),line_height:pixel(style.line_height.max(1.0)),pre:style.pre,wrap_pre:style.wrap_pre,align:style.text_align});
60        let leaf_style=Style{display:Display::Block,..Style::default()};
61        let id=self.tree.new_leaf_with_context(leaf_style,context.clone())?;
62        self.entries.push(Entry{taffy:id,parent:Some(parent),node_id:None,hidden:false,background:None,border_color:style.border_color,text:Some(context),input_data:None,image:None});Ok(id)
63    }
64    fn element(&mut self,id:NodeId,parent:Option<usize>,mut style:ComputedStyle)->Result<taffy::NodeId,taffy::TaffyError> {
65        style.font_size=style.font_size.min(self.max_font);
66        style.line_height=style.line_height.clamp(1.0,self.max_font*4.0);
67        let tid=self.tree.new_leaf(Style::default())?;let index=self.entries.len();
68        self.entries.push(Entry{taffy:tid,parent,node_id:Some(id),hidden:style.layout.display==Display::None,background:style.background,border_color:style.border_color,text:None,input_data:None,image:None});
69        if style.layout.display==Display::None{self.tree.set_style(tid,style.layout)?;return Ok(tid);}
70        if let Some(node)=self.arena.get(id) {
71            if matches!(node.tag(),Some("input"|"textarea"|"button"|"select")) {
72                let kind=if node.tag()==Some("input"){node.get_attribute("type").unwrap_or("text").to_ascii_lowercase()}
73                    else if node.tag()==Some("button"){node.get_attribute("type").unwrap_or("submit").to_ascii_lowercase()}
74                    else if node.tag()==Some("select"){String::from("select")}else{String::from("textarea")};
75                if kind=="hidden" {style.layout.display=Display::None;self.entries[index].hidden=true;}
76                let value=if node.tag()==Some("input"){String::from(node.get_attribute("value").unwrap_or(""))}
77                    else{node.get_attribute("value").map(String::from).unwrap_or_else(||self.arena.get_text_content(id))};
78                let checkbox=kind=="checkbox";
79                style.layout.size.width=length(if checkbox{20.0}else if kind=="submit"||kind=="button"{160.0}else{300.0f32.min(self.viewport.0 as f32)});
80                style.layout.size.height=length(if checkbox{20.0}else if kind=="textarea"{80.0}else{30.0});
81                self.entries[index].input_data=Some(crate::doc::FormInputLayout{input_type:kind,name:String::from(node.get_attribute("name").unwrap_or("")),value,placeholder:String::from(node.get_attribute("placeholder").unwrap_or("")),checked:node.get_attribute("checked").is_some(),is_focused:false});
82                self.tree.set_style(tid,style.layout)?;return Ok(tid);
83            }
84            if node.tag()==Some("img") {
85                if let Some((w,h,pixels))=self.assets.iter().find_map(|asset|match asset {
86                    crate::doc::DocNode::DomImage{node_id,width,height,pixels} if *node_id==id=>Some((*width,*height,pixels.clone())),_=>None}) {
87                    let width=(w as f32).min(self.viewport.0 as f32).max(1.0);
88                    style.layout.size.width=length(width);style.layout.size.height=length(h as f32*width/w.max(1) as f32);
89                    self.entries[index].image=Some((w,h,pixels));
90                } else {
91                    let alt=node.get_attribute("alt").unwrap_or("[Image unavailable]");
92                    let mut runs=Vec::new();append_text(&mut runs,alt,&style,None,id);
93                    let context=Rc::new(TextContext{runs,line_height:pixel(style.line_height),pre:false,wrap_pre:true,align:0});
94                    self.tree.set_node_context(tid,Some(context.clone()))?;self.entries[index].text=Some(context);
95                }
96                self.tree.set_style(tid,style.layout)?;return Ok(tid);
97            }
98        }
99        let mut children=Vec::new();let mut runs=Vec::new();
100        let arena=self.arena;
101        let mut child=arena.get(id).and_then(|n|n.first_child);
102        while let Some(cid)=child {
103            let Some(node)=arena.get(cid) else{break;};
104            match &node.data {
105                NodeData::Text(text)=>append_text(&mut runs,text,&style,None,id),
106                NodeData::Element{..}=>{
107                    let mut child_style=self.sheet.compute(self.arena,cid,Some(&style),self.viewport);
108                    child_style.font_size=child_style.font_size.min(self.max_font);
109                    child_style.line_height=child_style.line_height.clamp(1.0,self.max_font*4.0);
110                    if child_style.layout.display!=Display::None {
111                        if child_style.inline && !contains_replaced(self.arena,cid) {self.inline(cid,&child_style,None,&mut runs);}
112                        else {if !runs.is_empty(){children.push(self.text_leaf(&mut runs,&style,index)?);}children.push(self.element(cid,Some(index),child_style)?);}
113                    }
114                },_=>{}
115            }
116            child=node.next_sibling;
117        }
118        if !runs.is_empty(){children.push(self.text_leaf(&mut runs,&style,index)?);}
119        self.tree.set_children(tid,&children)?;self.tree.set_style(tid,style.layout)?;Ok(tid)
120    }
121    fn inline(&self,id:NodeId,style:&ComputedStyle,link:Option<&str>,runs:&mut Vec<TextRun>) {
122        let Some(node)=self.arena.get(id) else{return;};
123        let link=node.get_attribute("href").or(link);
124        if node.tag()==Some("br") {append_text(runs,"\u{2028}",style,link,id);return;}
125        let mut child=node.first_child;while let Some(cid)=child {let Some(n)=self.arena.get(cid) else{break;};match &n.data{
126            NodeData::Text(text)=>append_text(runs,text,style,link,id),
127            NodeData::Element{..}=>{let mut s=self.sheet.compute(self.arena,cid,Some(style),self.viewport);s.font_size=s.font_size.min(self.max_font);s.line_height=s.line_height.clamp(1.0,self.max_font*4.0);if s.layout.display!=Display::None{self.inline(cid,&s,link,runs);}},_=>{}
128        }child=n.next_sibling;}
129    }

141fn same_run(a:&TextRun,b:&TextRun)->bool {
142    a.node_id==b.node_id && a.px==b.px && a.line_height==b.line_height
143        && a.span.bold==b.span.bold && a.span.italic==b.span.italic

146pub fn append_run(runs:&mut Vec<TextRun>,run:TextRun) {
147    if let Some(last)=runs.last_mut().filter(|last|same_run(last,&run)) {
148        last.span.text.push_str(&run.span.text);
149    } else {runs.push(run);}
150}
151fn next_advance(fonts:&mut TextFonts,line:&TextLine,source:&TextRun,c:char)->f32 {
152    let kern=line.runs.last().filter(|r|same_run(r,source))
153        .and_then(|r|r.span.text.chars().next_back())
154        .map_or(0.0,|left|fonts.kerning(source.face(),left,c,source.px));
155    fonts.advance(source.face(),c,source.px)+kern
156}
157fn push_char(fonts:&mut TextFonts,line:&mut TextLine,source:&TextRun,c:char) {
158    let advance=next_advance(fonts,line,source,c);
159    line.baseline=line.baseline.max(fonts.ascender(source.face(),source.px));
160    line.height=line.height.max(ceil_pixel(source.line_height.max(fonts.line_height(source.face(),source.px))));
161    if let Some(last)=line.runs.last_mut().filter(|r|same_run(r,source)) {
162        last.span.text.push(c);last.width+=advance;
163    } else {
164        let mut span=StyledSpan::plain("");
165        span.bold=source.span.bold;span.italic=source.span.italic;span.code=source.span.code;
166        span.link=source.span.link.clone();span.color=source.span.color;span.text.push(c);
167        let mut run=TextRun::new(span,source.px,source.node_id,source.line_height);
168        run.x=line.width;run.width=advance;line.runs.push(run);
169    }
170    line.width+=advance;
171}
172fn finish_line(fonts:&mut TextFonts,line:&mut TextLine,lines:&mut Vec<TextLine>,height:u32) {
173    // The shared baseline also covers mixed-size ascenders and descenders.
174    let descent=line.runs.iter().map(|r|
175        fonts.line_height(r.face(),r.px)-fonts.ascender(r.face(),r.px))
176        .fold(0.0f32,f32::max);
177    line.height=line.height.max(ceil_pixel(line.baseline+descent));
178    lines.push(core::mem::replace(line,empty_line(height)));
179}
180fn flush_word<'a>(fonts:&mut TextFonts,word:&mut Vec<(&'a TextRun,char)>,line:&mut TextLine,
181    lines:&mut Vec<TextLine>,pending:&mut Option<&'a TextRun>,limit:f32,height:u32) {
182    if word.is_empty(){return;}
183    let mut width=0.0;let mut previous:Option<(&TextRun,char)>=None;
184    for &(r,c) in word.iter() {
185        width+=fonts.advance(r.face(),c,r.px);
186        if let Some((left,l))=previous.filter(|(left,_)|same_run(left,r)) {
187            width+=fonts.kerning(left.face(),l,c,left.px);
188        }
189        previous=Some((r,c));
190    }
191    let source=word[0].0;
192    let space_source=pending.filter(|_|line.width>0.0);
193    let space=space_source.map_or(0.0,|r|next_advance(fonts,line,r,' '));
194    let boundary=if let Some(r)=space_source {
195        if same_run(r,source) {fonts.kerning(source.face(),' ',word[0].1,source.px)}else{0.0}
196    } else if line.width>0.0 {
197        next_advance(fonts,line,source,word[0].1)-fonts.advance(source.face(),word[0].1,source.px)
198    } else {0.0};
199    if line.width>0.0 && line.width+space+width+boundary>limit {
200        finish_line(fonts,line,lines,height);
201    } else if let Some(r)=space_source {push_char(fonts,line,r,' ');}
202    for (source,c) in word.drain(..) {
203        if line.width>0.0 && line.width+next_advance(fonts,line,source,c)>limit {
204            finish_line(fonts,line,lines,height);
205        }
206        push_char(fonts,line,source,c);
207    }
208    *pending=None;
209}

211/// Shared measured wrapping for HTML, Markdown, plain text, and table cells.
212pub fn wrap_runs(fonts:&mut TextFonts,runs:&[TextRun],width:f32,height:u32,
213    pre:bool,wrap_pre:bool,align:u8)->Vec<TextLine> {
214    let limit=width.max(1.0);let mut lines=Vec::new();let mut line=empty_line(height);
215    let mut pending=None;let mut word=Vec::new();
216    for run in runs {for c in normalized_chars(&run.span.text) {
217        if c=='\u{2028}' || (pre && c=='\n') {
218            flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
219            finish_line(fonts,&mut line,&mut lines,height);pending=None;
220        } else if pre {
221            // Expand tabs into measured spaces, rather than attempting a tab glyph.
222            let count=if c=='\t' {4}else{1};let c=if c=='\t' {' '}else{c};
223            for _ in 0..count {
224                if wrap_pre && line.width>0.0 && line.width+next_advance(fonts,&line,run,c)>limit {
225                    finish_line(fonts,&mut line,&mut lines,height);
226                }
227                push_char(fonts,&mut line,run,c);
228            }
229        } else if c.is_whitespace() {
230            flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
231            pending.get_or_insert(run);
232        } else {word.push((run,c));}
233    }}
234    flush_word(fonts,&mut word,&mut line,&mut lines,&mut pending,limit,height);
235    if !line.runs.is_empty() || (pre && lines.is_empty()) {finish_line(fonts,&mut line,&mut lines,height);}
236    for line in &mut lines {
237        let offset=match align {1=>(width-line.width).max(0.0)*0.5,2=>(width-line.width).max(0.0),_=>0.0};
238        for run in &mut line.runs {run.x+=offset;}
239    }
240    lines
241}
242fn wrap(fonts:&mut TextFonts,context:&TextContext,width:f32)->Vec<TextLine> {
243    wrap_runs(fonts,&context.runs,width,context.line_height,context.pre,context.wrap_pre,context.align)
244}
245fn intrinsic_widths(fonts:&mut TextFonts,context:&TextContext)->(f32,f32) {
246    let mut maximum=0.0f32;let mut longest=0.0f32;
247    let mut width=0.0f32;let mut word=0.0f32;
248    let mut previous:Option<(&TextRun,char)>=None;let mut pending:Option<&TextRun>=None;
249    for run in &context.runs {for c in normalized_chars(&run.span.text) {
250        if c=='\u{2028}' || (context.pre && c=='\n') {
251            maximum=maximum.max(width);longest=longest.max(word);
252            width=0.0;word=0.0;previous=None;pending=None;continue;
253        }
254        if !context.pre && c.is_whitespace() {
255            longest=longest.max(word);word=0.0;pending.get_or_insert(run);continue;
256        }
257        if let Some(space_source)=pending.filter(|_|width>0.0) {
258            width+=fonts.advance(space_source.face(),' ',space_source.px);
259            if let Some((left,l))=previous.filter(|(left,_)|same_run(left,space_source)) {
260                width+=fonts.kerning(left.face(),l,' ',left.px);
261            }
262            previous=Some((space_source,' '));
263        }
264        pending=None;
265        let count=if context.pre && c=='\t' {4}else{1};
266        let c=if context.pre && c=='\t' {' '}else{c};
267        for _ in 0..count {
268            let glyph=fonts.advance(run.face(),c,run.px);
269            let kern=previous.filter(|(left,_)|same_run(left,run))
270                .map_or(0.0,|(left,l)|fonts.kerning(left.face(),l,c,left.px));
271            width+=glyph+kern;
272            if c.is_whitespace() {longest=longest.max(word);word=0.0;}
273            else {
274                word+=glyph+if previous.is_some_and(|(_,l)|!l.is_whitespace()) {kern}else{0.0};
275            }
276            previous=Some((run,c));
277        }
278    }}
279    maximum=maximum.max(width);longest=longest.max(word);
280    (maximum,if context.pre && !context.wrap_pre {maximum}else{longest})
281}
282fn measure(fonts:&mut TextFonts,context:&TextContext,known:Size<Option<f32>>,available:Size<AvailableSpace>)->Size<f32> {
283    let (max_width,longest)=intrinsic_widths(fonts,context);
284    let width=known.width.unwrap_or_else(||match available.width {
285        AvailableSpace::Definite(w)=>w.max(0.0),AvailableSpace::MinContent=>longest,AvailableSpace::MaxContent=>max_width,
286    });
287    let lines=wrap(fonts,context,width);
288    let actual=lines.iter().map(|l|l.width).fold(0.0f32,f32::max);
289    Size{width:known.width.unwrap_or(actual.min(width)),height:known.height.unwrap_or(lines.iter().map(|l|l.height).sum::<u32>() as f32)}
290}

292pub fn paint_lines(fonts:&mut TextFonts,surf:&mut ostd::display::ViSurface,lines:&[TextLine],
293    x:f32,y:f32,color:Color,clip:(i32,i32,i32,i32)) {
294    let mut top=y;
295    for line in lines {
296        for run in &line.runs {
297            let run_top=top+line.baseline-fonts.ascender(run.face(),run.px);
298            crate::draw::text(fonts,surf,x+run.x,run_top,&run.span.text,run.face(),run.px,
299                run.span.color.unwrap_or(color),clip);
300        }
301        top+=line.height as f32;
302    }
303}

316pub fn layout_with_nodes_zoom(fonts:&mut TextFonts,arena:&DocumentArena,external:&[(NodeId,String)],width:u32,height:u32,assets:&[crate::doc::DocNode],zoom:f32)->Result<WebLayout,taffy::TaffyError> {
317    let viewport=(width.max(1),height.max(1));let mut sheet=Stylesheet::default();collect_styles(arena,arena.root,external,&mut sheet,viewport);
318    let mut builder=Builder{arena,sheet,viewport,tree:TaffyTree::new(),entries:Vec::new(),assets,max_font:super::css::MAX_FONT_SIZE/zoom.max(1.0)};
319    let mut root_style=builder.sheet.compute(arena,arena.root,None,viewport);
320    root_style.layout.size.width=length(viewport.0 as f32);
321    let root=builder.element(arena.root,None,root_style)?;
322    builder.tree.compute_layout_with_measure(root,Size{width:AvailableSpace::Definite(viewport.0 as f32),height:AvailableSpace::MaxContent},|inputs,_,context,style| {
323        taffy::compute_leaf_layout(inputs,style,|_,_|0.0,|known,available|context.as_ref().map_or(Size::ZERO,|ctx|measure(fonts,ctx,known,available)))
324    })?;
325    let mut result=WebLayout::default();let mut origins:Vec<(i32,i32)>=Vec::new();
326    for entry in &builder.entries {
327        let l=builder.tree.layout(entry.taffy)?;let (px,py)=entry.parent.map_or((0,0),|p|origins[p]);let x=px.saturating_add(l.location.x as i32);let y=py.saturating_add(l.location.y as i32);origins.push((x,y));
328        if entry.hidden {continue;}
329        let border=[pixel(l.border.left),pixel(l.border.right),pixel(l.border.top),pixel(l.border.bottom)];
330        let (bx,by,bw,bh,lines)=if let Some(text)=&entry.text {(x,y,pixel(l.size.width),pixel(l.size.height),wrap(fonts,text,l.size.width))}else{(x,y,pixel(l.size.width),pixel(l.size.height),Vec::new())};
331        result.height=result.height.max(by.saturating_add(bh as i32));
332        result.boxes.push(WebBox{node_id:entry.node_id,x:bx,y:by,width:bw,height:bh,background:entry.background,border_color:entry.border_color,border,lines,input_data:entry.input_data.clone(),image:entry.image.clone(),highlights:Vec::new(),control_px:14.0});
333    }
334    // White default canvas is a web default, independent of the viewer chrome theme.
335    if let Some(root)=result.boxes.first_mut(){if root.background.is_none(){root.background=Some(Color::rgb(255,255,255));}root.height=root.height.max(height);}
336    Ok(result)
337}

339impl WebLayout {
340    pub fn scale(&mut self,scale:f32) {
341        self.height=(self.height as f32*scale) as i32;
342        for b in &mut self.boxes { b.x=(b.x as f32*scale) as i32;b.y=(b.y as f32*scale) as i32;b.width=(b.width as f32*scale) as u32;b.height=(b.height as f32*scale) as u32;
343            for width in &mut b.border {*width=(*width as f32*scale) as u32;} crate::doc::scale_lines(&mut b.lines,scale); }
344        for b in &mut self.boxes {b.control_px*=scale;}
345    }
346    pub fn render(&self,fonts:&mut TextFonts,surf:&mut ostd::display::ViSurface,view:(i32,i32,u32,u32),scroll:i32) {
347        let (vx,vy,vw,vh)=view;let clip=(vx,vy,vx.saturating_add(vw as i32),vy.saturating_add(vh as i32));
348        fn rect(surf:&mut ostd::display::ViSurface,x:i32,y:i32,w:u32,h:u32,color:Color,clip:(i32,i32,i32,i32)) {
349            let x0=x.max(clip.0);let y0=y.max(clip.1);let x1=x.saturating_add(w as i32).min(clip.2);let y1=y.saturating_add(h as i32).min(clip.3);
350            if x1>x0&&y1>y0 {crate::draw::fill_rect(surf,x0,y0,(x1-x0) as u32,(y1-y0) as u32,color);}
351        }
352        for b in &self.boxes {
353            let x=vx.saturating_add(b.x);let y=vy.saturating_add(b.y).saturating_sub(scroll);
354            if y.saturating_add(b.height as i32)<=clip.1||y>=clip.3 {continue;}
355            if let Some(bg)=b.background {rect(surf,x,y,b.width,b.height,bg,clip);}
356            let [left,right,top,bottom]=b.border;
357            rect(surf,x,y,left,b.height,b.border_color,clip);
358            rect(surf,x.saturating_add(b.width.saturating_sub(right) as i32),y,right,b.height,b.border_color,clip);
359            rect(surf,x,y,b.width,top,b.border_color,clip);
360            rect(surf,x,y.saturating_add(b.height.saturating_sub(bottom) as i32),b.width,bottom,b.border_color,clip);
361            if let Some((iw,ih,pixels))=&b.image {
362                crate::draw::draw_image_clipped(surf,x,y,b.width,b.height,pixels,(*iw*4) as usize,*ih,clip);continue;
363            }
364            if let Some(input)=&b.input_data {
365                let color=if input.is_focused{crate::draw::theme::ACCENT_CYAN}else{b.border_color};
366                rect(surf,x,y,b.width,b.height,crate::draw::theme::BG_INPUT,clip);
367                rect(surf,x,y,b.width,1,color,clip);rect(surf,x,y,1,b.height,color,clip);
368                rect(surf,x,y+b.height.saturating_sub(1) as i32,b.width,1,color,clip);
369                rect(surf,x+b.width.saturating_sub(1) as i32,y,1,b.height,color,clip);
370                if input.input_type=="checkbox" {
371                    if input.checked {rect(surf,x+4,y+4,b.width.saturating_sub(8),b.height.saturating_sub(8),crate::draw::theme::ACCENT_CYAN,clip);}
372                } else {
373                    let text=if !matches!(input.input_type.as_str(),"text"|"textarea"|"password"|"email"|"search"|"url"|"tel"|"submit"|"button") {
374                        "Unsupported control (read-only)"
375                    }else if input.value.is_empty(){if input.input_type=="submit"{"Submit"}else{&input.placeholder}}else{&input.value};
376                    let text_clip=(clip.0.max(x),clip.1.max(y),clip.2.min(x+b.width as i32),clip.3.min(y+b.height as i32));
377                    if input.input_type=="password" {
378                        let advance=fonts.measure(FontFace::UiRegular,"•",b.control_px);
379                        for index in 0..input.value.chars().count() {crate::draw::text(fonts,surf,x as f32+6.0+index as f32*advance,(y+4) as f32,"•",FontFace::UiRegular,b.control_px,crate::draw::theme::TEXT_PRIMARY,text_clip);}
380                    }else{crate::draw::text(fonts,surf,(x+6) as f32,(y+4) as f32,text,FontFace::UiRegular,b.control_px,crate::draw::theme::TEXT_PRIMARY,text_clip);}
381                }
382                continue;
383            }
384            for &(offset,height,active) in &b.highlights {
385                rect(surf,x,y+offset,b.lines.iter().map(|line|line.width as u32).max().unwrap_or(1),height,if active{Color::rgb(255,185,70)}else{Color::rgb(255,240,140)},clip);
386            }
387            paint_lines(fonts,surf,&b.lines,x as f32,y as f32,Color::rgb(32,32,32),clip);
388        }
389    }
390    pub fn search(&self,query:&str)->Vec<i32> {
391        let query=crate::doc::fold_text(query.trim());
392        if query.is_empty(){return Vec::new();}
393        let mut result=Vec::new();let mut text=String::new();
394        for b in &self.boxes {for line in &b.lines {text.clear();for run in &line.runs {text.push_str(&run.span.text);}if crate::doc::fold_text(&text).contains(&query){result.push(b.y);break;}}}result
395    }
396    pub fn hit_link(&self,x:i32,y:i32)->Option<String> {
397        for b in self.boxes.iter().rev() {
398            let mut line_y=b.y;
399            for line in &b.lines {
400                if y>=line_y && y<line_y+line.height as i32 {
401                    for run in &line.runs {
402                        let left=b.x as f32+run.x;
403                        if (x as f32)>=left && (x as f32)<left+run.width {
404                            if let Some(link)=&run.span.link {return Some(link.clone());}
405                        }
406                    }
407                }
408                line_y+=line.height as i32;
409            }
410        }
411        None
412    }
413    pub fn hit_node(&self,x:i32,y:i32)->Option<NodeId> {
414        for b in self.boxes.iter().rev() {
415            let mut line_y=b.y;
416            for line in &b.lines {
417                if y>=line_y&&y<line_y.saturating_add(line.height as i32) {
418                    for run in &line.runs {
419                        let left=b.x as f32+run.x;
420                        if (x as f32)>=left && (x as f32)<left+run.width {return run.node_id;}
421                    }
422                }
423                line_y=line_y.saturating_add(line.height as i32);
424            }
425            if x>=b.x&&x<b.x.saturating_add(b.width as i32)&&y>=b.y&&y<b.y.saturating_add(b.height as i32) {if let Some(id)=b.node_id{return Some(id);}}
426        }
427        None
428    }

451fn contains_replaced(arena:&DocumentArena,id:NodeId)->bool {
452    let Some(node)=arena.get(id)else{return false;};
453    if matches!(node.tag(),Some("input"|"textarea"|"button"|"img"|"select")){return true;}
454    let mut child=node.first_child;
455    while let Some(id)=child {if contains_replaced(arena,id){return true;}child=arena.get(id).and_then(|node|node.next_sibling);}
456    false
