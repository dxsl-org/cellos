17#[derive(Copy, Clone, Debug, PartialEq, Eq)]
18pub enum DocFormat {
19    Markdown,
20    Html,
21    PlainText,
22    Source(syntax::Language),
23}
24
25impl DocFormat {
26    pub fn detect_from_url_or_content(url: &str, content: &str) -> Self {
27        let path = url.split(['?', '#']).next().unwrap_or(url);
28        let lower = path.to_ascii_lowercase();
29
30        if let Some(language) = syntax::Language::from_url(path) {
31            return Self::Source(language);
32        }
33        if lower.ends_with(".md") || lower.ends_with(".markdown") {
34            return Self::Markdown;
35        }
36        if lower.ends_with(".html") || lower.ends_with(".htm") {
37            return Self::Html;
38        }
39        if lower.ends_with(".txt")
40            || lower.ends_with(".log")
41        {
42            return Self::PlainText;
43        }
44
45        // Content sniffing if extension is ambiguous or missing
46        let trimmed = content.trim_start();
47        if trimmed.starts_with("<!DOCTYPE")
48            || trimmed.starts_with("<html")
49            || trimmed.starts_with("<head")
50            || trimmed.starts_with("<body")
51        {
52            return Self::Html;
53        }
54
55        if trimmed.starts_with("# ") || trimmed.starts_with("## ") || trimmed.contains("\n# ") {
56            return Self::Markdown;
57        }
58
59        Self::PlainText
60    }
61}

63#[allow(dead_code)]
64pub fn parse_content(
65    format: DocFormat,
66    content: &str,
67) -> (Vec<DocNode>, Vec<html::ScriptSource>, Option<dom_arena::DocumentArena>) {
68    parse_content_with_base(format, content, None)
69}
70
71pub fn parse_content_with_base(
72    format: DocFormat,
73    content: &str,
74    base_url: Option<&str>,
75) -> (Vec<DocNode>, Vec<html::ScriptSource>, Option<dom_arena::DocumentArena>) {
76    match format {
77        DocFormat::Markdown => (markdown::parse_markdown_with_base(content, base_url), Vec::new(), None),
78        DocFormat::Html => {
79            let res = html::parse_html_with_base(content, base_url);
80            (res.nodes, res.scripts, Some(res.arena))
81        }
82        DocFormat::Source(language) => (
83            alloc::vec![DocNode::CodeBlock {
84                lang: String::from(language.name()),
85                lines: content.lines().map(String::from).collect(),
86            }],
87            Vec::new(),
88            None,
89        ),
90        DocFormat::PlainText => {
91            let mut lines = Vec::new();
92            for l in content.lines() {
93                lines.push(String::from(l));
94            }
95            (alloc::vec![DocNode::RawLines { lines }], Vec::new(), None)
96        }
97    }
98}

100pub fn parse_content_with_base_checked(
101    format: DocFormat, content: &str, base_url: Option<&str>,
102) -> Result<(Vec<DocNode>, Vec<html::ScriptSource>, Option<dom_arena::DocumentArena>), String> {
103    crate::budget::validate_text(content)?;
104    if matches!(format, DocFormat::Html) {
105        let output = html::parse_html_with_base_checked(content, base_url)?;
106        return Ok((output.nodes, output.scripts, Some(output.arena)));
107    }
108    if content.lines().count() > crate::budget::MAX_DOM_NODES {
109        return Err(String::from("Document exceeds node/line budget"));
110    }
111    if matches!(format, DocFormat::Markdown) {
112        crate::budget::validate_markdown(content)?;
113        let mut images = Vec::new();
114        let mut fenced=false;
115        for line in content.lines() {
116            let line = line.trim();
117            if line.starts_with("```"){fenced=!fenced;continue;}
118            if fenced{continue;}
119            if line.starts_with("![") && line.ends_with(')') {
120                if let Some(end) = line.find("](") { images.push(&line[end + 2..line.len() - 1]); }
121            }
122        }
123        validate_images(base_url, images.into_iter())?;
124    }
125    Ok(parse_content_with_base(format, content, base_url))
126}
127
128fn validate_images<'a>(base: Option<&str>, urls: impl Iterator<Item=&'a str>) -> Result<(), String> {
129    let mut count = 0usize; let mut bytes = 0usize;
130    for url in urls {
131        count += 1;
132        if count > crate::budget::MAX_IMAGES { return Err(String::from("Document exceeds image count budget")); }
133        if let Some(base) = base {
134            if let Ok(data) = crate::resources::load_binary(base, url) {
135                match crate::budget::image_bytes(&data) {
136                    Ok(size) => {
137                        bytes = bytes.checked_add(size).ok_or_else(|| String::from("Image budget overflow"))?;
138                        if bytes > crate::budget::MAX_IMAGE_TOTAL_BYTES { return Err(String::from("Document exceeds decoded image byte budget")); }
139                    }
140                    Err(error) if error.contains("budget") => return Err(error),
141                    Err(_) => {}
142                }
143            }
144        }
145    }
146    Ok(())
