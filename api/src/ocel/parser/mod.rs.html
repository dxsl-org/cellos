2//! Content parsers and auto-detection for Ocel.
3
4extern crate alloc;
5use alloc::string::String;

7
8pub mod html;

10
11use crate::doc::DocNode;
12
13#[derive(Copy, Clone, Debug, PartialEq, Eq)]
14pub enum DocFormat {
15    Markdown,
16    Html,
17    PlainText,
18}
19
20impl DocFormat {
21    pub fn detect_from_url_or_content(url: &str, content: &str) -> Self {
22        let lower = url.to_ascii_lowercase();
23
24        if lower.ends_with(".md") || lower.ends_with(".markdown") {
25            return Self::Markdown;
26        }
27        if lower.ends_with(".html") || lower.ends_with(".htm") {
28            return Self::Html;
29        }
30        if lower.ends_with(".txt")
31            || lower.ends_with(".log")
32            || lower.ends_with(".rs")
33            || lower.ends_with(".c")
34            || lower.ends_with(".h")
35            || lower.ends_with(".json")
36            || lower.ends_with(".toml")
37        {
38            return Self::PlainText;
39        }
40
41        // Content sniffing if extension is ambiguous or missing
42        let trimmed = content.trim_start();
43        if trimmed.starts_with("<!DOCTYPE")
44            || trimmed.starts_with("<html")
45            || trimmed.starts_with("<head")
46            || trimmed.starts_with("<body")
47        {
48            return Self::Html;
49        }
50
51        if trimmed.starts_with("# ") || trimmed.starts_with("## ") || trimmed.contains("\n# ") {
52            return Self::Markdown;
53        }
54
55        Self::PlainText
56    }
57}
58
59pub fn parse_content(
60    format: DocFormat,
61    content: &str,
62) -> (Vec<DocNode>, Vec<String>, Option<dom_arena::DocumentArena>) {
63    match format {
64        DocFormat::Markdown => (markdown::parse_markdown(content), Vec::new(), None),
65        DocFormat::Html => {
66            let res = html::parse_html(content);
67            (res.nodes, res.scripts, Some(res.arena))
68        }
69        DocFormat::PlainText => {
70            let mut lines = Vec::new();
71            for l in content.lines() {
72                lines.push(String::from(l));
73            }
74            (alloc::vec![DocNode::RawLines { lines }], Vec::new(), None)
75        }
76    }
77}
