17#[derive(Copy, Clone, Debug, PartialEq, Eq)]
18pub enum DocFormat {
19    Markdown,
20    Html,
21    PlainText,
22    Source(syntax::Language),
23    Feed,
24}
25
26impl DocFormat {
27    pub fn detect_from_url_or_content(url: &str, content: &str) -> Self {
28        let path = url.split(['?', '#']).next().unwrap_or(url);
29        let lower = path.to_ascii_lowercase();
30        let trimmed = content.trim_start();
31
32        if lower.ends_with(".rss") || lower.ends_with("/rss") || lower.ends_with(".atom") || lower.ends_with("/feed")
33            || ((lower.ends_with(".xml") || trimmed.starts_with("<?xml"))
34                && (trimmed.contains("<rss") || trimmed.contains("<feed") || trimmed.contains("<channel")))
35        {
36            return Self::Feed;
37        }
38
39        if let Some(language) = syntax::Language::from_url(path) {
40            return Self::Source(language);
41        }
42        if lower.ends_with(".json") || lower.ends_with(".jsonc") {
43            return Self::Source(syntax::Language::Json);
44        }
45        if lower.ends_with(".xml") || lower.ends_with(".svg") {
46            return Self::Source(syntax::Language::Xml);
47        }
48        if lower.ends_with(".md") || lower.ends_with(".markdown") {
49            return Self::Markdown;
50        }
51        if lower.ends_with(".html") || lower.ends_with(".htm") {
52            return Self::Html;
53        }
54        if lower.ends_with(".txt")
55            || lower.ends_with(".log")
56        {
57            return Self::PlainText;
58        }
59
60        // Content sniffing if extension is ambiguous or missing
61        let trimmed = content.trim_start();
62        if ["<!doctype", "<html", "<head", "<body"].iter().any(|prefix| {
63            trimmed.as_bytes().get(..prefix.len())
64                .is_some_and(|start| start.eq_ignore_ascii_case(prefix.as_bytes()))
65        }) {
66            return Self::Html;
67        }
68
69        if trimmed.starts_with("# ") || trimmed.starts_with("## ") || trimmed.contains("\n# ") {
70            return Self::Markdown;
71        }
72
73        Self::PlainText
74    }
75}

77#[allow(dead_code)]
78pub fn parse_content(
79    format: DocFormat,
80    content: &str,
81) -> (Vec<DocNode>, Vec<html::ScriptSource>, Option<dom_arena::DocumentArena>) {
82    parse_content_with_base(format, content, None)
83}
84
85pub fn parse_content_with_base(
86    format: DocFormat,
87    content: &str,
88    base_url: Option<&str>,
89) -> (Vec<DocNode>, Vec<html::ScriptSource>, Option<dom_arena::DocumentArena>) {
90    match format {
91        DocFormat::Markdown => (markdown::parse_markdown_with_base(content, base_url), Vec::new(), None),
92        DocFormat::Html => {
93            let res = html::parse_html_with_base(content, base_url);
94            (res.nodes, res.scripts, Some(res.arena))
95        }
96        DocFormat::Feed => (feed::parse_feed_to_nodes(content, base_url), Vec::new(), None),
97        DocFormat::Source(language) => (
98            alloc::vec![DocNode::CodeBlock {
99                lang: String::from(language.name()),
100                lines: content.lines().map(String::from).collect(),
101            }],
102            Vec::new(),
103            None,
104        ),
105        DocFormat::PlainText => {
106            let mut lines = Vec::new();
107            for l in content.lines() {
108                lines.push(String::from(l));
109            }
110            (alloc::vec![DocNode::RawLines { lines }], Vec::new(), None)
111        }
112    }
113}

115pub fn parse_content_with_base_checked(
116    format: DocFormat, content: &str, base_url: Option<&str>,
117) -> Result<(Vec<DocNode>, Vec<html::ScriptSource>, Option<dom_arena::DocumentArena>), String> {
118    crate::budget::validate_text(content)?;
119    if matches!(format, DocFormat::Html) {
120        let output = html::parse_html_with_base_checked(content, base_url)?;
121        return Ok((output.nodes, output.scripts, Some(output.arena)));
122    }
123    if matches!(format, DocFormat::Feed) {
124        return Ok((feed::parse_feed_to_nodes(content, base_url), Vec::new(), None));
125    }
126    if content.lines().count() > crate::budget::MAX_DOM_NODES {
127        return Err(String::from("Document exceeds node/line budget"));
128    }
129    if matches!(format, DocFormat::Markdown) {
130        crate::budget::validate_markdown(content)?;
131        let mut images = Vec::new();
132        let mut fenced=false;
133        for line in content.lines() {
134            let line = line.trim();
135            if line.starts_with("```"){fenced=!fenced;continue;}
136            if fenced{continue;}
137            if line.starts_with("![") && line.ends_with(')') {
138                if let Some(end) = line.find("](") { images.push(&line[end + 2..line.len() - 1]); }
139            }
140        }
141        validate_images(base_url, images.into_iter())?;
142    }
143    Ok(parse_content_with_base(format, content, base_url))
144}
145
146fn validate_images<'a>(base: Option<&str>, urls: impl Iterator<Item=&'a str>) -> Result<(), String> {
147    let mut count = 0usize; let mut bytes = 0usize;
148    for url in urls {
149        count += 1;
150        if count > crate::budget::MAX_IMAGES { return Err(String::from("Document exceeds image count budget")); }
151        if let Some(base) = base {
152            if let Ok(data) = crate::resources::load_binary(base, url) {
153                match crate::budget::image_bytes(&data) {
154                    Ok(size) => {
155                        bytes = bytes.checked_add(size).ok_or_else(|| String::from("Image budget overflow"))?;
156                        if bytes > crate::budget::MAX_IMAGE_TOTAL_BYTES { return Err(String::from("Document exceeds decoded image byte budget")); }
157                    }
158                    Err(error) if error.contains("budget") => return Err(error),
159                    Err(_) => {}
160                }
161            }
162        }
163    }
164    Ok(())
165}

171    #[test]
172    fn extensionless_html_is_case_insensitive() {
173        for prefix in ["<!doctype html>", "<!DOCTYPE HTML>", "<HtMl>", "<HEAD>", "<BoDy>"] {
174            let content = alloc::format!(" \n{prefix}<p>Example Domain</p>");
175            assert_eq!(DocFormat::detect_from_url_or_content("http://example.com/", &content), DocFormat::Html);
176        }
177        assert_eq!(DocFormat::detect_from_url_or_content("file:///source.txt", "<!doctype html>"), DocFormat::PlainText);
178        assert_eq!(DocFormat::detect_from_url_or_content("file:///app.json", "{}"), DocFormat::Source(super::syntax::Language::Json));
179        assert_eq!(DocFormat::detect_from_url_or_content("file:///tsconfig.jsonc", "{ /* comment */ }"), DocFormat::Source(super::syntax::Language::Json));
180        assert_eq!(DocFormat::detect_from_url_or_content("file:///manifest.xml", "<manifest></manifest>"), DocFormat::Source(super::syntax::Language::Xml));
181        assert_eq!(DocFormat::detect_from_url_or_content("https://news.ycombinator.com/rss", ""), DocFormat::Feed);
182        assert_eq!(DocFormat::detect_from_url_or_content("https://blog.test/feed.atom", ""), DocFormat::Feed);
183        assert_eq!(DocFormat::detect_from_url_or_content("https://news.test/feed.xml", "<rss version='2.0'><channel><title>News</title></channel></rss>"), DocFormat::Feed);
184    }

190                <channel>
191                    <title>CellOS Engineering Feed</title>
192                    <description>Updates from the microkernel development team</description>
193                    <link>https://cellos.org</link>
194                    <item>

196                        <link>https://cellos.org/posts/g2-launch</link>

199                    </item>
200                    <item>
201                        <title>Universal Reader Engine Released</title>
202                        <link>https://cellos.org/posts/reader-engine</link>
203                        <pubDate>Sun, 18 Oct 2026 15:30:00 GMT</pubDate>
204                        <description>Ocel gains Bionic Reading, RSS feed parsing and multi-page stitching.</description>

212        assert!(nodes.iter().any(|n| matches!(n, super::DocNode::ListItem { spans, .. } if spans.iter().any(|s| s.text == "Universal Reader Engine Released"))));
213        assert!(nodes.iter().any(|n| match n {
214            super::DocNode::Paragraph { spans } => spans.iter().any(|s| s.text.contains("Formal memory isolation")),
215            _ => false,
216        }));
217
218        let atom_xml = r#"
219            <feed xmlns="http://www.w3.org/2005/Atom">
220                <title>Hacker News Atom</title>
221                <subtitle>Technology news feed</subtitle>

229        "#;
230        let atom_nodes = super::feed::parse_feed_to_nodes(atom_xml, None);
231        assert!(atom_nodes.iter().any(|n| matches!(n, super::DocNode::Heading { text, .. } if text.contains("Hacker News Atom"))));
232        assert!(atom_nodes.iter().any(|n| matches!(n, super::DocNode::ListItem { spans, .. } if spans.iter().any(|s| s.text == "Show HN: Ocel Fast Reader" && s.link.as_deref() == Some("https://news.ycombinator.com/item?id=99999")))));
233}
