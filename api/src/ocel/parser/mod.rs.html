17#[derive(Copy, Clone, Debug, PartialEq, Eq)]
18pub enum DocFormat {
19    Markdown,
20    Html,
21    PlainText,
22    Source(syntax::Language),
23    Feed,
24}
25
26impl DocFormat {
27    pub fn detect_from_url_or_content(url: &str, content: &str) -> Self {
28        let path = url.split(['?', '#']).next().unwrap_or(url);
29        let lower = path.to_ascii_lowercase();
30        let trimmed = content.trim_start();
31
32        if lower.ends_with(".rss")
33            || lower.ends_with("/rss")
34            || lower.ends_with(".atom")
35            || lower.ends_with("/feed")
36            || ((lower.ends_with(".xml") || trimmed.starts_with("<?xml"))
37                && (trimmed.contains("<rss")
38                    || trimmed.contains("<feed")
39                    || trimmed.contains("<channel")))
40        {
41            return Self::Feed;
42        }
43
44        if let Some(language) = syntax::Language::from_url(path) {
45            return Self::Source(language);
46        }
47        if lower.ends_with(".json") || lower.ends_with(".jsonc") {
48            return Self::Source(syntax::Language::Json);
49        }
50        if lower.ends_with(".xml") || lower.ends_with(".svg") {
51            return Self::Source(syntax::Language::Xml);
52        }
53        if lower.ends_with(".md") || lower.ends_with(".markdown") {
54            return Self::Markdown;
55        }
56        if lower.ends_with(".html") || lower.ends_with(".htm") {
57            return Self::Html;
58        }
59        if lower.ends_with(".txt") || lower.ends_with(".log") {
60            return Self::PlainText;
61        }
62
63        // Content sniffing if extension is ambiguous or missing
64        let trimmed = content.trim_start();
65        if ["<!doctype", "<html", "<head", "<body"]
66            .iter()
67            .any(|prefix| {
68                trimmed
69                    .as_bytes()
70                    .get(..prefix.len())
71                    .is_some_and(|start| start.eq_ignore_ascii_case(prefix.as_bytes()))
72            })
73        {
74            return Self::Html;
75        }
76
77        if trimmed.starts_with("# ") || trimmed.starts_with("## ") || trimmed.contains("\n# ") {
78            return Self::Markdown;
79        }
80
81        Self::PlainText
82    }
83}

85/// Parsed document nodes, discovered scripts, and the DOM arena for HTML documents.
86pub type ParsedContent = (
87    Vec<DocNode>,
88    Vec<html::ScriptSource>,
89    Option<dom_arena::DocumentArena>,
90);
91
92#[allow(dead_code)]
93pub fn parse_content(format: DocFormat, content: &str) -> ParsedContent {
94    parse_content_with_base(format, content, None)
95}
96
97pub fn parse_content_with_base(
98    format: DocFormat,
99    content: &str,
100    base_url: Option<&str>,
101) -> ParsedContent {
102    match format {
103        DocFormat::Markdown => (
104            markdown::parse_markdown_with_base(content, base_url),
105            Vec::new(),
106            None,
107        ),
108        DocFormat::Html => {
109            let res = html::parse_html_with_base(content, base_url);
110            (res.nodes, res.scripts, Some(res.arena))
111        }
112        DocFormat::Feed => (
113            feed::parse_feed_to_nodes(content, base_url),
114            Vec::new(),
115            None,
116        ),
117        DocFormat::Source(language) => (
118            alloc::vec![DocNode::CodeBlock {
119                lang: String::from(language.name()),
120                lines: content.lines().map(String::from).collect(),
121            }],
122            Vec::new(),
123            None,
124        ),
125        DocFormat::PlainText => {
126            let mut lines = Vec::new();
127            for l in content.lines() {
128                lines.push(String::from(l));
129            }
130            (alloc::vec![DocNode::RawLines { lines }], Vec::new(), None)
131        }
132    }
133}

135pub fn parse_content_with_base_checked(
136    format: DocFormat,
137    content: &str,
138    base_url: Option<&str>,
139) -> Result<ParsedContent, String> {
140    crate::budget::validate_text(content)?;
141    if matches!(format, DocFormat::Html) {
142        let output = html::parse_html_with_base_checked(content, base_url)?;
143        return Ok((output.nodes, output.scripts, Some(output.arena)));
144    }
145    if matches!(format, DocFormat::Feed) {
146        return Ok((
147            feed::parse_feed_to_nodes(content, base_url),
148            Vec::new(),
149            None,
150        ));
151    }
152    if content.lines().count() > crate::budget::MAX_DOM_NODES {
153        return Err(String::from("Document exceeds node/line budget"));
154    }
155    if matches!(format, DocFormat::Markdown) {
156        crate::budget::validate_markdown(content)?;
157        let mut images = Vec::new();
158        let mut fenced = false;
159        for line in content.lines() {
160            let line = line.trim();
161            if line.starts_with("```") {
162                fenced = !fenced;
163                continue;
164            }
165            if fenced {
166                continue;
167            }
168            if line.starts_with("![") && line.ends_with(')') {
169                if let Some(end) = line.find("](") {
170                    images.push(&line[end + 2..line.len() - 1]);
171                }
172            }
173        }
174        validate_images(base_url, images.into_iter())?;
175    }
176    Ok(parse_content_with_base(format, content, base_url))
177}
178
179fn validate_images<'a>(
180    base: Option<&str>,
181    urls: impl Iterator<Item = &'a str>,
182) -> Result<(), String> {
183    let mut count = 0usize;
184    let mut bytes = 0usize;
185    for url in urls {
186        count += 1;
187        if count > crate::budget::MAX_IMAGES {
188            return Err(String::from("Document exceeds image count budget"));
189        }
190        if let Some(base) = base {
191            if let Ok(data) = crate::resources::load_binary(base, url) {
192                match crate::budget::image_bytes(&data) {
193                    Ok(size) => {
194                        bytes = bytes
195                            .checked_add(size)
196                            .ok_or_else(|| String::from("Image budget overflow"))?;
197                        if bytes > crate::budget::MAX_IMAGE_TOTAL_BYTES {
198                            return Err(String::from("Document exceeds decoded image byte budget"));
199                        }
200                    }
201                    Err(error) if error.contains("budget") => return Err(error),
202                    Err(_) => {}
203                }
204            }
205        }
206    }
207    Ok(())
208}

214    #[test]
215    fn extensionless_html_is_case_insensitive() {
216        for prefix in [
217            "<!doctype html>",
218            "<!DOCTYPE HTML>",
219            "<HtMl>",
220            "<HEAD>",
221            "<BoDy>",
222        ] {
223            let content = alloc::format!(" \n{prefix}<p>Example Domain</p>");
224            assert_eq!(
225                DocFormat::detect_from_url_or_content("http://example.com/", &content),
226                DocFormat::Html
227            );
228        }
229        assert_eq!(
230            DocFormat::detect_from_url_or_content("file:///source.txt", "<!doctype html>"),
231            DocFormat::PlainText
232        );
233        assert_eq!(
234            DocFormat::detect_from_url_or_content("file:///app.json", "{}"),
235            DocFormat::Source(super::syntax::Language::Json)
236        );
237        assert_eq!(
238            DocFormat::detect_from_url_or_content("file:///tsconfig.jsonc", "{ /* comment */ }"),
239            DocFormat::Source(super::syntax::Language::Json)
240        );
241        assert_eq!(
242            DocFormat::detect_from_url_or_content("file:///manifest.xml", "<manifest></manifest>"),
243            DocFormat::Source(super::syntax::Language::Xml)
244        );
245        assert_eq!(
246            DocFormat::detect_from_url_or_content("https://news.ycombinator.com/rss", ""),
247            DocFormat::Feed
248        );
249        assert_eq!(
250            DocFormat::detect_from_url_or_content("https://blog.test/feed.atom", ""),
251            DocFormat::Feed
252        );
253        assert_eq!(
254            DocFormat::detect_from_url_or_content(
255                "https://news.test/feed.xml",
256                "<rss version='2.0'><channel><title>News</title></channel></rss>"
257            ),
258            DocFormat::Feed
259        );
260    }

309        assert!(atom_nodes.iter().any(|n| matches!(n, super::DocNode::Heading { text, .. } if text.contains("Hacker News Atom"))));
310        assert!(atom_nodes.iter().any(|n| matches!(n, super::DocNode::ListItem { spans, .. } if spans.iter().any(|s| s.text == "Show HN: Ocel Fast Reader" && s.link.as_deref() == Some("https://news.ycombinator.com/item?id=99999")))));
311    }
