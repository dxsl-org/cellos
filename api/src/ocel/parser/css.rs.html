80/// Split only at top level, preserving quoted strings and functions.
81fn split_top(input:&str, delimiter:char) -> Vec<&str> {
82    let mut out=Vec::new(); let mut start=0; let mut depth=0usize; let mut quote=None; let mut escape=false;
83    for (i,c) in input.char_indices() {
84        if escape { escape=false; continue; }
85        if c=='\\' { escape=true; continue; }
86        if let Some(q)=quote { if c==q {quote=None;} continue; }
87        if c=='\''||c=='"' {quote=Some(c);continue;}
88        if c=='('||c=='[' {depth+=1;} else if c==')'||c==']' {depth=depth.saturating_sub(1);}
89        else if (c==delimiter||(delimiter==' '&&c.is_ascii_whitespace())) && depth==0 {out.push(input[start..i].trim());start=i+c.len_utf8();}
90    }
91    out.push(input[start..].trim());out
92}
93fn strip_var(value: &str) -> String {
94    let mut s = String::from(value);
95    while let Some(start) = s.find("var(") {
96        let mut depth = 0usize;
97        let mut close = None;
98        for (i, c) in s[start..].char_indices() {
99            if c == '(' { depth += 1; }
100            else if c == ')' {
101                depth = depth.saturating_sub(1);
102                if depth == 0 { close = Some(start + i); break; }
103            }
104        }
105        let Some(end) = close else { break; };
106        let inner = &s[start + 4..end];
107        if let Some((_, fallback)) = inner.split_once(',') {
108            let fb = String::from(fallback.trim());
109            s.replace_range(start..=end, &fb);
110        } else {
111            s.replace_range(start..=end, "");
112        }
113    }
114    s
115}
116fn declarations(input:&str)->Vec<Declaration> {
117    let mut out=Vec::new();
118    for part in split_top(input,';') {
119        let Some((name,value))=part.split_once(':') else {continue;};
120        let name=name.trim().to_ascii_lowercase(); let mut raw_val=value.trim(); let mut important=false;
121        if let Some(i)=raw_val.rfind('!') { if raw_val[i+1..].trim().eq_ignore_ascii_case("important") { important=true;raw_val=raw_val[..i].trim(); } }
122        let stripped=strip_var(raw_val);
123        let value=stripped.trim();
124        if value.is_empty() { continue; }
125        // Expand spacing shorthands before cascading, so shorthand/longhand compete correctly.
126        if matches!(name.as_str(),"margin"|"padding"|"border-width"|"inset") {
127            let words:Vec<_>=split_top(value,' ').into_iter().filter(|v|!v.is_empty()).collect(); if words.is_empty()||words.len()>4 {continue;}
128            let sides=[words[0],*words.get(1).unwrap_or(&words[0]),*words.get(2).unwrap_or(&words[0]),*words.get(3).unwrap_or(words.get(1).unwrap_or(&words[0]))];
129            for (side,val) in ["top","right","bottom","left"].into_iter().zip(sides) {
130                let key=if name=="border-width" {alloc::format!("border-{}-width",side)} else if name=="inset"{String::from(side)}else{alloc::format!("{}-{}",name,side)};
131                out.push(Declaration{name:key,value:String::from(val),important});
132            }
133        } else if matches!(name.as_str(),"border-top"|"border-bottom"|"border-left"|"border-right") {
134            let side=name.strip_prefix("border-").unwrap();
135            if value=="0" || value=="none" {
136                out.push(Declaration{name:alloc::format!("border-{}-width",side),value:String::from("0px"),important});
137            } else {
138                let words:Vec<_>=value.split_whitespace().collect();
139                if let Some(w)=words.first() {
140                    out.push(Declaration{name:alloc::format!("border-{}-width",side),value:String::from(*w),important});
141                }
142                if let Some(c)=words.iter().find(|w| color(w).is_some()) {
143                    out.push(Declaration{name:String::from("border-color"),value:String::from(*c),important});
144                }
145            }
146        } else if name=="background" {
147            if let Some(col) = extract_background_color(value) {
148                out.push(Declaration{name:String::from("background-color"),value:col,important});
149            }
150        } else if name=="border" {
151            if value=="0" || value=="none" {
152                for side in ["left","right","top","bottom"] {
153                    out.push(Declaration{name:alloc::format!("border-{}-width",side),value:String::from("0px"),important});
154                }
155            } else {
156                let words:Vec<_>=value.split_whitespace().collect();
157                if let Some(w)=words.first() {
158                    for side in ["left","right","top","bottom"] {
159                        out.push(Declaration{name:alloc::format!("border-{}-width",side),value:String::from(*w),important});
160                    }
161                }
162                if let Some(c)=words.iter().find(|w| color(w).is_some()) {
163                    out.push(Declaration{name:String::from("border-color"),value:String::from(*c),important});
164                }
165            }
166        } else if name=="gap"||name=="overflow" {
167            let p:Vec<_>=split_top(value,' ').into_iter().filter(|v|!v.is_empty()).collect();
168            let keys=if name=="gap"{["row-gap","column-gap"]}else{["overflow-x","overflow-y"]};
169            if (1..=2).contains(&p.len()) {for (key,v) in [(keys[0],p[0]),(keys[1],*p.get(1).unwrap_or(&p[0]))] {out.push(Declaration{name:String::from(key),value:String::from(v),important});}}
170        } else if name=="flex" {
171            let p:Vec<_>=split_top(value,' ').into_iter().filter(|v|!v.is_empty()).collect();
172            let expanded=match value {
173                "none"=>Some(("0","0","auto")), "auto"=>Some(("1","1","auto")),
174                _ if (1..=3).contains(&p.len()) && number(p[0]).is_some()=>Some((p[0],*p.get(1).unwrap_or(&"1"),*p.get(2).unwrap_or(&"0%"))),
175                _=>None,
176            };
177            if let Some((grow,shrink,basis))=expanded {for (key,v) in [("flex-grow",grow),("flex-shrink",shrink),("flex-basis",basis)] {out.push(Declaration{name:String::from(key),value:String::from(v),important});}}
178        } else {out.push(Declaration{name,value:String::from(value),important});}
179    }
180    out
181}
182fn extract_background_color(value: &str) -> Option<String> {
183    let v = value.trim();
184    if color(v).is_some() {
185        return Some(String::from(v));
186    }
187    for prefix in ["linear-gradient(", "radial-gradient("] {
188        if let Some(rest) = v.strip_prefix(prefix) {
189            if let Some(inner) = rest.strip_suffix(')') {
190                for part in split_top(inner, ',') {
191                    let part = part.trim();
192                    let first = part.split_whitespace().next().unwrap_or(part);
193                    if color(first).is_some() {
194                        return Some(String::from(first));
195                    }
196                }
197            }
198        }
199    }
200    for word in v.split_whitespace() {
201        if color(word).is_some() {
202            return Some(String::from(word));
203        }
204    }
205    None
206}
207fn strip_comments(input:&str)->String {
208    let mut out=String::new(); let mut it=input.chars().peekable();let mut quote=None;let mut escaped=false;
209    while let Some(c)=it.next() {
210        if escaped {out.push(c);escaped=false;continue;}
211        if c=='\\' {out.push(c);escaped=true;continue;}
212        if let Some(q)=quote {if c==q {quote=None;}out.push(c);continue;}
213        if c=='\''||c=='"' {quote=Some(c);out.push(c);continue;}
214        if c=='/' && it.peek()==Some(&'*') {it.next();while let Some(c)=it.next(){if c=='*'&&it.peek()==Some(&'/'){it.next();break;}}out.push(' ');} else {out.push(c);}
215    } out
216}

218    pub fn append(&mut self, css:&str, viewport:(u32,u32)) { self.parse_rules(&strip_comments(css),viewport); }
219    fn parse_rules(&mut self,css:&str,viewport:(u32,u32)) {
220        let mut start=0;let mut opening=None;let mut depth=0usize;let mut quote=None;let mut escaped=false;
221        for (i,c) in css.char_indices() {
222            if escaped {escaped=false;continue;} if c=='\\' {escaped=true;continue;}
223            if let Some(q)=quote {if c==q {quote=None;}continue;} if c=='\''||c=='"' {quote=Some(c);continue;}
224            if c=='{' {if depth==0 {opening=Some(i);}depth+=1;}
225            else if c=='}' && depth>0 {depth-=1;if depth==0 {let open=opening.take().unwrap();let header=css[start..open].trim();let body=&css[open+1..i];
226                if let Some(query)=header.strip_prefix("@media") {if media_matches(query,viewport) {self.parse_rules(body,viewport);}}
227                else if !header.starts_with('@') {self.rules.push(Rule {selectors:split_top(header,',').into_iter().map(String::from).collect(),declarations:declarations(body)});}
228                start=i+1;
229            }} else if c==';'&&depth==0 {start=i+1;}
230        }
231    }
232    pub fn compute(&self,arena:&DocumentArena,id:NodeId,parent:Option<&ComputedStyle>,viewport:(u32,u32))->ComputedStyle {
233        let Some(node)=arena.get(id) else {return ComputedStyle::initial(parent,"",self.calcs.clone());};
234        let mut style=ComputedStyle::initial(parent,node.tag().unwrap_or(""),self.calcs.clone());
235        let mut winners:BTreeMap<&str,((bool,u32,u32,u32,u32,usize),&str)>=BTreeMap::new();
236        let mut order=0;
237        for rule in &self.rules {
238            let specificity=rule.selectors.iter().filter_map(|s| selector_specificity(arena,id,s)).max();
239            if let Some((a,b,c))=specificity {for d in &rule.declarations {order+=1;if !valid_value(&d.name,&d.value,style.font_size,viewport){continue;}let rank=(d.important,0,a,b,c,order);if winners.get(d.name.as_str()).map_or(true,|(old,_)|rank>=*old) {winners.insert(&d.name,(rank,&d.value));}}}
240        }
241        let inline=node.get_attribute("style").map(declarations).unwrap_or_default();
242        for d in &inline {order+=1;if !valid_value(&d.name,&d.value,style.font_size,viewport){continue;}let rank=(d.important,1,0,0,0,order);if winners.get(d.name.as_str()).map_or(true,|(old,_)|rank>=*old) {winners.insert(&d.name,(rank,&d.value));}}
243        // Font size first: em dimensions and unitless line-height use the computed size.
244        if let Some((_,v))=winners.get("font-size") {if let Some(n)=resolve_length(v,parent.map_or(16.0,|p|p.font_size),viewport) {if n>0.0 {style.font_size=n.min(MAX_FONT_SIZE);style.line_height=style.font_size*1.25;}}}
245        for (name,(_,value)) in winners {apply(&mut style,name,value,viewport);}
246        if style.position==CssPosition::Static {style.layout.inset=Rect{left:auto(),right:auto(),top:auto(),bottom:auto()};}
247        let scrollable=|v|!matches!(v,Overflow::Visible|Overflow::Clip);
248        if scrollable(style.layout.overflow.x)||scrollable(style.layout.overflow.y) {
249            for axis in [&mut style.layout.overflow.x,&mut style.layout.overflow.y] {
250                if *axis==Overflow::Visible{*axis=Overflow::Scroll;}else if *axis==Overflow::Clip{*axis=Overflow::Hidden;}
251            }
252        }
253        if node.get_attribute("hidden").is_some() {style.layout.display=Display::None;}
254        style
255    }
256}

258fn media_matches(query:&str,viewport:(u32,u32))->bool {
259    split_top(query,',').into_iter().any(|alternative| {
260        let q=alternative.trim().to_ascii_lowercase();
261        let parts:Vec<_>=q.split("and").map(str::trim).collect();
262        parts.into_iter().all(|part| {
263            if matches!(part,"all"|"screen"|"only screen"|"") {return true;}
264            let Some(inner)=part.strip_prefix('(').and_then(|p|p.strip_suffix(')')) else {return false;};
265            let Some((feature,value))=inner.split_once(':') else {return false;};
266            if feature.trim()=="orientation" {return match value.trim(){"landscape"=>viewport.0>=viewport.1,"portrait"=>viewport.1>viewport.0,_=>false};}
267            let Some(n)=resolve_length(value.trim(),16.0,viewport) else {return false;};
268            match feature.trim(){"min-width"=>viewport.0 as f32>=n,"max-width"=>viewport.0 as f32<=n,"width"=>viewport.0 as f32==n,"min-height"=>viewport.1 as f32>=n,"max-height"=>viewport.1 as f32<=n,"height"=>viewport.1 as f32==n,_=>false}
269        })
270    })
271}

275fn ident_end(s:&str)->usize {s.char_indices().find(|(_,c)|!c.is_ascii_alphanumeric()&&*c!='-'&&*c!='_').map_or(s.len(),|(i,_)|i)}
276fn selector_parts(selector:&str)->Option<(Vec<&str>,Vec<Relation>)> {
277    let mut parts=Vec::new();let mut relations=Vec::new();let mut start=0;let mut depth=0usize;let mut quote=None;let mut pending=None;
278    for (i,c) in selector.char_indices() {
279        if let Some(q)=quote {if c==q {quote=None;}continue;}
280        if c=='\''||c=='"' {quote=Some(c);continue;}
281        if c=='['||c=='(' {depth+=1;continue;} if c==']'||c==')' {depth=depth.checked_sub(1)?;continue;}
282        if depth==0&&(c.is_ascii_whitespace()||matches!(c,'>'|'+'|'~')) {
283            if start<i {if !parts.is_empty(){relations.push(pending.take().unwrap_or(Relation::Descendant));}parts.push(&selector[start..i]);}
284            if !c.is_ascii_whitespace() {if pending.is_some_and(|r|!matches!(r,Relation::Descendant)){return None;}pending=Some(match c {'>'=>Relation::Child,'+'=>Relation::Adjacent,_=>Relation::Sibling});} else if pending.is_none(){pending=Some(Relation::Descendant);}
285            start=i+c.len_utf8();
286        }
287    }
288    if depth!=0||quote.is_some(){return None;} if start<selector.len(){if !parts.is_empty(){relations.push(pending.take().unwrap_or(Relation::Descendant));}parts.push(&selector[start..]);}
289    else if pending.is_some_and(|r|!matches!(r,Relation::Descendant)){return None;}
290    if parts.is_empty(){None}else{Some((parts,relations))}
291}

298fn matches_nth(index:usize,arg:&str)->bool {
299    let arg=arg.trim();
300    if arg=="odd"{return index%2==1;}
301    if arg=="even"{return index%2==0;}
302    if let Ok(n)=arg.parse::<usize>(){return index==n;}
303    let Some((a_str,b_str))=arg.split_once('n') else {return false;};
304    let a_str=a_str.trim();
305    let a:i64=if a_str.is_empty()||a_str=="+"{1} else if a_str=="-"{-1} else {a_str.parse().unwrap_or(0)};
306    let b:i64=if b_str.trim().is_empty(){0} else {b_str.chars().filter(|c|!c.is_ascii_whitespace()).collect::<String>().parse().unwrap_or(0)};
307    let idx=index as i64;
308    if a==0{idx==b}else{let diff=idx-b;(diff%a==0)&&(diff/a>=0)}
309}
310fn compound(arena:&DocumentArena,id:NodeId,selector:&str)->Option<(u32,u32,u32)> {
311    let node=arena.get(id)?;let tag=node.tag()?;let mut s=selector;let mut spec=(0,0,0);
312    if let Some(rest)=s.strip_prefix('*'){s=rest;}else{let n=ident_end(s);if n>0{if !tag.eq_ignore_ascii_case(&s[..n]){return None;}spec.2+=1;s=&s[n..];}}
313    while !s.is_empty() {
314        let prefix=s.chars().next()?;s=&s[prefix.len_utf8()..];
315        match prefix {
316            '#'|'.'=>{let n=ident_end(s);if n==0{return None;}let value=&s[..n];if prefix=='#'{if node.get_attribute("id")!=Some(value){return None;}spec.0+=1;}else{if !node.get_attribute("class").unwrap_or("").split_whitespace().any(|c|c==value){return None;}spec.1+=1;}s=&s[n..];},
317            '['=>{let n=s.find(']')?;let inner=s[..n].trim();let mut matched=false;
318                if let Some((key,value))=inner.split_once('='){let key=key.trim();let value=value.trim().trim_matches(['\'','"']);if let Some(key)=key.strip_suffix('~'){matched=node.get_attribute(key.trim()).is_some_and(|v|v.split_whitespace().any(|v|v==value));}else{matched=node.get_attribute(key)==Some(value);}}
319                else if ident_end(inner)==inner.len(){matched=node.get_attribute(inner).is_some();}
320                if !matched{return None;}spec.1+=1;s=&s[n+1..];},
321            ':'=>{
322                if s.starts_with(':') { return None; }
323                let n=ident_end(s);let pseudo=&s[..n];s=&s[n..];
324                let matched=match pseudo {
325                    "root"=>node.parent.is_some_and(|p|p==arena.root),
326                    "first-child"=>previous_element(arena,id).is_none(),
327                    "last-child"=>{let mut next=node.next_sibling;let mut last=true;while let Some(i)=next{let n=arena.get(i)?;if n.is_element(){last=false;break;}next=n.next_sibling;}last},
328                    "nth-child"=>{
329                        let rest=s.strip_prefix('(')?;let close=rest.find(')')?;let arg=&rest[..close];s=&rest[close+1..];
330                        let idx=child_element_index(arena,id);matches_nth(idx,arg)
331                    },
332                    "not"|"is"|"where"=>{let rest=s.strip_prefix('(')?;let close=rest.find(')')?;let inner=&rest[..close];s=&rest[close+1..];let result=compound(arena,id,inner);if pseudo=="not"{if result.is_some(){return None;}
333                            if !simple_selector_valid(inner){return None;}
334                        }else if result.is_none(){return None;}
335                        if pseudo!="where"{let specificity=syntactic_specificity(inner)?;spec.0+=specificity.0;spec.1+=specificity.1;spec.2+=specificity.2;}continue;},
336                    _=>return None,
337                };if !matched{return None;}spec.1+=1;
338            },_=>return None,
339        }
340    } Some(spec)
341}
342fn simple_selector_valid(s:&str)->bool {syntactic_specificity(s).is_some()}
343fn syntactic_specificity(mut s:&str)->Option<(u32,u32,u32)> {
344    let mut spec=(0,0,0);if let Some(rest)=s.strip_prefix('*'){s=rest;}else{let n=ident_end(s);if n>0{spec.2+=1;s=&s[n..];}}
345    while !s.is_empty(){let c=s.chars().next()?;s=&s[c.len_utf8()..];match c {
346        '#'|'.'=>{let n=ident_end(s);if n==0{return None;}if c=='#'{spec.0+=1;}else{spec.1+=1;}s=&s[n..];},
347        '['=>{let n=s.find(']')?;spec.1+=1;s=&s[n+1..];},
348        ':'=>{
349            if s.starts_with(':') { return None; }

373impl LinearCalc {

378fn parse_length(s:&str,font:f32,viewport:(u32,u32))->Option<LinearCalc> {
379    let s=s.trim();
380    if let Some(inner)=s.strip_prefix("calc(").and_then(|s|s.strip_suffix(')')) {
381        let mut total=LinearCalc::default();let mut pos=0;let bytes=inner.as_bytes();let mut first=true;
382        while pos<bytes.len() {
383            while pos<bytes.len()&&bytes[pos].is_ascii_whitespace(){pos+=1;}
384            if pos==bytes.len(){return None;}
385            let sign=if bytes[pos]==b'-'{pos+=1;-1.0}else if bytes[pos]==b'+'{pos+=1;1.0}else if first{1.0}else{return None;};
386            while pos<bytes.len()&&bytes[pos].is_ascii_whitespace(){pos+=1;}
387            let start=pos;
388            while pos<bytes.len()&&!matches!(bytes[pos],b'+'|b'-'){pos+=1;}
389            let term=parse_length(&inner[start..pos],font,viewport)?;
390            total.px+=sign*term.px;total.percent+=sign*term.percent;first=false;
391        }
392        return (!first&&total.px.is_finite()&&total.percent.is_finite()).then_some(total);
393    }
394    if let Some(v)=s.strip_suffix('%'){return Some(LinearCalc{px:0.0,percent:number(v)?/100.0,nonnegative:false});}
395    for (unit,factor) in [("rem",16.0),("px",1.0),("em",font),("vw",viewport.0 as f32/100.0),("vh",viewport.1 as f32/100.0)] {
396        if let Some(v)=s.strip_suffix(unit){return Some(LinearCalc{px:number(v)?*factor,percent:0.0,nonnegative:false});}
397    }
398    (s=="0").then_some(LinearCalc::default())
399}
400pub fn resolve_length(s:&str,font:f32,viewport:(u32,u32))->Option<f32>{
401    parse_length(s,font,viewport).filter(|v|v.percent==0.0).map(|v|v.px)

407fn compact_length<T:CssLength>(v:&str,font:f32,viewport:(u32,u32),calcs:&RefCell<Vec<LinearCalc>>)->Option<T> {
408    let value=parse_length(v,font,viewport)?;
409    if v.trim().ends_with('%') { return Some(percent(value.percent)); }
410    if value.percent==0.0 {return Some(length(value.px));}
411    if value.px==0.0 {return Some(percent(value.percent));}
412    let mut pool=calcs.borrow_mut();pool.push(value);
413    // Opaque, nonzero, eight-byte-aligned handles; no dereference or leaks.
414    Some(T::calc((pool.len()*8) as *const ()))
415}

479fn valid_tracks(v:&str,font:f32,viewport:(u32,u32))->bool {
480    if v=="none"{return true;}
481    if let Some(inner)=v.strip_prefix("repeat(").and_then(|v|v.strip_suffix(')')){
482        return inner.split_once(',').is_some_and(|(n,body)|n.trim().parse::<usize>().is_ok_and(|n|(1..=128).contains(&n))&&valid_tracks(body.trim(),font,viewport));
483    }
484    let mut any=false;
485    for word in split_top(v,' ').into_iter().filter(|v|!v.is_empty()) {
486        any=true;
487        if let Some(n)=word.strip_suffix("fr"){if !number(n).is_some_and(|n|n>=0.0){return false;}}
488        else if word!="auto"&&!valid_spacing(word,font,viewport){return false;}
489    }
490    any
491}
492fn tracks(v:&str,font:f32,viewport:(u32,u32),calcs:&RefCell<Vec<LinearCalc>>)->Option<Vec<GridTemplateComponent<String>>> {
493    if v=="none"{return Some(Vec::new());}
494    if let Some(inner)=v.strip_prefix("repeat(").and_then(|v|v.strip_suffix(')')){let (n,body)=inner.split_once(',')?;let count=n.trim().parse::<usize>().ok()?;if !(1..=128).contains(&count){return None;}let part=tracks(body.trim(),font,viewport,calcs)?;let mut out=Vec::new();for _ in 0..count{out.extend(part.iter().cloned());}return Some(out);}
495    let mut out=Vec::new();
496    for word in split_top(v,' ').into_iter().filter(|v|!v.is_empty()) {
497        let track:TrackSizingFunction=if let Some(n)=word.strip_suffix("fr"){let n=number(n)?;if n<0.0{return None;}fr(n)}
498            else if word=="auto"{auto()}else{TrackSizingFunction::from(spacing(word,font,viewport,calcs)?)};
499        out.push(GridTemplateComponent::Single(track));
500    }
501    (!out.is_empty()).then_some(out)
502}
