10const HEADER_LIMIT: usize = 16 * 1024;
11const URL_LIMIT: usize = 2048;
12const LOAD_TICKS: u64 = 45_000; // Normalized monotonic milliseconds, never CPU-loop counts.
13const CLOSE_TICKS: u64 = 3_000;

16#[derive(Clone, Debug)]
17pub struct Url { pub full: String, pub scheme: String, pub host: String, pub port: u16, pub path: String, authority: String, fragment: Option<String> }
18impl Url {
19    pub fn parse(input: &str) -> Result<Self, String> {
20        if input.len() > URL_LIMIT || input.bytes().any(|b| b <= 32 || b == 127) { return Err("Invalid or overlong network URL".into()); }
21        let fragment = input.split_once('#').map(|(_, fragment)| String::from(fragment));
22        let input = input.split('#').next().unwrap_or(input);
23        let (scheme, rest) = input.split_once("://").ok_or("Missing URL scheme")?;
24        let default = match scheme { "http" => 80, "https" => 443, "gemini" => 1965, _ => return Err("Unsupported network scheme".into()) };
25        let split = rest.find(|c| c == '/' || c == '?').unwrap_or(rest.len());
26        let authority = &rest[..split];
27        let (host, port) = match authority.split_once(':') {
28            Some((host, port)) => (host, port.parse::<u16>().map_err(|_| "Invalid URL port")?), None => (authority, default),
29        };
30        if port == 0 || host.is_empty() || host.len() > 253 || !host.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'.') {
31            return Err("Invalid hostname (IPv4/DNS names only; no credentials or IPv6)".into());
32        }
33        let path = match &rest[split..] { "" => String::from("/"), p if p.starts_with('?') => format!("/{}", p), p => String::from(p) };
34        let host = host.to_ascii_lowercase();
35        let authority = if port == default { host.clone() } else { format!("{}:{}", host, port) };
36        Ok(Self { full: format!("{}://{}{}", scheme, authority, path), scheme: scheme.into(), host, port, path, authority, fragment })
37    }
38    fn tls(&self) -> bool { self.scheme != "http" }
39    pub fn origin(&self) -> String { format!("{}://{}", self.scheme, self.authority) }
40    fn document_url(&self) -> String { match &self.fragment { Some(fragment) => format!("{}#{}", self.full, fragment), None => self.full.clone() } }
41    pub fn redirect(&self, location: &str) -> Result<Self, String> {
42        let target = if location.contains("://") { String::from(location) }
43            else if location.starts_with("//") { format!("{}:{}", self.scheme, location) }
44            else {
45                let mut target = crate::resources::resolve(&self.full, location)?;
46                if let Some((_, fragment)) = location.split_once('#') { target.push('#'); target.push_str(fragment); }
47                target
48            };
49        let mut next = Self::parse(&target)?;
50        if !location.contains('#') { next.fragment = self.fragment.clone(); }
51        if self.scheme == "https" && next.scheme != "https" || self.scheme == "gemini" && next.scheme != "gemini" || self.scheme != "gemini" && next.scheme == "gemini" {
52            return Err("Unsafe redirect protocol change".into());
53        }
54        Ok(next)
55    }
56}

58#[derive(Debug, PartialEq, Eq)]
59enum Framing { Headers, Fixed(usize), Eof, ChunkSize, ChunkData(usize), ChunkCrlf, Trailers, Complete }
60/// Incremental decoder: never treats a malformed or short response as success.
61pub struct ResponseDecoder {
62    framing: Framing, pending: Vec<u8>, pub body: Vec<u8>, limit: usize,
63    pub status: u16, pub location: Option<String>, pub media_type: String,
64    gemini: bool, header_bytes: usize, interim: usize,
65}
66impl ResponseDecoder {
67    pub fn new(limit: usize, gemini: bool) -> Self { Self { framing: Framing::Headers, pending: Vec::new(), body: Vec::new(), limit, status: 0, location: None, media_type: String::new(), gemini, header_bytes: 0, interim: 0 } }
68    fn append_pending(&mut self, count: usize) -> Result<(), String> {
69        if count > self.limit.saturating_sub(self.body.len()) { return Err("Response exceeds document/resource byte budget".into()); }
70        let needed = self.body.len() + count;
71        if needed > self.body.capacity() {
72            let capacity = needed.next_power_of_two().min(self.limit);
73            self.body.reserve_exact(capacity - self.body.len());
74        }
75        self.body.extend_from_slice(&self.pending[..count]);
76        self.pending.drain(..count);
77        Ok(())
78    }
79    pub fn complete(&self) -> bool { self.framing == Framing::Complete }
80    pub fn feed(&mut self, bytes: &[u8]) -> Result<(), String> {
81        // Each service reply is <=4096; incomplete framing lines are separately bounded.
82        if self.pending.len().saturating_add(bytes.len()) > HEADER_LIMIT + 4096 { return Err("Response framing exceeds header budget".into()); }
83        let needed = self.pending.len() + bytes.len();
84        if needed > self.pending.capacity() {
85            let capacity = needed.next_power_of_two().min(HEADER_LIMIT + 4096);
86            self.pending.reserve_exact(capacity - self.pending.len());
87        }
88        self.pending.extend_from_slice(bytes);
89        loop {
90            match self.framing {
91                Framing::Headers => {
92                    let delimiter: &[u8] = if self.gemini { b"\r\n" } else { b"\r\n\r\n" };
93                    let Some(end) = self.pending.windows(delimiter.len()).position(|w| w == delimiter) else {
94                        if self.pending.len() > if self.gemini { 1024 } else { HEADER_LIMIT } { return Err("Response headers too large".into()); }
95                        return Ok(());
96                    };
97                    if end + delimiter.len() > if self.gemini { 1024 } else { HEADER_LIMIT } { return Err("Response headers too large".into()); }
98                    let header = core::str::from_utf8(&self.pending[..end]).map_err(|_| "Non-UTF8 protocol header")?.to_string();
99                    self.pending.drain(..end + delimiter.len());
100                    self.headers(&header)?;
101                }
102                Framing::Fixed(left) => {
103                    let count = left.min(self.pending.len());
104                    self.append_pending(count)?;
105                    self.framing = if left == count { Framing::Complete } else { Framing::Fixed(left - count) };
106                    if left != count { return Ok(()); }
107                }
108                Framing::Eof => { self.append_pending(self.pending.len())?; return Ok(()); }
109                Framing::ChunkSize => {
110                    let Some(end) = self.pending.windows(2).position(|w| w == b"\r\n") else {
111                        if self.pending.len() > 1024 { return Err("Chunk size line too large".into()); } return Ok(());
112                    };
113                    if end > 1024 { return Err("Chunk size line too large".into()); }
114                    let line = core::str::from_utf8(&self.pending[..end]).map_err(|_| "Invalid chunk size")?;
115                    let size = line.split(';').next().unwrap_or("");
116                    if size.is_empty() || !size.bytes().all(|b| b.is_ascii_hexdigit()) { return Err("Invalid chunk size".into()); }
117                    let count = usize::from_str_radix(size, 16).map_err(|_| "Chunk size overflow")?;
118                    if count > self.limit.saturating_sub(self.body.len()) { return Err("Chunk exceeds body budget".into()); }
119                    self.pending.drain(..end + 2);
120                    self.framing = if count == 0 { Framing::Trailers } else { Framing::ChunkData(count) };
121                }
122                Framing::ChunkData(left) => {
123                    let count = left.min(self.pending.len());
124                    self.append_pending(count)?;
125                    self.framing = if left == count { Framing::ChunkCrlf } else { Framing::ChunkData(left - count) };
126                    if left != count { return Ok(()); }
127                }
128                Framing::ChunkCrlf => {
129                    if self.pending.len() < 2 { return Ok(()); }
130                    if &self.pending[..2] != b"\r\n" { return Err("Missing chunk delimiter".into()); }
131                    self.pending.drain(..2); self.framing = Framing::ChunkSize;
132                }
133                Framing::Trailers => {
134                    let Some(end) = self.pending.windows(2).position(|w| w == b"\r\n") else {
135                        if self.header_bytes.saturating_add(self.pending.len()) > HEADER_LIMIT { return Err("Chunk trailers too large".into()); }
136                        return Ok(());
137                    };
138                    self.header_bytes += end + 2;
139                    if self.header_bytes > HEADER_LIMIT { return Err("Chunk trailers too large".into()); }
140                    if end != 0 {
141                        let line = core::str::from_utf8(&self.pending[..end]).map_err(|_| "Invalid trailer")?;
142                        if !line.contains(':') || line.starts_with(' ') || line.starts_with('\t') { return Err("Malformed chunk trailer".into()); }
143                    }
144                    self.pending.drain(..end + 2);
145                    if end == 0 { self.framing = Framing::Complete; }
146                }
147                Framing::Complete => { if !self.pending.is_empty() { return Err("Unexpected bytes after framed response".into()); } return Ok(()); }
148            }
149        }
150    }
151    fn headers(&mut self, header: &str) -> Result<(), String> {
152        if self.gemini {
153            let bytes = header.as_bytes();
154            if bytes.len() < 3 || bytes[2] != b' ' || !bytes[..2].iter().all(u8::is_ascii_digit) { return Err("Malformed Gemini status".into()); }
155            self.status = ((bytes[0] - b'0') as u16) * 10 + (bytes[1] - b'0') as u16;
156            let meta = &header[3..];
157            if (30..40).contains(&self.status) { self.location = Some(meta.into()); self.pending.clear(); self.framing = Framing::Complete; }
158            else if (20..30).contains(&self.status) { self.media_type = meta.into(); self.framing = Framing::Eof; }
159            else { return Err(format!("Gemini status {}: {}", self.status, meta)); }
160            return Ok(());
161        }
162        let mut lines = header.split("\r\n");
163        let status = lines.next().ok_or("Missing HTTP status")?;
164        let mut fields = status.splitn(3, ' ');
165        if !matches!(fields.next(), Some("HTTP/1.0" | "HTTP/1.1")) { return Err("Unsupported/malformed HTTP version".into()); }
166        let code = fields.next().ok_or("Missing HTTP status code")?;
167        if code.len() != 3 || !code.bytes().all(|b| b.is_ascii_digit()) { return Err("Malformed HTTP status code".into()); }
168        self.status = code.parse().map_err(|_| "Invalid HTTP status")?;
169        let mut length: Option<usize> = None; let mut transfer = false;
170        for line in lines {
171            if line.starts_with(' ') || line.starts_with('\t') { return Err("Folded HTTP headers unsupported".into()); }
172            let (name, value) = line.split_once(':').ok_or("Malformed HTTP header")?;
173            if name.is_empty() || !name.bytes().all(|b| b.is_ascii_alphanumeric() || b"!#$%&'*+-.^_`|~".contains(&b)) || value.bytes().any(|b| b < 32 && b != 9 || b == 127) { return Err("Invalid HTTP header".into()); }
174            let value = value.trim();
175            if name.eq_ignore_ascii_case("content-length") {
176                if value.is_empty() || !value.bytes().all(|b| b.is_ascii_digit()) { return Err("Invalid Content-Length".into()); }
177                let n = value.parse::<usize>().map_err(|_| "Content-Length overflow")?;
178                if length.is_some_and(|old| old != n) { return Err("Conflicting Content-Length".into()); } length = Some(n);
179            } else if name.eq_ignore_ascii_case("transfer-encoding") {
180                if transfer || !value.eq_ignore_ascii_case("chunked") { return Err("Unsupported Transfer-Encoding".into()); } transfer = true;
181            } else if name.eq_ignore_ascii_case("location") {
182                if self.location.is_some() || value.is_empty() { return Err("Invalid duplicate/empty redirect location".into()); } self.location = Some(value.into());
183            } else if name.eq_ignore_ascii_case("content-type") { self.media_type = value.into(); }
184            else if name.eq_ignore_ascii_case("content-encoding") && !value.eq_ignore_ascii_case("identity") { return Err("Compressed HTTP content unsupported".into()); }
185        }
186        if transfer && length.is_some() { return Err("Ambiguous HTTP framing".into()); }
187        if (100..200).contains(&self.status) {
188            self.interim += 1; if self.interim > 4 || self.status == 101 { return Err("Unsupported HTTP interim response".into()); }
189            self.location = None; self.media_type.clear(); self.framing = Framing::Headers; return Ok(());
190        }
191        if matches!(self.status, 301 | 302 | 303 | 307 | 308) {
192            if self.location.is_none() { return Err("Redirect missing Location".into()); }
193            self.pending.clear(); self.framing = Framing::Complete; return Ok(());
194        }
195        if !(200..300).contains(&self.status) { return Err(format!("HTTP status {}", self.status)); }
196        self.location = None; // Location on a successful response is not a redirect.
197        if length.is_some_and(|n| n > self.limit) { return Err("Content-Length exceeds body budget".into()); }
198        self.framing = if self.status == 204 || self.status == 205 { Framing::Complete }
199            else if transfer { Framing::ChunkSize } else if let Some(n) = length { if n == 0 { Framing::Complete } else { Framing::Fixed(n) } } else { Framing::Eof };
200        Ok(())
201    }
202    pub fn eof(&mut self) -> Result<(), String> {
203        if self.framing == Framing::Eof { self.framing = Framing::Complete; Ok(()) }
204        else if self.complete() { Ok(()) } else { Err("Truncated protocol response before EOF".into()) }
205    }
206}

208/// Narrow exact-operation transport, monomorphized in production and scripted in
209/// regressions so timeouts/cancellation never depend on live network timing.
210pub trait Transport {
211    fn now(&self) -> u64;
212    fn endpoint(&mut self) -> Option<usize>;
213    fn submit(&mut self, endpoint: usize, bytes: &[u8]) -> Result<IpcOpId, IpcSubmitError>;
214    fn take(&mut self, operation: IpcOpId, reply: &mut [u8]) -> Result<IpcTakeResult, ipc::IpcError>;
215    fn cancel(&mut self, operation: IpcOpId);
216    fn close(&mut self, endpoint: usize, cap: u64, tls: bool);
217    fn cleanup(&mut self);
218    fn cleanup_busy(&self) -> bool { false }
219    fn cleanup_error(&self) -> Option<&'static str> { None }
220}
221pub struct KernelTransport { admitted: bool }
222static ACTIVE_JOBS: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
223impl KernelTransport { fn new() -> Self { Self { admitted: false } } }
224impl Drop for KernelTransport {
225    fn drop(&mut self) { if self.admitted { ACTIVE_JOBS.fetch_sub(1, core::sync::atomic::Ordering::Relaxed); } }
226}
227impl Transport for KernelTransport {
228    fn now(&self) -> u64 { sys_get_time_ms().unwrap_or(u64::MAX) }
229    fn endpoint(&mut self) -> Option<usize> {
230        sys_get_time_ms()?;
231        poll_cleanup();
232        if closing_count() >= 16 { return None; }
233        let endpoint = sys_lookup_service(api::syscall::service::NET)?;
234        if ACTIVE_JOBS.fetch_update(core::sync::atomic::Ordering::Relaxed, core::sync::atomic::Ordering::Relaxed,
235            |count| (count < crate::budget::MAX_TABS).then_some(count + 1)).is_err() { return None; }
236        self.admitted = true;
237        Some(endpoint)
238    }
239    fn submit(&mut self, endpoint: usize, bytes: &[u8]) -> Result<IpcOpId, IpcSubmitError> { ipc::submit(endpoint, bytes) }
240    fn take(&mut self, operation: IpcOpId, reply: &mut [u8]) -> Result<IpcTakeResult, ipc::IpcError> { ipc::take(operation, reply) }
241    fn cancel(&mut self, operation: IpcOpId) { let _ = ipc::cancel(operation); }
242    fn close(&mut self, endpoint: usize, cap: u64, tls: bool) { enqueue_close(endpoint, cap, tls); }
243    fn cleanup(&mut self) { poll_cleanup(); }
244    fn cleanup_busy(&self) -> bool { closing_count() >= 16 }
245    fn cleanup_error(&self) -> Option<&'static str> { cleanup_error() }
246}

248#[derive(Clone, Copy, Debug, PartialEq, Eq)]
249enum Stage { Resolve, Connect, WaitConnect, Send, Read, Done }
250#[derive(Debug)]
251pub struct FetchResult { pub url: String, pub body: Vec<u8>, pub media_type: String, pub gemini: bool }
252pub struct FetchJob<T: Transport = KernelTransport> {
253    url: Url, endpoint: usize, deadline: u64, stage: Stage, pending: Option<IpcOpId>, cap: Option<u64>,
254    request: Vec<u8>, sent: usize, sent_chunk: usize, decoder: ResponseDecoder, visited: Vec<String>,
255    same_origin: Option<String>,
256    address: Option<[u8; 4]>,
257    io: T,
258}
259impl FetchJob<KernelTransport> {
260    pub fn new(url: &str, limit: usize) -> Result<Self, String> {
261        recover_unaccepted_cleanup();
262        Self::with_transport(url, limit, None, KernelTransport::new())
263    }
264    pub fn same_origin(url: &str, limit: usize, origin: &str) -> Result<Self, String> { Self::with_transport(url, limit, Some(origin.into()), KernelTransport::new()) }
265}
266impl<T: Transport> FetchJob<T> {
267    fn with_transport(url: &str, limit: usize, same_origin: Option<String>, mut io: T) -> Result<Self, String> {
268        let url = Url::parse(url)?;
269        if same_origin.as_ref().is_some_and(|origin| *origin != url.origin()) { return Err("Cross-origin resource request blocked".into()); }
270        let endpoint = match io.endpoint() {
271            Some(endpoint) => endpoint,
272            None => return Err(io.cleanup_error().unwrap_or("Network service unavailable or cleanup queue busy; retry navigation after cleanup drains").into()),
273        };
274        let mut job = Self { decoder: ResponseDecoder::new(limit, url.scheme == "gemini"), visited: alloc::vec![url.full.clone()], url, endpoint, deadline: io.now().saturating_add(LOAD_TICKS), stage: Stage::Resolve, pending: None, cap: None, request: Vec::new(), sent: 0, sent_chunk: 0, same_origin, address: None, io };
275        job.make_request(); Ok(job)
276    }
277    fn make_request(&mut self) {
278        self.request = if self.url.scheme == "gemini" { format!("{}\r\n", self.url.full).into_bytes() }
279            else { format!("GET {} HTTP/1.1\r\nHost: {}\r\nUser-Agent: Ocel/0.2 (CellOS)\r\nAccept-Encoding: identity\r\nConnection: close\r\n\r\n", self.url.path, self.url.authority).into_bytes() };
280        self.sent = 0;
281    }
282    pub fn progress(&self) -> &'static str { match self.stage { Stage::Resolve => "Resolving host", Stage::Connect | Stage::WaitConnect => "Connecting", Stage::Send => "Sending request", Stage::Read => "Receiving document", Stage::Done => "Finished" } }
283    pub fn resident_bytes(&self) -> usize {
284        self.request.capacity() + self.decoder.body.capacity() + self.decoder.pending.capacity()
285            + self.visited.iter().map(String::capacity).sum::<usize>()
286            + self.url.full.capacity() + self.url.host.capacity() + self.url.path.capacity() + 4096
287    }
288    pub fn set_memory_budget(&mut self, bytes: usize) {
289        let overhead = self.resident_bytes().saturating_sub(self.decoder.body.capacity());
290        self.decoder.limit = self.decoder.limit.min(bytes.saturating_sub(overhead + HEADER_LIMIT + 4096));
291    }
292    fn request_bytes(&mut self, addr: Option<[u8; 4]>, buf: &mut [u8; IPC_BUF_SIZE]) -> Result<usize, String> {
293        let cap = self.cap.unwrap_or(0);
294        if self.stage == Stage::Connect && self.url.tls() {
295            buf[0] = 0x30;
296            buf[9..13].copy_from_slice(&addr.ok_or("Missing resolved address")?);
297            buf[13..15].copy_from_slice(&self.url.port.to_le_bytes()); buf[15..17].copy_from_slice(&(self.url.host.len() as u16).to_le_bytes()); buf[17..17 + self.url.host.len()].copy_from_slice(self.url.host.as_bytes());
298            return Ok(17 + self.url.host.len());
299        }
300        if self.url.tls() && self.stage == Stage::Send {
301            self.sent_chunk = (self.request.len() - self.sent).min(501);
302            buf[0] = 0x31; buf[1..9].copy_from_slice(&cap.to_le_bytes()); buf[9..11].copy_from_slice(&(self.sent_chunk as u16).to_le_bytes()); buf[11..11 + self.sent_chunk].copy_from_slice(&self.request[self.sent..self.sent + self.sent_chunk]);
303            return Ok(11 + self.sent_chunk);
304        }
305        if self.url.tls() && self.stage == Stage::Read {
306            buf[0] = 0x32; buf[1..9].copy_from_slice(&cap.to_le_bytes()); buf[9..13].copy_from_slice(&4094u32.to_le_bytes()); return Ok(13);
307        }
308        let req = match self.stage {
309            Stage::Resolve => NetRequest::Resolve { hostname: &self.url.host },
310            Stage::Connect => NetRequest::TcpConnect { addr: addr.ok_or("Missing address")?, port: self.url.port },
311            Stage::WaitConnect => NetRequest::SocketState { cap_id: cap as u32 },
312            Stage::Send => { self.sent_chunk = (self.request.len() - self.sent).min(480); NetRequest::TcpSendReady { cap_id: cap as u32, data: &self.request[self.sent..self.sent + self.sent_chunk] } },
313            Stage::Read => NetRequest::TcpRecvReady { cap_id: cap as u32, buf_len: 500 },
314            Stage::Done => return Err("Load already finished".into()),
315        };
316        api::ipc::encode(&req, buf).map(|b| b.len()).map_err(|_| "Cannot encode network request".into())
317    }
318    fn submit(&mut self) -> Result<(), String> {
319        let mut bytes = [0u8; IPC_BUF_SIZE];
320        let len = self.request_bytes(self.address, &mut bytes)?;
321        match self.io.submit(self.endpoint, &bytes[..len]) {
322            Ok(operation) => self.pending = Some(operation),
323            Err(IpcSubmitError::Busy) => {},
324            Err(error) => return Err(format!("Network IPC admission failed: {:?}", error)),
325        }
326        Ok(())
327    }
328    pub fn poll(&mut self) -> Poll<Result<FetchResult, String>> {
329        self.io.cleanup();
330        if self.stage == Stage::Done { return Poll::Pending; }
331        if let Some(error) = self.io.cleanup_error() { return self.fail(error.into()); }
332        if self.io.now() >= self.deadline { return self.fail("Network load deadline exceeded".into()); }
333        if self.stage == Stage::Resolve && self.pending.is_none() && self.io.cleanup_busy() { return Poll::Pending; }
334        if self.pending.is_none() { if let Err(e) = self.submit() { return self.fail(e); } return Poll::Pending; }
335        let mut reply = [0u8; IPC_BUF_SIZE];
336        let op = self.pending.unwrap();
337        let len = match self.io.take(op, &mut reply) {
338            Ok(IpcTakeResult::Pending) => return Poll::Pending,
339            Ok(IpcTakeResult::Terminal { status: IpcTerminal::Reply, len }) => { self.pending = None; len },
340            Ok(IpcTakeResult::Terminal { status, .. }) => { self.pending = None; return self.fail(format!("Network IPC terminated: {:?}", status)); },
341            Err(e) => return self.fail(format!("Network IPC take failed: {:?}", e)),
342        };
343        if let Err(e) = self.response(&reply[..len]) { return self.fail(e); }
344        if self.decoder.complete() {
345            if let Some(location) = self.decoder.location.take() {
346                if self.visited.len() > REDIRECT_LIMIT { return self.fail("Too many redirects".into()); }
347                let next = match self.url.redirect(&location) { Ok(next) => next, Err(e) => return self.fail(e) };
348                if self.same_origin.as_ref().is_some_and(|origin| *origin != next.origin()) { return self.fail("Cross-origin resource redirect blocked".into()); }
349                if self.visited.contains(&next.full) { return self.fail("Redirect loop".into()); }
350                self.close(); self.visited.push(next.full.clone()); self.url = next;
351                self.decoder = ResponseDecoder::new(self.decoder.limit, self.url.scheme == "gemini"); self.make_request(); self.stage = Stage::Resolve;
352                return Poll::Pending;
353            }
354            self.close(); self.stage = Stage::Done;
355            return Poll::Ready(Ok(FetchResult { url: self.url.document_url(), body: core::mem::take(&mut self.decoder.body), media_type: core::mem::take(&mut self.decoder.media_type), gemini: self.url.scheme == "gemini" }));
356        }
357        Poll::Pending
358    }
359    fn response(&mut self, bytes: &[u8]) -> Result<(), String> {
360        if self.stage == Stage::Resolve {
361            let addr = match api::ipc::decode::<NetResponse>(bytes) { Ok(NetResponse::Addr(addr)) => addr, _ => return Err(format!("DNS resolution failed for {}", self.url.host)) };
362            self.address = Some(addr); self.stage = Stage::Connect; return Ok(());
363        }
364        if self.stage == Stage::Connect {
365            let cap = if self.url.tls() {
366                u64::from_le_bytes(bytes.try_into().map_err(|_| "Malformed TLS connect reply")?)
367            } else { match api::ipc::decode::<NetResponse>(bytes) { Ok(NetResponse::CapId(cap)) => cap as u64, _ => return Err("TCP connect failed".into()) } };
368            if cap == 0 { return Err("TLS connection failed (network/certificate/time policy)".into()); }
369            self.cap = Some(cap); self.stage = if self.url.tls() { Stage::Send } else { Stage::WaitConnect }; return Ok(());
370        }
371        if self.stage == Stage::WaitConnect {
372            match api::ipc::decode::<NetResponse>(bytes) {
373                Ok(NetResponse::State(0x03)) => self.stage = Stage::Send,
374                Ok(NetResponse::State(0x01 | 0x02)) | Ok(NetResponse::NotReady) => {},
375                _ => return Err("TCP connection failed before establishment".into()),
376            }
377            return Ok(());
378        }
379        if self.stage == Stage::Send {
380            let count = if self.url.tls() { u32::from_le_bytes(bytes.try_into().map_err(|_| "Malformed TLS send reply")?) as usize }
381                else { match api::ipc::decode::<NetResponse>(bytes) { Ok(NetResponse::WriteProgress(n)) => n as usize, Ok(NetResponse::NotReady) => return Ok(()), _ => return Err("TCP write failed".into()) } };
382            if count > self.sent_chunk || count == 0 { return Err("Invalid/failed network write progress".into()); }
383            self.sent += count; if self.sent == self.request.len() { self.stage = Stage::Read; } return Ok(());
384        }
385        if self.url.tls() {
386            if bytes.len() < 2 { return Err("Malformed TLS read reply".into()); }
387            let count = u16::from_le_bytes([bytes[0], bytes[1]]) as usize;
388            if count == u16::MAX as usize { return Err("TLS read failed".into()); }
389            if count == u16::MAX as usize - 1 { return Ok(()); }
390            if bytes.len() != count + 2 { return Err("Truncated TLS read IPC reply".into()); }
391            if count == 0 { self.decoder.eof() } else { self.decoder.feed(&bytes[2..]) }
392        } else {
393            match api::ipc::decode::<NetResponse>(bytes) {
394                Ok(NetResponse::Data(data)) if !data.is_empty() => self.decoder.feed(data),
395                Ok(NetResponse::Eof) => self.decoder.eof(), Ok(NetResponse::NotReady) => Ok(()),
396                _ => Err("TCP read failed (not EOF)".into()),
397            }
398        }
399    }
400    fn fail(&mut self, error: String) -> Poll<Result<FetchResult, String>> { self.cancel(); Poll::Ready(Err(error)) }
401    fn close(&mut self) { if let Some(cap) = self.cap.take() { self.io.close(self.endpoint, cap, self.url.tls()); } }
402    pub fn cancel(&mut self) {
403        if let Some(op) = self.pending.take() {
404            self.io.cancel(op); let mut buf = [0u8; IPC_BUF_SIZE];
405            // A connect reply may have won the cancellation race: reclaim its cap.
406            if let Ok(IpcTakeResult::Terminal { status: IpcTerminal::Reply, len }) = self.io.take(op, &mut buf) {
407                if self.stage == Stage::Connect && self.cap.is_none() {
408                    let cap = if self.url.tls() { buf[..len].try_into().ok().map(u64::from_le_bytes) }
409                        else { match api::ipc::decode::<NetResponse>(&buf[..len]) { Ok(NetResponse::CapId(cap)) => Some(cap as u64), _ => None } };
410                    self.cap = cap.filter(|cap| *cap != 0);
411                }
412            }
413        }
414        self.close(); self.stage = Stage::Done;
415    }
416}
417impl<T: Transport> Drop for FetchJob<T> { fn drop(&mut self) { self.cancel(); } }

419struct Closing {
420    endpoint: usize, cap: u64, tls: bool, op: Option<IpcOpId>, deadline: u64,
421    // true: no request was accepted, safe explicit navigation recovery.
422    // false: terminal outcome uncertain; no accepted close is retried.
423    quarantine: Option<bool>,
424}
425const CLOSE_SLOTS: usize = 16 + crate::budget::MAX_TABS;
426struct CloseQueue { slots: [Option<Closing>; CLOSE_SLOTS], cursor: usize }
427static CLOSING: ostd::sync::Mutex<CloseQueue> = ostd::sync::Mutex::new(CloseQueue { slots: [const { None }; CLOSE_SLOTS], cursor: 0 });
428fn closing_count() -> usize { CLOSING.lock().slots.iter().filter(|slot| slot.is_some()).count() }
429fn cleanup_error() -> Option<&'static str> {
430    let queue = CLOSING.lock();
431    if queue.slots.iter().flatten().any(|job| job.quarantine == Some(false)) {
432        Some("Network socket cleanup outcome is uncertain; restart Ocel before further network navigation")
433    } else if queue.slots.iter().flatten().any(|job| job.quarantine == Some(true)) {
434        Some("Network socket cleanup admission timed out; retry navigation to recover retained sockets")
435    } else { None }
436}
437fn recover_unaccepted_cleanup() {
438    let now = sys_get_time_ms().unwrap_or(u64::MAX);
439    for job in CLOSING.lock().slots.iter_mut().flatten() { job.recover(now); }
440}
441impl Closing {
442    fn recover(&mut self, now: u64) {
443        if self.quarantine == Some(true) {
444            self.quarantine = None; self.deadline = now.saturating_add(CLOSE_TICKS);
445        }
446    }
447    /// true only after the cap is known closed or its provider has died.
448    fn advance<T: Transport>(&mut self, io: &mut T) -> bool {
449        if self.quarantine.is_some() { return false; }
450        let now = io.now();
451        if let Some(operation) = self.op {
452            if now >= self.deadline { io.cancel(operation); }
453            let mut reply = [0u8; IPC_BUF_SIZE];
454            return match io.take(operation, &mut reply) {
455                Ok(IpcTakeResult::Pending) => false,
456                Ok(IpcTakeResult::Terminal { status: IpcTerminal::Reply | IpcTerminal::PeerGone, .. }) => true,
457                _ => { self.op = None; self.quarantine = Some(false); false },
458            };
459        }
460        if now >= self.deadline {
461            self.quarantine = Some(true); return false;
462        }
463        let mut bytes = [0u8; IPC_BUF_SIZE];
464        let len = if self.tls { bytes[0] = 0x15; bytes[1..9].copy_from_slice(&self.cap.to_le_bytes()); 9 }
465            else { match api::ipc::encode(&NetRequest::TcpClose { cap_id: self.cap as u32 }, &mut bytes) {
466                Ok(bytes) => bytes.len(),
467                Err(_) => { self.quarantine = Some(false); return false; }
468            } };
469        match io.submit(self.endpoint, &bytes[..len]) {
470            Ok(operation) => {
471                self.op = Some(operation);
472                // Submission, not queue entry, starts the response deadline.
473                self.deadline = now.saturating_add(CLOSE_TICKS);
474                false
475            }
476            Err(IpcSubmitError::Busy) => false,
477            Err(IpcSubmitError::PeerGone) => true,
478            Err(_) => { self.quarantine = Some(false); false }
479        }
480    }
481}
482fn enqueue_close(endpoint: usize, cap: u64, tls: bool) {
483    {
484        let mut queue = CLOSING.lock();
485        if queue.slots.iter().flatten().any(|job| job.endpoint == endpoint && job.cap == cap) { return; }
486        // Pre-connect admission reserves a slot for every live job, including
487        // cancellation races; no socket can be created when the reserve is full.
488        let slot = queue.slots.iter_mut().find(|slot| slot.is_none()).expect("close admission invariant");
489        *slot = Some(Closing { endpoint, cap, tls, op: None, deadline: sys_get_time_ms().unwrap_or(0).saturating_add(CLOSE_TICKS), quarantine: None });
490    }
491    poll_cleanup();
492}
493pub fn poll_cleanup() {
494    let mut queue = CLOSING.lock();
495    // Four slots per UI turn, fixed storage and round-robin fairness.
496    for _ in 0..4 {
497        let index = queue.cursor; queue.cursor = (index + 1) % CLOSE_SLOTS;
498        let Some(job) = queue.slots[index].as_mut() else { continue; };
499        if job.advance(&mut KernelTransport::new()) { queue.slots[index] = None; }
500    }
501}
