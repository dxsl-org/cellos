24#[derive(Clone, Debug)]
25pub struct Url {
26    pub full: String,
27    pub scheme: String,
28    pub host: String,
29    pub port: u16,
30    pub path: String,
31    authority: String,
32    fragment: Option<String>,
33}
34impl Url {
35    pub fn parse(input: &str) -> Result<Self, String> {
36        if input.len() > URL_LIMIT || input.bytes().any(|b| b <= 32 || b == 127) {
37            return Err("Invalid or overlong network URL".into());
38        }
39        let fragment = input
40            .split_once('#')
41            .map(|(_, fragment)| String::from(fragment));
42        let input = input.split('#').next().unwrap_or(input);
43        let (scheme, rest) = input.split_once("://").ok_or("Missing URL scheme")?;
44        let default = match scheme {
45            "http" => 80,
46            "https" => 443,
47            "gemini" => 1965,
48            _ => return Err("Unsupported network scheme".into()),
49        };
50        let split = rest.find(['/', '?']).unwrap_or(rest.len());
51        let authority = &rest[..split];
52        let (host, port) = match authority.split_once(':') {
53            Some((host, port)) => (host, port.parse::<u16>().map_err(|_| "Invalid URL port")?),
54            None => (authority, default),
55        };
56        if port == 0
57            || host.is_empty()
58            || host.len() > 253
59            || !host
60                .bytes()
61                .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'.')
62        {
63            return Err("Invalid hostname (IPv4/DNS names only; no credentials or IPv6)".into());
64        }
65        let path = match &rest[split..] {
66            "" => String::from("/"),
67            p if p.starts_with('?') => format!("/{}", p),
68            p => String::from(p),
69        };
70        let host = host.to_ascii_lowercase();
71        let authority = if port == default {
72            host.clone()
73        } else {
74            format!("{}:{}", host, port)
75        };
76        Ok(Self {
77            full: format!("{}://{}{}", scheme, authority, path),
78            scheme: scheme.into(),
79            host,
80            port,
81            path,
82            authority,
83            fragment,
84        })
85    }
86    fn tls(&self) -> bool {
87        self.scheme != "http"
88    }
89    pub fn origin(&self) -> String {
90        format!("{}://{}", self.scheme, self.authority)
91    }
92    fn document_url(&self) -> String {
93        match &self.fragment {
94            Some(fragment) => format!("{}#{}", self.full, fragment),
95            None => self.full.clone(),
96        }
97    }
98    pub fn redirect(&self, location: &str) -> Result<Self, String> {
99        let target = if location.contains("://") {
100            String::from(location)
101        } else if location.starts_with("//") {
102            format!("{}:{}", self.scheme, location)
103        } else {
104            let mut target = crate::resources::resolve(&self.full, location)?;
105            if let Some((_, fragment)) = location.split_once('#') {
106                target.push('#');
107                target.push_str(fragment);
108            }
109            target
110        };
111        let mut next = Self::parse(&target)?;
112        if !location.contains('#') {
113            next.fragment = self.fragment.clone();
114        }
115        if self.scheme == "https" && next.scheme != "https"
116            || self.scheme == "gemini" && next.scheme != "gemini"
117            || self.scheme != "gemini" && next.scheme == "gemini"
118        {
119            return Err("Unsafe redirect protocol change".into());
120        }
121        Ok(next)
122    }
123}

126enum Framing {
127    Headers,
128    Fixed(usize),
129    Eof,
130    ChunkSize,
131    ChunkData(usize),
132    ChunkCrlf,
133    Trailers,
134    Complete,
135}
136/// Incremental decoder: never treats a malformed or short response as success.
137pub struct ResponseDecoder {
138    framing: Framing,
139    pending: Vec<u8>,
140    pub body: Vec<u8>,
141    limit: usize,
142    pub status: u16,
143    pub location: Option<String>,
144    pub media_type: String,
145    gemini: bool,
146    header_bytes: usize,
147    interim: usize,
148    pub headers: Vec<(String, String)>,
149    fetch: bool,
150    head: bool,
151}
152impl ResponseDecoder {
153    pub fn new(limit: usize, gemini: bool) -> Self {
154        Self {
155            framing: Framing::Headers,
156            pending: Vec::new(),
157            body: Vec::new(),
158            limit,
159            status: 0,
160            location: None,
161            media_type: String::new(),
162            gemini,
163            header_bytes: 0,
164            interim: 0,
165            headers: Vec::new(),
166            fetch: false,
167            head: false,
168        }
169    }
170    pub fn for_fetch(limit: usize, head: bool) -> Self {
171        Self {
172            fetch: true,
173            head,
174            ..Self::new(limit, false)
175        }
176    }
177    fn append_pending(&mut self, count: usize) -> Result<(), String> {
178        if count > self.limit.saturating_sub(self.body.len()) {
179            return Err("Response exceeds document/resource byte budget".into());
180        }
181        let needed = self.body.len() + count;
182        if needed > self.body.capacity() {
183            let capacity = needed.next_power_of_two().min(self.limit);
184            self.body.reserve_exact(capacity - self.body.len());
185        }
186        self.body.extend_from_slice(&self.pending[..count]);
187        self.pending.drain(..count);
188        Ok(())
189    }
190    pub fn complete(&self) -> bool {
191        self.framing == Framing::Complete
192    }
193    pub fn feed(&mut self, bytes: &[u8]) -> Result<(), String> {
194        // Each service reply is <=4096; incomplete framing lines are separately bounded.
195        if self.pending.len().saturating_add(bytes.len()) > HEADER_LIMIT + 4096 {
196            return Err("Response framing exceeds header budget".into());
197        }
198        let needed = self.pending.len() + bytes.len();
199        if needed > self.pending.capacity() {
200            let capacity = needed.next_power_of_two().min(HEADER_LIMIT + 4096);
201            self.pending.reserve_exact(capacity - self.pending.len());
202        }
203        self.pending.extend_from_slice(bytes);
204        loop {
205            match self.framing {
206                Framing::Headers => {
207                    let delimiter: &[u8] = if self.gemini { b"\r\n" } else { b"\r\n\r\n" };
208                    let Some(end) = self
209                        .pending
210                        .windows(delimiter.len())
211                        .position(|w| w == delimiter)
212                    else {
213                        if self.pending.len() > if self.gemini { 1024 } else { HEADER_LIMIT } {
214                            return Err("Response headers too large".into());
215                        }
216                        return Ok(());
217                    };
218                    if end + delimiter.len() > if self.gemini { 1024 } else { HEADER_LIMIT } {
219                        return Err("Response headers too large".into());
220                    }
221                    let header = core::str::from_utf8(&self.pending[..end])
222                        .map_err(|_| "Non-UTF8 protocol header")?
223                        .to_string();
224                    self.pending.drain(..end + delimiter.len());
225                    self.headers(&header)?;
226                }
227                Framing::Fixed(left) => {
228                    let count = left.min(self.pending.len());
229                    self.append_pending(count)?;
230                    self.framing = if left == count {
231                        Framing::Complete
232                    } else {
233                        Framing::Fixed(left - count)
234                    };
235                    if left != count {
236                        return Ok(());
237                    }
238                }
239                Framing::Eof => {
240                    self.append_pending(self.pending.len())?;
241                    return Ok(());
242                }
243                Framing::ChunkSize => {
244                    let Some(end) = self.pending.windows(2).position(|w| w == b"\r\n") else {
245                        if self.pending.len() > 1024 {
246                            return Err("Chunk size line too large".into());
247                        }
248                        return Ok(());
249                    };
250                    if end > 1024 {
251                        return Err("Chunk size line too large".into());
252                    }
253                    let line = core::str::from_utf8(&self.pending[..end])
254                        .map_err(|_| "Invalid chunk size")?;
255                    let size = line.split(';').next().unwrap_or("");
256                    if size.is_empty() || !size.bytes().all(|b| b.is_ascii_hexdigit()) {
257                        return Err("Invalid chunk size".into());
258                    }
259                    let count =
260                        usize::from_str_radix(size, 16).map_err(|_| "Chunk size overflow")?;
261                    if count > self.limit.saturating_sub(self.body.len()) {
262                        return Err("Chunk exceeds body budget".into());
263                    }
264                    self.pending.drain(..end + 2);
265                    self.framing = if count == 0 {
266                        Framing::Trailers
267                    } else {
268                        Framing::ChunkData(count)
269                    };
270                }
271                Framing::ChunkData(left) => {
272                    let count = left.min(self.pending.len());
273                    self.append_pending(count)?;
274                    self.framing = if left == count {
275                        Framing::ChunkCrlf
276                    } else {
277                        Framing::ChunkData(left - count)
278                    };
279                    if left != count {
280                        return Ok(());
281                    }
282                }
283                Framing::ChunkCrlf => {
284                    if self.pending.len() < 2 {
285                        return Ok(());
286                    }
287                    if &self.pending[..2] != b"\r\n" {
288                        return Err("Missing chunk delimiter".into());
289                    }
290                    self.pending.drain(..2);
291                    self.framing = Framing::ChunkSize;
292                }
293                Framing::Trailers => {
294                    let Some(end) = self.pending.windows(2).position(|w| w == b"\r\n") else {
295                        if self.header_bytes.saturating_add(self.pending.len()) > HEADER_LIMIT {
296                            return Err("Chunk trailers too large".into());
297                        }
298                        return Ok(());
299                    };
300                    self.header_bytes += end + 2;
301                    if self.header_bytes > HEADER_LIMIT {
302                        return Err("Chunk trailers too large".into());
303                    }
304                    if end != 0 {
305                        let line = core::str::from_utf8(&self.pending[..end])
306                            .map_err(|_| "Invalid trailer")?;
307                        if !line.contains(':') || line.starts_with(' ') || line.starts_with('\t') {
308                            return Err("Malformed chunk trailer".into());
309                        }
310                    }
311                    self.pending.drain(..end + 2);
312                    if end == 0 {
313                        self.framing = Framing::Complete;
314                    }
315                }
316                Framing::Complete => {
317                    if !self.pending.is_empty() {
318                        return Err("Unexpected bytes after framed response".into());
319                    }
320                    return Ok(());
321                }
322            }
323        }
324    }
325    fn headers(&mut self, header: &str) -> Result<(), String> {
326        if self.gemini {
327            let bytes = header.as_bytes();
328            if bytes.len() < 3 || bytes[2] != b' ' || !bytes[..2].iter().all(u8::is_ascii_digit) {
329                return Err("Malformed Gemini status".into());
330            }
331            self.status = ((bytes[0] - b'0') as u16) * 10 + (bytes[1] - b'0') as u16;
332            let meta = &header[3..];
333            if (30..40).contains(&self.status) {
334                self.location = Some(meta.into());
335                self.pending.clear();
336                self.framing = Framing::Complete;
337            } else if (20..30).contains(&self.status) {
338                self.media_type = meta.into();
339                self.framing = Framing::Eof;
340            } else {
341                return Err(format!("Gemini status {}: {}", self.status, meta));
342            }
343            return Ok(());
344        }
345        let mut lines = header.split("\r\n");
346        let status = lines.next().ok_or("Missing HTTP status")?;
347        let mut fields = status.splitn(3, ' ');
348        if !matches!(fields.next(), Some("HTTP/1.0" | "HTTP/1.1")) {
349            return Err("Unsupported/malformed HTTP version".into());
350        }
351        let code = fields.next().ok_or("Missing HTTP status code")?;
352        if code.len() != 3 || !code.bytes().all(|b| b.is_ascii_digit()) {
353            return Err("Malformed HTTP status code".into());
354        }
355        self.status = code.parse().map_err(|_| "Invalid HTTP status")?;
356        if !(100..600).contains(&self.status) {
357            return Err("Invalid HTTP status range".into());
358        }
359        let mut length: Option<usize> = None;
360        let mut transfer = false;
361        for line in lines {
362            if line.starts_with(' ') || line.starts_with('\t') {
363                return Err("Folded HTTP headers unsupported".into());
364            }
365            let (name, value) = line.split_once(':').ok_or("Malformed HTTP header")?;
366            if name.is_empty()
367                || !name
368                    .bytes()
369                    .all(|b| b.is_ascii_alphanumeric() || b"!#$%&'*+-.^_`|~".contains(&b))
370                || value.bytes().any(|b| b < 32 && b != 9 || b == 127)
371            {
372                return Err("Invalid HTTP header".into());
373            }
374            let value = value.trim();
375            if self.headers.len() >= 64 {
376                return Err("Too many HTTP headers".into());
377            }
378            self.headers.push((name.to_ascii_lowercase(), value.into()));
379            if name.eq_ignore_ascii_case("content-length") {
380                if value.is_empty() || !value.bytes().all(|b| b.is_ascii_digit()) {
381                    return Err("Invalid Content-Length".into());
382                }
383                let n = value
384                    .parse::<usize>()
385                    .map_err(|_| "Content-Length overflow")?;
386                if length.is_some_and(|old| old != n) {
387                    return Err("Conflicting Content-Length".into());
388                }
389                length = Some(n);
390            } else if name.eq_ignore_ascii_case("transfer-encoding") {
391                if transfer || !value.eq_ignore_ascii_case("chunked") {
392                    return Err("Unsupported Transfer-Encoding".into());
393                }
394                transfer = true;
395            } else if name.eq_ignore_ascii_case("location") {
396                if self.location.is_some() || value.is_empty() {
397                    return Err("Invalid duplicate/empty redirect location".into());
398                }
399                self.location = Some(value.into());
400            } else if name.eq_ignore_ascii_case("content-type") {
401                self.media_type = value.into();
402            } else if name.eq_ignore_ascii_case("content-encoding")
403                && !self.head
404                && !matches!(self.status, 204 | 205 | 304)
405                && !value.eq_ignore_ascii_case("identity")
406            {
407                return Err("Compressed HTTP content unsupported".into());
408            }
409        }
410        if transfer && length.is_some() {
411            return Err("Ambiguous HTTP framing".into());
412        }
413        if (100..200).contains(&self.status) {
414            self.interim += 1;
415            if self.interim > 4 || self.status == 101 {
416                return Err("Unsupported HTTP interim response".into());
417            }
418            self.location = None;
419            self.media_type.clear();
420            self.headers.clear();
421            self.framing = Framing::Headers;
422            return Ok(());
423        }
424        if matches!(self.status, 301 | 302 | 303 | 307 | 308) {
425            if self.location.is_none() {
426                return Err("Redirect missing Location".into());
427            }
428            self.pending.clear();
429            self.framing = Framing::Complete;
430            return Ok(());
431        }
432        if !self.fetch && !(200..300).contains(&self.status) {
433            return Err(format!("HTTP status {}", self.status));
434        }
435        self.location = None; // Location on a successful response is not a redirect.
436        if !self.head
437            && !matches!(self.status, 204 | 205 | 304)
438            && length.is_some_and(|n| n > self.limit)
439        {
440            return Err("Content-Length exceeds body budget".into());
441        }
442        self.framing = if self.head || matches!(self.status, 204 | 205 | 304) {
443            Framing::Complete
444        } else if transfer {
445            Framing::ChunkSize
446        } else if let Some(n) = length {
447            if n == 0 {
448                Framing::Complete
449            } else {
450                Framing::Fixed(n)
451            }
452        } else {
453            Framing::Eof
454        };
455        Ok(())
456    }
457    pub fn eof(&mut self) -> Result<(), String> {
458        if self.framing == Framing::Eof {
459            self.framing = Framing::Complete;
460            Ok(())
461        } else if self.complete() {
462            Ok(())
463        } else {
464            Err("Truncated protocol response before EOF".into())
465        }
466    }
467}

469/// Narrow exact-operation transport, monomorphized in production and scripted in
470/// regressions so timeouts/cancellation never depend on live network timing.
471pub trait Transport {
472    fn now(&self) -> u64;
473    fn endpoint(&mut self) -> Option<usize>;
474    fn submit(&mut self, endpoint: usize, bytes: &[u8]) -> Result<IpcOpId, IpcSubmitError>;
475    fn take(
476        &mut self,
477        operation: IpcOpId,
478        reply: &mut [u8],
479    ) -> Result<IpcTakeResult, ipc::IpcError>;
480    fn cancel(&mut self, operation: IpcOpId);
481    fn close(&mut self, endpoint: usize, cap: u64, tls: bool);
482    fn cleanup(&mut self);
483    fn cleanup_busy(&self) -> bool {
484        false
485    }
486    fn cleanup_error(&self) -> Option<&'static str> {
487        None
488    }
489}

508static ACTIVE_JOBS: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
509static DNS_CACHE: ostd::sync::Mutex<alloc::collections::BTreeMap<String, ([u8; 4], u64)>> =
510    ostd::sync::Mutex::new(alloc::collections::BTreeMap::new());
511
512fn cached_resolve(host: &str, now: u64) -> Option<[u8; 4]> {
513    let cache = DNS_CACHE.lock();
514    if let Some((addr, expiry)) = cache.get(host) {
515        if now < *expiry {
516            return Some(*addr);
517        }
518    }
519    None
520}
521
522fn cache_dns(host: String, addr: [u8; 4], now: u64) {
523    let mut cache = DNS_CACHE.lock();
524    if cache.len() >= 64 {
525        cache.clear();
526    }
527    cache.insert(host, (addr, now.saturating_add(300_000)));
528}

533    }

535impl Drop for KernelTransport {
536    fn drop(&mut self) {
537        if self.admitted {
538            ACTIVE_JOBS.fetch_sub(1, core::sync::atomic::Ordering::Relaxed);
539        }
540    }
541}
542impl Transport for KernelTransport {
543    fn now(&self) -> u64 {
544        sys_get_time_ms().unwrap_or(u64::MAX)
545    }
546    fn endpoint(&mut self) -> Option<usize> {
547        sys_get_time_ms()?;
548        poll_cleanup();
549        if closing_count() >= CLOSE_LIMIT {
550            return None;
551        }
552        let endpoint = provider_tid(api::syscall::service::NET)?;
553        if ACTIVE_JOBS
554            .fetch_update(
555                core::sync::atomic::Ordering::Relaxed,
556                core::sync::atomic::Ordering::Relaxed,
557                |count| (count < MAX_CONCURRENT_NET_JOBS).then_some(count + 1),
558            )
559            .is_err()
560        {
561            return None;
562        }
563        self.admitted = true;
564        Some(endpoint)
565    }
566    fn submit(&mut self, endpoint: usize, bytes: &[u8]) -> Result<IpcOpId, IpcSubmitError> {
567        ipc::submit(endpoint, bytes)
568    }
569    fn take(
570        &mut self,
571        operation: IpcOpId,
572        reply: &mut [u8],
573    ) -> Result<IpcTakeResult, ipc::IpcError> {
574        ipc::take(operation, reply)
575    }
576    fn cancel(&mut self, operation: IpcOpId) {
577        let _ = ipc::cancel(operation);
578    }
579    fn close(&mut self, endpoint: usize, cap: u64, tls: bool) {
580        enqueue_close(endpoint, cap, tls);
581    }
582    fn cleanup(&mut self) {
583        poll_cleanup();
584    }
585    fn cleanup_busy(&self) -> bool {
586        closing_count() >= CLOSE_LIMIT
587    }
588    fn cleanup_error(&self) -> Option<&'static str> {
589        cleanup_error()
590    }
591}

631impl FetchJob<KernelTransport> {
632    pub fn new(url: &str, limit: usize) -> Result<Self, String> {
633        recover_unaccepted_cleanup();
634        Self::with_transport(url, limit, None, KernelTransport::new())
635    }
636    #[allow(dead_code)]
637    pub fn same_origin(url: &str, limit: usize, origin: &str) -> Result<Self, String> {
638        Self::with_transport(url, limit, Some(origin.into()), KernelTransport::new())
639    }
640    pub fn same_site(url: &str, limit: usize, base_url: &str) -> Result<Self, String> {
641        let base = Url::parse(base_url)?;
642        let target = Url::parse(url)?;
643        if base.scheme == "https" && target.scheme == "http" {
644            return Err("Insecure mixed-content request blocked".into());
645        }
646        if !crate::resources::is_same_site(&base.host, &target.host) {
647            return Err("Cross-site resource request blocked".into());
648        }
649        let mut job = Self::new(url, limit)?;
650        job.same_site = Some(base.host);
651        Ok(job)
652    }
653    /// Native browser controller follows redirects so cookies and CORS are checked per hop.
654    pub fn browser(
655        url: &str,
656        limit: usize,
657        method: &str,
658        headers: &[(&str, &str)],
659        body: &[u8],
660    ) -> Result<Self, String> {
661        recover_unaccepted_cleanup();
662        let mut job = Self::prepare(url, limit, None, KernelTransport::new())?;
663        job.configure_browser(method, headers, body)?;
664        Ok(job)
665    }
666}
667impl<T: Transport> FetchJob<T> {
668    fn with_transport(
669        url: &str,
670        limit: usize,
671        same_origin: Option<String>,
672        io: T,
673    ) -> Result<Self, String> {
674        let mut job = Self::prepare(url, limit, same_origin, io)?;
675        job.make_request();
676        Ok(job)
677    }
678    fn prepare(
679        url: &str,
680        limit: usize,
681        same_origin: Option<String>,
682        mut io: T,
683    ) -> Result<Self, String> {
684        let url = Url::parse(url)?;
685        if same_origin
686            .as_ref()
687            .is_some_and(|origin| *origin != url.origin())
688        {
689            return Err("Cross-origin resource request blocked".into());
690        }
691        let endpoint = match io.endpoint() {
692            Some(endpoint) => endpoint,
693            None => return Err(io.cleanup_error().unwrap_or("Network service unavailable or cleanup queue busy; retry navigation after cleanup drains").into()),
694        };
695        let now = io.now();
696        let cached_addr = cached_resolve(&url.host, now);
697        let stage = if cached_addr.is_some() {
698            Stage::Connect
699        } else {
700            Stage::Resolve
701        };
702        Ok(Self {
703            decoder: ResponseDecoder::new(limit, url.scheme == "gemini"),
704            visited: alloc::vec![url.full.clone()],
705            url,
706            endpoint,
707            deadline: now.saturating_add(LOAD_TICKS),
708            stage,
709            pending: None,
710            cap: None,
711            request: Vec::new(),
712            sent: 0,
713            sent_chunk: 0,
714            same_origin,
715            same_site: None,
716            manual_redirects: false,
717            address: cached_addr,
718            reads: 0,
719            first_read_ms: 0,
720            io,
721        })
722    }
723    fn configure_browser(
724        &mut self,
725        method: &str,
726        headers: &[(&str, &str)],
727        body: &[u8],
728    ) -> Result<(), String> {
729        if !matches!(self.url.scheme.as_str(), "http" | "https")
730            || !matches!(method, "GET" | "HEAD" | "POST")
731            || method != "POST" && !body.is_empty()
732        {
733            return Err("Unsupported browser HTTP method/body".into());
734        }
735        if body.len() > 64 * 1024 || headers.len() > 32 {
736            return Err("Browser HTTP request exceeds budget".into());
737        }
738        let header_bytes = headers.iter().try_fold(0usize, |total, (name, value)| {
739            if name.is_empty()
740                || !name
741                    .bytes()
742                    .all(|b| b.is_ascii_alphanumeric() || b"!#$%&'*+-.^_`|~".contains(&b))
743                || value.bytes().any(|b| b < 32 || b == 127)
744                || [
745                    "host",
746                    "connection",
747                    "content-length",
748                    "transfer-encoding",
749                    "accept-encoding",
750                ]
751                .iter()
752                .any(|n| name.eq_ignore_ascii_case(n))
753            {
754                return Err(String::from("Invalid browser HTTP header"));
755            }
756            total
757                .checked_add(name.len() + value.len() + 4)
758                .filter(|n| *n <= 8192)
759                .ok_or_else(|| String::from("Browser HTTP headers exceed budget"))
760        })?;
761        use core::fmt::Write;
762        let mut request = String::with_capacity(
763            method.len()
764                + self.url.path.len()
765                + self.url.authority.len()
766                + header_bytes
767                + body.len()
768                + 160,
769        );
770        write!(&mut request, "{} {} HTTP/1.1\r\nHost: {}\r\nUser-Agent: Ocel/0.2 (CellOS)\r\nAccept-Encoding: identity\r\nConnection: close\r\n", method, self.url.path, self.url.authority).map_err(|_| "Cannot encode browser request")?;
771        for (name, value) in headers {
772            request.push_str(name);
773            request.push_str(": ");
774            request.push_str(value);
775            request.push_str("\r\n");
776        }
777        if method == "POST" {
778            write!(&mut request, "Content-Length: {}\r\n", body.len())
779                .map_err(|_| "Cannot encode browser request")?;
780        }
781        request.push_str("\r\n");
782        self.request = request.into_bytes();
783        self.request.extend_from_slice(body);
784        self.decoder = ResponseDecoder::for_fetch(self.decoder.limit, method == "HEAD");
785        self.manual_redirects = true;
786        Ok(())
787    }
788    fn make_request(&mut self) {
789        self.request = if self.url.scheme == "gemini" {
790            format!("{}\r\n", self.url.full).into_bytes()
791        } else {
792            format!("GET {} HTTP/1.1\r\nHost: {}\r\nUser-Agent: Ocel/0.2 (CellOS)\r\nAccept-Encoding: identity\r\nConnection: close\r\n\r\n", self.url.path, self.url.authority).into_bytes()
793        };
794        self.sent = 0;
795    }
796    pub fn progress(&self) -> &'static str {
797        match self.stage {
798            Stage::Resolve => "Resolving host",
799            Stage::Connect | Stage::WaitConnect => "Connecting",
800            Stage::Send => "Sending request",
801            Stage::Read => "Receiving document",
802            Stage::Done => "Finished",
803        }
804    }
805    pub fn resident_bytes(&self) -> usize {
806        self.request.capacity()
807            + self.decoder.body.capacity()
808            + self.decoder.pending.capacity()
809            + self.visited.iter().map(String::capacity).sum::<usize>()
810            + self.url.full.capacity()
811            + self.url.host.capacity()
812            + self.url.path.capacity()
813            + 4096
814            + self.decoder.headers.capacity() * core::mem::size_of::<(String, String)>()
815            + self
816                .decoder
817                .headers
818                .iter()
819                .map(|(k, v)| k.capacity() + v.capacity())
820                .sum::<usize>()
821    }
822    pub fn set_memory_budget(&mut self, bytes: usize) {
823        let overhead = self
824            .resident_bytes()
825            .saturating_sub(self.decoder.body.capacity());
826        self.decoder.limit = self
827            .decoder
828            .limit
829            .min(bytes.saturating_sub(overhead + HEADER_LIMIT + 4096));
830    }
831    fn request_bytes(
832        &mut self,
833        addr: Option<[u8; 4]>,
834        buf: &mut [u8; IPC_BUF_SIZE],
835    ) -> Result<usize, String> {
836        let cap = self.cap.unwrap_or(0);
837        if self.stage == Stage::Connect && self.url.tls() {
838            buf[0] = tls::CONNECT;
839            buf[9..13].copy_from_slice(&addr.ok_or("Missing resolved address")?);
840            buf[13..15].copy_from_slice(&self.url.port.to_le_bytes());
841            buf[15..17].copy_from_slice(&(self.url.host.len() as u16).to_le_bytes());
842            buf[tls::CONNECT_HEADER_LEN..tls::CONNECT_HEADER_LEN + self.url.host.len()]
843                .copy_from_slice(self.url.host.as_bytes());
844            return Ok(tls::CONNECT_HEADER_LEN + self.url.host.len());
845        }
846        if self.url.tls() && self.stage == Stage::Send {
847            self.sent_chunk = (self.request.len() - self.sent).min(tls::MAX_SEND);
848            buf[0] = tls::SEND;
849            buf[1..9].copy_from_slice(&cap.to_le_bytes());
850            buf[9..11].copy_from_slice(&(self.sent_chunk as u16).to_le_bytes());
851            buf[tls::SEND_HEADER_LEN..tls::SEND_HEADER_LEN + self.sent_chunk]
852                .copy_from_slice(&self.request[self.sent..self.sent + self.sent_chunk]);
853            return Ok(tls::SEND_HEADER_LEN + self.sent_chunk);
854        }
855        if self.url.tls() && self.stage == Stage::Read {
856            buf[0] = tls::RECV;
857            buf[1..9].copy_from_slice(&cap.to_le_bytes());
858            buf[9..13].copy_from_slice(&(tls::MAX_RECV_DATA as u32).to_le_bytes());
859            return Ok(tls::RECV_REQUEST_LEN);
860        }
861        let req = match self.stage {
862            Stage::Resolve => NetRequest::Resolve {
863                hostname: &self.url.host,
864            },
865            Stage::Connect => NetRequest::TcpConnect {
866                addr: addr.ok_or("Missing address")?,
867                port: self.url.port,
868            },
869            Stage::WaitConnect => NetRequest::SocketState { cap_id: cap as u32 },
870            Stage::Send => {
871                self.sent_chunk = (self.request.len() - self.sent).min(480);
872                NetRequest::TcpSendReady {
873                    cap_id: cap as u32,
874                    data: &self.request[self.sent..self.sent + self.sent_chunk],
875                }
876            }
877            Stage::Read => NetRequest::TcpRecvReady {
878                cap_id: cap as u32,
879                buf_len: 500,
880            },
881            Stage::Done => return Err("Load already finished".into()),
882        };
883        api::ipc::encode(&req, buf)
884            .map(|b| b.len())
885            .map_err(|_| "Cannot encode network request".into())
886    }
887    fn submit(&mut self) -> Result<(), String> {
888        let mut bytes = [0u8; IPC_BUF_SIZE];
889        let len = self.request_bytes(self.address, &mut bytes)?;
890        match self.io.submit(self.endpoint, &bytes[..len]) {
891            Ok(operation) => self.pending = Some(operation),
892            Err(IpcSubmitError::Busy) => {}
893            Err(error) => return Err(format!("Network IPC admission failed: {:?}", error)),
894        }
895        Ok(())
896    }
897    pub fn poll(&mut self) -> Poll<Result<FetchResult, String>> {
898        self.io.cleanup();
899        if self.stage == Stage::Done {
900            return Poll::Pending;
901        }
902        if let Some(error) = self.io.cleanup_error() {
903            return self.fail(error.into());
904        }
905        if self.io.now() >= self.deadline {
906            return self.fail("Network load deadline exceeded".into());
907        }
908        if self.stage == Stage::Resolve && self.pending.is_none() && self.io.cleanup_busy() {
909            return Poll::Pending;
910        }
911        if self.pending.is_none() {
912            if let Err(e) = self.submit() {
913                return self.fail(e);
914            }
915            return Poll::Pending;
916        }
917        let mut reply = [0u8; IPC_BUF_SIZE];
918        let op = self.pending.unwrap();
919        let len = match self.io.take(op, &mut reply) {
920            Ok(IpcTakeResult::Pending) => return Poll::Pending,
921            Ok(IpcTakeResult::Terminal {
922                status: IpcTerminal::Reply,
923                len,
924            }) => {
925                self.pending = None;
926                len
927            }
928            Ok(IpcTakeResult::Terminal { status, .. }) => {
929                self.pending = None;
930                return self.fail(format!("Network IPC terminated: {:?}", status));
931            }
932            Err(e) => return self.fail(format!("Network IPC take failed: {:?}", e)),
933        };
934        if let Err(e) = self.response(&reply[..len]) {
935            return self.fail(e);
936        }
937        if self.decoder.complete() {
938            if let Some(location) = self
939                .decoder
940                .location
941                .take()
942                .filter(|_| !self.manual_redirects)
943            {
944                if self.visited.len() > REDIRECT_LIMIT {
945                    return self.fail("Too many redirects".into());
946                }
947                let next = match self.url.redirect(&location) {
948                    Ok(next) => next,
949                    Err(e) => return self.fail(e),
950                };
951                if self
952                    .same_origin
953                    .as_ref()
954                    .is_some_and(|origin| *origin != next.origin())
955                {
956                    return self.fail("Cross-origin resource redirect blocked".into());
957                }
958                if self
959                    .same_site
960                    .as_ref()
961                    .is_some_and(|host| !crate::resources::is_same_site(host, &next.host))
962                {
963                    return self.fail("Cross-site resource redirect blocked".into());
964                }
965                if self.visited.contains(&next.full) {
966                    return self.fail("Redirect loop".into());
967                }
968                self.close();
969                self.visited.push(next.full.clone());
970                self.url = next;
971                self.decoder =
972                    ResponseDecoder::new(self.decoder.limit, self.url.scheme == "gemini");
973                self.make_request();
974                self.stage = Stage::Resolve;
975                return Poll::Pending;
976            }
977            crate::trace::stamp(&alloc::format!(
978                "body complete {} ({} bytes, {} read ops, {}ms since first read)",
979                self.url.full,
980                self.decoder.body.len(),
981                self.reads,
982                self.io.now().saturating_sub(self.first_read_ms)
983            ));
984            self.close();
985            self.stage = Stage::Done;
986            return Poll::Ready(Ok(FetchResult {
987                url: self.url.document_url(),
988                body: core::mem::take(&mut self.decoder.body),
989                media_type: core::mem::take(&mut self.decoder.media_type),
990                gemini: self.url.scheme == "gemini",
991                status: self.decoder.status,
992                headers: core::mem::take(&mut self.decoder.headers),
993            }));
994        }
995        // The next operation goes out in this turn: a reply and the following
996        // request no longer cost two UI turns each, which halves body transfer time.
997        if let Err(e) = self.submit() {
998            return self.fail(e);
999        }
1000        Poll::Pending
1001    }
1002    fn response(&mut self, bytes: &[u8]) -> Result<(), String> {
1003        if self.stage == Stage::Resolve {
1004            ostd::io::println(&alloc::format!(
1005                "[ocel/net] Resolve reply len={}",
1006                bytes.len()
1007            ));
1008            let addr = match api::ipc::decode::<NetResponse>(bytes) {
1009                Ok(NetResponse::Addr(addr)) => {
1010                    crate::trace::stamp(&alloc::format!(
1011                        "dns {} -> {}.{}.{}.{}",
1012                        self.url.host,
1013                        addr[0],
1014                        addr[1],
1015                        addr[2],
1016                        addr[3]
1017                    ));
1018                    addr
1019                }
1020                Ok(NetResponse::Err(code)) if code == NetError::Denied as u8 => {
1021                    return Err(NET_DENIED_MESSAGE.into())
1022                }
1023                _ => return Err(format!("DNS resolution failed for {}", self.url.host)),
1024            };
1025            cache_dns(self.url.host.clone(), addr, self.io.now());
1026            self.address = Some(addr);
1027            self.stage = Stage::Connect;
1028            return Ok(());
1029        }
1030        if self.stage == Stage::Connect {
1031            ostd::io::println(&alloc::format!(
1032                "[ocel/net] Connect reply len={}",
1033                bytes.len()
1034            ));
1035            let cap = if self.url.tls() {
1036                u64::from_le_bytes(
1037                    bytes
1038                        .try_into()
1039                        .map_err(|_| "Malformed TLS connect reply")?,
1040                )
1041            } else {
1042                match api::ipc::decode::<NetResponse>(bytes) {
1043                    Ok(NetResponse::CapId(cap)) => cap as u64,
1044                    Ok(NetResponse::Err(code)) if code == NetError::Denied as u8 => {
1045                        return Err(NET_DENIED_MESSAGE.into())
1046                    }
1047                    _ => return Err("TCP connect failed".into()),
1048                }
1049            };
1050            crate::trace::stamp(&alloc::format!(
1051                "connect{} cap={}",
1052                if self.url.tls() { "+tls" } else { "" },
1053                cap
1054            ));
1055            if cap == tls::CONNECT_FAILED {
1056                return Err("TLS connection failed (network/certificate/time policy)".into());
1057            }
1058            self.cap = Some(cap);
1059            self.stage = if self.url.tls() {
1060                Stage::Send
1061            } else {
1062                Stage::WaitConnect
1063            };
1064            return Ok(());
1065        }
1066        if self.stage == Stage::WaitConnect {
1067            match api::ipc::decode::<NetResponse>(bytes) {
1068                Ok(NetResponse::State(0x03)) => self.stage = Stage::Send,
1069                Ok(NetResponse::State(0x01 | 0x02)) | Ok(NetResponse::NotReady) => {}
1070                _ => return Err("TCP connection failed before establishment".into()),
1071            }
1072            return Ok(());
1073        }
1074        if self.stage == Stage::Send {
1075            let count = if self.url.tls() {
1076                u32::from_le_bytes(bytes.try_into().map_err(|_| "Malformed TLS send reply")?)
1077                    as usize
1078            } else {
1079                match api::ipc::decode::<NetResponse>(bytes) {
1080                    Ok(NetResponse::WriteProgress(n)) => n as usize,
1081                    Ok(NetResponse::NotReady) => return Ok(()),
1082                    _ => return Err("TCP write failed".into()),
1083                }
1084            };
1085            if count > self.sent_chunk || count == 0 {
1086                return Err("Invalid/failed network write progress".into());
1087            }
1088            self.sent += count;
1089            if self.sent == self.request.len() {
1090                self.stage = Stage::Read;
1091                crate::trace::stamp("request sent");
1092            }
1093            return Ok(());
1094        }
1095        if self.url.tls() {
1096            if bytes.len() < 2 {
1097                return Err("Malformed TLS read reply".into());
1098            }
1099            let count = u16::from_le_bytes([bytes[0], bytes[1]]) as usize;
1100            if count == tls::RECV_ERROR as usize {
1101                return Err("TLS read failed".into());
1102            }
1103            if count == u16::MAX as usize - 1 {
1104                return Ok(());
1105            }
1106            if bytes.len() != count + 2 {
1107                return Err("Truncated TLS read IPC reply".into());
1108            }
1109            self.reads += 1;
1110            if self.decoder.status == 0 {
1111                self.first_read_ms = self.io.now();
1112            }
1113            if count == tls::RECV_EOF as usize {
1114                self.decoder.eof()
1115            } else {
1116                self.decoder.feed(&bytes[2..])
1117            }
1118        } else {
1119            self.reads += 1;
1120            if self.decoder.status == 0 {
1121                self.first_read_ms = self.io.now();
1122            }
1123            match api::ipc::decode::<NetResponse>(bytes) {
1124                Ok(NetResponse::Data(data)) if !data.is_empty() => self.decoder.feed(data),
1125                Ok(NetResponse::Eof) => self.decoder.eof(),
1126                Ok(NetResponse::NotReady) => Ok(()),
1127                _ => Err("TCP read failed (not EOF)".into()),
1128            }
1129        }
1130    }
1131    fn fail(&mut self, error: String) -> Poll<Result<FetchResult, String>> {
1132        self.cancel();
1133        Poll::Ready(Err(error))
1134    }
1135    fn close(&mut self) {
1136        if let Some(cap) = self.cap.take() {
1137            self.io.close(self.endpoint, cap, self.url.tls());
1138        }
1139    }
1140    pub fn cancel(&mut self) {
1141        if let Some(op) = self.pending.take() {
1142            self.io.cancel(op);
1143            let mut buf = [0u8; IPC_BUF_SIZE];
1144            // A connect reply may have won the cancellation race: reclaim its cap.
1145            if let Ok(IpcTakeResult::Terminal {
1146                status: IpcTerminal::Reply,
1147                len,
1148            }) = self.io.take(op, &mut buf)
1149            {
1150                if self.stage == Stage::Connect && self.cap.is_none() {
1151                    let cap = if self.url.tls() {
1152                        buf[..len].try_into().ok().map(u64::from_le_bytes)
1153                    } else {
1154                        match api::ipc::decode::<NetResponse>(&buf[..len]) {
1155                            Ok(NetResponse::CapId(cap)) => Some(cap as u64),
1156                            _ => None,
1157                        }
1158                    };
1159                    self.cap = cap.filter(|cap| *cap != 0);
1160                }
1161            }
1162        }
1163        self.close();
1164        self.stage = Stage::Done;
1165    }
1166}
1167impl<T: Transport> Drop for FetchJob<T> {
1168    fn drop(&mut self) {
1169        self.cancel();
1170    }
1171}

1173struct Closing {
1174    endpoint: usize,
1175    cap: u64,
1176    tls: bool,
1177    op: Option<IpcOpId>,
1178    deadline: u64,
1179    // true: no request was accepted, safe explicit navigation recovery.
1180    // false: terminal outcome uncertain; no accepted close is retried.
1181    quarantine: Option<bool>,
1182}
1183#[cfg(feature = "desktop")]
1184const CLOSE_SLOTS: usize = 128;
1185#[cfg(not(feature = "desktop"))]
1186const CLOSE_SLOTS: usize = 32;
1187struct CloseQueue {
1188    slots: [Option<Closing>; CLOSE_SLOTS],
1189    cursor: usize,
1190}
1191static CLOSING: ostd::sync::Mutex<CloseQueue> = ostd::sync::Mutex::new(CloseQueue {
1192    slots: [const { None }; CLOSE_SLOTS],
1193    cursor: 0,
1194});
1195fn closing_count() -> usize {
1196    CLOSING
1197        .lock()
1198        .slots
1199        .iter()
1200        .filter(|slot| slot.is_some())
1201        .count()
1202}
1203fn cleanup_error() -> Option<&'static str> {
1204    let queue = CLOSING.lock();
1205    if queue
1206        .slots
1207        .iter()
1208        .flatten()
1209        .any(|job| job.quarantine == Some(false))
1210    {
1211        Some("Network socket cleanup outcome is uncertain; restart Ocel before further network navigation")
1212    } else if queue
1213        .slots
1214        .iter()
1215        .flatten()
1216        .any(|job| job.quarantine == Some(true))
1217    {
1218        Some("Network socket cleanup admission timed out; retry navigation to recover retained sockets")
1219    } else {
1220        None
1221    }
1222}
1223fn recover_unaccepted_cleanup() {
1224    let now = sys_get_time_ms().unwrap_or(u64::MAX);
1225    for job in CLOSING.lock().slots.iter_mut().flatten() {
1226        job.recover(now);
1227    }
1228}
1229impl Closing {
1230    fn recover(&mut self, now: u64) {
1231        if self.quarantine == Some(true) {
1232            self.quarantine = None;
1233            self.deadline = now.saturating_add(CLOSE_TICKS);
1234        }
1235    }
1236    /// true only after the cap is known closed or its provider has died.
1237    fn advance<T: Transport>(&mut self, io: &mut T) -> bool {
1238        if self.quarantine.is_some() {
1239            return false;
1240        }
1241        let now = io.now();
1242        if let Some(operation) = self.op {
1243            if now >= self.deadline {
1244                io.cancel(operation);
1245            }
1246            let mut reply = [0u8; IPC_BUF_SIZE];
1247            return match io.take(operation, &mut reply) {
1248                Ok(IpcTakeResult::Pending) => false,
1249                Ok(IpcTakeResult::Terminal {
1250                    status: IpcTerminal::Reply | IpcTerminal::PeerGone,
1251                    ..
1252                }) => true,
1253                _ => {
1254                    self.op = None;
1255                    self.quarantine = Some(false);
1256                    false
1257                }
1258            };
1259        }
1260        if now >= self.deadline {
1261            self.quarantine = Some(true);
1262            return false;
1263        }
1264        let mut bytes = [0u8; IPC_BUF_SIZE];
1265        let len = if self.tls {
1266            bytes[0] = tls::CLOSE;
1267            bytes[1..9].copy_from_slice(&self.cap.to_le_bytes());
1268            tls::HEADER_LEN
1269        } else {
1270            match api::ipc::encode(
1271                &NetRequest::TcpClose {
1272                    cap_id: self.cap as u32,
1273                },
1274                &mut bytes,
1275            ) {
1276                Ok(bytes) => bytes.len(),
1277                Err(_) => {
1278                    self.quarantine = Some(false);
1279                    return false;
1280                }
1281            }
1282        };
1283        match io.submit(self.endpoint, &bytes[..len]) {
1284            Ok(operation) => {
1285                self.op = Some(operation);
1286                // Submission, not queue entry, starts the response deadline.
1287                self.deadline = now.saturating_add(CLOSE_TICKS);
1288                false
1289            }
1290            Err(IpcSubmitError::Busy) => false,
1291            Err(IpcSubmitError::PeerGone) => true,
1292            Err(_) => {
1293                self.quarantine = Some(false);
1294                false
1295            }
1296        }
1297    }
1298}
1299fn enqueue_close(endpoint: usize, cap: u64, tls: bool) {
1300    {
1301        let mut queue = CLOSING.lock();
1302        if queue
1303            .slots
1304            .iter()
1305            .flatten()
1306            .any(|job| job.endpoint == endpoint && job.cap == cap)
1307        {
1308            return;
1309        }
1310        // Pre-connect admission reserves a slot for every live job, including
1311        // cancellation races; no socket can be created when the reserve is full.
1312        let slot = queue
1313            .slots
1314            .iter_mut()
1315            .find(|slot| slot.is_none())
1316            .expect("close admission invariant");
1317        *slot = Some(Closing {
1318            endpoint,
1319            cap,
1320            tls,
1321            op: None,
1322            deadline: sys_get_time_ms().unwrap_or(0).saturating_add(CLOSE_TICKS),
1323            quarantine: None,
1324        });
1325    }
1326    poll_cleanup();
1327}
1328pub fn poll_cleanup() {
1329    let mut queue = CLOSING.lock();
1330    // Four slots per UI turn, fixed storage and round-robin fairness.
1331    for _ in 0..CLEANUP_BATCH {
1332        let index = queue.cursor;
1333        queue.cursor = (index + 1) % CLOSE_SLOTS;
1334        let Some(job) = queue.slots[index].as_mut() else {
1335            continue;
1336        };
1337        if job.advance(&mut KernelTransport::new()) {
1338            queue.slots[index] = None;
1339        }
1340    }
1341}
