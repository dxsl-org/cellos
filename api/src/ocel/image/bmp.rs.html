2//! Pure-Rust, zero-dependency uncompressed BMP decoder for Ocel.
3
4extern crate alloc;
5use alloc::vec::Vec;
6
7pub struct DecodedImage {
8    pub width: u32,
9    pub height: u32,
10    pub pixels: Vec<u8>, // BGRA8888 packed
11}
12
13pub fn decode_bmp(data: &[u8]) -> Option<DecodedImage> {
14    if data.len() < 54 {
15        return None;
16    }
17
18    // 1. Check magic "BM"
19    if data[0] != b'B' || data[1] != b'M' {
20        return None;
21    }
22
23    // 2. Read pixel array offset
24    let pixel_offset = u32::from_le_bytes([data[10], data[11], data[12], data[13]]) as usize;
25
26    // 3. Read DIB header
27    let header_size = u32::from_le_bytes([data[14], data[15], data[16], data[17]]) as usize;
28    if header_size < 40 || data.len() < 14 + header_size {
29        return None;
30    }
31
32    let width_i = i32::from_le_bytes([data[18], data[19], data[20], data[21]]);
33    let height_i = i32::from_le_bytes([data[22], data[23], data[24], data[25]]);
34    if width_i <= 0 || width_i > 4096 || height_i.abs() > 4096 {
35        return None;
36    }
37
38    let width = width_i as u32;
39    let (height, bottom_up) = if height_i < 0 {
40        ((-height_i) as u32, false)
41    } else {
42        (height_i as u32, true)
43    };
44
45    let planes = u16::from_le_bytes([data[26], data[27]]);
46    if planes != 1 {
47        return None;
48    }
49
50    let bpp = u16::from_le_bytes([data[28], data[29]]);
51    if bpp != 24 && bpp != 32 {
52        return None;
53    }
54
55    let compression = u32::from_le_bytes([data[30], data[31], data[32], data[33]]);
56    if compression != 0 && compression != 3 {
57        // Only uncompressed BI_RGB (0) or BI_BITFIELDS (3)
58        return None;
59    }
60
61    if pixel_offset >= data.len() {
62        return None;
63    }
64
65    // 4. Decode scanlines
66    let stride = ((width * bpp as u32).div_ceil(32) * 4) as usize;
67    let mut pixels = alloc::vec![0u8; (width * height * 4) as usize];
68
69    let bytes_per_pixel = (bpp / 8) as usize;
70
71    for y in 0..height {
72        let src_row = if bottom_up { height - 1 - y } else { y };
73
74        let row_start = pixel_offset + (src_row as usize) * stride;
75        if row_start + (width as usize) * bytes_per_pixel > data.len() {
76            return None;
77        }
78
79        let dst_row_start = (y as usize) * (width as usize) * 4;
80
81        for x in 0..width {
82            let src_idx = row_start + (x as usize) * bytes_per_pixel;
83            let dst_idx = dst_row_start + (x as usize) * 4;
84
85            let b = data[src_idx];
86            let g = data[src_idx + 1];
87            let r = data[src_idx + 2];
88            let a = if bpp == 32 { data[src_idx + 3] } else { 255 };
89
90            pixels[dst_idx] = b;
91            pixels[dst_idx + 1] = g;
92            pixels[dst_idx + 2] = r;
93            pixels[dst_idx + 3] = a;
94        }
95    }
96
97    Some(DecodedImage {
98        width,
99        height,
100        pixels,
101    })
102}
