1//! AHCI 1.3.1 — HBA reset, port bring-up, IDENTIFY DEVICE, data path.
2//!
3//! PCI binding is done by the caller; this module resets the HBA, starts one
4//! SATA port with a device attached, completes a single IDENTIFY DEVICE,
5//! and then serves 512-byte sector transfers over the command list built here:
6//! READ DMA EXT (0x25) / WRITE DMA EXT (0x35) with a 48-bit LBA, and FLUSH CACHE
7//! EXT (0xEA). Every transfer is one PRDT entry over a caller-supplied authorized
8//! DMA IOVA, and the capacity is taken from the IDENTIFY response — a transfer
9//! past it is rejected with a typed error, never clamped.
10//!
11//! The q35 ICH9 AHCI is register-compatible with PCH AHCI; port count, remap and
12//! NCQ behaviour are not validated on the QEMU model (plan assumption A-02).
13//!
14//! Law 4 exception: Driver Cells may use `unsafe` for DMA memory access. MMIO
15//! goes through the bounds-checked `ostd::mmio::MmioRegion`; every `unsafe`
16//! block carries a `// SAFETY:` comment.
17//!
18//! Command completion is MSI-driven when `AllocPcieIrq` grants a line
19//! (`crate::irq`): the HBA is armed (`GHC.IE`, `PxIE`) only then, and the
20//! wait loops park between `PxCI` re-reads. Otherwise nothing is armed and the
21//! loops poll exactly as before.
22

33// ── Generic host control (ABAR offsets) ───────────────────────────────────────
34
35const REG_CAP: usize = 0x00;
36const REG_GHC: usize = 0x04;
37const REG_IS: usize = 0x08;
38const REG_PI: usize = 0x0C;
39const REG_VS: usize = 0x10;
40
41const GHC_HR: u32 = 1 << 0;
42const GHC_IE: u32 = 1 << 1;
43const GHC_AE: u32 = 1 << 31;
44
45// ── Port register block (base 0x100, stride 0x80) ─────────────────────────────
46
47const PORT_BASE: usize = 0x100;
48const PORT_STRIDE: usize = 0x80;
49
50const PX_CLB: usize = 0x00;
51const PX_CLBU: usize = 0x04;
52const PX_FB: usize = 0x08;
53const PX_FBU: usize = 0x0C;
54const PX_IS: usize = 0x10;
55const PX_IE: usize = 0x14;
56const PX_CMD: usize = 0x18;
57const PX_TFD: usize = 0x20;
58const PX_SIG: usize = 0x24;
59const PX_SSTS: usize = 0x28;
60const PX_SCTL: usize = 0x2C;
61const PX_SERR: usize = 0x30;
62const PX_CI: usize = 0x38;
63
64const CMD_ST: u32 = 1 << 0;
65const CMD_SUD: u32 = 1 << 1;
66const CMD_POD: u32 = 1 << 2;
67const CMD_FRE: u32 = 1 << 4;
68const CMD_FR: u32 = 1 << 14;
69const CMD_CR: u32 = 1 << 15;
70
71const SSTS_DET_MASK: u32 = 0x0F;
72const SSTS_DET_PRESENT: u32 = 0x03;
73const SSTS_IPM_MASK: u32 = 0x0F_00;
74const SSTS_IPM_ACTIVE: u32 = 0x01_00;
75
76const TFD_ERR: u32 = 1 << 0;
77/// Task file busy (BSY) — must be clear before PxCMD.ST is set.
78const TFD_BSY: u32 = 1 << 7;
79/// Task file data-request (DRQ) — must be clear before PxCMD.ST is set.
80const TFD_DRQ: u32 = 1 << 3;
81
82const IS_TFES: u32 = 1 << 30;
83/// Fatal port conditions that are **not** task-file errors: host-bus fatal (29),
84/// host-bus data error (28), interface fatal (27) and overflow (24). AHCI reports
85/// these independently of `TFES`, and a command can clear `PxCI` while one is
86/// latched — treating that as success would return stale DMA contents for a read
87/// and acknowledge a write that never reached the device.
88const IS_HBFS: u32 = 1 << 29;
89const IS_HBDS: u32 = 1 << 28;
90const IS_IFS: u32 = 1 << 27;
91const IS_OFS: u32 = 1 << 24;
92/// Every latched condition that makes a completed command untrustworthy.
93const IS_FATAL: u32 = IS_TFES | IS_HBFS | IS_HBDS | IS_IFS | IS_OFS;
94/// FIS-received conditions that can end a command: D2H Register (0), PIO Setup
95/// (1), DMA Setup (2) and Set Device Bits (3).
96const IS_DHRS: u32 = 1 << 0;
97const IS_PSS: u32 = 1 << 1;
98const IS_DSS: u32 = 1 << 2;
99const IS_SDBS: u32 = 1 << 3;
100/// `PxIE` in interrupt mode (enable bits share `PxIS` positions): every FIS that
101/// can end a command plus every fatal condition, so neither a completion nor a
102/// failure waits out a full park.
103const PX_IE_COMPLETION: u32 = IS_DHRS | IS_PSS | IS_DSS | IS_SDBS | IS_FATAL;
104
105/// AHCI port signature for an ATA (LBA) disk — the only signature IDENTIFY
106/// DEVICE is valid for.
107const SIG_ATA: u32 = 0x0000_0101;
108
109/// Every signature a device may publish in PxSIG once its initial D2H FIS has
110/// landed (AHCI 1.3.1 §3.3.9): ATA, ATAPI, SEMB, port multiplier.
111const SIG_KNOWN: [u32; 4] = [0x0000_0101, 0xEB14_0101, 0xC33C_0101, 0x9669_0101];
112
113// ── Command / FIS constants (AHCI 1.3.1 §4.2) ─────────────────────────────────
114
115/// Register Host-to-Device FIS type.
116const FIS_TYPE_H2D: u8 = 0x27;
117/// ATA IDENTIFY DEVICE command.
118const ATA_CMD_IDENTIFY: u8 = 0xEC;
119/// ATA READ DMA EXT — 48-bit LBA, DMA into a PRDT-listed buffer.
120const ATA_CMD_READ_DMA_EXT: u8 = 0x25;
121/// ATA WRITE DMA EXT — 48-bit LBA, DMA out of a PRDT-listed buffer.
122const ATA_CMD_WRITE_DMA_EXT: u8 = 0x35;
123/// ATA FLUSH CACHE EXT — 48-bit LBA, non-data; drains the device's write cache.
124const ATA_CMD_FLUSH_CACHE_EXT: u8 = 0xEA;
125/// H2D FIS length in DWORDs (20 bytes) for a command with no data payload in
126/// the FIS itself.
127const H2D_FIS_DWORDS: u32 = 5;
128/// One PRDT entry for a single 512-byte sector response or payload.
129const IDENTIFY_BYTES: u32 = 512;
130/// Fixed 512-byte logical sector.
131const SECTOR_BYTES: u32 = 512;
132/// Command-header DWORD0 bit 6: the host-to-device (write) flag of a data
133/// transfer. Bit 5 is the ATAPI flag (AHCI 1.3.1 §4.2.1), which is why this is
134/// not `1 << 5`.
135const CMD_HDR_W: u32 = 1 << 6;
136/// ATA device register value selecting LBA addressing.
137const ATA_DEVICE_LBA: u8 = 0x40;
138
139// ── Poll budgets ──────────────────────────────────────────────────────────────
140
141/// Bounded poll for register transitions that should complete in microseconds.
142/// Every poll is an MMIO read (a VM exit under QEMU), so the budget is kept
143/// small enough that a failing gate reports instead of blowing the boot window.
144const POLL_LIMIT: u64 = 2_000_000;
145/// Link-up after COMRESET can take longer than a register transition. Each poll
146/// is an MMIO read (a VM exit under QEMU), so the budget stays modest.
147const LINK_POLL_LIMIT: u64 = 200_000;
148/// Yield every this many polls so a long wait does not starve the system.
149const POLL_YIELD_EVERY: u64 = 8192;
150/// `PxCI` re-reads before the first park of a command wait, so a completion
151/// that lands within microseconds is not delayed by up to a tick on a busy BSP.
152const SPIN_BEFORE_PARK: u64 = 32;

154pub struct AhciController {
155    mmio: MmioRegion,
156    port: u8,
157    /// Set when a timeout or fatal port error could not be recovered by stopping
158    /// and restarting the engine. Every later request is refused with a typed
159    /// error: the single command slot's structures are shared with the next
160    /// request, so reusing them while the HBA may still own tag 0 would let a
161    /// retry race an in-flight DMA.
162    faulted: bool,
163    // The HBA fetches every one of these structures by DMA; all four are
164    // retained for the controller's lifetime so the device-visible IOVAs stay
165    // valid.
166    cmd_list: AuthorizedDma<DmaBuf>,
167    /// Retained for the controller's lifetime: the HBA writes received FISes
168    /// here continuously, so freeing it would be a use-after-free. It is only
169    /// read (via `iova()`) while the port is being started.
170    _fis: AuthorizedDma<DmaBuf>,
171    cmd_table: AuthorizedDma<DmaBuf>,
172    identify: AuthorizedDma<DmaBuf>,
173    /// MSI completion line; inactive means polled with the HBA unarmed.
174    irq: CompletionIrq,
175    /// Fault-injection builds only: what to do to the next issued command.
176    #[cfg(feature = "fault-inject")]
177    inject_next: crate::fault_inject::Inject,
178}

180impl AhciController {
181    /// Reset the HBA reachable via `mmio` (the AHCI ABAR), start the first port
182    /// with an ATA device, and complete IDENTIFY DEVICE.
183    ///
184    /// Returns a typed error on any device error; never panics.
185    pub fn new(mmio: MmioRegion, bdf: u32) -> ViResult<Self> {
186        // The Platform Cell scans ECAM concurrently with this cell's spawn, and
187        // its BAR size probe temporarily clears the PCI command register's
188        // memory-decode bit. A register read inside that window returns 0, so
189        // wait (bounded) for a sane CAP/PI before trusting any register — a
190        // zero CAP would otherwise look like "HBA reset already complete" and
191        // "no ports implemented".
192        // AHCI 1.3.1 §10.1.2: while CAP.SAM is 0 the host must enable AHCI mode
193        // (GHC.AE) before touching any register other than GHC. Best effort
194        // here; re-asserted after the Platform Cell's BAR-probe window below.
195        let ghc = rd(&mmio, REG_GHC)?;
196        if ghc & GHC_AE == 0 && ghc != 0xFFFF_FFFF {
197            wr(&mmio, REG_GHC, ghc | GHC_AE)?;
198        }
199
200        let (cap, version, pi) = {
201            let mut attempt = 0u32;
202            loop {
203                let cap = rd(&mmio, REG_CAP)?;
204                let version = rd(&mmio, REG_VS)?;
205                let pi = rd(&mmio, REG_PI)?;
206                if cap != 0 && cap != 0xFFFF_FFFF && pi != 0 {
207                    break (cap, version, pi);
208                }
209                attempt += 1;
210                if attempt >= 20_000 {
211                    let _ = print_fmt(format_args!(
212                        "[ahci] ABAR not readable: cap=0x{:08x} version=0x{:08x} pi=0x{:08x}\n",
213                        cap, version, pi
214                    ));
215                    return Err(ViError::IO);
216                }
217                ostd::task::yield_now();
218            }
219        };
220        // CAP.NP is bits 4:0 and holds "number of ports - 1".
221        let n_ports = ((cap & 0x1F) + 1) as u8;
222        let _ = print_fmt(format_args!(
223            "[ahci] controller bound bdf={:02x}:{:02x}.{} cap=0x{:08x} version=0x{:08x} pi=0x{:08x}\n",
224            (bdf >> 8) & 0xFF,
225            (bdf >> 3) & 0x1F,
226            bdf & 0x07,
227            cap,
228            version,
229            pi
230        ));
231
232        // 1. HBA reset. GHC.HR self-clears when the reset completes, and it is
233        // only defined while AE is set — writing HR alone would clear AHCI mode
234        // first. Keep AE in the same write, then re-assert it after the reset.
235        let ghc = rd(&mmio, REG_GHC)?;
236        wr(&mmio, REG_GHC, ghc | GHC_AE | GHC_HR)?;
237        let mut spins = 0u64;
238        loop {
239            if rd(&mmio, REG_GHC)? & GHC_HR == 0 {
240                break;
241            }
242            spins = poll_tick(spins, POLL_LIMIT, "[ahci] HBA reset never completed")?;
243        }
244        wr(&mmio, REG_GHC, GHC_AE)?;
245        let _ = print_fmt(format_args!("[ahci] HBA reset complete (AE enabled)\n"));
246
247        // 2. Command list, FIS receive area, command table and IDENTIFY buffer.
248        let cmd_list = alloc_dma(bdf, 1)?;
249        let fis = alloc_dma(bdf, 1)?;
250        let cmd_table = alloc_dma(bdf, 1)?;
251        let identify = alloc_dma(bdf, 1)?;
252        // DmaBuf is not guaranteed zeroed; the HBA reads the FIS receive area and
253        // the command table, so both must start in a defined state.
254        for buf in [&cmd_list, &fis, &cmd_table, &identify] {
255            // SAFETY: each buffer is a live, page-sized DMA allocation owned by
256            // this cell; write_bytes covers exactly its size.
257            unsafe { core::ptr::write_bytes(buf.inner().virt(), 0, buf.inner().size()) };
258        }
259
260        // 3. Reset is done; start a port with a device and keep it running.
261        let port = Self::start_port(&mmio, n_ports, pi, &cmd_list, &fis)?;
262        // Only after HBA reset (which clears GHC.IE) and port bring-up, so no
263        // interrupt is armed while the port is still being probed.
264        let irq = CompletionIrq::setup(bdf);
265
266        let mut ctrl = AhciController {
267            mmio,
268            port,
269            faulted: false,
270            cmd_list,
271            _fis: fis,
272            cmd_table,
273            identify,
274            irq,
275            #[cfg(feature = "fault-inject")]
276            inject_next: crate::fault_inject::Inject::None,
277        };
278        if ctrl.irq.active() {
279            ctrl.arm_interrupts(ctrl.port_offset())?;
280        }
281        ctrl.identify()?;
282        // Word 0 is the ATA general-configuration word: 0x0040 means a fixed,
283        // non-removable ATA device, which proves the 512-byte DMA landed (a
284        // completed-but-empty transfer would leave this zero). The model field
285        // is only set by QEMU when `model=` is passed, so it is reported but
286        // not required.
287        let w0 = dma_read_u16(&ctrl.identify, 0);
288        let model = ctrl.model();
289        let firmware = ata_string(&ctrl.identify, 23, 4);
290        let _ = print_fmt(format_args!(
291            "[ahci] IDENTIFY DEVICE ok port={} w0=0x{:04x} sectors={} model=\"{}\" fw=\"{}\"\n",
292            ctrl.port,
293            w0,
294            ctrl.sectors(),
295            model,
296            firmware
297        ));
298        Ok(ctrl)
299    }

310    /// LBA28 sector count (IDENTIFY words 60:61, i.e. byte offsets 120/122).
311    pub fn sectors(&self) -> u64 {
312        let lo = dma_read_u16(&self.identify, 60 * 2) as u64;
313        let hi = dma_read_u16(&self.identify, 61 * 2) as u64;
314        lo | (hi << 16)
315    }
316
317    /// Device capacity in 512-byte sectors, from the IDENTIFY response.
318    ///
319    /// Prefers the 48-bit field (IDENTIFY words 100:103) when the device
320    /// declares LBA48 support (word 83 bit 10); otherwise falls back to the
321    /// LBA28 field. The LBA28 field is clamped at `0x0FFF_FFFF` on a large
322    /// LBA48 device, so using it unconditionally would under-report a modern
323    /// disk and make an in-range LBA look out of range.
324    pub fn capacity_sectors(&self) -> u64 {
325        if dma_read_u16(&self.identify, 83 * 2) & (1 << 10) != 0 {
326            let mut cap = 0u64;
327            for i in 0..4usize {
328                cap |= (dma_read_u16(&self.identify, (100 + i) * 2) as u64) << (16 * i);
329            }
330            if cap != 0 {
331                return cap;
332            }
333        }
334        self.sectors()
335    }

344/// Split out so `new` can build the struct before the borrow-heavy IDENTIFY step.
345impl AhciController {
346    /// Program PxCLB/PxFB, enable the port, and issue IDENTIFY DEVICE.
347    fn identify(&mut self) -> ViResult<()> {
348        let p = self.port_offset();
349
350        // Command header slot 0: CFL=5 DWORDs, PRDTL=1 entry.
351        let ctba = self.cmd_table.iova();
352        dma_write_u32(&self.cmd_list, 0, H2D_FIS_DWORDS | (1 << 16));
353        dma_write_u32(&self.cmd_list, 4, 0);
354        dma_write_u32(&self.cmd_list, 8, ctba as u32);
355        dma_write_u32(&self.cmd_list, 12, (ctba >> 32) as u32);
356
357        // Command table: H2D register FIS at offset 0 (C=1, command = IDENTIFY),
358        // PRDT entry 0 at offset 128. The PRDT entry is `DBA(64) | reserved(32)
359        // | DBC(32)` — DBC is the *last* dword (offset 12), not offset 8 (QEMU
360        // `AHCI_SG`, Fuchsia `ahci_prd_t`), and holds byte-count-minus-one.
361        dma_write_u8(&self.cmd_table, 0, FIS_TYPE_H2D);
362        dma_write_u8(&self.cmd_table, 1, 0x80); // C bit: update the command register
363        dma_write_u8(&self.cmd_table, 2, ATA_CMD_IDENTIFY);
364        dma_write_u64(&self.cmd_table, 128, self.identify.iova());
365        dma_write_u32(&self.cmd_table, 136, 0); // reserved
366        dma_write_u32(&self.cmd_table, 140, IDENTIFY_BYTES - 1); // DBC = byte count - 1
367
368        // Clear latched errors, then issue. The fence makes the command
369        // structure writes visible before the HBA is told to fetch them.
370        wr(&self.mmio, p + PX_SERR, 0xFFFF_FFFF)?;
371        wr(&self.mmio, p + PX_IS, 0xFFFF_FFFF)?;
372        fence(Ordering::SeqCst);
373        wr(&self.mmio, p + PX_CI, 1)?;
374
375        let mut spins = 0u64;
376        let mut seen = 0u32;
377        loop {
378            if rd(&self.mmio, p + PX_CI)? & 1 == 0 {
379                break;
380            }
381            spins = self.completion_tick(p, spins, &mut seen, "[ahci] IDENTIFY never completed")?;
382        }
383
384        let is = rd(&self.mmio, p + PX_IS)? | seen;
385        let tfd = rd(&self.mmio, p + PX_TFD)?;
386        if self.irq.active() {
387            self.ack_interrupts(p)?;
388        }
389        if tfd & TFD_ERR != 0 || is & IS_FATAL != 0 {
390            let _ = print_fmt(format_args!(
391                "[ahci] IDENTIFY failed: PxIS=0x{:08x} PxTFD=0x{:08x}\n",
392                is, tfd
393            ));
394            return Err(ViError::IO);
395        }
396
397        // A completed-but-empty transfer must not be reported as a successful
398        // IDENTIFY: the command header's transferred byte count (PRDBC, command
399        // header DWORD1 at byte offset 4 — AHCI 1.3.1 §4.2.1; QEMU's
400        // `AHCICmdHdr.status` and Linux's `ahci_cmd_hdr.status`, at the same
401        // offset) must equal the requested 512 and the payload's
402        // general-configuration word must be a real value.
403        let prdbc = dma_read_u32(&self.cmd_list, 4);
404        let w0 = dma_read_u16(&self.identify, 0);
405        if prdbc != IDENTIFY_BYTES || w0 == 0 || w0 == 0xFFFF {
406            let _ = print_fmt(format_args!(
407                "[ahci] IDENTIFY payload invalid: PRDBC={} w0=0x{:04x}\n",
408                prdbc, w0
409            ));
410            return Err(ViError::IO);
411        }
412        Ok(())
413    }

415    fn port_offset(&self) -> usize {
416        PORT_BASE + self.port as usize * PORT_STRIDE
417    }

419    // ── Data path (phase 02b) ─────────────────────────────────────────────────
420
421    /// Read `count` consecutive 512-byte sectors from the given 48-bit LBA
422    /// into `iova` with one READ DMA EXT.
423    ///
424    /// `iova` must be the device-visible address of an authorized DMA buffer
425    /// of at least `count * 512` bytes (one PRDT entry covers it).
426    pub fn read_sectors(&mut self, lba: u64, count: u16, iova: u64) -> Result<(), BlockOutcome> {
427        self.ensure_live()?;
428        self.issue_dma(
429            ATA_CMD_READ_DMA_EXT,
430            lba,
431            count,
432            iova,
433            false,
434            "READ DMA EXT",
435        )
436    }

438    /// Write one 512-byte sector at the given 48-bit LBA from `iova`.
439    pub fn write_sector(&mut self, lba: u64, iova: u64) -> Result<(), BlockOutcome> {
440        self.ensure_live()?;
441        self.issue_dma(ATA_CMD_WRITE_DMA_EXT, lba, 1, iova, true, "WRITE DMA EXT")
442    }

444    /// Drain the device's write cache (FLUSH CACHE EXT, non-data).
445    pub fn flush(&mut self) -> Result<(), BlockOutcome> {
446        self.ensure_live()?;
447        let p = self.port_offset();
448        // Non-data command: PRDTL 0, and no PRDT entry programmed. The HBA only
449        // reads the FIS words it is told about via CFL, so the stale PRDT left
450        // by the previous data command is not fetched.
451        dma_write_u32(&self.cmd_list, 0, H2D_FIS_DWORDS);
452        dma_write_u32(&self.cmd_list, 4, 0);
453        dma_write_u32(&self.cmd_list, 8, self.cmd_table.iova() as u32);
454        dma_write_u32(&self.cmd_list, 12, (self.cmd_table.iova() >> 32) as u32);
455
456        dma_write_u8(&self.cmd_table, 0, FIS_TYPE_H2D);
457        dma_write_u8(&self.cmd_table, 1, 0x80); // C bit
458        dma_write_u8(&self.cmd_table, 2, ATA_CMD_FLUSH_CACHE_EXT);
459        dma_write_u8(&self.cmd_table, 7, ATA_DEVICE_LBA);
460        self.issue_and_wait(p, 0, "FLUSH CACHE EXT")
461    }

463    /// Program command slot 0 for a READ/WRITE DMA EXT of `count` sectors
464    /// (`1..=MAX_RUN_SECTORS`, one PRDT entry) and poll it to completion.
465    fn issue_dma(
466        &mut self,
467        cmd: u8,
468        lba: u64,
469        count: u16,
470        iova: u64,
471        write: bool,
472        what: &str,
473    ) -> Result<(), BlockOutcome> {
474        let p = self.port_offset();
475        let tag = 0u8;
476
477        // Reject a transfer beyond the IDENTIFY-reported capacity instead of
478        // letting the device answer with an opaque ABRT and no address.
479        let capacity = self.capacity_sectors();
480        if !(1..=MAX_RUN_SECTORS).contains(&count)
481            || lba
482                .checked_add(u64::from(count))
483                .is_none_or(|end| end > capacity)
484        {
485            let _ = print_fmt(format_args!(
486                "[ahci] port={} tag={} {} rejected: {} sectors at LBA {} past capacity {} sectors\n",
487                self.port, tag, what, count, lba, capacity
488            ));
489            return Err(BlockOutcome::NotIssued);
490        }
491        // Only 48 bits of LBA exist in the FIS; an LBA that needs more cannot be
492        // expressed, so it is out of range by definition.
493        if lba >> 48 != 0 {
494            let _ = print_fmt(format_args!(
495                "[ahci] port={} tag={} {} rejected: LBA {} exceeds 48 bits\n",
496                self.port, tag, what, lba
497            ));
498            return Err(BlockOutcome::NotIssued);
499        }
500
501        // Command header slot 0: CFL=5 DWORDs, direction bit, PRDTL=1 entry.
502        let header = H2D_FIS_DWORDS | (if write { CMD_HDR_W } else { 0 }) | (1 << 16);
503        dma_write_u32(&self.cmd_list, 0, header);
504        dma_write_u32(&self.cmd_list, 4, 0); // PRDBC cleared before issue
505        dma_write_u32(&self.cmd_list, 8, self.cmd_table.iova() as u32);
506        dma_write_u32(&self.cmd_list, 12, (self.cmd_table.iova() >> 32) as u32);
507
508        // H2D register FIS at offset 0 (C=1, 48-bit LBA fields), PRDT entry 0 at
509        // offset 128 (DBA 64-bit, reserved 32-bit, DBC = byte count - 1).
510        dma_write_u8(&self.cmd_table, 0, FIS_TYPE_H2D);
511        dma_write_u8(&self.cmd_table, 1, 0x80); // C bit: update the command register
512        dma_write_u8(&self.cmd_table, 2, cmd);
513        dma_write_u8(&self.cmd_table, 3, 0); // features high
514        dma_write_u8(&self.cmd_table, 4, (lba & 0xFF) as u8);
515        dma_write_u8(&self.cmd_table, 5, ((lba >> 8) & 0xFF) as u8);
516        dma_write_u8(&self.cmd_table, 6, ((lba >> 16) & 0xFF) as u8);
517        dma_write_u8(&self.cmd_table, 7, ATA_DEVICE_LBA);
518        dma_write_u8(&self.cmd_table, 8, ((lba >> 24) & 0xFF) as u8);
519        dma_write_u8(&self.cmd_table, 9, ((lba >> 32) & 0xFF) as u8);
520        dma_write_u8(&self.cmd_table, 10, ((lba >> 40) & 0xFF) as u8);
521        dma_write_u8(&self.cmd_table, 11, 0); // features exp
522        let [count_low, count_high] = count.to_le_bytes();
523        dma_write_u8(&self.cmd_table, 12, count_low); // sector count low
524        dma_write_u8(&self.cmd_table, 13, count_high); // sector count high
525        dma_write_u32(&self.cmd_table, 128, iova as u32);
526        dma_write_u32(&self.cmd_table, 132, (iova >> 32) as u32);
527        dma_write_u32(&self.cmd_table, 136, 0); // reserved
528        dma_write_u32(&self.cmd_table, 140, u32::from(count) * SECTOR_BYTES - 1); // DBC = byte count - 1
529
530        self.issue_and_wait(p, tag, what)
531    }
532
533    /// Clear latched port errors, kick PxCI bit `tag`, and poll it back to 0.
534    ///
535    /// The `PxCI` write is the issue point. A failure before it is `NotIssued`;
536    /// a timeout, device fault or register error after it is `Indeterminate`,
537    /// because the HBA may already have moved the data. A stall is never
538    /// reported as success.
539    fn issue_and_wait(&mut self, p: usize, tag: u8, what: &str) -> Result<(), BlockOutcome> {
540        #[cfg(feature = "fault-inject")]
541        let inject = core::mem::replace(&mut self.inject_next, crate::fault_inject::Inject::None);
542        #[cfg(feature = "fault-inject")]
543        if inject == crate::fault_inject::Inject::RefuseBeforeIssue {
544            let _ = print_fmt(format_args!(
545                "[ahci] fault-inject: {} refused before issue\n",
546                what
547            ));
548            return Err(BlockOutcome::NotIssued);
549        }
550        let not_issued = |_| BlockOutcome::NotIssued;
551        wr(&self.mmio, p + PX_SERR, 0xFFFF_FFFF).map_err(not_issued)?;
552        wr(&self.mmio, p + PX_IS, 0xFFFF_FFFF).map_err(not_issued)?;
553        // The fence makes the command structure writes visible before the HBA is
554        // told to fetch them.
555        fence(Ordering::SeqCst);
556        let bit = 1u32 << tag;
557        // `MmioRegion` refuses an out-of-bounds offset before touching the
558        // device, so an error here means nothing was issued.
559        wr(&self.mmio, p + PX_CI, bit).map_err(not_issued)?;
560        #[cfg(feature = "fault-inject")]
561        if let crate::fault_inject::Inject::ExitAfterIssue { lba } = inject {
562            let _ = print_fmt(format_args!(
563                "[ahci] fault-inject: driver exiting after PxCI at lba={}\n",
564                lba
565            ));
566            ostd::syscall::sys_exit(70);
567        }
568
569        let indeterminate = |_| BlockOutcome::Indeterminate;
570        let mut spins = 0u64;
571        let mut seen = 0u32;
572        loop {
573            if rd(&self.mmio, p + PX_CI).map_err(indeterminate)? & bit == 0 {
574                break;
575            }
576            match self.completion_tick(p, spins, &mut seen, "") {
577                Ok(next) => spins = next,
578                Err(_) => {
579                    let _ = print_fmt(format_args!(
580                        "[ahci] port={} tag={} {} timeout after {} polls\n",
581                        self.port, tag, what, POLL_LIMIT
582                    ));
583                    // Retire the slot before returning: tag 0's structures are
584                    // shared with the next request.
585                    self.recover_after_fault(p);
586                    return Err(BlockOutcome::Indeterminate);
587                }
588            }
589        }
590
591        #[cfg(feature = "fault-inject")]
592        if inject == crate::fault_inject::Inject::SuppressCompletion {
593            // The command really ran; the driver acts as if its completion was
594            // never observed, taking the timeout path.
595            let _ = print_fmt(format_args!(
596                "[ahci] fault-inject: {} completion suppressed after PxCI\n",
597                what
598            ));
599            self.recover_after_fault(p);
600            return Err(BlockOutcome::Indeterminate);
601        }
602
603        let is = rd(&self.mmio, p + PX_IS).map_err(indeterminate)? | seen;
604        let tfd = rd(&self.mmio, p + PX_TFD).map_err(indeterminate)?;
605        if self.irq.active() {
606            self.ack_interrupts(p).map_err(indeterminate)?;
607        }
608        if tfd & TFD_ERR != 0 || is & IS_FATAL != 0 {
609            let _ = print_fmt(format_args!(
610                "[ahci] port={} tag={} {} device fault: PxIS=0x{:08x} PxTFD=0x{:08x}\n",
611                self.port, tag, what, is, tfd
612            ));
613            self.recover_after_fault(p);
614            return Err(BlockOutcome::Indeterminate);
615        }
616        fence(Ordering::Acquire);
617        Ok(())
618    }
619
620    /// One step of a command-completion wait.
621    ///
622    /// Polled: exactly `poll_tick` against `POLL_LIMIT`. Interrupt mode, after
623    /// `SPIN_BEFORE_PARK` re-reads: park on the MSI, then acknowledge so the
624    /// next event raises a fresh message. Acknowledged `PxIS` bits accumulate in
625    /// `seen`, which the caller ORs into its post-completion `PxIS` read so a
626    /// fatal condition cleared here is still reported.
627    ///
628    /// A park stands in for one `poll_tick` yield window and spends that whole
629    /// window of `POLL_LIMIT`: the number of yield points before a timeout is
630    /// unchanged, so a lost interrupt ends in the same timeout, never a hang.
631    fn completion_tick(
632        &mut self,
633        p: usize,
634        spins: u64,
635        seen: &mut u32,
636        what: &str,
637    ) -> ViResult<u64> {
638        if spins >= SPIN_BEFORE_PARK && self.irq.active() {
639            if self.irq.park() {
640                *seen |= self.ack_interrupts(p)?;
641                let spins = spins + POLL_YIELD_EVERY;
642                if spins >= POLL_LIMIT {
643                    if !what.is_empty() {
644                        let _ = print_fmt(format_args!(
645                            "[ahci] timeout: {} ({} polls)\n",
646                            what, POLL_LIMIT
647                        ));
648                    }
649                    return Err(ViError::IO);
650                }
651                fence(Ordering::Acquire);
652                return Ok(spins);
653            }
654            // `WaitCompletion` refused the wait: nothing will hear the line.
655            self.disarm_interrupts(p);
656        }
657        poll_tick(spins, POLL_LIMIT, what)
658    }
659
660    /// Arm MSI delivery for port `p`: clear stale status (port, then HBA),
661    /// enable the completion and fatal conditions in `PxIE`, then set `GHC.IE`.
662    /// Called only while `self.irq` holds a line.
663    fn arm_interrupts(&self, p: usize) -> ViResult<()> {
664        wr(&self.mmio, p + PX_IS, 0xFFFF_FFFF)?;
665        wr(&self.mmio, REG_IS, 1u32 << self.port)?;
666        wr(&self.mmio, p + PX_IE, PX_IE_COMPLETION)?;
667        let ghc = rd(&self.mmio, REG_GHC)?;
668        wr(&self.mmio, REG_GHC, ghc | GHC_IE)
669    }
670
671    /// Back to the polled configuration (`GHC.IE` and `PxIE` clear) after the
672    /// kernel refused an `IRQ` wait. Best effort: the poll loop needs neither.
673    fn disarm_interrupts(&self, p: usize) {
674        if let Ok(ghc) = rd(&self.mmio, REG_GHC) {
675            let _ = wr(&self.mmio, REG_GHC, ghc & !GHC_IE);
676        }
677        let _ = wr(&self.mmio, p + PX_IE, 0);
678    }
679
680    /// Acknowledge port `p`: W1C the `PxIS` bits read, then this port's `IS`
681    /// bit — port first, so `IS` does not re-latch from a still-set `PxIS`.
682    /// Returns the acknowledged `PxIS` bits.
683    fn ack_interrupts(&self, p: usize) -> ViResult<u32> {
684        let is = rd(&self.mmio, p + PX_IS)?;
685        wr(&self.mmio, p + PX_IS, is)?;
686        wr(&self.mmio, REG_IS, 1u32 << self.port)?;
687        Ok(is)
688    }
689
690    /// Refuse a request on a controller that could not be recovered.
691    fn ensure_live(&self) -> Result<(), BlockOutcome> {
692        if self.faulted {
693            let _ = print_fmt(format_args!(
694                "[ahci] port={} controller faulted; request refused\n",
695                self.port
696            ));
697            return Err(BlockOutcome::NotIssued);
698        }
699        Ok(())
700    }
701
702    /// Retire the outstanding slot after a timeout or a fatal port error.
703    ///
704    /// Single-slot design: tag 0's command structures are shared with the next
705    /// request, so leaving the HBA owning the slot would let the next command
706    /// race a still-running DMA. Stop the engine (which retires the slot), clear
707    /// the latched error bits and restart it; if the restart does not come back
708    /// clean, poison the controller so nothing reuses those structures.
709    fn recover_after_fault(&mut self, p: usize) {
710        let _ = Self::stop_port_engine(&self.mmio, p);
711        let _ = wr(&self.mmio, p + PX_SERR, 0xFFFF_FFFF);
712        let _ = wr(&self.mmio, p + PX_IS, 0xFFFF_FFFF);
713        let restarted = {
714            let Self {
715                mmio,
716                cmd_list,
717                _fis,
718                ..
719            } = &*self;
720            Self::start_fis_receive(mmio, p, cmd_list, _fis).is_ok()
721                && Self::wait_task_file_idle(mmio, p).is_ok()
722                && Self::start_command_engine(mmio, p).is_ok()
723        };
724        if !restarted {
725            self.faulted = true;
726            let _ = print_fmt(format_args!(
727                "[ahci] port={} controller poisoned after an unrecoverable fault\n",
728                self.port
729            ));
730        } else if self.irq.active() {
731            // `start_command_engine` left PxIE at 0; re-arm so the next command
732            // is interrupt-driven again. A failure only costs park latency.
733            let _ = self.arm_interrupts(p);
734        }
735    }
736
737    /// Bring one implemented port up and return its index.
738    ///
739    /// PxSIG is only valid once the port is started (AHCI 1.3.1 §3.3.9: the
740    /// signature is valid when PxCMD.ST and PxCMD.FRE are set — and QEMU
741    /// returns 0xffff_ffff before that, including on q35), so the signature
742    /// check happens *after* the port engine is running. A started port whose
743    /// signature is not an ATA disk (the q35 ATAPI CD-ROM shares the
744    /// controller) is stopped again and the scan continues.
745    fn start_port(
746        mmio: &MmioRegion,
747        n_ports: u8,
748        pi: u32,
749        cmd_list: &AuthorizedDma<DmaBuf>,
750        fis: &AuthorizedDma<DmaBuf>,
751    ) -> ViResult<u8> {
752        for port in 0..n_ports {
753            if pi & (1u32 << port) == 0 {
754                continue;
755            }
756            let p = PORT_BASE + port as usize * PORT_STRIDE;
757
758            // Power the port and request spin-up before reading the link state;
759            // the HBA only reports DET/IPM once the port is powered.
760            let cmd = rd(mmio, p + PX_CMD)?;
761            wr(mmio, p + PX_CMD, cmd | CMD_POD | CMD_SUD)?;
762
763            let mut ssts = rd(mmio, p + PX_SSTS)?;
764            if ssts & SSTS_DET_MASK != SSTS_DET_PRESENT {
765                // DET=0 means no device on this port: nothing to reset, so do
766                // not spend the (MMIO-exit-bound) link budget on it. DET=1 means
767                // a device is detected but communication is not established, so
768                // COMRESET it.
769                if ssts & SSTS_DET_MASK == 0 {
770                    continue;
771                }
772                let sctl = rd(mmio, p + PX_SCTL)?;
773                wr(mmio, p + PX_SCTL, (sctl & !0xF) | 1)?;
774                delay_ms();
775                wr(mmio, p + PX_SCTL, sctl & !0xF)?;
776                let mut spins = 0u64;
777                loop {
778                    ssts = rd(mmio, p + PX_SSTS)?;
779                    if ssts & SSTS_DET_MASK == SSTS_DET_PRESENT
780                        && ssts & SSTS_IPM_MASK == SSTS_IPM_ACTIVE
781                    {
782                        break;
783                    }
784                    match poll_tick(spins, LINK_POLL_LIMIT, "") {
785                        Ok(next) => spins = next,
786                        Err(_) => {
787                            // An unresponsive port must not abort the scan for
788                            // the ports after it.
789                            let _ = print_fmt(format_args!(
790                                "[ahci] port {} link did not come up (PxSSTS=0x{:08x})\n",
791                                port, ssts
792                            ));
793                            break;
794                        }
795                    }
796                }
797            }
798            if ssts & SSTS_DET_MASK != SSTS_DET_PRESENT || ssts & SSTS_IPM_MASK != SSTS_IPM_ACTIVE {
799                continue;
800            }
801
802            Self::start_fis_receive(mmio, p, cmd_list, fis)?;
803            Self::wait_task_file_idle(mmio, p)?;
804            // AHCI 1.3.1 §3.3.9: PxSIG becomes valid once the device's initial
805            // D2H FIS lands, which with FRE on can precede the command engine.
806            // Prefer that order; QEMU's ICH9 model only publishes the signature
807            // after PxCMD.ST, so fall back to the started-engine read.
808            let sig = match Self::wait_device_signature(mmio, p) {
809                Some(sig) => {
810                    if sig != SIG_ATA {
811                        let _ = print_fmt(format_args!(
812                            "[ahci] port {} is not an ATA disk (SSTS=0x{:08x} SIG=0x{:08x}); skipping\n",
813                            port, ssts, sig
814                        ));
815                        Self::stop_port_engine(mmio, p)?;
816                        continue;
817                    }
818                    Self::start_command_engine(mmio, p)?;
819                    sig
820                }
821                None => {
822                    Self::start_command_engine(mmio, p)?;
823                    let sig = rd(mmio, p + PX_SIG)?;
824                    if sig != SIG_ATA {
825                        let _ = print_fmt(format_args!(
826                            "[ahci] port {} is not an ATA disk (SSTS=0x{:08x} SIG=0x{:08x}); skipping\n",
827                            port, ssts, sig
828                        ));
829                        Self::stop_port_engine(mmio, p)?;
830                        continue;
831                    }
832                    sig
833                }
834            };
835
836            let _ = print_fmt(format_args!(
837                "[ahci] port {} link up (SSTS=0x{:08x} SIG=0x{:08x})\n",
838                port, ssts, sig
839            ));
840            return Ok(port);
841        }
842
843        Err(ViError::NotFound)
844    }
845
846    /// Program PxCLB/PxFB and start FIS receive (PxCMD.FRE), leaving the command
847    /// engine stopped so the device's initial signature can be read first.
848    fn start_fis_receive(
849        mmio: &MmioRegion,
850        p: usize,
851        cmd_list: &AuthorizedDma<DmaBuf>,
852        fis: &AuthorizedDma<DmaBuf>,
853    ) -> ViResult<()> {
854        Self::stop_port_engine(mmio, p)?;
855
856        wr(mmio, p + PX_SERR, 0xFFFF_FFFF)?;
857        wr(mmio, p + PX_CLB, cmd_list.iova() as u32)?;
858        wr(mmio, p + PX_CLBU, (cmd_list.iova() >> 32) as u32)?;
859        wr(mmio, p + PX_FB, fis.iova() as u32)?;
860        wr(mmio, p + PX_FBU, (fis.iova() >> 32) as u32)?;
861
862        let cmd = rd(mmio, p + PX_CMD)?;
863        wr(mmio, p + PX_CMD, cmd | CMD_FRE | CMD_POD | CMD_SUD)?;
864        let mut spins = 0u64;
865        loop {
866            if rd(mmio, p + PX_CMD)? & CMD_FR != 0 {
867                break;
868            }
869            spins = poll_tick(spins, POLL_LIMIT, "[ahci] FIS receive never started")?;
870        }
871        Ok(())
872    }
873
874    /// Start the command engine (PxCMD.ST) and wait for CR.
875    fn start_command_engine(mmio: &MmioRegion, p: usize) -> ViResult<()> {
876        let cmd = rd(mmio, p + PX_CMD)?;
877        wr(mmio, p + PX_CMD, cmd | CMD_ST)?;
878        let mut spins = 0u64;
879        loop {
880            if rd(mmio, p + PX_CMD)? & CMD_CR != 0 {
881                break;
882            }
883            spins = poll_tick(spins, POLL_LIMIT, "[ahci] command list never started")?;
884        }
885
886        // Leave the port unarmed; `arm_interrupts` enables PxIE only when an
887        // MSI line is held.
888        wr(mmio, p + PX_IE, 0)?;
889        Ok(())
890    }
891
892    /// Poll PxTFD until BSY and DRQ are clear (bounded).
893    ///
894    /// AHCI 1.3.1 §10.3.1: software must not set PxCMD.ST while the task file
895    /// reports the device busy or holding a data request.
896    fn wait_task_file_idle(mmio: &MmioRegion, p: usize) -> ViResult<()> {
897        let mut spins = 0u64;
898        loop {
899            let tfd = rd(mmio, p + PX_TFD)?;
900            if tfd & (TFD_BSY | TFD_DRQ) == 0 {
901                return Ok(());
902            }
903            spins = poll_tick(spins, LINK_POLL_LIMIT, "[ahci] task file never went idle")?;
904        }
905    }
906
907    /// Poll PxSIG until a known device signature appears (bounded).
908    ///
909    /// Returns `None` when none appears before the budget expires — which is the
910    /// QEMU ICH9 model's behaviour — so the caller reads the signature after
911    /// starting the command engine instead of failing the port.
912    fn wait_device_signature(mmio: &MmioRegion, p: usize) -> Option<u32> {
913        let mut spins = 0u64;
914        loop {
915            let Ok(sig) = rd(mmio, p + PX_SIG) else {
916                return None;
917            };
918            if SIG_KNOWN.contains(&sig) {
919                return Some(sig);
920            }
921            match poll_tick(spins, LINK_POLL_LIMIT, "") {
922                Ok(next) => spins = next,
923                Err(_) => return None,
924            }
925        }
926    }
927
928    /// Clear PxCMD.ST then PxCMD.FRE, waiting for CR/FR to drop.
929    fn stop_port_engine(mmio: &MmioRegion, p: usize) -> ViResult<()> {
930        let cmd = rd(mmio, p + PX_CMD)?;
931        if cmd & CMD_ST != 0 {
932            wr(mmio, p + PX_CMD, cmd & !CMD_ST)?;
933            let mut spins = 0u64;
934            loop {
935                if rd(mmio, p + PX_CMD)? & CMD_CR == 0 {
936                    break;
937                }
938                spins = poll_tick(spins, POLL_LIMIT, "[ahci] port stop never completed")?;
939            }
940        }
941        let cmd = rd(mmio, p + PX_CMD)?;
942        if cmd & CMD_FRE != 0 {
943            wr(mmio, p + PX_CMD, cmd & !CMD_FRE)?;
944            let mut spins = 0u64;
945            loop {
946                if rd(mmio, p + PX_CMD)? & CMD_FR == 0 {
947                    break;
948                }
949                spins = poll_tick(spins, POLL_LIMIT, "[ahci] FIS receive never stopped")?;
950            }
951        }
952        Ok(())
953    }
954}

956// ── Register access (bounds-checked by MmioRegion) ────────────────────────────
957
958#[inline]
959fn rd(mmio: &MmioRegion, off: usize) -> ViResult<u32> {
960    mmio.read_u32(off)
961}
962
963#[inline]
964fn wr(mmio: &MmioRegion, off: usize, val: u32) -> ViResult<()> {
965    mmio.write_u32(off, val)
966}
967
968/// Count one poll iteration and enforce the budget.
969#[inline]
970fn poll_tick(mut spins: u64, limit: u64, what: &str) -> ViResult<u64> {
971    spins += 1;
972    if spins.is_multiple_of(POLL_YIELD_EVERY) {
973        ostd::task::yield_now();
974    }
975    if spins >= limit {
976        if !what.is_empty() {
977            let _ = print_fmt(format_args!("[ahci] timeout: {} ({} polls)\n", what, limit));
978        }
979        return Err(ViError::IO);
980    }
981    fence(Ordering::Acquire);
982    Ok(spins)
983}
984
985/// Hold COMRESET for at least the AHCI-mandated ~1 ms.
986///
987/// Uses the monotonic millisecond clock when the platform exposes one (`None`
988/// on a profile without a timer); an iteration count alone is not a time lower
989/// bound, since its duration depends on CPU frequency, `PAUSE`, and scheduler
990/// state — a fast physical CPU could deassert COMRESET too early.
991fn delay_ms() {
992    let start = ostd::syscall::sys_get_time_ms();
993    loop {
994        for _ in 0..16 {
995            ostd::task::yield_now();
996        }
997        for _ in 0..100_000 {
998            core::hint::spin_loop();
999        }
1000        match (start, ostd::syscall::sys_get_time_ms()) {
1001            (Some(start), Some(now)) if now.saturating_sub(start) >= 1 => return,
1002            (Some(_), Some(_)) => continue,
1003            // No monotonic clock: the yield/spin pass above is the whole bound.
1004            _ => return,
1005        }
1006    }
1007}

1011fn alloc_dma(bdf: u32, pages: usize) -> ViResult<AuthorizedDma<DmaBuf>> {
1012    let buf = DmaBuf::alloc(pages).ok_or(ViError::OutOfMemory)?;
1013    AuthorizedDma::authorize(buf, |b| b.authorize(bdf)).map_err(|_| ViError::PermissionDenied)
1014}

1016fn dma_write_u8(buf: &AuthorizedDma<DmaBuf>, off: usize, val: u8) {
1017    // SAFETY: `buf` is a live DMA allocation owned by this cell and `off` is
1018    // within it by construction (fixed command-table layout).
1019    unsafe { core::ptr::write_volatile(buf.inner().virt().add(off), val) };
1020}

1022fn dma_write_u32(buf: &AuthorizedDma<DmaBuf>, off: usize, val: u32) {
1023    // SAFETY: same contract as `dma_write_u8`; `off` is 4-byte aligned.
1024    unsafe {
1025        core::ptr::write_volatile(buf.inner().virt().add(off) as *mut u32, val);
1026    }
1027}

1029fn dma_write_u64(buf: &AuthorizedDma<DmaBuf>, off: usize, val: u64) {
1030    // SAFETY: same contract as `dma_write_u8`; `off` is 8-byte aligned.
1031    unsafe {
1032        core::ptr::write_volatile(buf.inner().virt().add(off) as *mut u64, val);
1033    }
1034}

1036fn dma_read_u16(buf: &AuthorizedDma<DmaBuf>, off: usize) -> u16 {
1037    // SAFETY: same contract as `dma_write_u8`; `off` is 2-byte aligned.
1038    unsafe { core::ptr::read_volatile(buf.inner().virt().add(off) as *const u16) }
1039}

1041fn dma_read_u32(buf: &AuthorizedDma<DmaBuf>, off: usize) -> u32 {
1042    // SAFETY: same contract as `dma_write_u8`; `off` is 4-byte aligned.
1043    unsafe { core::ptr::read_volatile(buf.inner().virt().add(off) as *const u32) }
1044}

1046/// Decode an ATA string field (`words` little-endian words starting at `word`).
1047///
1048/// ATA strings are byte-swapped inside each word, so the high byte holds the
1049/// earlier character; a NUL or non-ASCII byte is dropped so a malformed
1050/// response cannot produce garbage in the log.
1051fn ata_string(buf: &AuthorizedDma<DmaBuf>, word: usize, words: usize) -> String {
1052    let mut s = String::with_capacity(words * 2);
1053    for w in 0..words {
1054        let raw = dma_read_u16(buf, (word + w) * 2);
1055        for b in [(raw >> 8) as u8, (raw & 0xFF) as u8] {
1056            if (0x20..0x7F).contains(&b) {
1057                s.push(b as char);
1058            }
1059        }
1060    }
1061    let trimmed = s.trim();
1062    let mut out = String::with_capacity(trimmed.len());
1063    out.push_str(trimmed);
1064    out
1065}
