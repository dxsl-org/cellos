7#[derive(Clone, Copy, PartialEq, Eq)]
8pub struct Color {
9    pub b: u8,
10    pub g: u8,
11    pub r: u8,
12    pub a: u8,
13}
14
15impl Color {
16    pub const fn rgb(r: u8, g: u8, b: u8) -> Self {
17        Self { r, g, b, a: 255 }
18    }
19
20    #[allow(dead_code)]
21    pub const fn rgba(r: u8, g: u8, b: u8, a: u8) -> Self {
22        Self { r, g, b, a }
23    }
24
25    pub const fn as_bgra_bytes(self) -> [u8; 4] {
26        [self.b, self.g, self.r, self.a]
27    }
28}

47pub fn fill_rect(surf: &mut ViSurface, x: i32, y: i32, w: u32, h: u32, color: Color) {
48    let sw = surf.width() as i32;
49    let sh = surf.height() as i32;
50    let stride = surf.stride();
51    let pixels = surf.pixels_mut();
52
53    let x0 = x.max(0);
54    let y0 = y.max(0);
55    let x1 = (x + w as i32).min(sw);
56    let y1 = (y + h as i32).min(sh);
57    if x0 >= x1 || y0 >= y1 {
58        return;
59    }
60
61    let [b, g, r, a] = color.as_bgra_bytes();
62    for py in y0..y1 {
63        let row_start = py as usize * stride;
64        for px in x0..x1 {
65            let offset = row_start + px as usize * 4;
66            pixels[offset] = b;
67            pixels[offset + 1] = g;
68            pixels[offset + 2] = r;
69            pixels[offset + 3] = a;
70        }
71    }
72}

74pub fn stroke_rect(surf: &mut ViSurface, x: i32, y: i32, w: u32, h: u32, border: Color) {
75    if w == 0 || h == 0 {
76        return;
77    }
78    fill_rect(surf, x, y, w, 1, border);
79    fill_rect(surf, x, y + h as i32 - 1, w, 1, border);
80    fill_rect(surf, x, y, 1, h, border);
81    fill_rect(surf, x + w as i32 - 1, y, 1, h, border);
82}

84#[allow(clippy::too_many_arguments)]
85pub fn draw_text(
86    surf: &mut ViSurface,
87    fonts: &mut TextFonts,
88    x: f32,
89    y: f32,
90    text: &str,
91    face: FontFace,
92    px: f32,
93    color: Color,
94    clip: (i32, i32, i32, i32),
95) {
96    let width = surf.width();
97    let height = surf.height();
98    let stride = surf.stride();
99    fonts.draw_text(
100        surf.pixels_mut(),
101        width,
102        height,
103        stride,
104        x,
105        y,
106        text,
107        face,
108        px,
109        color.as_bgra_bytes(),
110        clip,
111    );
112}

114/// Fit a single line into a rectangle; long labels are clipped at its edge.
115#[allow(clippy::too_many_arguments)]
116pub fn draw_label(
117    surf: &mut ViSurface,
118    fonts: &mut TextFonts,
119    x: i32,
120    y: i32,
121    w: u32,
122    h: u32,
123    text: &str,
124    face: FontFace,
125    px: f32,
126    color: Color,
127    centered: bool,
128) {
129    if w == 0 || h == 0 {
130        return;
131    }
132    let text_x = if centered {
133        x as f32 + (w as f32 - fonts.measure(face, text, px).min(w as f32)) / 2.0
134    } else {
135        x as f32
136    };
137    let text_y = y as f32 + (h as f32 - fonts.line_height(face, px)) / 2.0;
138    draw_text(
139        surf,
140        fonts,
141        text_x,
142        text_y,
143        text,
144        face,
145        px,
146        color,
147        (x, y, x + w as i32, y + h as i32),
148    );
149}

151#[allow(clippy::too_many_arguments)]
152pub fn draw_button(
153    surf: &mut ViSurface,
154    fonts: &mut TextFonts,
155    x: i32,
156    y: i32,
157    w: u32,
158    h: u32,
159    text: &str,
160    bg: Color,
161    border: Color,
162    fg: Color,
163) {
164    fill_rect(surf, x, y, w, h, bg);
165    stroke_rect(surf, x, y, w, h, border);
166    draw_label(
167        surf,
168        fonts,
169        x + 6,
170        y + 1,
171        w.saturating_sub(12),
172        h.saturating_sub(2),
173        text,
174        FontFace::UiSemibold,
175        14.0,
176        fg,
177        true,
178    );
