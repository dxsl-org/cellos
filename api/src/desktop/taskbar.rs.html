12pub const TASKBAR_HEIGHT: u32 = 48;
13pub const MAX_VISIBLE_APPS: usize = 4;
14const BUTTON_Y: i32 = 6;
15const BUTTON_HEIGHT: u32 = 36;
16const APPS_X: i32 = 204;
17const APP_WIDTH: u32 = 148;
18const APP_STEP: i32 = APP_WIDTH as i32 + 6;
19const PAGE_WIDTH: u32 = 28;
20const PAGE_STEP: i32 = 34;
21const MORE_WIDTH: u32 = 64;
22
23/// Shared by painting and hit testing, including narrow-screen pagination.
24struct Layout {
25    tray_x: i32,
26    start: usize,
27    end: usize,
28    capacity: usize,
29    paged: bool,
30}
31
32impl Layout {
33    fn new(width: u32, total: usize, offset: usize) -> Self {
34        let tray_width = if width >= 600 { 212 } else { 64 };
35        let tray_x = (width as i32 - tray_width - 8).max(APPS_X);
36        let available = (tray_x - 12 - APPS_X).max(0);
37        // Reserve both arrows and More so page changes never move into the tray.
38        let capacity = ((available - 2 * PAGE_STEP - MORE_WIDTH as i32 - 6).max(0)
39            / APP_STEP) as usize;
40        let capacity = capacity.min(MAX_VISIBLE_APPS);
41        let start = if capacity == 0 {
42            0
43        } else {
44            offset.min(total.saturating_sub(1)) / capacity * capacity
45        };
46        Self {
47            tray_x,
48            start,
49            end: (start + capacity).min(total),
50            capacity,
51            paged: total > capacity,
52        }
53    }
54
55    fn apps_x(&self) -> i32 {
56        APPS_X + if self.paged { PAGE_STEP } else { 0 }
57    }
58
59    fn next_x(&self) -> i32 {
60        self.apps_x() + (self.end - self.start) as i32 * APP_STEP
61    }
62
63    fn more_x(&self) -> i32 {
64        self.tray_x - 12 - MORE_WIDTH as i32
65    }
66}

76pub struct TaskbarState {
77    pub mouse_x: i32,
78    pub mouse_y: i32,
79    pub page_offset: usize,
80    pub sub_taskbar_open: bool,
81    pub last_clock_min: u32,
82    pub clock_buf: [u8; 5],
83    pub net_online: bool,
84    pub vfs_online: bool,
85    pub ai_online: bool,
86}
87
88impl TaskbarState {
89    pub fn new() -> Self {
90        Self {
91            mouse_x: -1,
92            mouse_y: -1,
93            page_offset: 0,
94            sub_taskbar_open: false,
95            last_clock_min: u32::MAX,
96            clock_buf: *b"12:00",
97            net_online: false,
98            vfs_online: false,
99            ai_online: false,
100        }
101    }
102
103    pub fn update_daemons(&mut self) {
104        self.net_online = sys_lookup_service(service::NET).is_some();
105        self.vfs_online = sys_lookup_service(service::VFS).is_some();
106        self.ai_online = sys_lookup_service(15 /* AI */).is_some();
107
108        // Calculate time from monotonic ticks:
109        let ticks = sys_get_time();
110        let total_secs = ticks / 10_000_000;
111        let mins = ((total_secs / 60) % 60) as u32;
112        let hours = ((total_secs / 3600) % 24) as u32;
113
114        self.clock_buf[0] = b'0' + (hours / 10) as u8;
115        self.clock_buf[1] = b'0' + (hours % 10) as u8;
116        self.clock_buf[2] = b':';
117        self.clock_buf[3] = b'0' + (mins / 10) as u8;
118        self.clock_buf[4] = b'0' + (mins % 10) as u8;
119        self.last_clock_min = mins;
120    }
121}

123pub fn render(
124    surf: &mut ViSurface,
125    fonts: &mut TextFonts,
126    registry: &AppRegistry,
127    state: &TaskbarState,
128) {
129    let sw = surf.width();
130    let sh = surf.height();
131    let pinned = registry.pinned_apps();
132    let layout = Layout::new(sw, pinned.len(), state.page_offset);
133
134    fill_rect(surf, 0, 0, sw, sh, theme::TASKBAR_BG);
135    fill_rect(surf, 0, 0, sw, 1, theme::TASKBAR_BORDER);
136
137    for (x, w, text, border, fg) in [
138        (8, 84, "CellOS", theme::TASKBAR_BORDER, theme::ACCENT_CYAN),
139        (98, 94, "Search", theme::ACCENT_BLUE, theme::TEXT_PRIMARY),
140    ] {
141        let bg = if is_inside(state.mouse_x, state.mouse_y, x, BUTTON_Y, w, BUTTON_HEIGHT) {
142            theme::BTN_HOVER
143        } else {
144            theme::BTN_NORMAL
145        };
146        draw_button(surf, fonts, x, BUTTON_Y, w, BUTTON_HEIGHT, text, bg, border, fg);
147    }
148    fill_rect(surf, 198, 10, 1, 28, theme::TASKBAR_BORDER);
149
150    if layout.paged && layout.start > 0 {
151        render_page_button(surf, fonts, state, APPS_X, "<");
152    }
153    let mut cursor_x = layout.apps_x();
154    for app in &pinned[layout.start..layout.end] {
155        let bg = if is_inside(
156            state.mouse_x, state.mouse_y, cursor_x, BUTTON_Y, APP_WIDTH, BUTTON_HEIGHT,
157        ) {
158            theme::BTN_HOVER
159        } else {
160            theme::BTN_NORMAL
161        };
162        fill_rect(surf, cursor_x, BUTTON_Y, APP_WIDTH, BUTTON_HEIGHT, bg);
163        stroke_rect(surf, cursor_x, BUTTON_Y, APP_WIDTH, BUTTON_HEIGHT, theme::TASKBAR_BORDER);
164        draw_label(
165            surf, fonts, cursor_x + 6, BUTTON_Y, 28, BUTTON_HEIGHT, app.icon,
166            FontFace::UiRegular, 14.0, theme::ACCENT_CYAN, true,
167        );
168        draw_label(
169            surf, fonts, cursor_x + 40, BUTTON_Y, APP_WIDTH - 48, BUTTON_HEIGHT,
170            app.name, FontFace::UiRegular, 14.0, theme::TEXT_PRIMARY, false,
171        );
172        cursor_x += APP_STEP;
173    }
174    if layout.capacity > 0 && layout.paged && layout.end < pinned.len() {
175        render_page_button(surf, fonts, state, layout.next_x(), ">");
176    }
177    if layout.paged {
178        let x = layout.more_x();
179        let bg = if state.sub_taskbar_open {
180            theme::BTN_ACTIVE
181        } else if is_inside(state.mouse_x, state.mouse_y, x, BUTTON_Y, MORE_WIDTH, BUTTON_HEIGHT) {
182            theme::BTN_HOVER
183        } else {
184            theme::BTN_NORMAL
185        };
186        draw_button(
187            surf, fonts, x, BUTTON_Y, MORE_WIDTH, BUTTON_HEIGHT, "More", bg,
188            theme::TASKBAR_BORDER, theme::ACCENT_CYAN,
189        );
190    }
191
192    let mut right_x = layout.tray_x;
193    fill_rect(surf, right_x - 6, 10, 1, 28, theme::TASKBAR_BORDER);
194    if sw >= 600 {
195        for (label, online, online_color) in [
196            ("NET", state.net_online, theme::ACCENT_GREEN),
197            ("VFS", state.vfs_online, theme::ACCENT_GREEN),
198            ("AI", state.ai_online, theme::ACCENT_CYAN),
199        ] {
200            let color = if online { online_color } else { theme::TEXT_MUTED };
201            draw_label(
202                surf, fonts, right_x, BUTTON_Y, 42, BUTTON_HEIGHT, label,
203                FontFace::UiRegular, 14.0, color, true,
204            );
205            right_x += 48;
206        }
207    }
208    let clock = core::str::from_utf8(&state.clock_buf).unwrap_or("12:00");
209    draw_label(
210        surf, fonts, right_x, BUTTON_Y, 64, BUTTON_HEIGHT, clock,
211        FontFace::UiSemibold, 14.0, theme::TEXT_PRIMARY, true,
212    );
213    surf.damage_all();
214}

216fn render_page_button(
217    surf: &mut ViSurface,
218    fonts: &mut TextFonts,
219    state: &TaskbarState,
220    x: i32,
221    text: &str,
222) {
223    let bg = if is_inside(state.mouse_x, state.mouse_y, x, BUTTON_Y, PAGE_WIDTH, BUTTON_HEIGHT) {
224        theme::BTN_HOVER
225    } else {
226        theme::BTN_NORMAL
227    };
228    draw_button(
229        surf, fonts, x, BUTTON_Y, PAGE_WIDTH, BUTTON_HEIGHT, text, bg,
230        theme::TASKBAR_BORDER, theme::TEXT_PRIMARY,
231    );
232}

234pub fn handle_click(
235    registry: &mut AppRegistry,
236    state: &mut TaskbarState,
237    x: i32,
238    y: i32,
239    width: u32,
240) -> TaskbarAction {
241    if is_inside(x, y, 8, BUTTON_Y, 84, BUTTON_HEIGHT)
242        || is_inside(x, y, 98, BUTTON_Y, 94, BUTTON_HEIGHT)
243    {
244        return TaskbarAction::OpenSpotlight;
245    }
246    let pinned = registry.pinned_apps();
247    let layout = Layout::new(width, pinned.len(), state.page_offset);
248    if layout.paged && layout.start > 0
249        && is_inside(x, y, APPS_X, BUTTON_Y, PAGE_WIDTH, BUTTON_HEIGHT)
250    {
251        state.page_offset = layout.start.saturating_sub(layout.capacity);
252        return TaskbarAction::None;
253    }
254    let mut cursor_x = layout.apps_x();
255    for app in &pinned[layout.start..layout.end] {
256        if is_inside(x, y, cursor_x, BUTTON_Y, APP_WIDTH, BUTTON_HEIGHT) {
257            return TaskbarAction::LaunchApp(app.path);
258        }
259        cursor_x += APP_STEP;
260    }
261    if layout.capacity > 0 && layout.paged && layout.end < pinned.len()
262        && is_inside(x, y, layout.next_x(), BUTTON_Y, PAGE_WIDTH, BUTTON_HEIGHT)
263    {
264        state.page_offset = layout.end;
265        return TaskbarAction::None;
266    }
267    if layout.paged
268        && is_inside(x, y, layout.more_x(), BUTTON_Y, MORE_WIDTH, BUTTON_HEIGHT)
269    {
270        if layout.capacity == 0 {
271            return TaskbarAction::OpenSpotlight;
272        }
273        state.sub_taskbar_open = !state.sub_taskbar_open;
274        state.page_offset = if layout.end < pinned.len() { layout.end } else { 0 };
275        return TaskbarAction::ToggleSubTaskbar;
276    }
277    TaskbarAction::None
278}

280fn is_inside(mx: i32, my: i32, x: i32, y: i32, w: u32, h: u32) -> bool {
281    mx >= x && mx < x + w as i32 && my >= y && my < y + h as i32
282}

284#[cfg(test)]
285mod tests {
286    use super::*;
287
288    #[test]
289    fn narrow_taskbar_more_opens_application_search() {
290        let mut registry = AppRegistry::new();
291        let mut state = TaskbarState::new();
292        for width in [600, 640, 700, 727] {
293            let layout = Layout::new(width, registry.pinned_apps().len(), 0);
294            assert_eq!(layout.capacity, 0);
295            assert!(layout.paged);
296            assert!(layout.more_x() >= APPS_X);
297            assert!(layout.more_x() + (MORE_WIDTH as i32) < layout.tray_x);
298            assert!(matches!(
299                handle_click(&mut registry, &mut state, layout.more_x() + 1, BUTTON_Y + 1, width),
300                TaskbarAction::OpenSpotlight
301            ));
302        }
303    }
