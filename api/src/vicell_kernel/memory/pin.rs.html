1//! Async Pinning Registry — memory regions an in-flight asynchronous operation
2//! may still be reading or writing.
3//!
4//! # Contract
5//!
6//! A region is *pinned* while an agent outside the owning cell's control — a
7//! device programmed through the IOMMU, a kernel ISR, or a service holding a raw
8//! pointer into the region — may still touch its frames. While a region is
9//! pinned:
10//!
11//! * the owner may not free it: `GrantFree` and `GrantUnregister` refuse rather
12//!   than hand the frames back to the allocator, and
13//! * if the owner dies, the frames are **quarantined** — withheld from the frame
14//!   allocator instead of returned to it.
15//!
16//! Quarantine is the third option between the two failure modes. Returning a
17//! pinned frame leaves a device writing into whatever cell is allocated it next;
18//! blocking the death until the pin clears breaks the never-die property, since
19//! init restarts every permanent service and a cell dying mid-operation is the
20//! ordinary case. Nothing here ever delays a death: marking is a bounded scan of
21//! a fixed table and takes no other lock.
22//!
23//! Quarantine ends only on an explicit driver acknowledgement
24//! ([`acknowledge`]) — never on a timer, never implicitly. Frames with no
25//! acknowledgement stay withheld for the life of the boot, which leaks memory
26//! but cannot corrupt another cell.
27//!
28//! # Lock order
29//!
30//! `REGISTRY` is a **leaf**: no other lock is acquired while it is held, and it
31//! is never held across `FRAME_ALLOCATOR`, `KERNEL_ROOT` or `SCHEDULER`. The
32//! grant teardown path therefore runs
33//!
34//! ```text
35//! PAGE_GRANT_TABLE / REG_GRANT_TABLE → REGISTRY → (all released) → FRAME_ALLOCATOR → KERNEL_ROOT
36//! ```
37//!
38//! `FRAME_ALLOCATOR` is the OUTER lock of the last pair, contrary to what the
39//! older comments on `reap_grants_for_task` and `yield_cpu` claimed:
40//! `free_grant_pages` takes `FRAME_ALLOCATOR` first and holds it across
41//! `unmap_page`/`map_page`, each of which takes `KERNEL_ROOT` internally.
42//! Neither is ever acquired while `SCHEDULER` is held — the watchdog defers its
43//! reap list precisely so `yield_cpu` can drain it after dropping `SCHEDULER`.
44
45use crate::sync::Spinlock;
46use alloc::vec::Vec;

48const PAGE_SIZE: usize = 4096;
49
50/// Regions the kernel tracks as pinned at one time, across all cells.
51///
52/// A fixed table rather than a map: the count must not be driven by anything a
53/// caller supplies, and the teardown paths that scan it run on cell death.
54const MAX_PINNED_REGIONS: usize = 128;
55
56/// Per-owner ceiling, so one driver cell cannot starve the table.
57///
58/// Sized from the largest real consumer: the e1000 cell authorises 34 regions
59/// (two rings plus 16 TX and 16 RX buffers).
60const MAX_PINS_PER_TASK: usize = 48;
61
62/// Frame ranges the quarantine can hold at one time. Entries are only created
63/// by the reaper, one per grant a dying cell had pinned, so the bound is the
64/// number of grants that can be in flight across all dead cells at once.
65const MAX_QUARANTINED_REGIONS: usize = 64;
66
67/// Why a region could not be pinned.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum PinError {
70    /// Zero length, or `base + len` overflows the address space.
71    InvalidRange,
72    /// The global table is full.
73    TableFull,
74    /// This owner already holds [`MAX_PINS_PER_TASK`] regions.
75    TaskLimit,
76}
77
78/// Who holds a pin overlapping a queried region.
79#[derive(Debug, Clone, Copy)]
80pub struct PinHolder {
81    /// Page-aligned base of the pinned region.
82    pub base: usize,
83    /// Length of the pinned region in pages.
84    pub pages: usize,
85    /// Task id that owns the memory participating in the operation.
86    pub owner: usize,
87    /// Outstanding pin requests against this region. Released together by
88    /// [`acknowledge`]; there is no per-request release today.
89    pub holds: u32,
90    /// The owner has died and these frames are withheld from the allocator.
91    pub quarantined: bool,
92}

94#[derive(Clone, Copy)]
95struct PinEntry {
96    base: usize,
97    /// Zero marks a free slot.
98    pages: usize,
99    owner: usize,
100    holds: u32,
101    quarantined: bool,
102}

104/// Frames the reaper declined to return to the allocator, awaiting an
105/// acknowledgement from `owner`.
106#[derive(Clone, Copy)]
107struct FrameHold {
108    base: usize,
109    /// Zero marks a free slot.
110    pages: usize,
111    owner: usize,
112}
113
114const EMPTY_PIN: PinEntry = PinEntry {
115    base: 0,
116    pages: 0,
117    owner: 0,
118    holds: 0,
119    quarantined: false,
120};
121
122const EMPTY_HOLD: FrameHold = FrameHold {
123    base: 0,
124    pages: 0,
125    owner: 0,
126};
127
128struct Registry {
129    pins: [PinEntry; MAX_PINNED_REGIONS],
130    quarantine: [FrameHold; MAX_QUARANTINED_REGIONS],
131}
132
133/// One lock covers both tables: they are only ever consulted together, and a
134/// single leaf lock removes any ordering question between them.
135static REGISTRY: Spinlock<Registry> = Spinlock::new(Registry {
136    pins: [EMPTY_PIN; MAX_PINNED_REGIONS],
137    quarantine: [EMPTY_HOLD; MAX_QUARANTINED_REGIONS],
138});

140/// Page-aligned `(base, page_count)` covering `[base, base + len)`.
141fn span(base: usize, len: usize) -> Option<(usize, usize)> {
142    if len == 0 {
143        return None;
144    }
145    let start = base & !(PAGE_SIZE - 1);
146    let end = base.checked_add(len)?.checked_add(PAGE_SIZE - 1)? & !(PAGE_SIZE - 1);
147    Some((start, (end - start) / PAGE_SIZE))
148}
149
150fn overlaps(a: (usize, usize), b: (usize, usize)) -> bool {
151    a.0 < b.0 + b.1 * PAGE_SIZE && b.0 < a.0 + a.1 * PAGE_SIZE
152}

154/// Record `[base, base + len)` as participating in an in-flight operation owned
155/// by task `owner`, so the region cannot be freed underneath it.
156///
157/// Re-pinning a range already held by the same owner bumps its hold count
158/// instead of consuming a second slot.
159///
160/// # Errors
161/// [`PinError::InvalidRange`] for an empty or overflowing range,
162/// [`PinError::TaskLimit`] once the owner holds [`MAX_PINS_PER_TASK`] regions,
163/// [`PinError::TableFull`] when no slot is left. Callers must fail the operation
164/// closed: authorising a device against memory the kernel cannot protect is the
165/// use-after-free this registry exists to prevent.
166pub fn pin(base: usize, len: usize, owner: usize) -> Result<(), PinError> {
167    let (start, pages) = span(base, len).ok_or(PinError::InvalidRange)?;
168    let mut reg = REGISTRY.lock();
169    let mut owned = 0usize;
170    let mut free_slot = None;
171    let mut existing = None;
172    for (i, e) in reg.pins.iter().enumerate() {
173        if e.pages == 0 {
174            if free_slot.is_none() {
175                free_slot = Some(i);
176            }
177            continue;
178        }
179        if e.owner == owner {
180            owned += 1;
181            if e.base == start && e.pages == pages && !e.quarantined {
182                existing = Some(i);
183                break;
184            }
185        }
186    }
187    if let Some(i) = existing {
188        reg.pins[i].holds = reg.pins[i].holds.saturating_add(1);
189        return Ok(());
190    }
191    if owned >= MAX_PINS_PER_TASK {
192        return Err(PinError::TaskLimit);
193    }
194    let slot = free_slot.ok_or(PinError::TableFull)?;
195    reg.pins[slot] = PinEntry {
196        base: start,
197        pages,
198        owner,
199        holds: 1,
200        quarantined: false,
201    };
202    Ok(())
203}

205/// The first pin overlapping `[base, base + len)`, if any.
206///
207/// Teardown paths call this to decide whether frames may be released.
208pub fn holder_of(base: usize, len: usize) -> Option<PinHolder> {
209    let probe = span(base, len)?;
210    let reg = REGISTRY.lock();
211    reg.pins
212        .iter()
213        .find(|e| e.pages != 0 && overlaps(probe, (e.base, e.pages)))
214        .map(|e| PinHolder {
215            base: e.base,
216            pages: e.pages,
217            owner: e.owner,
218            holds: e.holds,
219            quarantined: e.quarantined,
220        })
221}

223/// Mark every pin owned by `tid` as quarantined and report how many.
224///
225/// Called from the grant reaper before the grant tables are swept. Marking does
226/// not free anything and does not wait: the dying task proceeds immediately.
227pub fn quarantine_task(tid: usize) -> usize {
228    let mut reg = REGISTRY.lock();
229    let mut marked = 0;
230    for e in reg.pins.iter_mut() {
231        if e.pages != 0 && e.owner == tid {
232            e.quarantined = true;
233            marked += 1;
234        }
235    }
236    marked
237}

239/// Take custody of `pages` frames at `base` that the reaper declined to free,
240/// charging them to the pin holder `owner` whose acknowledgement releases them.
241///
242/// Only the reaper calls this, and only for frames it was otherwise about to
243/// return to the allocator. That is what makes [`acknowledge`] safe to free
244/// blindly: a range that never came out of a grant — an MMIO window a driver
245/// authorised for DMA, or a grant whose ownership passed to a live grantee —
246/// never enters the quarantine and so is never handed to `deallocate_frame`.
247///
248/// Returns `false` when the quarantine is full. The caller must still not free
249/// the frames: they are leaked for the life of the boot, which is the safe
250/// direction.
251pub fn withhold_frames(base: usize, pages: usize, owner: usize) -> bool {
252    if pages == 0 {
253        return false;
254    }
255    let mut reg = REGISTRY.lock();
256    match reg.quarantine.iter_mut().find(|q| q.pages == 0) {
257        Some(slot) => {
258            *slot = FrameHold { base, pages, owner };
259            true
260        }
261        None => false,
262    }
263}
264
265/// Driver acknowledgement for `tid`: drop its pins and hand back the
266/// `(base, pages)` frame ranges the reaper placed in quarantine on its behalf.
267///
268/// The caller returns those frames to the allocator. Safe to call before the
269/// reaper as well as after — an acknowledgement that arrives first simply drops
270/// the pins, and the reaper then frees the frames itself.
271pub fn acknowledge(tid: usize) -> Vec<(usize, usize)> {
272    let mut released = Vec::new();
273    let mut reg = REGISTRY.lock();
274    for e in reg.pins.iter_mut() {
275        if e.pages != 0 && e.owner == tid {
276            *e = EMPTY_PIN;
277        }
278    }
279    for q in reg.quarantine.iter_mut() {
280        if q.pages != 0 && q.owner == tid {
281            released.push((q.base, q.pages));
282            *q = EMPTY_HOLD;
283        }
284    }
285    released
286}
287
288/// Pages currently withheld from the frame allocator.
289pub fn quarantined_pages() -> usize {
290    let reg = REGISTRY.lock();
291    reg.quarantine.iter().map(|q| q.pages).sum()
292}

294/// Release the registry lock unconditionally during fault teardown.
295///
296/// # Safety
297/// Only valid from `force_unlock_all_kernel_locks`, with interrupts disabled and
298/// no other context able to observe a half-written table.
299pub unsafe fn force_unlock() {
300    REGISTRY.force_unlock();
301}
