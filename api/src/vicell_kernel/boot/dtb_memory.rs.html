4mod ranges;
5use ranges::*;

7const PAGE_SIZE: usize = 4096;
8const MAX_RANGES: usize = super::MAX_MEMORY_MAP_ENTRIES;
9const MAX_BOUNDARIES: usize = MAX_RANGES * 2 + 4;
10
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub enum MapError {
13    NoMemory,
14    InvalidRange,
15    KernelOutsideMemory,
16    DynamicReservation,
17    TooManyRanges,
18    NoUsableMemory,
19}
20
21pub fn build(
22    tree: &fdt::Fdt<'_>,
23    kernel_base: usize,
24    kernel_end: usize,
25    output: &mut [MemoryMapEntry],
26) -> Result<usize, MapError> {
27    let kernel_end = align_up(kernel_end, PAGE_SIZE)?;
28    if kernel_base >= kernel_end {
29        return Err(MapError::InvalidRange);
30    }
31
32    let mut ram = [Range::default(); MAX_RANGES];
33    let mut ram_count = 0;
34    for node in tree.all_nodes() {
35        if node.name.split('@').next() != Some("memory") || !is_enabled(&node) {
36            continue;
37        }
38        let regions = node.reg().ok_or(MapError::InvalidRange)?;
39        for region in regions {
40            push_range(
41                &mut ram,
42                &mut ram_count,
43                Range::ram(
44                    region.starting_address as usize,
45                    region.size.ok_or(MapError::InvalidRange)?,
46                )?,
47            )?;
48        }
49    }
50    if ram_count == 0 {
51        return Err(MapError::NoMemory);
52    }
53    sort_ranges(&mut ram[..ram_count]);
54    ram_count = merge_ranges(&mut ram, ram_count);
55
56    let kernel = Range {
57        start: kernel_base,
58        end: kernel_end,
59    };
60    let kernel_ram = ram[..ram_count]
61        .iter()
62        .copied()
63        .find(|range| range.start <= kernel.start && kernel.end <= range.end)
64        .ok_or(MapError::KernelOutsideMemory)?;
65    let firmware = Range {
66        start: kernel_ram.start,
67        end: kernel_base,
68    };
69
70    let mut reserved = [Range::default(); MAX_RANGES];
71    let mut reserved_count = 0;
72    for reservation in tree.memory_reservations() {
73        push_range(
74            &mut reserved,
75            &mut reserved_count,
76            Range::protected(reservation.address() as usize, reservation.size())?,
77        )?;
78    }
79    if let Some(parent) = tree.find_node("/reserved-memory") {
80        for child in parent.children().filter(is_enabled) {
81            let Some(regions) = child.reg() else {
82                if child.property("size").is_some() {
83                    return Err(MapError::DynamicReservation);
84                }
85                continue;
86            };
87            let mut found = false;
88            for region in regions {
89                found = true;
90                push_range(
91                    &mut reserved,
92                    &mut reserved_count,
93                    Range::protected(
94                        region.starting_address as usize,
95                        region.size.ok_or(MapError::InvalidRange)?,
96                    )?,
97                )?;
98            }
99            if !found && child.property("size").is_some() {
100                return Err(MapError::DynamicReservation);
101            }
102        }
103    }
104
105    let mut output_count = 0;
106    let mut usable_count = 0;
107    for memory in ram[..ram_count].iter().copied() {
108        let mut boundaries = [0usize; MAX_BOUNDARIES];
109        let mut boundary_count = 0;
110        push_boundary(&mut boundaries, &mut boundary_count, memory.start)?;
111        push_boundary(&mut boundaries, &mut boundary_count, memory.end)?;
112        for protected in reserved[..reserved_count].iter().copied() {
113            add_boundaries(protected, memory, &mut boundaries, &mut boundary_count)?;
114        }
115        for protected in [firmware, kernel] {
116            add_boundaries(protected, memory, &mut boundaries, &mut boundary_count)?;
117        }
118        boundaries[..boundary_count].sort_unstable();
119        boundary_count = dedup(&mut boundaries, boundary_count);
120
121        for pair in boundaries[..boundary_count].windows(2) {
122            let segment = Range {
123                start: pair[0],
124                end: pair[1],
125            };
126            if segment.start == segment.end {
127                continue;
128            }
129            let ty = if segment.overlaps(kernel) {
130                MemoryType::Kernel
131            } else if reserved[..reserved_count]
132                .iter()
133                .copied()
134                .any(|range| range.overlaps(segment))
135            {
136                MemoryType::Reserved
137            } else if segment.overlaps(firmware) {
138                MemoryType::Bootloader
139            } else {
140                usable_count += 1;
141                MemoryType::Usable
142            };
143            emit(output, &mut output_count, segment, ty)?;
144        }
145    }
146    if usable_count == 0 {
147        return Err(MapError::NoUsableMemory);
148    }
149    Ok(output_count)
150}
