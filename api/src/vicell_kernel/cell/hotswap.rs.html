1//! Hot-swap support primitives shared by the supervisor-owned replacement path.
2//!
3//! The legacy syscall-400 whole-sequence orchestrator is retired. This module now
4//! keeps only the still-live freeze state, replacement binding, mailbox cutover,
5//! and `HotSwapReady` bookkeeping used by opcodes 401 and 413-422.
6
7use crate::sync::Spinlock;
8use alloc::collections::BTreeMap;
9use types::{CellId, ViError, ViResult};

11// ─── Freeze registry ─────────────────────────────────────────────────────────

13/// Global freeze set — cell ids whose incoming IPC should be queued rather than
14/// delivered.  Phase 02 will use this to buffer then flush messages to the new
15/// cell.  In Phase 01 the queuing is a no-op stub: existing callers simply fail
16/// to send if the cell is frozen (same as before).
17///
18/// Lock ordering: FROZEN (leaf) — never acquired while SCHEDULER is held.
19static FROZEN: Spinlock<alloc::collections::BTreeSet<u64>> =
20    Spinlock::new(alloc::collections::BTreeSet::new());
21
22/// One live replacement ceiling per frozen task id.
23///
24/// The record is written only after the supervisor captures the task's current
25/// CapSet and is cleared on every resume/kill terminal path. Missing records
26/// fail closed so replacement spawn never falls back to ambient caller caps.
27static SWAP_CEILINGS: Spinlock<BTreeMap<usize, crate::task::cap::CapSet>> =
28    Spinlock::new(BTreeMap::new());
29
30/// Force-release this module's lock during fault teardown.
31///
32/// # Safety
33/// Single-hart; called only from the fault/panic path with interrupts disabled.
34pub unsafe fn force_unlock_locks() {
35    FROZEN.force_unlock();
36    SWAP_CEILINGS.force_unlock();
37}
38
39/// Mark `cell_id` as frozen.  Subsequent `sys_send` calls to this cell will
40/// queue the message in the task's pending queue instead of delivering it.
41pub fn freeze(cell_id: CellId) {
42    FROZEN.lock().insert(cell_id.0);
43    log::info!("[hotswap] froze cell {}", cell_id.0);
44}
45
46/// Return true if `cell_id` is currently frozen.
47pub fn is_frozen(cell_id: CellId) -> bool {
48    FROZEN.lock().contains(&cell_id.0)
49}
50
51/// Remove `cell_id` from the freeze set and resume normal message delivery.
52pub fn unfreeze(cell_id: CellId) {
53    FROZEN.lock().remove(&cell_id.0);
54    log::info!("[hotswap] unfroze cell {}", cell_id.0);
55}

57/// Consume the live frozen task's ceiling for one replacement spawn attempt.
58///
59/// A freeze record is a one-shot authority token. Consuming it prevents a
60/// compromised supervisor from cloning multiple privileged replacements from
61/// one frozen task. A failed spawn must be rolled back with resume/refreeze.
62///
63/// Lock order is `SCHEDULER -> SWAP_CEILINGS`, shared by freeze, resume, exit,
64/// and consume. Holding both locks makes the live-Frozen check and token removal
65/// one atomic authority decision.
66pub(crate) fn take_frozen_replacement_ceiling(tid: usize) -> Option<crate::task::cap::CapSet> {
67    use crate::task::tcb::TaskState;
68
69    let scheduler = crate::task::SCHEDULER.lock();
70    let task = scheduler.as_ref()?.tasks.get(&tid)?;
71    if !matches!(task.state, TaskState::Frozen { .. }) {
72        return None;
73    }
74    SWAP_CEILINGS.lock().remove(&tid)
75}
76
77pub(crate) fn clear_swap_ceiling(tid: usize) {
78    SWAP_CEILINGS.lock().remove(&tid);
79}

81/// Bind a freshly spawned replacement to the frozen task whose one-shot
82/// capability ceiling authorized its creation.
83pub(crate) fn bind_replacement(source_tid: usize, target_tid: usize) -> bool {
84    let mut scheduler = crate::task::SCHEDULER.lock();
85    let Some(sched) = scheduler.as_mut() else {
86        return false;
87    };
88    let source_is_frozen = sched
89        .tasks
90        .get(&source_tid)
91        .is_some_and(|task| matches!(task.state, crate::task::tcb::TaskState::Frozen { .. }));
92    if !source_is_frozen {
93        return false;
94    }
95    let Some(target) = sched.tasks.get_mut(&target_tid) else {
96        return false;
97    };
98    if target.hotswap_source_tid.is_some() {
99        return false;
100    }
101    target.hotswap_source_tid = Some(source_tid);
102    true
103}

107/// Called from the `HotSwapReady` syscall handler (syscall 401) to record that
108/// the new cell has finished deserializing state.
109///
110/// Sets `Task::hotswap_ready = true` for `tid` under the SCHEDULER lock.
111pub fn set_task_hotswap_ready(tid: usize) {
112    let mut scheduler = crate::task::SCHEDULER.lock();
113    let Some(sched) = scheduler.as_mut() else {
114        return;
115    };
116    let Some(task) = sched.tasks.get_mut(&tid) else {
117        return;
118    };
119    task.hotswap_ready = true;
120    log::info!("[hotswap] task {} signalled HotSwapReady", tid);
121}

125fn enter_frozen_state(task: &mut crate::task::Task, swap_id: u64) {
126    task.hotswap_ingress_closed = false;
127    task.state = crate::task::tcb::TaskState::Frozen { swap_id };
128}

130/// Freeze one supervisor-managed task and snapshot its capability ceiling at
131/// the same scheduler transition point.
132pub(crate) fn freeze_task_with_ceiling(tid: usize, swap_id: u64) -> ViResult<()> {
133    use crate::task::tcb::TaskState;
134
135    // Lock order contract: SCHEDULER -> SWAP_CEILINGS. The task transition and
136    // ceiling publication are atomic to ResumeCell, KillCell, and replacement
137    // spawn, so no runnable task can retain a usable freeze record.
138    let mut scheduler = crate::task::SCHEDULER.lock();
139    let mut ceilings = SWAP_CEILINGS.lock();
140    if ceilings.contains_key(&tid) {
141        return Err(ViError::AlreadyExists);
142    }
143    let sched = scheduler.as_mut().ok_or(ViError::NotFound)?;
144    let task = sched.tasks.get_mut(&tid).ok_or(ViError::NotFound)?;
145    if task.is_critical || matches!(task.state, TaskState::Frozen { .. }) {
146        return Err(ViError::PermissionDenied);
147    }
148    let ceiling = crate::task::cap::CapSet::of_task(task);
149    enter_frozen_state(task, swap_id);
150    ceilings.insert(tid, ceiling);
151    crate::task::hart_local::ready::remove_from_all(tid);
152    Ok(())
153}

155/// Roll back a Frozen task to `TaskState::Ready` and re-queue it.
156///
157/// Called on swap abort so the old cell resumes from where it left off.
158pub(crate) fn unfreeze_task(tid: usize) {
159    use crate::task::tcb::TaskState;
160    let mut scheduler = crate::task::SCHEDULER.lock();
161    // Keep the SCHEDULER -> SWAP_CEILINGS order used by every record lifecycle.
162    clear_swap_ceiling(tid);
163    if let Some(sched) = scheduler.as_mut() {
164        for task in sched.tasks.values_mut() {
165            if task.hotswap_source_tid == Some(tid) {
166                task.hotswap_source_tid = None;
167            }
168        }
169        if let Some(task) = sched.tasks.get_mut(&tid) {
170            if matches!(task.state, TaskState::Frozen { .. }) {
171                task.hotswap_ingress_closed = false;
172                task.state = TaskState::Ready;
173                sched.push_ready(tid);
174            }
175        }
176    }
177}

179/// Atomically move a frozen provider's accepted IPC to its ready replacement.
180///
181/// The caller must hold `SupervisorCap`. This helper owns the cutover lock
182/// order: `SCHEDULER -> service registry`. A failed copy or registry compare
183/// removes every target append and leaves the source mailbox open and intact.
184pub(crate) fn commit_hotswap_barrier(
185    source_tid: usize,
186    target_tid: usize,
187    service_id: u16,
188) -> ViResult<()> {
189    use crate::task::pending_mailbox::{PendingMsg, PendingMsgData};
190    use crate::task::tcb::{TaskState, HOTSWAP_MSG_QUEUE_DEPTH};
191
192    let mut scheduler = crate::task::SCHEDULER.lock();
193    let sched = scheduler.as_mut().ok_or(ViError::NotFound)?;
194    let (source_len, target_start, target_cell, wake_sender) = {
195        let source = sched.tasks.get(&source_tid).ok_or(ViError::NotFound)?;
196        if !matches!(source.state, TaskState::Frozen { .. }) || source.hotswap_ingress_closed {
197            return Err(ViError::PermissionDenied);
198        }
199        let target = sched.tasks.get(&target_tid).ok_or(ViError::NotFound)?;
200        if matches!(
201            target.state,
202            TaskState::Frozen { .. } | TaskState::Terminated
203        ) || !target.hotswap_ready
204            || target.hotswap_source_tid != Some(source_tid)
205        {
206            return Err(ViError::PermissionDenied);
207        }
208        if target.pending_msgs.len() + source.pending_msgs.len() > HOTSWAP_MSG_QUEUE_DEPTH {
209            return Err(ViError::WouldBlock);
210        }
211        let wake_sender = match target.state {
212            TaskState::Recv { mask, .. } => source
213                .pending_msgs
214                .iter()
215                .find(|msg| mask == 0 || mask == msg.sender_tid)
216                .map(|msg| msg.sender_tid),
217            _ => None,
218        };
219        (
220            source.pending_msgs.len(),
221            target.pending_msgs.len(),
222            target.cell_id.0 as usize,
223            wake_sender,
224        )
225    };
226    if !crate::cell::service_registry::paused_matches(service_id, source_tid) {
227        return Err(ViError::WouldBlock);
228    }
229
230    for index in 0..source_len {
231        let copied = {
232            let source = sched.tasks.get(&source_tid).ok_or(ViError::NotFound)?;
233            let message = &source.pending_msgs.as_slice()[index];
234            PendingMsgData::try_copy(message.data.as_slice(), target_cell).map(|data| PendingMsg {
235                sender_tid: message.sender_tid,
236                data,
237                enqueued_tick: message.enqueued_tick,
238            })
239        };
240        let pushed = copied.and_then(|message| {
241            sched
242                .tasks
243                .get_mut(&target_tid)
244                .ok_or(())?
245                .pending_msgs
246                .try_push(message)
247        });
248        if pushed.is_err() {
249            rollback_mailbox_appends(sched, target_tid, target_start);
250            return Err(ViError::WouldBlock);
251        }
252    }
253
254    let mut source_mailbox = {
255        let source = sched.tasks.get_mut(&source_tid).ok_or(ViError::NotFound)?;
256        source.hotswap_ingress_closed = true;
257        core::mem::take(&mut source.pending_msgs)
258    };
259
260    if !crate::cell::service_registry::commit_paused(service_id, source_tid, target_tid) {
261        if let Some(source) = sched.tasks.get_mut(&source_tid) {
262            source.pending_msgs = core::mem::take(&mut source_mailbox);
263            source.hotswap_ingress_closed = false;
264        }
265        rollback_mailbox_appends(sched, target_tid, target_start);
266        return Err(ViError::WouldBlock);
267    }
268
269    if let Some(target) = sched.tasks.get_mut(&target_tid) {
270        target.hotswap_source_tid = None;
271    }
272    if let Some(sender_tid) = wake_sender {
273        if let Some(target) = sched.tasks.get_mut(&target_tid) {
274            target.state = TaskState::Ready;
275            target.current_caller = Some(sender_tid);
276        }
277        sched.push_ready(target_tid);
278    }
279    Ok(())
280}
281
282fn rollback_mailbox_appends(
283    sched: &mut crate::task::scheduler::Scheduler,
284    target_tid: usize,
285    target_start: usize,
286) {
287    if let Some(target) = sched.tasks.get_mut(&target_tid) {
288        while target.pending_msgs.len() > target_start {
289            let last = target.pending_msgs.len() - 1;
290            drop(target.pending_msgs.remove(last));
291        }
292    }
293}

295/// Terminate `tid` via the internal path, bypassing the Frozen kill-guard.
296///
297/// Used at the end of a successful swap to terminate the old cell.  The old cell
298/// is Frozen at this point; the regular `ForceExit` syscall would reject the
299/// request with `PermissionDenied`.
300///
301/// Mirrors the cleanup sequence from the `ForceExit` handler — must remain in sync.
302pub(crate) fn exit_task_internal(tid: usize, cell_id: CellId) {
303    // Resource cleanup (same order as ForceExit handler).
304    crate::cell::cap_registry::CAP_TABLE
305        .lock()
306        .revoke_all_for(cell_id);
307    crate::memory::cell_quota::deregister(cell_id);
308    crate::task::drivers::driver_cell::deregister_block_driver(tid);
309    crate::task::drivers::driver_cell::deregister_nic_driver(tid);
310    crate::resource_registry::release_for(cell_id);
311    crate::resource_registry::release_bdfs_for(tid);
312    // Keyed by task id, as at every other call site: this is also the point that
313    // releases quarantined frames, so a cell id here would look up nothing and
314    // leak them on every swap. The two agree for a loader-spawned cell, which is
315    // what let the mismatch go unnoticed.
316    crate::task::drivers::iommu::cleanup_cell(tid as u64);
317
318    if let Some(sched) = crate::task::SCHEDULER.lock().as_mut() {
319        // 0xAAAA_AAAA = hot-swap sentinel (distinguishes from clean exit 0 or watchdog MAX).
320        sched.exit_task(tid, 0xAAAA_AAAAusize);
321    }
322
323    // Grant pages are freed outside SCHEDULER lock (lock-order safety).
324    // SAFETY: reap_grants_for_task is pub(crate); hotswap.rs is in the same crate.
325    crate::task::syscall::reap_grants_for_task(tid);
326
327    crate::audit::log_event(
328        crate::audit::AuditEvent::CellExit,
329        &crate::audit::encode_u32x2(tid as u32, 0xAA00_0000u32), // hot-swap marker
330    );
