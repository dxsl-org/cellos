1//! Per-path capability ceiling for the cells named in the boot manifest.
2//!
3//! This is the ceiling applied to a **kernel/boot-initiated** spawn
4//! (`Spawner::Root`) and the source of the root authority's own capability
5//! write. It replaces a single "full authority" constant.
6//!
7//! **Invariant — the table is per-path, never a union.** A ceiling only ever
8//! constrains when it is *smaller* than the request; the union of every row is
9//! by definition a superset of every row, so a union-shaped ceiling would admit
10//! everything while looking like a restriction. Each row therefore states the
11//! authority that ONE path is expected to hold, and nothing else. `self_test`
12//! pins that property so a later edit cannot quietly collapse the table into a
13//! union.
14//!
15//! A path with no row is **unknown**, and an unknown path gets `CapSet::EMPTY`
16//! (fail-closed). `lookup` returns `None` for it so the diagnostics can say
17//! "missing entry" rather than "entry grants nothing" — the two look identical
18//! in the resulting CapSet and need very different fixes.
19//!
20//! Rows are derived from what each cell's `__ViCell_manifest` requests plus the
21//! path-triggered caps `CapSet::with_path_caps` layers on. A row is therefore
22//! the *expected* authority of that binary, not a guess.
23
24use crate::resource_registry::{DEV_GPIO, DEV_UART};
25use crate::task::cap::CapSet;
26
27mod selftest;

29/// Boot self-test of the table's load-bearing properties — that it is per-path
30/// rather than a union, and that no row is over-tightened. See `selftest`.
31pub use selftest::run as self_test;
32
33/// Block regions VFS must be able to hold: P1 | P4 | SRV | cell-store.
34///
35/// The 4th bit is wider than the manifest request (`0b111`) on purpose: the
36/// cell-store region is still granted after the policy intersect in the loader,
37/// and the ceiling intersect runs BEFORE policy — so a `0b111` ceiling here
38/// would zero the 4th bit before the policy blob ever gets a say.
39const VFS_REGIONS: u8 = 0b1111;
40
41/// The two console peripherals the interactive shell holds so it can delegate
42/// them to the peripheral demos it launches.
43const CONSOLE_MMIO: u8 = DEV_GPIO | DEV_UART;
44
45/// The expected ceiling for `path`, or `None` when `path` is not a boot cell.
46///
47/// Every row carries the reason that path holds those caps. Prefer a row that is
48/// marginally too generous over a missing one: one cap too many is a follow-up,
49/// a cell that cannot start is a dead system.
50pub fn lookup(path: &str) -> Option<CapSet> {
51    let caps = match path {
52        // Root authority. Not spawned from a path at all — the kernel writes
53        // these caps directly — but it is listed here so there is exactly one
54        // table describing boot authority. It is the widest row because a
55        // spawner's caps bound its children, and this cell brings up every
56        // service below. `platform` is deliberately absent: the Platform Cell is
57        // kernel-spawned, and `CapSet::apply_to` never writes `platform_cap`
58        // (the singleton latch owns it), so the root authority never actually
59        // held it and cannot delegate it.
60        "/bin/init" => CapSet {
61            block_io: true,             // delegated to /bin/vfs
62            network: true,              // delegated to /bin/net, /bin/net-broker
63            spawn: true,                // delegated to /bin/shell, /bin/supervisor
64            hypervisor: true,           // delegated to /bin/silo, /bin/hypervisor
65            mmio_devices: CONSOLE_MMIO, // delegated to /bin/shell
66            block_regions: VFS_REGIONS, // delegated to /bin/vfs
67            pcie_driver: true,          // delegated to the block/NIC/GPU/input drivers
68            platform: false,
69            supervisor: true, // delegated to /bin/supervisor
70        },
71        // PCIe ECAM enumeration + BAR registration. Kernel-spawned before init.
72        // Its manifest declares nothing; `with_path_caps` is the request signal,
73        // and `try_grant_platform` still enforces one holder ever.
74        "/bin/platform" => CapSet {
75            platform: true,
76            ..CapSet::EMPTY
77        },
78        // Filesystem service: raw block syscalls plus its partitions.
79        "/bin/vfs" => CapSet {
80            block_io: true,
81            block_regions: VFS_REGIONS,
82            ..CapSet::EMPTY
83        },
84        // Network stack and the cluster broker; both declare `network` only.
85        "/bin/net" | "/bin/net-broker" => CapSet {
86            network: true,
87            ..CapSet::EMPTY
88        },
89        // Interactive shell: spawns cells, and is the ceiling for the GPIO/UART
90        // peripheral demos it launches.
91        "/bin/shell" => CapSet {
92            spawn: true,
93            mmio_devices: CONSOLE_MMIO,
94            ..CapSet::EMPTY
95        },
96        // Hotswap orchestration: re-spawns cells (spawn) and freezes/resumes
97        // them (supervisor, from `with_path_caps`).
98        "/bin/supervisor" => CapSet {
99            spawn: true,
100            supervisor: true,
101            ..CapSet::EMPTY
102        },
103        // Both declare `hypervisor`. `CapSet::from_manifest` additionally gates
104        // that bit on the CPU actually reporting H-ext/EL2/VMX, so this row
105        // cannot conjure it on hardware that lacks virtualisation.
106        "/bin/silo" | "/bin/hypervisor" => CapSet {
107            hypervisor: true,
108            ..CapSet::EMPTY
109        },
110        // Driver cells: each claims a BAR/MMIO range and authorises DMA, which
111        // needs `pcie_driver`. Their manifests declare nothing — the install
112        // path is the request signal (`with_path_caps`).
113        "/bin/block" | "/bin/nvme" | "/bin/e1000" | "/bin/virtio-net" | "/bin/virtio-gpu"
114        | "/bin/input" => CapSet {
115            pcie_driver: true,
116            ..CapSet::EMPTY
117        },
118        // Known boot cells that need no authority at all: pure IPC clients. They
119        // are listed rather than left to the unknown-path fallback so a denial
120        // report can distinguish "needs nothing" from "row missing".
121        "/bin/config" | "/bin/compositor" | "/bin/fb-console" | "/bin/silo-test"
122        | "/bin/vfs-test" | "/bin/srv-test" => CapSet::EMPTY,
123        _ => return None,
124    };
125    Some(caps)
126}
127
128/// The ceiling for `path`; an unknown path yields `CapSet::EMPTY` (fail-closed).
129pub fn boot_ceiling(path: &str) -> CapSet {
130    lookup(path).unwrap_or(CapSet::EMPTY)
131}

133/// Report, at `error` level, every cap the boot ceiling refused for `path`.
134///
135/// Written for a reader who has only a boot log: it prints whether the path is
136/// missing from the table at all, the three sets involved, and one line per
137/// refused cap named exactly as the table field to add. Nothing here needs to be
138/// re-derived from the source to act on it.
139pub fn log_refusal(path: &str, requested: CapSet, ceiling: CapSet, granted: CapSet) {
140    if lookup(path).is_none() {
141        log::error!(
142            "[loader] boot-ceiling: {:?} has NO ROW in boot_ceiling::lookup — ceiling is EMPTY. Add a row.",
143            path
144        );
145    } else {
146        log::error!(
147            "[loader] boot-ceiling: {:?} narrowed by its row — widen the row if the cap is legitimate.",
148            path
149        );
150    }
151    log::error!("[loader]   requested = {:?}", requested);
152    log::error!("[loader]   ceiling   = {:?}", ceiling);
153    log::error!("[loader]   granted   = {:?}", granted);
154    if requested.block_io && !granted.block_io {
155        log::error!("[loader]   refused: block_io");
156    }
157    if requested.network && !granted.network {
158        log::error!("[loader]   refused: network");
159    }
160    if requested.spawn && !granted.spawn {
161        log::error!("[loader]   refused: spawn");
162    }
163    if requested.hypervisor && !granted.hypervisor {
164        log::error!("[loader]   refused: hypervisor");
165    }
166    if requested.pcie_driver && !granted.pcie_driver {
167        log::error!("[loader]   refused: pcie_driver");
168    }
169    if requested.platform && !granted.platform {
170        log::error!("[loader]   refused: platform");
171    }
172    if requested.supervisor && !granted.supervisor {
173        log::error!("[loader]   refused: supervisor");
174    }
175    if requested.mmio_devices != granted.mmio_devices {
176        log::error!(
177            "[loader]   refused: mmio_devices {:#07b}",
178            requested.mmio_devices & !granted.mmio_devices
179        );
180    }
181    if requested.block_regions != granted.block_regions {
182        log::error!(
183            "[loader]   refused: block_regions {:#06b}",
184            requested.block_regions & !granted.block_regions
185        );
186    }
187}
