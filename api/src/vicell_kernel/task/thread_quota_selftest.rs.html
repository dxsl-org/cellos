1//! Boot self-test for thread-stack quota accounting.
2//!
3//! A thread's kernel stack is real memory held for the thread's whole life. If it
4//! is not charged to the cell that asked for the thread, a cell can grow its true
5//! footprint by a quarter of a megabyte at a time while every number the kernel
6//! reports about it stays still — and the only remaining bound is the thread count,
7//! which says nothing about memory.
8//!
9//! Two directions are proven here, because getting one right and the other wrong is
10//! worse than neither: a charge that is never released turns the quota into a slow
11//! leak that eventually refuses legitimate work, and a release without a charge
12//! makes the whole ledger decorative.
13//!
14//! Runs in the same single-hart window as the other task self-tests — after
15//! `task::init()` and before `smp::start_secondaries()` — so the real thread it
16//! spawns cannot be picked up by another hart between creation and teardown. The
17//! scheduler and the quota tables are left exactly as they were found.
18
19use super::syscall::{handle_syscall, Syscall, SyscallError};
20use super::tcb::Task;
21use crate::memory::cell_quota;
22use types::CellId;
23
24/// Synthetic tid outside any range the boot sequence has assigned yet.
25const PARENT_TID: usize = 9201;
26
27/// Quota slot to exercise. Must be `< MAX_CELLS` for the charge to be tracked at
28/// all (higher ids are uncapped by design), and is the last slot precisely because
29/// cell ids are handed out from `next_task_id` upward — nothing has reached it at
30/// this point in boot, and the test deregisters it before anything can.
31const QUOTA_CELL: u64 = (cell_quota::MAX_CELLS - 1) as u64;
32
33/// Bytes one thread stack occupies: the usable pages plus the guard page.
34fn stack_charge_bytes() -> usize {
35    (crate::task::STACK_PAGES + 1) * crate::memory::paging::PAGE_SIZE
36}
37
38fn insert_parent() {
39    let parent = alloc::boxed::Box::new(Task::new(
40        PARENT_TID,
41        CellId(QUOTA_CELL),
42        "selftest",
43        alloc::vec::Vec::new(),
44    ));
45    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
46        sched.tasks.insert(PARENT_TID, parent);
47    }
48}
49
50fn remove(tid: usize) {
51    if let Some(sched) = super::SCHEDULER.lock().as_mut() {
52        sched.tasks.remove(&tid);
53    }
54    super::hart_local::ready::remove_from_all(tid);
55}
56
57/// Drop every reapable zombie OUTSIDE the scheduler lock, which is what actually
58/// returns a dead thread's stack frames to the allocator.
59fn reap() {
60    let dead = super::SCHEDULER
61        .lock()
62        .as_mut()
63        .map(|s| s.take_reapable_zombies())
64        .unwrap_or_default();
65    drop(dead);
66}
67
68/// Charge appears at spawn and is gone once the thread dies.
69fn charge_then_release(expected: usize) -> bool {
70    cell_quota::register(CellId(QUOTA_CELL), cell_quota::DEFAULT_QUOTA_BYTES);
71    insert_parent();
72
73    let mut ok = true;
74    let spawned = handle_syscall(
75        PARENT_TID,
76        Syscall::Spawn {
77            entry: 0x1000,
78            arg: 0,
79        },
80    );
81
82    match spawned {
83        Ok(tid) if tid != 0 => {
84            let charged = cell_quota::in_use(CellId(QUOTA_CELL));
85            if charged != expected {
86                ok = false;
87                log::error!(
88                    "[selftest] THREAD-QUOTA: FAIL — charged {} bytes, expected {}",
89                    charged,
90                    expected
91                );
92            }
93            // Death by the funnel every real death path uses, not by lifting the
94            // task out of the table: the refund is only correct if it happens there.
95            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
96                sched.exit_task(tid, 0);
97            }
98            let after = cell_quota::in_use(CellId(QUOTA_CELL));
99            if after != 0 {
100                ok = false;
101                log::error!(
102                    "[selftest] THREAD-QUOTA: FAIL — {} bytes still charged after exit",
103                    after
104                );
105            }
106            reap();
107        }
108        other => {
109            ok = false;
110            log::error!("[selftest] THREAD-QUOTA: FAIL — spawn returned {:?}", other);
111        }
112    }
113
114    remove(PARENT_TID);
115    cell_quota::deregister(CellId(QUOTA_CELL));
116    ok
117}

119/// A cell whose quota cannot absorb another stack is refused, and the refusal
120/// leaves no charge behind.
121fn refused_when_unaffordable() -> bool {
122    // One page of quota — below a stack by three orders of magnitude, so the
123    // refusal cannot be an artefact of a borderline comparison.
124    cell_quota::register(CellId(QUOTA_CELL), crate::memory::paging::PAGE_SIZE);
125    insert_parent();
126
127    let mut ok = true;
128    let refused = handle_syscall(
129        PARENT_TID,
130        Syscall::Spawn {
131            entry: 0x1000,
132            arg: 0,
133        },
134    );
135    if !matches!(refused, Err(SyscallError::TryAgain)) {
136        ok = false;
137        log::error!(
138            "[selftest] THREAD-QUOTA: FAIL — unaffordable spawn returned {:?}, expected TryAgain",
139            refused
140        );
141        if let Ok(stray) = refused {
142            remove(stray);
143        }
144    }
145    let leaked = cell_quota::in_use(CellId(QUOTA_CELL));
146    if leaked != 0 {
147        ok = false;
148        log::error!(
149            "[selftest] THREAD-QUOTA: FAIL — refused spawn left {} bytes charged",
150            leaked
151        );
152    }
153
154    remove(PARENT_TID);
155    cell_quota::deregister(CellId(QUOTA_CELL));
156    ok
157}

159/// Returns true iff a thread stack is charged, released, and enforced as specified.
160/// Logs a decisive serial line.
161///
162/// Transparent to the boot sequence: the successful spawn advances `next_task_id`,
163/// so the counter is snapshotted on entry and restored on exit — the first real cell
164/// still gets the tid it would have without this test, keeping cell-id assignment
165/// stable whether or not the test is compiled in.
166pub fn self_test() -> bool {
167    let saved_next_tid = super::SCHEDULER.lock().as_ref().map(|s| s.next_task_id);
168
169    let expected = stack_charge_bytes();
170    let ok = charge_then_release(expected) & refused_when_unaffordable();
171
172    if let (Some(sched), Some(n)) = (super::SCHEDULER.lock().as_mut(), saved_next_tid) {
173        sched.next_task_id = n;
174    }
175
176    if ok {
177        log::info!("[selftest] THREAD-QUOTA: PASS (charged, released, enforced)");
178    } else {
179        log::error!("[selftest] THREAD-QUOTA: FAIL");
180    }
181    ok
182}
