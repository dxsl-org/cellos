1//! The wait side of the completion queue: `WaitCompletion`.
2//!
3//! Calling this **is** the submission. A hardware condition is not a discrete
4//! operation a cell submits, so there is nothing else that could reserve a slot
5//! from the waiting cell's own context — and reserving anywhere else would put
6//! the refusal in an interrupt handler, which has nothing to do with it. The
7//! wait therefore reserves, records the reservation against the source, and only
8//! then parks.
9//!
10//! The order of those steps is the whole correctness argument. Arming before
11//! checking whether a frame is already pending means the two possible arrivals
12//! are both covered: one that lands after the arm completes the reservation, and
13//! one that landed before it left the level flag set, which the check consumes.
14//! There is no third case, and no window in which a frame is visible to neither.
15//!
16//! A wait that ends without a result releases its slot before returning. A
17//! reservation that outlived its waiter would consume queue capacity that never
18//! comes back, and the queue is the cell's bound on outstanding work.
19
20use super::completion::{self, Completion, CompletionQueue};
21use super::syscall::{validate_user_buf, SyscallError};
22use super::tcb::TaskState;
23use super::waker;
24use api::completion::{ViCompletion, COMPLETION_LEN};

26/// Removes this waiter's registration on every return path without erasing a
27/// newer waiter that may have replaced it on the shared per-cell queue.
28struct WaiterRegistration<'a> {
29    queue: &'a CompletionQueue,
30    tid: usize,
31}
32
33impl<'a> WaiterRegistration<'a> {
34    fn new(queue: &'a CompletionQueue, tid: usize) -> Self {
35        queue.register_waiter(tid);
36        Self { queue, tid }
37    }
38}
39
40impl Drop for WaiterRegistration<'_> {
41    fn drop(&mut self) {
42        let _ = self.queue.clear_waiter(self.tid);
43    }
44}

46/// Reserve a slot on the caller's completion queue, wait for it to be filled,
47/// and write the result to `out_ptr`.
48///
49/// `mask` names exactly one source; `deadline` is an absolute tick count, or
50/// `None` to wait indefinitely.
51///
52/// # Returns
53/// `1` when a completion was written to `out_ptr`, `0` when the wait ended with
54/// nothing to report.
55///
56/// # Errors
57/// - `InvalidInput` — `mask` does not name exactly one source this kernel
58///   serves, or `out_ptr` is null.
59/// - `BufferTooSmall` — `out_ptr` cannot hold a whole record.
60/// - `TryAgain` — the caller's queue has no free slot. This is backpressure, and
61///   it is reported here, to the cell that caused it, rather than to the
62///   interrupt handler that would otherwise have had nowhere to put a result.
63/// - `Unknown` — the caller is not a live task.
64///
65/// # Panics
66/// Never panics.
67pub fn wait_completion(
68    caller_id: usize,
69    mask: u32,
70    deadline: Option<u64>,
71    out_ptr: usize,
72) -> Result<usize, SyscallError> {
73    // One bit only: a submission is made against one source, and a mask naming
74    // two would leave the second one with no reservation of its own.
75    if mask != api::syscall::events::NET_RX {
76        return Err(SyscallError::InvalidInput);
77    }
78    validate_user_buf(out_ptr, COMPLETION_LEN, COMPLETION_LEN)?;
79
80    let queue = {
81        let mut guard = super::SCHEDULER.lock();
82        let sched = guard.as_mut().ok_or(SyscallError::Unknown)?;
83        completion::queue_for(sched, caller_id).ok_or(SyscallError::Unknown)?
84    };
85    let slot = queue.reserve().ok_or(SyscallError::TryAgain)?;
86    let _waiter = WaiterRegistration::new(&queue, caller_id);
87    waker::arm_net_rx(queue.clone(), slot);
88
89    // Lost-wakeup guard, first half: a frame that arrived while nobody held a
90    // reservation is remembered in the level flag. Consuming it only *after* the
91    // arm is what makes the check total — a frame arriving during the check has
92    // a reservation to complete instead.
93    if waker::consume_pending(mask) != 0
94        && matches!(waker::disarm_net_rx(&queue), waker::DisarmResult::Owned(_))
95    {
96        // The result is reported straight out of the reservation rather than
97        // pushed through the queue and pulled back: appending would raise a
98        // wake request for a task that is not parked, and that request would
99        // still be outstanding to cancel the *next* park.
100        if queue.release(slot) {
101            return Ok(write_completion(
102                out_ptr,
103                Completion {
104                    slot,
105                    result: mask as isize,
106                },
107            ));
108        }
109    }
110
111    // Second half: anything the source landed while the steps above ran is
112    // delivered without parking at all.
113    if let Some(done) = queue.drain() {
114        return Ok(write_completion(out_ptr, done));
115    }
116
117    // `WaitEvent` with no event bits: this waiter's result comes from the queue,
118    // so the sweep must not consume a fired bit on its behalf and report an
119    // empty wake. The deadline is the one thing the sweep is still asked for.
120    {
121        let mut guard = super::SCHEDULER.lock();
122        if let Some(task) = guard.as_mut().and_then(|s| s.tasks.get_mut(&caller_id)) {
123            task.state = TaskState::WaitEvent { mask: 0, deadline };
124        }
125    }
126    super::yield_cpu();
127
128    if let Some(done) = queue.drain() {
129        return Ok(write_completion(out_ptr, done));
130    }
131
132    // Nothing landed: the deadline passed, or the wake was spurious. Either way
133    // the reservation goes back, and the race with a frame arriving at this
134    // exact moment is settled by whoever takes the slot.
135    loop {
136        match waker::disarm_net_rx(&queue) {
137            waker::DisarmResult::Owned(own) if queue.release(own) => return Ok(0),
138            // The source owns the slot but has not published it yet. This can
139            // only run concurrently on another hart, so wait until the queue
140            // record becomes visible before clearing the waiter registration.
141            waker::DisarmResult::Completing => core::hint::spin_loop(),
142            // Either the source completed the slot as it was being withdrawn,
143            // or another waiter displaced it. Both publish a result before
144            // leaving the Completing state.
145            _ => {
146                return match queue.drain() {
147                    Some(done) => Ok(write_completion(out_ptr, done)),
148                    None => Ok(0),
149                };
150            }
151        }
152        if let Some(done) = queue.drain() {
153            return Ok(write_completion(out_ptr, done));
154        }
155    }
156}

158/// Write `done` into the caller's buffer.
159///
160/// Precondition: `out_ptr` has been through `validate_user_buf` for
161/// [`COMPLETION_LEN`] bytes in this call.
162fn write_completion(out_ptr: usize, done: Completion) -> usize {
163    let record = ViCompletion {
164        slot: done.slot.index() as u32,
165        result: done.result as i64,
166    }
167    .to_bytes();
168    // SAFETY: `out_ptr` is a non-null caller buffer validated for exactly
169    // COMPLETION_LEN bytes at entry to `wait_completion`, and `record` is a
170    // distinct kernel stack array of that same length, so the two cannot
171    // overlap. A byte copy has no alignment requirement.
172    unsafe { core::ptr::copy_nonoverlapping(record.as_ptr(), out_ptr as *mut u8, COMPLETION_LEN) };
173    1
174}
