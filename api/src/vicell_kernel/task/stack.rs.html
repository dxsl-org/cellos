1//! Stack Management for Tasks.
2//!
3//! Handles allocation, deallocation, and guard pages for Kernel and User stacks.
4//! Complies with Rule 2 (Owned Buffers / Memory Safety) and Rule 8 (Resource Management).
5
6use crate::memory::frame::{FrameAllocator, FRAME_ALLOCATOR};
7use crate::memory::paging::{self, Flags, PAGE_SIZE};
8use log::{error, trace};
9use types::{VAddr, ViError};

11/// Hand `total_pages` frames starting at `base` back to `allocator`, restoring each
12/// to the boot identity mapping (kernel RWX) first.
13///
14/// Invariant this upholds: every frame on the allocator's free list is
15/// identity-mapped kernel-RWX. The cell loader zeroes a freshly allocated frame
16/// through its identity address, so a frame released while unmapped (as the guard
17/// frame is) store-faults for its next owner, and one released carrying USER flags
18/// hands that owner the wrong permissions. Unmap-then-map normalises the PTE
19/// whatever state the frame was left in.
20///
21/// The caller must already hold `FRAME_ALLOCATOR`; only `KERNEL_ROOT` (a leaf) is
22/// taken here, which is the documented order.
23fn release_frames(allocator: &mut FrameAllocator, base: VAddr, total_pages: usize) {
24    let kernel_rwx = Flags::from_bits(
25        Flags::VALID | Flags::READ | Flags::WRITE | Flags::EXECUTE | Flags::ACCESSED | Flags::DIRTY,
26    );
27    for i in 0..total_pages {
28        let frame = base + (i * PAGE_SIZE);
29        let _ = paging::unmap_page(frame);
30        let _ = paging::map_page(allocator, frame, frame, kernel_rwx);
31        allocator.deallocate_frame(frame);
32    }
33    paging::tlb_flush_all();
34}

36/// Represents an allocated Stack.
37/// Implements Drop to automatically free pages.
38#[derive(Debug)]
39pub struct Stack {
40    /// Base address (lowest address) of the allocated range.
41    /// This includes the guard page at the bottom if present.
42    pub base: VAddr,
43    /// Number of usable pages (excluding guard page).
44    pub pages: usize,
45    /// Whether this stack has a guard page.
46    pub has_guard: bool,
47    /// Top of the stack (initial SP).
48    pub top: VAddr,
49}
50
51impl Stack {
52    /// Allocate a new Kernel Stack of `pages` usable pages, plus a guard page
53    /// below it.
54    ///
55    /// # Errors
56    /// - `OutOfMemory` — no contiguous run of `pages + 1` frames exists, or a
57    ///   page-table mapping could not be installed.
58    /// - `NotSupported` — the guard page could not be established. No stack is
59    ///   returned in that case; see [`Self::allocate`].
60    pub fn new_kernel(pages: usize) -> Result<Self, ViError> {
61        Self::allocate(pages, true, false)
62    }
63
64    /// Allocate a new User Stack of `pages` usable pages, plus a guard page below
65    /// it. Usable pages are mapped USER RW.
66    ///
67    /// # Errors
68    /// Same as [`Self::new_kernel`].
69    pub fn new_user(pages: usize) -> Result<Self, ViError> {
70        Self::allocate(pages, true, true)
71    }
72
73    /// Internal allocation logic.
74    ///
75    /// Contract: this either returns a stack whose guard page is *verified* absent
76    /// from the page tables, or it returns an error having released every frame it
77    /// took. There is deliberately no third outcome. An unguarded stack is not a
78    /// degraded stack — in a single address space the frame below it belongs to
79    /// another cell, so an overflow that should have trapped instead corrupts a
80    /// neighbour with no fault and no log, and the victim dies later somewhere
81    /// unrelated. A log line at allocation time is not a mitigation: nothing reads
82    /// it in the microseconds before the overflow.
83    fn allocate(pages: usize, guard: bool, user_mode: bool) -> Result<Self, ViError> {
84        let total_pages = if guard { pages + 1 } else { pages };
85
86        let mut frame_guard = FRAME_ALLOCATOR.lock();
87        let allocator = frame_guard.as_mut().ok_or_else(|| {
88            error!(
89                "Stack alloc failed: frame allocator unavailable (user={}, pages={})",
90                user_mode, total_pages
91            );
92            ViError::OutOfMemory
93        })?;
94
95        // A stack must be ONE contiguous run of frames, and that is a consequence of
96        // SAS rather than a shortcut: everything is identity-mapped (VA == PA) and
97        // there is no virtual-address allocator, so virtual contiguity can only come
98        // from physical contiguity.
99        //
100        // The cost: this can fail on a fragmented allocator while plenty of memory is
101        // free, and no amount of lazy commit helps — a guard page cannot make VA != PA.
102        // So callers must read `OutOfMemory` here as "no run this long exists right
103        // now", which is recoverable, and never as grounds to panic. Lifting the
104        // constraint means adding a VA allocator so stack pages can be scattered.
105        let base_frame = allocator.allocate_contiguous(total_pages).ok_or_else(|| {
106            error!(
107                "Stack alloc failed: no contiguous run (user={}, pages={}, bytes={})",
108                user_mode,
109                total_pages,
110                total_pages * PAGE_SIZE
111            );
112            ViError::OutOfMemory
113        })?;
114
115        let base_addr = base_frame; // Identity Map
116
117        // 2. Map Pages
118        // If Guard Page is requested, the bottom page (base_addr) is NOT mapped (or mapped as invalid).
119        // Ideally unmapped.
120
121        let usable_start_idx = if guard { 1 } else { 0 };
122
123        // SAS identity map: all RAM is already identity-mapped RWX by
124        // init_kernel_paging. The usable pages are re-mapped below, then the guard
125        // frame (base_addr) is unmapped after the loop so an overflow traps.
126
127        // Usable Pages
128        let flags = if user_mode {
129            // User Read/Write (Exec?)
130            Flags::from_bits(
131                Flags::VALID
132                    | Flags::READ
133                    | Flags::WRITE
134                    | Flags::USER
135                    | Flags::ACCESSED
136                    | Flags::DIRTY,
137            )
138        } else {
139            // Kernel Read/Write
140            Flags::from_bits(
141                Flags::VALID | Flags::READ | Flags::WRITE | Flags::ACCESSED | Flags::DIRTY,
142            )
143        };
144
145        for i in usable_start_idx..total_pages {
146            let addr = base_addr + (i * PAGE_SIZE);
147            if paging::map_page(allocator, addr, addr, flags).is_err() {
148                // The run is not yet owned by a Stack, so nothing will drop it —
149                // release it here or the frames are lost until reboot.
150                error!("Stack alloc failed: cannot map page 0x{:X}", addr);
151                release_frames(allocator, base_addr, total_pages);
152                return Err(ViError::OutOfMemory);
153            }
154        }
155
156        // Guard page: drop the bottom frame's pre-existing identity mapping so a
157        // stack overflow (a write below base_addr+PAGE_SIZE) faults instead of
158        // silently corrupting the neighbouring frame. The spawn paths zero only
159        // the usable pages (skipping base_addr), so nothing legitimately writes to
160        // the guard frame. The frame stays owned by this Stack (freed in Drop);
161        // only its PTE is cleared. unmap_page locks KERNEL_ROOT (not FRAME_ALLOCATOR,
162        // which we still hold) — no deadlock.
163        //
164        // The guard is verified by translation, not by the unmap's return code: on
165        // some arches `unmap_page` reports success for a page it never touched
166        // (paging root absent, or the PTE was not a 4 KiB leaf). Asking the page
167        // tables whether the frame still resolves is the only answer that matches
168        // what the hardware will do on overflow.
169        if guard {
170            let unmap_ok = paging::unmap_page(base_addr).is_ok();
171            paging::tlb_flush_all();
172            if !unmap_ok || paging::virt_to_phys(base_addr).is_some() {
173                error!(
174                    "Stack alloc refused: guard frame 0x{:X} still mapped (unmap_ok={})",
175                    base_addr, unmap_ok
176                );
177                release_frames(allocator, base_addr, total_pages);
178                return Err(ViError::NotSupported);
179            }
180        }
181
182        // Calculate Top (Stack grows down)
183        // Top is at the END of the allocated range.
184        let top = base_addr + (total_pages * PAGE_SIZE);
185
186        trace!(
187            "Allocated Stack: Base=0x{:X}, Top=0x{:X}, Pages={}, User={}",
188            base_addr,
189            top,
190            pages,
191            user_mode
192        );
193
194        Ok(Stack {
195            base: base_addr,
196            pages,
197            has_guard: guard,
198            top,
199        })
200    }
201
202    /// Total physical bytes reserved for this stack, including the guard page.
203    pub fn allocated_bytes(&self) -> usize {
204        let total_pages = if self.has_guard {
205            self.pages + 1
206        } else {
207            self.pages
208        };
209        total_pages * PAGE_SIZE
210    }
211
212    /// Task-owned usable stack bytes, excluding any kernel-only guard reservation.
213    pub fn usable_bytes(&self) -> usize {
214        self.pages * PAGE_SIZE
215    }
216}
217
218impl Drop for Stack {
219    fn drop(&mut self) {
220        trace!("Dropping Stack at 0x{:X}", self.base);
221
222        let total_pages = if self.has_guard {
223            self.pages + 1
224        } else {
225            self.pages
226        };
227
228        let mut frame_guard = FRAME_ALLOCATOR.lock();
229        if let Some(allocator) = frame_guard.as_mut() {
230            release_frames(allocator, self.base, total_pages);
231        }
232    }
233}

235/// Physical frames mapped for a cell's ELF segments, recorded as `(vaddr, frame)`
236/// at load time so they can be reclaimed when the cell dies.
237///
238/// `Stack::drop` only frees stacks; without this a cell's code/data frames leak
239/// on every death (a supervised service restarted repeatedly would grow to OOM).
240/// Segment frames are allocated exclusively for the cell by `load_segments`
241/// (IPC/shared buffers use separate frames), so freeing them on death is safe.
242#[derive(Debug)]
243pub struct CellSegments {
244    pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
245    /// VA base allocated by `va_alloc::alloc_cell_va` for PIE cells; `0` for
246    /// fixed-VA cells.  Returned to the allocator's free list on drop.
247    pie_va_base: usize,
248}
249
250impl CellSegments {
251    pub fn new(
252        pages: alloc::vec::Vec<(types::VAddr, types::PhysAddr)>,
253        pie_va_base: usize,
254    ) -> Self {
255        Self { pages, pie_va_base }
256    }
257
258    /// Total physical bytes reserved for this cell's ELF segment frames.
259    pub fn allocated_bytes(&self) -> usize {
260        self.pages.len() * PAGE_SIZE
261    }
262
263    /// Unmap this cell's segment VAs immediately at death — WITHOUT freeing the
264    /// frames (those are freed lazily when the zombie is reaped, in `drop`).
265    ///
266    /// Frees the address space right away so (a) a respawn can reuse the fixed VA
267    /// and (b) the load-time overwrite guard (`load_segments`) only ever observes
268    /// LIVE cells' (and kernel MMIO) mappings, never a dead-but-unreaped cell's.
269    /// Locks only `KERNEL_ROOT` (a leaf), so it is safe under the SCHEDULER lock.
270    pub fn eager_unmap(&self) {
271        let mut unmapped_any = false;
272        for &(vaddr, frame) in &self.pages {
273            // Only unmap a VA that still resolves to OUR frame (it won't if a
274            // respawn already re-pointed it — leave the new mapping intact).
275            if paging::virt_to_phys(vaddr) == Some(frame) {
276                let _ = paging::unmap_page(vaddr);
277                unmapped_any = true;
278            }
279        }
280        if unmapped_any {
281            paging::tlb_flush_all();
282        }
283    }
284}
285
286impl Drop for CellSegments {
287    fn drop(&mut self) {
288        let mut frame_guard = FRAME_ALLOCATOR.lock();
289        if let Some(allocator) = frame_guard.as_mut() {
290            for &(vaddr, frame) in &self.pages {
291                // Only unmap if this VA still resolves to OUR frame. Cells load at
292                // fixed VAs, so a supervised cell respawned at the same VA before we
293                // are reaped will have re-pointed this VA at the NEW instance's frame
294                // — unmapping it then would crash the new cell. Skip the unmap in
295                // that case; the old frame is still ours to free either way.
296                if paging::virt_to_phys(vaddr) == Some(frame) {
297                    let _ = paging::unmap_page(vaddr);
298                }
299                allocator.deallocate_frame(frame);
300            }
301        }
302        // Return the PIE VA slot to the allocator so it can be reused.
303        if self.pie_va_base != 0 {
304            crate::loader::va_alloc::free_cell_va(self.pie_va_base);
305        }
306    }
