1//! Shared VirtIO MMIO slot enumeration.
2//!
3//! `virtio_blk`, `input_irq_ack`, and Driver Cells use `virtio_slots()` to
4//! iterate all VirtIO MMIO slots for the current platform.
5//!
6//! AArch64: scans all 32 slots at 0x0a000000, stride 0x200 (QEMU virt layout).
7//! QEMU assigns devices to slots in an implementation-defined order so we must
8//! probe all 32.  The identity map in paging.rs covers the full 0x0a004000 range.
9//!
10//! RISC-V: reads DTB-confirmed slots from `platform::PLATFORM`.
11//! PCIe-only arches expose no VirtIO-MMIO slots.
12
13extern crate alloc;
14use alloc::vec::Vec;
15
16/// A VirtIO MMIO slot with base address and IRQ.
17pub struct VirtioSlot {
18    pub base: usize,
19    pub irq: u32,
20}
21
22/// Iterator over all VirtIO MMIO slots for the current platform.
23pub fn virtio_slots() -> impl Iterator<Item = VirtioSlot> {
24    #[cfg(target_arch = "aarch64")]
25    {
26        // QEMU ARM virt: 32 VirtIO MMIO slots at 0x0a000000, 512 bytes each, SPI 16+i.
27        // All 32 slots are identity-mapped by init_kernel_paging (0x0a000000..0x0a004000).
28        const BASE: usize = 0x0a00_0000;
29        const STRIDE: usize = 0x200;
30        let slots: Vec<VirtioSlot> = (0..32_usize)
31            .map(|i| VirtioSlot {
32                base: BASE + i * STRIDE,
33                irq: 16 + i as u32,
34            })
35            .collect();
36        slots.into_iter()
37    }
38    #[cfg(target_arch = "riscv64")]
39    {
40        let slots: Vec<VirtioSlot> = crate::platform::with(|p| {
41            p.virtio_mmio
42                .iter()
43                .filter_map(|e| {
44                    e.as_ref().map(|e| VirtioSlot {
45                        base: e.base,
46                        irq: e.irq,
47                    })
48                })
49                .collect()
50        });
51        slots.into_iter()
52    }
53    #[cfg(not(any(target_arch = "aarch64", target_arch = "riscv64")))]
54    {
55        Vec::new().into_iter()
56    }
57}

59/// Execute one VirtIO MMIO ACK sequence with RV64 SUM enabled only for that scope.
60///
61/// On RV64, supervisor-mode MMIO ACK helpers may touch user-accessible identity-mapped
62/// device pages while trap handling runs with SUM clear. This helper snapshots whether
63/// SUM was already set, raises it only for `ack`, and restores the prior state
64/// immediately afterwards. Other architectures execute `ack` unchanged.
65pub(crate) fn with_riscv_sum_for_virtio_ack<T>(ack: impl FnOnce() -> T) -> T {
66    #[cfg(target_arch = "riscv64")]
67    {
68        const SUM_MASK: usize = 1usize << 18;
69        let sstatus: usize;
70        // SAFETY: this helper is used only for short-lived VirtIO MMIO ACK sequences.
71        // `csrr/csrs/csrc sstatus` are privileged RV64 supervisor CSR operations; the
72        // code records the prior SUM bit, sets only SUM for the duration of `ack`, and
73        // restores the exact prior SUM state before returning. It does not widen trap
74        // entry globally or alter SSIE/SEIE/STIE.
75        unsafe {
76            core::arch::asm!("csrr {value}, sstatus", value = out(reg) sstatus, options(nostack));
77            if sstatus & SUM_MASK == 0 {
78                core::arch::asm!("csrs sstatus, {mask}", mask = in(reg) SUM_MASK, options(nostack));
79            }
80        }
81        let result = ack();
82        unsafe {
83            if sstatus & SUM_MASK == 0 {
84                core::arch::asm!("csrc sstatus, {mask}", mask = in(reg) SUM_MASK, options(nostack));
85            }
86        }
87        result
88    }
89    #[cfg(not(target_arch = "riscv64"))]
90    {
91        ack()
92    }
93}

95/// Read VirtIO `InterruptStatus` and ACK the exact bits it reports.
96///
97/// Returns the observed status bits; `0` means there was nothing pending to ACK.
98pub(crate) fn ack_virtio_interrupt_status(mmio_base: usize) -> u32 {
99    let status = with_riscv_sum_for_virtio_ack(|| {
100        // SAFETY: caller provides a live, identity-mapped VirtIO MMIO base. The
101        // status register is read-only, and the scoped SUM helper contains the
102        // RV64 privilege widening to this one MMIO access sequence only.
103        unsafe { core::ptr::read_volatile((mmio_base + 0x60) as *const u32) }
104    });
105    if status != 0 {
106        with_riscv_sum_for_virtio_ack(|| {
107            // SAFETY: same MMIO contract as the status read above; the ACK register
108            // consumes exactly the pending status bits returned by the device.
109            unsafe { core::ptr::write_volatile((mmio_base + 0x64) as *mut u32, status) };
110        });
111    }
112    status
113}

115/// Acknowledge an interrupt from an as-yet unclaimed VirtIO MMIO slot.
116///
117/// Driver Cells may need several synchronous queue transactions to initialise
118/// before they can enter `sys_wait_irq`. Leaving those level-triggered
119/// interrupts asserted starves the Cell before registration completes.
120fn ack_unclaimed(irq: u32) -> bool {
121    #[cfg(target_arch = "aarch64")]
122    let base = (16..48)
123        .contains(&irq)
124        .then_some(0x0a00_0000usize + (irq as usize - 16) * 0x200);
125    #[cfg(target_arch = "riscv64")]
126    let base = (1..9)
127        .contains(&irq)
128        .then_some(0x1000_1000usize + (irq as usize - 1) * 0x1000);
129    // x86_64 routes VirtIO through PCI MSI-X, not a fixed MMIO IRQ window, so
130    // there is no slot base to derive and nothing to acknowledge here.
131    #[cfg(not(any(target_arch = "aarch64", target_arch = "riscv64")))]
132    let base: Option<usize> = {
133        let _ = irq;
134        None
135    };
136
137    let Some(base) = base else {
138        return false;
139    };
140    ack_virtio_interrupt_status(base) != 0
141}

143/// VirtIO MMIO IRQ dispatcher — called from the arch trap handlers (riscv64/aarch64)
144/// when any VirtIO MMIO IRQ fires. Routes the IRQ to whichever Driver Cell registered
145/// for it via `sys_wait_irq` (the block / net / gpu cells all rely on this), ACKs the
146/// input slot when the input service Cell isn't up yet, and warns on an unclaimed slot.
147///
148/// Relocated here from the deleted kernel `virtio_blk` driver (G2 loader redesign
149/// phase 06): it is VirtIO-common, not block-specific. The former kernel-block ACK
150/// branch is gone — the virtio-blk Driver Cell now ACKs its own IRQ via the
151/// `sys_wait_irq` path above.
152#[no_mangle]
153pub extern "Rust" fn vi_handle_virtio_irq(irq: u32) {
154    let nic_irq = crate::task::drivers::driver_cell::owns_registered_nic_irq(irq);
155
156    // Driver Cell IRQ routing: a Cell registered for this IRQ via sys_wait_irq —
157    // signal it (sets IRQ_PENDING + writes VirtIO InterruptACK) and return.
158    if crate::task::drivers::irq_wait::has_waiter(irq as u8) {
159        crate::task::drivers::irq_wait::signal_irq(irq as u8);
160        if nic_irq {
161            crate::task::waker::signal_net_rx();
162        }
163        return;
164    }
165    // Input (keyboard) slot: ACK to prevent an interrupt storm before the input
166    // service Cell is up; event routing lives entirely in that Cell.
167    if crate::task::drivers::input_irq_ack::ack_if_input(irq) {
168        return;
169    }
170    if ack_unclaimed(irq) {
171        if nic_irq {
172            crate::task::waker::signal_net_rx();
173        }
174        return;
175    }
176    // Unknown VirtIO slot — no device registered. InterruptStatus is already cleared
177    // by plic_complete in the trap handler.
178    log::warn!(
179        "[virtio] unhandled IRQ {} — no registered device for this slot",
180        irq
181    );
