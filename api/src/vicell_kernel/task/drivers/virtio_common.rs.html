1//! Shared VirtIO MMIO slot enumeration.
2//!
3//! `virtio_blk`, `input_irq_ack`, and Driver Cells use `virtio_slots()` to
4//! iterate all VirtIO MMIO slots for the current platform.
5//!
6//! AArch64: scans all 32 slots at 0x0a000000, stride 0x200 (QEMU virt layout).
7//! QEMU assigns devices to slots in an implementation-defined order so we must
8//! probe all 32.  The identity map in paging.rs covers the full 0x0a004000 range.
9//!
10//! Other arches: reads DTB-confirmed slots from `platform::PLATFORM`.
11
12extern crate alloc;
13use alloc::vec::Vec;
14
15/// A VirtIO MMIO slot with base address and IRQ.
16pub struct VirtioSlot {
17    pub base: usize,
18    pub irq: u32,
19}
20
21/// Iterator over all VirtIO MMIO slots for the current platform.
22pub fn virtio_slots() -> impl Iterator<Item = VirtioSlot> {
23    #[cfg(target_arch = "aarch64")]
24    {
25        // QEMU ARM virt: 32 VirtIO MMIO slots at 0x0a000000, 512 bytes each, SPI 16+i.
26        // All 32 slots are identity-mapped by init_kernel_paging (0x0a000000..0x0a004000).
27        const BASE: usize = 0x0a00_0000;
28        const STRIDE: usize = 0x200;
29        let slots: Vec<VirtioSlot> = (0..32_usize)
30            .map(|i| VirtioSlot {
31                base: BASE + i * STRIDE,
32                irq: 16 + i as u32,
33            })
34            .collect();
35        slots.into_iter()
36    }
37    #[cfg(not(target_arch = "aarch64"))]
38    {
39        let slots: Vec<VirtioSlot> = crate::platform::with(|p| {
40            p.virtio_mmio
41                .iter()
42                .filter_map(|e| {
43                    e.as_ref().map(|e| VirtioSlot {
44                        base: e.base,
45                        irq: e.irq,
46                    })
47                })
48                .collect()
49        });
50        slots.into_iter()
51    }
52}

54/// Acknowledge an interrupt from an as-yet unclaimed VirtIO MMIO slot.
55///
56/// Driver Cells may need several synchronous queue transactions to initialise
57/// before they can enter `sys_wait_irq`. Leaving those level-triggered
58/// interrupts asserted starves the Cell before registration completes.
59fn ack_unclaimed(irq: u32) -> bool {
60    #[cfg(target_arch = "aarch64")]
61    let base = (16..48)
62        .contains(&irq)
63        .then_some(0x0a00_0000usize + (irq as usize - 16) * 0x200);
64    #[cfg(target_arch = "riscv64")]
65    let base = (1..9)
66        .contains(&irq)
67        .then_some(0x1000_1000usize + (irq as usize - 1) * 0x1000);
68    // x86_64 routes VirtIO through PCI MSI-X, not a fixed MMIO IRQ window, so
69    // there is no slot base to derive and nothing to acknowledge here.
70    #[cfg(not(any(target_arch = "aarch64", target_arch = "riscv64")))]
71    let base: Option<usize> = {
72        let _ = irq;
73        None
74    };
75
76    let Some(base) = base else {
77        return false;
78    };
79    // SAFETY: the architecture-specific ranges above are the identity-mapped
80    // VirtIO MMIO windows. InterruptStatus is read-only and InterruptACK accepts
81    // exactly the status bits returned by the device.
82    let status = unsafe { core::ptr::read_volatile((base + 0x60) as *const u32) };
83    if status == 0 {
84        return false;
85    }
86    unsafe { core::ptr::write_volatile((base + 0x64) as *mut u32, status) };
87    true
88}

90/// VirtIO MMIO IRQ dispatcher — called from the arch trap handlers (riscv64/aarch64)
91/// when any VirtIO MMIO IRQ fires. Routes the IRQ to whichever Driver Cell registered
92/// for it via `sys_wait_irq` (the block / net / gpu cells all rely on this), ACKs the
93/// input slot when the input service Cell isn't up yet, and warns on an unclaimed slot.
94///
95/// Relocated here from the deleted kernel `virtio_blk` driver (G2 loader redesign
96/// phase 06): it is VirtIO-common, not block-specific. The former kernel-block ACK
97/// branch is gone — the virtio-blk Driver Cell now ACKs its own IRQ via the
98/// `sys_wait_irq` path above.
99#[no_mangle]
100pub extern "Rust" fn vi_handle_virtio_irq(irq: u32) {
101    // Driver Cell IRQ routing: a Cell registered for this IRQ via sys_wait_irq —
102    // signal it (sets IRQ_PENDING + writes VirtIO InterruptACK) and return.
103    if crate::task::drivers::irq_wait::has_waiter(irq as u8) {
104        crate::task::drivers::irq_wait::signal_irq(irq as u8);
105        return;
106    }
107    // Input (keyboard) slot: ACK to prevent an interrupt storm before the input
108    // service Cell is up; event routing lives entirely in that Cell.
109    if crate::task::drivers::input_irq_ack::ack_if_input(irq) {
110        return;
111    }
112    if ack_unclaimed(irq) {
113        return;
114    }
115    // Unknown VirtIO slot — no device registered. InterruptStatus is already cleared
116    // by plic_complete in the trap handler.
117    log::warn!(
118        "[virtio] unhandled IRQ {} — no registered device for this slot",
119        irq
120    );
121}
