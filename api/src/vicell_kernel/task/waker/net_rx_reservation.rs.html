1//! Serialized ownership of the single NET_RX completion reservation.
2
3use super::{CompletionQueue, SlotId};
4use crate::sync::Spinlock;
5use alloc::sync::Arc;
6
7enum Reservation {
8    /// No slot is armed. The last queue handle stays here so finishing an ISR
9    /// completion never drops its final `Arc` in interrupt context.
10    Idle(Option<Arc<CompletionQueue>>),
11    Armed {
12        queue: Arc<CompletionQueue>,
13        slot: SlotId,
14    },
15    Completing {
16        queue: Arc<CompletionQueue>,
17        slot: SlotId,
18    },
19}
20
21static RESERVATION: Spinlock<Reservation> = Spinlock::new(Reservation::Idle(None));
22
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum DisarmResult {
25    Owned(SlotId),
26    Completing,
27    NotOwned,
28}
29
30pub struct PendingCompletion {
31    queue: Arc<CompletionQueue>,
32    slot: SlotId,
33}
34
35pub fn arm(queue: Arc<CompletionQueue>, slot: SlotId) {
36    let arriving = queue.cell().0;
37    let displaced = loop {
38        let mut state = RESERVATION.lock();
39        match &mut *state {
40            Reservation::Idle(retained) => {
41                let retired = retained.take();
42                *state = Reservation::Armed { queue, slot };
43                drop(state);
44                // Replacing an idle queue happens in syscall context, where
45                // releasing its final reference may safely reach the allocator.
46                drop(retired);
47                return;
48            }
49            Reservation::Armed {
50                queue: previous,
51                slot: previous_slot,
52            } => {
53                let pending = PendingCompletion {
54                    queue: Arc::clone(previous),
55                    slot: *previous_slot,
56                };
57                *state = Reservation::Completing {
58                    queue: Arc::clone(previous),
59                    slot: *previous_slot,
60                };
61                break pending;
62            }
63            Reservation::Completing { .. } => {
64                drop(state);
65                core::hint::spin_loop();
66            }
67        }
68    };
69
70    let displaced_cell = displaced.queue.cell().0;
71    let _ = displaced.queue.complete_from(
72        displaced.slot,
73        api::completion::source::NET_RX,
74        api::completion::RESULT_ABANDONED as isize,
75    );
76    let mut state = RESERVATION.lock();
77    *state = Reservation::Armed { queue, slot };
78    log::warn!(
79        "[net-rx] cell {} took over the reservation held by cell {}",
80        arriving,
81        displaced_cell
82    );
83}

85pub fn disarm(queue: &Arc<CompletionQueue>) -> DisarmResult {
86    let mut state = RESERVATION.lock();
87    match &*state {
88        Reservation::Armed { queue: armed, slot } if Arc::ptr_eq(armed, queue) => {
89            let owned = *slot;
90            *state = Reservation::Idle(Some(Arc::clone(armed)));
91            DisarmResult::Owned(owned)
92        }
93        Reservation::Completing {
94            queue: completing, ..
95        } if Arc::ptr_eq(completing, queue) => DisarmResult::Completing,
96        _ => DisarmResult::NotOwned,
97    }
98}

100pub fn begin_signal() -> Option<PendingCompletion> {
101    let mut state = RESERVATION.lock();
102    let pending = match &*state {
103        Reservation::Armed { queue, slot } => PendingCompletion {
104            queue: Arc::clone(queue),
105            slot: *slot,
106        },
107        Reservation::Idle(_) | Reservation::Completing { .. } => return None,
108    };
109    *state = Reservation::Completing {
110        queue: Arc::clone(&pending.queue),
111        slot: pending.slot,
112    };
113    Some(pending)
114}
115
116pub fn finish_signal(pending: PendingCompletion, result: isize) {
117    let _ = pending
118        .queue
119        .complete_from(pending.slot, api::completion::source::NET_RX, result);
120    let mut state = RESERVATION.lock();
121    if matches!(
122        &*state,
123        Reservation::Completing { queue, slot }
124            if Arc::ptr_eq(queue, &pending.queue) && *slot == pending.slot
125    ) {
126        *state = Reservation::Idle(Some(Arc::clone(&pending.queue)));
127    }
128}
