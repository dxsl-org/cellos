1//! Per-cell completion queue: the kernel-owned place an asynchronous result lands.
2//!
3//! The queue is heap memory the kernel allocates and holds through the task
4//! record. It is deliberately **not** a grant: a grant can be unregistered or
5//! freed by the cell that owns it, and a queue that can vanish under an
6//! in-flight operation turns every completion into a write to whatever was
7//! handed the frame next. Because the queue is kernel-owned, appending needs no
8//! address resolution and no allocator, which is what lets the append path hold
9//! one leaf lock and nothing else.
10//!
11//! Two rules make a completion infallible:
12//!
13//! - **A slot is reserved when the operation is submitted**, from the submitting
14//!   context, where a refusal is an ordinary error the caller can act on. By the
15//!   time anything is in flight its landing place already exists.
16//! - **Appending never allocates and never grows the queue.** The drainable ring
17//!   holds slot indexes and is exactly as long as the slot array, so it cannot
18//!   overflow: a slot contributes at most one entry between reservation and
19//!   release.
20//!
21//! Waking the task that is waiting is a *separate*, deferred step. An append may
22//! run in interrupt context, and making a task runnable needs the scheduler lock,
23//! which must never be taken from there. So an append only raises a flag; the
24//! flag is turned into a scheduler wake by [`deliver_pending_wakes`], called from
25//! `yield_cpu` after `SCHEDULER` is free — the same deferral the grant reap uses
26//! for work that cannot run under the sweep's locks.
27//!
28//! Lock order: this module's leaf lock is taken alone. It is never held across a
29//! call that takes `SCHEDULER`, `FRAME_ALLOCATOR` or `KERNEL_ROOT`, and none of
30//! those is held when it is taken.
31
32use super::scheduler::Scheduler;
33use super::tcb::TaskState;
34use crate::sync::Spinlock;
35use alloc::sync::Arc;

40/// Operations one cell may have outstanding at once.
41///
42/// The cap is the backpressure: a cell that submits without draining is refused
43/// at submission rather than served, so the failure is visible, attributable and
44/// confined to the cell that caused it.
45pub const QUEUE_CAPACITY: usize = 32;
46
47/// Names one submission within the cell that reserved it.
48///
49/// A slot — not a task id — is the identifier a completion carries. Task ids are
50/// reused, do not survive a restart, and cannot tell two operations of one task
51/// apart. A slot is meaningful only inside the cell that owns the queue and dies
52/// with it.
53#[derive(Clone, Copy, PartialEq, Eq, Debug)]
54pub struct SlotId(u16);
55
56impl SlotId {
57    /// Position of this slot in its queue. Always `< QUEUE_CAPACITY`.
58    pub fn index(self) -> usize {
59        self.0 as usize
60    }
61}
62
63/// A finished operation, ready to be handed to whoever was waiting.
64#[derive(Clone, Copy, PartialEq, Eq, Debug)]
65pub struct Completion {
66    pub slot: SlotId,
67    /// One public source bit from `api::completion::source`, or zero for
68    /// kernel-only queue self-tests that do not model a public producer.
69    pub source: u32,
70    /// Operation result. Negative values are reserved for errors; the exact
71    /// encoding belongs with the first operation migrated onto the queue, not
72    /// here, so nothing is frozen before a real caller exists.
73    pub result: isize,
74}
75
76#[derive(Clone, Copy, PartialEq, Eq)]
77enum Slot {
78    /// No submission owns this index.
79    Free,
80    /// A submission owns this index; its result has not arrived.
81    Reserved,
82    /// The result arrived and has not been drained.
83    Done { source: u32, result: isize },
84}
85
86struct Ring {
87    slots: [Slot; QUEUE_CAPACITY],
88    /// Indexes of slots holding an undrained result, oldest first.
89    drainable: [u16; QUEUE_CAPACITY],
90    head: usize,
91    len: usize,
92}
93
94/// Raised by any append, cleared by [`deliver_pending_wakes`]. A single relaxed
95/// load keeps the common tick — no completions anywhere — at one atomic read.
96static WAKES_PENDING: AtomicBool = AtomicBool::new(false);
97
98/// The landing ground for one cell's completions.
99///
100/// Shared by every thread of the cell through `Arc`, so the last reference to
101/// disappear frees it. A completion source that still holds a reference keeps
102/// the queue alive past the cell's death: the append then lands in a queue
103/// nobody drains, which leaks a bounded allocation rather than corrupting
104/// whatever took the memory next.
105pub struct CompletionQueue {
106    cell: CellId,
107    ring: Spinlock<Ring>,
108    /// Task to make runnable when a completion arrives; 0 means nobody is
109    /// waiting. See [`CompletionQueue::register_waiter`] for the contract.
110    waiter: AtomicUsize,
111    wake_requested: AtomicBool,
112}
113
114impl CompletionQueue {
115    fn new(cell: CellId) -> Self {
116        Self {
117            cell,
118            ring: Spinlock::new(Ring {
119                slots: [Slot::Free; QUEUE_CAPACITY],
120                drainable: [0; QUEUE_CAPACITY],
121                head: 0,
122                len: 0,
123            }),
124            waiter: AtomicUsize::new(0),
125            wake_requested: AtomicBool::new(false),
126        }
127    }
128
129    /// Cell this queue belongs to.
130    pub fn cell(&self) -> CellId {
131        self.cell
132    }
133
134    /// Take a slot for an operation about to be submitted.
135    ///
136    /// # Errors
137    /// `None` when every slot is taken. Call this from the submitting cell's own
138    /// context and turn it into an ordinary error there: refusing an interrupt
139    /// handler that is trying to report completion would deadlock the system it
140    /// was serving, so the refusal has to happen here, before anything is in
141    /// flight.
142    pub fn reserve(&self) -> Option<SlotId> {
143        let mut ring = self.ring.lock();
144        let index = ring.slots.iter().position(|s| *s == Slot::Free)?;
145        ring.slots[index] = Slot::Reserved;
146        Some(SlotId(index as u16))
147    }
148
149    /// Record the result of the operation holding `slot`.
150    ///
151    /// Takes this queue's leaf lock and nothing else, so it is safe from
152    /// interrupt context. Waking the waiter is deliberately not done here — see
153    /// [`deliver_pending_wakes`].
154    ///
155    /// # Returns
156    /// `true` once the result is stored. `false` only for a kernel-side protocol
157    /// violation — completing a slot that was never reserved, or completing one
158    /// twice — which is logged and leaves the queue untouched. A slot obtained
159    /// from [`reserve`](Self::reserve) and completed once always succeeds, which
160    /// is the property the whole design rests on.
161    #[must_use = "a refused completion means a slot was misused and the waiter will hang"]
162    pub fn complete(&self, slot: SlotId, result: isize) -> bool {
163        self.complete_from(slot, api::completion::source::UNSPECIFIED, result)
164    }
165
166    /// Record a public source and result for the operation holding `slot`.
167    ///
168    /// Production completion sources must use this method so the userspace
169    /// record identifies why the parked executor woke. The zero-source
170    /// [`complete`](Self::complete) wrapper remains for kernel-only queue tests.
171    #[must_use = "a refused completion means a slot was misused and the waiter will hang"]
172    pub fn complete_from(&self, slot: SlotId, source: u32, result: isize) -> bool {
173        let index = slot.index();
174        // The whole lock-holding region is this block, and it contains no call
175        // that could reach another lock — not the logger, which is why the
176        // refusal below reports after the guard is dropped.
177        let refused = {
178            let mut ring = self.ring.lock();
179            match ring.slots.get(index) {
180                Some(Slot::Reserved) => {
181                    ring.slots[index] = Slot::Done { source, result };
182                    // Cannot overflow: `drainable` is as long as `slots`, and a
183                    // slot is only pushed on the Reserved -> Done edge, which
184                    // happens once per reservation.
185                    let position = (ring.head + ring.len) % QUEUE_CAPACITY;
186                    ring.drainable[position] = index as u16;
187                    ring.len += 1;
188                    // Publish the wake while the ring transition is serialized.
189                    // A drain that empties the ring can then cancel this exact
190                    // request without erasing a later completion's wake.
191                    self.wake_requested.store(true, Ordering::Release);
192                    None
193                }
194                Some(Slot::Free) => Some("free"),
195                Some(Slot::Done { .. }) => Some("already done"),
196                None => Some("out of range"),
197            }
198        };
199
200        if let Some(why) = refused {
201            log::error!(
202                "[completion] cell {} slot {} completed while {}",
203                self.cell.0,
204                index,
205                why
206            );
207            return false;
208        }
209
210        // The queue-local request is already visible with the entry. Raising the
211        // global scan flag afterwards may cause a harmless empty scan if the
212        // submitter drains first, but can never hide a completion.
213        WAKES_PENDING.store(true, Ordering::Release);
214        true
215    }
216
217    /// Withdraw a reservation whose operation never started.
218    ///
219    /// This is not a completion and must never be expressed as one: completing
220    /// a slot raises a wake request, and a request raised by the submitter
221    /// itself — which is already running — is still outstanding when that same
222    /// task parks next, so the park is cancelled the instant it begins. A
223    /// submitter that withdraws and re-submits in a loop would then never
224    /// sleep at all.
225    ///
226    /// # Returns
227    /// `true` when the slot was reserved and is now free. `false` when it
228    /// already holds a result: the submitter lost the race with the source and
229    /// must [`drain`](Self::drain) that result rather than discard it.
230    #[must_use = "a refused withdrawal means a result is waiting and would be lost"]
231    pub fn release(&self, slot: SlotId) -> bool {
232        let mut ring = self.ring.lock();
233        match ring.slots.get(slot.index()) {
234            Some(Slot::Reserved) => {
235                ring.slots[slot.index()] = Slot::Free;
236                true
237            }
238            _ => false,
239        }
240    }
241
242    /// Take the oldest undrained completion, releasing its slot for reuse.
243    ///
244    /// Holds the same single leaf lock as [`complete`](Self::complete); the
245    /// report of an inconsistency is deliberately made after the guard drops.
246    pub fn drain(&self) -> Option<Completion> {
247        let taken = {
248            let mut ring = self.ring.lock();
249            if ring.len == 0 {
250                return None;
251            }
252            let index = ring.drainable[ring.head] as usize;
253            ring.head = (ring.head + 1) % QUEUE_CAPACITY;
254            ring.len -= 1;
255            if ring.len == 0 {
256                self.wake_requested.store(false, Ordering::Release);
257            }
258            match ring.slots[index] {
259                Slot::Done { source, result } => {
260                    ring.slots[index] = Slot::Free;
261                    Ok(Completion {
262                        slot: SlotId(index as u16),
263                        source,
264                        result,
265                    })
266                }
267                // Unreachable while the ring and the slot array agree; treat a
268                // disagreement as an empty drain rather than a stale read.
269                _ => Err(index),
270            }
271        };
272
273        match taken {
274            Ok(completion) => Some(completion),
275            Err(index) => {
276                log::error!(
277                    "[completion] cell {} drained slot {} that holds no result",
278                    self.cell.0,
279                    index
280                );
281                None
282            }
283        }
284    }
285
286    /// Slots taken by submissions that have not yet reported a result.
287    pub fn reserved(&self) -> usize {
288        let ring = self.ring.lock();
289        ring.slots.iter().filter(|s| **s == Slot::Reserved).count()
290    }
291
292    /// Completions recorded and not yet drained.
293    pub fn drainable(&self) -> usize {
294        self.ring.lock().len
295    }
296
297    /// Name `tid` as the task to make runnable when a completion arrives.
298    ///
299    /// # Contract
300    /// The registrant must be parked in a state whose only wake condition is
301    /// this queue. Registering a task that is also parked on IPC or a timer
302    /// makes it runnable early, and the state it parked in is what those paths
303    /// match on. Registration is kernel-internal and has no caller outside the
304    /// boot self-test; the park state belongs with the first operation migrated
305    /// onto the queue, where a real caller can pin its shape.
306    pub fn register_waiter(&self, tid: usize) {
307        self.waiter.store(tid, Ordering::Release);
308    }
309
310    /// Stop waking `tid` for this queue if it still owns the registration.
311    ///
312    /// Returns `false` when another waiter replaced `tid`; the older wait must
313    /// not erase the newer waiter's registration while unwinding.
314    pub fn clear_waiter(&self, tid: usize) -> bool {
315        self.waiter
316            .compare_exchange(tid, 0, Ordering::AcqRel, Ordering::Acquire)
317            .is_ok()
318    }
319
320    /// Consume a pending wake request, yielding the task to make runnable.
321    fn take_wake_request(&self) -> Option<usize> {
322        if !self.wake_requested.swap(false, Ordering::AcqRel) {
323            return None;
324        }
325        match self.waiter.load(Ordering::Acquire) {
326            0 => None,
327            tid => Some(tid),
328        }
329    }
330}

332/// The queue serving `tid`'s cell, created on first use.
333///
334/// Threads of one cell share one queue, so the handle is looked up across the
335/// cell before a new one is allocated. Returns `None` only when `tid` names no
336/// live task.
337///
338/// Call from a submitting context with `SCHEDULER` held. The append path never
339/// comes here — it uses the handle it was given, which is the reason it needs no
340/// lock but its own.
341pub fn queue_for(sched: &mut Scheduler, tid: usize) -> Option<Arc<CompletionQueue>> {
342    let cell = sched.tasks.get(&tid)?.cell_id;
343    if let Some(existing) = sched.tasks.get(&tid).and_then(|t| t.completion.clone()) {
344        return Some(existing);
345    }
346    let queue = sched
347        .tasks
348        .values()
349        .filter(|t| t.cell_id == cell)
350        .find_map(|t| t.completion.clone())
351        .unwrap_or_else(|| Arc::new(CompletionQueue::new(cell)));
352    sched.tasks.get_mut(&tid)?.completion = Some(queue.clone());
353    Some(queue)
354}

356/// Whether any append has asked for a wake since the last delivery.
357pub fn wakes_pending() -> bool {
358    WAKES_PENDING.load(Ordering::Acquire)
359}
360
361/// A state a completion wake may leave, versus one it must not disturb.
362///
363/// `Ready`/`Running` are already runnable, a `Terminated` task has nowhere to go,
364/// and `Frozen` is held deliberately by hot-swap — unfreezing it here would run a
365/// cell mid-swap.
366fn is_parked(state: &TaskState) -> bool {
367    !matches!(
368        state,
369        TaskState::Ready | TaskState::Running | TaskState::Terminated | TaskState::Frozen { .. }
370    )
371}

373/// Turn flagged completions into scheduler wakes.
374///
375/// Runs where `SCHEDULER` may be taken and neither `FRAME_ALLOCATOR` nor
376/// `KERNEL_ROOT` is held — that is, from `yield_cpu` and not from an append.
377/// Requires `SCHEDULER` to be held by the caller.
378pub fn deliver_pending_wakes(sched: &mut Scheduler) {
379    // Cleared before the scan: an append landing during the scan re-raises the
380    // flag, so at worst its wake waits for the next tick. Clearing afterwards
381    // would swallow it entirely.
382    if !WAKES_PENDING.swap(false, Ordering::AcqRel) {
383        return;
384    }
385
386    // Reached through the task table, so a queue kept alive only by a completion
387    // source outlives its cell unreachable — which is the intended end state: its
388    // registered waiter is dead, and the alternative, a wake for a tid the table
389    // has since reissued, would make a stranger runnable.
390    let mut requested: Vec<usize> = Vec::new();
391    for task in sched.tasks.values() {
392        if let Some(queue) = task.completion.as_ref() {
393            if let Some(tid) = queue.take_wake_request() {
394                requested.push(tid);
395            }
396        }
397    }
398
399    let mut runnable: Vec<usize> = Vec::new();
400    for tid in requested {
401        if let Some(task) = sched.tasks.get_mut(&tid) {
402            if is_parked(&task.state) {
403                task.state = TaskState::Ready;
404                runnable.push(tid);
405            }
406        }
407    }
408    for tid in runnable {
409        sched.push_ready(tid);
410    }
