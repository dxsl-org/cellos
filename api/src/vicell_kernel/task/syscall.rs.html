14/// Set of physical frames currently issued via `ShmAlloc`.
15/// `ShmMap` accepts only handles that appear here, preventing a malicious
16/// cell from mapping arbitrary kernel/cell-owned frames into its address
17/// space via a forged handle.
18///
19/// NOTE: This is still a single global pool — any cell that knows a peer's
20/// outstanding handle can map it. A per-owner ACL is the proper fix; this
21/// table is the minimum bar to stop "ShmMap kernel_text_phys" attacks.
22static SHM_HANDLES: Spinlock<Option<BTreeSet<usize>>> = Spinlock::new(None);
23
24fn shm_handles_lock() -> &'static Spinlock<Option<BTreeSet<usize>>> {
25    &SHM_HANDLES
26}
27
28fn shm_register(handle: usize) {
29    let mut guard = shm_handles_lock().lock();
30    if guard.is_none() {
31        *guard = Some(BTreeSet::new());
32    }
33    if let Some(set) = guard.as_mut() {
34        set.insert(handle);
35    }
36}
37
38fn shm_is_valid(handle: usize) -> bool {
39    let guard = shm_handles_lock().lock();
40    guard.as_ref().is_some_and(|set| set.contains(&handle))
41}

45/// Kernel-managed zero-copy memory region.
46///
47/// Distinct from `tcb::GrantEntry` which tracks per-task grants from the kernel.
48/// Owner and grantee are tracked by raw task id (usize) — same as `caller_id`
49/// from `current_task_id()`. We intentionally avoid `CellId` wrappers here
50/// because `caller_id` IS a task id, not a Cell id (F7).
51struct PageGrant {
52    base: usize,                           // physical address of the first allocated page
53    size: usize,                           // total byte count (multiple of 4096)
54    owner: usize,                          // task id of the GrantAlloc caller
55    shared_to: Option<(usize, GrantPerm)>, // current grantee task id + permission
56}

89/// Allocate `n_pages` contiguous physical frames, map them USER RW, and zero them.
90///
91/// Returns the physical base address on success, or `None` on OOM or partial map.
92/// Lock order: FRAME_ALLOCATOR (alloc) → FRAME_ALLOCATOR (map_page) → release →
93///             FRAME_ALLOCATOR (partial-failure dealloc).
94fn alloc_grant_pages(n_pages: usize) -> Option<usize> {
95    use crate::memory::frame::FRAME_ALLOCATOR;
96    use crate::memory::paging::Flags;
97    const PAGE_SIZE: usize = 4096;
98
99    let user_flags =
100        Flags::VALID | Flags::READ | Flags::WRITE | Flags::USER | Flags::ACCESSED | Flags::DIRTY;
101
102    let paddr = {
103        let mut g = FRAME_ALLOCATOR.lock();
104        g.as_mut().and_then(|a| a.allocate_contiguous(n_pages))?
105    };
106
107    let mut mapped = 0usize;
108    {
109        let mut guard = FRAME_ALLOCATOR.lock();
110        if let Some(alloc) = guard.as_mut() {
111            for i in 0..n_pages {
112                let v = paddr + i * PAGE_SIZE;
113                if crate::memory::paging::map_page(alloc, v, v, Flags::from_bits(user_flags))
114                    .is_ok()
115                {
116                    mapped += 1;
117                } else {
118                    break;
119                }
120            }
121        }
122    }
123
124    if mapped < n_pages {
125        // Partial map: restore the kernel identity mapping for the pages we
126        // re-mapped USER (NOT a bare unmap — every Usable frame must stay
127        // identity-mapped for the loader's identity-address zeroing), then free.
128        let kernel_rwx = Flags::VALID
129            | Flags::READ
130            | Flags::WRITE
131            | Flags::EXECUTE
132            | Flags::ACCESSED
133            | Flags::DIRTY;
134        let mut fa = FRAME_ALLOCATOR.lock();
135        if let Some(a) = fa.as_mut() {
136            for i in 0..mapped {
137                let f = paddr + i * PAGE_SIZE;
138                let _ = crate::memory::paging::unmap_page(f);
139                let _ = crate::memory::paging::map_page(a, f, f, Flags::from_bits(kernel_rwx));
140            }
141            for k in 0..n_pages {
142                a.deallocate_frame(paddr + k * PAGE_SIZE);
143            }
144        }
145        crate::memory::paging::tlb_flush_all();
146        return None;
147    }
148
149    // Zero every mapped page before handing to user: prevents stale data from a
150    // previously-freed grant leaking to a different cell (info-disclosure under G2).
151    // SAFETY: frames are identity-mapped USER RW; SUM=1 allows S-mode writes.
152    unsafe {
153        core::ptr::write_bytes(paddr as *mut u8, 0, n_pages * PAGE_SIZE);
154    }
155
156    Some(paddr)
157}

159/// Restore the boot identity mapping (kernel RWX) and deallocate `n_pages`
160/// physical frames starting at `base`.
161///
162/// Grant frames are identity-mapped at boot (RWX kernel) and re-mapped USER RW by
163/// `alloc_grant_pages`. On free we must RESTORE the kernel identity mapping — NOT
164/// unmap it: in the SAS model every Usable frame must stay identity-mapped so the
165/// cell loader can zero a reused frame through its identity address
166/// (`phys_to_virt(frame)`). Unmapping here left freed grant frames with no PTE →
167/// a later cell load store-faulted while zeroing BSS (and read wrong pages).
168/// Dropping the USER bit also prevents a stale cell from touching a reused frame.
169///
170/// Lock order: FRAME_ALLOCATOR → (map_page/unmap_page → KERNEL_ROOT). Mirrors
171/// `Stack::drop`. Must NOT be called while already holding FRAME_ALLOCATOR.
172fn free_grant_pages(base: usize, n_pages: usize) {
173    use crate::memory::frame::FRAME_ALLOCATOR;
174    use crate::memory::paging::Flags;
175    const PAGE_SIZE: usize = 4096;
176    let kernel_rwx = Flags::from_bits(
177        Flags::VALID | Flags::READ | Flags::WRITE | Flags::EXECUTE | Flags::ACCESSED | Flags::DIRTY,
178    );
179
180    let mut fa = FRAME_ALLOCATOR.lock();
181    if let Some(alloc) = fa.as_mut() {
182        for i in 0..n_pages {
183            let f = base + i * PAGE_SIZE;
184            let _ = crate::memory::paging::unmap_page(f);
185            let _ = crate::memory::paging::map_page(alloc, f, f, kernel_rwx);
186            alloc.deallocate_frame(f);
187        }
188    }
189    crate::memory::paging::tlb_flush_all();
190}

192// ── Grant Reaper ──────────────────────────────────────────────────────────────

194/// Reclaim all grant pages owned or held by a dying task.
195///
196/// Called from every task-exit code path (Exit syscall, ForceExit, scheduler watchdog, fault handler).
197/// Two effects:
198///   1. Owner death  — remove entry, unmap pages, return frames to allocator.
199///   2. Grantee death — clear `shared_to` so the owner's grant becomes unshared.
200///
201/// Lock order: PAGE_GRANT_TABLE collect → unmap (KERNEL_ROOT) → FRAME_ALLOCATOR.
202/// Never holds FRAME_ALLOCATOR when calling free_grant_pages.
203pub(crate) fn reap_grants_for_task(dead_tid: usize) {
204    const PAGE_SIZE: usize = 4096;
205
206    // ── PAGE_GRANT_TABLE pass ─────────────────────────────────────────────────
207    let owned: alloc::vec::Vec<PageGrant> = {
208        let mut tbl = grant_table_lock().lock();
209        let Some(map) = tbl.as_mut() else { return };
210        // Clear grantee references (no removal needed — owner keeps the entry).
211        for grant in map.values_mut() {
212            if grant.shared_to.is_some_and(|(tid, _)| tid == dead_tid) {
213                grant.shared_to = None;
214            }
215        }
216        // Collect and remove owned entries.
217        let owned_keys: alloc::vec::Vec<usize> = map
218            .iter()
219            .filter(|(_, g)| g.owner == dead_tid)
220            .map(|(k, _)| *k)
221            .collect();
222        owned_keys.iter().filter_map(|k| map.remove(k)).collect()
223    }; // PAGE_GRANT_TABLE lock released
224
225    for grant in &owned {
226        free_grant_pages(grant.base, grant.size / PAGE_SIZE);
227    }
228
229    // ── REG_GRANT_TABLE pass ──────────────────────────────────────────────────
230    let reg_owned: alloc::vec::Vec<RegGrant> = {
231        let mut tbl = reg_grant_table_lock().lock();
232        let Some(map) = tbl.as_mut() else { return };
233        // Clear grantee references when the grantee dies.
234        for grant in map.values_mut() {
235            if grant.shared_to.is_some_and(|(tid, _)| tid == dead_tid) {
236                grant.shared_to = None;
237            }
238        }
239        let mut owned_keys = alloc::vec::Vec::new();
240        let mut orphan_keys = alloc::vec::Vec::new();
241        for (&key, grant) in map.iter_mut() {
242            if grant.owner == dead_tid {
243                if let Some((grantee, _)) = grant.shared_to {
244                    // Transfer ownership to the sole grantee. The grantee's
245                    // owner-death cleanup drops its cached pointer before calling
246                    // GrantUnregister, allowing the pages to be reclaimed without
247                    // exposing an owner-dead, permanently leaked grant.
248                    grant.owner = grantee;
249                } else {
250                    owned_keys.push(key);
251                }
252            } else if grant.owner == 0 && grant.shared_to.is_none() {
253                orphan_keys.push(key);
254            }
255        }
256        owned_keys.extend(orphan_keys);
257        owned_keys.iter().filter_map(|k| map.remove(k)).collect()
258    }; // REG_GRANT_TABLE lock released
259
260    for reg in &reg_owned {
261        free_grant_pages(reg.base, reg.size / PAGE_SIZE);
262    }
263}

306/// Per-cell block-I/O range gate (Milestone 2.5 P03).
307///
308/// Replaces the old global `sector >= CELL_TABLE_BASE_LBA` check: the caller's
309/// `block_regions` bitmask (from its manifest PART_* bits, or the legacy VFS
310/// grant) defines which MBR partitions its raw block syscalls may address.
311/// Deny-by-default: a sector outside every granted partition is rejected —
312/// which structurally protects P2 (cell table) and P3 (snapshot), since no
313/// bit exists for them. Logs every denial (silent denials cost us a day once).
314fn check_block_access(caller_id: usize, sector: u64, count: u64) -> bool {
315    use crate::loader::disk_layout as dl;
316    let regions = super::SCHEDULER
317        .lock()
318        .as_ref()
319        .and_then(|s| s.tasks.get(&caller_id))
320        .map(|t| t.block_regions)
321        .unwrap_or(0);
322    let end = match sector.checked_add(count) {
323        Some(e) => e,
324        None => return false,
325    };
326    const GRANTABLE: [(u8, u64, u64); 4] = [
327        (0b001, dl::PART_FAT32_BASE_LBA, dl::PART_FAT32_SECTORS), // P1 (PART_DATA)
328        (0b010, dl::PART_LFS_BASE_LBA, dl::PART_LFS_SECTORS),     // P4 (PART_LFS)
329        (0b100, dl::PART_SRV_BASE_LBA, dl::PART_SRV_SECTORS), // P5 (SRV/RedoxFS, co-granted w/ LFS)
330        (
331            0b1000,
332            dl::PART_CELLSTORE_BASE_LBA,
333            dl::PART_CELLSTORE_SECTORS,
334        ), // P6 (cell-store, read-only; VFS /bin overlay)
335    ];
336    for (bit, base, size) in GRANTABLE {
337        if regions & bit != 0 && sector >= base && end <= base + size {
338            return true;
339        }
340    }
341    log::warn!(
342        "[blk] sector {}..{} denied for tid {} (regions={:#04b})",
343        sector,
344        end,
345        caller_id,
346        regions
347    );
348    false
349}

375/// Validate a user-supplied (ptr, len) buffer descriptor.
376///
377/// Rejects: NULL pointer, zero-length when expected non-empty, lengths above
378/// `max`, and pointer+length arithmetic overflow.
379///
380/// Does NOT walk the page table to confirm the U-bit. The trap handler enables
381/// SUM only for the duration of `handle_syscall`, so a kernel-space `ptr`
382/// supplied by user code will fault on access — but the fault is far more
383/// graceful when we reject obvious garbage up front.
384#[inline]
385fn validate_user_buf(ptr: usize, len: usize, max: usize) -> Result<(), SyscallError> {
386    if ptr == 0 {
387        return Err(SyscallError::InvalidInput);
388    }
389    if len > max {
390        return Err(SyscallError::BufferTooSmall);
391    }
392    if ptr.checked_add(len).is_none() {
393        return Err(SyscallError::InvalidInput);
394    }
395    Ok(())
396}

398fn write_optional_usize(ptr: usize, value: usize) -> Result<(), SyscallError> {
399    if ptr == 0 {
400        return Ok(());
401    }
402    validate_user_buf(ptr, core::mem::size_of::<usize>(), MAX_USER_BUF)?;
403    // SAFETY: the caller pointer and exact write size were validated above.
404    unsafe { (ptr as *mut usize).write(value) };
405    Ok(())
406}

469fn collect_process_rows<T>(
470    row_capacity: usize,
471    mut build: impl FnMut(&crate::task::tcb::Task) -> T,
472) -> Vec<T> {
473    let mut rows = Vec::new();
474    if row_capacity == 0 {
475        return rows;
476    }
477    if let Some(sched) = super::SCHEDULER.lock().as_ref() {
478        let cap = core::cmp::min(row_capacity, sched.tasks.len());
479        rows.reserve(cap);
480        for task in sched.tasks.values().take(cap) {
481            rows.push(build(task));
482        }
483    }
484    rows
485}

1001/// Dispatches a system call to the appropriate handler.
1002///
1003/// `caller_id` is the ID of the task invoking the syscall.
1004pub fn handle_syscall(caller_id: usize, syscall: Syscall) -> SyscallResult {
1005    // Syscall allowlist enforcement: reject if this syscall's bit is not set in
1006    // the per-Cell bitset loaded from ELF section `__ViCell_syscalls`.
1007    // Cells without that section default to u64::MAX (permit-all, backwards compat).
1008    if let Some(vi) = syscall_to_vi(&syscall) {
1009        if let Some(bit) = vi.allowlist_bit() {
1010            let allowed = get_syscall_allowlist(caller_id);
1011            if (allowed >> bit) & 1 == 0 {
1012                log::warn!(
1013                    "[kernel] syscall {:?} denied for tid {} (allowlist={:#018x})",
1014                    vi,
1015                    caller_id,
1016                    allowed
1017                );
1018                crate::audit::log_event(
1019                    crate::audit::AuditEvent::SyscallDenied,
1020                    &crate::audit::encode_u32x2(caller_id as u32, bit as u32),
1021                );
1022                return Err(SyscallError::PermissionDenied);
1023            }
1024        }
1025    }
1026
1027    match syscall {
1028        // --- Hubris ABI Implementation ---
1029        Syscall::Send {
1030            target,
1031            msg_ptr,
1032            msg_len,
1033        } => {
1034            crate::audit::log_event(
1035                crate::audit::AuditEvent::IpcSend,
1036                &crate::audit::encode_u32x2(caller_id as u32, target as u32),
1037            );
1038            let res = super::ipc_send(caller_id, target, msg_ptr, msg_len);
1039            match res {
1040                Ok(0) => Ok(0),
1041                Ok(1) => {
1042                    super::yield_cpu(); // Blocked
1043                    if let Some(sched) = super::SCHEDULER.lock().as_ref() {
1044                        return Ok(sched
1045                            .tasks
1046                            .get(&caller_id)
1047                            .and_then(|t| t.reply_value)
1048                            .unwrap_or(0));
1049                    }
1050                    Ok(0)
1051                }
1052                Err(_) => Err(SyscallError::InvalidCommand),
1053                _ => Ok(0),
1054            }
1055        }
1056        Syscall::TrySend {
1057            target,
1058            msg_ptr,
1059            msg_len,
1060        } => {
1061            // Non-blocking: deliver if target in Recv, else drop. Never blocks.
1062            // (Input-service sends fall back to a bounded queue — see ipc_try_send.)
1063            match super::ipc_try_send(caller_id, target, msg_ptr, msg_len) {
1064                Ok(()) => Ok(0),           // delivered (or queued for the input service)
1065                Err(()) => Ok(usize::MAX), // dropped (target not ready or gone)
1066            }
1067        }
1068        Syscall::Recv {
1069            mask,
1070            buf_ptr,
1071            buf_len,
1072        } => {
1073            // A NotifyOnExit death that arrived while we were busy (not parked in
1074            // Recv) was queued; deliver it now without blocking. The dead tid is
1075            // returned as the "sender" so a supervisor never misses a child death.
1076            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
1077                if let Some(t) = sched.tasks.get_mut(&caller_id) {
1078                    if !t.pending_deaths.is_empty() {
1079                        let (dead_tid, reason) = t.pending_deaths.remove(0);
1080                        // Deliver the exit reason as the recv payload (NotifyOnExit
1081                        // contract) so a supervisor can apply a restart policy.
1082                        if buf_len >= core::mem::size_of::<u64>()
1083                            && validate_user_buf(buf_ptr, core::mem::size_of::<u64>(), MAX_USER_BUF)
1084                                .is_ok()
1085                        {
1086                            // SAFETY: buf_ptr is validated as this caller's user buffer;
1087                            // the caller is mid-syscall so the write is exclusive in the SAS.
1088                            unsafe {
1089                                core::ptr::write_unaligned(buf_ptr as *mut u64, reason as u64);
1090                            }
1091                        }
1092                        return Ok(dead_tid);
1093                    }
1094                }
1095            }
1096
1097            // Hot-swap pending-message drain — guaranteed delivery path.
1098            //
1099            // The Step 5 drain in hotswap.rs tries to deliver buffered messages via
1100            // ipc_send, which only succeeds when the new cell is already in TaskState::Recv.
1101            // But the new cell calls sys_hotswap_ready from inside its Restore handler and
1102            // hasn't re-entered sys_recv yet at that moment, so the Step 5 drain races.
1103            //
1104            // This path is the guaranteed fallback: when the new cell finally calls
1105            // sys_recv for the first time (returning from the Restore handler through its
1106            // app_entry loop), any messages that slipped through Step 5 are delivered here
1107            // before blocking.  FIFO order matches the original sender queue order.
1108            //
1109            // Invariants:
1110            // - Lock is held for the entire drain + delivery so no concurrent send can
1111            //   observe a partially-drained queue.
1112            // - We copy from msg.data (an owned Box<[u8]>) before releasing the lock,
1113            //   avoiding any lifetime dependency on the task struct.
1114            // - current_caller is set so the cell sees the original sender's TID, not
1115            //   the hotswap orchestrator's TID, preserving the IPC identity contract.
1116            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
1117                if let Some(t) = sched.tasks.get_mut(&caller_id) {
1118                    // The drain MUST honour the recv mask: a masked recv (e.g. the
1119                    // shell awaiting a VFS reply) must NOT consume a queued input
1120                    // key event — that poisoned every VFS request/reply pair after
1121                    // the input-queue change (the shell decoded a keystroke as the
1122                    // VFS reply → "vwrite failed", and the real reply desynced the
1123                    // next conversation). Non-matching messages stay queued for the
1124                    // wildcard read loop that wants them.
1125                    let slot = t
1126                        .pending_msgs
1127                        .iter()

4186/// Per-cell syscall allowlist gate.
4187///
4188/// Reads the caller's `syscall_allowlist` bitmask and returns
4189/// `Err(SyscallError::PermissionDenied)` if the opcode's bit is not set.
4190/// The SCHEDULER lock is acquired and released here — callers must NOT hold it.
4191fn check_allowlist(syscall_id: usize, caller_id: usize) -> Result<(), SyscallError> {
4192    let sc = ViSyscall::from(syscall_id);
4193    let bit = sc.allowlist_bit();
4194    // Bit 36 gates raw block-I/O opcodes (500/501/503) and BlkReadAsync (212).
4195    let blk_io_bit: Option<u8> = if matches!(syscall_id, 500 | 501 | 503 | 212) {
4196        Some(36)
4197    } else {
4198        None
4199    };
4200
4201    // Raw opcodes with a dedicated `map_syscall` fallback mapping. These are
4202    // intentionally absent from `ViSyscall` (Law 1: keeps experimental ids out
4203    // of the stable ABI) so they decode as `Unknown` — but they are NOT unknown:
4204    // 500/501/503 are gated by bit 36 below + the ZST BlockIoCap at the handler;
4205    // 502 and the legacy FD ops (3/100/106/107/108/110/111) predate the bitmap
4206    // and stay always-permitted, matching their pre-Phase-31b behavior.
4207    let known_raw = matches!(
4208        syscall_id,
4209        3 | 100 | 106 | 107 | 108 | 110 | 111 | 500 | 501 | 502 | 503
4210    );
4211
4212    let allowlist = super::SCHEDULER
4213        .lock()
4214        .as_ref()
4215        .and_then(|s| s.tasks.get(&caller_id))
4216        .map(|t| t.syscall_allowlist)
4217        .unwrap_or(0); // task absent → deny-all for safety
4218
4219    // Deny truly-unknown opcodes that land in the legacy inner-match fallback —
4220    // their allowlist_bit() returns None, so without this guard they bypass the
4221    // check. Exit (60) and Yield (104) are always permitted unconditionally.
4222    // Known-raw ids are exempt: blocking them here made every allowlist-declaring
4223    // cell lose raw block I/O (broke the VFS FAT32 mount silently since Phase 31b).
4224    // Every deny below logs: a silent dispatch-level denial cost a full day of
4225    // triage when the shell's missing `Read` bit bricked serial input with no
4226    // kernel output at all.
4227    if sc == ViSyscall::Unknown
4228        && !known_raw
4229        && !matches!(syscall_id, 60 | 104)
4230        && allowlist != u64::MAX
4231    {
4232        log::warn!(
4233            "[kernel] unknown opcode {} denied for tid {} (allowlist={:#018x})",
4234            syscall_id,
4235            caller_id,
4236            allowlist
4237        );
4238        return Err(SyscallError::PermissionDenied);
4239    }
4240    if let Some(b) = bit {
4241        if allowlist & (1u64 << b) == 0 {
4242            log::warn!(
4243                "[kernel] syscall {:?} (bit {}) denied for tid {} (allowlist={:#018x})",
4244                sc,
4245                b,
4246                caller_id,
4247                allowlist
4248            );
4249            return Err(SyscallError::PermissionDenied);
4250        }
4251    }
4252    if let Some(b) = blk_io_bit {
4253        if allowlist & (1u64 << b) == 0 {
4254            log::warn!(
4255                "[kernel] raw block opcode {} denied for tid {} (no bit 36)",
4256                syscall_id,
4257                caller_id
4258            );
4259            return Err(SyscallError::PermissionDenied);
4260        }
4261    }
4262    Ok(())
4263}

4268pub extern "Rust" fn ViCell_syscall_dispatch(frame: &mut ViTrapFrame) {
4269    let syscall_id = frame.regs[17];
4270
4271    // Watchdog progress signal: a syscall proves the caller is making progress
4272    // (ViCell cells are poll-based — try_recv/yield every loop iteration), so
4273    // reset its CPU-monopoly counter.
4274    {
4275        let hart_id = super::hart_local::current_hart_id();

4278            if let Some(sched) = super::SCHEDULER.lock().as_mut() {
4279                if let Some(t) = sched.tasks.get_mut(&cid) {
4280                    t.run_ticks = 0;
4281                    t.rt_overrun_warned = false;
4282                }
4283            }
4284        }
4285    }
4286
4287    let a0 = frame.regs[10];
4288    let a1 = frame.regs[11];
4289    let a2 = frame.regs[12];
4290    let a3 = frame.regs[13];
4291

4293        frame.regs[10] = usize::MAX;
4294        return;
4295    };
4296
4297    let caller_id = super::current_task_id();
4298
4299    if check_allowlist(syscall_id, caller_id).is_err() {
4300        frame.regs[10] = usize::MAX;
4301        return;
4302    }
4303
4304    // SAFETY: csrs/csrc sstatus SUM (bit 18) enables S-mode access to user pages
4305    // for the duration of handle_syscall. Disabled immediately after to prevent
4306    // inadvertent user-page reads on subsequent kernel faults.
4307    #[cfg(target_arch = "riscv64")]
4308    unsafe {
4309        core::arch::asm!("csrs sstatus, {0}", in(reg) 0x40000usize);

4311

4313
4314    #[cfg(target_arch = "riscv64")]
4315    unsafe {
4316        core::arch::asm!("csrc sstatus, {0}", in(reg) 0x40000usize);
4317    }
4318
4319    match result {
4320        Ok(val) => frame.regs[10] = val,
4321        Err(_) => frame.regs[10] = usize::MAX,
4322    }
4323}
