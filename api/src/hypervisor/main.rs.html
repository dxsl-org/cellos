4//! Hypervisor Service Cell — boots an ARM64 Linux guest via EL2 Stage-2 MMU.
5//!
6//! Reads vmlinuz + initrd.gz from VIFS1, builds a minimal DTB, maps 128 MiB
7//! of guest RAM, loads all images, and runs the VmExit dispatch loop.
8#[cfg(all(feature = "volatile-disk", feature = "hostile-backend-recovery"))]
9compile_error!("volatile-disk cannot provide hostile backend recovery");
10#[cfg(all(feature = "volatile-disk", feature = "ubuntu-wide-guest"))]
11compile_error!("ubuntu-wide-guest requires the persistent /mnt/sd/guest_disk.img backend");
12#[cfg(all(feature = "ubuntu-wide-guest", not(target_arch = "x86_64")))]
13compile_error!("ubuntu-wide-guest is qualified only on the x86_64 PVH path");

131/// Entry: dispatch to the arch personality that has a VMM backend.
132#[no_mangle]
133pub fn main() -> ! {
134    #[cfg(target_arch = "aarch64")]
135    boot_arm::boot_arm();
136    #[cfg(target_arch = "x86_64")]
137    boot_x86::run();
138    #[cfg(not(any(target_arch = "aarch64", target_arch = "x86_64")))]
139    ostd::io::println("[hv] no VMM personality for this architecture");
140    quiesce()
141}
142
143pub(crate) fn quiesce() -> ! {
144    ostd::io::println("[hv] service quiesced");
145    loop {
146        let _ = ostd::syscall::sys_wait_for_event(0, 0);
147    }
148}
