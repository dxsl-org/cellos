1//! tokio-smoke - Demonstrating Tokio I/O driver pattern on CellOS via vimio.
2
3#![no_std]
4#![no_main]
5#![forbid(unsafe_code)]
6
7extern crate alloc;
8
9use core::time::Duration;
10use ostd::io::println;
11use ostd::syscall::sys_lookup_service;
12use vimio::{Events, Interest, Selector, Token, Waker};
13
14api::declare_manifest!(
15    block_io = false,
16    network = true,
17    spawn = false,
18    gpio = false,
19    uart = false,
20    hypervisor = false,
21    i2c = false,
22    spi = false
23);
24
25api::declare_syscalls![Send, Recv, TrySend, TryRecv, RecvTimeout, Log, LookupService, StateRestore, GetTime];
26
27#[cfg(target_os = "none")]
28ostd::declare_custom_heap!(2 * 1024 * 1024);
29
30const SERVER_TOKEN: Token = Token(1);
31const WAKER_TOKEN: Token = Token(2);
32const HTTP_PORT: u16 = 8088;
33
34#[cfg(not(test))]
35ostd::cell_main!(cell_main);
36
37fn cell_main() {
38    println("[tokio-smoke] Starting Tokio-style I/O Driver on CellOS (vimio)...");
39
40    let selector = match Selector::new() {
41        Ok(s) => s,
42        Err(_) => {
43            println("[tokio-smoke] Failed to create Selector!");
44            ostd::syscall::sys_exit(1);
45        }
46    };
47
48    println("[tokio-smoke] Selector created successfully.");
49
50    // Create Tokio-style Waker
51    let waker = match Waker::new(&selector, WAKER_TOKEN) {
52        Ok(w) => w,
53        Err(_) => {
54            println("[tokio-smoke] Failed to create Waker!");
55            ostd::syscall::sys_exit(1);
56        }
57    };
58
59    println("[tokio-smoke] Waker bound with Token(2). Testing wake() invocation...");
60    if waker.wake().is_ok() {
61        let mut events = Events::with_capacity(8);
62        selector.select(&mut events, Some(Duration::from_millis(500))).ok();
63
64        let woken = events.iter().any(|e| e.token == WAKER_TOKEN);
65        if woken {
66            println("[tokio-smoke] Waker event received by Selector PASS.");
67        } else {
68            println("[tokio-smoke] Waker event not received!");
69        }
70    }
71
72    // Connect to network service if online
73    if let Some(net_tid) = sys_lookup_service(api::syscall::service::NET) {
74        println(&alloc::format!("[tokio-smoke] Network service available at tid: {}", net_tid));
75
76        let req = api::ipc::NetRequest::TcpListen { port: HTTP_PORT };
77        let mut req_buf = [0u8; 128];
78        let mut resp_buf = [0u8; 128];
79
80        if let Ok(encoded) = api::ipc::encode(&req, &mut req_buf) {
81            let _ = ostd::syscall::sys_send(net_tid, encoded);
82            let _ = ostd::syscall::sys_recv(net_tid, &mut resp_buf);
83
84            if let Ok(api::ipc::NetResponse::CapId(listener_cap)) = api::ipc::decode::<api::ipc::NetResponse<'_>>(&resp_buf) {
85                println(&alloc::format!("[tokio-smoke] Listening on :{} with cap_id={}", HTTP_PORT, listener_cap));
86                if selector.register(listener_cap, SERVER_TOKEN, Interest::READABLE).is_ok() {
87                    println("[tokio-smoke] Registered listener socket with Selector.");
88                }
89            }
90        }
91    } else {
92        println("[tokio-smoke] Network service offline; testing driver loop in standalone mode.");
93    }
94
95    println("[tokio-smoke] Running driver select() event loop (Tokio current_thread pattern)...");
96
97    let mut events = Events::with_capacity(16);
98    let mut ticks = 0;
99
100    while ticks < 3 {
101        ticks += 1;
102        match selector.select(&mut events, Some(Duration::from_millis(50))) {
103            Ok(()) => {
104                for event in events.iter() {
105                    println(&alloc::format!("[tokio-smoke] Dispatched Event: token={:?} readable={} writable={}",
106                        event.token, event.readable, event.writable));
107                }
108            }
109            Err(_) => {
110                println("[tokio-smoke] Selector select error");
111                break;
112            }
113        }
114    }
115
116    println("[tokio-smoke] PASS: Tokio current_thread I/O driver verification complete!");
117    ostd::syscall::sys_exit(0);
118}
