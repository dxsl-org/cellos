3use core::mem::MaybeUninit;

5pub(crate) fn try_init_array<T, E, const N: usize>(
6    mut initialize: impl FnMut(usize) -> Result<T, E>,
7) -> Result<[T; N], E> {
8    struct InitGuard<'a, T> {
9        storage: &'a mut [MaybeUninit<T>],
10        initialized: usize,
11    }
12
13    impl<T> Drop for InitGuard<'_, T> {
14        fn drop(&mut self) {
15            for value in &mut self.storage[..self.initialized] {
16                unsafe { value.assume_init_drop() };
17            }
18        }
19    }
20
21    let mut storage: [MaybeUninit<T>; N] = unsafe { MaybeUninit::uninit().assume_init() };
22    let mut guard = InitGuard {
23        storage: &mut storage,
24        initialized: 0,
25    };
26    for index in 0..N {
27        guard.storage[index].write(initialize(index)?);
28        guard.initialized += 1;
29    }
30    let array = unsafe { (guard.storage.as_ptr() as *const [T; N]).read() };
31    guard.initialized = 0;
32    Ok(array)
33}

35pub(crate) const TX_SLOTS: usize = 16;
36pub(crate) const RX_SLOTS: usize = 16;
37
38#[derive(Clone, Copy, Debug, Eq, PartialEq)]
39pub(crate) enum DmaSlot {
40    TxRing,
41    TxBuffer(usize),
42    RxRing,
43    RxBuffer(usize),
44}
45
46#[derive(Clone, Copy, Debug, Eq, PartialEq)]
47pub(crate) enum InitialDmaProgram {
48    TxRingBase(u64),
49    RxRingBase(u64),
50    RxBuffer { index: usize, iova: u64 },
51}
52
53#[derive(Clone, Copy)]
54pub(crate) struct DmaIovas {
55    pub(crate) tx_ring: u64,
56    pub(crate) rx_ring: u64,
57    pub(crate) tx_bufs: [u64; TX_SLOTS],
58    pub(crate) rx_bufs: [u64; RX_SLOTS],
59}
60
61pub(crate) fn for_each_initial_dma_program(
62    layout: &DmaIovas,
63    mut emit: impl FnMut(InitialDmaProgram),
64) {
65    emit(InitialDmaProgram::TxRingBase(layout.tx_ring));
66    emit(InitialDmaProgram::RxRingBase(layout.rx_ring));
67    for (index, &iova) in layout.rx_bufs.iter().enumerate() {
68        emit(InitialDmaProgram::RxBuffer { index, iova });
69    }
70}
71
72pub(crate) fn authorize_dma_layout<E>(
73    mut authorize: impl FnMut(DmaSlot) -> Result<u64, E>,
74) -> Result<DmaIovas, E> {
75    let tx_ring = authorize(DmaSlot::TxRing)?;
76    let mut tx_bufs = [0u64; TX_SLOTS];
77    for (index, slot) in tx_bufs.iter_mut().enumerate() {
78        *slot = authorize(DmaSlot::TxBuffer(index))?;
79    }
80    let rx_ring = authorize(DmaSlot::RxRing)?;
81    let mut rx_bufs = [0u64; RX_SLOTS];
82    for (index, slot) in rx_bufs.iter_mut().enumerate() {
83        *slot = authorize(DmaSlot::RxBuffer(index))?;
84    }
85    Ok(DmaIovas {
86        tx_ring,
87        rx_ring,
88        tx_bufs,
89        rx_bufs,
90    })
91}

93pub(crate) fn with_authorized_dma_layout<R, E, T>(
94    resources: R,
95    mut authorize: impl FnMut(&R, DmaSlot) -> Result<u64, E>,
96    initialize: impl FnOnce(R, DmaIovas) -> T,
97) -> Result<T, E> {
98    let layout = authorize_dma_layout(|slot| authorize(&resources, slot))?;
99    Ok(initialize(resources, layout))
