1//! Private Phase04 slice S3 driver. It links the same `FrontendHost` the GUI Terminal
2//! uses and drives a real `cellbox sh` through it: the shell controls its own GUI-kind
3//! session and this Cell is only its frontend (Phase 05 S5-3 annex option C; keyboard in
4//! via `submit_*` on the frontend attachment, bytes out via `drain_*` of the rings the
5//! shell relays to), printing `P04 PASS|FAIL <name>` observations. The host runner supplies
6//! the independent verdict from serial output. Not part of any ordinary image: init
7//! starts it only from the runner's private autoload plan, and the kernel carries its
8//! `-> /bin/cellbox` launch edge only under `phase04-driver-profile`.
9//!
10//! Secrets: the driver never reads, stores or prints any. Every typed command is a fixed
11//! script, the shells it launches get an empty environment, and a FAIL line carries only a
12//! status name or a short printable slice of the scripted (non-secret) transcript. The one
13//! token it types into `read -s` is a fixed fixture constant; that scenario's FAIL lines
14//! carry statuses and counts only, so the runner can require the token absent from the UART.

89/// Must match the private runner's generated file, including a UTF-8 sequence split
90/// over cat's 512-byte read boundary. Never decode command payloads for assertions.
91fn fixture_bytes(len: usize) -> Vec<u8> {
92    let mut bytes: Vec<u8> = (0..len).map(|i| ((i * 73 + 19) % 256) as u8).collect();
93    for (index, chunk) in bytes.chunks_mut(512).enumerate() {
94        let sentinel = format!("P04-CHUNK-{index:04}-NO-UART!");
95        chunk[8..8 + sentinel.len()].copy_from_slice(sentinel.as_bytes());
96    }
97    bytes[31] = 0;
98    bytes[32] = 0xff;
99    bytes[511..514].copy_from_slice(&[0xe2, 0x82, 0xac]);
100    bytes
101}

117fn check(name: &str, ok: bool, why: &str) {
118    CHECKS.fetch_add(1, Ordering::Relaxed);
119    if ok {
120        println(&format!("P04 PASS {name}"));
121    } else {
122        FAILURES.fetch_add(1, Ordering::Relaxed);
123        println(&format!("P04 FAIL {name} {}", clip(why)));
124    }
125}

127/// Hex encodes observed bytes solely as fixture evidence, never as a data fallback.
128fn evidence(name: &str, bytes: &[u8]) {
129    const HEX: &[u8; 16] = b"0123456789abcdef";
130    println(&format!("P04 BYTES {name} len={}", bytes.len()));
131    for (index, chunk) in bytes.chunks(128).enumerate() {
132        let mut hex = String::with_capacity(chunk.len() * 2);
133        for byte in chunk {
134            hex.push(HEX[(byte >> 4) as usize] as char);
135            hex.push(HEX[(byte & 15) as usize] as char);
136        }
137        println(&format!("P04 HEX {name} offset={} {hex}", index * 128));
138    }
139}

168/// Remove CSI/ESC sequences; everything else is kept byte-for-byte as text.
169fn strip_ansi(bytes: &[u8]) -> String {
170    let mut out = Vec::with_capacity(bytes.len());
171    let mut i = 0;
172    while i < bytes.len() {
173        if bytes[i] == 0x1b {
174            i += 1;
175            if i < bytes.len() && bytes[i] == b'[' {
176                i += 1;
177                while i < bytes.len() && !(0x40..=0x7e).contains(&bytes[i]) {
178                    i += 1;
179                }
180            }
181            i += 1;
182            continue;
183        }
184        out.push(bytes[i]);
185        i += 1;
186    }
187    String::from_utf8_lossy(&out).into_owned()
188}

196/// One frontend host (and the shell controlling its own session behind it), the
197/// transcripts the driver collected from it and the grid model of its stdout.

199    host: FrontendHost,
200    out: Vec<u8>,
201    err: Vec<u8>,
202    screen: Screen,
203}

205impl Term {
206    fn open(rows: u16, cols: u16) -> Result<Term, Status> {
207        let host = FrontendHost::open(rows, cols)?;
208        Ok(Term {
209            host,
210            out: Vec::new(),
211            err: Vec::new(),
212            screen: Screen::new(usize::from(cols), usize::from(rows)),
213        })
214    }
215
216    fn launch_shell(&mut self) -> Result<(), Status> {
217        // No directory grant: a granted cwd would make the shell path-sealed and the VFS
218        // refuses every path-addressed request of a sealed cell. The shell then creates
219        // its session for this frontend; input is accepted once it is attached.
220        self.host.launch(SHELL, &["sh"], &[])?;
221        self.host.wait_attached(60_000)
222    }
223
224    fn pump(&mut self) {
225        // The model follows the session size the host last set (resize/re-attach).
226        self.screen
227            .fit(usize::from(self.host.cols()), usize::from(self.host.rows()));
228        let mut buf = [0u8; 512];
229        for stderr in [false, true] {
230            for _ in 0..16 {
231                let read = if stderr {
232                    self.host.drain_stderr(&mut buf)
233                } else {
234                    self.host.drain_stdout(&mut buf)
235                };
236                match read {
237                    Ok(n) if n > 0 => {
238                        if !stderr {
239                            self.screen.feed(&buf[..n]);
240                        }
241                        let sink = if stderr { &mut self.err } else { &mut self.out };
242                        sink.extend_from_slice(&buf[..n]);
243                        if sink.len() > CAP {
244                            let cut = sink.len() - CAP / 2;
245                            sink.drain(..cut);
246                        }
247                    }
248                    _ => break,
249                }
250            }
251        }
252    }
253
254    fn at_prompt(&self) -> bool {
255        strip_ansi(&self.out).ends_with(PROMPT)
256    }
257
258    /// Wait until the transcript ends with the prompt. `Err` names why not.
259    fn wait_prompt(&mut self, ms: u64) -> Result<(), String> {
260        let end = now_ms() + ms;
261        let mut spins = 0u32;
262        loop {
263            self.pump();
264            if self.at_prompt() {
265                return Ok(());
266            }
267            spins += 1;
268            if spins.is_multiple_of(32) {
269                match self.host.poll_child() {
270                    Ok(ChildPoll::Exited) => {
271                        self.pump();
272                        if self.at_prompt() {
273                            return Ok(());
274                        }
275                        return Err(String::from("shell-exited"));
276                    }
277                    Ok(ChildPoll::Running) => {}
278                    Err(status) => return Err(format!("poll {status:?}")),
279                }
280            }
281            if now_ms() >= end {
282                let tail = strip_ansi(&self.out);
283                let tail: String = tail
284                    .chars()
285                    .rev()
286                    .take(80)
287                    .collect::<Vec<_>>()
288                    .into_iter()
289                    .rev()
290                    .collect();
291                return Err(format!("prompt-timeout tail={tail:?}"));
292            }
293            let _ = sys_set_timer(1);
294        }
295    }
296
297    /// Type `cmd` and press Enter without waiting (transcripts are reset first).
298    fn send(&mut self, cmd: &str) -> Result<(), String> {
299        self.pump();
300        self.out.clear();
301        self.err.clear();
302        self.host
303            .submit_text(cmd)
304            .map_err(|s| format!("submit_text {s:?}"))?;
305        self.host
306            .submit_key(ENTER, 0, 1)
307            .map_err(|s| format!("submit_key {s:?}"))
308    }
309
310    /// Output lines of the finished command: the echoed command line and the trailing
311    /// prompt are removed, ANSI sequences stripped.
312    fn result(&self, cmd: &str) -> Vec<String> {
313        let mut lines = lines_of(&strip_ansi(&self.out));
314        if lines
315            .last()
316            .map(|l| l.starts_with(PROMPT.trim_end()))
317            .unwrap_or(false)
318        {
319            lines.pop();
320        }
321        if lines.first().map(String::as_str) == Some(cmd) {
322            lines.remove(0);
323        }
324        lines
325    }
326
327    /// Remove only the exact editor echo and prompt, leaving arbitrary data intact.
328    fn raw_result(&self, cmd: &str) -> Option<&[u8]> {
329        self.out
330            .strip_prefix(cmd.as_bytes())?
331            .strip_prefix(b"\n")?
332            .strip_suffix(PROMPT.as_bytes())
333    }
334
335    fn run(&mut self, cmd: &str, ms: u64) -> Result<Vec<String>, String> {
336        self.send(cmd)?;
337        self.wait_prompt(ms)?;
338        // Diagnostics of the scripted command (never secrets: fixed scripts only).
339        let diagnostic = self.stderr_text();
340        if !diagnostic.trim().is_empty() {
341            println(&format!(
342                "P04 INFO stderr cmd={} text={}",
343                clip(cmd),
344                clip(diagnostic.trim())
345            ));
346        }
347        Ok(self.result(cmd))
348    }
349
350    fn stderr_text(&self) -> String {
351        strip_ansi(&self.err)
352    }
353}

374fn cell_main() {
375    println("P04 START");
376    let mut a = match Term::open(24, 80) {
377        Ok(term) => term,
378        Err(status) => {
379            check("shell-prompt", false, &format!("open {status:?}"));
380            finish();
381        }
382    };
383    let launched = a.launch_shell();
384    let prompt = match launched {
385        Ok(()) => a.wait_prompt(60_000),
386        Err(status) => Err(format!("launch {status:?}")),
387    };
388    check(
389        "shell-prompt",
390        prompt.is_ok(),
391        prompt.as_ref().err().map(String::as_str).unwrap_or(""),
392    );
393    // Phase 09 lane (`scripts/qemu-phase09-ssh-test.sh`): only the ssh checks run.
394    #[cfg(all(feature = "phase09-ssh", not(feature = "kssh")))]
395    if prompt.is_ok() {
396        ssh::ssh_checks(&mut a);
397    }
398    #[cfg(feature = "kssh")]
399    if prompt.is_ok() {
400        ssh::kssh::kssh_checks(&mut a);
401    }
402    #[cfg(not(feature = "phase09-ssh"))]
403    if prompt.is_ok() {
404        builtins(&mut a);
405        typed_pipeline(&mut a);
406        file_create_ls(&mut a);
407        externals(&mut a);
408        soft_reset(&mut a);
409        stock_backpressure(&mut a);
410        stock_negative_grants(&mut a);
411        exit_status(&mut a);
412        rg::rg_checks(&mut a);
413        rg::rg_fixture_checks(&mut a);
414        edit::edit_checks(&mut a);
415        ctrl_c(&mut a);
416        endless_local_sink(&mut a);
417        secret_namespace(&mut a);
418        secret_read(&mut a);
419        hostile_output(&mut a);
420        paste_injection(&mut a);
421        net_rights(&mut a);
422        config_rights(&mut a);
423        sandbox_rights(&mut a);
424        // Phase 10 S10-0: `/bin/cellbox <command>` launched directly, and the in-shell
425        // explicit form.
426        oneshot::oneshot_checks(&mut a);
427        resize(&mut a);
428        frontend(&mut a);

435        repeat_open_close(&mut a);

437    }

440    finish();
441}

443fn finish() -> ! {
444    let failures = FAILURES.load(Ordering::Relaxed);
445    let checks = CHECKS.load(Ordering::Relaxed);
446    if failures == 0 {
447        println(&format!("P04-TERMINAL: PASS checks={checks}"));
448    } else {
449        println(&format!(
450            "P04-TERMINAL: FAIL failures={failures} checks={checks}"
451        ));
452    }
453    sys_exit(if failures == 0 { 0 } else { 1 });
454}

457    let echo = a.run("echo p04-a'b'c", 30_000);
458    check(
459        "builtin-echo",

470

474            .find(|x| x.starts_with("Mem (KiB):"))

491        "builtin-date-changes",

497    let has = |l: &Vec<String>, name: &str| {

499            .iter()
500            .any(|r| r.contains(&format!("\"name\":\"{name}\"")))

503        "builtin-ps-records",

507    // C5a: this image boots without `ui`; the data commands above ran with no UI task
508    // anywhere in the system.
509    let ui = ["compositor", "desktop", "fb-console"];
510    check(
511        "headless-no-ui-tasks",
512        matches!(&ps, Ok(l) if !rows_of(l).is_empty() && !ui.iter().any(|name| has(l, name))),
513        &format!("{:?}", ps.as_ref().map(|l| rows_of(l).len())),
514    );
515}

557/// Contents of the private fixture `/srv/cellos/p04-secret` in the boot ramdisk (VIFS1).
558/// The VFS reserves `/srv/cellos/**` to the live KMS service, so no shell read path,
559/// including the kernel BootFS fallback, may ever deliver these bytes.
560const SECRET_TOKEN: &str = "p04-kms-namespace-fixture-token";
561
562/// Reads, copies, listings and writes of the KMS namespace from the shell are refused
563/// (nonzero status) and never deliver the fixture bytes; a readable ramdisk file still
564/// reads through the same commands.
565fn secret_namespace(a: &mut Term) {
566    const DENIED: [&str; 9] = [
567        "cat /srv/cellos/p04-secret",
568        "grep p04 /srv/cellos/p04-secret",
569        "head -n 1 /srv/cellos/p04-secret",
570        "wc /srv/cellos/p04-secret",
571        "sort /srv/cellos/p04-secret",
572        "from jsonl < /srv/cellos/p04-secret | to jsonl",
573        "cp /srv/cellos/p04-secret /tmp/p04-leak",
574        "ls /srv/cellos",
575        "echo p04-w > /srv/cellos/p04-w",
576    ];
577    let mut leaked = false;
578    for (index, cmd) in DENIED.iter().enumerate() {
579        let run = a.run(cmd, 30_000);
580        let shown =
581            strip_ansi(&a.out).contains(SECRET_TOKEN) || a.stderr_text().contains(SECRET_TOKEN);
582        leaked |= shown;
583        let st = a.run("echo sk=$?", 30_000);
584        let refused =
585            matches!(&st, Ok(l) if l.len() == 1 && l[0].starts_with("sk=") && l[0] != "sk=0");
586        check(
587            &format!("secret-denied-{index}"),
588            run.is_ok() && refused && !shown,
589            &format!(
590                "cmd={cmd} run={:?} st={st:?} shown={shown}",
591                run.map(|l| l.len())
592            ),
593        );
594    }
595    // A copy that was refused left nothing behind to read.
596    let leak = a.run("cat /tmp/p04-leak", 30_000);
597    let copied = strip_ansi(&a.out).contains(SECRET_TOKEN);
598    leaked |= copied;
599    check("secret-no-copy", !copied, &format!("{leak:?}"));
600    check(
601        "secret-never-shown",
602        !leaked,
603        "fixture bytes reached a shell stream",
604    );
605
606    // Positive control: the ramdisk fallback still serves a policy-readable file.
607    let wc = a.run("wc /p04-list/alpha-entry", 30_000);
608    let st = a.run("echo sw=$?", 30_000);
609    check(
610        "ramdisk-read-still-works",
611        matches!(&wc, Ok(l) if !l.is_empty())
612            && matches!(&st, Ok(l) if l == &[String::from("sw=0")]),
613        &format!("{wc:?} {st:?}"),
614    );
615}

617/// S5-4a item 1: a fixed fixture token (not a real secret) typed into `read -s` through

635/// `read -s` on the shell's own GUI-kind session: the shell takes its echo-off lease, the
636/// typed token is never echoed (the exact stdout is the command echo, the newline the reader
637/// prints on Enter and the prompt), `${#SECRET}` shows only its length, `history` shows only
638/// the typed command lines, and no drained byte of the scenario contains the token.
639fn secret_read(a: &mut Term) {
640    let token = READ_SECRET_TOKEN.as_bytes();
641    let mut seen: Vec<u8> = Vec::new();
642    let cleared = a.run("history -c", 30_000);
643    seen.extend_from_slice(&a.out);
644    seen.extend_from_slice(&a.err);
645    let cmd = "read -s SECRET";
646    let sent = a.send(cmd);
647    // Typed text is read only after the command line was submitted; give the shell time to
648    // take its echo-off lease first (keys queued before it would still reach the silent
649    // reader: the line editor is not reading while a command runs).
650    sleep_ms(1500);
651    let typed = a
652        .host
653        .submit_text(READ_SECRET_TOKEN)
654        .and_then(|_| a.host.submit_key(ENTER, 0, 1));
655    let back = a.wait_prompt(30_000).is_ok();
656    seen.extend_from_slice(&a.out);
657    seen.extend_from_slice(&a.err);
658    let silent = a.out == format!("{cmd}\n\n{PROMPT}").as_bytes() && a.err.is_empty();
659    check(
660        "secret-read-silent",
661        cleared.is_ok()
662            && sent.is_ok()
663            && typed.is_ok()
664            && back
665            && silent
666            && occurrences(&seen, token) == 0,
667        &format!(
668            "cleared={} sent={} typed={} back={back} silent={silent} out_len={} err_len={}",
669            cleared.is_ok(),
670            sent.is_ok(),
671            status(&typed),
672            a.out.len(),
673            a.err.len()
674        ),
675    );
676
677    let length = a.run("echo ${#SECRET}", 30_000);
678    seen.extend_from_slice(&a.out);
679    seen.extend_from_slice(&a.err);
680    let want = format!("{}", token.len());
681    // Only digits are ever echoed into a FAIL line.
682    let shown: Vec<String> = length
683        .as_ref()
684        .map(|lines| {
685            lines
686                .iter()
687                .map(|l| {
688                    if l.bytes().all(|b| b.is_ascii_digit()) {
689                        l.clone()
690                    } else {
691                        String::from("<non-numeric>")
692                    }
693                })
694                .collect()
695        })
696        .unwrap_or_default();
697    check(
698        "secret-read-length",
699        matches!(&length, Ok(lines) if lines.len() == 1 && lines[0] == want),
700        &format!("ok={} want={want} lines={shown:?}", length.is_ok()),
701    );
702
703    let history = a.run("history", 30_000);
704    seen.extend_from_slice(&a.out);
705    seen.extend_from_slice(&a.err);
706    let entries = history
707        .as_ref()
708        .ok()
709        .and_then(|lines| history_entries(lines));
710    let typed_only = ["read -s SECRET", "echo ${#SECRET}", "history"]
711        .map(String::from)
712        .to_vec();
713    check(
714        "secret-read-history",
715        entries.as_ref() == Some(&typed_only),
716        &format!(
717            "ok={} entries={:?}",
718            history.is_ok(),
719            entries.as_ref().map(Vec::len)
720        ),
721    );
722    let unset = a.run("unset SECRET", 30_000);
723    seen.extend_from_slice(&a.out);
724    seen.extend_from_slice(&a.err);
725    check(
726        "secret-read-never-drained",
727        unset.is_ok() && occurrences(&seen, token) == 0,
728        &format!("unset={} drained_bytes={}", unset.is_ok(), seen.len()),
729    );
730}

789/// One input-owner request on `op` built here (bypassing the SDK's own validation).
790fn raw_input(
791    op: ObjectRefV1,
792    selector: InputOp,
793    body: &[u8],
794    reply: &mut [u8],
795) -> Result<(Status, usize), Status> {
796    let owner = ostd::session::service_binding(api::syscall::service::INPUT)?;
797    let header = ostd::session::service_call(
798        owner,
799        api::syscall::service::INPUT,
800        op,
801        selector as u16,
802        true,
803        body,
804        reply,
805    )?;
806    Ok((header.status, header.total_len as usize))
807}

809/// PreparePaste / WritePaste / CommitPaste of `text` sent straight to the input owner, then
810/// AbortPaste. Returns the (write, commit, abort) statuses.
811fn raw_paste(
812    op: ObjectRefV1,
813    session: ObjectRefV1,
814    attachment: ObjectRefV1,
815    text: &[u8],
816) -> Result<(Status, Status, Status), Status> {
817    let mut body = [0u8; 48];
818    let mut w = Writer::new(&mut body);
819    w.reference(session)?;
820    w.reference(attachment)?;
821    w.u64(0)?;
822    w.u32(text.len() as u32)?;
823    w.u32(0)?;
824    let mut reply = [0u8; HEADER_LEN + 16];
825    let (prepared, n) = raw_input(op, InputOp::PreparePaste, &body, &mut reply)?;
826    if prepared != Status::Ok {
827        return Err(prepared);
828    }
829    let paste = ObjectRefV1::from_bytes(&reply[HEADER_LEN..n])?;
830    let mut body = [0u8; 600];
831    let mut w = Writer::new(&mut body);
832    w.reference(paste)?;
833    w.u32(0)?;
834    w.u32(text.len() as u32)?;
835    w.bytes(text)?;
836    let len = w.position();
837    let (written, _) = raw_input(
838        op,
839        InputOp::WritePaste,
840        &body[..len],
841        &mut [0u8; HEADER_LEN + 8],
842    )?;
843    let (committed, _) = raw_input(
844        op,
845        InputOp::CommitPaste,
846        &paste.to_bytes(),
847        &mut [0u8; HEADER_LEN + 8],
848    )?;
849    let (aborted, _) = raw_input(
850        op,
851        InputOp::AbortPaste,
852        &paste.to_bytes(),
853        &mut [0u8; HEADER_LEN],
854    )?;
855    Ok((written, committed, aborted))
856}

1029/// The `cellos-vt1` soft reset cellbox owes after such a child.
1030const SOFT_RESET: &[u8] = b"\x1b[?1049l\x1b[?25h\x1b[?2004l\x1b[0m";
1031/// `/bin/p03-filter rows` output.
1032const ROWS_JSONL: &[u8] = b"{\"a\":1}\n{\"a\":2}\n";

1034fn occurrences(haystack: &[u8], needle: &[u8]) -> usize {
1035    haystack
1036        .windows(needle.len())
1037        .filter(|window| *window == needle)
1038        .count()
1039}

1041/// No soft reset on either stream of the last command.
1042fn no_reset(a: &Term) -> bool {
1043    occurrences(&a.out, SOFT_RESET) == 0 && occurrences(&a.err, SOFT_RESET) == 0
1044}

1046/// S5-2 foreground-loss reset: the shell follows the screen modes of the child bytes it
1047/// relays. A child that exits with the alternate screen, a hidden cursor and bracketed
1048/// paste on is followed by exactly one soft reset on the shell's stdout, after the child's
1049/// bytes and before the next prompt, as a single command and as a pipeline stage; builtin
1050/// output, a stock external and a plain fixture mode are followed by none.
1051fn soft_reset(a: &mut Term) {
1052    let expected = [TUI_ABANDON, SOFT_RESET].concat();
1053    for (name, cmd, status_var) in [
1054        ("tui-abandon", "/bin/p03-filter tui-abandon", "sttui"),
1055        (
1056            "tui-pipeline",
1057            "/bin/p03-filter tui-abandon | /bin/p03-filter cat",
1058            "sttuipipe",
1059        ),
1060    ] {
1061        let run = a.run(cmd, 40_000);
1062        let once = occurrences(&a.out, SOFT_RESET) == 1 && occurrences(&a.err, SOFT_RESET) == 0;
1063        let exact =
1064            run.is_ok() && a.raw_result(cmd) == Some(expected.as_slice()) && a.err.is_empty();
1065        stream_evidence(name, a, cmd);
1066        let st = a.run(&format!("echo {status_var}=$?"), 30_000);
1067        let after = no_reset(a);
1068        check(
1069            &format!("soft-reset-after-{name}"),
1070            exact && once && after && matches!(&st, Ok(l) if l == &[format!("{status_var}=0")]),
1071            &format!("run={run:?} exact={exact} once={once} next-clean={after} st={st:?}"),
1072        );
1073    }
1074
1075    let plain: [(&str, &str, &[u8]); 3] = [
1076        ("builtin-echo", "echo p04-plain-reset", b"p04-plain-reset\n"),
1077        (
1078            "external-echo",
1079            "/bin/echo p04-plain-reset",
1080            b"p04-plain-reset\n",
1081        ),
1082        ("filter", "/bin/p03-filter rows", ROWS_JSONL),
1083    ];
1084    for (name, cmd, output) in plain {
1085        let run = a.run(cmd, 40_000);
1086        let exact = run.is_ok() && a.raw_result(cmd) == Some(output) && a.err.is_empty();
1087        let clean = no_reset(a);
1088        if name == "filter" {
1089            stream_evidence("plain-filter", a, cmd);
1090        }
1091        check(
1092            &format!("soft-reset-none-{name}"),
1093            exact && clean,
1094            &format!("run={run:?} exact={exact} clean={clean}"),
1095        );
1096    }
1097}

1099/// Withhold host reads so the real stock producer fills the pipeline and host
1100/// rings; resume with an odd-sized read, then compare every raw byte.
1101fn stock_backpressure(a: &mut Term) {
1102    let cmd = format!("/bin/cat {LARGE_PATH}");
1103    let sent = a.send(&cmd);
1104    sleep_ms(1500);
1105    let mut first = [0u8; 137];
1106    let read = a.host.drain_stdout(&mut first);
1107    if let Ok(n) = read {
1108        a.out.extend_from_slice(&first[..n]);
1109    }
1110    let resumed = a.wait_prompt(90_000);
1111    let expected = fixture_bytes(LARGE_LEN);
1112    let exact =
1113        resumed.is_ok() && a.raw_result(&cmd) == Some(expected.as_slice()) && a.err.is_empty();
1114    stream_evidence("large-cat", a, &cmd);
1115    let st = a.run("echo stlarge=$?", 30_000);
1116    check(
1117        "stock-full-ring-drain-exact",
1118        sent.is_ok()
1119            && matches!(read, Ok(137))
1120            && exact
1121            && matches!(&st, Ok(l) if l == &[String::from("stlarge=0")]),
1122        "withheld drain/resume: every binary byte once and successful stock exit",
1123    );
1124
1125    // ls emits 60-byte names plus separate newline writes, deliberately not a
1126    // divisor of the 4096-byte ring. Withheld drains force its helper to retain
1127    // the remainder of a partially accepted name rather than replay that name.
1128    let names: Vec<String> = (0..256)
1129        .map(|i| format!("p04-pressure-{i:03}-{}", "x".repeat(43)))
1130        .collect();
1131    let mut listing = Vec::from(b".\n..\n".as_slice());
1132    listing.extend(
1133        names
1134            .iter()
1135            .flat_map(|name| name.bytes().chain(core::iter::once(b'\n'))),
1136    );
1137    let ls_cmd = "/bin/ls /p04-pressure";
1138    let sent = a.send(ls_cmd);
1139    sleep_ms(1500);
1140    let back = a.wait_prompt(60_000);
1141    let exact = a.raw_result(ls_cmd) == Some(listing.as_slice()) && a.err.is_empty();
1142    stream_evidence("pressure-ls", a, ls_cmd);
1143    let st = a.run("echo stpressure=$?", 30_000);
1144    check(
1145        "stock-fragmented-ring-drain-exact",
1146        sent.is_ok()
1147            && back.is_ok()
1148            && exact
1149            && matches!(&st, Ok(l) if l == &[String::from("stpressure=0")]),
1150        "stock helper non-aligned fragments across full ring without replay",
1151    );
1152
1153    // A real external downstream closes without reading the stock cat's output.
1154    let close_start = now_ms();
1155    let broken = a.run(
1156        &format!("/bin/cat {LARGE_PATH} | /bin/p03-filter exit 0"),
1157        15_000,

1222/// Query the canonical terminal record of a directly prepared child; do not
1223/// confuse a publication refusal or owner cancellation with a helper exit.
1224fn direct_child_exit(op: Operation, child: ChildRefV1) -> Result<i32, Status> {
1225    let end = now_ms() + 60_000;
1226    loop {
1227        let mut reply = [0u8; HEADER_LEN + 24];
1228        let h = ostd::session::kernel_call(
1229            KernelOp::QueryChild,
1230            op.reference().raw(),
1231            false,
1232            &child.to_bytes(),
1233            &mut reply,
1234        )?;
1235        if h.status != Status::Ok {
1236            return Err(h.status);
1237        }
1238        let mut body = Reader::new(&reply[HEADER_LEN..h.total_len as usize]);
1239        let state = body.u32()?;
1240        let reason = body.u32()?;
1241        let code = body.i32()?;
1242        if body.u32()? != 0 {
1243            return Err(Status::Malformed);
1244        }
1245        let _sequence = body.u64()?;
1246        body.finish()?;
1247        if state == 3 {
1248            return if reason == 0 {
1249                Ok(code)
1250            } else {
1251                Err(Status::Cancelled)
1252            };
1253        }
1254        if now_ms() >= end {
1255            return Err(Status::TimedOut);
1256        }
1257        let _ = sys_set_timer(1);
1258    }
1259}

1261fn drain_endpoint(endpoint: Endpoint) -> Vec<u8> {
1262    let mut bytes = Vec::new();
1263    let mut chunk = [0u8; 512];

1266            Ok(n) => bytes.extend_from_slice(&chunk[..n]),
1267            _ => break,
1268        }
1269    }
1270    bytes

1273/// The amendment admits only this stock /bin/echo Path edge in private images.
1274/// Missing slots reach a real committed child; forged refs must be refused by
1275/// admission, a different observation which never claims helper execution.
1276fn stock_negative_grants(a: &mut Term) {
1277    a.pump();
1278    let baseline = (a.out.len(), a.err.len());
1279    for (name, stdout, stderr, invalid) in [
1280        ("negative-missing-stdout", false, true, false),
1281        ("negative-missing-stderr", true, false, false),
1282        ("negative-missing-both", false, false, false),
1283        ("negative-invalid-stdout", true, true, true),
1284    ] {
1285        let attempt = (|| -> Result<(), Status> {
1286            let op = Operation::create(None)?;
1287            let (out_read, out_write) = op.create_stream(64)?;
1288            let (err_read, err_write) = op.create_stream(64)?;
1289            let payload = format!("p04-forbidden-{name}");
1290            let grants_count = usize::from(stdout) + usize::from(stderr);
1291            let context_len = 48 + 7 + payload.len() + 1 + grants_count * 40;
1292            let outcome = (|| -> Result<(), Status> {
1293                let mut prepared = op.prepare_child("/bin/echo", context_len)?;
1294                let mut grants = Vec::new();
1295                for (slot, enabled, source) in [
1296                    (GrantSlot::Stdout, stdout, out_write),
1297                    (GrantSlot::Stderr, stderr, err_write),
1298                ] {
1299                    if enabled {
1300                        let reference = if invalid && slot == GrantSlot::Stdout {
1301                            ObjectRefV1 {
1302                                id: u64::MAX - 17,
1303                                generation: 1,
1304                            }
1305                        } else {
1306                            source
1307                                .delegate(&prepared, EndpointRights::WRITE_ONLY)?
1308                                .reference()
1309                                .raw()
1310                        };
1311                        grants.push(Grant::endpoint(
1312                            slot,
1313                            reference,
1314                            EndpointRights::WRITE_ONLY.bits() as u16,
1315                        )?);
1316                    }
1317                }
1318                let context =
1319                    ChildContext::new(op.reference().raw(), &[&payload], b"", &[], &grants)?;
1320                let published = prepared
1321                    .write_context(&context)
1322                    .and_then(|_| prepared.commit());
1323                match published {
1324                    Ok(child) => {
1325                        let result = direct_child_exit(op, child);
1326                        println(&format!(
1327                            "P04 INFO negative-child name={name} exit={result:?} published=true"
1328                        ));
1329                        if invalid {
1330                            Err(Status::Malformed)
1331                        } else {
1332                            match result {
1333                                Ok(code) if code != 0 => Ok(()),
1334                                Ok(_) => Err(Status::Malformed),
1335                                Err(s) => Err(s),
1336                            }
1337                        }
1338                    }
1339                    Err(s)
1340                        if invalid
1341                            && matches!(
1342                                s,
1343                                Status::Denied
1344                                    | Status::Stale
1345                                    | Status::Revoked
1346                                    | Status::NotFound
1347                                    | Status::WrongOwner
1348                                    | Status::WrongSession
1349                                    | Status::Malformed
1350                            ) =>
1351                    {
1352                        println(&format!(
1353                            "P04 INFO negative-invalid-ref admission={s:?} helper-executed=false"
1354                        ));
1355                        Ok(())
1356                    }
1357                    Err(s) => Err(s),
1358                }
1359            })();
1360            let _ = (out_write.close(), err_write.close());
1361            let observed_out = drain_endpoint(out_read);
1362            let observed_err = drain_endpoint(err_read);
1363            evidence(&format!("{name}.stdout"), &observed_out);
1364            evidence(&format!("{name}.stderr"), &observed_err);
1365            let quiet = observed_out.is_empty() && observed_err.is_empty();
1366            let _ = op.cancel(2);
1367            let _ = (out_read.close(), err_read.close(), op.close());
1368            if !quiet {
1369                return Err(Status::Malformed);
1370            }
1371            outcome
1372        })();
1373        check(name, attempt.is_ok(), &status(&attempt));
1374    }
1375    a.pump();
1376    check(
1377        "negative-grants-host-isolation",
1378        baseline == (a.out.len(), a.err.len()),
1379        "direct negative child contaminated unrelated host streams",
1380    );
1381}

1430/// the host streams, so the driver only frames the run between two INFO lines and checks

1437        let prefix = format!("NETPROBE {op} ");
1438        let seen: Vec<&String> = lines.iter().filter(|l| l.starts_with(&prefix)).collect();

1440            &format!("net-denied-{op}"),
1441            seen.len() == 1 && &seen[0][prefix.len()..] == "denied",
1442            &format!("{seen:?} run={:?}", probe.as_ref().map(Vec::len)),
1443        );
1444    }
1445    let st = a.run("echo stnp=$?", 30_000);
1446    let probe_lines: Vec<&String> = lines
1447        .iter()
1448        .filter(|l| l.starts_with("NETPROBE "))
1449        .collect();

1451        "net-probe-done",
1452        probe_lines.len() == NET_PROBE_OPS.len() + 1
1453            && probe_lines
1454                .last()

1489        "net-curl-admitted",
1490        curl.is_ok()
1491            && matches!(&st, Ok(l) if l.len() == 1 && l[0].starts_with("stcurl="))
1492            && !streams.contains("HTTP/")
1493            && !refused,
1494        &format!("{curl:?} st={st:?} streams={}", streams.trim()),

1811/// An endless typed producer with a LOCAL terminal sink (no external tail): rows must
1812/// stream to the terminal while the pipeline runs, Ctrl-C must return the prompt promptly
1813/// with status 130, the shell must survive, and the cell heap must never be exhausted.
1814fn endless_local_sink(a: &mut Term) {
1815    const CMD: &str = "top --records --frames 100000 | to jsonl";
1816    let mut oom = false;
1817    let sent = a.send(CMD);
1818    let mut seen = false;
1819    let mut early_prompt = false;
1820    let end = now_ms() + 90_000;
1821    while sent.is_ok() && now_ms() < end {
1822        a.pump();
1823        oom |= oom_text(a);
1824        if strip_ansi(&a.out).contains("{\"") {
1825            seen = true;
1826            break;
1827        }
1828        if a.at_prompt() {
1829            early_prompt = true;
1830            break;
1831        }
1832        let _ = sys_set_timer(1);
1833    }
1834    // Rows keep arriving while the pipeline is still running: discard what was seen and
1835    // require fresh rows (and no prompt) after a pause.
1836    let mut more = false;
1837    if seen {
1838        a.pump();
1839        a.out.clear();
1840        let until = now_ms() + 1_500;
1841        while now_ms() < until {
1842            a.pump();
1843            oom |= oom_text(a);
1844            let _ = sys_set_timer(1);
1845        }
1846        a.pump();
1847        oom |= oom_text(a);
1848        more = strip_ansi(&a.out).contains("{\"") && !a.at_prompt();
1849    }

1851        "endless-local-sink-streams-live",
1852        sent.is_ok() && seen && more,
1853        &format!("sent={sent:?} seen={seen} more={more} early_prompt={early_prompt}"),
1854    );
1855
1856    // Same key path as the other Ctrl-C scenarios.
1857    let started = now_ms();
1858    let key = a.host.submit_key(0x63, CTRL, 1);
1859    let back = a.wait_prompt(15_000);
1860    let elapsed = now_ms().saturating_sub(started);
1861    println(&format!("P04 INFO endless-local-sink elapsed_ms={elapsed}"));
1862    oom |= oom_text(a);
1863    let st = a.run("echo ccsink=$?", 30_000);
1864    oom |= oom_text(a);
1865    check(
1866        "endless-local-sink-ctrl-c-130",
1867        key.is_ok()
1868            && back.is_ok()
1869            && elapsed < 15_000
1870            && matches!(&st, Ok(l) if l == &[String::from("ccsink=130")]),
1871        &format!(
1872            "key={} back={back:?} elapsed={elapsed} st={st:?}",
1873            status(&key)
1874        ),
1875    );
1876    let alive = a.run("echo alive", 30_000);
1877    oom |= oom_text(a);
1878    check(
1879        "endless-local-sink-shell-alive",
1880        matches!(&alive, Ok(l) if l == &[String::from("alive")]),

1882    );
1883    check(
1884        "endless-local-sink-no-oom",
1885        !oom,
1886        "OOM/heap exhausted text on the terminal streams",
1887    );

1922/// Keep pumping until `term`'s stdout has been quiet for a moment, so a frame drawn in
1923/// several ring chunks is complete in the grid model (at most 3 s).
1924fn settle(term: &mut Term) {
1925    const QUIET_MS: u64 = 250;
1926    let end = now_ms() + 3_000;
1927    let mut seen = term.screen.bytes;
1928    let mut quiet_since = now_ms();
1929    while now_ms() < end {
1930        term.pump();
1931        if term.screen.bytes != seen {
1932            seen = term.screen.bytes;
1933            quiet_since = now_ms();
1934        } else if now_ms().saturating_sub(quiet_since) >= QUIET_MS {
1935            return;
1936        }
1937        let _ = sys_set_timer(1);
1938    }
1939}

1941/// Pump `term` until `pred` yields a value on its grid model, re-checked on the settled
1942/// frame. `Err` carries a structure-only description of the screen.
1943fn wait_screen<T>(
1944    term: &mut Term,
1945    ms: u64,
1946    what: &str,
1947    mut pred: impl FnMut(&Screen) -> Option<T>,
1948) -> Result<T, String> {
1949    let end = now_ms() + ms;
1950    loop {
1951        term.pump();
1952        if pred(&term.screen).is_some() {
1953            settle(term);
1954            if let Some(value) = pred(&term.screen) {
1955                return Ok(value);
1956            }
1957        }
1958        if now_ms() >= end {
1959            return Err(format!("no {what}: {}", term.screen.summary()));
1960        }
1961        let _ = sys_set_timer(1);
1962    }
1963}

1975fn raw_submit_text(

1977    session: ObjectRefV1,
1978    attachment: ObjectRefV1,
1979    text: &[u8],

1985    w.reference(attachment)?;

1988    w.u16(0)?;
1989    w.u32((24 + text.len()) as u32)?;
1990    w.u64(0)?;
1991    w.u64(0)?;
1992    w.u16(2)?;
1993    w.u16(text.len() as u16)?;

2393/// `format_bytes` text (`512B`, `1.5K`, `12.0M`) as comparable tenths of bytes.
2394fn bytes_value(text: &str) -> Option<u128> {
2395    let shift = match text.chars().last()? {
2396        'B' => 0,
2397        'K' => 10,
2398        'M' => 20,
2399        'G' => 30,
2400        'T' => 40,
2401        'P' => 50,
2402        'E' => 60,
2403        _ => return None,
2404    };
2405    let number = &text[..text.len() - 1];
2406    let tenths = match number.split_once('.') {
2407        Some((whole, tenth)) if tenth.len() == 1 => {
2408            whole.parse::<u128>().ok()? * 10 + tenth.parse::<u128>().ok()?
2409        }
2410        None if shift == 0 => number.parse::<u128>().ok()? * 10,
2411        _ => return None,
2412    };
2413    Some(tenths << shift)
2414}

2416/// An Ok CPU% cell (`12.3`) as tenths; markers (`new`, `reset`, `n/a`) are `None`.
2417fn tenths_value(text: &str) -> Option<u64> {
2418    let (whole, tenth) = text.split_once('.')?;
2419    if whole.is_empty() || tenth.len() != 1 || !whole.bytes().all(|b| b.is_ascii_digit()) {
2420        return None;
2421    }
2422    Some(whole.parse::<u64>().ok()? * 10 + tenth.parse::<u64>().ok()?)
2423}

2680/// §3.5 keys on the Cells view, judged on settled frames: `o` sorts by the next shown
2681/// column (QUOTA, descending); three more reach CELL (cell_id ascending); `r` reverses it
2682/// and `r` again restores it. On that stable order PageUp/Down/PageDown/Up put the
2683/// highlight on rows 0, 1, the last visible row and the one above it. `f`/Space/Enter on
2684/// the first field drops exactly one column, the same again restores it.
2685fn keys_walk(t: &mut Term) -> Result<String, String> {
2686    let rows = sort_key(t, "o")?;
2687    let quotas: Vec<u128> = rows
2688        .iter()
2689        .filter_map(|r| r.get(4).and_then(|v| bytes_value(v)))
2690        .collect();
2691    if quotas.len() < 2 || !quotas.windows(2).all(|pair| pair[0] >= pair[1]) {
2692        return Err(format!("o: QUOTA not descending {quotas:?}"));
2693    }
2694    sort_key(t, "o")?;
2695    sort_key(t, "o")?;
2696    let ascending = cell_ids(&sort_key(t, "o")?);
2697    let reversed = cell_ids(&sort_key(t, "r")?);
2698    let rows = sort_key(t, "r")?;
2699    let again = cell_ids(&rows);
2700    let strict = |ids: &Option<Vec<u64>>, up: bool| {
2701        ids.as_ref().is_some_and(|v| {
2702            v.len() >= 2
2703                && v.windows(2)
2704                    .all(|p| if up { p[0] < p[1] } else { p[0] > p[1] })
2705        })
2706    };
2707    if !strict(&ascending, true) || !strict(&reversed, false) || !strict(&again, true) {
2708        return Err(format!(
2709            "CELL order o={ascending:?} r={reversed:?} r={again:?}"
2710        ));
2711    }
2712    let n = rows.len();
2713    if n < 3 {
2714        return Err(format!("only {n} rows"));
2715    }
2716    nav(t, PAGE_UP, 0)?;
2717    nav(t, DOWN, 1)?;
2718    nav(t, PAGE_DOWN, n - 1)?;
2719    nav(t, UP, n - 2)?;
2720    let columns = t
2721        .screen
2722        .frame()
2723        .map(|f| f.header().len())
2724        .ok_or_else(|| String::from("no frame"))?;
2725    let fewer = toggle_first_field(t, columns)?;
2726    let restored = toggle_first_field(t, fewer)?;
2727    if fewer + 1 != columns || restored != columns {
2728        return Err(format!(
2729            "fields: {columns} -> {fewer} -> {restored} columns"
2730        ));
2731    }
2732    Ok(format!("rows={n} columns={columns}->{fewer}->{restored}"))
2733}

2800/// Host u starts `top`; t quits it (restore + prompt + status 0), compares its drill, then

2803fn top_sessions(t: &mut Term, drill: Result<(String, Vec<u64>), String>) {

2882/// While u idles in `top` for at least 5 s, t's `top -b -n 2 -d 2` shows u's Cell twice

2892        .as_ref()
2893        .map(|lines| {
2894            lines
2895                .iter()
2896                .map(|l| {
2897                    l.split_whitespace()
2898                        .map(String::from)
2899                        .collect::<Vec<String>>()
2900                })
2901                .filter(|r| r.first() == Some(&id))

2906        .last()

2920            "cell={id} rows={} cpu_tenths={cpu:?} threshold_tenths={IDLE_COST_MAX_TENTHS} (threshold is an implementation choice) run={:?}",

3056/// Poll until the shell child has exited (output stays readable through the pump).
3057fn wait_exit(term: &mut Term, ms: u64) -> Result<(), String> {
3058    let end = now_ms() + ms;
3059    loop {
3060        term.pump();
3061        match term.host.poll_child() {
3062            Ok(ChildPoll::Exited) => {
3063                term.pump();
3064                return Ok(());
3065            }
3066            Ok(ChildPoll::Running) => {}
3067            Err(status) => return Err(format!("poll {status:?}")),
3068        }
3069        if now_ms() >= end {
3070            return Err(String::from("exit-timeout"));
3071        }
3072        let _ = sys_set_timer(1);
3073    }
3074}
