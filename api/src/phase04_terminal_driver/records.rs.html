1//! Pure readers for the Phase 06 S6-5 `top-load` scenarios: `top --records … | to jsonl`
2//! envelope lines (`top.cell.v1`/`top.task.v1` rows, Begin `meta`, End), the fixture's
3//! `TOPLOAD` acknowledgement lines, the `s2tw-probe trunc` line, and the comparisons the
4//! scenarios make on them. Host-tested; nothing here touches a host or a syscall.
5//!
6//! The JSON reader only looks up a key of one envelope line. Every key it is used with is
7//! unique within its line (`"tasks":` never matches `"total_tasks":`: the quote is part of
8//! the key), and no value it reads contains an escaped quote.

14/// The raw value token of `"key":` in `json`: a string without its quotes, or a number,
15/// `true`, `false`, `null` (or an opening brace for a nested record).
16pub fn raw<'a>(json: &'a str, key: &str) -> Option<&'a str> {
17    let needle = format!("\"{key}\":");
18    let start = json.find(&needle)? + needle.len();
19    let rest = &json[start..];
20    if let Some(body) = rest.strip_prefix('"') {
21        return body.find('"').map(|end| &body[..end]);
22    }
23    let end = rest.find([',', '}', ']']).unwrap_or(rest.len());
24    Some(&rest[..end])
25}
26
27/// How many times `"key":` occurs in `json`.
28pub fn key_count(json: &str, key: &str) -> usize {
29    json.matches(&format!("\"{key}\":")).count()
30}
31
32pub fn uint(json: &str, key: &str) -> Option<u64> {
33    raw(json, key)?.parse().ok()
34}
35
36pub fn text<'a>(json: &'a str, key: &str) -> Option<&'a str> {
37    let needle = format!("\"{key}\":\"");
38    json.contains(&needle).then(|| raw(json, key)).flatten()
39}
40
41pub fn boolean(json: &str, key: &str) -> Option<bool> {
42    match raw(json, key)? {
43        "true" => Some(true),
44        "false" => Some(false),
45        _ => None,
46    }
47}

49/// One frame of an envelope stream: the Begin line, its row lines and the End line.
50#[derive(Clone, Debug, Default)]
51pub struct RecFrame {
52    pub begin: String,
53    pub rows: Vec<String>,
54    pub end: String,
55}
56
57impl RecFrame {
58    pub fn total_tasks(&self) -> Option<u64> {
59        uint(&self.begin, "total_tasks")
60    }
61
62    pub fn online_cpus(&self) -> Option<u64> {
63        uint(&self.begin, "online_cpus")
64    }
65
66    /// `gone_cells`/`gone_tasks` of the Begin `meta` (`None` for Null or absent).
67    pub fn gone(&self, key: &str) -> Option<u64> {
68        uint(&self.begin, key)
69    }
70
71    pub fn complete(&self) -> Option<bool> {
72        boolean(&self.end, "complete")
73    }
74
75    pub fn dropped(&self) -> Option<u64> {
76        uint(&self.end, "dropped")
77    }
78
79    pub fn cells(&self) -> Vec<CellRow> {
80        self.rows.iter().filter_map(|row| CellRow::parse(row)).collect()
81    }
82
83    pub fn tasks(&self) -> Vec<TaskRow> {
84        self.rows.iter().filter_map(|row| TaskRow::parse(row)).collect()
85    }
86
87    /// The Cell row of incarnation `id`.
88    pub fn cell(&self, id: Ident) -> Option<CellRow> {
89        self.cells().into_iter().find(|row| row.ident == id)
90    }
91}

93/// Split envelope lines into frames; every line between a Begin and its End must be a row
94/// and every frame must end. Lines that are not envelopes are ignored.
95pub fn frames(lines: &[String]) -> Result<Vec<RecFrame>, String> {
96    let mut out = Vec::new();
97    let mut open: Option<RecFrame> = None;
98    for line in lines.iter().filter(|l| l.starts_with("{\"frame\":")) {
99        match (raw(line, "frame"), open.as_mut()) {
100            (Some("begin"), None) => open = Some(RecFrame { begin: line.clone(), ..RecFrame::default() }),
101            (Some("row"), Some(frame)) => frame.rows.push(line.clone()),
102            (Some("end"), Some(frame)) => {
103                frame.end = line.clone();
104                out.extend(open.take());
105            }
106            (kind, _) => return Err(format!("envelope {kind:?} out of order after {} frames", out.len())),
107        }
108    }
109    match open {
110        Some(_) => Err(format!("unterminated frame after {} frames", out.len())),
111        None => Ok(out),
112    }
113}

115/// A Cell incarnation: `cell_id:generation`.
116#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
117pub struct Ident {
118    pub id: u64,
119    pub generation: u64,
120}
121
122impl Ident {
123    /// `ID:GEN` text.
124    pub fn parse(text: &str) -> Option<Ident> {
125        let (id, generation) = text.split_once(':')?;
126        Some(Ident { id: id.parse().ok()?, generation: generation.parse().ok()? })
127    }
128}
129
130impl core::fmt::Display for Ident {
131    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
132        write!(f, "{}:{}", self.id, self.generation)
133    }
134}

136/// The fields of a `top.cell.v1` row the scenarios compare.
137#[derive(Clone, Debug, PartialEq, Eq)]
138pub struct CellRow {
139    pub ident: Ident,
140    pub root_tid: u64,
141    pub name: String,
142    pub tasks: u64,
143    pub cpu_ns: Option<u64>,
144    pub tenths: Option<u64>,
145    pub quality: String,
146    pub quota: Option<u64>,
147    pub limit: Option<u64>,
148}

150impl CellRow {
151    pub fn parse(row: &str) -> Option<CellRow> {
152        if !row.starts_with("{\"frame\":\"row\"") || key_count(row, "root_tid") != 1 {
153            return None;
154        }
155        Some(CellRow {
156            ident: Ident { id: uint(row, "cell_id")?, generation: uint(row, "generation")? },
157            root_tid: uint(row, "root_tid")?,
158            name: String::from(text(row, "name")?),
159            tasks: uint(row, "tasks")?,
160            cpu_ns: uint(row, "cpu_ns"),
161            tenths: uint(row, "cpu_tenths"),
162            quality: String::from(text(row, "cpu_quality")?),
163            quota: uint(row, "quota_bytes"),
164            limit: uint(row, "limit_bytes"),
165        })
166    }
167}

169/// The fields of a `top.task.v1` row the scenarios compare.
170#[derive(Clone, Debug, PartialEq, Eq)]
171pub struct TaskRow {
172    pub tid: u64,
173    /// `None` for an ownerless kernel task.
174    pub owner: Option<Ident>,
175    pub cpu_ns: Option<u64>,
176    pub tenths: Option<u64>,

178    pub stack_bytes: u64,
179    pub stack_in_quota: bool,

182impl TaskRow {
183    pub fn parse(row: &str) -> Option<TaskRow> {
184        if !row.starts_with("{\"frame\":\"row\"") || key_count(row, "tid") != 1 {
185            return None;
186        }
187        let owner = match (uint(row, "cell_id"), uint(row, "generation")) {
188            (Some(id), Some(generation)) => Some(Ident { id, generation }),
189            _ => None,
190        };
191        Some(TaskRow {
192            tid: uint(row, "tid")?,
193            owner,
194            cpu_ns: uint(row, "cpu_ns"),
195            tenths: uint(row, "cpu_tenths"),
196            quality: String::from(text(row, "cpu_quality")?),
197            stack_bytes: uint(row, "stack_bytes")?,
198            stack_in_quota: boolean(row, "stack_in_quota")?,
199        })
200    }

203/// The last complete line of `transcript` that starts with `TOPLOAD ` (the fixture's
204/// acknowledgement), if any.
205pub fn last_ack(transcript: &str) -> Option<&str> {
206    let complete = &transcript[..transcript.rfind('\n')? + 1];
207    complete.lines().rev().map(|l| l.trim_end_matches('\r')).find(|l| l.starts_with("TOPLOAD "))
208}

215/// The `tids=a,b,…` list of an acknowledgement line (empty list for `tids=`).
216pub fn ack_tids(line: &str) -> Option<Vec<u64>> {
217    let list = line.split_whitespace().find_map(|word| word.strip_prefix("tids="))?;
218    if list.is_empty() {
219        return Some(Vec::new());
220    }
221    list.split(',').map(|tid| tid.parse().ok()).collect()
222}

224/// `(emitted_cells, total_cells, emitted_tasks, total_tasks, complete)` of the probe's
225/// `other:cells-E/T-tasks-e/U-complete-C` result.
226pub fn trunc_counts(result: &str) -> Option<(u64, u64, u64, u64, u64)> {
227    let rest = result.strip_prefix("other:cells-")?;
228    let (cells, rest) = rest.split_once("-tasks-")?;
229    let (tasks, complete) = rest.split_once("-complete-")?;
230    let pair = |text: &str| -> Option<(u64, u64)> {
231        let (a, b) = text.split_once('/')?;
232        Some((a.parse().ok()?, b.parse().ok()?))
233    };
234    let (e, t) = pair(cells)?;
235    let (et, u) = pair(tasks)?;
236    Some((e, t, et, u, complete.parse().ok()?))
237}

239/// Σ `cpu_tenths` over a Cells frame's rows that have one, and the bound
240/// `online_cpus × 1000 + rows` (no CPU billed twice; one tenth of rounding per row).
241pub fn cpu_sum(frame: &RecFrame) -> Option<(u64, u64)> {
242    let cells = frame.cells();
243    let sum = cells.iter().filter_map(|row| row.tenths).sum();
244    Some((sum, frame.online_cpus()? * 1000 + cells.len() as u64))
245}

247/// #12: a task stream across the exit of `tid`. Returns the index of the last frame that
248/// still lists it.
249pub fn exit_gone_tasks(frames: &[RecFrame], tid: u64) -> Result<usize, String> {
250    let listed: Vec<bool> = frames.iter().map(|f| f.tasks().iter().any(|t| t.tid == tid)).collect();
251    if listed.first() != Some(&true) {
252        return Err(format!("tid {tid} not in the first frame: listed={listed:?}"));
253    }
254    let gone = listed.iter().position(|&l| !l).ok_or_else(|| format!("tid {tid} never left: listed={listed:?}"))?;
255    if listed[gone..].iter().any(|&l| l) {
256        return Err(format!("tid {tid} came back: listed={listed:?}"));
257    }
258    let count = frames[gone].gone("gone_tasks");
259    let (before, after) = (frames[gone - 1].total_tasks(), frames[gone].total_tasks());
260    let exact = matches!((before, after), (Some(b), Some(a)) if a + 1 == b);
261    let ok = match count {
262        Some(n) if exact => n == 1,
263        Some(n) => n >= 1,
264        None => false,
265    };
266    if !ok {
267        return Err(format!("frame {gone} gone_tasks={count:?} total_tasks {before:?}->{after:?}"));
268    }
269    if let Some(reset) = frames.iter().flat_map(|f| f.tasks()).find(|t| t.quality == "reset") {
270        return Err(format!("tid {} reset", reset.tid));
271    }
272    Ok(gone - 1)
273}

275/// #13: a Cells stream across a thread exit of `cell`: its task count falls from `from` to
276/// `from - 1`, its `cpu_ns` never falls, its quality stays `ok`, and after the exit its
277/// `cpu_ns` is at least `floor`. Returns the `cpu_ns` series.
278pub fn exit_retains_cpu(frames: &[RecFrame], cell: Ident, from: u64, floor: u64) -> Result<Vec<u64>, String> {
279    let rows: Vec<CellRow> = frames.iter().map(|f| f.cell(cell)).collect::<Option<_>>().ok_or_else(|| format!("{cell} missing from a frame"))?;
280    let tasks: Vec<u64> = rows.iter().map(|r| r.tasks).collect();
281    let cpu: Vec<u64> = rows.iter().map(|r| r.cpu_ns).collect::<Option<_>>().ok_or_else(|| format!("{cell} cpu_ns Null"))?;
282    let qualities: Vec<&str> = rows.iter().map(|r| r.quality.as_str()).collect();
283    let why = || format!("tasks={tasks:?} cpu_ns={cpu:?} quality={qualities:?} floor={floor}");
284    let fell = tasks.first() == Some(&from)
285        && tasks.last() == Some(&(from - 1))
286        && tasks.windows(2).all(|p| p[1] <= p[0]);
287    let monotonic = cpu.windows(2).all(|p| p[1] >= p[0]);
288    let ok = qualities.iter().all(|&q| q == "ok");
289    let retained = rows.iter().zip(&cpu).filter(|(r, _)| r.tasks < from).all(|(_, &ns)| ns >= floor);
290    if fell && monotonic && ok && retained {
291        Ok(cpu)
292    } else {
293        Err(why())
294    }
295}

297/// #15: a Cells stream across a quit and relaunch of `old` as `new`: `old` listed, then a
298/// frame without it carrying `gone_cells ≥ 1`, then `new` first with quality `new`, then
299/// `new` with `ok`; never both in one frame and never `reset` on either.
300pub fn relaunch_sequence(frames: &[RecFrame], old: Ident, new: Ident) -> Result<String, String> {
301    let rows: Vec<(Option<CellRow>, Option<CellRow>)> = frames.iter().map(|f| (f.cell(old), f.cell(new))).collect();
302    let shape: Vec<String> = rows
303        .iter()
304        .zip(frames)
305        .map(|((o, n), f)| {
306            format!(
307                "{}{}g{}",
308                o.as_ref().map_or("-", |r| if r.quality == "reset" { "O!" } else { "O" }),
309                n.as_ref().map_or(String::from("-"), |r| format!("N{}", r.quality)),
310                f.gone("gone_cells").map_or(String::from("?"), |n| format!("{n}"))
311            )
312        })
313        .collect();
314    let why = || format!("old={old} new={new} frames={}", shape.join(","));
315    if rows.iter().any(|(o, n)| o.is_some() && n.is_some()) {
316        return Err(format!("both generations in one frame: {}", why()));
317    }
318    if rows.iter().flat_map(|(o, n)| [o, n]).flatten().any(|r| r.quality == "reset") {
319        return Err(format!("reset: {}", why()));
320    }
321    let present = rows.iter().position(|(o, _)| o.is_some()).ok_or_else(why)?;
322    let gone = present + rows[present..].iter().position(|(o, _)| o.is_none()).ok_or_else(why)?;
323    if rows[gone..].iter().any(|(o, _)| o.is_some()) || !frames[gone].gone("gone_cells").is_some_and(|n| n >= 1) {
324        return Err(why());
325    }
326    let first = rows.iter().position(|(_, n)| n.is_some()).ok_or_else(why)?;
327    let later_ok = rows[first + 1..].iter().any(|(_, n)| n.as_ref().is_some_and(|r| r.quality == "ok"));
328    let first_new = rows[first].1.as_ref().is_some_and(|r| r.quality == "new");
329    if first < gone || !first_new || !later_ok {
330        return Err(why());
331    }
332    Ok(why())
333}

335/// #14: `quota_bytes` once on every Cell row and on no task row; `cell`'s quota covers the
336/// stacks of its in-quota tasks and stays within its limit.
337pub fn quota_once(cells: &RecFrame, tasks: &RecFrame, cell: Ident) -> Result<String, String> {
338    if let Some(row) = cells.rows.iter().find(|r| key_count(r, "quota_bytes") != 1) {
339        return Err(format!("Cell row without exactly one quota_bytes: {row}"));
340    }
341    if tasks.rows.iter().any(|r| key_count(r, "quota_bytes") != 0) {
342        return Err(String::from("a task row carries quota_bytes"));
343    }
344    let row = cells.cell(cell).ok_or_else(|| format!("{cell} not in the Cells frame"))?;
345    let members = tasks.tasks();
346    if members.is_empty() || members.iter().any(|t| t.owner != Some(cell)) {
347        return Err(format!("task frame of {cell} has {} rows, owners not all {cell}", members.len()));
348    }
349    let stacks: u64 = members.iter().filter(|t| t.stack_in_quota).map(|t| t.stack_bytes).sum();
350    let detail = format!("quota={:?} limit={:?} in_quota_stacks={stacks} tasks={}", row.quota, row.limit, members.len());
351    match (row.quota, row.limit) {
352        (Some(quota), Some(limit)) if quota >= stacks && quota <= limit => Ok(detail),
353        _ => Err(detail),
354    }
355}
