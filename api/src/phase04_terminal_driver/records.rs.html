1//! Pure readers for the Phase 06 S6-5 `top-load` scenarios: `top --records … | to jsonl`
2//! envelope lines (`top.cell.v1`/`top.task.v1` rows, Begin `meta`, End), the fixture's
3//! `TOPLOAD` acknowledgement lines, the `s2tw-probe trunc` line, and the comparisons the
4//! scenarios make on them. Host-tested; nothing here touches a host or a syscall.
5//!
6//! The JSON reader only looks up a key of one envelope line. Every key it is used with is
7//! unique within its line (`"tasks":` never matches `"total_tasks":`: the quote is part of
8//! the key), and no value it reads contains an escaped quote.

14/// The raw value token of `"key":` in `json`: a string without its quotes, or a number,
15/// `true`, `false`, `null` (or an opening brace for a nested record).
16pub fn raw<'a>(json: &'a str, key: &str) -> Option<&'a str> {
17    let needle = format!("\"{key}\":");
18    let start = json.find(&needle)? + needle.len();
19    let rest = &json[start..];
20    if let Some(body) = rest.strip_prefix('"') {
21        return body.find('"').map(|end| &body[..end]);
22    }
23    let end = rest.find([',', '}', ']']).unwrap_or(rest.len());
24    Some(&rest[..end])
25}
26
27/// How many times `"key":` occurs in `json`.
28pub fn key_count(json: &str, key: &str) -> usize {
29    json.matches(&format!("\"{key}\":")).count()
30}
31
32pub fn uint(json: &str, key: &str) -> Option<u64> {
33    raw(json, key)?.parse().ok()
34}
35
36pub fn text<'a>(json: &'a str, key: &str) -> Option<&'a str> {
37    let needle = format!("\"{key}\":\"");
38    json.contains(&needle).then(|| raw(json, key)).flatten()
39}
40
41pub fn boolean(json: &str, key: &str) -> Option<bool> {
42    match raw(json, key)? {
43        "true" => Some(true),
44        "false" => Some(false),
45        _ => None,
46    }
47}

49/// One frame of an envelope stream: the Begin line, its row lines and the End line.
50#[derive(Clone, Debug, Default)]
51pub struct RecFrame {
52    pub begin: String,
53    pub rows: Vec<String>,
54    pub end: String,
55}
56
57impl RecFrame {
58    pub fn total_tasks(&self) -> Option<u64> {
59        uint(&self.begin, "total_tasks")
60    }
61
62    pub fn online_cpus(&self) -> Option<u64> {
63        uint(&self.begin, "online_cpus")
64    }
65
66    /// `gone_cells`/`gone_tasks` of the Begin `meta` (`None` for Null or absent).
67    pub fn gone(&self, key: &str) -> Option<u64> {
68        uint(&self.begin, key)
69    }
70
71    pub fn complete(&self) -> Option<bool> {
72        boolean(&self.end, "complete")
73    }
74
75    pub fn dropped(&self) -> Option<u64> {
76        uint(&self.end, "dropped")
77    }
78
79    pub fn cells(&self) -> Vec<CellRow> {
80        self.rows
81            .iter()
82            .filter_map(|row| CellRow::parse(row))
83            .collect()
84    }
85
86    pub fn tasks(&self) -> Vec<TaskRow> {
87        self.rows
88            .iter()
89            .filter_map(|row| TaskRow::parse(row))
90            .collect()
91    }
92
93    /// The Cell row of incarnation `id`.
94    pub fn cell(&self, id: Ident) -> Option<CellRow> {
95        self.cells().into_iter().find(|row| row.ident == id)
96    }
97}

99/// Split envelope lines into frames; every line between a Begin and its End must be a row
100/// and every frame must end. Lines that are not envelopes are ignored.
101pub fn frames(lines: &[String]) -> Result<Vec<RecFrame>, String> {
102    let mut out = Vec::new();
103    let mut open: Option<RecFrame> = None;
104    for line in lines.iter().filter(|l| l.starts_with("{\"frame\":")) {
105        match (raw(line, "frame"), open.as_mut()) {
106            (Some("begin"), None) => {
107                open = Some(RecFrame {
108                    begin: line.clone(),
109                    ..RecFrame::default()
110                })
111            }
112            (Some("row"), Some(frame)) => frame.rows.push(line.clone()),
113            (Some("end"), Some(frame)) => {
114                frame.end = line.clone();
115                out.extend(open.take());
116            }
117            (kind, _) => {
118                return Err(format!(
119                    "envelope {kind:?} out of order after {} frames",
120                    out.len()
121                ))
122            }
123        }
124    }
125    match open {
126        Some(_) => Err(format!("unterminated frame after {} frames", out.len())),
127        None => Ok(out),
128    }
129}

131/// A Cell incarnation: `cell_id:generation`.
132#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
133pub struct Ident {
134    pub id: u64,
135    pub generation: u64,
136}
137
138impl Ident {
139    /// `ID:GEN` text.
140    pub fn parse(text: &str) -> Option<Ident> {
141        let (id, generation) = text.split_once(':')?;
142        Some(Ident {
143            id: id.parse().ok()?,
144            generation: generation.parse().ok()?,
145        })
146    }
147}
148
149impl core::fmt::Display for Ident {
150    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
151        write!(f, "{}:{}", self.id, self.generation)
152    }
153}

169impl CellRow {

171        if !row.starts_with("{\"frame\":\"row\"") || key_count(row, "root_tid") != 1 {
172            return None;
173        }
174        Some(CellRow {
175            ident: Ident {
176                id: uint(row, "cell_id")?,
177                generation: uint(row, "generation")?,
178            },
179            root_tid: uint(row, "root_tid")?,
180            name: String::from(text(row, "name")?),
181            tasks: uint(row, "tasks")?,
182            cpu_ns: uint(row, "cpu_ns"),
183            tenths: uint(row, "cpu_tenths"),
184            quality: String::from(text(row, "cpu_quality")?),
185            quota: uint(row, "quota_bytes"),
186            limit: uint(row, "limit_bytes"),
187        })
188    }

217            tenths: uint(row, "cpu_tenths"),
218            quality: String::from(text(row, "cpu_quality")?),
219            stack_bytes: uint(row, "stack_bytes")?,
220            stack_in_quota: boolean(row, "stack_in_quota")?,
221        })
222    }

255/// `(emitted_cells, total_cells, emitted_tasks, total_tasks, complete)` of the probe's
256/// `other:cells-E/T-tasks-e/U-complete-C` result.
257pub fn trunc_counts(result: &str) -> Option<(u64, u64, u64, u64, u64)> {
258    let rest = result.strip_prefix("other:cells-")?;
259    let (cells, rest) = rest.split_once("-tasks-")?;
260    let (tasks, complete) = rest.split_once("-complete-")?;
261    let pair = |text: &str| -> Option<(u64, u64)> {
262        let (a, b) = text.split_once('/')?;
263        Some((a.parse().ok()?, b.parse().ok()?))
264    };
265    let (e, t) = pair(cells)?;
266    let (et, u) = pair(tasks)?;
267    Some((e, t, et, u, complete.parse().ok()?))
268}

270/// Σ `cpu_tenths` over a Cells frame's rows that have one, and the bound
271/// `online_cpus × 1000 + rows` (no CPU billed twice; one tenth of rounding per row).
272pub fn cpu_sum(frame: &RecFrame) -> Option<(u64, u64)> {
273    let cells = frame.cells();
274    let sum = cells.iter().filter_map(|row| row.tenths).sum();
275    Some((sum, frame.online_cpus()? * 1000 + cells.len() as u64))
276}

278/// #12: a task stream across the exit of `tid`. Returns the index of the last frame that
279/// still lists it.
280pub fn exit_gone_tasks(frames: &[RecFrame], tid: u64) -> Result<usize, String> {
281    let listed: Vec<bool> = frames
282        .iter()
283        .map(|f| f.tasks().iter().any(|t| t.tid == tid))
284        .collect();
285    if listed.first() != Some(&true) {
286        return Err(format!(
287            "tid {tid} not in the first frame: listed={listed:?}"
288        ));
289    }
290    let gone = listed
291        .iter()
292        .position(|&l| !l)
293        .ok_or_else(|| format!("tid {tid} never left: listed={listed:?}"))?;
294    if listed[gone..].iter().any(|&l| l) {
295        return Err(format!("tid {tid} came back: listed={listed:?}"));
296    }
297    let count = frames[gone].gone("gone_tasks");
298    let (before, after) = (frames[gone - 1].total_tasks(), frames[gone].total_tasks());
299    let exact = matches!((before, after), (Some(b), Some(a)) if a + 1 == b);
300    let ok = match count {
301        Some(n) if exact => n == 1,
302        Some(n) => n >= 1,
303        None => false,
304    };
305    if !ok {
306        return Err(format!(
307            "frame {gone} gone_tasks={count:?} total_tasks {before:?}->{after:?}"
308        ));
309    }
310    if let Some(reset) = frames
311        .iter()
312        .flat_map(|f| f.tasks())
313        .find(|t| t.quality == "reset")
314    {
315        return Err(format!("tid {} reset", reset.tid));
316    }
317    Ok(gone - 1)
318}

320/// #13: a Cells stream across a thread exit of `cell`: its task count falls from `from` to
321/// `from - 1`, its `cpu_ns` never falls, its quality stays `ok`, and after the exit its
322/// `cpu_ns` is at least `floor`. Returns the `cpu_ns` series.
323pub fn exit_retains_cpu(
324    frames: &[RecFrame],
325    cell: Ident,
326    from: u64,
327    floor: u64,
328) -> Result<Vec<u64>, String> {
329    let rows: Vec<CellRow> = frames
330        .iter()
331        .map(|f| f.cell(cell))
332        .collect::<Option<_>>()
333        .ok_or_else(|| format!("{cell} missing from a frame"))?;
334    let tasks: Vec<u64> = rows.iter().map(|r| r.tasks).collect();
335    let cpu: Vec<u64> = rows
336        .iter()
337        .map(|r| r.cpu_ns)
338        .collect::<Option<_>>()
339        .ok_or_else(|| format!("{cell} cpu_ns Null"))?;
340    let qualities: Vec<&str> = rows.iter().map(|r| r.quality.as_str()).collect();
341    let why = || format!("tasks={tasks:?} cpu_ns={cpu:?} quality={qualities:?} floor={floor}");
342    let fell = tasks.first() == Some(&from)
343        && tasks.last() == Some(&(from - 1))
344        && tasks.windows(2).all(|p| p[1] <= p[0]);
345    let monotonic = cpu.windows(2).all(|p| p[1] >= p[0]);
346    let ok = qualities.iter().all(|&q| q == "ok");
347    let retained = rows
348        .iter()
349        .zip(&cpu)
350        .filter(|(r, _)| r.tasks < from)
351        .all(|(_, &ns)| ns >= floor);
352    if fell && monotonic && ok && retained {
353        Ok(cpu)
354    } else {
355        Err(why())
356    }
357}

359/// #15: a Cells stream across a quit and relaunch of `old` as `new`: `old` listed, then a
360/// frame without it carrying `gone_cells ≥ 1`, then `new` first with quality `new`, then
361/// `new` with `ok`; never both in one frame and never `reset` on either.
362pub fn relaunch_sequence(frames: &[RecFrame], old: Ident, new: Ident) -> Result<String, String> {
363    let rows: Vec<(Option<CellRow>, Option<CellRow>)> =
364        frames.iter().map(|f| (f.cell(old), f.cell(new))).collect();
365    let shape: Vec<String> = rows
366        .iter()
367        .zip(frames)
368        .map(|((o, n), f)| {
369            format!(
370                "{}{}g{}",
371                o.as_ref()
372                    .map_or("-", |r| if r.quality == "reset" { "O!" } else { "O" }),
373                n.as_ref()
374                    .map_or(String::from("-"), |r| format!("N{}", r.quality)),
375                f.gone("gone_cells")
376                    .map_or(String::from("?"), |n| format!("{n}"))
377            )
378        })
379        .collect();
380    let why = || format!("old={old} new={new} frames={}", shape.join(","));
381    if rows.iter().any(|(o, n)| o.is_some() && n.is_some()) {
382        return Err(format!("both generations in one frame: {}", why()));
383    }
384    if rows
385        .iter()
386        .flat_map(|(o, n)| [o, n])
387        .flatten()
388        .any(|r| r.quality == "reset")
389    {
390        return Err(format!("reset: {}", why()));
391    }
392    let present = rows.iter().position(|(o, _)| o.is_some()).ok_or_else(why)?;
393    let gone = present
394        + rows[present..]
395            .iter()
396            .position(|(o, _)| o.is_none())
397            .ok_or_else(why)?;
398    if rows[gone..].iter().any(|(o, _)| o.is_some())
399        || !frames[gone].gone("gone_cells").is_some_and(|n| n >= 1)
400    {
401        return Err(why());
402    }
403    let first = rows.iter().position(|(_, n)| n.is_some()).ok_or_else(why)?;
404    let later_ok = rows[first + 1..]
405        .iter()
406        .any(|(_, n)| n.as_ref().is_some_and(|r| r.quality == "ok"));
407    let first_new = rows[first].1.as_ref().is_some_and(|r| r.quality == "new");
408    if first < gone || !first_new || !later_ok {
409        return Err(why());
410    }
411    Ok(why())
412}

414/// #14: `quota_bytes` once on every Cell row and on no task row; `cell`'s quota covers the
415/// stacks of its in-quota tasks and stays within its limit.
416pub fn quota_once(cells: &RecFrame, tasks: &RecFrame, cell: Ident) -> Result<String, String> {
417    if let Some(row) = cells.rows.iter().find(|r| key_count(r, "quota_bytes") != 1) {
418        return Err(format!("Cell row without exactly one quota_bytes: {row}"));
419    }
420    if tasks.rows.iter().any(|r| key_count(r, "quota_bytes") != 0) {
421        return Err(String::from("a task row carries quota_bytes"));
422    }
423    let row = cells
424        .cell(cell)
425        .ok_or_else(|| format!("{cell} not in the Cells frame"))?;
426    let members = tasks.tasks();
427    if members.is_empty() || members.iter().any(|t| t.owner != Some(cell)) {
428        return Err(format!(
429            "task frame of {cell} has {} rows, owners not all {cell}",
430            members.len()
431        ));
432    }
433    let stacks: u64 = members
434        .iter()
435        .filter(|t| t.stack_in_quota)
436        .map(|t| t.stack_bytes)
437        .sum();
438    let detail = format!(
439        "quota={:?} limit={:?} in_quota_stacks={stacks} tasks={}",
440        row.quota,
441        row.limit,
442        members.len()
443    );
444    match (row.quota, row.limit) {
445        (Some(quota), Some(limit)) if quota >= stacks && quota <= limit => Ok(detail),
446        _ => Err(detail),
447    }
448}

628        let owner = Some((5, 9));
629        let frame = |sample, total, gone, tids: &[u64]| {
630            (
631                begin(sample, total, Some(0), gone),
632                tids.iter()
633                    .map(|&t| task(t, owner, 1, "ok", 0, false))
634                    .collect::<Vec<_>>(),
635            )
636        };
637        let good = stream(&[
638            frame(1, 70, Some(0), &[7, 8]),
639            frame(2, 70, Some(0), &[7, 8]),

641            frame(4, 69, Some(0), &[8]),
642        ]);
643        assert_eq!(exit_gone_tasks(&good, 7), Ok(1));
644        // Total fell by one but two tasks were counted gone.

646        assert!(exit_gone_tasks(&wrong, 7).is_err());
647        // Another task appeared meanwhile: at least one gone is enough.
648        let busy = stream(&[frame(1, 70, Some(0), &[7, 8]), frame(2, 70, Some(1), &[8])]);

653            frame(3, 70, Some(0), &[7]),
654        ]);

656        let late = stream(&[frame(1, 70, Some(0), &[8]), frame(2, 69, Some(1), &[8])]);
657        assert!(exit_gone_tasks(&late, 7).is_err());
658        let null = stream(&[frame(1, 70, Some(0), &[7]), frame(2, 69, None, &[])]);
659        assert!(exit_gone_tasks(&null, 7).is_err());
