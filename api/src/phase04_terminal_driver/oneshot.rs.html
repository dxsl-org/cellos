2//! directly with a prepared context, as `stock_negative_grants` launches `/bin/echo`, on

9/// Ring of each stream this driver creates for a one-shot (bytes).
10const RING: u32 = 4096;
11/// Bound for one ordinary one-shot, launch to exit record (TCG).
12const RUN_MS: u64 = 120_000;
13/// A refused one-shot of [`ONE_SHOT_REFUSED`] or a consumer close must end well before the
14/// 30 s write deadline of the child.
15const PROMPT_EXIT_MS: u64 = 20_000;
16const FRONTEND_ONLY_SH: &[u8] = b"cellbox: a terminal frontend runs only 'sh'\n";
17const RG_FX_HD: &str = "/mnt/sd/rg/hd";

19/// How one direct one-shot launch is wired and driven.
20struct Spec<'a> {
21    argv: &'a [&'a str],
22    stdout: bool,
23    stderr: bool,
24    /// `Some`: a Stdin grant fed with these bytes, then closed (end of stream).
25    stdin: Option<&'a [u8]>,
26    /// Close the stdout reader once this many bytes arrived (a consumer that stops).
27    close_stdout_after: Option<usize>,
28    /// Cancel the one-shot's Operation this long after the commit.
29    cancel_after_ms: Option<u64>,
30    timeout_ms: u64,
31}
32
33impl<'a> Spec<'a> {
34    fn new(argv: &'a [&'a str]) -> Self {
35        Spec { argv, stdout: true, stderr: true, stdin: None, close_stdout_after: None, cancel_after_ms: None, timeout_ms: RUN_MS }
36    }
37}

39/// What a one-shot did: its terminal record, both streams and the elapsed time.
40struct Ran {
41    /// `Ok(code)` for a normal exit; `Err(reason)` for a cancelled/killed child.
42    exit: Result<i32, u32>,
43    out: Vec<u8>,
44    err: Vec<u8>,
45    elapsed_ms: u64,
46}
47
48impl Ran {
49    fn code(&self) -> Option<i32> {
50        self.exit.ok()
51    }
52
53    fn quiet_err(&self) -> bool {
54        self.err.is_empty()
55    }
56
57    fn summary(&self) -> String {
58        format!("exit={:?} out={} err={} ms={}", self.exit, hexs(&self.out), hexs(&self.err), self.elapsed_ms)
59    }
60}

62/// Short hex of observed bytes for FAIL lines (payload text never reaches the UART).
63fn hexs(bytes: &[u8]) -> String {
64    const HEX: &[u8; 16] = b"0123456789abcdef";
65    let mut out = format!("{}:", bytes.len());
66    for byte in bytes.iter().take(96) {
67        out.push(HEX[(byte >> 4) as usize] as char);
68        out.push(HEX[(byte & 15) as usize] as char);
69    }
70    out
71}

73/// One non-blocking QueryChild: `Some((reason, code))` once the child has ended.
74fn query_child(op: Operation, child: ChildRefV1) -> Result<Option<(u32, i32)>, Status> {
75    let mut reply = [0u8; HEADER_LEN + 24];
76    let h = ostd::session::kernel_call(KernelOp::QueryChild, op.reference().raw(), false, &child.to_bytes(), &mut reply)?;
77    if h.status != Status::Ok {
78        return Err(h.status);
79    }
80    let mut body = Reader::new(&reply[HEADER_LEN..h.total_len as usize]);
81    let state = body.u32()?;
82    let reason = body.u32()?;
83    let code = body.i32()?;
84    if body.u32()? != 0 {
85        return Err(Status::Malformed);
86    }
87    let _sequence = body.u64()?;
88    body.finish()?;
89    Ok((state == 3).then_some((reason, code)))
90}

92/// Move everything readable now from `endpoint` into `sink`; `false` at end of stream.
93fn drain_now(endpoint: Endpoint, sink: &mut Vec<u8>) -> bool {
94    let mut chunk = [0u8; 512];
95    for _ in 0..64 {
96        match endpoint.read(&mut chunk, 0) {
97            Ok(n) => sink.extend_from_slice(&chunk[..n]),
98            Err(Status::WouldBlock | Status::TimedOut) => return true,
99            Err(_) => return false,
100        }
101    }
102    true
103}

105/// Launch `/bin/cellbox` with `spec.argv` as its context argv (no program path, no cwd,
106/// no environment), drive its streams and wait for its terminal record.
107fn launch(spec: &Spec) -> Result<Ran, Status> {
108    let op = Operation::create(None)?;
109    let result = (|| -> Result<Ran, Status> {
110        let (out_read, out_write) = op.create_stream(RING)?;
111        let (err_read, err_write) = op.create_stream(RING)?;
112        let stdin = match spec.stdin {
113            Some(_) => Some(op.create_stream(RING)?),
114            None => None,
115        };
116        let placeholder = ObjectRefV1 { id: 1, generation: 1 };
117        let mut wanted: Vec<(GrantSlot, Endpoint, EndpointRights)> = Vec::new();
118        if spec.stdout {
119            wanted.push((GrantSlot::Stdout, out_write, EndpointRights::WRITE_ONLY));
120        }
121        if spec.stderr {
122            wanted.push((GrantSlot::Stderr, err_write, EndpointRights::WRITE_ONLY));
123        }
124        if let Some((reader, _)) = stdin {
125            wanted.push((GrantSlot::Stdin, reader, EndpointRights::READ_ONLY));
126        }
127        let sizing: Vec<Grant> = wanted
128            .iter()
129            .map(|(slot, _, rights)| Grant::endpoint(*slot, placeholder, rights.bits() as u16))
130            .collect::<Result<_, _>>()?;
131        let context_len = ChildContext::new(op.reference().raw(), spec.argv, b"", &[], &sizing)?.as_bytes().len();
132        let mut prepared = op.prepare_child(SHELL, context_len)?;
133        let mut grants = Vec::new();
134        for (slot, source, rights) in &wanted {
135            let delegated = source.delegate(&prepared, *rights)?;
136            grants.push(Grant::endpoint(*slot, delegated.reference().raw(), rights.bits() as u16)?);
137        }
138        let context = ChildContext::new(op.reference().raw(), spec.argv, b"", &[], &grants)?;
139        prepared.write_context(&context)?;
140        // Stdin that fits the ring goes in before the commit, and its writer is closed
141        // then: the child may run (and, not reading stdin, exit) before this driver runs
142        // again, and a writer still open when the child's reader goes away is a producer
143        // whose consumer closed — the kernel cancels its operation (stop reason 3), with
144        // this driver's output rings. A larger stdin is fed below, to a consumer.
145        let mut feed = None;
146        if let (Some((_, writer)), Some(bytes)) = (stdin, spec.stdin) {
147            let mut rest = bytes;
148            while !rest.is_empty() {
149                match writer.write(rest, 0) {
150                    Ok(n) => rest = &rest[n..],
151                    Err(_) => break,
152                }
153            }
154            if rest.is_empty() {
155                let _ = writer.close();
156            } else {
157                feed = Some((writer, rest));
158            }
159        }
160        let child = prepared.commit()?;
161        let start = now_ms();
162        // Only the child holds the writers (and the stdin reader) now: its exit ends both
163        // output streams.
164        let _ = (out_write.close(), err_write.close());
165        if let Some((reader, _)) = stdin {
166            let _ = reader.close();
167        }
168        let mut out = Vec::new();
169        let mut err = Vec::new();
170        let mut out_open = Some(out_read);
171        let mut cancelled = false;
172        let exit = loop {
173            // Close the feed as soon as its last byte is in.
174            if let Some((writer, rest)) = feed.as_mut() {
175                let done = match writer.write(rest, 0) {
176                    Ok(n) => {
177                        *rest = &rest[n..];
178                        rest.is_empty()
179                    }
180                    Err(Status::WouldBlock | Status::TimedOut) => false,
181                    Err(_) => true,
182                };
183                if done {
184                    let _ = writer.close();
185                    feed = None;
186                }
187            }
188            if let Some(reader) = out_open {
189                drain_now(reader, &mut out);
190                if spec.close_stdout_after.is_some_and(|after| out.len() >= after) {
191                    let _ = reader.close();
192                    out_open = None;
193                }
194            }
195            drain_now(err_read, &mut err);
196            if let Some(ms) = spec.cancel_after_ms {
197                if !cancelled && now_ms() >= start + ms {
198                    cancelled = true;
199                    op.cancel(2)?;
200                }
201            }
202            if let Some((reason, code)) = query_child(op, child)? {
203                break if reason == 0 { Ok(code) } else { Err(reason) };
204            }
205            if now_ms() >= start + spec.timeout_ms {
206                return Err(Status::TimedOut);
207            }
208            let _ = sys_set_timer(1);
209        };
210        let elapsed_ms = now_ms().saturating_sub(start);
211        if let Some(reader) = out_open {
212            // The child has ended: what it wrote is in the ring, then end of stream.
213            for _ in 0..256 {
214                let before = out.len();
215                if !drain_now(reader, &mut out) || out.len() == before {
216                    break;
217                }
218            }
219            let _ = reader.close();
220        }
221        drain_now(err_read, &mut err);
222        let _ = err_read.close();
223        if let Some((writer, _)) = feed {
224            let _ = writer.close();
225        }
226        Ok(Ran { exit, out, err, elapsed_ms })
227    })();
228    let _ = op.cancel(2);
229    let _ = op.close();
230    // Every run, for the record (observed bytes as hex only).
231    println(&format!("P04 INFO oneshot-run argv0={} argc={} {}", spec.argv[0], spec.argv.len(), ran(&result)));
232    result
233}

269/// The shell's view of tasks named `name`: their tids, sorted (the shell runs `ps`
270/// in process, so the listing itself adds no task).
271fn tasks_named(a: &mut Term, name: &str) -> Result<Vec<String>, String> {
272    let lines = a.run(&format!("ps --records | where name '==' '\"{name}\"' | select tid | to jsonl"), 30_000)?;
273    let mut tids: Vec<String> = rows_of(&lines).into_iter().cloned().collect();
274    tids.sort();
275    Ok(tids)
276}

278fn task_set(a: &mut Term) -> Result<Vec<Vec<String>>, String> {

488/// The S2-K secret namespace and a write into it are refused exactly as in the shell:
489/// same nonzero status, the fixture bytes on no stream, nothing left behind; a one-shot
490/// cannot even reach the S2-T/W sandbox probe (no launch, 127).
491fn denials_same_as_shell(a: &mut Term) {
492    let mut bad = Vec::new();
493    let token = SECRET_TOKEN.as_bytes();
494    for cmd in [
495        "cat /srv/cellos/p04-secret",
496        "grep p04 /srv/cellos/p04-secret",
497        "head -n 1 /srv/cellos/p04-secret",
498        "wc /srv/cellos/p04-secret",
499        "sort /srv/cellos/p04-secret",
500        // The shell echoes these command lines in INFO lines: no `p04os-` token here.
501        "cp /srv/cellos/p04-secret /tmp/p04-osleak",
502        "ls /srv/cellos",
503        "vwrite /srv/cellos/p04-osw p04-osw",
504    ] {
505        let shell_out = a.run(cmd, 30_000);
506        let shell_err = a.stderr_text().replace('\r', "");
507        let shell_status = a.run("echo sos=$?", 30_000);
508        let argv: Vec<&str> = cmd.split(' ').collect();
509        let r = run(&argv);
510        // Same status and the same bytes as the shell (its transcript has the echoed line
511        // and prompt removed); a refused read or copy is nonzero, the legacy `vwrite`
512        // helper reports its refused write with status 0.
513        let same = match (&shell_status, &shell_out, &r) {
514            (Ok(l), Ok(shell_lines), Ok(r)) => {
515                let text = String::from_utf8_lossy(&r.out);
516                let ours: Vec<&str> = text.lines().filter(|line| !line.is_empty()).collect();
517                let theirs: Vec<&str> = shell_lines.iter().map(String::as_str).filter(|line| !line.is_empty()).collect();
518                l.len() == 1
519                    && r.code().is_some_and(|code| l[0] == format!("sos={code}") && (code != 0 || argv[0] == "vwrite"))
520                    && ours == theirs
521                    && r.err == shell_err.as_bytes()
522                    && !(r.out.is_empty() && r.err.is_empty())
523            }
524            _ => false,
525        };
526        let leaked = matches!(&r, Ok(r) if r.out.windows(token.len()).any(|w| w == token) || r.err.windows(token.len()).any(|w| w == token));
527        if !same || leaked {
528            bad.push(format!("{cmd}: shell={shell_status:?} shell_err={} {} leaked={leaked}", hexs(shell_err.as_bytes()), ran(&r)));
529        }
530    }
531    let leak = run(&["cat", "/tmp/p04-osleak"]);
532    let probe = run(&["/data/bin/s2tw-probe", "sb"]);
533    let clean = matches!(&leak, Ok(r) if r.code() != Some(0) && !r.out.windows(token.len()).any(|w| w == token))
534        && is(&probe, 127, b"", b"cellbox: /data/bin/s2tw-probe: unknown command\n");
535    check(
536        "oneshot-denials-same-as-shell",
537        bad.is_empty() && clean,
538        &format!("{bad:?} leak {} probe {}", ran(&leak), ran(&probe)),
539    );
540}

564/// A reader that closes after the first chunk stops a streamed `rg` at once, long before
565/// the 30 s write deadline: the kernel stops a producer whose final consumer closed
566/// (stop reason 3, which a parent shell reports as 141) instead of letting it write on.
567fn consumer_close() {
568    // > 64 KiB of matching lines, written to RamFS by a one-shot `tee` over its stdin.
569    let mut big = Vec::new();
570    let mut n = 0u32;
571    while big.len() <= 72 * 1024 {
572        big.extend_from_slice(format!("p04os-line {n:06} needle\n").as_bytes());
573        n += 1;
574    }
575    let path = "/tmp/p04os-big.txt";
576    let tee = launch(&Spec { stdin: Some(&big), ..Spec::new(&["tee", path]) });
577    let tee_ok = matches!(&tee, Ok(r) if r.code() == Some(0) && r.out == big && r.quiet_err());
578    let r = launch(&Spec { close_stdout_after: Some(1), timeout_ms: 60_000, ..Spec::new(&["rg", "needle", path]) });
579    let reported = match &r {
580        Ok(Ran { exit: Err(3), .. }) => Some(141),
581        Ok(Ran { exit: Err(_), .. }) => Some(130),
582        Ok(Ran { exit: Ok(code), .. }) => Some(*code),
583        Err(_) => None,
584    };
585    println(&format!("P04 INFO oneshot-consumer-close lines={n} reported={reported:?} {}", ran(&r)));
586    check(
587        "oneshot-consumer-close",
588        tee_ok
589            && matches!(&r, Ok(r) if r.exit == Err(3) && !r.out.is_empty() && r.out.len() < 64 * 1024 && r.elapsed_ms < PROMPT_EXIT_MS)
590            && reported == Some(141),
591        &format!("tee_ok={tee_ok} rg {}", ran(&r)),
592    );
593}

595/// Cancelling the one-shot's Operation during `sleep 30` ends it promptly with a
596/// cancellation record (reported as 130), and no `cellbox` task is left behind.
597fn cancel(a: &mut Term) {
598    let before = tasks_named(a, "cellbox");
599    let r = launch(&Spec { cancel_after_ms: Some(1500), timeout_ms: 25_000, ..Spec::new(&["sleep", "30"]) });
600    sleep_ms(2000);
601    let after = tasks_named(a, "cellbox");
602    let reported = match &r {
603        Ok(Ran { exit: Err(_), .. }) => Some(130),
604        Ok(Ran { exit: Ok(code), .. }) => Some(*code),
605        Err(_) => None,
606    };
607    println(&format!("P04 INFO oneshot-cancel {} reported={reported:?}", ran(&r)));
608    check(
609        "oneshot-cancel",
610        matches!(&r, Ok(r) if r.exit.is_err() && r.elapsed_ms < PROMPT_EXIT_MS) && reported == Some(130)
611            && before.is_ok() && before == after,
612        &format!("{} cellbox {before:?} -> {after:?}", ran(&r)),
613    );
614}

616/// A terminal frontend runs only `sh`: any other argv exits 2 with one line on the
617/// holder's stderr writer and never attaches.
618fn frontend_argv_refused() {
619    let result = (|| -> Result<(Option<i32>, Vec<u8>, Vec<u8>), String> {
620        let mut t = Term::open(24, 80).map_err(|s| format!("open {s:?}"))?;
621        t.host.launch(SHELL, &["ls", "/p04-list"], &[]).map_err(|s| format!("launch {s:?}"))?;
622        let exited = wait_exit(&mut t, 60_000);
623        let code = t.host.exit_code();
624        let (out, err) = (core::mem::take(&mut t.out), core::mem::take(&mut t.err));
625        let _ = t.host.close();
626        exited?;
627        Ok((code, out, err))
628    })();
629    check(
630        "oneshot-frontend-argv-refused",
631        matches!(&result, Ok((Some(2), out, err)) if out.is_empty() && err.as_slice() == FRONTEND_ONLY_SH),
632        &match &result {
633            Ok((code, out, err)) => format!("code={code:?} out={} err={}", hexs(out), hexs(err)),
634            Err(why) => why.clone(),
635        },
636    );
637}

639/// In `cellbox sh`: `cellbox CMD` and `/bin/cellbox CMD` are the short forms in process;
640/// a refused name answers like the one-shot, with no effect.
641fn shell_explicit_prefix(a: &mut Term) {
642    let short = a.run("echo p04os-hi", 30_000);
643    let long = a.run("cellbox echo p04os-hi", 30_000);
644    let ls_short = a.run("ls /p04-list", 30_000);
645    let ls_long = a.run("/bin/cellbox ls /p04-list", 30_000);
646    let st_ls = a.run("echo stexl=$?", 30_000);
647    let pwd_before = a.run("pwd", 30_000);
648    let cd = a.run("cellbox cd /p04-list", 30_000);
649    let cd_err = a.stderr_text();
650    let st_cd = a.run("echo stexcd=$?", 30_000);
651    let pwd_after = a.run("pwd", 30_000);
652    let typed = a.run("cellbox ps --records | select tid | take 1 | to jsonl", 30_000);
653    let st_typed = a.run("echo stexty=$?", 30_000);
654    let ok = matches!((&short, &long), (Ok(s), Ok(l)) if s == l && s == &[String::from("p04os-hi")])
655        && matches!((&ls_short, &ls_long), (Ok(s), Ok(l)) if s == l && s.iter().any(|x| x == "alpha-entry"))
656        && matches!(&st_ls, Ok(l) if l == &[String::from("stexl=0")])
657        && cd.is_ok() && cd_err.contains("cellbox: cd changes or reads shell state")
658        && matches!(&st_cd, Ok(l) if l == &[String::from("stexcd=2")])
659        && pwd_before.is_ok() && pwd_before == pwd_after
660        && matches!(&typed, Ok(l) if rows_of(l).len() == 1)
661        && matches!(&st_typed, Ok(l) if l == &[String::from("stexty=0")]);
662    check(
663        "shell-explicit-prefix",
664        ok,
665        &format!("echo={} ls={} st_ls={st_ls:?} cd_st={st_cd:?} pwd={} typed={:?} st_typed={st_typed:?}",
666            short == long, ls_short == ls_long, pwd_before == pwd_after, typed.map(|l| rows_of(&l).len())),
667    );
