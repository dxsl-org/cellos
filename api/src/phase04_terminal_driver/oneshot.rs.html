1//! Phase 10 S10-0 (`phase-10-s0-proposal.md` §2 S10-0.8): `/bin/cellbox <command>` launched
2//! directly with a prepared context, as `stock_negative_grants` launches `/bin/echo`, on
3//! this fixture's reviewed `-> /bin/cellbox` edge. Every one-shot payload carries the
4//! `p04os-` token, which the runner requires absent from the UART; FAIL lines print
5//! observed bytes as hex only.

9/// Ring of each stream this driver creates for a one-shot (bytes).
10const RING: u32 = 4096;
11/// Bound for one ordinary one-shot, launch to exit record (TCG).
12const RUN_MS: u64 = 120_000;
13/// A refused one-shot of [`ONE_SHOT_REFUSED`] or a consumer close must end well before the
14/// 30 s write deadline of the child.
15const PROMPT_EXIT_MS: u64 = 20_000;
16const FRONTEND_ONLY_SH: &[u8] = b"cellbox: a terminal frontend runs only 'sh'\n";
17const RG_FX_HD: &str = "/mnt/sd/rg/hd";

19/// How one direct one-shot launch is wired and driven.
20struct Spec<'a> {
21    argv: &'a [&'a str],
22    stdout: bool,
23    stderr: bool,
24    /// `Some`: a Stdin grant fed with these bytes, then closed (end of stream).
25    stdin: Option<&'a [u8]>,
26    /// Close the stdout reader once this many bytes arrived (a consumer that stops).
27    close_stdout_after: Option<usize>,
28    /// Cancel the one-shot's Operation this long after the commit.
29    cancel_after_ms: Option<u64>,
30    timeout_ms: u64,
31}
32
33impl<'a> Spec<'a> {
34    fn new(argv: &'a [&'a str]) -> Self {
35        Spec {
36            argv,
37            stdout: true,
38            stderr: true,
39            stdin: None,
40            close_stdout_after: None,
41            cancel_after_ms: None,
42            timeout_ms: RUN_MS,
43        }
44    }
45}

47/// What a one-shot did: its terminal record, both streams and the elapsed time.
48struct Ran {
49    /// `Ok(code)` for a normal exit; `Err(reason)` for a cancelled/killed child.
50    exit: Result<i32, u32>,
51    out: Vec<u8>,
52    err: Vec<u8>,
53    elapsed_ms: u64,
54}
55
56impl Ran {
57    fn code(&self) -> Option<i32> {
58        self.exit.ok()
59    }
60
61    fn quiet_err(&self) -> bool {
62        self.err.is_empty()
63    }
64
65    fn summary(&self) -> String {
66        format!(
67            "exit={:?} out={} err={} ms={}",
68            self.exit,
69            hexs(&self.out),
70            hexs(&self.err),
71            self.elapsed_ms
72        )
73    }
74}

76/// Short hex of observed bytes for FAIL lines (payload text never reaches the UART).
77fn hexs(bytes: &[u8]) -> String {
78    const HEX: &[u8; 16] = b"0123456789abcdef";
79    let mut out = format!("{}:", bytes.len());
80    for byte in bytes.iter().take(96) {
81        out.push(HEX[(byte >> 4) as usize] as char);
82        out.push(HEX[(byte & 15) as usize] as char);
83    }
84    out
85}

87/// One non-blocking QueryChild: `Some((reason, code))` once the child has ended.
88fn query_child(op: Operation, child: ChildRefV1) -> Result<Option<(u32, i32)>, Status> {
89    let mut reply = [0u8; HEADER_LEN + 24];
90    let h = ostd::session::kernel_call(
91        KernelOp::QueryChild,
92        op.reference().raw(),
93        false,
94        &child.to_bytes(),
95        &mut reply,
96    )?;
97    if h.status != Status::Ok {
98        return Err(h.status);
99    }
100    let mut body = Reader::new(&reply[HEADER_LEN..h.total_len as usize]);
101    let state = body.u32()?;
102    let reason = body.u32()?;
103    let code = body.i32()?;
104    if body.u32()? != 0 {
105        return Err(Status::Malformed);
106    }
107    let _sequence = body.u64()?;
108    body.finish()?;
109    Ok((state == 3).then_some((reason, code)))
110}

112/// Move everything readable now from `endpoint` into `sink`; `false` at end of stream.
113fn drain_now(endpoint: Endpoint, sink: &mut Vec<u8>) -> bool {
114    let mut chunk = [0u8; 512];
115    for _ in 0..64 {
116        match endpoint.read(&mut chunk, 0) {
117            Ok(n) => sink.extend_from_slice(&chunk[..n]),
118            Err(Status::WouldBlock | Status::TimedOut) => return true,
119            Err(_) => return false,
120        }
121    }
122    true
123}

125/// Launch `/bin/cellbox` with `spec.argv` as its context argv (no program path, no cwd,
126/// no environment), drive its streams and wait for its terminal record.
127fn launch(spec: &Spec) -> Result<Ran, Status> {
128    let op = Operation::create(None)?;
129    let result = (|| -> Result<Ran, Status> {
130        let (out_read, out_write) = op.create_stream(RING)?;
131        let (err_read, err_write) = op.create_stream(RING)?;
132        let stdin = match spec.stdin {
133            Some(_) => Some(op.create_stream(RING)?),
134            None => None,
135        };
136        let placeholder = ObjectRefV1 {
137            id: 1,
138            generation: 1,
139        };
140        let mut wanted: Vec<(GrantSlot, Endpoint, EndpointRights)> = Vec::new();
141        if spec.stdout {
142            wanted.push((GrantSlot::Stdout, out_write, EndpointRights::WRITE_ONLY));
143        }
144        if spec.stderr {
145            wanted.push((GrantSlot::Stderr, err_write, EndpointRights::WRITE_ONLY));
146        }
147        if let Some((reader, _)) = stdin {
148            wanted.push((GrantSlot::Stdin, reader, EndpointRights::READ_ONLY));
149        }
150        let sizing: Vec<Grant> = wanted
151            .iter()
152            .map(|(slot, _, rights)| Grant::endpoint(*slot, placeholder, rights.bits() as u16))
153            .collect::<Result<_, _>>()?;
154        let context_len = ChildContext::new(op.reference().raw(), spec.argv, b"", &[], &sizing)?
155            .as_bytes()
156            .len();
157        let mut prepared = op.prepare_child(SHELL, context_len)?;
158        let mut grants = Vec::new();
159        for (slot, source, rights) in &wanted {
160            let delegated = source.delegate(&prepared, *rights)?;
161            grants.push(Grant::endpoint(
162                *slot,
163                delegated.reference().raw(),
164                rights.bits() as u16,
165            )?);
166        }
167        let context = ChildContext::new(op.reference().raw(), spec.argv, b"", &[], &grants)?;
168        prepared.write_context(&context)?;
169        // Stdin that fits the ring goes in before the commit, and its writer is closed
170        // then: the child may run (and, not reading stdin, exit) before this driver runs
171        // again, and a writer still open when the child's reader goes away is a producer
172        // whose consumer closed — the kernel cancels its operation (stop reason 3), with
173        // this driver's output rings. A larger stdin is fed below, to a consumer.
174        let mut feed = None;
175        if let (Some((_, writer)), Some(bytes)) = (stdin, spec.stdin) {
176            let mut rest = bytes;
177            while !rest.is_empty() {
178                match writer.write(rest, 0) {
179                    Ok(n) => rest = &rest[n..],
180                    Err(_) => break,
181                }
182            }
183            if rest.is_empty() {
184                let _ = writer.close();
185            } else {
186                feed = Some((writer, rest));
187            }
188        }
189        let child = prepared.commit()?;
190        let start = now_ms();
191        // Only the child holds the writers (and the stdin reader) now: its exit ends both
192        // output streams.
193        let _ = (out_write.close(), err_write.close());
194        if let Some((reader, _)) = stdin {
195            let _ = reader.close();
196        }
197        let mut out = Vec::new();
198        let mut err = Vec::new();
199        let mut out_open = Some(out_read);
200        let mut cancelled = false;
201        let exit = loop {
202            // Close the feed as soon as its last byte is in.
203            if let Some((writer, rest)) = feed.as_mut() {
204                let done = match writer.write(rest, 0) {
205                    Ok(n) => {
206                        *rest = &rest[n..];
207                        rest.is_empty()
208                    }
209                    Err(Status::WouldBlock | Status::TimedOut) => false,
210                    Err(_) => true,
211                };
212                if done {
213                    let _ = writer.close();
214                    feed = None;
215                }
216            }
217            if let Some(reader) = out_open {
218                drain_now(reader, &mut out);
219                if spec
220                    .close_stdout_after
221                    .is_some_and(|after| out.len() >= after)
222                {
223                    let _ = reader.close();
224                    out_open = None;
225                }
226            }
227            drain_now(err_read, &mut err);
228            if let Some(ms) = spec.cancel_after_ms {
229                if !cancelled && now_ms() >= start + ms {
230                    cancelled = true;
231                    op.cancel(2)?;
232                }
233            }
234            if let Some((reason, code)) = query_child(op, child)? {
235                break if reason == 0 { Ok(code) } else { Err(reason) };
236            }
237            if now_ms() >= start + spec.timeout_ms {
238                return Err(Status::TimedOut);
239            }
240            let _ = sys_set_timer(1);
241        };
242        let elapsed_ms = now_ms().saturating_sub(start);
243        if let Some(reader) = out_open {
244            // The child has ended: what it wrote is in the ring, then end of stream.
245            for _ in 0..256 {
246                let before = out.len();
247                if !drain_now(reader, &mut out) || out.len() == before {
248                    break;
249                }
250            }
251            let _ = reader.close();
252        }
253        drain_now(err_read, &mut err);
254        let _ = err_read.close();
255        if let Some((writer, _)) = feed {
256            let _ = writer.close();
257        }
258        Ok(Ran {
259            exit,
260            out,
261            err,
262            elapsed_ms,
263        })
264    })();
265    let _ = op.cancel(2);
266    let _ = op.close();
267    // Every run, for the record (observed bytes as hex only).
268    println(&format!(
269        "P04 INFO oneshot-run argv0={} argc={} {}",
270        spec.argv[0],
271        spec.argv.len(),
272        ran(&result)
273    ));
274    result
275}

308fn typed_jsonl(bytes: &[u8]) -> bool {
309    let rows = lines(bytes);
310    !rows.is_empty()
311        && bytes.ends_with(b"\n")
312        && rows
313            .iter()
314            .all(|row| row.starts_with(b"{\"cellbox\":1,") && row.ends_with(b"}"))
315}

321/// The shell's view of tasks named `name`: their tids, sorted (the shell runs `ps`
322/// in process, so the listing itself adds no task).
323fn tasks_named(a: &mut Term, name: &str) -> Result<Vec<String>, String> {
324    let lines = a.run(
325        &format!("ps --records | where name '==' '\"{name}\"' | select tid | to jsonl"),
326        30_000,
327    )?;
328    let mut tids: Vec<String> = rows_of(&lines).into_iter().cloned().collect();
329    tids.sort();
330    Ok(tids)
331}

333fn task_set(a: &mut Term) -> Result<Vec<Vec<String>>, String> {
334    ["cellbox", "echo", "ls", "hypervisor"]

392/// The one-shot lists the fixture directory exactly as the shell's `ls` does.
393fn ls_fixture(a: &mut Term) {

395    let r = run(&["ls", "/p04-list"]);

402        bytes.shrink_to_fit();
403        bytes

406    check(
407        "oneshot-ls-fixture",
408        names
409            && matches!((&r, &expected), (Ok(ran), Some(bytes)) if ran.code() == Some(0) && &ran.out == bytes && ran.quiet_err()),
410        &format!("{} shell={:?}", ran(&r), shell.map(|l| l.len())),
411    );

414/// `cat` with no operand copies its stdin: UTF-8 (NUL bytes included) byte for byte; a

417    let text: &[u8] = b"p04os-cat\x00nul \xe2\x82\xac euro\n\x00second line\n";

422        "oneshot-cat-stdin-bytes",

424        &format!("utf8 {} dash {} raw {}", ran(&utf8), ran(&dash), ran(&raw)),

458/// `rg` over the S7-4 fixture: typed records by default, the search text with `--text`,

597fn mutate_effect(a: &mut Term) {
598    let dir = "/tmp/p04os-dir";
599    let file = "/tmp/p04os-dir/p04os-file";
600    let mkdir = run(&["mkdir", dir]);
601    let touch = run(&["touch", file]);
602    let listed = run(&["ls", dir]);
603    let shell = a.run(&format!("ls {dir}"), 30_000);
604    let rm = run(&["rm", file]);
605    let gone = run(&["ls", dir]);
606    let has = |r: &Result<Ran, Status>| matches!(r, Ok(r) if lines(&r.out).iter().any(|l| *l == b"p04os-file"));
607    check(
608        "oneshot-mutate-effect",
609        matches!(&mkdir, Ok(r) if r.code() == Some(0))
610            && matches!(&touch, Ok(r) if r.code() == Some(0))
611            && has(&listed)
612            && matches!(&shell, Ok(l) if l.iter().any(|l| l == "p04os-file"))
613            && matches!(&rm, Ok(r) if r.code() == Some(0))
614            && matches!(&gone, Ok(r) if r.code() == Some(0))
615            && !has(&gone),
616        &format!(
617            "mkdir {} touch {} ls {} shell={} rm {} after {}",
618            ran(&mkdir),
619            ran(&touch),
620            ran(&listed),
621            shell.is_ok(),
622            ran(&rm),
623            ran(&gone)
624        ),
625    );
626}

628/// The S2-K secret namespace and a write into it are refused exactly as in the shell:
629/// same nonzero status, the fixture bytes on no stream, nothing left behind; a one-shot
630/// cannot even reach the S2-T/W sandbox probe (no launch, 127).
631fn denials_same_as_shell(a: &mut Term) {
632    let mut bad = Vec::new();
633    let token = SECRET_TOKEN.as_bytes();
634    for cmd in [
635        "cat /srv/cellos/p04-secret",
636        "grep p04 /srv/cellos/p04-secret",
637        "head -n 1 /srv/cellos/p04-secret",
638        "wc /srv/cellos/p04-secret",
639        "sort /srv/cellos/p04-secret",
640        // The shell echoes these command lines in INFO lines: no `p04os-` token here.
641        "cp /srv/cellos/p04-secret /tmp/p04-osleak",
642        "ls /srv/cellos",
643        "vwrite /srv/cellos/p04-osw p04-osw",
644    ] {
645        let shell_out = a.run(cmd, 30_000);
646        let shell_err = a.stderr_text().replace('\r', "");
647        let shell_status = a.run("echo sos=$?", 30_000);
648        let argv: Vec<&str> = cmd.split(' ').collect();
649        let r = run(&argv);
650        // Same status and the same bytes as the shell (its transcript has the echoed line
651        // and prompt removed); a refused read or copy is nonzero, the legacy `vwrite`
652        // helper reports its refused write with status 0.
653        let same = match (&shell_status, &shell_out, &r) {
654            (Ok(l), Ok(shell_lines), Ok(r)) => {
655                let text = String::from_utf8_lossy(&r.out);
656                let ours: Vec<&str> = text.lines().filter(|line| !line.is_empty()).collect();
657                let theirs: Vec<&str> = shell_lines
658                    .iter()
659                    .map(String::as_str)
660                    .filter(|line| !line.is_empty())
661                    .collect();
662                l.len() == 1
663                    && r.code().is_some_and(|code| {
664                        l[0] == format!("sos={code}") && (code != 0 || argv[0] == "vwrite")
665                    })
666                    && ours == theirs
667                    && r.err == shell_err.as_bytes()
668                    && !(r.out.is_empty() && r.err.is_empty())
669            }
670            _ => false,
671        };
672        let leaked = matches!(&r, Ok(r) if r.out.windows(token.len()).any(|w| w == token) || r.err.windows(token.len()).any(|w| w == token));
673        if !same || leaked {
674            bad.push(format!(
675                "{cmd}: shell={shell_status:?} shell_err={} {} leaked={leaked}",
676                hexs(shell_err.as_bytes()),
677                ran(&r)
678            ));
679        }
680    }
681    let leak = run(&["cat", "/tmp/p04-osleak"]);
682    let probe = run(&["/data/bin/s2tw-probe", "sb"]);
683    let clean = matches!(&leak, Ok(r) if r.code() != Some(0) && !r.out.windows(token.len()).any(|w| w == token))
684        && is(
685            &probe,
686            127,
687            b"",
688            b"cellbox: /data/bin/s2tw-probe: unknown command\n",
689        );
690    check(
691        "oneshot-denials-same-as-shell",
692        bad.is_empty() && clean,
693        &format!("{bad:?} leak {} probe {}", ran(&leak), ran(&probe)),
694    );
695}

715/// Without a Stdout grant nothing is written anywhere: not stderr, not the UART (the
716/// runner checks the `p04os-` token).
717fn missing_stdout() {

719        stdout: false,

721    });

730            && matches!(&both, Ok(r) if r.code().is_some_and(|c| c != 0) && r.out.is_empty() && r.err.is_empty()),
731        &format!("stderr-only {} none {}", ran(&r), ran(&both)),
732    );

735/// A reader that closes after the first chunk stops a streamed `rg` at once, long before
736/// the 30 s write deadline: the kernel stops a producer whose final consumer closed
737/// (stop reason 3, which a parent shell reports as 141) instead of letting it write on.
738fn consumer_close() {
739    // > 64 KiB of matching lines, written to RamFS by a one-shot `tee` over its stdin.
740    let mut big = Vec::new();
741    let mut n = 0u32;
742    while big.len() <= 72 * 1024 {
743        big.extend_from_slice(format!("p04os-line {n:06} needle\n").as_bytes());
744        n += 1;
745    }
746    let path = "/tmp/p04os-big.txt";
747    let tee = launch(&Spec {
748        stdin: Some(&big),
749        ..Spec::new(&["tee", path])
750    });
751    let tee_ok = matches!(&tee, Ok(r) if r.code() == Some(0) && r.out == big && r.quiet_err());
752    let r = launch(&Spec {
753        close_stdout_after: Some(1),
754        timeout_ms: 60_000,
755        ..Spec::new(&["rg", "needle", path])
756    });
757    let reported = match &r {
758        Ok(Ran { exit: Err(3), .. }) => Some(141),
759        Ok(Ran { exit: Err(_), .. }) => Some(130),
760        Ok(Ran { exit: Ok(code), .. }) => Some(*code),
761        Err(_) => None,
762    };
763    println(&format!(
764        "P04 INFO oneshot-consumer-close lines={n} reported={reported:?} {}",
765        ran(&r)
766    ));
767    check(
768        "oneshot-consumer-close",
769        tee_ok
770            && matches!(&r, Ok(r) if r.exit == Err(3) && !r.out.is_empty() && r.out.len() < 64 * 1024 && r.elapsed_ms < PROMPT_EXIT_MS)
771            && reported == Some(141),
772        &format!("tee_ok={tee_ok} rg {}", ran(&r)),
773    );
774}

776/// Cancelling the one-shot's Operation during `sleep 30` ends it promptly with a
777/// cancellation record (reported as 130), and no `cellbox` task is left behind.
778fn cancel(a: &mut Term) {
779    let before = tasks_named(a, "cellbox");
780    let r = launch(&Spec {
781        cancel_after_ms: Some(1500),
782        timeout_ms: 25_000,
783        ..Spec::new(&["sleep", "30"])
784    });
785    sleep_ms(2000);
786    let after = tasks_named(a, "cellbox");
787    let reported = match &r {
788        Ok(Ran { exit: Err(_), .. }) => Some(130),
789        Ok(Ran { exit: Ok(code), .. }) => Some(*code),
790        Err(_) => None,
791    };
792    println(&format!(
793        "P04 INFO oneshot-cancel {} reported={reported:?}",
794        ran(&r)
795    ));
796    check(
797        "oneshot-cancel",
798        matches!(&r, Ok(r) if r.exit.is_err() && r.elapsed_ms < PROMPT_EXIT_MS)
799            && reported == Some(130)
800            && before.is_ok()
801            && before == after,
802        &format!("{} cellbox {before:?} -> {after:?}", ran(&r)),
803    );
804}

806/// Exit code, stdout and stderr bytes of one refused frontend launch.
807type FrontendExit = (Option<i32>, Vec<u8>, Vec<u8>);
808
809/// A terminal frontend runs only `sh`: any other argv exits 2 with one line on the
810/// holder's stderr writer and never attaches.
811fn frontend_argv_refused() {
812    let result = (|| -> Result<FrontendExit, String> {
813        let mut t = Term::open(24, 80).map_err(|s| format!("open {s:?}"))?;
814        t.host
815            .launch(SHELL, &["ls", "/p04-list"], &[])
816            .map_err(|s| format!("launch {s:?}"))?;
817        let exited = wait_exit(&mut t, 60_000);
818        let code = t.host.exit_code();
819        let (out, err) = (core::mem::take(&mut t.out), core::mem::take(&mut t.err));
820        let _ = t.host.close();
821        exited?;
822        Ok((code, out, err))
823    })();
824    check(
825        "oneshot-frontend-argv-refused",
826        matches!(&result, Ok((Some(2), out, err)) if out.is_empty() && err.as_slice() == FRONTEND_ONLY_SH),
827        &match &result {
828            Ok((code, out, err)) => format!("code={code:?} out={} err={}", hexs(out), hexs(err)),
829            Err(why) => why.clone(),
830        },
831    );
832}

834/// In `cellbox sh`: `cellbox CMD` and `/bin/cellbox CMD` are the short forms in process;
835/// a refused name answers like the one-shot, with no effect.
836fn shell_explicit_prefix(a: &mut Term) {
837    let short = a.run("echo p04os-hi", 30_000);
838    let long = a.run("cellbox echo p04os-hi", 30_000);
839    let ls_short = a.run("ls /p04-list", 30_000);
840    let ls_long = a.run("/bin/cellbox ls /p04-list", 30_000);
841    let st_ls = a.run("echo stexl=$?", 30_000);
842    let pwd_before = a.run("pwd", 30_000);
843    let cd = a.run("cellbox cd /p04-list", 30_000);
844    let cd_err = a.stderr_text();
845    let st_cd = a.run("echo stexcd=$?", 30_000);
846    let pwd_after = a.run("pwd", 30_000);
847    let typed = a.run(
848        "cellbox ps --records | select tid | take 1 | to jsonl",
849        30_000,
850    );
851    let st_typed = a.run("echo stexty=$?", 30_000);
852    let ok = matches!((&short, &long), (Ok(s), Ok(l)) if s == l && s == &[String::from("p04os-hi")])
853        && matches!((&ls_short, &ls_long), (Ok(s), Ok(l)) if s == l && s.iter().any(|x| x == "alpha-entry"))
854        && matches!(&st_ls, Ok(l) if l == &[String::from("stexl=0")])
855        && cd.is_ok()
856        && cd_err.contains("cellbox: cd changes or reads shell state")
857        && matches!(&st_cd, Ok(l) if l == &[String::from("stexcd=2")])
858        && pwd_before.is_ok()
859        && pwd_before == pwd_after
860        && matches!(&typed, Ok(l) if rows_of(l).len() == 1)
861        && matches!(&st_typed, Ok(l) if l == &[String::from("stexty=0")]);
862    check(
863        "shell-explicit-prefix",
864        ok,
865        &format!(
866            "echo={} ls={} st_ls={st_ls:?} cd_st={st_cd:?} pwd={} typed={:?} st_typed={st_typed:?}",
867            short == long,
868            ls_short == ls_long,
869            pwd_before == pwd_after,
870            typed.map(|l| rows_of(&l).len())
871        ),
872    );
