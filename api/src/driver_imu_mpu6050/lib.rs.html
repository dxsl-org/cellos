37impl<I2C: ViI2c> Mpu6050<I2C> {
38    pub fn new(i2c: I2C, addr: u8) -> Self {
39        Self { i2c, addr }
40    }
41
42    pub fn init(&mut self) -> Result<u8, I2C::Error> {
43        let mut who_am_i = [0u8; 1];
44        self.i2c
45            .write_read(self.addr, &[REG_WHO_AM_I], &mut who_am_i)?;
46        let id = who_am_i[0];
47        if id != WHO_AM_I_EXPECTED {
48            return Ok(id);
49        }
50
51        // Wake device from sleep mode (clear SLEEP bit 6 in PWR_MGMT_1)
52        self.i2c.write(self.addr, &[REG_PWR_MGMT_1, 0x00])?;
53        // Set sample rate divider (1 kHz / (1 + 7) = 125 Hz)
54        self.i2c.write(self.addr, &[REG_SMPLRT_DIV, 0x07])?;
55        // DLPF cfg = 3 (44 Hz Accel, 42 Hz Gyro bandwidth)
56        self.i2c.write(self.addr, &[REG_CONFIG, 0x03])?;
57        // Gyro range = ±500 deg/s (FS_SEL = 1)
58        self.i2c.write(self.addr, &[REG_GYRO_CONFIG, 0x08])?;
59        // Accel range = ±2g (AFS_SEL = 0)
60        self.i2c.write(self.addr, &[REG_ACCEL_CONFIG, 0x00])?;
61
62        Ok(id)
63    }
64
65    pub fn read_sensor(&mut self) -> Result<ImuData, I2C::Error> {
66        let mut buf = [0u8; 14];
67        self.i2c
68            .write_read(self.addr, &[REG_ACCEL_XOUT_H], &mut buf)?;
69
70        let accel_x = i16::from_be_bytes([buf[0], buf[1]]);
71        let accel_y = i16::from_be_bytes([buf[2], buf[3]]);
72        let accel_z = i16::from_be_bytes([buf[4], buf[5]]);
73        let temp_raw = i16::from_be_bytes([buf[6], buf[7]]);
74        let gyro_x = i16::from_be_bytes([buf[8], buf[9]]);
75        let gyro_y = i16::from_be_bytes([buf[10], buf[11]]);
76        let gyro_z = i16::from_be_bytes([buf[12], buf[13]]);
77
78        Ok(ImuData {
79            accel_x,
80            accel_y,
81            accel_z,
82            temp_raw,
83            gyro_x,
84            gyro_y,
85            gyro_z,
86        })
87    }
