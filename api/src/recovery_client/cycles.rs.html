13/// E6: corrupt, missing, wrong-schema and wrong-revision state each end in
14/// RecoveryFailed with no published replacement; healing restores service.
15pub fn restore_refusals(client: &mut Client) {
16    for (kind, label) in [
17        (corrupt::CHECKSUM, "corrupt"),
18        (corrupt::MISSING, "missing"),
19        (corrupt::SCHEMA, "wrong-schema"),
20        (corrupt::REVISION, "wrong-revision"),
21    ] {
22        let before = client.get();
23        let old = client.worker;
24        if client.authority_call(Msg::Corrupt { kind }) != Some(Msg::Ack) {
25            fail(&format!("E6 {label}: inject"));
26        }
27        client.wait_status(label, |s| {
28            s.state == state::RECOVERY_FAILED && s.worker == 0
29        });
30        if common::lookup(WORKER_SERVICE).is_some() {
31            fail(&format!("E6 {label}: a replacement was published"));
32        }
33        println(&format!("[recovery-client] E6 PASS {label}: RecoveryFailed, no Ready (worker service unpublished)"));
34        if client.authority_call(Msg::Heal) != Some(Msg::Ack) {
35            fail(&format!("E6 {label}: heal"));
36        }
37        client.await_replacement_after_failure(old);
38        let after = client.get();
39        if (after.0, after.1) != (before.0, before.1) {
40            fail(&format!("E6 {label}: healed state differs"));
41        }
42    }
43}

45/// E11: every published replacement is a new `(cell, generation)`.
46pub fn identities(client: &Client) {
47    let seen = &client.incarnations;
48    let mut distinct: Vec<(u64, u64)> = Vec::new();
49    for id in seen {
50        if distinct.contains(id) {
51            fail(&format!("E11: incarnation {id:?} published twice"));
52        }
53        distinct.push(*id);
54    }
55    if distinct.len() < 2 {
56        fail("E11: no replacement observed");
57    }
58    println(&format!("[recovery-client] E11 PASS {} published worker incarnations, all distinct (cell, generation); stale epochs refused (E5)",
59        distinct.len()));
60}

62/// E12: ten crash/restore cycles. Settled free frames must not drift and every
63/// retirement must be Contained with nothing quarantined or leaked.
64pub fn ten_cycles(client: &mut Client) -> u32 {
65    let ledger = client.status().ledger;
66    let mut settled: Vec<u64> = Vec::new();
67    let mut remote = 0;
68    for cycle in 1..=10 {
69        let before = free_frames();
70        let v = client.crash(crash::SPIN, 0);
71        common::sleep(20);
72        let _ = client.get();
73        let after = free_frames();
74        let status = client.status();
75        println(&format!(
76            "[recovery-client] E12 cycle={cycle} free_before={before} free_after={after} state={} threads={} remote_harts={:#x} grants={} qpages={} leaked={} shm={} ledger={}",
77            state_name(&v), v.member_threads, v.remote_hart_mask, v.grants_reclaimed, v.quarantined_pages,
78            v.leaked_pages, v.shm_frames_released, status.ledger
79        ));
80        if state_name(&v) != "Contained"
81            || v.quarantined_pages != 0
82            || v.leaked_pages != 0
83            || status.ledger != ledger
84        {
85            fail(&format!("E12 cycle {cycle}: unreconciled retirement"));
86        }
87        remote += u32::from(v.remote_hart_mask != 0);
88        settled.push(after);
89    }
90    let steady = &settled[1..];
91    let (low, high) = (
92        steady.iter().min().copied().unwrap_or(0),
93        steady.iter().max().copied().unwrap_or(0),
94    );
95    if high != low {
96        fail(&format!(
97            "E12: settled free frames drift {low}..{high} over cycles 2-10"
98        ));
99    }
100    println(&format!("[recovery-client] E12 PASS 10 cycles: all Contained, 0 quarantined, 0 leaked, settled free frames {low} constant (cycles 2-10), ledger {ledger}"));
101    remote
102}
