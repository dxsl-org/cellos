12fn expect_contained(step: &str, v: &api::retirement::ViRetirementV1) {
13    if state_name(v) != "Contained" {
14        fail(&format!(
15            "{step}: retirement {} reasons={:#x}",
16            state_name(v),
17            v.reasons
18        ));
19    }
20}

22/// E1 fault before commit, E2 fault after commit before reply.
23pub fn commit_faults(client: &mut Client) {
24    let (base, _, _) = client.get();
25    let old = client.worker;
26    if client
27        .mutate(1001, METHOD_INC, 1, Fault::BeforeCommit)
28        .is_some()
29    {
30        fail("E1: crashed worker replied");
31    }
32    expect_contained("E1", &client.await_replacement(old));
33    match client.mutate(1001, METHOD_INC, 1, Fault::None) {
34        Some(Msg::Done {
35            replayed: false,
36            result,
37            ..
38        }) if result == base + 1 => {}
39        other => fail(&format!("E1 resubmit: {other:?}")),
40    }
41    if client.get().0 != base + 1 {
42        fail("E1: committed value");
43    }
44    println(&format!("[recovery-client] E1 PASS fault-before-commit: old revision kept, resubmit applied once (counter {})", base + 1));
45
46    let old = client.worker;
47    if client
48        .mutate(1002, METHOD_INC, 1, Fault::AfterCommit)
49        .is_some()
50    {
51        fail("E2: crashed worker replied");
52    }
53    expect_contained("E2", &client.await_replacement(old));
54    match client.mutate(1002, METHOD_INC, 1, Fault::None) {
55        Some(Msg::Done {
56            replayed: true,
57            result,
58            ..
59        }) if result == base + 2 => {}
60        other => fail(&format!("E2 resubmit: {other:?}")),
61    }
62    if client.get().0 != base + 2 {
63        fail("E2: reapplied");
64    }
65    println(&format!("[recovery-client] E2 PASS fault-after-commit: recorded outcome returned, not reapplied (counter {})", base + 2));
66}

68/// E3 same id different payload, E4 same id different method (authority).
69pub fn conflicts(client: &mut Client) {
70    let before = client.get();
71    for (step, method, amount) in [("E3", METHOD_INC, 5), ("E4", METHOD_ADD_TWICE, 1)] {
72        match client.mutate(1002, method, amount, Fault::None) {
73            Some(Msg::Failed {
74                refusal: Refusal::Conflict,
75                ..
76            }) => {}
77            other => fail(&format!("{step}: {other:?}")),
78        }
79        if client.get() != before {
80            fail(&format!("{step}: state changed"));
81        }
82    }
83    println("[recovery-client] E3 PASS same request id with a different payload refused (Conflict), state unchanged");
84    println("[recovery-client] E4 PASS same request id with a different method/authority refused (Conflict), state unchanged");
85}

87/// E5: a live writer fenced at the authority cannot commit afterwards.
88pub fn fenced_writer(client: &mut Client) {
89    let (counter, revision, epoch) = client.get();
90    let old = client.worker;
91    match client.mutate(1003, METHOD_INC, 1, Fault::Stall) {
92        Some(Msg::Staged { request_id: 1003 }) => {}
93        other => fail(&format!("E5 stage: {other:?}")),
94    }
95    if client.authority_call(Msg::Fence) != Some(Msg::Ack) {
96        fail("E5: fence");
97    }
98    // `Go` is its own request: the stalled worker answers it with the commit outcome.
99    match crate::common::call(old, Msg::Go, 400) {
100        Some(Msg::Failed {
101            refusal: Refusal::StaleEpoch,
102            ..
103        }) => {}
104        other => fail(&format!("E5 delayed commit: {other:?}")),
105    }
106    expect_contained("E5", &client.await_replacement(old));
107    let (after, after_revision, after_epoch) = client.get();
108    if (after, after_revision) != (counter, revision) || after_epoch <= epoch {
109        fail("E5: stale writer changed state or epoch did not advance");
110    }
111    println(&format!("[recovery-client] E5 PASS old writer refused StaleEpoch after fence; revision {revision} kept; epoch {epoch} -> {after_epoch}"));
112}

114/// E7: helper thread spinning on another hart while the root faults.
115pub fn whole_cell(client: &mut Client) -> u64 {
116    let v = client.crash(crash::SPIN, 0);
117    expect_contained("E7", &v);
118    if v.member_threads < 2 {
119        fail(&format!(
120            "E7: only {} member thread(s) retired",
121            v.member_threads
122        ));
123    }
124    println(&format!("[recovery-client] E7 PASS whole-Cell retirement: threads={} remote_harts={:#x} state=Contained",
125        v.member_threads, v.remote_hart_mask));
126    v.remote_hart_mask
127}

129/// E10: the dead writer leaves a set lock word in a page it borrowed writable.
130pub fn abandoned_lock(client: &mut Client) {
131    let grant = match client.authority_call(Msg::Lend) {
132        Some(Msg::Lent { grant }) => grant,
133        other => fail(&format!("E10 lend: {other:?}")),
134    };
135    let before = client.get();
136    let v = client.crash(crash::LOCK_AND_DIE, grant);
137    if v.writable_borrows == 0 || v.reasons & api::retirement::reason::WRITABLE_BORROW == 0 {
138        fail(&format!(
139            "E10: writable borrow not reported ({:#x})",
140            v.reasons
141        ));
142    }
143    let status = client.status();
144    if status.lock_word != 1 {
145        fail(&format!("E10: lock word {}", status.lock_word));
146    }
147    let after = client.get();
148    if (after.0, after.1) != (before.0, before.1) {
149        fail("E10: committed state changed");
150    }
151    println(&format!("[recovery-client] E10 PASS writable borrow reported ({}), lock word 1 left held and page discarded, state restored from committed revision {}",
152        v.writable_borrows, after.1));
153}
