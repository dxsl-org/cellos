1//! Separate client invocations: E4 foreign principal, E8 missing borrower
2//! acknowledgement (needs the `recovery-fixture` VFS), E9 untracked export
3//! (terminal for the boot: the provider then needs node recovery).

14/// E4 (second principal): reusing another caller's request id is a separate
15/// key; it neither replays nor conflicts with that caller's binding.
16pub fn foreign(client: &mut Client, request_id: u64) {
17    let (counter, _, _) = client.get();
18    match client.mutate(request_id, METHOD_INC, 1, Fault::None) {
19        Some(Msg::Done {
20            replayed: false,
21            result,
22            ..
23        }) if result == counter + 1 => {}
24        other => fail(&format!("E4 foreign principal: {other:?}")),
25    }
26    println(&format!("[recovery-client] E4 PASS foreign principal reusing request id {request_id}: independent binding, applied once (counter {})", counter + 1));
27}

29/// E8: the worker dies while VFS still holds its grant lease. The pages stay
30/// quarantined (charged, counted) until VFS completes, then return.
31pub fn missing_ack(client: &mut Client) {
32    let v = client.crash(crash::HOLD_LEASE, 0);
33    if state_name(&v) != "Quarantined" || (v.vfs_leases_pending == 0 && v.quarantined_pages == 0) {
34        fail(&format!(
35            "E8: retirement {} vfs={} qpages={}",
36            state_name(&v),
37            v.vfs_leases_pending,
38            v.quarantined_pages
39        ));
40    }
41    let held = free_frames();
42    println(&format!("[recovery-client] E8 quarantined: state=Quarantined vfs_leases={} qpages={} reasons={:#x} free_while_held={held}",
43        v.vfs_leases_pending, v.quarantined_pages, v.reasons));
44    for _ in 0..200 {
45        common::sleep(10);
46        let now = free_frames();
47        if now >= held + v.quarantined_pages {
48            println(&format!("[recovery-client] E8 PASS missing borrower ack quarantined {} page(s); released after VFS completed (free {held} -> {now})",
49                v.quarantined_pages));
50            return;
51        }
52    }
53    fail("E8: quarantined pages never returned");
54}

56/// E9: an untracked SHM export makes containment unproven: no restart, no
57/// Ready, and healing is refused (node recovery required).
58pub fn untracked_export(client: &mut Client) {
59    if client.worker_call(Msg::Crash {
60        mode: crash::SHM_EXPORT,
61        arg: 0,
62    }) != Some(Msg::Ack)
63    {
64        fail("E9: crash not acknowledged");
65    }
66    let status = client.wait_status("node recovery required", |s| {
67        s.state == state::NODE_RECOVERY_REQUIRED && s.worker == 0
68    });
69    let Some(v) = status.verdict else {
70        fail("E9: no verdict")
71    };
72    if state_name(&v) != "Unproven"
73        || v.reasons & abi::retirement::reason::SHM_UNTRACKED == 0
74        || v.shm_frames_released == 0
75    {
76        fail(&format!(
77            "E9: retirement {} reasons={:#x}",
78            state_name(&v),
79            v.reasons
80        ));
81    }
82    common::sleep(100);
83    if common::lookup(WORKER_SERVICE).is_some() {
84        fail("E9: a replacement was published");
85    }
86    match client.authority_call(Msg::Heal) {
87        Some(Msg::Refused(Refusal::Denied)) => {}
88        other => fail(&format!("E9 heal: {other:?}")),
89    }
90    println(&format!("[recovery-client] E9 PASS untracked SHM export ({} frame(s)): Unproven, no restart, no Ready, heal refused (node recovery required)",
91        v.shm_frames_released));
92}
