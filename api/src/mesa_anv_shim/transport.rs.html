1//! How the shim reaches the Intel GPU driver Cell.
2
3use intel_gpu_core::uapi::ioctl::{EAGAIN, EINVAL, EIO, ENODEV};
4use ostd::ipc::{Completion, IpcError, IpcSubmitError, IpcTerminal, PendingCall};
5
6/// Scheduler ticks (10 ms) per wait round.
7const WAIT_ROUND_TICKS: u64 = 100;
8
9/// One request/reply exchange with the `XE_GPU` service.
10pub trait Transport {
11    /// Send `request` and wait for the reply; returns the reply length. Errors are errnos.
12    fn call(&mut self, request: &[u8], reply: &mut [u8]) -> Result<usize, i32>;
13}
14
15/// `XE_GPU` over kernel-bounded IPC operations (`IpcSubmit` + `IpcTake`, CP1 D-G6): the
16/// driver never blocks on this client, and its death ends the wait with `ENODEV`.
17pub struct IpcTransport {
18    provider: usize,
19}
20
21impl IpcTransport {
22    /// The live `XE_GPU` provider (`service::XE_GPU`, registered by init per CP1 D-G8).
23    pub fn connect() -> Result<Self, i32> {
24        ostd::service::provider_tid(abi::syscall::service::XE_GPU)
25            .map(|provider| Self { provider })
26            .ok_or(ENODEV)
27    }
28}
29
30impl Transport for IpcTransport {
31    fn call(&mut self, request: &[u8], reply: &mut [u8]) -> Result<usize, i32> {
32        let call = PendingCall::submit(self.provider, request).map_err(|e| match e {
33            IpcSubmitError::Busy => EAGAIN,
34            IpcSubmitError::PeerGone => ENODEV,
35            IpcSubmitError::InvalidRequest => EINVAL,
36        })?;
37        // No caller deadline: the kernel ends every operation with a terminal (its own 30 s
38        // deadline at the latest), so re-park until it arrives.
39        loop {
40            match call.wait_and_take(reply, WAIT_ROUND_TICKS, 1) {
41                Ok(None) => {}
42                Ok(Some(Completion {
43                    terminal: IpcTerminal::Reply,
44                    len,
45                })) => return Ok(len),
46                Ok(Some(Completion {
47                    terminal: IpcTerminal::PeerGone,
48                    ..
49                }))
50                | Err(IpcError::PeerGone) => return Err(ENODEV),
51                Ok(Some(_)) | Err(_) => return Err(EIO),
52            }
53        }
54    }
55}
