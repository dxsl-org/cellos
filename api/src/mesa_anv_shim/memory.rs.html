1//! Client memory behind GEM objects (CP1 D-G1 option B): the shim allocates a buffer in
2//! this Cell, exports it to the `XE_GPU` driver (`ExportDmaRange`, 437), and revokes and
3//! frees it once the object is gone (`RevokeDmaExport`, 439).
4
5use alloc::collections::BTreeMap;
6use ostd::dma::DmaBuf;
7use ostd::syscall::DmaExportError;
8
9/// A buffer of this Cell's own memory.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub struct ClientBuffer {
12    /// Allocator-private name.
13    pub id: usize,
14    /// Where the CPU reaches byte 0 in this Cell.
15    pub addr: u64,
16    pub size: usize,
17}
18
19/// Why a revoke did not finish.
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum RevokeError {
22    /// The kernel has not confirmed the device drain yet; the range stays locked and the
23    /// kernel retries on its own. Ask again later.
24    Pending,
25}
26
27/// Allocation and export of client memory. Production: [`DmaClientMemory`]; host tests
28/// supply a fake.
29pub trait ClientMemory {
30    /// `size` bytes (a page multiple) of zero-filled memory, or `None` if this Cell cannot
31    /// back them.
32    fn alloc(&mut self, size: usize) -> Option<ClientBuffer>;
33    /// Export `buf` to the GPU driver; `device_write` lets the device write it.
34    fn export(&mut self, buf: &ClientBuffer, device_write: bool) -> Option<u64>;
35    /// Revoke an export as its owner. `Ok` once the device can no longer reach it, or
36    /// when the lease is already gone (driver or kernel revoked it).
37    fn revoke(&mut self, export_id: u64) -> Result<(), RevokeError>;
38    /// Return the memory. Only called after its export is revoked.
39    fn free(&mut self, buf: ClientBuffer);
40}
41
42/// [`ClientMemory`] from Grants: `DmaBuf::alloc` (zeroed, physically contiguous) and
43/// `DmaBuf::export` to `service::XE_GPU`. A Tier-2 private-root Cell gets no Grant and no
44/// export (D-G0 large memory is not implemented), so its allocations fail.
45#[derive(Default)]
46pub struct DmaClientMemory {
47    buffers: BTreeMap<usize, DmaBuf>,
48}
49
50impl ClientMemory for DmaClientMemory {
51    fn alloc(&mut self, size: usize) -> Option<ClientBuffer> {
52        let buf = DmaBuf::alloc(size.div_ceil(4096))?;
53        let handle = ClientBuffer {
54            id: buf.phys(),
55            addr: buf.phys() as u64,
56            size,
57        };
58        self.buffers.insert(handle.id, buf);
59        Some(handle)
60    }
61
62    fn export(&mut self, buf: &ClientBuffer, device_write: bool) -> Option<u64> {
63        self.buffers
64            .get(&buf.id)?
65            .export(0, buf.size, abi::syscall::service::XE_GPU, device_write)
66            .ok()
67    }
68
69    fn revoke(&mut self, export_id: u64) -> Result<(), RevokeError> {
70        match ostd::dma::revoke_export(export_id) {
71            // Revoked now, or already gone with the driver or by its own revoke.
72            Ok(()) | Err(DmaExportError::NotFound) => Ok(()),
73            // Undrained, or refused: the lease may still be live, so the memory must not
74            // be freed yet.
75            Err(_) => Err(RevokeError::Pending),
76        }
77    }
78
79    fn free(&mut self, buf: ClientBuffer) {
80        if let Some(buf) = self.buffers.remove(&buf.id) {
81            buf.free();
82        }
83    }
84}
