1//! Broadcom BCM57xx MII PHY communication under APE PHY lock.
2
3use crate::ape::{self, with_ape_lock};
4use crate::regs::{
5    RegisterFile, MICOMM_BUSY, MICOMM_CMD_READ, MICOMM_CMD_WRITE, MICOMM_DATA_MASK,
6    MICOMM_PHY_SHIFT, MICOMM_READFAIL, MICOMM_REG_SHIFT, MI_COMM,
7};
8use types::{ViError, ViResult};
9
10pub(crate) const MII_BMCR: u32 = 0;
11pub(crate) const BMCR_SPEED1000: u16 = 1 << 6;
12pub(crate) const BMCR_FULL_DUPLEX: u16 = 1 << 8;
13pub(crate) const BMCR_ANRESTART: u16 = 1 << 9;
14pub(crate) const BMCR_ANENABLE: u16 = 1 << 12;
15
16pub(crate) const BCM57XX_PHY_ADDR: u32 = 1;
17
18const MI_POLLS: u32 = 50_000;
19
20fn mi_access(regs: &impl RegisterFile, cmd: u32) -> ViResult<u16> {
21    regs.wr32(MI_COMM, cmd | MICOMM_BUSY)?;
22    for _ in 0..MI_POLLS {
23        let val = regs.rd32(MI_COMM)?;
24        if val & MICOMM_BUSY == 0 {
25            if val & MICOMM_READFAIL != 0 {
26                return Err(ViError::IO);
27            }
28            return Ok((val & MICOMM_DATA_MASK) as u16);
29        }
30        core::hint::spin_loop();
31    }
32    Err(ViError::IO)
33}
34
35pub(crate) fn phy_read(regs: &impl RegisterFile, func: u8, reg: u32) -> ViResult<u16> {
36    let locknum = ape::phy_lock_for_func(func);
37    with_ape_lock(regs, func, locknum, || {
38        let cmd = MICOMM_CMD_READ
39            | (BCM57XX_PHY_ADDR << MICOMM_PHY_SHIFT)
40            | ((reg & 0x1F) << MICOMM_REG_SHIFT);
41        mi_access(regs, cmd)
42    })
43}
44
45pub(crate) fn phy_write(regs: &impl RegisterFile, func: u8, reg: u32, data: u16) -> ViResult<()> {
46    let locknum = ape::phy_lock_for_func(func);
47    with_ape_lock(regs, func, locknum, || {
48        let cmd = MICOMM_CMD_WRITE
49            | (BCM57XX_PHY_ADDR << MICOMM_PHY_SHIFT)
50            | ((reg & 0x1F) << MICOMM_REG_SHIFT)
51            | (data as u32 & MICOMM_DATA_MASK);
52        mi_access(regs, cmd).map(|_| ())
53    })
54}
55
56pub(crate) fn restart_autoneg(regs: &impl RegisterFile, func: u8) -> ViResult<()> {
57    let bmcr = phy_read(regs, func, MII_BMCR)?;
58    phy_write(
59        regs,
60        func,
61        MII_BMCR,
62        bmcr | BMCR_ANENABLE | BMCR_ANRESTART | BMCR_SPEED1000 | BMCR_FULL_DUPLEX,
63    )
64}
