1//! Broadcom BCM57xx controller bring-up, APE handshake, and data path.

3use crate::ape::{self, with_ape_lock};
4use crate::dma_layout::{
5    for_each_initial_dma_program, try_init_array, with_authorized_dma_layout, DmaIovas, DmaSlot,

8use crate::phy;
9use crate::regs::*;
10use crate::rings::{self, BcmRxDesc, BcmTxDesc};

12use ostd::dma::DmaBuf;

16pub use crate::rings::BUF_SIZE;
17
18const RESET_POLLS: u32 = 1_000_000;
19
20pub struct BcmController {
21    mmio: MmioRegion,
22    tx_ring: DmaBuf,
23    rx_ring: DmaBuf,
24    #[allow(dead_code)]
25    status_block: DmaBuf,
26    tx_bufs: [DmaBuf; TX_SLOTS],
27    rx_bufs: [DmaBuf; RX_SLOTS],
28    dma_iovas: DmaIovas,
29    tx_next: usize,
30    rx_head: usize,
31    pub mac: [u8; 6],
32    pub link_up: bool,
33    pub func: u8,
34}
35
36unsafe impl Send for BcmController {}
37
38impl Drop for BcmController {
39    fn drop(&mut self) {
40        // Unload notification to APE processor prevents dropping BMC NC-SI link.
41        let _ = ape::notify_driver_state(&self.mmio, self.func, ape::APE_HOST_DRVR_STATE_UNLOAD);
42    }
43}
44
45impl BcmController {
46    pub fn new(mmio: MmioRegion, bdf: u32) -> ViResult<Self> {
47        let func = (bdf & 0x07) as u8;
48        // 1. Notify APE processor that driver is starting.
49        ape::notify_driver_state(&mmio, func, ape::APE_HOST_DRVR_STATE_START)?;
50
51        // 2. Perform GRC core reset under GRC lock.
52        with_ape_lock(&mmio, func, ape::APE_LOCK_GRC, || {
53            let mut val = mmio.read_u32(MISC_CFG)?;
54            val |= MISCCFG_RESET_CORE_CLOCKS;
55            mmio.write_u32(MISC_CFG, val)?;
56
57            for _ in 0..RESET_POLLS {
58                if mmio.read_u32(MISC_CFG)? & MISCCFG_RESET_CORE_CLOCKS == 0 {
59                    return Ok(());
60                }
61                core::hint::spin_loop();
62            }
63            Err(ViError::IO)
64        })?;
65
66        // 3. Read station MAC address from MAC_ADDR1 registers.
67        let lo = mmio.read_u32(MAC_ADDR1_LO)?;
68        let hi = mmio.read_u32(MAC_ADDR1_HI)?;
69        let mac = [
70            (lo >> 24) as u8,
71            (lo >> 16) as u8,
72            (lo >> 8) as u8,
73            lo as u8,
74            (hi >> 8) as u8,
75            hi as u8,
76        ];
77        if mac == [0u8; 6] {
78            return Err(ViError::IO);
79        }
80
81        // 4. Restart PHY autonegotiation under PHY lock.
82        phy::restart_autoneg(&mmio, func)?;
83
84        // 5. Allocate and authorize DMA.
85        let tx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
86        let tx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
87        let rx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
88        let rx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
89        let status_block = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
90
91        with_authorized_dma_layout(
92            (mmio, tx_ring, rx_ring, status_block, tx_bufs, rx_bufs),
93            |r, s| {
94                match s {
95                    DmaSlot::TxRing => r.1.authorize(bdf),
96                    DmaSlot::RxRing => r.2.authorize(bdf),
97                    DmaSlot::StatusBlock => r.3.authorize(bdf),
98                    DmaSlot::TxBuffer(i) => r.4[i].authorize(bdf),
99                    DmaSlot::RxBuffer(i) => r.5[i].authorize(bdf),
100                }
101                .map_err(|_| ViError::PermissionDenied)
102            },
103            |(mmio, tx_ring, rx_ring, status_block, tx_bufs, rx_bufs), dma_iovas| {
104                rings::clear(&tx_ring);
105                rings::clear(&rx_ring);
106                rings::clear(&status_block);
107                let mut ctrl = BcmController {
108                    mmio,
109                    tx_ring,
110                    rx_ring,
111                    status_block,
112                    tx_bufs,
113                    rx_bufs,
114                    dma_iovas,
115                    tx_next: 0,
116                    rx_head: 0,
117                    mac,
118                    link_up: false,
119                    func,
120                };
121                for_each_initial_dma_program(&dma_iovas, |p| {
122                    let _ = ctrl.apply(p);
123                });
124                ctrl.finish_init();
125                Ok(ctrl)
126            },
127        )?
128    }
129
130    fn apply(&mut self, program: InitialDmaProgram) -> ViResult<()> {
131        match program {
132            InitialDmaProgram::TxRingBase(_) => {}
133            InitialDmaProgram::RxRingBase(_) => {}
134            InitialDmaProgram::StatusBlockBase(_) => {}
135            InitialDmaProgram::RxBuffer { index, iova } => {
136                rings::write_rx_desc(
137                    &self.rx_ring,
138                    index,
139                    BcmRxDesc {
140                        addr_hi: (iova >> 32) as u32,
141                        addr_lo: iova as u32,
142                        idx_len: (index as u32) << 16 | (BUF_SIZE as u32),
143                        type_flags: 0,
144                    },
145                );
146            }
147        }
148        Ok(())
149    }
150
151    fn finish_init(&mut self) {
152        let _ = self.mmio.write_u32(RX_STD_PRODIDX_REG, RX_SLOTS as u32);
153        self.link_up = true;
154    }
155
156    pub fn send_frame(&mut self, frame: &[u8]) -> ViResult<()> {
157        if frame.is_empty() || frame.len() > BUF_SIZE {
158            return Err(ViError::InvalidArgument);
159        }
160        let slot = self.tx_next;
161        let buf = &self.tx_bufs[slot];
162        unsafe {
163            core::ptr::copy_nonoverlapping(frame.as_ptr(), buf.virt(), frame.len());
164        }
165        compiler_fence(Ordering::Release);
166
167        let iova = self.dma_iovas.tx_bufs[slot];
168        rings::write_tx_desc(
169            &self.tx_ring,
170            slot,
171            BcmTxDesc {
172                addr_hi: (iova >> 32) as u32,
173                addr_lo: iova as u32,
174                len_flags: ((frame.len() as u32) << 16)
175                    | rings::BCMD_FLAGS_END
176                    | rings::BCMD_FLAGS_PACKET_END,
177                vlan_tag: 0,
178            },
179        );
180        compiler_fence(Ordering::Release);
181
182        self.tx_next = (slot + 1) % TX_SLOTS;
183        self.mmio.write_u32(TX_PRODIDX_REG, self.tx_next as u32)?;
184
185        compiler_fence(Ordering::Acquire);
186        Ok(())
187    }
188
189    pub fn poll_rx(&mut self, target: &mut [u8]) -> ViResult<usize> {
190        let slot = self.rx_head;
191        let buf = &self.rx_bufs[slot];
192        let len = BUF_SIZE.min(target.len());
193        unsafe {
194            core::ptr::copy_nonoverlapping(buf.virt() as *const u8, target.as_mut_ptr(), len);
195        }
196        self.rx_head = (slot + 1) % RX_SLOTS;
197        self.mmio
198            .write_u32(RX_STD_PRODIDX_REG, self.rx_head as u32)?;
199        Ok(len)
200    }
