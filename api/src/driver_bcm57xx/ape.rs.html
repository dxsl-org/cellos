43const APE_LOCK_POLLS: u32 = 50_000;

45fn lock_bit(func: u8, locknum: usize) -> u32 {
46    match locknum {
47        APE_LOCK_PHY0 | APE_LOCK_PHY1 | APE_LOCK_PHY2 | APE_LOCK_PHY3 => APE_LOCK_REQ_DRIVER0,
48        _ => {
49            if func == 0 {
50                APE_LOCK_REQ_DRIVER0
51            } else {
52                1 << func
53            }
54        }
55    }
56}
57
58pub(crate) fn phy_lock_for_func(func: u8) -> usize {
59    match func {
60        0 => APE_LOCK_PHY0,
61        1 => APE_LOCK_PHY1,
62        2 => APE_LOCK_PHY2,
63        _ => APE_LOCK_PHY3,
64    }
65}
66
67pub(crate) fn acquire_ape_lock(regs: &impl RegisterFile, func: u8, locknum: usize) -> ViResult<()> {
68    let bit = lock_bit(func, locknum);
69    let off = 4 * locknum;
70    regs.wr32(APE_PER_LOCK_REQ + off, bit)?;
71
72    for _ in 0..APE_LOCK_POLLS {
73        let status = regs.rd32(APE_PER_LOCK_GRANT + off)?;
74        if status == bit {
75            return Ok(());
76        }
77        core::hint::spin_loop();
78    }
79    // Revoke request on timeout
80    let _ = regs.wr32(APE_PER_LOCK_GRANT + off, bit);
81    Err(ViError::WouldBlock)
82}
83
84pub(crate) fn release_ape_lock(regs: &impl RegisterFile, func: u8, locknum: usize) -> ViResult<()> {
85    let bit = lock_bit(func, locknum);
86    let off = 4 * locknum;
87    regs.wr32(APE_PER_LOCK_GRANT + off, bit)
88}
89
90pub(crate) fn with_ape_lock<R>(
91    regs: &impl RegisterFile,
92    func: u8,
93    locknum: usize,
94    op: impl FnOnce() -> ViResult<R>,
95) -> ViResult<R> {
96    acquire_ape_lock(regs, func, locknum)?;
97    let result = op();
98    let release_res = release_ape_lock(regs, func, locknum);
99    match (result, release_res) {
100        (Ok(val), Ok(())) => Ok(val),
101        (Err(e), _) => Err(e),
102        (Ok(_), Err(e)) => Err(e),
103    }
104}

106pub(crate) fn notify_driver_state(regs: &impl RegisterFile, func: u8, state: u32) -> ViResult<()> {
107    let event = match state {
108        APE_HOST_DRVR_STATE_START => {
109            regs.wr32(APE_HOST_SEG_SIG, APE_HOST_SEG_SIG_MAGIC)?;
110            regs.wr32(APE_HOST_SEG_LEN, APE_HOST_SEG_LEN_MAGIC)?;
111            regs.wr32(APE_HOST_DRVR_STATE, APE_HOST_DRVR_STATE_START)?;
112            APE_EVENT_STATUS_STATE_START
113        }
114        APE_HOST_DRVR_STATE_UNLOAD => {
115            regs.wr32(APE_HOST_DRVR_STATE, APE_HOST_DRVR_STATE_UNLOAD)?;
116            APE_EVENT_STATUS_STATE_UNLOAD
117        }
118        _ => return Ok(()),
119    };
120
121    let _ = with_ape_lock(regs, func, APE_LOCK_MEM, || {
122        let full_event = event
123            | APE_EVENT_STATUS_DRIVER_EVNT
124            | APE_EVENT_STATUS_STATE_CHNGE
125            | APE_EVENT_STATUS_EVENT_PENDING;
126        regs.wr32(APE_EVENT_STATUS, full_event)?;
127        regs.wr32(APE_EVENT, APE_EVENT_1)
128    });
129    Ok(())
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135    use crate::fake::FakeRegs;
136
137    #[test]
138    fn ape_lock_acquire_and_release_flow() {
139        let fake = FakeRegs::new();
140        with_ape_lock(&fake, 0, APE_LOCK_GRC, || Ok(())).expect("APE lock succeeds");
141        let off = 4 * APE_LOCK_GRC;
142        assert_eq!(fake.get(APE_PER_LOCK_GRANT + off), 0);
143    }
144
145    #[test]
146    fn ape_lock_released_on_error() {
147        let fake = FakeRegs::new();
148        let res: ViResult<()> = with_ape_lock(&fake, 0, APE_LOCK_GRC, || Err(ViError::IO));
149        assert_eq!(res, Err(ViError::IO));
150        let off = 4 * APE_LOCK_GRC;
151        assert_eq!(fake.get(APE_PER_LOCK_GRANT + off), 0);
152    }
153
154    #[test]
155    fn driver_state_change_writes_magic_and_state() {
156        let fake = FakeRegs::new();
157        notify_driver_state(&fake, 0, APE_HOST_DRVR_STATE_START).expect("notify start succeeds");
158        assert_eq!(fake.get(APE_HOST_SEG_SIG), APE_HOST_SEG_SIG_MAGIC);
159        assert_eq!(fake.get(APE_HOST_DRVR_STATE), APE_HOST_DRVR_STATE_START);
160
161        notify_driver_state(&fake, 0, APE_HOST_DRVR_STATE_UNLOAD).expect("notify unload succeeds");
162        assert_eq!(fake.get(APE_HOST_DRVR_STATE), APE_HOST_DRVR_STATE_UNLOAD);
163    }
