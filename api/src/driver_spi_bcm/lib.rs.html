23/// Polling-only BCM SPI0 master, Mode 0, native CS0 baseline.
24#[cfg(feature = "runtime")]
25pub struct BcmSpi0 {
26    core: BcmSpiCore<MmioBus>,
27}
28
29#[cfg(feature = "runtime")]
30impl BcmSpi0 {
31    pub fn open() -> Result<Self, ViError> {
32        let mmio = request_region(BCM2837.mmio.spi0_base, BCM2837.mmio.spi0_grant_size)?;
33        Ok(Self {
34            core: BcmSpiCore::new(MmioBus::new(mmio)),
35        })
36    }
37}
38
39#[cfg(feature = "runtime")]
40impl ViSpi for BcmSpi0 {
41    type Error = SpiError;
42
43    fn cs_select(&mut self) -> Result<(), Self::Error> {
44        self.core.cs_select()
45    }
46
47    fn cs_deselect(&mut self) -> Result<(), Self::Error> {
48        self.core.cs_deselect()
49    }
50
51    fn transfer(&mut self, tx: &[u8], rx: &mut [u8]) -> Result<(), Self::Error> {
52        self.core.transfer(tx, rx)
53    }
54
55    fn write(&mut self, bytes: &[u8]) -> Result<(), Self::Error> {
56        self.core.write(bytes)
57    }
58}
59
60#[cfg(feature = "runtime")]
61impl Drop for BcmSpi0 {
62    fn drop(&mut self) {
63        let _ = self.core.cs_deselect();
64    }
