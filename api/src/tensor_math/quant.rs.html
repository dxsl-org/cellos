1//! Q8_0 (GGML) block layout and decoders.
2//!
3//! A Q8_0 block is [`Q8_0_BLOCK_BYTES`] bytes: a little-endian IEEE-754 binary16 scale
4//! followed by [`Q8_0_BLOCK_WEIGHTS`] signed 8-bit weights, dequantized as
5//! `w = scale * q`. Blocks are the unit of decoding — a partial block is rejected — so a
6//! quantized tensor always has `cols % Q8_0_BLOCK_WEIGHTS == 0` weights per row.
7//!
8//! Invariant: every decoder validates its input length *before* reading and never indexes
9//! outside the buffer it was handed. Malformed bytes produce [`MathError::ShapeMismatch`]
10//! rather than a panic, so hostile model files cannot take the engine down.
11
12use super::MathError;
13
14/// Weights stored in one Q8_0 block.
15pub const Q8_0_BLOCK_WEIGHTS: usize = 32;
16
17/// Bytes occupied by one Q8_0 block (2-byte f16 scale + 32 i8 weights).
18pub const Q8_0_BLOCK_BYTES: usize = 34;
19
20/// Decode one Q8_0 block (2-byte f16 scale + 32 i8) into `out`.
21///
22/// `block` must hold at least [`Q8_0_BLOCK_BYTES`] bytes; trailing bytes are ignored so the
23/// function can be applied straight to a sub-slice of a tensor buffer.
24pub fn q8_0_block_to_f32(block: &[u8], out: &mut [f32; Q8_0_BLOCK_WEIGHTS]) -> Result<(), MathError> {
25    let raw = block
26        .get(..Q8_0_BLOCK_BYTES)
27        .ok_or(MathError::ShapeMismatch)?;
28    let mut bytes = raw.iter();
29    // Both `next()` calls are `Some` because `raw` is exactly `Q8_0_BLOCK_BYTES` long.
30    let (Some(&lo), Some(&hi)) = (bytes.next(), bytes.next()) else {
31        return Err(MathError::ShapeMismatch);
32    };
33    let scale = f16_to_f32(u16::from_le_bytes([lo, hi]));
34    for (slot, &weight) in out.iter_mut().zip(bytes) {
35        *slot = scale * f32::from(weight as i8);
36    }
37    Ok(())
38}
39
40/// IEEE-754 binary16 bits → f32 (subnormals, inf, nan included).
41///
42/// The conversion is exact: every binary16 value, including subnormals and NaN payloads, has
43/// an exact binary32 representation.
44pub fn f16_to_f32(bits: u16) -> f32 {
45    let sign = (bits as u32 & 0x8000) << 16;
46    let exponent = (bits as u32 >> 10) & 0x1f;
47    let mantissa = bits as u32 & 0x03ff;
48    let out = if exponent == 0 {
49        if mantissa == 0 {
50            sign
51        } else {
52            // Subnormal half: value = mantissa * 2^-24. Shift the leading one up to the f32
53            // implicit-bit position; bits shifted out are discarded by the mask.
54            let leading = 31 - mantissa.leading_zeros();
55            let exponent32 = leading + 103; // (leading - 24) + bias(127)
56            let mantissa32 = (mantissa << (23 - leading)) & 0x007f_ffff;
57            sign | (exponent32 << 23) | mantissa32
58        }
59    } else if exponent == 0x1f {
60        // Infinity when the mantissa is zero, otherwise NaN with the payload preserved.
61        sign | 0x7f80_0000 | (mantissa << 13)
62    } else {
63        sign | ((exponent + 112) << 23) | (mantissa << 13)
64    };
65    f32::from_bits(out)
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn f16_decodes_the_edge_cases() {
74        assert_eq!(f16_to_f32(0x0000), 0.0);
75        assert_eq!(f16_to_f32(0x8000).to_bits(), (-0.0f32).to_bits());
76        assert_eq!(f16_to_f32(0x3C00), 1.0);
77        assert_eq!(f16_to_f32(0xC000), -2.0);
78        assert_eq!(f16_to_f32(0xC100), -2.5);
79        assert_eq!(f16_to_f32(0x3555), 0.333_251_95);
80        assert_eq!(f16_to_f32(0x7C00), f32::INFINITY);
81        assert_eq!(f16_to_f32(0xFC00), f32::NEG_INFINITY);
82        assert!(f16_to_f32(0x7E00).is_nan());
83        assert!(f16_to_f32(0xFE00).is_nan());
84        assert!(f16_to_f32(0x7C01).is_nan());
85        // Largest finite half and the subnormal range.
86        assert_eq!(f16_to_f32(0x7BFF), 65504.0);
87        assert_eq!(f16_to_f32(0x0001), 2f32.powi(-24));
88        assert_eq!(f16_to_f32(0x0200), 2f32.powi(-15));
89        assert_eq!(f16_to_f32(0x03FF), 1023.0 * 2f32.powi(-24));
90        assert_eq!(f16_to_f32(0x0400), 2f32.powi(-14));
91    }
92
93    #[test]
94    fn q8_0_block_decodes_scale_and_weights() {
95        let mut block = [0u8; Q8_0_BLOCK_BYTES];
96        block[..2].copy_from_slice(&0x3800u16.to_le_bytes()); // 0.5
97        for (i, byte) in block[2..].iter_mut().enumerate() {
98            *byte = (i as i32 - 16) as i8 as u8;
99        }
100        let mut out = [f32::NAN; Q8_0_BLOCK_WEIGHTS];
101        q8_0_block_to_f32(&block, &mut out).expect("well-formed block decodes");
102        for (i, value) in out.iter().enumerate() {
103            assert_eq!(*value, 0.5 * (i as i32 - 16) as f32);
104        }
105        assert_eq!(out[0], -8.0);
106        assert_eq!(out[16], 0.0);
107        assert_eq!(out[31], 7.5);
108    }
109
110    #[test]
111    fn q8_0_block_tolerates_trailing_bytes_and_rejects_short_ones() {
112        let mut long = [0u8; Q8_0_BLOCK_BYTES + 5];
113        long[..2].copy_from_slice(&0x3C00u16.to_le_bytes()); // 1.0
114        long[2] = 7;
115        let mut out = [0.0f32; Q8_0_BLOCK_WEIGHTS];
116        q8_0_block_to_f32(&long, &mut out).expect("a longer slice still decodes one block");
117        assert_eq!(out[0], 7.0);
118
119        for len in [0usize, 1, 2, 33] {
120            let short = [0u8; 33];
121            let mut out = [0.0f32; Q8_0_BLOCK_WEIGHTS];
122            assert_eq!(
123                q8_0_block_to_f32(&short[..len], &mut out),
124                Err(MathError::ShapeMismatch),
125                "a {len}-byte block must be rejected"
126            );
127        }
128    }
