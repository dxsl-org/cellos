1//! Q8_0 (GGML) block layout and decoders.
2//!
3//! A Q8_0 block is [`Q8_0_BLOCK_BYTES`] bytes: a little-endian IEEE-754 binary16 scale
4//! followed by [`Q8_0_BLOCK_WEIGHTS`] signed 8-bit weights, dequantized as
5//! `w = scale * q`. Blocks are the unit of decoding — a partial block is rejected — so a
6//! quantized tensor always has `cols % Q8_0_BLOCK_WEIGHTS == 0` weights per row.
7//!
8//! Invariant: every decoder validates its input length *before* reading and never indexes
9//! outside the buffer it was handed. Malformed bytes produce [`MathError::ShapeMismatch`]
10//! rather than a panic, so hostile model files cannot take the engine down.
11
12use super::MathError;
13
14/// Weights stored in one Q8_0 block.
15pub const Q8_0_BLOCK_WEIGHTS: usize = 32;
16
17/// Bytes occupied by one Q8_0 block (2-byte f16 scale + 32 i8 weights).
18pub const Q8_0_BLOCK_BYTES: usize = 34;
19
20/// Decode one Q8_0 block (2-byte f16 scale + 32 i8) into `out`.
21///
22/// `block` must hold at least [`Q8_0_BLOCK_BYTES`] bytes; trailing bytes are ignored so the
23/// function can be applied straight to a sub-slice of a tensor buffer.
24pub fn q8_0_block_to_f32(
25    block: &[u8],
26    out: &mut [f32; Q8_0_BLOCK_WEIGHTS],
27) -> Result<(), MathError> {
28    let raw = block
29        .get(..Q8_0_BLOCK_BYTES)
30        .ok_or(MathError::ShapeMismatch)?;
31    let mut bytes = raw.iter();
32    // Both `next()` calls are `Some` because `raw` is exactly `Q8_0_BLOCK_BYTES` long.
33    let (Some(&lo), Some(&hi)) = (bytes.next(), bytes.next()) else {
34        return Err(MathError::ShapeMismatch);
35    };
36    let scale = f16_to_f32(u16::from_le_bytes([lo, hi]));
37    for (slot, &weight) in out.iter_mut().zip(bytes) {
38        *slot = scale * f32::from(weight as i8);
39    }
40    Ok(())
41}

43/// IEEE-754 binary16 bits → f32 (subnormals, inf, nan included).
44///
45/// The conversion is exact: every binary16 value, including subnormals and NaN payloads, has
46/// an exact binary32 representation.
47pub fn f16_to_f32(bits: u16) -> f32 {
48    let sign = (bits as u32 & 0x8000) << 16;
49    let exponent = (bits as u32 >> 10) & 0x1f;
50    let mantissa = bits as u32 & 0x03ff;
51    let out = if exponent == 0 {
52        if mantissa == 0 {
53            sign
54        } else {
55            // Subnormal half: value = mantissa * 2^-24. Shift the leading one up to the f32
56            // implicit-bit position; bits shifted out are discarded by the mask.
57            let leading = 31 - mantissa.leading_zeros();
58            let exponent32 = leading + 103; // (leading - 24) + bias(127)
59            let mantissa32 = (mantissa << (23 - leading)) & 0x007f_ffff;
60            sign | (exponent32 << 23) | mantissa32
61        }
62    } else if exponent == 0x1f {
63        // Infinity when the mantissa is zero, otherwise NaN with the payload preserved.
64        sign | 0x7f80_0000 | (mantissa << 13)
65    } else {
66        sign | ((exponent + 112) << 23) | (mantissa << 13)
67    };
68    f32::from_bits(out)
69}

71/// f32 → IEEE-754 binary16 bits, round-to-nearest-even — the inverse of [`f16_to_f32`], used to
72/// store the per-block scale of a freshly quantized activation row.
73///
74/// Magnitudes above the largest finite half (65504) saturate instead of rounding to infinity: the
75/// only caller converts a block peak (a finite f32 by construction), and an infinite scale would
76/// turn a finite activation block into NaN.
77pub fn f32_to_f16(value: f32) -> u16 {
78    let bits = value.to_bits();
79    let sign = ((bits >> 16) & 0x8000) as u16;
80    let exponent = ((bits >> 23) & 0xff) as i32;
81    let mantissa = bits & 0x007f_ffff;
82
83    if exponent == 0xff {
84        // Infinity keeps its sign; every NaN collapses to one quiet NaN payload.
85        return sign | if mantissa == 0 { 0x7c00 } else { 0x7e00 };
86    }
87    if exponent == 0 {
88        // An f32 subnormal is far below the smallest half subnormal (2^-24).
89        return sign;
90    }
91
92    // `value = wide * 2^(exponent - 150)` with the implicit leading one made explicit.
93    let half_exponent = exponent - 112;
94    if half_exponent >= 0x1f {
95        return sign | 0x7bff;
96    }
97    if half_exponent <= 0 {
98        // Subnormal half: `value = k * 2^-24`, so `k = round(wide * 2^(exponent - 126))`.
99        let shift = 126 - exponent;
100        if shift > 24 {
101            return sign;
102        }
103        let wide = mantissa | 0x0080_0000;
104        let mut k = wide >> shift;
105        let remainder = wide & ((1 << shift) - 1);
106        let halfway = 1 << (shift - 1);
107        if remainder > halfway || (remainder == halfway && k & 1 == 1) {
108            k += 1;
109        }
110        // Rounding up out of the subnormal range lands exactly on the smallest normal (0x0400).
111        return sign | k as u16;
112    }
113
114    let mut rounded = mantissa;
115    let remainder = mantissa & 0x1fff;
116    if remainder > 0x1000 || (remainder == 0x1000 && mantissa & 0x2000 != 0) {
117        rounded += 0x2000;
118        if rounded & 0x0080_0000 != 0 {
119            // The mantissa carried into the exponent and is zero afterwards.
120            let carried = half_exponent + 1;
121            if carried >= 0x1f {
122                return sign | 0x7bff;
123            }
124            return sign | ((carried as u16) << 10);
125        }
126    }
127    sign | ((half_exponent as u16) << 10) | ((rounded >> 13) as u16)
128}

130/// Quantize `x` into Q8_0 blocks, writing `q8_0_row_bytes(x.len())` bytes into `out`.
131///
132/// The rule is GGML's, with one deliberate difference: the quantizer divides by the *stored*
133/// (f16-rounded) scale, not by the unrounded `amax/127` GGML keeps in a register. Quantizing against
134/// the scale a reader will actually multiply by keeps the per-weight error inside half a step, and it
135/// is the convention the fixture's weight packer already uses. Rounding is half-away-from-zero
136/// (`roundf`), clamped to the `i8` range, so `amax` maps to ±127.
137///
138/// A block whose peak is zero (or too small to survive the f16 scale) quantizes to all zeros with a
139/// zero scale: it dequantizes to zero rather than to NaN. A NaN element also quantizes to zero
140/// (Rust's saturating float-to-int cast) instead of propagating, because a NaN *scale* would poison
141/// every weight in the row; a block that is entirely NaN has a zero peak and so quantizes to zero.
142pub fn q8_0_row_from_f32(x: &[f32], out: &mut [u8]) -> Result<(), MathError> {
143    if x.is_empty() {
144        return Err(MathError::Empty);
145    }
146    let Some(row_bytes) = crate::q8_0_row_bytes(x.len()) else {
147        return Err(MathError::NotDivisible);
148    };
149    if out.len() != row_bytes {
150        return Err(MathError::ShapeMismatch);
151    }
152    for (block, slot) in x
153        .chunks_exact(Q8_0_BLOCK_WEIGHTS)
154        .zip(out.chunks_exact_mut(Q8_0_BLOCK_BYTES))
155    {
156        let mut peak = 0.0f32;
157        for value in block {
158            peak = peak.max(value.abs());
159        }
160        let scale_bits = f32_to_f16(peak / 127.0);
161        let scale = f16_to_f32(scale_bits);
162        slot[..2].copy_from_slice(&scale_bits.to_le_bytes());
163        for (byte, value) in slot[2..].iter_mut().zip(block) {
164            let quantized = if scale > 0.0 {
165                let scaled = value / scale;
166                let rounded = if scaled >= 0.0 {
167                    (scaled + 0.5) as i32
168                } else {
169                    (scaled - 0.5) as i32
170                };
171                rounded.clamp(i32::from(i8::MIN), i32::from(i8::MAX)) as i8
172            } else {
173                0
174            };
175            *byte = quantized as u8;
176        }
177    }
178    Ok(())
179}

185    fn f16_decodes_the_edge_cases() {
186        assert_eq!(f16_to_f32(0x0000), 0.0);
187        assert_eq!(f16_to_f32(0x8000).to_bits(), (-0.0f32).to_bits());
188        assert_eq!(f16_to_f32(0x3C00), 1.0);
189        assert_eq!(f16_to_f32(0xC000), -2.0);
190        assert_eq!(f16_to_f32(0xC100), -2.5);
191        assert_eq!(f16_to_f32(0x3555), 0.333_251_95);
192        assert_eq!(f16_to_f32(0x7C00), f32::INFINITY);
193        assert_eq!(f16_to_f32(0xFC00), f32::NEG_INFINITY);
194        assert!(f16_to_f32(0x7E00).is_nan());
195        assert!(f16_to_f32(0xFE00).is_nan());
196        assert!(f16_to_f32(0x7C01).is_nan());
197        // Largest finite half and the subnormal range.
198        assert_eq!(f16_to_f32(0x7BFF), 65504.0);
199        assert_eq!(f16_to_f32(0x0001), 2f32.powi(-24));
200        assert_eq!(f16_to_f32(0x0200), 2f32.powi(-15));
201        assert_eq!(f16_to_f32(0x03FF), 1023.0 * 2f32.powi(-24));
202        assert_eq!(f16_to_f32(0x0400), 2f32.powi(-14));
203    }

253    #[test]
254    fn f16_encode_is_the_exact_inverse_of_the_decoder() {
255        // Every finite half (and both infinities) round-trips bit-exactly; NaN payloads collapse to
256        // one quiet NaN of the same sign by design, so they are checked separately.
257        for bits in 0u16..=u16::MAX {
258            let value = f16_to_f32(bits);
259            if value.is_nan() {
260                assert_eq!(
261                    f32_to_f16(value),
262                    (bits & 0x8000) | 0x7e00,
263                    "NaN half {bits:#06x} re-encodes as a quiet NaN of the same sign"
264                );
265                continue;
266            }
267            assert_eq!(
268                f32_to_f16(value),
269                bits,
270                "half {bits:#06x} ({value}) must re-encode to its own bits"
271            );
272        }
273    }

275    #[test]
276    fn f16_encode_rounds_to_nearest_even_and_saturates() {
277        assert_eq!(f32_to_f16(0.0), 0x0000);
278        assert_eq!(f32_to_f16(-0.0), 0x8000);
279        assert_eq!(f32_to_f16(1.0), 0x3C00);
280        assert_eq!(f32_to_f16(0.5), 0x3800);
281        assert_eq!(f32_to_f16(-2.0), 0xC000);
282        assert_eq!(f32_to_f16(65504.0), 0x7BFF);
283        assert_eq!(
284            f32_to_f16(65520.0),
285            0x7BFF,
286            "past the top: saturate, never inf"
287        );
288        assert_eq!(f32_to_f16(f32::INFINITY), 0x7C00);
289        assert_eq!(f32_to_f16(f32::NEG_INFINITY), 0xFC00);
290        assert_eq!(f32_to_f16(2f32.powi(-24)), 0x0001, "smallest subnormal");
291        assert_eq!(
292            f32_to_f16(2f32.powi(-25)),
293            0x0000,
294            "exactly half a step: ties to even"
295        );
296        assert_eq!(
297            f32_to_f16(3.0 * 2f32.powi(-25)),
298            0x0002,
299            "1.5 steps rounds up to even"
300        );
301        assert_eq!(f32_to_f16(1e-30), 0x0000, "below the range: signed zero");
302        assert_eq!(f32_to_f16(-1e-30), 0x8000);
303        // Halfway cases around 1.0: the step to the next half is 2^-10, so 1 + 2^-11 is exactly the
304        // midpoint between 0x3C00 (even mantissa) and 0x3C01, and 1 + 3·2^-11 is the midpoint
305        // between 0x3C01 and 0x3C02 (even mantissa again). Ties therefore go up, then stay.
306        assert_eq!(f32_to_f16(1.0 + 2f32.powi(-11)), 0x3C00);
307        assert_eq!(f32_to_f16(1.0 + 2f32.powi(-11) + 2f32.powi(-23)), 0x3C01);
308        assert_eq!(
309            f32_to_f16(1.0 + 2f32.powi(-10)),
310            0x3C01,
311            "exact half, no rounding at all"
312        );
313        assert_eq!(f32_to_f16(1.0 + 3.0 * 2f32.powi(-11)), 0x3C02);
314        assert_eq!(f32_to_f16(1.0 + 2f32.powi(-9)), 0x3C02);
315    }

317    #[test]
318    fn q8_0_row_quantizes_a_peak_to_full_range() {
319        let mut x: Vec<f32> = (0..32).map(|i| (i as f32 - 16.0) * 0.25).collect();
320        x[0] = -4.0;
321        x[31] = 4.0;
322        let mut row = [0u8; Q8_0_BLOCK_BYTES];
323        q8_0_row_from_f32(&x, &mut row).expect("32 weights is one block");
324
325        let scale_bits = u16::from_le_bytes([row[0], row[1]]);
326        let scale = f16_to_f32(scale_bits);
327        // The block peak is ±4.0: GGML's rule stores amax/127 as f16.
328        assert_eq!(scale_bits, f32_to_f16(4.0 / 127.0));
329        assert_eq!(i32::from(row[2] as i8), -127);
330        assert_eq!(i32::from(row[Q8_0_BLOCK_BYTES - 1] as i8), 127);
331
332        // Every dequantized weight is within half a step of the value that went in.
333        let mut decoded = [0.0f32; Q8_0_BLOCK_WEIGHTS];
334        q8_0_block_to_f32(&row, &mut decoded).expect("the row we just wrote decodes");
335        for (original, quantized) in x.iter().zip(decoded) {
336            assert!(
337                (original - quantized).abs() <= scale * 0.5 + f32::EPSILON,
338                "{original} quantized to {quantized} (scale {scale})"
339            );
340        }
341    }

343    #[test]
344    fn q8_0_row_handles_zero_nan_and_shape_errors() {
345        let zeros = [0.0f32; 32];
346        let mut row = [0xffu8; Q8_0_BLOCK_BYTES];
347        q8_0_row_from_f32(&zeros, &mut row).expect("a zero block is representable");
348        assert_eq!(
349            &row, &[0u8; Q8_0_BLOCK_BYTES],
350            "zero in, zero scale and zero weights out"
351        );
352
353        let mut with_nan = [1.0f32; 32];
354        with_nan[7] = f32::NAN;
355        q8_0_row_from_f32(&with_nan, &mut row).expect("NaN elements do not fail the quantizer");
356        let mut decoded = [0.0f32; Q8_0_BLOCK_WEIGHTS];
357        q8_0_block_to_f32(&row, &mut decoded).expect("decodes");
358        assert_eq!(decoded[7], 0.0, "a NaN element quantizes to zero");
359        let scale = f16_to_f32(u16::from_le_bytes([row[0], row[1]]));
360        assert_eq!(
361            u16::from_le_bytes([row[0], row[1]]),
362            f32_to_f16(1.0 / 127.0),
363            "a NaN peak is ignored, not propagated"
364        );
365        assert!(
366            (decoded[0] - 1.0).abs() <= scale * 0.5 + f32::EPSILON,
367            "the finite elements still quantize normally (got {})",
368            decoded[0]
369        );
370
371        let mut many = [0u8; 2 * Q8_0_BLOCK_BYTES];
372        assert_eq!(
373            q8_0_row_from_f32(&[1.0f32; 32], &mut many),
374            Err(MathError::ShapeMismatch),
375            "out must be exactly the row's byte length"
376        );
377        assert_eq!(
378            q8_0_row_from_f32(&[1.0f32; 33], &mut [0u8; Q8_0_BLOCK_BYTES + 2]),
379            Err(MathError::NotDivisible),
380            "a partial block is a shape error"
381        );
382        assert_eq!(
383            q8_0_row_from_f32(&[], &mut []),
384            Err(MathError::Empty),
385            "an empty row is undefined"
386        );
387    }
