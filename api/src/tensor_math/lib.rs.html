1//! f32 tensor kernels for the Cellos native CPU inference engine (Spec 24 §4).
2//!
3//! The crate is `no_std` and allocation-free: every kernel reads its operands from slices and
4//! writes into a caller-supplied output slice, so the engine can run it out of a static arena
5//! with no allocator in the loop. The bare-metal targets ship no `libm`, so the few
6//! transcendental functions needed (`sqrt`, `exp`, `pow`, `cos`, `sin`) come from the `libm`
7//! crate.
8//!
9//! # Invariants
10//!
11//! - **Shapes are validated, arithmetic is not.** Every kernel checks its operand lengths
12//!   before it touches memory and reports a [`MathError`] instead of panicking, indexing out
13//!   of bounds, or leaving `out` partially written. Malformed *numeric* input (NaN, ±inf) is
14//!   not a shape error: it propagates, and [`softmax_in_place`] still returns a probability
15//!   vector for it.
16//! - **Check precedence** is [`MathError::Empty`] → [`MathError::NotDivisible`] →
17//!   [`MathError::ShapeMismatch`], so a caller can rely on which error it gets.
18//! - An operation that is undefined over a zero-length operand is [`MathError::Empty`]
19//!   (`matvec`/`matvec_q8_0` with no rows or no columns, `rms_norm`, `dot`, `rope_normal` over
20//!   an empty vector). A pure element-wise combine over empty slices is the identity and
21//!   returns `Ok`.
22//! - Quantized operands live in [`quant`], which documents the Q8_0 block layout and the
23//!   bounds checks its decoders perform.
24
25#![cfg_attr(not(test), no_std)]

31/// Errors that are the caller's fault (shape mismatch), never arithmetic faults.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum MathError {
34    /// Operand lengths disagree with the declared shape.
35    ShapeMismatch,
36    /// A length is not a whole number of blocks (`cols % 32 != 0` for Q8_0, an odd-length
37    /// RoPE head).
38    NotDivisible,
39    /// The operation is undefined over a zero-length operand.
40    Empty,
41}
42
43impl fmt::Display for MathError {
44    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
45        let message = match self {
46            MathError::ShapeMismatch => "operand shape mismatch",
47            MathError::NotDivisible => "length is not a whole number of blocks",
48            MathError::Empty => "operation is undefined over an empty operand",
49        };
50        f.write_str(message)
51    }
52}

56/// `out[r] = dot(W[r], x)`, `W` row-major `rows × cols`, `x.len() == cols`.
57///
58/// Rows are summed in four independent f32 lanes that are added together at the end; that
59/// order is shared with [`matvec_q8_0`], so dequantized Q8_0 weights reproduce the dense
60/// result exactly rather than merely approximately.
61pub fn matvec(
62    out: &mut [f32],
63    w: &[f32],
64    x: &[f32],
65    rows: usize,
66    cols: usize,
67) -> Result<(), MathError> {
68    if rows == 0 || cols == 0 {
69        return Err(MathError::Empty);
70    }
71    let Some(elements) = rows.checked_mul(cols) else {
72        return Err(MathError::ShapeMismatch);
73    };
74    if w.len() != elements || x.len() != cols || out.len() != rows {
75        return Err(MathError::ShapeMismatch);
76    }
77    for (row, slot) in w.chunks_exact(cols).zip(out.iter_mut()) {
78        *slot = dot_f32(row, x);
79    }
80    Ok(())
81}

83/// Same as [`matvec`] with Q8_0-quantized rows; `w` is `rows × row_bytes` where
84/// `row_bytes = cols / 32 * 34` (cols must be a multiple of 32).
85///
86/// Each 34-byte block is decoded to f32 on the stack and accumulated in the same lane order as
87/// [`matvec`], so `matvec_q8_0(out, q, x, r, c)` and `matvec(out, dequantize(q), x, r, c)`
88/// agree bit for bit when the dequantized weights are produced from the same block bytes.
89///
90/// The engine ships [`matvec_q8_0_int8`] instead, because decoding a block to f32 costs more per MAC
91/// than the integer dot product (measured: 6.8 vs 12.8 GFLOP/s for the decode, and the decode is the
92/// dense kernel's equal — see `libs/ai-engine/benches/cpu_engine.rs`). This kernel is kept as the
93/// f32 reference the integer kernel is bounded against, and as the benchmark's comparison row; its
94/// own staging block still beats converting i8 inside the accumulation loop (10.3 vs 7.5 GFLOP/s at
95/// `-O3`), so do not "simplify" it into the loop.
96pub fn matvec_q8_0(
97    out: &mut [f32],
98    w: &[u8],
99    x: &[f32],
100    rows: usize,
101    cols: usize,
102) -> Result<(), MathError> {
103    if rows == 0 || cols == 0 {
104        return Err(MathError::Empty);
105    }
106    let Some(row_bytes) = q8_0_row_bytes(cols) else {
107        return Err(MathError::NotDivisible);
108    };
109    let Some(bytes) = rows.checked_mul(row_bytes) else {
110        return Err(MathError::ShapeMismatch);
111    };
112    if w.len() != bytes || x.len() != cols || out.len() != rows {
113        return Err(MathError::ShapeMismatch);
114    }
115    for (row, slot) in w.chunks_exact(row_bytes).zip(out.iter_mut()) {
116        let mut acc = [0.0f32; 4];
117        let blocks = row.chunks_exact(quant::Q8_0_BLOCK_BYTES);
118        let vectors = x.chunks_exact(quant::Q8_0_BLOCK_WEIGHTS);
119        for (block, vector) in blocks.zip(vectors) {
120            let mut decoded = [0.0f32; quant::Q8_0_BLOCK_WEIGHTS];
121            quant::q8_0_block_to_f32(block, &mut decoded)?;
122            accumulate_products(&mut acc, &decoded, vector);
123        }
124        *slot = reduce_lanes(acc);
125    }
126    Ok(())
127}

129/// Row bytes for a Q8_0 tensor with `cols` columns, or `None` when `cols` is not a multiple of
130/// [`quant::Q8_0_BLOCK_WEIGHTS`].
131pub fn q8_0_row_bytes(cols: usize) -> Option<usize> {
132    if !cols.is_multiple_of(quant::Q8_0_BLOCK_WEIGHTS) {
133        return None;
134    }
135    (cols / quant::Q8_0_BLOCK_WEIGHTS).checked_mul(quant::Q8_0_BLOCK_BYTES)
136}

138/// Same as [`matvec_q8_0`] with a Q8_0-quantized activation row instead of an f32 vector.
139///
140/// `x_q8` is `q8_0_row_bytes(x.len())` bytes as produced by [`quant::q8_0_row_from_f32`], so the
141/// activation is rounded once per projection and each block of 32 weights is then a single integer
142/// dot product:
143///
144/// ```text
145/// block part = d_w · d_a · Σ (wᵢ · aᵢ)      (Σ in i32, so it is exact)
146/// ```
147///
148/// `|Σ| ≤ 32 · 128² = 524_288`, four orders of magnitude inside `i32`, so the integer part never
149/// overflows and carries no rounding at all; the only error against [`matvec_q8_0`] is the half-step
150/// of the activation quantization. The per-block parts of one row are summed in four f32 lanes and
151/// collapsed with [`reduce_lanes`], the same order both f32 kernels use.
152///
153/// This is the shipped engine path: it removes the 32-element f32 decode (i8→f32 convert, scale
154/// multiply, store) that made the f32-staging kernel ~2× slower per MAC than the dense one, and it
155/// removes every f32 multiply and add from the inner loop — which is what decides cost on the cell,
156/// where softfloat is emulated.
157pub fn matvec_q8_0_int8(
158    out: &mut [f32],
159    w: &[u8],
160    x_q8: &[u8],
161    rows: usize,
162    cols: usize,
163) -> Result<(), MathError> {
164    if rows == 0 || cols == 0 {
165        return Err(MathError::Empty);
166    }
167    let Some(row_bytes) = q8_0_row_bytes(cols) else {
168        return Err(MathError::NotDivisible);
169    };
170    let Some(bytes) = rows.checked_mul(row_bytes) else {
171        return Err(MathError::ShapeMismatch);
172    };
173    if w.len() != bytes || x_q8.len() != row_bytes || out.len() != rows {
174        return Err(MathError::ShapeMismatch);
175    }
176    for (row, slot) in w.chunks_exact(row_bytes).zip(out.iter_mut()) {
177        let mut acc = [0.0f32; 4];
178        for (index, (block, vector)) in row
179            .chunks_exact(quant::Q8_0_BLOCK_BYTES)
180            .zip(x_q8.chunks_exact(quant::Q8_0_BLOCK_BYTES))
181            .enumerate()
182        {
183            let (Some(&w_lo), Some(&w_hi)) = (block.first(), block.get(1)) else {
184                return Err(MathError::ShapeMismatch);
185            };
186            let (Some(&a_lo), Some(&a_hi)) = (vector.first(), vector.get(1)) else {
187                return Err(MathError::ShapeMismatch);
188            };
189            let d_w = quant::f16_to_f32(u16::from_le_bytes([w_lo, w_hi]));
190            let d_a = quant::f16_to_f32(u16::from_le_bytes([a_lo, a_hi]));
191            let mut sum = 0i32;
192            for (&weight, &activation) in block[2..].iter().zip(&vector[2..]) {
193                sum += i32::from(weight as i8) * i32::from(activation as i8);
194            }
195            acc[index % 4] += (d_w * d_a) * sum as f32;
196        }
197        *slot = reduce_lanes(acc);
198    }
199    Ok(())
200}

202/// `out[i] = x[i] / sqrt(mean(x²) + eps) * weight[i]`.
203///
204/// Computed with one reciprocal square root, so the stored value is
205/// `(x[i] * (1 / sqrt(mean + eps))) * weight[i]`; `eps` is the caller's numerical floor (use
206/// `0.0` for an exact normalisation).
207pub fn rms_norm(out: &mut [f32], x: &[f32], weight: &[f32], eps: f32) -> Result<(), MathError> {
208    if x.is_empty() {
209        return Err(MathError::Empty);
210    }
211    if out.len() != x.len() || weight.len() != x.len() {
212        return Err(MathError::ShapeMismatch);
213    }
214    let mut sum_of_squares = 0.0f32;
215    for &value in x {
216        sum_of_squares += value * value;
217    }
218    let mean = sum_of_squares / x.len() as f32;
219    let scale = 1.0 / libm::sqrtf(mean + eps);
220    for ((slot, &value), &factor) in out.iter_mut().zip(x).zip(weight) {
221        *slot = (value * scale) * factor;
222    }
223    Ok(())
224}

226/// In-place Llama NORMAL RoPE over one head vector: pairs `(i, i + half)` rotated by
227/// `pos * freq_base^(-2i/dim)`.
228///
229/// The rotation angle is evaluated in f64 (the frequency table is the numerically sensitive
230/// part of RoPE) and rounded to f32 for the rotation itself. `pos == 0` returns immediately, so
231/// position zero is the exact identity for every input including non-finite ones.
232pub fn rope_normal(head: &mut [f32], pos: usize, freq_base: f32) -> Result<(), MathError> {
233    if head.is_empty() {
234        return Err(MathError::Empty);
235    }
236    let dim = head.len();
237    if !dim.is_multiple_of(2) {
238        return Err(MathError::NotDivisible);
239    }
240    if pos == 0 {
241        // cos(0) = 1 and sin(0) = 0 exactly, for any `freq_base`.
242        return Ok(());
243    }
244    let half = dim / 2;
245    let (low, high) = head.split_at_mut(half);
246    for (i, (low_slot, high_slot)) in low.iter_mut().zip(high.iter_mut()).enumerate() {
247        let exponent = -2.0 * (i as f64) / (dim as f64);
248        let angle = pos as f64 * libm::pow(freq_base as f64, exponent);
249        let cos = libm::cos(angle) as f32;
250        let sin = libm::sin(angle) as f32;
251        let (x0, x1) = (*low_slot, *high_slot);
252        *low_slot = x0 * cos - x1 * sin;
253        *high_slot = x0 * sin + x1 * cos;
254    }
255    Ok(())
256}

258/// Numerically stable in-place softmax (subtract max).
259///
260/// The output always sums to 1 and is never NaN: when the input carries no finite ranking
261/// information (every element `-inf`, or a NaN anywhere) the uniform distribution `1/n` is
262/// written instead. An empty slice is a no-op.
263pub fn softmax_in_place(x: &mut [f32]) {
264    if x.is_empty() {
265        return;
266    }
267    let mut max = f32::NEG_INFINITY;
268    for &value in x.iter() {
269        if value > max {
270            max = value;
271        }
272    }
273    if !max.is_finite() {
274        fill_uniform(x);
275        return;
276    }
277    let mut sum = 0.0f32;
278    for slot in x.iter_mut() {
279        let weight = libm::expf(*slot - max);
280        *slot = weight;
281        sum += weight;
282    }
283    if !(sum.is_finite() && sum > 0.0) {
284        fill_uniform(x);
285        return;
286    }
287    let inverse = 1.0 / sum;
288    for slot in x.iter_mut() {
289        *slot *= inverse;
290    }
291}

293/// SiLU(x) = x * sigmoid(x) = x / (1 + exp(-x)).
294pub fn silu(x: f32) -> f32 {
295    x / (1.0 + libm::expf(-x))

298/// `gate[i] = silu(gate[i]) * up[i]` — the Llama-family SwiGLU feed-forward activation.
299///
300/// The gate projection is the one that goes through SiLU (llama.cpp `ggml_swiglu`: `silu(gate) * up`).
301/// Swapping the two branches changes the model output, so this direction is pinned by the engine's
302/// golden-oracle test, not only by this crate's unit tests.
303pub fn swiglu_in_place(gate: &mut [f32], up: &[f32]) -> Result<(), MathError> {
304    if gate.len() != up.len() {
305        return Err(MathError::ShapeMismatch);
306    }
307    for (slot, &value) in gate.iter_mut().zip(up) {
308        *slot = silu(*slot) * value;
309    }
310    Ok(())
311}

317    }
318    if a.len() != b.len() {
319        return Err(MathError::ShapeMismatch);
320    }

346/// Index of the largest value (first wins on ties); `None` for an empty slice.
347///
348/// NaN never compares greater, so an all-NaN slice yields `Some(0)`.
349pub fn argmax(x: &[f32]) -> Option<usize> {
350    let mut best: Option<(usize, f32)> = None;
351    for (index, &value) in x.iter().enumerate() {
352        // "Keep the incumbent unless the candidate is strictly greater" is what NaN requires:
353        // NaN never compares greater, so an all-NaN slice keeps index 0.
354        let replace = match best {
355            Some((_, best_value)) => value > best_value,
356            None => true,
357        };
358        if replace {
359            best = Some((index, value));
360        }
361    }
362    best.map(|(index, _)| index)
363}

365/// Deterministic PRNG (xorshift64*) used by sampling.
366///
367/// The state is never zero: seeding with 0 is remapped to the golden-ratio constant, so any
368/// seed produces a usable, non-degenerate stream. The stream is a pure function of the seed.
369pub struct Rng(u64);
370
371impl Rng {
372    /// Creates a generator from `seed`. Two generators built from the same seed produce the
373    /// same sequence.
374    pub fn new(seed: u64) -> Self {
375        // xorshift64* has a fixed point at zero; substitute the golden ratio instead.
376        Rng(if seed == 0 {
377            0x9E37_79B9_7F4A_7C15
378        } else {
379            seed
380        })
381    }
382
383    /// Next 32 bits of the output stream (the high half of the xorshift64* output).
384    pub fn next_u32(&mut self) -> u32 {
385        let mut x = self.0;
386        x ^= x >> 12;
387        x ^= x << 25;
388        x ^= x >> 27;
389        self.0 = x;
390        (x.wrapping_mul(0x2545_F491_4F6C_DD1D) >> 32) as u32
391    }
392
393    /// Uniform in [0, 1): 24 bits of mantissa precision, so the value is always `< 1`.
394    pub fn next_f32(&mut self) -> f32 {
395        const INV_TWO_POW_24: f32 = 1.0 / 16_777_216.0;
396        (self.next_u32() >> 8) as f32 * INV_TWO_POW_24
397    }
398}

400/// Greedy or top-k sampling.
401///
402/// `temperature <= 0.0` (or NaN) selects [`argmax`]. Otherwise the logits are divided by
403/// `temperature`, restricted to the `k` largest (all when `k == 0`), softmaxed, and one index is
404/// drawn from `rng`. Ties at the cut-off are broken towards the lower index, matching
405/// [`argmax`]. Returns `None` only for an empty `logits` slice.
406///
407/// `logits` is modified in place — scaled by the temperature, with everything outside the
408/// candidate set set to `-inf` — and ends up holding the sampling distribution. The selection
409/// uses `O(1)` auxiliary memory and `O(k · n)` time; `k` is the caller's bounded candidate count.
410pub fn sample_top_k(
411    logits: &mut [f32],
412    k: usize,
413    temperature: f32,
414    rng: &mut Rng,
415) -> Option<usize> {
416    if logits.is_empty() {
417        return None;
418    }
419    // A non-positive or NaN temperature means greedy decoding (see the crate docs).
420    if temperature.is_nan() || temperature <= 0.0 {
421        return argmax(logits);
422    }
423    let inverse_temperature = 1.0 / temperature;
424    for value in logits.iter_mut() {
425        *value *= inverse_temperature;
426    }
427
428    let keep = if k == 0 {
429        logits.len()
430    } else {
431        k.min(logits.len())
432    };
433    if keep < logits.len() {
434        let threshold = kth_largest(logits, keep);
435        let mut above = 0usize;
436        for &value in logits.iter() {
437            if value > threshold {
438                above += 1;
439            }
440        }
441        let mut ties_allowed = keep.saturating_sub(above);
442        for value in logits.iter_mut() {
443            if *value > threshold {
444                continue;
445            }
446            if *value == threshold && ties_allowed > 0 {
447                ties_allowed -= 1;
448                continue;
449            }
450            *value = f32::NEG_INFINITY;
451        }
452    }
453
454    softmax_in_place(logits);
455    let mut mass = 0.0f32;
456    for &probability in logits.iter() {
457        if probability > 0.0 {
458            mass += probability;
459        }
460    }
461    let target = rng.next_f32() * mass;
462    let mut cumulative = 0.0f32;
463    let mut fallback = 0usize;
464    for (index, &probability) in logits.iter().enumerate() {
465        if probability > 0.0 {
466            fallback = index;
467            cumulative += probability;
468            if cumulative > target {
469                return Some(index);
470            }
471        }
472    }
473    // Only reachable when rounding puts `target` inside the last bucket's error.
474    Some(fallback)
475}

477/// `k`-th largest value of `x`, duplicates counted (1-based), or `-inf` when `x` holds fewer
478/// than `k` values above `-inf`. NaN elements rank nowhere and are never returned.
479fn kth_largest(x: &[f32], k: usize) -> f32 {
480    let mut upper: Option<f32> = None;
481    let mut counted = 0usize;
482    loop {
483        let mut next = f32::NEG_INFINITY;
484        let mut found = false;
485        for &value in x.iter() {
486            let eligible = match upper {
487                Some(bound) => value < bound,
488                None => true,
489            };
490            if eligible && value > next {
491                next = value;
492                found = true;
493            }
494        }
495        if !found {
496            return next;
497        }
498        for &value in x.iter() {
499            if value == next {
500                counted += 1;
501            }
502        }
503        if counted >= k {
504            return next;
505        }
506        upper = Some(next);
507    }
508}

510/// Writes the uniform distribution over `x`; `x` must be non-empty.
511fn fill_uniform(x: &mut [f32]) {
512    debug_assert!(!x.is_empty());
513    let uniform = 1.0 / x.len() as f32;
514    for slot in x.iter_mut() {
515        *slot = uniform;
516    }
517}

519/// Four-lane f32 accumulation of `a[i] * b[i]`; `a` and `b` must be equally long.
520#[inline]
521fn accumulate_products(acc: &mut [f32; 4], a: &[f32], b: &[f32]) {
522    debug_assert_eq!(a.len(), b.len());
523    for (left, right) in a.chunks_exact(4).zip(b.chunks_exact(4)) {
524        acc[0] += left[0] * right[0];
525        acc[1] += left[1] * right[1];
526        acc[2] += left[2] * right[2];
527        acc[3] += left[3] * right[3];
528    }
529    for (left, right) in a
530        .chunks_exact(4)
531        .remainder()
532        .iter()
533        .zip(b.chunks_exact(4).remainder())
534    {
535        acc[0] += left * right;
536    }
537}
538
539/// Collapses the four accumulation lanes in a fixed order.
540#[inline]
541fn reduce_lanes(acc: [f32; 4]) -> f32 {
542    (acc[0] + acc[1]) + (acc[2] + acc[3])
543}
544
545/// Sequential f32 inner product over equally long slices.
546#[inline]
547fn dot_f32(a: &[f32], b: &[f32]) -> f32 {
548    let mut acc = [0.0f32; 4];
549    accumulate_products(&mut acc, a, b);
550    reduce_lanes(acc)
551}

564    /// Packs 32 f32 weights into one Q8_0 block with an f16-exact power-of-two scale, and
565    /// returns the block's bytes alongside the weights it dequantizes to.
566    fn quantize_block(
567        values: &[f32],
568        scale_bits: u16,
569    ) -> ([u8; quant::Q8_0_BLOCK_BYTES], Vec<f32>) {
570        let scale = quant::f16_to_f32(scale_bits);
571        let mut block = [0u8; quant::Q8_0_BLOCK_BYTES];
572        block[..2].copy_from_slice(&scale_bits.to_le_bytes());
573        let mut dequantized = Vec::with_capacity(quant::Q8_0_BLOCK_WEIGHTS);
574        for (index, &value) in values.iter().enumerate() {
575            let quantized = if scale == 0.0 {
576                0.0
577            } else {
578                (value / scale).round().clamp(-127.0, 127.0)
579            };
580            block[2 + index] = quantized as i8 as u8;
581            dequantized.push(scale * quantized);
582        }
583        (block, dequantized)
584    }

656    fn check_q8_0_equivalence(rows: usize, cols: usize, seed: u64) {
657        let blocks_per_row = cols / quant::Q8_0_BLOCK_WEIGHTS;
658        let row_bytes = q8_0_row_bytes(cols).expect("cols is a multiple of 32");
659        let scale_bits = [0x2000u16, 0x1C00, 0x2400, 0x1800]; // 2^-7, 2^-9, 2^-6, 2^-11
660        let mut rng = Rng::new(seed);
661        let mut weights = vec![0u8; rows * row_bytes];
662        let mut dense = vec![0.0f32; rows * cols];
663        let mut magnitude = 0.0f32;
664        for r in 0..rows {
665            for b in 0..blocks_per_row {
666                let bits = scale_bits[(r + b) % scale_bits.len()];
667                let amplitude = quant::f16_to_f32(bits) * 126.0;
668                let values = random_values(&mut rng, quant::Q8_0_BLOCK_WEIGHTS, amplitude);
669                let (block, dequantized) = quantize_block(&values, bits);
670                let byte_offset = r * row_bytes + b * quant::Q8_0_BLOCK_BYTES;
671                weights[byte_offset..byte_offset + quant::Q8_0_BLOCK_BYTES].copy_from_slice(&block);
672                let float_offset = r * cols + b * quant::Q8_0_BLOCK_WEIGHTS;
673                dense[float_offset..float_offset + quant::Q8_0_BLOCK_WEIGHTS]
674                    .copy_from_slice(&dequantized);
675            }
676        }
677        let x = random_values(&mut rng, cols, 1.0);
678        let mut quantized_out = vec![0.0f32; rows];
679        matvec_q8_0(&mut quantized_out, &weights, &x, rows, cols).expect("shapes agree");
680        let mut dense_out = vec![0.0f32; rows];
681        matvec(&mut dense_out, &dense, &x, rows, cols).expect("shapes agree");
682
683        for r in 0..rows {
684            magnitude = magnitude.max(dense_out[r].abs());
685            assert!(
686                (quantized_out[r] - dense_out[r]).abs() < 1e-4,
687                "rows {rows}, cols {cols}, row {r}: {} vs {}",
688                quantized_out[r],
689                dense_out[r]
690            );
691        }
692        assert!(
693            magnitude > 1e-3,
694            "the fixture must produce non-trivial activations"
695        );
696    }

734    #[test]
735    fn matvec_q8_0_int8_stays_within_the_activation_quantization_bound() {
736        // The only difference from the f32-staging kernel is that the activation is rounded to i8:
737        // each element moves by at most d_a/2, which the row scales by |wᵢ| = d_w·|qᵢ|. Summing
738        // d_w·d_a·0.5·Σ|qᵢ| over the row's blocks is therefore a hard bound on the difference.
739        for &(rows, cols, seed) in &[(1usize, 32usize, 11u64), (3, 64, 12), (5, 160, 13)] {
740            let blocks_per_row = cols / quant::Q8_0_BLOCK_WEIGHTS;
741            let row_bytes = q8_0_row_bytes(cols).expect("cols is a multiple of 32");
742            let scale_bits = [0x2000u16, 0x1C00, 0x2400, 0x1800]; // 2^-7, 2^-9, 2^-6, 2^-11
743            let mut rng = Rng::new(seed);
744            let mut weights = vec![0u8; rows * row_bytes];
745            let mut bound = 0.0f32;
746            for r in 0..rows {
747                for b in 0..blocks_per_row {
748                    let bits = scale_bits[(r + b) % scale_bits.len()];
749                    let amplitude = quant::f16_to_f32(bits) * 126.0;
750                    let values = random_values(&mut rng, quant::Q8_0_BLOCK_WEIGHTS, amplitude);
751                    let (block, _) = quantize_block(&values, bits);
752                    let byte_offset = r * row_bytes + b * quant::Q8_0_BLOCK_BYTES;
753                    weights[byte_offset..byte_offset + quant::Q8_0_BLOCK_BYTES]
754                        .copy_from_slice(&block);
755                }
756            }
757            let x = random_values(&mut rng, cols, 1.0);
758            let mut x_q8 = vec![0u8; row_bytes];
759            quant::q8_0_row_from_f32(&x, &mut x_q8).expect("cols is a whole number of blocks");
760            for r in 0..rows {
761                for b in 0..blocks_per_row {
762                    let base = r * row_bytes + b * quant::Q8_0_BLOCK_BYTES;
763                    let d_w =
764                        quant::f16_to_f32(u16::from_le_bytes([weights[base], weights[base + 1]]));
765                    let d_a = quant::f16_to_f32(u16::from_le_bytes([
766                        x_q8[b * quant::Q8_0_BLOCK_BYTES],
767                        x_q8[b * quant::Q8_0_BLOCK_BYTES + 1],
768                    ]));
769                    let sum_abs: f32 = weights[base + 2..base + quant::Q8_0_BLOCK_BYTES]
770                        .iter()
771                        .map(|byte| (*byte as i8 as f32).abs())
772                        .sum();
773                    bound += d_w * d_a * 0.5 * sum_abs;
774                }
775            }
776
777            let mut int8_out = vec![0.0f32; rows];
778            matvec_q8_0_int8(&mut int8_out, &weights, &x_q8, rows, cols).expect("shapes agree");
779            let mut staging_out = vec![0.0f32; rows];
780            matvec_q8_0(&mut staging_out, &weights, &x, rows, cols).expect("shapes agree");
781
782            for r in 0..rows {
783                let error = (int8_out[r] - staging_out[r]).abs();
784                assert!(
785                    error <= bound + 1e-5,
786                    "rows {rows}, cols {cols}, row {r}: error {error} exceeds the bound {bound}"
787                );
788                assert!(
789                    int8_out[r] != 0.0 && staging_out[r] != 0.0,
790                    "row {r} must be non-trivial for the comparison to mean anything"
791                );
792            }
793        }
794    }

1171                sample_top_k(&mut mirror, 3, 1.0, &mut b),
1172                drawn.last().copied()

1175        for index in &drawn {

1177        }

1181        );
