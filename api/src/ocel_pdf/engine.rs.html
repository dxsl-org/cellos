2//! The only unsafe boundary: C owns MuPDF and catches all of its exceptions.
3use alloc::{string::String, vec::Vec};
4use core::{ffi::{c_char, c_int, c_void}, ptr::NonNull};
5use ocel_pdf_proto::{MAX_DOCUMENT_BYTES, MAX_PAGE_PIXELS, PIXEL_CHUNK_BYTES};
6
7extern "C" {
8    fn ocel_pdf_open(input: *const u8, length: usize, pages: *mut u32,
9        error: *mut c_char, capacity: usize) -> *mut c_void;
10    fn ocel_pdf_close(document: *mut c_void);
11    fn ocel_pdf_render(document: *mut c_void, page: u32, max_width: u32, max_height: u32,
12        width: *mut u32, height: *mut u32, bytes: *mut u32,
13        error: *mut c_char, capacity: usize) -> c_int;
14    fn ocel_pdf_read_pixels(document: *mut c_void, offset: usize, output: *mut u8, length: usize) -> c_int;
15}
16
17pub struct Document {
18    native: NonNull<c_void>,
19    // fz_open_memory borrows these bytes. Drop closes native BEFORE Rust drops
20    // the vector. Moving the Document never moves the vector's allocation.
21    _input: Vec<u8>,
22    pub pages: u32,
23    raster_bytes: usize,
24}
25
26fn message(error: &[u8]) -> String {
27    let length = error.iter().position(|&c| c == 0).unwrap_or(error.len());
28    String::from_utf8_lossy(&error[..length]).into_owned()
29}
30
31impl Document {
32    pub fn open(input: Vec<u8>) -> Result<Self, String> {
33        if input.is_empty() || input.len() > MAX_DOCUMENT_BYTES {
34            return Err(String::from("Invalid PDF document size"));
35        }
36        let mut pages = 0;
37        let mut error = [0u8; 256];
38        // SAFETY: inputs remain alive for the whole document lifetime; outputs
39        // are writable and C returns normally after catching any exception.
40        let pointer = unsafe { ocel_pdf_open(input.as_ptr(), input.len(), &mut pages,
41            error.as_mut_ptr().cast(), error.len()) };
42        let native = NonNull::new(pointer).ok_or_else(|| message(&error))?;
43        Ok(Self { native, _input: input, pages, raster_bytes: 0 })
44    }
45
46    pub fn render(&mut self, page: u32, max_width: u32, max_height: u32) -> Result<(u32, u32, u32), String> {
47        if page >= self.pages || max_width == 0 || max_height == 0 ||
48            max_width as usize > MAX_PAGE_PIXELS || max_height as usize > MAX_PAGE_PIXELS {
49            return Err(String::from("Invalid PDF page or raster dimensions"));
50        }
51        self.raster_bytes = 0;
52        let (mut width, mut height, mut bytes) = (0, 0, 0);
53        let mut error = [0u8; 256];
54        // SAFETY: native is live, exclusively accessed, and C bounds output.
55        let status = unsafe { ocel_pdf_render(self.native.as_ptr(), page, max_width, max_height,
56            &mut width, &mut height, &mut bytes, error.as_mut_ptr().cast(), error.len()) };
57        if status != 0 { return Err(message(&error)); }
58        if width == 0 || height == 0 || width > max_width || height > max_height ||
59            (width as usize).checked_mul(height as usize).filter(|&n| n <= MAX_PAGE_PIXELS)
60                .and_then(|n| n.checked_mul(4)) != Some(bytes as usize) {
61            return Err(String::from("Invalid MuPDF raster result"));
62        }
63        self.raster_bytes = bytes as usize;
64        Ok((width, height, bytes))
65    }
66
67    pub fn read_pixels(&self, offset: usize, length: usize) -> Result<Vec<u8>, String> {
68        if self.raster_bytes == 0 || length == 0 || length > PIXEL_CHUNK_BYTES ||
69            offset.checked_add(length).filter(|&end| end <= self.raster_bytes).is_none() {
70            return Err(String::from("Invalid PDF raster range"));
71        }
72        let mut output = Vec::new();
73        output.try_reserve_exact(length).map_err(|_| String::from("Pixel allocation failed"))?;
74        output.resize(length, 0);
75        // SAFETY: C validates offset/length independently; output is initialized
76        // writable storage of exactly length, and native remains owned by self.
77        if unsafe { ocel_pdf_read_pixels(self.native.as_ptr(), offset, output.as_mut_ptr(), length) } != 0 {
78            return Err(String::from("PDF raster is unavailable"));
79        }
80        Ok(output)
81    }
82}
83
84impl Drop for Document {
85    fn drop(&mut self) {
86        // SAFETY: this pointer is exclusively owned and closed exactly once.
87        unsafe { ocel_pdf_close(self.native.as_ptr()); }
88    }
89}

91#[no_mangle]
92pub extern "C" fn ocel_pdf_exit(status: c_int) -> ! {
93    ostd::syscall::sys_exit(status as usize)
