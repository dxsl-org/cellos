31fn read_document(path: &str) -> Result<Vec<u8>, String> {
32    if path.is_empty() || path.len() > 512 || path.as_bytes().contains(&0) {
33        return Err(String::from("Invalid PDF file path"));
34    }
35    let mut file = ostd::fs::File::open(path).map_err(|_| String::from("Cannot open PDF file"))?;
36    let mut bytes = Vec::new();
37    // Reserve once: no repeated copying of the bounded document. The extra
38    // byte is read separately to reject oversized inputs rather than truncate.
39    bytes.try_reserve_exact(MAX_DOCUMENT_BYTES).map_err(|_| String::from("PDF input allocation failed"))?;
40    let result = (|| {
41        let mut chunk = [0u8; 4096];
42        loop {
43            let remaining = MAX_DOCUMENT_BYTES - bytes.len();
44            let count = file.read(&mut chunk[..remaining.min(4096).max(1)])
45                .map_err(|_| String::from("Cannot read PDF file"))?;
46            if count == 0 { break; }
47            if count > remaining { return Err(String::from("PDF exceeds 2 MiB input limit")); }
48            bytes.extend_from_slice(&chunk[..count]);
49        }
50        if bytes.is_empty() { return Err(String::from("PDF file is empty")); }
51        Ok(bytes)
52    })();
53    let _ = file.close();
54    result
55}

57fn handle(request: Request, owner: usize, active: &mut Option<OwnedDocument>, next: &mut u32) -> Result<Response, String> {
58    match request {
59        Request::Ping => Ok(Response::Ready),
60        Request::Open { path } => {
61            if active.is_some() { return Err(String::from("PDF service busy: close the active document first")); }
62            if *next == 0 { return Err(String::from("PDF handle space exhausted")); }
63            let document = engine::Document::open(read_document(&path)?)?;
64            let pages = document.pages;
65            let handle = *next;
66            *next = next.checked_add(1).unwrap_or(0);
67            *active = Some(OwnedDocument { owner, handle, document });
68            Ok(Response::Opened { document: handle, pages })
69        }
70        Request::RenderPage { document, page, max_width, max_height } => {
71            let live = active.as_mut().filter(|d| d.owner == owner && d.handle == document)
72                .ok_or_else(|| String::from("Unknown PDF document handle"))?;
73            let (width, height, bytes) = live.document.render(page, max_width, max_height)?;
74            Ok(Response::Rendered { width, height, bytes })
75        }
76        Request::ReadPixels { document, offset, length } => {
77            let live = active.as_ref().filter(|d| d.owner == owner && d.handle == document)
78                .ok_or_else(|| String::from("Unknown PDF document handle"))?;
79            let bytes = live.document.read_pixels(offset as usize, length as usize)?;
80            Ok(Response::Pixels { offset, bytes })
81        }
82        Request::Close { document } => {
83            if !active.as_ref().is_some_and(|d| d.owner == owner && d.handle == document) {
84                return Err(String::from("Unknown PDF document handle"));
85            }
86            *active = None;
87            Ok(Response::Closed)
88        }
89    }
90}

92fn cell_main() {
93    init_custom_heap();
94    ostd::io::println("[ocel-pdf] MuPDF 1.26.1 native PDF renderer ready (AGPL-3.0-or-later)");
95    // Registered by init. This service never spawns cells or self-registers.
96    let mut active = None;
97    let mut next_handle = 1;
98    let mut receive = [0u8; IPC_BYTES];
99    let mut send = [0u8; IPC_BYTES];
100    loop {
101        receive.fill(0);
102        match sys_recv(0, &mut receive) {
103            SyscallResult::Ok(owner) if owner > 0 => {
104                let operation = ostd::ipc::current();
105                let response = match postcard::from_bytes::<Request>(&receive) {
106                    Ok(request) => handle(request, owner, &mut active, &mut next_handle)
107                        .unwrap_or_else(|message| Response::Error { message }),
108                    Err(_) => Response::Error { message: String::from("Malformed PDF request") },
109                };
110                if let Ok(encoded) = postcard::to_slice(&response, &mut send) {
111                    if let Some(operation) = operation {
112                        let _ = ostd::ipc::reply(operation, encoded);
113                    } else {
114                        let _ = sys_send(owner, encoded);
115                    }
116                }
117            }
118            _ => sys_yield(),
119        }
120    }
121}
