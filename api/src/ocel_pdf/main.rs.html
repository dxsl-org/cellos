30fn read_document(path: &str) -> Result<Vec<u8>, String> {
31    if path.is_empty() || path.len() > 512 || path.as_bytes().contains(&0) {
32        return Err(String::from("Invalid PDF file path"));
33    }
34    let mut file = ostd::fs::File::open(path).map_err(|_| String::from("Cannot open PDF file"))?;
35    let mut bytes = Vec::new();
36    // Reserve once: no repeated copying of the bounded document. The extra
37    // byte is read separately to reject oversized inputs rather than truncate.
38    bytes.try_reserve_exact(MAX_DOCUMENT_BYTES).map_err(|_| String::from("PDF input allocation failed"))?;
39    let result = (|| {
40        let mut chunk = [0u8; 4096];
41        loop {
42            let remaining = MAX_DOCUMENT_BYTES - bytes.len();
43            let count = file.read(&mut chunk[..remaining.min(4096).max(1)])
44                .map_err(|_| String::from("Cannot read PDF file"))?;
45            if count == 0 { break; }
46            if count > remaining { return Err(String::from("PDF exceeds 2 MiB input limit")); }
47            bytes.extend_from_slice(&chunk[..count]);
48        }
49        if bytes.is_empty() { return Err(String::from("PDF file is empty")); }
50        Ok(bytes)
51    })();
52    let _ = file.close();
53    result
54}

56fn handle(request: Request, owner: usize, active: &mut Option<OwnedDocument>, next: &mut u32) -> Result<Response, String> {
57    let now = ostd::syscall::sys_get_scheduler_ticks().ok_or_else(|| String::from("PDF session clock unavailable"))?;
58    if active.as_ref().is_some_and(|d| d.lease.expired(now)) { *active = None; }
59    match request {
60        Request::Ping => Ok(Response::Ready),
61        Request::Open { path } => {
62            if active.is_some() { return Err(String::from("PDF service busy: close the active document first")); }
63            if *next == 0 { return Err(String::from("PDF handle space exhausted")); }
64            let document = engine::Document::open(read_document(&path)?)?;
65            let pages = document.pages;
66            let handle = *next;
67            *next = next.checked_add(1).unwrap_or(0);
68            *active = Some(OwnedDocument { lease: DocumentOwner::new(owner, handle, now), document });
69            Ok(Response::Opened { document: handle, pages })
70        }
71        Request::RenderPage { document, page, max_width, max_height } => {
72            let live = active.as_mut().filter(|d| d.lease.matches(owner, document))
73                .ok_or_else(|| String::from("Unknown PDF document handle"))?;
74            live.lease.renew(owner, document, now);
75            let result = live.document.render(page, max_width, max_height);
76            // Any native failure discards this owner's document and cache. Never
77            // let an abandoned or partially failed render stay busy indefinitely.
78            let (width, height, bytes) = match result {
79                Ok(result) => result,
80                Err(error) => { *active = None; return Err(error); }
81            };
82            Ok(Response::Rendered { width, height, bytes })
83        }
84        Request::ReadPixels { document, offset, length } => {
85            let live = active.as_mut().filter(|d| d.lease.matches(owner, document))
86                .ok_or_else(|| String::from("Unknown PDF document handle"))?;
87            live.lease.renew(owner, document, now);
88            let bytes = live.document.read_pixels(offset as usize, length as usize)?;
89            Ok(Response::Pixels { offset, bytes })
90        }
91        Request::Close { document } => {
92            if !active.as_ref().is_some_and(|d| d.lease.matches(owner, document)) {
93                return Err(String::from("Unknown PDF document handle"));
94            }
95            *active = None;
96            Ok(Response::Closed)
97        }
98    }
99}

101fn cell_main() {
102    init_custom_heap();
103    engine::Document::init_deadline_clock();
104    ostd::io::println("[ocel-pdf] MuPDF 1.26.1 native PDF renderer ready (AGPL-3.0-or-later)");
105    // Registered by init. This service never spawns cells or self-registers.
106    let mut active = None;
107    let mut next_handle = 1;
108    let mut receive = [0u8; IPC_BYTES];
109    let mut send = [0u8; IPC_BYTES];
110    loop {
111        receive.fill(0);
112        match sys_recv(0, &mut receive) {
113            SyscallResult::Ok(owner) if owner > 0 => {
114                let operation = ostd::ipc::current();
115                let mut requested_document = None;
116                let response = match postcard::from_bytes::<Request>(&receive) {
117                    Ok(request) => {
118                        requested_document = match &request {
119                            Request::RenderPage { document, .. } | Request::ReadPixels { document, .. }
120                            | Request::Close { document } => Some(*document),
121                            Request::Open { .. } | Request::Ping => None,
122                        };
123                        handle(request, owner, &mut active, &mut next_handle)
124                            .unwrap_or_else(|message| Response::Error { message })
125                    }
126                    Err(_) => Response::Error { message: String::from("Malformed PDF request") },
127                };
128                if let Response::Opened { document, .. } = &response { requested_document = Some(*document); }
129                if let Ok(encoded) = postcard::to_slice(&response, &mut send) {
130                    if let Some(operation) = operation {
131                        if ostd::ipc::reply(operation, encoded).is_err()
132                            && active.as_ref().is_some_and(|d| requested_document.is_some_and(|handle| d.lease.matches(owner, handle))) {
133                            // Exact IPC cancellation abandons the result, not the
134                            // native stack. Cleanup occurs after C returned.
135                            active = None;
136                        }
137                    } else {
138                        let _ = sys_send(owner, encoded);
139                    }
140                }
141            }
142            _ => sys_yield(),
143        }
144    }
145}
