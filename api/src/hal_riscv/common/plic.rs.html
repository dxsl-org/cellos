1//! PLIC (Platform-Level Interrupt Controller) Driver for RISC-V.
2//! Reference: https://github.com/riscv/riscv-plic-spec/blob/master/riscv-plic.adoc
3
4use core::sync::atomic::{AtomicUsize, Ordering};
5
6pub const PLIC_PRIORITY_BASE: usize = 0x0;
7pub const PLIC_PENDING_BASE: usize = 0x1000;
8pub const PLIC_ENABLE_BASE: usize = 0x2000;
9pub const PLIC_THRESHOLD_AND_CLAIM_BASE: usize = 0x20_0000;
10
11// PLIC context numbering is SoC policy. The kernel resolves the current
12// physical hart through `hal/soc/riscv` and passes the selected context into
13// this shared register-access mechanism.
14
15/// Runtime PLIC base address. Updated before `init()` via `set_plic_base()`.
16static PLIC_RUNTIME_BASE: AtomicUsize = AtomicUsize::new(0);
17
18/// Override the PLIC base address before `init()` is called (called from kernel
19/// after DTB parsing populates `platform::PlatformInfo`).
20pub fn set_plic_base(base: usize) {
21    PLIC_RUNTIME_BASE.store(base, Ordering::Relaxed);
22}
23
24pub struct Plic;
25
26impl Plic {
27    pub const fn new() -> Self {
28        Self
29    }
30
31    fn base() -> usize {
32        let base = PLIC_RUNTIME_BASE.load(Ordering::Relaxed);
33        assert!(
34            base != 0,
35            "PLIC base must be selected before register access"
36        );
37        base
38    }
39
40    /// Set priority for a specific IRQ.
41    /// Priority: 0 (disabled) to 7 (highest).
42    pub fn set_priority(&self, irq: u32, priority: u32) {
43        let addr = Self::base() + PLIC_PRIORITY_BASE + (irq as usize) * 4;
44        // SAFETY: `addr` points into the identity-mapped PLIC MMIO aperture
45        // selected by `set_plic_base()`. The caller provides a device IRQ id.
46        unsafe {
47            (addr as *mut u32).write_volatile(priority);
48        }
49    }
50
51    /// Enable interrupt for a specific Context.
52    pub fn enable(&self, context: usize, irq: u32) {
53        let addr = Self::base() + PLIC_ENABLE_BASE + (context * 0x80) + ((irq as usize / 32) * 4);
54        let mask = 1 << (irq % 32);
55        // SAFETY: same MMIO aperture contract as `set_priority`; read-modify-write
56        // is required by the PLIC enable register layout.
57        unsafe {
58            let ptr = addr as *mut u32;
59            ptr.write_volatile(ptr.read_volatile() | mask);
60        }
61    }
62
63    /// Set priority threshold for a specific Context.
64    /// Interrupts <= threshold are masked.
65    pub fn set_threshold(&self, context: usize, threshold: u32) {
66        let addr = Self::base() + PLIC_THRESHOLD_AND_CLAIM_BASE + (context * 0x1000);
67        // SAFETY: `addr` points at this context's threshold register inside the
68        // configured PLIC MMIO region.
69        unsafe {
70            (addr as *mut u32).write_volatile(threshold);
71        }
72    }
73
74    /// Claim an interrupt for a specific Context.
75    /// Returns the IRQ number, or 0 if none.
76    pub fn claim(&self, context: usize) -> u32 {
77        let addr = Self::base() + PLIC_THRESHOLD_AND_CLAIM_BASE + (context * 0x1000) + 4;
78        // SAFETY: `addr` points at this context's claim register inside the
79        // configured PLIC MMIO region.
80        unsafe { (addr as *mut u32).read_volatile() }
81    }
82
83    /// Complete an interrupt for a specific Context.
84    pub fn complete(&self, context: usize, irq: u32) {
85        let addr = Self::base() + PLIC_THRESHOLD_AND_CLAIM_BASE + (context * 0x1000) + 4;
86        // SAFETY: `addr` points at this context's completion register inside the
87        // configured PLIC MMIO region.
88        unsafe {
89            (addr as *mut u32).write_volatile(irq);
90        }
91    }
92}
93
94// Global PLIC instance (zero-size; all state in PLIC_RUNTIME_BASE).
95pub static PLIC: Plic = Plic::new();
96
97/// Initialize the active S-mode PLIC context and enable the provided IRQs.
98///
99/// Uses the base address set by `set_plic_base()` — call that first from the
100/// kernel after platform discovery. Missing or empty runtime IRQ lists fail
101/// closed by enabling no device IRQs.
102pub fn init(context: usize, irqs: &[u32]) {
103    PLIC.set_threshold(context, 0);
104    for &irq in irqs {
105        if irq == 0 {
106            continue;
107        }
108        PLIC.set_priority(irq, 1);
109        PLIC.enable(context, irq);
110    }
111}
112
113/// Claim the highest-priority pending IRQ from one S-mode context.
114pub fn claim(context: usize) -> Option<u32> {
115    let irq = PLIC.claim(context);
116    (irq != 0).then_some(irq)
117}
118
119/// Notify the PLIC that `context` has finished handling `irq`.
120pub fn complete(context: usize, irq: u32) {
121    if irq == 0 {
122        return;
123    }
124    PLIC.complete(context, irq);
125}
