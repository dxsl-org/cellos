7// SBI Extension IDs
8const SBI_EID_TIMER: usize = 0x54494D45;
9const SBI_EID_LEGACY_SET_TIMER: usize = 0x00;
10const SBI_EID_DBCN: usize = 0x4442434E;
11const SBI_FID_DBCN_WRITE: usize = 0;
12const SBI_EID_BASE: usize = 0x10;
13const SBI_FID_PROBE_EXTENSION: usize = 3;
14const SBI_EID_RFENCE: usize = 0x5246_4E43;
15const SBI_FID_REMOTE_SFENCE_VMA: usize = 1;
16
17#[inline(always)]
18fn sbi_call(eid: usize, fid: usize, arg0: usize, arg1: usize, arg2: usize) -> (usize, usize) {
19    let mut error;
20    let mut value;
21    unsafe {
22        core::arch::asm!(
23            "ecall",
24            in("a7") eid,
25            in("a6") fid,
26            in("a0") arg0,
27            in("a1") arg1,
28            in("a2") arg2,
29            lateout("a0") error,
30            lateout("a1") value,
31            options(nostack)
32        );
33    }
34    (error, value)
35}

37/// SBI call variant for extensions whose fourth argument occupies `a3`.
38#[inline(always)]
39fn sbi_call4(
40    eid: usize,
41    fid: usize,
42    arg0: usize,
43    arg1: usize,
44    arg2: usize,
45    arg3: usize,
46) -> (usize, usize) {
47    let mut error;
48    let mut value;
49    // SAFETY: SBI defines this ecall register convention; inputs are plain
50    // machine words and the firmware owns all privileged side effects.
51    unsafe {
52        core::arch::asm!(
53            "ecall",
54            in("a7") eid,
55            in("a6") fid,
56            in("a0") arg0,
57            in("a1") arg1,
58            in("a2") arg2,
59            in("a3") arg3,
60            lateout("a0") error,
61            lateout("a1") value,
62            options(nostack)
63        );
64    }
65    (error, value)
66}

68/// Whether firmware advertises the RFENCE extension required before RV64 SMP.
69pub fn sbi_rfence_available() -> Result<bool, usize> {
70    let (error, value) = sbi_call(SBI_EID_BASE, SBI_FID_PROBE_EXTENSION, SBI_EID_RFENCE, 0, 0);
71    if error == 0 {
72        Ok(value != 0)
73    } else {
74        Err(error)
75    }
76}

78/// Synchronously invalidate a virtual-address range on selected remote harts.
79///
80/// `hart_mask` uses the SBI hart-mask encoding with `hart_mask_base = 0`.
81/// `start_addr` and `size` must describe page-aligned addresses and lengths.
82pub fn sbi_remote_sfence_vma(
83    hart_mask: usize,
84    hart_mask_base: usize,
85    start_addr: usize,
86    size: usize,
87) -> Result<(), usize> {
88    let (error, _) = sbi_call4(
89        SBI_EID_RFENCE,
90        SBI_FID_REMOTE_SFENCE_VMA,
91        hart_mask,
92        hart_mask_base,
93        start_addr,
94        size,
95    );
96    if error == 0 {
97        Ok(())
98    } else {
99        Err(error)
100    }
101}

103/// Write a character to debug console (DBCN)
104pub fn console_putchar(c: u8) -> Result<(), usize> {
105    let byte = c;
106    let ptr = &byte as *const u8 as usize;
107    // console_write(num_bytes=1, base_addr_lo=ptr, base_addr_hi=0)
108    let (error, _value) = sbi_call(SBI_EID_DBCN, SBI_FID_DBCN_WRITE, 1, ptr, 0);
109
110    if error == 0 {
111        Ok(())
112    } else {
113        Err(error)
114    }
115}

117const SBI_FID_DBCN_READ: usize = 1;
118
119/// Read a character from debug console (DBCN)
120pub fn console_getchar() -> isize {
121    let mut byte: u8 = 0;
122    let ptr = &mut byte as *mut u8 as usize;
123    let (error, value) = sbi_call(SBI_EID_DBCN, SBI_FID_DBCN_READ, 1, ptr, 0);
124
125    if error == 0 && value == 1 {
126        byte as isize
127    } else {
128        -1
129    }
130}

132const SBI_FID_SET_TIMER: usize = 0;
133
134/// Set the next timer interrupt deadline.
135///
136/// On RV32 the SBI spec (v1.0 §5.1) requires passing `stime_value_lo` in
137/// `a0` and `stime_value_hi` in `a1` because `usize` is 32-bit — the 64-bit
138/// value cannot fit in a single argument register.
139pub fn set_timer(stime_value: u64) {
140    #[cfg(target_arch = "riscv64")]
141    sbi_call(SBI_EID_TIMER, SBI_FID_SET_TIMER, stime_value as usize, 0, 0);
142    #[cfg(target_arch = "riscv32")]
143    {
144        let lo = stime_value as usize; // bits 31:0
145        let hi = (stime_value >> 32) as usize; // bits 63:32
146        sbi_call(SBI_EID_TIMER, SBI_FID_SET_TIMER, lo, hi, 0);
147    }
148}

150// SBI System Reset (SRST) extension — used by the panic handler to reboot a
151// dead kernel instead of freezing (a robot must come back up, not brick).
152const SBI_EID_SRST: usize = 0x53525354;
153const SBI_FID_SYSTEM_RESET: usize = 0;
154/// SRST reset_type: graceful power-off.
155pub const SBI_RESET_SHUTDOWN: usize = 0x0;
156/// SRST reset_type: full re-init (re-runs firmware + kernel from scratch).
157pub const SBI_RESET_COLD_REBOOT: usize = 0x1;
158
159/// Request a system reset via SBI SRST.
160///
161/// On success the machine resets and this never returns; it returns only if the
162/// firmware does not implement SRST, letting the caller fall back to a halt loop.
163pub fn system_reset(reset_type: usize, reset_reason: usize) {
164    sbi_call(
165        SBI_EID_SRST,
166        SBI_FID_SYSTEM_RESET,
167        reset_type,
168        reset_reason,
169        0,
170    );
171}

173// --- SBI Hart State Management (HSM) extension  SBI spec v2.0 §9 ---
174const SBI_EID_HSM: usize = 0x48534D;
175const SBI_FID_HART_START: usize = 0;
176const SBI_FID_HART_GET_STATUS: usize = 2;
177
178/// Request OpenSBI to start a hart that is in STOPPED state.
179///
180/// `start_addr` is the PHYSICAL address of the secondary's first instruction
181/// (must be 4-byte aligned; hart starts bare — SATP=0).  `opaque` is passed
182/// in the secondary's `a1` register on entry (SBI HSM §9.1.1) and is
183/// conventionally the kernel stack top for that hart.
184///
185/// # Safety
186/// `start_addr` must be a valid, aligned physical address reachable without paging.
187pub fn sbi_hart_start(hart_id: usize, start_addr: usize, opaque: usize) -> Result<(), usize> {
188    let (error, _) = sbi_call(SBI_EID_HSM, SBI_FID_HART_START, hart_id, start_addr, opaque);
189    if error == 0 {
190        Ok(())
191    } else {
192        Err(error)
193    }
194}
195
196/// Query the current HSM state of a hart.
197///
198/// Returns: 0 = Started, 1 = Stopped, 2 = Start Pending, 3 = Stop Pending,
199/// 4 = Suspended, 5 = Suspend Pending, 6 = Resume Pending.
200pub fn sbi_hart_get_status(hart_id: usize) -> Result<usize, usize> {
201    let (error, value) = sbi_call(SBI_EID_HSM, SBI_FID_HART_GET_STATUS, hart_id, 0, 0);
202    if error == 0 {
203        Ok(value)
204    } else {
205        Err(error)
206    }
207}

209// --- SBI IPI (Send Inter-Processor Interrupt) extension  SBI spec v2.0 §4 ---
210const SBI_EID_IPI: usize = 0x735049;
211const SBI_FID_SEND_IPI: usize = 0;
212
213/// Send a software IPI to all harts selected by `hart_mask`.
214///
215/// Bit N in `hart_mask` targets hart `hart_mask_base + N`.  Pass
216/// `hart_mask_base = 0` for absolute hart IDs (e.g. `1 << 1` to ping hart 1).
217/// The IPI sets SSIP on each target; the target's trap handler sees scause=1.
218pub fn sbi_send_ipi(hart_mask: usize, hart_mask_base: usize) -> Result<(), usize> {
219    let (error, _) = sbi_call(SBI_EID_IPI, SBI_FID_SEND_IPI, hart_mask, hart_mask_base, 0);
220    if error == 0 {
221        Ok(())
222    } else {
223        Err(error)
224    }
