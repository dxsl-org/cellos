1//! SATP and ASID operations used exclusively by private native-domain roots.
2//!
3//! The scheduler must not write SATP directly. These operations keep the PTE-store,
4//! SATP, and `sfence.vma` ordering at the architecture boundary.
5
6#[cfg(feature = "test-hooks")]
7use core::sync::atomic::{AtomicUsize, Ordering};
8
9#[cfg(feature = "test-hooks")]
10static SATP_WRITES: AtomicUsize = AtomicUsize::new(0);
11#[cfg(feature = "test-hooks")]
12static ASID_FLUSHES: AtomicUsize = AtomicUsize::new(0);
13
14/// Test-only observation of scheduler domain activation, never an admission signal.
15#[cfg(feature = "test-hooks")]
16pub fn switch_counters() -> (usize, usize) {
17    (
18        SATP_WRITES.load(Ordering::Acquire),
19        ASID_FLUSHES.load(Ordering::Acquire),
20    )
21}
22
23#[cfg(feature = "test-hooks")]
24pub fn reset_switch_counters() {
25    SATP_WRITES.store(0, Ordering::Release);
26    ASID_FLUSHES.store(0, Ordering::Release);
27}
28
29/// Records a scheduler-proven SATP + ASID fence pair for test fixtures.
30#[inline]
31pub fn observe_switch_activation() {
32    #[cfg(feature = "test-hooks")]
33    {
34        SATP_WRITES.fetch_add(1, Ordering::Relaxed);
35        ASID_FLUSHES.fetch_add(1, Ordering::Relaxed);
36    }
37}

40#[derive(Clone, Copy, Debug, Eq, PartialEq)]
41pub enum DomainPagingError {
42    Firmware(usize),
43}
44
45/// Install an Sv39 root and invalidate stale translations for its ASID.
46///
47/// `root_ppn` is a physical page number, not a byte address.
48#[inline]
49pub fn activate_address_space(root_ppn: usize, asid: usize) {
50    let satp = (8usize << 60) | ((asid & 0xffff) << 44) | (root_ppn & ((1usize << 44) - 1));
51    // SAFETY: this is called only after the private root and its supervisor
52    // entry pages have been completed; the fence orders every preceding PTE store.
53    unsafe {
54        core::arch::asm!(
55            "fence rw, rw",
56            "csrw satp, {satp}",
57            "sfence.vma zero, {asid}",
58            satp = in(reg) satp,
59            asid = in(reg) asid,
60            options(nostack),
61        );
62    }
63    #[cfg(feature = "test-hooks")]
64    {
65        SATP_WRITES.fetch_add(1, Ordering::Relaxed);
66        ASID_FLUSHES.fetch_add(1, Ordering::Relaxed);
67    }
68}
69
70/// Invalidate local translations tagged with `asid`.
71#[inline]
72pub fn flush_asid(asid: usize) {
73    // SAFETY: `sfence.vma` is valid in S-mode and does not access Rust memory.
74    unsafe {
75        core::arch::asm!("sfence.vma zero, {asid}", asid = in(reg) asid, options(nostack));
76    }
77    #[cfg(feature = "test-hooks")]
78    ASID_FLUSHES.fetch_add(1, Ordering::Relaxed);
79}

81/// Invalidate a selected hart set before its ASID can be recycled.
82///
83/// SBI RFENCE has no single-ASID call in the deployed firmware baseline, so this
84/// deliberately performs the stronger all-ASID remote invalidation.
85pub fn flush_asid_remote(hart_mask: usize, _asid: usize) -> Result<(), DomainPagingError> {
86    if hart_mask == 0 {
87        return Ok(());
88    }
89    sbi::sbi_remote_sfence_vma(hart_mask, 0, 0, usize::MAX).map_err(DomainPagingError::Firmware)
90}
91
92/// Invalidate every local translation before an ASID epoch wraps.
93#[inline]
94pub fn flush_all() {
95    // SAFETY: see `flush_asid`; x0 selects every virtual address and ASID.
96    unsafe {
97        core::arch::asm!("sfence.vma zero, zero", options(nostack));
98    }
