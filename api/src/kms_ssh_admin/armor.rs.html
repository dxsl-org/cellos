1//! Strict decode of one armored `OPENSSH PRIVATE KEY` block (k7-cp1-draft §3.3).
2//!
3//! The cell never parses key contents: the `openssh-key-v1\0` magic and every structural
4//! check stay in KMS (D-K4). This module only turns pasted text into the binary blob KMS
5//! is given, and refuses everything that is not exactly one armored block, before any KMS
6//! frame exists.
7
8use base64ct::{Base64, Encoding};
9
10use crate::secret::Secret;
11
12pub const BEGIN: &[u8] = b"-----BEGIN OPENSSH PRIVATE KEY-----";
13pub const END: &[u8] = b"-----END OPENSSH PRIVATE KEY-----";
14/// Raw text cap: a 65,536-byte blob is 87,384 base64 bytes plus line breaks and the two
15/// marker lines (O-K7-4).
16pub const TEXT_MAX: usize = 98_304;
17/// The ABI cap on one blob (`KMS_SSH_IMPORT_MAX_LEN`).
18pub const BLOB_MAX: usize = ostd::clients::kms::KMS_SSH_IMPORT_MAX_LEN;
19/// Base64 text of a `BLOB_MAX`-byte blob, canonical padding included.
20const BODY_MAX: usize = BLOB_MAX.div_ceil(3) * 4;
21
22/// Why a pasted text is not one armored key block. One fixed message per variant; no
23/// message ever carries input bytes.
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum ArmorError {
26    /// More than [`TEXT_MAX`] bytes.
27    TextTooLong,
28    /// No bytes at all (or only line breaks).
29    Empty,
30    /// A bare carriage return not followed by a line feed.
31    BareCarriageReturn,
32    /// The first line is not the exact BEGIN marker.
33    MissingBegin,
34    /// No exact END marker line closes the block.
35    MissingEnd,
36    /// A second block, a second BEGIN or an END marker inside the body.
37    SecondBlock,
38    /// Bytes other than line breaks after the END line.
39    TextAfterBlock,
40    /// An empty line, a space, a header line or any byte outside the base64 alphabet
41    /// between the markers.
42    BadBodyLine,
43    /// Nothing between the markers.
44    EmptyBody,
45    /// Not canonical standard base64 (length, padding or trailing bits).
46    BadBase64,
47    /// The decoded blob is empty or longer than [`BLOB_MAX`].
48    BadLength,
49}
50
51impl ArmorError {
52    pub const fn message(self) -> &'static str {
53        match self {
54            Self::TextTooLong => "key text is longer than 98304 bytes",
55            Self::Empty => "no key text",
56            Self::BareCarriageReturn => "line breaks must be LF or CRLF",
57            Self::MissingBegin => "the key text must start with the BEGIN OPENSSH PRIVATE KEY line",
58            Self::MissingEnd => "the key text must end with the END OPENSSH PRIVATE KEY line",
59            Self::SecondBlock => "exactly one key block is accepted",
60            Self::TextAfterBlock => "nothing may follow the END line",
61            Self::BadBodyLine => "the key body must be non-empty lines of base64 only",
62            Self::EmptyBody => "the key body is empty",
63            Self::BadBase64 => "the key body is not canonical base64",
64            Self::BadLength => "the decoded key must be 1 to 65536 bytes",
65        }
66    }
67}

69/// Split `text` into lines (LF or CRLF terminated; a final unterminated line counts) and
70/// call `each` with every line without its terminator. A lone CR is an error.
71fn for_each_line<'a>(
72    text: &'a [u8],
73    mut each: impl FnMut(&'a [u8]) -> Result<(), ArmorError>,
74) -> Result<(), ArmorError> {
75    let mut start = 0;
76    while start < text.len() {
77        let rest = &text[start..];
78        let (line, advance) = match rest.iter().position(|&b| b == b'\n') {
79            Some(at) => {
80                let line = &rest[..at];
81                (line.strip_suffix(b"\r").unwrap_or(line), at + 1)
82            }
83            None => (rest, rest.len()),
84        };
85        if line.contains(&b'\r') {
86            return Err(ArmorError::BareCarriageReturn);
87        }
88        each(line)?;
89        start += advance;
90    }
91    Ok(())
92}

94/// The standard-alphabet byte set (padding is checked by the decoder).
95const fn base64_byte(b: u8) -> bool {
96    b.is_ascii_alphanumeric() || b == b'+' || b == b'/' || b == b'='
97}

99/// Decode the single armored block in `text` to the binary OpenSSH blob.
100///
101/// Accepted: optional empty lines, the exact BEGIN line, one or more non-empty body lines
102/// of the standard base64 alphabet (any width), the exact END line, optional empty lines.
103/// Line breaks are LF or CRLF. Padding must be canonical and only at the end. Everything
104/// else is refused.
105pub fn decode(text: &[u8]) -> Result<Secret, ArmorError> {
106    if text.len() > TEXT_MAX {
107        return Err(ArmorError::TextTooLong);
108    }
109    enum At {
110        BeforeBegin,
111        Body,
112        AfterEnd,
113    }
114    let mut at = At::BeforeBegin;
115    let mut saw_text = false;
116    let mut saw_marker = false;
117    // The compacted base64 text; it cannot exceed what a 65,536-byte blob needs.
118    let mut body = Secret::with_capacity(BODY_MAX);
119    let mut body_lines = 0usize;
120    for_each_line(text, |line| {
121        saw_text |= !line.is_empty();
122        match at {
123            At::BeforeBegin => {
124                if line.is_empty() {
125                    return Ok(());
126                }
127                if line != BEGIN {
128                    return Err(ArmorError::MissingBegin);
129                }
130                saw_marker = true;
131                at = At::Body;
132            }
133            At::Body => {
134                if line == END {
135                    at = At::AfterEnd;
136                } else if line == BEGIN {
137                    return Err(ArmorError::SecondBlock);
138                } else if line.is_empty() || !line.iter().copied().all(base64_byte) {
139                    return Err(ArmorError::BadBodyLine);
140                } else if !body.push_slice(line) {
141                    // More base64 than a maximal blob needs.
142                    return Err(ArmorError::BadLength);
143                } else {
144                    body_lines += 1;
145                }
146            }
147            At::AfterEnd => {
148                if line == BEGIN {
149                    return Err(ArmorError::SecondBlock);
150                }
151                if !line.is_empty() {
152                    return Err(ArmorError::TextAfterBlock);
153                }
154            }
155        }
156        Ok(())
157    })?;
158    match at {
159        At::AfterEnd => {}
160        At::Body => return Err(ArmorError::MissingEnd),
161        At::BeforeBegin if saw_text || saw_marker => return Err(ArmorError::MissingBegin),
162        At::BeforeBegin => return Err(ArmorError::Empty),
163    }
164    if body_lines == 0 {
165        return Err(ArmorError::EmptyBody);
166    }
167    let encoded = body.as_slice();
168    // Canonical standard base64: the length is a multiple of four and `=` pads only the
169    // last quantum. `base64ct` refuses a bad length, misplaced or excess padding, bytes
170    // outside the alphabet and non-zero trailing bits. A 65,536 and a 65,537 byte blob
171    // share one base64 length, so the destination is sized by the upper bound and the
172    // real length is checked after the decode.
173    if !encoded.len().is_multiple_of(4) {
174        return Err(ArmorError::BadBase64);
175    }
176    let mut blob = Secret::zeroed(encoded.len() / 4 * 3);
177    let written = Base64::decode(encoded, blob.as_mut_slice())
178        .map_err(|_| ArmorError::BadBase64)?
179        .len();
180    blob.truncate(written);
181    if blob.is_empty() || blob.len() > BLOB_MAX {
182        return Err(ArmorError::BadLength);
183    }
184    Ok(blob)
185}

187/// Armor `blob` the way `ssh-keygen` writes it (70 columns, LF). Used by the tests to
188/// build synthetic fixtures; never by the cell.
189#[cfg(test)]
190pub fn encode(blob: &[u8], width: usize, newline: &str) -> std::string::String {
191    let mut out = std::string::String::new();
192    out.push_str(core::str::from_utf8(BEGIN).unwrap());
193    out.push_str(newline);
194    let mut text = std::vec![0u8; blob.len().div_ceil(3) * 4];
195    let encoded = Base64::encode(blob, &mut text).unwrap();
196    for chunk in encoded.as_bytes().chunks(width) {
197        out.push_str(core::str::from_utf8(chunk).unwrap());
198        out.push_str(newline);
199    }
200    out.push_str(core::str::from_utf8(END).unwrap());
201    out.push_str(newline);
202    out
203}
