1//! `/bin/kms-ssh-admin` (KSSH K7): the admin surface for the single KMS-custodied SSH
2//! identity. Dedicated cell, launched only by the shell's reviewed path edge, holding the
3//! kernel-attested `CALLER_FLAG_KMS_SSH_ADMIN` and never the sign right.
4//!
5//! ```text
6//! kms-ssh-admin status [NAME]   leaf status, or one key's facts           (op 19)
7//! kms-ssh-admin import NAME     armored key on the echo-off lease         (ops 15, 16*, 17; 18)
8//! kms-ssh-admin delete NAME     typed confirmation, then                  (ops 19, 20)
9//! ```
10//!
11//! Exit codes: 0 ok, 1 refused by the key service (or unreachable), 2 usage or key text that
12//! is not one armored OpenSSH private key, 124 key service timeout, 125 no terminal or
13//! session lost, 130 cancelled. DEV_POSTURE: anyone at the foreground session is the admin.

69#[derive(Debug, PartialEq, Eq)]
70pub enum Command<'a> {
71    Status(Option<&'a str>),
72    Import(&'a str),
73    Delete(&'a str),
74}
75
76#[derive(Debug, PartialEq, Eq)]
77pub enum UsageError {
78    Grammar,
79    Name,
80}
81
82impl UsageError {
83    pub const fn message(&self) -> &'static str {
84        match self {
85            Self::Grammar => report::USAGE,
86            Self::Name => {
87                "kms-ssh-admin: invalid key name (1 to 64 characters from A-Z a-z 0-9 _ -)\n"
88            }
89        }
90    }
91}
92
93/// The whole grammar. The name is checked here with the same validator KMS uses, so a name
94/// KMS would refuse never leaves the cell. Nothing here reads key material.
95pub fn parse<'a>(args: &[&'a str]) -> Result<Command<'a>, UsageError> {
96    let name = |text: &'a str| {
97        if validate_ssh_key_name(text.as_bytes()) {
98            Ok(text)
99        } else {
100            Err(UsageError::Name)
101        }
102    };
103    match args {
104        ["status"] => Ok(Command::Status(None)),
105        ["status", text] => Ok(Command::Status(Some(name(text)?))),
106        ["import", text] => Ok(Command::Import(name(text)?)),
107        ["delete", text] => Ok(Command::Delete(name(text)?)),
108        _ => Err(UsageError::Grammar),
109    }
110}
111
112/// One run: parse, connect to the key service, execute. `connect` is only called after a
113/// valid command line, so a usage error never depends on the key service.
114pub fn run<K: Kms>(
115    args: &[&str],
116    console: &mut dyn Console,
117    connect: impl FnOnce() -> Result<K, KmsClientError>,
118) -> i32 {
119    let command = match parse(args) {
120        Ok(command) => command,
121        Err(error) => {
122            console.err(error.message());
123            return EXIT_USAGE;
124        }
125    };
126    let kms = match connect() {
127        Ok(kms) => kms,
128        Err(error) => {
129            let (code, message) = report::kms_failure(error);
130            console.err(&message);
131            return code;
132        }
133    };
134    match command {
135        Command::Status(name) => status::run(name, console, &kms),
136        Command::Import(name) => import::run(name, console, &kms),
137        Command::Delete(name) => delete::run(name, console, &kms),
138    }
139}

144#[cfg(not(test))]
145fn cell_main() {
146    let code = match ostd::child_context::ChildContext::read_own() {
147        Ok(context) => entry(&context),
148        Err(_) => report::EXIT_NO_TERMINAL,
149    };
150    ostd::syscall::sys_exit(code as u32 as usize)
151}
152
153/// The foreground-child entry. Without a terminal session (a pipeline stage has no control
154/// grant; a bare spawn has no context session) there is nobody to read the key from or to
155/// confirm a delete with: refuse with 125.
156#[cfg(not(test))]
157fn entry(context: &ostd::child_context::ChildContext) -> i32 {
158    use alloc::string::String;
159    use alloc::vec::Vec;
160    use ostd::child_context::GrantSlot;
161    use terminal_sdk::session::TerminalSession;
162
163    let has_control = context
164        .grants()
165        .any(|grant| grant.slot == GrantSlot::Control);
166    let session = if has_control {
167        TerminalSession::from_own_context(context).ok()
168    } else {
169        None
170    };
171    let Some(session) = session else {
172        console::refuse_without_terminal(context);
173        return report::EXIT_NO_TERMINAL;
174    };
175    let args: Vec<String> = context.argv().map(String::from).collect();
176    let words: Vec<&str> = args.iter().map(String::as_str).collect();
177    let mut tty = console::Tty::new(session);
178    let code = run(&words, &mut tty, ostd::clients::kms::KmsClient::connect);
179    if tty.lost() && code == report::EXIT_OK {
180        report::EXIT_NO_TERMINAL
181    } else {
182        code
183    }
