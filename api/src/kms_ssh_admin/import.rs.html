1//! `kms-ssh-admin import NAME`: read the armored key on the echo-off lease, decode it
2//! strictly, and drive Begin / Chunk x n / Commit (ops 15-17, Abort 18 on failure).
3//!
4//! The key is imported as is: a wrapped key's passphrase never enters this cell and its
5//! decryptability is not checked here (K2 O-15). The cell prints the blob length and a
6//! SHA-256 prefix so the operator can compare it with the host's decoded blob.
7
8use ostd::clients::kms::{ssh_chunk_count, KmsClientError, SshKeyStatusPayload, KMS_SSH_CHUNK_LEN};
9
10use crate::armor::{self, TEXT_MAX};
11use crate::console::{Block, Console};
12use crate::kms::Kms;
13use crate::report::{self, EXIT_CANCELLED, EXIT_NO_TERMINAL, EXIT_OK, EXIT_USAGE};
14use crate::secret::Secret;

16/// What a completed import tells the operator.
17pub struct Imported {
18    pub status: SshKeyStatusPayload,
19    pub blob_len: usize,
20    pub digest: [u8; 32],
21}
22
23/// Drive one import of `blob` (1..=65,536 bytes) as `name`. Aborts the pending import
24/// (best effort) on any failure after Begin; a timeout aborts nothing, because the client
25/// is poisoned and could not send it.
26pub fn drive(kms: &dyn Kms, name: &[u8], blob: &[u8]) -> Result<Imported, KmsClientError> {
27    let total = u32::try_from(blob.len()).map_err(|_| KmsClientError::InvalidPayload)?;
28    if total == 0 || blob.len() > armor::BLOB_MAX {
29        return Err(KmsClientError::InvalidPayload);
30    }
31    let digest = ostd::session::sha256(blob);
32    let begin = kms.import_begin(name, total, &digest)?;
33    let committed = stage_and_commit(kms, &begin, total, blob, &digest);
34    if let Err(error) = &committed {
35        if *error != KmsClientError::Timeout && begin.handle != 0 {
36            let _ = kms.import_abort(begin.handle);
37        }
38    }
39    Ok(Imported {
40        status: committed?,
41        blob_len: blob.len(),
42        digest,
43    })
44}
45
46fn stage_and_commit(
47    kms: &dyn Kms,
48    begin: &ostd::clients::kms::SshTransferBeginResponsePayload,
49    total: u32,
50    blob: &[u8],
51    digest: &[u8; 32],
52) -> Result<SshKeyStatusPayload, KmsClientError> {
53    // Begin's reply: a nonzero handle, our length, the canonical count and capacity.
54    let count = ssh_chunk_count(total);
55    if begin.handle == 0
56        || begin.total_len != total
57        || begin.chunk_count != count
58        || usize::from(begin.chunk_capacity) != KMS_SSH_CHUNK_LEN
59    {
60        return Err(KmsClientError::MismatchedResponse);
61    }
62    let mut offset = 0usize;
63    for index in 0..count {
64        let take = (blob.len() - offset).min(KMS_SSH_CHUNK_LEN);
65        let ack = kms.import_chunk(begin.handle, index, &blob[offset..offset + take])?;
66        offset += take;
67        if ack.handle != begin.handle
68            || ack.next_chunk_index != index + 1
69            || ack.received_len as usize != offset
70        {
71            return Err(KmsClientError::MismatchedResponse);
72        }
73    }
74    if offset != blob.len() {
75        return Err(KmsClientError::MismatchedResponse);
76    }
77    let status = kms.import_commit(begin.handle, count, digest)?;
78    if status.key_present != 1 {
79        return Err(KmsClientError::MismatchedResponse);
80    }
81    Ok(status)
82}

84pub fn run(name: &str, console: &mut dyn Console, kms: &dyn Kms) -> i32 {
85    console.err(report::IMPORT_PROMPT);
86    let mut text = Secret::with_capacity(TEXT_MAX);
87    let read = console.read_key_block(&mut text);
88    // The echo-off line break the operator cannot see.
89    console.err("\n");
90    match read {
91        Block::Done => {}
92        Block::Cancelled => {
93            text.wipe();
94            console.err(report::IMPORT_CANCELLED);
95            return EXIT_CANCELLED;
96        }
97        Block::TooLong => {
98            text.wipe();
99            console.err(report::TEXT_TOO_LONG);
100            return EXIT_USAGE;
101        }
102        Block::Lost => {
103            text.wipe();
104            return EXIT_NO_TERMINAL;
105        }
106    }
107    let decoded = armor::decode(text.as_slice());
108    // The key text is not needed past the decode.
109    drop(text);
110    let blob = match decoded {
111        Ok(blob) => blob,
112        Err(error) => {
113            console.err(&report::line(error.message()));
114            return EXIT_USAGE;
115        }
116    };
117    match drive(kms, name.as_bytes(), blob.as_slice()) {
118        Ok(done) => {
119            console.out(&report::imported(
120                name,
121                &done.status,
122                done.blob_len,
123                &done.digest,
124            ));
125            if console.lost() {
126                EXIT_NO_TERMINAL
127            } else {
128                EXIT_OK
129            }
130        }
131        Err(error) => {
132            let (code, message) = report::kms_failure(error);
133            console.err(&message);
134            code
135        }
136    }
137}
