1//! Crash recovery and fail-closed ledger initialization.
2
3use super::fence::EnforcementFence;
4use super::ledger::{ApplicationLedger, ResourceVector};
5
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub enum RecoveryStatus {
8    Recovered,
9    UninitializedStorage,
10    CorruptedMedia,
11    FenceRegression,
12}
13
14pub fn recover_ledger(
15    has_valid_genesis: bool,
16    is_corrupt: bool,
17    stored_fence: u64,
18    boot_floor: u64,
19    ledger_id: [u8; 32],
20    genesis_id: [u8; 32],
21    max_vector: ResourceVector,
22    ceiling_vector: ResourceVector,
23) -> Result<(ApplicationLedger, EnforcementFence), RecoveryStatus> {
24    if !has_valid_genesis {
25        return Err(RecoveryStatus::UninitializedStorage);
26    }
27    if is_corrupt {
28        return Err(RecoveryStatus::CorruptedMedia);
29    }
30    if stored_fence < boot_floor {
31        return Err(RecoveryStatus::FenceRegression);
32    }
33    let ledger = ApplicationLedger::new(ledger_id, genesis_id, max_vector, ceiling_vector, stored_fence)
34        .map_err(|_| RecoveryStatus::CorruptedMedia)?;
35    let fence = EnforcementFence::new(stored_fence);
36    Ok((ledger, fence))
37}
