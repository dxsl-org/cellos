1use crate::PropertyBuffer;
2use ostd::mmio::MmioRegion;
3use types::{ViError, ViResult};
4
5const MBOX_READ: usize = 0x00;
6const MBOX_STATUS: usize = 0x18;
7const MBOX_WRITE: usize = 0x20;
8
9const MBOX_FULL: u32 = 0x8000_0000;
10const MBOX_EMPTY: u32 = 0x4000_0000;
11
12const CHANNEL_PROPERTY: u32 = 8;
13const VC_UNCACHED_ALIAS: u32 = 0xC000_0000;
14
15pub struct BcmMailbox {
16    region: MmioRegion,
17}
18
19impl BcmMailbox {
20    pub fn new(region: MmioRegion) -> Self {
21        Self { region }
22    }
23
24    pub fn open() -> ViResult<Self> {
25        let base = hal_soc_bcm27xx::BCM2837.mmio.mailbox_base;
26        let size = hal_soc_bcm27xx::BCM2837.mmio.mailbox_grant_size;
27        let region = ostd::mmio::request_region(base, size)?;
28        Ok(Self::new(region))
29    }
30
31    pub fn call<const N: usize>(&self, buffer: &mut PropertyBuffer<N>) -> ViResult<()> {
32        let ptr = buffer.data.as_mut_ptr() as usize;
33        let bus_addr = (ptr as u32) | VC_UNCACHED_ALIAS;
34
35        // 1. Wait until mailbox is not full
36        let mut timeout = 1_000_000;
37        while (self.region.read::<u32>(MBOX_STATUS)? & MBOX_FULL) != 0 {
38            timeout -= 1;
39            if timeout == 0 {
40                return Err(ViError::IO);
41            }
42        }
43
44        // 2. Write buffer bus address with channel
45        let write_val = (bus_addr & !0xF) | (CHANNEL_PROPERTY & 0xF);
46        self.region.write::<u32>(MBOX_WRITE, write_val)?;
47
48        // 3. Wait for response on matching channel
49        timeout = 1_000_000;
50        loop {
51            while (self.region.read::<u32>(MBOX_STATUS)? & MBOX_EMPTY) != 0 {
52                timeout -= 1;
53                if timeout == 0 {
54                    return Err(ViError::IO);
55                }
56            }
57
58            let resp = self.region.read::<u32>(MBOX_READ)?;
59            if (resp & 0xF) == (CHANNEL_PROPERTY & 0xF) {
60                if (resp & !0xF) == (bus_addr & !0xF) {
61                    break;
62                }
63            }
64        }
65
66        // 4. Verify request/response success code (0x8000_0000)
67        if buffer.data[1] == 0x8000_0000 {
68            Ok(())
69        } else {
70            Err(ViError::InvalidInput)
71        }
72    }
73}
