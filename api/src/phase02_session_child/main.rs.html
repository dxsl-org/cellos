2//! through the real owner syscalls and printed as a `P02C` marker.

55fn now() -> u64 {
56    ostd::session::monotonic_ns().unwrap_or(0)
57}

59fn sleep_ms(ms: u64) {
60    let end = now() + ms * 1_000_000;
61    while now() < end {
62        let _ = sys_set_timer(1);
63    }
64}

77/// The committed endpoint a parent delegated into `slot`.
78fn granted(context: &ChildContext, slot: GrantSlot) -> Option<Endpoint> {
79    let operation = OperationRef::from_raw(context.operation()).ok()?;
80    let grant = context.grants().find(|g| g.slot == slot && g.kind == 1)?;
81    Some(Endpoint::new(EndpointRef::from_raw(grant.object).ok()?, operation))
82}

88fn cell_main() {
89    let context = match ChildContext::read_own() {
90        Ok(context) => context,
91        Err(status) => {
92            mark(&format!("FAIL read_own status={status:?}"));
93            sys_exit(2);
94        }
95    };
96    let argv = argv_of(&context);
97    let mode = argv.first().map(String::as_str).unwrap_or("");
98    match mode {
99        "ctx" => ctx(&context, &argv),
100        "raw-panic" => raw_panic(&context),
101        "hang" => hang(&context),
102        "timeout" => timeout(&context),
103        "writer" => writer(&context, &argv),
104        "descend" => descend(&context, &argv),
105        "blocked-writer" => blocked_writer(&context),
106        "blocked-reader" => blocked_reader(&context),
107        "forged" => forged(&argv),
108        "orphan" => orphan(&context),
109        "panic" => panic!("P02C deliberate fault"),
110        other => {
111            mark(&format!("FAIL unknown mode {other}"));
112            sys_exit(2);
113        }
114    }
115    sys_exit(0);
116}

118/// Report exactly what the kernel handed this child at its first instruction,
119/// on both owner-enforced output streams, then consume typed input to EOF.
120fn ctx(context: &ChildContext, argv: &[String]) {
121    let cwd = core::str::from_utf8(context.cwd()).unwrap_or("<non-utf8>");
122    let mut env = String::new();
123    for (key, value) in context.environment() {
124        env.push_str(&format!("{key}={value};"));
125    }
126    let summary = format!(
127        "ctx|argc={}|argv={:?}|cwd={}|cwdgrant={}|env={}",
128        argv.len(),
129        argv,
130        cwd,
131        context.cwd_grant().is_some(),
132        env
133    );
134    mark(&summary);
135    let mut session = match TerminalSession::from_own_context(context) {
136        Ok(session) => session,
137        Err(status) => {
138            mark(&format!("FAIL from_own_context status={status:?}"));
139            sys_exit(2);
140        }
141    };
142    let _ = session.write_stdout(summary.as_bytes(), 0);
143    let _ = session.write_stderr(b"diag|ctx", 0);
144    let deadline = now() + 8_000_000_000;
145    loop {
146        match session.next_event(deadline) {
147            Ok(event) => match event.kind {
148                EventKind::Text(text) => {
149                    mark(&format!("ctx text epoch={} seq={}", event.epoch, event.sequence));
150                    let _ = session.write_stdout(format!("got|{text}").as_bytes(), 0);
151                }
152                EventKind::Eof => {
153                    mark("ctx eof");
154                    let _ = session.write_stdout(b"eof|seen", 0);
155                    return;
156                }
157                _ => {}
158            },
159            Err(status) => {
160                mark(&format!("ctx event status={status:?}"));
161                return;
162            }
163        }
164    }
165}

167/// Fault while a scoped raw mode lease is held: restoration must not need us.
168fn raw_panic(context: &ChildContext) {
169    let mut session = match TerminalSession::from_own_context(context) {
170        Ok(session) => session,
171        Err(status) => {
172            mark(&format!("FAIL raw from_own_context status={status:?}"));
173            return;
174        }
175    };
176    let guard = session.set_mode(1, 1);
177    mark(&format!("raw lease acquired={}", guard.is_ok()));
178    let guard = guard.ok();
179    sleep_ms(200);
180    core::mem::forget(guard);
181    panic!("P02C fault while raw");
182}

184/// A foreground child whose bounded typed read times out and then exits.
185fn timeout(context: &ChildContext) {
186    let mut session = match TerminalSession::from_own_context(context) {
187        Ok(session) => session,
188        Err(status) => {
189            mark(&format!("FAIL timeout from_own_context status={status:?}"));
190            return;
191        }
192    };
193    let outcome = session.next_event(now() + 700_000_000).err();
194    mark(&format!("timeout status={outcome:?}"));
195}

197/// Block in a typed read until the owner cancels this operation.
198fn hang(context: &ChildContext) {
199    let mut session = match TerminalSession::from_own_context(context) {
200        Ok(session) => session,
201        Err(status) => {
202            mark(&format!("FAIL hang from_own_context status={status:?}"));
203            return;
204        }
205    };
206    mark("hang ready");
207    let status = session.next_event(u64::MAX).err();
208    mark(&format!("hang woke status={status:?}"));
209}

211fn writer(context: &ChildContext, argv: &[String]) {
212    let Some(out) = granted(context, GrantSlot::Stdout) else {
213        mark("FAIL writer no stdout grant");
214        return;
215    };
216    let delay: u64 = argv.get(1).and_then(|v| v.parse().ok()).unwrap_or(0);
217    let tag = argv.get(2).map(String::as_str).unwrap_or("w");
218    sleep_ms(delay);
219    match out.write(tag.as_bytes(), 0) {
220        Ok(n) => mark(&format!("writer {tag} wrote={n}")),
221        Err(status) => mark(&format!("writer {tag} status={status:?}")),
222    }
223}

225/// Create a child-owned descendant that retains a delegated writer after this
226/// child has exited.
227fn descend(context: &ChildContext, argv: &[String]) {
228    let Some(out) = granted(context, GrantSlot::Stdout) else {
229        mark("FAIL descend no stdout grant");
230        return;
231    };
232    let delay = argv.get(1).map(String::as_str).unwrap_or("300");
233    let parent = match OperationRef::from_raw(context.operation()) {
234        Ok(parent) => parent,
235        Err(status) => {
236            mark(&format!("descend bad operation status={status:?}"));
237            return;
238        }
239    };
240    match spawn_writer(out, delay, "gc", parent) {
241        Ok(()) => {
242            mark("descend grandchild committed");
243            // Drop this child's own writer so only the descendant retains one,
244            // then stay alive: a creator's death drains the operations it owns.
245            let _ = out.close();
246            sleep_ms(2500);
247        }
248        Err(status) => mark(&format!("descend spawn status={status:?}")),
249    }
250}

252fn spawn_writer(out: Endpoint, delay: &str, tag: &str, parent: OperationRef) -> Result<(), Status> {
253    let operation = Operation::create(Some(parent))?;
254    let argv = ["writer", delay, tag];
255    // Length is exact because the dummy grant has the same encoded size.
256    let dummy = Grant::endpoint(
257        GrantSlot::Stdout,
258        ObjectRefV1 { id: 1, generation: 1 },
259        30,
260    )?;
261    let probe = ChildContext::new(operation.reference().raw(), &argv, b"", &[], &[dummy])?;
262    let mut prepared = operation.prepare_child(CHILD_PATH, probe.as_bytes().len())?;
263    let delegated = out.delegate(&prepared, EndpointRights::WRITE_DELEGATE)?;
264    let grant = Grant::endpoint(GrantSlot::Stdout, delegated.reference().raw(), 30)?;
265    let context = ChildContext::new(operation.reference().raw(), &argv, b"", &[], &[grant])?;
266    prepared.write_context(&context)?;
267    prepared.commit()?;
268    Ok(())
269}

271/// Fill the ring, then block in a write with no deadline until the consumer
272/// closes or the operation is cancelled.
273fn blocked_writer(context: &ChildContext) {
274    let Some(out) = granted(context, GrantSlot::Stdout) else {
275        mark("FAIL blocked-writer no stdout grant");
276        return;
277    };
278    let block = [0x5au8; 512];
279    let mut total = 0usize;
280    mark("blocked-writer ready");
281    loop {
282        match out.write(&block, u64::MAX) {
283            Ok(n) => total += n,
284            Err(status) => {
285                mark(&format!("blocked-writer woke status={status:?} total={total}"));
286                return;
287            }
288        }
289    }
290}

305/// Use references the parent holds privately; every use must be refused.
306fn forged(argv: &[String]) {
307    let operation = argv.get(1).and_then(|_| parse_ref(&argv[1], &argv[2]));
308    let endpoint = argv.get(3).and_then(|_| parse_ref(&argv[3], &argv[4]));
309    let (Some(operation), Some(endpoint)) = (operation, endpoint) else {
310        mark("FAIL forged malformed argv");
311        return;
312    };
313    let (Ok(operation), Ok(endpoint)) = (OperationRef::from_raw(operation), EndpointRef::from_raw(endpoint)) else {
314        mark("FAIL forged refs");
315        return;
316    };
317    let foreign = Endpoint::new(endpoint, operation);
318    let mut buffer = [0u8; 8];
319    let read = foreign.read(&mut buffer, 0);
320    let write = foreign.write(b"x", 0);
321    let duplicate = foreign.duplicate(EndpointRights::WRITE_ONLY);
322    let close = foreign.close();
323    let cancel = Operation::from_ref(operation).cancel(1);
324    mark(&format!(
325        "forged read_denied={} write_denied={} dup_denied={} close_denied={} cancel_denied={}",
326        read.is_err(),
327        write.is_err(),
328        duplicate.is_err(),
329        close.is_err(),
330        cancel.is_err()
331    ));
332    mark(&format!("forged detail read={read:?} dup={:?}", duplicate.map(|_| ())));
333}

335/// Observe the death of the parent that held the only writer.
336fn orphan(context: &ChildContext) {
337    let Some(input) = granted(context, GrantSlot::Stdin) else {
338        mark("FAIL orphan no stdin grant");
339        return;
340    };
341    let mut buffer = [0u8; 16];
342    mark("orphan ready");
343    // Heartbeat until the owner reports a terminal condition. If the parent's
344    // death retires this child outright, the heartbeat simply stops; the host
345    // runner treats that cessation as the observation.
346    for tick in 0..400u32 {
347        match input.read(&mut buffer, now() + 100_000_000) {
348            Ok(n) => mark(&format!("orphan bytes={n}")),
349            Err(Status::TimedOut) => mark(&format!("orphan tick {tick}")),
350            Err(status) => {
351                mark(&format!("orphan woke after parent exit status={status:?}"));
352                return;
353            }
354        }
355    }
