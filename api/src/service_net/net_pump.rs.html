1//! The net service's turn work, run before every receive.

3use ostd::io::println;
4use smoltcp::wire::{EthernetAddress, HardwareAddress, IpCidr};
5
6use super::net_state::NetState;
7use super::{
8    apply_link_local, dotted_ipv4, drain_ready_waits, graceful_deadline_line, now_instant,
9    poll_resolves, reap_aborted, reap_graceful_closes, sys_get_time, MTIME_TICKS_PER_MS,
10    SMOLTCP_MAINTENANCE_TICKS,
11};
12use crate::dhcp::{poll_dhcp, DhcpState};
13use crate::link_local::LinkLocal;

15impl NetState<'_> {
16    /// The turn work before every receive: late MAC binding, NIC drain, DHCP,
17    /// link-local, one stack poll, TLS and DNS progress, close reapers and
18    /// ready waits.
19    pub(super) fn pump(&mut self) {
20        if !self.mac_synchronized {
21            if let Some(mac) = self.device.query_driver_mac() {
22                println(&alloc::format!(
23                    "[net] hardware MAC bound from NIC driver: {:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
24                    mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]
25                ));
26                self.iface
27                    .set_hardware_addr(HardwareAddress::Ethernet(EthernetAddress(mac)));
28                self.device.set_host_mac(mac);
29                self.mac_synchronized = true;
30                if self.dhcp_state == DhcpState::Pending {
31                    self.sockets
32                        .get_mut::<smoltcp::socket::dhcpv4::Socket>(self.dhcp_handle)
33                        .reset();
34                }
35            }
36        }
37        self.probe.turn(&self.device);
38        let drained = self.device.pump_rx_split();
39        self.probe.rx_drained(drained);
40        if self.dhcp_state == DhcpState::Pending {
41            self.poll_dhcp();
42        }
43        self.poll_link_local();
44
45        let now = sys_get_time();
46        // Poll on every turn, not only on the 100 ms maintenance cadence:
47        // smoltcp advances a handshake one step per poll, and a listener
48        // stuck in SynReceived cannot report ACCEPT. Waiting for the
49        // maintenance tick added up to 100 ms per handshake step.
50        self.iface
51            .poll(now_instant(), &mut self.device, &mut self.sockets);
52        if now.wrapping_sub(self.last_poll_ticks) >= SMOLTCP_MAINTENANCE_TICKS {
53            self.last_poll_ticks = now;
54        }
55        #[cfg(feature = "qemu-dev-https")]
56        self.dev_clock
57            .poll(&mut self.iface, &mut self.sockets, self.local_ip != [0; 4]);
58        self.tls_pending.poll(
59            &mut self.iface,
60            &mut self.device,
61            &mut self.sockets,
62            &mut self.table,
63            &mut self.tls_table,
64        );
65        poll_resolves(&self.resolver, &mut self.pending_dns, &mut self.sockets);
66        let graceful_aborted =
67            reap_graceful_closes(&mut self.table, &mut self.sockets, sys_get_time());
68        if let Some(line) = graceful_deadline_line(graceful_aborted) {
69            println(&line);
70        }
71        reap_aborted(&mut self.table, &mut self.sockets, sys_get_time());
72        drain_ready_waits(&mut self.pending_ready, &mut self.sockets, &self.table);
73    }
74
75    fn poll_dhcp(&mut self) {
76        let lease = poll_dhcp(
77            self.dhcp_handle,
78            &mut self.iface,
79            &mut self.sockets,
80            &mut self.device,
81            now_instant(),
82        );
83        self.dhcp_state = lease.state;
84        if self.dhcp_state == DhcpState::Acquired {
85            // The lease has just overwritten the IPv4 slot, claimed
86            // link-local address included.
87            self.link_local = None;
88            if let Some(IpCidr::Ipv4(cidr)) = self
89                .iface
90                .ip_addrs()
91                .iter()
92                .find(|address| matches!(address, IpCidr::Ipv4(_)))
93            {
94                self.local_ip.copy_from_slice(cidr.address().as_bytes());
95                println(&alloc::format!(
96                    "[net] IP address: {}",
97                    dotted_ipv4(self.local_ip)
98                ));
99            }
100        } else if self.link_local.is_none()
101            && sys_get_time().wrapping_sub(self.dhcp_start_ticks) >= 4_000 * MTIME_TICKS_PER_MS
102        {
103            let HardwareAddress::Ethernet(addr) = self.iface.hardware_addr();
104            self.link_local = Some(LinkLocal::new(addr.0, sys_get_time() / MTIME_TICKS_PER_MS));
105        }
106        // The lease names the resolver to use from here on (option 6);
107        // without it the DNS socket keeps SLIRP's built-in default.
108        if let Some(server) = lease.dns_server {
109            self.resolver.set_server(&mut self.sockets, server);
110            println(&alloc::format!("[net] DNS server: {}", dotted_ipv4(server)));
111        }
112    }
113
114    /// The ARP tap is drained every turn so it never holds stale packets; they
115    /// feed the claim while one runs.
116    fn poll_link_local(&mut self) {
117        let now_ms = sys_get_time() / MTIME_TICKS_PER_MS;
118        while let Some(packet) = self.device.take_arp() {
119            if let Some(engine) = self.link_local.as_mut() {
120                engine.on_arp(packet, now_ms);
121            }
122        }
123        if let Some(engine) = self.link_local.as_mut() {
124            engine.poll(now_ms);
125            while let Some(event) = engine.next_event() {
126                apply_link_local(event, &mut self.iface, &mut self.device, &mut self.local_ip);
127            }
128        }
129    }
