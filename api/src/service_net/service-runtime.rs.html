42/// One `[net-loop]` line per this many raw timer ticks (`loop-trace` images).
43///
44/// It used to be ten times shorter (`200 * MTIME_TICKS_PER_MS`), which made the
45/// trace print about once per loop turn — the noise an operator notices first,
46/// and enough UART time to change the thing it was measuring. The tick rate is
47/// per-arch, and `MTIME_TICKS_PER_MS` is only a 10 MHz assumption, so this is a
48/// nominal 2 s: measured on QEMU raspi3b it lands at about one line per four
49/// turns (8962 → 2245 lines over the same 900 s window, same ~8960 turns).
50const LOOP_TRACE_INTERVAL_TICKS: u64 = 2_000 * MTIME_TICKS_PER_MS;
51// A 10-tick deadline can fire just after 9 complete tick periods when the
52// submission lands immediately before a scheduler tick. The exclusive ceiling
53// therefore has to be the preceding 9 periods, not the 100 ms maintenance budget.
54#[cfg(any(
55    test,
56    all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge"))
57))]
58const IDLE_IPC_WAKE_PROOF_CEILING_TICKS: u64 =
59    (NET_RX_MAINTENANCE_WAIT_SCHEDULER_TICKS - 1) * SCHEDULER_TICK_MS * MTIME_TICKS_PER_MS;

71/// Classify the ticks a recordless wait burned *before it returned*.
72///
73/// `wait_return_ticks` is the wait's own latency, not the latency of the IPC
74/// that followed it: only the return can be below the exclusive ceiling (a
75/// deadline return lands at whole tick periods, at or above it), and the
76/// waiter's path back to `TryRecv` afterwards is not part of the wake.
77#[cfg(any(
78    test,
79    all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge"))
80))]
81pub(crate) const fn classify_idle_ipc_wake(wait_return_ticks: u64) -> IdleIpcWakeClassification {
82    if wait_return_ticks < IDLE_IPC_WAKE_PROOF_CEILING_TICKS {
83        IdleIpcWakeClassification::Pass
84    } else {
85        IdleIpcWakeClassification::Inconclusive
86    }
87}

104/// Dotted-quad text for log lines (`10.0.2.15`).
105pub(crate) fn dotted_ipv4(ip: [u8; 4]) -> String {
106    let mut text = String::new();
107    for (index, octet) in ip.iter().enumerate() {
108        if index > 0 {
109            text.push('.');
110        }
111        let mut number = *octet as u32;
112        let mut digits = [0u8; 3];
113        let mut digit_index = 3;
114        loop {
115            digit_index -= 1;
116            digits[digit_index] = b'0' + (number % 10) as u8;
117            number /= 10;
118            if number == 0 {
119                break;
120            }
121        }
122        for digit in &digits[digit_index..] {
123            text.push(*digit as char);
124        }
125    }
126    text
127}

149fn handle_resolve(
150    buffer: &[u8],
151    sender: usize,
152    owner: SocketOwner,
153    iface: &mut Interface,
154    sockets: &mut SocketSet<'_>,
155    resolver: &Resolver,
156    pending: &mut heapless::Vec<PendingDns, DNS_QUERY_LIMIT>,
157) -> bool {
158    let Ok(NetRequest::Resolve { hostname }) = ipc::decode::<NetRequest<'_>>(buffer) else {
159        return false;
160    };
161    if let Some(ip) = crate::dns::static_lookup(hostname, resolver.server()) {
162        send_ready(sender, NetResponse::Addr(ip));
163        return true;
164    }
165    if pending.is_full() {
166        send_ready(sender, NetResponse::Err(0xFD));
167        return true;
168    }
169    let Some(query) = resolver.start(hostname, iface, sockets) else {
170        send_ready(sender, NetResponse::Err(0xFF));
171        return true;
172    };
173    // Legacy callers wait at Recv; async callers use an exact reply token.
174    let _ = pending.push(PendingDns {
175        owner, sender, op_id: ostd::ipc::current(), query: Some(query),
176        answer: None, deadline: sys_get_time().saturating_add(DNS_QUERY_DEADLINE_TICKS),
177    });
178    true
179}

181fn poll_resolves(
182    resolver: &Resolver,
183    pending: &mut heapless::Vec<PendingDns, DNS_QUERY_LIMIT>,
184    sockets: &mut SocketSet<'_>,
185) {
186    let now = sys_get_time();
187    let mut index = 0;
188    while index < pending.len() {
189        let state = &mut pending[index];
190        if state.answer.is_none() {
191            if let Some(query) = state.query {
192                if let Some(result) = resolver.poll_result(query, sockets) {
193                    state.query = None;
194                    state.answer = Some(result);
195                } else if now >= state.deadline {
196                    resolver.cancel(query, sockets);
197                    state.query = None;
198                    state.answer = Some(None);
199                }
200            }
201        }
202        let Some(answer) = state.answer else {
203            index += 1;
204            continue;
205        };
206        let mut out = [0u8; IPC_BUF_SIZE];
207        let response = match answer {
208            Some(ip) => NetResponse::Addr(ip),
209            None => NetResponse::Err(0xFF),
210        };
211        let sent = ipc::encode(&response, &mut out).is_ok_and(|bytes| match state.op_id {
212            Some(op_id) => {
213                let _ = ostd::ipc::reply(op_id, bytes);
214                true // A cancelled op cannot be delivered to another caller.
215            }
216            None => matches!(ostd::syscall::sys_try_send(state.sender, bytes), SyscallResult::Ok(0)),
217        });
218        if sent || now >= state.deadline.saturating_add(10_000_000) {
219            pending.swap_remove(index);
220        } else {
221            index += 1;
222        }
223    }
224}

227const WAITSET_DEADLINE_TICKS: u64 = 50_000_000;

229struct PendingReady {
230    op_id: usize,
231    interests: heapless::Vec<u8, {MAX_INTERESTS * ipc::NET_READY_RECORD_BYTES}>,
232    cursor: u16,
233    deadline: u64,
234}

236fn send_ready(sender: usize, response: NetResponse<'_>) {
237    let mut buf = [0u8; IPC_BUF_SIZE];
238    if let Ok(bytes) = ipc::encode(&response, &mut buf) {
239        let _ = ostd::syscall::sys_send(sender, bytes);
240    }
241}

243/// One deferred IPC token represents up to 256 socket interests. Register only
244/// after authorization; recheck before parking and again on every net turn.
245fn handle_ready_wait(
246    buffer: &[u8],
247    sender: usize,
248    owner: SocketOwner,
249    sockets: &mut SocketSet<'_>,
250    table: &SocketTable,
251    pending: &mut BTreeMap<SocketOwner, PendingReady>,
252) -> bool {
253    let Ok(NetRequest::TcpReady { interests, cursor, wait: true }) =
254        ipc::decode::<NetRequest<'_>>(buffer)
255    else {
256        return false;
257    };
258    let result = readiness(interests, cursor, owner, sockets, table);
259    match result {
260        Err(()) => send_ready(sender, NetResponse::Err(0xFF)),
261        Ok((events, next_cursor)) if !events.is_empty() => {
262            let mut packed = [0u8; handlers::tcp::READY_PAGE * ipc::NET_READY_RECORD_BYTES];
263            send_ready(sender, NetResponse::TcpReady {
264                events: handlers::tcp::pack_events(&events, &mut packed), next_cursor,
265            });
266        }
267        Ok(_) => {
268            let Some(op_id) = ostd::ipc::current() else {
269                send_ready(sender, NetResponse::Err(0xFD));
270                return true;
271            };
272            if let Some(previous) = pending.remove(&owner) {
273                let mut out = [0u8; IPC_BUF_SIZE];
274                if let Ok(bytes) = ipc::encode(&NetResponse::Err(0xFD), &mut out) {
275                    let _ = ostd::ipc::reply(previous.op_id, bytes);
276                }
277            }
278            if pending.len() >= WAITSET_LIMIT {
279                send_ready(sender, NetResponse::Err(0xFD));
280                return true;
281            }
282            let mut owned = heapless::Vec::new();
283            for &byte in interests {
284                if owned.push(byte).is_err() {
285                    send_ready(sender, NetResponse::Err(0xFF));
286                    return true;
287                }
288            }
289            pending.insert(owner, PendingReady {
290                op_id,
291                interests: owned,
292                cursor,
293                deadline: sys_get_time().saturating_add(WAITSET_DEADLINE_TICKS),
294            });
295        }
296    }
297    true
298}

300/// FIN queues output before the close. Closed completes normally; the finite
301/// deadline is an explicit abort of a peer that never acknowledges or closes.
302fn reap_graceful_closes(table: &mut SocketTable, sockets: &mut SocketSet<'_>) {
303    let now = sys_get_time();
304    for _ in 0..8 {
305        let next = table.graceful_closes().find_map(|(cap, deadline)| {
306            let done = table
307                .get_unchecked(cap)
308                .is_none_or(|h| sockets.get_mut::<smoltcp::socket::tcp::Socket>(h).state()
309                    == smoltcp::socket::tcp::State::Closed);
310            (done || now >= deadline).then_some(cap)
311        });
312        let Some(cap) = next else { break };
313        if let Some(h) = table.remove_internal(cap) {
314            sockets.remove(h);
315        }
316    }
317}

319fn drain_ready_waits(
320    pending: &mut BTreeMap<SocketOwner, PendingReady>,
321    sockets: &mut SocketSet<'_>,
322    table: &SocketTable,
323) {
324    let now = sys_get_time();
325    // Bound the owner scan: registration admits at most WAITSET_LIMIT records.
326    let mut completed = heapless::Vec::<SocketOwner, WAITSET_LIMIT>::new();
327    for (&owner, wait) in pending.iter() {
328        let result = readiness(&wait.interests, wait.cursor, owner, sockets, table);
329        let response = match result {
330            Err(()) => Some(NetResponse::Err(0xFF)),
331            Ok((events, next_cursor)) if !events.is_empty() => {
332                let mut packed = [0u8; handlers::tcp::READY_PAGE * ipc::NET_READY_RECORD_BYTES];
333                let mut out = [0u8; IPC_BUF_SIZE];
334                let response = NetResponse::TcpReady {
335                    events: handlers::tcp::pack_events(&events, &mut packed), next_cursor,
336                };
337                if let Ok(bytes) = ipc::encode(&response, &mut out) {
338                    let _ = ostd::ipc::reply(wait.op_id, bytes);
339                }
340                let _ = completed.push(owner);
341                continue;
342            }
343            Ok(_) if now >= wait.deadline => Some(NetResponse::NotReady),
344            Ok(_) => None,
345        };
346        if let Some(response) = response {
347            let mut out = [0u8; IPC_BUF_SIZE];
348            if let Ok(bytes) = ipc::encode(&response, &mut out) {
349                let _ = ostd::ipc::reply(wait.op_id, bytes);
350            }
351            let _ = completed.push(owner);
352        }
353    }
354    for owner in completed {
355        pending.remove(&owner);
356    }
357}
358fn release_owner(
359    owner: SocketOwner,
360    sockets: &mut SocketSet<'_>,
361    table: &mut SocketTable,
362    tls_table: &mut BTreeMap<u64, TlsSocketEntry>,
363    tls_pending: &mut crate::tls_handler::TlsPending,
364    pending_ready: &mut BTreeMap<SocketOwner, PendingReady>,
365    pending_dns: &mut heapless::Vec<PendingDns, DNS_QUERY_LIMIT>,
366    resolver: &Resolver,
367    watched: &mut BTreeMap<SocketOwner, (usize, u64)>,
368) {
369    pending_ready.remove(&owner);
370    let mut index = 0;
371    while index < pending_dns.len() {
372        if pending_dns[index].owner == owner {
373            if let Some(query) = pending_dns[index].query {
374                resolver.cancel(query, sockets);
375            }
376            pending_dns.swap_remove(index);
377        } else {
378            index += 1;
379        }
380    }
381    tls_pending.cancel_owner(owner, sockets, table);
382    while let Some(cap) = table.owned_cap(owner) {
383        tls_table.remove(&cap);
384        if let Some(handle) = table.remove_internal(cap) {
385            sockets.remove(handle);
386        }
387    }
388    if let Some((_, token)) = watched.remove(&owner) {
389        ostd::syscall::sys_cancel_cell_owner_watch(token);
390    }
391}

394#[cfg(target_os = "none")]
395pub(crate) fn run() {
396    println("[net] Network Service v0.1: smoltcp + NIC Driver Cell + DHCP");
397
398    let mut device = VirtioNetDevice::new();
399    let config = Config::new(HardwareAddress::Ethernet(MAC));
400    let mut iface = Interface::new(config, &mut device, now_instant());
401    iface.update_ip_addrs(|addresses| {
402        let _ = addresses.push(IpCidr::new(IpAddress::v4(0, 0, 0, 0), 0));
403    });
404
405    // Allocate slots incrementally: the 290-cap profile cannot hold the
406    // complete socket table on its small cell stack.
407    let mut socket_storage = alloc::vec::Vec::with_capacity(SOCKET_SET_STORAGE);
408    socket_storage.resize_with(SOCKET_SET_STORAGE, || SocketStorage::EMPTY);
409    let mut sockets = SocketSet::new(&mut socket_storage[..]);
410    let mut table = SocketTable::new();
411    let mut tls_table: BTreeMap<u64, TlsSocketEntry> = BTreeMap::new();
412    let mut tls_pending = crate::tls_handler::TlsPending::default();
413    let mut pending_ready: BTreeMap<SocketOwner, PendingReady> = BTreeMap::new();
414    let mut pending_dns = heapless::Vec::<PendingDns, DNS_QUERY_LIMIT>::new();
415    // Root TID and death-watch token. Worker TID exit is not root death.
416    let mut watched: BTreeMap<SocketOwner, (usize, u64)> = BTreeMap::new();
417    // One DNS socket serves every `NetRequest::Resolve`; the server starts as
418    // SLIRP's and is replaced by whatever the DHCP lease names.
419    let mut resolver = Resolver::install(&mut sockets);
420
421    #[cfg(not(feature = "hypervisor-bridge"))]
422    let dhcp_handle = add_dhcp_socket(&mut sockets);
423    #[cfg(not(feature = "hypervisor-bridge"))]
424    let mut dhcp_state = DhcpState::Pending;
425
426    let mut buffer = [0u8; IPC_BUF_SIZE];
427    #[cfg(not(feature = "hypervisor-bridge"))]
428    let mut last_poll_ticks = sys_get_time();
429    let mut local_ip = [0u8; 4];
430    #[cfg(not(feature = "hypervisor-bridge"))]
431    let mut net_rx_producer_proved = false;
432    #[cfg(not(feature = "hypervisor-bridge"))]
433    let mut pending_net_rx_proof = false;
434    #[cfg(not(feature = "hypervisor-bridge"))]
435    let mut ipc_burst_grace = 0;
436    #[cfg(all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge")))]
437    let mut idle_ipc_wake_oracle = crate::idle_ipc_wake_oracle::IdleIpcWakeOracle::new();
438
439    #[cfg(not(feature = "hypervisor-bridge"))]
440    println("[net] Starting DHCP...");
441
442    #[cfg(feature = "loop-trace")]
443    let mut trace_turns: u64 = 0;
444    #[cfg(feature = "loop-trace")]
445    let mut trace_recv: u64 = 0;
446    #[cfg(feature = "loop-trace")]
447    let mut trace_last = sys_get_time();
448
449    loop {
450        ostd::syscall::sys_heartbeat(500);
451        // `loop-trace` image: one line every ~2 s with what the turn is spending
452        // its time on. A stopped heartbeat is the signal that the service is
453        // parked inside a driver command rather than looping.
454        #[cfg(feature = "loop-trace")]
455        {
456            trace_turns += 1;
457            let now = sys_get_time();
458            if now.wrapping_sub(trace_last) >= LOOP_TRACE_INTERVAL_TICKS {
459                trace_last = now;
460                println(&alloc::format!(
461                    "[net-loop] turns={} recv={} l2send={} l2recv={} drv_cmd={} drv_to={} rx_q={} guest_q={}",
462                    trace_turns,
463                    trace_recv,
464                    crate::handlers::L2_SEND_REQUESTS.load(Ordering::Relaxed),
465                    crate::handlers::L2_RECV_REQUESTS.load(Ordering::Relaxed),
466                    crate::interface::DRV_COMMANDS.load(Ordering::Relaxed),
467                    crate::interface::DRV_TIMEOUTS.load(Ordering::Relaxed),
468                    device.rx_queue_len(),
469                    device.guest_rx_queue_len(),
470                ));
471            }
472        }
473        #[cfg(not(feature = "hypervisor-bridge"))]
474        {
475            let drained = device.pump_rx_split();
476            if pending_net_rx_proof {
477                if !net_rx_producer_proved && drained > 0 {
478                    println("[net-rx-producer] irq->completion PASS");
479                    net_rx_producer_proved = true;
480                }
481                pending_net_rx_proof = false;
482            }
483
484            if dhcp_state == DhcpState::Pending {
485                let lease = poll_dhcp(
486                    dhcp_handle,
487                    &mut iface,
488                    &mut sockets,
489                    &mut device,
490                    now_instant(),
491                );
492                dhcp_state = lease.state;
493                if dhcp_state == DhcpState::Acquired {
494                    if let Some(IpCidr::Ipv4(cidr)) = iface
495                        .ip_addrs()
496                        .iter()
497                        .find(|address| matches!(address, IpCidr::Ipv4(_)))
498                    {
499                        local_ip.copy_from_slice(cidr.address().as_bytes());
500                        println(&alloc::format!(
501                            "[net] IP address: {}",
502                            crate::service_runtime::dotted_ipv4(local_ip)
503                        ));
504                    }
505                }
506                // The lease names the resolver to use from here on (option 6);
507                // without it the DNS socket keeps SLIRP's built-in default.
508                if let Some(server) = lease.dns_server {
509                    resolver.set_server(&mut sockets, server);
510                    println(&alloc::format!(
511                        "[net] DNS server: {}",
512                        crate::service_runtime::dotted_ipv4(server)
513                    ));
514                }
515            }
516
517            let now = sys_get_time();
518            // Poll on every turn, not only on the 100 ms maintenance cadence:
519            // smoltcp advances a handshake one step per poll, and a listener
520            // stuck in SynReceived cannot report ACCEPT. Waiting for the
521            // maintenance tick added up to 100 ms per handshake step.
522            iface.poll(now_instant(), &mut device, &mut sockets);
523            if now.wrapping_sub(last_poll_ticks) >= SMOLTCP_MAINTENANCE_TICKS {
524                last_poll_ticks = now;
525            }
526        }
527        tls_pending.poll(&mut iface, &mut device, &mut sockets, &mut table, &mut tls_table);
528        poll_resolves(&resolver, &mut pending_dns, &mut sockets);
529        reap_graceful_closes(&mut table, &mut sockets);
530        drain_ready_waits(&mut pending_ready, &mut sockets, &table);
531
532        buffer.fill(0);
533        let receive_result = sys_try_recv_attested(0, &mut buffer);
534        match receive_result {
535            SyscallResult::Ok(sender) if sender > 0 => {
536                #[cfg(feature = "loop-trace")]
537                {
538                    trace_recv += 1;
539                }
540                let Some(identity) = api::caller_identity::CallerIdentity::from_recv_buf(&buffer)
541                else {
542                    if let Some(owner) = watched.iter().find_map(|(&owner, &(root_tid, _))|
543                        (root_tid == sender).then_some(owner))
544                    {
545                        release_owner(owner, &mut sockets, &mut table, &mut tls_table,
546                            &mut tls_pending, &mut pending_ready, &mut pending_dns, &resolver,
547                            &mut watched);
548                    }
549                    #[cfg(all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge")))]
550                    idle_ipc_wake_oracle.record_ipc_miss();
551                    continue;
552                };
553                if identity.cell_id == 0 || identity.generation == 0 {
554                    #[cfg(all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge")))]
555                    idle_ipc_wake_oracle.record_ipc_miss();
556                    continue;
557                }
558                let owner = SocketOwner {
559                    cell_id: identity.cell_id,
560                    generation: identity.generation,
561                };
562                // Owner watch is a cleanup optimisation: without it a dead
563                // client's sockets are only reclaimed when the net cell
564                // restarts, so a refusal must never deny live traffic.
565                if !watched.contains_key(&owner) {
566                    let watch = ostd::syscall::sys_watch_cell_owner(owner.cell_id, owner.generation);
567                    if let Some((principal, token)) = watch {
568                        while let Some(stale) = watched.keys().copied().find(|candidate|
569                            candidate.cell_id == owner.cell_id && *candidate != owner)
570                        {
571                            release_owner(stale, &mut sockets, &mut table, &mut tls_table,
572                                &mut tls_pending, &mut pending_ready, &mut pending_dns, &resolver,
573                                &mut watched);
574                        }
575                        watched.insert(owner, (principal.root_tid as usize, token));
576                    } else if !WATCH_REFUSED.swap(true, core::sync::atomic::Ordering::Relaxed) {
577                        println("[net] owner death watch unavailable; socket reclaim deferred");
578                    }
579                }
580                #[cfg(all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge")))]
581                idle_ipc_wake_oracle.record_ipc_drain(sys_get_time(), sender);
582                // A ready wait is registered only after a network progress
583                // pass. The retained interest batch is rechecked before idle
584                // sleep and on every NET_RX / maintenance wake.
585                iface.poll(now_instant(), &mut device, &mut sockets);
586                if !handle_ready_wait(&buffer, sender, owner, &mut sockets, &table, &mut pending_ready)
587                    && !handle_resolve(&buffer, sender, owner, &mut iface, &mut sockets,
588                        &resolver, &mut pending_dns)
589                {
590                    handlers::handle_request(
591                        &buffer, sender, owner, &mut iface, &mut device, &mut sockets,
592                        &mut table, &mut tls_table, &mut tls_pending, &resolver, &local_ip,
593                    );
594                }
595                drain_ready_waits(&mut pending_ready, &mut sockets, &table);
596                #[cfg(not(feature = "hypervisor-bridge"))]
597                {
598                    ipc_burst_grace = IPC_BURST_GRACE_YIELDS;
599                }
600            }
601            _ => {
602                #[cfg(all(feature = "ipc-wake-oracle", not(feature = "hypervisor-bridge")))]
603                idle_ipc_wake_oracle.record_ipc_miss();
604                #[cfg(feature = "hypervisor-bridge")]
605                {
606                    let _ = sys_wait_completion(NET_RX, 1);
607                }
608                #[cfg(not(feature = "hypervisor-bridge"))]
609                if consume_ipc_burst_grace(&mut ipc_burst_grace) {
610                    // Let the caller run after a reply before parking on NET_RX;
611                    // sequential IPC bursts then avoid one timer quantum per call.
612                    sys_yield();
613                } else {
614                    #[cfg(feature = "ipc-wake-oracle")]
615                    let wait_started_ticks = sys_get_time();
616                    #[cfg(feature = "ipc-wake-oracle")]
617                    let completion = match sys_wait_completion_detailed(
618                        NET_RX,
619                        NET_RX_MAINTENANCE_WAIT_SCHEDULER_TICKS,
620                    ) {
621                        WaitCompletionResult::NoRecord => {
622                            idle_ipc_wake_oracle.arm(
623                                wait_started_ticks,
624                                sys_get_time().wrapping_sub(wait_started_ticks),
625                                SMOLTCP_MAINTENANCE_TICKS,
626                                IDLE_IPC_WAKE_PROOF_CEILING_TICKS,
627                            );
628                            None
629                        }
630                        WaitCompletionResult::Completion(completion) => {
631                            idle_ipc_wake_oracle.clear();
632                            Some(completion)
633                        }
634                        WaitCompletionResult::ErrorOrInvalid(_) => {
635                            idle_ipc_wake_oracle.clear();
636                            None
637                        }
638                    };
639                    #[cfg(not(feature = "ipc-wake-oracle"))]
640                    let completion =
641                        sys_wait_completion(NET_RX, NET_RX_MAINTENANCE_WAIT_SCHEDULER_TICKS);
642                    if let Some(completion) = completion {
643                        pending_net_rx_proof =
644                            completion.source == NET_RX && completion.result == NET_RX as i64;
645                    }
646                }
647                // A recordless return means queued IPC interrupted the wait early or
648                // the finite maintenance budget elapsed. Either way, retry the loop
649                // without claiming NET_RX producer proof.
650            }
651        }
652    }
653}
