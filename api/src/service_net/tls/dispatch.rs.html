25const TCP_TIMEOUT: u64 = 150_000_000; // 15 seconds (sys_get_time: 10 MHz)
26const TLS_TIMEOUT: u64 = 300_000_000; // 30 seconds for one TLS operation
27const REPLY_TIMEOUT: u64 = 300_000_000; // legacy receiver may not yet be scheduled

29type Handshake = Pin<Box<dyn Future<Output = Result<TlsSocketEntry, embedded_tls::TlsError>>>>;
30type Sending = Pin<Box<dyn Future<Output = (TlsSocketEntry, Result<usize, embedded_tls::TlsError>)>>>;
31type Receiving = Pin<Box<dyn Future<Output = (TlsSocketEntry, Vec<u8>, Result<usize, embedded_tls::TlsError>)>>>;

33enum State {

52impl ReplyTo {
53    pub fn capture(sender: usize) -> Self {
54        match ostd::ipc::current() {
55            Some(token) => Self::Async(token),
56            None => Self::Legacy(sender),
57        }

89fn failed(request: &Request) -> &'static [u8] {
90    match request {
91        Request::Send(_) => &[0, 0, 0, 0],
92        Request::Recv(_) => &[0, 0],
93    }
94}

96impl TlsPending {
97    pub fn is_empty(&self) -> bool {
98        self.operations.is_empty() && self.replies.is_empty()
99    }
100
101    pub fn contains_cap(&self, cap: u64) -> bool {
102        self.operations.contains_key(&cap) || self.replies.iter().any(|reply| reply.cap == Some(cap))
103    }

105    pub fn connect(&mut self, cap: u64, handle: SocketHandle, owner: SocketOwner,
106        reply: ReplyTo, hostname: String) {
107        self.operations.insert(cap, Pending {
108            owner, reply, handle,
109            deadline: sys_get_time().saturating_add(TCP_TIMEOUT),
110            state: State::Connecting(hostname), queue: VecDeque::new(),
111        });
112    }
113
114    pub fn enqueue(&mut self, cap: u64, owner: SocketOwner, reply: ReplyTo,
115        request: Request, tls_table: &mut BTreeMap<u64, TlsSocketEntry>,
116        table: &SocketTable) {
117        if let Some(op) = self.operations.get_mut(&cap) {
118            if op.owner == owner {
119                op.queue.push_back(Queued { reply, request });
120            } else {
121                self.answer(reply, owner, None, failed(&request));
122            }
123        } else if let Some(handle) = table.get(cap, owner) {
124            if let Some(entry) = tls_table.remove(&cap) {
125                let state = start(entry, request);
126                self.operations.insert(cap, Pending {
127                    owner, reply, handle,
128                    deadline: sys_get_time().saturating_add(TLS_TIMEOUT),
129                    state, queue: VecDeque::new(),
130                });
131            } else {
132                self.answer(reply, owner, None, failed(&request));
133            }
134        } else {
135            self.answer(reply, owner, None, failed(&request));
136        }
137    }
138
139    /// Async callers have kernel operation tokens. Legacy callers use their
140    /// original sender TID; a nonblocking send is retried until their masked
141    /// receive is ready, never holding the net dispatcher hostage.
142    fn answer(&mut self, reply: ReplyTo, owner: SocketOwner, cap: Option<u64>, bytes: &[u8]) -> bool {
143        match reply {
144            ReplyTo::Async(token) => ostd::ipc::reply(token, bytes).is_ok(),
145            ReplyTo::Legacy(sender) => {
146                self.replies.push_back(LegacyReply {
147                    sender, owner, cap, bytes: bytes.to_vec(),
148                    deadline: sys_get_time().saturating_add(REPLY_TIMEOUT),
149                });
150                true
151            }
152        }
153    }
154
155    fn flush_replies(&mut self, sockets: &mut SocketSet<'_>, table: &mut SocketTable,
156        tls_table: &mut BTreeMap<u64, TlsSocketEntry>) {
157        let count = self.replies.len();
158        for _ in 0..count {
159            let Some(reply) = self.replies.pop_front() else { break; };
160            if matches!(sys_try_send(reply.sender, &reply.bytes), SyscallResult::Ok(0)) {
161                continue;
162            }
163            if sys_get_time() < reply.deadline {
164                self.replies.push_back(reply);
165                continue;
166            }
167            if let Some(cap) = reply.cap {
168                self.cancel_cap(cap);
169                tls_table.remove(&cap);
170                if let Some(handle) = table.remove_internal(cap) {
171                    sockets.remove(handle);
172                }
173            }
174        }
175    }

177    /// Cancel the pending future (which owns keys) BEFORE removing its TCP socket.
178    /// Call before SocketTable::remove_owner and the TLS table's owner cleanup.
179    pub fn cancel_owner(&mut self, owner: SocketOwner, sockets: &mut SocketSet<'_>,
180        table: &mut SocketTable) {
181        let mut cursor = None;
182        while let Some(cap) = self.next_key(cursor) {
183            cursor = Some(cap);
184            if self.operations.get(&cap).is_some_and(|op| op.owner == owner) {
185                let op = self.operations.remove(&cap).unwrap();
186                drop(op);
187                if let Some(handle) = table.remove_internal(cap) {
188                    sockets.remove(handle);
189                }
190            }
191        }
192        self.replies.retain(|reply| reply.owner != owner);
193    }
194
195    /// A close from the owner cancels any in-flight and queued calls on this cap.
196    /// The caller still performs normal SocketTable and TLS table removal.
197    pub fn cancel_cap(&mut self, cap: u64) {
198        self.replies.retain(|reply| reply.cap != Some(cap));
199        if let Some(op) = self.operations.remove(&cap) {
200            let Pending { reply, owner, state, queue, .. } = op;
201            let response: &[u8] = match &state {
202                State::Connecting(_) | State::Handshake(_) => &[0; 8],
203                State::Send(_) => &[0; 4],
204                State::Recv(_) => &[0; 2],
205                State::Vacant => &[],
206            };
207            drop(state);
208            self.answer(reply, owner, None, response);
209            for queued in queue {
210                self.answer(queued.reply, owner, None, failed(&queued.request));
211            }
212        }
213    }

215    fn next_key(&self, after: Option<u64>) -> Option<u64> {
216        match after {
217            None => self.operations.first_key_value().map(|(&cap, _)| cap),
218            Some(cap) => self.operations.range((Excluded(cap), Unbounded)).next().map(|(&cap, _)| cap),
219        }
220    }

222    /// Poll every pending operation once. The caller must continue regular NIC
223    /// pumping and smoltcp polling even when no request arrives.
224    pub fn poll(&mut self, iface: &mut Interface, device: &mut VirtioNetDevice,
225        sockets: &mut SocketSet<'_>, table: &mut SocketTable,
226        tls_table: &mut BTreeMap<u64, TlsSocketEntry>) {
227        let mut cursor = None;
228        while let Some(cap) = self.next_key(cursor) {
229            cursor = Some(cap);
230            let mut op = self.operations.remove(&cap).unwrap();
231            if table.get(cap, op.owner) != Some(op.handle) || sys_get_time() >= op.deadline {
232                self.fail(cap, op, sockets, table);
233                continue;
234            }
235            let result = match &mut op.state {
236                State::Connecting(_) => match sockets.get_mut::<tcp::Socket>(op.handle).state() {
237                    tcp::State::Established => Step::TcpConnected,
238                    tcp::State::Closed | tcp::State::CloseWait => Step::Failed,
239                    _ => Step::Waiting,
240                },
241                State::Handshake(future) => {
242                    match poll_future(future, iface, device, sockets) {
243                        Poll::Ready(Ok(entry)) => Step::Connected(entry),
244                        Poll::Ready(Err(_)) => Step::Failed,
245                        Poll::Pending => Step::Waiting,
246                    }
247                }
248                State::Send(future) => match poll_future(future, iface, device, sockets) {
249                    Poll::Ready((entry, result)) => Step::Sent(entry, result),
250                    Poll::Pending => Step::Waiting,
251                },
252                State::Recv(future) => match poll_future(future, iface, device, sockets) {
253                    Poll::Ready((entry, data, result)) => Step::Received(entry, data, result),
254                    Poll::Pending => Step::Waiting,
255                },
256                State::Vacant => unreachable!(),
257            };
258            match result {
259                Step::Waiting => { self.operations.insert(cap, op); }
260                Step::TcpConnected => {
261                    table.set_state(cap, SocketState::Connected);
262                    let State::Connecting(hostname) = core::mem::replace(&mut op.state, State::Vacant) else {
263                        unreachable!();
264                    };
265                    op.state = State::Handshake(Box::pin(TlsSocketEntry::handshake(op.handle, hostname)));
266                    op.deadline = sys_get_time().saturating_add(TLS_TIMEOUT);
267                    self.operations.insert(cap, op);
268                }
269                Step::Failed => self.fail(cap, op, sockets, table),
270                Step::Connected(entry) => {
271                    // Drop the finished future before moving the established
272                    // connection into the live table.
273                    op.state = State::Vacant;
274                    if self.answer(op.reply, op.owner, Some(cap), &cap.to_le_bytes()) {
275                        tls_table.insert(cap, entry);
276                        self.start_queued(cap, op, tls_table);
277                    } else {
278                        drop(entry);
279                        for queued in op.queue {
280                            self.answer(queued.reply, op.owner, None, failed(&queued.request));
281                        }
282                        if let Some(handle) = table.remove_internal(cap) { sockets.remove(handle); }
283                    }
284                }
285                Step::Sent(entry, result) => {
286                    op.state = State::Vacant;
287                    let count = result.unwrap_or(0) as u32;
288                    if self.answer(op.reply, op.owner, Some(cap), &count.to_le_bytes()) {
289                        tls_table.insert(cap, entry);
290                        self.start_queued(cap, op, tls_table);
291                    } else {
292                        drop(entry);
293                        for queued in op.queue { self.answer(queued.reply, op.owner, None, failed(&queued.request)); }
294                        if let Some(handle) = table.remove_internal(cap) { sockets.remove(handle); }
295                    }
296                }
297                Step::Received(entry, data, result) => {
298                    op.state = State::Vacant;
299                    let response = match result {
300                        Ok(len) => encode_tls_recv_reply(&data[..len]),
301                        Err(_) => alloc::vec![0; 2],
302                    };
303                    if self.answer(op.reply, op.owner, Some(cap), &response) {
304                        tls_table.insert(cap, entry);
305                        self.start_queued(cap, op, tls_table);
306                    } else {
307                        drop(entry);
308                        for queued in op.queue { self.answer(queued.reply, op.owner, None, failed(&queued.request)); }
309                        if let Some(handle) = table.remove_internal(cap) { sockets.remove(handle); }
310                    }
311                }
312            }
313        }
314        self.flush_replies(sockets, table, tls_table);
315    }

317    fn start_queued(&mut self, cap: u64, mut op: Pending,
318        tls_table: &mut BTreeMap<u64, TlsSocketEntry>) {
319        if let Some(next) = op.queue.pop_front() {
320            let entry = tls_table.remove(&cap).expect("completed TLS entry");
321            op.reply = next.reply;
322            op.deadline = sys_get_time().saturating_add(TLS_TIMEOUT);
323            op.state = start(entry, next.request);
324            self.operations.insert(cap, op);
325        }
326    }
327
328    fn fail(&mut self, cap: u64, op: Pending, sockets: &mut SocketSet<'_>, table: &mut SocketTable) {
329        let Pending { reply, owner, state, queue, .. } = op;
330        let response: &[u8] = match &state {
331            State::Connecting(_) | State::Handshake(_) => &[0; 8],
332            State::Send(_) => &[0; 4],
333            State::Recv(_) => &[0; 2],
334            State::Vacant => &[],
335        };
336        // The suspended future owns TLS keys and record buffers: destroy it
337        // before the smoltcp socket can be removed.
338        drop(state);
339        self.answer(reply, owner, None, response);
340        for queued in queue { self.answer(queued.reply, owner, None, failed(&queued.request)); }
341        if let Some(handle) = table.remove_internal(cap) { sockets.remove(handle); }
342    }
343}
344
345enum Step {
346    Waiting,
347    TcpConnected,
348    Failed,
349    Connected(TlsSocketEntry),
350    Sent(TlsSocketEntry, Result<usize, embedded_tls::TlsError>),
351    Received(TlsSocketEntry, Vec<u8>, Result<usize, embedded_tls::TlsError>),
352}
353
354fn start(mut entry: TlsSocketEntry, request: Request) -> State {
355    match request {
356        Request::Send(data) => State::Send(Box::pin(async move {
357            let result = entry.send(&data).await;
358            (entry, result)
359        })),
360        Request::Recv(size) => State::Recv(Box::pin(async move {
361            let mut data = alloc::vec![0u8; size];
362            let result = entry.recv(&mut data).await;
363            (entry, data, result)
364        })),
365    }
366}

368fn poll_future<T>(future: &mut Pin<Box<dyn Future<Output = T>>>, iface: &mut Interface,
369    device: &mut VirtioNetDevice, sockets: &mut SocketSet<'_>) -> Poll<T> {
370    // The transport leaves Pending on absent data/space; the outer service
371    // dispatcher, not a self-waking busy loop, is responsible for the next poll.
372    let mut context = Context::from_waker(Waker::noop());
373    unsafe { set_tls_context(iface, device, sockets as *mut SocketSet<'_> as *mut ()) };
374    let result = future.as_mut().poll(&mut context);
375    clear_tls_context();
376    result
377}
