1//! Strict DER framing and identity binding for the mounted client chain.
2//!
3//! Bounds are frozen: at most [`RELAY_CHAIN_MAX_CERTS`] certificates and
4//! [`RELAY_CHAIN_MAX_LEN`] bytes. The leaf must carry the clientAuth EKU
5//! and must not carry the serverAuth EKU; the served SPKI from KMS
6//! (opcode 14) must hash to the deployment NodeId.
7
8use sha2::{Digest, Sha256};
9use types::kms::{RELAY_CHAIN_MAX_CERTS, RELAY_CHAIN_MAX_LEN};
10
11/// id-ce-extKeyUsage extension OID content (2.5.29.37).
12const EXTENDED_KEY_USAGE_OID: [u8; 3] = [0x55, 0x1D, 0x25];
13/// clientAuth extended-key-usage OID content (1.3.6.1.5.5.7.3.2).
14const CLIENT_AUTH_EKU: [u8; 8] = [0x2B, 0x06, 0x01, 0x05, 0x05, 0x07, 0x03, 0x02];
15/// serverAuth extended-key-usage OID content (1.3.6.1.5.5.7.3.1).
16const SERVER_AUTH_EKU: [u8; 8] = [0x2B, 0x06, 0x01, 0x05, 0x05, 0x07, 0x03, 0x01];
17
18#[derive(Debug, PartialEq, Eq)]
19pub enum ChainError {
20    /// Too many certificates for the frozen bound.
21    TooManyCertificates,
22    /// Chain exceeds the frozen byte budget.
23    TooLong,
24    /// A certificate is not well-framed DER SEQUENCE data.
25    MalformedDer,
26    /// The same certificate appears twice in the chain.
27    DuplicateCertificate,
28    /// Leaf lacks clientAuth or carries serverAuth.
29    LeafUsageNotAllowed,
30    /// Leaf SPKI or NodeId facts disagree.
31    IdentityMismatch,
32}

34/// Read one DER TLV header; returns `(tag, content_len, header_len)`.
35fn read_tlv(bytes: &[u8]) -> Option<(u8, usize, usize)> {
36    let (&tag, rest) = bytes.split_first()?;
37    let &first = rest.first()?;
38    if first < 0x80 {
39        return Some((tag, first as usize, 2));
40    }
41    let count = (first & 0x7F) as usize;
42    if count == 0 || count > 4 || rest.len() < 1 + count {
43        return None;
44    }
45    if rest[1] == 0 {
46        return None;
47    }
48    let mut len = 0usize;
49    for &byte in &rest[1..1 + count] {
50        len = (len << 8) | byte as usize;
51    }
52    // Canonical DER forbids non-minimal long-form lengths.
53    (len >= 0x80).then_some((tag, len, 2 + count))
54}
55/// Consume exactly one DER TLV with `expected_tag`.
56///
57/// The returned pair is `(full_tlv, content)`. All slices borrow the bounded
58/// input chain; parsing never allocates.
59fn take_tlv<'a>(
60    bytes: &mut &'a [u8],
61    expected_tag: u8,
62) -> Result<(&'a [u8], &'a [u8]), ChainError> {
63    let input = *bytes;
64    let (tag, len, head) = read_tlv(input).ok_or(ChainError::MalformedDer)?;
65    let total = head.checked_add(len).ok_or(ChainError::MalformedDer)?;
66    if tag != expected_tag || total > input.len() {
67        return Err(ChainError::MalformedDer);
68    }
69    let (full, rest) = input.split_at(total);
70    *bytes = rest;
71    Ok((full, &full[head..]))
72}

75struct LeafProfile<'a> {
76    spki: &'a [u8],
77}

79fn validate_eku(encoded: &[u8]) -> Result<(), ChainError> {
80    let mut outer = encoded;
81    let (_, mut oids) = take_tlv(&mut outer, 0x30)?;
82    if !outer.is_empty() {
83        return Err(ChainError::MalformedDer);
84    }
85    let mut client_auth = false;
86    let mut server_auth = false;
87    while !oids.is_empty() {
88        let (_, oid) = take_tlv(&mut oids, 0x06)?;
89        client_auth |= oid == CLIENT_AUTH_EKU;
90        server_auth |= oid == SERVER_AUTH_EKU;
91    }
92    if !client_auth || server_auth {
93        return Err(ChainError::LeafUsageNotAllowed);
94    }
95    Ok(())
96}

98fn validate_extensions(encoded: &[u8]) -> Result<(), ChainError> {
99    let mut explicit = encoded;
100    let (_, mut extensions) = take_tlv(&mut explicit, 0x30)?;
101    if !explicit.is_empty() {
102        return Err(ChainError::MalformedDer);
103    }
104    let mut found_eku = false;
105    while !extensions.is_empty() {
106        let (_, mut extension) = take_tlv(&mut extensions, 0x30)?;
107        let (_, oid) = take_tlv(&mut extension, 0x06)?;
108        if extension.first() == Some(&0x01) {
109            let (_, critical) = take_tlv(&mut extension, 0x01)?;
110            if critical.len() != 1 || !matches!(critical[0], 0x00 | 0xFF) {
111                return Err(ChainError::MalformedDer);
112            }
113        }
114        let (_, value) = take_tlv(&mut extension, 0x04)?;
115        if !extension.is_empty() {
116            return Err(ChainError::MalformedDer);
117        }
118        if oid == EXTENDED_KEY_USAGE_OID {
119            if found_eku {
120                return Err(ChainError::MalformedDer);
121            }
122            validate_eku(value)?;
123            found_eku = true;
124        }
125    }
126    if !found_eku {
127        return Err(ChainError::LeafUsageNotAllowed);
128    }
129    Ok(())
130}

132/// Walk the X.509 Certificate/TBSCertificate structure and return the canonical
133/// leaf SPKI slice after enforcing the client-only EKU profile.
134fn leaf_profile(leaf: &[u8]) -> Result<LeafProfile<'_>, ChainError> {
135    let mut outer = leaf;
136    let (_, mut certificate) = take_tlv(&mut outer, 0x30)?;
137    if !outer.is_empty() {
138        return Err(ChainError::MalformedDer);
139    }
140    let (_, mut tbs) = take_tlv(&mut certificate, 0x30)?;
141    take_tlv(&mut certificate, 0x30)?;
142    take_tlv(&mut certificate, 0x03)?;
143    if !certificate.is_empty() {
144        return Err(ChainError::MalformedDer);
145    }
146
147    // Relay leaves are X.509 v3. The explicit version is INTEGER 2.
148    let (_, version) = take_tlv(&mut tbs, 0xA0)?;
149    let mut version = version;
150    let (_, version_number) = take_tlv(&mut version, 0x02)?;
151    if !version.is_empty() || version_number != [0x02] {
152        return Err(ChainError::MalformedDer);
153    }
154    take_tlv(&mut tbs, 0x02)?; // serialNumber
155    take_tlv(&mut tbs, 0x30)?; // signature
156    take_tlv(&mut tbs, 0x30)?; // issuer
157    take_tlv(&mut tbs, 0x30)?; // validity
158    take_tlv(&mut tbs, 0x30)?; // subject
159    let (spki, _) = take_tlv(&mut tbs, 0x30)?;
160
161    let mut extensions_seen = false;
162    let mut issuer_uid_seen = false;
163    let mut subject_uid_seen = false;
164    while let Some(tag) = tbs.first().copied() {
165        match tag {
166            0x81 if !issuer_uid_seen && !extensions_seen => {
167                take_tlv(&mut tbs, 0x81)?;
168                issuer_uid_seen = true;
169            }
170            0x82 if !subject_uid_seen && !extensions_seen => {
171                take_tlv(&mut tbs, 0x82)?;
172                subject_uid_seen = true;
173            }
174            0xA3 if !extensions_seen => {
175                let (_, extensions) = take_tlv(&mut tbs, 0xA3)?;
176                validate_extensions(extensions)?;
177                extensions_seen = true;
178            }
179            _ => return Err(ChainError::MalformedDer),
180        }
181    }
182    if !extensions_seen {
183        return Err(ChainError::LeafUsageNotAllowed);
184    }
185    Ok(LeafProfile { spki })
186}

188/// Frame-check a concatenated DER chain and enforce the frozen bounds.
189///
190/// Returns the per-certificate DER slices. Duplicate certificates (exact
191/// byte equality) are rejected so a stale re-issued leaf cannot shadow a
192/// rotation.
193pub fn frame_chain(
194    chain: &[u8],
195) -> Result<heapless::Vec<&[u8], RELAY_CHAIN_MAX_CERTS>, ChainError> {
196    if chain.len() > RELAY_CHAIN_MAX_LEN {
197        return Err(ChainError::TooLong);
198    }
199    let mut certs = heapless::Vec::<&[u8], RELAY_CHAIN_MAX_CERTS>::new();
200    let mut rest = chain;
201    while !rest.is_empty() {
202        if certs.len() == RELAY_CHAIN_MAX_CERTS {
203            return Err(ChainError::TooManyCertificates);
204        }
205        let (tag, len, head) = read_tlv(rest).ok_or(ChainError::MalformedDer)?;
206        let total = head.checked_add(len).ok_or(ChainError::MalformedDer)?;
207        if tag != 0x30 || total > rest.len() {
208            return Err(ChainError::MalformedDer);
209        }
210        let (cert, tail) = rest.split_at(total);
211        certs
212            .push(cert)
213            .map_err(|_| ChainError::TooManyCertificates)?;
214        rest = tail;
215    }
216    for index in 0..certs.len() {
217        for later in &certs[index + 1..] {
218            if certs[index] == *later {
219                return Err(ChainError::DuplicateCertificate);
220            }
221        }
222    }
223    Ok(certs)
224}

226/// Validate leaf usage: clientAuth must occur inside EKU, and serverAuth must
227/// not occur there. OID-shaped bytes in other certificate fields/extensions
228/// have no effect.
229pub fn validate_leaf_usage(leaf: &[u8]) -> Result<(), ChainError> {
230    leaf_profile(leaf).map(|_| ())
231}
232
233/// Full acceptance pipeline for one mounted chain.
234pub fn validate_chain(chain: &[u8]) -> Result<(), ChainError> {
235    let certs = frame_chain(chain)?;
236    match certs.split_first() {
237        Some((leaf, _)) => validate_leaf_usage(leaf),
238        None => Err(ChainError::MalformedDer),
239    }
240}

242/// Validate a mounted active chain against opcode 14 and both active NodeId
243/// metadata sources.
244///
245/// This function must not authorize opcode 13 staging: the frozen opcode-14
246/// response exposes only the active generation, not an enrollment's pending
247/// key. Initial enrollment and renewal therefore remain fail-closed until the
248/// supervisor can supply an authenticated pending-key binding without changing
249/// the provisioning ABI.
250pub fn validate_active_chain(
251    chain: &[u8],
252    active_spki_sec1: &[u8; 65],
253    active_spki_sha256: &[u8; 32],
254    kms_metadata_node_id: &[u8; 32],
255    manifest_node_id: &[u8; 32],
256) -> Result<(), ChainError> {
257    let certs = frame_chain(chain)?;
258    let leaf = certs.first().ok_or(ChainError::MalformedDer)?;
259    let profile = leaf_profile(leaf)?;
260    let (canonical_spki, canonical_len) =
261        types::kms::p256_spki_der(active_spki_sec1).ok_or(ChainError::IdentityMismatch)?;
262    if profile.spki != &canonical_spki[..canonical_len] {
263        return Err(ChainError::IdentityMismatch);
264    }
265    let leaf_digest: [u8; 32] = Sha256::digest(profile.spki).into();
266    if leaf_digest != *active_spki_sha256
267        || leaf_digest != *kms_metadata_node_id
268        || leaf_digest != *manifest_node_id
269    {
270        return Err(ChainError::IdentityMismatch);
271    }
272    Ok(())
273}

282    fn tlv(tag: u8, content: &[u8]) -> Vec<u8> {
283        let mut out = Vec::new();
284        out.push(tag);
285        if content.len() < 0x80 {
286            out.push(content.len() as u8);
287        } else {
288            assert!(content.len() <= u8::MAX as usize);
289            out.extend_from_slice(&[0x81, content.len() as u8]);
290        }
291        out.extend_from_slice(content);
292        out
293    }

295    fn append(target: &mut Vec<u8>, encoded: Vec<u8>) {
296        target.extend_from_slice(&encoded);
297    }

299    fn extension(oid: &[u8], value: &[u8]) -> Vec<u8> {
300        let mut body = Vec::new();
301        append(&mut body, tlv(0x06, oid));
302        append(&mut body, tlv(0x04, value));
303        tlv(0x30, &body)
304    }

306    fn relay_leaf(
307        spki_sec1: &[u8; 65],
308        eku_oids: Option<&[&[u8]]>,
309        unrelated_value: Option<&[u8]>,
310    ) -> Vec<u8> {
311        let (spki, spki_len) = types::kms::p256_spki_der(spki_sec1).unwrap();
312        let mut extensions = Vec::new();
313        if let Some(value) = unrelated_value {
314            append(&mut extensions, extension(&[0x2A, 0x03], value));
315        }
316        if let Some(oids) = eku_oids {
317            let mut usages = Vec::new();
318            for oid in oids {
319                append(&mut usages, tlv(0x06, oid));
320            }
321            append(
322                &mut extensions,
323                extension(&EXTENDED_KEY_USAGE_OID, &tlv(0x30, &usages)),
324            );
325        }
326
327        let mut tbs = Vec::new();
328        append(&mut tbs, tlv(0xA0, &tlv(0x02, &[0x02])));
329        append(&mut tbs, tlv(0x02, &[0x01]));
330        append(&mut tbs, tlv(0x30, &[]));
331        append(&mut tbs, tlv(0x30, &[]));
332        append(&mut tbs, tlv(0x30, &[]));
333        append(&mut tbs, tlv(0x30, &[]));
334        tbs.extend_from_slice(&spki[..spki_len]);
335        append(&mut tbs, tlv(0xA3, &tlv(0x30, &extensions)));
336
337        let mut certificate = Vec::new();
338        append(&mut certificate, tlv(0x30, &tbs));
339        append(&mut certificate, tlv(0x30, &[]));
340        append(&mut certificate, tlv(0x03, &[0x00]));
341        tlv(0x30, &certificate)
342    }

344    fn framed_cert(body_len: u8) -> heapless::Vec<u8, 512> {
345        let mut out = heapless::Vec::new();
346        out.push(0x30).unwrap();

352    #[test]
353    fn bounds_and_duplicates_are_enforced() {
354        let single = framed_cert(4);
355        assert!(frame_chain(&single).is_ok());
356        let mut duplicate: heapless::Vec<u8, 512> = single.clone();
357        duplicate.extend(single.iter().copied());
358        assert_eq!(
359            frame_chain(&duplicate),
360            Err(ChainError::DuplicateCertificate)
361        );
362        let mut four: heapless::Vec<u8, 512> = heapless::Vec::new();
363        for _ in 0..RELAY_CHAIN_MAX_CERTS + 1 {
364            four.extend(framed_cert(4).iter().copied());
365        }
366        assert_eq!(frame_chain(&four), Err(ChainError::TooManyCertificates));
367        assert_eq!(frame_chain(&[0x30, 0x82]), Err(ChainError::MalformedDer));
368        let big = [0u8; RELAY_CHAIN_MAX_LEN + 1];
369        assert_eq!(frame_chain(&big), Err(ChainError::TooLong));
370    }
