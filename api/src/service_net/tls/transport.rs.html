1//! Nonblocking smoltcp transport used by the async TLS connection.
2//!
3//! Only one net service turn runs at a time. The dispatcher installs the three
4//! borrowed smoltcp objects immediately before polling a TLS future and clears
5//! them again before returning to IPC dispatch. A transport operation that cannot
6//! proceed returns Pending; the service loop, not a transport spin loop, polls it
7//! again on the next turn.
8
9use crate::interface::VirtioNetDevice;
10use core::future::poll_fn;
11use core::sync::atomic::{AtomicPtr, Ordering};
12use core::task::Poll;
13use embedded_io::{ErrorKind, ErrorType};
14use embedded_io_async::{Read, Write};
15use smoltcp::{
16    iface::{Interface, SocketHandle, SocketSet},
17    socket::tcp,
18};
19
20static TLS_IFACE: AtomicPtr<()> = AtomicPtr::new(core::ptr::null_mut());
21static TLS_DEVICE: AtomicPtr<()> = AtomicPtr::new(core::ptr::null_mut());
22static TLS_SOCKETS: AtomicPtr<()> = AtomicPtr::new(core::ptr::null_mut());
23
24/// Install borrowed context for the duration of ONE synchronous Future::poll.
25///
26/// # Safety
27/// All pointers must remain valid, exclusive, and point at the same socket set
28/// until `clear_tls_context` is called. No future may be polled after its socket
29/// has been removed.
30pub unsafe fn set_tls_context(iface: *mut Interface, device: *mut VirtioNetDevice, sockets: *mut ()) {
31    TLS_IFACE.store(iface.cast(), Ordering::Relaxed);
32    TLS_DEVICE.store(device.cast(), Ordering::Relaxed);
33    TLS_SOCKETS.store(sockets, Ordering::Relaxed);
34}
35
36/// Remove borrowed pointers before the dispatcher may mutate the socket set.
37pub fn clear_tls_context() {
38    TLS_IFACE.store(core::ptr::null_mut(), Ordering::Relaxed);
39    TLS_DEVICE.store(core::ptr::null_mut(), Ordering::Relaxed);
40    TLS_SOCKETS.store(core::ptr::null_mut(), Ordering::Relaxed);
41}
42
43pub struct SmoltcpTlsTransport {
44    pub handle: SocketHandle,
45}
46
47impl SmoltcpTlsTransport {
48    pub fn new(handle: SocketHandle) -> Self {
49        Self { handle }
50    }
51
52    unsafe fn sockets() -> &'static mut SocketSet<'static> {
53        &mut *(TLS_SOCKETS.load(Ordering::Relaxed) as *mut SocketSet<'static>)
54    }
55
56    unsafe fn poll_network() {
57        let iface = &mut *(TLS_IFACE.load(Ordering::Relaxed) as *mut Interface);
58        let device = &mut *(TLS_DEVICE.load(Ordering::Relaxed) as *mut VirtioNetDevice);
59        let sockets = Self::sockets();
60        iface.poll(crate::service_runtime::now_instant(), device, sockets);
61    }
62}
63
64impl ErrorType for SmoltcpTlsTransport {
65    type Error = ErrorKind;
66}
67
68impl Read for SmoltcpTlsTransport {
69    async fn read(&mut self, buf: &mut [u8]) -> Result<usize, ErrorKind> {
70        if buf.is_empty() {
71            return Ok(0);
72        }
73        poll_fn(|_| {
74            // SAFETY: the dispatcher installs a valid context for this poll.
75            let socket = unsafe { Self::sockets().get_mut::<tcp::Socket>(self.handle) };
76            if socket.can_recv() {
77                return Poll::Ready(socket.recv_slice(buf).map_err(|_| ErrorKind::Other));
78            }
79            if !socket.may_recv() {
80                return Poll::Ready(Err(ErrorKind::ConnectionReset));
81            }
82            Poll::Pending
83        })
84        .await
85    }
86}
87
88impl Write for SmoltcpTlsTransport {
89    async fn write(&mut self, buf: &[u8]) -> Result<usize, ErrorKind> {
90        if buf.is_empty() {
91            return Ok(0);
92        }
93        poll_fn(|_| {
94            let socket = unsafe { Self::sockets().get_mut::<tcp::Socket>(self.handle) };
95            if socket.can_send() {
96                let n = socket.send_slice(buf).map_err(|_| ErrorKind::Other)?;
97                // Promptly transmit a record; never wait for the remote ACK here.
98                unsafe { Self::poll_network() };
99                return Poll::Ready(Ok(n));
100            }
101            if !socket.may_send() {
102                return Poll::Ready(Err(ErrorKind::BrokenPipe));
103            }
104            Poll::Pending
105        })
106        .await
107    }
108
109    async fn flush(&mut self) -> Result<(), ErrorKind> {
110        unsafe { Self::poll_network() };
111        Ok(())
112    }
