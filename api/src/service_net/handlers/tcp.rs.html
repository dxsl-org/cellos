19pub(crate) fn handle_tcp_request(
20    req: &NetRequest<'_>,
21    sender: usize,
22    owner: SocketOwner,
23    iface: &mut Interface,
24    sockets: &mut SocketSet<'_>,
25    table: &mut SocketTable,
26    tls_table: &mut BTreeMap<u64, TlsSocketEntry>,
27) -> bool {
28    use NetResponse as R;
29    match req {
30        NetRequest::TcpConnect { addr, port } => {
31            let (handle, cap) = match make_tcp(sockets, table, owner) {
32                Ok(t) => t,
33                Err(_) => {
34                    send_typed(sender, R::Err(0xFF));
35                    return true;
36                }
37            };
38            let remote = IpEndpoint::new(IpAddress::v4(addr[0], addr[1], addr[2], addr[3]), *port);
39            if sockets
40                .get_mut::<tcp::Socket>(handle)
41                .connect(iface.context(), remote, next_ephemeral_port())
42                .is_err()
43            {
44                table.remove_internal(cap);
45                sockets.remove(handle);
46                send_typed(sender, R::Err(0xFF));
47                return true;
48            }
49            table.set_state(cap, SocketState::Connecting);
50            // A cancelled exact call cannot retain a socket capability the
51            // client never received. Legacy callers still get their reply.
52            let mut response = [0u8; api::ipc::IPC_BUF_SIZE];
53            let delivered = api::ipc::encode(&R::CapId(cap as u32), &mut response)
54                .ok()
55                .is_some_and(|bytes| {
56                    if let Some(token) = ostd::ipc::current() {
57                        ostd::ipc::reply(token, bytes).is_ok()
58                    } else {
59                        ostd::ipc::post(sender, bytes).is_ok()
60                    }
61                });
62            if !delivered {
63                table.remove_internal(cap);
64                sockets.remove(handle);
65            }
66            true
67        }
68        NetRequest::TcpSend { cap_id, data } => {
69            let cap = *cap_id as u64;
70            if table.is_udp(cap, owner) {
71                send_typed(sender, R::Data(&0u32.to_le_bytes()));
72                return true;
73            }
74            try_promote(table, sockets, cap, owner);
75            let n = if let Some(h) = table.get(cap, owner) {
76                let s = sockets.get_mut::<tcp::Socket>(h);
77                if s.can_send() {
78                    s.send_slice(data).unwrap_or(0)
79                } else {
80                    0
81                }
82            } else {
83                0
84            };
85            send_typed(sender, R::Data(&(n as u32).to_le_bytes()));
86            true
87        }
88        NetRequest::TcpRecv { cap_id, buf_len } => {
89            let cap = *cap_id as u64;
90            if table.is_udp(cap, owner) {
91                send_typed(sender, R::Data(&[]));
92                return true;
93            }
94            try_promote(table, sockets, cap, owner);
95            let buf_len = (*buf_len as usize).min(NET_TCP_INLINE_DATA_MAX);
96            if let Some(h) = table.get(cap, owner) {
97                let s = sockets.get_mut::<tcp::Socket>(h);
98                if s.can_recv() {
99                    let _ = s.recv(|data| {
100                        let n = data.len().min(buf_len);
101                        send_typed(sender, R::Data(&data[..n]));
102                        (n, ())
103                    });
104                } else if !s.may_recv() {
105                    send_typed(sender, R::Err(0xFF));
106                } else {
107                    send_typed(sender, R::Data(&[]));
108                }
109            } else {
110                send_typed(sender, R::Data(&[]));
111            }
112            true
113        }
114        NetRequest::TcpClose { cap_id } => {
115            let cap = *cap_id as u64;
116            if let Some(h) = table.remove(cap, owner) {
117                sockets.remove(h);
118                tls_table.remove(&cap);
119                send_typed(sender, R::Ok);
120            } else {
121                send_typed(sender, R::Err(0xFF));
122            }
123            true
124        }
125        NetRequest::TcpListen { port } => {
126            let (handle, cap) = match make_tcp(sockets, table, owner) {
127                Ok(t) => t,
128                Err(_) => {
129                    send_typed(sender, R::Err(0xFF));
130                    return true;
131                }
132            };
133            if sockets
134                .get_mut::<tcp::Socket>(handle)
135                .listen(*port)
136                .is_err()
137            {
138                table.remove_internal(cap);
139                sockets.remove(handle);
140                send_typed(sender, R::Err(0xFF));
141                return true;
142            }
143            table.set_state(cap, SocketState::Listening);
144            table.set_listen_port(cap, *port);
145            send_typed(sender, R::CapId(cap as u32));
146            true
147        }
148        NetRequest::TcpAccept { cap_id } => {
149            let cap = *cap_id as u64;
150            if table.is_udp(cap, owner)
151                || table.get_state(cap, owner) != Some(SocketState::Listening)
152            {
153                send_typed(sender, R::Err(0xFF));
154                return true;
155            }
156            let handle = match table.get(cap, owner) {
157                Some(h) => h,
158                None => {
159                    send_typed(sender, R::Err(0xFF));
160                    return true;
161                }
162            };
163            if sockets.get_mut::<tcp::Socket>(handle).state() != tcp::State::Established {
164                send_typed(sender, R::Err(0xFE));
165                return true;
166            }
167            let listen_port = match table.get_listen_port(cap, owner) {
168                Some(p) => p,
169                None => {
170                    send_typed(sender, R::Err(0xFF));
171                    return true;
172                }
173            };
174            // Prepare a replacement listener first. If its listen fails or
175            // capacity is exhausted, the established handle stays attached to
176            // the old cap and the caller can retry without a leaked stream cap.
177            if !table.can_insert(owner) {
178                send_typed(sender, R::Err(0xFF));
179                return true;
180            }
181            let mut replacement = tcp::Socket::new(
182                tcp::SocketBuffer::new(alloc::vec![0u8; 4096]),
183                tcp::SocketBuffer::new(alloc::vec![0u8; 4096]),
184            );
185            if replacement.listen(listen_port).is_err() {
186                send_typed(sender, R::Err(0xFF));
187                return true;
188            }
189            let nh = sockets.add(replacement);
190            match table.insert_with_state(handle, SocketState::Connected, owner) {
191                Ok(stream_cap) => {
192                    table.update_handle(cap, nh);
193                    send_typed(sender, R::CapId(stream_cap as u32));
194                }
195                Err(_) => {
196                    sockets.remove(nh);
197                    send_typed(sender, R::Err(0xFF));
198                }
199            }
200            true
201        }
202        NetRequest::TcpRecvReady { cap_id, buf_len } => {
203            let cap = *cap_id as u64;
204            if table.is_udp(cap, owner) || *buf_len == 0 {
205                send_typed(sender, R::Err(0xFF));
206            } else if let Some(h) = table.get(cap, owner) {
207                let s = sockets.get_mut::<tcp::Socket>(h);
208                if s.can_recv() {
209                    let limit = (*buf_len as usize).min(NET_TCP_INLINE_DATA_MAX);
210                    let _ = s.recv(|data| {
211                        let n = data.len().min(limit);
212                        send_typed(
213                            sender,
214                            if n > 0 {
215                                R::Data(&data[..n])
216                            } else {
217                                R::NotReady
218                            },
219                        );
220                        (n, ())
221                    });
222                } else if !s.may_recv() {
223                    send_typed(sender, R::Eof);
224                } else {
225                    send_typed(sender, R::NotReady);
226                }
227            } else {
228                send_typed(sender, R::Err(0xFF));
229            }
230            true
231        }
232        NetRequest::TcpSendReady { cap_id, data } => {
233            let cap = *cap_id as u64;
234            if table.is_udp(cap, owner) || table.is_graceful_closing(cap) {
235                send_typed(sender, R::Err(0xFF));
236            } else if let Some(h) = table.get(cap, owner) {
237                let s = sockets.get_mut::<tcp::Socket>(h);
238                if !s.may_send() {
239                    send_typed(sender, R::Eof);
240                } else if !s.can_send() {
241                    send_typed(sender, R::NotReady);
242                } else {
243                    let n = s.send_slice(data).unwrap_or(0);
244                    send_typed(
245                        sender,
246                        if n == 0 && !data.is_empty() {
247                            R::NotReady
248                        } else {
249                            R::WriteProgress(n as u32)
250                        },
251                    );
252                }
253            } else {
254                send_typed(sender, R::Err(0xFF));
255            }
256            true
257        }
258        NetRequest::TcpCloseGraceful { cap_id } => {
259            let cap = *cap_id as u64;
260            if table.is_udp(cap, owner) {
261                send_typed(sender, R::Err(0xFF));
262            } else if let Some(h) = table.get(cap, owner) {
263                if !table.is_graceful_closing(cap) {
264                    sockets.get_mut::<tcp::Socket>(h).close();
265                    table.begin_graceful_close(
266                        cap,
267                        crate::service_runtime::sys_get_time().saturating_add(100_000_000),
268                    );
269                }
270                send_typed(sender, R::Ok);
271            } else {
272                send_typed(sender, R::Err(0xFF));
273            }
274            true
275        }
276        NetRequest::TcpReady {
277            interests,
278            cursor,
279            wait,
280        } => {
281            if *wait {
282                // The runtime owns deferred waits; a direct dispatch cannot
283                // retain the receive token and must refuse rather than poll.
284                send_typed(sender, R::Err(0xFD));
285            } else {
286                match readiness(interests, *cursor, owner, sockets, table) {
287                    Ok((events, next_cursor)) => {
288                        let mut packed = [0u8; READY_PAGE * api::ipc::NET_READY_RECORD_BYTES];
289                        let bytes = pack_events(&events, &mut packed);
290                        send_typed(
291                            sender,
292                            R::TcpReady {
293                                events: bytes,
294                                next_cursor,
295                            },
296                        );
297                    }
298                    Err(()) => send_typed(sender, R::Err(0xFF)),
299                }
300            }
301            true
302        }
303        NetRequest::SocketState { cap_id } => {
304            let cap = *cap_id as u64;
305            if table.is_udp(cap, owner) {
306                send_typed(sender, R::State(0x00));
307                return true;
308            }
309            let byte = if let Some(h) = table.get(cap, owner) {
310                tcp_state_byte(sockets.get_mut::<tcp::Socket>(h).state())
311            } else {
312                0x00
313            };
314            send_typed(sender, R::State(byte));
315            true
316        }
317        _ => false,
318    }
319}

321pub(crate) const READY_PAGE: usize = 32;
322pub(crate) const MAX_INTERESTS: usize = 256;
323pub(crate) type ReadyEvents = heapless::Vec<TcpReadyEvent, READY_PAGE>;
324
325/// Level-triggered readiness after interface progress. Every cap is validated
326/// before returning any events; foreign owners cannot probe another owner's
327/// socket even if its cap is mixed into an otherwise valid batch.
328pub(crate) fn readiness(
329    interests: &[u8],
330    cursor: u16,
331    owner: SocketOwner,
332    sockets: &mut SocketSet<'_>,
333    table: &SocketTable,
334) -> Result<(ReadyEvents, u16), ()> {
335    if interests.is_empty()
336        || interests.len() > MAX_INTERESTS * api::ipc::NET_READY_RECORD_BYTES
337        || !interests
338            .len()
339            .is_multiple_of(api::ipc::NET_READY_RECORD_BYTES)
340    {
341        return Err(());
342    }
343    let count = interests.len() / api::ipc::NET_READY_RECORD_BYTES;
344    for raw in interests.chunks_exact(api::ipc::NET_READY_RECORD_BYTES) {
345        let interest = TcpInterest::decode(raw).ok_or(())?;
346        if interest.mask == 0
347            || interest.mask
348                & !(net_ready::ACCEPT
349                    | net_ready::READ
350                    | net_ready::WRITE
351                    | net_ready::EOF
352                    | net_ready::ERROR)
353                != 0
354            || table.is_udp(interest.cap_id as u64, owner)
355            || table.get(interest.cap_id as u64, owner).is_none()
356        {
357            return Err(());
358        }
359    }
360    let mut events = ReadyEvents::new();
361    let start = cursor as usize % count;
362    let mut next = start;
363    for _ in 0..count {
364        let interest = TcpInterest::decode(
365            &interests[next * api::ipc::NET_READY_RECORD_BYTES
366                ..(next + 1) * api::ipc::NET_READY_RECORD_BYTES],
367        )
368        .ok_or(())?;
369        let cap = interest.cap_id as u64;
370        let handle = table.get(cap, owner).ok_or(())?;
371        let socket = sockets.get_mut::<tcp::Socket>(handle);
372        let mut ready = 0;
373        if table.get_listen_port(cap, owner).is_some() {
374            if socket.state() == tcp::State::Established && interest.mask & net_ready::ACCEPT != 0 {
375                ready |= net_ready::ACCEPT;
376            }
377            if socket.state() == tcp::State::Closed {
378                ready |= net_ready::ERROR;
379            }
380        } else {
381            if socket.can_recv() && interest.mask & net_ready::READ != 0 {
382                ready |= net_ready::READ;
383            }
384            if socket.can_send()
385                && !table.is_graceful_closing(cap)
386                && interest.mask & net_ready::WRITE != 0
387            {
388                ready |= net_ready::WRITE;
389            }
390            // CLOSE-WAIT (peer shutdown of its write half) can still serve an
391            // HTTP response. EOF-only monitors indicate unusable TX, not FIN.
392            if !socket.may_send() && !socket.is_active() {
393                ready |= net_ready::EOF;
394            }
395            if socket.state() == tcp::State::Closed && !table.is_graceful_closing(cap) {
396                ready |= net_ready::ERROR;
397            }
398        }
399        if ready != 0 {
400            events
401                .push(TcpReadyEvent {
402                    cap_id: interest.cap_id,
403                    ready,
404                })
405                .map_err(|_| ())?;
406        }
407        next = (next + 1) % count;
408        if events.len() == READY_PAGE {
409            break;
410        }
411    }
412    Ok((events, next as u16))
413}

415pub(crate) fn pack_events<'a>(
416    events: &[TcpReadyEvent],
417    out: &'a mut [u8; READY_PAGE * api::ipc::NET_READY_RECORD_BYTES],
418) -> &'a [u8] {
419    let len = events.len() * api::ipc::NET_READY_RECORD_BYTES;
420    for (event, chunk) in events
421        .iter()
422        .zip(out[..len].chunks_exact_mut(api::ipc::NET_READY_RECORD_BYTES))
423    {
424        chunk[..4].copy_from_slice(&event.cap_id.to_le_bytes());
425        chunk[4] = event.ready;
426    }
427    &out[..len]
