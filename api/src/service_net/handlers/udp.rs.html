15pub(crate) fn handle_udp_request(
16    req: &NetRequest<'_>,
17    sender: usize,
18    owner: SocketOwner,
19    iface: &mut Interface,
20    device: &mut VirtioNetDevice,
21    sockets: &mut SocketSet<'_>,
22    table: &mut SocketTable,
23) -> bool {
24    use NetResponse as R;
25    match req {
26        NetRequest::UdpCreate => {
27            let handle = sockets.add(udp::Socket::new(
28                udp::PacketBuffer::new(alloc::vec![udp::PacketMetadata::EMPTY; 4], alloc::vec![0u8; 1024]),
29                udp::PacketBuffer::new(alloc::vec![udp::PacketMetadata::EMPTY; 4], alloc::vec![0u8; 1024]),
30            ));
31            match table.insert(handle, owner) {
32                Ok(cap) => {
33                    table.mark_udp(cap);
34                    send_typed(sender, R::CapId(cap as u32));
35                }
36                Err(_) => {
37                    sockets.remove(handle);
38                    send_typed(sender, R::Err(0xFF));
39                }
40            }
41            true
42        }
43        NetRequest::UdpBind { cap_id, port } => {
44            let cap = *cap_id as u64;
45            let port = if *port == 0 { next_ephemeral_port() } else { *port };
46            let ok = if let Some(h) = table.get(cap, owner) {
47                sockets.get_mut::<udp::Socket>(h).bind(port).is_ok()
48            } else {
49                false
50            };
51            if ok {
52                table.set_state(cap, SocketState::Listening);
53                send_typed(sender, R::Ok);
54            } else {
55                send_typed(sender, R::Err(0xFF));
56            }
57            true
58        }
59        NetRequest::UdpSend { cap_id, addr, port, data } => {
60            let cap = *cap_id as u64;
61            let ep = IpEndpoint::new(IpAddress::v4(addr[0], addr[1], addr[2], addr[3]), *port);
62            let n = if let Some(h) = table.get(cap, owner) {
63                if sockets.get_mut::<udp::Socket>(h).send_slice(data, ep).is_ok() {
64                    iface.poll(now_instant(), device, sockets);
65                    data.len()
66                } else {
67                    0
68                }
69            } else {
70                0
71            };
72            send_typed(sender, R::Data(&(n as u32).to_le_bytes()));
73            true
74        }
75        NetRequest::UdpRecv { cap_id, buf_len } => {
76            let cap = *cap_id as u64;
77            let buf_len = (*buf_len as usize).min(512);
78            let result = if let Some(h) = table.get(cap, owner) {
79                let s = sockets.get_mut::<udp::Socket>(h);
80                if s.can_recv() {
81                    let mut raw = alloc::vec![0u8; buf_len];
82                    s.recv_slice(&mut raw).ok().map(|(n, meta)| {
83                        let IpAddress::Ipv4(src_ip) = meta.endpoint.addr;
84                        let mut reply = alloc::vec![0u8; 6 + n];
85                        reply[0..4].copy_from_slice(src_ip.as_bytes());
86                        reply[4..6].copy_from_slice(&meta.endpoint.port.to_le_bytes());
87                        reply[6..6 + n].copy_from_slice(&raw[..n]);
88                        reply
89                    })
90                } else {
91                    None
92                }
93            } else {
94                None
95            };
96            match result {
97                Some(reply) => {
98                    let mut rb = [0u8; IPC_BUF_SIZE];
99                    if let Ok(s) = ipc::encode(&R::Data(&reply), &mut rb) {
100                        sys_send(sender, s);
101                    }
102                }
103                None => send_typed(sender, R::Data(&[])),
104            }
105            true
106        }
107        _ => false,
108    }
109}
