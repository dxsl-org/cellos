1//! Coarse per-Cell network rights, enforced before any effect.
2//!
3//! The kernel exports the sender Cell's reviewed net rights as
4//! `CALLER_FLAG_NET_*` bits of the attested caller trailer. Every request that
5//! creates network reach — a name lookup, an egress socket, an inbound port or
6//! group, a raw frame — needs its right. A request that only acts on a socket
7//! the caller already owns is not gated: owning one already required a gated
8//! request, and the socket table rejects a foreign cap with its existing error.
9//! `GetLocalIp` stays ungated (it discloses only the local address).
10//!
11//! A denial is `NetResponse::Err(NET_ERR_DENIED)` and nothing else: the runtime
12//! answers it before the owner watch, both intercepts and dispatch, so no socket,
13//! DNS query, owner watch or frame exists because of a denied request.
14
15use api::caller_identity::{
16    CALLER_FLAG_NET_CONNECT, CALLER_FLAG_NET_L2, CALLER_FLAG_NET_LISTEN, CALLER_FLAG_NET_RESOLVE,
17};
18use api::ipc::{self, NetRequest, NetResponse, NET_ERR_DENIED};
19
20use crate::tls_wire::{parse_raw_tls_request, RawTlsRequest, TLS_CLOSE_OP};

22/// One coarse network right.
23#[derive(Clone, Copy, Debug, Eq, PartialEq)]
24pub(crate) enum Right {
25    /// D: name resolution.
26    Resolve,
27    /// N: egress sockets.
28    Connect,
29    /// Nl: inbound ports and multicast groups.
30    Listen,
31    /// L2: raw Ethernet frames.
32    L2,
33}
34
35impl Right {
36    /// The attested caller flag that carries this right.
37    pub(crate) const fn caller_flag(self) -> u32 {
38        match self {
39            Self::Resolve => CALLER_FLAG_NET_RESOLVE,
40            Self::Connect => CALLER_FLAG_NET_CONNECT,
41            Self::Listen => CALLER_FLAG_NET_LISTEN,
42            Self::L2 => CALLER_FLAG_NET_L2,
43        }
44    }
45}
46
47/// The right a typed request needs, or `None` when it is ungated.
48///
49/// No `_` arm: a new `NetRequest` variant does not compile until it is
50/// classified here.
51pub(crate) fn required_right(request: &NetRequest<'_>) -> Option<Right> {
52    match request {
53        NetRequest::Resolve { .. } => Some(Right::Resolve),
54        NetRequest::TcpConnect { .. } | NetRequest::UdpCreate | NetRequest::UdpSend { .. } => {
55            Some(Right::Connect)
56        }
57        NetRequest::TcpListen { .. }
58        | NetRequest::UdpBind { .. }
59        | NetRequest::MulticastJoin { .. } => Some(Right::Listen),
60        NetRequest::L2Send { .. } | NetRequest::L2Recv { .. } => Some(Right::L2),
61        NetRequest::TcpSend { .. }
62        | NetRequest::TcpRecv { .. }
63        | NetRequest::TcpClose { .. }
64        | NetRequest::TcpAccept { .. }
65        | NetRequest::UdpRecv { .. }
66        | NetRequest::SocketState { .. }
67        | NetRequest::GetLocalIp
68        | NetRequest::MulticastLeave { .. }
69        | NetRequest::TcpReady { .. }
70        | NetRequest::TcpRecvReady { .. }
71        | NetRequest::TcpSendReady { .. }
72        | NetRequest::TcpCloseGraceful { .. } => None,
73    }
74}

76/// The right a legacy raw TLS request needs, or `None` when it is ungated.
77///
78/// `Connect` opens an outbound TCP socket to any address, so it is egress like
79/// `TcpConnect`; the rest act on a TLS cap the caller already owns.
80pub(crate) fn raw_tls_required_right(request: &RawTlsRequest<'_>) -> Option<Right> {
81    match request {
82        RawTlsRequest::Connect { .. } => Some(Right::Connect),
83        RawTlsRequest::Close { .. } | RawTlsRequest::Send { .. } | RawTlsRequest::Recv { .. } => {
84            None
85        }
86    }
87}

89/// How a denied request is answered: in the wire format its caller reads.
90#[derive(Clone, Copy, Debug, Eq, PartialEq)]
91pub(crate) enum Denial {
92    /// `NetResponse::Err(NET_ERR_DENIED)`.
93    Typed,
94    /// Raw TLS connect has no error code on its wire: cap `0` is its existing
95    /// "connect failed" reply.
96    RawTlsConnect,
97}
98
99/// Bytes enough for either denial reply.
100pub(crate) const DENIAL_REPLY_LEN: usize = 8;
101
102impl Denial {
103    /// Encode the denial reply into `out`.
104    pub(crate) fn reply(self, out: &mut [u8; DENIAL_REPLY_LEN]) -> &[u8] {
105        match self {
106            Self::Typed => match ipc::encode(&NetResponse::Err(NET_ERR_DENIED), out) {
107                Ok(bytes) => bytes,
108                // Two postcard bytes always fit; an empty reply is still a refusal.
109                Err(_) => &[],
110            },
111            Self::RawTlsConnect => {
112                *out = 0u64.to_le_bytes();
113                &out[..]
114            }
115        }
116    }
117}

119/// Admit or deny one attested request by the caller's flags.
120///
121/// Classification mirrors dispatch exactly: a buffer that decodes as a
122/// `NetRequest` is typed (both intercepts and `handle_request` decode it the same
123/// way); otherwise a first byte at or above `TLS_CLOSE_OP` goes to the raw TLS
124/// handler. Anything else reaches no handler and has no effect to gate.
125pub(crate) fn admit(buffer: &[u8], flags: u32) -> Result<(), Denial> {
126    let (right, denial) = match ipc::decode::<NetRequest<'_>>(buffer) {
127        Ok(request) => (required_right(&request), Denial::Typed),
128        Err(_) if buffer.first().copied().unwrap_or(0) >= TLS_CLOSE_OP => {
129            match parse_raw_tls_request(buffer) {
130                Ok(request) => (raw_tls_required_right(&request), Denial::RawTlsConnect),
131                Err(_) => (None, Denial::Typed),
132            }
133        }
134        Err(_) => (None, Denial::Typed),
135    };
136    match right {
137        Some(right) if flags & right.caller_flag() == 0 => Err(denial),
138        _ => Ok(()),
139    }
140}
