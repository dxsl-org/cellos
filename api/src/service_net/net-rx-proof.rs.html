1//! When the net service may claim `[net-rx-producer] irq->completion PASS`.
2//!
3//! The claim needs a NET_RX completion record, which only the NIC interrupt
4//! produces, tied to a frame the service really received. The frame may be
5//! polled on either side of that record. Replies to the service's own
6//! transmissions usually arrive inside the turn that sent the request, and
7//! the RX poll drains them before the service parks. Their interrupt then
8//! surfaces only through the level flag on a later wait. A `restrict=on`
9//! QEMU network carries nothing else (the DHCP OFFER and ACK), so a rule that
10//! accepted only frames drained *after* the record never passed there.
11
12#[derive(Debug, Default)]
13pub(super) struct NetRxProof {
14    proved: bool,
15    /// Frames were drained since the last NET_RX completion was consumed.
16    rx_since_completion: bool,
17    /// A NET_RX completion awaits the next turn's drain; one turn only.
18    completion_pending: bool,
19}
20
21impl NetRxProof {
22    /// One turn drained `frames` frames. Returns true exactly once: on the
23    /// call that completes the proof.
24    pub(super) fn drained(&mut self, frames: usize) -> bool {
25        let completion_before = core::mem::take(&mut self.completion_pending);
26        if frames == 0 {
27            return false;
28        }
29        self.rx_since_completion = true;
30        self.prove(completion_before)
31    }
32
33    /// A wait returned a NET_RX completion record. Returns true exactly once:
34    /// on the call that completes the proof.
35    pub(super) fn completion(&mut self) -> bool {
36        let frames_before = core::mem::take(&mut self.rx_since_completion);
37        self.completion_pending = true;
38        self.prove(frames_before)
39    }
40
41    fn prove(&mut self, tied: bool) -> bool {
42        if tied && !self.proved {
43            self.proved = true;
44            return true;
45        }
46        false
47    }
48}

50#[cfg(test)]
51mod tests {
52    use super::NetRxProof;
53
54    #[test]
55    fn frame_drained_before_the_completion_proves() {
56        let mut proof = NetRxProof::default();
57        assert!(!proof.drained(1));
58        assert!(proof.completion());
59    }
60
61    #[test]
62    fn frame_drained_after_the_completion_proves() {
63        let mut proof = NetRxProof::default();
64        assert!(!proof.completion());
65        assert!(proof.drained(1));
66    }
67
68    #[test]
69    fn completion_without_any_frame_does_not_prove() {
70        let mut proof = NetRxProof::default();
71        assert!(!proof.completion());
72        assert!(!proof.drained(0));
73        assert!(!proof.completion());
74        assert!(!proof.drained(0));
75    }
76
77    #[test]
78    fn completion_expires_after_one_empty_turn() {
79        let mut proof = NetRxProof::default();
80        assert!(!proof.completion());
81        assert!(!proof.drained(0));
82        assert!(
83            !proof.drained(1),
84            "a later frame is not tied to the stale record"
85        );
86    }
87
88    #[test]
89    fn frames_count_once_toward_a_completion() {
90        let mut proof = NetRxProof::default();
91        assert!(!proof.drained(0));
92        assert!(!proof.completion());
93        assert!(!proof.drained(0));
94        assert!(!proof.completion(), "no frame since the previous record");
95    }
96
97    #[test]
98    fn proof_is_reported_once() {
99        let mut proof = NetRxProof::default();
100        assert!(!proof.drained(1));
101        assert!(proof.completion());
102        assert!(!proof.drained(1));
103        assert!(!proof.completion());
104    }
105}
