1//! Service-level DNS resolution.
2//!
3//! The net cell owns exactly one smoltcp `dns::Socket`; every consumer resolves
4//! through `NetRequest::Resolve`, so literal parsing, the SLIRP alias table and
5//! the wire query live in one place instead of one copy per tool. Resolution
6//! order:
7//!
8//! 1. **IPv4 literal** — returned without touching the wire.
9//! 2. **SLIRP alias** — `gateway`/`host` → 10.0.2.2 (host loopback), `dns` →
10//!    the leased DNS server, `localhost` → 127.0.0.1.
11//! 3. **UDP A-record query** to the DNS server from the DHCP lease (option 6),
12//!    falling back to SLIRP's 10.0.2.3 when the lease carries none.
13//!
14//! Wire queries retain a bounded smoltcp DNS slot and are polled by the net
15//! reactor alongside TCP; a slow resolver never parks other clients.
16
17use alloc::vec;
18use smoltcp::iface::{Interface, SocketHandle, SocketSet};
19use smoltcp::socket::dns;
20use smoltcp::wire::{DnsQueryType, IpAddress};
21
22/// QEMU SLIRP's built-in resolver — the fallback when the DHCP lease carries no
23/// DNS option. SLIRP forwards it to the host's own resolvers.
24pub const SLIRP_DNS_SERVER: [u8; 4] = [10, 0, 2, 3];
25
26/// SLIRP's alias for the host loopback: guest traffic to it lands on the host's
27/// 127.0.0.1, which is how the integration tests reach host mocks.
28const SLIRP_HOST: [u8; 4] = [10, 0, 2, 2];
29
30
31/// Resolve without the wire: IPv4 literals and the SLIRP names.
32///
33/// `server` is the DNS server currently in force, so `dns` answers with the
34/// address the resolver would actually query.
35pub fn static_lookup(hostname: &str, server: [u8; 4]) -> Option<[u8; 4]> {
36    match hostname {
37        "dns" => return Some(server),
38        "gateway" | "host" => return Some(SLIRP_HOST),
39        "localhost" => return Some([127, 0, 0, 1]),
40        _ => {}
41    }
42    parse_ipv4(hostname)
43}
44
45/// Strict dotted-quad parser: exactly four decimal octets, no leading zeros,
46/// each ≤ 255. Anything else is a hostname and goes to the resolver.
47fn parse_ipv4(s: &str) -> Option<[u8; 4]> {
48    let mut octets = [0u8; 4];
49    let mut count = 0;
50    for part in s.split('.') {
51        if count == 4 {
52            return None;
53        }
54        octets[count] = parse_octet(part)?;
55        count += 1;
56    }
57    (count == 4).then_some(octets)
58}
59
60fn parse_octet(s: &str) -> Option<u8> {
61    if s.is_empty() || s.len() > 3 || (s.len() > 1 && s.starts_with('0')) {
62        return None;
63    }
64    let mut n: u16 = 0;
65    for ch in s.bytes() {
66        if !ch.is_ascii_digit() {
67            return None;
68        }
69        n = n * 10 + (ch - b'0') as u16;
70    }
71    (n <= 255).then_some(n as u8)
72}

74/// The net cell's resolver: one `dns::Socket` plus the server it queries.
75pub struct Resolver {
76    handle: SocketHandle,
77    server: [u8; 4],
78}
79
80impl Resolver {
81    /// Install the DNS socket in the interface's socket set.
82    pub fn install(sockets: &mut SocketSet<'_>) -> Self {
83        let server = SLIRP_DNS_SERVER;
84        let handle = sockets.add(dns::Socket::new(&[ip_address(server)], vec![]));
85        Self { handle, server }
86    }
87
88    /// The DNS server in force (lease option 6, else SLIRP's).
89    pub fn server(&self) -> [u8; 4] {
90        self.server
91    }
92
93    /// Adopt the DNS server handed out by DHCP.
94    pub fn set_server(&mut self, sockets: &mut SocketSet<'_>, server: [u8; 4]) {
95        if server == self.server {
96            return;
97        }
98        self.server = server;
99        sockets
100            .get_mut::<dns::Socket>(self.handle)
101            .update_servers(&[ip_address(server)]);
102    }
103
104    /// Begin a wire query without parking the net service loop. The returned
105    /// handle owns one smoltcp DNS slot until poll_result or cancel.
106    pub fn start(
107        &self,
108        hostname: &str,
109        iface: &mut Interface,
110        sockets: &mut SocketSet<'_>,
111    ) -> Option<dns::QueryHandle> {
112        sockets
113            .get_mut::<dns::Socket>(self.handle)
114            .start_query(iface.context(), hostname, DnsQueryType::A)
115            .ok()
116    }
117
118    pub fn poll_result(
119        &self,
120        query: dns::QueryHandle,
121        sockets: &mut SocketSet<'_>,
122    ) -> Option<Option<[u8; 4]>> {
123        match sockets.get_mut::<dns::Socket>(self.handle).get_query_result(query) {
124            Ok(addresses) => Some(first_ipv4(&addresses)),
125            Err(dns::GetQueryResultError::Failed) => Some(None),
126            Err(dns::GetQueryResultError::Pending) => None,
127        }
128    }
129
130    pub fn cancel(&self, query: dns::QueryHandle, sockets: &mut SocketSet<'_>) {
131        sockets.get_mut::<dns::Socket>(self.handle).cancel_query(query);
132    }
133
134}
135
136fn first_ipv4(addresses: &[IpAddress]) -> Option<[u8; 4]> {
137    addresses.iter().find_map(|address| match address {
138        IpAddress::Ipv4(v4) => Some(v4.0),
139        #[allow(unreachable_patterns)] // reason: proto-ipv6 is not enabled here
140        _ => None,
141    })
142}
143
144fn ip_address(ip: [u8; 4]) -> IpAddress {
145    IpAddress::v4(ip[0], ip[1], ip[2], ip[3])
146}
