1//! Service-level DNS resolution.
2//!
3//! The net cell owns exactly one smoltcp `dns::Socket`; every consumer resolves
4//! through `NetRequest::Resolve`, so literal parsing, the SLIRP alias table and
5//! the wire query live in one place instead of one copy per tool. Resolution
6//! order:
7//!
8//! 1. **IPv4 literal** — returned without touching the wire.
9//! 2. **SLIRP alias** — `gateway`/`host` → 10.0.2.2 (host loopback), `dns` →
10//!    the leased DNS server, `localhost` → 127.0.0.1.
11//! 3. **UDP A-record query** to the DNS server from the DHCP lease (option 6),
12//!    falling back to SLIRP's 10.0.2.3 when the lease carries none.
13//!
14//! The query is synchronous: the service is a single-threaded message loop, so a
15//! lookup parks request handling for at most [`RESOLVE_BUDGET`]. Other sockets
16//! keep progressing while it waits because the wait loop polls the whole
17//! interface, not just the DNS socket.
18
19use alloc::vec;
20use ostd::syscall::sys_wait_completion;
21use smoltcp::iface::{Interface, SocketHandle, SocketSet};
22use smoltcp::socket::dns;
23use smoltcp::time::Duration;
24use smoltcp::wire::{DnsQueryType, IpAddress};
25
26use crate::interface::VirtioNetDevice;
27use crate::service_runtime::now_instant;

29/// QEMU SLIRP's built-in resolver — the fallback when the DHCP lease carries no
30/// DNS option. SLIRP forwards it to the host's own resolvers.
31pub const SLIRP_DNS_SERVER: [u8; 4] = [10, 0, 2, 3];
32
33/// SLIRP's alias for the host loopback: guest traffic to it lands on the host's
34/// 127.0.0.1, which is how the integration tests reach host mocks.
35const SLIRP_HOST: [u8; 4] = [10, 0, 2, 2];
36
37/// Ceiling on one A-record lookup. smoltcp retransmits at 1 s, then 2 s; a
38/// server that has not answered within this window is treated as unreachable.
39const RESOLVE_BUDGET: Duration = Duration::from_millis(3_000);
40
41/// Wake cadence while a query is in flight — one scheduler tick (10 ms).
42const RX_POLL_TICKS: u64 = 1;
43
44/// Iteration backstop for [`Resolver::resolve`].
45///
46/// The budget above is wall-clock. At one wake per tick this is twice what the
47/// budget can consume, so it only bites if the clock itself stops advancing —
48/// and then the service still answers `Err` instead of parking its message loop
49/// forever (which the watchdog would read as a hung cell).
50const RESOLVE_POLL_CEILING: usize = 600;
51
52/// Resolve without the wire: IPv4 literals and the SLIRP names.
53///
54/// `server` is the DNS server currently in force, so `dns` answers with the
55/// address the resolver would actually query.
56pub fn static_lookup(hostname: &str, server: [u8; 4]) -> Option<[u8; 4]> {
57    match hostname {
58        "dns" => return Some(server),
59        "gateway" | "host" => return Some(SLIRP_HOST),
60        "localhost" => return Some([127, 0, 0, 1]),
61        _ => {}
62    }
63    parse_ipv4(hostname)
64}
65
66/// Strict dotted-quad parser: exactly four decimal octets, no leading zeros,
67/// each ≤ 255. Anything else is a hostname and goes to the resolver.
68fn parse_ipv4(s: &str) -> Option<[u8; 4]> {
69    let mut octets = [0u8; 4];
70    let mut count = 0;
71    for part in s.split('.') {
72        if count == 4 {
73            return None;
74        }
75        octets[count] = parse_octet(part)?;
76        count += 1;
77    }
78    (count == 4).then_some(octets)
79}
80
81fn parse_octet(s: &str) -> Option<u8> {
82    if s.is_empty() || s.len() > 3 || (s.len() > 1 && s.starts_with('0')) {
83        return None;
84    }
85    let mut n: u16 = 0;
86    for ch in s.bytes() {
87        if !ch.is_ascii_digit() {
88            return None;
89        }
90        n = n * 10 + (ch - b'0') as u16;
91    }
92    (n <= 255).then_some(n as u8)
93}

95/// The net cell's resolver: one `dns::Socket` plus the server it queries.
96pub struct Resolver {
97    handle: SocketHandle,
98    server: [u8; 4],
99}
100
101impl Resolver {
102    /// Install the DNS socket in the interface's socket set.
103    pub fn install(sockets: &mut SocketSet<'_>) -> Self {
104        let server = SLIRP_DNS_SERVER;
105        let handle = sockets.add(dns::Socket::new(&[ip_address(server)], vec![]));
106        Self { handle, server }
107    }
108
109    /// The DNS server in force (lease option 6, else SLIRP's).
110    pub fn server(&self) -> [u8; 4] {
111        self.server
112    }
113
114    /// Adopt the DNS server handed out by DHCP.
115    pub fn set_server(&mut self, sockets: &mut SocketSet<'_>, server: [u8; 4]) {
116        if server == self.server {
117            return;
118        }
119        self.server = server;
120        sockets
121            .get_mut::<dns::Socket>(self.handle)
122            .update_servers(&[ip_address(server)]);
123    }
124
125    /// Resolve `hostname` to an IPv4 address, or `None` if the server does not
126    /// answer within [`RESOLVE_BUDGET`] (the caller replies `Err`).
127    ///
128    /// Blocks: drives the interface in a bounded loop so the query, its
129    /// retransmits and any RX frames are all serviced.
130    pub fn resolve(
131        &self,
132        hostname: &str,
133        iface: &mut Interface,
134        device: &mut VirtioNetDevice,
135        sockets: &mut SocketSet<'_>,
136    ) -> Option<[u8; 4]> {
137        let query = {
138            let socket = sockets.get_mut::<dns::Socket>(self.handle);
139            socket
140                .start_query(iface.context(), hostname, DnsQueryType::A)
141                .ok()?
142        };
143
144        let deadline = now_instant() + RESOLVE_BUDGET;
145        for _ in 0..RESOLVE_POLL_CEILING {
146            device.pump_rx_split();
147            iface.poll(now_instant(), device, sockets);
148            match sockets
149                .get_mut::<dns::Socket>(self.handle)
150                .get_query_result(query)
151            {
152                Ok(addresses) => return first_ipv4(&addresses),
153                Err(dns::GetQueryResultError::Failed) => return None,
154                Err(dns::GetQueryResultError::Pending) => {}
155            }
156            if now_instant() >= deadline {
157                break;
158            }
159            wait_for_frame();
160        }
161        sockets
162            .get_mut::<dns::Socket>(self.handle)
163            .cancel_query(query);
164        None
165    }
166}
167
168fn first_ipv4(addresses: &[IpAddress]) -> Option<[u8; 4]> {
169    addresses.iter().find_map(|address| match address {
170        IpAddress::Ipv4(v4) => Some(v4.0),
171        #[allow(unreachable_patterns)] // reason: proto-ipv6 is not enabled here
172        _ => None,
173    })
174}
175
176fn ip_address(ip: [u8; 4]) -> IpAddress {
177    IpAddress::v4(ip[0], ip[1], ip[2], ip[3])
178}
179
180/// Park until an RX frame arrives or one scheduler tick elapses. The timeout is
181/// the contract: a dropped reply must not park the service loop forever.
182fn wait_for_frame() {
183    let _ = sys_wait_completion(api::syscall::events::NET_RX, RX_POLL_TICKS);
184}
