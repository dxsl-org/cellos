1//! RFC 3927 IPv4 link-local address claiming: probe, announce, defend.
2//!
3//! A pure state machine on a millisecond clock. The service loop feeds it the
4//! ARP packets the device tapped off its RX path plus the current time, and
5//! applies the [`Event`]s it emits (frames to send, the address to configure or
6//! drop, log lines). Deadlines are checked once per loop turn, and a turn parks
7//! for at most 100 ms, so the second-scale RFC intervals stay within one turn.
8
9use alloc::{collections::VecDeque, vec::Vec};
10
11use crate::arp::{ArpOp, ArpPacket};
12
13/// RFC 3927 §9 constants, in milliseconds where they are times.
14pub(crate) const PROBE_WAIT_MS: u64 = 1_000;
15pub(crate) const PROBE_NUM: u8 = 3;
16pub(crate) const PROBE_MIN_MS: u64 = 1_000;
17pub(crate) const PROBE_MAX_MS: u64 = 2_000;
18pub(crate) const ANNOUNCE_WAIT_MS: u64 = 2_000;
19pub(crate) const ANNOUNCE_NUM: u8 = 2;
20pub(crate) const ANNOUNCE_INTERVAL_MS: u64 = 2_000;
21pub(crate) const MAX_CONFLICTS: u32 = 10;
22pub(crate) const RATE_LIMIT_INTERVAL_MS: u64 = 60_000;
23pub(crate) const DEFEND_INTERVAL_MS: u64 = 10_000;
24/// The same peer observation is logged at most once per this interval, so a
25/// chatty neighbour cannot flood the console but a rebooted one shows again.
26const PEER_LOG_INTERVAL_MS: u64 = 10_000;
27/// Peers remembered for log de-duplication and pending verification.
28const PEER_SLOTS: usize = 8;
29
30/// First candidate, derived from `mac` so a host comes back with the same
31/// address (§2.1). Only 169.254.1.0–169.254.254.255 is assignable: the first
32/// and last /24 of 169.254/16 are reserved, so a MAC octet of 0 or 255 is remapped.
33pub(crate) fn first_candidate(mac: [u8; 6]) -> [u8; 4] {
34    let third = match mac[4] {
35        0 | 255 => 1 + (mac[3] ^ mac[4]) % 254,
36        octet => octet,
37    };
38    [169, 254, third, mac[5]]
39}
40
41/// What a neighbour's ARP packet was.
42#[derive(Clone, Copy, Debug, PartialEq, Eq)]
43pub(crate) enum PeerKind {
44    Announce,
45    Request,
46    Reply,
47}
48
49/// One step the service loop must apply.
50#[derive(Clone, Copy, Debug, PartialEq, Eq)]
51pub(crate) enum Event {
52    /// Transmit this ARP packet.
53    Send(ArpPacket),
54    /// Probing starts for `ip`, the `attempt`-th candidate of this run.
55    Probing { ip: [u8; 4], attempt: u32 },
56    /// Probing found no owner: configure `ip`; announcements follow.
57    Claim([u8; 4]),
58    /// Announcements done: `ip` is the interface address.
59    Bound([u8; 4]),
60    /// `mac` uses or probes for the candidate `ip`; another one is chosen.
61    Conflict { ip: [u8; 4], mac: [u8; 6] },
62    /// `mac` claimed our `ip`; it was defended with one announcement.
63    Defended { ip: [u8; 4], mac: [u8; 6] },
64    /// `mac` claimed our `ip` again within the defend interval: unconfigure it.
65    Release { ip: [u8; 4], mac: [u8; 6] },
66    /// A neighbour at link-local `ip`/`mac` was heard.
67    Peer {
68        ip: [u8; 4],
69        mac: [u8; 6],
70        kind: PeerKind,
71    },
72}
73
74#[derive(Clone, Copy, Debug, PartialEq, Eq)]
75enum Phase {
76    /// `sent` probes are out; the next one (after the last: the claim) is due at `due_ms`.
77    Probing {
78        sent: u8,
79        due_ms: u64,
80    },
81    /// The address is configured; `sent` announcements are out.
82    Announcing {
83        sent: u8,
84        due_ms: u64,
85    },
86    Bound,
87}
88
89struct Peer {
90    ip: [u8; 4],
91    mac: [u8; 6],
92    /// Last log time per [`PeerKind`].
93    logged_ms: [Option<u64>; 3],
94    /// Announced while we had no address to ask from; asked once we claim one.
95    verify: bool,
96}
97
98/// One interface's link-local claim.
99pub(crate) struct LinkLocal {
100    mac: [u8; 6],
101    rng: u64,
102    candidate: [u8; 4],
103    phase: Phase,
104    attempt: u32,
105    conflicts: u32,
106    last_conflict_ms: Option<u64>,
107    peers: Vec<Peer>,
108    events: VecDeque<Event>,
109}
110
111impl LinkLocal {
112    /// Start probing the MAC-derived candidate. The pseudo-random sequence for
113    /// later candidates and delays is seeded from `mac` too (§2.1), so it
114    /// repeats per host and differs between hosts.
115    pub(crate) fn new(mac: [u8; 6], now_ms: u64) -> Self {
116        let seed = mac.iter().fold(0u64, |acc, &b| (acc << 8) | u64::from(b));
117        let mut engine = Self {
118            mac,
119            rng: seed,
120            candidate: [0; 4],
121            phase: Phase::Bound,
122            attempt: 0,
123            conflicts: 0,
124            last_conflict_ms: None,
125            peers: Vec::new(),
126            events: VecDeque::new(),
127        };
128        engine.start_probing(first_candidate(mac), now_ms);
129        engine
130    }
131
132    /// The configured address (announcing or bound), if any.
133    pub(crate) fn claimed(&self) -> Option<[u8; 4]> {
134        (!matches!(self.phase, Phase::Probing { .. })).then_some(self.candidate)
135    }
136
137    /// Next event to apply, oldest first.
138    pub(crate) fn next_event(&mut self) -> Option<Event> {
139        self.events.pop_front()
140    }
141
142    /// Advance the probe/announce timers.
143    pub(crate) fn poll(&mut self, now_ms: u64) {
144        match self.phase {
145            Phase::Probing { sent, due_ms } if now_ms >= due_ms && sent < PROBE_NUM => {
146                self.send(ArpPacket::probe(self.mac, self.candidate));
147                let gap = if sent + 1 < PROBE_NUM {
148                    self.random(PROBE_MIN_MS, PROBE_MAX_MS)
149                } else {
150                    ANNOUNCE_WAIT_MS
151                };
152                self.phase = Phase::Probing {
153                    sent: sent + 1,
154                    due_ms: now_ms + gap,
155                };
156            }
157            Phase::Probing { due_ms, .. } if now_ms >= due_ms => {
158                self.events.push_back(Event::Claim(self.candidate));
159                self.announce_step(0, now_ms);
160                self.verify_pending();
161            }
162            Phase::Announcing { sent, due_ms } if now_ms >= due_ms => {
163                self.announce_step(sent, now_ms)
164            }
165            _ => {}
166        }
167    }
168
169    /// Account for one ARP packet from the wire (§2.2.1 probing conflicts,
170    /// §2.5 defense) and note link-local neighbours.
171    pub(crate) fn on_arp(&mut self, packet: ArpPacket, now_ms: u64) {
172        // QEMU's multicast netdev (and some switches) echo our own frames back;
173        // a packet from our own hardware address is never a conflict.
174        if packet.sender_mac == self.mac {
175            return;
176        }
177        let rival_probe = packet.is_probe() && packet.target_ip == self.candidate;
178        match self.phase {
179            Phase::Probing { .. } if packet.sender_ip == self.candidate || rival_probe => {
180                self.conflicts += 1;
181                let ip = self.candidate;
182                self.events.push_back(Event::Conflict {
183                    ip,
184                    mac: packet.sender_mac,
185                });
186                let next = self.next_candidate();
187                self.start_probing(next, now_ms);
188                return;
189            }
190            Phase::Announcing { .. } | Phase::Bound if packet.sender_ip == self.candidate => {
191                self.defend_or_yield(packet.sender_mac, now_ms);
192                return;
193            }
194            // smoltcp drops requests whose sender is 0.0.0.0, so the prober would
195            // never learn the address is taken: answer for it.
196            Phase::Announcing { .. } | Phase::Bound if rival_probe => {
197                self.send(ArpPacket::reply_to_probe(
198                    self.mac,
199                    self.candidate,
200                    packet.sender_mac,
201                ));
202            }
203            _ => {}
204        }
205        self.observe_peer(&packet, now_ms);
206    }
207
208    fn start_probing(&mut self, ip: [u8; 4], now_ms: u64) {
209        self.candidate = ip;
210        self.attempt += 1;
211        let wait = if self.conflicts >= MAX_CONFLICTS {
212            RATE_LIMIT_INTERVAL_MS
213        } else {
214            self.random(0, PROBE_WAIT_MS)
215        };
216        self.phase = Phase::Probing {
217            sent: 0,
218            due_ms: now_ms + wait,
219        };
220        self.events.push_back(Event::Probing {
221            ip,
222            attempt: self.attempt,
223        });
224    }
225
226    fn announce_step(&mut self, sent: u8, now_ms: u64) {
227        self.send(ArpPacket::announce(self.mac, self.candidate));
228        self.phase = if sent + 1 >= ANNOUNCE_NUM {
229            self.events.push_back(Event::Bound(self.candidate));
230            Phase::Bound
231        } else {
232            Phase::Announcing {
233                sent: sent + 1,
234                due_ms: now_ms + ANNOUNCE_INTERVAL_MS,
235            }
236        };
237    }
238
239    /// §2.5 (b): defend once; a second conflict within DEFEND_INTERVAL gives
240    /// the address up and restarts with a new candidate.
241    fn defend_or_yield(&mut self, rival: [u8; 6], now_ms: u64) {
242        let ip = self.candidate;
243        let recent = self
244            .last_conflict_ms
245            .is_some_and(|at| now_ms.saturating_sub(at) < DEFEND_INTERVAL_MS);
246        if recent {
247            self.events.push_back(Event::Release { ip, mac: rival });
248            self.conflicts += 1;
249            self.last_conflict_ms = None;
250            let next = self.next_candidate();
251            self.start_probing(next, now_ms);
252        } else {
253            self.last_conflict_ms = Some(now_ms);
254            self.send(ArpPacket::announce(self.mac, ip));
255            self.events.push_back(Event::Defended { ip, mac: rival });
256        }
257    }
258
259    /// Log a link-local neighbour, and ask an announcing one for its address:
260    /// its reply proves a live stack behind the address and fills our
261    /// neighbour cache before the first datagram needs it.
262    fn observe_peer(&mut self, packet: &ArpPacket, now_ms: u64) {
263        let ip = packet.sender_ip;
264        if ip[..2] != [169, 254] || ip == self.candidate {
265            return;
266        }
267        let claimed = self.claimed();
268        let kind = match packet.op {
269            // Only a reply addressed to us answers a request of ours. Hosts on
270            // a looping segment also answer their own announcements, and that
271            // overheard reply says nothing about a path back to us.
272            ArpOp::Reply if claimed == Some(packet.target_ip) && packet.target_mac == self.mac => {
273                PeerKind::Reply
274            }
275            ArpOp::Reply => return,
276            ArpOp::Request if packet.is_announcement() => PeerKind::Announce,
277            ArpOp::Request => PeerKind::Request,
278        };
279        let mac = packet.sender_mac;
280        let index = match self.peers.iter().position(|p| p.ip == ip && p.mac == mac) {
281            Some(index) => index,
282            None => {
283                if self.peers.len() >= PEER_SLOTS {
284                    self.peers.remove(0);
285                }
286                self.peers.push(Peer {
287                    ip,
288                    mac,
289                    logged_ms: [None; 3],
290                    verify: false,
291                });
292                self.peers.len() - 1
293            }
294        };
295        let peer = &mut self.peers[index];
296        let last = &mut peer.logged_ms[kind as usize];
297        if last.is_none_or(|at| now_ms.saturating_sub(at) >= PEER_LOG_INTERVAL_MS) {
298            *last = Some(now_ms);
299            self.events.push_back(Event::Peer { ip, mac, kind });
300        }
301        if kind == PeerKind::Announce {
302            match claimed {
303                Some(own) => self
304                    .events
305                    .push_back(Event::Send(ArpPacket::request(self.mac, own, ip))),
306                None => peer.verify = true,
307            }
308        }
309    }
310
311    fn verify_pending(&mut self) {
312        let (mac, own) = (self.mac, self.candidate);
313        for peer in self.peers.iter_mut().filter(|p| p.verify) {
314            peer.verify = false;
315            self.events
316                .push_back(Event::Send(ArpPacket::request(mac, own, peer.ip)));
317        }
318    }
319
320    fn send(&mut self, packet: ArpPacket) {
321        self.events.push_back(Event::Send(packet));
322    }
323
324    fn next_candidate(&mut self) -> [u8; 4] {
325        loop {
326            let r = self.next_u64();
327            let ip = [169, 254, 1 + (r % 254) as u8, (r >> 32) as u8];
328            if ip != self.candidate {
329                return ip;
330            }
331        }
332    }
333
334    /// Uniform-enough value in `lo..=hi`.
335    fn random(&mut self, lo: u64, hi: u64) -> u64 {
336        lo + self.next_u64() % (hi - lo + 1)
337    }
338
339    /// SplitMix64: tiny, and statistically fine for spreading candidates.
340    fn next_u64(&mut self) -> u64 {
341        self.rng = self.rng.wrapping_add(0x9E37_79B9_7F4A_7C15);
342        let mut z = self.rng;
343        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
344        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
345        z ^ (z >> 31)
346    }
