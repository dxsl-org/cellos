1//! CapId-keyed socket handle table for the net service cell.
2//!
3//! Maps kernel-issued `CapId`s (u64) to smoltcp `SocketHandle`s so that
4//! any consumer cell can reference an open socket across IPC calls without
5//! exposing smoltcp-internal handles.
6
7extern crate alloc;
8
9use crate::socket_state::SocketState;
10use alloc::collections::{BTreeMap, BTreeSet};
11use smoltcp::iface::SocketHandle;
12use types::ViError;
13
14/// 256 slow HTTP connections, one simultaneous fast probe and the listener;
15/// leave 32 slots for unrelated TCP/UDP/TLS consumers.
16pub const HTTP_OWNER_BUDGET: usize = 258;
17pub const OTHER_OWNER_RESERVE: usize = 32;
18pub const MAX_SOCKETS: usize = HTTP_OWNER_BUDGET + OTHER_OWNER_RESERVE;
19
20/// Socket-set storage includes management sockets (DHCP and DNS) which are
21/// driven directly, without consumer capabilities.
22pub const SOCKET_SET_STORAGE: usize = MAX_SOCKETS + 2;
23
24/// Attested owner of a socket capability: bound to CellId and cell generation.
25#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
26pub struct SocketOwner {
27    pub cell_id: u64,
28    pub generation: u64,
29}
30
31/// Maps a `CapId` to a smoltcp `SocketHandle` and connection state.
32#[derive(Default)]
33pub struct SocketTable {
34    entries: BTreeMap<u64, SocketHandle>,
35    states: BTreeMap<u64, SocketState>,
36    owners: BTreeMap<u64, SocketOwner>,
37    /// Bound listen port per cap — needed so ACCEPT can renew the listener on
38    /// the same port after the original socket transitions to Established.
39    listen_ports: BTreeMap<u64, u16>,
40    /// Tracks which caps hold UDP sockets so TCP-only opcodes can reject them
41    /// before calling `sockets.get_mut::<tcp::Socket>()`, which panics on a
42    /// wrong-type handle.
43    udp_caps: BTreeSet<u64>,
44    graceful_closes: BTreeMap<u64, u64>,
45    next_cap: u64,
46}
47
48impl SocketTable {
49    pub fn new() -> Self {
50        Self {
51            entries: BTreeMap::new(),
52            states: BTreeMap::new(),
53            owners: BTreeMap::new(),
54            listen_ports: BTreeMap::new(),
55            udp_caps: BTreeSet::new(),
56            graceful_closes: BTreeMap::new(),
57            next_cap: 1,
58        }
59    }
60
61    /// One owner cannot consume the capacity reserved for other services.
62    /// A refused insertion never allocates a cap or mutates the socket table.
63    pub fn can_insert(&self, owner: SocketOwner) -> bool {
64        self.entries.len() < MAX_SOCKETS
65            && self.next_cap <= u32::MAX as u64
66            && self.owners.values().filter(|candidate| **candidate == owner).count()
67                < HTTP_OWNER_BUDGET
68    }
69
70    /// Allocate a new `CapId` and associate it with `handle` and attested `owner`.
71    ///
72    /// # Errors
73    /// Returns `ViError::OutOfMemory` if `MAX_SOCKETS` is already reached.
74    pub fn insert(&mut self, handle: SocketHandle, owner: SocketOwner) -> Result<u64, ViError> {
75        if !self.can_insert(owner) {
76            return Err(ViError::OutOfMemory);
77        }
78        let cap = self.next_cap;
79        self.next_cap += 1;
80        self.entries.insert(cap, handle);
81        self.states.insert(cap, SocketState::Created);
82        self.owners.insert(cap, owner);
83        Ok(cap)
84    }
85
86    /// Allocate a new `CapId` for `handle` with an explicit initial state and attested owner.
87    ///
88    /// Unlike `insert` (which defaults to `Created`), this sets `state` directly.
89    /// Only call from ACCEPT — the handle is already Established and must be
90    /// surfaced as `Connected` to the consumer.
91    ///
92    /// # Errors
93    /// Returns `ViError::OutOfMemory` if `MAX_SOCKETS` is already reached.
94    pub fn insert_with_state(
95        &mut self,
96        handle: SocketHandle,
97        state: SocketState,
98        owner: SocketOwner,
99    ) -> Result<u64, ViError> {
100        if !self.can_insert(owner) {
101            return Err(ViError::OutOfMemory);
102        }
103        let cap = self.next_cap;
104        self.next_cap += 1;
105        self.entries.insert(cap, handle);
106        self.states.insert(cap, state);
107        self.owners.insert(cap, owner);
108        Ok(cap)
109    }
110
111    /// Check whether `caller` is the recorded owner of `cap`.
112    pub fn is_owner(&self, cap: u64, caller: SocketOwner) -> bool {
113        self.owners.get(&cap).copied() == Some(caller)
114    }
115
116    /// Look up the smoltcp `SocketHandle` for a given `CapId`, checking caller ownership.
117    pub fn get(&self, cap: u64, caller: SocketOwner) -> Option<SocketHandle> {
118        if self.is_owner(cap, caller) {
119            self.entries.get(&cap).copied()
120        } else {
121            None
122        }
123    }
124
125    /// Internal lifecycle inspection for a cap already in the close ledger.
126    pub fn get_unchecked(&self, cap: u64) -> Option<SocketHandle> {
127        self.entries.get(&cap).copied()
128    }
129
130    /// Read the connection state for `cap`, checking caller ownership.
131    pub fn get_state(&self, cap: u64, caller: SocketOwner) -> Option<SocketState> {
132        if self.is_owner(cap, caller) {
133            self.states.get(&cap).copied()
134        } else {
135            None
136        }
137    }
138
139    /// Update the connection state for `cap`.
140    pub fn set_state(&mut self, cap: u64, state: SocketState) {
141        if self.entries.contains_key(&cap) {
142            self.states.insert(cap, state);
143        }
144    }
145
146    /// Record the port a listening socket is bound to, so ACCEPT can renew it.
147    pub fn set_listen_port(&mut self, cap: u64, port: u16) {
148        if self.entries.contains_key(&cap) {
149            self.listen_ports.insert(cap, port);
150        }
151    }
152
153    /// Read the bound listen port for `cap`, checking caller ownership.
154    pub fn get_listen_port(&self, cap: u64, caller: SocketOwner) -> Option<u16> {
155        if self.is_owner(cap, caller) {
156            self.listen_ports.get(&cap).copied()
157        } else {
158            None
159        }
160    }
161    /// Repoint an existing cap at a new smoltcp handle.
162    ///
163    /// Used by ACCEPT to swap the exhausted listening handle for a fresh one
164    /// without changing the cap the consumer holds.
165    pub fn update_handle(&mut self, cap: u64, new_handle: SocketHandle) {
166        if self.entries.contains_key(&cap) {
167            self.entries.insert(cap, new_handle);
168        }
169    }
170
171    /// Mark a cap as holding a UDP socket.
172    ///
173    /// TCP-only opcodes (CONNECT, SEND, RECV, etc.) check this to avoid calling
174    /// `sockets.get_mut::<tcp::Socket>()`, which panics on a wrong-type handle.
175    pub fn mark_udp(&mut self, cap: u64) {
176        self.udp_caps.insert(cap);
177    }
178
179    /// Returns `true` if `cap` holds a UDP socket owned by `caller`.
180    pub fn is_udp(&self, cap: u64, caller: SocketOwner) -> bool {
181        self.is_owner(cap, caller) && self.udp_caps.contains(&cap)
182    }
183
184    /// Queue FIN once and retain the first drain deadline.
185    pub fn begin_graceful_close(&mut self, cap: u64, deadline_ticks: u64) {
186        self.graceful_closes.entry(cap).or_insert(deadline_ticks);
187    }
188
189    pub fn is_graceful_closing(&self, cap: u64) -> bool {
190        self.graceful_closes.contains_key(&cap)
191    }
192
193    pub fn graceful_closes(&self) -> impl Iterator<Item = (u64, u64)> + '_ {
194        self.graceful_closes.iter().map(|(&cap, &deadline)| (cap, deadline))
195    }
196
197    pub fn owned_cap(&self, owner: SocketOwner) -> Option<u64> {
198        self.owners.iter().find_map(|(&cap, &candidate)| {
199            (candidate == owner).then_some(cap)
200        })
201    }
202
203    /// Root cell death invalidates every cap of the exact owner generation.
204    pub fn remove_owner(&mut self, owner: SocketOwner, sockets: &mut smoltcp::iface::SocketSet<'_>) {
205        while let Some(cap) = self.owned_cap(owner) {
206            if let Some(handle) = self.remove_internal(cap) {
207                sockets.remove(handle);
208            }
209        }
210    }
211
212    /// Remove a socket from the table if requested by its registered owner.
213    pub fn remove(&mut self, cap: u64, caller: SocketOwner) -> Option<SocketHandle> {
214        if !self.is_owner(cap, caller) {
215            return None;
216        }
217        self.remove_internal(cap)
218    }
219
220    /// Remove a socket unconditionally (for internal error cleanup within the net cell).
221    pub fn remove_internal(&mut self, cap: u64) -> Option<SocketHandle> {
222        self.owners.remove(&cap);
223        self.states.remove(&cap);
224        self.listen_ports.remove(&cap);
225        self.udp_caps.remove(&cap);
226        self.graceful_closes.remove(&cap);
227        self.entries.remove(&cap)
228    }
229
230    #[cfg(test)]
231    pub(crate) fn next_cap_for_test(&self) -> u64 {
232        self.next_cap
233    }
234}
