1//! CapId-keyed socket handle table for the net service cell.
2//!
3//! Maps kernel-issued `CapId`s (u64) to smoltcp `SocketHandle`s so that
4//! any consumer cell can reference an open socket across IPC calls without
5//! exposing smoltcp-internal handles.
6
7extern crate alloc;
8
9use crate::socket_state::SocketState;
10use alloc::collections::{BTreeMap, BTreeSet};
11use smoltcp::iface::SocketHandle;
12use types::ViError;
13
14/// Maximum simultaneous sockets (including the DHCP management socket).
15pub const MAX_SOCKETS: usize = 18; // 16 user + 1 DHCP + 1 ARP
16
17/// Attested owner of a socket capability: bound to CellId and cell generation.
18#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
19pub struct SocketOwner {
20    pub cell_id: u64,
21    pub generation: u64,
22}
23
24/// Maps a `CapId` to a smoltcp `SocketHandle` and connection state.
25#[derive(Default)]
26pub struct SocketTable {
27    entries: BTreeMap<u64, SocketHandle>,
28    states: BTreeMap<u64, SocketState>,
29    owners: BTreeMap<u64, SocketOwner>,
30    /// Bound listen port per cap — needed so ACCEPT can renew the listener on
31    /// the same port after the original socket transitions to Established.
32    listen_ports: BTreeMap<u64, u16>,
33    /// Tracks which caps hold UDP sockets so TCP-only opcodes can reject them
34    /// before calling `sockets.get_mut::<tcp::Socket>()`, which panics on a
35    /// wrong-type handle.
36    udp_caps: BTreeSet<u64>,
37    next_cap: u64,
38}
39
40impl SocketTable {
41    pub fn new() -> Self {
42        Self {
43            entries: BTreeMap::new(),
44            states: BTreeMap::new(),
45            owners: BTreeMap::new(),
46            listen_ports: BTreeMap::new(),
47            udp_caps: BTreeSet::new(),
48            next_cap: 1,
49        }
50    }
51
52    /// Allocate a new `CapId` and associate it with `handle` and attested `owner`.
53    ///
54    /// # Errors
55    /// Returns `ViError::OutOfMemory` if `MAX_SOCKETS` is already reached.
56    pub fn insert(&mut self, handle: SocketHandle, owner: SocketOwner) -> Result<u64, ViError> {
57        if self.entries.len() >= MAX_SOCKETS {
58            return Err(ViError::OutOfMemory);
59        }
60        let cap = self.next_cap;
61        self.next_cap += 1;
62        self.entries.insert(cap, handle);
63        self.states.insert(cap, SocketState::Created);
64        self.owners.insert(cap, owner);
65        Ok(cap)
66    }
67
68    /// Allocate a new `CapId` for `handle` with an explicit initial state and attested owner.
69    ///
70    /// Unlike `insert` (which defaults to `Created`), this sets `state` directly.
71    /// Only call from ACCEPT — the handle is already Established and must be
72    /// surfaced as `Connected` to the consumer.
73    ///
74    /// # Errors
75    /// Returns `ViError::OutOfMemory` if `MAX_SOCKETS` is already reached.
76    pub fn insert_with_state(
77        &mut self,
78        handle: SocketHandle,
79        state: SocketState,
80        owner: SocketOwner,
81    ) -> Result<u64, ViError> {
82        if self.entries.len() >= MAX_SOCKETS {
83            return Err(ViError::OutOfMemory);
84        }
85        let cap = self.next_cap;
86        self.next_cap += 1;
87        self.entries.insert(cap, handle);
88        self.states.insert(cap, state);
89        self.owners.insert(cap, owner);
90        Ok(cap)
91    }
92
93    /// Check whether `caller` is the recorded owner of `cap`.
94    pub fn is_owner(&self, cap: u64, caller: SocketOwner) -> bool {
95        self.owners.get(&cap).copied() == Some(caller)
96    }
97
98    /// Look up the smoltcp `SocketHandle` for a given `CapId`, checking caller ownership.
99    pub fn get(&self, cap: u64, caller: SocketOwner) -> Option<SocketHandle> {
100        if self.is_owner(cap, caller) {
101            self.entries.get(&cap).copied()
102        } else {
103            None
104        }
105    }
106
107    /// Read the connection state for `cap`, checking caller ownership.
108    pub fn get_state(&self, cap: u64, caller: SocketOwner) -> Option<SocketState> {
109        if self.is_owner(cap, caller) {
110            self.states.get(&cap).copied()
111        } else {
112            None
113        }
114    }
115
116    /// Update the connection state for `cap`.
117    pub fn set_state(&mut self, cap: u64, state: SocketState) {
118        if self.entries.contains_key(&cap) {
119            self.states.insert(cap, state);
120        }
121    }
122
123    /// Record the port a listening socket is bound to, so ACCEPT can renew it.
124    pub fn set_listen_port(&mut self, cap: u64, port: u16) {
125        if self.entries.contains_key(&cap) {
126            self.listen_ports.insert(cap, port);
127        }
128    }
129
130    /// Read the bound listen port for `cap`, checking caller ownership.
131    pub fn get_listen_port(&self, cap: u64, caller: SocketOwner) -> Option<u16> {
132        if self.is_owner(cap, caller) {
133            self.listen_ports.get(&cap).copied()
134        } else {
135            None
136        }
137    }
138    /// Repoint an existing cap at a new smoltcp handle.
139    ///
140    /// Used by ACCEPT to swap the exhausted listening handle for a fresh one
141    /// without changing the cap the consumer holds.
142    pub fn update_handle(&mut self, cap: u64, new_handle: SocketHandle) {
143        if self.entries.contains_key(&cap) {
144            self.entries.insert(cap, new_handle);
145        }
146    }
147
148    /// Mark a cap as holding a UDP socket.
149    ///
150    /// TCP-only opcodes (CONNECT, SEND, RECV, etc.) check this to avoid calling
151    /// `sockets.get_mut::<tcp::Socket>()`, which panics on a wrong-type handle.
152    pub fn mark_udp(&mut self, cap: u64) {
153        self.udp_caps.insert(cap);
154    }
155
156    /// Returns `true` if `cap` holds a UDP socket owned by `caller`.
157    pub fn is_udp(&self, cap: u64, caller: SocketOwner) -> bool {
158        self.is_owner(cap, caller) && self.udp_caps.contains(&cap)
159    }
160
161    /// Remove a socket from the table if requested by its registered owner.
162    pub fn remove(&mut self, cap: u64, caller: SocketOwner) -> Option<SocketHandle> {
163        if !self.is_owner(cap, caller) {
164            return None;
165        }
166        self.remove_internal(cap)
167    }
168
169    /// Remove a socket unconditionally (for internal error cleanup within the net cell).
170    pub fn remove_internal(&mut self, cap: u64) -> Option<SocketHandle> {
171        self.owners.remove(&cap);
172        self.states.remove(&cap);
173        self.listen_ports.remove(&cap);
174        self.udp_caps.remove(&cap);
175        self.entries.remove(&cap)
176    }
177
178    #[cfg(test)]
179    pub(crate) fn next_cap_for_test(&self) -> u64 {
180        self.next_cap
181    }
182}
