1//! Boot entropy for the smoltcp stack: the interface RNG seed and the
2//! ephemeral-port cursor.
3//!
4//! smoltcp draws TCP initial sequence numbers, DNS transaction ids, DNS source
5//! ports and IPv4 ids from `Config::random_seed`. Left at its default of 0,
6//! every boot replays the same sequence and an off-path peer can predict them
7//! (spoofed connections, RST/data injection, DNS poisoning). Connect-side source
8//! ports were a plain counter from 49152 for the same reason.
9//!
10//! Both come from [`ViRng`], the same fail-closed source TLS uses: a kernel that
11//! serves no entropy stops the service here rather than running a predictable
12//! stack. A development kernel built with `dev-weak-rng` still serves bytes.
13
14use core::sync::atomic::{AtomicU16, Ordering};
15
16use ostd::sync::Mutex;
17use rand_core::RngCore;
18use smoltcp::{iface::Config, wire::HardwareAddress};
19
20use crate::tls::rng::ViRng;
21
22/// IANA dynamic-port range (RFC 6335), inclusive.
23const EPHEMERAL_FIRST: u16 = 49152;
24const EPHEMERAL_SPAN: u32 = (u16::MAX - EPHEMERAL_FIRST) as u32 + 1;
25
26/// Largest single cursor advance (RFC 6056 §3.3.5, random increments). Small
27/// enough that a port is not handed out again until ~500 allocations later,
28/// large enough that one observed port does not name the next.
29const MAX_PORT_STEP: u32 = 64;
30
31/// Port stream drawn after the interface seed; `None` until [`seed_stack`] runs.
32static PORT_RNG: Mutex<Option<ViRng>> = Mutex::new(None);
33/// Next ephemeral port. Single-threaded service, so load/store is not racing.
34static NEXT_PORT: AtomicU16 = AtomicU16::new(EPHEMERAL_FIRST);
35
36/// Draw the smoltcp `Config::random_seed` and start the ephemeral-port cursor at
37/// a random point. Call once, before the interface exists.
38///
39/// # Panics
40/// When the kernel serves no entropy (see [`ViRng::new`]).
41fn seed_stack() -> u64 {
42    let mut rng = ViRng::new();
43    let seed = rng.next_u64();
44    NEXT_PORT.store(
45        advance_port(EPHEMERAL_FIRST, rng.next_u32()),
46        Ordering::Relaxed,
47    );
48    *PORT_RNG.lock() = Some(rng);
49    seed
50}
51
52/// The smoltcp interface configuration the service runs with: `mac`, and a
53/// seed drawn by [`seed_stack`].
54///
55/// # Panics
56/// When the kernel serves no entropy (see [`ViRng::new`]).
57pub(crate) fn seeded_config(mac: HardwareAddress) -> Config {
58    let mut config = Config::new(mac);
59    config.random_seed = seed_stack();
60    config
61}
62
63/// Hand out the next connect-side source port.
64pub(crate) fn next_ephemeral_port() -> u16 {
65    let draw = PORT_RNG.lock().as_mut().map(|rng| rng.next_u32());
66    let port = NEXT_PORT.load(Ordering::Relaxed);
67    NEXT_PORT.store(advance_port(port, port_step(draw)), Ordering::Relaxed);
68    port
69}
70
71/// Cursor advance for one allocation: 1..=[`MAX_PORT_STEP`] from the port
72/// stream, or 1 before the stack is seeded (host tests only — `run()` seeds
73/// before any socket exists).
74fn port_step(draw: Option<u32>) -> u32 {
75    draw.map_or(1, |draw| 1 + draw % MAX_PORT_STEP)
76}
77
78/// `port + step`, wrapped inside the dynamic range. A port below the range
79/// (never stored, but cheap to rule out) is treated as its first port.
80fn advance_port(port: u16, step: u32) -> u16 {
81    let offset = u32::from(port.saturating_sub(EPHEMERAL_FIRST));
82    // Both terms are below EPHEMERAL_SPAN, so the wrapped sum fits in u16.
83    EPHEMERAL_FIRST + ((offset + step % EPHEMERAL_SPAN) % EPHEMERAL_SPAN) as u16
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89    use smoltcp::wire::EthernetAddress;
90
91    #[test]
92    fn service_interface_config_is_seeded() {
93        // Regression: the interface was built with `Config::new` alone, whose
94        // `random_seed` is 0 on every boot.
95        let mac = HardwareAddress::Ethernet(EthernetAddress([0x52, 0x54, 0, 0x12, 0x34, 0x56]));
96        assert_ne!(Config::new(mac).random_seed, seeded_config(mac).random_seed);
97    }
98
99    #[test]
100    fn advance_stays_inside_the_dynamic_range() {
101        assert_eq!(advance_port(EPHEMERAL_FIRST, 0), EPHEMERAL_FIRST);
102        assert_eq!(advance_port(u16::MAX, 1), EPHEMERAL_FIRST);
103        assert_eq!(advance_port(u16::MAX - 1, 1), u16::MAX);
104        assert_eq!(
105            advance_port(u16::MAX, EPHEMERAL_SPAN + 2),
106            EPHEMERAL_FIRST + 1
107        );
108        assert_eq!(advance_port(1024, 3), EPHEMERAL_FIRST + 3);
109    }
110
111    #[test]
112    fn steps_are_bounded_random_increments() {
113        assert_eq!(port_step(None), 1);
114        assert_eq!(port_step(Some(0)), 1);
115        assert_eq!(port_step(Some(MAX_PORT_STEP - 1)), MAX_PORT_STEP);
116        assert_eq!(port_step(Some(u32::MAX)), 1 + u32::MAX % MAX_PORT_STEP);
117    }
118}
