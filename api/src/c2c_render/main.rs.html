88fn options(args: &[String]) -> Result<Options, String> {
89    let mut result = Options {
90        mode: Mode::Compare,
91        config: RenderConfig {
92            width: 256,
93            height: 192,
94            samples: 64,
95            max_bounces: 8,
96            seed: 1,
97        },
98        tile_size: 16,
99        workers: 1,
100        output: String::from("/tmp/c2c-render.ppm"),
101    };
102    let mut index = 0;
103    if let Some(first) = args.first().filter(|arg| !arg.starts_with('-')) {
104        result.mode = match first.as_str() {
105            "baseline" => Mode::Baseline,
106            "copy" => Mode::Copy,
107            "shared" => Mode::Shared,
108            "compare" => Mode::Compare,
109            "tier2" => Mode::Tier2,
110            "compare-tier2" => Mode::CompareTier2,
111            "remote" | "tier3" => return Err(String::from(
112                "remote/Tier3 rendering is NotSupported; no remote readiness or transport bypass",
113            )),
114            _ => return Err(format!("unknown mode: {}", first)),
115        };
116        index = 1;
117    }
118    while index < args.len() {
119        let flag = args[index].as_str();
120        let value = args
121            .get(index + 1)
122            .ok_or_else(|| format!("missing value for {}", flag))?;
123        let number = || {
124            value
125                .parse::<u32>()
126                .map_err(|_| format!("invalid integer for {}: {}", flag, value))
127        };
128        match flag {
129            "--width" => result.config.width = number()?,
130            "--height" => result.config.height = number()?,
131            "--samples" => result.config.samples = number()?,
132            "--depth" => result.config.max_bounces = number()?,
133            "--seed" => {
134                result.config.seed = value
135                    .parse::<u64>()
136                    .map_err(|_| String::from("invalid seed"))?
137            }
138            "--tile-size" => result.tile_size = number()?,
139            "--workers" => result.workers = number()?,
140            "--output" => result.output = value.clone(),
141            _ => return Err(format!("unknown option: {}", flag)),
142        }
143        index += 2;
144    }
145    result
146        .config
147        .validate()
148        .map_err(|error| format!("invalid render configuration: {:?}", error))?;
149    let bytes = result
150        .config
151        .rgb_bytes()
152        .map_err(|error| format!("image size: {:?}", error))?;
153    if bytes > MAX_IMAGE_BYTES {
154        return Err(String::from(
155            "RGB image exceeds 4 MiB application memory limit",
156        ));
157    }
158    if result.tile_size == 0 || result.tile_size > 18 {
159        return Err(String::from(
160            "tile-size must be 1..18 to keep copied RGB tiles <=1024 bytes",
161        ));
162    }
163    if !(1..=4).contains(&result.workers) {
164        return Err(String::from("workers must be 1..4"));
165    }
166    let count = result
167        .config
168        .width
169        .div_ceil(result.tile_size)
170        .checked_mul(result.config.height.div_ceil(result.tile_size))
171        .ok_or_else(|| String::from("tile count overflow"))?;
172    if count > MAX_TILES {
173        return Err(String::from("more than 65536 tiles; increase tile-size"));
174    }
175    if !result.output.starts_with('/') || result.output.len() > 96 {
176        return Err(String::from(
177            "output must be an absolute VFS path <=96 bytes",
178        ));
179    }
180    Ok(result)
181}

183struct Worker {
184    tid: usize,
185    active: bool,
186}
187
188impl Worker {
189    fn spawn(scene_hash: u64, path: &'static str) -> Result<Self, String> {
190        if !sys_set_spawn_args("") {
191            return Err(String::from("could not clear worker spawn arguments"));
192        }
193        let tid = match sys_spawn_from_path(path) {
194            SyscallResult::Ok(tid) if tid > 0 => tid,
195            result => return Err(format!("spawn {} failed: {:?}", path, result)),
196        };
197        let worker = Self { tid, active: true };
198        match worker.call(&Request::Ready).map(|(response, _)| response) {
199            Ok(Response::Ready { scene_hash: actual }) if actual == scene_hash => Ok(worker),
200            result => Err(format!(
201                "worker startup/fingerprint validation failed: {:?}",
202                result
203            )),
204        }
205    }
206
207    fn call(&self, request: &Request) -> Result<(Response, usize), String> {
208        let mut send = [0u8; wire::ipc::IPC_BUF_SIZE];
209        let mut receive = [0u8; wire::ipc::IPC_BUF_SIZE];
210        // Request admission and completion belong to the exact returned worker
211        // TID and one kernel operation, never a global service or arbitrary recv.
212        let encoded = wire::ipc::encode(request, &mut send)
213            .map_err(|_| String::from("request exceeds IPC frame"))?;
214        let request_bytes = encoded.len();
215        let call = ostd::ipc::PendingCall::submit(self.tid, encoded)
216            .map_err(|error| format!("worker {} admission failed: {:?}", self.tid, error))?;
217        let completion = match call.wait_and_take(&mut receive, 0, 1) {
218            Ok(Some(completion)) => completion,
219            result => {
220                let _ = call.cancel();
221                let _ = call.try_take(&mut receive);
222                return Err(format!(
223                    "worker {} operation failed: {:?}",
224                    self.tid, result
225                ));
226            }
227        };
228        if completion.terminal != ostd::ipc::IpcTerminal::Reply {
229            return Err(format!(
230                "worker {} operation terminal: {:?}",
231                self.tid, completion.terminal
232            ));
233        }
234        let raw = &receive[..completion.len];
235        let response =
236            wire::ipc::decode(raw).map_err(|_| String::from("invalid worker response encoding"))?;
237        Ok((response, request_bytes + raw.len()))
238    }
239
240    fn require_shared_refusal(&self, config: RenderConfig, scene_hash: u64) -> Result<(), String> {
241        // A valid one-pixel request with no real grant tests the worker boundary,
242        // not grant allocation or malformed geometry. This is setup traffic.
243        let request = Request::Render {
244            config,
245            tile: Tile {
246                id: 0,
247                x: 0,
248                y: 0,
249                width: 1,
250                height: 1,
251            },
252            scene_hash,
253            output: Output::Shared {
254                grant_id: u64::MAX,
255                bytes: 3,
256            },
257        };
258        match self.call(&request).map(|(response, _)| response) {
259            Ok(Response::Error(RenderError::InvalidOutput)) => {
260                ostd::io::println("c2c-render: TIER2 SHARED OUTPUT REFUSED");
261                Ok(())
262            }
263            result => Err(format!("Tier2 Shared output refusal failed: {:?}", result)),
264        }
265    }
266
267    fn terminate(&mut self) -> bool {
268        if !self.active {
269            return true;
270        }
271        if matches!(sys_force_exit(self.tid), SyscallResult::Ok(_)) {
272            // ForceExit itself is nonblocking. Wait is published by the kernel
273            // only after saved contexts have stopped executing on every hart.
274            // The forced-exit sentinel is mapped to Err by ostd, not a failure
275            // of this quiescence barrier for our known, successfully killed child.
276            let _ = sys_wait(self.tid);
277            self.active = false;
278            true
279        } else {
280            false
281        }
282    }
283
284    fn stop(&mut self) -> Result<(), String> {
285        match self.call(&Request::Stop) {
286            Ok((Response::Stopped, _)) => {
287                let result = sys_wait(self.tid);
288                self.active = false;
289                match result {
290                    SyscallResult::Ok(0) => Ok(()),
291                    // This exact child acknowledged Stop and retains no grant.
292                    // With Wait admitted by our manifest, the kernel's only
293                    // error here means the child has already been quiescently
294                    // reaped (syscall.rs Wait's InvalidDriverId branch).
295                    SyscallResult::Err(ostd::syscall::SyscallError::Unknown) => Ok(()),
296                    _ => Err(format!("worker clean exit failed: {:?}", result)),
297                }
298            }
299            result => {
300                let _ = self.terminate();
301                Err(format!("worker stop failed: {:?}", result))
302            }
303        }
304    }
305}
306
307impl Drop for Worker {
308    fn drop(&mut self) {
309        if self.active && !self.terminate() {
310            ostd::io::println("c2c-render: ERROR cannot terminate worker; coordinator retained alive to preserve ownership");
311            loop {
312                sys_yield();
313            }
314        }
315    }
316}

325struct Slot {
326    worker: Worker,
327    grant: Option<GrantHandle<u8>>,
328    pending: Option<ActiveTile>,
329    send: [u8; wire::ipc::IPC_BUF_SIZE],
330    receive: [u8; wire::ipc::IPC_BUF_SIZE],
331    completed: u32,
332}
333
334impl Slot {
335    fn submit(
336        &mut self,
337        config: RenderConfig,
338        tile: Tile,
339        scene_hash: u64,
340        clock: Clock,
341    ) -> Result<(), String> {
342        let output = match self.grant.as_ref() {
343            Some(grant) => Output::Shared {
344                grant_id: grant.id() as u64,
345                bytes: tile
346                    .rgb_bytes()
347                    .map_err(|error| format!("tile bytes: {:?}", error))?
348                    as u32,
349            },
350            None => Output::Copied,
351        };
352        let request = Request::Render {
353            config,
354            tile,
355            scene_hash,
356            output,
357        };
358        let encoded = wire::ipc::encode(&request, &mut self.send)
359            .map_err(|_| String::from("request exceeds IPC frame"))?;
360        let request_bytes = encoded.len();
361        let started_ns = clock.now_ns();
362        let call = ostd::ipc::PendingCall::submit(self.worker.tid, encoded)
363            .map_err(|error| format!("worker {} admission failed: {:?}", self.worker.tid, error))?;
364        // Attach every accepted operation before doing anything else fallible.
365        self.pending = Some(ActiveTile {
366            call,
367            tile,
368            started_ns,
369            request_bytes,
370        });
371        Ok(())
372    }
373}

375struct Pool {
376    slots: Vec<Slot>,
377}
378
379impl Pool {
380    fn start(mode: Mode, opts: &Options, scene_hash: u64, tiles: u32) -> Result<Self, String> {
381        let count = if mode == Mode::Baseline {
382            0
383        } else {
384            opts.workers.min(tiles)
385        };
386        let mut pool = Self {
387            slots: Vec::with_capacity(count as usize),
388        };
389        let path = if mode == Mode::Tier2 {
390            "/bin/c2c-render-domain-worker"
391        } else {
392            "/bin/c2c-render-worker"
393        };
394        for _ in 0..count {
395            let worker = Worker::spawn(scene_hash, path)?;
396            // Pool ownership precedes allocation/sharing and every later failure.
397            pool.slots.push(Slot {
398                worker,
399                grant: None,
400                pending: None,
401                send: [0; wire::ipc::IPC_BUF_SIZE],
402                receive: [0; wire::ipc::IPC_BUF_SIZE],
403                completed: 0,
404            });
405            let slot = pool.slots.last_mut().unwrap();
406            if mode == Mode::Tier2 && opts.mode == Mode::CompareTier2 {
407                slot.worker
408                    .require_shared_refusal(opts.config, scene_hash)?;
409            }
410            if mode == Mode::Shared {
411                slot.grant = Some(
412                    GrantHandle::<u8>::alloc(COPIED_TILE_MAX_BYTES)
413                        .ok_or_else(|| String::from("tile grant allocation failed"))?,
414                );
415                if !ostd::syscall::sys_grant_share(
416                    slot.grant.as_ref().unwrap().id(),
417                    slot.worker.tid,
418                    2,
419                ) {
420                    return Err(String::from("sharing tile grant with worker failed"));
421                }
422            }
423        }
424        Ok(pool)
425    }
426
427    fn finish(&mut self) -> Result<(), String> {
428        for slot in &mut self.slots {
429            slot.worker.stop()?;
430        }
431        Ok(())
432    }
433}
434
435impl Drop for Pool {
436    fn drop(&mut self) {
437        let mut drained = true;
438        // Cancellation settles tokens, NOT execution. Retain all grants until
439        // every potentially writing child has crossed its quiescence barrier.
440        for slot in &mut self.slots {
441            if let Some(active) = slot.pending.as_ref() {
442                let _ = active.call.cancel();
443                match active.call.try_take(&mut slot.receive) {
444                    Ok(Some(_)) | Err(ostd::ipc::IpcError::InvalidOperation) => {}
445                    _ => drained = false,
446                }
447            }
448        }
449        let mut quiescent = true;
450        for slot in &mut self.slots {
451            // Do not short circuit: attempt termination for ALL holders.
452            if !slot.worker.terminate() {
453                quiescent = false;
454            }
455        }
456        if !quiescent || !drained {
457            ostd::io::println("c2c-render: ERROR pool cleanup incomplete; coordinator retained alive to preserve all worker/grant/operation ownership");
458            loop {
459                sys_yield();
460            }
461        }
462        // Only now may field destruction release grants and operation handles.
463    }
464}

466/// Largest number of distinct processor ids one render can observe: the x86
467/// kernel bound `MAX_HARTS` (= the HAL's `MAX_CPUS`, 64). Narrower native
468/// targets observe fewer and stay within it.
469const MAX_PROCESSOR_IDS: usize = 64;
470
471/// Distinct hardware endpoint observations, bounded by [`MAX_PROCESSOR_IDS`].
472#[derive(Clone, Copy)]
473struct ProcessorIds {
474    ids: [Option<u32>; MAX_PROCESSOR_IDS],
475}
476
477impl Default for ProcessorIds {
478    fn default() -> Self {
479        Self {
480            ids: [None; MAX_PROCESSOR_IDS],
481        }
482    }
483}
484
485impl ProcessorIds {
486    fn observe(&mut self, id: Option<u32>) -> Result<(), String> {
487        let Some(id) = id else {
488            return Ok(());
489        };
490        if self.ids.contains(&Some(id)) {
491            return Ok(());
492        }
493        let empty = self
494            .ids
495            .iter_mut()
496            .find(|slot| slot.is_none())
497            .ok_or_else(|| {
498                format!(
499                    "processor observations exceed the {}-CPU kernel bound",
500                    MAX_PROCESSOR_IDS
501                )
502            })?;
503        *empty = Some(id);
504        self.ids.sort_unstable();
505        Ok(())
506    }
507}
508
509impl core::fmt::Display for ProcessorIds {
510    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
511        let mut separator = "";
512        for id in self.ids.iter().flatten() {
513            write!(f, "{}{}", separator, id)?;
514            separator = ";";
515        }
516        if separator.is_empty() {
517            f.write_str("-")?;
518        }
519        Ok(())
520    }
521}

546/// Validated `Response::Done` fields: stats, compute ns, borrowed pixels, and the
547/// processor IDs observed before/after rendering.
548type ValidatedDone<'a> = (RenderStats, u64, &'a [u8], Option<u32>, Option<u32>);
549
550fn validate_done(
551    raw: &[u8],
552    tile: Tile,
553    config: RenderConfig,
554    scene_hash: u64,
555    shared: bool,
556) -> Result<ValidatedDone<'_>, String> {
557    // Postcard encodes Response::Done as discriminant 0 followed by these
558    // fields. Vec<u8>'s length-prefixed byte sequence can be borrowed directly:
559    // copied pixels stay in this slot's reusable IPC buffer, without allocating
560    // or copying another tile Vec. Tests encode the actual protocol Response.
561    let (
562        (
563            variant,
564            tile_id,
565            actual,
566            stats,
567            compute_ns,
568            pixels,
569            processor_id_before,
570            processor_id_after,
571        ),
572        remaining,
573    ) = postcard::take_from_bytes::<(
574        u32,
575        u32,
576        u64,
577        RenderStats,
578        u64,
579        &[u8],
580        Option<u32>,
581        Option<u32>,
582    )>(raw)
583    .map_err(|_| String::from("invalid worker Done response encoding"))?;
584    let samples = u64::from(tile.width) * u64::from(tile.height) * u64::from(config.samples);
585    let rays_limit = samples
586        .checked_mul(u64::from(config.max_bounces))
587        .and_then(|value| value.checked_mul(2))
588        .ok_or_else(|| String::from("ray bound overflow"))?;
589    let bytes = tile
590        .rgb_bytes()
591        .map_err(|error| format!("tile bytes: {:?}", error))?;
592    if variant != 0
593        || !remaining.is_empty()
594        || tile_id != tile.id
595        || actual != scene_hash
596        || stats.samples != samples
597        || stats.rays < samples
598        || stats.rays > rays_limit
599        || (shared && !pixels.is_empty())
600        || (!shared && pixels.len() != bytes)
601    {
602        return Err(String::from(
603            "worker response ID/fingerprint/stats/pixel-length mismatch",
604        ));
605    }
606    Ok((
607        stats,
608        compute_ns,
609        pixels,
610        processor_id_before,
611        processor_id_after,
612    ))
613}

615fn assemble(image: &mut [u8], config: RenderConfig, tile: Tile, pixels: &[u8]) {
616    let row_bytes = tile.width as usize * 3;
617    for row in 0..tile.height as usize {
618        let start = ((tile.y as usize + row) * config.width as usize + tile.x as usize) * 3;
619        image[start..start + row_bytes]
620            .copy_from_slice(&pixels[row * row_bytes..(row + 1) * row_bytes]);
621    }
622}

624fn tile_at(id: u32, config: RenderConfig, tile_size: u32) -> Tile {
625    let columns = config.width.div_ceil(tile_size);
626    let x = (id % columns) * tile_size;
627    let y = (id / columns) * tile_size;
628    Tile {
629        id,
630        x,
631        y,
632        width: tile_size.min(config.width - x),
633        height: tile_size.min(config.height - y),
634    }
635}

637impl Measurement {
638    fn accumulate(
639        &mut self,
640        stats: RenderStats,
641        duration: u64,
642        wire_bytes: u64,
643        pixel_bytes: u64,
644        latency_ns: u64,
645    ) -> Result<(), String> {
646        self.stats.rays = self
647            .stats
648            .rays
649            .checked_add(stats.rays)
650            .ok_or_else(|| String::from("ray total overflow"))?;
651        self.stats.samples = self
652            .stats
653            .samples
654            .checked_add(stats.samples)
655            .ok_or_else(|| String::from("sample total overflow"))?;
656        self.compute_ns = self
657            .compute_ns
658            .checked_add(duration)
659            .ok_or_else(|| String::from("compute duration overflow"))?;
660        self.wire_bytes = self
661            .wire_bytes
662            .checked_add(wire_bytes)
663            .ok_or_else(|| String::from("wire byte total overflow"))?;
664        self.pixel_payload_bytes = self
665            .pixel_payload_bytes
666            .checked_add(pixel_bytes)
667            .ok_or_else(|| String::from("pixel byte total overflow"))?;
668        self.latency.push(latency_ns);
669        Ok(())
670    }

673fn render(mode: Mode, opts: &Options, scene: &Scene, clock: Clock) -> Result<Measurement, String> {
674    let started = clock.now_ns();
675    let config = opts.config;
676    let scene_hash = scene.fingerprint();
677    let bytes = config
678        .rgb_bytes()
679        .map_err(|error| format!("image size: {:?}", error))?;
680    let tiles = config.width.div_ceil(opts.tile_size) * config.height.div_ceil(opts.tile_size);
681    let mut measured = Measurement {
682        mode,
683        wall_ns: 0,
684        setup_ns: 0,
685        compute_ns: 0,
686        wire_bytes: 0,
687        pixel_payload_bytes: 0,
688        stats: RenderStats::default(),
689        latency: Vec::with_capacity(tiles as usize),
690        pixels: vec![0; bytes],
691        workers_started: 0,
692        peak_outstanding: 0,
693        worker_records: [WorkerRecord::default(); 4],
694        processor_ids: ProcessorIds::default(),
695    };
696    let mut scratch = if mode == Mode::Baseline {
697        vec![0; COPIED_TILE_MAX_BYTES]
698    } else {
699        Vec::new()
700    };
701    let mut pool = Pool::start(mode, opts, scene_hash, tiles)?;
702    measured.workers_started = pool.slots.len() as u32;
703    measured.setup_ns = clock.elapsed_ns(started);
704    if mode == Mode::Baseline {
705        for id in 0..tiles {
706            let tile = tile_at(id, config, opts.tile_size);
707            let bytes = tile
708                .rgb_bytes()
709                .map_err(|error| format!("tile bytes: {:?}", error))?;
710            let processor_id_before = ostd::system_info::processor_id();
711            let tile_started = clock.now_ns();
712            let stats = render_tile(scene, config, tile, &mut scratch[..bytes])
713                .map_err(|error| format!("baseline render: {:?}", error))?;
714            let duration = clock.elapsed_ns(tile_started);
715            let processor_id_after = ostd::system_info::processor_id();
716            measured.processor_ids.observe(processor_id_before)?;
717            measured.processor_ids.observe(processor_id_after)?;
718            measured.accumulate(stats, duration, 0, 0, duration)?;
719            assemble(&mut measured.pixels, config, tile, &scratch[..bytes]);
720        }
721    } else {
722        let mut next_tile = 0;
723        let mut completed = 0;
724        let mut outstanding = 0;
725        while completed < tiles {
726            // Fill ALL available slots before consuming even an already-retained
727            // terminal. This records accepted, not-yet-consumed operations, not
728            // simultaneous computation or an observed processor count.
729            for slot in &mut pool.slots {
730                if slot.pending.is_none() && next_tile < tiles {
731                    slot.submit(
732                        config,
733                        tile_at(next_tile, config, opts.tile_size),
734                        scene_hash,
735                        clock,
736                    )?;
737                    next_tile += 1;
738                    outstanding += 1;
739                    measured.peak_outstanding = measured.peak_outstanding.max(outstanding);
740                }
741            }
742            let mut progress = false;
743            // Scan every worker before waiting; a slow first slot never blocks
744            // consuming later workers or refilling their available slots.
745            for (index, slot) in pool.slots.iter_mut().enumerate() {
746                let Some(active) = slot.pending.as_ref() else {
747                    continue;
748                };
749                let completion = match active.call.try_take(&mut slot.receive) {
750                    Ok(Some(completion)) => completion,
751                    Ok(None) => continue,
752                    Err(error) => {
753                        return Err(format!(
754                            "worker {} operation failed: {:?}",
755                            slot.worker.tid, error,
756                        ))
757                    }
758                };
759                let active = slot.pending.take().unwrap();
760                outstanding -= 1;
761                if completion.terminal != ostd::ipc::IpcTerminal::Reply {
762                    return Err(format!(
763                        "worker {} operation terminal: {:?}",
764                        slot.worker.tid, completion.terminal,
765                    ));
766                }
767                let (stats, duration, pixels, processor_id_before, processor_id_after) =
768                    validate_done(
769                        &slot.receive[..completion.len],
770                        active.tile,
771                        config,
772                        scene_hash,
773                        slot.grant.is_some(),
774                    )?;
775                for id in [processor_id_before, processor_id_after] {
776                    measured.processor_ids.observe(id)?;
777                    measured.worker_records[index].processor_ids.observe(id)?;
778                }
779                measured.accumulate(
780                    stats,
781                    duration,
782                    (active.request_bytes + completion.len) as u64,
783                    pixels.len() as u64,
784                    clock.elapsed_ns(active.started_ns),
785                )?;
786                if let Some(grant) = slot.grant.as_mut() {
787                    let bytes = active
788                        .tile
789                        .rgb_bytes()
790                        .map_err(|error| format!("tile bytes: {:?}", error))?;
791                    // Only this slot's validated reply permits reading/reusing
792                    // its grant. Image assembly is local, not pixel IPC traffic.
793                    grant.with_bytes(|pixels| {
794                        assemble(&mut measured.pixels, config, active.tile, &pixels[..bytes]);
795                    });
796                } else {
797                    assemble(&mut measured.pixels, config, active.tile, pixels);
798                }
799                slot.completed += 1;
800                completed += 1;
801                progress = true;
802            }
803            if !progress && !ostd::ipc::wait(0) {
804                return Err(String::from("worker pool IPC wait failed"));
805            }
806        }
807    }
808    pool.finish()?;
809    for (record, slot) in measured.worker_records.iter_mut().zip(&pool.slots) {
810        record.tid = slot.worker.tid;
811        record.tiles = slot.completed;
812    }
813    drop(pool);
814    measured.wall_ns = clock.elapsed_ns(started);
815    measured.latency.sort_unstable();
816    Ok(measured)
817}

880/// One VFS endpoint and the buffers its request/reply exchanges reuse.
881struct Vfs {
882    tid: usize,
883    send: [u8; IPC_BUF_SIZE],
884    receive: [u8; IPC_BUF_SIZE],
885}
886
887impl Vfs {
888    fn call(&mut self, request: &VfsRequest<'_>) -> Result<VfsResponse<'_>, IpcError> {
889        ostd::ipc::service_call_typed(self.tid, request, &mut self.send, &mut self.receive)
890    }
891
892    /// One step of writing `path` that the VFS must answer `Ok`.
893    fn write(&mut self, path: &str, request: &VfsRequest<'_>) -> Result<(), String> {
894        match self.call(request) {
895            Ok(VfsResponse::Ok) => Ok(()),
896            result => Err(write_failure(path, result)),
897        }
898    }
899}
900
901fn write_failure(path: &str, result: Result<VfsResponse<'_>, IpcError>) -> String {
902    match result {
903        Ok(response) => format!(
904            "VFS write {} failed: {:?}; partial image may remain",
905            path, response
906        ),
907        Err(error) => format!(
908            "VFS write {} operation failed: {:?}; completion not proven",
909            path, error
910        ),
911    }
912}
913
914/// Check the completed file size, not just a successful submission.
915fn verify_size(expected: u64, result: Result<VfsResponse<'_>, IpcError>) -> Result<(), String> {
916    match result {
917        Ok(VfsResponse::Stat {
918            size,
919            is_dir: false,
920        }) if size == expected => Ok(()),
921        result => Err(format!(
922            "saved image size verification failed: expected {}, got {:?}",
923            expected, result
924        )),
925    }
926}

928/// Save `pixels` as a binary PPM at `path`, then check the stored size.
929///
930/// A Cell launched with a cwd grant (every shell-launched foreground child) is
931/// sealed by the VFS: every path-string request is refused with `ERR_DENIED`,
932/// and the grant is the only directory it may reach. The image is then saved
933/// by handle: `OpenDir` down from the granted cwd and [`save_ppm_at`] the leaf,
934/// so `path` must lie at or below the cwd. A Cell launched without that grant
935/// still names the path.
936fn save_ppm(path: &str, config: RenderConfig, pixels: &[u8]) -> Result<(), String> {
937    let tid = ostd::service::provider_tid(ostd::service::service::VFS)
938        .ok_or_else(|| String::from("VFS service unavailable; image not saved"))?;
939    let mut vfs = Vfs {
940        tid,
941        send: [0; IPC_BUF_SIZE],
942        receive: [0; IPC_BUF_SIZE],
943    };
944    let header = format!("P6\n{} {}\n255\n", config.width, config.height);
945    let expected = (header.len() + pixels.len()) as u64;
946    let context = ostd::child_context::ChildContext::read_own().ok();
947    let granted = context
948        .as_ref()
949        .and_then(|c| Some((c.cwd_grant()?, core::str::from_utf8(c.cwd()).ok()?)));
950    let Some((grant, cwd)) = granted else {
951        vfs.write(
952            path,
953            &VfsRequest::Write {
954                path,
955                content: header.as_bytes(),
956            },
957        )?;
958        for chunk in pixels.chunks(3072) {
959            vfs.write(
960                path,
961                &VfsRequest::Append {
962                    path,
963                    content: chunk,
964                },
965            )?;
966        }
967        return verify_size(expected, vfs.call(&VfsRequest::Stat(path)));
968    };
969    let relative = below(cwd, path).ok_or_else(|| {
970        format!(
971            "output {} is outside the working directory {}; image not saved",
972            path, cwd
973        )
974    })?;
975    let mut names = relative
976        .split('/')
977        .filter(|name| !name.is_empty() && *name != ".");
978    let name = names
979        .next_back()
980        .ok_or_else(|| format!("output {} names no file; image not saved", path))?;
981    let mut dir = ViDirHandle(grant.id);
982    // First handle derived from the grant; closing it revokes every later one
983    // and every file handle opened through them.
984    let mut derived = None;
985    let mut opened = Ok(());
986    for component in names {
987        match vfs.call(&VfsRequest::OpenDir {
988            dir,
989            name: component,
990        }) {
991            Ok(VfsResponse::DirHandle(sub)) => {
992                derived.get_or_insert(sub);
993                dir = sub;
994            }
995            result => {
996                opened = Err(format!(
997                    "VFS open directory {} of {} failed: {:?}; image not saved",
998                    component, path, result
999                ));
1000                break;
1001            }
1002        }
1003    }
1004    let result = opened.and_then(|()| {
1005        save_ppm_at(
1006            &mut vfs,
1007            dir,
1008            name,
1009            path,
1010            header.as_bytes(),
1011            pixels,
1012            expected,
1013        )
1014    });
1015    if let Some(first) = derived {
1016        let _ = vfs.call(&VfsRequest::CloseDir { dir: first });
1017    }
1018    result
1019}

1021/// Largest `WriteGrant` payload the VFS accepts (`MAX_GRANT_WRITE` in
1022/// `cells/services/vfs/src/grant_write.rs`).
1023const VFS_GRANT_WRITE_MAX: usize = 4096;
1024
1025/// Write `header` then `pixels` as `name` inside `dir`, and check the stored size.
1026///
1027/// The VFS has no handle-addressed append. `WriteAt` creates or truncates the
1028/// file with the header; the pixels then go through a file handle
1029/// (`OpenFileAt`) as `WriteGrant` writes at explicit offsets, each starting at
1030/// the size the previous one left. `WriteGrant` is used rather than the 64 KiB
1031/// `WriteHandleGrant` because it rewrites the whole file through the backend, so
1032/// it also serves `/tmp`: the RAM backend has no positional write and answers
1033/// `WriteHandleGrant` with `ERR_IO`.
1034fn save_ppm_at(
1035    vfs: &mut Vfs,
1036    dir: ViDirHandle,
1037    name: &str,
1038    path: &str,
1039    header: &[u8],
1040    pixels: &[u8],
1041    expected: u64,
1042) -> Result<(), String> {
1043    vfs.write(
1044        path,
1045        &VfsRequest::WriteAt {
1046            dir,
1047            name,
1048            content: header,
1049        },
1050    )?;
1051    if !pixels.is_empty() {
1052        let file = match vfs.call(&VfsRequest::OpenFileAt { dir, name }) {
1053            Ok(VfsResponse::FileHandle(file)) => file,
1054            result => return Err(write_failure(path, result)),
1055        };
1056        let written = write_pixels(vfs, file, path, header.len() as u64, pixels);
1057        let _ = vfs.call(&VfsRequest::CloseFile { file });
1058        written?;
1059    }
1060    verify_size(expected, vfs.call(&VfsRequest::StatAt { dir, name }))
1061}
1062
1063/// Write `pixels` to `file` from byte `offset` through one grant shared
1064/// read-only with the VFS, at most [`VFS_GRANT_WRITE_MAX`] bytes per request.
1065fn write_pixels(
1066    vfs: &mut Vfs,
1067    file: ViVfsFileHandle,
1068    path: &str,
1069    mut offset: u64,
1070    pixels: &[u8],
1071) -> Result<(), String> {
1072    let chunk_max = pixels.len().min(VFS_GRANT_WRITE_MAX);
1073    let mut grant = GrantHandle::<u8>::alloc(chunk_max).ok_or_else(|| {
1074        format!(
1075            "VFS write {} grant allocation failed; partial image may remain",
1076            path
1077        )
1078    })?;
1079    if !ostd::syscall::sys_grant_share(grant.id(), vfs.tid, 0) {
1080        return Err(format!(
1081            "VFS write {} grant share failed; partial image may remain",
1082            path
1083        ));
1084    }
1085    for chunk in pixels.chunks(chunk_max) {
1086        // The VFS reads the grant only while serving the request below, and
1087        // this task is blocked in that call, so refilling never races a read.
1088        grant.with_bytes_mut(|bytes| bytes[..chunk.len()].copy_from_slice(chunk));
1089        let request = VfsRequest::WriteGrant {
1090            cap: file.0,
1091            offset,
1092            grant: grant.id(),
1093            bytes: chunk.len(),
1094        };
1095        match vfs.call(&request) {
1096            Ok(VfsResponse::GrantDone { bytes }) if bytes == chunk.len() => {}
1097            result => return Err(write_failure(path, result)),
1098        }
1099        offset += chunk.len() as u64;
1100    }
1101    Ok(())
1102}

1104/// `target` relative to the directory `cwd`, or `None` when an absolute
1105/// `target` lies outside it. A relative `target` is already relative to it.
1106fn below<'a>(cwd: &str, target: &'a str) -> Option<&'a str> {
1107    if !target.starts_with('/') {
1108        return Some(target);
1109    }
1110    let rest = target.strip_prefix(cwd.trim_end_matches('/'))?;
1111    (rest.is_empty() || rest.starts_with('/')).then_some(rest)
1112}

1114fn run(args: &[String]) -> Result<(), String> {
1115    let opts = options(args)?;
1116    let clock = Clock::new()?;
1117    let scene_started = clock.now_ns();
1118    let scene = Scene::demo();
1119    let scene_setup_ns = clock.elapsed_ns(scene_started);
1120    let scene_hash = scene.fingerprint();
1121    ostd::io::println("mode,scene_hash,width,height,samples_per_pixel,max_bounces,seed,tile_size,tiles,timer_hz,scene_setup_ns,wall_ns,setup_ns,compute_ns,render_wire_bytes,pixel_payload_bytes,rays,total_samples,RGB_bytes,pixel_hash,tile_p50_ns,tile_p95_ns,tile_p99_ns,workers_requested,workers_started,peak_outstanding,processor_ids");
1122    let first = if matches!(opts.mode, Mode::Compare | Mode::CompareTier2) {
1123        Mode::Baseline
1124    } else {
1125        opts.mode
1126    };
1127    let baseline = render(first, &opts, &scene, clock)?;
1128    report(&baseline, &opts, scene_hash, scene_setup_ns, clock);
1129    if matches!(opts.mode, Mode::Compare | Mode::CompareTier2) {
1130        let modes: &[Mode] = if opts.mode == Mode::CompareTier2 {
1131            &[Mode::Tier2]
1132        } else {
1133            &[Mode::Copy, Mode::Shared]
1134        };
1135        for &mode in modes {
1136            let measured = render(mode, &opts, &scene, clock)?;
1137            report(&measured, &opts, scene_hash, scene_setup_ns, clock);
1138            if measured.pixels != baseline.pixels || measured.stats != baseline.stats {
1139                let mismatch = measured
1140                    .pixels
1141                    .iter()
1142                    .zip(&baseline.pixels)
1143                    .position(|(actual, expected)| actual != expected);
1144                return Err(format!("COMPARE mismatch in {} at RGB byte {:?}; baseline rays/samples {:?}, actual {:?}", mode.name(), mismatch, baseline.stats, measured.stats));
1145            }
1146        }
1147    }
1148    let save_started = clock.now_ns();
1149    save_ppm(&opts.output, opts.config, &baseline.pixels)?;
1150    ostd::io::println(&format!(
1151        "c2c-render: saved {} bytes={} save_ns={}",
1152        opts.output,
1153        baseline.pixels.len(),
1154        clock.elapsed_ns(save_started)
1155    ));
1156    match opts.mode {
1157        Mode::Compare => ostd::io::println("c2c-render: COMPARE OK baseline=copy=shared"),
1158        Mode::CompareTier2 => ostd::io::println("c2c-render: COMPARE OK baseline=tier2"),
1159        _ => ostd::io::println("c2c-render: RENDER OK"),
1160    }
1161    Ok(())
1162}

1164fn cell_main() {
1165    let args = ostd::args();
1166    if args.iter().any(|arg| arg == "--help" || arg == "-h") {
1167        ostd::io::println(USAGE);
1168        sys_exit(0);

1170    match run(&args) {
1171        Ok(()) => sys_exit(0),
1172        Err(error) => {
1173            ostd::io::println(&format!("c2c-render: ERROR {}", error));
1174            sys_exit(1);
1175        }

1324    fn processor_observations_deduplicate_and_refuse_capacity_loss() {
1325        let mut ids = ProcessorIds::default();
1326        ids.observe(None).unwrap();
1327        assert_eq!(ids.to_string(), "-");
1328        ids.observe(Some(u32::MAX)).unwrap();
1329        ids.observe(Some(u32::MAX)).unwrap();
1330        ids.observe(None).unwrap();
1331        ids.observe(Some(0)).unwrap();
1332        assert_eq!(ids.to_string(), "0;4294967295");
1333        // Fill to the kernel bound out of order; the display stays sorted.
1334        for id in (1..MAX_PROCESSOR_IDS as u32 - 1).rev() {
1335            ids.observe(Some(id)).unwrap();
1336        }
1337        let full: Vec<String> = (0..MAX_PROCESSOR_IDS as u32 - 1)
1338            .chain([u32::MAX])
1339            .map(|id| id.to_string())
1340            .collect();
1341        let full = full.join(";");
1342        assert_eq!(ids.to_string(), full);
1343        let overflow = ids.observe(Some(MAX_PROCESSOR_IDS as u32 - 1)).unwrap_err();
1344        assert!(overflow.contains("64-CPU"), "{overflow}");
1345        assert_eq!(ids.to_string(), full);
1346        ids.observe(Some(0)).unwrap();
1347        ids.observe(Some(u32::MAX)).unwrap();
1348        ids.observe(None).unwrap();
1349        assert_eq!(ids.to_string(), full);
