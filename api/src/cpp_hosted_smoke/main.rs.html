1//! `cpp-hosted` witness for x86_64 Tier-2 (ADR-0034, runtime plan R4).
2//!
3//! The checks live in `cpp/witness.cpp` and print `CPP-HOSTED: <name> OK|FAIL`
4//! markers, then `CPP-HOSTED: <pass>/<total> pass`. When every check passed,
5//! the Cell ends in a deliberately uncaught exception: libc++abi names it on
6//! stderr, `abort()` reaches mlibc's panic path, and the Cell exits with
7//! status 1. A run therefore always exits non-zero; the markers, the
8//! terminate line and the kernel staying up are the verdict
9//! (`scripts/x86/qemu-cpp-hosted-smoke.sh`).
10
11#![no_std]
12#![no_main]
13extern crate api;
14extern crate mlibc_shim;
15extern crate ostd;
16
17use ostd::syscall::{sys_exit, sys_log};
18
19api::declare_manifest!(
20    block_io = false,
21    network = false,
22    spawn = false,
23    tier = api::manifest::PROTECTION_CLASS_UNTRUSTED
24);
25api::declare_syscalls![
26    Write, Log, Exit, GetTime, Yield, Spawn, FutexWait, FutexWake, Open, Close, GetRandom
27];
28
29#[cfg(cpp_hosted_witness)]
30extern "C" {
31    /// Runs every check; returns the number of failed checks. `noexcept` on
32    /// the C++ side: no exception reaches this frame (ADR-0034 decision 4).
33    fn cpp_hosted_witness() -> i32;
34    /// Throws past a `noexcept` boundary: std::terminate, then abort().
35    fn cpp_hosted_uncaught();
36}
37
38#[cfg(cpp_hosted_witness)]
39#[no_mangle]
40pub fn main() {
41    // SAFETY: integer-only C functions with no arguments (rule H1); mlibc and
42    // the C++ static constructors ran in `__cellos_libc_start` before `main`.
43    let failed = unsafe { cpp_hosted_witness() };
44    if failed != 0 {
45        sys_exit(1);
46    }
47    // SAFETY: as above; it never returns when the runtime is correct.
48    unsafe { cpp_hosted_uncaught() };
49    let _ = sys_log("CPP-HOSTED: uncaught FAIL (returned)");
50    sys_exit(1);
51}
52
53/// Built without the C++ runtime (see build.rs): fail loudly instead of passing.
54#[cfg(not(cpp_hosted_witness))]
55#[no_mangle]
56pub fn main() {
57    let _ = sys_log("CPP-HOSTED: NOT-BUILT");
58    sys_exit(1);
59}
