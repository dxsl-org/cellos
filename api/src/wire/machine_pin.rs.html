2//! Machine pin frame: cellbox and init ↔ the `machine-pin` cell over
3//! `service::MACHINE_PIN` (ADR-0029 §3, amended in Phase 13).
4//!
5//! Class C record (`docs/abi-boundary-records.md`). Fixed little-endian
6//! frames, identified by magic, version exactly [`FRAME_VERSION`]; unknown
7//! operations, kinds, states and codes, non-zero reserved bytes and non-zero
8//! bytes past the frame are refused before any effect. A request fits a
9//! 64-byte receive payload; caller identity never comes from the frame but
10//! from the kernel's attested receive trailer.
11//!
12//! Request ([`REQUEST_BYTES`]): 0 magic `MACHPINQ` · 8 version · 9 op (1
13//! Status, 2 Confirm, 3 Gate, 4 ConsoleShell) · 10 reserved [6] · 16 argument
14//! [32] (Confirm: digest; ConsoleShell: tid as u64 LE in 16..24, rest zero;
15//! zero for Status and Gate).
16//! Response ([`RESPONSE_BYTES`]): 0 magic `MACHPINR` · 8 version · 9 kind (1
17//! Status, 2 Recorded, 3 Refused, 4 Gate) · 10 code ([`PinState`] for Status
18//! and Gate, [`Refusal`] for Refused, zero for Recorded) · 11 safe mode (Gate:
19//! 0 or 1; otherwise zero) · 12 reserved [4] · 16 digest [32] (Status:
20//! current; Recorded: the new pin; otherwise zero) · 48 digest [32] (Status:
21//! pinned, zero when none; otherwise zero).
22//!
23//! Ops 3 and 4 and kind 4 were appended in Phase 13, when the endpoint left
24//! init: init asks `Gate` as a client and tells the cell the console shell's
25//! tid with a one-way `ConsoleShell`, which the cell accepts only from init.
26
27/// Version byte of both frames.
28pub const FRAME_VERSION: u8 = 1;
29/// Exact request length.
30pub const REQUEST_BYTES: usize = 48;
31/// Exact response length.
32pub const RESPONSE_BYTES: usize = 80;
33const REQUEST_MAGIC: &[u8; 8] = b"MACHPINQ";
34const RESPONSE_MAGIC: &[u8; 8] = b"MACHPINR";
35
36/// What the machine-pin cell is asked to do.
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum MachinePinRequest {
39    /// Report this boot's gate outcome and both digests. Answered in every mode.
40    Status,
41    /// Re-pin the current facts. Honoured only in safe mode, only from the
42    /// console shell init started, and only when `digest` is the current one.
43    Confirm { digest: [u8; 32] },
44    /// Init's question after it launched the cell: this boot's outcome and
45    /// whether it withholds services. Answered in every mode.
46    Gate,
47    /// One-way, from init only: the live console shell init started in safe
48    /// mode, the only sender `Confirm` is honoured from. Never answered.
49    ConsoleShell { tid: u64 },
50}
51
52/// This boot's gate outcome.
53#[derive(Clone, Copy, Debug, PartialEq, Eq)]
54#[repr(u8)]
55pub enum PinState {
56    /// The stored record equals the current facts.
57    Match = 1,
58    /// No record existed; the current facts were recorded.
59    FirstBootRecorded = 2,
60    /// The stored record decodes but differs: safe mode.
61    Mismatch = 3,
62    /// The stored record does not decode or has an unknown version: safe mode.
63    RecordUnreadable = 4,
64    /// The pin store could not be read or written.
65    StoreUnavailable = 5,
66    /// The kernel gave no facts, or the endpoint was never registered.
67    FactsUnavailable = 6,
68}
69
70/// Why a [`MachinePinRequest::Confirm`] was refused.
71#[derive(Clone, Copy, Debug, PartialEq, Eq)]
72#[repr(u8)]
73pub enum Refusal {
74    /// The boot is not in safe mode.
75    NotSafeMode = 1,
76    /// The sender is not the live console shell init started.
77    NotConsoleShell = 2,
78    /// The digest is not the current facts' digest.
79    DigestStale = 3,
80    /// Writing or reading back the record failed.
81    StoreWriteFailed = 4,
82}
83
84/// The machine-pin cell's answer.
85#[derive(Clone, Copy, Debug, PartialEq, Eq)]
86pub enum MachinePinResponse {
87    /// `pinned` is zero when no readable record exists.
88    Status {
89        state: PinState,
90        current: [u8; 32],
91        pinned: [u8; 32],
92    },
93    /// The current facts are now the pin; reboot to start services.
94    Recorded { digest: [u8; 32] },
95    /// Nothing was written.
96    Refused(Refusal),
97    /// This boot's outcome and whether init must withhold every non-boot
98    /// service (the answer to [`MachinePinRequest::Gate`]).
99    Gate { state: PinState, safe_mode: bool },
100}
101
102impl PinState {
103    fn decode(value: u8) -> Option<Self> {
104        Some(match value {
105            1 => Self::Match,
106            2 => Self::FirstBootRecorded,
107            3 => Self::Mismatch,
108            4 => Self::RecordUnreadable,
109            5 => Self::StoreUnavailable,
110            6 => Self::FactsUnavailable,
111            _ => return None,
112        })
113    }
114
115    /// Lower-case name for console output.
116    pub const fn name(self) -> &'static str {
117        match self {
118            Self::Match => "match",
119            Self::FirstBootRecorded => "first-boot-recorded",
120            Self::Mismatch => "mismatch",
121            Self::RecordUnreadable => "record-unreadable",
122            Self::StoreUnavailable => "store-unavailable",
123            Self::FactsUnavailable => "facts-unavailable",
124        }
125    }
126}
127
128impl Refusal {
129    fn decode(value: u8) -> Option<Self> {
130        Some(match value {
131            1 => Self::NotSafeMode,
132            2 => Self::NotConsoleShell,
133            3 => Self::DigestStale,
134            4 => Self::StoreWriteFailed,
135            _ => return None,
136        })
137    }
138
139    /// The variant name, for console output.
140    pub const fn name(self) -> &'static str {
141        match self {
142            Self::NotSafeMode => "NotSafeMode",
143            Self::NotConsoleShell => "NotConsoleShell",
144            Self::DigestStale => "DigestStale",
145            Self::StoreWriteFailed => "StoreWriteFailed",
146        }
147    }
148}

150fn header_valid(frame: &[u8], magic: &[u8; 8], len: usize) -> bool {
151    frame.len() >= len
152        && &frame[..8] == magic
153        && frame[8] == FRAME_VERSION
154        && frame[len..].iter().all(|byte| *byte == 0)
155}
156
157fn digest_at(frame: &[u8], at: usize) -> [u8; 32] {
158    let mut digest = [0; 32];
159    digest.copy_from_slice(&frame[at..at + 32]);
160    digest
161}
162
163fn zero(bytes: &[u8]) -> bool {
164    bytes.iter().all(|byte| *byte == 0)
165}

167impl MachinePinRequest {
168    /// Wire bytes.
169    pub fn encode(self) -> [u8; REQUEST_BYTES] {
170        let mut frame = [0; REQUEST_BYTES];
171        frame[..8].copy_from_slice(REQUEST_MAGIC);
172        frame[8] = FRAME_VERSION;
173        match self {
174            Self::Status => frame[9] = 1,
175            Self::Confirm { digest } => {
176                frame[9] = 2;
177                frame[16..48].copy_from_slice(&digest);
178            }
179            Self::Gate => frame[9] = 3,
180            Self::ConsoleShell { tid } => {
181                frame[9] = 4;
182                frame[16..24].copy_from_slice(&tid.to_le_bytes());
183            }
184        }
185        frame
186    }
187
188    /// The request in `frame`, which may be a longer zero-padded payload;
189    /// `None` for anything else.
190    pub fn decode(frame: &[u8]) -> Option<Self> {
191        if !header_valid(frame, REQUEST_MAGIC, REQUEST_BYTES) || !zero(&frame[10..16]) {
192            return None;
193        }
194        match frame[9] {
195            1 if zero(&frame[16..48]) => Some(Self::Status),
196            2 => Some(Self::Confirm {
197                digest: digest_at(frame, 16),
198            }),
199            3 if zero(&frame[16..48]) => Some(Self::Gate),
200            4 if zero(&frame[24..48]) => Some(Self::ConsoleShell {
201                tid: u64::from_le_bytes(frame[16..24].try_into().ok()?),
202            }),
203            _ => None,
204        }
205    }
206
207    /// Whether `frame` claims to be a request of this protocol (magic only),
208    /// so a malformed claim is reported rather than silently ignored.
209    pub fn is_claimed(frame: &[u8]) -> bool {
210        frame.len() >= 8 && &frame[..8] == REQUEST_MAGIC
211    }
212}
213
214impl MachinePinResponse {
215    /// Wire bytes.
216    pub fn encode(self) -> [u8; RESPONSE_BYTES] {
217        let mut frame = [0; RESPONSE_BYTES];
218        frame[..8].copy_from_slice(RESPONSE_MAGIC);
219        frame[8] = FRAME_VERSION;
220        match self {
221            Self::Status {
222                state,
223                current,
224                pinned,
225            } => {
226                frame[9] = 1;
227                frame[10] = state as u8;
228                frame[16..48].copy_from_slice(&current);
229                frame[48..80].copy_from_slice(&pinned);
230            }
231            Self::Recorded { digest } => {
232                frame[9] = 2;
233                frame[16..48].copy_from_slice(&digest);
234            }
235            Self::Refused(refusal) => {
236                frame[9] = 3;
237                frame[10] = refusal as u8;
238            }
239            Self::Gate { state, safe_mode } => {
240                frame[9] = 4;
241                frame[10] = state as u8;
242                frame[11] = u8::from(safe_mode);
243            }
244        }
245        frame
246    }
247
248    /// The response in `frame`; `None` for anything else.
249    pub fn decode(frame: &[u8]) -> Option<Self> {
250        if !header_valid(frame, RESPONSE_MAGIC, RESPONSE_BYTES) || !zero(&frame[12..16]) {
251            return None;
252        }
253        // Byte 11 is the Gate safe-mode flag and zero in every other kind.
254        if frame[9] != 4 && frame[11] != 0 {
255            return None;
256        }
257        match frame[9] {
258            1 => Some(Self::Status {
259                state: PinState::decode(frame[10])?,
260                current: digest_at(frame, 16),
261                pinned: digest_at(frame, 48),
262            }),
263            2 if frame[10] == 0 && zero(&frame[48..80]) => Some(Self::Recorded {
264                digest: digest_at(frame, 16),
265            }),
266            3 if zero(&frame[16..80]) => Some(Self::Refused(Refusal::decode(frame[10])?)),
267            4 if frame[11] <= 1 && zero(&frame[16..80]) => Some(Self::Gate {
268                state: PinState::decode(frame[10])?,
269                safe_mode: frame[11] == 1,
270            }),
271            _ => None,
272        }
273    }
274}

280    #[test]
281    fn requests_round_trip_and_fit_inits_payload() {
282        const { assert!(REQUEST_BYTES <= 64) };
283        for request in [
284            MachinePinRequest::Status,
285            MachinePinRequest::Confirm { digest: [7; 32] },
286            MachinePinRequest::Gate,
287            MachinePinRequest::ConsoleShell {
288                tid: 0x0102_0304_0506_0708,
289            },
290        ] {
291            let frame = request.encode();
292            assert_eq!(MachinePinRequest::decode(&frame), Some(request));
293            let mut padded = [0u8; 64];
294            padded[..REQUEST_BYTES].copy_from_slice(&frame);
295            assert_eq!(MachinePinRequest::decode(&padded), Some(request));
296            assert!(MachinePinRequest::is_claimed(&padded));
297        }
298    }

300    #[test]
301    fn malformed_requests_are_refused() {
302        let base = MachinePinRequest::Confirm { digest: [7; 32] }.encode();
303        let refuse = |mutate: &dyn Fn(&mut [u8; 64])| {
304            let mut frame = [0u8; 64];
305            frame[..REQUEST_BYTES].copy_from_slice(&base);
306            mutate(&mut frame);
307            assert_eq!(MachinePinRequest::decode(&frame), None);
308        };
309        refuse(&|f| f[0] = b'X');
310        refuse(&|f| f[8] = 2);
311        refuse(&|f| f[9] = 0);
312        refuse(&|f| f[9] = 3);
313        refuse(&|f| f[10] = 1);
314        refuse(&|f| f[15] = 1);
315        refuse(&|f| f[48] = 1);
316        refuse(&|f| f[63] = 1);
317        refuse(&|f| {
318            f[9] = 1;
319        });
320        assert_eq!(MachinePinRequest::decode(&base[..47]), None);
321        assert!(!MachinePinRequest::is_claimed(b"OCLSREQ1"));
322    }

395    /// The Phase 13 frames, pinned byte for byte: changing them breaks init
396    /// and the machine-pin cell at once, so the layout is part of the record.
397    #[test]
398    fn appended_frames_are_byte_pinned() {
399        let mut gate = [0u8; REQUEST_BYTES];
400        gate[..10].copy_from_slice(b"MACHPINQ\x01\x03");
401        assert_eq!(MachinePinRequest::Gate.encode(), gate);
402
403        let mut shell = [0u8; REQUEST_BYTES];
404        shell[..10].copy_from_slice(b"MACHPINQ\x01\x04");
405        shell[16..24].copy_from_slice(&[0x2A, 0, 0, 0, 0, 0, 0, 0]);
406        assert_eq!(MachinePinRequest::ConsoleShell { tid: 42 }.encode(), shell);
407
408        let mut answer = [0u8; RESPONSE_BYTES];
409        answer[..12].copy_from_slice(b"MACHPINR\x01\x04\x03\x01");
410        assert_eq!(
411            MachinePinResponse::Gate {
412                state: PinState::Mismatch,
413                safe_mode: true,
414            }
415            .encode(),
416            answer
417        );
418    }
