2//! Net service error bytes carried in `NetResponse::Err(u8)`; the requests are
3//! the postcard `NetRequest`/`NetResponse` in `ipc.rs`, which also owns the
4//! readiness frames re-exported here.
5
6pub use crate::ipc::{
7    net_ready, NET_READY_FRAME_BYTES, NET_READY_OPCODE, NET_READY_RECORD_BYTES, REACTOR_WAKE_OPCODE,
8};
9
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11#[repr(u8)]
12pub enum NetError {
13    /// The port is already bound.
14    AddrInUse = 0xFB,
15    /// The caller lacks the right for the request.
16    Denied = 0xFC,
17    /// A queue or slot is full, or the wait was refused.
18    Busy = 0xFD,
19    /// Not ready: no connection to accept yet, or the socket is not established.
20    NotReady = 0xFE,
21    /// Generic failure. On `TcpRecv` the same byte is [`TCP_RECV_EOF`].
22    Failure = 0xFF,
23}
24
25/// `TcpRecv` end of stream. Shares its byte with [`NetError::Failure`]; only
26/// the request kind tells them apart.
27pub const TCP_RECV_EOF: u8 = 0xFF;
28
29/// `NetRequest` postcard variants stay below this index: the raw TLS frames
30/// are routed by first byte `>=` [`super::tls::CLOSE`] when postcard decode
31/// fails, so variant 0x15 would be taken for `TLS_CLOSE`.
32pub const NET_REQUEST_VARIANT_LIMIT: u8 = super::tls::CLOSE;
