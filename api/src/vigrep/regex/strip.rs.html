16/// Return an HIR that is guaranteed to never match the given line terminator,
17/// if possible.
18///
19/// If the transformation isn't possible, then an error is returned.
20///
21/// In general, if a literal line terminator occurs anywhere in the HIR, then
22/// this will return an error. However, if the line terminator occurs within
23/// a character class with at least one other character (that isn't also a line
24/// terminator), then the line terminator is simply stripped from that class.
25///
26/// If the given line terminator is not ASCII, then this function returns an
27/// error.
28///
29/// Note that as of regex 1.9, this routine could theoretically be implemented
30/// without returning an error. Namely, for example, we could turn
31/// `foo\nbar` into `foo[a&&b]bar`. That is, replace line terminator with a
32/// sub-expression that can never match anything. Thus, ripgrep would accept
33/// such regexes and just silently not match anything. Regex versions prior
34/// to 1.8 don't support such constructs. I ended up deciding to leave the
35/// existing behavior of returning an error instead. For example:
36///
37/// ```text
38/// $ echo -n 'foo\nbar\n' | rg 'foo\nbar'
39/// the literal '"\n"' is not allowed in a regex
40///
41/// Consider enabling multiline mode with the --multiline flag (or -U for short).
42/// When multiline mode is enabled, new line characters can be matched.
43/// ```
44///
45/// This looks like a good error message to me, and even suggests a flag that
46/// the user can use instead.
47pub(crate) fn strip_from_match(
48    expr: Hir,
49    line_term: LineTerminator,
50) -> Result<Hir, Error> {
51    if line_term.is_crlf() {
52        let expr1 = strip_from_match_ascii(expr, b'\r')?;
53        strip_from_match_ascii(expr1, b'\n')
54    } else {
55        strip_from_match_ascii(expr, line_term.as_byte())
56    }
57}
58
59/// The implementation of strip_from_match. The given byte must be ASCII.
60/// This function returns an error otherwise. It also returns an error if
61/// it couldn't remove `\n` from the given regex without leaving an empty
62/// character class in its place.
63fn strip_from_match_ascii(expr: Hir, byte: u8) -> Result<Hir, Error> {
64    if !byte.is_ascii() {
65        return Err(Error::new(ErrorKind::InvalidLineTerminator(byte)));
66    }
67    let ch = char::from(byte);
68    let invalid = || Err(Error::new(ErrorKind::NotAllowed(ch.to_string())));
69    Ok(match expr.into_kind() {
70        HirKind::Empty => Hir::empty(),
71        HirKind::Literal(hir::Literal(lit)) => {
72            if lit.iter().find(|&&b| b == byte).is_some() {
73                return invalid();
74            }
75            Hir::literal(lit)
76        }
77        HirKind::Class(hir::Class::Unicode(mut cls)) => {
78            if cls.ranges().is_empty() {
79                return Ok(Hir::class(hir::Class::Unicode(cls)));
80            }
81            let remove = hir::ClassUnicode::new(Some(
82                hir::ClassUnicodeRange::new(ch, ch),
83            ));
84            cls.difference(&remove);
85            if cls.ranges().is_empty() {
86                return invalid();
87            }
88            Hir::class(hir::Class::Unicode(cls))
89        }
90        HirKind::Class(hir::Class::Bytes(mut cls)) => {
91            if cls.ranges().is_empty() {
92                return Ok(Hir::class(hir::Class::Bytes(cls)));
93            }
94            let remove = hir::ClassBytes::new(Some(
95                hir::ClassBytesRange::new(byte, byte),
96            ));
97            cls.difference(&remove);
98            if cls.ranges().is_empty() {
99                return invalid();
100            }
101            Hir::class(hir::Class::Bytes(cls))
102        }
103        HirKind::Look(x) => Hir::look(x),
104        HirKind::Repetition(mut x) => {
105            x.sub = Box::new(strip_from_match_ascii(*x.sub, byte)?);
106            Hir::repetition(x)
107        }
108        HirKind::Capture(mut x) => {
109            x.sub = Box::new(strip_from_match_ascii(*x.sub, byte)?);
110            Hir::capture(x)
111        }
112        HirKind::Concat(xs) => {
113            let xs = xs
114                .into_iter()
115                .map(|e| strip_from_match_ascii(e, byte))
116                .collect::<Result<Vec<Hir>, Error>>()?;
117            Hir::concat(xs)
118        }
119        HirKind::Alternation(xs) => {
120            let xs = xs
121                .into_iter()
122                .map(|e| strip_from_match_ascii(e, byte))
123                .collect::<Result<Vec<Hir>, Error>>()?;
124            Hir::alternation(xs)
125        }
126    })
127}

129#[cfg(test)]
130mod tests {
131    use regex_syntax::Parser;
132
133    use super::{LineTerminator, strip_from_match};
134    use crate::regex::error::Error;
135
136    fn roundtrip(pattern: &str, byte: u8) -> String {
137        roundtrip_line_term(pattern, LineTerminator::byte(byte)).unwrap()
138    }
139
140    fn roundtrip_crlf(pattern: &str) -> String {
141        roundtrip_line_term(pattern, LineTerminator::crlf()).unwrap()
142    }
143
144    fn roundtrip_err(pattern: &str, byte: u8) -> Result<String, Error> {
145        roundtrip_line_term(pattern, LineTerminator::byte(byte))
146    }
147
148    fn roundtrip_line_term(
149        pattern: &str,
150        line_term: LineTerminator,
151    ) -> Result<String, Error> {
152        let expr1 = Parser::new().parse(pattern).unwrap();
153        let expr2 = strip_from_match(expr1, line_term)?;
154        Ok(expr2.to_string())
155    }
156
157    #[test]
158    fn various() {
159        assert_eq!(roundtrip(r"[a\n]", b'\n'), "a");
160        assert_eq!(roundtrip(r"[a\n]", b'a'), "\n");
161        assert_eq!(roundtrip_crlf(r"[a\n]"), "a");
162        assert_eq!(roundtrip_crlf(r"[a\r]"), "a");
163        assert_eq!(roundtrip_crlf(r"[a\r\n]"), "a");
164
165        assert_eq!(roundtrip(r"(?-u)\s", b'a'), r"(?-u:[\x09-\x0D\x20])");
166        assert_eq!(roundtrip(r"(?-u)\s", b'\n'), r"(?-u:[\x09\x0B-\x0D\x20])");
167
168        assert!(roundtrip_err(r"\n", b'\n').is_err());
169        assert!(roundtrip_err(r"abc\n", b'\n').is_err());
170        assert!(roundtrip_err(r"\nabc", b'\n').is_err());
171        assert!(roundtrip_err(r"abc\nxyz", b'\n').is_err());
172        assert!(roundtrip_err(r"\x0A", b'\n').is_err());
173        assert!(roundtrip_err(r"\u000A", b'\n').is_err());
174        assert!(roundtrip_err(r"\U0000000A", b'\n').is_err());
175        assert!(roundtrip_err(r"\u{A}", b'\n').is_err());
176        assert!(roundtrip_err("\n", b'\n').is_err());
177    }
