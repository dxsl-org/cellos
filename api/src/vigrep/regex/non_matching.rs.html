17/// Return a confirmed set of non-matching bytes from the given expression.
18pub(crate) fn non_matching_bytes(expr: &Hir) -> ByteSet {
19    let mut set = ByteSet::full();
20    remove_matching_bytes(expr, &mut set);
21    set
22}
23
24/// Remove any bytes from the given set that can occur in a matched produced by
25/// the given expression.
26fn remove_matching_bytes(expr: &Hir, set: &mut ByteSet) {
27    match *expr.kind() {
28        HirKind::Empty
29        | HirKind::Look(Look::WordAscii | Look::WordAsciiNegate)
30        | HirKind::Look(Look::WordUnicode | Look::WordUnicodeNegate)
31        | HirKind::Look(Look::WordStartAscii | Look::WordStartUnicode)
32        | HirKind::Look(Look::WordEndAscii | Look::WordEndUnicode)
33        | HirKind::Look(
34            Look::WordStartHalfAscii | Look::WordStartHalfUnicode,
35        )
36        | HirKind::Look(Look::WordEndHalfAscii | Look::WordEndHalfUnicode) => {
37        }
38        HirKind::Look(Look::Start | Look::End) => {
39            // FIXME: This is wrong, but not doing this leads to incorrect
40            // results because of how anchored searches are implemented in
41            // the 'grep-searcher' crate.
42            set.remove(b'\n');
43        }
44        HirKind::Look(Look::StartLF | Look::EndLF) => {
45            set.remove(b'\n');
46        }
47        HirKind::Look(Look::StartCRLF | Look::EndCRLF) => {
48            set.remove(b'\r');
49            set.remove(b'\n');
50        }
51        HirKind::Literal(hir::Literal(ref lit)) => {
52            for &b in lit.iter() {
53                set.remove(b);
54            }
55        }
56        HirKind::Class(hir::Class::Unicode(ref cls)) => {
57            for range in cls.iter() {
58                // This is presumably faster than encoding every codepoint
59                // to UTF-8 and then removing those bytes from the set.
60                for seq in Utf8Sequences::new(range.start(), range.end()) {
61                    for byte_range in seq.as_slice() {
62                        set.remove_all(byte_range.start, byte_range.end);
63                    }
64                }
65            }
66        }
67        HirKind::Class(hir::Class::Bytes(ref cls)) => {
68            for range in cls.iter() {
69                set.remove_all(range.start(), range.end());
70            }
71        }
72        HirKind::Repetition(ref x) => {
73            remove_matching_bytes(&x.sub, set);
74        }
75        HirKind::Capture(ref x) => {
76            remove_matching_bytes(&x.sub, set);
77        }
78        HirKind::Concat(ref xs) => {
79            for x in xs {
80                remove_matching_bytes(x, set);
81            }
82        }
83        HirKind::Alternation(ref xs) => {
84            for x in xs {
85                remove_matching_bytes(x, set);
86            }
87        }
88    }
89}

91#[cfg(test)]
92mod tests {
93    use {crate::matcher::ByteSet, regex_syntax::ParserBuilder};
94
95    use super::non_matching_bytes;
96
97    fn extract(pattern: &str) -> ByteSet {
98        let expr =
99            ParserBuilder::new().utf8(false).build().parse(pattern).unwrap();
100        non_matching_bytes(&expr)
101    }
102
103    fn sparse(set: &ByteSet) -> Vec<u8> {
104        let mut sparse_set = vec![];
105        for b in (0..256).map(|b| b as u8) {
106            if set.contains(b) {
107                sparse_set.push(b);
108            }
109        }
110        sparse_set
111    }
112
113    fn sparse_except(except: &[u8]) -> Vec<u8> {
114        let mut except_set = vec![false; 256];
115        for &b in except {
116            except_set[b as usize] = true;
117        }
118
119        let mut set = vec![];
120        for b in (0..256).map(|b| b as u8) {
121            if !except_set[b as usize] {
122                set.push(b);
123            }
124        }
125        set
126    }
127
128    #[test]
129    fn dot() {
130        assert_eq!(
131            sparse(&extract(".")),
132            vec![
133                b'\n', 192, 193, 245, 246, 247, 248, 249, 250, 251, 252, 253,
134                254, 255,
135            ]
136        );
137        assert_eq!(
138            sparse(&extract("(?s).")),
139            vec![
140                192, 193, 245, 246, 247, 248, 249, 250, 251, 252, 253, 254,
141                255,
142            ]
143        );
144        assert_eq!(sparse(&extract("(?-u).")), vec![b'\n']);
145        assert_eq!(sparse(&extract("(?s-u).")), vec![]);
146    }
147
148    #[test]
149    fn literal() {
150        assert_eq!(sparse(&extract("a")), sparse_except(&[b'a']));
151        assert_eq!(sparse(&extract("☃")), sparse_except(&[0xE2, 0x98, 0x83]));
152        assert_eq!(sparse(&extract(r"\xFF")), sparse_except(&[0xC3, 0xBF]));
153        assert_eq!(sparse(&extract(r"(?-u)\xFF")), sparse_except(&[0xFF]));
154    }
155
156    #[test]
157    fn anchor() {
158        // FIXME: The first four tests below should correspond to a full set
159        // of bytes for the non-matching bytes I think.
160        assert_eq!(sparse(&extract(r"^")), sparse_except(&[b'\n']));
161        assert_eq!(sparse(&extract(r"$")), sparse_except(&[b'\n']));
162        assert_eq!(sparse(&extract(r"\A")), sparse_except(&[b'\n']));
163        assert_eq!(sparse(&extract(r"\z")), sparse_except(&[b'\n']));
164        assert_eq!(sparse(&extract(r"(?m)^")), sparse_except(&[b'\n']));
165        assert_eq!(sparse(&extract(r"(?m)$")), sparse_except(&[b'\n']));
166    }
