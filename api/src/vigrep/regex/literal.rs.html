19/// A type that encapsulates "inner" literal extractiong from a regex.
20///
21/// It uses a huge pile of heuristics to try to pluck out literals from a regex
22/// that are in turn used to build a simpler regex that is more amenable to
23/// optimization.
24///
25/// The main idea underlying the validity of this technique is the fact
26/// that ripgrep searches individuals lines and not across lines. (Unless
27/// -U/--multiline is enabled.) Namely, we can pluck literals out of the regex,
28/// search for them, find the bounds of the line in which that literal occurs
29/// and then run the original regex on only that line. This overall works
30/// really really well in throughput oriented searches because it potentially
31/// allows ripgrep to spend a lot more time in a fast vectorized routine for
32/// finding literals as opposed to the (much) slower regex engine.
33///
34/// This optimization was far more important in the old days, but since then,
35/// Rust's regex engine has actually grown its own (albeit limited) support for
36/// inner literal optimizations. So this technique doesn't apply as much as it
37/// used to.
38///
39/// A good example of a regex where this particular extractor helps is
40/// `\s+(Sherlock|[A-Z]atso[a-z]|Moriarty)\s+`. The `[A-Z]` before the `atso`
41/// in particular is what inhibits the regex engine's own inner literal
42/// optimizations from kicking in. This particular regex also did not have any
43/// inner literals extracted in the old implementation (ripgrep <=13). So this
44/// particular implementation represents a strict improvement from both the old
45/// implementation and from the regex engine's own optimizations. (Which could
46/// in theory be improved still.)
47#[derive(Clone, Debug)]
48pub(crate) struct InnerLiterals {
49    seq: Seq,
50}
51
52impl InnerLiterals {
53    /// Create a set of inner literals from the given HIR expression.
54    ///
55    /// If no line terminator was configured, then this always declines to
56    /// extract literals because the inner literal optimization may not be
57    /// valid.
58    ///
59    /// Note that this requires the actual regex that will be used for a search
60    /// because it will query some state about the compiled regex. That state
61    /// may influence inner literal extraction.
62    pub(crate) fn new(chir: &ConfiguredHIR, re: &Regex) -> InnerLiterals {
63        // If there's no line terminator, then the inner literal optimization
64        // at this level is not valid.
65        if chir.config().line_terminator.is_none() {
66            log::trace!(
67                "skipping inner literal extraction, \
68                 no line terminator is set"
69            );
70            return InnerLiterals::none();
71        }
72        // If we believe the regex is already accelerated, then just let
73        // the regex engine do its thing. We'll skip the inner literal
74        // optimization.
75        //
76        // ... but only if the regex doesn't have any Unicode word boundaries.
77        // If it does, there's enough of a chance of the regex engine falling
78        // back to a slower engine that it's worth trying our own inner literal
79        // optimization.
80        if re.is_accelerated() {
81            if !chir.hir().properties().look_set().contains_word_unicode() {
82                log::trace!(
83                    "skipping inner literal extraction, \
84                     existing regex is believed to already be accelerated",
85                );
86                return InnerLiterals::none();
87            }
88        }
89        // In this case, we pretty much know that the regex engine will handle
90        // it as best as possible, even if it isn't reported as accelerated.
91        if chir.hir().properties().is_alternation_literal() {
92            log::trace!(
93                "skipping inner literal extraction, \
94                 found alternation of literals, deferring to regex engine",
95            );
96            return InnerLiterals::none();
97        }
98        let seq = Extractor::new().extract_untagged(chir.hir());
99        InnerLiterals { seq }
100    }
101
102    /// Returns a infinite set of inner literals, such that it can never
103    /// produce a matcher.
104    pub(crate) fn none() -> InnerLiterals {
105        InnerLiterals { seq: Seq::infinite() }
106    }
107
108    /// If it is deemed advantageous to do so (via various suspicious
109    /// heuristics), this will return a single regular expression pattern that
110    /// matches a subset of the language matched by the regular expression that
111    /// generated these literal sets. The idea here is that the pattern
112    /// returned by this method is much cheaper to search for. i.e., It is
113    /// usually a single literal or an alternation of literals.
114    pub(crate) fn one_regex(&self) -> Result<Option<Regex>, Error> {
115        let Some(lits) = self.seq.literals() else { return Ok(None) };
116        if lits.is_empty() {
117            return Ok(None);
118        }
119        let mut alts = vec![];
120        for lit in lits.iter() {
121            alts.push(Hir::literal(lit.as_bytes()));
122        }
123        let hir = Hir::alternation(alts);
124        log::debug!("extracted fast line regex: {:?}", hir.to_string());
125        let re = Regex::builder()
126            .configure(Regex::config().utf8_empty(false))
127            .build_from_hir(&hir)
128            .map_err(Error::regex)?;
129        Ok(Some(re))
130    }
131}

133/// An inner literal extractor.
134///
135/// This is a somewhat stripped down version of the extractor from
136/// regex-syntax. The main difference is that we try to identify a "best" set
137/// of required literals while traversing the HIR.
138#[derive(Debug)]
139struct Extractor {
140    limit_class: usize,
141    limit_repeat: usize,
142    limit_literal_len: usize,
143    limit_total: usize,
144}
145
146impl Extractor {
147    /// Create a new inner literal extractor with a default configuration.
148    fn new() -> Extractor {
149        Extractor {
150            limit_class: 10,
151            limit_repeat: 10,
152            limit_literal_len: 100,
153            limit_total: 64,
154        }
155    }
156
157    /// Execute the extractor at the top-level and return an untagged sequence
158    /// of literals.
159    fn extract_untagged(&self, hir: &Hir) -> Seq {
160        let mut seq = self.extract(hir);
161        log::trace!("extracted inner literals: {:?}", seq.seq);
162        seq.seq.optimize_for_prefix_by_preference();
163        log::trace!(
164            "extracted inner literals after optimization: {:?}",
165            seq.seq
166        );
167        if !seq.is_good() {
168            log::trace!(
169                "throwing away inner literals because they might be slow"
170            );
171            seq.make_infinite();
172        }
173        seq.seq
174    }
175
176    /// Execute the extractor and return a sequence of literals.
177    fn extract(&self, hir: &Hir) -> TSeq {
178        use regex_syntax::hir::HirKind::*;
179
180        match *hir.kind() {
181            Empty | Look(_) => TSeq::singleton(self::Literal::exact(vec![])),
182            Literal(hir::Literal(ref bytes)) => {
183                let mut seq =
184                    TSeq::singleton(self::Literal::exact(bytes.to_vec()));
185                self.enforce_literal_len(&mut seq);
186                seq
187            }
188            Class(hir::Class::Unicode(ref cls)) => {
189                self.extract_class_unicode(cls)
190            }
191            Class(hir::Class::Bytes(ref cls)) => self.extract_class_bytes(cls),
192            Repetition(ref rep) => self.extract_repetition(rep),
193            Capture(hir::Capture { ref sub, .. }) => self.extract(sub),
194            Concat(ref hirs) => self.extract_concat(hirs.iter()),
195            Alternation(ref hirs) => self.extract_alternation(hirs.iter()),
196        }
197    }
198
199    /// Extract a sequence from the given concatenation. Sequences from each of
200    /// the child HIR expressions are combined via cross product.
201    ///
202    /// This short circuits once the cross product turns into a sequence
203    /// containing only inexact literals.
204    fn extract_concat<'a, I: Iterator<Item = &'a Hir>>(&self, it: I) -> TSeq {
205        let mut seq = TSeq::singleton(self::Literal::exact(vec![]));
206        let mut prev: Option<TSeq> = None;
207        for hir in it {
208            // If every element in the sequence is inexact, then a cross
209            // product will always be a no-op. Thus, there is nothing else we
210            // can add to it and can quit early. Note that this also includes
211            // infinite sequences.
212            if seq.is_inexact() {
213                // If a concatenation has an empty sequence anywhere, then
214                // it's impossible for the concatenantion to ever match. So we
215                // can just quit now.
216                if seq.is_empty() {
217                    return seq;
218                }
219                if seq.is_really_good() {
220                    return seq;
221                }
222                prev = Some(match prev {
223                    None => seq,
224                    Some(prev) => prev.choose(seq),
225                });
226                seq = TSeq::singleton(self::Literal::exact(vec![]));
227                seq.make_not_prefix();
228            }
229            // Note that 'cross' also dispatches based on whether we're
230            // extracting prefixes or suffixes.
231            seq = self.cross(seq, self.extract(hir));
232        }
233        if let Some(prev) = prev { prev.choose(seq) } else { seq }
234    }
235
236    /// Extract a sequence from the given alternation.
237    ///
238    /// This short circuits once the union turns into an infinite sequence.
239    fn extract_alternation<'a, I: Iterator<Item = &'a Hir>>(
240        &self,
241        it: I,
242    ) -> TSeq {
243        let mut seq = TSeq::empty();
244        for hir in it {
245            // Once our 'seq' is infinite, every subsequent union
246            // operation on it will itself always result in an
247            // infinite sequence. Thus, it can never change and we can
248            // short-circuit.
249            if !seq.is_finite() {
250                break;
251            }
252            seq = self.union(seq, &mut self.extract(hir));
253        }
254        seq
255    }

257    /// Extract a sequence of literals from the given repetition. We do our
258    /// best, Some examples:
259    ///
260    ///   'a*'    => [inexact(a), exact("")]
261    ///   'a*?'   => [exact(""), inexact(a)]
262    ///   'a+'    => [inexact(a)]
263    ///   'a{3}'  => [exact(aaa)]
264    ///   'a{3,5} => [inexact(aaa)]
265    ///
266    /// The key here really is making sure we get the 'inexact' vs 'exact'
267    /// attributes correct on each of the literals we add. For example, the
268    /// fact that 'a*' gives us an inexact 'a' and an exact empty string means
269    /// that a regex like 'ab*c' will result in [inexact(ab), exact(ac)]
270    /// literals being extracted, which might actually be a better prefilter
271    /// than just 'a'.
272    fn extract_repetition(&self, rep: &hir::Repetition) -> TSeq {
273        let mut subseq = self.extract(&rep.sub);
274        match *rep {
275            hir::Repetition { min: 0, max, greedy, .. } => {
276                // When 'max=1', we can retain exactness, since 'a?' is
277                // equivalent to 'a|'. Similarly below, 'a??' is equivalent to
278                // '|a'.
279                if max != Some(1) {
280                    subseq.make_inexact();
281                }
282                let mut empty = TSeq::singleton(Literal::exact(vec![]));
283                if !greedy {
284                    core::mem::swap(&mut subseq, &mut empty);
285                }
286                self.union(subseq, &mut empty)
287            }
288            hir::Repetition { min, max: Some(max), .. } if min == max => {
289                assert!(min > 0); // handled above
290                let limit =
291                    u32::try_from(self.limit_repeat).unwrap_or(u32::MAX);
292                let mut seq = TSeq::singleton(Literal::exact(vec![]));
293                for _ in 0..core::cmp::min(min, limit) {
294                    if seq.is_inexact() {
295                        break;
296                    }
297                    seq = self.cross(seq, subseq.clone());
298                }
299                if usize::try_from(min).is_err() || min > limit {
300                    seq.make_inexact();
301                }
302                seq
303            }
304            hir::Repetition { min, max: Some(max), .. } if min < max => {
305                assert!(min > 0); // handled above
306                let limit =
307                    u32::try_from(self.limit_repeat).unwrap_or(u32::MAX);
308                let mut seq = TSeq::singleton(Literal::exact(vec![]));
309                for _ in 0..core::cmp::min(min, limit) {
310                    if seq.is_inexact() {
311                        break;
312                    }
313                    seq = self.cross(seq, subseq.clone());
314                }
315                seq.make_inexact();
316                seq
317            }
318            hir::Repetition { .. } => {
319                subseq.make_inexact();
320                subseq
321            }
322        }
323    }

325    /// Convert the given Unicode class into a sequence of literals if the
326    /// class is small enough. If the class is too big, return an infinite
327    /// sequence.
328    fn extract_class_unicode(&self, cls: &hir::ClassUnicode) -> TSeq {
329        if self.class_over_limit_unicode(cls) {
330            return TSeq::infinite();
331        }
332        let mut seq = TSeq::empty();
333        for r in cls.iter() {
334            for ch in r.start()..=r.end() {
335                seq.push(Literal::from(ch));
336            }
337        }
338        self.enforce_literal_len(&mut seq);
339        seq
340    }

342    /// Convert the given byte class into a sequence of literals if the class
343    /// is small enough. If the class is too big, return an infinite sequence.
344    fn extract_class_bytes(&self, cls: &hir::ClassBytes) -> TSeq {
345        if self.class_over_limit_bytes(cls) {
346            return TSeq::infinite();
347        }
348        let mut seq = TSeq::empty();
349        for r in cls.iter() {
350            for b in r.start()..=r.end() {
351                seq.push(Literal::from(b));
352            }
353        }
354        self.enforce_literal_len(&mut seq);
355        seq
356    }

358    /// Returns true if the given Unicode class exceeds the configured limits
359    /// on this extractor.
360    fn class_over_limit_unicode(&self, cls: &hir::ClassUnicode) -> bool {
361        let mut count = 0;
362        for r in cls.iter() {
363            if count > self.limit_class {
364                return true;
365            }
366            count += r.len();
367        }
368        count > self.limit_class
369    }
370
371    /// Returns true if the given byte class exceeds the configured limits on
372    /// this extractor.
373    fn class_over_limit_bytes(&self, cls: &hir::ClassBytes) -> bool {
374        let mut count = 0;
375        for r in cls.iter() {
376            if count > self.limit_class {
377                return true;
378            }
379            count += r.len();
380        }
381        count > self.limit_class
382    }

384    /// Compute the cross product of the two sequences if the result would be
385    /// within configured limits. Otherwise, make `seq2` infinite and cross the
386    /// infinite sequence with `seq1`.
387    fn cross(&self, mut seq1: TSeq, mut seq2: TSeq) -> TSeq {
388        if !seq2.prefix {
389            return seq1.choose(seq2);
390        }
391        if seq1
392            .max_cross_len(&seq2)
393            .map_or(false, |len| len > self.limit_total)
394        {
395            seq2.make_infinite();
396        }
397        seq1.cross_forward(&mut seq2);
398        assert!(seq1.len().map_or(true, |x| x <= self.limit_total));
399        self.enforce_literal_len(&mut seq1);
400        seq1
401    }
402
403    /// Union the two sequences if the result would be within configured
404    /// limits. Otherwise, make `seq2` infinite and union the infinite sequence
405    /// with `seq1`.
406    fn union(&self, mut seq1: TSeq, seq2: &mut TSeq) -> TSeq {
407        if seq1.max_union_len(seq2).map_or(false, |len| len > self.limit_total)
408        {
409            // We try to trim our literal sequences to see if we can make
410            // room for more literals. The idea is that we'd rather trim down
411            // literals already in our sequence if it means we can add a few
412            // more and retain a finite sequence. Otherwise, we'll union with
413            // an infinite sequence and that infects everything and effectively
414            // stops literal extraction in its tracks.
415            //
416            // We do we keep 4 bytes here? Well, it's a bit of an abstraction
417            // leakage. Downstream, the literals may wind up getting fed to
418            // the Teddy algorithm, which supports searching literals up to
419            // length 4. So that's why we pick that number here. Arguably this
420            // should be a tuneable parameter, but it seems a little tricky to
421            // describe. And I'm still unsure if this is the right way to go
422            // about culling literal sequences.
423            seq1.keep_first_bytes(4);
424            seq2.keep_first_bytes(4);
425            seq1.dedup();
426            seq2.dedup();
427            if seq1
428                .max_union_len(seq2)
429                .map_or(false, |len| len > self.limit_total)
430            {
431                seq2.make_infinite();
432            }
433        }
434        seq1.union(seq2);
435        assert!(seq1.len().map_or(true, |x| x <= self.limit_total));
436        seq1.prefix = seq1.prefix && seq2.prefix;
437        seq1
438    }

440    /// Applies the literal length limit to the given sequence. If none of the
441    /// literals in the sequence exceed the limit, then this is a no-op.
442    fn enforce_literal_len(&self, seq: &mut TSeq) {
443        seq.keep_first_bytes(self.limit_literal_len);
444    }

552    /// Tags this sequence as "not a prefix." When this happens, this sequence
553    /// can't be crossed as a suffix of another sequence.
554    fn make_not_prefix(&mut self) {
555        self.prefix = false;
556    }

558    /// Returns true if it's believed that the sequence given is "good" for
559    /// acceleration. This is useful for determining whether a sequence of
560    /// literals has any shot of being fast.
561    fn is_good(&self) -> bool {
562        if self.has_poisonous_literal() {
563            return false;
564        }
565        let Some(min) = self.min_literal_len() else { return false };
566        let Some(len) = self.len() else { return false };
567        // If we have some very short literals, then let's require that our
568        // sequence is itself very small.
569        if min <= 1 {
570            return len <= 3;
571        }
572        min >= 2 && len <= 64
573    }
574
575    /// Returns true if it's believed that the sequence given is "really
576    /// good" for acceleration. This is useful for short circuiting literal
577    /// extraction.
578    fn is_really_good(&self) -> bool {
579        if self.has_poisonous_literal() {
580            return false;
581        }
582        let Some(min) = self.min_literal_len() else { return false };
583        let Some(len) = self.len() else { return false };
584        min >= 3 && len <= 8
585    }
586
587    /// Returns true if the given sequence contains a poisonous literal.
588    fn has_poisonous_literal(&self) -> bool {
589        let Some(lits) = self.literals() else { return false };
590        lits.iter().any(is_poisonous)
591    }
592
593    /// Compare the two sequences and return the one that is believed to be
594    /// best according to a hodge podge of heuristics.
595    fn choose(self, other: TSeq) -> TSeq {
596        let (mut seq1, mut seq2) = (self, other);
597        // Whichever one we pick, by virtue of picking one, we choose
598        // to not take the other. So we must consider the result inexact.
599        seq1.make_inexact();
600        seq2.make_inexact();
601
602        if !seq1.is_finite() {
603            return seq2;
604        } else if !seq2.is_finite() {
605            return seq1;
606        }
607        if seq1.has_poisonous_literal() {
608            return seq2;
609        } else if seq2.has_poisonous_literal() {
610            return seq1;
611        }
612        let Some(min1) = seq1.min_literal_len() else { return seq2 };
613        let Some(min2) = seq2.min_literal_len() else { return seq1 };
614        if min1 < min2 {
615            return seq2;
616        } else if min2 < min1 {
617            return seq1;
618        }
619        // OK because we know both sequences are finite, otherwise they wouldn't
620        // have a minimum literal length.
621        let len1 = seq1.len().unwrap();
622        let len2 = seq2.len().unwrap();
623        if len1 < len2 {
624            return seq2;
625        } else if len2 < len1 {
626            return seq1;
627        }
628        // We could do extra stuff like looking at a background frequency
629        // distribution of bytes and picking the one that looks more rare, but for
630        // now we just pick one.
631        seq1
632    }
633}

641/// Returns true if it is believe that this literal is likely to match very
642/// frequently, and is thus not a good candidate for a prefilter.
643fn is_poisonous(lit: &Literal) -> bool {
644    use regex_syntax::hir::literal::rank;
645
646    lit.is_empty() || (lit.len() == 1 && rank(lit.as_bytes()[0]) >= 250)
647}

896    fn impossible() {
897        // N.B. The extractor in this module "optimizes" the sequence and makes
898        // it infinite if it isn't "good." An empty sequence (generated by a
899        // concatenantion containing an expression that can never match) is
900        // considered "not good." Since infinite sequences are not actionably
901        // and disable optimizations, this winds up being okay.
902        //
903        // The literal extractor in regex-syntax doesn't combine these two
904        // steps and makes the caller choose to optimize. That is, it returns
905        // the sequences as they are. Which in this case, for some of the tests
906        // below, would be an empty Seq and not an infinite Seq.
907        assert_eq!(Seq::infinite(), e(r"[a&&b]"));
908        assert_eq!(Seq::infinite(), e(r"a[a&&b]"));
909        assert_eq!(Seq::infinite(), e(r"[a&&b]b"));
910        assert_eq!(Seq::infinite(), e(r"a[a&&b]b"));
911        assert_eq!(exact(["a", "b"]), e(r"a|[a&&b]|b"));
912        assert_eq!(exact(["a", "b"]), e(r"a|c[a&&b]|b"));
913        assert_eq!(exact(["a", "b"]), e(r"a|[a&&b]d|b"));
914        assert_eq!(exact(["a", "b"]), e(r"a|c[a&&b]d|b"));

919    // This tests patterns that contain something that defeats literal
920    // detection, usually because it would blow some limit on the total number
921    // of literals that can be returned.
922    //
923    // The main idea is that when literal extraction sees something that
924    // it knows will blow a limit, it replaces it with a marker that says
925    // "any literal will match here." While not necessarily true, the
926    // over-estimation is just fine for the purposes of literal extraction,
927    // because the imprecision doesn't matter: too big is too big.
928    //
929    // This is one of the trickier parts of literal extraction, since we need
930    // to make sure all of our literal extraction operations correctly compose
931    // with the markers.
932    //
933    // Note that unlike in regex-syntax, some of these have "inner" literals
934    // extracted where a prefix or suffix would otherwise not be found.
935    #[test]
936    fn anything() {
937        assert_eq!(Seq::infinite(), e(r"."));
938        assert_eq!(Seq::infinite(), e(r"(?s)."));
939        assert_eq!(Seq::infinite(), e(r"[A-Za-z]"));
940        assert_eq!(Seq::infinite(), e(r"[A-Z]"));
941        assert_eq!(Seq::infinite(), e(r"[A-Z]{0}"));
942        assert_eq!(Seq::infinite(), e(r"[A-Z]?"));
943        assert_eq!(Seq::infinite(), e(r"[A-Z]*"));
944        assert_eq!(Seq::infinite(), e(r"[A-Z]+"));
945        assert_eq!(seq([I("1")]), e(r"1[A-Z]"));
946        assert_eq!(seq([I("1")]), e(r"1[A-Z]2"));
947        assert_eq!(seq([I("123")]), e(r"[A-Z]+123"));
948        assert_eq!(seq([I("123")]), e(r"[A-Z]+123[A-Z]+"));
949        assert_eq!(Seq::infinite(), e(r"1|[A-Z]|3"));
950        assert_eq!(seq([E("1"), I("2"), E("3")]), e(r"1|2[A-Z]|3"),);
951        assert_eq!(seq([E("1"), I("2"), E("3")]), e(r"1|[A-Z]2[A-Z]|3"),);
952        assert_eq!(seq([E("1"), I("2"), E("3")]), e(r"1|[A-Z]2|3"),);
953        assert_eq!(seq([E("1"), I("2"), E("4")]), e(r"1|2[A-Z]3|4"),);
954        assert_eq!(seq([I("2")]), e(r"(?:|1)[A-Z]2"));

989    #[test]
990    fn optimize() {
991        // This gets a common prefix that isn't too short.
992        let s = e(r"foobarfoobar|foobar|foobarzfoobar|foobarfoobar");
993        assert_eq!(seq([I("foobar")]), s);
994
995        // This also finds a common prefix, but since it's only one byte, it
996        // prefers the multiple literals.
997        let s = e(r"abba|akka|abccba");
998        assert_eq!(exact(["abba", "akka", "abccba"]), s);
999
1000        let s = e(r"sam|samwise");
1001        assert_eq!(seq([E("sam")]), s);
1002
1003        // The empty string is poisonous, so our seq becomes infinite, even
1004        // though all literals are exact.
1005        let s = e(r"foobarfoo|foo||foozfoo|foofoo");
1006        assert_eq!(Seq::infinite(), s);
1007
1008        // A space is also poisonous, so our seq becomes infinite. But this
1009        // only gets triggered when we don't have a completely exact sequence.
1010        // When the sequence is exact, spaces are okay, since we presume that
1011        // any prefilter will match a space more quickly than the regex engine.
1012        // (When the sequence is exact, there's a chance of the prefilter being
1013        // used without needing the regex engine at all.)
1014        let s = e(r"foobarfoo|foo| |foofoo");
1015        assert_eq!(Seq::infinite(), s);
1016    }
