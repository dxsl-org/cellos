23/// Config represents the configuration of a regex matcher in this crate.
24/// The configuration is itself a rough combination of the knobs found in
25/// the `regex` crate itself, along with additional `grep-matcher` specific
26/// options.
27///
28/// The configuration can be used to build a "configured" HIR expression. A
29/// configured HIR expression is an HIR expression that is aware of the
30/// configuration which generated it, and provides transformation on that HIR
31/// such that the configuration is preserved.
32#[derive(Clone, Debug)]
33pub(crate) struct Config {
34    pub(crate) case_insensitive: bool,
35    pub(crate) case_smart: bool,
36    pub(crate) multi_line: bool,
37    pub(crate) dot_matches_new_line: bool,
38    pub(crate) swap_greed: bool,
39    pub(crate) ignore_whitespace: bool,
40    pub(crate) unicode: bool,
41    pub(crate) octal: bool,
42    pub(crate) size_limit: usize,
43    pub(crate) dfa_size_limit: usize,
44    pub(crate) nest_limit: u32,
45    pub(crate) line_terminator: Option<LineTerminator>,
46    pub(crate) ban: Option<u8>,
47    pub(crate) crlf: bool,
48    pub(crate) word: bool,
49    pub(crate) fixed_strings: bool,
50    pub(crate) whole_line: bool,
51}
52
53impl Default for Config {
54    fn default() -> Config {
55        Config {
56            case_insensitive: false,
57            case_smart: false,
58            multi_line: false,
59            dot_matches_new_line: false,
60            swap_greed: false,
61            ignore_whitespace: false,
62            unicode: true,
63            octal: false,
64            // These size limits are much bigger than what's in the regex
65            // crate by default.
66            size_limit: 100 * (1 << 20),
67            dfa_size_limit: 1000 * (1 << 20),
68            nest_limit: 250,
69            line_terminator: None,
70            ban: None,
71            crlf: false,
72            word: false,
73            fixed_strings: false,
74            whole_line: false,
75        }
76    }

80    /// Use this configuration to build an HIR from the given patterns. The HIR
81    /// returned corresponds to a single regex that is an alternation of the
82    /// patterns given.
83    pub(crate) fn build_many<P: AsRef<str>>(
84        &self,
85        patterns: &[P],
86    ) -> Result<ConfiguredHIR, Error> {
87        ConfiguredHIR::new(self.clone(), patterns)
88    }

90    /// Accounting for the `smart_case` config knob, return true if and only if
91    /// this pattern should be matched case insensitively.
92    fn is_case_insensitive(&self, analysis: &AstAnalysis) -> bool {
93        if self.case_insensitive {
94            return true;
95        }
96        if !self.case_smart {
97            return false;
98        }
99        analysis.any_literal() && !analysis.any_uppercase()
100    }
101
102    /// Returns whether the given patterns should be treated as "fixed strings"
103    /// literals. This is different from just querying the `fixed_strings` knob
104    /// in that if the knob is false, this will still return true in some cases
105    /// if the patterns are themselves indistinguishable from literals.
106    ///
107    /// The main idea here is that if this returns true, then it is safe
108    /// to build an `regex_syntax::hir::Hir` value directly from the given
109    /// patterns as an alternation of `hir::Literal` values.
110    fn is_fixed_strings<P: AsRef<str>>(&self, patterns: &[P]) -> bool {
111        // When these are enabled, we really need to parse the patterns and
112        // let them go through the standard HIR translation process in order
113        // for case folding transforms to be applied.
114        if self.case_insensitive || self.case_smart {
115            return false;
116        }
117        // Even if whole_line or word is enabled, both of those things can
118        // be implemented by wrapping the Hir generated by an alternation of
119        // fixed string literals. So for here at least, we don't care about the
120        // word or whole_line settings.
121        if self.fixed_strings {
122            // ... but if any literal contains a line terminator, then we've
123            // got to bail out because this will ultimately result in an error.
124            if let Some(lineterm) = self.line_terminator {
125                for p in patterns.iter() {
126                    if has_line_terminator(lineterm, p.as_ref()) {
127                        return false;
128                    }
129                }
130            }
131            return true;
132        }
133        // In this case, the only way we can hand construct the Hir is if none
134        // of the patterns contain meta characters. If they do, then we need to
135        // send them through the standard parsing/translation process.
136        for p in patterns.iter() {
137            let p = p.as_ref();
138            if p.chars().any(regex_syntax::is_meta_character) {
139                return false;
140            }
141            // Same deal as when fixed_strings is set above. If the pattern has
142            // a line terminator anywhere, then we need to bail out and let
143            // an error occur.
144            if let Some(lineterm) = self.line_terminator {
145                if has_line_terminator(lineterm, p) {
146                    return false;
147                }
148            }
149        }
150        true
151    }
152}

154/// A "configured" HIR expression, which is aware of the configuration which
155/// produced this HIR.
156///
157/// Since the configuration is tracked, values with this type can be
158/// transformed into other HIR expressions (or regular expressions) in a way
159/// that preserves the configuration. For example, the `fast_line_regex`
160/// method will apply literal extraction to the inner HIR and use that to build
161/// a new regex that matches the extracted literals in a way that is
162/// consistent with the configuration that produced this HIR. For example, the
163/// size limits set on the configured HIR will be propagated out to any
164/// subsequently constructed HIR or regular expression.
165#[derive(Clone, Debug)]
166pub(crate) struct ConfiguredHIR {
167    config: Config,
168    hir: Hir,
169}
170
171impl ConfiguredHIR {
172    /// Parse the given patterns into a single HIR expression that represents
173    /// an alternation of the patterns given.
174    fn new<P: AsRef<str>>(
175        config: Config,
176        patterns: &[P],
177    ) -> Result<ConfiguredHIR, Error> {
178        let hir = if config.is_fixed_strings(patterns) {
179            let mut alts = vec![];
180            for p in patterns.iter() {
181                alts.push(Hir::literal(p.as_ref().as_bytes()));
182            }
183            log::debug!(
184                "assembling HIR from {} fixed string literals",
185                alts.len()
186            );
187            let hir = Hir::alternation(alts);
188            hir
189        } else {
190            let mut alts = vec![];
191            for p in patterns.iter() {
192                alts.push(if config.fixed_strings {
193                    format!("(?:{})", regex_syntax::escape(p.as_ref()))
194                } else {
195                    format!("(?:{})", p.as_ref())
196                });
197            }
198            let pattern = alts.join("|");
199            let ast = ast::parse::ParserBuilder::new()
200                .nest_limit(config.nest_limit)
201                .octal(config.octal)
202                .ignore_whitespace(config.ignore_whitespace)
203                .build()
204                .parse(&pattern)
205                .map_err(Error::generic)?;
206            let analysis = AstAnalysis::from_ast(&ast);
207            let mut hir = hir::translate::TranslatorBuilder::new()
208                .utf8(false)
209                .case_insensitive(config.is_case_insensitive(&analysis))
210                .multi_line(config.multi_line)
211                .dot_matches_new_line(config.dot_matches_new_line)
212                .crlf(config.crlf)
213                .swap_greed(config.swap_greed)
214                .unicode(config.unicode)
215                .build()
216                .translate(&pattern, &ast)
217                .map_err(Error::generic)?;
218            if let Some(byte) = config.ban {
219                ban::check(&hir, byte)?;
220            }
221            // We don't need to do this for the fixed-strings case above
222            // because is_fixed_strings will return false if any pattern
223            // contains a line terminator. Therefore, we don't need to strip
224            // it.
225            //
226            // We go to some pains to avoid doing this in the fixed-strings
227            // case because this can result in building a new HIR when ripgrep
228            // is given a huge set of literals to search for. And this can
229            // actually take a little time. It's not huge, but it's noticeable.
230            hir = match config.line_terminator {
231                None => hir,
232                Some(line_term) => strip_from_match(hir, line_term)?,
233            };
234            hir
235        };
236        Ok(ConfiguredHIR { config, hir })
237    }

239    /// Return a reference to the underlying configuration.
240    pub(crate) fn config(&self) -> &Config {
241        &self.config
242    }
243
244    /// Return a reference to the underlying HIR.
245    pub(crate) fn hir(&self) -> &Hir {
246        &self.hir
247    }
248
249    /// Convert this HIR to a regex that can be used for matching.
250    pub(crate) fn to_regex(&self) -> Result<Regex, Error> {
251        let meta = Regex::config()
252            .utf8_empty(false)
253            .nfa_size_limit(Some(self.config.size_limit))
254            // Cellos adaptation (S7-3, D76-2): upstream fixes the one-pass
255            // DFA limit at 10 MiB; here it follows the configured NFA size
256            // limit so a small `size_limit` bounds every compiled engine.
257            .onepass_size_limit(Some(self.config.size_limit))
258            // Same deal here. The default limit for full DFAs is VERY small,
259            // but with ripgrep we can afford to spend a bit more time on
260            // building them I think.
261            .dfa_size_limit(Some(1 * (1 << 20)))
262            .dfa_state_limit(Some(1_000))
263            .hybrid_cache_capacity(self.config.dfa_size_limit);
264        Regex::builder()
265            .configure(meta)
266            .build_from_hir(&self.hir)
267            .map_err(Error::regex)
268    }

271    pub(crate) fn non_matching_bytes(&self) -> ByteSet {
272        non_matching_bytes(&self.hir)
273    }

275    /// Returns the line terminator configured on this expression.
276    ///
277    /// When we have beginning/end anchors (NOT line anchors), the fast line
278    /// searching path isn't quite correct. Or at least, doesn't match the slow
279    /// path. Namely, the slow path strips line terminators while the fast path
280    /// does not. Since '$' (when multi-line mode is disabled) doesn't match at
281    /// line boundaries, the existence of a line terminator might cause it to
282    /// not match when it otherwise would with the line terminator stripped.
283    ///
284    /// Since searching with text anchors is exceptionally rare in the context
285    /// of line oriented searching (multi-line mode is basically always
286    /// enabled), we just disable this optimization when there are text
287    /// anchors. We disable it by not returning a line terminator, since
288    /// without a line terminator, the fast search path can't be executed.
289    ///
290    /// Actually, the above is no longer quite correct. Later on, another
291    /// optimization was added where if the line terminator was in the set of
292    /// bytes that was guaranteed to never be part of a match, then the higher
293    /// level search infrastructure assumes that the fast line-by-line search
294    /// path can still be taken. This optimization applies when multi-line
295    /// search (not multi-line mode) is enabled. In that case, there is no
296    /// configured line terminator since the regex is permitted to match a
297    /// line terminator. But if the regex is guaranteed to never match across
298    /// multiple lines despite multi-line search being requested, we can still
299    /// do the faster and more flexible line-by-line search. This is why the
300    /// non-matching extraction routine removes `\n` when `\A` and `\z` are
301    /// present even though that's not quite correct...
302    ///
303    /// See: <https://github.com/BurntSushi/ripgrep/issues/2260>
304    pub(crate) fn line_terminator(&self) -> Option<LineTerminator> {
305        if self.hir.properties().look_set().contains_anchor_haystack() {
306            None
307        } else {
308            self.config.line_terminator
309        }
310    }
311
312    /// Turns this configured HIR into an equivalent one, but where it must
313    /// match at the start and end of a line.
314    pub(crate) fn into_whole_line(self) -> ConfiguredHIR {
315        let line_anchor_start = Hir::look(self.line_anchor_start());
316        let line_anchor_end = Hir::look(self.line_anchor_end());
317        let hir =
318            Hir::concat(vec![line_anchor_start, self.hir, line_anchor_end]);
319        ConfiguredHIR { config: self.config, hir }
320    }
321
322    /// Turns this configured HIR into an equivalent one, but where it must
323    /// match at word boundaries.
324    pub(crate) fn into_word(self) -> ConfiguredHIR {
325        let hir = Hir::concat(vec![
326            Hir::look(if self.config.unicode {
327                hir::Look::WordStartHalfUnicode
328            } else {
329                hir::Look::WordStartHalfAscii
330            }),
331            self.hir,
332            Hir::look(if self.config.unicode {
333                hir::Look::WordEndHalfUnicode
334            } else {
335                hir::Look::WordEndHalfAscii
336            }),
337        ]);
338        ConfiguredHIR { config: self.config, hir }
339    }
340
341    /// Returns the "start line" anchor for this configuration.
342    fn line_anchor_start(&self) -> hir::Look {
343        if self.config.crlf {
344            hir::Look::StartCRLF
345        } else {
346            hir::Look::StartLF
347        }
348    }
349
350    /// Returns the "end line" anchor for this configuration.
351    fn line_anchor_end(&self) -> hir::Look {
352        if self.config.crlf { hir::Look::EndCRLF } else { hir::Look::EndLF }
353    }
354}

356/// Returns true if the given literal string contains any byte from the line
357/// terminator given.
358fn has_line_terminator(lineterm: LineTerminator, literal: &str) -> bool {
359    if lineterm.is_crlf() {
360        literal.as_bytes().iter().copied().any(|b| b == b'\r' || b == b'\n')
361    } else {
362        literal.as_bytes().iter().copied().any(|b| b == lineterm.as_byte())
363    }
