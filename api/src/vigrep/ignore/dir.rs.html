6// This module provides a data structure, `Ignore`, that connects "directory
7// traversal" with "ignore matchers." Specifically, it knows about gitignore
8// semantics and precedence, and is organized based on directory hierarchy.
9// Namely, every matcher logically corresponds to ignore rules from a single
10// directory, and points to the matcher for its corresponding parent directory.
11// In this sense, `Ignore` is a *persistent* data structure.
12//
13// This design was specifically chosen to make it possible to use this data
14// structure in a parallel directory iterator.
15//
16// My initial intention was to expose this module as part of this crate's
17// public API, but I think the data structure's public API is too complicated
18// with non-obvious failure modes. Alas, such things haven't been documented
19// well.
20//
21// vigrep: only the precedence logic is kept, re-expressed for the serial
22// walker in `crate::walk`:
23//
24// * The caller lists a directory and passes the bytes of the ignore files it
25//   found there to `Ignore::add_child` (upstream's `add_child_with_entries`
26//   probed and opened them itself). Nothing here touches a filesystem.
27// * Rules are discovered from the walk root downward only. Upstream's
28//   `add_parents` (ancestor-above-root discovery and its cross-root cache),
29//   global gitignore, `.git/info/exclude`, explicit `--ignore-file` matchers,
30//   file types and repository (`.git`/`.jj`) detection are not ported, so
31//   `.gitignore` rules apply as with upstream's `require_git(false)` and are
32//   not stopped by a nested repository.
33// * Every error from loading an ignore file is reported. Upstream drops I/O
34//   errors here (`maybe_push_ignore_io`), which made a non-UTF-8 line end
35//   rule loading silently.
36// * The precedence is upstream's: overrides (command globs) first; then, per
37//   category, the deepest directory's match wins, and the categories combine
38//   as custom ignore files (`.rgignore`) > `.ignore` > `.gitignore`; hidden
39//   entries are ignored only when nothing else matched.

53/// IgnoreMatch represents information about where a match came from when using
54/// the `Ignore` matcher.
55#[derive(Clone, Debug)]
56#[allow(dead_code)]
57pub(crate) struct IgnoreMatch<'a>(IgnoreMatchInner<'a>);
58
59/// IgnoreMatchInner describes precisely where the match information came from.
60/// This is private to allow expansion to more matchers in the future.
61#[derive(Clone, Debug)]
62#[allow(dead_code)]
63enum IgnoreMatchInner<'a> {
64    Override(overrides::Glob<'a>),
65    Gitignore(&'a gitignore::Glob),
66    Hidden,
67}
68
69impl<'a> IgnoreMatch<'a> {
70    fn overrides(x: overrides::Glob<'a>) -> IgnoreMatch<'a> {
71        IgnoreMatch(IgnoreMatchInner::Override(x))
72    }
73
74    fn gitignore(x: &'a gitignore::Glob) -> IgnoreMatch<'a> {
75        IgnoreMatch(IgnoreMatchInner::Gitignore(x))
76    }
77
78    fn hidden() -> IgnoreMatch<'static> {
79        IgnoreMatch(IgnoreMatchInner::Hidden)
80    }
81}

103#[derive(Clone, Debug)]
104struct IgnoreInner {
105    /// An override matcher (default is empty).
106    overrides: Arc<Override>,
107    /// The parent directory to match next.
108    ///
109    /// If this is the root directory or there are otherwise no more
110    /// directories to match, then `parent` is `None`.

112    /// Ignore files used in addition to `.ignore`
113    custom_ignore_filenames: Arc<Vec<Vec<u8>>>,
114    /// The matcher for custom ignore files
115    custom_ignore_matcher: Gitignore,
116    /// The matcher for .ignore files.

119    git_ignore_matcher: Gitignore,

124impl Ignore {
125    /// The names of the ignore files this matcher reads in each directory, in
126    /// no particular order. The walker reads the ones present in a listing
127    /// and hands their contents to [`Ignore::add_child`].
128    pub(crate) fn ignore_file_names(&self) -> Vec<&[u8]> {
129        let mut names: Vec<&[u8]> = self
130            .inner
131            .custom_ignore_filenames
132            .iter()
133            .map(|n| n.as_slice())
134            .collect();
135        if self.inner.opts.ignore {
136            names.push(b".ignore");
137        }
138        if self.inner.opts.git_ignore {
139            names.push(b".gitignore");
140        }
141        names
142    }

144    /// Create a new `Ignore` matcher for the given child directory.
145    ///
146    /// `files` holds the `(file name, contents)` of the ignore files found
147    /// in `dir`; entries whose name is not one of
148    /// [`Ignore::ignore_file_names`] are not used.
149    ///
150    /// Since building the matcher may require reading from multiple
151    /// files, it's possible that this method partially succeeds. Therefore,
152    /// a matcher is always returned (which may match nothing) and an error is
153    /// returned if it exists.
154    pub(crate) fn add_child<P: AsRef<[u8]>>(
155        &self,
156        dir: P,
157        files: &[(&[u8], &[u8])],
158    ) -> (Ignore, Option<Error>) {
159        let (ig, err) =
160            self.add_child_path_with_found_ignore_files(dir.as_ref(), files);
161        (Ignore { inner: Arc::new(ig) }, err)
162    }
163
164    fn add_child_path_with_found_ignore_files(
165        &self,
166        dir: &[u8],
167        files: &[(&[u8], &[u8])],
168    ) -> (IgnoreInner, Option<Error>) {
169        let mut errs = PartialErrorBuilder::default();
170        let custom_ig_matcher =
171            if self.inner.custom_ignore_filenames.is_empty() {
172                Gitignore::empty()
173            } else {
174                let custom_ignore_names: Vec<&[u8]> = self
175                    .inner
176                    .custom_ignore_filenames
177                    .iter()
178                    .map(|n| n.as_slice())
179                    .collect();
180                let (m, err) = create_gitignore(
181                    dir,
182                    files,
183                    &custom_ignore_names,
184                    self.inner.opts.ignore_case_insensitive,
185                );
186                errs.maybe_push(err);
187                m
188            };
189        let ig_matcher = if !self.inner.opts.ignore {
190            Gitignore::empty()
191        } else {
192            let (m, err) = create_gitignore(
193                dir,
194                files,
195                &[b".ignore"],
196                self.inner.opts.ignore_case_insensitive,
197            );
198            errs.maybe_push(err);
199            m
200        };
201        let gi_matcher = if !self.inner.opts.git_ignore {
202            Gitignore::empty()
203        } else {
204            let (m, err) = create_gitignore(
205                dir,
206                files,
207                &[b".gitignore"],
208                self.inner.opts.ignore_case_insensitive,
209            );
210            errs.maybe_push(err);
211            m
212        };
213        let ig = IgnoreInner {
214            overrides: self.inner.overrides.clone(),
215            parent: Some(self.inner.clone()),
216            custom_ignore_filenames: self
217                .inner
218                .custom_ignore_filenames
219                .clone(),
220            custom_ignore_matcher: custom_ig_matcher,
221            ignore_matcher: ig_matcher,
222            git_ignore_matcher: gi_matcher,
223            opts: self.inner.opts,
224        };
225        (ig, errs.into_error_option())
226    }

228    /// Returns true if at least one type of ignore rule should be matched.
229    fn has_any_ignore_rules(&self) -> bool {
230        let opts = self.inner.opts;
231        let has_custom_ignore_files =
232            !self.inner.custom_ignore_filenames.is_empty();
233
234        opts.ignore || opts.git_ignore || has_custom_ignore_files
235    }

237    /// Like `matched`, but for a walk entry: hidden entries are ignored when
238    /// nothing else matched.
239    ///
240    /// vigrep: takes the entry's path and kind instead of a walk `DirEntry`.
241    pub(crate) fn matched_dir_entry<'a>(
242        &'a self,
243        path: &[u8],
244        is_dir: bool,
245    ) -> Match<IgnoreMatch<'a>> {
246        let m = self.matched(path, is_dir);
247        if m.is_none() && self.inner.opts.hidden && is_hidden(path) {
248            return Match::Ignore(IgnoreMatch::hidden());
249        }
250        m
251    }
252
253    /// Returns a match indicating whether the given file path should be
254    /// ignored or not.
255    ///
256    /// The match contains information about its origin.
257    fn matched<'a, P: AsRef<[u8]>>(
258        &'a self,
259        path: P,
260        is_dir: bool,
261    ) -> Match<IgnoreMatch<'a>> {
262        // We need to be careful with our path. If it has a leading ./, then
263        // strip it because it causes nothing but trouble.
264        let mut path = path.as_ref();
265        if let Some(p) = strip_prefix("./", path) {
266            path = p;
267        }
268        // Match against the override patterns. If an override matches
269        // regardless of whether it's whitelist/ignore, then we quit and
270        // return that result immediately. Overrides have the highest
271        // precedence.
272        if !self.inner.overrides.is_empty() {
273            let mat = self
274                .inner
275                .overrides
276                .matched(path, is_dir)
277                .map(IgnoreMatch::overrides);
278            if !mat.is_none() {
279                return mat;
280            }
281        }
282        let mut whitelisted = Match::None;
283        if self.has_any_ignore_rules() {
284            let mat = self.matched_ignore(path, is_dir);
285            if mat.is_ignore() {
286                return mat;
287            } else if mat.is_whitelist() {
288                whitelisted = mat;
289            }
290        }
291        whitelisted
292    }
293
294    /// Performs matching only on the ignore files for this directory and
295    /// all parent directories.
296    fn matched_ignore<'a>(
297        &'a self,
298        path: &[u8],
299        is_dir: bool,
300    ) -> Match<IgnoreMatch<'a>> {
301        let (mut m_custom_ignore, mut m_ignore, mut m_gi) =
302            (Match::None, Match::None, Match::None);
303        for ig in self.parents() {
304            if m_custom_ignore.is_none() {
305                m_custom_ignore = ig
306                    .inner
307                    .custom_ignore_matcher
308                    .matched(path, is_dir)
309                    .map(IgnoreMatch::gitignore);
310            }
311            if m_ignore.is_none() {
312                m_ignore = ig
313                    .inner
314                    .ignore_matcher
315                    .matched(path, is_dir)
316                    .map(IgnoreMatch::gitignore);
317            }
318            if m_gi.is_none() {
319                m_gi = ig
320                    .inner
321                    .git_ignore_matcher
322                    .matched(path, is_dir)
323                    .map(IgnoreMatch::gitignore);
324            }
325        }
326
327        m_custom_ignore.or(m_ignore).or(m_gi)
328    }

330    /// Returns an iterator over parent ignore matchers, including this one.
331    pub(crate) fn parents(&self) -> Parents<'_> {
332        Parents(Some(IgnoreRef { inner: &self.inner }))
333    }

341/// An iterator over all parents of an ignore matcher, including itself.
342pub(crate) struct Parents<'a>(Option<IgnoreRef<'a>>);
343
344impl<'a> Iterator for Parents<'a> {
345    type Item = IgnoreRef<'a>;
346
347    fn next(&mut self) -> Option<IgnoreRef<'a>> {
348        match self.0.take() {
349            None => None,
350            Some(ig) => {
351                self.0 = ig
352                    .inner
353                    .parent
354                    .as_deref()
355                    .map(|inner| IgnoreRef { inner });
356                Some(ig)
357            }
358        }
359    }

388    /// Builds a new `Ignore` matcher.
389    ///
390    /// The matcher returned won't match anything until ignore rules from
391    /// directories are added to it.

394            inner: Arc::new(IgnoreInner {
395                overrides: self.overrides.clone(),
396                parent: None,

398                    self.custom_ignore_filenames.clone(),

400                custom_ignore_matcher: Gitignore::empty(),
401                ignore_matcher: Gitignore::empty(),
402                git_ignore_matcher: Gitignore::empty(),
403                opts: self.opts,

405        }

478/// Creates a new gitignore matcher for the directory given.
479///
480/// The matcher is meant to match files below `dir`.
481/// Ignore globs are extracted from each of the named files found in `files`
482/// in the order of `names` (earlier names have lower precedence than later
483/// names).
484///
485/// vigrep: `files` holds the contents the caller read; every error is
486/// reported (upstream drops I/O errors here).
487pub(crate) fn create_gitignore(
488    dir: &[u8],
489    files: &[(&[u8], &[u8])],
490    names: &[&[u8]],
491    case_insensitive: bool,
492) -> (Gitignore, Option<Error>) {
493    let mut builder = GitignoreBuilder::new(dir);
494    let mut errs = PartialErrorBuilder::default();
495    builder.case_insensitive(case_insensitive).unwrap();
496    for name in names {
497        for &(file_name, contents) in files {
498            if file_name == *name {
499                errs.maybe_push(builder.add_bytes(join(dir, name), contents));
500            }
501        }
502    }
503    let gi = match builder.build() {
504        Ok(gi) => gi,
505        Err(err) => {
506            errs.push(err);
507            GitignoreBuilder::new(dir).build().unwrap()
508        }
509    };
510    (gi, errs.into_error_option())
511}

686    // This tests that .gitignore files beyond a .git barrier aren't
687    // matched, but .ignore files are.
688    //
689    // vigrep: there is no repository detection, so the nested `foo/.git`
690    // is not a barrier and the ancestor `.gitignore` still applies.
691    #[test]
692    fn stops_at_git_dir() {
693        let ig0 = IgnoreBuilder::new().build();
694        let (ig1, err) = ig0.add_child(
695            TD,
696            &files(&[(".git", ""), (".gitignore", "foo"), (".ignore", "bar")]),
697        );
698        assert!(err.is_none());
699        let foo = [TD.as_bytes(), b"/foo"].concat();
700        let (ig2, err) = ig1.add_child(&foo, &files(&[(".git", "")]));
701        assert!(err.is_none());
702
703        assert!(ig1.matched("foo", false).is_ignore());
704        assert!(ig2.matched("foo", false).is_ignore());
705
706        assert!(ig1.matched("bar", false).is_ignore());
707        assert!(ig2.matched("bar", false).is_ignore());
708    }

710    // vigrep: a non-UTF-8 rule line is an error. Upstream reads the file
711    // with `BufRead::lines`, gets an I/O error and drops it, so rule loading
712    // stopped silently at that line.
713    #[test]
714    fn invalid_utf8_rule_is_reported() {
715        let ig0 = IgnoreBuilder::new().build();
716        let (ig, err) = ig0.add_child(
717            TD,
718            &[(b".gitignore".as_slice(), b"foo\n\xFF\nbar".as_slice())],

720        let err = err.expect("an error");
721        assert!(err.is_io());
722        assert!(ig.matched("foo", false).is_ignore());
723        assert!(ig.matched("bar", false).is_none());

727    fn deeper_directory_overrides_ancestor() {
728        let ig0 = IgnoreBuilder::new().build();
729        let (ig1, _) = ig0.add_child(TD, &files(&[(".gitignore", "*.log")]));
730        let sub = [TD.as_bytes(), b"/sub"].concat();
731        let (ig2, _) =
732            ig1.add_child(&sub, &files(&[(".gitignore", "!keep.log")]));
733        let keep = [sub.as_slice(), b"/keep.log"].concat();
734        let other = [sub.as_slice(), b"/other.log"].concat();
