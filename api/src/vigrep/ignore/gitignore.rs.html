6/*!
7The gitignore module provides a way to match globs from a gitignore file
8against file paths.
9
10Note that this module implements the specification as described in the
11`gitignore` man page from scratch. That is, this module does *not* shell out to
12the `git` command line tool.
13
14vigrep: paths are bytes and rules come from bytes supplied by the caller
15([`GitignoreBuilder::add_bytes`], [`GitignoreBuilder::add_line`]); nothing
16here opens a file, reads git configuration or consults the environment, so
17upstream's `Gitignore::new`, `Gitignore::global`, `build_global` and
18`gitconfig_excludes_path` are not ported.
19*/

36/// Glob represents a single glob in a gitignore file.
37///
38/// This is used to report information about the highest precedent glob that
39/// matched in one or more gitignore files.
40#[derive(Clone, Debug)]
41pub struct Glob {
42    /// The file path that this glob was extracted from.
43    from: Option<Vec<u8>>,
44    /// The original glob string.
45    original: String,
46    /// The actual glob string used to convert to a regex.
47    actual: String,
48    /// Whether this is a whitelisted glob or not.
49    is_whitelist: bool,
50    /// Whether this glob should only match directories or not.
51    is_only_dir: bool,
52}
53
54impl Glob {
55    /// Returns the file path that defined this glob.
56    pub fn from(&self) -> Option<&[u8]> {
57        self.from.as_deref()
58    }
59
60    /// The original glob as it was defined in a gitignore file.
61    pub fn original(&self) -> &str {
62        &self.original
63    }
64
65    /// The actual glob that was compiled to respect gitignore
66    /// semantics.
67    pub fn actual(&self) -> &str {
68        &self.actual
69    }
70
71    /// Whether this was a whitelisted glob or not.
72    pub fn is_whitelist(&self) -> bool {
73        self.is_whitelist
74    }
75
76    /// Whether this glob must match a directory or not.
77    pub fn is_only_dir(&self) -> bool {
78        self.is_only_dir
79    }
80
81    /// Returns true if and only if this glob has a `**/` prefix.
82    fn has_doublestar_prefix(&self) -> bool {
83        self.actual.starts_with("**/") || self.actual == "**"
84    }
85}

87/// Gitignore is a matcher for the globs in one or more gitignore files
88/// in the same directory.
89#[derive(Clone, Debug)]
90pub struct Gitignore {
91    set: GlobSet,
92    root: Vec<u8>,
93    globs: Vec<Glob>,
94    num_ignores: u64,
95    num_whitelists: u64,
96    matches: Option<Arc<Pool<Vec<usize>>>>,
97}
98
99impl Gitignore {
100    /// Creates a new empty gitignore matcher that never matches anything.
101    ///
102    /// Its path is empty.
103    pub fn empty() -> Gitignore {
104        Gitignore {
105            set: GlobSet::empty(),
106            root: Vec::new(),
107            globs: vec![],
108            num_ignores: 0,
109            num_whitelists: 0,
110            matches: None,
111        }
112    }
113
114    /// Returns the directory containing this gitignore matcher.
115    ///
116    /// All matches are done relative to this path.
117    pub fn path(&self) -> &[u8] {
118        &self.root
119    }
120
121    /// Returns true if and only if this gitignore has zero globs, and
122    /// therefore never matches any file path.
123    pub fn is_empty(&self) -> bool {
124        self.set.is_empty()
125    }
126
127    /// Returns the total number of globs, which should be equivalent to
128    /// `num_ignores + num_whitelists`.
129    pub fn len(&self) -> usize {
130        self.set.len()
131    }
132
133    /// Returns the total number of ignore globs.
134    pub fn num_ignores(&self) -> u64 {
135        self.num_ignores
136    }
137
138    /// Returns the total number of whitelisted globs.
139    pub fn num_whitelists(&self) -> u64 {
140        self.num_whitelists
141    }
142
143    /// Returns whether the given path (file or directory) matched a pattern in
144    /// this gitignore matcher.
145    ///
146    /// `is_dir` should be true if the path refers to a directory and false
147    /// otherwise.
148    ///
149    /// The given path is matched relative to the path given when building
150    /// the matcher. Specifically, before matching `path`, its prefix (as
151    /// determined by a common suffix of the directory containing this
152    /// gitignore) is stripped. If there is no common suffix/prefix overlap,
153    /// then `path` is assumed to be relative to this matcher.
154    pub fn matched<P: AsRef<[u8]>>(
155        &self,
156        path: P,
157        is_dir: bool,
158    ) -> Match<&Glob> {
159        if self.is_empty() {
160            return Match::None;
161        }
162        self.matched_stripped(self.strip(path.as_ref()), is_dir)
163    }
164
165    /// Returns whether the given path (file or directory, and expected to be
166    /// under the root) or any of its parent directories (up to the root)
167    /// matched a pattern in this gitignore matcher.
168    ///
169    /// NOTE: This method is more expensive than walking the directory hierarchy
170    /// top-to-bottom and matching the entries. But, is easier to use in cases
171    /// when a list of paths are available without a hierarchy.
172    ///
173    /// `is_dir` should be true if the path refers to a directory and false
174    /// otherwise.
175    ///
176    /// The given path is matched relative to the path given when building
177    /// the matcher. Specifically, before matching `path`, its prefix (as
178    /// determined by a common suffix of the directory containing this
179    /// gitignore) is stripped. If there is no common suffix/prefix overlap,
180    /// then `path` is assumed to be relative to this matcher.
181    ///
182    /// # Panics
183    ///
184    /// This method panics if the given file path is not under the root path
185    /// of this matcher.
186    pub fn matched_path_or_any_parents<P: AsRef<[u8]>>(
187        &self,
188        path: P,
189        is_dir: bool,
190    ) -> Match<&Glob> {
191        if self.is_empty() {
192            return Match::None;
193        }
194        let mut path = self.strip(path.as_ref());
195        assert!(!has_root(path), "path is expected to be under the root");
196
197        match self.matched_stripped(path, is_dir) {
198            Match::None => (), // walk up
199            a_match => return a_match,
200        }
201        while let Some(parent) = parent(path) {
202            match self.matched_stripped(parent, /* is_dir */ true) {
203                Match::None => path = parent, // walk up
204                a_match => return a_match,
205            }
206        }
207        Match::None
208    }
209
210    /// Like matched, but takes a path that has already been stripped.
211    fn matched_stripped<P: AsRef<[u8]>>(
212        &self,
213        path: P,
214        is_dir: bool,
215    ) -> Match<&Glob> {
216        if self.is_empty() {
217            return Match::None;
218        }
219        let path = path.as_ref();
220        let mut matches = self.matches.as_ref().unwrap().get();
221        let candidate = Candidate::new(path);
222        self.set.matches_candidate_into(&candidate, &mut *matches);
223        for &i in matches.iter().rev() {
224            let glob = &self.globs[i];
225            if !glob.is_only_dir() || is_dir {
226                return if glob.is_whitelist() {
227                    Match::Whitelist(glob)
228                } else {
229                    Match::Ignore(glob)
230                };
231            }
232        }
233        Match::None
234    }
235
236    /// Strips the given path such that it's suitable for matching with this
237    /// gitignore matcher.
238    fn strip<'a, P: 'a + AsRef<[u8]> + ?Sized>(
239        &'a self,
240        path: &'a P,
241    ) -> &'a [u8] {
242        let mut path = path.as_ref();
243        // A leading ./ is completely superfluous. We also strip it from
244        // our gitignore root path, so we need to strip it from our candidate
245        // path too.
246        if let Some(p) = strip_prefix("./", path) {
247            path = p;
248        }
249        // Strip any common prefix between the candidate path and the root
250        // of the gitignore, to make sure we get relative matching right.
251        // BUT, a file name might not have any directory components to it,
252        // in which case, we don't want to accidentally strip any part of the
253        // file name.
254        //
255        // As an additional special case, if the root is just `.`, then we
256        // shouldn't try to strip anything, e.g., when path begins with a `.`.
257        if !same_path(&self.root, b".") && !is_file_name(path) {
258            if let Some(p) = strip_prefix(&self.root, path) {
259                path = p;
260                // If we're left with a leading slash, get rid of it.
261                if let Some(p) = strip_prefix("/", path) {
262                    path = p;
263                }
264            }
265        }
266        path
267    }
268}

270/// Builds a matcher for a single set of globs from a .gitignore file.
271#[derive(Clone, Debug)]
272pub struct GitignoreBuilder {
273    builder: GlobSetBuilder,
274    root: Vec<u8>,
275    globs: Vec<Glob>,
276    case_insensitive: bool,
277    allow_unclosed_class: bool,
278}
279
280impl GitignoreBuilder {
281    /// Create a new builder for a gitignore file.
282    ///
283    /// The path given should be the path at which the globs for this gitignore
284    /// file should be matched. Note that paths are always matched relative
285    /// to the root path given here. Generally, the root path should correspond
286    /// to the *directory* containing a `.gitignore` file.
287    pub fn new<P: AsRef<[u8]>>(root: P) -> GitignoreBuilder {
288        let root = root.as_ref();
289        GitignoreBuilder {
290            builder: GlobSetBuilder::new(),
291            root: strip_prefix("./", root).unwrap_or(root).to_vec(),
292            globs: vec![],
293            case_insensitive: false,
294            allow_unclosed_class: true,
295        }
296    }
297
298    /// Builds a new matcher from the globs added so far.
299    ///
300    /// Once a matcher is built, no new globs can be added to it.
301    pub fn build(&self) -> Result<Gitignore, Error> {
302        let nignore = self.globs.iter().filter(|g| !g.is_whitelist()).count();
303        let nwhite = self.globs.iter().filter(|g| g.is_whitelist()).count();
304        let set = self
305            .builder
306            .build()
307            .map_err(|err| Error::Glob { glob: None, err: err.to_string() })?;
308        Ok(Gitignore {
309            set,
310            root: self.root.clone(),
311            globs: self.globs.clone(),
312            num_ignores: nignore as u64,
313            num_whitelists: nwhite as u64,
314            matches: Some(Arc::new(Pool::new(|| vec![]))),
315        })
316    }
317
318    /// Add each glob from the contents of the gitignore file at the path
319    /// given.
320    ///
321    /// vigrep: upstream's `add(path)` opens and reads the file; here the
322    /// caller supplies the bytes it read (e.g. through a VFS handle) and
323    /// `path` only labels globs and errors. Lines are split as
324    /// `BufRead::lines` does (LF, with one CR before it removed); a line that
325    /// is not valid UTF-8 ends processing with an I/O error, as upstream.
326    ///
327    /// The file given should be formatted as a `gitignore` file.
328    ///
329    /// Note that partial errors can be returned. For example, if there was
330    /// a problem adding one glob, an error for that will be returned, but
331    /// all other valid globs will still be added.
332    pub fn add_bytes<P: AsRef<[u8]>>(
333        &mut self,
334        path: P,
335        contents: &[u8],
336    ) -> Option<Error> {
337        let path = path.as_ref();
338        log::debug!("adding gitignore file: {}", bstr::BStr::new(path));
339        let mut errs = PartialErrorBuilder::default();
340        for (i, line) in lines(contents).enumerate() {
341            let lineno = (i + 1) as u64;
342            let line = match core::str::from_utf8(line) {
343                Ok(line) => line,
344                Err(_) => {
345                    let err = crate::io::Error::new(
346                        crate::io::ErrorKind::InvalidData,
347                        "stream did not contain valid UTF-8",
348                    );
349                    errs.push(Error::Io(err).tagged(path, lineno));
350                    break;
351                }
352            };
353
354            // Match Git's handling of .gitignore files that begin with the Unicode BOM
355            const UTF8_BOM: &str = "\u{feff}";
356            let line =
357                if i == 0 { line.trim_start_matches(UTF8_BOM) } else { &line };
358
359            if let Err(err) = self.add_line(Some(path.to_vec()), &line) {
360                errs.push(err.tagged(path, lineno));
361            }
362        }
363        errs.into_error_option()
364    }

366    /// Add each glob line from the string given.
367    ///
368    /// If this string came from a particular `gitignore` file, then its path
369    /// should be provided here.
370    ///
371    /// The string given should be formatted as a `gitignore` file.
372    #[cfg(test)]
373    fn add_str(
374        &mut self,
375        from: Option<Vec<u8>>,
376        gitignore: &str,
377    ) -> Result<&mut GitignoreBuilder, Error> {
378        for line in gitignore.lines() {
379            self.add_line(from.clone(), line)?;
380        }
381        Ok(self)
382    }

384    /// Add a line from a gitignore file to this builder.
385    ///
386    /// If this line came from a particular `gitignore` file, then its path
387    /// should be provided here.
388    ///
389    /// If the line could not be parsed as a glob, then an error is returned.
390    pub fn add_line(
391        &mut self,
392        from: Option<Vec<u8>>,
393        mut line: &str,
394    ) -> Result<&mut GitignoreBuilder, Error> {
395        #![allow(deprecated)]
396
397        if line.starts_with("#") {
398            return Ok(self);
399        }
400        if !line.ends_with("\\ ") {
401            line = line.trim_right();
402        }
403        if line.is_empty() {
404            return Ok(self);
405        }
406        let mut glob = Glob {
407            from,
408            original: line.to_string(),
409            actual: String::new(),
410            is_whitelist: false,
411            is_only_dir: false,
412        };
413        let mut is_absolute = false;
414        if line.starts_with("\\!") || line.starts_with("\\#") {
415            line = &line[1..];
416            is_absolute = line.chars().nth(0) == Some('/');
417        } else {
418            if line.starts_with("!") {
419                glob.is_whitelist = true;
420                line = &line[1..];
421            }
422            if line.starts_with("/") {
423                // `man gitignore` says that if a glob starts with a slash,
424                // then the glob can only match the beginning of a path
425                // (relative to the location of gitignore). We achieve this by
426                // simply banning wildcards from matching /.
427                line = &line[1..];
428                is_absolute = true;
429            }
430        }
431        // If it ends with a slash, then this should only match directories,
432        // but the slash should otherwise not be used while globbing.
433        if line.as_bytes().last() == Some(&b'/') {
434            glob.is_only_dir = true;
435            line = &line[..line.len() - 1];
436            // If the slash was escaped, then remove the escape.
437            // See: https://github.com/BurntSushi/ripgrep/issues/2236
438            if line.as_bytes().last() == Some(&b'\\') {
439                line = &line[..line.len() - 1];
440            }
441        }
442        glob.actual = line.to_string();
443        // If there is a literal slash, then this is a glob that must match the
444        // entire path name. Otherwise, we should let it match anywhere, so use
445        // a **/ prefix.
446        if !is_absolute && !line.chars().any(|c| c == '/') {
447            // ... but only if we don't already have a **/ prefix.
448            if !glob.has_doublestar_prefix() {
449                glob.actual = format!("**/{}", glob.actual);
450            }
451        }
452        // If the glob ends with `/**`, then we should only match everything
453        // inside a directory, but not the directory itself. Standard globs
454        // will match the directory. So we add `/*` to force the issue.
455        if glob.actual.ends_with("/**") {
456            glob.actual = format!("{}/*", glob.actual);
457        }
458        let parsed = GlobBuilder::new(&glob.actual)
459            .literal_separator(true)
460            .case_insensitive(self.case_insensitive)
461            .backslash_escape(true)
462            .allow_unclosed_class(self.allow_unclosed_class)
463            .build()
464            .map_err(|err| Error::Glob {
465                glob: Some(glob.original.clone()),
466                err: err.kind().to_string(),
467            })?;
468        self.builder.add(parsed);
469        self.globs.push(glob);
470        Ok(self)
471    }

473    /// Toggle whether the globs should be matched case insensitively or not.
474    ///
475    /// When this option is changed, only globs added after the change will be
476    /// affected.
477    ///
478    /// This is disabled by default.
479    pub fn case_insensitive(
480        &mut self,
481        yes: bool,
482    ) -> Result<&mut GitignoreBuilder, Error> {
483        // TODO: This should not return a `Result`. Fix this in the next semver
484        // release.
485        self.case_insensitive = yes;
486        Ok(self)
487    }

489    /// Toggle whether unclosed character classes are allowed. When allowed,
490    /// a `[` without a matching `]` is treated literally instead of resulting
491    /// in a parse error.
492    ///
493    /// For example, if this is set then the glob `[abc` will be treated as the
494    /// literal string `[abc` instead of returning an error.
495    ///
496    /// By default, this is true in order to match established `gitignore`
497    /// semantics. Generally speaking, enabling this leads to worse failure
498    /// modes since the glob parser becomes more permissive. You might want to
499    /// enable this when compatibility (e.g., with POSIX glob implementations)
500    /// is more important than good error messages.
501    pub fn allow_unclosed_class(
502        &mut self,
503        yes: bool,
504    ) -> &mut GitignoreBuilder {
505        self.allow_unclosed_class = yes;
506        self
507    }

510/// vigrep: the line splitting of `BufRead::lines`: LF-terminated lines with a
511/// single CR before the LF removed; a final unterminated line is yielded and
512/// an empty input yields nothing.
513fn lines(mut contents: &[u8]) -> impl Iterator<Item = &[u8]> {
514    core::iter::from_fn(move || {
515        if contents.is_empty() {
516            return None;
517        }
518        let (mut line, rest) = match memchr::memchr(b'\n', contents) {
519            Some(i) => (&contents[..i], &contents[i + 1..]),
520            None => (contents, &contents[contents.len()..]),
521        };
522        if contents.len() > line.len() && line.last() == Some(&b'\r') {
523            line = &line[..line.len() - 1];
524        }
525        contents = rest;
526        Some(line)
527    })
528}
