13/// The default buffer capacity that we use for the line buffer.
14pub(crate) const DEFAULT_BUFFER_CAPACITY: usize = 64 * (1 << 10); // 64 KB

16/// The behavior of a searcher in the face of long lines and big contexts.
17///
18/// When searching data incrementally using a fixed size buffer, this controls
19/// the amount of *additional* memory to allocate beyond the size of the buffer
20/// to accommodate lines (which may include the lines in a context window, when
21/// enabled) that do not fit in the buffer.
22///
23/// The default is to eagerly allocate without a limit.
24#[derive(Clone, Copy, Debug)]
25pub(crate) enum BufferAllocation {
26    /// Attempt to expand the size of the buffer until either at least the next
27    /// line fits into memory or until all available memory is exhausted.
28    ///
29    /// This is the default.
30    Eager,
31    /// Limit the amount of additional memory allocated to the given size. If
32    /// a line is found that requires more memory than is allowed here, then
33    /// stop reading and return an error.
34    Error(usize),
35}
36
37impl Default for BufferAllocation {
38    fn default() -> BufferAllocation {
39        BufferAllocation::Eager
40    }
41}
42
43/// Create a new error to be used when a configured allocation limit has been
44/// reached.
45pub(crate) fn alloc_error(limit: usize) -> io::Error {
46    let msg = format!("configured allocation limit ({}) exceeded", limit);
47    io::Error::new(io::ErrorKind::Other, msg)
48}

50/// The behavior of binary detection in the line buffer.
51///
52/// Binary detection is the process of _heuristically_ identifying whether a
53/// given chunk of data is binary or not, and then taking an action based on
54/// the result of that heuristic. The motivation behind detecting binary data
55/// is that binary data often indicates data that is undesirable to search
56/// using textual patterns. Of course, there are many cases in which this isn't
57/// true, which is why binary detection is disabled by default.
58#[derive(Clone, Copy, Debug, Eq, PartialEq)]
59pub(crate) enum BinaryDetection {
60    /// No binary detection is performed. Data reported by the line buffer may
61    /// contain arbitrary bytes.
62    None,
63    /// The given byte is searched in all contents read by the line buffer. If
64    /// it occurs, then the data is considered binary and the line buffer acts
65    /// as if it reached EOF. The line buffer guarantees that this byte will
66    /// never be observable by callers.
67    Quit(u8),
68    /// The given byte is searched in all contents read by the line buffer. If
69    /// it occurs, then it is replaced by the line terminator. The line buffer
70    /// guarantees that this byte will never be observable by callers.
71    Convert(u8),
72}
73
74impl Default for BinaryDetection {
75    fn default() -> BinaryDetection {
76        BinaryDetection::None
77    }
78}
79
80impl BinaryDetection {
81    /// Returns true if and only if the detection heuristic demands that
82    /// the line buffer stop read data once binary data is observed.
83    fn is_quit(&self) -> bool {
84        match *self {
85            BinaryDetection::Quit(_) => true,
86            _ => false,
87        }
88    }
89}

91/// The configuration of a buffer. This contains options that are fixed once
92/// a buffer has been constructed.
93#[derive(Clone, Copy, Debug)]
94struct Config {
95    /// The number of bytes to attempt to read at a time.
96    capacity: usize,
97    /// The line terminator.
98    lineterm: u8,
99    /// The behavior for handling long lines.
100    buffer_alloc: BufferAllocation,
101    /// When set, the presence of the given byte indicates binary content.
102    binary: BinaryDetection,
103}
104
105impl Default for Config {
106    fn default() -> Config {
107        Config {
108            capacity: DEFAULT_BUFFER_CAPACITY,
109            lineterm: b'\n',
110            buffer_alloc: BufferAllocation::default(),
111            binary: BinaryDetection::default(),
112        }
113    }
114}

116/// A builder for constructing line buffers.
117#[derive(Clone, Debug, Default)]
118pub(crate) struct LineBufferBuilder {
119    config: Config,
120}
121
122impl LineBufferBuilder {
123    /// Create a new builder for a buffer.
124    pub(crate) fn new() -> LineBufferBuilder {
125        LineBufferBuilder { config: Config::default() }
126    }
127
128    /// Create a new line buffer from this builder's configuration.
129    pub(crate) fn build(&self) -> LineBuffer {
130        LineBuffer {
131            config: self.config,
132            buf: vec![0; self.config.capacity],
133            pos: 0,
134            last_lineterm: 0,
135            end: 0,
136            absolute_byte_offset: 0,
137            binary_byte_offset: None,
138        }
139    }
140
141    /// Set the default capacity to use for a buffer.
142    ///
143    /// In general, the capacity of a buffer corresponds to the amount of data
144    /// to hold in memory, and the size of the reads to make to the underlying
145    /// reader.
146    ///
147    /// This is set to a reasonable default and probably shouldn't be changed
148    /// unless there's a specific reason to do so.
149    pub(crate) fn capacity(
150        &mut self,
151        capacity: usize,
152    ) -> &mut LineBufferBuilder {
153        self.config.capacity = capacity;
154        self
155    }
156
157    /// Set the line terminator for the buffer.
158    ///
159    /// Every buffer has a line terminator, and this line terminator is used
160    /// to determine how to roll the buffer forward. For example, when a read
161    /// to the buffer's underlying reader occurs, the end of the data that is
162    /// read is likely to correspond to an incomplete line. As a line buffer,
163    /// callers should not access this data since it is incomplete. The line
164    /// terminator is how the line buffer determines the part of the read that
165    /// is incomplete.
166    ///
167    /// By default, this is set to `b'\n'`.
168    pub(crate) fn line_terminator(
169        &mut self,
170        lineterm: u8,
171    ) -> &mut LineBufferBuilder {
172        self.config.lineterm = lineterm;
173        self
174    }
175
176    /// Set the maximum amount of additional memory to allocate for long lines.
177    ///
178    /// In order to enable line oriented search, a fundamental requirement is
179    /// that, at a minimum, each line must be able to fit into memory. This
180    /// setting controls how big that line is allowed to be. By default, this
181    /// is set to `BufferAllocation::Eager`, which means a line buffer will
182    /// attempt to allocate as much memory as possible to fit a line, and will
183    /// only be limited by available memory.
184    ///
185    /// Note that this setting only applies to the amount of *additional*
186    /// memory to allocate, beyond the capacity of the buffer. That means that
187    /// a value of `0` is sensible, and in particular, will guarantee that a
188    /// line buffer will never allocate additional memory beyond its initial
189    /// capacity.
190    pub(crate) fn buffer_alloc(
191        &mut self,
192        behavior: BufferAllocation,
193    ) -> &mut LineBufferBuilder {
194        self.config.buffer_alloc = behavior;
195        self
196    }
197
198    /// Whether to enable binary detection or not. Depending on the setting,
199    /// this can either cause the line buffer to report EOF early or it can
200    /// cause the line buffer to clean the data.
201    ///
202    /// By default, this is disabled. In general, binary detection should be
203    /// viewed as an imperfect heuristic.
204    pub(crate) fn binary_detection(
205        &mut self,
206        detection: BinaryDetection,
207    ) -> &mut LineBufferBuilder {
208        self.config.binary = detection;
209        self
210    }
211}

213/// A line buffer reader efficiently reads a line oriented buffer from an
214/// arbitrary reader.
215#[derive(Debug)]
216pub(crate) struct LineBufferReader<'b, R> {
217    rdr: R,
218    line_buffer: &'b mut LineBuffer,
219}
220
221impl<'b, R: io::Read> LineBufferReader<'b, R> {
222    /// Create a new buffered reader that reads from `rdr` and uses the given
223    /// `line_buffer` as an intermediate buffer.
224    ///
225    /// This does not change the binary detection behavior of the given line
226    /// buffer.
227    pub(crate) fn new(
228        rdr: R,
229        line_buffer: &'b mut LineBuffer,
230    ) -> LineBufferReader<'b, R> {
231        line_buffer.clear();
232        LineBufferReader { rdr, line_buffer }
233    }
234
235    /// The absolute byte offset which corresponds to the starting offsets
236    /// of the data returned by `buffer` relative to the beginning of the
237    /// underlying reader's contents. As such, this offset does not generally
238    /// correspond to an offset in memory. It is typically used for reporting
239    /// purposes. It can also be used for counting the number of bytes that
240    /// have been searched.
241    pub(crate) fn absolute_byte_offset(&self) -> u64 {
242        self.line_buffer.absolute_byte_offset()
243    }
244
245    /// If binary data was detected, then this returns the absolute byte offset
246    /// at which binary data was initially found.
247    pub(crate) fn binary_byte_offset(&self) -> Option<u64> {
248        self.line_buffer.binary_byte_offset()
249    }
250
251    /// Fill the contents of this buffer by discarding the part of the buffer
252    /// that has been consumed. The free space created by discarding the
253    /// consumed part of the buffer is then filled with new data from the
254    /// reader.
255    ///
256    /// If EOF is reached, then `false` is returned. Otherwise, `true` is
257    /// returned. (Note that if this line buffer's binary detection is set to
258    /// `Quit`, then the presence of binary data will cause this buffer to
259    /// behave as if it had seen EOF at the first occurrence of binary data.)
260    ///
261    /// This forwards any errors returned by the underlying reader, and will
262    /// also return an error if the buffer must be expanded past its allocation
263    /// limit, as governed by the buffer allocation strategy.
264    pub(crate) fn fill(&mut self) -> Result<bool, io::Error> {
265        self.line_buffer.fill(&mut self.rdr)
266    }
267
268    /// Return the contents of this buffer.
269    pub(crate) fn buffer(&self) -> &[u8] {
270        self.line_buffer.buffer()
271    }
272
273    /// Return the buffer as a BStr, used for convenient equality checking
274    /// in tests only.
275    #[cfg(test)]
276    fn bstr(&self) -> &bstr::BStr {
277        self.buffer().as_bstr()
278    }
279
280    /// Consume the number of bytes provided. This must be less than or equal
281    /// to the number of bytes returned by `buffer`.
282    pub(crate) fn consume(&mut self, amt: usize) {
283        self.line_buffer.consume(amt);
284    }
285
286    /// Consumes the remainder of the buffer. Subsequent calls to `buffer` are
287    /// guaranteed to return an empty slice until the buffer is refilled.
288    ///
289    /// This is a convenience function for `consume(buffer.len())`.
290    #[cfg(test)]
291    fn consume_all(&mut self) {
292        self.line_buffer.consume_all();
293    }
294}

296/// A line buffer manages a (typically fixed) buffer for holding lines.
297///
298/// Callers should create line buffers sparingly and reuse them when possible.
299/// Line buffers cannot be used directly, but instead must be used via the
300/// LineBufferReader.
301#[derive(Clone, Debug)]
302pub(crate) struct LineBuffer {
303    /// The configuration of this buffer.
304    config: Config,
305    /// The primary buffer with which to hold data.
306    buf: Vec<u8>,
307    /// The current position of this buffer. This is always a valid sliceable
308    /// index into `buf`, and its maximum value is the length of `buf`.
309    pos: usize,
310    /// The end position of searchable content in this buffer. This is either
311    /// set to just after the final line terminator in the buffer, or to just
312    /// after the end of the last byte emitted by the reader when the reader
313    /// has been exhausted.
314    last_lineterm: usize,
315    /// The end position of the buffer. This is always greater than or equal to
316    /// last_lineterm. The bytes between last_lineterm and end, if any, always
317    /// correspond to a partial line.
318    end: usize,
319    /// The absolute byte offset corresponding to `pos`. This is most typically
320    /// not a valid index into addressable memory, but rather, an offset that
321    /// is relative to all data that passes through a line buffer (since
322    /// construction or since the last time `clear` was called).
323    ///
324    /// When the line buffer reaches EOF, this is set to the position just
325    /// after the last byte read from the underlying reader. That is, it
326    /// becomes the total count of bytes that have been read.
327    absolute_byte_offset: u64,
328    /// If binary data was found, this records the absolute byte offset at
329    /// which it was first detected.
330    binary_byte_offset: Option<u64>,
331}
332
333impl LineBuffer {
334    /// Set the binary detection method used on this line buffer.
335    ///
336    /// This permits dynamically changing the binary detection strategy on
337    /// an existing line buffer without needing to create a new one.
338    pub(crate) fn set_binary_detection(&mut self, binary: BinaryDetection) {
339        self.config.binary = binary;
340    }
341
342    /// Reset this buffer, such that it can be used with a new reader.
343    fn clear(&mut self) {
344        self.pos = 0;
345        self.last_lineterm = 0;
346        self.end = 0;
347        self.absolute_byte_offset = 0;
348        self.binary_byte_offset = None;
349    }
350
351    /// The absolute byte offset which corresponds to the starting offsets
352    /// of the data returned by `buffer` relative to the beginning of the
353    /// reader's contents. As such, this offset does not generally correspond
354    /// to an offset in memory. It is typically used for reporting purposes,
355    /// particularly in error messages.
356    ///
357    /// This is reset to `0` when `clear` is called.
358    fn absolute_byte_offset(&self) -> u64 {
359        self.absolute_byte_offset
360    }
361
362    /// If binary data was detected, then this returns the absolute byte offset
363    /// at which binary data was initially found.
364    fn binary_byte_offset(&self) -> Option<u64> {
365        self.binary_byte_offset
366    }
367
368    /// Return the contents of this buffer.
369    fn buffer(&self) -> &[u8] {
370        &self.buf[self.pos..self.last_lineterm]
371    }
372
373    /// Return the contents of the free space beyond the end of the buffer as
374    /// a mutable slice.
375    fn free_buffer(&mut self) -> &mut [u8] {
376        &mut self.buf[self.end..]
377    }
378
379    /// Consume the number of bytes provided. This must be less than or equal
380    /// to the number of bytes returned by `buffer`.
381    fn consume(&mut self, amt: usize) {
382        assert!(amt <= self.buffer().len());
383        self.pos += amt;
384        self.absolute_byte_offset += amt as u64;
385    }
386
387    /// Consumes the remainder of the buffer. Subsequent calls to `buffer` are
388    /// guaranteed to return an empty slice until the buffer is refilled.
389    ///
390    /// This is a convenience function for `consume(buffer.len())`.
391    #[cfg(test)]
392    fn consume_all(&mut self) {
393        let amt = self.buffer().len();
394        self.consume(amt);
395    }
396
397    /// Fill the contents of this buffer by discarding the part of the buffer
398    /// that has been consumed. The free space created by discarding the
399    /// consumed part of the buffer is then filled with new data from the given
400    /// reader.
401    ///
402    /// Callers should provide the same reader to this line buffer in
403    /// subsequent calls to fill. A different reader can only be used
404    /// immediately following a call to `clear`.
405    ///
406    /// If EOF is reached, then `false` is returned. Otherwise, `true` is
407    /// returned. (Note that if this line buffer's binary detection is set to
408    /// `Quit`, then the presence of binary data will cause this buffer to
409    /// behave as if it had seen EOF.)
410    ///
411    /// This forwards any errors returned by `rdr`, and will also return an
412    /// error if the buffer must be expanded past its allocation limit, as
413    /// governed by the buffer allocation strategy.
414    fn fill<R: io::Read>(&mut self, mut rdr: R) -> Result<bool, io::Error> {
415        // If the binary detection heuristic tells us to quit once binary data
416        // has been observed, then we no longer read new data and reach EOF
417        // once the current buffer has been consumed.
418        if self.config.binary.is_quit() && self.binary_byte_offset.is_some() {
419            return Ok(!self.buffer().is_empty());
420        }
421
422        self.roll();
423        assert_eq!(self.pos, 0);
424        loop {
425            self.ensure_capacity()?;
426            let readlen = rdr.read(self.free_buffer().as_bytes_mut())?;
427            if readlen == 0 {
428                // We're only done reading for good once the caller has
429                // consumed everything.
430                self.last_lineterm = self.end;
431                return Ok(!self.buffer().is_empty());
432            }
433
434            // Get a mutable view into the bytes we've just read. These are
435            // the bytes that we do binary detection on, and also the bytes we
436            // search to find the last line terminator. We need a mutable slice
437            // in the case of binary conversion.
438            let oldend = self.end;
439            self.end += readlen;
440            let newbytes = &mut self.buf[oldend..self.end];
441
442            // Binary detection.
443            match self.config.binary {
444                BinaryDetection::None => {} // nothing to do
445                BinaryDetection::Quit(byte) => {
446                    if let Some(i) = newbytes.find_byte(byte) {
447                        self.end = oldend + i;
448                        self.last_lineterm = self.end;
449                        self.binary_byte_offset =
450                            Some(self.absolute_byte_offset + self.end as u64);
451                        // If the first byte in our buffer is a binary byte,
452                        // then our buffer is empty and we should report as
453                        // such to the caller.
454                        return Ok(self.pos < self.end);
455                    }
456                }
457                BinaryDetection::Convert(byte) => {
458                    if let Some(i) =
459                        replace_bytes(newbytes, byte, self.config.lineterm)
460                    {
461                        // Record only the first binary offset.
462                        if self.binary_byte_offset.is_none() {
463                            self.binary_byte_offset = Some(
464                                self.absolute_byte_offset
465                                    + (oldend + i) as u64,
466                            );
467                        }
468                    }
469                }
470            }
471
472            // Update our `last_lineterm` positions if we read one.
473            if let Some(i) = newbytes.rfind_byte(self.config.lineterm) {
474                self.last_lineterm = oldend + i + 1;
475                return Ok(true);
476            }
477            // At this point, if we couldn't find a line terminator, then we
478            // don't have a complete line. Therefore, we try to read more!
479        }
480    }
481
482    /// Roll the unconsumed parts of the buffer to the front.
483    ///
484    /// This operation is idempotent.
485    ///
486    /// After rolling, `last_lineterm` and `end` point to the same location,
487    /// and `pos` is always set to `0`.
488    fn roll(&mut self) {
489        if self.pos == self.end {
490            self.pos = 0;
491            self.last_lineterm = 0;
492            self.end = 0;
493            return;
494        }
495
496        let roll_len = self.end - self.pos;
497        self.buf.copy_within(self.pos..self.end, 0);
498        self.pos = 0;
499        self.last_lineterm = roll_len;
500        self.end = roll_len;
501    }
502
503    /// Ensures that the internal buffer has a non-zero amount of free space
504    /// in which to read more data. If there is no free space, then more is
505    /// allocated. If the allocation must exceed the configured limit, then
506    /// this returns an error.
507    fn ensure_capacity(&mut self) -> Result<(), io::Error> {
508        if !self.free_buffer().is_empty() {
509            return Ok(());
510        }
511        // `len` is used for computing the next allocation size. The capacity
512        // is permitted to start at `0`, so we make sure it's at least `1`.
513        let len = core::cmp::max(1, self.buf.len());
514        let additional = match self.config.buffer_alloc {
515            BufferAllocation::Eager => len * 2,
516            BufferAllocation::Error(limit) => {
517                let used = self.buf.len() - self.config.capacity;
518                let n = core::cmp::min(len * 2, limit - used);
519                if n == 0 {
520                    return Err(alloc_error(self.config.capacity + limit));
521                }
522                n
523            }
524        };
525        assert!(additional > 0);
526        let newlen = self.buf.len() + additional;
527        self.buf.resize(newlen, 0);
528        assert!(!self.free_buffer().is_empty());
529        Ok(())
530    }
531}

533/// Replaces `src` with `replacement` in bytes, and return the offset of the
534/// first replacement, if one exists.
535fn replace_bytes(
536    mut bytes: &mut [u8],
537    src: u8,
538    replacement: u8,
539) -> Option<usize> {
540    if src == replacement {
541        return None;
542    }
543    let first_pos = bytes.find_byte(src)?;
544    bytes[first_pos] = replacement;
545    bytes = &mut bytes[first_pos + 1..];
546    while let Some(i) = bytes.find_byte(src) {
547        bytes[i] = replacement;
548        bytes = &mut bytes[i + 1..];
549
550        // To search for adjacent `src` bytes we use a different strategy.
551        // Since binary data tends to have long runs of NUL terminators,
552        // it is faster to compare one-byte-at-a-time than to stop and start
553        // memchr (through `find_byte`) for every byte in a sequence.
554        while bytes.get(0) == Some(&src) {
555            bytes[0] = replacement;
556            bytes = &mut bytes[1..];
557        }
558    }
559    Some(first_pos)
560}

713    #[test]
714    fn buffer_limited_capacity1() {
715        let bytes = "homer\nlisa\nmaggie";
716        let mut linebuf = LineBufferBuilder::new()
717            .capacity(1)
718            .buffer_alloc(BufferAllocation::Error(5))
719            .build();
720        let mut rdr = LineBufferReader::new(bytes.as_bytes(), &mut linebuf);
721
722        assert!(rdr.fill().unwrap());
723        assert_eq!(rdr.bstr(), "homer\n");
724        rdr.consume_all();
725
726        assert!(rdr.fill().unwrap());
727        assert_eq!(rdr.bstr(), "lisa\n");
728        rdr.consume_all();
729
730        // This returns an error because while we have just enough room to
731        // store maggie in the buffer, we *don't* have enough room to read one
732        // more byte, so we don't know whether we're at EOF or not, and
733        // therefore must give up.
734        assert!(rdr.fill().is_err());
735
736        // We can mush on though!
737        assert_eq!(rdr.bstr(), "m");
738        rdr.consume_all();
739
740        assert!(rdr.fill().unwrap());
741        assert_eq!(rdr.bstr(), "aggie");
742        rdr.consume_all();
743
744        assert!(!rdr.fill().unwrap());
745    }
