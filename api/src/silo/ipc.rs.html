1//! Attested KMS-only ingress for the development Silo provider.
2
3use api::caller_identity::CallerIdentity;
4use api::syscall::service;
5use ostd::syscall::{sys_lookup_service, sys_recv_attested, sys_try_send, SyscallResult};
6use service_silo::protocol::{PeerIdentity, ProtocolState};
7use types::silo::DEVELOPMENT_SILO_FRAME_LEN;
8
9use crate::guest::GuestSession;
10
11/// Serve requests forever; only a kernel-attested live KMS instance can bind.
12pub fn run(mut guest: GuestSession) -> ! {
13    let mut state = ProtocolState::new();
14    loop {
15        let mut buffer = [0u8; api::ipc::IPC_BUF_SIZE];
16        let sender = match sys_recv_attested(0, &mut buffer) {
17            SyscallResult::Ok(sender) if sender > 0 => sender,
18            _ => continue,
19        };
20        let peer = CallerIdentity::from_recv_buf(&buffer).map(|identity| PeerIdentity {
21            sender_tid: identity.sender_tid as usize,
22            cell_id: identity.cell_id,
23            generation: identity.generation,
24        });
25        let response = state.process(
26            &mut guest,
27            sender,
28            sys_lookup_service(service::KMS),
29            peer,
30            &buffer[..DEVELOPMENT_SILO_FRAME_LEN],
31        );
32        if let Some(response) = response {
33            let _ = sys_try_send(sender, &response.encode());
34        }
35    }
36}
