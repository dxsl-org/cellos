3use crate::vmm;
4use api::hypervisor::ViVmExit;
5use service_silo::layout::{HVC_SILO_DONE, HVC_SILO_FAULT, HVC_SILO_READY};

8const MAX_EXITS_PER_OPERATION: usize = 64;
9
10/// Non-secret diagnostic fields for an unexpected, always-fatal VM exit.
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub enum SiloUnexpectedExit {
13    MmioRead {
14        ipa: u64,
15        size: u8,
16        reg: u8,
17    },
18    MmioWrite {
19        ipa: u64,
20        size: u8,
21    },
22    Hvc(UnexpectedHvc),
23    SysReg {
24        op0: u8,
25        op1: u8,
26        crn: u8,
27        crm: u8,
28        op2: u8,
29        rt: u8,
30        is_write: bool,
31    },
32    Shutdown,
33    Unknown {
34        ec: u32,
35        iss: u32,
36        pc: Option<u64>,
37    },
38    PortIn {
39        port: u16,
40        size: u8,
41        reg: u8,
42    },
43    PortOut {
44        port: u16,
45        size: u8,
46    },
47    Hlt,
48    Msr {
49        index: u32,
50        is_write: bool,
51    },
52}

54/// A bounded VMM-side failure, distinct from a guest-declared Silo fault.
55#[derive(Debug, Clone, Copy, PartialEq, Eq)]
56pub enum SiloVmmFault {
57    RunVcpu,
58    UnexpectedExit(SiloUnexpectedExit),
59    ExitBudgetExceeded,
60}
61
62/// Result of one bounded guest execution interval.
63#[derive(Debug, Clone, Copy, PartialEq, Eq)]
64pub enum SiloRunResult {
65    Done,
66    /// Guest-declared protocol/crypto fault code carried in HVC x1.
67    GuestFault(u64),
68    VmmFault(SiloVmmFault),
69}
70
71/// Run until the purpose-specific completion HVC or a bounded failure occurs.
72pub fn run_until_done(vm_id: usize, vcpu_id: usize) -> SiloRunResult {
73    let mut exit = ViVmExit::Unknown { ec: 0, iss: 0 };
74    for _ in 0..MAX_EXITS_PER_OPERATION {
75        if vmm::run_vcpu(vm_id, vcpu_id, &mut exit) == usize::MAX {
76            return SiloRunResult::VmmFault(SiloVmmFault::RunVcpu);
77        }
78        match exit {
79            ViVmExit::Hvc { imm: 0, regs }
80                if regs[0] == HVC_SILO_READY || regs[0] == HVC_SILO_DONE =>
81            {
82                return SiloRunResult::Done;
83            }
84            ViVmExit::Hvc { imm: 0, regs } if regs[0] == HVC_SILO_FAULT => {
85                return SiloRunResult::GuestFault(regs[1]);
86            }
87            ViVmExit::Wfi | ViVmExit::Preempted => {}
88            unexpected => {
89                return SiloRunResult::VmmFault(SiloVmmFault::UnexpectedExit(
90                    diagnose_unexpected_exit(vm_id, vcpu_id, unexpected),
91                ));
92            }
93        }
94    }
95    SiloRunResult::VmmFault(SiloVmmFault::ExitBudgetExceeded)
96}

98fn diagnose_unexpected_exit(vm_id: usize, vcpu_id: usize, exit: ViVmExit) -> SiloUnexpectedExit {
99    match exit {
100        ViVmExit::MmioRead { ipa, size, reg } => SiloUnexpectedExit::MmioRead { ipa, size, reg },
101        ViVmExit::MmioWrite { ipa, size, .. } => SiloUnexpectedExit::MmioWrite { ipa, size },
102        ViVmExit::Hvc { imm, regs } => SiloUnexpectedExit::Hvc(diagnose_hvc(imm, regs)),
103        ViVmExit::SysReg {
104            op0,
105            op1,
106            crn,
107            crm,
108            op2,
109            rt,
110            is_write,
111        } => SiloUnexpectedExit::SysReg {
112            op0,
113            op1,
114            crn,
115            crm,
116            op2,
117            rt,
118            is_write,
119        },
120        ViVmExit::Shutdown => SiloUnexpectedExit::Shutdown,
121        ViVmExit::Unknown { ec, iss } => {
122            let mut regs = [0u64; 32];
123            let pc = if vmm::read_vcpu_regs(vm_id, vcpu_id, &mut regs) == 0 {
124                Some(regs[31])
125            } else {
126                None
127            };
128            regs.fill(0);
129            core::hint::black_box(&regs);
130            SiloUnexpectedExit::Unknown { ec, iss, pc }
131        }
132        ViVmExit::PortIn { port, size, reg } => SiloUnexpectedExit::PortIn { port, size, reg },
133        ViVmExit::PortOut { port, size, .. } => SiloUnexpectedExit::PortOut { port, size },
134        ViVmExit::Hlt => SiloUnexpectedExit::Hlt,
135        ViVmExit::Msr {
136            index, is_write, ..
137        } => SiloUnexpectedExit::Msr { index, is_write },
138        ViVmExit::Wfi | ViVmExit::Preempted => unreachable!(),
139    }
140}
