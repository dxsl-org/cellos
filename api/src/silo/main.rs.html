52fn silo_main() {
53    println("[silo] development/reference provider starting");
54    let guest_binary = match artifact::admitted_guest() {
55        Ok(bytes) => bytes,
56        Err(error) => {
57            println(&alloc::format!(
58                "[silo] guest artifact rejected: {:?}",
59                error
60            ));
61            return;
62        }
63    };
64    let vm_id = vmm::create_vm(GUEST_RAM_PAGES);
65    if vm_id == 0 || vm_id == usize::MAX {
66        println("[silo] create_vm failed");
67        return;
68    }
69    if vmm::map_guest_memory(vm_id, GUEST_IPA_BASE, GUEST_RAM_BYTES, true) != 0 {
70        println("[silo] map_guest_memory failed");
71        return;
72    }
73    let vcpu_id = vmm::create_vcpu(vm_id, GUEST_IPA_BASE);
74    if vcpu_id == 0 || vcpu_id == usize::MAX {
75        println("[silo] create_vcpu failed");
76        return;
77    }
78    // Test-hook kernels seed a smoke program during vCPU creation. Load the
79    // admitted guest afterward so the bytes executed always match its digest.
80    if vmm::write_guest_memory(vm_id, GUEST_IPA_BASE, guest_binary) != guest_binary.len() {
81        println("[silo] guest load failed");
82        return;
83    }
84    let session = match guest::GuestSession::initialize(vm_id, vcpu_id) {
85        Ok(session) => session,
86        Err(guest::GuestError::GuestFault(fault)) => {
87            match (
88                fault.response,
89                fault.response.and_then(|response| response.mailbox_code),
90            ) {
91                (Some(response), Some(mailbox_code)) => println(&alloc::format!(
92                    "[silo] one-time initialization guest fault: hvc_code=0x{:02x} \
93                     mailbox_code=0x{:02x} request_seq={} response_seq={} status=0x{:02x}",
94                    fault.hvc_code,
95                    mailbox_code,
96                    response.request_seq,
97                    response.response_seq,
98                    response.status,
99                )),
100                (Some(response), None) => println(&alloc::format!(
101                    "[silo] one-time initialization guest fault: hvc_code=0x{:02x} \
102                     mailbox_code=unavailable request_seq={} response_seq={} status=0x{:02x}",
103                    fault.hvc_code,
104                    response.request_seq,
105                    response.response_seq,
106                    response.status,
107                )),
108                (None, _) => println(&alloc::format!(
109                    "[silo] one-time initialization guest fault: hvc_code=0x{:02x} \
110                     mailbox_response=unavailable",
111                    fault.hvc_code,
112                )),
113            }
114            return;
115        }
116        Err(guest::GuestError::VmmFault(run_loop::SiloVmmFault::UnexpectedExit(
117            run_loop::SiloUnexpectedExit::Unknown { ec, iss, pc },
118        ))) => {
119            match pc {
120                Some(pc) => println(&alloc::format!(
121                    "[silo] one-time initialization VMM fault: \
122                     UnexpectedExit::Unknown ec=0x{:02x} iss=0x{:08x} pc=0x{:016x}",
123                    ec,
124                    iss,
125                    pc,
126                )),
127                None => println(&alloc::format!(
128                    "[silo] one-time initialization VMM fault: \
129                     UnexpectedExit::Unknown ec=0x{:02x} iss=0x{:08x} pc=unavailable",
130                    ec,
131                    iss,
132                )),
133            }
134            return;
135        }
136        Err(guest::GuestError::VmmFault(fault)) => {
137            println(&alloc::format!(
138                "[silo] one-time initialization VMM fault: {:?}",
139                fault
140            ));
141            return;
142        }
143        Err(error) => {
144            println(&alloc::format!(
145                "[silo] one-time initialization failed: {:?}",
146                error
147            ));
148            return;
149        }
150    };
151    if ostd::service::register(api::syscall::service::SILO, 0).is_err() {
152        println("[silo] readiness registration denied");
153        return;
154    }
155    println("[silo] DEV_REFERENCE ready and registered; accepting only live KMS");
156    ipc::run(session)
157}
