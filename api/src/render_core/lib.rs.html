23#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
24pub struct RenderConfig {
25    pub width: u32,
26    pub height: u32,
27    pub samples: u32,
28    pub max_bounces: u32,
29    pub seed: u64,
30}
31
32impl RenderConfig {
33    pub fn validate(self) -> Result<(), RenderError> {
34        if self.width == 0
35            || self.height == 0
36            || self.width > MAX_DIMENSION
37            || self.height > MAX_DIMENSION
38            || self.samples == 0
39            || self.samples > MAX_SAMPLES
40            || self.max_bounces == 0
41            || self.max_bounces > MAX_BOUNCES
42        {
43            return Err(RenderError::InvalidConfig);
44        }
45        self.rgb_bytes()?;
46        Ok(())
47    }
48
49    pub fn rgb_bytes(self) -> Result<usize, RenderError> {
50        byte_length(self.width, self.height).ok_or(RenderError::InvalidConfig)
51    }
52}

54#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
55pub struct Tile {
56    pub id: u32,
57    pub x: u32,
58    pub y: u32,
59    pub width: u32,
60    pub height: u32,
61}
62
63impl Tile {
64    pub fn validated(self, config: RenderConfig) -> Result<Self, RenderError> {
65        config.validate()?;
66        if self.width == 0
67            || self.height == 0
68            || self
69                .x
70                .checked_add(self.width)
71                .map_or(true, |end| end > config.width)
72            || self
73                .y
74                .checked_add(self.height)
75                .map_or(true, |end| end > config.height)
76        {
77            return Err(RenderError::InvalidTile);
78        }
79        self.rgb_bytes()?;
80        Ok(self)
81    }
82
83    pub fn rgb_bytes(self) -> Result<usize, RenderError> {
84        byte_length(self.width, self.height).ok_or(RenderError::InvalidTile)
85    }
86}
87
88fn byte_length(width: u32, height: u32) -> Option<usize> {
89    usize::try_from(
90        u64::from(width)
91            .checked_mul(u64::from(height))?
92            .checked_mul(3)?,
93    )
94    .ok()
95}

268/// Writes RGB8 pixels in tile-local row-major order. Every pixel owns its full sample
269/// budget and random stream; tile IDs, worker order and partition size have no effect.
270pub fn render_tile(
271    scene: &Scene,
272    config: RenderConfig,
273    tile: Tile,
274    output: &mut [u8],
275) -> Result<RenderStats, RenderError> {
276    let tile = tile.validated(config)?;
277    if output.len() != tile.rgb_bytes()? {
278        return Err(RenderError::OutputLength);
279    }
280    let mut stats = RenderStats::default();
281    let aspect = f64::from(config.width) / f64::from(config.height);
282    let origin = CAMERA_ORIGIN;
283    let forward = (CAMERA_TARGET - origin).unit();
284    let right = forward.cross(Vec3::new(0.0, 1.0, 0.0)).unit();
285    let up = right.cross(forward);
286    let half_height = libm::tan(CAMERA_FOV * core::f64::consts::PI / 360.0);
287    let mut index = 0;
288    for y in tile.y..tile.y + tile.height {
289        for x in tile.x..tile.x + tile.width {
290            let mut color = Vec3::ZERO;
291            let pixel = u64::from(y) * u64::from(config.width) + u64::from(x);
292            for sample in 0..config.samples {
293                let mut rng = Rng(mix(config.seed
294                    ^ mix(pixel)
295                    ^ mix(u64::from(sample).wrapping_add(0x73c4_891d))));
296                let sx = (2.0 * (f64::from(x) + rng.uniform()) / f64::from(config.width) - 1.0)
297                    * aspect
298                    * half_height;
299                let sy = (1.0 - 2.0 * (f64::from(y) + rng.uniform()) / f64::from(config.height))
300                    * half_height;
301                color += trace(
302                    scene,
303                    Ray {
304                        origin,
305                        direction: (forward + right * sx + up * sy).unit(),
306                    },
307                    config.max_bounces,
308                    &mut rng,
309                    &mut stats,
310                );
311                stats.samples += 1;
312            }
313            color = color / f64::from(config.samples);
314            output[index] = encode(color.x);
315            output[index + 1] = encode(color.y);
316            output[index + 2] = encode(color.z);
317            index += 3;
318        }
319    }
320    Ok(stats)
321}

323fn trace(
324    scene: &Scene,
325    mut ray: Ray,
326    max_bounces: u32,
327    rng: &mut Rng,
328    stats: &mut RenderStats,
329) -> Vec3 {
330    let mut radiance = Vec3::ZERO;
331    let mut throughput = Vec3::ONE;
332    let mut specular = true;
333    for bounce in 0..max_bounces {
334        stats.rays += 1;
335        let Some(hit) = scene.acceleration.hit(ray, 1e-6, f64::INFINITY) else {
336            radiance += throughput * Vec3::new(0.018, 0.025, 0.04);
337            break;
338        };
339        let direction;
340        match hit.material {
341            Material::Emissive(emission) => {
342                // Diffuse paths already estimate the light with next-event sampling.
343                if specular && hit.front {
344                    radiance += throughput * emission;
345                }
346                break;
347            }
348            Material::Diffuse(albedo) => {
349                radiance += throughput * direct_light(scene, hit, albedo, rng, stats);
350                throughput = throughput * albedo;
351                direction = cosine_direction(hit.normal, rng);
352                specular = false;
353            }
354            Material::Metal(albedo) => {
355                direction = ray.direction - hit.normal * (2.0 * ray.direction.dot(hit.normal));
356                if direction.dot(hit.normal) <= 0.0 {
357                    break;
358                }
359                throughput = throughput * albedo;
360                specular = true;
361            }
362            Material::Dielectric(index) => {
363                let ratio = if hit.front { 1.0 / index } else { index };
364                let cosine = (-ray.direction.dot(hit.normal)).clamp(0.0, 1.0);
365                let sin_squared = (1.0 - cosine * cosine).max(0.0);
366                let base_reflectance = (1.0 - ratio) / (1.0 + ratio);
367                let r0 = base_reflectance * base_reflectance;
368                let fresnel = r0 + (1.0 - r0) * libm::pow(1.0 - cosine, 5.0);
369                if ratio * ratio * sin_squared > 1.0 || rng.uniform() < fresnel {
370                    direction = ray.direction + hit.normal * (2.0 * cosine);
371                } else {
372                    let perpendicular = (ray.direction + hit.normal * cosine) * ratio;
373                    direction = perpendicular
374                        - hit.normal * libm::sqrt((1.0 - perpendicular.length_squared()).max(0.0));
375                }
376                specular = true;
377            }
378        }
379        ray = Ray {
380            origin: offset(hit, direction),
381            direction: direction.unit(),
382        };
383        if bounce >= 4 {
384            let survival = throughput.max_component().clamp(0.05, 0.95);
385            if rng.uniform() >= survival {
386                break;
387            }
388            throughput = throughput / survival;
389        }
390    }
391    radiance
392}

394fn direct_light(
395    scene: &Scene,
396    hit: Hit,
397    albedo: Vec3,
398    rng: &mut Rng,
399    stats: &mut RenderStats,
400) -> Vec3 {
401    let light = scene.light;
402    let point = light.corner + light.edge_u * rng.uniform() + light.edge_v * rng.uniform();
403    let delta = point - hit.point;
404    let distance_squared = delta.length_squared();
405    if distance_squared <= 1e-12 {
406        return Vec3::ZERO;
407    }
408    let distance = libm::sqrt(distance_squared);
409    let direction = delta / distance;
410    let cosine_surface = hit.normal.dot(direction).max(0.0);
411    let cosine_light = light.normal.dot(-direction).max(0.0);
412    if cosine_surface == 0.0 || cosine_light == 0.0 {
413        return Vec3::ZERO;
414    }
415    stats.rays += 1;
416    let origin = offset(hit, direction);
417    // Use the offset-to-light distance, so the sampled emitter cannot self-occlude.
418    let shadow_delta = point - origin;
419    let shadow_distance = libm::sqrt(shadow_delta.length_squared());
420    let shadow = Ray {
421        origin,
422        direction: shadow_delta / shadow_distance,
423    };
424    if scene
425        .acceleration
426        .hit(shadow, 1e-6, shadow_distance - 2e-5)
427        .is_some()
428    {
429        return Vec3::ZERO;
430    }
431    albedo
432        * light.emission
433        * (cosine_surface * cosine_light * light.area / (core::f64::consts::PI * distance_squared))
434}

436fn offset(hit: Hit, direction: Vec3) -> Vec3 {
437    let sign = if direction.dot(hit.normal) >= 0.0 {
438        1.0
439    } else {
440        -1.0
441    };
442    hit.point + hit.normal * (sign * 1e-5)
443}

445fn cosine_direction(normal: Vec3, rng: &mut Rng) -> Vec3 {
446    let angle = 2.0 * core::f64::consts::PI * rng.uniform();
447    let radius_squared = rng.uniform();
448    let radius = libm::sqrt(radius_squared);
449    let helper = if normal.x.abs() > 0.9 {
450        Vec3::new(0.0, 1.0, 0.0)
451    } else {
452        Vec3::new(1.0, 0.0, 0.0)
453    };
454    let tangent = helper.cross(normal).unit();
455    let bitangent = normal.cross(tangent);
456    tangent * (radius * libm::cos(angle))
457        + bitangent * (radius * libm::sin(angle))
458        + normal * libm::sqrt(1.0 - radius_squared)
459}

461fn encode(linear: f64) -> u8 {
462    // Finite nonnegative radiance, Reinhard exposure mapping and gamma-2 display encoding.
463    if !linear.is_finite() || linear <= 0.0 {
464        return 0;
465    }
466    (libm::sqrt(linear / (1.0 + linear)) * 255.0 + 0.5).clamp(0.0, 255.0) as u8
467}

469fn hash_word(mut hash: u64, word: u64) -> u64 {
470    for byte in word.to_le_bytes() {
471        hash = (hash ^ u64::from(byte)).wrapping_mul(0x100000001b3);
472    }
473    hash
474}

476fn mix(mut value: u64) -> u64 {
477    value = (value ^ (value >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
478    value = (value ^ (value >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
479    value ^ (value >> 31)
480}

482struct Rng(u64);
483impl Rng {
484    fn uniform(&mut self) -> f64 {
485        self.0 = self.0.wrapping_add(0x9e37_79b9_7f4a_7c15);
486        ((mix(self.0) >> 11) as f64) * (1.0 / 9007199254740992.0)
487    }
488}

507        let scene = Scene::demo();
508        let config = config();
509        let full = Tile {
510            id: 0,

512            y: 0,
513            width: config.width,
514            height: config.height,
515        };
516        let mut reference = vec![0; config.rgb_bytes().unwrap()];
517        let reference_stats = render_tile(&scene, config, full, &mut reference).unwrap();
518        let mut assembled = vec![0; reference.len()];
519        let mut total = RenderStats::default();
520        for y in (0..config.height).rev() {
521            for x in (0..config.width).rev() {
522                let tile = Tile {
523                    id: u32::MAX,
524                    x,
525                    y,
526                    width: 1,
527                    height: 1,
528                };
529                let mut pixel = [0; 3];
530                let stats = render_tile(&scene, config, tile, &mut pixel).unwrap();
531                let index = ((y * config.width + x) * 3) as usize;
532                assembled[index..index + 3].copy_from_slice(&pixel);
533                total.samples += stats.samples;
534                total.rays += stats.rays;
535            }
536        }
537        assert_eq!(reference, assembled);
538        assert_eq!(reference_stats, total);
539        assert_eq!(total.samples, 19 * 13 * 4);
540        assert!(total.rays >= total.samples);
