40fn filled_stat(byte: u8) -> core::mem::MaybeUninit<Stat> {
41    let mut value = core::mem::MaybeUninit::<Stat>::uninit();
42    unsafe {
43        core::ptr::write_bytes(
44            value.as_mut_ptr() as *mut u8,
45            byte,
46            core::mem::size_of::<Stat>(),
47        );
48    }
49    value
50}

52fn stat_bytes(value: &core::mem::MaybeUninit<Stat>) -> &[u8] {
53    unsafe {
54        core::slice::from_raw_parts(value.as_ptr() as *const u8, core::mem::size_of::<Stat>())
55    }
56}

58fn validate_open_fstat(fd: i32) -> bool {
59    let mut wire = api::syscall::ViFstatV1::default();
60    if ostd::syscall::sys_fstat(fd as usize, &mut wire) != Ok(api::syscall::VI_FSTAT_V1_LEN)
61        || wire.kind != api::syscall::VI_FSTAT_KIND_REGULAR
62        || wire.access != api::syscall::VI_FSTAT_ACCESS_READ
63        || wire.reserved != [0; 2]
64        || wire.size == 0
65    {
66        return false;
67    }
68    let Ok(expected_size) = c_long::try_from(wire.size) else {
69        return false;
70    };
71
72    let mut actual = filled_stat(0xA5);
73    if unsafe { fstat(fd, actual.as_mut_ptr()) } != 0 {
74        return false;
75    }
76    let actual_ref = unsafe { &*actual.as_ptr() };
77    if actual_ref.st_mode != 0o100000 || actual_ref.st_size != expected_size {
78        return false;
79    }
80
81    let mut expected = filled_stat(0);
82    unsafe {
83        core::ptr::addr_of_mut!((*expected.as_mut_ptr()).st_mode).write(0o100000);
84        core::ptr::addr_of_mut!((*expected.as_mut_ptr()).st_size).write(expected_size);
85    }
86    if stat_bytes(&actual) != stat_bytes(&expected) {
87        return false;
88    }
89
90    let mut invalid = filled_stat(0xA5);
91    let invalid_ret = unsafe { fstat(-1, invalid.as_mut_ptr()) };
92    invalid_ret == -1
93        && stat_bytes(&invalid).iter().all(|byte| *byte == 0xA5)
94        && unsafe { fstat(fd, core::ptr::null_mut()) } == -1
95}

97pub(super) fn test_fstat() {
98    const PATH: &[u8] = b"/BIN/INIT\0";
99    let fd = unsafe { open(PATH.as_ptr() as *const c_char, 0, 0) };
100    if fd < 0 {
101        println("[posix-shim] POSIX-FSTAT-OPEN: FAIL");
102        return;
103    }
104    println("[posix-shim] POSIX-FSTAT-OPEN: OK");
105
106    let valid = validate_open_fstat(fd);
107    let close_ret = unsafe { close(fd) };
108    if valid && close_ret == 0 {
109        println("[posix-shim] POSIX-FSTAT: OK");
110    } else {
111        println(&format!(
112            "[posix-shim] POSIX-FSTAT: FAIL valid={valid} close={close_ret}"
113        ));
114    }
115}

117pub(super) fn test_stat() {
118    const PATH: &[u8] = b"/BIN/INIT\0";
119    let mut actual = filled_stat(0xA5);
120    let ret = unsafe { stat(PATH.as_ptr() as *const c_char, actual.as_mut_ptr()) };
121    if ret != 0 {
122        println("[posix-shim] POSIX-STAT: FAIL ret != 0");
123        return;
124    }
125    let actual_ref = unsafe { &*actual.as_ptr() };
126    if actual_ref.st_mode != 0o100000 || actual_ref.st_size == 0 {
127        println(&format!(
128            "[posix-shim] POSIX-STAT: FAIL mode={:#o} size={}",
129            actual_ref.st_mode, actual_ref.st_size
130        ));
131        return;
132    }
133
134    // Negative tests:
135    let mut invalid = filled_stat(0xA5);
136    let null_path = unsafe { stat(core::ptr::null(), invalid.as_mut_ptr()) };
137    let null_buf = unsafe { stat(PATH.as_ptr() as *const c_char, core::ptr::null_mut()) };
138    let non_existent = unsafe {
139        stat(
140            b"/NONEXISTENT_FILE_12345\0".as_ptr() as *const c_char,
141            invalid.as_mut_ptr(),
142        )
143    };
144
145    if null_path == -1
146        && null_buf == -1
147        && non_existent == -1
148        && stat_bytes(&invalid).iter().all(|byte| *byte == 0xA5)
149    {
150        println("[posix-shim] POSIX-STAT: OK");
151    } else {
152        println("[posix-shim] POSIX-STAT: FAIL negative checks");
153    }
154}

156pub(super) fn test_unlink() {
157    // Calling unlink on NULL or non-existent file must fail closed (-1)
158    let null_ret = unsafe { unlink(core::ptr::null()) };
159    if null_ret != -1 {
160        println("[posix-shim] POSIX-UNLINK: FAIL null_ret != -1");
161        return;
162    }
163
164    let non_existent =
165        unsafe { unlink(b"/tmp/nonexistent_posix_unlink_12345\0".as_ptr() as *const c_char) };
166    if non_existent != -1 {
167        println("[posix-shim] POSIX-UNLINK: FAIL non_existent != -1");
168        return;
169    }
170
171    // Positive create -> stat (exists) -> unlink (success) -> stat (gone) -> unlink (fails)
172    const SMOKE_PATH: &str = "/tmp/posix_unlink_smoke.txt";
173    const SMOKE_CSTR: &[u8] = b"/tmp/posix_unlink_smoke.txt\0";
174
175    let mut vfs = ostd::clients::VfsClient::new();
176    if vfs
177        .write_file(SMOKE_PATH, b"posix unlink smoke data\n")
178        .is_err()
179    {
180        println("[posix-shim] POSIX-UNLINK: FAIL create file");
181        return;
182    }
183
184    // Verify stat confirms file exists
185    let mut st = filled_stat(0);
186    let stat_before = unsafe { stat(SMOKE_CSTR.as_ptr() as *const c_char, st.as_mut_ptr()) };
187    if stat_before != 0 {
188        println("[posix-shim] POSIX-UNLINK: FAIL stat before unlink");
189        return;
190    }
191
192    // Call unlink -> must return 0
193    let unlink_ret = unsafe { unlink(SMOKE_CSTR.as_ptr() as *const c_char) };
194    if unlink_ret != 0 {
195        println(&format!(
196            "[posix-shim] POSIX-UNLINK: FAIL unlink ret={unlink_ret}"
197        ));
198        return;
199    }
200
201    // Verify stat confirms file is gone
202    let mut st_after = filled_stat(0xA5);
203    let stat_after = unsafe { stat(SMOKE_CSTR.as_ptr() as *const c_char, st_after.as_mut_ptr()) };
204    if stat_after != -1 {
205        println("[posix-shim] POSIX-UNLINK: FAIL file still exists after unlink");
206        return;
207    }
208
209    // Second unlink on same file must fail closed (-1)
210    let unlink_again = unsafe { unlink(SMOKE_CSTR.as_ptr() as *const c_char) };
211    if unlink_again != -1 {
212        println("[posix-shim] POSIX-UNLINK: FAIL second unlink succeeded");
213        return;
214    }
215
216    println("[posix-shim] POSIX-UNLINK: OK");
217}

219pub(super) fn test_mkdir_rmdir() {
220    const PATH: &[u8] = b"/srv/posix_mkdir_rmdir_smoke\0";
221    const MISSING: &[u8] = b"/srv/posix_mkdir_rmdir_missing\0";
222    const INVALID_UTF8: &[u8] = b"/srv/posix_mkdir_rmdir_\xFF\0";
223
224    if unsafe { mkdir(core::ptr::null(), 0o755) } != -1 || unsafe { rmdir(core::ptr::null()) } != -1
225    {
226        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL null path succeeded");
227        return;
228    }
229    if unsafe { mkdir(INVALID_UTF8.as_ptr() as *const c_char, 0o755) } != -1
230        || unsafe { rmdir(INVALID_UTF8.as_ptr() as *const c_char) } != -1
231    {
232        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL invalid UTF-8 succeeded");
233        return;
234    }
235    if unsafe { rmdir(MISSING.as_ptr() as *const c_char) } != -1 {
236        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL missing directory removed");
237        return;
238    }
239    const FILE_PATH: &str = "/srv/posix_mkdir_rmdir_regular_file";
240    const FILE_CSTR: &[u8] = b"/srv/posix_mkdir_rmdir_regular_file\0";
241    let mut vfs = ostd::clients::VfsClient::new();
242    if vfs
243        .write_file(FILE_PATH, b"rmdir must not delete this file\n")
244        .is_err()
245    {
246        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL create regular file");
247        return;
248    }
249    if unsafe { rmdir(FILE_CSTR.as_ptr() as *const c_char) } != -1 {
250        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL rmdir regular file succeeded");
251        return;
252    }
253    let mut regular = filled_stat(0xA5);
254    if unsafe { stat(FILE_CSTR.as_ptr() as *const c_char, regular.as_mut_ptr()) } != 0
255        || unsafe { (*regular.as_ptr()).st_mode } != 0o100000
256    {
257        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL regular file was removed");
258        return;
259    }
260    if unsafe { unlink(FILE_CSTR.as_ptr() as *const c_char) } != 0 {
261        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL cleanup regular file");
262        return;
263    }
264    if unsafe { mkdir(PATH.as_ptr() as *const c_char, 0o755) } != 0 {
265        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL mkdir");
266        return;
267    }
268    let mut before = filled_stat(0xA5);
269    if unsafe { stat(PATH.as_ptr() as *const c_char, before.as_mut_ptr()) } != 0
270        || unsafe { (*before.as_ptr()).st_mode } != 0o040000
271    {
272        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL directory stat");
273        return;
274    }
275
276    if unsafe { rmdir(PATH.as_ptr() as *const c_char) } != 0 {
277        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL rmdir");
278        return;
279    }
280    let mut after = filled_stat(0xA5);
281    if unsafe { stat(PATH.as_ptr() as *const c_char, after.as_mut_ptr()) } != -1
282        || !stat_bytes(&after).iter().all(|byte| *byte == 0xA5)
283    {
284        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL removed directory stat");
285        return;
286    }
287    if unsafe { rmdir(PATH.as_ptr() as *const c_char) } != -1 {
288        println("[posix-shim] POSIX-MKDIR-RMDIR: FAIL second rmdir succeeded");
289        return;
290    }
291
292    println("[posix-shim] POSIX-MKDIR-RMDIR: OK");
293}

295pub(super) fn test_rename() {
296    // Calling rename on NULL pointers must fail closed (-1)
297    let null_both = unsafe { rename(core::ptr::null(), core::ptr::null()) };
298    if null_both != -1 {
299        println("[posix-shim] POSIX-RENAME: FAIL null_both != -1");
300        return;
301    }
302
303    let non_existent_src = unsafe {
304        rename(
305            b"/srv/nonexistent_src_12345\0".as_ptr() as *const c_char,
306            b"/srv/nonexistent_dst_12345\0".as_ptr() as *const c_char,
307        )
308    };
309    if non_existent_src != -1 {
310        println("[posix-shim] POSIX-RENAME: FAIL non_existent_src != -1");
311        return;
312    }
313
314    // Calling rename on non-/srv path must fail closed (-1) (VFS backend constraint)
315    let non_srv = unsafe {
316        rename(
317            b"/tmp/non_srv_src\0".as_ptr() as *const c_char,
318            b"/tmp/non_srv_dst\0".as_ptr() as *const c_char,
319        )
320    };
321    if non_srv != -1 {
322        println("[posix-shim] POSIX-RENAME: FAIL non_srv != -1");
323        return;
324    }
325
326    // Positive create -> stat (exists) -> rename -> stat old (gone) -> stat new (exists) -> unlink (cleanup)
327    const RN_SRC_PATH: &str = "/srv/posix_rename_src.txt";
328    const RN_SRC_CSTR: &[u8] = b"/srv/posix_rename_src.txt\0";
329    const RN_DST_CSTR: &[u8] = b"/srv/posix_rename_dst.txt\0";
330
331    let mut vfs = ostd::clients::VfsClient::new();
332    if vfs
333        .write_file(RN_SRC_PATH, b"posix rename smoke data\n")
334        .is_err()
335    {
336        println("[posix-shim] POSIX-RENAME: FAIL create src file");
337        return;
338    }
339
340    // Verify stat confirms src file exists
341    let mut st = filled_stat(0);
342    let stat_before = unsafe { stat(RN_SRC_CSTR.as_ptr() as *const c_char, st.as_mut_ptr()) };
343    if stat_before != 0 {
344        println("[posix-shim] POSIX-RENAME: FAIL stat before rename");
345        return;
346    }
347
348    // Call rename -> must return 0
349    let rn_ret = unsafe {
350        rename(
351            RN_SRC_CSTR.as_ptr() as *const c_char,
352            RN_DST_CSTR.as_ptr() as *const c_char,
353        )
354    };
355    if rn_ret != 0 {
356        println(&format!(
357            "[posix-shim] POSIX-RENAME: FAIL rename ret={rn_ret}"
358        ));
359        return;
360    }
361
362    // Verify stat confirms src file is gone
363    let mut st_old = filled_stat(0xAA);
364    let stat_old = unsafe { stat(RN_SRC_CSTR.as_ptr() as *const c_char, st_old.as_mut_ptr()) };
365    if stat_old != -1 {
366        println("[posix-shim] POSIX-RENAME: FAIL src file still exists after rename");
367        return;
368    }
369
370    // Verify stat confirms dst file exists
371    let mut st_new = filled_stat(0x55);
372    let stat_new = unsafe { stat(RN_DST_CSTR.as_ptr() as *const c_char, st_new.as_mut_ptr()) };
373    if stat_new != 0 {
374        println("[posix-shim] POSIX-RENAME: FAIL dst file does not exist after rename");
375        return;
376    }
377
378    // Cleanup dst file via unlink
379    let unlink_ret = unsafe { unlink(RN_DST_CSTR.as_ptr() as *const c_char) };
380    if unlink_ret != 0 {
381        println("[posix-shim] POSIX-RENAME: FAIL cleanup unlink dst");
382        return;
383    }
384
385    println("[posix-shim] POSIX-RENAME: OK");
386}

388/// Exercise the published raw Rename ABI. Its kernel RPC must retain this
389/// cell's attested VfsMutate identity; a queue acknowledgement is insufficient.
390pub(super) fn test_raw_rename() {
391    const SRC_PATH: &str = "/srv/raw_rename_src.txt";
392    const SRC_CSTR: &[u8] = b"/srv/raw_rename_src.txt\0";
393    const DST_CSTR: &[u8] = b"/srv/raw_rename_dst.txt\0";
394
395    let mut vfs = ostd::clients::VfsClient::new();
396    if ostd::syscall::sys_rename(
397        "/srv/raw_rename_missing.txt",
398        "/srv/raw_rename_missing_dst.txt",
399    )
400    .is_ok()
401    {
402        println("[posix-shim] RAW-RENAME: FAIL missing source succeeded");
403        return;
404    }
405
406    if vfs
407        .write_file(SRC_PATH, b"raw rename smoke data\n")
408        .is_err()
409    {
410        println("[posix-shim] RAW-RENAME: FAIL create src file");
411        return;
412    }
413    if ostd::syscall::sys_rename(SRC_PATH, "/srv/raw_rename_dst.txt").is_err() {
414        println("[posix-shim] RAW-RENAME: FAIL syscall");
415        return;
416    }
417
418    let mut old = filled_stat(0xAA);
419    if unsafe { stat(SRC_CSTR.as_ptr() as *const c_char, old.as_mut_ptr()) } != -1 {
420        println("[posix-shim] RAW-RENAME: FAIL src file still exists");
421        return;
422    }
423    let mut new = filled_stat(0x55);
424    if unsafe { stat(DST_CSTR.as_ptr() as *const c_char, new.as_mut_ptr()) } != 0 {
425        println("[posix-shim] RAW-RENAME: FAIL dst file missing");
426        return;
427    }
428    if unsafe { unlink(DST_CSTR.as_ptr() as *const c_char) } != 0 {
429        println("[posix-shim] RAW-RENAME: FAIL cleanup");
430        return;
431    }
432    println("[posix-shim] RAW-RENAME: OK");
