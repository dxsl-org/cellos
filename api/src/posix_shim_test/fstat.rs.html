34fn filled_stat(byte: u8) -> core::mem::MaybeUninit<Stat> {
35    let mut value = core::mem::MaybeUninit::<Stat>::uninit();
36    unsafe {
37        core::ptr::write_bytes(
38            value.as_mut_ptr() as *mut u8,
39            byte,
40            core::mem::size_of::<Stat>(),
41        );
42    }
43    value
44}

46fn stat_bytes(value: &core::mem::MaybeUninit<Stat>) -> &[u8] {
47    unsafe {
48        core::slice::from_raw_parts(value.as_ptr() as *const u8, core::mem::size_of::<Stat>())
49    }
50}

52fn validate_open_fstat(fd: i32) -> bool {
53    let mut wire = api::syscall::ViFstatV1::default();
54    if ostd::syscall::sys_fstat(fd as usize, &mut wire) != Ok(api::syscall::VI_FSTAT_V1_LEN)
55        || wire.kind != api::syscall::VI_FSTAT_KIND_REGULAR
56        || wire.access != api::syscall::VI_FSTAT_ACCESS_READ
57        || wire.reserved != [0; 2]
58        || wire.size == 0
59    {
60        return false;
61    }
62    let Ok(expected_size) = c_long::try_from(wire.size) else {
63        return false;
64    };
65
66    let mut actual = filled_stat(0xA5);
67    if unsafe { fstat(fd, actual.as_mut_ptr()) } != 0 {
68        return false;
69    }
70    let actual_ref = unsafe { &*actual.as_ptr() };
71    if actual_ref.st_mode != 0o100000 || actual_ref.st_size != expected_size {
72        return false;
73    }
74
75    let mut expected = filled_stat(0);
76    unsafe {
77        core::ptr::addr_of_mut!((*expected.as_mut_ptr()).st_mode).write(0o100000);
78        core::ptr::addr_of_mut!((*expected.as_mut_ptr()).st_size).write(expected_size);
79    }
80    if stat_bytes(&actual) != stat_bytes(&expected) {
81        return false;
82    }
83
84    let mut invalid = filled_stat(0xA5);
85    let invalid_ret = unsafe { fstat(-1, invalid.as_mut_ptr()) };
86    invalid_ret == -1
87        && stat_bytes(&invalid).iter().all(|byte| *byte == 0xA5)
88        && unsafe { fstat(fd, core::ptr::null_mut()) } == -1
89}

91pub(super) fn test_fstat() {
92    const PATH: &[u8] = b"/BIN/INIT\0";
93    let fd = unsafe { open(PATH.as_ptr() as *const c_char, 0, 0) };
94    if fd < 0 {
95        println("[posix-shim] POSIX-FSTAT-OPEN: FAIL");
96        return;
97    }
98    println("[posix-shim] POSIX-FSTAT-OPEN: OK");
99
100    let valid = validate_open_fstat(fd);
101    let close_ret = unsafe { close(fd) };
102    if valid && close_ret == 0 {
103        println("[posix-shim] POSIX-FSTAT: OK");
104    } else {
105        println(&format!(
106            "[posix-shim] POSIX-FSTAT: FAIL valid={valid} close={close_ret}"
107        ));
108    }
109}

111pub(super) fn test_stat() {
112    const PATH: &[u8] = b"/BIN/INIT\0";
113    let mut actual = filled_stat(0xA5);
114    let ret = unsafe { stat(PATH.as_ptr() as *const c_char, actual.as_mut_ptr()) };
115    if ret != 0 {
116        println("[posix-shim] POSIX-STAT: FAIL ret != 0");
117        return;
118    }
119    let actual_ref = unsafe { &*actual.as_ptr() };
120    if actual_ref.st_mode != 0o100000 || actual_ref.st_size == 0 {
121        println(&format!(
122            "[posix-shim] POSIX-STAT: FAIL mode={:#o} size={}",
123            actual_ref.st_mode, actual_ref.st_size
124        ));
125        return;
126    }
127
128    // Negative tests:
129    let mut invalid = filled_stat(0xA5);
130    let null_path = unsafe { stat(core::ptr::null(), invalid.as_mut_ptr()) };
131    let null_buf = unsafe { stat(PATH.as_ptr() as *const c_char, core::ptr::null_mut()) };
132    let non_existent = unsafe {
133        stat(
134            b"/NONEXISTENT_FILE_12345\0".as_ptr() as *const c_char,
135            invalid.as_mut_ptr(),
136        )
137    };
138
139    if null_path == -1
140        && null_buf == -1
141        && non_existent == -1
142        && stat_bytes(&invalid).iter().all(|byte| *byte == 0xA5)
143    {
144        println("[posix-shim] POSIX-STAT: OK");
145    } else {
146        println("[posix-shim] POSIX-STAT: FAIL negative checks");
147    }
148}

150pub(super) fn test_unlink() {
151    // Calling unlink on NULL or non-existent file must fail closed (-1)
152    let null_ret = unsafe { unlink(core::ptr::null()) };
153    if null_ret != -1 {
154        println("[posix-shim] POSIX-UNLINK: FAIL null_ret != -1");
155        return;
156    }
157
158    let non_existent = unsafe {
159        unlink(b"/tmp/nonexistent_posix_unlink_12345\0".as_ptr() as *const c_char)
160    };
161    if non_existent != -1 {
162        println("[posix-shim] POSIX-UNLINK: FAIL non_existent != -1");
163        return;
164    }
165
166    // Positive create -> stat (exists) -> unlink (success) -> stat (gone) -> unlink (fails)
167    const SMOKE_PATH: &str = "/tmp/posix_unlink_smoke.txt";
168    const SMOKE_CSTR: &[u8] = b"/tmp/posix_unlink_smoke.txt\0";
169
170    let mut vfs = ostd::clients::VfsClient::new();
171    if vfs.write_file(SMOKE_PATH, b"posix unlink smoke data\n").is_err() {
172        println("[posix-shim] POSIX-UNLINK: FAIL create file");
173        return;
174    }
175
176    // Verify stat confirms file exists
177    let mut st = filled_stat(0);
178    let stat_before = unsafe { stat(SMOKE_CSTR.as_ptr() as *const c_char, st.as_mut_ptr()) };
179    if stat_before != 0 {
180        println("[posix-shim] POSIX-UNLINK: FAIL stat before unlink");
181        return;
182    }
183
184    // Call unlink -> must return 0
185    let unlink_ret = unsafe { unlink(SMOKE_CSTR.as_ptr() as *const c_char) };
186    if unlink_ret != 0 {
187        println(&format!(
188            "[posix-shim] POSIX-UNLINK: FAIL unlink ret={unlink_ret}"
189        ));
190        return;
191    }
192
193    // Verify stat confirms file is gone
194    let mut st_after = filled_stat(0xA5);
195    let stat_after = unsafe { stat(SMOKE_CSTR.as_ptr() as *const c_char, st_after.as_mut_ptr()) };
196    if stat_after != -1 {
197        println("[posix-shim] POSIX-UNLINK: FAIL file still exists after unlink");
198        return;
199    }
200
201    // Second unlink on same file must fail closed (-1)
202    let unlink_again = unsafe { unlink(SMOKE_CSTR.as_ptr() as *const c_char) };
203    if unlink_again != -1 {
204        println("[posix-shim] POSIX-UNLINK: FAIL second unlink succeeded");
205        return;
206    }
207
208    println("[posix-shim] POSIX-UNLINK: OK");
