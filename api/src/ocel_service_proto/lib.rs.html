2//! Allocation-free, allowlisted engine activation. Caller identity never comes
3//! from these frames: init obtains it from the kernel's receive trailer.
4#![no_std]
5#![forbid(unsafe_code)]
6
7pub const FRAME_BYTES: usize = 32;
8pub const MAX_LEASES: usize = 16;
9const REQUEST_MAGIC: &[u8; 8] = b"OCLSREQ1";
10const RESPONSE_MAGIC: &[u8; 8] = b"OCLSRSP1";
11
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13#[repr(u8)]
14pub enum Engine { JavaScript = 1, Pdf = 2 }
15impl Engine {
16    pub const fn path(self) -> &'static str {
17        match self { Self::JavaScript => "/bin/ocel-quickjs", Self::Pdf => "/bin/ocel-pdf" }
18    }
19    fn decode(value: u8) -> Option<Self> {
20        match value { 1 => Some(Self::JavaScript), 2 => Some(Self::Pdf), _ => None }
21    }
22}
23
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub enum Request {
26    Acquire { engine: Engine },
27    Release { engine: Engine, lease: u64 },
28}
29impl Request {
30    pub const fn engine(self) -> Engine {
31        match self { Self::Acquire { engine } | Self::Release { engine, .. } => engine }
32    }
33    pub fn encode(self) -> [u8; FRAME_BYTES] {
34        let mut frame = [0; FRAME_BYTES];
35        frame[..8].copy_from_slice(REQUEST_MAGIC);
36        frame[9] = self.engine() as u8;
37        match self {
38            Self::Acquire { .. } => frame[8] = 1,
39            Self::Release { lease, .. } => { frame[8] = 2; frame[16..24].copy_from_slice(&lease.to_le_bytes()); }
40        }
41        frame
42    }
43    pub fn decode(frame: &[u8]) -> Option<Self> {
44        if frame.len() < FRAME_BYTES || &frame[..8] != REQUEST_MAGIC
45            || frame[10..16].iter().any(|byte| *byte != 0)
46            || frame[24..32].iter().any(|byte| *byte != 0) { return None; }
47        let engine = Engine::decode(frame[9])?;
48        let lease = u64::from_le_bytes(frame[16..24].try_into().ok()?);
49        match (frame[8], lease) {
50            (1, 0) => Some(Self::Acquire { engine }),
51            (2, lease) if lease != 0 => Some(Self::Release { engine, lease }),
52            _ => None,
53        }
54    }
55}

57#[derive(Clone, Copy, Debug, PartialEq, Eq)]
58#[repr(u8)]
59pub enum Failure { Unavailable = 2, Denied = 3, Busy = 4, Failed = 5 }
60impl Failure {
61    fn decode(value: u8) -> Option<Self> {
62        match value { 2 => Some(Self::Unavailable), 3 => Some(Self::Denied), 4 => Some(Self::Busy), 5 => Some(Self::Failed), _ => None }
63    }
64}

66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67pub enum Response {
68    Ready { engine: Engine, lease: u64, tid: u64 },
69    Released { engine: Engine },
70    Error { engine: Engine, failure: Failure },
71}
72impl Response {
73    pub const fn engine(self) -> Engine {
74        match self { Self::Ready { engine, .. } | Self::Released { engine } | Self::Error { engine, .. } => engine }
75    }
76    pub fn encode(self) -> [u8; FRAME_BYTES] {
77        let mut frame = [0; FRAME_BYTES];
78        frame[..8].copy_from_slice(RESPONSE_MAGIC);
79        frame[9] = self.engine() as u8;
80        match self {
81            Self::Ready { lease, tid, .. } => {
82                frame[16..24].copy_from_slice(&lease.to_le_bytes());
83                frame[24..32].copy_from_slice(&tid.to_le_bytes());
84            }
85            Self::Released { .. } => frame[8] = 1,
86            Self::Error { failure, .. } => frame[8] = failure as u8,
87        }
88        frame
89    }
90    pub fn decode(frame: &[u8]) -> Option<Self> {
91        if frame.len() < FRAME_BYTES || &frame[..8] != RESPONSE_MAGIC
92            || frame[10..16].iter().any(|byte| *byte != 0) { return None; }
93        let engine = Engine::decode(frame[9])?;
94        let lease = u64::from_le_bytes(frame[16..24].try_into().ok()?);
95        let tid = u64::from_le_bytes(frame[24..32].try_into().ok()?);
96        match (frame[8], lease, tid) {
97            (0, lease, tid) if lease != 0 && tid != 0 => Some(Self::Ready { engine, lease, tid }),
98            (1, 0, 0) => Some(Self::Released { engine }),
99            (status, 0, 0) => Some(Self::Error { engine, failure: Failure::decode(status)? }),
100            _ => None,
101        }
102    }
103}

105#[cfg(test)]
106mod tests {
107    use super::*;
108    #[test]
109    fn malformed_selectors_and_lease_shapes_fail_closed() {
110        let mut acquire = Request::Acquire { engine: Engine::Pdf }.encode();
111        for selector in [0, 3, 255] { acquire[9] = selector; assert!(Request::decode(&acquire).is_none()); }
112        acquire[9] = Engine::Pdf as u8;
113        acquire[16] = 1;
114        assert!(Request::decode(&acquire).is_none());
115        let release = Request::Release { engine: Engine::Pdf, lease: 0 }.encode();
116        assert!(Request::decode(&release).is_none());
117        let mut valid = Request::Acquire { engine: Engine::JavaScript }.encode();
118        valid[10] = 1;
119        assert!(Request::decode(&valid).is_none());
120        assert!(Request::decode(&valid[..31]).is_none());
121    }
122    #[test]
123    fn provider_response_cannot_hide_invalid_handles_or_status() {
124        for (lease, tid) in [(0, 1), (1, 0), (0, 0)] {
125            assert!(Response::decode(&Response::Ready { engine: Engine::Pdf, lease, tid }.encode()).is_none());
126        }
127        let mut error = Response::Error { engine: Engine::Pdf, failure: Failure::Unavailable }.encode();
128        error[24] = 1;
129        assert!(Response::decode(&error).is_none());
130        error[24] = 0; error[8] = 255;
131        assert!(Response::decode(&error).is_none());
132    }
