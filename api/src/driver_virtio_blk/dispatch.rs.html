1//! IPC request dispatcher — translates `DrvRequest` messages into VirtIO-blk I/O.
2//!
3//! Wire format: `types::block_wire`, shared with the AHCI/NVMe cells so VFS
4//! speaks one block protocol.
5
6use crate::device::BlkDevice;
7use types::block_wire::{self, BlockOutcome, ReadRunRequest, SECTOR_SIZE};
8
9pub const REPLY_SIZE: usize = block_wire::MAX_REPLY_SIZE;
10
11#[derive(PartialEq, Eq)]
12enum DrvOp {
13    Read,
14    Write,
15    Flush,
16}
17
18fn parse_op(data: &[u8]) -> Option<(DrvOp, u64)> {
19    if data.len() < 2 {
20        return None;
21    }
22    let op = match u16::from_le_bytes([data[0], data[1]]) {
23        block_wire::OP_READ => DrvOp::Read,
24        block_wire::OP_WRITE => DrvOp::Write,
25        block_wire::OP_FLUSH => return Some((DrvOp::Flush, 0)),
26        _ => return None,
27    };
28    if data.len() < 10 {
29        return None;
30    }
31    let sector = u64::from_le_bytes(data[2..10].try_into().ok()?);
32    Some((op, sector))
33}
34
35fn status(out: &mut [u8; REPLY_SIZE], outcome: BlockOutcome) -> usize {
36    out[0] = outcome.to_byte();
37    1
38}
39
40/// Handle one incoming IPC message. Writes the reply into `out` and returns the
41/// number of bytes to send back: the outcome byte plus the sectors of a
42/// completed read, else 1.
43pub fn handle(dev: &mut BlkDevice, data: &[u8], out: &mut [u8; REPLY_SIZE]) -> usize {
44    if let Some(request) = ReadRunRequest::decode(data) {
45        // One virtio request for the whole run (at most 3584 bytes, inside one
46        // bounce page on either transport); the device refuses a sector past
47        // its capacity itself.
48        return block_wire::serve_read_run(&request, u64::MAX, out, |lba, run| {
49            dev.read_sectors(lba, run)
50        });
51    }
52    let Some((op, sector)) = parse_op(data) else {
53        return status(out, BlockOutcome::NotIssued);
54    };
55
56    match op {
57        DrvOp::Read => {
58            let mut sec = [0u8; SECTOR_SIZE];
59            match dev.read_sectors(sector, &mut sec) {
60                Ok(()) => {
61                    out[0] = BlockOutcome::Completed.to_byte();
62                    out[1..block_wire::REPLY_SIZE].copy_from_slice(&sec);
63                    block_wire::REPLY_SIZE
64                }
65                Err(outcome) => status(out, outcome),
66            }
67        }
68        DrvOp::Write => {
69            if data.len() < 10 + SECTOR_SIZE {
70                return status(out, BlockOutcome::NotIssued);
71            }
72            let mut sec = [0u8; SECTOR_SIZE];
73            sec.copy_from_slice(&data[10..10 + SECTOR_SIZE]);
74            match dev.write_sector(sector, &sec) {
75                Ok(()) => status(out, BlockOutcome::Completed),
76                Err(outcome) => status(out, outcome),
77            }
78        }
79        DrvOp::Flush => match dev.flush() {
80            Ok(()) => status(out, BlockOutcome::Completed),
81            Err(outcome) => status(out, outcome),
82        },
83    }
84}
