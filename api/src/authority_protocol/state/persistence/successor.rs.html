7/// Verify that `next` is one legal persisted Phase-2 transition after `previous`.
8pub fn verify_protected_successor(
9    previous: &ProtectedAuthorityRecord,
10    next: &ProtectedAuthorityRecord,
11) -> Result<(), AuthorityFault> {
12    let valid = previous.invariants_hold()
13        && next.invariants_hold()
14        && previous.revision.checked_add(1) == Some(next.revision)
15        && previous.device_id == next.device_id
16        && previous.authority_id == next.authority_id
17        && previous.authority_epoch == next.authority_epoch
18        && previous.state_epoch == next.state_epoch
19        && previous.approved_loader_digest == next.approved_loader_digest
20        && previous.boot_floor <= next.boot_floor
21        && previous.generation_floor <= next.generation_floor
22        && previous.last_request_sequence <= next.last_request_sequence
23        && state_advances(previous, next)
24        && relay_advances(previous, next);
25    valid
26        .then_some(())
27        .ok_or(AuthorityFault::PersistenceFailure)
28}

30fn state_advances(previous: &ProtectedAuthorityRecord, next: &ProtectedAuthorityRecord) -> bool {
31    if previous.mode == AuthorityMode::Sealed {
32        return next.mode == AuthorityMode::Sealed
33            && previous.boot == next.boot
34            && previous.time == next.time
35            && previous.pending_time == next.pending_time
36            && previous.time_floors == next.time_floors;
37    }
38    if next.mode == AuthorityMode::Sealed {
39        return previous.boot == next.boot
40            && next.time == TimeState::Unavailable
41            && next.pending_time.is_none()
42            && previous.time_floors == next.time_floors;
43    }
44    let boot_advanced = match next.boot {
45        BootState::Open { epoch } => {
46            epoch == next.boot_floor
47                && previous.boot_floor.checked_add(1) == Some(epoch)
48                && next.mode == AuthorityMode::Serving
49                && previous.boot != next.boot
50        }
51        BootState::Closed => false,
52    };
53    if !boot_advanced && (previous.mode != next.mode || previous.boot != next.boot) {
54        return false;
55    }
56    if boot_advanced {
57        next.time == TimeState::Unavailable
58            && next.pending_time.is_none()
59            && previous.time_floors == next.time_floors
60    } else {
61        time_advances(previous, next)
62    }
63}

65fn time_advances(previous: &ProtectedAuthorityRecord, next: &ProtectedAuthorityRecord) -> bool {
66    if previous.time == next.time && previous.pending_time == next.pending_time {
67        return previous.time_floors == next.time_floors;
68    }
69    match (
70        previous.time,
71        previous.pending_time,
72        next.time,
73        next.pending_time,
74    ) {
75        (TimeState::Unavailable, None, TimeState::Unavailable, Some(_)) => {
76            previous.time_floors == next.time_floors
77        }
78        (
79            TimeState::Unavailable,
80            Some(challenge),
81            TimeState::Valid {
82                source_epoch,
83                sequence,
84                time_request_id,
85                purpose,
86                ..
87            },
88            None,
89        ) => {
90            time_request_id == challenge.time_request_id
91                && purpose == challenge.purpose
92                && source_epoch == next.time_floors.source_epoch
93                && sequence == next.time_floors.source_sequence
94                && time_floors_strictly_advance(previous.time_floors, next.time_floors)
95        }
96        (TimeState::Valid { .. }, None, TimeState::Unavailable, None) => {
97            previous.time_floors == next.time_floors
98        }
99        _ => false,
100    }
101}

103fn relay_advances(previous: &ProtectedAuthorityRecord, next: &ProtectedAuthorityRecord) -> bool {
104    let old = previous.relay;
105    let new = next.relay;
106    if old == new && previous.previous_active == next.previous_active {
107        return previous.generation_floor == next.generation_floor;
108    }
109    match (old, new) {
110        (
111            RelayProfileState::Empty,
112            RelayProfileState::Pending {
113                generation,
114                pending_slot,
115                ..
116            },
117        ) => enrolled(previous, next, None, generation, pending_slot),
118        (
119            RelayProfileState::Active(active),
120            RelayProfileState::Pending {
121                generation,
122                pending_slot,
123                ..
124            },
125        ) => enrolled(previous, next, Some(active), generation, pending_slot),
126        (
127            RelayProfileState::Pending {
128                generation,
129                csr_handle,
130                pending_slot,
131            },
132            RelayProfileState::Uploading(upload),
133        ) => {
134            stable(previous, next)
135                && generation == upload.generation
136                && csr_handle == upload.csr_handle
137                && pending_slot == upload.pending_slot
138                && upload.next_index == 0
139                && previous.previous_active == next.previous_active
140        }
141        (RelayProfileState::Uploading(old), RelayProfileState::Uploading(new)) => {
142            stable(previous, next)
143                && old.next_index.checked_add(1) == Some(new.next_index)
144                && (crate::ProfileUploadIntent {
145                    next_index: new.next_index,
146                    ..old
147                }) == new
148                && previous.previous_active == next.previous_active
149        }
150        (RelayProfileState::Uploading(upload), RelayProfileState::Staged(intent)) => {
151            stable(previous, next)
152                && upload.complete()
153                && upload.matches_relay_intent(&intent)
154                && previous.previous_active == next.previous_active
155        }
156        (RelayProfileState::Staged(old), RelayProfileState::ReceiptConsumed(new))
157        | (RelayProfileState::ReceiptConsumed(old), RelayProfileState::Prepared(new)) => {
158            stable(previous, next) && old == new && previous.previous_active == next.previous_active
159        }
160        (RelayProfileState::Prepared(old), RelayProfileState::Promoted { intent: new, .. }) => {
161            stable(previous, next) && old == new && previous.previous_active == next.previous_active
162        }
163        (RelayProfileState::Promoted { intent: old, .. }, RelayProfileState::Active(new)) => {
164            stable(previous, next) && old == new && next.previous_active.is_none()
165        }
166        (
167            RelayProfileState::Pending { .. }
168            | RelayProfileState::Uploading(_)
169            | RelayProfileState::Staged(_)
170            | RelayProfileState::ReceiptConsumed(_),
171            RelayProfileState::Empty,
172        ) => {
173            stable(previous, next)
174                && previous.previous_active.is_none()
175                && next.previous_active.is_none()
176        }
177        (
178            RelayProfileState::Pending { .. }
179            | RelayProfileState::Uploading(_)
180            | RelayProfileState::Staged(_)
181            | RelayProfileState::ReceiptConsumed(_),
182            RelayProfileState::Active(new),
183        ) => {
184            stable(previous, next)
185                && previous.previous_active == Some(new)
186                && next.previous_active.is_none()
187        }
188        _ => false,
189    }
