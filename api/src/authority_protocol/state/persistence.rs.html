52pub trait ProtectedStore {

67pub fn verify_protected_record<V: ProtectedRecordVerifier>(
68    record: ProtectedAuthorityRecord,
69    verifier: &V,
70) -> Result<VerifiedProtectedRecord, AuthorityFault> {
71    if !record.invariants_hold() || !verifier.verify(&record) {
72        return Err(AuthorityFault::PersistenceFailure);
73    }
74    Ok(VerifiedProtectedRecord(record))
75}

111impl<S: ProtectedStore> AuthorityState<S> {
112    pub fn restore(
113        store: S,
114        verified: &VerifiedProtectedRecord,
115        boot_challenge: [u8; DIGEST_LEN],
116    ) -> Self {
117        let record = *verified.record();
118        Self {
119            mode: if record.mode == AuthorityMode::Sealed {
120                AuthorityMode::Sealed
121            } else {
122                AuthorityMode::Ready
123            },
124            boot: BootState::Closed,
125            time: TimeState::Unavailable,
126            relay: record.relay,
127            device_id: record.device_id,
128            authority_id: record.authority_id,
129            authority_epoch: record.authority_epoch,
130            boot_challenge,
131            boot_floor: record.boot_floor,
132            generation_floor: record.generation_floor,
133            state_epoch: record.state_epoch,
134            approved_loader_digest: record.approved_loader_digest,
135            last_request_sequence: record.last_request_sequence,
136            previous_active: record.previous_active,
137            pending_time: None,
138            time_floors: record.time_floors,
139            protected_revision: record.revision,
140            store,
141        }
142    }
143    pub fn into_store(self) -> S {
144        self.store
145    }
146
147    pub(super) fn persist(&mut self) -> Result<(), AuthorityFault> {
148        let expected = self.protected_revision;
149        let next_revision = match expected.checked_add(1) {
150            Some(value) => value,
151            None => {
152                self.mode = AuthorityMode::Sealed;
153                self.store.seal_on_conflict(expected);
154                return Err(AuthorityFault::PersistenceFailure);
155            }
156        };
157        let next = self.snapshot(next_revision);
158        if self.store.compare_and_swap(expected, &next) {
159            self.protected_revision = next_revision;
160            Ok(())
161        } else {
162            self.mode = AuthorityMode::Sealed;
163            self.store.seal_on_conflict(expected);
164            Err(AuthorityFault::PersistenceFailure)
165        }
166    }
167
168    fn snapshot(&self, revision: u64) -> ProtectedAuthorityRecord {
169        ProtectedAuthorityRecord {
170            revision,
171            mode: self.mode,
172            boot: self.boot,
173            time: self.time,
174            relay: self.relay,
175            device_id: self.device_id,
176            authority_id: self.authority_id,
177            authority_epoch: self.authority_epoch,
178            boot_floor: self.boot_floor,
179            generation_floor: self.generation_floor,
180            state_epoch: self.state_epoch,
181            approved_loader_digest: self.approved_loader_digest,
182            last_request_sequence: self.last_request_sequence,
183            previous_active: self.previous_active,
184            pending_time: self.pending_time,
185            time_floors: self.time_floors,
186        }
187    }
