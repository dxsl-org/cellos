29pub trait ProtectedStore {
30    fn compare_and_swap(&mut self, expected_revision: u64, next: &ProtectedAuthorityRecord)

33    /// Irreversibly seal the external counter domain before returning.

35}

44pub fn verify_protected_record<V: ProtectedRecordVerifier>(
45    record: ProtectedAuthorityRecord,
46    verifier: &V,
47) -> Result<VerifiedProtectedRecord, AuthorityFault> {
48    if !record.invariants_hold() || !verifier.verify(&record) {
49        return Err(AuthorityFault::PersistenceFailure);
50    }
51    Ok(VerifiedProtectedRecord(record))
52}

66impl<S: ProtectedStore> AuthorityState<S> {
67    pub fn restore(
68        store: S,
69        verified: &VerifiedProtectedRecord,
70        boot_challenge: [u8; DIGEST_LEN],
71    ) -> Self {
72        let record = *verified.record();
73        Self {
74            mode: if record.mode == AuthorityMode::Sealed {
75                AuthorityMode::Sealed
76            } else {
77                AuthorityMode::Ready
78            },
79            boot: BootState::Closed,
80            time: TimeState::Unavailable,
81            relay: record.relay,
82            device_id: record.device_id,
83            authority_id: record.authority_id,
84            authority_epoch: record.authority_epoch,
85            boot_challenge,
86            boot_floor: record.boot_floor,
87            generation_floor: record.generation_floor,
88            state_epoch: record.state_epoch,
89            approved_loader_digest: record.approved_loader_digest,
90            last_request_sequence: record.last_request_sequence,
91            previous_active: record.previous_active,
92            pending_time: None,
93            time_floors: record.time_floors,
94            protected_revision: record.revision,
95            store,
96        }
97    }
98    pub fn into_store(self) -> S {
99        self.store
100    }
101
102    pub(super) fn persist(&mut self) -> Result<(), AuthorityFault> {
103        let expected = self.protected_revision;
104        let next_revision = match expected.checked_add(1) {
105            Some(value) => value,
106            None => {
107                self.mode = AuthorityMode::Sealed;
108                self.store.seal_on_conflict(expected);
109                return Err(AuthorityFault::PersistenceFailure);
110            }
111        };
112        let next = self.snapshot(next_revision);
113        if self.store.compare_and_swap(expected, &next) {
114            self.protected_revision = next_revision;
115            Ok(())
116        } else {
117            self.mode = AuthorityMode::Sealed;
118            self.store.seal_on_conflict(expected);
119            Err(AuthorityFault::PersistenceFailure)
120        }
121    }
122
123    fn snapshot(&self, revision: u64) -> ProtectedAuthorityRecord {
124        ProtectedAuthorityRecord {
125            revision,
126            mode: self.mode,
127            boot: self.boot,
128            time: self.time,
129            relay: self.relay,
130            device_id: self.device_id,
131            authority_id: self.authority_id,
132            authority_epoch: self.authority_epoch,
133            boot_floor: self.boot_floor,
134            generation_floor: self.generation_floor,
135            state_epoch: self.state_epoch,
136            approved_loader_digest: self.approved_loader_digest,
137            last_request_sequence: self.last_request_sequence,
138            previous_active: self.previous_active,
139            pending_time: self.pending_time,
140            time_floors: self.time_floors,
141        }
142    }
