5impl TypedRequest {
6    pub fn decode_payload(operation: Operation, input: &[u8]) -> Result<Self, WireError> {
7        let mut reader = Reader::new(input);
8        let context = read_context(&mut reader, operation)?;
9        let request = match operation {
10            Operation::OpenBoot => Self::OpenBoot(OpenBootRequest {
11                context,
12                loader_digest: reader.array()?,
13            }),
14            Operation::ReadCommittedRelayState => {
15                Self::ReadCommittedRelayState(ReadCommittedRelayStateRequest { context })
16            }
17            Operation::RequestSignedTime => Self::RequestSignedTime(RequestSignedTimeRequest {
18                context,
19                purpose: read_purpose(&mut reader)?,
20            }),
21            Operation::AcceptSignedTime => {
22                Self::AcceptSignedTime(read_accept_time(context, &mut reader)?)
23            }
24            Operation::BeginRelayEnrollment => {
25                Self::BeginRelayEnrollment(BeginRelayEnrollmentRequest {
26                    context,
27                    hostname: reader.bounded()?,
28                })
29            }
30            Operation::ReadRelayCsrChunk => Self::ReadRelayCsrChunk(ReadRelayCsrChunkRequest {
31                context,
32                csr_handle: reader.u64()?,
33                chunk_index: reader.u32()?,
34            }),
35            Operation::ValidateAndStageRelayProfile => {
36                Self::ValidateAndStageRelayProfile(read_profile(context, &mut reader)?)
37            }
38            Operation::ConsumeStagedRelayProfile => {
39                let (generation, policy_epoch, profile_digest) = read_relay_tuple(&mut reader)?;
40                Self::ConsumeStagedRelayProfile(ConsumeStagedRelayProfileRequest {
41                    context,
42                    generation,
43                    policy_epoch,
44                    profile_digest,
45                })
46            }
47            Operation::CommitRelayGeneration => {
48                let (generation, policy_epoch, profile_digest) = read_relay_tuple(&mut reader)?;
49                Self::CommitRelayGeneration(CommitRelayGenerationRequest {
50                    context,
51                    generation,
52                    policy_epoch,
53                    profile_digest,
54                })
55            }
56            Operation::AbortRelayEnrollment => {
57                Self::AbortRelayEnrollment(AbortRelayEnrollmentRequest {
58                    context,
59                    generation: reader.u64()?,
60                })
61            }
62            Operation::GetRelayActivePublicKey => {
63                Self::GetRelayActivePublicKey(GetRelayActivePublicKeyRequest { context })
64            }
65            Operation::SignTls13ClientCertificateVerify => {
66                Self::SignTls13ClientCertificateVerify(SignTls13ClientCertificateVerifyRequest {
67                    context,
68                    transcript_hash: reader.array()?,
69                    relay_generation: reader.u64()?,
70                    active_profile_digest: reader.array()?,
71                    public_request_id: reader.u64()?,
72                })
73            }
74        };
75        reader.finish()?;
76        Ok(request)
77    }
78}

80fn read_accept_time(
81    context: RequestContext,
82    reader: &mut Reader<'_>,
83) -> Result<AcceptSignedTimeRequest, WireError> {
84    let value = AcceptSignedTimeRequest {
85        context,
86        time_request_id: reader.array()?,
87        purpose: read_purpose(reader)?,
88        source_epoch: reader.u64()?,
89        source_sequence: reader.u64()?,
90        unix_seconds: reader.i64()?,
91        expires_at: reader.u64()?,
92        nonce: reader.array()?,
93        source_signature: reader.bounded()?,
94    };
95    if !is_strict_p256_der_signature(value.source_signature.as_slice()) {
96        return Err(WireError::InvalidLength);
97    }
98    Ok(value)
99}

101fn read_profile(
102    context: RequestContext,
103    reader: &mut Reader<'_>,
104) -> Result<ValidateAndStageRelayProfileRequest, WireError> {
105    let value = ValidateAndStageRelayProfileRequest {
106        context,
107        generation: reader.u64()?,
108        policy_epoch: reader.u64()?,
109        pending_slot: reader.u8()?,
110        pending_spki_digest: reader.array()?,
111        profile_digest: reader.array()?,
112        tpm_public_digest: reader.array()?,
113        profile: reader.bounded()?,
114    };
115    if value.pending_slot > 1 || value.profile.is_empty() {
116        return Err(WireError::InvalidLength);
117    }
118    Ok(value)
119}
120
121fn read_purpose(reader: &mut Reader<'_>) -> Result<u8, WireError> {
122    let value = reader.u8()?;
123    TimePurpose::try_from(value).map_err(|_| WireError::InvalidLength)?;
124    Ok(value)
125}

127fn read_relay_tuple(reader: &mut Reader<'_>) -> Result<(u64, u64, [u8; DIGEST_LEN]), WireError> {
128    Ok((reader.u64()?, reader.u64()?, reader.array()?))
