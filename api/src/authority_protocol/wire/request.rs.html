7// Fixed inline storage is the no-allocation wire contract; boxing is forbidden.

14    AcceptSignedTime(AcceptSignedTimeRequest),

19    CommitRelayGeneration(CommitRelayGenerationRequest),

24pub(crate) trait CanonicalBody {
25    fn canonical_body(&self) -> ([u8; DIGEST_LEN], usize);
26}
27
28macro_rules! canonical_body {
29    ($( $type:ty => $variant:ident ),+ $(,)?) => {$(
30        impl CanonicalBody for $type {
31            fn canonical_body(&self) -> ([u8; DIGEST_LEN], usize) {
32                let request = TypedRequest::$variant(*self);
33                request.canonical_body()
34            }
35        }
36    )+};
37}

54impl TypedRequest {
55    pub const fn operation(&self) -> Operation {
56        match self {
57            Self::OpenBoot(_) => Operation::OpenBoot,
58            Self::ReadCommittedRelayState(_) => Operation::ReadCommittedRelayState,
59            Self::RequestSignedTime(_) => Operation::RequestSignedTime,
60            Self::AcceptSignedTime(_) => Operation::AcceptSignedTime,
61            Self::BeginRelayEnrollment(_) => Operation::BeginRelayEnrollment,
62            Self::ReadRelayCsrChunk(_) => Operation::ReadRelayCsrChunk,
63            Self::ValidateAndStageRelayProfile(_) => Operation::ValidateAndStageRelayProfile,
64            Self::ConsumeStagedRelayProfile(_) => Operation::ConsumeStagedRelayProfile,
65            Self::CommitRelayGeneration(_) => Operation::CommitRelayGeneration,
66            Self::AbortRelayEnrollment(_) => Operation::AbortRelayEnrollment,
67            Self::GetRelayActivePublicKey(_) => Operation::GetRelayActivePublicKey,
68            Self::SignTls13ClientCertificateVerify(_) => {
69                Operation::SignTls13ClientCertificateVerify
70            }
71        }
72    }
73    pub const fn context(&self) -> &RequestContext {
74        match self {
75            Self::OpenBoot(value) => &value.context,
76            Self::ReadCommittedRelayState(value) => &value.context,
77            Self::RequestSignedTime(value) => &value.context,
78            Self::AcceptSignedTime(value) => &value.context,
79            Self::BeginRelayEnrollment(value) => &value.context,
80            Self::ReadRelayCsrChunk(value) => &value.context,
81            Self::ValidateAndStageRelayProfile(value) => &value.context,
82            Self::ConsumeStagedRelayProfile(value) => &value.context,
83            Self::CommitRelayGeneration(value) => &value.context,
84            Self::AbortRelayEnrollment(value) => &value.context,
85            Self::GetRelayActivePublicKey(value) => &value.context,
86            Self::SignTls13ClientCertificateVerify(value) => &value.context,
87        }
88    }
89
90    pub fn context_mut(&mut self) -> &mut RequestContext {
91        match self {
92            Self::OpenBoot(value) => &mut value.context,
93            Self::ReadCommittedRelayState(value) => &mut value.context,
94            Self::RequestSignedTime(value) => &mut value.context,
95            Self::AcceptSignedTime(value) => &mut value.context,
96            Self::BeginRelayEnrollment(value) => &mut value.context,
97            Self::ReadRelayCsrChunk(value) => &mut value.context,
98            Self::ValidateAndStageRelayProfile(value) => &mut value.context,
99            Self::ConsumeStagedRelayProfile(value) => &mut value.context,
100            Self::CommitRelayGeneration(value) => &mut value.context,
101            Self::AbortRelayEnrollment(value) => &mut value.context,
102            Self::GetRelayActivePublicKey(value) => &mut value.context,
103            Self::SignTls13ClientCertificateVerify(value) => &mut value.context,
104        }
105    }
106
107    pub fn canonical_body_digest(&self) -> [u8; DIGEST_LEN] {
108        self.canonical_body().0
109    }
110
111    pub fn canonical_payload_len(&self) -> usize {
112        self.canonical_body().1
113    }
114
115    fn canonical_body(&self) -> ([u8; DIGEST_LEN], usize) {
116        let mut payload = [0u8; FRAME_MAX_PAYLOAD];
117        let length = self
118            .encode_payload(&mut payload)
119            .expect("typed request fits protocol maximum");
120        (
121            crate::sha256(&payload[REQUEST_CONTEXT_WIRE_LEN..length]),
122            length,
123        )
124    }
