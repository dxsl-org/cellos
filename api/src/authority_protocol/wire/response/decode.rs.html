5impl TypedResponse {
6    pub fn decode_payload(operation: Operation, input: &[u8]) -> Result<Self, WireError> {
7        let mut reader = Reader::new(input);
8        let binding = read_binding(&mut reader, operation)?;
9        let response = match operation {
10            Operation::OpenBoot => Self::OpenBoot(OpenBootResponse {
11                binding,
12                boot_epoch: reader.u64()?,
13                state_epoch: reader.u64()?,
14                approved_loader_digest: reader.array()?,
15            }),
16            Operation::ReadCommittedRelayState => {
17                let (generation, policy_epoch, profile_digest) = read_relay_tuple(&mut reader)?;
18                Self::ReadCommittedRelayState(ReadCommittedRelayStateResponse {
19                    binding,
20                    generation,
21                    policy_epoch,
22                    profile_digest,
23                })
24            }
25            Operation::RequestSignedTime => Self::RequestSignedTime(RequestSignedTimeResponse {
26                binding,
27                time_request_id: reader.array()?,
28                purpose: read_purpose(&mut reader)?,
29                nonce: reader.array()?,
30            }),
31            Operation::AcceptSignedTime => Self::AcceptSignedTime(AcceptSignedTimeResponse {
32                binding,
33                time_request_id: reader.array()?,
34                purpose: read_purpose(&mut reader)?,
35                source_epoch: reader.u64()?,
36                source_sequence: reader.u64()?,
37                expires_at: reader.u64()?,
38            }),
39            Operation::BeginRelayEnrollment => {
40                Self::BeginRelayEnrollment(BeginRelayEnrollmentResponse {
41                    binding,
42                    generation: reader.u64()?,
43                    policy_epoch: reader.u64()?,
44                    csr_handle: reader.u64()?,
45                    csr_len: reader.u32()?,
46                    csr_digest: reader.array()?,
47                })
48            }
49            Operation::ReadRelayCsrChunk => Self::ReadRelayCsrChunk(ReadRelayCsrChunkResponse {
50                binding,
51                chunk_index: reader.u32()?,
52                chunk: reader.bounded()?,
53            }),
54            Operation::ValidateAndStageRelayProfile => {
55                Self::ValidateAndStageRelayProfile(ValidateAndStageRelayProfileResponse {
56                    binding,
57                    receipt: StagedProfileReceipt {
58                        device_id: reader.array()?,
59                        authority_id: reader.array()?,
60                        authority_epoch: reader.u64()?,
61                        generation: reader.u64()?,
62                        policy_epoch: reader.u64()?,
63                        pending_slot: reader.u8()?,
64                        pending_spki_digest: reader.array()?,
65                        profile_digest: reader.array()?,
66                        boot_epoch: reader.u64()?,
67                        validation_request_id: reader.u64()?,
68                    },
69                })
70            }
71            Operation::ConsumeStagedRelayProfile => {
72                Self::ConsumeStagedRelayProfile(ConsumeStagedRelayProfileResponse {
73                    binding,
74                    generation: reader.u64()?,
75                })
76            }
77            Operation::CommitRelayGeneration => {
78                let (generation, policy_epoch, profile_digest) = read_relay_tuple(&mut reader)?;
79                Self::CommitRelayGeneration(CommitRelayGenerationResponse {
80                    binding,
81                    generation,
82                    policy_epoch,
83                    profile_digest,
84                })
85            }
86            Operation::AbortRelayEnrollment => {
87                Self::AbortRelayEnrollment(AbortRelayEnrollmentResponse {
88                    binding,
89                    generation: reader.u64()?,
90                })
91            }
92            Operation::GetRelayActivePublicKey => {
93                let value = GetRelayActivePublicKeyResponse {
94                    binding,
95                    generation: reader.u64()?,
96                    public_key: reader.array()?,
97                    public_key_digest: reader.array()?,
98                };
99                if value.public_key[0] != 4 {
100                    return Err(WireError::InvalidLength);
101                }
102                Self::GetRelayActivePublicKey(value)
103            }
104            Operation::SignTls13ClientCertificateVerify => {
105                Self::SignTls13ClientCertificateVerify(SignTls13ClientCertificateVerifyResponse {
106                    binding,
107                    signature: reader.array()?,
108                })
109            }
110        };
111        reader.finish()?;
112        Ok(response)
113    }
114}

116fn read_purpose(reader: &mut Reader<'_>) -> Result<u8, WireError> {
117    let value = reader.u8()?;
118    TimePurpose::try_from(value).map_err(|_| WireError::InvalidLength)?;
119    Ok(value)
120}
121
122fn read_relay_tuple(reader: &mut Reader<'_>) -> Result<(u64, u64, [u8; DIGEST_LEN]), WireError> {
123    Ok((reader.u64()?, reader.u64()?, reader.array()?))
