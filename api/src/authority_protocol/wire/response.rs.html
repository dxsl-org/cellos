31pub fn verify_typed_response<A: ResponseAuthenticator>(
32    response: TypedResponse,
33    header: &FrameHeader,
34    expected: &ExpectedResponseBinding,
35    authenticator: &A,
36) -> Result<ValidatedResponse, AuthorityFault> {
37    let binding = response.binding();
38    let mut payload = [0u8; FRAME_MAX_PAYLOAD];
39    let length = response
40        .encode_payload(&mut payload)
41        .map_err(|_| AuthorityFault::Malformed)?;
42    let body_digest = crate::sha256(&payload[RESPONSE_BINDING_WIRE_LEN..length]);
43    if header.class != FrameClass::Response
44        || header.operation != response.operation()
45        || header.payload_len as usize != length
46        || response.operation() != expected.operation
47        || header.request_id != expected.request_id
48        || binding.device_id != expected.device_id
49        || binding.authority_id != expected.authority_id
50        || binding.boot_epoch != expected.boot_epoch
51        || binding.request_id != expected.request_id
52        || binding.operation != response.operation()
53        || !constant_time_eq(&binding.payload_digest, &body_digest)
54        || !constant_time_eq(&binding.authority_signature[..16], &header.authenticator)
55        || !authenticator.verify(
56            &binding.authentication_input(),
57            &binding.authority_signature,
58        )
59    {
60        return Err(AuthorityFault::ChallengeMismatch);
61    }
62    Ok(ValidatedResponse(response))
63}
