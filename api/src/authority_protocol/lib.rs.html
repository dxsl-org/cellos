63/// Fixed-capacity, non-allocating byte sequence.
64#[derive(Debug, Clone, Copy, PartialEq, Eq)]
65pub struct Bounded<const N: usize> {
66    len: u16,
67    buf: [u8; N],
68}
69
70impl<const N: usize> Bounded<N> {
71    /// Copy `src` in full; return `None` instead of truncating when oversized.
72    pub fn from_slice(src: &[u8]) -> Option<Self> {
73        if src.len() > N || src.len() > u16::MAX as usize {
74            return None;
75        }
76        let mut buf = [0u8; N];
77        buf[..src.len()].copy_from_slice(src);
78        Some(Self {
79            len: src.len() as u16,
80            buf,
81        })
82    }
83
84    /// Return the occupied prefix only.
85    pub fn as_slice(&self) -> &[u8] {
86        &self.buf[..self.len as usize]
87    }
88
89    /// Return the occupied byte count.
90    pub const fn len(&self) -> usize {
91        self.len as usize
92    }
93
94    /// Report whether no bytes are occupied.
95    pub const fn is_empty(&self) -> bool {
96        self.len == 0
97    }
98}
99
100/// Constant-time equality for fixed authentication values.
101pub fn constant_time_eq(a: &[u8], b: &[u8]) -> bool {
102    if a.len() != b.len() {
103        return false;
104    }
105    let mut diff = 0u8;
106    for index in 0..a.len() {
107        diff |= a[index] ^ b[index];
108    }
109    diff == 0
110}

112pub(crate) fn sha256(bytes: &[u8]) -> [u8; 32] {
113    Sha256::digest(bytes).into()
114}
