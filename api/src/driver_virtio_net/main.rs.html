1//! VirtIO NIC Driver Cell — Tier-1 Privileged Driver Cell.
2//!
3//! Finds a virtio Network device, initialises the virtqueues via `virtio-drivers`,
4//! registers as the system NIC via `sys_register_nic_driver`, then serves Tx/Rx/GetMac
5//! IPC from the net service.  This cell replaces the kernel's `virtio_net.rs` driver
6//! per the Kernel Boundary Law (2026-06-23).
7//!
8//! # Transports
9//! - riscv64 / aarch64 (`mmio`): probes the QEMU virt VirtIO MMIO slots and claims
10//!   the slot via `sys_request_mmio`.
11//! - x86_64 (`pci_x86`): claims a `1af4:1041` (modern) or `1af4:1000` (transitional)
12//!   PCI function, drives it through `virtio-pci-transport` with DMA authorised for
13//!   its BDF, and polls (no interrupts until phase 04b).
14//!
15//! # Fallback
16//! If no VirtIO NIC is found, or the PCI function is refused (logged by name), the
17//! cell exits cleanly (code 0).  The e1000/igb Driver Cells remain.
18//!
19//! # Law 4 exception
20//! `src/mmio.rs` and `src/pci_x86/hal.rs` use `unsafe` for the `virtio_drivers::Hal`
21//! impls, DMA bounce copies and MMIO probe reads; `src/pci_x86/raw_net.rs` for the
22//! raw queue calls that borrow buffers across the device's ownership window. All
23//! other modules are safe code.
24

37use device::NetDevice;
38use dispatch::{handle, NicReply, REPLY_BUF};
39use ostd::app::{AppContext, AppEvent};
40use ostd::service::{sender_is_live_provider, service};
41use ostd::sync::Mutex;
42use ostd::syscall::{sys_register_nic_driver, sys_try_send};
43
44static STATE: Mutex<Option<NetDevice>> = Mutex::new(None);
45
46fn handler(_ctx: &mut AppContext, event: AppEvent) {
47    match event {
48        AppEvent::Init => {
49            // Find and initialise the platform's virtio Network device.
50            let Some(dev) = device::find_and_init_net() else {
51                // No usable virtio NIC on this platform — exit gracefully.
52                // The e1000/igb Driver Cells handle the other PCIe NICs.
53                ostd::syscall::sys_exit(0);
54            };
55
56            // Register only after state is ready because successful publication
57            // allows the net service to route Tx/Rx immediately.
58            *STATE.lock() = Some(dev);
59            if sys_register_nic_driver().is_err() {
60                *STATE.lock() = None;
61                ostd::io::println("[virtio-net] NIC driver registration failed");
62                ostd::syscall::sys_exit(1);
63            }
64            ostd::io::println("[virtio-net] ready");
65        }
66
67        // The net service speaks the raw NIC wire protocol (no 0xAC App-SDK
68        // envelope), so requests arrive as RawMessage. Accept Message too for
69        // envelope-wrapped senders — the dispatch payload layout is identical.
70        AppEvent::Message { sender_tid, data } | AppEvent::RawMessage { sender_tid, data } => {
71            let operation = ostd::ipc::current();
72            let reply = |bytes: &[u8]| {
73                if let Some(operation) = operation {
74                    let _ = ostd::ipc::reply(operation, bytes);
75                } else {
76                    let _ = sys_try_send(sender_tid, bytes);
77                }
78            };
79            // Only the live net service may drive the NIC; any other cell could
80            // inject or read raw frames. `sender_tid` is kernel-stamped and the
81            // registry is asked per request, so a restarted net service — or one
82            // init has not registered yet — is followed without a cached tid.
83            // The refusal is the ordinary failure byte the net service retries on.
84            if !sender_is_live_provider(service::NET, sender_tid) {
85                reply(&[1u8]);
86                return;
87            }
88            let mut out_buf = [0u8; REPLY_BUF];
89            if let Some(dev) = STATE.lock().as_mut() {
90                match handle(dev, data.as_ref(), &mut out_buf) {
91                    NicReply::Status(code) => {
92                        reply(&[code]);
93                    }
94                    NicReply::Frame { len, buf } => {
95                        reply(&buf[..2 + len]);
96                    }
97                    NicReply::Mac(mac) => {
98                        reply(&mac);
99                    }
100                }
101            } else {
102                reply(&[1u8]);
103            }
104        }
105
106        AppEvent::Shutdown | AppEvent::ShutdownWith { .. } => {
107            ostd::syscall::sys_exit(0);
108        }
109
110        _ => {}
111    }
112}
113
114ostd::run_app!(handler);
