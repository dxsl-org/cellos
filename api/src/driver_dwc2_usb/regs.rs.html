142// HCSPLT bits (Host Channel Split Control). A channel that talks to a full- or
143// low-speed device behind a high-speed hub addresses it through these fields;
144// the bit layout is from the core's own header (dwc2_core.h).
145pub const HCSPLT_PRTADDR_MASK: u32 = 0x7F; // hub port, bits 6:0
146pub const HCSPLT_HUBADDR_SHIFT: u32 = 7; // hub device address, bits 13:7
147pub const HCSPLT_HUBADDR_MASK: u32 = 0x7F << HCSPLT_HUBADDR_SHIFT;
148pub const HCSPLT_COMPSPLT: u32 = 1 << 16; // 1 = complete split, 0 = start split
149pub const HCSPLT_SPLTENA: u32 = 1 << 31; // split enable
150/// Which part of the payload a split transaction carries, bits 15:14.
151///
152/// Zero is `MID` -- the middle of a payload -- and this driver never wrote the
153/// field, so every split it issued asked the hub for the middle of a transfer
154/// that is one packet long, which is nothing the hub can return.
155pub const HCSPLT_XACTPOS_SHIFT: u32 = 14;
156pub const HCSPLT_XACTPOS_ALL: u32 = 3 << HCSPLT_XACTPOS_SHIFT;
157
158/// `HCCHAR.ODDFRM` — the frame the transfer goes on the wire is an odd one.
159///
160/// The core latches this at channel start and refuses a transfer whose parity
161/// disagrees with the frame it lands in. Behind a hub that breaks the pairing
162/// between a start-split and the complete-split that collects it, and the
163/// channel reports a transaction error for a sequence the bus never saw.
164pub const HCCHAR_ODDFRM: u32 = 1 << 29;
165
166/// `HCCHAR.LSPDDEV` — the target is a low-speed device.
167///
168/// Not a cosmetic hint: a low-speed device only understands transactions
169/// preceded by a full-speed preamble. Behind a hub the same bit tells the hub's
170/// transaction translator to use that preamble, and without it the hub runs a
171/// full-speed transaction to a device that cannot hear it.
172pub const HCCHAR_LSPDDEV: u32 = 1 << 17;
173
174/// `HFNUM.FRNUM` — the USB frame counter, in 125 us frames.
175pub const HFNUM_FRNUM_MASK: u32 = 0xFFFF;

177/// Mask of W1C bits in HPRT0. When writing to HPRT0 (e.g. to set PRTRST or PRTPWR),
178/// these bits must be written as ZERO to prevent inadvertently clearing port events!
179pub const HPRT0_W1C_MASK: u32 =
180    HPRT0_PRTCONNDET | HPRT0_PRTENA | HPRT0_PRTENCHNG | HPRT0_PRTOVRCURRCHNG;
