10/// How host channels move payload bytes.
11///
12/// The BCM2837 DWC2 supports both; which one a given environment implements is
13/// not discoverable from the register file, so the driver tries FIFO first and
14/// falls back to DMA. FIFO is the board-proven path (the LAN9514 Ethernet runs
15/// there); DMA is what QEMU's `hcd-dwc2` model implements, since it sources
16/// payloads from guest memory at `HCDMA` and ignores host-channel FIFO writes
17/// entirely.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum TransferMode {
20    /// Programmed I/O: payload goes through the channel's data FIFO.
21    Fifo,
22    /// The core reads/writes payloads directly from/to guest memory at `HCDMA`.
23    Dma,
24}
25
26/// Bytes reserved per host channel inside the DMA scratch region.
27///
28/// Sized for the largest payload this driver moves: a full configuration
29/// descriptor (≤512 bytes) and an Ethernet frame. One slot per channel keeps two
30/// in-flight channels from aliasing the same memory.
31pub const DMA_SLOT_BYTES: usize = 4096;
32/// Host channels that get a DMA slot (DWC2 implements up to 16; this driver uses 8).
33const DMA_CHANNELS: usize = 8;
34
35/// Control-endpoint max packet size assumed before a device reports its own.
36///
37/// USB 2.0 §5.5.3 fixes low-speed control endpoints at 8 bytes and allows
38/// full-speed ones 8/16/32/64, so 8 is the only value safe to assume for the
39/// first descriptor read. Declaring 64 there makes the host expect 64-byte
40/// packets from a device that sends 8, and the read fails.
41pub const CONTROL_MPS_LOW_FULL_SPEED: u8 = 8;
42
43/// HPRT0 `PRTSPD` encoding (DWC2 databook): the value `reset_port` returns.
44pub const PORT_SPEED_HIGH: u32 = 0;
45pub const PORT_SPEED_FULL: u32 = 1;
46pub const PORT_SPEED_LOW: u32 = 2;
47
48/// Control-endpoint packet size to assume for a device on a port at `speed`.
49///
50/// USB 2.0 5.5.3: low speed is 8 bytes, full speed is 8/16/32/64, and
51/// **high speed is 64**. The assumption has to follow the negotiated link
52/// speed, and `read_device_descriptor` replaces it with the device's real
53/// `bMaxPacketSize0` as soon as it has read one.
54pub fn initial_control_mps(speed: u32) -> u8 {
55    match speed {
56        PORT_SPEED_HIGH => 64,
57        _ => CONTROL_MPS_LOW_FULL_SPEED,
58    }
59}

61pub struct UsbHostEngine<'a> {
62    mmio: &'a MmioRegion,
63    /// Max packet size for control transfers on endpoint 0. Updated from the
64    /// device descriptor's `bMaxPacketSize0` once it has been read.
65    ctrl_mps: Cell<u8>,
66    mode: Cell<TransferMode>,
67    /// Per-channel DMA scratch, allocated on first use in DMA mode.
68    dma: RefCell<Option<DmaBuf>>,
69    /// `HCINT` captured at the last channel failure.
70    ///
71    /// `wait_channel` collapses every hardware outcome into `ViError::IO`, and
72    /// a device refusing a request, the bus failing to carry it, and the core
73    /// never finishing look identical from the caller's side. Keeping the raw
74    /// register lets `report_failure` name which one actually happened.
75    last_hcint: Cell<u32>,
76}

94    /// Adopt a device's control-endpoint max packet size.
95    ///
96    /// Clamped to the legal 8/16/32/64 set: a device reporting 0 or a bogus
97    /// value must not produce a zero-length packet size that hangs the engine.
98    pub fn set_control_mps(&self, mps: u8) {
99        let clamped = match mps {
100            0..=8 => 8,
101            9..=16 => 16,
102            17..=32 => 32,
103            _ => 64,
104        };
105        self.ctrl_mps.set(clamped);

113    /// Select the payload-transfer mode.
114    ///
115    /// Switching to DMA allocates the scratch region on first use; a failed
116    /// allocation leaves the engine in FIFO mode rather than half-configured.
117    pub fn set_mode(&self, mode: TransferMode) -> bool {
118        if mode == TransferMode::Dma && self.dma.borrow().is_none() {
119            let Some(buf) = DmaBuf::alloc(DMA_SLOT_BYTES * DMA_CHANNELS / 4096) else {
120                return false;
121            };
122            *self.dma.borrow_mut() = Some(buf);
123        }
124        self.mode.set(mode);
125        true
126    }

128    /// Guest-physical base of `ch`'s DMA slot, or `None` outside DMA mode.
129    ///
130    /// In SAS grant pages are identity-mapped, so the value programmed into
131    /// `HCDMA` is the same address the CPU uses.
132    fn dma_slot(&self, ch: usize) -> Option<usize> {
133        if self.mode.get() != TransferMode::Dma || ch >= DMA_CHANNELS {
134            return None;
135        }
136        let guard = self.dma.borrow();
137        guard.as_ref().map(|b| b.phys() + ch * DMA_SLOT_BYTES)
138    }

140    /// Bus address of a RAM physical address on this SoC.
141    ///
142    /// A bus master does not see ARM physical addresses. On the BCM283x the
143    /// Raspberry Pi device tree's `_DMA` resource states the translation
144    /// outright: "Bus 0xC0000000 -> CPU 0x00000000", over the first gigabyte
145    /// and marked NonCacheable. U-Boot's working dwc2 driver programs `HCDMA`
146    /// through exactly this translation (`phys_to_bus`).
147    ///
148    /// Programming the raw ARM physical address asks the core to read and write
149    /// a different location, and the failure is quiet: the address still lands
150    /// in mapped memory, so no AHB error is raised, and the CPU-side read-back
151    /// of the slot looks correct because the CPU and the core are simply
152    /// looking at different bytes.
153    #[inline]
154    fn bus_address(phys: usize) -> u32 {
155        (phys | 0xC000_0000) as u32
156    }
157
158    /// Point the channel's DMA engine at `offset` bytes into its slot.
159    ///
160    /// Programmed **once per transfer, and always before `CHENA`**. The DWC2
161    /// advances `HCDMA` itself as it moves each packet (`hcdma += actual`), so a
162    /// multi-packet transfer walks the slot on its own; re-arming mid-transfer
163    /// both defeats that walk and races the core's asynchronous completion.
164    ///
165    /// The ordering against `CHENA` is not a preference. The core latches its
166    /// DMA address when the channel is enabled, so a start that arms `HCDMA`
167    /// afterwards -- or never -- sends the first transaction of the transfer to
168    /// whatever address the previous one left behind. On a SETUP that is a
169    /// request the device ACKs and then STALLs one stage later, and the retry
170    /// succeeds because by then the register holds the right address.
171    fn program_hcdma(&self, ch: usize, offset: usize) {
172        if let Some(base) = self.dma_slot(ch) {
173            let addr = base + offset.min(DMA_SLOT_BYTES);
174            self.write32(hcdma(ch), Self::bus_address(addr));
175        }
176    }
177
178    /// Publish CPU writes in `[offset, offset+len)` to the device.
179    ///
180    /// A refused sync is reported, never skipped: without the clean the core
181    /// reads stale memory, so the device receives a request whose bytes are not
182    /// the ones staged and answers with a STALL -- a failure that looks like a
183    /// protocol problem and is actually a missing cache operation.
184    fn cache_clean(&self, ch: usize, offset: usize, len: usize) {
185        let guard = self.dma.borrow();
186        if let Some(buf) = guard.as_ref() {
187            let start = ch * DMA_SLOT_BYTES + offset.min(DMA_SLOT_BYTES);
188            let end = (start + len).min((ch + 1) * DMA_SLOT_BYTES);
189            let span = end.saturating_sub(start);
190            match buf.begin_cache_sync(start, span) {
191                Some(token) => {
192                    if !buf.complete_cache_sync(token) {
193                        ostd::io::println("[dwc2] WARN: cache sync completion refused");
194                    }
195                }
196                None => {
197                    ostd::io::println(
198                        "[dwc2] WARN: cache sync refused - device may read stale memory",
199                    );
200                }
201            }
202        }
203    }
204
205    /// Drop stale cache lines over a range the device just wrote.
206    fn cache_invalidate(&self, ch: usize, offset: usize, len: usize) {
207        self.cache_clean(ch, offset, len);
208    }

210    /// Copy an outgoing payload into the channel slot at `offset` and publish it.
211    ///
212    /// This is one of the crate's two `allow(unsafe_code)` islands (see
213    /// `scripts/unsafe-allowlist.toml`): the destination is a grant region this
214    /// cell owns for its whole lifetime, `n` is bounded by the slot size and the
215    /// source length, and the region is not aliased by any other live reference.
216    #[allow(unsafe_code)]
217    fn stage_out(&self, ch: usize, offset: usize, data: &[u8]) {
218        let guard = self.dma.borrow();
219        let n = data.len().min(DMA_SLOT_BYTES - offset.min(DMA_SLOT_BYTES));
220        if let Some(buf) = guard.as_ref() {
221            // SAFETY: destination is this cell's grant slot at a bounded
222            // offset; `n` is bounded by the remaining slot and by `data`.
223            unsafe {
224                core::ptr::copy_nonoverlapping(
225                    data.as_ptr(),
226                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset),
227                    n,
228                );
229            }
230        }
231        drop(guard);
232        self.cache_clean(ch, offset, n);
233    }

235    /// The third `allow(unsafe_code)` island: a volatile read of this cell's
236    /// own grant slot at a bounded offset, used only for the trace below.
237    #[allow(unsafe_code)]
238    /// Print the first bytes staged in a channel slot.
239    ///
240    /// Proves what the core is about to hand the device. If these are not the
241    /// request the caller built, the fault is in staging or cache maintenance;
242    /// if they are, the fault is between memory and the bus.
243    pub fn trace_slot(&self, ch: usize, label: &str, len: usize) {
244        let guard = self.dma.borrow();
245        let Some(buf) = guard.as_ref() else {
246            return;
247        };
248        ostd::io::print("[dwc2] slot ");
249        ostd::io::print(label);
250        ostd::io::print(" =");
251        let n = len.min(16).min(DMA_SLOT_BYTES);
252        for i in 0..n {
253            // SAFETY: reading this cell's own grant slot at a bounded offset.
254            let byte = unsafe {
255                core::ptr::read_volatile(
256                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES).wrapping_add(i),
257                )
258            };
259            ostd::io::print(" ");
260            print_hex_val(byte as u32);
261        }
262        ostd::io::println("");
263    }

265    /// Copy a received payload out of the channel slot at `offset`.
266    ///
267    /// The second `allow(unsafe_code)` island: same owned-grant argument as
268    /// [`Self::stage_out`], with `n` additionally bounded by the destination.
269    #[allow(unsafe_code)]
270    fn collect_in(&self, ch: usize, offset: usize, dst: &mut [u8], len: usize) {
271        self.cache_invalidate(ch, offset, len);
272        let guard = self.dma.borrow();
273        if let Some(buf) = guard.as_ref() {
274            let n = len
275                .min(dst.len())
276                .min(DMA_SLOT_BYTES.saturating_sub(offset.min(DMA_SLOT_BYTES)));
277            // SAFETY: source is this cell's grant slot at a bounded offset;
278            // `n` is bounded by the slot remainder and the destination.
279            unsafe {
280                core::ptr::copy_nonoverlapping(
281                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset) as *const u8,
282                    dst.as_mut_ptr(),
283                    n,
284                );
285            }
286        }
287    }

311    /// Execute a standard 8-byte USB SETUP packet on Channel 0.
312    pub fn send_setup(&self, dev_addr: u8, setup: &[u8; 8]) -> ViResult<()> {
313        let ch = 0;
314        self.write32(hcsplt(ch), 0);
315        self.write32(hcintmsk(ch), 0x07FF);
316        // 1. Clear pending channel interrupts
317        self.write32(hcint(ch), 0xFFFF_FFFF);
318        // 2. Configure Transfer Size (HCTSIZ):
319        // XFERSIZE = 8, PKTCNT = 1, PID = 3 (SETUP)
320        let sctsiz = 8 | (1 << 19) | (3 << 29);
321        self.write32(hctsiz(ch), sctsiz);
322
323        // 3. Hand the 8 setup bytes to the core *before* the channel starts. In
324        //    DMA mode the core reads them from guest memory at HCDMA; the FIFO is
325        //    not consulted at all.
326        //
327        //    This order is the whole ballgame. Enabling the channel first makes
328        //    the core latch the previous transfer's HCDMA, so this SETUP is read
329        //    from the wrong place, the device ACKs eight bytes it never looked
330        //    at, and the data stage that follows is STALLed. The retry passes,
331        //    because the failed attempt has since written the right address --
332        //    which is exactly why every first attempt failed and every second one
333        //    worked.
334        if self.dma_slot(ch).is_some() {
335            self.program_hcdma(ch, 0);
336            self.stage_out(ch, 0, setup);
337            static SETUP_TRACED: core::sync::atomic::AtomicBool =
338                core::sync::atomic::AtomicBool::new(false);
339            if !SETUP_TRACED.swap(true, core::sync::atomic::Ordering::Relaxed) {
340                self.trace_slot(ch, "setup", setup.len());
341            }
342        } else {
343            let w0 = u32::from_le_bytes([setup[0], setup[1], setup[2], setup[3]]);
344            let w1 = u32::from_le_bytes([setup[4], setup[5], setup[6], setup[7]]);
345            self.write_fifo(ch, w0);
346            self.write_fifo(ch, w1);
347        }
348
349        // 4. Start the channel last: HCCHAR fields (bits 0-10 MPS, 11-14 EPNUM,
350        //    15 EPDIR, 18-19 EPTYPE, 20 MC, 22-28 DEVADDR, 31 CHENA): EPNUM = 0,
351        //    EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
352        let scchar =
353            (self.control_mps() as u32) | (1 << 20) | ((dev_addr as u32) << 22) | (1 << 31);
354        self.write32(hcchar(ch), scchar);
355
356        // 5. Poll for completion
357        self.wait_channel(ch)
358    }

360    /// Receive the DATA IN phase of a control transfer on Channel 0.
361    ///
362    /// Returns the bytes actually received. Packets are sized by the control
363    /// endpoint's MPS and the transfer ends at the first **short** packet
364    /// (USB 2.0 §8.5.3.1) — a device answering an 18-byte descriptor request
365    /// with its real 8-byte report must not be read as if it had 18 bytes
366    /// available, which is what a fixed-size read would do.
367    pub fn recv_data(&self, dev_addr: u8, buf: &mut [u8]) -> ViResult<usize> {
368        let ch = 0;
369        let mps = (self.control_mps() as usize).max(1);
370        let mut received = 0;
371        let mut toggle = 2u32; // PID 2 = DATA1 for the first data packet
372                               // Arm the DMA engine once; the core advances it per packet.
373        self.program_hcdma(ch, 0);
374
375        while received < buf.len() {
376            let chunk = (buf.len() - received).min(mps);
377            self.prepare_channel(ch);
378            self.write32(hcsplt(ch), 0);
379            self.write32(hcintmsk(ch), 0x07FF);
380            let sctsiz = (chunk as u32) | (1 << 19) | (toggle << 29);
381            self.write32(hctsiz(ch), sctsiz);
382
383            // HCCHAR: EPNUM = 0, EPDIR = 1 (IN), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
384            let scchar =
385                (mps as u32) | (1 << 15) | (1 << 20) | ((dev_addr as u32) << 22) | (1 << 31);
386            self.write32(hcchar(ch), scchar);
387
388            self.wait_channel(ch)?;
389
390            // The core writes the *remaining* count back into HCTSIZ.
391            let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
392            let got = chunk.saturating_sub(remaining).min(chunk);
393
394            let words = got.div_ceil(4);
395            for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
396                for (j, byte) in word.to_le_bytes().iter().enumerate() {
397                    let offset = i * 4 + j;
398                    let idx = received + offset;
399                    if idx < buf.len() && offset < got {
400                        buf[idx] = *byte;
401                    }
402                }
403            }
404
405            received += got;
406            if got < chunk {
407                break;
408            }
409            toggle = if toggle == 2 { 0 } else { 2 }; // DATA1 <-> DATA0
410        }
411
412        if self.dma_slot(ch).is_some() && received > 0 {
413            let mut tmp = [0u8; DMA_SLOT_BYTES];
414            let n = received.min(buf.len()).min(DMA_SLOT_BYTES);
415            self.collect_in(ch, 0, &mut tmp, n);
416            buf[..n].copy_from_slice(&tmp[..n]);
417        }
418
419        Ok(received)
420    }

422    /// Send STATUS handshake on Channel 0 (0-byte packet with DATA1).
423    pub fn send_status(&self, dev_addr: u8, is_in: bool) -> ViResult<()> {
424        let ch = 0;
425        self.prepare_channel(ch);
426        self.write32(hcsplt(ch), 0);
427        self.write32(hcintmsk(ch), 0x07FF);
428        // XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1)
429        let sctsiz = (1 << 19) | (2 << 29);
430        self.write32(hctsiz(ch), sctsiz);
431
432        // A zero-length handshake moves no bytes, but the channel still latches
433        // an address when it starts and the previous transfer's is not a
434        // position this stage is allowed to begin from.
435        self.program_hcdma(ch, 0);
436
437        // HCCHAR: EPNUM = 0, EPTYPE = 0 (Control), MPS = 64, MC = 1, CHENA = 1, EPDIR from the caller.
438        let epdir = if is_in { 1 } else { 0 };
439        let scchar = (self.control_mps() as u32)
440            | (epdir << 15)
441            | (1 << 20)
442            | ((dev_addr as u32) << 22)
443            | (1 << 31);
444        self.write32(hcchar(ch), scchar);
445
446        self.wait_channel(ch)
447    }

449    /// Execute a complete synchronous USB Control Transfer.
450    pub fn control_transfer(
451        &self,
452        dev_addr: u8,
453        req_type: u8,
454        req: u8,
455        val: u16,
456        idx: u16,
457        buf: &mut [u8],
458    ) -> ViResult<usize> {
459        let is_in = (req_type & 0x80) != 0;
460        let length = buf.len() as u16;
461
462        let setup = [
463            req_type,
464            req,
465            (val & 0xFF) as u8,
466            ((val >> 8) & 0xFF) as u8,
467            (idx & 0xFF) as u8,
468            ((idx >> 8) & 0xFF) as u8,
469            (length & 0xFF) as u8,
470            ((length >> 8) & 0xFF) as u8,
471        ];
472
473        // Each phase reports its own failure: a control transfer that fails
474        // silently is the hardest kind of USB bug to find, and the request
475        // fields are the only thing that identifies which one it was.
476        if let Err(e) = self.send_setup(dev_addr, &setup) {
477            self.report_failure("setup", dev_addr, req_type, req, val, idx, length, e);
478            return Err(e);
479        }
480
481        // 2. DATA phase (optional)
482        let mut actual = 0;
483        if length > 0 {
484            let result = if is_in {
485                self.recv_data(dev_addr, buf)
486            } else {
487                self.send_data(dev_addr, buf).map(|()| buf.len())
488            };
489            match result {
490                Ok(n) => actual = n,
491                Err(e) => {
492                    self.report_failure("data", dev_addr, req_type, req, val, idx, length, e);
493                    return Err(e);
494                }
495            }
496        }
497
498        // 3. STATUS phase (handshake in opposite direction)
499        if let Err(e) = self.send_status(dev_addr, !is_in) {
500            self.report_failure("status", dev_addr, req_type, req, val, idx, length, e);
501            return Err(e);
502        }
503
504        Ok(actual)
505    }

507    /// Log one failed control-transfer phase with the request that caused it.
508    #[allow(clippy::too_many_arguments)]
509    fn report_failure(
510        &self,
511        phase: &str,
512        dev_addr: u8,
513        req_type: u8,
514        req: u8,
515        val: u16,
516        idx: u16,
517        length: u16,
518        error: ViError,
519    ) {
520        ostd::io::print("[dwc2] control ");
521        ostd::io::print(phase);
522        ostd::io::print(" failed: bmRequestType=0x");
523        print_hex_val(req_type as u32);
524        ostd::io::print(" bRequest=0x");
525        print_hex_val(req as u32);
526        ostd::io::print(" wValue=0x");
527        print_hex_val(val as u32);
528        ostd::io::print(" wIndex=0x");
529        print_hex_val(idx as u32);
530        ostd::io::print(" wLength=0x");
531        print_hex_val(length as u32);
532        ostd::io::print(" addr=");
533        print_hex_val(dev_addr as u32);
534        ostd::io::print(" err=");
535        match error {
536            ViError::IO => {
537                // The raw outcome, not a bucket: a STALL, an AHB error and a
538                // core that never finished all used to print the same word.
539                ostd::io::print("IO - ");
540                ostd::io::println(self.describe_hcint(self.last_hcint.get()));
541            }
542            ViError::WouldBlock => ostd::io::println("WouldBlock (NAK)"),
543            _ => ostd::io::println("other"),
544        }
545    }

547    /// Send DATA OUT phase on Channel 0.
548    fn send_data(&self, dev_addr: u8, data: &[u8]) -> ViResult<()> {
549        let ch = 0;
550        let mps = (self.control_mps() as usize).max(1);
551        let mut sent = 0;
552        let mut toggle = 2; // PID 2 = DATA1
553                            // Stage the whole payload once and arm DMA once: the core reads it
554                            // packet by packet, advancing HCDMA itself.
555        if self.dma_slot(ch).is_some() {
556            self.stage_out(ch, 0, data);
557            self.program_hcdma(ch, 0);
558        }
559
560        while sent < data.len() {
561            let chunk = (data.len() - sent).min(mps);
562            self.prepare_channel(ch);
563            self.write32(hcsplt(ch), 0);
564            self.write32(hcintmsk(ch), 0x07FF);
565            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
566            self.write32(hctsiz(ch), sctsiz);
567
568            // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
569            let scchar = (mps as u32) | (1 << 20) | ((dev_addr as u32) << 22) | (1 << 31);
570            self.write32(hcchar(ch), scchar);
571
572            if self.dma_slot(ch).is_some() {
573                // Payload already staged and HCDMA already armed.
574            } else {
575                let words = chunk.div_ceil(4);
576                for i in 0..words {
577                    let mut b = [0u8; 4];
578                    for (j, byte) in b.iter_mut().enumerate() {
579                        let offset = i * 4 + j;
580                        if sent + offset < data.len() && offset < chunk {
581                            *byte = data[sent + offset];
582                        }
583                    }
584                    self.write_fifo(ch, u32::from_le_bytes(b));
585                }
586            }
587
588            self.wait_channel(ch)?;
589
590            sent += chunk;
591            toggle = if toggle == 2 { 0 } else { 2 };
592        }
593
594        Ok(())
595    }

597    /// Transmit a raw Ethernet packet via Bulk OUT (Channel 2, EP 2).
598    pub fn bulk_transmit(&self, dev_addr: u8, ep_num: u8, packet: &[u8]) -> ViResult<()> {
599        let ch = 2;
600        let mut sent = 0;
601        let mut toggle = 0; // Starts at DATA0
602
603        let use_dma = self.dma_slot(ch).is_some();
604        if use_dma {
605            self.stage_out(ch, 0, packet);
606            self.program_hcdma(ch, 0);
607        }
608        while sent < packet.len() {
609            let chunk = (packet.len() - sent).min(512); // 512 bytes for High-Speed Bulk
610
611            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
612            // HCCHAR: EPDIR = 0 (OUT), EPTYPE = 2 (Bulk), MC = 1 packet, MPS = 512 (HS Bulk).
613            let scchar = 512
614                | ((ep_num as u32) << 11)
615                | (2 << 18)
616                | (1 << 20)
617                | ((dev_addr as u32) << 22)
618                | (1 << 31);
619
620            let mut retries = 0;
621            loop {
622                self.prepare_channel(ch);
623                self.write32(hcsplt(ch), 0);
624                self.write32(hcintmsk(ch), 0x07FF);
625                self.write32(hctsiz(ch), sctsiz);
626                self.write32(hcchar(ch), scchar);
627
628                if use_dma {
629                    // Staged and armed before the loop; the core walks it.
630                } else {
631                    let words_count = chunk.div_ceil(4);
632                    for i in 0..words_count {
633                        let mut b = [0u8; 4];
634                        for (j, byte) in b.iter_mut().enumerate() {
635                            let offset = i * 4 + j;
636                            let idx = sent + offset;
637                            if idx < packet.len() && offset < chunk {
638                                *byte = packet[idx];
639                            }
640                        }
641                        self.write_fifo(ch, u32::from_le_bytes(b));
642                    }
643                }
644
645                match self.wait_channel(ch) {
646                    Ok(()) => break,
647                    Err(ViError::WouldBlock) => {
648                        retries += 1;
649                        if retries > 50 {
650                            ostd::io::println("[dwc2] bulk_transmit: exceeded 50 NAK retries");
651                            return Err(ViError::IO);
652                        }
653                        sys_yield();
654                    }
655                    Err(e) => return Err(e),
656                }
657            }
658
659            sent += chunk;
660            toggle = if toggle == 0 { 2 } else { 0 };
661        }
662
663        Ok(())
664    }

666    /// Receive a raw packet via Bulk IN (Channel 1, EP 1). Returns received length or 0 if nothing.
667    pub fn bulk_receive(&self, dev_addr: u8, ep_num: u8, buf: &mut [u8]) -> ViResult<usize> {
668        let ch = 1;
669        self.prepare_channel(ch);
670        self.write32(hcsplt(ch), 0);
671        self.write32(hcintmsk(ch), 0x07FF);
672        let want = buf.len().min(512);
673        let sctsiz = (want as u32) | (1 << 19);
674        self.write32(hctsiz(ch), sctsiz);
675
676        let scchar = 512
677            | ((ep_num as u32) << 11)
678            | (1 << 15) // IN
679            | (2 << 18) // Bulk
680            | (1 << 20) // MC = 1
681            | ((dev_addr as u32) << 22)
682            | (1 << 31);
683        self.program_hcdma(ch, 0);
684        self.write32(hcchar(ch), scchar);
685
686        // Non-blocking wait: check if transfer completed or NAK
687        let mut count = 0;
688        while count < 1000 {
689            let int = self.read32(hcint(ch));
690            if int & (1 << 0) != 0 {
691                // The core reports the actual length in HCTSIZ either way, but
692                // only DMA deposits the payload outside the FIFO.
693                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
694                let got = want.saturating_sub(remaining);
695                if self.dma_slot(ch).is_some() {
696                    self.collect_in(ch, 0, buf, got);
697                } else {
698                    let words = want.div_ceil(4);
699                    for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
700                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
701                            let idx = i * 4 + j;
702                            if idx < buf.len() && idx < want {
703                                buf[idx] = *byte;
704                            }
705                        }
706                    }
707                }
708                return Ok(got);
709            }
710            if int & (1 << 4) != 0 {
711                // NAK: device has no packet right now
712                return Ok(0);
713            }
714            count += 1;
715            sys_yield();
716        }
717
718        Ok(0)
719    }

721    /// Receive up to one packet via Interrupt IN on `ch`.
722    ///
723    /// HID input reports ride interrupt endpoints. The device NAKs until it has
724    /// a report, so a NAK is a normal "no data yet" answer, not an error — the
725    /// caller polls. Returns the *actual* transferred length, read back from
726    /// `HCTSIZ` rather than assumed: a HID report can be shorter than the buffer
727    /// (an 8-byte keyboard report into a 64-byte buffer), and using the request
728    /// length would hand the decoder trailing garbage.
729    ///
730    /// `ch` must not collide with the control (0) or Ethernet bulk (1, 2)
731    /// channels.
732    pub fn interrupt_receive(
733        &self,
734        ch: usize,
735        dev_addr: u8,
736        ep_num: u8,
737        mps: u16,
738        buf: &mut [u8],
739    ) -> ViResult<usize> {
740        let want = buf.len().min(mps.max(1) as usize).min(1024);
741        if want == 0 {
742            return Ok(0);
743        }
744
745        self.prepare_channel(ch);
746        self.write32(hcsplt(ch), 0);
747        self.write32(hcintmsk(ch), 0x07FF);
748
749        // HCTSIZ: XFERSIZE (= want), PKTCNT = 1, PID = DATA0 (0). The core
750        // overwrites XFERSIZE with the remaining count as it transfers.
751        let sctsiz = (want as u32) | (1 << 19);
752        self.write32(hctsiz(ch), sctsiz);
753
754        // HCCHAR: MPS from the endpoint descriptor, EPNUM, IN direction,
755        // EPTYPE = 3 (Interrupt), MC = 1, DEVADDR, CHENA.
756        let scchar = (mps as u32 & 0x7FF)
757            | ((ep_num as u32) << 11)
758            | (1 << 15) // EPDIR = IN
759            | (3 << 18) // EPTYPE = Interrupt
760            | (1 << 20) // MC = 1
761            | ((dev_addr as u32) << 22)
762            | (1 << 31); // CHENA
763        self.program_hcdma(ch, 0);
764        self.write32(hcchar(ch), scchar);
765
766        let mut polls = 0u32;
767        while polls < 2_000 {
768            let int = self.read32(hcint(ch));
769
770            // NAK — device has no report queued. Halt and report "no data".
771            if int & (1 << 4) != 0 {
772                self.halt_channel(ch);
773                return Ok(0);
774            }
775            // Errors (STALL / babble / transaction error) end the poll cycle.
776            if int & ((1 << 2) | (1 << 3) | (1 << 7)) != 0 {
777                self.halt_channel(ch);
778                return Err(ViError::IO);
779            }
780
781            let complete = int & (1 << 0) != 0 || int & (1 << 1) != 0 || int & (1 << 5) != 0;
782            if complete {
783                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
784                let got = want.saturating_sub(remaining);
785                self.halt_channel(ch);
786
787                if got > 0 {
788                    if self.dma_slot(ch).is_some() {
789                        // DMA deposited the report in the channel slot.
790                        let mut tmp = [0u8; DMA_SLOT_BYTES];
791                        self.collect_in(ch, 0, &mut tmp, got);
792                        let n = got.min(buf.len());
793                        buf[..n].copy_from_slice(&tmp[..n]);
794                    } else {
795                        // Drain ceil(got/4) FIFO words; only the first `got`
796                        // bytes are meaningful (the tail word is padding).
797                        let words = got.div_ceil(4);
798                        for w in 0..words {
799                            let word = self.read_fifo(ch);
800                            for (j, byte) in word.to_le_bytes().iter().enumerate() {
801                                let idx = w * 4 + j;
802                                if idx < got && idx < buf.len() {
803                                    buf[idx] = *byte;
804                                }
805                            }
806                        }
807                    }
808                } else if self.dma_slot(ch).is_none() {
809                    // A zero-length packet still has a FIFO word to retire on
810                    // some revisions; drain one so the channel is clean.
811                    let _ = self.read_fifo(ch);
812                }
813                return Ok(got);
814            }
815
816            polls += 1;
817            sys_yield();
818        }
819
820        self.halt_channel(ch);
821        Ok(0)
822    }

824    /// Name the host-channel interrupt bit that ended a transfer.
825    ///
826    /// The DWC2 encodes one cause per bit and they need different fixes:
827    /// `AHBERR` is the core failing to reach memory at `HCDMA`, `XACTERR` is a
828    /// bus protocol error, `BBLERR` is the device sending more than the
829    /// programmed packet size, `DTERR` is a data-toggle mismatch, and `STALL`
830    /// is the device itself refusing. Collapsing them into one "IO error" hides
831    /// which of those actually happened.
832    fn describe_hcint(&self, int: u32) -> &'static str {
833        if int & (1 << 2) != 0 {
834            return "AHBERR - core could not reach memory at HCDMA";
835        }
836        if int & (1 << 8) != 0 {
837            return "BBLERR - babble, device exceeded the packet size";
838        }
839        if int & (1 << 3) != 0 {
840            return "STALL - endpoint refused the request";
841        }
842        if int & (1 << 7) != 0 {
843            return "XACTERR - transaction error";
844        }
845        if int & (1 << 10) != 0 {
846            return "DTERR - data toggle mismatch";
847        }
848        if int & (1 << 9) != 0 {
849            return "FRMOVRN - frame overrun";
850        }
851        if int & (1 << 6) != 0 {
852            return "NYET - not ready";
853        }
854        if int & (1 << 4) != 0 {
855            return "NAK";
856        }
857        "unknown"
858    }

869    /// Dump the channel and core registers that explain a stalled transfer.
870    ///
871    /// A timeout with no status bit set (`HCINT == 0`) is ambiguous from the
872    /// outside: the channel may never have been issued, may have been issued and
873    /// silently abandoned by the core, or the port may have dropped underneath
874    /// it. These registers distinguish those cases.
875    pub fn dump_transfer_state(&self, ch: usize, why: &str) {
876        ostd::io::print("[dwc2] state at ");
877        ostd::io::print(why);
878        ostd::io::println(":");
879        for (name, offset) in [
880            ("HCCHAR", hcchar(ch)),
881            ("HCTSIZ", hctsiz(ch)),
882            ("HCDMA", hcdma(ch)),
883            ("HCINTMSK", hcintmsk(ch)),
884        ] {
885            self.dump_reg(name, self.read32(offset));
886        }
887        self.dump_reg("HCINT", self.read32(hcint(ch)));
888        self.dump_reg("HAINT", self.read32(HAINT));
889        self.dump_reg("GINTSTS", self.read32(GINTSTS));
890        self.dump_reg("GINTMSK", self.read32(GINTMSK));
891        self.dump_reg("HPRT0", self.read32(HPRT0));
892        self.dump_reg("GRSTCTL", self.read32(GRSTCTL));
893        self.dump_reg("GNPTXSTS", self.read32(GNPTXSTS));
894        self.dump_reg("MODE", self.mode.get() as u32);
895    }

897    /// Log a channel error once per distinct cause, with the register state.
898    fn report_channel_error(&self, ch: usize, int: u32) {
899        static REPORTED: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
900        if REPORTED.fetch_add(1, core::sync::atomic::Ordering::Relaxed) < 4 {
901            ostd::io::print("[dwc2] channel error: ");
902            ostd::io::println(self.describe_hcint(int));
903            self.dump_transfer_state(ch, "error");
904        }
905    }

907    /// Leave `ch` disabled before arming it, whatever state it was left in.
908    ///
909    /// A channel that is still enabled from an abandoned transfer ignores the
910    /// next `CHENA`, so each transfer starts by making sure the core has really
911    /// stopped and its interrupt bits are cleared.
912    fn prepare_channel(&self, ch: usize) {
913        if self.read32(hcchar(ch)) & (1 << 31) != 0 {
914            self.halt_channel(ch);
915        }
916        self.write32(hcint(ch), 0xFFFF_FFFF);
917    }

919    /// Wait for channel transfer completion or error with timeout.
920    ///
921    /// BCM2837 DWC2 Slave mode: the core does NOT reliably set XFERCOMPL
922    /// or CHHLTD after a successful transaction.  ACK (bit 5) from the
923    /// device is the definitive completion signal; we halt the channel
924    /// manually after seeing it.
925    fn wait_channel(&self, ch: usize) -> ViResult<()> {
926        let mut count = 0;
927        while count < 50_000 {
928            let int = self.read32(hcint(ch));
929
930            // ── Hardware-generated halt ───────────────────────────────
931            if int & (1 << 1) != 0 {
932                // CHHLTD
933                self.write32(hcint(ch), 0xFFFF_FFFF);
934                // Babble (8) and data-toggle error (10) end a transfer just as
935                // surely as the bits that were already checked.
936                if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
937                    self.report_channel_error(ch, int);
938                    self.last_hcint.set(int);
939                    return Err(ViError::IO);
940                }
941                if int & (1 << 4) != 0 {
942                    self.last_hcint.set(int);
943                    return Err(ViError::WouldBlock);
944                }
945                return Ok(());
946            }
947
948            // ── Transfer complete ─────────────────────────────────────
949            if int & (1 << 0) != 0 {
950                self.halt_channel(ch);
951                return Ok(());
952            }
953
954            // ── ACK = device accepted the packet (BCM2837 primary path)
955            if int & (1 << 5) != 0 {
956                self.halt_channel(ch);
957                return Ok(());
958            }
959
960            // ── Error conditions ──────────────────────────────────────
961            if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
962                self.report_channel_error(ch, int);
963                self.halt_channel(ch);
964                self.last_hcint.set(int);
965                return Err(ViError::IO);
966            }
967            if int & (1 << 4) != 0 {
968                self.halt_channel(ch);
969                self.last_hcint.set(int);
970                return Err(ViError::WouldBlock);
971            }
972
973            count += 1;
974            sys_yield();
975        }
976        // Timeout — capture hcint BEFORE halt clears it
977        let int = self.read32(hcint(ch));
978        let char_val = self.read32(hcchar(ch));
979        self.halt_channel(ch);
980        static TIMEOUT_COUNT: core::sync::atomic::AtomicUsize =
981            core::sync::atomic::AtomicUsize::new(0);
982        let attempt = TIMEOUT_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
983        if attempt < 3 {
984            ostd::io::print("[dwc2] TIMEOUT ch=");
985            print_hex_val(ch as u32);
986            ostd::io::print(" hcint=0x");
987            print_hex_val(int);
988            ostd::io::print(" hcchar=0x");
989            print_hex_val(char_val);
990            ostd::io::println("");
991            self.dump_transfer_state(ch, "timeout");
992        }
993        self.last_hcint.set(int);
994        Err(ViError::IO)
995    }

997    /// Explicitly halt a host channel (required in DWC2 Slave mode).
998    ///
999    /// `CHDIS` and `CHENA` must never be set together. The core reads
1000    /// `CHENA=1` as a fresh enable, so raising it while requesting a disable
1001    /// **re-arms the channel with the parameters still in `HCCHAR`/`HCTSIZ`** --
1002    /// a second, unwanted transaction launched from a completed one. Only
1003    /// `CHDIS` is raised; the core clears `CHENA` and reports `CHHLTD`.
1004    fn halt_channel(&self, ch: usize) {
1005        let reg = hcchar(ch);
1006        let mut val = self.read32(reg);
1007        val |= 1 << 30; // CHDIS
1008        val &= !(1 << 31); // CHENA
1009        self.write32(reg, val);
1010        // Wait for CHHLTD (bit 1) with a short timeout
1011        for _ in 0..10_000 {
1012            if self.read32(hcint(ch)) & (1 << 1) != 0 {
1013                break;
1014            }
1015            sys_yield();
1016        }
1017        self.write32(hcint(ch), 0xFFFF_FFFF);
1018    }
1019}

1021fn print_hex_val(val: u32) {
1022    const HEX: &[u8; 16] = b"0123456789ABCDEF";
1023    let mut buf = [0u8; 8];
1024    for i in 0..8 {
1025        buf[7 - i] = HEX[((val >> (i * 4)) & 0xF) as usize];
1026    }
1027    if let Ok(s) = core::str::from_utf8(&buf) {
1028        ostd::io::print(s);
1029    }
