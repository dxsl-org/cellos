10/// How host channels move payload bytes.
11///
12/// The BCM2837 DWC2 supports both; which one a given environment implements is
13/// not discoverable from the register file, so the driver tries FIFO first and
14/// falls back to DMA. FIFO is the board-proven path (the LAN9514 Ethernet runs
15/// there); DMA is what QEMU's `hcd-dwc2` model implements, since it sources
16/// payloads from guest memory at `HCDMA` and ignores host-channel FIFO writes
17/// entirely.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum TransferMode {
20    /// Programmed I/O: payload goes through the channel's data FIFO.
21    Fifo,
22    /// The core reads/writes payloads directly from/to guest memory at `HCDMA`.
23    Dma,
24}
25
26/// Bytes reserved per host channel inside the DMA scratch region.
27///
28/// Sized for the largest payload this driver moves: a full configuration
29/// descriptor (≤512 bytes) and an Ethernet frame. One slot per channel keeps two
30/// in-flight channels from aliasing the same memory.
31pub const DMA_SLOT_BYTES: usize = 4096;
32/// Host channels that get a DMA slot (DWC2 implements up to 16; this driver uses 8).
33const DMA_CHANNELS: usize = 8;

35/// Control-endpoint max packet size assumed before a device reports its own.
36///
37/// USB 2.0 §5.5.3 fixes low-speed control endpoints at 8 bytes and allows
38/// full-speed ones 8/16/32/64, so 8 is the only value safe to assume for the
39/// first descriptor read. Declaring 64 there makes the host expect 64-byte
40/// packets from a device that sends 8, and the read fails.
41pub const CONTROL_MPS_LOW_FULL_SPEED: u8 = 8;
42
43/// HPRT0 `PRTSPD` encoding (DWC2 databook): the value `reset_port` returns.
44pub const PORT_SPEED_HIGH: u32 = 0;
45pub const PORT_SPEED_FULL: u32 = 1;
46pub const PORT_SPEED_LOW: u32 = 2;
47
48/// Control-endpoint packet size to assume for a device on a port at `speed`.
49///
50/// USB 2.0 5.5.3: low speed is 8 bytes, full speed is 8/16/32/64, and
51/// **high speed is 64**. The assumption has to follow the negotiated link
52/// speed, and `read_device_descriptor` replaces it with the device's real
53/// `bMaxPacketSize0` as soon as it has read one.
54pub fn initial_control_mps(speed: u32) -> u8 {
55    match speed {
56        PORT_SPEED_HIGH => 64,
57        _ => CONTROL_MPS_LOW_FULL_SPEED,
58    }
59}

61/// Maximum complete-splits issued while the accepted start-split is still live.
62const SPLIT_ATTEMPTS: usize = 4;
63/// Spin budget for one split poll attempt.
64///
65/// The split path keeps `split_pending` across calls, so a shorter spin only
66/// means the poll resumes on the next iteration. The full `SPIN_POLLS` budget
67/// here cost hundreds of milliseconds per HID poll, and the NIC driver cell
68/// shares its loop with HID: a NIC round trip (client -> cell -> front-end ->
69/// cell -> USB -> reply) then exceeded the client's reply timeout even though
70/// the frame was transmitted.
71const SPLIT_SPIN_POLLS: usize = 20_000;
72
73/// U-Boot's working DWC2 path abandons a split after four raw HFNUM ticks.
74const PERIODIC_SPLIT_MICROFRAMES: u32 = 4;
75
76/// Microframes in one full-speed frame.
77///
78/// `HFNUM` counts microframes, and a hub pairs the two halves of a split inside
79/// one millisecond frame, so the counter has to be shifted to ask that question.
80/// Both reference drivers derive the frame the same way.
81const FULL_FRAME_SHIFT: u32 = 3;
82
83/// Register reads a non-yielding channel wait will make before giving up.
84///
85/// The channel is left to report for itself (a frame check here once halted a
86/// split exactly as its low-speed half was finishing), so this is the only bound:
87/// it stops a core that never reports anything from spinning forever. Callers
88/// that must also stay inside a split's pairing window check the microframe
89/// counter themselves -- `run_packet` does that per attempt.
90const SPIN_POLLS: usize = 200_000;
91
92/// Poll budget for the ordinary `wait_channel`.
93/// Host microframes a *yielding* channel wait may spend before giving up.
94///
95/// 800 microframes is 100 ms of bus time — far more than a high-speed bulk
96/// transfer needs (1514 bytes is ~26 us on the wire) and more than a split
97/// sequence needs, but small enough that a stuck channel cannot hold the driver's
98/// loop. The count `WAIT_POLLS` it replaces could not express that: each poll
99/// yields, and a yield costs ~20 ms here.
100const WAIT_MICROFRAMES: u32 = 800;
101
102/// Yields the same wait may spend. A yield costs ~20 ms here, so eight of them is
103/// the same ~100 ms expressed in the unit the wait actually spends — and it is
104/// what bounds the wait when the microframe counter itself wraps (it is 16 bits
105/// wide, so an eight-second stall can mask back to a small elapsed value).
106const WAIT_YIELDS: u32 = 8;
107
108/// Retries a bulk OUT makes against a NAK before giving up.
109///
110/// Each attempt costs the channel wait plus a yield (~20 ms on the board), so 50 of
111/// them was a second per give-up and the console showed a line for each — while the
112/// Net Cell, which retries the frame itself, waited out its whole deadline inside
113/// the driver. Eight keeps a retry inside that deadline.
114const BULK_NAK_RETRIES: usize = 8;
115
116/// Register reads spent waiting for a Tx FIFO flush to complete.
117///
118/// The core clears `GRSTCTL.TXFFLSH` in microseconds; this only has to be long
119/// enough not to miss it.
120const TX_FLUSH_POLLS: usize = 1_000;
121
122/// Poll budget for waiting out a channel halt.
123///
124/// The core clears `CHENA` in microseconds; this only has to be long enough not
125/// to be missed, and every iteration of it is a yield handed to the rest of the
126/// system.
127const CHANNEL_HALT_POLLS: usize = 4_000;
128
129/// Complete-split attempts for a synchronous control transfer.
130///
131/// One per microframe, which is the most a split pairing can use: the window
132/// U-Boot bounds a split by is four raw `HFNUM` ticks past its start-split, so a
133/// fifth attempt could only land after the pairing is gone. The board trace is
134/// what set this: two attempts issued inside the start-split's own microframe
135/// both came back NYET and the low-speed keyboard never enumerated.
136const SPLIT_COMPLETE_ATTEMPTS: usize = 4;
137
138/// Register reads spent waiting for the microframe counter to move.
139///
140/// A microframe is 125 us and one `HFNUM` read costs a couple hundred
141/// nanoseconds, so this covers a boundary with room to spare while still
142/// returning if the counter never moves. The wait is a spin on purpose: a
143/// `sys_yield()` hands the CPU away for about twenty milliseconds, which is
144/// sixty-four microframes -- the pairing is over long before control comes back.
145const MICROFRAME_POLLS: usize = 2_000;

147/// Where a device sits when it has to be reached through a hub.
148///
149/// A high-speed hub does not pass full- and low-speed traffic through, so a
150/// device behind one is addressed in two transactions: a **start-split** the hub
151/// buffers, then a **complete-split** that collects the result. The core
152/// expresses both through `HCSPLT`.
153///
154/// This is per device, not per transfer, which is why the engine carries it the
155/// same way it carries `ctrl_mps`: both are facts about whoever is being
156/// addressed right now.
157#[derive(Clone, Copy, Debug, PartialEq, Eq)]
158pub struct Split {
159    /// The address the hub itself was assigned.
160    pub hub_addr: u8,
161    /// The downstream port on that hub holding the device.
162    pub port: u8,
163    /// The device is low-speed and needs the preamble its speed requires.
164    pub low_speed: bool,
165}

186/// Devices whose endpoint data toggles this engine tracks (USB device addresses).
187const TOGGLE_DEVICES: usize = 128;
188/// Endpoints per device (USB 2.0 allows 16 per direction).
189const TOGGLE_ENDPOINTS: usize = 16;
190
191/// Data toggle (`HCTSIZ.PID`) to use next, per device, endpoint and direction.
192///
193/// USB 2.0 §8.6: the data toggle belongs to the endpoint, not to a transfer, so it
194/// survives from one URB to the next — and the DWC2 reports the value a transfer
195/// actually ended on back in `HCTSIZ`. U-Boot's `dwc2.c` keeps exactly this table
196/// (`in_data_toggle`/`out_data_toggle`, filled from the register in
197/// `wait_for_chhltd`) because without it every packet after the first carries the
198/// wrong toggle. The driver here used a constant DATA0 for bulk and interrupt IN
199/// and restarted DATA0 for every bulk OUT call, which is what the board reported
200/// as `[lan9514] first bulk-IN failure: DTERR - data toggle mismatch` — and why
201/// the receive path delivered a handful of frames and then froze (`nic_rx=5`, then
202/// nothing, while the keyboard dropped reports).
203struct DataToggles {
204    in_toggle: [[u8; TOGGLE_ENDPOINTS]; TOGGLE_DEVICES],
205    out_toggle: [[u8; TOGGLE_ENDPOINTS]; TOGGLE_DEVICES],
206}
207
208impl DataToggles {
209    const fn new() -> Self {
210        Self {
211            in_toggle: [[0; TOGGLE_ENDPOINTS]; TOGGLE_DEVICES],
212            out_toggle: [[0; TOGGLE_ENDPOINTS]; TOGGLE_DEVICES],
213        }
214    }
215}

217/// Index a device address inside the toggle table.
218///
219/// `None` for an address or endpoint outside the tracked range: such a transfer
220/// runs with DATA0 rather than aliasing another device's toggle.
221fn toggle_slot(dev_addr: u8, ep_num: u8) -> Option<(usize, usize)> {
222    let dev = dev_addr as usize;
223    let ep = ep_num as usize & 0x0F;
224    (dev < TOGGLE_DEVICES && ep < TOGGLE_ENDPOINTS).then_some((dev, ep))
225}

227pub struct UsbHostEngine<'a> {
228    mmio: &'a MmioRegion,
229    /// Max packet size for control transfers on endpoint 0. Updated from the

233    /// Per-channel DMA scratch, allocated on first use in DMA mode.
234    dma: RefCell<Option<DmaBuf>>,
235    /// Hub the device currently being addressed sits behind, when it is not on
236    /// the root port itself.
237    split: Cell<Option<Split>>,
238    /// `HCINT` captured at the last channel outcome, success or failure.
239    ///
240    /// `wait_channel` collapses every hardware result into an error kind, and a
241    /// device refusing a request, the bus failing to carry it, and the core
242    /// never finishing look identical from the caller's side. Keeping the raw
243    /// register lets `report_failure` name which one happened -- and lets the
244    /// split paths tell a completed complete-split from one that ended on a
245    /// handshake that carries no data.

248    last_split_timing: Cell<SplitTiming>,
249    /// Endpoint data toggles; see [`DataToggles`].
250    toggles: RefCell<DataToggles>,
251}

253/// Transfer faults by cause, for the driver's `loop-trace` report.
254///
255/// Errors are the slow path, so counting them costs nothing on a working
256/// transfer, and the *mix* is what names the fault: a board run whose NIC
257/// transactions all fail wants to know whether they end in `XACTERR` (the bus
258/// protocol, i.e. the split schedule) or in the poll budget (a transfer the
259/// driver waited out), because the two need different fixes. Zero in a build
260/// that never prints them.
261pub static FAULT_XACTERR: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
262pub static FAULT_STALL: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
263pub static FAULT_OTHER: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
264pub static FAULT_TIMEOUT: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
265
266/// Fault counts as `[xacterr, stall, other, timeout]`.
267pub fn fault_counts() -> [usize; 4] {
268    use core::sync::atomic::Ordering;
269    [
270        FAULT_XACTERR.load(Ordering::Relaxed),
271        FAULT_STALL.load(Ordering::Relaxed),
272        FAULT_OTHER.load(Ordering::Relaxed),
273        FAULT_TIMEOUT.load(Ordering::Relaxed),
274    ]
275}

291    /// `HCTSIZ.PID` bits for the next transfer to `(dev_addr, ep_num)`.
292    fn next_pid(&self, dev_addr: u8, ep_num: u8, dir_in: bool) -> u32 {
293        let Some((dev, ep)) = toggle_slot(dev_addr, ep_num) else {
294            return 0; // DATA0
295        };
296        let toggles = self.toggles.borrow();
297        let pid = if dir_in {
298            toggles.in_toggle[dev][ep]
299        } else {
300            toggles.out_toggle[dev][ep]
301        };
302        (pid as u32) << 29
303    }
304
305    /// Adopt the data toggle the core ended the transfer on.
306    ///
307    /// Read back from `HCTSIZ.PID` rather than flipped locally: the core reports
308    /// the value the transfer actually completed with, which is what the endpoint
309    /// expects next. U-Boot does the same in `wait_for_chhltd`.
310    fn adopt_pid(&self, ch: usize, dev_addr: u8, ep_num: u8, dir_in: bool) {
311        let Some((dev, ep)) = toggle_slot(dev_addr, ep_num) else {
312            return;
313        };
314        let pid = ((self.read32(hctsiz(ch)) >> 29) & 0x3) as u8;
315        let mut toggles = self.toggles.borrow_mut();
316        if dir_in {
317            toggles.in_toggle[dev][ep] = pid;
318        } else {
319            toggles.out_toggle[dev][ep] = pid;
320        }
321    }
322
323    /// Whether the last failure was a data-toggle mismatch.
324    ///
325    /// A mismatch means the stored toggle was stale — the endpoint advanced and the
326    /// table did not (a port reset under a device, a device that re-enumerated) —
327    /// and the fix is to flip the stored value, not to retry it unchanged: the same
328    /// value fails the same way forever.
329    fn last_was_dterr(&self) -> bool {
330        self.last_hcint.get() & (1 << 10) != 0
331    }
332
333    /// Flip one endpoint's stored toggle; see [`Self::last_was_dterr`].
334    fn flip_pid(&self, dev_addr: u8, ep_num: u8, dir_in: bool) {
335        let Some((dev, ep)) = toggle_slot(dev_addr, ep_num) else {
336            return;
337        };
338        let mut toggles = self.toggles.borrow_mut();
339        let slot = if dir_in {
340            &mut toggles.in_toggle[dev][ep]
341        } else {
342            &mut toggles.out_toggle[dev][ep]
343        };
344        *slot = match *slot {
345            0 => 2, // DATA0 -> DATA1
346            _ => 0, // DATA1 -> DATA0
347        };
348    }

356        };
357        let mut toggles = self.toggles.borrow_mut();
358        if dir_in {
359            toggles.in_toggle[dev][ep] = 0;
360        } else {
361            toggles.out_toggle[dev][ep] = 0;

402    /// Current USB frame number, used to bound a complete-split.

407    /// The millisecond frame a periodic split is paired within.
408    ///
409    /// `HFNUM` counts microframes, so the raw counter advances every 125 us and
410    /// cannot answer a question about millisecond frames. Both reference drivers
411    /// shift it right three for exactly this check.
412    #[inline]
413    fn full_frame(&self) -> u32 {
414        self.frame_number() >> FULL_FRAME_SHIFT
415    }

417    /// The millisecond frame, for callers outside this module.
418    ///
419    /// An endpoint states its interval in milliseconds, so anything comparing
420    /// against that number has to count the same unit. `frame_number` counts
421    /// microframes, and a gate that compares the two directly opens eight times
422    /// too often and at whatever phase the serving loop happens to arrive at.
423    pub fn full_frame_number(&self) -> u32 {
424        self.full_frame()
425    }

427    /// The last channel outcome reported `XFERCOMPL`.
428    ///
429    /// A complete-split that ends on ACK has delivered nothing: Linux is
430    /// explicit that ACK belongs to the start-split and "should not occur in
431    /// CSPLIT". Reading the raw bit is the only way to tell, because every
432    /// completion collapses into the same `Ok` on the way out.
433    fn last_reported_complete(&self) -> bool {
434        self.last_hcint.get() & (1 << 0) != 0
435    }

437    /// The last channel failure was NYET rather than NAK.
438    ///
439    /// Both arrive as `WouldBlock` and they mean opposite things for a split: a
440    /// hub answering NYET has not finished the transaction and is worth asking
441    /// again, while a NAK ends the pairing and the next attempt has to begin
442    /// with a fresh start-split.
443    fn last_was_nyet(&self) -> bool {
444        self.last_hcint.get() & (1 << 6) != 0
445    }

447    /// The last channel failure was the device stalling the endpoint.
448    ///
449    /// Only this one means the endpoint is halted and needs clearing. A NYET is
450    /// the hub asking for time, a NAK is a device with nothing to send, and a bare
451    /// halt is the core ending a periodic channel at its frame boundary -- none of
452    /// them is a stall, and clearing one that was never set costs a control
453    /// transfer on every poll.
454    pub fn last_was_stall(&self) -> bool {
455        self.last_hcint.get() & (1 << 3) != 0
456    }

458    /// `HCCHAR` for this instant, with `ODDFRM` set from the frame the transfer
459    /// will be sent in.
460    ///
461    /// The core latches the parity of the frame a transfer lands in when the
462    /// channel starts, and a transfer whose parity disagrees is refused. The
463    /// frame read here is the one just started, and the transfer goes out in the
464    /// next, which is why the parity is inverted -- the same reading U-Boot
465    /// takes before it starts a channel.
466    #[inline]
467    fn start_hcchar(&self, base: u32) -> u32 {
468        // The low bit of the counter, which is what Linux uses too: it compares
469        // `wire_frame & 1` where wire_frame lives in the same units HFNUM reports,
470        // not in millisecond frames. Reading it through the frame shift would be
471        // the wrong bit.
472        if self.frame_number() & 1 == 0 {
473            base | HCCHAR_ODDFRM
474        } else {
475            base & !HCCHAR_ODDFRM
476        }
477    }

479    /// Start a channel and, behind a hub, run the split handshake that reaches
480    /// the device at all.
481    ///
482    /// Every packet takes two passes over the channel: the start-split hands the
483    /// transaction to the hub, and the complete-split collects what it buffered.
484    /// A hub that has not finished answers the complete-split with NYET, which is
485    /// why that is a retry here rather than a failure.
486    ///
487    /// Without this a full- or low-speed device on a hub port is never reached:
488    /// nothing on the bus answers, and the channel comes back with XACTERR. That
489    /// reads like a protocol error and is really a missing transaction.
490    ///
491    /// `arm(complete)` stages the payload and starts the channel for one pass.
492    fn run_packet(&self, ch: usize, arm: impl Fn(bool)) -> ViResult<()> {
493        if self.split.get().is_none() {
494            arm(false);
495            return self.wait_channel(ch);
496        }
497
498        // The log's NYET tells us the hub answered CSPLIT but not whether it
499        // had time to complete the low-speed transfer. Capture both sides of
500        // each channel wait, without printing until the entire packet has ended:
501        // sys_yield() can move this thread across many 125-us microframes.
502        let mut timing = SplitTiming::EMPTY;
503        let before_arm = self.frame_number();
504        arm(false);
505        let after_arm = self.frame_number();
506        let split_reg = self.read32(hcsplt(ch));
507        let start = self.wait_channel(ch);
508        timing.samples[0] = [
509            before_arm,
510            after_arm,
511            self.frame_number(),
512            self.last_hcint.get(),
513            self.read32(hcchar(ch)),
514            split_reg,
515        ];
516        timing.passes = 1;
517        if let Err(e) = start {
518            self.last_split_timing.set(timing);
519            return Err(e);
520        }
521
522        // A hub publishes the result of a split in the microframe *after* the one
523        // that carried the start-split, so a complete-split issued in the same
524        // microframe can only ever be answered NYET. The board trace shows exactly
525        // that for the low-speed keyboard on hub ports 3 and 5:
526        //
527        //     split ss  hfnum=0000184A->0000184A->0000184A hcint=0x00000022
528        //     split cs1 hfnum=0000184A->0000184A->0000184A hcint=0x00000042
529        //     split cs2 hfnum=0000184A->0000184A->0000184A hcint=0x00000042
530        //
531        // Both retries were spent inside the start-split's own microframe, so both
532        // came back NYET and enumeration gave up with the keyboard's descriptor
533        // still sitting in the hub. Each retry here waits for the counter to move
534        // first, then uses the tight non-yielding wait: yielding would hand the CPU
535        // away for ~20 ms and put the pair tens of frames past the window U-Boot
536        // bounds a split by -- it abandons one whose complete-split is still NYET
537        // more than four raw `HFNUM` ticks after its start-split.
538        let started = self.frame_number();
539        let mut issued_in = started;
540        for attempt in 0..SPLIT_COMPLETE_ATTEMPTS {
541            if !self.wait_microframe(issued_in, MICROFRAME_POLLS) {
542                break;
543            }
544            issued_in = self.frame_number();
545            if !split_window_open(started, issued_in) {
546                break;
547            }
548            arm(true);
549            timing.attempts = attempt + 1;
550            let after_arm = self.frame_number();
551            let split_reg = self.read32(hcsplt(ch));
552            let outcome = self.wait_channel_spin(ch, SPIN_POLLS);
553            if timing.passes < timing.samples.len() {
554                timing.samples[timing.passes] = [
555                    issued_in,
556                    after_arm,
557                    self.frame_number(),
558                    self.last_hcint.get(),
559                    self.read32(hcchar(ch)),
560                    split_reg,
561                ];
562                timing.passes += 1;
563            }
564            match outcome {
565                Ok(()) if self.last_reported_complete() => return Ok(()),
566                Ok(()) | Err(ViError::WouldBlock) if self.last_was_nyet() => {}
567                Err(e) => {
568                    self.last_split_timing.set(timing);
569                    return Err(e);
570                }
571                Ok(()) => {
572                    self.last_split_timing.set(timing);
573                    return Err(ViError::IO);
574                }
575            }
576        }
577        self.last_split_timing.set(timing);
578        Err(ViError::IO)
579    }

615    /// Guest-physical base of `ch`'s DMA slot, or `None` outside DMA mode.
616    ///
617    /// In SAS grant pages are identity-mapped, so the value programmed into
618    /// `HCDMA` is the same address the CPU uses.
619    fn dma_slot(&self, ch: usize) -> Option<usize> {
620        if self.mode.get() != TransferMode::Dma || ch >= DMA_CHANNELS {
621            return None;
622        }
623        let guard = self.dma.borrow();
624        guard.as_ref().map(|b| b.phys() + ch * DMA_SLOT_BYTES)
625    }

627    /// Bus address of a RAM physical address on this SoC.
628    ///
629    /// A bus master does not see ARM physical addresses. On the BCM283x the
630    /// Raspberry Pi device tree's `_DMA` resource states the translation
631    /// outright: "Bus 0xC0000000 -> CPU 0x00000000", over the first gigabyte
632    /// and marked NonCacheable. U-Boot's working dwc2 driver programs `HCDMA`
633    /// through exactly this translation (`phys_to_bus`).
634    ///
635    /// Programming the raw ARM physical address asks the core to read and write
636    /// a different location, and the failure is quiet: the address still lands
637    /// in mapped memory, so no AHB error is raised, and the CPU-side read-back
638    /// of the slot looks correct because the CPU and the core are simply
639    /// looking at different bytes.
640    #[inline]
641    fn bus_address(phys: usize) -> u32 {
642        (phys | 0xC000_0000) as u32
643    }

645    /// Point the channel's DMA engine at `offset` bytes into its slot.
646    ///
647    /// Programmed **once per transfer, and always before `CHENA`**. The DWC2
648    /// advances `HCDMA` itself as it moves each packet (`hcdma += actual`), so a
649    /// multi-packet transfer walks the slot on its own; re-arming mid-transfer
650    /// both defeats that walk and races the core's asynchronous completion.
651    ///
652    /// The ordering against `CHENA` is not a preference. The core latches its
653    /// DMA address when the channel is enabled, so a start that arms `HCDMA`
654    /// afterwards -- or never -- sends the first transaction of the transfer to
655    /// whatever address the previous one left behind. On a SETUP that is a
656    /// request the device ACKs and then STALLs one stage later, and the retry
657    /// succeeds because by then the register holds the right address.
658    fn program_hcdma(&self, ch: usize, offset: usize) {
659        if let Some(base) = self.dma_slot(ch) {
660            let addr = base + offset.min(DMA_SLOT_BYTES);
661            self.write32(hcdma(ch), Self::bus_address(addr));
662        }
663    }

665    /// Publish CPU writes in `[offset, offset+len)` to the device.
666    ///
667    /// A refused sync is reported, never skipped: without the clean the core
668    /// reads stale memory, so the device receives a request whose bytes are not
669    /// the ones staged and answers with a STALL -- a failure that looks like a
670    /// protocol problem and is actually a missing cache operation.
671    fn cache_clean(&self, ch: usize, offset: usize, len: usize) {
672        let guard = self.dma.borrow();
673        if let Some(buf) = guard.as_ref() {
674            let start = ch * DMA_SLOT_BYTES + offset.min(DMA_SLOT_BYTES);
675            let end = (start + len).min((ch + 1) * DMA_SLOT_BYTES);
676            let span = end.saturating_sub(start);
677            if span == 0 {
678                // Nothing to publish. The kernel rejects a zero-length sync as
679                // an invalid range, so asking for one turns a no-op into a
680                // reported failure -- and a transfer that legitimately moves no
681                // bytes (an idle bulk-IN poll) would warn on every poll.
682                return;
683            }
684            match buf.begin_cache_sync(start, span) {
685                Some(token) => {
686                    if !buf.complete_cache_sync(token) {
687                        ostd::io::println("[dwc2] WARN: cache sync completion refused");
688                    }
689                }
690                None => {
691                    ostd::io::println(
692                        "[dwc2] WARN: cache sync refused - device may read stale memory",
693                    );
694                }
695            }
696        }
697    }

704    /// Copy an outgoing payload into the channel slot at `offset` and publish it.
705    ///
706    /// This is one of the crate's two `allow(unsafe_code)` islands (see
707    /// `scripts/unsafe-allowlist.toml`): the destination is a grant region this
708    /// cell owns for its whole lifetime, `n` is bounded by the slot size and the
709    /// source length, and the region is not aliased by any other live reference.
710    #[allow(unsafe_code)]
711    fn stage_out(&self, ch: usize, offset: usize, data: &[u8]) {
712        let guard = self.dma.borrow();
713        let n = data.len().min(DMA_SLOT_BYTES - offset.min(DMA_SLOT_BYTES));
714        if let Some(buf) = guard.as_ref() {
715            // SAFETY: destination is this cell's grant slot at a bounded
716            // offset; `n` is bounded by the remaining slot and by `data`.
717            unsafe {
718                core::ptr::copy_nonoverlapping(
719                    data.as_ptr(),
720                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset),
721                    n,
722                );
723            }
724        }
725        drop(guard);
726        self.cache_clean(ch, offset, n);
727    }

729    /// Copy a received payload out of the channel slot at `offset`.
730    ///
731    /// The second `allow(unsafe_code)` island: same owned-grant argument as
732    /// [`Self::stage_out`], with `n` additionally bounded by the destination.
733    #[allow(unsafe_code)]
734    fn collect_in(&self, ch: usize, offset: usize, dst: &mut [u8], len: usize) {
735        self.cache_invalidate(ch, offset, len);
736        let guard = self.dma.borrow();
737        if let Some(buf) = guard.as_ref() {
738            let n = len
739                .min(dst.len())
740                .min(DMA_SLOT_BYTES.saturating_sub(offset.min(DMA_SLOT_BYTES)));
741            // SAFETY: source is this cell's grant slot at a bounded offset;
742            // `n` is bounded by the slot remainder and the destination.
743            unsafe {
744                core::ptr::copy_nonoverlapping(
745                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset) as *const u8,
746                    dst.as_mut_ptr(),
747                    n,
748                );
749            }
750        }
751    }

775    /// Execute a standard 8-byte USB SETUP packet on Channel 0.
776    pub fn send_setup(&self, dev_addr: u8, setup: &[u8; 8]) -> ViResult<()> {
777        let ch = 0;
778        self.prepare_channel(ch);
779        self.write32(hcintmsk(ch), 0x07FF);
780
781        // HCTSIZ: XFERSIZE = 8, PKTCNT = 1, PID = 3 (SETUP).
782        // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1.
783        let sctsiz = 8 | (1 << 19) | (3 << 29);
784        let scchar = (self.control_mps() as u32)
785            | self.device_flags()
786            | (1 << 20)
787            | ((dev_addr as u32) << 22)
788            | (1 << 31);
789        let dma = self.dma_slot(ch).is_some();
790
791        // The payload is staged and HCDMA armed *before* CHENA on every pass:
792        // the core latches its DMA address when the channel starts, so enabling
793        // first reads the previous transfer's address and the device ACKs eight
794        // bytes it never looked at.
795        self.run_packet(ch, |complete| {
796            self.write32(hcsplt(ch), self.hcsplt_value(complete));
797            self.write32(hctsiz(ch), sctsiz);
798            if dma {
799                self.program_hcdma(ch, 0);
800                self.stage_out(ch, 0, setup);
801            } else {
802                self.write_fifo(
803                    ch,
804                    u32::from_le_bytes([setup[0], setup[1], setup[2], setup[3]]),
805                );
806                self.write_fifo(
807                    ch,
808                    u32::from_le_bytes([setup[4], setup[5], setup[6], setup[7]]),
809                );
810            }
811            self.write32(hcint(ch), 0xFFFF_FFFF);
812            self.write32(hcchar(ch), self.start_hcchar(scchar));
813        })
814    }

816    /// Receive the DATA IN phase of a control transfer on Channel 0.
817    ///
818    /// Returns the bytes actually received. Packets are sized by the control
819    /// endpoint's MPS and the transfer ends at the first **short** packet
820    /// (USB 2.0 §8.5.3.1) — a device answering an 18-byte descriptor request
821    /// with its real 8-byte report must not be read as if it had 18 bytes
822    /// available, which is what a fixed-size read would do.
823    pub fn recv_data(&self, dev_addr: u8, buf: &mut [u8]) -> ViResult<usize> {
824        let ch = 0;
825        let mps = (self.control_mps() as usize).max(1);
826        let mut received = 0;
827        let mut toggle = 2u32; // PID 2 = DATA1 for the first data packet
828                               // Arm the DMA engine once; the core advances it per packet.
829        self.program_hcdma(ch, 0);
830
831        while received < buf.len() {
832            let chunk = (buf.len() - received).min(mps);
833            self.prepare_channel(ch);
834            self.write32(hcintmsk(ch), 0x07FF);
835
836            // HCCHAR: EPNUM = 0, EPDIR = 1 (IN), EPTYPE = 0 (Control), MC = 1.
837            let sctsiz = (chunk as u32) | (1 << 19) | (toggle << 29);
838            let scchar = (mps as u32)
839                | self.device_flags()
840                | (1 << 15)
841                | (1 << 20)
842                | ((dev_addr as u32) << 22)
843                | (1 << 31);
844
845            self.run_packet(ch, |complete| {
846                self.write32(hcsplt(ch), self.hcsplt_value(complete));
847                self.write32(hctsiz(ch), sctsiz);
848                // Directly, the address is armed once before the loop and the
849                // core walks the slot itself, so re-arming per packet would
850                // overwrite what it has already written. Behind a hub the core
851                // moves a single packet per pass, so the address is re-armed
852                // every pass and advanced by what has arrived so far.
853                if complete || self.split.get().is_some() {
854                    self.program_hcdma(ch, received);
855                }
856                self.write32(hcint(ch), 0xFFFF_FFFF);
857                self.write32(hcchar(ch), self.start_hcchar(scchar));
858            })?;
859
860            // The core writes the *remaining* count back into HCTSIZ.
861            let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
862            let got = chunk.saturating_sub(remaining).min(chunk);
863
864            let words = got.div_ceil(4);
865            for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
866                for (j, byte) in word.to_le_bytes().iter().enumerate() {
867                    let offset = i * 4 + j;
868                    let idx = received + offset;
869                    if idx < buf.len() && offset < got {
870                        buf[idx] = *byte;
871                    }
872                }
873            }
874
875            received += got;
876            if got < chunk {
877                break;
878            }
879            toggle = if toggle == 2 { 0 } else { 2 }; // DATA1 <-> DATA0
880        }
881
882        if self.dma_slot(ch).is_some() && received > 0 {
883            let mut tmp = [0u8; DMA_SLOT_BYTES];
884            let n = received.min(buf.len()).min(DMA_SLOT_BYTES);
885            self.collect_in(ch, 0, &mut tmp, n);
886            buf[..n].copy_from_slice(&tmp[..n]);
887        }
888
889        Ok(received)
890    }

892    /// Send STATUS handshake on Channel 0 (0-byte packet with DATA1).
893    pub fn send_status(&self, dev_addr: u8, is_in: bool) -> ViResult<()> {
894        let ch = 0;
895        self.prepare_channel(ch);
896        self.write32(hcsplt(ch), 0);
897        self.write32(hcintmsk(ch), 0x07FF);
898        // XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1)
899        // HCTSIZ: XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1).
900        // HCCHAR: EPNUM = 0, EPTYPE = 0 (Control), MPS = 64, MC = 1, EPDIR from
901        // the caller.
902        let sctsiz = (1 << 19) | (2 << 29);
903        let epdir = if is_in { 1 } else { 0 };
904        let scchar = (self.control_mps() as u32)
905            | self.device_flags()
906            | (epdir << 15)
907            | (1 << 20)
908            | ((dev_addr as u32) << 22)
909            | (1 << 31);
910
911        self.run_packet(ch, |complete| {
912            self.write32(hcsplt(ch), self.hcsplt_value(complete));
913            self.write32(hctsiz(ch), sctsiz);
914            // A zero-length handshake moves no bytes, but the channel still
915            // latches an address when it starts and the previous transfer's is
916            // not a position this stage may begin from.
917            self.program_hcdma(ch, 0);
918            self.write32(hcint(ch), 0xFFFF_FFFF);
919            self.write32(hcchar(ch), self.start_hcchar(scchar));
920        })
921    }

923    /// Execute a complete synchronous USB Control Transfer.
924    pub fn control_transfer(
925        &self,
926        dev_addr: u8,
927        req_type: u8,
928        req: u8,
929        val: u16,
930        idx: u16,
931        buf: &mut [u8],
932    ) -> ViResult<usize> {
933        let is_in = (req_type & 0x80) != 0;
934        let length = buf.len() as u16;
935
936        let setup = [
937            req_type,
938            req,
939            (val & 0xFF) as u8,
940            ((val >> 8) & 0xFF) as u8,
941            (idx & 0xFF) as u8,
942            ((idx >> 8) & 0xFF) as u8,
943            (length & 0xFF) as u8,
944            ((length >> 8) & 0xFF) as u8,
945        ];
946
947        // Each phase reports its own failure: a control transfer that fails
948        // silently is the hardest kind of USB bug to find, and the request
949        // fields are the only thing that identifies which one it was.
950        if let Err(e) = self.send_setup(dev_addr, &setup) {
951            self.report_failure("setup", dev_addr, req_type, req, val, idx, length, e);
952            return Err(e);
953        }
954
955        // 2. DATA phase (optional)
956        let mut actual = 0;
957        if length > 0 {
958            let result = if is_in {
959                self.recv_data(dev_addr, buf)
960            } else {
961                self.send_data(dev_addr, buf).map(|()| buf.len())
962            };
963            match result {
964                Ok(n) => actual = n,
965                Err(e) => {
966                    self.report_failure("data", dev_addr, req_type, req, val, idx, length, e);
967                    return Err(e);
968                }
969            }
970        }
971
972        // 3. STATUS phase (handshake in opposite direction)
973        if let Err(e) = self.send_status(dev_addr, !is_in) {
974            self.report_failure("status", dev_addr, req_type, req, val, idx, length, e);
975            return Err(e);
976        }
977
978        Ok(actual)
979    }

1071    /// Send DATA OUT phase on Channel 0.
1072    fn send_data(&self, dev_addr: u8, data: &[u8]) -> ViResult<()> {
1073        let ch = 0;
1074        let mps = (self.control_mps() as usize).max(1);
1075        let mut sent = 0;
1076        let mut toggle = 2; // PID 2 = DATA1
1077                            // Stage the whole payload once and arm DMA once: the core reads it
1078                            // packet by packet, advancing HCDMA itself.
1079        if self.dma_slot(ch).is_some() {
1080            self.stage_out(ch, 0, data);
1081            self.program_hcdma(ch, 0);
1082        }
1083
1084        while sent < data.len() {
1085            let chunk = (data.len() - sent).min(mps);
1086            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
1087
1088            // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
1089            let scchar = (mps as u32)
1090                | self.device_flags()
1091                | (1 << 20)
1092                | ((dev_addr as u32) << 22)
1093                | (1 << 31);
1094
1095            // Through `run_packet`, like the setup and status stages: a data
1096            // stage that programs `HCSPLT` itself is only right for a device on
1097            // the root port. Behind a hub the channel would address the
1098            // low-speed device on the high-speed bus, which nothing answers --
1099            // the request reaches the keyboard as XACTERR and its LEDs never
1100            // move. Staging the payload on both passes matches what U-Boot does
1101            // for an OUT split: the start-split delivers the bytes and the
1102            // complete-split collects the handshake.
1103            self.run_packet(ch, |complete| {
1104                self.write32(hcsplt(ch), self.hcsplt_value(complete));
1105                self.write32(hcintmsk(ch), 0x07FF);
1106                self.write32(hctsiz(ch), sctsiz);
1107
1108                if self.dma_slot(ch).is_some() {
1109                    // Payload already staged and HCDMA already armed.
1110                } else {
1111                    let words = chunk.div_ceil(4);
1112                    for i in 0..words {
1113                        let mut b = [0u8; 4];
1114                        for (j, byte) in b.iter_mut().enumerate() {
1115                            let offset = i * 4 + j;
1116                            if sent + offset < data.len() && offset < chunk {
1117                                *byte = data[sent + offset];
1118                            }
1119                        }
1120                        self.write_fifo(ch, u32::from_le_bytes(b));
1121                    }
1122                }
1123
1124                self.write32(hcint(ch), 0xFFFF_FFFF);
1125                self.write32(hcchar(ch), self.start_hcchar(scchar));
1126            })?;
1127
1128            sent += chunk;
1129            toggle = if toggle == 2 { 0 } else { 2 };
1130        }
1131
1132        Ok(())
1133    }

1135    /// Transmit a raw Ethernet packet via Bulk OUT (Channel 2, EP 2).
1136    pub fn bulk_transmit(&self, dev_addr: u8, ep_num: u8, packet: &[u8]) -> ViResult<()> {
1137        let ch = 2;
1138        if self.port_refuses() {
1139            return Err(ViError::IO);
1140        }
1141        let mut sent = 0;
1142        let mut halt_cleared = false;
1143        let mut toggle_flipped = false;
1144
1145        let use_dma = self.dma_slot(ch).is_some();
1146        if use_dma {
1147            self.stage_out(ch, 0, packet);
1148        }
1149        while sent < packet.len() {
1150            let chunk = (packet.len() - sent).min(512); // 512 bytes for High-Speed Bulk
1151
1152            // HCCHAR: EPDIR = 0 (OUT), EPTYPE = 2 (Bulk), MC = 1 packet, MPS = 512 (HS Bulk).
1153            let scchar = 512
1154                | ((ep_num as u32) << 11)
1155                | (2 << 18)
1156                | (1 << 20)
1157                | ((dev_addr as u32) << 22)
1158                | (1 << 31);
1159
1160            let mut retries = 0;
1161            loop {
1162                self.prepare_channel(ch);
1163                self.write32(hcsplt(ch), 0);
1164                self.write32(hcintmsk(ch), 0x07FF);
1165                // The packet's PID comes from the endpoint's own toggle, which
1166                // survives across transfers (see `DataToggles`), and it is read here
1167                // -- inside the retry loop -- so a toggle the recovery paths flip or
1168                // reset is what the next attempt actually arms with. Restarting at
1169                // DATA0 for every frame gave every second and later frame the wrong
1170                // one, and a receiver discards a mismatched packet even when the
1171                // handshake completes.
1172                let sctsiz = (chunk as u32) | (1 << 19) | self.next_pid(dev_addr, ep_num, false);
1173                self.write32(hctsiz(ch), sctsiz);
1174                if use_dma {
1175                    // Point the core at the start of *this* chunk on every attempt,
1176                    // including retries: the core advances `HCDMA` as it fetches each
1177                    // packet into the FIFO, so a retry that only rewrote `HCTSIZ`
1178                    // would fetch the packet after this one as if it were this chunk.
1179                    // U-Boot and Linux re-issue the packet from its start too.
1180                    self.program_hcdma(ch, sent);
1181                }
1182                self.write32(hcchar(ch), self.start_hcchar(scchar));
1183
1184                if !use_dma {
1185                    let words_count = chunk.div_ceil(4);
1186                    for i in 0..words_count {
1187                        let mut b = [0u8; 4];
1188                        for (j, byte) in b.iter_mut().enumerate() {
1189                            let offset = i * 4 + j;
1190                            let idx = sent + offset;
1191                            if idx < packet.len() && offset < chunk {
1192                                *byte = packet[idx];
1193                            }
1194                        }
1195                        self.write_fifo(ch, u32::from_le_bytes(b));
1196                    }
1197                }
1198
1199                match self.wait_channel(ch) {
1200                    Ok(()) => {
1201                        // Adopt the toggle this packet ended on, for the next packet
1202                        // of this frame and for the next frame after it.
1203                        self.adopt_pid(ch, dev_addr, ep_num, false);
1204                        break;
1205                    }
1206                    Err(ViError::WouldBlock) => {
1207                        retries += 1;
1208                        if retries > BULK_NAK_RETRIES {
1209                            static FIRST_GIVE_UP: core::sync::atomic::AtomicBool =
1210                                core::sync::atomic::AtomicBool::new(false);
1211                            if !FIRST_GIVE_UP.swap(true, core::sync::atomic::Ordering::Relaxed) {
1212                                ostd::io::println(
1213                                    "[dwc2] bulk_transmit: the endpoint kept answering NAK",
1214                                );
1215                            }
1216                            // Giving up abandons the packet the core staged in the
1217                            // Tx FIFO, so clear it out like any other abort.
1218                            self.flush_tx_fifo();
1219                            return Err(ViError::IO);
1220                        }
1221                        sys_yield();
1222                    }
1223                    // The device refused the endpoint. A bulk endpoint stays halted
1224                    // until the host clears it (see `clear_endpoint_halt`), so a
1225                    // plain retry would fail identically: clear, re-arm and try the
1226                    // same packet once more.
1227                    Err(error) if self.last_was_stall() && !halt_cleared => {
1228                        halt_cleared = true;
1229                        if !self.clear_endpoint_halt(dev_addr, ep_num, false) {
1230                            return Err(error);
1231                        }
1232                        // Clearing the halt also resets the endpoint's data toggle to
1233                        // DATA0 (USB 2.0 §8.6), so the retry must go back to it or the
1234                        // device sees a toggle mismatch (`DTERR`) instead.
1235                        self.clear_endpoint_toggle(dev_addr, ep_num, false);
1236                        static FIRST_CLEAR: core::sync::atomic::AtomicBool =
1237                            core::sync::atomic::AtomicBool::new(false);
1238                        if !FIRST_CLEAR.swap(true, core::sync::atomic::Ordering::Relaxed) {
1239                            ostd::io::println(
1240                                "[dwc2] cleared a halted bulk-OUT endpoint after STALL",
1241                            );
1242                        }
1243                    }
1244                    // A stale toggle: flip it and re-arm once (see `bulk_receive`).
1245                    Err(error) if self.last_was_dterr() && !toggle_flipped => {
1246                        toggle_flipped = true;
1247                        self.flip_pid(dev_addr, ep_num, false);
1248                        static FIRST_DTERR: core::sync::atomic::AtomicBool =
1249                            core::sync::atomic::AtomicBool::new(false);
1250                        if !FIRST_DTERR.swap(true, core::sync::atomic::Ordering::Relaxed) {
1251                            ostd::io::println("[dwc2] flipped a stale bulk-OUT toggle after DTERR");
1252                        }
1253                        let _ = error;
1254                    }
1255                    Err(e) => return Err(e),
1256                }
1257            }
1258
1259            sent += chunk;
1260        }
1261
1262        Ok(())
1263    }

1396    /// Receive a raw packet via Bulk IN (Channel 1, EP 1). Returns received length or 0 if nothing.
1397    pub fn bulk_receive(&self, dev_addr: u8, ep_num: u8, buf: &mut [u8]) -> ViResult<usize> {
1398        let ch = 1;
1399        if self.port_refuses() {
1400            return Err(ViError::IO);
1401        }
1402        let want = buf.len().min(512);
1403        // HCCHAR: EPDIR = 1 (IN), EPTYPE = 2 (Bulk), MC = 1 packet, MPS = 512 (HS Bulk).
1404        let scchar = 512
1405            | ((ep_num as u32) << 11)
1406            | (1 << 15) // IN
1407            | (2 << 18) // Bulk
1408            | (1 << 20) // MC = 1
1409            | ((dev_addr as u32) << 22)
1410            | (1 << 31);
1411
1412        let mut halt_cleared = false;
1413        let mut toggle_flipped = false;
1414        loop {
1415            self.prepare_channel(ch);
1416            self.write32(hcsplt(ch), 0);
1417            self.write32(hcintmsk(ch), 0x07FF);
1418            // PID from the endpoint's own toggle, not a constant DATA0: the toggle
1419            // belongs to the endpoint and survives across transfers (see
1420            // `DataToggles`), and a stale one answers every later packet with
1421            // `DTERR`.
1422            let sctsiz = (want as u32) | (1 << 19) | self.next_pid(dev_addr, ep_num, true);
1423            self.write32(hctsiz(ch), sctsiz);
1424            self.program_hcdma(ch, 0);
1425            self.write32(hcchar(ch), self.start_hcchar(scchar));
1426
1427            // A poll, not a wait: the device answers NAK immediately when it has
1428            // nothing, so a long budget here only ever happens when the core reports
1429            // nothing at all — and the old 1000 yields is twenty seconds of the
1430            // serving loop that the Net Cell's request deadline is sitting inside.
1431            // Same two units as `wait_channel`: host microframes and yields.
1432            let started = self.frame_number();
1433            let mut yields = 0;
1434            let outcome = loop {
1435                let int = self.read32(hcint(ch));
1436                if int & (1 << 0) != 0 {
1437                    // The core reports the actual length in HCTSIZ either way, but
1438                    // only DMA deposits the payload outside the FIFO. Read the
1439                    // endpoint's new toggle out of the same register first.
1440                    self.adopt_pid(ch, dev_addr, ep_num, true);
1441                    let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1442                    let got = want.saturating_sub(remaining);
1443                    if self.dma_slot(ch).is_some() {
1444                        self.collect_in(ch, 0, buf, got);
1445                    } else {
1446                        let words = want.div_ceil(4);
1447                        for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
1448                            for (j, byte) in word.to_le_bytes().iter().enumerate() {
1449                                let idx = i * 4 + j;
1450                                if idx < buf.len() && idx < want {
1451                                    buf[idx] = *byte;
1452                                }
1453                            }
1454                        }
1455                    }
1456                    break Ok(got);
1457                }
1458                if int & (1 << 4) != 0 {
1459                    // NAK: device has no packet right now.
1460                    break Ok(0);
1461                }
1462                // Error conditions the old read loop ignored (only XFRC and NAK were
1463                // tested): a refused or broken transfer was reported as "no frame",
1464                // which hid a halted endpoint behind an ordinary idle poll and made
1465                // every later poll fail the same silent way.
1466                if int & ((1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
1467                    self.last_hcint.set(int);
1468                    self.halt_channel(ch);
1469                    break Err(ViError::IO);
1470                }
1471                if yields >= WAIT_YIELDS
1472                    || self.frame_number().wrapping_sub(started) & HFNUM_FRNUM_MASK
1473                        >= WAIT_MICROFRAMES
1474                {
1475                    break Ok(0);
1476                }
1477                yields += 1;
1478                sys_yield();
1479            };
1480
1481            match outcome {
1482                Ok(n) => return Ok(n),
1483                // The device refused this endpoint. Clear the halt and re-arm once:
1484                // until the halt is cleared, no later transfer can succeed.
1485                Err(error) if self.last_was_stall() && !halt_cleared => {
1486                    halt_cleared = true;
1487                    if !self.clear_endpoint_halt(dev_addr, ep_num, true) {
1488                        return Err(error);
1489                    }
1490                    static FIRST_CLEAR: core::sync::atomic::AtomicBool =
1491                        core::sync::atomic::AtomicBool::new(false);
1492                    if !FIRST_CLEAR.swap(true, core::sync::atomic::Ordering::Relaxed) {
1493                        ostd::io::println("[dwc2] cleared a halted bulk-IN endpoint after STALL");
1494                    }
1495                }
1496                // A stale toggle: flip it and re-arm once, or the same packet fails
1497                // with `DTERR` for as long as the device and the table disagree.
1498                Err(error) if self.last_was_dterr() && !toggle_flipped => {
1499                    toggle_flipped = true;
1500                    self.flip_pid(dev_addr, ep_num, true);
1501                    static FIRST_DTERR: core::sync::atomic::AtomicBool =
1502                        core::sync::atomic::AtomicBool::new(false);
1503                    if !FIRST_DTERR.swap(true, core::sync::atomic::Ordering::Relaxed) {
1504                        ostd::io::println("[dwc2] flipped a stale bulk-IN toggle after DTERR");
1505                    }
1506                    let _ = error;
1507                }
1508                Err(error) => return Err(error),
1509            }
1510        }
1511    }

1513    /// Receive up to one packet via Interrupt IN on `ch`.
1514    ///
1515    /// HID input reports ride interrupt endpoints. The device NAKs until it has
1516    /// a report, so a NAK is a normal "no data yet" answer, not an error — the
1517    /// caller polls. Returns the *actual* transferred length, read back from
1518    /// `HCTSIZ` rather than assumed: a HID report can be shorter than the buffer
1519    /// (an 8-byte keyboard report into a 64-byte buffer), and using the request
1520    /// length would hand the decoder trailing garbage.
1521    ///
1522    /// `ch` must not collide with the control (0) or Ethernet bulk (1, 2)
1523    /// channels.
1524    pub fn interrupt_receive(
1525        &self,
1526        ch: usize,
1527        dev_addr: u8,
1528        ep_num: u8,
1529        mps: u16,
1530        buf: &mut [u8],
1531        split_pending: &mut bool,
1532    ) -> ViResult<usize> {
1533        let want = buf.len().min(mps.max(1) as usize).min(1024);
1534        if want == 0 {
1535            return Ok(0);
1536        }
1537
1538        // Behind a hub the poll is a pair of transactions, one half per call.
1539        if self.split.get().is_some() {
1540            return self.poll_split(ch, dev_addr, ep_num, mps, want, buf, split_pending);
1541        }
1542
1543        self.prepare_channel(ch);
1544        self.write32(hcsplt(ch), 0);
1545        self.write32(hcintmsk(ch), 0x07FF);
1546
1547        // HCTSIZ: XFERSIZE (= want), PKTCNT = 1, and the PID this endpoint's
1548        // toggle says (DATA0 only for the first report; see `DataToggles`). The core
1549        // overwrites XFERSIZE with the remaining count as it transfers.
1550        let sctsiz = (want as u32) | (1 << 19) | self.next_pid(dev_addr, ep_num, true);
1551        self.write32(hctsiz(ch), sctsiz);
1552
1553        // HCCHAR: MPS from the endpoint descriptor, EPNUM, IN direction, DEVCTL,
1554        // and EPTYPE = Interrupt, which is what the endpoint is.
1555        //
1556        // EPTYPE was briefly Bulk here, to move the transfer out of the core's
1557        // periodic class while it was thought the missing periodic frame list was
1558        // the problem. It is not: a low-speed device has no bulk endpoints at all.
1559        // USB 2.0 gives low speed control and interrupt transfers and nothing else,
1560        // and the hub's translator refuses a bulk transaction aimed at one -- which
1561        // the trace shows as a STALL on the very first poll of both endpoints, long
1562        // before the device has had any reason to refuse anything.
1563        let scchar = (mps as u32 & 0x7FF)
1564            | ((ep_num as u32) << 11)
1565            | self.device_flags()
1566            | (1 << 15) // EPDIR = IN
1567            | (3 << 18) // EPTYPE = Interrupt: the only type a low-speed endpoint has
1568            | (1 << 20) // MC = 1
1569            | ((dev_addr as u32) << 22)
1570            | (1 << 31); // CHENA
1571        self.program_hcdma(ch, 0);
1572        self.write32(hcchar(ch), self.start_hcchar(scchar));
1573
1574        let mut polls = 0u32;
1575        while polls < 2_000 {
1576            let int = self.read32(hcint(ch));
1577
1578            // NAK — device has no report queued. Halt and report "no data".
1579            if int & (1 << 4) != 0 {
1580                self.halt_channel(ch);
1581                return Ok(0);
1582            }
1583            // Errors (STALL / babble / transaction error / toggle mismatch) end the
1584            // poll cycle. `DTERR` used to be missing here, so a mismatched report
1585            // simply ended the poll with "no data" and the device's toggle and ours
1586            // stayed out of step.
1587            if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 10)) != 0 {
1588                if int & (1 << 10) != 0 {
1589                    self.flip_pid(dev_addr, ep_num, true);
1590                }
1591                self.halt_channel(ch);
1592                return Err(ViError::IO);
1593            }
1594
1595            let complete = int & (1 << 0) != 0 || int & (1 << 1) != 0 || int & (1 << 5) != 0;
1596            if complete {
1597                // The report that just arrived advanced this endpoint's toggle.
1598                self.adopt_pid(ch, dev_addr, ep_num, true);
1599                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1600                let got = want.saturating_sub(remaining);
1601                self.halt_channel(ch);
1602
1603                if got > 0 {
1604                    if self.dma_slot(ch).is_some() {
1605                        // DMA deposited the report in the channel slot.
1606                        let mut tmp = [0u8; DMA_SLOT_BYTES];
1607                        self.collect_in(ch, 0, &mut tmp, got);
1608                        let n = got.min(buf.len());
1609                        buf[..n].copy_from_slice(&tmp[..n]);
1610                    } else {
1611                        // Drain ceil(got/4) FIFO words; only the first `got`
1612                        // bytes are meaningful (the tail word is padding).
1613                        let words = got.div_ceil(4);
1614                        for w in 0..words {
1615                            let word = self.read_fifo(ch);
1616                            for (j, byte) in word.to_le_bytes().iter().enumerate() {
1617                                let idx = w * 4 + j;
1618                                if idx < got && idx < buf.len() {
1619                                    buf[idx] = *byte;
1620                                }
1621                            }
1622                        }
1623                    }
1624                } else if self.dma_slot(ch).is_none() {
1625                    // A zero-length packet still has a FIFO word to retire on
1626                    // some revisions; drain one so the channel is clean.
1627                    let _ = self.read_fifo(ch);
1628                }
1629                return Ok(got);
1630            }
1631
1632            polls += 1;
1633            sys_yield();
1634        }
1635
1636        self.halt_channel(ch);
1637        Ok(0)
1638    }

1640    /// Poll one interrupt-IN packet through a high-speed hub's translator.
1641    ///
1642    /// DWC2 runs the start- and complete-splits as separate host-channel
1643    /// transactions. In DMA mode each transaction is complete only once CHHLTD
1644    /// arrives. After an accepted start-split the complete-split is armed
1645    /// immediately; ODDFRM schedules it into the following microframe. A NYET is
1646    /// retried while the original start-split remains within U-Boot's
1647    /// board-proven four-microframe budget.
1648    ///
1649    /// The three ways a complete-split can end mean different things:
1650    ///
1651    /// * `XFERCOMPL` is the report, and ends the pairing;
1652    /// * `NYET` is the hub still working, so the pairing stays live;
1653    /// * `NAK` ends the pairing with nothing.
1654    ///
1655    /// ACK carries no data here -- Linux is explicit that it "should not occur in
1656    /// CSPLIT" -- so it is not taken for a result.
1657    #[allow(clippy::too_many_arguments)]
1658    fn poll_split(
1659        &self,
1660        ch: usize,
1661        dev_addr: u8,
1662        ep_num: u8,
1663        mps: u16,
1664        want: usize,
1665        buf: &mut [u8],
1666        pending: &mut bool,
1667    ) -> ViResult<usize> {
1668        self.prepare_channel(ch);
1669        self.write32(hcintmsk(ch), 0x07FF);
1670
1671        // HCTSIZ: XFERSIZE = one max packet, PKTCNT = 1, and the PID this
1672        // endpoint's toggle says (see `DataToggles`). A split carries exactly one
1673        // packet, whatever the caller asked for, so the same value is used for both
1674        // halves of the pair and adopted only once the data has arrived.
1675        //
1676        // HCCHAR: MPS, EPNUM, IN, and EPTYPE = Interrupt, for the same reason as
1677        // the channel above: a low-speed endpoint has no other type.
1678        let sctsiz = (want as u32) | (1 << 19) | self.next_pid(dev_addr, ep_num, true);
1679        let scchar = (mps as u32 & 0x7FF)
1680            | ((ep_num as u32) << 11)
1681            | self.device_flags()
1682            | (1 << 15)
1683            | (3 << 18)
1684            | (3 << 20)
1685            | ((dev_addr as u32) << 22)
1686            | (1 << 31);
1687        let arm = |complete: bool| {
1688            self.write32(hcsplt(ch), self.hcsplt_value(complete));
1689            self.write32(hctsiz(ch), sctsiz);
1690            self.program_hcdma(ch, 0);
1691            self.write32(hcint(ch), 0xFFFF_FFFF);
1692            self.write32(hcchar(ch), self.start_hcchar(scchar));
1693        };
1694
1695        // Match U-Boot's working channel loop on this controller: complete the
1696        // start-split channel, then arm the complete-split immediately. The old
1697        // path forced SSPLIT into microframe 0 and waited until microframe 5 for
1698        // CSPLIT. That schedule is the reverse of USB interrupt-IN split
1699        // scheduling and left only the frame boundary for retries.
1700
1701        arm(false);
1702        let outcome = self.wait_channel_spin(ch, SPLIT_SPIN_POLLS);
1703        if !matches!(outcome, Ok(())) {
1704            *pending = false;
1705            return match outcome {
1706                Err(e) => Err(e),
1707                _ => Ok(0),
1708            };
1709        }
1710
1711        let started = self.frame_number();
1712        for _ in 0..SPLIT_ATTEMPTS {
1713            arm(true);
1714            let outcome = self.wait_channel_spin(ch, SPLIT_SPIN_POLLS);
1715
1716            match outcome {
1717                Ok(()) if !self.last_reported_complete() => {
1718                    // ACK on a complete-split carries no data.
1719                    *pending = false;
1720                    return Ok(0);
1721                }
1722                Ok(()) => {
1723                    *pending = false;
1724                    // The report arrived: this endpoint's toggle has advanced to
1725                    // what the next poll must ask for.
1726                    self.adopt_pid(ch, dev_addr, ep_num, true);
1727                    let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1728                    let got = want.saturating_sub(remaining);
1729                    if got == 0 {
1730                        return Ok(0);
1731                    }
1732                    if self.dma_slot(ch).is_some() {
1733                        let mut tmp = [0u8; DMA_SLOT_BYTES];
1734                        self.collect_in(ch, 0, &mut tmp, got.min(DMA_SLOT_BYTES));
1735                        let n = got.min(buf.len());
1736                        buf[..n].copy_from_slice(&tmp[..n]);
1737                        return Ok(n);
1738                    }
1739                    let words = got.div_ceil(4);
1740                    for w in 0..words {
1741                        let word = self.read_fifo(ch);
1742                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
1743                            let idx = w * 4 + j;
1744                            if idx < got && idx < buf.len() {
1745                                buf[idx] = *byte;
1746                            }
1747                        }
1748                    }
1749                    return Ok(got.min(buf.len()));
1750                }
1751                Err(ViError::WouldBlock) => {
1752                    if !self.last_was_nyet() {
1753                        *pending = false;
1754                        return Ok(0);
1755                    }
1756                    if !split_window_open(started, self.frame_number()) {
1757                        *pending = false;
1758                        return Ok(0);
1759                    }
1760                }
1761                Err(e) => {
1762                    *pending = false;
1763                    return Err(e);
1764                }
1765            }
1766        }
1767
1768        // The hub never finished. Starting over on the next poll is the only
1769        // thing left: the frame this pairing belonged to is gone with it.
1770        *pending = false;
1771        Ok(0)
1772    }

1774    /// Name the host-channel interrupt bit that ended a transfer.
1775    ///
1776    /// The DWC2 encodes one cause per bit and they need different fixes:
1777    /// `AHBERR` is the core failing to reach memory at `HCDMA`, `XACTERR` is a
1778    /// bus protocol error, `BBLERR` is the device sending more than the
1779    /// programmed packet size, `DTERR` is a data-toggle mismatch, and `STALL`
1780    /// is the device itself refusing. Collapsing them into one "IO error" hides
1781    /// which of those actually happened.
1782    fn describe_hcint(&self, int: u32) -> &'static str {
1783        if int & (1 << 2) != 0 {
1784            return "AHBERR - core could not reach memory at HCDMA";
1785        }
1786        if int & (1 << 8) != 0 {
1787            return "BBLERR - babble, device exceeded the packet size";
1788        }
1789        if int & (1 << 3) != 0 {
1790            return "STALL - endpoint refused the request";
1791        }
1792        if int & (1 << 7) != 0 {
1793            return "XACTERR - transaction error";
1794        }
1795        if int & (1 << 10) != 0 {
1796            return "DTERR - data toggle mismatch";
1797        }
1798        if int & (1 << 9) != 0 {
1799            return "FRMOVRN - frame overrun";
1800        }
1801        if int & (1 << 6) != 0 {
1802            return "NYET - not ready";
1803        }
1804        if int & (1 << 4) != 0 {
1805            return "NAK";
1806        }
1807        "unknown"
1808    }

1819    /// Dump the channel and core registers that explain a stalled transfer.
1820    ///
1821    /// A timeout with no status bit set (`HCINT == 0`) is ambiguous from the
1822    /// outside: the channel may never have been issued, may have been issued and
1823    /// silently abandoned by the core, or the port may have dropped underneath
1824    /// it. These registers distinguish those cases.
1825    pub fn dump_transfer_state(&self, ch: usize, why: &str) {
1826        ostd::io::print("[dwc2] state at ");
1827        ostd::io::print(why);
1828        ostd::io::println(":");
1829        for (name, offset) in [
1830            ("HCCHAR", hcchar(ch)),
1831            ("HCTSIZ", hctsiz(ch)),
1832            ("HCDMA", hcdma(ch)),
1833            ("HCINTMSK", hcintmsk(ch)),
1834            ("HCSPLT", hcsplt(ch)),
1835            ("HFNUM", HFNUM),
1836        ] {
1837            self.dump_reg(name, self.read32(offset));
1838        }
1839        self.dump_reg("HCINT", self.read32(hcint(ch)));
1840        self.dump_reg("HAINT", self.read32(HAINT));
1841        // `HAINT` names every channel with an unserviced interrupt, and the dump
1842        // above covers only the channel this transfer used. The board's timeouts
1843        // all read `HAINT=0x2` with the dumped channel's own `HCINT` zero, which
1844        // cannot be told apart from a stale aggregate without reading the channel
1845        // that owns the bit.
1846        let haint = self.read32(HAINT);
1847        for pending in 0..DMA_CHANNELS {
1848            if haint & (1 << pending) == 0 {
1849                continue;
1850            }
1851            ostd::io::print("[dwc2]   HAINT ch=");
1852            print_hex_val(pending as u32);
1853            ostd::io::print(" HCCHAR=0x");
1854            print_hex_val(self.read32(hcchar(pending)));
1855            ostd::io::print(" HCINT=0x");
1856            print_hex_val(self.read32(hcint(pending)));
1857            ostd::io::println("");
1858        }
1859        self.dump_reg("GINTSTS", self.read32(GINTSTS));
1860        self.dump_reg("GINTMSK", self.read32(GINTMSK));
1861        self.dump_reg("HPRT0", self.read32(HPRT0));
1862        self.dump_reg("GRSTCTL", self.read32(GRSTCTL));
1863        self.dump_reg("GNPTXSTS", self.read32(GNPTXSTS));
1864        self.dump_reg("MODE", self.mode.get() as u32);
1865    }

1899    /// Wait for channel transfer completion or error with timeout.
1900    ///
1901    /// In DMA mode `CHHLTD` is the completion event. Handshake bits can become
1902    /// visible first, but the channel still owns its registers until the halt
1903    /// arrives; [`Self::channel_outcome`] keeps waiting rather than letting the
1904    /// caller re-arm the channel against that delayed completion.
1905    /// Wait for channel completion or error, bounded by host time rather than by
1906    /// a poll count.
1907    ///
1908    /// This wait yields, and a yield costs the board around twenty milliseconds —
1909    /// so a *count* budget is not a time budget. `WAIT_POLLS` (50 000) was minutes
1910    /// of host time, and the board measured a single driver turn at **404 s**: the
1911    /// driver is out of `Recv` for that whole turn, so the net service's 200 ms
1912    /// offer and the hypervisor's 2 s L2 deadline both expire long before it comes
1913    /// back, and the guest's frame is lost with them. A transfer that has not
1914    /// reported inside [`WAIT_MICROFRAMES`] is stuck; failing it fast is what lets
1915    /// the loop turn again, and every caller above retries.
1916    fn wait_channel(&self, ch: usize) -> ViResult<()> {
1917        let started = self.frame_number();
1918        let mut yields = 0u32;
1919        loop {
1920            if let Some(outcome) = self.channel_outcome(ch) {
1921                return outcome;
1922            }
1923            yields += 1;
1924            if yields >= WAIT_YIELDS
1925                || self.frame_number().wrapping_sub(started) & HFNUM_FRNUM_MASK >= WAIT_MICROFRAMES
1926            {
1927                break;
1928            }
1929            sys_yield();
1930        }
1931        self.channel_timeout(ch)
1932    }

1934    /// Wait for a channel without handing the CPU away.
1935    ///
1936    /// A periodic split has to keep both halves inside one millisecond frame. The
1937    /// yielding wait gives the CPU away for around twenty milliseconds per poll,
1938    /// which put the complete-split twenty frames past the start-split it belongs
1939    /// to -- outside the window the hub pairs them within, so it came back as a
1940    /// transaction error however many times it was tried.
1941    ///
1942    /// The bound is the frame itself rather than a count: past that the pairing is
1943    /// gone and starting over on the next poll is the only useful thing left. This
1944    /// runs only when a device has something to report.
1945    /// The channel is left to report for itself. The frame check this used to
1946    /// carry halted the channel the moment the frame turned over, which is exactly
1947    /// when a full- or low-speed transaction behind a hub is finishing -- and the
1948    /// halt it issued was then read straight back as the outcome, so every split
1949    /// came back as a halt carrying no status:
1950    ///
1951    ///     half=ssplit outcome=NAK hcint=0x00000002
1952    ///     half=csplit outcome=NAK hcint=0x00000002
1953    ///
1954    /// The core halts a periodic channel at its own frame boundary and says so, so
1955    /// nothing here needs to do it, and nothing here should.
1956    fn wait_channel_spin(&self, ch: usize, polls: usize) -> ViResult<()> {
1957        for _ in 0..polls {
1958            if let Some(outcome) = self.channel_outcome(ch) {
1959                return outcome;
1960            }
1961        }
1962        self.channel_timeout(ch)
1963    }

1980    /// Classify a channel's interrupt register, or `None` while it is still running.
1981    fn channel_outcome(&self, ch: usize) -> Option<ViResult<()>> {
1982        let int = self.read32(hcint(ch));
1983        if int == 0 {
1984            return None;
1985        }
1986
1987        // Buffer-DMA completion is reported by CHHLTD. ACK, NAK, NYET, and even
1988        // XFERCOMPL may be visible before it; treating that prefix as the final
1989        // result races the old channel's delayed halt against the next arm. The
1990        // observed sequence was ACK (0x20), NYET (0x40), then a stale bare
1991        // CHHLTD (0x02) immediately after re-arm. U-Boot waits for CHHLTD before
1992        // interpreting the same bits, and Linux masks every DMA channel event
1993        // except CHHLTD and AHBERR for this reason.
1994        if self.mode.get() == TransferMode::Dma && int & ((1 << 1) | (1 << 2)) == 0 {
1995            return None;
1996        }
1997
1998        // ── Hardware-generated halt ───────────────────────────────
1999        if int & (1 << 1) != 0 {
2000            // CHHLTD
2001            self.write32(hcint(ch), 0xFFFF_FFFF);
2002            // Babble (8) and data-toggle error (10) end a transfer just as
2003            // surely as the bits that were already checked.
2004            if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
2005                self.report_channel_error(ch, int);
2006                // The core has halted this channel (CHHLTD), so an OUT transfer's
2007                // half-fetched packet can be cleared out of the Tx FIFO here.
2008                self.flush_tx_fifo_after_abort(ch);
2009                self.last_hcint.set(int);
2010                return Some(Err(ViError::IO));
2011            }
2012            // NAK and NYET both mean "ask again", and both have to be reported
2013            // rather than folded into success: a hub answering a complete-split
2014            // with NYET has not produced a result yet, and returning Ok there
2015            // hands the caller an empty buffer as if the transfer had happened.
2016            if int & ((1 << 4) | (1 << 6)) != 0 {
2017                self.last_hcint.set(int);
2018                return Some(Err(ViError::WouldBlock));
2019            }
2020            // A halt carrying nothing else is not a result. CHHLTD says the channel
2021            // stopped -- at a frame boundary, or because it was disabled -- and
2022            // answering "done" to it hands the caller an empty buffer for a transfer
2023            // that never completed. Linux is explicit that the only completion is
2024            // XFERCOMPL, with ACK belonging to the start-split.
2025            if int & ((1 << 0) | (1 << 5)) != 0 {
2026                self.last_hcint.set(int);
2027                return Some(Ok(()));
2028            }
2029            self.last_hcint.set(int);
2030            return Some(Err(ViError::WouldBlock));
2031        }
2032
2033        // ── Transfer complete ─────────────────────────────────────
2034        if int & (1 << 0) != 0 {
2035            self.halt_channel(ch);
2036            self.last_hcint.set(int);
2037            return Some(Ok(()));
2038        }
2039
2040        // ── ACK = the packet was accepted
2041        //
2042        // No halt here. On a split, an ACK to the start-split means the hub has
2043        // taken the transaction and its translator is running it -- the pair is
2044        // live at that moment and the complete-split is what collects it.
2045        // Disabling the channel in between asks the core to abandon a transfer
2046        // that is still in flight, and the trace shows where that lands: the
2047        // channel is armed for the complete-split with CHDIS already set, and it
2048        // comes back halted with no status at all, on the very first poll, on both
2049        // endpoints, before the device has had any chance to refuse anything. A
2050        // transfer the core has finished needs no halt; a transfer being abandoned
2051        // gets one from its caller.
2052        if int & (1 << 5) != 0 {
2053            // A transfer that is not part of a split is over when the packet is
2054            // accepted, and halting it is the ordinary end of it. A split is a
2055            // different thing: the ACK belongs to the start-split, the hub's
2056            // translator is running the transaction because of it, and the pair is
2057            // live. Disabling the channel in that moment asks the core to abandon
2058            // a transfer that is still in flight, and the trace shows where that
2059            // lands -- the channel armed for the complete-split with CHDIS already
2060            // set, coming back halted with no status at all, on the very first
2061            // poll, on both endpoints, before the device has refused anything.
2062            if self.split.get().is_none() {
2063                self.halt_channel(ch);
2064            }
2065            self.last_hcint.set(int);
2066            return Some(Ok(()));
2067        }
2068
2069        // ── Error conditions ──────────────────────────────────────
2070        if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
2071            {
2072                use core::sync::atomic::Ordering;
2073                let counter = if int & (1 << 7) != 0 {
2074                    &FAULT_XACTERR
2075                } else if int & (1 << 3) != 0 {
2076                    &FAULT_STALL
2077                } else {
2078                    &FAULT_OTHER
2079                };
2080                counter.fetch_add(1, Ordering::Relaxed);
2081            }
2082            self.report_channel_error(ch, int);
2083            self.halt_channel(ch);
2084            self.last_hcint.set(int);
2085            return Some(Err(ViError::IO));
2086        }
2087        if int & ((1 << 4) | (1 << 6)) != 0 {
2088            self.halt_channel(ch);
2089            self.last_hcint.set(int);
2090            return Some(Err(ViError::WouldBlock));
2091        }
2092
2093        None
2094    }

2096    /// Give up on a channel that reported nothing within its budget.
2097    fn channel_timeout(&self, ch: usize) -> ViResult<()> {
2098        // Timeout — capture hcint BEFORE halt clears it
2099        let int = self.read32(hcint(ch));
2100        let char_val = self.read32(hcchar(ch));
2101        // Read the direction before the halt clears EPDIR, then clear the Tx FIFO
2102        // an abandoned OUT transfer may have left a packet in.
2103        let was_out = char_val & HCCHAR_EPDIR == 0;
2104        self.halt_channel(ch);
2105        if was_out {
2106            self.flush_tx_fifo();
2107        }
2108        FAULT_TIMEOUT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
2109        static TIMEOUT_COUNT: core::sync::atomic::AtomicUsize =
2110            core::sync::atomic::AtomicUsize::new(0);
2111        let attempt = TIMEOUT_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
2112        if attempt < 3 {
2113            ostd::io::print("[dwc2] TIMEOUT ch=");
2114            print_hex_val(ch as u32);
2115            ostd::io::print(" hcint=0x");
2116            print_hex_val(int);
2117            ostd::io::print(" hcchar=0x");
2118            print_hex_val(char_val);
2119            ostd::io::println("");
2120            self.dump_transfer_state(ch, "timeout");
2121        }
2122        self.last_hcint.set(int);
2123        Err(ViError::IO)
2124    }

2126    /// Halt an active host channel before it can be reused.
2127    ///
2128    /// DWC2 accepts a halt request as `CHDIS|CHENA` while the channel is
2129    /// active, then clears `CHENA` when stopping has completed. Both U-Boot and
2130    /// Linux use that sequence and wait for `CHENA` to clear. Writing `CHDIS`
2131    /// while clearing `CHENA` is not an active-channel halt request.
2132    ///
2133    /// A channel which has already stopped needs no request: setting `CHENA` in
2134    /// that state can start a spurious transaction. Its pending W1C status is
2135    /// cleared below so it cannot be mistaken for a later transfer's result.
2136    fn halt_channel(&self, ch: usize) {
2137        let reg = hcchar(ch);
2138        let current = self.read32(reg);
2139        if let Some(request) = active_halt_request(current) {
2140            self.write32(reg, request);
2141
2142            // This is deliberately a short non-yielding wait: yielding costs
2143            // multiple USB frames. The next prepare_channel() retries the halt
2144            // if a faulty core leaves CHENA asserted beyond this bound.
2145            for _ in 0..CHANNEL_HALT_POLLS {
2146                if self.read32(reg) & HCCHAR_CHENA == 0 {
2147                    break;
2148                }
2149            }
2150        }
2151        self.write32(hcint(ch), 0xFFFF_FFFF);
2152    }

2155/// Whether a `(phase, cause)` control-transfer failure still needs reporting.
2156///
2157/// Keeps one bit per kind in [`REPORTED_FAILURES`]; the first caller for a kind
2158/// prints it and every later one is silent (see `UsbHostEngine::report_failure`).
2159fn claim_failure_report(phase: &str, error: &ViError, hcint: u32) -> bool {
2160    /// Phases a control transfer is logged with; a channel error uses the next
2161    /// slot so the two do not share bits.
2162    const PHASES: u64 = 4;
2163    /// Causes [`failure_cause`] can return.
2164    const CAUSES: u64 = 12;
2165    let phase_index = match phase {
2166        "setup" => 0,
2167        "data" => 1,
2168        "status" => 2,
2169        _ => 3,
2170    };
2171    let index = (phase_index * CAUSES + failure_cause(error, hcint)).min(PHASES * CAUSES - 1);
2172    let bit = 1u64 << index;
2173    REPORTED_FAILURES.fetch_or(bit, core::sync::atomic::Ordering::Relaxed) & bit == 0
2174}
2175
2176/// Which cause a failure ended in, as a small index.
2177///
2178/// The order follows [`UsbHostEngine::describe_hcint`], so two failures that
2179/// print the same sentence share a report: the `NAK` of an interrupt-IN poll and
2180/// the `XACTERR` of a register read with a low-speed split in flight are different
2181/// kinds and both deserve their first line, but the second of each does not.
2182fn failure_cause(error: &ViError, hcint: u32) -> u64 {
2183    if matches!(error, ViError::WouldBlock) {
2184        // NYET is the hub still working; NAK is the device having nothing to send.
2185        return if hcint & (1 << 6) != 0 { 1 } else { 0 };
2186    }
2187    if !matches!(error, ViError::IO) {
2188        return 2;
2189    }
2190    for (index, bit) in [2u32, 8, 3, 7, 10, 9, 6, 4].into_iter().enumerate() {
2191        if hcint & (1 << bit) != 0 {
2192            return index as u64 + 3;
2193        }
2194    }
2195    // An `IO` with no status bit at all (an abandoned channel, a poll budget).
2196    11
2197}

2199/// Kinds of transfer failure the console has already named, one bit each.
2200///
2201/// See `claim_failure_report`: the bit index is `phase * 12 + cause`, and the last
2202/// phase is the channel-error path, so the whole set fits in one word.
2203static REPORTED_FAILURES: core::sync::atomic::AtomicU64 = core::sync::atomic::AtomicU64::new(0);

2205/// Build a DWC2 halt request only for a channel that is actually active.
2206///
2207/// `CHDIS|CHENA` requests the active-channel halt; a stopped channel must not
2208/// receive `CHENA`, because that would begin a new transaction.
2209#[inline]
2210fn active_halt_request(hcchar: u32) -> Option<u32> {
2211    (hcchar & HCCHAR_CHENA != 0).then_some((hcchar | HCCHAR_CHDIS | HCCHAR_CHENA) & !HCCHAR_EPDIR)
2212}

2214/// Whether a split's pairing window is still open `now` raw `HFNUM` ticks after
2215/// the start-split it belongs to.
2216///
2217/// U-Boot abandons a split whose complete-split is still NYET more than four
2218/// ticks past that point, on the grounds that the hub has dropped the pairing by
2219/// then. The counter is 16 bits and wraps about every eight seconds, so the
2220/// difference has to wrap with it -- a stale start must read as closed, never as
2221/// a fresh window.
2222#[inline]
2223fn split_window_open(started: u32, now: u32) -> bool {
2224    now.wrapping_sub(started) & HFNUM_FRNUM_MASK <= PERIODIC_SPLIT_MICROFRAMES
2225}

2227/// `HCSPLT` for a transfer addressed through `split`, or 0 when there is none.
2228///
2229/// Every transfer to a device behind a hub needs this, not just the periodic
2230/// poll: a channel armed without it addresses the low-speed device on the
2231/// high-speed bus, which nothing answers and the core reports as XACTERR. The
2232/// value is built here rather than read from the engine so a stage that forgets
2233/// to program it can be caught by a test instead of by a keyboard whose LEDs
2234/// never light.
2235#[inline]
2236fn hcsplt_for(split: Option<Split>, complete: bool) -> u32 {
2237    let Some(split) = split else {
2238        return 0;
2239    };
2240    let mut value = HCSPLT_SPLTENA
2241        | ((split.hub_addr as u32 & HCSPLT_HUBADDR_MASK >> HCSPLT_HUBADDR_SHIFT)
2242            << HCSPLT_HUBADDR_SHIFT)
2243        | (split.port as u32 & HCSPLT_PRTADDR_MASK)
2244        // Every split this driver issues carries one packet, so the whole
2245        // payload is what the complete-split collects. Linux programs this
2246        // field for both halves, next to the split address it is easy to
2247        // mistake for the whole of HCSPLT.
2248        | HCSPLT_XACTPOS_ALL;
2249    if complete {
2250        value |= HCSPLT_COMPSPLT;
2251    }
2252    value
2253}
