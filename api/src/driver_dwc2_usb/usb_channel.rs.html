10/// How host channels move payload bytes.
11///
12/// The BCM2837 DWC2 supports both; which one a given environment implements is
13/// not discoverable from the register file, so the driver tries FIFO first and
14/// falls back to DMA. FIFO is the board-proven path (the LAN9514 Ethernet runs
15/// there); DMA is what QEMU's `hcd-dwc2` model implements, since it sources
16/// payloads from guest memory at `HCDMA` and ignores host-channel FIFO writes
17/// entirely.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum TransferMode {
20    /// Programmed I/O: payload goes through the channel's data FIFO.
21    Fifo,
22    /// The core reads/writes payloads directly from/to guest memory at `HCDMA`.
23    Dma,
24}
25
26/// Bytes reserved per host channel inside the DMA scratch region.
27///
28/// Sized for the largest payload this driver moves: a full configuration
29/// descriptor (≤512 bytes) and an Ethernet frame. One slot per channel keeps two
30/// in-flight channels from aliasing the same memory.
31pub const DMA_SLOT_BYTES: usize = 4096;
32/// Host channels that get a DMA slot (DWC2 implements up to 16; this driver uses 8).
33const DMA_CHANNELS: usize = 8;
34
35/// Control-endpoint max packet size assumed before a device reports its own.
36///
37/// USB 2.0 §5.5.3 fixes low-speed control endpoints at 8 bytes and allows
38/// full-speed ones 8/16/32/64, so 8 is the only value safe to assume for the
39/// first descriptor read. Declaring 64 there makes the host expect 64-byte
40/// packets from a device that sends 8, and the read fails.
41pub const CONTROL_MPS_LOW_FULL_SPEED: u8 = 8;
42
43/// HPRT0 `PRTSPD` encoding (DWC2 databook): the value `reset_port` returns.
44pub const PORT_SPEED_HIGH: u32 = 0;
45pub const PORT_SPEED_FULL: u32 = 1;
46pub const PORT_SPEED_LOW: u32 = 2;
47
48/// Control-endpoint packet size to assume for a device on a port at `speed`.
49///
50/// USB 2.0 5.5.3: low speed is 8 bytes, full speed is 8/16/32/64, and
51/// **high speed is 64**. The assumption has to follow the negotiated link
52/// speed, and `read_device_descriptor` replaces it with the device's real
53/// `bMaxPacketSize0` as soon as it has read one.
54pub fn initial_control_mps(speed: u32) -> u8 {
55    match speed {
56        PORT_SPEED_HIGH => 64,
57        _ => CONTROL_MPS_LOW_FULL_SPEED,
58    }
59}

86    /// Adopt a device's control-endpoint max packet size.
87    ///
88    /// Clamped to the legal 8/16/32/64 set: a device reporting 0 or a bogus
89    /// value must not produce a zero-length packet size that hangs the engine.
90    pub fn set_control_mps(&self, mps: u8) {
91        let clamped = match mps {
92            0..=8 => 8,
93            9..=16 => 16,
94            17..=32 => 32,
95            _ => 64,
96        };
97        self.ctrl_mps.set(clamped);
98    }

100    /// Current payload-transfer mode.

102        self.mode.get()
103    }

105    /// Select the payload-transfer mode.
106    ///
107    /// Switching to DMA allocates the scratch region on first use; a failed
108    /// allocation leaves the engine in FIFO mode rather than half-configured.
109    pub fn set_mode(&self, mode: TransferMode) -> bool {
110        if mode == TransferMode::Dma && self.dma.borrow().is_none() {
111            let Some(buf) = DmaBuf::alloc(DMA_SLOT_BYTES * DMA_CHANNELS / 4096) else {
112                return false;
113            };
114            *self.dma.borrow_mut() = Some(buf);
115        }
116        self.mode.set(mode);
117        true
118    }

120    /// Guest-physical base of `ch`'s DMA slot, or `None` outside DMA mode.
121    ///
122    /// In SAS grant pages are identity-mapped, so the value programmed into
123    /// `HCDMA` is the same address the CPU uses.
124    fn dma_slot(&self, ch: usize) -> Option<usize> {
125        if self.mode.get() != TransferMode::Dma || ch >= DMA_CHANNELS {
126            return None;
127        }
128        let guard = self.dma.borrow();
129        guard.as_ref().map(|b| b.phys() + ch * DMA_SLOT_BYTES)
130    }
131
132    /// Point the channel's DMA engine at `offset` bytes into its slot.
133    ///
134    /// Programmed **once per transfer**, never per packet. The DWC2 advances
135    /// `HCDMA` itself as it moves each packet (`hcdma += actual`), so a
136    /// multi-packet transfer walks the slot on its own. Re-arming mid-transfer
137    /// both defeats that walk and races the core's asynchronous completion,
138    /// which latches `HCDMA` when it retires the packet.
139    fn program_hcdma(&self, ch: usize, offset: usize) {
140        if let Some(base) = self.dma_slot(ch) {
141            let addr = base + offset.min(DMA_SLOT_BYTES);
142            self.write32(hcdma(ch), addr as u32);
143        }
144    }
145
146    /// Publish CPU writes in `[offset, offset+len)` to the device.
147    fn cache_clean(&self, ch: usize, offset: usize, len: usize) {
148        let guard = self.dma.borrow();
149        if let Some(buf) = guard.as_ref() {
150            let start = ch * DMA_SLOT_BYTES + offset.min(DMA_SLOT_BYTES);
151            let end = (start + len).min((ch + 1) * DMA_SLOT_BYTES);
152            if let Some(token) = buf.begin_cache_sync(start, end.saturating_sub(start)) {
153                let _ = buf.complete_cache_sync(token);
154            }
155        }
156    }
157
158    /// Drop stale cache lines over a range the device just wrote.
159    fn cache_invalidate(&self, ch: usize, offset: usize, len: usize) {
160        self.cache_clean(ch, offset, len);
161    }
162
163    /// Copy an outgoing payload into the channel slot at `offset` and publish it.
164    ///
165    /// This is one of the crate's two `allow(unsafe_code)` islands (see
166    /// `scripts/unsafe-allowlist.toml`): the destination is a grant region this
167    /// cell owns for its whole lifetime, `n` is bounded by the slot size and the
168    /// source length, and the region is not aliased by any other live reference.
169    #[allow(unsafe_code)]
170    fn stage_out(&self, ch: usize, offset: usize, data: &[u8]) {
171        let guard = self.dma.borrow();
172        let n = data.len().min(DMA_SLOT_BYTES - offset.min(DMA_SLOT_BYTES));
173        if let Some(buf) = guard.as_ref() {
174            // SAFETY: destination is this cell's grant slot at a bounded
175            // offset; `n` is bounded by the remaining slot and by `data`.
176            unsafe {
177                core::ptr::copy_nonoverlapping(
178                    data.as_ptr(),
179                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset),
180                    n,
181                );
182            }
183        }
184        drop(guard);
185        self.cache_clean(ch, offset, n);
186    }
187
188    /// Copy a received payload out of the channel slot at `offset`.
189    ///
190    /// The second `allow(unsafe_code)` island: same owned-grant argument as
191    /// [`Self::stage_out`], with `n` additionally bounded by the destination.
192    #[allow(unsafe_code)]
193    fn collect_in(&self, ch: usize, offset: usize, dst: &mut [u8], len: usize) {
194        self.cache_invalidate(ch, offset, len);
195        let guard = self.dma.borrow();
196        if let Some(buf) = guard.as_ref() {
197            let n = len
198                .min(dst.len())
199                .min(DMA_SLOT_BYTES.saturating_sub(offset.min(DMA_SLOT_BYTES)));
200            // SAFETY: source is this cell's grant slot at a bounded offset;
201            // `n` is bounded by the slot remainder and the destination.
202            unsafe {
203                core::ptr::copy_nonoverlapping(
204                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset) as *const u8,
205                    dst.as_mut_ptr(),
206                    n,
207                );
208            }
209        }
210    }

234    /// Execute a standard 8-byte USB SETUP packet on Channel 0.
235    pub fn send_setup(&self, dev_addr: u8, setup: &[u8; 8]) -> ViResult<()> {
236        let ch = 0;
237        self.write32(hcsplt(ch), 0);
238        self.write32(hcintmsk(ch), 0x07FF);
239        // 1. Clear pending channel interrupts
240        self.write32(hcint(ch), 0xFFFF_FFFF);
241        // 2. Configure Transfer Size (HCTSIZ):
242        // XFERSIZE = 8, PKTCNT = 1, PID = 3 (SETUP)
243        let sctsiz = 8 | (1 << 19) | (3 << 29);
244        self.write32(hctsiz(ch), sctsiz);
245
246        // 3. Configure Channel Characteristics (HCCHAR):
247        // HCCHAR fields (bits 0-10 MPS, 11-14 EPNUM, 15 EPDIR, 18-19 EPTYPE, 20 MC, 22-28 DEVADDR,
248        // 31 CHENA): EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
249        let scchar =
250            (self.control_mps() as u32) | (1 << 20) | ((dev_addr as u32) << 22) | (1 << 31);
251        self.write32(hcchar(ch), scchar);
252
253        // 4. Hand the 8 setup bytes to the core. In DMA mode it reads them from
254        //    guest memory at HCDMA; the FIFO is not consulted at all.
255        if self.dma_slot(ch).is_some() {
256            self.program_hcdma(ch, 0);
257            self.stage_out(ch, 0, setup);
258        } else {
259            let w0 = u32::from_le_bytes([setup[0], setup[1], setup[2], setup[3]]);
260            let w1 = u32::from_le_bytes([setup[4], setup[5], setup[6], setup[7]]);
261            self.write_fifo(ch, w0);
262            self.write_fifo(ch, w1);
263        }
264
265        // 5. Poll for completion
266        self.wait_channel(ch)
267    }

269    /// Receive the DATA IN phase of a control transfer on Channel 0.
270    ///
271    /// Returns the bytes actually received. Packets are sized by the control
272    /// endpoint's MPS and the transfer ends at the first **short** packet
273    /// (USB 2.0 §8.5.3.1) — a device answering an 18-byte descriptor request
274    /// with its real 8-byte report must not be read as if it had 18 bytes
275    /// available, which is what a fixed-size read would do.
276    pub fn recv_data(&self, dev_addr: u8, buf: &mut [u8]) -> ViResult<usize> {
277        let ch = 0;
278        let mps = (self.control_mps() as usize).max(1);
279        let mut received = 0;
280        let mut toggle = 2u32; // PID 2 = DATA1 for the first data packet
281                               // Arm the DMA engine once; the core advances it per packet.
282        self.program_hcdma(ch, 0);
283
284        while received < buf.len() {
285            let chunk = (buf.len() - received).min(mps);
286            self.prepare_channel(ch);
287            self.write32(hcsplt(ch), 0);
288            self.write32(hcintmsk(ch), 0x07FF);
289            let sctsiz = (chunk as u32) | (1 << 19) | (toggle << 29);
290            self.write32(hctsiz(ch), sctsiz);
291
292            // HCCHAR: EPNUM = 0, EPDIR = 1 (IN), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
293            let scchar =
294                (mps as u32) | (1 << 15) | (1 << 20) | ((dev_addr as u32) << 22) | (1 << 31);
295            self.write32(hcchar(ch), scchar);
296
297            self.wait_channel(ch)?;
298
299            // The core writes the *remaining* count back into HCTSIZ.
300            let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
301            let got = chunk.saturating_sub(remaining).min(chunk);
302
303            let words = got.div_ceil(4);
304            for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
305                for (j, byte) in word.to_le_bytes().iter().enumerate() {
306                    let offset = i * 4 + j;
307                    let idx = received + offset;
308                    if idx < buf.len() && offset < got {
309                        buf[idx] = *byte;
310                    }
311                }
312            }
313
314            received += got;
315            if got < chunk {
316                break;
317            }
318            toggle = if toggle == 2 { 0 } else { 2 }; // DATA1 <-> DATA0
319        }
320
321        if self.dma_slot(ch).is_some() && received > 0 {
322            let mut tmp = [0u8; DMA_SLOT_BYTES];
323            let n = received.min(buf.len()).min(DMA_SLOT_BYTES);
324            self.collect_in(ch, 0, &mut tmp, n);
325            buf[..n].copy_from_slice(&tmp[..n]);
326        }
327
328        Ok(received)
329    }

331    /// Send STATUS handshake on Channel 0 (0-byte packet with DATA1).
332    pub fn send_status(&self, dev_addr: u8, is_in: bool) -> ViResult<()> {
333        let ch = 0;
334        self.prepare_channel(ch);
335        self.write32(hcsplt(ch), 0);
336        self.write32(hcintmsk(ch), 0x07FF);
337        // XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1)
338        let sctsiz = (1 << 19) | (2 << 29);
339        self.write32(hctsiz(ch), sctsiz);
340
341        // HCCHAR: EPNUM = 0, EPTYPE = 0 (Control), MPS = 64, MC = 1, CHENA = 1, EPDIR from the caller.
342        let epdir = if is_in { 1 } else { 0 };
343        let scchar = (self.control_mps() as u32)
344            | (epdir << 15)
345            | (1 << 20)
346            | ((dev_addr as u32) << 22)
347            | (1 << 31);
348        self.write32(hcchar(ch), scchar);
349
350        self.wait_channel(ch)
351    }

353    /// Execute a complete synchronous USB Control Transfer.
354    pub fn control_transfer(
355        &self,
356        dev_addr: u8,
357        req_type: u8,
358        req: u8,
359        val: u16,
360        idx: u16,
361        buf: &mut [u8],
362    ) -> ViResult<usize> {
363        let is_in = (req_type & 0x80) != 0;
364        let length = buf.len() as u16;
365
366        let setup = [
367            req_type,
368            req,
369            (val & 0xFF) as u8,
370            ((val >> 8) & 0xFF) as u8,
371            (idx & 0xFF) as u8,
372            ((idx >> 8) & 0xFF) as u8,
373            (length & 0xFF) as u8,
374            ((length >> 8) & 0xFF) as u8,
375        ];
376
377        // Each phase reports its own failure: a control transfer that fails
378        // silently is the hardest kind of USB bug to find, and the request
379        // fields are the only thing that identifies which one it was.
380        if let Err(e) = self.send_setup(dev_addr, &setup) {
381            self.report_failure("setup", dev_addr, req_type, req, val, idx, length, e);
382            return Err(e);
383        }
384
385        // 2. DATA phase (optional)
386        let mut actual = 0;
387        if length > 0 {
388            let result = if is_in {
389                self.recv_data(dev_addr, buf)
390            } else {
391                self.send_data(dev_addr, buf).map(|()| buf.len())
392            };
393            match result {
394                Ok(n) => actual = n,
395                Err(e) => {
396                    self.report_failure("data", dev_addr, req_type, req, val, idx, length, e);
397                    return Err(e);
398                }
399            }
400        }
401
402        // 3. STATUS phase (handshake in opposite direction)
403        if let Err(e) = self.send_status(dev_addr, !is_in) {
404            self.report_failure("status", dev_addr, req_type, req, val, idx, length, e);
405            return Err(e);
406        }
407
408        Ok(actual)
409    }

446    /// Send DATA OUT phase on Channel 0.
447    fn send_data(&self, dev_addr: u8, data: &[u8]) -> ViResult<()> {
448        let ch = 0;
449        let mps = (self.control_mps() as usize).max(1);
450        let mut sent = 0;
451        let mut toggle = 2; // PID 2 = DATA1
452                            // Stage the whole payload once and arm DMA once: the core reads it
453                            // packet by packet, advancing HCDMA itself.
454        if self.dma_slot(ch).is_some() {
455            self.stage_out(ch, 0, data);
456            self.program_hcdma(ch, 0);
457        }
458
459        while sent < data.len() {
460            let chunk = (data.len() - sent).min(mps);
461            self.prepare_channel(ch);
462            self.write32(hcsplt(ch), 0);
463            self.write32(hcintmsk(ch), 0x07FF);
464            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
465            self.write32(hctsiz(ch), sctsiz);
466
467            // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
468            let scchar = (mps as u32) | (1 << 20) | ((dev_addr as u32) << 22) | (1 << 31);
469            self.write32(hcchar(ch), scchar);
470
471            if self.dma_slot(ch).is_some() {
472                // Payload already staged and HCDMA already armed.
473            } else {
474                let words = chunk.div_ceil(4);
475                for i in 0..words {
476                    let mut b = [0u8; 4];
477                    for (j, byte) in b.iter_mut().enumerate() {
478                        let offset = i * 4 + j;
479                        if sent + offset < data.len() && offset < chunk {
480                            *byte = data[sent + offset];
481                        }
482                    }
483                    self.write_fifo(ch, u32::from_le_bytes(b));
484                }
485            }
486
487            self.wait_channel(ch)?;
488
489            sent += chunk;
490            toggle = if toggle == 2 { 0 } else { 2 };
491        }
492
493        Ok(())
494    }

496    /// Transmit a raw Ethernet packet via Bulk OUT (Channel 2, EP 2).
497    pub fn bulk_transmit(&self, dev_addr: u8, ep_num: u8, packet: &[u8]) -> ViResult<()> {
498        let ch = 2;
499        let mut sent = 0;
500        let mut toggle = 0; // Starts at DATA0
501
502        let use_dma = self.dma_slot(ch).is_some();
503        if use_dma {
504            self.stage_out(ch, 0, packet);
505            self.program_hcdma(ch, 0);
506        }
507        while sent < packet.len() {
508            let chunk = (packet.len() - sent).min(512); // 512 bytes for High-Speed Bulk
509
510            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
511            // HCCHAR: EPDIR = 0 (OUT), EPTYPE = 2 (Bulk), MC = 1 packet, MPS = 512 (HS Bulk).
512            let scchar = 512
513                | ((ep_num as u32) << 11)
514                | (2 << 18)
515                | (1 << 20)
516                | ((dev_addr as u32) << 22)
517                | (1 << 31);
518
519            let mut retries = 0;
520            loop {
521                self.prepare_channel(ch);
522                self.write32(hcsplt(ch), 0);
523                self.write32(hcintmsk(ch), 0x07FF);
524                self.write32(hctsiz(ch), sctsiz);
525                self.write32(hcchar(ch), scchar);
526
527                if use_dma {
528                    // Staged and armed before the loop; the core walks it.
529                } else {
530                    let words_count = chunk.div_ceil(4);
531                    for i in 0..words_count {
532                        let mut b = [0u8; 4];
533                        for (j, byte) in b.iter_mut().enumerate() {
534                            let offset = i * 4 + j;
535                            let idx = sent + offset;
536                            if idx < packet.len() && offset < chunk {
537                                *byte = packet[idx];
538                            }
539                        }
540                        self.write_fifo(ch, u32::from_le_bytes(b));
541                    }
542                }
543
544                match self.wait_channel(ch) {
545                    Ok(()) => break,
546                    Err(ViError::WouldBlock) => {
547                        retries += 1;
548                        if retries > 50 {
549                            ostd::io::println("[dwc2] bulk_transmit: exceeded 50 NAK retries");
550                            return Err(ViError::IO);
551                        }
552                        sys_yield();
553                    }
554                    Err(e) => return Err(e),
555                }
556            }
557
558            sent += chunk;
559            toggle = if toggle == 0 { 2 } else { 0 };
560        }
561
562        Ok(())
563    }

565    /// Receive a raw packet via Bulk IN (Channel 1, EP 1). Returns received length or 0 if nothing.
566    pub fn bulk_receive(&self, dev_addr: u8, ep_num: u8, buf: &mut [u8]) -> ViResult<usize> {
567        let ch = 1;
568        self.prepare_channel(ch);
569        self.write32(hcsplt(ch), 0);
570        self.write32(hcintmsk(ch), 0x07FF);
571        let want = buf.len().min(512);
572        let sctsiz = (want as u32) | (1 << 19);
573        self.write32(hctsiz(ch), sctsiz);
574
575        let scchar = 512
576            | ((ep_num as u32) << 11)
577            | (1 << 15) // IN
578            | (2 << 18) // Bulk
579            | (1 << 20) // MC = 1
580            | ((dev_addr as u32) << 22)
581            | (1 << 31);
582        self.write32(hcchar(ch), scchar);
583        self.program_hcdma(ch, 0);
584
585        // Non-blocking wait: check if transfer completed or NAK
586        let mut count = 0;
587        while count < 1000 {
588            let int = self.read32(hcint(ch));
589            if int & (1 << 0) != 0 {
590                // The core reports the actual length in HCTSIZ either way, but
591                // only DMA deposits the payload outside the FIFO.
592                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
593                let got = want.saturating_sub(remaining);
594                if self.dma_slot(ch).is_some() {
595                    self.collect_in(ch, 0, buf, got);
596                } else {
597                    let words = want.div_ceil(4);
598                    for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
599                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
600                            let idx = i * 4 + j;
601                            if idx < buf.len() && idx < want {
602                                buf[idx] = *byte;
603                            }
604                        }
605                    }
606                }
607                return Ok(got);
608            }
609            if int & (1 << 4) != 0 {
610                // NAK: device has no packet right now
611                return Ok(0);
612            }
613            count += 1;
614            sys_yield();
615        }
616
617        Ok(0)
618    }

620    /// Receive up to one packet via Interrupt IN on `ch`.
621    ///
622    /// HID input reports ride interrupt endpoints. The device NAKs until it has
623    /// a report, so a NAK is a normal "no data yet" answer, not an error — the
624    /// caller polls. Returns the *actual* transferred length, read back from
625    /// `HCTSIZ` rather than assumed: a HID report can be shorter than the buffer
626    /// (an 8-byte keyboard report into a 64-byte buffer), and using the request
627    /// length would hand the decoder trailing garbage.
628    ///
629    /// `ch` must not collide with the control (0) or Ethernet bulk (1, 2)
630    /// channels.
631    pub fn interrupt_receive(
632        &self,
633        ch: usize,
634        dev_addr: u8,
635        ep_num: u8,
636        mps: u16,
637        buf: &mut [u8],
638    ) -> ViResult<usize> {
639        let want = buf.len().min(mps.max(1) as usize).min(1024);
640        if want == 0 {
641            return Ok(0);
642        }
643
644        self.prepare_channel(ch);
645        self.write32(hcsplt(ch), 0);
646        self.write32(hcintmsk(ch), 0x07FF);
647
648        // HCTSIZ: XFERSIZE (= want), PKTCNT = 1, PID = DATA0 (0). The core
649        // overwrites XFERSIZE with the remaining count as it transfers.
650        let sctsiz = (want as u32) | (1 << 19);
651        self.write32(hctsiz(ch), sctsiz);
652
653        // HCCHAR: MPS from the endpoint descriptor, EPNUM, IN direction,
654        // EPTYPE = 3 (Interrupt), MC = 1, DEVADDR, CHENA.
655        let scchar = (mps as u32 & 0x7FF)
656            | ((ep_num as u32) << 11)
657            | (1 << 15) // EPDIR = IN
658            | (3 << 18) // EPTYPE = Interrupt
659            | (1 << 20) // MC = 1
660            | ((dev_addr as u32) << 22)
661            | (1 << 31); // CHENA
662        self.write32(hcchar(ch), scchar);
663        self.program_hcdma(ch, 0);
664
665        let mut polls = 0u32;
666        while polls < 2_000 {
667            let int = self.read32(hcint(ch));
668
669            // NAK — device has no report queued. Halt and report "no data".
670            if int & (1 << 4) != 0 {
671                self.halt_channel(ch);
672                return Ok(0);
673            }
674            // Errors (STALL / babble / transaction error) end the poll cycle.
675            if int & ((1 << 2) | (1 << 3) | (1 << 7)) != 0 {
676                self.halt_channel(ch);
677                return Err(ViError::IO);
678            }
679
680            let complete = int & (1 << 0) != 0 || int & (1 << 1) != 0 || int & (1 << 5) != 0;
681            if complete {
682                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
683                let got = want.saturating_sub(remaining);
684                self.halt_channel(ch);
685
686                if got > 0 {
687                    if self.dma_slot(ch).is_some() {
688                        // DMA deposited the report in the channel slot.
689                        let mut tmp = [0u8; DMA_SLOT_BYTES];
690                        self.collect_in(ch, 0, &mut tmp, got);
691                        let n = got.min(buf.len());
692                        buf[..n].copy_from_slice(&tmp[..n]);
693                    } else {
694                        // Drain ceil(got/4) FIFO words; only the first `got`
695                        // bytes are meaningful (the tail word is padding).
696                        let words = got.div_ceil(4);
697                        for w in 0..words {
698                            let word = self.read_fifo(ch);
699                            for (j, byte) in word.to_le_bytes().iter().enumerate() {
700                                let idx = w * 4 + j;
701                                if idx < got && idx < buf.len() {
702                                    buf[idx] = *byte;
703                                }
704                            }
705                        }
706                    }
707                } else if self.dma_slot(ch).is_none() {
708                    // A zero-length packet still has a FIFO word to retire on
709                    // some revisions; drain one so the channel is clean.
710                    let _ = self.read_fifo(ch);
711                }
712                return Ok(got);
713            }
714
715            polls += 1;
716            sys_yield();
717        }
718
719        self.halt_channel(ch);
720        Ok(0)
721    }
722
723    /// Name the host-channel interrupt bit that ended a transfer.
724    ///
725    /// The DWC2 encodes one cause per bit and they need different fixes:
726    /// `AHBERR` is the core failing to reach memory at `HCDMA`, `XACTERR` is a
727    /// bus protocol error, `BBLERR` is the device sending more than the
728    /// programmed packet size, `DTERR` is a data-toggle mismatch, and `STALL`
729    /// is the device itself refusing. Collapsing them into one "IO error" hides
730    /// which of those actually happened.
731    fn describe_hcint(&self, int: u32) -> &'static str {
732        if int & (1 << 2) != 0 {
733            return "AHBERR - core could not reach memory at HCDMA";
734        }
735        if int & (1 << 8) != 0 {
736            return "BBLERR - babble, device exceeded the packet size";
737        }
738        if int & (1 << 3) != 0 {
739            return "STALL - endpoint refused the request";
740        }
741        if int & (1 << 7) != 0 {
742            return "XACTERR - transaction error";
743        }
744        if int & (1 << 10) != 0 {
745            return "DTERR - data toggle mismatch";
746        }
747        if int & (1 << 9) != 0 {
748            return "FRMOVRN - frame overrun";
749        }
750        if int & (1 << 6) != 0 {
751            return "NYET - not ready";
752        }
753        if int & (1 << 4) != 0 {
754            return "NAK";
755        }
756        "unknown"
757    }

759    /// Print one register as `[dwc2]   NAME=0xVALUE`.
760    fn dump_reg(&self, name: &str, val: u32) {
761        ostd::io::print("[dwc2]   ");
762        ostd::io::print(name);
763        ostd::io::print("=0x");
764        print_hex_val(val);
765        ostd::io::println("");
766    }

768    /// Dump the channel and core registers that explain a stalled transfer.
769    ///
770    /// A timeout with no status bit set (`HCINT == 0`) is ambiguous from the
771    /// outside: the channel may never have been issued, may have been issued and
772    /// silently abandoned by the core, or the port may have dropped underneath
773    /// it. These registers distinguish those cases.
774    pub fn dump_transfer_state(&self, ch: usize, why: &str) {
775        ostd::io::print("[dwc2] state at ");
776        ostd::io::print(why);
777        ostd::io::println(":");
778        for (name, offset) in [
779            ("HCCHAR", hcchar(ch)),
780            ("HCTSIZ", hctsiz(ch)),
781            ("HCINT", hcint(ch)),
782            ("HCINTMSK", hcintmsk(ch)),
783        ] {
784            self.dump_reg(name, self.read32(offset));
785        }
786        self.dump_reg("HAINT", self.read32(HAINT));
787        self.dump_reg("GINTSTS", self.read32(GINTSTS));
788        self.dump_reg("GINTMSK", self.read32(GINTMSK));
789        self.dump_reg("HPRT0", self.read32(HPRT0));
790        self.dump_reg("GRSTCTL", self.read32(GRSTCTL));
791        self.dump_reg("GNPTXSTS", self.read32(GNPTXSTS));
792        self.dump_reg("MODE", self.mode.get() as u32);
793    }
794
795    /// Log a channel error once per distinct cause, with the register state.
796    fn report_channel_error(&self, ch: usize, int: u32) {
797        static REPORTED: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
798        if REPORTED.fetch_add(1, core::sync::atomic::Ordering::Relaxed) < 4 {
799            ostd::io::print("[dwc2] channel error: ");
800            ostd::io::println(self.describe_hcint(int));
801            self.dump_transfer_state(ch, "error");
802        }
803    }
804
805    /// Leave `ch` disabled before arming it, whatever state it was left in.
806    ///
807    /// A channel that is still enabled from an abandoned transfer ignores the
808    /// next `CHENA`, so each transfer starts by making sure the core has really
809    /// stopped and its interrupt bits are cleared.
810    fn prepare_channel(&self, ch: usize) {
811        if self.read32(hcchar(ch)) & (1 << 31) != 0 {
812            self.halt_channel(ch);
813        }
814        self.write32(hcint(ch), 0xFFFF_FFFF);
815    }
816
817    /// Wait for channel transfer completion or error with timeout.
818    ///
819    /// BCM2837 DWC2 Slave mode: the core does NOT reliably set XFERCOMPL
820    /// or CHHLTD after a successful transaction.  ACK (bit 5) from the
821    /// device is the definitive completion signal; we halt the channel
822    /// manually after seeing it.
823    fn wait_channel(&self, ch: usize) -> ViResult<()> {
824        let mut count = 0;
825        while count < 50_000 {
826            let int = self.read32(hcint(ch));
827
828            // ── Hardware-generated halt ───────────────────────────────
829            if int & (1 << 1) != 0 {
830                // CHHLTD
831                self.write32(hcint(ch), 0xFFFF_FFFF);
832                // Babble (8) and data-toggle error (10) end a transfer just as
833                // surely as the bits that were already checked.
834                if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
835                    self.report_channel_error(ch, int);
836                    return Err(ViError::IO);
837                }
838                if int & (1 << 4) != 0 {
839                    return Err(ViError::WouldBlock);
840                }
841                return Ok(());
842            }
843
844            // ── Transfer complete ─────────────────────────────────────
845            if int & (1 << 0) != 0 {
846                self.halt_channel(ch);
847                return Ok(());
848            }
849
850            // ── ACK = device accepted the packet (BCM2837 primary path)
851            if int & (1 << 5) != 0 {
852                self.halt_channel(ch);
853                return Ok(());
854            }
855
856            // ── Error conditions ──────────────────────────────────────
857            if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
858                self.report_channel_error(ch, int);
859                self.halt_channel(ch);
860                return Err(ViError::IO);
861            }
862            if int & (1 << 4) != 0 {
863                self.halt_channel(ch);
864                return Err(ViError::WouldBlock);
865            }
866
867            count += 1;
868            sys_yield();
869        }
870        // Timeout — capture hcint BEFORE halt clears it
871        let int = self.read32(hcint(ch));
872        let char_val = self.read32(hcchar(ch));
873        self.halt_channel(ch);
874        static TIMEOUT_COUNT: core::sync::atomic::AtomicUsize =
875            core::sync::atomic::AtomicUsize::new(0);
876        let attempt = TIMEOUT_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
877        if attempt < 3 {
878            ostd::io::print("[dwc2] TIMEOUT ch=");
879            print_hex_val(ch as u32);
880            ostd::io::print(" hcint=0x");
881            print_hex_val(int);
882            ostd::io::print(" hcchar=0x");
883            print_hex_val(char_val);
884            ostd::io::println("");
885            self.dump_transfer_state(ch, "timeout");
886        }
887        Err(ViError::IO)
888    }
889
890    /// Explicitly halt a host channel (required in DWC2 Slave mode).
891    ///
892    /// `CHDIS` and `CHENA` must never be set together. The core reads
893    /// `CHENA=1` as a fresh enable, so raising it while requesting a disable
894    /// **re-arms the channel with the parameters still in `HCCHAR`/`HCTSIZ`** --
895    /// a second, unwanted transaction launched from a completed one. Only
896    /// `CHDIS` is raised; the core clears `CHENA` and reports `CHHLTD`.
897    fn halt_channel(&self, ch: usize) {
898        let reg = hcchar(ch);
899        let mut val = self.read32(reg);
900        val |= 1 << 30; // CHDIS
901        val &= !(1 << 31); // CHENA
902        self.write32(reg, val);
903        // Wait for CHHLTD (bit 1) with a short timeout
904        for _ in 0..10_000 {
905            if self.read32(hcint(ch)) & (1 << 1) != 0 {
906                break;
907            }
908            sys_yield();
909        }
910        self.write32(hcint(ch), 0xFFFF_FFFF);
911    }
912}

914fn print_hex_val(val: u32) {
915    const HEX: &[u8; 16] = b"0123456789ABCDEF";
916    let mut buf = [0u8; 8];
917    for i in 0..8 {
918        buf[7 - i] = HEX[((val >> (i * 4)) & 0xF) as usize];
919    }
920    if let Ok(s) = core::str::from_utf8(&buf) {
921        ostd::io::print(s);
922    }
