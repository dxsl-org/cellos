10/// How host channels move payload bytes.
11///
12/// The BCM2837 DWC2 supports both; which one a given environment implements is
13/// not discoverable from the register file, so the driver tries FIFO first and
14/// falls back to DMA. FIFO is the board-proven path (the LAN9514 Ethernet runs
15/// there); DMA is what QEMU's `hcd-dwc2` model implements, since it sources
16/// payloads from guest memory at `HCDMA` and ignores host-channel FIFO writes
17/// entirely.
18#[derive(Clone, Copy, Debug, PartialEq, Eq)]
19pub enum TransferMode {
20    /// Programmed I/O: payload goes through the channel's data FIFO.
21    Fifo,
22    /// The core reads/writes payloads directly from/to guest memory at `HCDMA`.
23    Dma,
24}
25
26/// Bytes reserved per host channel inside the DMA scratch region.
27///
28/// Sized for the largest payload this driver moves: a full configuration
29/// descriptor (≤512 bytes) and an Ethernet frame. One slot per channel keeps two
30/// in-flight channels from aliasing the same memory.
31pub const DMA_SLOT_BYTES: usize = 4096;
32/// Host channels that get a DMA slot (DWC2 implements up to 16; this driver uses 8).
33const DMA_CHANNELS: usize = 8;
34
35/// Control-endpoint max packet size assumed before a device reports its own.
36///
37/// USB 2.0 §5.5.3 fixes low-speed control endpoints at 8 bytes and allows
38/// full-speed ones 8/16/32/64, so 8 is the only value safe to assume for the
39/// first descriptor read. Declaring 64 there makes the host expect 64-byte
40/// packets from a device that sends 8, and the read fails.
41pub const CONTROL_MPS_LOW_FULL_SPEED: u8 = 8;
42
43/// HPRT0 `PRTSPD` encoding (DWC2 databook): the value `reset_port` returns.
44pub const PORT_SPEED_HIGH: u32 = 0;
45pub const PORT_SPEED_FULL: u32 = 1;
46pub const PORT_SPEED_LOW: u32 = 2;
47
48/// Control-endpoint packet size to assume for a device on a port at `speed`.
49///
50/// USB 2.0 5.5.3: low speed is 8 bytes, full speed is 8/16/32/64, and
51/// **high speed is 64**. The assumption has to follow the negotiated link
52/// speed, and `read_device_descriptor` replaces it with the device's real
53/// `bMaxPacketSize0` as soon as it has read one.
54pub fn initial_control_mps(speed: u32) -> u8 {
55    match speed {
56        PORT_SPEED_HIGH => 64,
57        _ => CONTROL_MPS_LOW_FULL_SPEED,
58    }
59}

61/// Complete-splits a periodic poll will issue before giving the hub up.
62const SPLIT_ATTEMPTS: usize = 2;
63
64/// Microframes in one full-speed frame.
65///
66/// `HFNUM` counts microframes, and a hub pairs the two halves of a split inside
67/// one millisecond frame, so the counter has to be shifted to ask that question.
68/// Both reference drivers derive the frame the same way.
69const FULL_FRAME_SHIFT: u32 = 3;
70
71/// Register reads a non-yielding channel wait will make before giving up.
72///
73/// The frame check inside that wait is the real bound; this only stops a core
74/// that never reports anything from spinning forever.
75const SPIN_POLLS: usize = 200_000;
76
77/// Poll budget for the ordinary `wait_channel`.
78const WAIT_POLLS: usize = 50_000;
79
80/// Poll budget for waiting out a channel halt.
81///
82/// The core clears `CHENA` in microseconds; this only has to be long enough not
83/// to be missed, and every iteration of it is a yield handed to the rest of the
84/// system.
85const CHANNEL_HALT_POLLS: usize = 64;
86
87/// Complete-split attempts a synchronous transfer will make.
88///
89/// A control transfer has a caller waiting on its result, so its split is run to
90/// completion here rather than spread across calls the way the interrupt poll is.
91/// The attempts are issued back to back and bounded by frames rather than spun
92/// on, because the caller cannot be handed "not yet".
93const SPLIT_COMPLETE_ATTEMPTS: usize = 8;
94
95/// Frames a complete-split may still belong to the start-split that began it.
96///
97/// The pairing is only meaningful inside the hub's frame budget, so a
98/// complete-split issued after this many is started over instead of being taken
99/// for a result. `HFNUM` counts microframes at high speed, and a low-speed
100/// transaction behind the hub's translator takes up to a full millisecond.
101const SPLIT_COMPLETE_FRAMES: u32 = 12;
102
103/// Where a device sits when it has to be reached through a hub.
104///
105/// A high-speed hub does not pass full- and low-speed traffic through, so a
106/// device behind one is addressed in two transactions: a **start-split** the hub
107/// buffers, then a **complete-split** that collects the result. The core
108/// expresses both through `HCSPLT`.
109///
110/// This is per device, not per transfer, which is why the engine carries it the
111/// same way it carries `ctrl_mps`: both are facts about whoever is being
112/// addressed right now.
113#[derive(Clone, Copy, Debug, PartialEq, Eq)]
114pub struct Split {
115    /// The address the hub itself was assigned.
116    pub hub_addr: u8,
117    /// The downstream port on that hub holding the device.
118    pub port: u8,
119    /// The device is low-speed and needs the preamble its speed requires.
120    pub low_speed: bool,
121}

124    mmio: &'a MmioRegion,
125    /// Max packet size for control transfers on endpoint 0. Updated from the
126    /// device descriptor's `bMaxPacketSize0` once it has been read.
127    ctrl_mps: Cell<u8>,
128    mode: Cell<TransferMode>,
129    /// Per-channel DMA scratch, allocated on first use in DMA mode.
130    dma: RefCell<Option<DmaBuf>>,
131    /// Hub the device currently being addressed sits behind, when it is not on
132    /// the root port itself.

138    /// never finishing look identical from the caller's side. Keeping the raw
139    /// register lets `report_failure` name which one happened -- and lets the
140    /// split paths tell a completed complete-split from one that ended on a
141    /// handshake that carries no data.

143}

189    /// `HCSPLT` for the current context; `complete` selects the second pass.
190    fn hcsplt_value(&self, complete: bool) -> u32 {
191        let Some(split) = self.split.get() else {
192            return 0;
193        };
194        let mut value = HCSPLT_SPLTENA
195            | ((split.hub_addr as u32 & HCSPLT_HUBADDR_MASK >> HCSPLT_HUBADDR_SHIFT)
196                << HCSPLT_HUBADDR_SHIFT)
197            | (split.port as u32 & HCSPLT_PRTADDR_MASK);
198        if complete {
199            value |= HCSPLT_COMPSPLT;
200        }
201        value
202    }

204    /// Current USB frame number, used to bound a complete-split.
205    pub fn frame_number(&self) -> u32 {
206        self.read32(HFNUM) & HFNUM_FRNUM_MASK
207    }

209    /// The millisecond frame a periodic split is paired within.
210    ///
211    /// `HFNUM` counts microframes, so the raw counter advances every 125 us and
212    /// cannot answer a question about millisecond frames. Both reference drivers
213    /// shift it right three for exactly this check.
214    #[inline]
215    fn full_frame(&self) -> u32 {
216        self.frame_number() >> FULL_FRAME_SHIFT
217    }
218
219    /// Wait until the current frame is nearly over, without yielding.
220    ///
221    /// The hub does not pass the full- or low-speed transaction through: it runs
222    /// one of its own and buffers the result, and that transaction takes a whole
223    /// millisecond frame. A complete-split issued one microframe after its
224    /// start-split therefore asks before there is anything to hand back, and the
225    /// answer is a channel that halts carrying nothing. The complete-split has to
226    /// arrive late in the frame the start-split opened, because the hub stops
227    /// pairing the two the moment that frame ends.
228    ///
229    /// Returns false once the frame has moved, which is the signal that the
230    /// pairing is gone and the next poll has to start over. Both the frame that
231    /// must not move and the position inside it come from the shifted counter: it
232    /// advances every microframe, so comparing it raw would fail on the first read.
233    fn await_frame_end(&self) -> bool {
234        let frame = self.full_frame();
235        let last_microframe = (1 << FULL_FRAME_SHIFT) - 1;
236        for _ in 0..SPIN_POLLS {
237            let counter = self.read32(HFNUM) & HFNUM_FRNUM_MASK;
238            if counter >> FULL_FRAME_SHIFT != frame {
239                return false;
240            }
241            if counter & last_microframe == last_microframe {
242                return true;
243            }
244        }
245        false
246    }
247
248    /// The last channel outcome reported `XFERCOMPL`.
249    ///
250    /// A complete-split that ends on ACK has delivered nothing: Linux is
251    /// explicit that ACK belongs to the start-split and "should not occur in
252    /// CSPLIT". Reading the raw bit is the only way to tell, because every
253    /// completion collapses into the same `Ok` on the way out.
254    fn last_reported_complete(&self) -> bool {
255        self.last_hcint.get() & (1 << 0) != 0
256    }
257
258    /// The last channel failure was NYET rather than NAK.
259    ///
260    /// Both arrive as `WouldBlock` and they mean opposite things for a split: a
261    /// hub answering NYET has not finished the transaction and is worth asking
262    /// again, while a NAK ends the pairing and the next attempt has to begin
263    /// with a fresh start-split.
264    fn last_was_nyet(&self) -> bool {
265        self.last_hcint.get() & (1 << 6) != 0
266    }

268    /// `HCCHAR` for this instant, with `ODDFRM` set from the frame the transfer
269    /// will be sent in.
270    ///
271    /// The core latches the parity of the frame a transfer lands in when the
272    /// channel starts, and a transfer whose parity disagrees is refused. The
273    /// frame read here is the one just started, and the transfer goes out in the
274    /// next, which is why the parity is inverted -- the same reading U-Boot
275    /// takes before it starts a channel.
276    #[inline]
277    fn start_hcchar(&self, base: u32) -> u32 {
278        if self.frame_number() & 1 == 0 {
279            base | HCCHAR_ODDFRM
280        } else {
281            base & !HCCHAR_ODDFRM
282        }
283    }

285    /// Start a channel and, behind a hub, run the split handshake that reaches
286    /// the device at all.
287    ///
288    /// Every packet takes two passes over the channel: the start-split hands the
289    /// transaction to the hub, and the complete-split collects what it buffered.
290    /// A hub that has not finished answers the complete-split with NYET, which is
291    /// why that is a retry here rather than a failure.
292    ///
293    /// Without this a full- or low-speed device on a hub port is never reached:
294    /// nothing on the bus answers, and the channel comes back with XACTERR. That
295    /// reads like a protocol error and is really a missing transaction.
296    ///
297    /// `arm(complete)` stages the payload and starts the channel for one pass.
298    fn run_packet(&self, ch: usize, arm: impl Fn(bool)) -> ViResult<()> {
299        if self.split.get().is_none() {
300            arm(false);
301            return self.wait_channel(ch);
302        }
303
304        arm(false);
305        self.wait_channel(ch)?;
306
307        let started = self.frame_number();
308        for _ in 0..SPLIT_COMPLETE_ATTEMPTS {
309            // Put the complete-split straight back out. The reference is explicit
310            // that a non-periodic split is asked again immediately rather than
311            // waited on, and everything a yield would add here is time the pairing
312            // does not survive -- which is why the stalled-endpoint recovery this
313            // path carries could never clear a stall.
314            arm(true);
315            match self.wait_channel_spin(ch, false) {
316                // Only a real transfer completion is the result. An ACK here is
317                // the hub acknowledging the request, not delivering it.
318                Ok(()) if self.last_reported_complete() => return Ok(()),
319                Ok(()) => {}
320                Err(ViError::WouldBlock) => {}
321                Err(e) => return Err(e),
322            }
323            if self.frame_number().wrapping_sub(started) & HFNUM_FRNUM_MASK > SPLIT_COMPLETE_FRAMES
324            {
325                return Err(ViError::IO);
326            }
327        }
328        Err(ViError::IO)
329    }

350    /// Select the payload-transfer mode.
351    ///
352    /// Switching to DMA allocates the scratch region on first use; a failed
353    /// allocation leaves the engine in FIFO mode rather than half-configured.
354    pub fn set_mode(&self, mode: TransferMode) -> bool {
355        if mode == TransferMode::Dma && self.dma.borrow().is_none() {
356            let Some(buf) = DmaBuf::alloc(DMA_SLOT_BYTES * DMA_CHANNELS / 4096) else {
357                return false;

361        self.mode.set(mode);

365    /// Guest-physical base of `ch`'s DMA slot, or `None` outside DMA mode.
366    ///
367    /// In SAS grant pages are identity-mapped, so the value programmed into
368    /// `HCDMA` is the same address the CPU uses.
369    fn dma_slot(&self, ch: usize) -> Option<usize> {
370        if self.mode.get() != TransferMode::Dma || ch >= DMA_CHANNELS {
371            return None;
372        }
373        let guard = self.dma.borrow();
374        guard.as_ref().map(|b| b.phys() + ch * DMA_SLOT_BYTES)
375    }

377    /// Bus address of a RAM physical address on this SoC.
378    ///
379    /// A bus master does not see ARM physical addresses. On the BCM283x the
380    /// Raspberry Pi device tree's `_DMA` resource states the translation
381    /// outright: "Bus 0xC0000000 -> CPU 0x00000000", over the first gigabyte
382    /// and marked NonCacheable. U-Boot's working dwc2 driver programs `HCDMA`
383    /// through exactly this translation (`phys_to_bus`).
384    ///
385    /// Programming the raw ARM physical address asks the core to read and write
386    /// a different location, and the failure is quiet: the address still lands
387    /// in mapped memory, so no AHB error is raised, and the CPU-side read-back
388    /// of the slot looks correct because the CPU and the core are simply
389    /// looking at different bytes.
390    #[inline]
391    fn bus_address(phys: usize) -> u32 {
392        (phys | 0xC000_0000) as u32
393    }

395    /// Point the channel's DMA engine at `offset` bytes into its slot.
396    ///
397    /// Programmed **once per transfer, and always before `CHENA`**. The DWC2
398    /// advances `HCDMA` itself as it moves each packet (`hcdma += actual`), so a
399    /// multi-packet transfer walks the slot on its own; re-arming mid-transfer
400    /// both defeats that walk and races the core's asynchronous completion.
401    ///
402    /// The ordering against `CHENA` is not a preference. The core latches its
403    /// DMA address when the channel is enabled, so a start that arms `HCDMA`
404    /// afterwards -- or never -- sends the first transaction of the transfer to
405    /// whatever address the previous one left behind. On a SETUP that is a
406    /// request the device ACKs and then STALLs one stage later, and the retry
407    /// succeeds because by then the register holds the right address.
408    fn program_hcdma(&self, ch: usize, offset: usize) {
409        if let Some(base) = self.dma_slot(ch) {
410            let addr = base + offset.min(DMA_SLOT_BYTES);
411            self.write32(hcdma(ch), Self::bus_address(addr));
412        }
413    }

415    /// Publish CPU writes in `[offset, offset+len)` to the device.
416    ///
417    /// A refused sync is reported, never skipped: without the clean the core
418    /// reads stale memory, so the device receives a request whose bytes are not
419    /// the ones staged and answers with a STALL -- a failure that looks like a
420    /// protocol problem and is actually a missing cache operation.
421    fn cache_clean(&self, ch: usize, offset: usize, len: usize) {
422        let guard = self.dma.borrow();
423        if let Some(buf) = guard.as_ref() {
424            let start = ch * DMA_SLOT_BYTES + offset.min(DMA_SLOT_BYTES);
425            let end = (start + len).min((ch + 1) * DMA_SLOT_BYTES);
426            let span = end.saturating_sub(start);
427            if span == 0 {
428                // Nothing to publish. The kernel rejects a zero-length sync as
429                // an invalid range, so asking for one turns a no-op into a
430                // reported failure -- and a transfer that legitimately moves no
431                // bytes (an idle bulk-IN poll) would warn on every poll.
432                return;
433            }
434            match buf.begin_cache_sync(start, span) {
435                Some(token) => {
436                    if !buf.complete_cache_sync(token) {
437                        ostd::io::println("[dwc2] WARN: cache sync completion refused");
438                    }
439                }
440                None => {
441                    ostd::io::println(
442                        "[dwc2] WARN: cache sync refused - device may read stale memory",
443                    );
444                }
445            }
446        }
447    }

454    /// Copy an outgoing payload into the channel slot at `offset` and publish it.
455    ///
456    /// This is one of the crate's two `allow(unsafe_code)` islands (see
457    /// `scripts/unsafe-allowlist.toml`): the destination is a grant region this
458    /// cell owns for its whole lifetime, `n` is bounded by the slot size and the
459    /// source length, and the region is not aliased by any other live reference.
460    #[allow(unsafe_code)]
461    fn stage_out(&self, ch: usize, offset: usize, data: &[u8]) {
462        let guard = self.dma.borrow();
463        let n = data.len().min(DMA_SLOT_BYTES - offset.min(DMA_SLOT_BYTES));
464        if let Some(buf) = guard.as_ref() {
465            // SAFETY: destination is this cell's grant slot at a bounded
466            // offset; `n` is bounded by the remaining slot and by `data`.
467            unsafe {
468                core::ptr::copy_nonoverlapping(
469                    data.as_ptr(),
470                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset),
471                    n,
472                );
473            }
474        }
475        drop(guard);
476        self.cache_clean(ch, offset, n);
477    }

479    /// The third `allow(unsafe_code)` island: a volatile read of this cell's
480    /// own grant slot at a bounded offset, used only for the trace below.
481    #[allow(unsafe_code)]
482    /// Print the first bytes staged in a channel slot.
483    ///
484    /// Proves what the core is about to hand the device. If these are not the
485    /// request the caller built, the fault is in staging or cache maintenance;
486    /// if they are, the fault is between memory and the bus.
487    pub fn trace_slot(&self, ch: usize, label: &str, len: usize) {
488        let guard = self.dma.borrow();
489        let Some(buf) = guard.as_ref() else {
490            return;
491        };
492        ostd::io::print("[dwc2] slot ");
493        ostd::io::print(label);
494        ostd::io::print(" =");
495        let n = len.min(16).min(DMA_SLOT_BYTES);
496        for i in 0..n {
497            // SAFETY: reading this cell's own grant slot at a bounded offset.
498            let byte = unsafe {
499                core::ptr::read_volatile(
500                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES).wrapping_add(i),
501                )
502            };
503            ostd::io::print(" ");
504            print_hex_val(byte as u32);
505        }
506        ostd::io::println("");
507    }

509    /// Copy a received payload out of the channel slot at `offset`.
510    ///
511    /// The second `allow(unsafe_code)` island: same owned-grant argument as
512    /// [`Self::stage_out`], with `n` additionally bounded by the destination.
513    #[allow(unsafe_code)]
514    fn collect_in(&self, ch: usize, offset: usize, dst: &mut [u8], len: usize) {
515        self.cache_invalidate(ch, offset, len);
516        let guard = self.dma.borrow();
517        if let Some(buf) = guard.as_ref() {
518            let n = len
519                .min(dst.len())
520                .min(DMA_SLOT_BYTES.saturating_sub(offset.min(DMA_SLOT_BYTES)));
521            // SAFETY: source is this cell's grant slot at a bounded offset;
522            // `n` is bounded by the slot remainder and the destination.
523            unsafe {
524                core::ptr::copy_nonoverlapping(
525                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset) as *const u8,
526                    dst.as_mut_ptr(),
527                    n,
528                );
529            }
530        }
531    }

555    /// Execute a standard 8-byte USB SETUP packet on Channel 0.
556    pub fn send_setup(&self, dev_addr: u8, setup: &[u8; 8]) -> ViResult<()> {
557        let ch = 0;
558        self.prepare_channel(ch);
559        self.write32(hcintmsk(ch), 0x07FF);
560
561        // HCTSIZ: XFERSIZE = 8, PKTCNT = 1, PID = 3 (SETUP).
562        // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1.
563        let sctsiz = 8 | (1 << 19) | (3 << 29);
564        let scchar = (self.control_mps() as u32)
565            | self.device_flags()
566            | (1 << 20)
567            | ((dev_addr as u32) << 22)
568            | (1 << 31);
569        let dma = self.dma_slot(ch).is_some();
570
571        // Stage once up front so the trace has something to show. Each pass
572        // stages again, which is also what hands the bytes to the core.
573        if dma {
574            self.stage_out(ch, 0, setup);
575            static SETUP_TRACED: core::sync::atomic::AtomicBool =
576                core::sync::atomic::AtomicBool::new(false);
577            if !SETUP_TRACED.swap(true, core::sync::atomic::Ordering::Relaxed) {
578                self.trace_slot(ch, "setup", setup.len());
579            }
580        }
581
582        // The payload is staged and HCDMA armed *before* CHENA on every pass:
583        // the core latches its DMA address when the channel starts, so enabling
584        // first reads the previous transfer's address and the device ACKs eight
585        // bytes it never looked at.
586        self.run_packet(ch, |complete| {
587            self.write32(hcsplt(ch), self.hcsplt_value(complete));
588            self.write32(hctsiz(ch), sctsiz);
589            if dma {
590                self.program_hcdma(ch, 0);
591                self.stage_out(ch, 0, setup);
592            } else {
593                self.write_fifo(
594                    ch,
595                    u32::from_le_bytes([setup[0], setup[1], setup[2], setup[3]]),
596                );
597                self.write_fifo(
598                    ch,
599                    u32::from_le_bytes([setup[4], setup[5], setup[6], setup[7]]),
600                );
601            }
602            self.write32(hcint(ch), 0xFFFF_FFFF);
603            self.write32(hcchar(ch), self.start_hcchar(scchar));
604        })
605    }

607    /// Receive the DATA IN phase of a control transfer on Channel 0.
608    ///
609    /// Returns the bytes actually received. Packets are sized by the control
610    /// endpoint's MPS and the transfer ends at the first **short** packet
611    /// (USB 2.0 §8.5.3.1) — a device answering an 18-byte descriptor request
612    /// with its real 8-byte report must not be read as if it had 18 bytes
613    /// available, which is what a fixed-size read would do.
614    pub fn recv_data(&self, dev_addr: u8, buf: &mut [u8]) -> ViResult<usize> {
615        let ch = 0;
616        let mps = (self.control_mps() as usize).max(1);
617        let mut received = 0;
618        let mut toggle = 2u32; // PID 2 = DATA1 for the first data packet
619                               // Arm the DMA engine once; the core advances it per packet.
620        self.program_hcdma(ch, 0);
621
622        while received < buf.len() {
623            let chunk = (buf.len() - received).min(mps);
624            self.prepare_channel(ch);
625            self.write32(hcintmsk(ch), 0x07FF);
626
627            // HCCHAR: EPNUM = 0, EPDIR = 1 (IN), EPTYPE = 0 (Control), MC = 1.
628            let sctsiz = (chunk as u32) | (1 << 19) | (toggle << 29);
629            let scchar = (mps as u32)
630                | self.device_flags()
631                | (1 << 15)
632                | (1 << 20)
633                | ((dev_addr as u32) << 22)
634                | (1 << 31);
635
636            self.run_packet(ch, |complete| {
637                self.write32(hcsplt(ch), self.hcsplt_value(complete));
638                self.write32(hctsiz(ch), sctsiz);
639                // Directly, the address is armed once before the loop and the
640                // core walks the slot itself, so re-arming per packet would
641                // overwrite what it has already written. Behind a hub the core
642                // moves a single packet per pass, so the address is re-armed
643                // every pass and advanced by what has arrived so far.
644                if complete || self.split.get().is_some() {
645                    self.program_hcdma(ch, received);
646                }
647                self.write32(hcint(ch), 0xFFFF_FFFF);
648                self.write32(hcchar(ch), self.start_hcchar(scchar));
649            })?;
650
651            // The core writes the *remaining* count back into HCTSIZ.
652            let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
653            let got = chunk.saturating_sub(remaining).min(chunk);
654
655            let words = got.div_ceil(4);
656            for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
657                for (j, byte) in word.to_le_bytes().iter().enumerate() {
658                    let offset = i * 4 + j;
659                    let idx = received + offset;
660                    if idx < buf.len() && offset < got {
661                        buf[idx] = *byte;
662                    }
663                }
664            }
665
666            received += got;
667            if got < chunk {
668                break;
669            }
670            toggle = if toggle == 2 { 0 } else { 2 }; // DATA1 <-> DATA0
671        }
672
673        if self.dma_slot(ch).is_some() && received > 0 {
674            let mut tmp = [0u8; DMA_SLOT_BYTES];
675            let n = received.min(buf.len()).min(DMA_SLOT_BYTES);
676            self.collect_in(ch, 0, &mut tmp, n);
677            buf[..n].copy_from_slice(&tmp[..n]);
678        }
679
680        Ok(received)
681    }

683    /// Send STATUS handshake on Channel 0 (0-byte packet with DATA1).
684    pub fn send_status(&self, dev_addr: u8, is_in: bool) -> ViResult<()> {
685        let ch = 0;
686        self.prepare_channel(ch);
687        self.write32(hcsplt(ch), 0);
688        self.write32(hcintmsk(ch), 0x07FF);
689        // XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1)
690        // HCTSIZ: XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1).
691        // HCCHAR: EPNUM = 0, EPTYPE = 0 (Control), MPS = 64, MC = 1, EPDIR from
692        // the caller.
693        let sctsiz = (1 << 19) | (2 << 29);
694        let epdir = if is_in { 1 } else { 0 };
695        let scchar = (self.control_mps() as u32)
696            | self.device_flags()
697            | (epdir << 15)
698            | (1 << 20)
699            | ((dev_addr as u32) << 22)
700            | (1 << 31);
701
702        self.run_packet(ch, |complete| {
703            self.write32(hcsplt(ch), self.hcsplt_value(complete));
704            self.write32(hctsiz(ch), sctsiz);
705            // A zero-length handshake moves no bytes, but the channel still
706            // latches an address when it starts and the previous transfer's is
707            // not a position this stage may begin from.
708            self.program_hcdma(ch, 0);
709            self.write32(hcint(ch), 0xFFFF_FFFF);
710            self.write32(hcchar(ch), self.start_hcchar(scchar));
711        })
712    }

714    /// Execute a complete synchronous USB Control Transfer.
715    pub fn control_transfer(
716        &self,
717        dev_addr: u8,
718        req_type: u8,
719        req: u8,
720        val: u16,
721        idx: u16,
722        buf: &mut [u8],
723    ) -> ViResult<usize> {
724        let is_in = (req_type & 0x80) != 0;
725        let length = buf.len() as u16;
726
727        let setup = [
728            req_type,
729            req,
730            (val & 0xFF) as u8,
731            ((val >> 8) & 0xFF) as u8,
732            (idx & 0xFF) as u8,
733            ((idx >> 8) & 0xFF) as u8,
734            (length & 0xFF) as u8,
735            ((length >> 8) & 0xFF) as u8,
736        ];
737
738        // Each phase reports its own failure: a control transfer that fails
739        // silently is the hardest kind of USB bug to find, and the request
740        // fields are the only thing that identifies which one it was.
741        if let Err(e) = self.send_setup(dev_addr, &setup) {
742            self.report_failure("setup", dev_addr, req_type, req, val, idx, length, e);
743            return Err(e);
744        }
745
746        // 2. DATA phase (optional)
747        let mut actual = 0;
748        if length > 0 {
749            let result = if is_in {
750                self.recv_data(dev_addr, buf)
751            } else {
752                self.send_data(dev_addr, buf).map(|()| buf.len())
753            };
754            match result {
755                Ok(n) => actual = n,
756                Err(e) => {
757                    self.report_failure("data", dev_addr, req_type, req, val, idx, length, e);
758                    return Err(e);
759                }
760            }
761        }
762
763        // 3. STATUS phase (handshake in opposite direction)
764        if let Err(e) = self.send_status(dev_addr, !is_in) {
765            self.report_failure("status", dev_addr, req_type, req, val, idx, length, e);
766            return Err(e);
767        }
768
769        Ok(actual)
770    }

812    /// Send DATA OUT phase on Channel 0.
813    fn send_data(&self, dev_addr: u8, data: &[u8]) -> ViResult<()> {
814        let ch = 0;
815        let mps = (self.control_mps() as usize).max(1);
816        let mut sent = 0;
817        let mut toggle = 2; // PID 2 = DATA1
818                            // Stage the whole payload once and arm DMA once: the core reads it
819                            // packet by packet, advancing HCDMA itself.
820        if self.dma_slot(ch).is_some() {
821            self.stage_out(ch, 0, data);
822            self.program_hcdma(ch, 0);
823        }
824
825        while sent < data.len() {
826            let chunk = (data.len() - sent).min(mps);
827            self.prepare_channel(ch);
828            self.write32(hcsplt(ch), 0);
829            self.write32(hcintmsk(ch), 0x07FF);
830            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
831            self.write32(hctsiz(ch), sctsiz);
832
833            // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
834            let scchar = (mps as u32)
835                | self.device_flags()
836                | (1 << 20)
837                | ((dev_addr as u32) << 22)
838                | (1 << 31);
839            self.write32(hcchar(ch), self.start_hcchar(scchar));
840
841            if self.dma_slot(ch).is_some() {
842                // Payload already staged and HCDMA already armed.
843            } else {
844                let words = chunk.div_ceil(4);
845                for i in 0..words {
846                    let mut b = [0u8; 4];
847                    for (j, byte) in b.iter_mut().enumerate() {
848                        let offset = i * 4 + j;
849                        if sent + offset < data.len() && offset < chunk {
850                            *byte = data[sent + offset];
851                        }
852                    }
853                    self.write_fifo(ch, u32::from_le_bytes(b));
854                }
855            }
856
857            self.wait_channel(ch)?;
858
859            sent += chunk;
860            toggle = if toggle == 2 { 0 } else { 2 };
861        }
862
863        Ok(())
864    }

866    /// Transmit a raw Ethernet packet via Bulk OUT (Channel 2, EP 2).
867    pub fn bulk_transmit(&self, dev_addr: u8, ep_num: u8, packet: &[u8]) -> ViResult<()> {
868        let ch = 2;
869        let mut sent = 0;
870        let mut toggle = 0; // Starts at DATA0
871
872        let use_dma = self.dma_slot(ch).is_some();
873        if use_dma {
874            self.stage_out(ch, 0, packet);
875            self.program_hcdma(ch, 0);
876        }
877        while sent < packet.len() {
878            let chunk = (packet.len() - sent).min(512); // 512 bytes for High-Speed Bulk
879
880            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
881            // HCCHAR: EPDIR = 0 (OUT), EPTYPE = 2 (Bulk), MC = 1 packet, MPS = 512 (HS Bulk).
882            let scchar = 512
883                | ((ep_num as u32) << 11)
884                | (2 << 18)
885                | (1 << 20)
886                | ((dev_addr as u32) << 22)
887                | (1 << 31);
888
889            let mut retries = 0;
890            loop {
891                self.prepare_channel(ch);
892                self.write32(hcsplt(ch), 0);
893                self.write32(hcintmsk(ch), 0x07FF);
894                self.write32(hctsiz(ch), sctsiz);
895                self.write32(hcchar(ch), self.start_hcchar(scchar));
896
897                if use_dma {
898                    // Staged and armed before the loop; the core walks it.
899                } else {
900                    let words_count = chunk.div_ceil(4);
901                    for i in 0..words_count {
902                        let mut b = [0u8; 4];
903                        for (j, byte) in b.iter_mut().enumerate() {
904                            let offset = i * 4 + j;
905                            let idx = sent + offset;
906                            if idx < packet.len() && offset < chunk {
907                                *byte = packet[idx];
908                            }
909                        }
910                        self.write_fifo(ch, u32::from_le_bytes(b));
911                    }
912                }
913
914                match self.wait_channel(ch) {
915                    Ok(()) => break,
916                    Err(ViError::WouldBlock) => {
917                        retries += 1;
918                        if retries > 50 {
919                            ostd::io::println("[dwc2] bulk_transmit: exceeded 50 NAK retries");
920                            return Err(ViError::IO);
921                        }
922                        sys_yield();
923                    }
924                    Err(e) => return Err(e),
925                }
926            }
927
928            sent += chunk;
929            toggle = if toggle == 0 { 2 } else { 0 };
930        }
931
932        Ok(())
933    }

935    /// Receive a raw packet via Bulk IN (Channel 1, EP 1). Returns received length or 0 if nothing.
936    pub fn bulk_receive(&self, dev_addr: u8, ep_num: u8, buf: &mut [u8]) -> ViResult<usize> {
937        let ch = 1;
938        self.prepare_channel(ch);
939        self.write32(hcsplt(ch), 0);
940        self.write32(hcintmsk(ch), 0x07FF);
941        let want = buf.len().min(512);
942        let sctsiz = (want as u32) | (1 << 19);
943        self.write32(hctsiz(ch), sctsiz);
944
945        let scchar = 512
946            | ((ep_num as u32) << 11)
947            | (1 << 15) // IN
948            | (2 << 18) // Bulk
949            | (1 << 20) // MC = 1
950            | ((dev_addr as u32) << 22)
951            | (1 << 31);
952        self.program_hcdma(ch, 0);
953        self.write32(hcchar(ch), self.start_hcchar(scchar));
954
955        // Non-blocking wait: check if transfer completed or NAK
956        let mut count = 0;
957        while count < 1000 {
958            let int = self.read32(hcint(ch));
959            if int & (1 << 0) != 0 {
960                // The core reports the actual length in HCTSIZ either way, but
961                // only DMA deposits the payload outside the FIFO.
962                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
963                let got = want.saturating_sub(remaining);
964                if self.dma_slot(ch).is_some() {
965                    self.collect_in(ch, 0, buf, got);
966                } else {
967                    let words = want.div_ceil(4);
968                    for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
969                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
970                            let idx = i * 4 + j;
971                            if idx < buf.len() && idx < want {
972                                buf[idx] = *byte;
973                            }
974                        }
975                    }
976                }
977                return Ok(got);
978            }
979            if int & (1 << 4) != 0 {
980                // NAK: device has no packet right now
981                return Ok(0);
982            }
983            count += 1;
984            sys_yield();
985        }
986
987        Ok(0)
988    }

990    /// Receive up to one packet via Interrupt IN on `ch`.
991    ///
992    /// HID input reports ride interrupt endpoints. The device NAKs until it has
993    /// a report, so a NAK is a normal "no data yet" answer, not an error — the
994    /// caller polls. Returns the *actual* transferred length, read back from
995    /// `HCTSIZ` rather than assumed: a HID report can be shorter than the buffer
996    /// (an 8-byte keyboard report into a 64-byte buffer), and using the request
997    /// length would hand the decoder trailing garbage.
998    ///
999    /// `ch` must not collide with the control (0) or Ethernet bulk (1, 2)
1000    /// channels.
1001    pub fn interrupt_receive(
1002        &self,
1003        ch: usize,
1004        dev_addr: u8,
1005        ep_num: u8,
1006        mps: u16,
1007        buf: &mut [u8],
1008        split_pending: &mut bool,
1009    ) -> ViResult<usize> {
1010        let want = buf.len().min(mps.max(1) as usize).min(1024);
1011        if want == 0 {
1012            return Ok(0);
1013        }
1014
1015        // Behind a hub the poll is a pair of transactions, one half per call.
1016        if self.split.get().is_some() {
1017            return self.poll_split(ch, dev_addr, ep_num, mps, want, buf, split_pending);
1018        }
1019
1020        self.prepare_channel(ch);
1021        self.write32(hcsplt(ch), 0);
1022        self.write32(hcintmsk(ch), 0x07FF);
1023
1024        // HCTSIZ: XFERSIZE (= want), PKTCNT = 1, PID = DATA0 (0). The core
1025        // overwrites XFERSIZE with the remaining count as it transfers.
1026        let sctsiz = (want as u32) | (1 << 19);
1027        self.write32(hctsiz(ch), sctsiz);
1028
1029        // HCCHAR: MPS from the endpoint descriptor, EPNUM, IN direction,
1030        // EPTYPE = 3 (Interrupt), MC = 1, DEVADDR, CHENA.
1031        let scchar = (mps as u32 & 0x7FF)
1032            | ((ep_num as u32) << 11)
1033            | self.device_flags()
1034            | (1 << 15) // EPDIR = IN
1035            | (3 << 18) // EPTYPE = Interrupt
1036            | (1 << 20) // MC = 1
1037            | ((dev_addr as u32) << 22)
1038            | (1 << 31); // CHENA
1039        self.program_hcdma(ch, 0);
1040        self.write32(hcchar(ch), self.start_hcchar(scchar));
1041
1042        let mut polls = 0u32;
1043        while polls < 2_000 {
1044            let int = self.read32(hcint(ch));
1045
1046            // NAK — device has no report queued. Halt and report "no data".
1047            if int & (1 << 4) != 0 {
1048                self.halt_channel(ch);
1049                return Ok(0);
1050            }
1051            // Errors (STALL / babble / transaction error) end the poll cycle.
1052            if int & ((1 << 2) | (1 << 3) | (1 << 7)) != 0 {
1053                self.halt_channel(ch);
1054                return Err(ViError::IO);
1055            }
1056
1057            let complete = int & (1 << 0) != 0 || int & (1 << 1) != 0 || int & (1 << 5) != 0;
1058            if complete {
1059                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1060                let got = want.saturating_sub(remaining);
1061                self.halt_channel(ch);
1062
1063                if got > 0 {
1064                    if self.dma_slot(ch).is_some() {
1065                        // DMA deposited the report in the channel slot.
1066                        let mut tmp = [0u8; DMA_SLOT_BYTES];
1067                        self.collect_in(ch, 0, &mut tmp, got);
1068                        let n = got.min(buf.len());
1069                        buf[..n].copy_from_slice(&tmp[..n]);
1070                    } else {
1071                        // Drain ceil(got/4) FIFO words; only the first `got`
1072                        // bytes are meaningful (the tail word is padding).
1073                        let words = got.div_ceil(4);
1074                        for w in 0..words {
1075                            let word = self.read_fifo(ch);
1076                            for (j, byte) in word.to_le_bytes().iter().enumerate() {
1077                                let idx = w * 4 + j;
1078                                if idx < got && idx < buf.len() {
1079                                    buf[idx] = *byte;
1080                                }
1081                            }
1082                        }
1083                    }
1084                } else if self.dma_slot(ch).is_none() {
1085                    // A zero-length packet still has a FIFO word to retire on
1086                    // some revisions; drain one so the channel is clean.
1087                    let _ = self.read_fifo(ch);
1088                }
1089                return Ok(got);
1090            }
1091
1092            polls += 1;
1093            sys_yield();
1094        }
1095
1096        self.halt_channel(ch);
1097        Ok(0)
1098    }

1100    /// One half of an interrupt IN split, for a device behind a hub.
1101    ///
1102    /// A split is a pair of transactions the hub has to see in different
1103    /// microframes: a start-split that hands it the request, and a complete-split
1104    /// issued later that collects the result. One half is issued per call, with
1105    /// the pairing carried in `pending`, so the gap between them is the gap
1106    /// between two polls rather than a wait taken here.
1107    ///
1108    /// Waiting here is what this used to do, and it cost the machine: `HFNUM`
1109    /// counts frames of a millisecond, not microframes, so crossing one boundary
1110    /// meant spinning for a whole millisecond of system calls -- every poll, on
1111    /// every interface, exactly when a device had something to say.
1112    ///
1113    /// The three ways a complete-split can end mean different things:
1114    ///
1115    /// * `XFERCOMPL` is the report, and ends the pairing;
1116    /// * `NYET` is the hub still working, so the pairing stays live;
1117    /// * `NAK` ends the pairing with nothing.
1118    ///
1119    /// ACK carries no data here -- Linux is explicit that it "should not occur in
1120    /// CSPLIT" -- so it is not taken for a result.
1121    #[allow(clippy::too_many_arguments)]
1122    fn poll_split(
1123        &self,
1124        ch: usize,
1125        dev_addr: u8,
1126        ep_num: u8,
1127        mps: u16,
1128        want: usize,
1129        buf: &mut [u8],
1130        pending: &mut bool,
1131    ) -> ViResult<usize> {
1132        self.prepare_channel(ch);
1133        self.write32(hcintmsk(ch), 0x07FF);
1134
1135        // HCTSIZ: XFERSIZE = one max packet, PKTCNT = 1, PID = DATA0. A split
1136        // carries exactly one packet, whatever the caller asked for.
1137        //
1138        // HCCHAR: MPS, EPNUM, IN, EPTYPE = 3 (Interrupt), and MC = 3. The
1139        // multicount is how many times the core retries the transaction itself,
1140        // and for a periodic split it is more than one: the hub answers NYET
1141        // until it has run the full- or low-speed transaction, so a single
1142        // attempt meets that NYET and ends the transfer with nothing.
1143        let sctsiz = (want as u32) | (1 << 19);
1144        let scchar = (mps as u32 & 0x7FF)
1145            | ((ep_num as u32) << 11)
1146            | self.device_flags()
1147            | (1 << 15)
1148            | (3 << 18)
1149            | (3 << 20)
1150            | ((dev_addr as u32) << 22)
1151            | (1 << 31);
1152        let arm = |complete: bool| {
1153            self.write32(hcsplt(ch), self.hcsplt_value(complete));
1154            self.write32(hctsiz(ch), sctsiz);
1155            self.program_hcdma(ch, 0);
1156            self.write32(hcint(ch), 0xFFFF_FFFF);
1157            self.write32(hcchar(ch), self.start_hcchar(scchar));
1158        };
1159
1160        // Both halves are issued here, one microframe apart and inside a single
1161        // frame. The second half cannot be handed to a later poll: this loop's
1162        // RecvTimeout puts hundreds of frames between polls and the hub pairs a
1163        // split for less than one, which is what made every complete-split here
1164        // answer NYET.
1165        // Both halves are issued here, one microframe apart and inside a single
1166        // frame, and nothing is written to the console between them. A line of
1167        // console output costs about seven milliseconds at this baud, which is
1168        // seven of the frames the hub pairs a split within -- printing the first
1169        // half before issuing the second is what kept the pair apart. Every report
1170        // below therefore sits on a path where no further channel work depends on
1171        // it, and the retry path has none at all.
1172        let frame = self.frame_number();
1173        arm(false);
1174        let outcome = self.wait_channel_spin(ch, true);
1175        if !matches!(outcome, Ok(())) {
1176            *pending = false;
1177            report_split_progress(
1178                false,
1179                &outcome,
1180                self.last_hcint.get(),
1181                self.last_was_nyet(),
1182                want,
1183                frame,
1184            );
1185            return match outcome {
1186                Err(e) => Err(e),
1187                _ => Ok(0),
1188            };
1189        }
1190
1191        for _ in 0..SPLIT_ATTEMPTS {
1192            if !self.await_frame_end() {
1193                // The frame moved, so the pairing is gone whether or not the hub
1194                // ever answered.
1195                *pending = false;
1196                return Ok(0);
1197            }
1198
1199            arm(true);
1200            let outcome = self.wait_channel_spin(ch, true);
1201            let int = self.last_hcint.get();
1202
1203            match outcome {
1204                Ok(()) if !self.last_reported_complete() => {
1205                    // ACK on a complete-split carries no data.
1206                    *pending = false;
1207                    report_split_progress(true, &outcome, int, self.last_was_nyet(), want, frame);
1208                    return Ok(0);
1209                }
1210                Ok(()) => {
1211                    *pending = false;
1212                    let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1213                    let got = want.saturating_sub(remaining);
1214                    if got == 0 {
1215                        report_split_progress(
1216                            true,
1217                            &outcome,
1218                            int,
1219                            self.last_was_nyet(),
1220                            want,
1221                            frame,
1222                        );
1223                        return Ok(0);
1224                    }
1225                    if self.dma_slot(ch).is_some() {
1226                        let mut tmp = [0u8; DMA_SLOT_BYTES];
1227                        self.collect_in(ch, 0, &mut tmp, got.min(DMA_SLOT_BYTES));
1228                        let n = got.min(buf.len());
1229                        buf[..n].copy_from_slice(&tmp[..n]);
1230                        return Ok(n);
1231                    }
1232                    let words = got.div_ceil(4);
1233                    for w in 0..words {
1234                        let word = self.read_fifo(ch);
1235                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
1236                            let idx = w * 4 + j;
1237                            if idx < got && idx < buf.len() {
1238                                buf[idx] = *byte;
1239                            }
1240                        }
1241                    }
1242                    return Ok(got.min(buf.len()));
1243                }
1244                // Still working, or NAK which ends the pairing. Either way the
1245                // next poll starts over rather than asking again here -- and a
1246                // hub that answered NYET is asked again below, with nothing
1247                // written in between for the same reason as above.
1248                Err(ViError::WouldBlock) => {
1249                    if !self.last_was_nyet() {
1250                        *pending = false;
1251                        report_split_progress(
1252                            true,
1253                            &outcome,
1254                            int,
1255                            self.last_was_nyet(),
1256                            want,
1257                            frame,
1258                        );
1259                        return Ok(0);
1260                    }
1261                }
1262                Err(e) => {
1263                    *pending = false;
1264                    report_split_progress(true, &outcome, int, self.last_was_nyet(), want, frame);
1265                    return Err(e);
1266                }
1267            }
1268        }
1269
1270        // The hub never finished. Starting over on the next poll is the only
1271        // thing left: the frame this pairing belonged to is gone with it.
1272        *pending = false;
1273        Ok(0)
1274    }

1276    /// Name the host-channel interrupt bit that ended a transfer.
1277    ///
1278    /// The DWC2 encodes one cause per bit and they need different fixes:
1279    /// `AHBERR` is the core failing to reach memory at `HCDMA`, `XACTERR` is a
1280    /// bus protocol error, `BBLERR` is the device sending more than the
1281    /// programmed packet size, `DTERR` is a data-toggle mismatch, and `STALL`
1282    /// is the device itself refusing. Collapsing them into one "IO error" hides
1283    /// which of those actually happened.
1284    fn describe_hcint(&self, int: u32) -> &'static str {
1285        if int & (1 << 2) != 0 {
1286            return "AHBERR - core could not reach memory at HCDMA";
1287        }
1288        if int & (1 << 8) != 0 {
1289            return "BBLERR - babble, device exceeded the packet size";
1290        }
1291        if int & (1 << 3) != 0 {
1292            return "STALL - endpoint refused the request";
1293        }
1294        if int & (1 << 7) != 0 {
1295            return "XACTERR - transaction error";
1296        }
1297        if int & (1 << 10) != 0 {
1298            return "DTERR - data toggle mismatch";
1299        }
1300        if int & (1 << 9) != 0 {
1301            return "FRMOVRN - frame overrun";
1302        }
1303        if int & (1 << 6) != 0 {
1304            return "NYET - not ready";
1305        }
1306        if int & (1 << 4) != 0 {
1307            return "NAK";
1308        }
1309        "unknown"
1310    }

1373    /// Wait for channel transfer completion or error with timeout.
1374    ///
1375    /// BCM2837 DWC2 Slave mode: the core does NOT reliably set XFERCOMPL
1376    /// or CHHLTD after a successful transaction.  ACK (bit 5) from the
1377    /// device is the definitive completion signal; we halt the channel
1378    /// manually after seeing it.
1379    fn wait_channel(&self, ch: usize) -> ViResult<()> {
1380        self.wait_channel_with(ch, WAIT_POLLS)
1381    }

1383    /// Wait for channel completion or error, spending at most `budget` polls.
1384    ///
1385    /// The budget is a real cost rather than a formality: every poll yields, and
1386    /// a transfer that never reports anything spends the whole of it before it is
1387    /// given up on. Callers on the driver's own serving thread want a small one.
1388    fn wait_channel_with(&self, ch: usize, budget: usize) -> ViResult<()> {
1389        for _ in 0..budget {
1390            if let Some(outcome) = self.channel_outcome(ch) {
1391                return outcome;
1392            }
1393            sys_yield();
1394        }
1395        self.channel_timeout(ch)
1396    }

1398    /// Wait for a channel without handing the CPU away.
1399    ///
1400    /// A periodic split has to keep both halves inside one millisecond frame. The
1401    /// yielding wait gives the CPU away for around twenty milliseconds per poll,
1402    /// which put the complete-split twenty frames past the start-split it belongs
1403    /// to -- outside the window the hub pairs them within, so it came back as a
1404    /// transaction error however many times it was tried.
1405    ///
1406    /// The bound is the frame itself rather than a count: past that the pairing is
1407    /// gone and starting over on the next poll is the only useful thing left. This
1408    /// runs only when a device has something to report.
1409    /// `keep_frame` bounds the wait to the millisecond frame it started in, which
1410    /// is what a periodic split needs. A control or bulk transfer legitimately
1411    /// spans frames, so its callers ask for the unbounded wait and keep their own
1412    /// budget.
1413    fn wait_channel_spin(&self, ch: usize, keep_frame: bool) -> ViResult<()> {
1414        let frame = self.full_frame();
1415        for _ in 0..SPIN_POLLS {
1416            if let Some(outcome) = self.channel_outcome(ch) {
1417                return outcome;
1418            }
1419            if keep_frame && self.full_frame() != frame {
1420                self.halt_channel(ch);
1421                self.last_hcint.set(self.read32(hcint(ch)));
1422                return Err(ViError::WouldBlock);
1423            }
1424        }
1425        self.channel_timeout(ch)
1426    }

1428    /// Classify a channel's interrupt register, or `None` while it is still running.
1429    fn channel_outcome(&self, ch: usize) -> Option<ViResult<()>> {
1430        let int = self.read32(hcint(ch));
1431        if int == 0 {
1432            return None;
1433        }
1434
1435        // ── Hardware-generated halt ───────────────────────────────
1436        if int & (1 << 1) != 0 {
1437            // CHHLTD
1438            self.write32(hcint(ch), 0xFFFF_FFFF);
1439            // Babble (8) and data-toggle error (10) end a transfer just as
1440            // surely as the bits that were already checked.
1441            if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
1442                self.report_channel_error(ch, int);
1443                self.last_hcint.set(int);
1444                return Some(Err(ViError::IO));
1445            }
1446            // NAK and NYET both mean "ask again", and both have to be reported
1447            // rather than folded into success: a hub answering a complete-split
1448            // with NYET has not produced a result yet, and returning Ok there
1449            // hands the caller an empty buffer as if the transfer had happened.
1450            if int & ((1 << 4) | (1 << 6)) != 0 {
1451                self.last_hcint.set(int);
1452                return Some(Err(ViError::WouldBlock));
1453            }
1454            self.last_hcint.set(int);
1455            return Some(Ok(()));
1456        }
1457
1458        // ── Transfer complete ─────────────────────────────────────
1459        if int & (1 << 0) != 0 {
1460            self.halt_channel(ch);
1461            self.last_hcint.set(int);
1462            return Some(Ok(()));
1463        }
1464
1465        // ── ACK = device accepted the packet (BCM2837 primary path)
1466        if int & (1 << 5) != 0 {
1467            self.halt_channel(ch);
1468            self.last_hcint.set(int);
1469            return Some(Ok(()));
1470        }
1471
1472        // ── Error conditions ──────────────────────────────────────
1473        if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
1474            self.report_channel_error(ch, int);
1475            self.halt_channel(ch);
1476            self.last_hcint.set(int);
1477            return Some(Err(ViError::IO));
1478        }
1479        if int & ((1 << 4) | (1 << 6)) != 0 {
1480            self.halt_channel(ch);
1481            self.last_hcint.set(int);
1482            return Some(Err(ViError::WouldBlock));
1483        }
1484
1485        None
1486    }

1488    /// Give up on a channel that reported nothing within its budget.
1489    fn channel_timeout(&self, ch: usize) -> ViResult<()> {
1490        // Timeout — capture hcint BEFORE halt clears it
1491        let int = self.read32(hcint(ch));
1492        let char_val = self.read32(hcchar(ch));
1493        self.halt_channel(ch);
1494        static TIMEOUT_COUNT: core::sync::atomic::AtomicUsize =
1495            core::sync::atomic::AtomicUsize::new(0);
1496        let attempt = TIMEOUT_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
1497        if attempt < 3 {
1498            ostd::io::print("[dwc2] TIMEOUT ch=");
1499            print_hex_val(ch as u32);
1500            ostd::io::print(" hcint=0x");
1501            print_hex_val(int);
1502            ostd::io::print(" hcchar=0x");
1503            print_hex_val(char_val);
1504            ostd::io::println("");
1505            self.dump_transfer_state(ch, "timeout");
1506        }
1507        self.last_hcint.set(int);
1508        Err(ViError::IO)
1509    }

1511    /// Explicitly halt a host channel (required in DWC2 Slave mode).
1512    ///
1513    /// `CHDIS` and `CHENA` must never be set together. The core reads
1514    /// `CHENA=1` as a fresh enable, so raising it while requesting a disable
1515    /// **re-arms the channel with the parameters still in `HCCHAR`/`HCTSIZ`** --
1516    /// a second, unwanted transaction launched from a completed one. Only
1517    /// `CHDIS` is raised; the core clears `CHENA` and reports `CHHLTD`.
1518    fn halt_channel(&self, ch: usize) {
1519        let reg = hcchar(ch);
1520        let mut val = self.read32(reg);
1521        val |= 1 << 30; // CHDIS
1522        val &= !(1 << 31); // CHENA
1523        self.write32(reg, val);
1524        // Wait for CHHLTD (bit 1) with a short timeout. The core clears CHENA in
1525        // microseconds, so this only has to be long enough not to be missed --
1526        // and it runs on the serving thread, so it must not be long enough to be
1527        // felt if the core never answers.
1528        for _ in 0..CHANNEL_HALT_POLLS {
1529            if self.read32(hcint(ch)) & (1 << 1) != 0 {
1530                break;
1531            }
1532            sys_yield();
1533        }
1534        self.write32(hcint(ch), 0xFFFF_FFFF);
1535    }
1536}

1538/// Report the first split poll in full, once.
1539///
1540/// Report how each half of a split ended, a few times per half.
1541///
1542/// Which half ran and how it ended is the whole question when a split does not
1543/// deliver: a start-split the hub refuses, a complete-split it is not ready for,
1544/// and a complete-split that ends without data are three different faults, and
1545/// nothing else in the driver tells them apart. Both halves are reported
1546/// separately -- a single shared flag reports one half and hides the other, which
1547/// is exactly the one worth seeing. The frame number comes along so the gap
1548/// between the halves is visible rather than inferred.
1549fn report_split_progress(
1550    complete: bool,
1551    outcome: &ViResult<()>,
1552    hcint: u32,
1553    nyet: bool,
1554    want: usize,
1555    frame: u32,
1556) {
1557    static SEEN: [core::sync::atomic::AtomicUsize; 2] =
1558        [const { core::sync::atomic::AtomicUsize::new(0) }; 2];
1559    let slot = usize::from(complete);
1560    if SEEN[slot].fetch_add(1, core::sync::atomic::Ordering::Relaxed) >= 3 {
1561        return;
1562    }
1563    ostd::io::print("[dwc2] split: half=");
1564    ostd::io::print(if complete { "csplit" } else { "ssplit" });
1565    ostd::io::print(" outcome=");
1566    match outcome {
1567        Ok(()) => ostd::io::print("Ok"),
1568        Err(ViError::WouldBlock) if nyet => ostd::io::print("NYET"),
1569        Err(ViError::WouldBlock) => ostd::io::print("NAK"),
1570        Err(_) => ostd::io::print("error"),
1571    }
1572    ostd::io::print(" hcint=0x");
1573    print_hex_val(hcint);
1574    ostd::io::print(" frame=0x");
1575    print_hex_val(frame);
1576    ostd::io::print(" want=");
1577    print_usize_val(want);
1578    ostd::io::println("");
1579}

1581fn print_usize_val(v: usize) {
1582    let mut out = [0u8; 20];
1583    let mut n = v;
1584    let mut len = 0;
1585    if n == 0 {
1586        ostd::io::print("0");
1587        return;
1588    }
1589    while n > 0 && len < out.len() {
1590        out[len] = b'0' + (n % 10) as u8;
1591        n /= 10;
1592        len += 1;
1593    }
1594    out[..len].reverse();
1595    if let Ok(s) = core::str::from_utf8(&out[..len]) {
1596        ostd::io::print(s);
1597    }
1598}

1604        buf[7 - i] = HEX[((val >> (i * 4)) & 0xF) as usize];
1605    }
1606    if let Ok(s) = core::str::from_utf8(&buf) {
1607        ostd::io::print(s);
1608    }
