10/// Register snapshots taken around a split, recorded as it runs and written out
11/// afterwards.
12///
13/// Nothing can be printed inside a split: one line of console output costs about
14/// seven milliseconds at this baud, which is seven of the frames a split has to
15/// live in, and printing between the halves is what kept them from pairing at all.
16/// The registers are captured here instead and read back once the transfer is
17/// over, which is the closest thing to watching the bus that the console allows.
18mod trace {
19    use core::sync::atomic::{AtomicBool, AtomicU32, AtomicUsize, Ordering};
20    use ostd::io::print;
21
22    /// Deep enough to hold a full enumeration and the first polls after it. At
23    /// twenty-four the ring was filled by enumeration alone, so the registers of
24    /// the first interrupt poll -- the one that finds an endpoint already halted --
25    /// were pushed out before anything read them.
26    pub const RECORDS: usize = 160;
27    const WORDS: usize = 5;
28    /// Tag of the record the dump stops after, so a single event can be followed.
29    pub const TAG_ARM_SSPLIT: u32 = 1;
30    pub const TAG_AFTER_SSPLIT: u32 = 2;
31    pub const TAG_ARM_CSPLIT: u32 = 3;
32    pub const TAG_AFTER_CSPLIT: u32 = 4;
33
34    static CELLS: [AtomicU32; RECORDS * WORDS] = [const { AtomicU32::new(0) }; RECORDS * WORDS];
35    static NEXT: AtomicUsize = AtomicUsize::new(0);
36    static DUMPED: AtomicBool = AtomicBool::new(false);
37
38    /// Record `tag` beside four registers. Never blocks and never allocates.
39    pub fn record(tag: u32, a: u32, b: u32, c: u32, d: u32) {
40        let i = NEXT.fetch_add(1, Ordering::Relaxed);
41        if i >= RECORDS {
42            return;
43        }
44        let base = i * WORDS;
45        for (off, v) in [tag, a, b, c, d].into_iter().enumerate() {
46            CELLS[base + off].store(v, Ordering::Relaxed);
47        }
48    }
49
50    fn hex(v: u32) {
51        const D: &[u8; 16] = b"0123456789abcdef";
52        let mut buf = [0u8; 8];
53        let mut i = 0;
54        while i < 8 {
55            buf[i] = D[((v >> (28 - i * 4)) & 0xF) as usize];
56            i += 1;
57        }
58        if let Ok(t) = core::str::from_utf8(&buf) {
59            print(t);
60        }
61    }
62
63    fn decimal(v: usize) {
64        let mut buf = [0u8; 20];
65        let mut n = v;
66        let mut len = 0;
67        if n == 0 {
68            print("0");
69            return;
70        }
71        while n > 0 && len < buf.len() {
72            buf[len] = b'0' + (n % 10) as u8;
73            n /= 10;
74            len += 1;
75        }
76        buf[..len].reverse();
77        if let Ok(t) = core::str::from_utf8(&buf[..len]) {
78            print(t);
79        }
80    }
81
82    /// Begin a fresh recording.
83    ///
84    /// Enumeration is hundreds of records on its own, so a ring read after it
85    /// describes enumeration and never the polls that come next -- which is the
86    /// question now. Starting over when the polls begin puts them at the front.
87    pub fn reset() {
88        NEXT.store(0, Ordering::Relaxed);
89        DUMPED.store(false, Ordering::Relaxed);
90    }
91
92    /// Write the recording out, once, outside any transfer.
93    pub fn dump_once() {
94        if DUMPED.swap(true, Ordering::Relaxed) {
95            return;
96        }
97        let n = NEXT.load(Ordering::Relaxed).min(RECORDS);
98        print("[dwc2-trace] records=");
99        decimal(n);
100        print(" (tag hcint hcchar hcsplt hctsiz)\n");
101        for i in 0..n {
102            let base = i * WORDS;
103            print("  ");
104            hex(CELLS[base].load(Ordering::Relaxed));
105            print(" ");
106            hex(CELLS[base + 1].load(Ordering::Relaxed));
107            print(" ");
108            hex(CELLS[base + 2].load(Ordering::Relaxed));
109            print(" ");
110            hex(CELLS[base + 3].load(Ordering::Relaxed));
111            print(" ");
112            hex(CELLS[base + 4].load(Ordering::Relaxed));
113            print("\n");
114        }
115    }

118/// How host channels move payload bytes.
119///
120/// The BCM2837 DWC2 supports both; which one a given environment implements is
121/// not discoverable from the register file, so the driver tries FIFO first and
122/// falls back to DMA. FIFO is the board-proven path (the LAN9514 Ethernet runs
123/// there); DMA is what QEMU's `hcd-dwc2` model implements, since it sources
124/// payloads from guest memory at `HCDMA` and ignores host-channel FIFO writes
125/// entirely.
126#[derive(Clone, Copy, Debug, PartialEq, Eq)]
127pub enum TransferMode {
128    /// Programmed I/O: payload goes through the channel's data FIFO.
129    Fifo,
130    /// The core reads/writes payloads directly from/to guest memory at `HCDMA`.
131    Dma,
132}
133
134/// Bytes reserved per host channel inside the DMA scratch region.
135///
136/// Sized for the largest payload this driver moves: a full configuration
137/// descriptor (≤512 bytes) and an Ethernet frame. One slot per channel keeps two
138/// in-flight channels from aliasing the same memory.
139pub const DMA_SLOT_BYTES: usize = 4096;
140/// Host channels that get a DMA slot (DWC2 implements up to 16; this driver uses 8).
141const DMA_CHANNELS: usize = 8;
142
143/// Control-endpoint max packet size assumed before a device reports its own.
144///
145/// USB 2.0 §5.5.3 fixes low-speed control endpoints at 8 bytes and allows
146/// full-speed ones 8/16/32/64, so 8 is the only value safe to assume for the
147/// first descriptor read. Declaring 64 there makes the host expect 64-byte
148/// packets from a device that sends 8, and the read fails.
149pub const CONTROL_MPS_LOW_FULL_SPEED: u8 = 8;
150
151/// HPRT0 `PRTSPD` encoding (DWC2 databook): the value `reset_port` returns.
152pub const PORT_SPEED_HIGH: u32 = 0;
153pub const PORT_SPEED_FULL: u32 = 1;
154pub const PORT_SPEED_LOW: u32 = 2;
155
156/// Control-endpoint packet size to assume for a device on a port at `speed`.
157///
158/// USB 2.0 5.5.3: low speed is 8 bytes, full speed is 8/16/32/64, and
159/// **high speed is 64**. The assumption has to follow the negotiated link
160/// speed, and `read_device_descriptor` replaces it with the device's real
161/// `bMaxPacketSize0` as soon as it has read one.
162pub fn initial_control_mps(speed: u32) -> u8 {
163    match speed {
164        PORT_SPEED_HIGH => 64,
165        _ => CONTROL_MPS_LOW_FULL_SPEED,
166    }
167}

169/// Complete-splits a periodic poll will issue before giving the hub up.
170const SPLIT_ATTEMPTS: usize = 2;
171
172/// Microframes in one full-speed frame.
173///
174/// `HFNUM` counts microframes, and a hub pairs the two halves of a split inside
175/// one millisecond frame, so the counter has to be shifted to ask that question.
176/// Both reference drivers derive the frame the same way.
177const FULL_FRAME_SHIFT: u32 = 3;
178
179/// Register reads a non-yielding channel wait will make before giving up.
180///
181/// The frame check inside that wait is the real bound; this only stops a core
182/// that never reports anything from spinning forever.
183const SPIN_POLLS: usize = 200_000;
184
185/// Poll budget for the ordinary `wait_channel`.
186const WAIT_POLLS: usize = 50_000;
187
188/// Poll budget for waiting out a channel halt.
189///
190/// The core clears `CHENA` in microseconds; this only has to be long enough not
191/// to be missed, and every iteration of it is a yield handed to the rest of the
192/// system.
193const CHANNEL_HALT_POLLS: usize = 4_000;
194
195/// Complete-split attempts a synchronous transfer will make.
196///
197/// A control transfer has a caller waiting on its result, so its split is run to
198/// completion here rather than spread across calls the way the interrupt poll is.
199/// The attempts are issued back to back and bounded by frames rather than spun
200/// on, because the caller cannot be handed "not yet".
201const SPLIT_COMPLETE_ATTEMPTS: usize = 2;
202
203/// Frames a complete-split may still belong to the start-split that began it.
204///
205/// The pairing is only meaningful inside the hub's frame budget, so a
206/// complete-split issued after this many is started over instead of being taken
207/// for a result. `HFNUM` counts microframes at high speed, and a low-speed
208/// transaction behind the hub's translator takes up to a full millisecond.
209const SPLIT_COMPLETE_FRAMES: u32 = 12;

211/// Where a device sits when it has to be reached through a hub.
212///
213/// A high-speed hub does not pass full- and low-speed traffic through, so a
214/// device behind one is addressed in two transactions: a **start-split** the hub
215/// buffers, then a **complete-split** that collects the result. The core
216/// expresses both through `HCSPLT`.
217///
218/// This is per device, not per transfer, which is why the engine carries it the
219/// same way it carries `ctrl_mps`: both are facts about whoever is being
220/// addressed right now.
221#[derive(Clone, Copy, Debug, PartialEq, Eq)]
222pub struct Split {
223    /// The address the hub itself was assigned.
224    pub hub_addr: u8,
225    /// The downstream port on that hub holding the device.
226    pub port: u8,
227    /// The device is low-speed and needs the preamble its speed requires.
228    pub low_speed: bool,
229}

231pub struct UsbHostEngine<'a> {
232    mmio: &'a MmioRegion,
233    /// Max packet size for control transfers on endpoint 0. Updated from the
234    /// device descriptor's `bMaxPacketSize0` once it has been read.
235    ctrl_mps: Cell<u8>,
236    mode: Cell<TransferMode>,
237    /// Per-channel DMA scratch, allocated on first use in DMA mode.
238    dma: RefCell<Option<DmaBuf>>,
239    /// Hub the device currently being addressed sits behind, when it is not on
240    /// the root port itself.
241    split: Cell<Option<Split>>,
242    /// `HCINT` captured at the last channel outcome, success or failure.
243    ///
244    /// `wait_channel` collapses every hardware result into an error kind, and a
245    /// device refusing a request, the bus failing to carry it, and the core
246    /// never finishing look identical from the caller's side. Keeping the raw
247    /// register lets `report_failure` name which one happened -- and lets the
248    /// split paths tell a completed complete-split from one that ended on a
249    /// handshake that carries no data.
250    last_hcint: Cell<u32>,
251}

284    /// `HCCHAR` bits that describe the device rather than the endpoint.
285    ///
286    /// A low-speed device needs `HCCHAR.LSPDDEV` on every transfer, including
287    /// the ones a hub relays, because that is what selects the preamble. It is

297    /// `HCSPLT` for the current context; `complete` selects the second pass.
298    fn hcsplt_value(&self, complete: bool) -> u32 {
299        let Some(split) = self.split.get() else {
300            return 0;
301        };
302        let mut value = HCSPLT_SPLTENA
303            | ((split.hub_addr as u32 & HCSPLT_HUBADDR_MASK >> HCSPLT_HUBADDR_SHIFT)
304                << HCSPLT_HUBADDR_SHIFT)
305            | (split.port as u32 & HCSPLT_PRTADDR_MASK)
306            // Every split this driver issues carries one packet, so the whole
307            // payload is what the complete-split collects. Linux programs this
308            // field for both halves, next to the split address it is easy to
309            // mistake for the whole of HCSPLT.
310            | HCSPLT_XACTPOS_ALL;
311        if complete {
312            value |= HCSPLT_COMPSPLT;
313        }
314        value
315    }

317    /// Current USB frame number, used to bound a complete-split.
318    pub fn frame_number(&self) -> u32 {
319        self.read32(HFNUM) & HFNUM_FRNUM_MASK
320    }

322    /// The millisecond frame a periodic split is paired within.
323    ///
324    /// `HFNUM` counts microframes, so the raw counter advances every 125 us and
325    /// cannot answer a question about millisecond frames. Both reference drivers
326    /// shift it right three for exactly this check.
327    #[inline]
328    fn full_frame(&self) -> u32 {
329        self.frame_number() >> FULL_FRAME_SHIFT
330    }
331
332    /// Wait until a frame has just started, without yielding.
333    ///
334    /// The hub's translator runs a periodic transaction inside a frame slot, and
335    /// the start-split is what places it there -- so the start-split has to go out
336    /// at the beginning of a frame, with the complete-split collecting the result
337    /// at the end of that same frame. Issuing the start-split wherever the poll
338    /// happened to fall leaves the translator without the frame it was asked to
339    /// work in, and it answers every complete-split with NYET because the
340    /// transaction never had the time to run.
341    ///
342    /// Returns false if the frame moves while waiting, meaning the poll is starting
343    /// too close to a boundary to do anything useful with the frame it found.
344    fn await_frame_start(&self) -> bool {
345        let frame = self.full_frame();
346        let last_microframe = (1 << FULL_FRAME_SHIFT) - 1;
347        for _ in 0..SPIN_POLLS {
348            let counter = self.read32(HFNUM) & HFNUM_FRNUM_MASK;
349            if counter >> FULL_FRAME_SHIFT != frame {
350                // A frame has just turned over, which is the moment asked for.
351                return true;
352            }
353            if counter & last_microframe == 0 {
354                // First microframe of the frame: nothing asked for it to end.
355                return true;
356            }
357        }
358        false
359    }
360
361    /// Wait until the current frame is nearly over, without yielding.
362    ///
363    /// The hub does not pass the full- or low-speed transaction through: it runs
364    /// one of its own and buffers the result, and that transaction takes a whole
365    /// millisecond frame. A complete-split issued one microframe after its
366    /// start-split therefore asks before there is anything to hand back, and the
367    /// answer is a channel that halts carrying nothing. The complete-split has to
368    /// arrive late in the frame the start-split opened, because the hub stops
369    /// pairing the two the moment that frame ends.
370    ///
371    /// Returns false once the frame has moved, which is the signal that the
372    /// pairing is gone and the next poll has to start over. Both the frame that
373    /// must not move and the position inside it come from the shifted counter: it
374    /// advances every microframe, so comparing it raw would fail on the first read.
375    ///
376    /// The point waited for leaves room for the complete-split itself. Waiting for
377    /// the last microframe of the frame leaves it none, and the core halts the
378    /// channel at the frame boundary before the hub can answer -- a halt carrying
379    /// no status at all, which is what `hcint=0x02` on its own is.
380    fn await_frame_end(&self) -> bool {
381        let frame = self.full_frame();
382        let microframe: u32 = (1 << FULL_FRAME_SHIFT) - 1;
383        let leave_for_complete_split = microframe.saturating_sub(2);
384        for _ in 0..SPIN_POLLS {
385            let counter = self.read32(HFNUM) & HFNUM_FRNUM_MASK;
386            if counter >> FULL_FRAME_SHIFT != frame {
387                return false;
388            }
389            if counter & microframe >= leave_for_complete_split {
390                return true;
391            }
392        }
393        false
394    }

396    /// The last channel outcome reported `XFERCOMPL`.
397    ///
398    /// A complete-split that ends on ACK has delivered nothing: Linux is
399    /// explicit that ACK belongs to the start-split and "should not occur in
400    /// CSPLIT". Reading the raw bit is the only way to tell, because every
401    /// completion collapses into the same `Ok` on the way out.
402    fn last_reported_complete(&self) -> bool {
403        self.last_hcint.get() & (1 << 0) != 0
404    }

406    /// The last channel failure was NYET rather than NAK.
407    ///
408    /// Both arrive as `WouldBlock` and they mean opposite things for a split: a
409    /// hub answering NYET has not finished the transaction and is worth asking
410    /// again, while a NAK ends the pairing and the next attempt has to begin
411    /// with a fresh start-split.
412    fn last_was_nyet(&self) -> bool {
413        self.last_hcint.get() & (1 << 6) != 0
414    }

416    /// `HCCHAR` for this instant, with `ODDFRM` set from the frame the transfer
417    /// will be sent in.
418    ///
419    /// The core latches the parity of the frame a transfer lands in when the
420    /// channel starts, and a transfer whose parity disagrees is refused. The
421    /// frame read here is the one just started, and the transfer goes out in the
422    /// next, which is why the parity is inverted -- the same reading U-Boot
423    /// takes before it starts a channel.
424    #[inline]
425    fn start_hcchar(&self, base: u32) -> u32 {
426        if self.frame_number() & 1 == 0 {
427            base | HCCHAR_ODDFRM
428        } else {
429            base & !HCCHAR_ODDFRM
430        }
431    }

433    /// Start a channel and, behind a hub, run the split handshake that reaches
434    /// the device at all.
435    ///
436    /// Every packet takes two passes over the channel: the start-split hands the
437    /// transaction to the hub, and the complete-split collects what it buffered.
438    /// A hub that has not finished answers the complete-split with NYET, which is
439    /// why that is a retry here rather than a failure.
440    ///
441    /// Without this a full- or low-speed device on a hub port is never reached:
442    /// nothing on the bus answers, and the channel comes back with XACTERR. That
443    /// reads like a protocol error and is really a missing transaction.
444    ///
445    /// `arm(complete)` stages the payload and starts the channel for one pass.
446    fn run_packet(&self, ch: usize, arm: impl Fn(bool)) -> ViResult<()> {
447        if self.split.get().is_none() {
448            arm(false);
449            return self.wait_channel(ch);
450        }
451
452        arm(false);
453        // The yielding wait, which is what this path had when enumeration last
454        // worked. The same-frame rule that made the periodic path's wait short is
455        // a periodic rule -- NetBSD guards it behind INT and ISOC -- and a
456        // non-periodic complete-split is free to go out later: the hub buffers
457        // what its translator fetched, so patience here collects it, and a tight
458        // budget spends the transfer before the translator is done.
459        self.wait_channel(ch)?;
460
461        let started = self.frame_number();
462        trace::record(
463            trace::TAG_ARM_SSPLIT,
464            self.read32(HFNUM),
465            self.read32(hcchar(ch)),
466            self.read32(hcsplt(ch)),
467            self.read32(hctsiz(ch)),
468        );
469        trace::record(
470            trace::TAG_AFTER_SSPLIT,
471            self.last_hcint.get(),
472            self.read32(hcchar(ch)),
473            self.read32(hcsplt(ch)),
474            self.read32(hctsiz(ch)),
475        );
476        for _ in 0..SPLIT_COMPLETE_ATTEMPTS {
477            arm(true);
478            let outcome = self.wait_channel(ch);
479            trace::record(
480                trace::TAG_AFTER_CSPLIT,
481                self.last_hcint.get(),
482                self.read32(hcchar(ch)),
483                self.read32(hcsplt(ch)),
484                self.read32(hctsiz(ch)),
485            );
486            match outcome {
487                // Only a real transfer completion is the result. An ACK here is
488                // the hub acknowledging the request, not delivering it.
489                Ok(()) if self.last_reported_complete() => return Ok(()),
490                Ok(()) => {}
491                Err(ViError::WouldBlock) => {}
492                Err(e) => return Err(e),
493            }
494            if self.frame_number().wrapping_sub(started) & HFNUM_FRNUM_MASK > SPLIT_COMPLETE_FRAMES
495            {
496                trace::dump_once();
497                return Err(ViError::IO);
498            }
499        }
500        trace::dump_once();
501        Err(ViError::IO)
502    }

523    /// Select the payload-transfer mode.
524    ///
525    /// Switching to DMA allocates the scratch region on first use; a failed
526    /// allocation leaves the engine in FIFO mode rather than half-configured.
527    pub fn set_mode(&self, mode: TransferMode) -> bool {
528        if mode == TransferMode::Dma && self.dma.borrow().is_none() {
529            let Some(buf) = DmaBuf::alloc(DMA_SLOT_BYTES * DMA_CHANNELS / 4096) else {
530                return false;
531            };
532            *self.dma.borrow_mut() = Some(buf);
533        }
534        self.mode.set(mode);
535        true
536    }

538    /// Guest-physical base of `ch`'s DMA slot, or `None` outside DMA mode.
539    ///
540    /// In SAS grant pages are identity-mapped, so the value programmed into
541    /// `HCDMA` is the same address the CPU uses.
542    fn dma_slot(&self, ch: usize) -> Option<usize> {
543        if self.mode.get() != TransferMode::Dma || ch >= DMA_CHANNELS {
544            return None;
545        }
546        let guard = self.dma.borrow();
547        guard.as_ref().map(|b| b.phys() + ch * DMA_SLOT_BYTES)
548    }

550    /// Bus address of a RAM physical address on this SoC.
551    ///
552    /// A bus master does not see ARM physical addresses. On the BCM283x the
553    /// Raspberry Pi device tree's `_DMA` resource states the translation
554    /// outright: "Bus 0xC0000000 -> CPU 0x00000000", over the first gigabyte
555    /// and marked NonCacheable. U-Boot's working dwc2 driver programs `HCDMA`
556    /// through exactly this translation (`phys_to_bus`).
557    ///
558    /// Programming the raw ARM physical address asks the core to read and write
559    /// a different location, and the failure is quiet: the address still lands
560    /// in mapped memory, so no AHB error is raised, and the CPU-side read-back
561    /// of the slot looks correct because the CPU and the core are simply
562    /// looking at different bytes.
563    #[inline]
564    fn bus_address(phys: usize) -> u32 {
565        (phys | 0xC000_0000) as u32
566    }

568    /// Point the channel's DMA engine at `offset` bytes into its slot.
569    ///
570    /// Programmed **once per transfer, and always before `CHENA`**. The DWC2
571    /// advances `HCDMA` itself as it moves each packet (`hcdma += actual`), so a
572    /// multi-packet transfer walks the slot on its own; re-arming mid-transfer
573    /// both defeats that walk and races the core's asynchronous completion.
574    ///
575    /// The ordering against `CHENA` is not a preference. The core latches its
576    /// DMA address when the channel is enabled, so a start that arms `HCDMA`
577    /// afterwards -- or never -- sends the first transaction of the transfer to
578    /// whatever address the previous one left behind. On a SETUP that is a
579    /// request the device ACKs and then STALLs one stage later, and the retry
580    /// succeeds because by then the register holds the right address.
581    fn program_hcdma(&self, ch: usize, offset: usize) {
582        if let Some(base) = self.dma_slot(ch) {
583            let addr = base + offset.min(DMA_SLOT_BYTES);
584            self.write32(hcdma(ch), Self::bus_address(addr));
585        }
586    }

588    /// Publish CPU writes in `[offset, offset+len)` to the device.
589    ///
590    /// A refused sync is reported, never skipped: without the clean the core
591    /// reads stale memory, so the device receives a request whose bytes are not
592    /// the ones staged and answers with a STALL -- a failure that looks like a
593    /// protocol problem and is actually a missing cache operation.
594    fn cache_clean(&self, ch: usize, offset: usize, len: usize) {
595        let guard = self.dma.borrow();
596        if let Some(buf) = guard.as_ref() {
597            let start = ch * DMA_SLOT_BYTES + offset.min(DMA_SLOT_BYTES);
598            let end = (start + len).min((ch + 1) * DMA_SLOT_BYTES);
599            let span = end.saturating_sub(start);
600            if span == 0 {
601                // Nothing to publish. The kernel rejects a zero-length sync as
602                // an invalid range, so asking for one turns a no-op into a
603                // reported failure -- and a transfer that legitimately moves no
604                // bytes (an idle bulk-IN poll) would warn on every poll.
605                return;
606            }
607            match buf.begin_cache_sync(start, span) {
608                Some(token) => {
609                    if !buf.complete_cache_sync(token) {
610                        ostd::io::println("[dwc2] WARN: cache sync completion refused");
611                    }
612                }
613                None => {
614                    ostd::io::println(
615                        "[dwc2] WARN: cache sync refused - device may read stale memory",
616                    );
617                }
618            }
619        }
620    }

622    /// Drop stale cache lines over a range the device just wrote.
623    fn cache_invalidate(&self, ch: usize, offset: usize, len: usize) {
624        self.cache_clean(ch, offset, len);

627    /// Copy an outgoing payload into the channel slot at `offset` and publish it.
628    ///
629    /// This is one of the crate's two `allow(unsafe_code)` islands (see
630    /// `scripts/unsafe-allowlist.toml`): the destination is a grant region this
631    /// cell owns for its whole lifetime, `n` is bounded by the slot size and the
632    /// source length, and the region is not aliased by any other live reference.
633    #[allow(unsafe_code)]
634    fn stage_out(&self, ch: usize, offset: usize, data: &[u8]) {
635        let guard = self.dma.borrow();
636        let n = data.len().min(DMA_SLOT_BYTES - offset.min(DMA_SLOT_BYTES));
637        if let Some(buf) = guard.as_ref() {
638            // SAFETY: destination is this cell's grant slot at a bounded
639            // offset; `n` is bounded by the remaining slot and by `data`.
640            unsafe {
641                core::ptr::copy_nonoverlapping(
642                    data.as_ptr(),
643                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset),
644                    n,
645                );
646            }
647        }
648        drop(guard);
649        self.cache_clean(ch, offset, n);
650    }

652    /// The third `allow(unsafe_code)` island: a volatile read of this cell's
653    /// own grant slot at a bounded offset, used only for the trace below.
654    #[allow(unsafe_code)]
655    /// Print the first bytes staged in a channel slot.
656    ///
657    /// Proves what the core is about to hand the device. If these are not the
658    /// request the caller built, the fault is in staging or cache maintenance;
659    /// if they are, the fault is between memory and the bus.
660    pub fn trace_slot(&self, ch: usize, label: &str, len: usize) {
661        let guard = self.dma.borrow();
662        let Some(buf) = guard.as_ref() else {
663            return;
664        };
665        ostd::io::print("[dwc2] slot ");
666        ostd::io::print(label);
667        ostd::io::print(" =");
668        let n = len.min(16).min(DMA_SLOT_BYTES);
669        for i in 0..n {
670            // SAFETY: reading this cell's own grant slot at a bounded offset.
671            let byte = unsafe {
672                core::ptr::read_volatile(
673                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES).wrapping_add(i),
674                )
675            };
676            ostd::io::print(" ");
677            print_hex_val(byte as u32);
678        }
679        ostd::io::println("");
680    }

682    /// Copy a received payload out of the channel slot at `offset`.
683    ///
684    /// The second `allow(unsafe_code)` island: same owned-grant argument as
685    /// [`Self::stage_out`], with `n` additionally bounded by the destination.
686    #[allow(unsafe_code)]
687    fn collect_in(&self, ch: usize, offset: usize, dst: &mut [u8], len: usize) {
688        self.cache_invalidate(ch, offset, len);
689        let guard = self.dma.borrow();
690        if let Some(buf) = guard.as_ref() {
691            let n = len
692                .min(dst.len())
693                .min(DMA_SLOT_BYTES.saturating_sub(offset.min(DMA_SLOT_BYTES)));
694            // SAFETY: source is this cell's grant slot at a bounded offset;
695            // `n` is bounded by the slot remainder and the destination.
696            unsafe {
697                core::ptr::copy_nonoverlapping(
698                    buf.virt().wrapping_add(ch * DMA_SLOT_BYTES + offset) as *const u8,
699                    dst.as_mut_ptr(),
700                    n,
701                );
702            }
703        }
704    }

728    /// Execute a standard 8-byte USB SETUP packet on Channel 0.
729    pub fn send_setup(&self, dev_addr: u8, setup: &[u8; 8]) -> ViResult<()> {
730        let ch = 0;
731        self.prepare_channel(ch);
732        self.write32(hcintmsk(ch), 0x07FF);
733
734        // HCTSIZ: XFERSIZE = 8, PKTCNT = 1, PID = 3 (SETUP).
735        // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1.
736        let sctsiz = 8 | (1 << 19) | (3 << 29);
737        let scchar = (self.control_mps() as u32)
738            | self.device_flags()
739            | (1 << 20)
740            | ((dev_addr as u32) << 22)
741            | (1 << 31);
742        let dma = self.dma_slot(ch).is_some();
743
744        // Stage once up front so the trace has something to show. Each pass
745        // stages again, which is also what hands the bytes to the core.
746        if dma {
747            self.stage_out(ch, 0, setup);
748            static SETUP_TRACED: core::sync::atomic::AtomicBool =
749                core::sync::atomic::AtomicBool::new(false);
750            if !SETUP_TRACED.swap(true, core::sync::atomic::Ordering::Relaxed) {
751                self.trace_slot(ch, "setup", setup.len());
752            }
753        }
754
755        // The payload is staged and HCDMA armed *before* CHENA on every pass:
756        // the core latches its DMA address when the channel starts, so enabling
757        // first reads the previous transfer's address and the device ACKs eight
758        // bytes it never looked at.
759        self.run_packet(ch, |complete| {
760            self.write32(hcsplt(ch), self.hcsplt_value(complete));
761            self.write32(hctsiz(ch), sctsiz);
762            if dma {
763                self.program_hcdma(ch, 0);
764                self.stage_out(ch, 0, setup);
765            } else {
766                self.write_fifo(
767                    ch,
768                    u32::from_le_bytes([setup[0], setup[1], setup[2], setup[3]]),
769                );
770                self.write_fifo(
771                    ch,
772                    u32::from_le_bytes([setup[4], setup[5], setup[6], setup[7]]),
773                );
774            }
775            self.write32(hcint(ch), 0xFFFF_FFFF);
776            self.write32(hcchar(ch), self.start_hcchar(scchar));
777        })
778    }

780    /// Receive the DATA IN phase of a control transfer on Channel 0.
781    ///
782    /// Returns the bytes actually received. Packets are sized by the control
783    /// endpoint's MPS and the transfer ends at the first **short** packet
784    /// (USB 2.0 §8.5.3.1) — a device answering an 18-byte descriptor request
785    /// with its real 8-byte report must not be read as if it had 18 bytes
786    /// available, which is what a fixed-size read would do.
787    pub fn recv_data(&self, dev_addr: u8, buf: &mut [u8]) -> ViResult<usize> {
788        let ch = 0;
789        let mps = (self.control_mps() as usize).max(1);
790        let mut received = 0;
791        let mut toggle = 2u32; // PID 2 = DATA1 for the first data packet
792                               // Arm the DMA engine once; the core advances it per packet.
793        self.program_hcdma(ch, 0);
794
795        while received < buf.len() {
796            let chunk = (buf.len() - received).min(mps);
797            self.prepare_channel(ch);
798            self.write32(hcintmsk(ch), 0x07FF);
799
800            // HCCHAR: EPNUM = 0, EPDIR = 1 (IN), EPTYPE = 0 (Control), MC = 1.
801            let sctsiz = (chunk as u32) | (1 << 19) | (toggle << 29);
802            let scchar = (mps as u32)
803                | self.device_flags()
804                | (1 << 15)
805                | (1 << 20)
806                | ((dev_addr as u32) << 22)
807                | (1 << 31);
808
809            self.run_packet(ch, |complete| {
810                self.write32(hcsplt(ch), self.hcsplt_value(complete));
811                self.write32(hctsiz(ch), sctsiz);
812                // Directly, the address is armed once before the loop and the
813                // core walks the slot itself, so re-arming per packet would
814                // overwrite what it has already written. Behind a hub the core
815                // moves a single packet per pass, so the address is re-armed
816                // every pass and advanced by what has arrived so far.
817                if complete || self.split.get().is_some() {
818                    self.program_hcdma(ch, received);
819                }
820                self.write32(hcint(ch), 0xFFFF_FFFF);
821                self.write32(hcchar(ch), self.start_hcchar(scchar));
822            })?;
823
824            // The core writes the *remaining* count back into HCTSIZ.
825            let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
826            let got = chunk.saturating_sub(remaining).min(chunk);
827
828            let words = got.div_ceil(4);
829            for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
830                for (j, byte) in word.to_le_bytes().iter().enumerate() {
831                    let offset = i * 4 + j;
832                    let idx = received + offset;
833                    if idx < buf.len() && offset < got {
834                        buf[idx] = *byte;
835                    }
836                }
837            }
838
839            received += got;
840            if got < chunk {
841                break;
842            }
843            toggle = if toggle == 2 { 0 } else { 2 }; // DATA1 <-> DATA0
844        }
845
846        if self.dma_slot(ch).is_some() && received > 0 {
847            let mut tmp = [0u8; DMA_SLOT_BYTES];
848            let n = received.min(buf.len()).min(DMA_SLOT_BYTES);
849            self.collect_in(ch, 0, &mut tmp, n);
850            buf[..n].copy_from_slice(&tmp[..n]);
851        }
852
853        Ok(received)
854    }

856    /// Send STATUS handshake on Channel 0 (0-byte packet with DATA1).
857    pub fn send_status(&self, dev_addr: u8, is_in: bool) -> ViResult<()> {
858        let ch = 0;
859        self.prepare_channel(ch);
860        self.write32(hcsplt(ch), 0);
861        self.write32(hcintmsk(ch), 0x07FF);
862        // XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1)
863        // HCTSIZ: XFERSIZE = 0, PKTCNT = 1, PID = 2 (DATA1).
864        // HCCHAR: EPNUM = 0, EPTYPE = 0 (Control), MPS = 64, MC = 1, EPDIR from
865        // the caller.
866        let sctsiz = (1 << 19) | (2 << 29);
867        let epdir = if is_in { 1 } else { 0 };
868        let scchar = (self.control_mps() as u32)
869            | self.device_flags()
870            | (epdir << 15)
871            | (1 << 20)
872            | ((dev_addr as u32) << 22)
873            | (1 << 31);
874
875        self.run_packet(ch, |complete| {
876            self.write32(hcsplt(ch), self.hcsplt_value(complete));
877            self.write32(hctsiz(ch), sctsiz);
878            // A zero-length handshake moves no bytes, but the channel still
879            // latches an address when it starts and the previous transfer's is
880            // not a position this stage may begin from.
881            self.program_hcdma(ch, 0);
882            self.write32(hcint(ch), 0xFFFF_FFFF);
883            self.write32(hcchar(ch), self.start_hcchar(scchar));
884        })
885    }

887    /// Execute a complete synchronous USB Control Transfer.
888    pub fn control_transfer(
889        &self,
890        dev_addr: u8,
891        req_type: u8,
892        req: u8,
893        val: u16,
894        idx: u16,
895        buf: &mut [u8],
896    ) -> ViResult<usize> {
897        let is_in = (req_type & 0x80) != 0;
898        let length = buf.len() as u16;
899
900        let setup = [
901            req_type,
902            req,
903            (val & 0xFF) as u8,
904            ((val >> 8) & 0xFF) as u8,
905            (idx & 0xFF) as u8,
906            ((idx >> 8) & 0xFF) as u8,
907            (length & 0xFF) as u8,
908            ((length >> 8) & 0xFF) as u8,
909        ];
910
911        // Each phase reports its own failure: a control transfer that fails
912        // silently is the hardest kind of USB bug to find, and the request
913        // fields are the only thing that identifies which one it was.
914        if let Err(e) = self.send_setup(dev_addr, &setup) {
915            self.report_failure("setup", dev_addr, req_type, req, val, idx, length, e);
916            return Err(e);
917        }
918
919        // 2. DATA phase (optional)
920        let mut actual = 0;
921        if length > 0 {
922            let result = if is_in {
923                self.recv_data(dev_addr, buf)
924            } else {
925                self.send_data(dev_addr, buf).map(|()| buf.len())
926            };
927            match result {
928                Ok(n) => actual = n,
929                Err(e) => {
930                    self.report_failure("data", dev_addr, req_type, req, val, idx, length, e);
931                    return Err(e);
932                }
933            }
934        }
935
936        // 3. STATUS phase (handshake in opposite direction)
937        if let Err(e) = self.send_status(dev_addr, !is_in) {
938            self.report_failure("status", dev_addr, req_type, req, val, idx, length, e);
939            return Err(e);
940        }
941
942        Ok(actual)
943    }

985    /// Send DATA OUT phase on Channel 0.
986    fn send_data(&self, dev_addr: u8, data: &[u8]) -> ViResult<()> {
987        let ch = 0;
988        let mps = (self.control_mps() as usize).max(1);
989        let mut sent = 0;
990        let mut toggle = 2; // PID 2 = DATA1
991                            // Stage the whole payload once and arm DMA once: the core reads it
992                            // packet by packet, advancing HCDMA itself.
993        if self.dma_slot(ch).is_some() {
994            self.stage_out(ch, 0, data);
995            self.program_hcdma(ch, 0);
996        }
997
998        while sent < data.len() {
999            let chunk = (data.len() - sent).min(mps);
1000            self.prepare_channel(ch);
1001            self.write32(hcsplt(ch), 0);
1002            self.write32(hcintmsk(ch), 0x07FF);
1003            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
1004            self.write32(hctsiz(ch), sctsiz);
1005
1006            // HCCHAR: EPNUM = 0, EPDIR = 0 (OUT), EPTYPE = 0 (Control), MC = 1, CHENA = 1.
1007            let scchar = (mps as u32)
1008                | self.device_flags()
1009                | (1 << 20)
1010                | ((dev_addr as u32) << 22)
1011                | (1 << 31);
1012            self.write32(hcchar(ch), self.start_hcchar(scchar));
1013
1014            if self.dma_slot(ch).is_some() {
1015                // Payload already staged and HCDMA already armed.
1016            } else {
1017                let words = chunk.div_ceil(4);
1018                for i in 0..words {
1019                    let mut b = [0u8; 4];
1020                    for (j, byte) in b.iter_mut().enumerate() {
1021                        let offset = i * 4 + j;
1022                        if sent + offset < data.len() && offset < chunk {
1023                            *byte = data[sent + offset];
1024                        }
1025                    }
1026                    self.write_fifo(ch, u32::from_le_bytes(b));
1027                }
1028            }
1029
1030            self.wait_channel(ch)?;
1031
1032            sent += chunk;
1033            toggle = if toggle == 2 { 0 } else { 2 };
1034        }
1035
1036        Ok(())
1037    }

1039    /// Transmit a raw Ethernet packet via Bulk OUT (Channel 2, EP 2).
1040    pub fn bulk_transmit(&self, dev_addr: u8, ep_num: u8, packet: &[u8]) -> ViResult<()> {
1041        let ch = 2;
1042        let mut sent = 0;
1043        let mut toggle = 0; // Starts at DATA0
1044
1045        let use_dma = self.dma_slot(ch).is_some();
1046        if use_dma {
1047            self.stage_out(ch, 0, packet);
1048            self.program_hcdma(ch, 0);
1049        }
1050        while sent < packet.len() {
1051            let chunk = (packet.len() - sent).min(512); // 512 bytes for High-Speed Bulk
1052
1053            let sctsiz = (chunk as u32) | (1 << 19) | ((toggle as u32) << 29);
1054            // HCCHAR: EPDIR = 0 (OUT), EPTYPE = 2 (Bulk), MC = 1 packet, MPS = 512 (HS Bulk).
1055            let scchar = 512
1056                | ((ep_num as u32) << 11)
1057                | (2 << 18)
1058                | (1 << 20)
1059                | ((dev_addr as u32) << 22)
1060                | (1 << 31);
1061
1062            let mut retries = 0;
1063            loop {
1064                self.prepare_channel(ch);
1065                self.write32(hcsplt(ch), 0);
1066                self.write32(hcintmsk(ch), 0x07FF);
1067                self.write32(hctsiz(ch), sctsiz);
1068                self.write32(hcchar(ch), self.start_hcchar(scchar));
1069
1070                if use_dma {
1071                    // Staged and armed before the loop; the core walks it.
1072                } else {
1073                    let words_count = chunk.div_ceil(4);
1074                    for i in 0..words_count {
1075                        let mut b = [0u8; 4];
1076                        for (j, byte) in b.iter_mut().enumerate() {
1077                            let offset = i * 4 + j;
1078                            let idx = sent + offset;
1079                            if idx < packet.len() && offset < chunk {
1080                                *byte = packet[idx];
1081                            }
1082                        }
1083                        self.write_fifo(ch, u32::from_le_bytes(b));
1084                    }
1085                }
1086
1087                match self.wait_channel(ch) {
1088                    Ok(()) => break,
1089                    Err(ViError::WouldBlock) => {
1090                        retries += 1;
1091                        if retries > 50 {
1092                            ostd::io::println("[dwc2] bulk_transmit: exceeded 50 NAK retries");
1093                            return Err(ViError::IO);
1094                        }
1095                        sys_yield();
1096                    }
1097                    Err(e) => return Err(e),
1098                }
1099            }
1100
1101            sent += chunk;
1102            toggle = if toggle == 0 { 2 } else { 0 };
1103        }
1104
1105        Ok(())
1106    }

1108    /// Receive a raw packet via Bulk IN (Channel 1, EP 1). Returns received length or 0 if nothing.
1109    pub fn bulk_receive(&self, dev_addr: u8, ep_num: u8, buf: &mut [u8]) -> ViResult<usize> {
1110        let ch = 1;
1111        self.prepare_channel(ch);
1112        self.write32(hcsplt(ch), 0);
1113        self.write32(hcintmsk(ch), 0x07FF);
1114        let want = buf.len().min(512);
1115        let sctsiz = (want as u32) | (1 << 19);
1116        self.write32(hctsiz(ch), sctsiz);
1117
1118        let scchar = 512
1119            | ((ep_num as u32) << 11)
1120            | (1 << 15) // IN
1121            | (2 << 18) // Bulk
1122            | (1 << 20) // MC = 1
1123            | ((dev_addr as u32) << 22)
1124            | (1 << 31);
1125        self.program_hcdma(ch, 0);
1126        self.write32(hcchar(ch), self.start_hcchar(scchar));
1127
1128        // Non-blocking wait: check if transfer completed or NAK
1129        let mut count = 0;
1130        while count < 1000 {
1131            let int = self.read32(hcint(ch));
1132            if int & (1 << 0) != 0 {
1133                // The core reports the actual length in HCTSIZ either way, but
1134                // only DMA deposits the payload outside the FIFO.
1135                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1136                let got = want.saturating_sub(remaining);
1137                if self.dma_slot(ch).is_some() {
1138                    self.collect_in(ch, 0, buf, got);
1139                } else {
1140                    let words = want.div_ceil(4);
1141                    for (i, word) in (0..words).map(|i| (i, self.read_fifo(ch))) {
1142                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
1143                            let idx = i * 4 + j;
1144                            if idx < buf.len() && idx < want {
1145                                buf[idx] = *byte;
1146                            }
1147                        }
1148                    }
1149                }
1150                return Ok(got);
1151            }
1152            if int & (1 << 4) != 0 {
1153                // NAK: device has no packet right now
1154                return Ok(0);
1155            }
1156            count += 1;
1157            sys_yield();
1158        }
1159
1160        Ok(0)
1161    }

1163    /// Receive up to one packet via Interrupt IN on `ch`.
1164    ///
1165    /// HID input reports ride interrupt endpoints. The device NAKs until it has
1166    /// a report, so a NAK is a normal "no data yet" answer, not an error — the
1167    /// caller polls. Returns the *actual* transferred length, read back from
1168    /// `HCTSIZ` rather than assumed: a HID report can be shorter than the buffer
1169    /// (an 8-byte keyboard report into a 64-byte buffer), and using the request
1170    /// length would hand the decoder trailing garbage.
1171    ///
1172    /// `ch` must not collide with the control (0) or Ethernet bulk (1, 2)
1173    /// channels.
1174    pub fn interrupt_receive(
1175        &self,
1176        ch: usize,
1177        dev_addr: u8,
1178        ep_num: u8,
1179        mps: u16,
1180        buf: &mut [u8],
1181        split_pending: &mut bool,
1182    ) -> ViResult<usize> {
1183        let want = buf.len().min(mps.max(1) as usize).min(1024);
1184        if want == 0 {
1185            return Ok(0);
1186        }
1187
1188        // Behind a hub the poll is a pair of transactions, one half per call.
1189        if self.split.get().is_some() {
1190            return self.poll_split(ch, dev_addr, ep_num, mps, want, buf, split_pending);
1191        }
1192
1193        self.prepare_channel(ch);
1194        self.write32(hcsplt(ch), 0);
1195        self.write32(hcintmsk(ch), 0x07FF);
1196
1197        // HCTSIZ: XFERSIZE (= want), PKTCNT = 1, PID = DATA0 (0). The core
1198        // overwrites XFERSIZE with the remaining count as it transfers.
1199        let sctsiz = (want as u32) | (1 << 19);
1200        self.write32(hctsiz(ch), sctsiz);
1201
1202        // HCCHAR: MPS from the endpoint descriptor, EPNUM, IN direction, DEVCTL,
1203        // and EPTYPE = Interrupt, which is what the endpoint is.
1204        //
1205        // EPTYPE was briefly Bulk here, to move the transfer out of the core's
1206        // periodic class while it was thought the missing periodic frame list was
1207        // the problem. It is not: a low-speed device has no bulk endpoints at all.
1208        // USB 2.0 gives low speed control and interrupt transfers and nothing else,
1209        // and the hub's translator refuses a bulk transaction aimed at one -- which
1210        // the trace shows as a STALL on the very first poll of both endpoints, long
1211        // before the device has had any reason to refuse anything.
1212        let scchar = (mps as u32 & 0x7FF)
1213            | ((ep_num as u32) << 11)
1214            | self.device_flags()
1215            | (1 << 15) // EPDIR = IN
1216            | (3 << 18) // EPTYPE = Interrupt: the only type a low-speed endpoint has
1217            | (1 << 20) // MC = 1
1218            | ((dev_addr as u32) << 22)
1219            | (1 << 31); // CHENA
1220        self.program_hcdma(ch, 0);
1221        self.write32(hcchar(ch), self.start_hcchar(scchar));
1222
1223        let mut polls = 0u32;
1224        while polls < 2_000 {
1225            let int = self.read32(hcint(ch));
1226
1227            // NAK — device has no report queued. Halt and report "no data".
1228            if int & (1 << 4) != 0 {
1229                self.halt_channel(ch);
1230                return Ok(0);
1231            }
1232            // Errors (STALL / babble / transaction error) end the poll cycle.
1233            if int & ((1 << 2) | (1 << 3) | (1 << 7)) != 0 {
1234                self.halt_channel(ch);
1235                return Err(ViError::IO);
1236            }
1237
1238            let complete = int & (1 << 0) != 0 || int & (1 << 1) != 0 || int & (1 << 5) != 0;
1239            if complete {
1240                let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1241                let got = want.saturating_sub(remaining);
1242                self.halt_channel(ch);
1243
1244                if got > 0 {
1245                    if self.dma_slot(ch).is_some() {
1246                        // DMA deposited the report in the channel slot.
1247                        let mut tmp = [0u8; DMA_SLOT_BYTES];
1248                        self.collect_in(ch, 0, &mut tmp, got);
1249                        let n = got.min(buf.len());
1250                        buf[..n].copy_from_slice(&tmp[..n]);
1251                    } else {
1252                        // Drain ceil(got/4) FIFO words; only the first `got`
1253                        // bytes are meaningful (the tail word is padding).
1254                        let words = got.div_ceil(4);
1255                        for w in 0..words {
1256                            let word = self.read_fifo(ch);
1257                            for (j, byte) in word.to_le_bytes().iter().enumerate() {
1258                                let idx = w * 4 + j;
1259                                if idx < got && idx < buf.len() {
1260                                    buf[idx] = *byte;
1261                                }
1262                            }
1263                        }
1264                    }
1265                } else if self.dma_slot(ch).is_none() {
1266                    // A zero-length packet still has a FIFO word to retire on
1267                    // some revisions; drain one so the channel is clean.
1268                    let _ = self.read_fifo(ch);
1269                }
1270                return Ok(got);
1271            }
1272
1273            polls += 1;
1274            sys_yield();
1275        }
1276
1277        self.halt_channel(ch);
1278        Ok(0)
1279    }

1281    /// One half of an interrupt IN split, for a device behind a hub.
1282    ///
1283    /// A split is a pair of transactions the hub has to see in different
1284    /// microframes: a start-split that hands it the request, and a complete-split
1285    /// issued later that collects the result. One half is issued per call, with
1286    /// the pairing carried in `pending`, so the gap between them is the gap
1287    /// between two polls rather than a wait taken here.
1288    ///
1289    /// Waiting here is what this used to do, and it cost the machine: `HFNUM`
1290    /// counts frames of a millisecond, not microframes, so crossing one boundary
1291    /// meant spinning for a whole millisecond of system calls -- every poll, on
1292    /// every interface, exactly when a device had something to say.
1293    ///
1294    /// The three ways a complete-split can end mean different things:
1295    ///
1296    /// * `XFERCOMPL` is the report, and ends the pairing;
1297    /// * `NYET` is the hub still working, so the pairing stays live;
1298    /// * `NAK` ends the pairing with nothing.
1299    ///
1300    /// ACK carries no data here -- Linux is explicit that it "should not occur in
1301    /// CSPLIT" -- so it is not taken for a result.
1302    #[allow(clippy::too_many_arguments)]
1303    fn poll_split(
1304        &self,
1305        ch: usize,
1306        dev_addr: u8,
1307        ep_num: u8,
1308        mps: u16,
1309        want: usize,
1310        buf: &mut [u8],
1311        pending: &mut bool,
1312    ) -> ViResult<usize> {
1313        self.prepare_channel(ch);
1314        self.write32(hcintmsk(ch), 0x07FF);
1315
1316        // HCTSIZ: XFERSIZE = one max packet, PKTCNT = 1, PID = DATA0. A split
1317        // carries exactly one packet, whatever the caller asked for.
1318        //
1319        // HCCHAR: MPS, EPNUM, IN, and EPTYPE = Interrupt, for the same reason as
1320        // the channel above: a low-speed endpoint has no other type.
1321        let sctsiz = (want as u32) | (1 << 19);
1322        let scchar = (mps as u32 & 0x7FF)
1323            | ((ep_num as u32) << 11)
1324            | self.device_flags()
1325            | (1 << 15)
1326            | (3 << 18)
1327            | (3 << 20)
1328            | ((dev_addr as u32) << 22)
1329            | (1 << 31);
1330        let arm = |complete: bool| {
1331            self.write32(hcsplt(ch), self.hcsplt_value(complete));
1332            self.write32(hctsiz(ch), sctsiz);
1333            self.program_hcdma(ch, 0);
1334            self.write32(hcint(ch), 0xFFFF_FFFF);
1335            self.write32(hcchar(ch), self.start_hcchar(scchar));
1336        };
1337
1338        // Both halves are issued here, one microframe apart and inside a single
1339        // frame. The second half cannot be handed to a later poll: this loop's
1340        // RecvTimeout puts hundreds of frames between polls and the hub pairs a
1341        // split for less than one, which is what made every complete-split here
1342        // answer NYET.
1343        // Both halves are issued here, one microframe apart and inside a single
1344        // frame, and nothing is written to the console between them. A line of
1345        // console output costs about seven milliseconds at this baud, which is
1346        // seven of the frames the hub pairs a split within -- printing the first
1347        // half before issuing the second is what kept the pair apart. Every report
1348        // below therefore sits on a path where no further channel work depends on
1349        // it, and the retry path has none at all.
1350        // The start-split opens the frame the hub will run the transaction in, so
1351        // it is issued at the frame's start; the complete-split below collects the
1352        // result at the frame's end.
1353        if !self.await_frame_start() {
1354            *pending = false;
1355            return Ok(0);
1356        }
1357
1358        let frame = self.frame_number();
1359        trace::record(
1360            trace::TAG_ARM_SSPLIT,
1361            self.read32(HFNUM),
1362            self.read32(hcchar(ch)),
1363            self.read32(hcsplt(ch)),
1364            self.read32(hctsiz(ch)),
1365        );
1366        arm(false);
1367        let outcome = self.wait_channel_spin(ch, SPIN_POLLS);
1368        trace::record(
1369            trace::TAG_AFTER_SSPLIT,
1370            self.last_hcint.get(),
1371            self.read32(hcchar(ch)),
1372            self.read32(hcsplt(ch)),
1373            self.read32(hctsiz(ch)),
1374        );
1375        if !matches!(outcome, Ok(())) {
1376            *pending = false;
1377            report_split_progress(
1378                false,
1379                &outcome,
1380                self.last_hcint.get(),
1381                self.last_was_nyet(),
1382                want,
1383                frame,
1384            );
1385            return match outcome {
1386                Err(e) => Err(e),
1387                _ => Ok(0),
1388            };
1389        }
1390
1391        for _ in 0..SPLIT_ATTEMPTS {
1392            if !self.await_frame_end() {
1393                // The frame moved, so the pairing is gone whether or not the hub
1394                // ever answered.
1395                *pending = false;
1396                return Ok(0);
1397            }
1398
1399            trace::record(
1400                trace::TAG_ARM_CSPLIT,
1401                self.read32(HFNUM),
1402                self.read32(hcchar(ch)),
1403                self.read32(hcsplt(ch)),
1404                self.read32(hctsiz(ch)),
1405            );
1406            arm(true);
1407            let outcome = self.wait_channel_spin(ch, SPIN_POLLS);
1408            let int = self.last_hcint.get();
1409            trace::record(
1410                trace::TAG_AFTER_CSPLIT,
1411                int,
1412                self.read32(hcchar(ch)),
1413                self.read32(hcsplt(ch)),
1414                self.read32(hctsiz(ch)),
1415            );
1416
1417            match outcome {
1418                Ok(()) if !self.last_reported_complete() => {
1419                    // ACK on a complete-split carries no data.
1420                    *pending = false;
1421                    report_split_progress(true, &outcome, int, self.last_was_nyet(), want, frame);
1422                    trace::dump_once();
1423                    return Ok(0);
1424                }
1425                Ok(()) => {
1426                    *pending = false;
1427                    let remaining = (self.read32(hctsiz(ch)) & 0x7FFFF) as usize;
1428                    let got = want.saturating_sub(remaining);
1429                    if got == 0 {
1430                        report_split_progress(
1431                            true,
1432                            &outcome,
1433                            int,
1434                            self.last_was_nyet(),
1435                            want,
1436                            frame,
1437                        );
1438                        return Ok(0);
1439                    }
1440                    if self.dma_slot(ch).is_some() {
1441                        let mut tmp = [0u8; DMA_SLOT_BYTES];
1442                        self.collect_in(ch, 0, &mut tmp, got.min(DMA_SLOT_BYTES));
1443                        let n = got.min(buf.len());
1444                        buf[..n].copy_from_slice(&tmp[..n]);
1445                        return Ok(n);
1446                    }
1447                    let words = got.div_ceil(4);
1448                    for w in 0..words {
1449                        let word = self.read_fifo(ch);
1450                        for (j, byte) in word.to_le_bytes().iter().enumerate() {
1451                            let idx = w * 4 + j;
1452                            if idx < got && idx < buf.len() {
1453                                buf[idx] = *byte;
1454                            }
1455                        }
1456                    }
1457                    return Ok(got.min(buf.len()));
1458                }
1459                // Still working, or NAK which ends the pairing. Either way the
1460                // next poll starts over rather than asking again here -- and a
1461                // hub that answered NYET is asked again below, with nothing
1462                // written in between for the same reason as above.
1463                Err(ViError::WouldBlock) => {
1464                    if !self.last_was_nyet() {
1465                        *pending = false;
1466                        report_split_progress(
1467                            true,
1468                            &outcome,
1469                            int,
1470                            self.last_was_nyet(),
1471                            want,
1472                            frame,
1473                        );
1474                        return Ok(0);
1475                    }
1476                }
1477                Err(e) => {
1478                    *pending = false;
1479                    report_split_progress(true, &outcome, int, self.last_was_nyet(), want, frame);
1480                    return Err(e);
1481                }
1482            }
1483        }
1484
1485        // The hub never finished. Starting over on the next poll is the only
1486        // thing left: the frame this pairing belonged to is gone with it.
1487        *pending = false;
1488        trace::dump_once();
1489        Ok(0)
1490    }

1492    /// Name the host-channel interrupt bit that ended a transfer.
1493    ///
1494    /// The DWC2 encodes one cause per bit and they need different fixes:
1495    /// `AHBERR` is the core failing to reach memory at `HCDMA`, `XACTERR` is a
1496    /// bus protocol error, `BBLERR` is the device sending more than the
1497    /// programmed packet size, `DTERR` is a data-toggle mismatch, and `STALL`
1498    /// is the device itself refusing. Collapsing them into one "IO error" hides
1499    /// which of those actually happened.
1500    fn describe_hcint(&self, int: u32) -> &'static str {
1501        if int & (1 << 2) != 0 {
1502            return "AHBERR - core could not reach memory at HCDMA";
1503        }
1504        if int & (1 << 8) != 0 {
1505            return "BBLERR - babble, device exceeded the packet size";
1506        }
1507        if int & (1 << 3) != 0 {
1508            return "STALL - endpoint refused the request";
1509        }
1510        if int & (1 << 7) != 0 {
1511            return "XACTERR - transaction error";
1512        }
1513        if int & (1 << 10) != 0 {
1514            return "DTERR - data toggle mismatch";
1515        }
1516        if int & (1 << 9) != 0 {
1517            return "FRMOVRN - frame overrun";
1518        }
1519        if int & (1 << 6) != 0 {
1520            return "NYET - not ready";
1521        }
1522        if int & (1 << 4) != 0 {
1523            return "NAK";
1524        }
1525        "unknown"
1526    }

1599    /// Wait for channel completion or error, spending at most `budget` polls.
1600    ///
1601    /// The budget is a real cost rather than a formality: every poll yields, and
1602    /// a transfer that never reports anything spends the whole of it before it is
1603    /// given up on. Callers on the driver's own serving thread want a small one.
1604    fn wait_channel_with(&self, ch: usize, budget: usize) -> ViResult<()> {
1605        for _ in 0..budget {
1606            if let Some(outcome) = self.channel_outcome(ch) {
1607                return outcome;
1608            }
1609            sys_yield();
1610        }
1611        self.channel_timeout(ch)
1612    }

1614    /// Wait for a channel without handing the CPU away.
1615    ///
1616    /// A periodic split has to keep both halves inside one millisecond frame. The
1617    /// yielding wait gives the CPU away for around twenty milliseconds per poll,
1618    /// which put the complete-split twenty frames past the start-split it belongs
1619    /// to -- outside the window the hub pairs them within, so it came back as a
1620    /// transaction error however many times it was tried.
1621    ///
1622    /// The bound is the frame itself rather than a count: past that the pairing is
1623    /// gone and starting over on the next poll is the only useful thing left. This
1624    /// runs only when a device has something to report.
1625    /// The channel is left to report for itself. The frame check this used to
1626    /// carry halted the channel the moment the frame turned over, which is exactly
1627    /// when a full- or low-speed transaction behind a hub is finishing -- and the
1628    /// halt it issued was then read straight back as the outcome, so every split
1629    /// came back as a halt carrying no status:
1630    ///
1631    ///     half=ssplit outcome=NAK hcint=0x00000002
1632    ///     half=csplit outcome=NAK hcint=0x00000002
1633    ///
1634    /// The core halts a periodic channel at its own frame boundary and says so, so
1635    /// nothing here needs to do it, and nothing here should.
1636    fn wait_channel_spin(&self, ch: usize, polls: usize) -> ViResult<()> {
1637        for _ in 0..polls {
1638            if let Some(outcome) = self.channel_outcome(ch) {
1639                return outcome;
1640            }
1641        }
1642        self.channel_timeout(ch)
1643    }

1645    /// Classify a channel's interrupt register, or `None` while it is still running.
1646    fn channel_outcome(&self, ch: usize) -> Option<ViResult<()>> {
1647        let int = self.read32(hcint(ch));
1648        if int == 0 {
1649            return None;
1650        }
1651
1652        // ── Hardware-generated halt ───────────────────────────────
1653        if int & (1 << 1) != 0 {
1654            // CHHLTD
1655            self.write32(hcint(ch), 0xFFFF_FFFF);
1656            // Babble (8) and data-toggle error (10) end a transfer just as
1657            // surely as the bits that were already checked.
1658            if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
1659                self.report_channel_error(ch, int);
1660                self.last_hcint.set(int);
1661                return Some(Err(ViError::IO));
1662            }
1663            // NAK and NYET both mean "ask again", and both have to be reported
1664            // rather than folded into success: a hub answering a complete-split
1665            // with NYET has not produced a result yet, and returning Ok there
1666            // hands the caller an empty buffer as if the transfer had happened.
1667            if int & ((1 << 4) | (1 << 6)) != 0 {
1668                self.last_hcint.set(int);
1669                return Some(Err(ViError::WouldBlock));
1670            }
1671            // A halt carrying nothing else is not a result. CHHLTD says the channel
1672            // stopped -- at a frame boundary, or because it was disabled -- and
1673            // answering "done" to it hands the caller an empty buffer for a transfer
1674            // that never completed. Linux is explicit that the only completion is
1675            // XFERCOMPL, with ACK belonging to the start-split.
1676            if int & ((1 << 0) | (1 << 5)) != 0 {
1677                self.last_hcint.set(int);
1678                return Some(Ok(()));
1679            }
1680            self.last_hcint.set(int);
1681            return Some(Err(ViError::WouldBlock));
1682        }
1683
1684        // ── Transfer complete ─────────────────────────────────────
1685        if int & (1 << 0) != 0 {
1686            self.halt_channel(ch);
1687            self.last_hcint.set(int);
1688            return Some(Ok(()));
1689        }
1690
1691        // ── ACK = the packet was accepted
1692        //
1693        // No halt here. On a split, an ACK to the start-split means the hub has
1694        // taken the transaction and its translator is running it -- the pair is
1695        // live at that moment and the complete-split is what collects it.
1696        // Disabling the channel in between asks the core to abandon a transfer
1697        // that is still in flight, and the trace shows where that lands: the
1698        // channel is armed for the complete-split with CHDIS already set, and it
1699        // comes back halted with no status at all, on the very first poll, on both
1700        // endpoints, before the device has had any chance to refuse anything. A
1701        // transfer the core has finished needs no halt; a transfer being abandoned
1702        // gets one from its caller.
1703        if int & (1 << 5) != 0 {
1704            // A transfer that is not part of a split is over when the packet is
1705            // accepted, and halting it is the ordinary end of it. A split is a
1706            // different thing: the ACK belongs to the start-split, the hub's
1707            // translator is running the transaction because of it, and the pair is
1708            // live. Disabling the channel in that moment asks the core to abandon
1709            // a transfer that is still in flight, and the trace shows where that
1710            // lands -- the channel armed for the complete-split with CHDIS already
1711            // set, coming back halted with no status at all, on the very first
1712            // poll, on both endpoints, before the device has refused anything.
1713            if self.split.get().is_none() {
1714                self.halt_channel(ch);
1715            }
1716            self.last_hcint.set(int);
1717            return Some(Ok(()));
1718        }
1719
1720        // ── Error conditions ──────────────────────────────────────
1721        if int & ((1 << 2) | (1 << 3) | (1 << 7) | (1 << 8) | (1 << 10)) != 0 {
1722            self.report_channel_error(ch, int);
1723            self.halt_channel(ch);
1724            self.last_hcint.set(int);
1725            return Some(Err(ViError::IO));
1726        }
1727        if int & ((1 << 4) | (1 << 6)) != 0 {
1728            self.halt_channel(ch);
1729            self.last_hcint.set(int);
1730            return Some(Err(ViError::WouldBlock));
1731        }
1732
1733        None
1734    }
1735
1736    /// Give up on a channel that reported nothing within its budget.
1737    fn channel_timeout(&self, ch: usize) -> ViResult<()> {
1738        // Timeout — capture hcint BEFORE halt clears it
1739        let int = self.read32(hcint(ch));
1740        let char_val = self.read32(hcchar(ch));
1741        self.halt_channel(ch);
1742        static TIMEOUT_COUNT: core::sync::atomic::AtomicUsize =
1743            core::sync::atomic::AtomicUsize::new(0);
1744        let attempt = TIMEOUT_COUNT.fetch_add(1, core::sync::atomic::Ordering::Relaxed);
1745        if attempt < 3 {
1746            ostd::io::print("[dwc2] TIMEOUT ch=");
1747            print_hex_val(ch as u32);
1748            ostd::io::print(" hcint=0x");
1749            print_hex_val(int);
1750            ostd::io::print(" hcchar=0x");
1751            print_hex_val(char_val);
1752            ostd::io::println("");
1753            self.dump_transfer_state(ch, "timeout");
1754        }
1755        self.last_hcint.set(int);
1756        Err(ViError::IO)
1757    }
1758
1759    /// Explicitly halt a host channel (required in DWC2 Slave mode).
1760    ///
1761    /// `CHDIS` and `CHENA` must never be set together. The core reads
1762    /// `CHENA=1` as a fresh enable, so raising it while requesting a disable
1763    /// **re-arms the channel with the parameters still in `HCCHAR`/`HCTSIZ`** --
1764    /// a second, unwanted transaction launched from a completed one. Only
1765    /// `CHDIS` is raised; the core clears `CHENA` and reports `CHHLTD`.
1766    fn halt_channel(&self, ch: usize) {
1767        let reg = hcchar(ch);
1768        let mut val = self.read32(reg);
1769        val |= 1 << 30; // CHDIS
1770        val &= !(1 << 31); // CHENA
1771        self.write32(reg, val);
1772        // Wait for CHHLTD (bit 1), and only then clear it. Clearing the interrupt
1773        // unconditionally returns while the core is still finishing the halt, and
1774        // the CHHLTD it sets afterwards survives into the next arm -- where the
1775        // wait reads it and reports a halt that belongs to the transfer before.
1776        // The trace made that plain: the records alternate between an ACK, whose
1777        // channel carries CHDIS because this function set it, and a halt whose
1778        // channel carries no CHDIS at all, which is this function's own halt
1779        // arriving late.
1780        //
1781        // The wait reads the register rather than yielding: a yield is around
1782        // twenty milliseconds, and four of them is the whole budget a split has.
1783        for _ in 0..CHANNEL_HALT_POLLS {
1784            if self.read32(hcint(ch)) & (1 << 1) != 0 {
1785                break;
1786            }
1787        }
1788        self.write32(hcint(ch), 0xFFFF_FFFF);
1789    }
1790}

1792/// Report the first split poll in full, once.
1793///
1794/// Report how each half of a split ended, a few times per half.
1795///
1796/// Which half ran and how it ended is the whole question when a split does not
1797/// deliver: a start-split the hub refuses, a complete-split it is not ready for,
1798/// and a complete-split that ends without data are three different faults, and
1799/// nothing else in the driver tells them apart. Both halves are reported
1800/// separately -- a single shared flag reports one half and hides the other, which
1801/// is exactly the one worth seeing. The frame number comes along so the gap
1802/// between the halves is visible rather than inferred.
1803fn report_split_progress(
1804    complete: bool,
1805    outcome: &ViResult<()>,
1806    hcint: u32,
1807    nyet: bool,
1808    want: usize,
1809    frame: u32,
1810) {
1811    static SEEN: [core::sync::atomic::AtomicUsize; 2] =
1812        [const { core::sync::atomic::AtomicUsize::new(0) }; 2];
1813    let slot = usize::from(complete);
1814    if SEEN[slot].fetch_add(1, core::sync::atomic::Ordering::Relaxed) >= 3 {
1815        return;
1816    }
1817    ostd::io::print("[dwc2] split: half=");
1818    ostd::io::print(if complete { "csplit" } else { "ssplit" });
1819    ostd::io::print(" outcome=");
1820    match outcome {
1821        Ok(()) => ostd::io::print("Ok"),
1822        Err(ViError::WouldBlock) if nyet => ostd::io::print("NYET"),
1823        Err(ViError::WouldBlock) => ostd::io::print("NAK"),
1824        Err(_) => ostd::io::print("error"),
1825    }
1826    ostd::io::print(" hcint=0x");
1827    print_hex_val(hcint);
1828    ostd::io::print(" frame=0x");
1829    print_hex_val(frame);
1830    ostd::io::print(" want=");
1831    print_usize_val(want);
1832    ostd::io::println("");
1833}

1835fn print_usize_val(v: usize) {
1836    let mut out = [0u8; 20];
1837    let mut n = v;
1838    let mut len = 0;
1839    if n == 0 {
1840        ostd::io::print("0");
1841        return;
1842    }
1843    while n > 0 && len < out.len() {
1844        out[len] = b'0' + (n % 10) as u8;
1845        n /= 10;
1846        len += 1;
1847    }
1848    out[..len].reverse();
1849    if let Ok(s) = core::str::from_utf8(&out[..len]) {
1850        ostd::io::print(s);
1851    }
1852}
