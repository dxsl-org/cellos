1//! USB HID class driver over the DWC2 host controller.
2//!
3//! Brings up whatever HID device is attached to a LAN9514 downstream port and
4//! feeds its reports into the input service. The full path is:
5//!
6//! ```text
7//!   port connect → SET_ADDRESS → GET_DESCRIPTOR(DEVICE)
8//!                → GET_DESCRIPTOR(CONFIGURATION) → walk interfaces
9//!                → per HID interface: GET_DESCRIPTOR(REPORT) → parse
10//!                → SET_CONFIGURATION → SET_IDLE(0) → SET_PROTOCOL
11//!                → poll Interrupt IN → decode → forward to input service
12//! ```
13//!
14//! **Composite devices.** A wireless receiver presents several interfaces — its
15//! keyboard and its mouse — and each carries its own report descriptor and
16//! interrupt endpoint. They are enumerated independently and polled round-robin;
17//! treating only the first interface would surface a keyboard with no pointer
18//! (or the reverse), which is the usual failure mode for combo dongles.
19//!
20//! **Report vs boot protocol.** The descriptor path runs when a report
21//! descriptor parses, because it is the only way to decode a device that uses
22//! report IDs or a non-standard layout. Boot protocol (fixed 8-byte keyboard /
23//! 3-byte mouse) is the fallback for a descriptor that is absent or unparseable,
24//! and it is what most combo receivers implement anyway so a BIOS can use them.
25

40/// First host channel reserved for HID. Channels 0-2 are control and the
41/// Ethernet bulk pair, so HID devices take 3..=6 (four concurrently attached).
42pub const HID_CHANNEL_BASE: usize = 3;
43/// Maximum HID interfaces driven at once across all ports.
44pub const MAX_HID_INTERFACES: usize = 4;
45/// Buffer size for one interrupt IN report.
46const REPORT_BUF: usize = 64;
47/// Control-transfer attempts before an enumeration stage gives up.
48///
49/// A port reset and a mode switch are both followed by a brief window where the
50/// first SETUP is lost, so one attempt is not enough to tell a dead device from
51/// a slow one.
52const ENUM_ATTEMPTS: usize = 3;
53
54/// One enumerated HID interface with its own endpoint and decoder.
55pub struct HidInterface {
56    pub channel: usize,
57    pub dev_addr: u8,
58    pub interface: u8,
59    pub endpoint: EndpointDesc,
60    /// Descriptor-driven decoder, when the report descriptor parsed.
61    pub decoder: Option<HidDecoder>,
62    /// Boot-protocol state, used when `decoder` is `None`.
63    pub boot_state: BootState,
64    pub kind: HidKind,
65    /// Packets received (diagnostics).
66    pub reports_seen: u32,
67    /// How this device is reached: through the hub, or directly.
68    ///
69    /// Carried per interface because polling sets the context on the engine
70    /// before each poll, and a full- or low-speed device behind a hub is not
71    /// reachable without it.
72    pub split: Option<crate::usb_channel::Split>,
73    /// Frame number of the last poll, so the endpoint's interval is honoured.
74    pub last_poll_frame: u32,
75}

79const HUB_PORT_SPEED_LOW: u8 = 1;

81/// Probe `port` for a device, reset it, and assign the next free address.
82///
83/// Returns the device's interfaces so the caller can decide what it is: a hub
84/// port may hold a HID device, and on the LAN9514 the first port holds the
85/// Ethernet controller. Classification is deliberately the caller's job — this
86/// function only brings the device to a state where descriptors are readable.
87///
88/// Returns `None` when the port is empty or the device does not answer; a flaky
89/// port must not take the whole controller down.
90pub fn attach_port(
91    engine: &UsbHostEngine<'_>,
92    hub: &UsbHub<'_>,
93    port: u16,
94    next_addr: &mut u8,
95) -> Option<(u8, Vec<InterfaceDesc>, Option<crate::usb_channel::Split>)> {
96    // Everything before the device answers is traffic to the hub itself, which
97    // sits on the root port and must not be reached through a split. Whatever
98    // the previous port left behind is cleared before the first of them.
99    engine.set_split(None);
100
101    if !hub.is_port_connected(port) {
102        return None;
103    }
104
105    print("[usb-hid] device on hub port ");
106    print_u8(port as u8);
107    println("");
108
109    if hub.reset_port(port).is_err() {
110        println("[usb-hid] WARN: port reset failed");
111        hub.clear_port_change(port, crate::hub::C_PORT_RESET);
112        return None;
113    }
114    hub.clear_port_change(port, crate::hub::C_PORT_CONNECTION);
115
116    let addr = *next_addr;
117    if addr == 0 {
118        println("[usb-hid] WARN: address space exhausted");
119        return None;
120    }
121
122    // A device behind the hub can be slower than the hub itself, and the core
123    // will not run a high-speed channel programmed with 8-byte packets (nor
124    // frame a full-speed device that answers in 8 bytes at 64). Take EP0's
125    // starting size from this port's negotiated speed.
126    let hub_speed = hub.port_speed(port);
127    let ep0_mps = match hub_speed {
128        HUB_PORT_SPEED_HIGH => {
129            crate::usb_channel::initial_control_mps(crate::usb_channel::PORT_SPEED_HIGH)
130        }
131        _ => crate::usb_channel::initial_control_mps(crate::usb_channel::PORT_SPEED_FULL),
132    };
133    engine.set_control_mps(ep0_mps);
134
135    // A full- or low-speed device behind this high-speed hub can only be reached
136    // through it: the host asks the hub to buffer each transaction and then asks
137    // for the result. Nothing on the bus answers otherwise, and the channel
138    // reports XACTERR, which reads like a protocol error and is really a missing
139    // transaction. A high-speed device is addressed directly and needs none of
140    // this.
141    let split = if hub_speed == HUB_PORT_SPEED_HIGH {
142        None
143    } else {
144        Some(crate::usb_channel::Split {
145            hub_addr: hub.address(),
146            port: port as u8,
147            low_speed: hub_speed == HUB_PORT_SPEED_LOW,
148        })
149    };
150    engine.set_split(split);
151
152    // Bring the device up. Every failure past this point has to clear the split
153    // context on its way out: leaving it set routes the *next* transfer -- which
154    // is hub traffic for the next port -- through a hub port, and the hub then
155    // reports transaction errors for requests it never received.
156    let brought_up = (|| -> Option<Vec<InterfaceDesc>> {
157        // The device answers at address 0 until SET_ADDRESS latches.
158        let device = read_device_descriptor(engine, 0)?;
159        print("[usb-hid] vendor:product ");
160        print_hex16(device.vendor_id);
161        print(":");
162        print_hex16(device.product_id);
163        println("");
164
165        UsbHub::set_address(engine, addr).ok()?;
166        let (config_value, descriptors) = read_configuration(engine, addr)?;
167
168        // Activate the configuration before any interface-level request: SET_IDLE
169        // and SET_PROTOCOL are only answered in the configured state.
170        UsbHub::set_configuration(engine, addr, config_value).ok()?;
171
172        Some(descriptors)
173    })();
174
175    match brought_up {
176        Some(descriptors) => {
177            // Only a device that came all the way up consumes its address.
178            *next_addr = next_addr.saturating_add(1);
179            Some((addr, descriptors, split))
180        }
181        None => {
182            engine.set_split(None);
183            None
184        }
185    }
186}

188/// Read a device descriptor from `addr`, retrying the short-read case.
189///
190/// USB 2.0 §9.6.1 lets a device answer the first 8 bytes of its device
191/// descriptor and expect a second request for the full 18, so one short reply is
192/// normal and only a second short reply means failure.
193pub fn read_device_descriptor(
194    engine: &UsbHostEngine<'_>,
195    addr: u8,
196) -> Option<usb_desc::DeviceDescriptor> {
197    // The first SETUP after a port reset is regularly lost, and a failed
198    // transfer must be retried rather than abandoned: an abandoned control
199    // transaction leaves the device mid-stage, and its *next* transaction then
200    // stalls in the status phase. A `.ok()?` here would end the whole
201    // enumeration on that one transient failure.
202    for attempt in 0..ENUM_ATTEMPTS {
203        let mut buf = [0u8; 18];
204        let result = engine.control_transfer(
205            addr,
206            RT_DEV_TO_HOST_STANDARD,
207            usb_desc::REQ_GET_DESCRIPTOR,
208            (usb_desc::DT_DEVICE as u16) << 8,
209            0,
210            &mut buf,
211        );
212
213        match result {
214            // USB 2.0 §9.6.1 lets a device answer the first 8 bytes of its
215            // device descriptor and expect a second request for the full 18;
216            // a short reply is therefore normal and worth retrying.
217            Ok(n) if n >= 18 => {
218                let device = usb_desc::DeviceDescriptor::parse(&buf)?;
219                // The device descriptor is the only place a device reports its
220                // control-endpoint packet size, and every later transfer on this
221                // engine depends on it — adopt it before returning.
222                engine.set_control_mps(device.max_packet_size_0);
223                return Some(device);
224            }
225            _ => {
226                if attempt + 1 < ENUM_ATTEMPTS {
227                    delay_ms(10);
228                }
229            }
230        }
231    }
232    None
233}

235/// Read a device's active configuration, returning its config value and the
236/// interfaces it declares.
237///
238/// The 9-byte header is read first because `wTotalLength` is authoritative: a
239/// fixed-size read either truncates a large configuration or mis-frames a small
240/// one, and either way the interface walk then reads the wrong bytes.
241pub fn read_configuration(
242    engine: &UsbHostEngine<'_>,
243    addr: u8,
244) -> Option<(u8, Vec<InterfaceDesc>)> {
245    let mut header = [0u8; 9];
246    let mut header_len = 0usize;
247    for attempt in 0..ENUM_ATTEMPTS {
248        header = [0u8; 9];
249        match engine.control_transfer(
250            addr,
251            RT_DEV_TO_HOST_STANDARD,
252            usb_desc::REQ_GET_DESCRIPTOR,
253            (usb_desc::DT_CONFIGURATION as u16) << 8,
254            0,
255            &mut header,
256        ) {
257            Ok(n) if n >= 9 => {
258                header_len = n;
259                break;
260            }
261            _ => {
262                if attempt + 1 < ENUM_ATTEMPTS {
263                    delay_ms(10);
264                }
265            }
266        }
267    }
268    if header_len < 9 {
269        return None;
270    }
271    let total = usb_desc::configuration_total_length(&header)? as usize;
272    if !(9..=512).contains(&total) {
273        return None;
274    }
275
276    let mut cfg = alloc::vec![0u8; total];
277    let mut cn = 0usize;
278    for attempt in 0..ENUM_ATTEMPTS {
279        match engine.control_transfer(
280            addr,
281            RT_DEV_TO_HOST_STANDARD,
282            usb_desc::REQ_GET_DESCRIPTOR,
283            (usb_desc::DT_CONFIGURATION as u16) << 8,
284            0,
285            &mut cfg,
286        ) {
287            Ok(n) if n >= 9 => {
288                cn = n;
289                break;
290            }
291            _ => {
292                if attempt + 1 < ENUM_ATTEMPTS {
293                    delay_ms(10);
294                }
295            }
296        }
297    }
298    if cn < 9 {
299        return None;
300    }
301
302    let config_value = cfg.get(5).copied().unwrap_or(1);
303    Some((
304        config_value,
305        usb_desc::parse_configuration(&cfg[..cn.min(total)]),
306    ))
307}

309/// What the root device turned out to be.
310pub enum RootClass {
311    /// A hub — its downstream ports must be scanned for devices.
312    Hub,
313    /// A HID device sitting directly on the root port.
314    Hid,
315    /// Something this driver does not claim.
316    Other,
317}

319/// Classify a device from its device and interface descriptors.
320///
321/// A hub declares class 0x09 either at device level (typical for a wired hub
322/// like the LAN9514) or per interface, so both are checked. Interface classes
323/// decide everything else, because §9.6.1 leaves the device-level class at 0 for
324/// a device that mixes classes — which a combo receiver does.
325pub fn classify(device: &usb_desc::DeviceDescriptor, interfaces: &[InterfaceDesc]) -> RootClass {
326    const CLASS_HUB: u8 = 0x09;
327    if device.class == CLASS_HUB || interfaces.iter().any(|i| i.class == CLASS_HUB) {
328        return RootClass::Hub;
329    }
330    if interfaces.iter().any(|i| i.is_hid()) {
331        return RootClass::Hid;
332    }
333    RootClass::Other
334}

336/// Bring up one HID interface: fetch its report descriptor, choose a decode
337/// path, and select the protocol the decoder expects.
338fn start_interface(
339    engine: &UsbHostEngine<'_>,
340    dev_addr: u8,
341    iface: &InterfaceDesc,
342) -> Option<HidInterface> {
343    let ep = *iface.interrupt_in.first()?;
344    let kind = match iface.protocol {
345        usb_desc::PROTOCOL_KEYBOARD => HidKind::Keyboard,
346        usb_desc::PROTOCOL_MOUSE => HidKind::Mouse,
347        _ => HidKind::Unknown,
348    };
349
350    print("[usb-hid] HID interface ");
351    print_u8(iface.number);
352    print(" class=");
353    print_u8(iface.class);
354    print(" boot=");
355    print_u8(iface.subclass);
356    println("");
357
358    // ── Report descriptor → full decoder ─────────────────────────────────────
359    //
360    // The HID descriptor's wDescriptorLength is authoritative; the same request
361    // with a short buffer is what truncated descriptors come from.
362    let mut decoder = None;
363    if let Some(len) = iface.report_descriptor_len {
364        let len = len as usize;
365        if len > 0 && len <= 1024 {
366            let mut rd = alloc::vec![0u8; len];
367            let got = engine
368                .control_transfer(
369                    dev_addr,
370                    RT_DEV_TO_HOST_STANDARD,
371                    usb_desc::REQ_GET_DESCRIPTOR,
372                    (usb_desc::DT_HID_REPORT as u16) << 8,
373                    iface.number as u16,
374                    &mut rd,
375                )
376                .unwrap_or(0);
377            if got > 0 {
378                let map = crate::hid::parse_report_descriptor(&rd[..got]);
379                if !map.reports.is_empty() {
380                    let parsed_kind = HidKind::from_map(&map);
381                    println("[usb-hid] report descriptor parsed");
382                    decoder = Some(HidDecoder::new(map));
383                    // A descriptor that names no collection still decodes; the
384                    // interface's own protocol then supplies the class.
385                    let _ = parsed_kind;
386                } else {
387                    println("[usb-hid] WARN: report descriptor declared no fields");
388                }
389            }
390        }
391    }
392
393    // ── Idle + protocol ──────────────────────────────────────────────────────
394    //
395    // SET_IDLE(0) stops the device resending an unchanged report forever, which
396    // would otherwise flood the poll loop and the input service.
397    let _ = engine.control_transfer(
398        dev_addr,
399        RT_CLASS_INTERFACE_OUT,
400        usb_desc::REQ_HID_SET_IDLE,
401        0,
402        iface.number as u16,
403        &mut [],
404    );
405
406    // Keep report protocol when a descriptor drives decoding (boot protocol
407    // would change the report shape and desynchronise it). Only a device using
408    // the boot fallback is switched to boot protocol.
409    if iface.supports_boot() && decoder.is_none() {
410        let _ = engine.control_transfer(
411            dev_addr,
412            RT_CLASS_INTERFACE_OUT,
413            usb_desc::REQ_HID_SET_PROTOCOL,
414            0, // 0 = boot protocol
415            iface.number as u16,
416            &mut [],
417        );
418    }
419
420    Some(HidInterface {
421        split: engine.split(),
422        last_poll_frame: 0,
423        channel: 0, // assigned by the caller
424        dev_addr,
425        interface: iface.number,
426        endpoint: ep,
427        decoder,
428        boot_state: BootState::default(),
429        kind,
430        reports_seen: 0,
431    })
432}

434/// Start every drivable HID interface of one already-configured device.
435///
436/// Returns how many were started. `channel_cursor` is the next free host
437/// channel so two devices never share one.
438pub fn start_hid_interfaces(
439    engine: &UsbHostEngine<'_>,
440    addr: u8,
441    interfaces: &[InterfaceDesc],
442    channel_cursor: usize,
443    out: &mut Vec<HidInterface>,
444) -> usize {
445    let mut started = 0usize;
446    for iface in interfaces.iter().filter(|i| i.is_hid() && i.alternate == 0) {
447        if out.len() >= MAX_HID_INTERFACES {
448            break;
449        }
450        let Some(mut hid) = start_interface(engine, addr, iface) else {
451            continue;
452        };
453        hid.channel = channel_cursor + out.len();
454        out.push(hid);
455        started += 1;
456    }
457    started
458}

460/// Enumerate every HID device behind `hub`, filling `out`.
461///
462/// Ports are scanned in order and each device takes the next free host channel.
463/// `next_addr` carries the USB address counter across ports so two devices never
464/// collide on one address.
465pub fn enumerate(
466    engine: &UsbHostEngine<'_>,
467    hub: &UsbHub<'_>,
468    port_count: u8,
469    next_addr: &mut u8,
470    out: &mut Vec<HidInterface>,
471) {
472    for port in 1..=port_count as u16 {
473        if out.len() >= MAX_HID_INTERFACES {
474            return;
475        }
476        let Some((addr, descriptors, split)) = attach_port(engine, hub, port, next_addr) else {
477            continue;
478        };
479        // The interfaces are brought up through the same path the device was.
480        engine.set_split(split);
481        let started = start_hid_interfaces(engine, addr, &descriptors, HID_CHANNEL_BASE, out);
482        // Back to direct addressing: the next thing talked to may be the hub
483        // itself, and a stale context would send that through a hub port.
484        engine.set_split(None);
485        if started == 0 {
486            println("[usb-hid] port holds no drivable HID interface");
487        }
488    }
489}

491/// Poll one interface and append the events it produced.
492///
493/// A NAK (no report queued) is the normal idle case and yields nothing.
494pub fn poll_interface(
495    engine: &UsbHostEngine<'_>,
496    iface: &mut HidInterface,
497    out: &mut Vec<EvdevEvent>,
498) {
499    // An interrupt endpoint states how often it wants to be asked, and the host
500    // is expected to honour it. Polling as fast as the CPU runs issues thousands
501    // of split pairs per microframe, and each start-split replaces whatever the
502    // hub still had buffered for this endpoint -- so a report that arrives
503    // between two polls is overwritten before the complete-split collects it,
504    // and the endpoint reads as permanently idle.
505    // A full- or low-speed device states its interval in frames, and the frame
506    // counter runs in microframes behind a high-speed hub.
507    let now = engine.frame_number();
508    let mut interval = (iface.endpoint.interval as u32).max(1);
509    if iface.split.is_some() {
510        interval = interval.saturating_mul(8);
511    }
512    if iface.last_poll_frame != 0 && now.wrapping_sub(iface.last_poll_frame) & 0xFFFF < interval {
513        return;
514    }
515    iface.last_poll_frame = now;
516
517    let mut buf = [0u8; REPORT_BUF];
518    // Poll the device the way it is wired, then restore direct addressing so a
519    // later transfer to the hub or the controller's own device is not routed
520    // through a hub port.
521    engine.set_split(iface.split);
522    let result = engine.interrupt_receive(
523        iface.channel,
524        iface.dev_addr,
525        iface.endpoint.number(),
526        iface.endpoint.max_packet_size,
527        &mut buf,
528    );
529    engine.set_split(None);
530    let got = match result {
531        Ok(n) => n,
532        Err(_) => {
533            // A STALL leaves the endpoint halted until the host clears it.
534            // Dropping the device here would lose a working keyboard over one
535            // bad report, so clear the stall and keep polling.
536            let _ = engine.control_transfer(
537                iface.dev_addr,
538                0x02, // Standard endpoint OUT
539                0x01, // CLEAR_FEATURE
540                0,    // ENDPOINT_HALT
541                iface.endpoint.address as u16,
542                &mut [],
543            );
544            return;
545        }
546    };
547    if got == 0 {
548        return;
549    }
550
551    iface.reports_seen = iface.reports_seen.saturating_add(1);
552
553    match iface.decoder.as_mut() {
554        Some(dec) => dec.process(&buf[..got], out),
555        None => {
556            // Boot fallback: the interface protocol says which shape to expect.
557            let kind = match iface.kind {
558                HidKind::Unknown => detect_boot_kind(&buf[..got]),
559                k => k,
560            };
561            decode_boot_report(kind, &buf[..got], &mut iface.boot_state, out);
562        }
563    }
564}

566/// Guess a boot report's class from its length when the interface said nothing.
567///
568/// A boot keyboard report is always 8 bytes; a boot mouse is 3 or 4.
569fn detect_boot_kind(report: &[u8]) -> HidKind {
570    match report.len() {
571        8 => HidKind::Keyboard,
572        3 | 4 => HidKind::Mouse,
573        _ => HidKind::Unknown,
574    }
575}

577/// Send one event to the input service.
578///
579/// `tid` is the input service's current tid; a failed send means it is not
580/// accepting right now (restarting, or not yet up), which the caller tolerates
581/// by dropping the event rather than spinning.
582pub fn forward_event(tid: usize, ev: &EvdevEvent) -> bool {
583    if tid == 0 {
584        return false;
585    }
586    let mut buf = [0u8; INPUT_EVENT_LEN];
587    ev.encode(&mut buf);
588    matches!(ostd::syscall::sys_send(tid, &buf), SyscallResult::Ok(_))
589}

591/// Register this cell as a raw event producer with the input service.
592///
593/// One typed request, sent before any event. After it the service routes this
594/// sender's short `[opcode][code][value]` messages down its event path.
595pub fn register_as_source(tid: usize, kind: u8) -> bool {
596    if tid == 0 {
597        return false;
598    }
599    let mut buf = [0u8; 32];
600    match api::ipc::encode(
601        &api::ipc::InputRequest::RegisterEventSource { kind },
602        &mut buf,
603    ) {
604        Ok(encoded) => matches!(
605            ostd::syscall::sys_try_send(tid, encoded),
606            SyscallResult::Ok(_)
607        ),
608        Err(_) => false,
609    }
610}

614fn print_hex16(v: u16) {
615    const HEX: &[u8; 16] = b"0123456789abcdef";
616    let mut out = [0u8; 4];
617    for (i, slot) in out.iter_mut().enumerate() {
618        let nibble = (v >> (12 - i * 4)) & 0xF;
619        *slot = HEX[nibble as usize];
620    }
621    if let Ok(s) = core::str::from_utf8(&out) {
622        print(s);
623    }
624}
625
626fn print_u8(v: u8) {
627    let mut out = [0u8; 3];
628    let mut n = v;
629    let mut len = 0;
630    if n == 0 {
631        print("0");
632        return;
633    }
634    while n > 0 && len < out.len() {
635        out[len] = b'0' + (n % 10);
636        n /= 10;
637        len += 1;
638    }
639    out[..len].reverse();
640    if let Ok(s) = core::str::from_utf8(&out[..len]) {
641        print(s);
642    }
643}
