1//! USB HID class driver over the DWC2 host controller.
2//!
3//! Brings up whatever HID device is attached to a LAN9514 downstream port and
4//! feeds its reports into the input service. The full path is:
5//!
6//! ```text
7//!   port connect → SET_ADDRESS → GET_DESCRIPTOR(DEVICE)
8//!                → GET_DESCRIPTOR(CONFIGURATION) → walk interfaces
9//!                → per HID interface: GET_DESCRIPTOR(REPORT) → parse
10//!                → SET_CONFIGURATION → SET_IDLE(0) → SET_PROTOCOL
11//!                → poll Interrupt IN → decode → forward to input service
12//! ```
13//!
14//! **Composite devices.** A wireless receiver presents several interfaces — its
15//! keyboard and its mouse — and each carries its own report descriptor and
16//! interrupt endpoint. They are enumerated independently and polled round-robin;
17//! treating only the first interface would surface a keyboard with no pointer
18//! (or the reverse), which is the usual failure mode for combo dongles.
19//!
20//! **Report vs boot protocol.** The descriptor path runs when a report
21//! descriptor parses, because it is the only way to decode a device that uses
22//! report IDs or a non-standard layout. Boot protocol (fixed 8-byte keyboard /
23//! 3-byte mouse) is the fallback for a descriptor that is absent or unparseable,
24//! and it is what most combo receivers implement anyway so a BIOS can use them.
25

39/// First host channel reserved for HID. Channels 0-2 are control and the
40/// Ethernet bulk pair, so HID devices take 3..=6 (four concurrently attached).
41pub const HID_CHANNEL_BASE: usize = 3;
42/// Maximum HID interfaces driven at once across all ports.
43pub const MAX_HID_INTERFACES: usize = 4;
44/// Buffer size for one interrupt IN report.
45const REPORT_BUF: usize = 64;
46/// Control-transfer attempts before an enumeration stage gives up.
47///
48/// A port reset and a mode switch are both followed by a brief window where the
49/// first SETUP is lost, so one attempt is not enough to tell a dead device from
50/// a slow one.
51const ENUM_ATTEMPTS: usize = 3;
52
53/// One enumerated HID interface with its own endpoint and decoder.
54pub struct HidInterface {
55    pub channel: usize,
56    pub dev_addr: u8,
57    pub interface: u8,
58    pub endpoint: EndpointDesc,
59    /// Descriptor-driven decoder, when the report descriptor parsed.
60    pub decoder: Option<HidDecoder>,
61    /// Boot-protocol state, used when `decoder` is `None`.
62    pub boot_state: BootState,
63    pub kind: HidKind,
64    /// Packets received (diagnostics).
65    pub reports_seen: u32,
66}

68/// Probe `port` for a device, reset it, and assign the next free address.
69///
70/// Returns the device's interfaces so the caller can decide what it is: a hub
71/// port may hold a HID device, and on the LAN9514 the first port holds the
72/// Ethernet controller. Classification is deliberately the caller's job — this
73/// function only brings the device to a state where descriptors are readable.
74///
75/// Returns `None` when the port is empty or the device does not answer; a flaky
76/// port must not take the whole controller down.
77pub fn attach_port(
78    engine: &UsbHostEngine<'_>,
79    hub: &UsbHub<'_>,
80    port: u16,
81    next_addr: &mut u8,
82) -> Option<(u8, Vec<InterfaceDesc>)> {
83    if !hub.is_port_connected(port) {
84        return None;
85    }
86
87    print("[usb-hid] device on hub port ");
88    print_u8(port as u8);
89    println("");
90
91    if hub.reset_port(port).is_err() {
92        println("[usb-hid] WARN: port reset failed");
93        hub.clear_port_change(port, crate::hub::C_PORT_RESET);
94        return None;
95    }
96    hub.clear_port_change(port, crate::hub::C_PORT_CONNECTION);
97
98    let addr = *next_addr;
99    if addr == 0 {
100        println("[usb-hid] WARN: address space exhausted");
101        return None;
102    }
103
104    // A device behind the hub can be slower than the hub itself, and the core
105    // will not run a high-speed channel programmed with 8-byte packets (nor
106    // frame a full-speed device that answers in 8 bytes at 64). Take EP0's
107    // starting size from this port's negotiated speed.
108    let hub_speed = hub.port_speed(port);
109    let ep0_mps = match hub_speed {
110        2 => crate::usb_channel::initial_control_mps(crate::usb_channel::PORT_SPEED_HIGH),
111        _ => crate::usb_channel::initial_control_mps(crate::usb_channel::PORT_SPEED_FULL),
112    };
113    engine.set_control_mps(ep0_mps);
114
115    // The device answers at address 0 until SET_ADDRESS latches.
116    let device = read_device_descriptor(engine, 0)?;
117    print("[usb-hid] vendor:product ");
118    print_hex16(device.vendor_id);
119    print(":");
120    print_hex16(device.product_id);
121    println("");
122
123    UsbHub::set_address(engine, addr).ok()?;
124    *next_addr = next_addr.saturating_add(1);
125
126    let (config_value, descriptors) = read_configuration(engine, addr)?;
127
128    // Activate the configuration before any interface-level request: SET_IDLE
129    // and SET_PROTOCOL are only answered in the configured state.
130    UsbHub::set_configuration(engine, addr, config_value).ok()?;
131
132    Some((addr, descriptors))
133}

135/// Read a device descriptor from `addr`, retrying the short-read case.
136///
137/// USB 2.0 §9.6.1 lets a device answer the first 8 bytes of its device
138/// descriptor and expect a second request for the full 18, so one short reply is
139/// normal and only a second short reply means failure.
140pub fn read_device_descriptor(
141    engine: &UsbHostEngine<'_>,
142    addr: u8,
143) -> Option<usb_desc::DeviceDescriptor> {
144    // The first SETUP after a port reset is regularly lost, and a failed
145    // transfer must be retried rather than abandoned: an abandoned control
146    // transaction leaves the device mid-stage, and its *next* transaction then
147    // stalls in the status phase. A `.ok()?` here would end the whole
148    // enumeration on that one transient failure.
149    for attempt in 0..ENUM_ATTEMPTS {
150        let mut buf = [0u8; 18];
151        let result = engine.control_transfer(
152            addr,
153            RT_DEV_TO_HOST_STANDARD,
154            usb_desc::REQ_GET_DESCRIPTOR,
155            (usb_desc::DT_DEVICE as u16) << 8,
156            0,
157            &mut buf,
158        );
159
160        match result {
161            // USB 2.0 §9.6.1 lets a device answer the first 8 bytes of its
162            // device descriptor and expect a second request for the full 18;
163            // a short reply is therefore normal and worth retrying.
164            Ok(n) if n >= 18 => {
165                let device = usb_desc::DeviceDescriptor::parse(&buf)?;
166                // The device descriptor is the only place a device reports its
167                // control-endpoint packet size, and every later transfer on this
168                // engine depends on it — adopt it before returning.
169                engine.set_control_mps(device.max_packet_size_0);
170                return Some(device);
171            }
172            _ => {
173                if attempt + 1 < ENUM_ATTEMPTS {
174                    for _ in 0..200 {
175                        sys_yield();
176                    }
177                }
178            }
179        }
180    }
181    None
182}

184/// Read a device's active configuration, returning its config value and the
185/// interfaces it declares.
186///
187/// The 9-byte header is read first because `wTotalLength` is authoritative: a
188/// fixed-size read either truncates a large configuration or mis-frames a small
189/// one, and either way the interface walk then reads the wrong bytes.
190pub fn read_configuration(
191    engine: &UsbHostEngine<'_>,
192    addr: u8,
193) -> Option<(u8, Vec<InterfaceDesc>)> {
194    let mut header = [0u8; 9];
195    let mut header_len = 0usize;
196    for attempt in 0..ENUM_ATTEMPTS {
197        header = [0u8; 9];
198        match engine.control_transfer(
199            addr,
200            RT_DEV_TO_HOST_STANDARD,
201            usb_desc::REQ_GET_DESCRIPTOR,
202            (usb_desc::DT_CONFIGURATION as u16) << 8,
203            0,
204            &mut header,
205        ) {
206            Ok(n) if n >= 9 => {
207                header_len = n;
208                break;
209            }
210            _ => {
211                if attempt + 1 < ENUM_ATTEMPTS {
212                    for _ in 0..200 {
213                        sys_yield();
214                    }
215                }
216            }
217        }
218    }
219    if header_len < 9 {
220        return None;
221    }
222    let total = usb_desc::configuration_total_length(&header)? as usize;
223    if !(9..=512).contains(&total) {
224        return None;
225    }
226
227    let mut cfg = alloc::vec![0u8; total];
228    let mut cn = 0usize;
229    for attempt in 0..ENUM_ATTEMPTS {
230        match engine.control_transfer(
231            addr,
232            RT_DEV_TO_HOST_STANDARD,
233            usb_desc::REQ_GET_DESCRIPTOR,
234            (usb_desc::DT_CONFIGURATION as u16) << 8,
235            0,
236            &mut cfg,
237        ) {
238            Ok(n) if n >= 9 => {
239                cn = n;
240                break;
241            }
242            _ => {
243                if attempt + 1 < ENUM_ATTEMPTS {
244                    for _ in 0..200 {
245                        sys_yield();
246                    }
247                }
248            }
249        }
250    }
251    if cn < 9 {
252        return None;
253    }
254
255    let config_value = cfg.get(5).copied().unwrap_or(1);
256    Some((
257        config_value,
258        usb_desc::parse_configuration(&cfg[..cn.min(total)]),
259    ))
260}

262/// What the root device turned out to be.
263pub enum RootClass {
264    /// A hub — its downstream ports must be scanned for devices.
265    Hub,
266    /// A HID device sitting directly on the root port.
267    Hid,
268    /// Something this driver does not claim.
269    Other,
270}

272/// Classify a device from its device and interface descriptors.
273///
274/// A hub declares class 0x09 either at device level (typical for a wired hub
275/// like the LAN9514) or per interface, so both are checked. Interface classes
276/// decide everything else, because §9.6.1 leaves the device-level class at 0 for
277/// a device that mixes classes — which a combo receiver does.
278pub fn classify(device: &usb_desc::DeviceDescriptor, interfaces: &[InterfaceDesc]) -> RootClass {
279    const CLASS_HUB: u8 = 0x09;
280    if device.class == CLASS_HUB || interfaces.iter().any(|i| i.class == CLASS_HUB) {
281        return RootClass::Hub;
282    }
283    if interfaces.iter().any(|i| i.is_hid()) {
284        return RootClass::Hid;
285    }
286    RootClass::Other
287}

289/// Bring up one HID interface: fetch its report descriptor, choose a decode
290/// path, and select the protocol the decoder expects.
291fn start_interface(
292    engine: &UsbHostEngine<'_>,
293    dev_addr: u8,
294    iface: &InterfaceDesc,
295) -> Option<HidInterface> {
296    let ep = *iface.interrupt_in.first()?;
297    let kind = match iface.protocol {
298        usb_desc::PROTOCOL_KEYBOARD => HidKind::Keyboard,
299        usb_desc::PROTOCOL_MOUSE => HidKind::Mouse,
300        _ => HidKind::Unknown,
301    };
302
303    print("[usb-hid] HID interface ");
304    print_u8(iface.number);
305    print(" class=");
306    print_u8(iface.class);
307    print(" boot=");
308    print_u8(iface.subclass);
309    println("");
310
311    // ── Report descriptor → full decoder ─────────────────────────────────────
312    //
313    // The HID descriptor's wDescriptorLength is authoritative; the same request
314    // with a short buffer is what truncated descriptors come from.
315    let mut decoder = None;
316    if let Some(len) = iface.report_descriptor_len {
317        let len = len as usize;
318        if len > 0 && len <= 1024 {
319            let mut rd = alloc::vec![0u8; len];
320            let got = engine
321                .control_transfer(
322                    dev_addr,
323                    RT_DEV_TO_HOST_STANDARD,
324                    usb_desc::REQ_GET_DESCRIPTOR,
325                    (usb_desc::DT_HID_REPORT as u16) << 8,
326                    iface.number as u16,
327                    &mut rd,
328                )
329                .unwrap_or(0);
330            if got > 0 {
331                let map = crate::hid::parse_report_descriptor(&rd[..got]);
332                if !map.reports.is_empty() {
333                    let parsed_kind = HidKind::from_map(&map);
334                    println("[usb-hid] report descriptor parsed");
335                    decoder = Some(HidDecoder::new(map));
336                    // A descriptor that names no collection still decodes; the
337                    // interface's own protocol then supplies the class.
338                    let _ = parsed_kind;
339                } else {
340                    println("[usb-hid] WARN: report descriptor declared no fields");
341                }
342            }
343        }
344    }
345
346    // ── Idle + protocol ──────────────────────────────────────────────────────
347    //
348    // SET_IDLE(0) stops the device resending an unchanged report forever, which
349    // would otherwise flood the poll loop and the input service.
350    let _ = engine.control_transfer(
351        dev_addr,
352        RT_CLASS_INTERFACE_OUT,
353        usb_desc::REQ_HID_SET_IDLE,
354        0,
355        iface.number as u16,
356        &mut [],
357    );
358
359    // Keep report protocol when a descriptor drives decoding (boot protocol
360    // would change the report shape and desynchronise it). Only a device using
361    // the boot fallback is switched to boot protocol.
362    if iface.supports_boot() && decoder.is_none() {
363        let _ = engine.control_transfer(
364            dev_addr,
365            RT_CLASS_INTERFACE_OUT,
366            usb_desc::REQ_HID_SET_PROTOCOL,
367            0, // 0 = boot protocol
368            iface.number as u16,
369            &mut [],
370        );
371    }
372
373    Some(HidInterface {
374        channel: 0, // assigned by the caller
375        dev_addr,
376        interface: iface.number,
377        endpoint: ep,
378        decoder,
379        boot_state: BootState::default(),
380        kind,
381        reports_seen: 0,
382    })
383}

385/// Start every drivable HID interface of one already-configured device.
386///
387/// Returns how many were started. `channel_cursor` is the next free host
388/// channel so two devices never share one.
389pub fn start_hid_interfaces(
390    engine: &UsbHostEngine<'_>,
391    addr: u8,
392    interfaces: &[InterfaceDesc],
393    channel_cursor: usize,
394    out: &mut Vec<HidInterface>,
395) -> usize {
396    let mut started = 0usize;
397    for iface in interfaces.iter().filter(|i| i.is_hid() && i.alternate == 0) {
398        if out.len() >= MAX_HID_INTERFACES {
399            break;
400        }
401        let Some(mut hid) = start_interface(engine, addr, iface) else {
402            continue;
403        };
404        hid.channel = channel_cursor + out.len();
405        out.push(hid);
406        started += 1;
407    }
408    started
409}

411/// Enumerate every HID device behind `hub`, filling `out`.
412///
413/// Ports are scanned in order and each device takes the next free host channel.
414/// `next_addr` carries the USB address counter across ports so two devices never
415/// collide on one address.
416pub fn enumerate(
417    engine: &UsbHostEngine<'_>,
418    hub: &UsbHub<'_>,
419    port_count: u8,
420    next_addr: &mut u8,
421    out: &mut Vec<HidInterface>,
422) {
423    for port in 1..=port_count as u16 {
424        if out.len() >= MAX_HID_INTERFACES {
425            return;
426        }
427        let Some((addr, descriptors)) = attach_port(engine, hub, port, next_addr) else {
428            continue;
429        };
430        let started = start_hid_interfaces(engine, addr, &descriptors, HID_CHANNEL_BASE, out);
431        if started == 0 {
432            println("[usb-hid] port holds no drivable HID interface");
433        }
434    }
435}

437/// Poll one interface and append the events it produced.
438///
439/// A NAK (no report queued) is the normal idle case and yields nothing.
440pub fn poll_interface(
441    engine: &UsbHostEngine<'_>,
442    iface: &mut HidInterface,
443    out: &mut Vec<EvdevEvent>,
444) {
445    let mut buf = [0u8; REPORT_BUF];
446    let got = match engine.interrupt_receive(
447        iface.channel,
448        iface.dev_addr,
449        iface.endpoint.number(),
450        iface.endpoint.max_packet_size,
451        &mut buf,
452    ) {
453        Ok(n) => n,
454        Err(_) => {
455            // A STALL leaves the endpoint halted until the host clears it.
456            // Dropping the device here would lose a working keyboard over one
457            // bad report, so clear the stall and keep polling.
458            let _ = engine.control_transfer(
459                iface.dev_addr,
460                0x02, // Standard endpoint OUT
461                0x01, // CLEAR_FEATURE
462                0,    // ENDPOINT_HALT
463                iface.endpoint.address as u16,
464                &mut [],
465            );
466            return;
467        }
468    };
469    if got == 0 {
470        return;
471    }
472
473    iface.reports_seen = iface.reports_seen.saturating_add(1);
474
475    match iface.decoder.as_mut() {
476        Some(dec) => dec.process(&buf[..got], out),
477        None => {
478            // Boot fallback: the interface protocol says which shape to expect.
479            let kind = match iface.kind {
480                HidKind::Unknown => detect_boot_kind(&buf[..got]),
481                k => k,
482            };
483            decode_boot_report(kind, &buf[..got], &mut iface.boot_state, out);
484        }
485    }
486}

488/// Guess a boot report's class from its length when the interface said nothing.
489///
490/// A boot keyboard report is always 8 bytes; a boot mouse is 3 or 4.
491fn detect_boot_kind(report: &[u8]) -> HidKind {
492    match report.len() {
493        8 => HidKind::Keyboard,
494        3 | 4 => HidKind::Mouse,
495        _ => HidKind::Unknown,
496    }
497}

499/// Send one event to the input service.
500///
501/// `tid` is the input service's current tid; a failed send means it is not
502/// accepting right now (restarting, or not yet up), which the caller tolerates
503/// by dropping the event rather than spinning.
504pub fn forward_event(tid: usize, ev: &EvdevEvent) -> bool {
505    if tid == 0 {
506        return false;
507    }
508    let mut buf = [0u8; INPUT_EVENT_LEN];
509    ev.encode(&mut buf);
510    matches!(ostd::syscall::sys_send(tid, &buf), SyscallResult::Ok(_))
511}

513/// Register this cell as a raw event producer with the input service.
514///
515/// One typed request, sent before any event. After it the service routes this
516/// sender's short `[opcode][code][value]` messages down its event path.
517pub fn register_as_source(tid: usize, kind: u8) -> bool {
518    if tid == 0 {
519        return false;
520    }
521    let mut buf = [0u8; 32];
522    match api::ipc::encode(
523        &api::ipc::InputRequest::RegisterEventSource { kind },
524        &mut buf,
525    ) {
526        Ok(encoded) => matches!(
527            ostd::syscall::sys_try_send(tid, encoded),
528            SyscallResult::Ok(_)
529        ),
530        Err(_) => false,
531    }
532}

536fn print_hex16(v: u16) {
537    const HEX: &[u8; 16] = b"0123456789abcdef";
538    let mut out = [0u8; 4];
539    for (i, slot) in out.iter_mut().enumerate() {
540        let nibble = (v >> (12 - i * 4)) & 0xF;
541        *slot = HEX[nibble as usize];
542    }
543    if let Ok(s) = core::str::from_utf8(&out) {
544        print(s);
545    }
546}
547
548fn print_u8(v: u8) {
549    let mut out = [0u8; 3];
550    let mut n = v;
551    let mut len = 0;
552    if n == 0 {
553        print("0");
554        return;
555    }
556    while n > 0 && len < out.len() {
557        out[len] = b'0' + (n % 10);
558        n /= 10;
559        len += 1;
560    }
561    out[..len].reverse();
562    if let Ok(s) = core::str::from_utf8(&out[..len]) {
563        print(s);
564    }
565}
