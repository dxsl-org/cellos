1//! USB HID class driver over the DWC2 host controller.
2//!
3//! Brings up whatever HID device is attached to a LAN9514 downstream port and
4//! feeds its reports into the input service. The full path is:
5//!
6//! ```text
7//!   port connect → SET_ADDRESS → GET_DESCRIPTOR(DEVICE)
8//!                → GET_DESCRIPTOR(CONFIGURATION) → walk interfaces
9//!                → per HID interface: GET_DESCRIPTOR(REPORT) → parse
10//!                → SET_CONFIGURATION → SET_IDLE(0) → SET_PROTOCOL
11//!                → poll Interrupt IN → decode → forward to input service
12//! ```
13//!
14//! **Composite devices.** A wireless receiver presents several interfaces — its
15//! keyboard and its mouse — and each carries its own report descriptor and
16//! interrupt endpoint. They are enumerated independently and polled round-robin;
17//! treating only the first interface would surface a keyboard with no pointer
18//! (or the reverse), which is the usual failure mode for combo dongles.
19//!
20//! **Report vs boot protocol.** The descriptor path runs when a report
21//! descriptor parses, because it is the only way to decode a device that uses
22//! report IDs or a non-standard layout. Boot protocol (fixed 8-byte keyboard /
23//! 3-byte mouse) is the fallback for a descriptor that is absent or unparseable,
24//! and it is what most combo receivers implement anyway so a BIOS can use them.
25

40/// First host channel reserved for HID. Channels 0-2 are control and the
41/// Ethernet bulk pair, so HID devices take 3..=6 (four concurrently attached).
42pub const HID_CHANNEL_BASE: usize = 3;
43/// Maximum HID interfaces driven at once across all ports.
44pub const MAX_HID_INTERFACES: usize = 4;
45/// Buffer size for one interrupt IN report.
46const REPORT_BUF: usize = 64;
47/// Control-transfer attempts before an enumeration stage gives up.
48///
49/// A port reset and a mode switch are both followed by a brief window where the
50/// first SETUP is lost, so one attempt is not enough to tell a dead device from
51/// a slow one.
52const ENUM_ATTEMPTS: usize = 3;
53
54/// One enumerated HID interface with its own endpoint and decoder.
55pub struct HidInterface {
56    pub channel: usize,
57    pub dev_addr: u8,
58    pub interface: u8,
59    pub endpoint: EndpointDesc,
60    /// Descriptor-driven decoder, when the report descriptor parsed.
61    pub decoder: Option<HidDecoder>,
62    /// Boot-protocol state, used when `decoder` is `None`.
63    pub boot_state: BootState,
64    pub kind: HidKind,
65    /// Packets received (diagnostics).
66    pub reports_seen: u32,
67    /// How this device is reached: through the hub, or directly.
68    ///
69    /// Carried per interface because polling sets the context on the engine
70    /// before each poll, and a full- or low-speed device behind a hub is not
71    /// reachable without it.
72    pub split: Option<crate::usb_channel::Split>,
73    /// Frame number of the last poll, so the endpoint's interval is honoured.
74    pub last_poll_frame: u32,
75    /// A start-split is outstanding and the next poll collects its result.
76    ///
77    /// The two halves of a split reach the hub in different microframes, and this
78    /// driver issues one half per call rather than waiting out that gap on the
79    /// machine's time.
80    pub split_pending: bool,
81}

84const HUB_PORT_SPEED_HIGH: u8 = 2;
85const HUB_PORT_SPEED_LOW: u8 = 1;

87/// Probe `port` for a device, reset it, and assign the next free address.
88///
89/// Returns the device's interfaces so the caller can decide what it is: a hub
90/// port may hold a HID device, and on the LAN9514 the first port holds the
91/// Ethernet controller. Classification is deliberately the caller's job — this
92/// function only brings the device to a state where descriptors are readable.
93///
94/// Returns `None` when the port is empty or the device does not answer; a flaky
95/// port must not take the whole controller down.
96pub fn attach_port(
97    engine: &UsbHostEngine<'_>,
98    hub: &UsbHub<'_>,
99    port: u16,
100    next_addr: &mut u8,
101) -> Option<(u8, Vec<InterfaceDesc>, Option<crate::usb_channel::Split>)> {
102    // Everything before the device answers is traffic to the hub itself, which
103    // sits on the root port and must not be reached through a split. Whatever
104    // the previous port left behind is cleared before the first of them.
105    engine.set_split(None);
106
107    if !hub.is_port_connected(port) {
108        return None;
109    }
110
111    print("[usb-hid] device on hub port ");
112    print_u8(port as u8);
113    println("");
114
115    if hub.reset_port(port).is_err() {
116        println("[usb-hid] WARN: port reset failed");
117        hub.clear_port_change(port, crate::hub::C_PORT_RESET);
118        return None;
119    }
120    hub.clear_port_change(port, crate::hub::C_PORT_CONNECTION);
121
122    let addr = *next_addr;
123    if addr == 0 {
124        println("[usb-hid] WARN: address space exhausted");
125        return None;
126    }
127
128    // A device behind the hub can be slower than the hub itself, and the core
129    // will not run a high-speed channel programmed with 8-byte packets (nor
130    // frame a full-speed device that answers in 8 bytes at 64). Take EP0's
131    // starting size from this port's negotiated speed.
132    let hub_speed = hub.port_speed(port);
133    let ep0_mps = match hub_speed {
134        HUB_PORT_SPEED_HIGH => {
135            crate::usb_channel::initial_control_mps(crate::usb_channel::PORT_SPEED_HIGH)
136        }
137        _ => crate::usb_channel::initial_control_mps(crate::usb_channel::PORT_SPEED_FULL),
138    };
139    engine.set_control_mps(ep0_mps);
140
141    // A full- or low-speed device behind this high-speed hub can only be reached
142    // through it: the host asks the hub to buffer each transaction and then asks
143    // for the result. Nothing on the bus answers otherwise, and the channel
144    // reports XACTERR, which reads like a protocol error and is really a missing
145    // transaction. A high-speed device is addressed directly and needs none of
146    // this.
147    let split = if hub_speed == HUB_PORT_SPEED_HIGH {
148        None
149    } else {
150        Some(crate::usb_channel::Split {
151            hub_addr: hub.address(),
152            port: port as u8,
153            low_speed: hub_speed == HUB_PORT_SPEED_LOW,
154        })
155    };
156    engine.set_split(split);
157
158    // Bring the device up. Every failure past this point has to clear the split
159    // context on its way out: leaving it set routes the *next* transfer -- which
160    // is hub traffic for the next port -- through a hub port, and the hub then
161    // reports transaction errors for requests it never received.
162    let brought_up = (|| -> Option<Vec<InterfaceDesc>> {
163        // The device answers at address 0 until SET_ADDRESS latches.
164        let device = read_device_descriptor(engine, 0)?;
165        print("[usb-hid] vendor:product ");
166        print_hex16(device.vendor_id);
167        print(":");
168        print_hex16(device.product_id);
169        println("");
170
171        UsbHub::set_address(engine, addr).ok()?;
172        let (config_value, descriptors) = read_configuration(engine, addr)?;
173
174        // Activate the configuration before any interface-level request: SET_IDLE
175        // and SET_PROTOCOL are only answered in the configured state.
176        UsbHub::set_configuration(engine, addr, config_value).ok()?;
177
178        Some(descriptors)
179    })();
180
181    match brought_up {
182        Some(descriptors) => {
183            // Only a device that came all the way up consumes its address.
184            *next_addr = next_addr.saturating_add(1);
185            Some((addr, descriptors, split))
186        }
187        None => {
188            engine.set_split(None);
189            None
190        }
191    }
192}

194/// Read a device descriptor from `addr`, retrying the short-read case.
195///
196/// USB 2.0 §9.6.1 lets a device answer the first 8 bytes of its device
197/// descriptor and expect a second request for the full 18, so one short reply is
198/// normal and only a second short reply means failure.
199pub fn read_device_descriptor(
200    engine: &UsbHostEngine<'_>,
201    addr: u8,
202) -> Option<usb_desc::DeviceDescriptor> {
203    // The first SETUP after a port reset is regularly lost, and a failed
204    // transfer must be retried rather than abandoned: an abandoned control
205    // transaction leaves the device mid-stage, and its *next* transaction then
206    // stalls in the status phase. A `.ok()?` here would end the whole
207    // enumeration on that one transient failure.
208    for attempt in 0..ENUM_ATTEMPTS {
209        let mut buf = [0u8; 18];
210        let result = engine.control_transfer(
211            addr,
212            RT_DEV_TO_HOST_STANDARD,
213            usb_desc::REQ_GET_DESCRIPTOR,
214            (usb_desc::DT_DEVICE as u16) << 8,
215            0,
216            &mut buf,
217        );
218
219        match result {
220            // USB 2.0 §9.6.1 lets a device answer the first 8 bytes of its
221            // device descriptor and expect a second request for the full 18;
222            // a short reply is therefore normal and worth retrying.
223            Ok(n) if n >= 18 => {
224                let device = usb_desc::DeviceDescriptor::parse(&buf)?;
225                // The device descriptor is the only place a device reports its
226                // control-endpoint packet size, and every later transfer on this
227                // engine depends on it — adopt it before returning.
228                engine.set_control_mps(device.max_packet_size_0);
229                return Some(device);
230            }
231            _ => {
232                if attempt + 1 < ENUM_ATTEMPTS {
233                    delay_ms(10);
234                }
235            }
236        }
237    }
238    None
239}

241/// Read a device's active configuration, returning its config value and the
242/// interfaces it declares.
243///
244/// The 9-byte header is read first because `wTotalLength` is authoritative: a
245/// fixed-size read either truncates a large configuration or mis-frames a small
246/// one, and either way the interface walk then reads the wrong bytes.
247pub fn read_configuration(
248    engine: &UsbHostEngine<'_>,
249    addr: u8,
250) -> Option<(u8, Vec<InterfaceDesc>)> {
251    let mut header = [0u8; 9];
252    let mut header_len = 0usize;
253    for attempt in 0..ENUM_ATTEMPTS {
254        header = [0u8; 9];
255        match engine.control_transfer(
256            addr,
257            RT_DEV_TO_HOST_STANDARD,
258            usb_desc::REQ_GET_DESCRIPTOR,
259            (usb_desc::DT_CONFIGURATION as u16) << 8,
260            0,
261            &mut header,
262        ) {
263            Ok(n) if n >= 9 => {
264                header_len = n;
265                break;
266            }
267            _ => {
268                if attempt + 1 < ENUM_ATTEMPTS {
269                    delay_ms(10);
270                }
271            }
272        }
273    }
274    if header_len < 9 {
275        return None;
276    }
277    let total = usb_desc::configuration_total_length(&header)? as usize;
278    if !(9..=512).contains(&total) {
279        return None;
280    }
281
282    let mut cfg = alloc::vec![0u8; total];
283    let mut cn = 0usize;
284    for attempt in 0..ENUM_ATTEMPTS {
285        match engine.control_transfer(
286            addr,
287            RT_DEV_TO_HOST_STANDARD,
288            usb_desc::REQ_GET_DESCRIPTOR,
289            (usb_desc::DT_CONFIGURATION as u16) << 8,
290            0,
291            &mut cfg,
292        ) {
293            Ok(n) if n >= 9 => {
294                cn = n;
295                break;
296            }
297            _ => {
298                if attempt + 1 < ENUM_ATTEMPTS {
299                    delay_ms(10);
300                }
301            }
302        }
303    }
304    if cn < 9 {
305        return None;
306    }
307
308    let config_value = cfg.get(5).copied().unwrap_or(1);
309    Some((
310        config_value,
311        usb_desc::parse_configuration(&cfg[..cn.min(total)]),
312    ))
313}

315/// What the root device turned out to be.
316pub enum RootClass {
317    /// A hub — its downstream ports must be scanned for devices.
318    Hub,
319    /// A HID device sitting directly on the root port.
320    Hid,
321    /// Something this driver does not claim.
322    Other,
323}
324
325/// Classify a device from its device and interface descriptors.
326///
327/// A hub declares class 0x09 either at device level (typical for a wired hub
328/// like the LAN9514) or per interface, so both are checked. Interface classes
329/// decide everything else, because §9.6.1 leaves the device-level class at 0 for
330/// a device that mixes classes — which a combo receiver does.
331pub fn classify(device: &usb_desc::DeviceDescriptor, interfaces: &[InterfaceDesc]) -> RootClass {
332    const CLASS_HUB: u8 = 0x09;
333    if device.class == CLASS_HUB || interfaces.iter().any(|i| i.class == CLASS_HUB) {
334        return RootClass::Hub;
335    }
336    if interfaces.iter().any(|i| i.is_hid()) {
337        return RootClass::Hid;
338    }
339    RootClass::Other
340}

342/// Bring up one HID interface: fetch its report descriptor, choose a decode
343/// path, and select the protocol the decoder expects.
344fn start_interface(
345    engine: &UsbHostEngine<'_>,
346    dev_addr: u8,
347    iface: &InterfaceDesc,
348) -> Option<HidInterface> {
349    let ep = *iface.interrupt_in.first()?;
350    let kind = match iface.protocol {
351        usb_desc::PROTOCOL_KEYBOARD => HidKind::Keyboard,
352        usb_desc::PROTOCOL_MOUSE => HidKind::Mouse,
353        _ => HidKind::Unknown,
354    };
355
356    print("[usb-hid] HID interface ");
357    print_u8(iface.number);
358    print(" class=");
359    print_u8(iface.class);
360    print(" boot=");
361    print_u8(iface.subclass);
362    println("");
363
364    // ── Report descriptor → full decoder ─────────────────────────────────────
365    //
366    // The HID descriptor's wDescriptorLength is authoritative; the same request
367    // with a short buffer is what truncated descriptors come from.
368    let mut decoder = None;
369    if let Some(len) = iface.report_descriptor_len {
370        let len = len as usize;
371        if len > 0 && len <= 1024 {
372            let mut rd = alloc::vec![0u8; len];
373            let got = engine
374                .control_transfer(
375                    dev_addr,
376                    RT_DEV_TO_HOST_STANDARD,
377                    usb_desc::REQ_GET_DESCRIPTOR,
378                    (usb_desc::DT_HID_REPORT as u16) << 8,
379                    iface.number as u16,
380                    &mut rd,
381                )
382                .unwrap_or(0);
383            if got > 0 {
384                let map = crate::hid::parse_report_descriptor(&rd[..got]);
385                if !map.reports.is_empty() {
386                    let parsed_kind = HidKind::from_map(&map);
387                    println("[usb-hid] report descriptor parsed");
388                    decoder = Some(HidDecoder::new(map));
389                    // A descriptor that names no collection still decodes; the
390                    // interface's own protocol then supplies the class.
391                    let _ = parsed_kind;
392                } else {
393                    println("[usb-hid] WARN: report descriptor declared no fields");
394                }
395            }
396        }
397    }
398
399    // ── Idle + protocol ──────────────────────────────────────────────────────
400    //
401    // SET_IDLE(0) stops the device resending an unchanged report forever, which
402    // would otherwise flood the poll loop and the input service.
403    let _ = engine.control_transfer(
404        dev_addr,
405        RT_CLASS_INTERFACE_OUT,
406        usb_desc::REQ_HID_SET_IDLE,
407        0,
408        iface.number as u16,
409        &mut [],
410    );
411
412    // Keep report protocol when a descriptor drives decoding (boot protocol
413    // would change the report shape and desynchronise it). Only a device using
414    // the boot fallback is switched to boot protocol.
415    if iface.supports_boot() && decoder.is_none() {
416        let _ = engine.control_transfer(
417            dev_addr,
418            RT_CLASS_INTERFACE_OUT,
419            usb_desc::REQ_HID_SET_PROTOCOL,
420            0, // 0 = boot protocol
421            iface.number as u16,
422            &mut [],
423        );
424    }
425
426    Some(HidInterface {
427        split: engine.split(),
428        last_poll_frame: 0,
429        split_pending: false,
430        channel: 0, // assigned by the caller
431        dev_addr,
432        interface: iface.number,
433        endpoint: ep,
434        decoder,
435        boot_state: BootState::default(),
436        kind,
437        reports_seen: 0,
438    })
439}

441/// Start every drivable HID interface of one already-configured device.
442///
443/// Returns how many were started. `channel_cursor` is the next free host
444/// channel so two devices never share one.
445pub fn start_hid_interfaces(
446    engine: &UsbHostEngine<'_>,
447    addr: u8,
448    interfaces: &[InterfaceDesc],
449    channel_cursor: usize,
450    out: &mut Vec<HidInterface>,
451) -> usize {
452    let mut started = 0usize;
453    for iface in interfaces.iter().filter(|i| i.is_hid() && i.alternate == 0) {
454        if out.len() >= MAX_HID_INTERFACES {
455            break;
456        }
457        let Some(mut hid) = start_interface(engine, addr, iface) else {
458            continue;
459        };
460        hid.channel = channel_cursor + out.len();
461        out.push(hid);
462        started += 1;
463    }
464    started
465}

467/// Enumerate every HID device behind `hub`, filling `out`.
468///
469/// Ports are scanned in order and each device takes the next free host channel.
470/// `next_addr` carries the USB address counter across ports so two devices never
471/// collide on one address.
472pub fn enumerate(
473    engine: &UsbHostEngine<'_>,
474    hub: &UsbHub<'_>,
475    port_count: u8,
476    next_addr: &mut u8,
477    out: &mut Vec<HidInterface>,
478) {
479    for port in 1..=port_count as u16 {
480        if out.len() >= MAX_HID_INTERFACES {
481            return;
482        }
483        let Some((addr, descriptors, split)) = attach_port(engine, hub, port, next_addr) else {
484            continue;
485        };
486        // The interfaces are brought up through the same path the device was.
487        engine.set_split(split);
488        let started = start_hid_interfaces(engine, addr, &descriptors, HID_CHANNEL_BASE, out);
489        // Back to direct addressing: the next thing talked to may be the hub
490        // itself, and a stale context would send that through a hub port.
491        engine.set_split(None);
492        if started == 0 {
493            println("[usb-hid] port holds no drivable HID interface");
494        }
495    }
496}

498/// Poll one interface and append the events it produced.
499///
500/// A NAK (no report queued) is the normal idle case and yields nothing.
501pub fn poll_interface(
502    engine: &UsbHostEngine<'_>,
503    iface: &mut HidInterface,
504    out: &mut Vec<EvdevEvent>,
505) {
506    // An interrupt endpoint states how often it wants to be asked, and the host
507    // is expected to honour it. Polling as fast as the CPU runs issues thousands
508    // of split pairs per microframe, and each start-split replaces whatever the
509    // hub still had buffered for this endpoint -- so a report that arrives
510    // between two polls is overwritten before the complete-split collects it,
511    // and the endpoint reads as permanently idle.
512    // An endpoint states its interval in frames, and `HFNUM` counts the same
513    // frames, so the value is used as given.
514    //
515    // A poll with half a split outstanding is exempt: its other half is what
516    // collects the report, and the hub only pairs the two for a few frames.
517    let now = engine.frame_number();
518    let interval = (iface.endpoint.interval as u32).max(1);
519    if !iface.split_pending
520        && iface.last_poll_frame != 0
521        && now.wrapping_sub(iface.last_poll_frame) & 0xFFFF < interval
522    {
523        return;
524    }
525    iface.last_poll_frame = now;
526
527    let mut buf = [0u8; REPORT_BUF];
528    // Poll the device the way it is wired, then restore direct addressing so a
529    // later transfer to the hub or the controller's own device is not routed
530    // through a hub port.
531    engine.set_split(iface.split);
532    let result = engine.interrupt_receive(
533        iface.channel,
534        iface.dev_addr,
535        iface.endpoint.number(),
536        iface.endpoint.max_packet_size,
537        &mut buf,
538        &mut iface.split_pending,
539    );
540    engine.set_split(None);
541    let got = match result {
542        Ok(n) => n,
543        Err(_) => {
544            // A STALL leaves the endpoint halted until the host clears it.
545            // Dropping the device here would lose a working keyboard over one
546            // bad report, so clear the stall and keep polling.
547            let _ = engine.control_transfer(
548                iface.dev_addr,
549                0x02, // Standard endpoint OUT
550                0x01, // CLEAR_FEATURE
551                0,    // ENDPOINT_HALT
552                iface.endpoint.address as u16,
553                &mut [],
554            );
555            return;
556        }
557    };
558    if got == 0 {
559        return;
560    }
561
562    iface.reports_seen = iface.reports_seen.saturating_add(1);
563    // Whether the reports keep arriving is the first thing to know when input
564    // stops reaching the shell: a driver that has gone quiet and a shell that
565    // has stopped consuming look identical from the console, and they are
566    // different bugs. The first few are listed, then a periodic tally.
567    if iface.reports_seen <= 4 || iface.reports_seen.is_multiple_of(512) {
568        print("[usb-hid] iface ");
569        print_u8(iface.interface);
570        print(" reports=");
571        print_usize_hid(iface.reports_seen as usize);
572        print(" len=");
573        print_usize_hid(got);
574        println("");
575    }
576
577    match iface.decoder.as_mut() {
578        Some(dec) => dec.process(&buf[..got], out),
579        None => {
580            // Boot fallback: the interface protocol says which shape to expect.
581            let kind = match iface.kind {
582                HidKind::Unknown => detect_boot_kind(&buf[..got]),
583                k => k,
584            };
585            decode_boot_report(kind, &buf[..got], &mut iface.boot_state, out);
586        }
587    }
588}

590/// Guess a boot report's class from its length when the interface said nothing.
591///
592/// A boot keyboard report is always 8 bytes; a boot mouse is 3 or 4.
593fn detect_boot_kind(report: &[u8]) -> HidKind {
594    match report.len() {
595        8 => HidKind::Keyboard,
596        3 | 4 => HidKind::Mouse,
597        _ => HidKind::Unknown,
598    }
599}

601/// Send one event to the input service.
602///
603/// `tid` is the input service's current tid; a failed send means it is not
604/// accepting right now (restarting, or not yet up), which the caller tolerates
605/// by dropping the event rather than spinning.
606pub fn forward_event(tid: usize, ev: &EvdevEvent) -> bool {
607    if tid == 0 {
608        return false;
609    }
610    let mut buf = [0u8; INPUT_EVENT_LEN];
611    ev.encode(&mut buf);
612    matches!(ostd::syscall::sys_send(tid, &buf), SyscallResult::Ok(_))
613}

615/// Register this cell as a raw event producer with the input service.
616///
617/// One typed request, sent before any event. After it the service routes this
618/// sender's short `[opcode][code][value]` messages down its event path.
619pub fn register_as_source(tid: usize, kind: u8) -> bool {
620    if tid == 0 {
621        return false;
622    }
623    let mut buf = [0u8; 32];
624    match api::ipc::encode(
625        &api::ipc::InputRequest::RegisterEventSource { kind },
626        &mut buf,
627    ) {
628        Ok(encoded) => matches!(
629            ostd::syscall::sys_try_send(tid, encoded),
630            SyscallResult::Ok(_)
631        ),
632        Err(_) => false,
633    }
634}

638fn print_usize_hid(v: usize) {
639    let mut out = [0u8; 20];
640    let mut n = v;
641    let mut len = 0;
642    if n == 0 {
643        print("0");
644        return;
645    }
646    while n > 0 && len < out.len() {
647        out[len] = b'0' + (n % 10) as u8;
648        n /= 10;
649        len += 1;
650    }
651    out[..len].reverse();
652    if let Ok(s) = core::str::from_utf8(&out[..len]) {
653        print(s);
654    }
655}
656
657fn print_hex16(v: u16) {
658    const HEX: &[u8; 16] = b"0123456789abcdef";
659    let mut out = [0u8; 4];
660    for (i, slot) in out.iter_mut().enumerate() {
661        let nibble = (v >> (12 - i * 4)) & 0xF;
662        *slot = HEX[nibble as usize];
663    }
664    if let Ok(s) = core::str::from_utf8(&out) {
665        print(s);
666    }
667}
668
669fn print_u8(v: u8) {
670    let mut out = [0u8; 3];
671    let mut n = v;
672    let mut len = 0;
673    if n == 0 {
674        print("0");
675        return;
676    }
677    while n > 0 && len < out.len() {
678        out[len] = b'0' + (n % 10);
679        n /= 10;
680        len += 1;
681    }
682    out[..len].reverse();
683    if let Ok(s) = core::str::from_utf8(&out[..len]) {
684        print(s);
685    }
686}
