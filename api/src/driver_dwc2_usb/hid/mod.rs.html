1//! HID input: report-descriptor parsing, report decoding, and evdev translation.
2//!
3//! The pipeline is:
4//!
5//! ```text
6//!   GET_DESCRIPTOR(REPORT)  -> report::parse_report_descriptor  -> HidReportMap
7//!   Interrupt IN payload    -> decode::decode_auto              -> [HidValue]
8//!   [HidValue]              -> HidDecoder                       -> [EvdevEvent]
9//!   [EvdevEvent]            -> EvdevEvent::encode               -> input-service bytes
10//! ```
11//!
12//! A HID device mixes two field kinds and both must be handled, because which
13//! one a key uses is a descriptor author's choice:
14//!
15//! * **Array** fields carry the usage itself — a keyboard's six key slots, so a
16//!   key's *absence* is its release and the decoder derives edges by diffing.
17//! * **Variable** fields carry a value per entry — modifier bitmaps, mouse
18//!   button bits, and relative axes. Here a release *is* reported (value 0).
19//!
20//! The two kinds keep separate held-state so an array diff can never emit a
21//! release for a key that only ever appears as a variable field.
22
23pub mod decode;
24pub mod keymap;
25pub mod leds;
26pub mod report;

36// ─── Input-service wire opcodes (must match cells/services/input) ─────────────
37/// `EV_KEY` — a key or mouse button changed state.
38pub const EV_KEY: u8 = 0;
39/// `EV_REL` — a relative pointer axis moved.
40pub const EV_REL: u8 = 1;
41/// `EV_ABS` — an absolute pointer position (unused by this driver).
42pub const EV_ABS: u8 = 2;
43/// A HID event with a stable logical-device identity.
44pub const EV_DEVICE: u8 = 0x05;
45
46/// Host-local logical HID interface identity.
47#[derive(Clone, Copy, Debug, PartialEq, Eq)]
48pub struct HidDeviceId(pub u32);
49
50/// Length of a device-identified input event.
51///
52/// Layout: `[EV_DEVICE][device:u32][event_type][code:u32][value:i32]`.
53pub const DEVICE_EVENT_LEN: usize = 14;
54
55/// Serialise an event from one logical HID interface.
56pub fn encode_device_event(
57    device: HidDeviceId,
58    opcode: u8,
59    code: u32,
60    value: i32,
61    buf: &mut [u8; DEVICE_EVENT_LEN],
62) {
63    buf[0] = EV_DEVICE;
64    buf[1..5].copy_from_slice(&device.0.to_le_bytes());
65    buf[5] = opcode;
66    buf[6..10].copy_from_slice(&code.to_le_bytes());
67    buf[10..14].copy_from_slice(&value.to_le_bytes());
68}
69
70/// One decoded event, in evdev terms.
71#[derive(Clone, Copy, Debug, PartialEq, Eq)]
72pub enum EvdevEvent {
73    Key { code: u32, pressed: bool },
74    Rel { code: u32, value: i32 },
75}
76
77impl EvdevEvent {
78    /// Serialise an event with its originating HID logical-interface identity.
79    pub fn encode_device(&self, device: HidDeviceId, buf: &mut [u8; DEVICE_EVENT_LEN]) {
80        match *self {
81            EvdevEvent::Key { code, pressed } => {
82                encode_device_event(device, EV_KEY, code, if pressed { 1 } else { 0 }, buf)
83            }
84            EvdevEvent::Rel { code, value } => {
85                encode_device_event(device, EV_REL, code, value, buf)
86            }
87        }
88    }
89}

91/// Which device class a HID interface serves.
92#[derive(Clone, Copy, Debug, PartialEq, Eq)]
93pub enum HidKind {
94    Keyboard,
95    Mouse,
96    /// A combo dongle that packs both into one interface.
97    Combo,
98    Unknown,
99}
100
101impl HidKind {
102    /// Classify from the parsed report map's collections.
103    pub fn from_map(map: &HidReportMap) -> Self {
104        let k = map.has_collection(HidCollection::Keyboard);
105        let m = map.has_collection(HidCollection::Mouse);
106        match (k, m) {
107            (true, true) => HidKind::Combo,
108            (true, false) => HidKind::Keyboard,
109            (false, true) => HidKind::Mouse,
110            (false, false) => HidKind::Unknown,
111        }
112    }
113}

115/// Stateful decoder holding the previous report so edges can be derived.
116pub struct HidDecoder {
117    map: HidReportMap,
118    /// evdev codes currently down from **Array** fields (keyboard key slots).
119    held_array_keys: Vec<u32>,
120    /// `(evdev code, pressed)` from **Variable** fields — modifiers, buttons.
121    held_variable: Vec<(u32, bool)>,
122    /// Relative axes are noisy at rest; suppress explicit zero deltas.
123    suppress_zero_rel: bool,
124}
125
126impl HidDecoder {
127    pub fn new(map: HidReportMap) -> Self {
128        Self {
129            map,
130            held_array_keys: Vec::new(),
131            held_variable: Vec::new(),
132            suppress_zero_rel: true,
133        }
134    }
135
136    pub fn map(&self) -> &HidReportMap {
137        &self.map
138    }
139
140    pub fn kind(&self) -> HidKind {
141        HidKind::from_map(&self.map)
142    }
143
144    /// Feed one raw report and receive the events it implies.
145    pub fn process(&mut self, report: &[u8], out: &mut Vec<EvdevEvent>) {
146        let values = decode_auto(&self.map, report);
147
148        self.process_array_keys(&values, out);
149        self.process_variable_fields(&values, out);
150    }
151
152    /// Array fields: the report lists keys currently down, so diff the set.
153    fn process_array_keys(&mut self, values: &[HidValue], out: &mut Vec<EvdevEvent>) {
154        let mut now_held: Vec<u32> = Vec::new();
155        for v in values.iter().filter(|v| !v.is_variable) {
156            if v.page != keymap::PAGE_KEYBOARD {
157                continue;
158            }
159            if let Some(code) = keymap::keyboard_to_evdev(v.usage) {
160                if !now_held.contains(&code) {
161                    now_held.push(code);
162                }
163            }
164        }
165
166        for code in &now_held {
167            if !self.held_array_keys.contains(code) {
168                out.push(EvdevEvent::Key {
169                    code: *code,
170                    pressed: true,
171                });
172            }
173        }
174        for code in &self.held_array_keys {
175            if !now_held.contains(code) {
176                out.push(EvdevEvent::Key {
177                    code: *code,
178                    pressed: false,
179                });
180            }
181        }
182        self.held_array_keys = now_held;
183    }
184
185    /// Variable fields: every entry reports its own value, including releases.
186    fn process_variable_fields(&mut self, values: &[HidValue], out: &mut Vec<EvdevEvent>) {
187        for v in values.iter().filter(|v| v.is_variable) {
188            // Relative axes move the pointer; nothing else uses value semantics.
189            if v.page == keymap::PAGE_GENERIC_DESKTOP && v.collection == HidCollection::Mouse {
190                if let Some(rel) = keymap::generic_desktop_to_rel(v.usage) {
191                    if v.value != 0 || !self.suppress_zero_rel {
192                        out.push(EvdevEvent::Rel {
193                            code: rel,
194                            value: v.value,
195                        });
196                    }
197                }
198                continue;
199            }
200
201            // Modifier bitmaps and mouse buttons are edge-producing.
202            let Some((code, _is_button)) = keymap::hid_field_to_evdev(v.page, v.usage) else {
203                continue;
204            };
205            let pressed = v.value != 0;
206            let prev = self
207                .held_variable
208                .iter()
209                .find(|(c, _)| *c == code)
210                .map(|(_, p)| *p);
211            if prev == Some(pressed) {
212                continue;
213            }
214            match self.held_variable.iter_mut().find(|(c, _)| *c == code) {
215                Some(entry) => entry.1 = pressed,
216                None => self.held_variable.push((code, pressed)),
217            }
218            out.push(EvdevEvent::Key { code, pressed });
219        }
220    }
221}

223/// Held state for the boot-protocol fallback path.
224///
225/// Kept separate per device class so a keyboard report's diff cannot release a
226/// mouse button (they travel in different report shapes but share this decoder).
227#[derive(Default)]
228pub struct BootState {
229    keys: Vec<u32>,
230    buttons: Vec<u32>,
231}
232
233/// Decode a HID **boot protocol** report without any descriptor.
234///
235/// Boot protocol is the fixed layout a device must also support so a BIOS can
236/// use it (`HID 1.11` Appendix B): keyboards send 8 bytes — modifier bitmap,
237/// reserved, then six usage slots — and mice send 3 (or 4 with a wheel).
238///
239/// Used as the fallback when a device's report descriptor is missing or fails
240/// to parse. A combo receiver usually implements it for both halves, which is
241/// exactly why it is worth keeping alongside the descriptor path.
242pub fn decode_boot_report(
243    kind: HidKind,
244    report: &[u8],
245    state: &mut BootState,
246    out: &mut Vec<EvdevEvent>,
247) {
248    match kind {
249        HidKind::Keyboard | HidKind::Combo if report.len() >= 8 => {
250            let mut now_held: Vec<u32> = Vec::new();
251            for code in keymap::boot_modifiers(report[0]) {
252                if !now_held.contains(&code) {
253                    now_held.push(code);
254                }
255            }
256            for &usage in &report[2..8] {
257                if usage == 0 {
258                    continue;
259                }
260                if let Some(code) = keymap::keyboard_to_evdev(usage as u32) {
261                    if !now_held.contains(&code) {
262                        now_held.push(code);
263                    }
264                }
265            }
266            for code in &now_held {
267                if !state.keys.contains(code) {
268                    out.push(EvdevEvent::Key {
269                        code: *code,
270                        pressed: true,
271                    });
272                }
273            }
274            for code in &state.keys {
275                if !now_held.contains(code) {
276                    out.push(EvdevEvent::Key {
277                        code: *code,
278                        pressed: false,
279                    });
280                }
281            }
282            state.keys = now_held;
283        }
284        HidKind::Mouse if report.len() >= 3 => {
285            let buttons = report[0];
286            for (bit, code) in [
287                keymap::BTN_LEFT,
288                keymap::BTN_RIGHT,
289                keymap::BTN_MIDDLE,
290                keymap::BTN_SIDE,
291                keymap::BTN_EXTRA,
292            ]
293            .iter()
294            .enumerate()
295            {
296                let pressed = buttons & (1 << bit) != 0;
297                let prev = state.buttons.contains(code);
298                if pressed == prev {
299                    continue;
300                }
301                if pressed {
302                    state.buttons.push(*code);
303                } else {
304                    state.buttons.retain(|c| c != code);
305                }
306                out.push(EvdevEvent::Key {
307                    code: *code,
308                    pressed,
309                });
310            }
311            // Boot mouse deltas are signed 8-bit.
312            let dx = report[1] as i8 as i32;
313            let dy = report[2] as i8 as i32;
314            if dx != 0 {
315                out.push(EvdevEvent::Rel {
316                    code: keymap::REL_X,
317                    value: dx,
318                });
319            }
320            if dy != 0 {
321                out.push(EvdevEvent::Rel {
322                    code: keymap::REL_Y,
323                    value: dy,
324                });
325            }
326            if report.len() >= 4 {
327                let wheel = report[3] as i8 as i32;
328                if wheel != 0 {
329                    out.push(EvdevEvent::Rel {
330                        code: keymap::REL_WHEEL,
331                        value: wheel,
332                    });
333                }
334            }
335        }
336        _ => {}
337    }
