1//! HID report descriptor parser (HID 1.11 §6.2.2).
2//!
3//! Walks the item stream a HID device returns for `GET_DESCRIPTOR(REPORT)` and
4//! builds a flat field map: which bit range of which report carries which
5//! usage. This is what lets the driver support arbitrary HID devices instead of
6//! only the fixed boot-protocol reports.
7//!
8//! Main items (`Input`/`Output`/`Feature`) consume the accumulated local usages
9//! and emit a field spanning `report_count` entries of `report_size` bits each,
10//! in declaration order within their `report_id`. Global items mutate state that
11//! persists until changed (with `Push`/`Pop`); local items are cleared after
12//! every main item, per spec.
13
14extern crate alloc;

18/// The top-level collection a field belongs to.
19///
20/// A combo receiver exposes a keyboard interface and a mouse interface, and may
21/// pack both into one report, so routing needs the collection, not just the
22/// usage page.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum HidCollection {
25    Other,
26    Keyboard,
27    Mouse,
28    Consumer,
29}
30
31/// One decoded field declaration from the report descriptor.
32///
33/// A single `Input` main item can span several entries (a mouse's X/Y/Wheel, a
34/// keyboard's six key slots), and its local usage list or range says what each
35/// entry means. Both forms are common in the wild, so the field keeps the whole
36/// list rather than collapsing to its first element.
37#[derive(Clone, Debug)]
38pub struct HidField {
39    pub report_id: u8,
40    /// Bit offset within the report payload (after any report-ID byte).
41    pub offset_bits: u32,
42    pub size_bits: u32,
43    pub count: u32,
44    /// Explicit usages from local `Usage` items, page in the high 16 bits.
45    pub usages: Vec<u32>,
46    /// Range start from `Usage Minimum` (page in the high 16 bits).
47    pub usage_min: Option<u32>,
48    /// Range end from `Usage Maximum` (page in the high 16 bits).
49    pub usage_max: Option<u32>,
50    /// `true` for `Variable` (one value per entry), `false` for `Array`
51    /// (the entry carries the usage index itself — how keyboards report keys).
52    pub is_variable: bool,
53    pub logical_min: i32,
54    pub logical_max: i32,
55    pub collection: HidCollection,
56}
57
58impl HidField {
59    /// Usage for entry `index` of a `Variable` field.
60    ///
61    /// Explicit usages win; otherwise the entry maps into the declared range.
62    /// Returns `None` when the descriptor said nothing usable about this entry,
63    /// so the caller drops it instead of inventing a usage.
64    pub fn variable_usage(&self, index: u32) -> Option<u32> {
65        if let Some(u) = self.usages.get(index as usize) {
66            return Some(*u);
67        }
68        let (min, max) = (self.usage_min?, self.usage_max?);
69        if index > max.saturating_sub(min) {
70            return None;
71        }
72        Some(min.saturating_add(index))
73    }
74
75    /// Usage an `Array` entry's value denotes.
76    ///
77    /// Array values are indices into the declared usage range; a keyboard
78    /// declares `Usage Minimum 0`, so its value is already the usage ID.
79    pub fn array_usage(&self, value: u32) -> Option<u32> {
80        match (self.usage_min, self.usage_max) {
81            (Some(min), Some(max)) => {
82                let base = min & 0xFFFF;
83                let limit = max & 0xFFFF;
84                if value >= base && value <= limit {
85                    Some((min & 0xFFFF_0000) | value)
86                } else if value <= limit.saturating_sub(base) {
87                    Some((min & 0xFFFF_0000) | base.saturating_add(value))
88                } else {
89                    None
90                }
91            }
92            _ => None,
93        }
94    }
95}

97/// Every field of one report ID.
98#[derive(Clone, Debug)]
99pub struct ReportLayout {
100    pub report_id: u8,
101    pub fields: Vec<HidField>,
102    /// Total payload bits (excluding the report-ID byte).
103    pub total_bits: u32,
104}
105
106/// Parse result for a whole report descriptor.
107#[derive(Clone, Debug, Default)]
108pub struct HidReportMap {
109    pub reports: Vec<ReportLayout>,
110    /// True when the descriptor declares any Report ID — reports then carry a
111    /// leading report-ID byte that the caller must strip.
112    pub uses_report_ids: bool,
113}
114
115impl HidReportMap {
116    pub fn layout(&self, report_id: u8) -> Option<&ReportLayout> {
117        self.reports.iter().find(|r| r.report_id == report_id)
118    }
119
120    /// The report ID a device sends when only one report exists.
121    pub fn sole_report_id(&self) -> Option<u8> {
122        if self.reports.len() == 1 {
123            Some(self.reports[0].report_id)
124        } else {
125            None
126        }
127    }
128
129    pub fn has_collection(&self, c: HidCollection) -> bool {
130        self.reports
131            .iter()
132            .any(|r| r.fields.iter().any(|f| f.collection == c))
133    }
134}

155/// Sign-extend an item's payload, which HID encodes as a signed integer.
156fn sign_extend(raw: u32, size: usize) -> i32 {
157    match size {
158        1 => raw as u8 as i8 as i32,
159        2 => raw as u16 as i16 as i32,
160        _ => raw as i32,
161    }
162}
163
164/// Sign-extend a logical bound, clamped to the field's own bit width.
165///
166/// A `Logical Minimum` of `0x8000` with `Report Size` 16 is −32768, but devices
167/// also declare bounds wider than the field; clamping keeps the decode in range.
168fn clamp_logical(value: i32, size_bits: u32) -> i32 {
169    if size_bits == 0 || size_bits >= 32 {
170        return value;
171    }
172    let half = 1i64 << (size_bits - 1);
173    let max = half - 1;
174    let min = -half;
175    (value as i64).clamp(min, max) as i32
176}

178/// Parse a HID report descriptor into a field map.
179///
180/// Unknown items are skipped by size, which is what makes this tolerant of
181/// vendor-specific descriptors: an unrecognised tag cannot desynchronise the
182/// stream because the length prefix is always authoritative.
183pub fn parse_report_descriptor(data: &[u8]) -> HidReportMap {
184    let mut map = HidReportMap::default();
185    let mut globals = Globals::default();
186    let mut globals_stack: Vec<Globals> = Vec::new();
187    let mut locals = Locals::default();
188    // Collection stack: (usage_page, usage) of each open collection.
189    let mut collections: Vec<(u32, u32)> = Vec::new();
190
191    let mut i = 0usize;
192    while i < data.len() {
193        let prefix = data[i];
194        i += 1;
195
196        // Long item (0xFE): [0xFE][bDataSize][bLongItemTag][data…]
197        if prefix == 0xFE {
198            if i + 1 >= data.len() {
199                break;
200            }
201            let dsize = data[i] as usize;
202            i += 2;
203            i = i.saturating_add(dsize).min(data.len());
204            continue;
205        }
206
207        let size = match prefix & 0x03 {
208            0 => 0usize,
209            1 => 1,
210            2 => 2,
211            _ => 4,
212        };
213        let item_type = (prefix >> 2) & 0x03;
214        let tag = (prefix >> 4) & 0x0F;
215
216        if i + size > data.len() {
217            break;
218        }
219        let mut raw: u32 = 0;
220        for (k, byte) in data[i..i + size].iter().enumerate() {
221            raw |= (*byte as u32) << (8 * k);
222        }
223        i += size;
224
225        match item_type {
226            // ── Main items ─────────────────────────────────────────────────
227            0 => match tag {
228                0x8 | 0x9 | 0xB => {
229                    // Input / Output / Feature. Only Input carries device→host
230                    // data; Output (LEDs) and Feature are not decoded.
231                    if tag == 0x8 && globals.report_size > 0 && globals.report_count > 0 {
232                        let flags = raw;
233                        let is_variable = flags & 0x02 != 0;
234
235                        let report_id = globals.report_id;
236                        let offset = match map.reports.iter().find(|r| r.report_id == report_id) {
237                            Some(r) => r.total_bits,
238                            None => {
239                                map.reports.push(ReportLayout {
240                                    report_id,
241                                    fields: Vec::new(),
242                                    total_bits: 0,
243                                });
244                                0
245                            }
246                        };
247
248                        let collection = classify_collection(&collections);
249                        let field = HidField {
250                            report_id,
251                            offset_bits: offset,
252                            size_bits: globals.report_size,
253                            count: globals.report_count,
254                            usages: locals.usages.clone(),
255                            usage_min: locals.usage_min,
256                            usage_max: locals.usage_max,
257                            is_variable,
258                            logical_min: globals.logical_min,
259                            logical_max: globals.logical_max,
260                            collection,
261                        };
262
263                        let bits = globals.report_size.saturating_mul(globals.report_count);
264                        if let Some(r) = map.reports.iter_mut().find(|r| r.report_id == report_id) {
265                            r.fields.push(field);
266                            r.total_bits = r.total_bits.saturating_add(bits);
267                        }
268                    }
269                    locals = Locals::default();
270                }
271                0xA => {
272                    // Collection start
273                    let page = locals
274                        .usages
275                        .first()
276                        .map(|u| u >> 16)
277                        .unwrap_or(globals.usage_page);
278                    let usage = locals.usages.first().map(|u| u & 0xFFFF).unwrap_or(0);
279                    collections.push((page, usage));
280                    locals = Locals::default();
281                }
282                0xC => {
283                    // End Collection
284                    collections.pop();
285                    locals = Locals::default();
286                }
287                _ => locals = Locals::default(),
288            },
289
290            // ── Global items ───────────────────────────────────────────────
291            1 => match tag {
292                0x0 => globals.usage_page = raw,
293                0x1 => globals.logical_min = sign_extend(raw, size),
294                0x2 => globals.logical_max = sign_extend(raw, size),
295                0x7 => globals.report_size = raw,
296                0x8 => {
297                    globals.report_id = raw as u8;
298                    if raw != 0 {
299                        map.uses_report_ids = true;
300                    }
301                }
302                0x9 => globals.report_count = raw,
303                0xA => globals_stack.push(globals),
304                0xB => {
305                    if let Some(g) = globals_stack.pop() {
306                        globals = g;
307                    }
308                }
309                // Physical Min/Max, Unit, Unit Exponent: parsed but unused —
310                // they describe scaling, not layout.
311                0x3..=0x6 => {}
312                _ => {}
313            },
314
315            // ── Local items ────────────────────────────────────────────────
316            2 => match tag {
317                0x0 => {
318                    // A bare Usage after a Usage Page inherits that page unless
319                    // it is a 32-bit extended usage (high half carries its own).
320                    let u = if size == 4 {
321                        raw
322                    } else {
323                        (globals.usage_page << 16) | raw
324                    };
325                    locals.usages.push(u);
326                }
327                0x1 => {
328                    locals.usage_min = Some((globals.usage_page << 16) | (raw & 0xFFFF));
329                }
330                0x2 => {
331                    locals.usage_max = Some((globals.usage_page << 16) | (raw & 0xFFFF));
332                }
333                // Designator / String index items do not affect the bit layout.
334                _ => {}
335            },
336
337            // ── Reserved ───────────────────────────────────────────────────
338            _ => {}
339        }
340    }
341
342    // Logical bounds were parsed before `report_size` may have been finalised;
343    // clamp them now against each field's own width.
344    for r in map.reports.iter_mut() {
345        for f in r.fields.iter_mut() {
346            f.logical_min = clamp_logical(f.logical_min, f.size_bits);
347            f.logical_max = clamp_logical(f.logical_max, f.size_bits);
348        }
349    }
350
351    map
352}

354/// Classify the innermost recognised collection in the open stack.
355fn classify_collection(open: &[(u32, u32)]) -> HidCollection {
356    use crate::hid::keymap::{
357        GD_KEYBOARD, GD_MOUSE, GD_POINTER, PAGE_CONSUMER, PAGE_GENERIC_DESKTOP,
358    };
359    for (page, usage) in open.iter().rev() {
360        match (*page, *usage) {
361            (PAGE_GENERIC_DESKTOP, GD_KEYBOARD) => return HidCollection::Keyboard,
362            (PAGE_GENERIC_DESKTOP, GD_MOUSE) | (PAGE_GENERIC_DESKTOP, GD_POINTER) => {
363                return HidCollection::Mouse
364            }
365            (PAGE_CONSUMER, _) => return HidCollection::Consumer,
366            _ => {}
367        }
368    }
369    HidCollection::Other
370}
