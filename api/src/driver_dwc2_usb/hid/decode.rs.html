1//! HID input report decoder.
2//!
3//! Turns a raw report payload into `(usage_page, usage, value)` triples using
4//! the layout [`super::report`] extracted from the report descriptor. Handles
5//! both field kinds a real device mixes:
6//!
7//! * **Variable** fields — one value per entry (mouse X/Y/Wheel, button bits).
8//! * **Array** fields — the entry carries the usage (a keyboard's key slots).
9//!
10//! Report IDs are stripped here, so callers pass the payload as the device sent
11//! it and the decoder splits it by the layout's declarations.
12//!
13//! Note on "no change": an all-zero report is meaningful (every key released),
14//! so the decoder reports every entry it can resolve and lets the caller diff
15//! against its previous state. It never filters zeroes itself — a filter here
16//! would swallow releases.
17

24/// One decoded entry.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub struct HidValue {
27    pub page: u32,
28    pub usage: u32,
29    /// Raw field value, sign-extended per the field's bit width.
30    pub value: i32,
31    pub collection: super::report::HidCollection,
32    /// `true` when the entry came from a `Variable` field (a value), `false`
33    /// when from an `Array` field (the entry *is* the usage). Modifier keys and
34    /// mouse buttons are Variable; a keyboard's key slots are Array.
35    pub is_variable: bool,
36}

38/// Extract a `size_bits`-wide little-endian bit field from `data`.
39///
40/// Returns `None` when the field runs past the payload, which is how a
41/// truncated or mis-framed report is rejected instead of read out of bounds.
42fn extract_bits(data: &[u8], offset_bits: u32, size_bits: u32) -> Option<u32> {
43    if size_bits == 0 || size_bits > 32 {
44        return None;
45    }
46    let end = offset_bits.checked_add(size_bits)?;
47    if (end as usize).div_ceil(8) > data.len() {
48        return None;
49    }
50    let mut acc: u64 = 0;
51    for bit in 0..size_bits {
52        let src = offset_bits + bit;
53        let byte = data[(src / 8) as usize];
54        if byte & (1 << (src % 8)) != 0 {
55            acc |= 1u64 << bit;
56        }
57    }
58    Some(acc as u32)
59}

61/// Sign-extend a raw field value using the field's declared logical range.
62///
63/// HID packs signed values two's-complement into `report_size` bits, so the
64/// width — not the logical maximum — decides the sign bit.
65fn to_signed(raw: u32, size_bits: u32) -> i32 {
66    if size_bits == 0 || size_bits >= 32 {
67        return raw as i32;
68    }
69    let sign_bit = 1u32 << (size_bits - 1);
70    if raw & sign_bit != 0 {
71        (raw as i64 - (1i64 << size_bits)) as i32
72    } else {
73        raw as i32
74    }
75}

77/// Decode one report payload against its layout.
78///
79/// `report_id` selects the layout; pass the device's report-ID byte. For devices
80/// that use a single unnumbered report, pass 0.
81pub fn decode_report(map: &HidReportMap, report_id: u8, payload: &[u8]) -> Vec<HidValue> {
82    let Some(layout) = map.layout(report_id) else {
83        return Vec::new();
84    };
85    let mut out = Vec::new();
86    for field in &layout.fields {
87        decode_field(field, payload, &mut out);
88    }
89    out
90}

92/// Decode a report whose payload may or may not carry a leading report-ID byte.
93///
94/// HID devices that declare `Report ID` prefix every report with it; devices
95/// that do not must not have one stripped. Getting this wrong shifts the whole
96/// bit stream, so the decision is made from the parsed descriptor.
97pub fn decode_auto(map: &HidReportMap, report: &[u8]) -> Vec<HidValue> {
98    if map.uses_report_ids {
99        let Some((&id, rest)) = report.split_first() else {
100            return Vec::new();
101        };
102        decode_report(map, id, rest)
103    } else {
104        let id = map.sole_report_id().unwrap_or(0);
105        decode_report(map, id, report)
106    }
107}

109fn decode_field(field: &HidField, payload: &[u8], out: &mut Vec<HidValue>) {
110    if field.is_variable {
111        for index in 0..field.count {
112            let offset = field.offset_bits + index * field.size_bits;
113            let Some(raw) = extract_bits(payload, offset, field.size_bits) else {
114                return;
115            };
116            let Some(usage) = field.variable_usage(index) else {
117                continue;
118            };
119            out.push(HidValue {
120                page: usage >> 16,
121                usage: usage & 0xFFFF,
122                value: to_signed(raw, field.size_bits),
123                collection: field.collection,
124                is_variable: true,
125            });
126        }
127    } else {
128        // Array field: each entry is a usage index. Duplicate entries are
129        // dropped — a keyboard lists a key once even if it appears twice.
130        let mut seen: Vec<u32> = Vec::new();
131        for index in 0..field.count {
132            let offset = field.offset_bits + index * field.size_bits;
133            let Some(raw) = extract_bits(payload, offset, field.size_bits) else {
134                return;
135            };
136            let Some(usage) = field.array_usage(raw) else {
137                continue;
138            };
139            let usage_code = usage & 0xFFFF;
140            if usage_code == 0 || seen.contains(&usage_code) {
141                continue;
142            }
143            seen.push(usage_code);
144            out.push(HidValue {
145                page: usage >> 16,
146                usage: usage_code,
147                value: 1,
148                collection: field.collection,
149                is_variable: false,
150            });
151        }
152    }
153}
