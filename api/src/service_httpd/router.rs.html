1//! Route classification is pure: all network and backend work belongs to the reactor.
2use crate::net_ipc::{request_complete_len, RequestReadError};
3
4pub(crate) enum Route<'a> {
5    Index,
6    Status,
7    ApiStatus,
8    ApiCells,
9    File(&'a str),
10    Files(&'a str),
11    #[cfg(target_os = "none")]
12    Infer { body: &'a [u8], max_tokens: u16 },
13    Restart,
14    NotFound,
15}
16
17pub(crate) fn classify<'a>(raw: &'a [u8], file_to_serve: Option<&'a str>) -> Result<Route<'a>, RequestReadError> {
18    let len = request_complete_len(raw)?.ok_or(RequestReadError::Incomplete)?;
19    if raw.len() < len { return Err(RequestReadError::Incomplete); }
20    if let Some(file) = file_to_serve { return Ok(Route::File(file)); }
21    let mut headers = [httparse::EMPTY_HEADER; 16];
22    let mut req = httparse::Request::new(&mut headers);
23    #[cfg_attr(not(target_os = "none"), allow(unused_variables))]
24    let header_end = match req.parse(&raw[..len]) {
25        Ok(httparse::Status::Complete(n)) => n,
26        _ => return Err(RequestReadError::BadRequest),
27    };
28    let path = req.path.ok_or(RequestReadError::BadRequest)?;
29    let path_only = path.split('?').next().unwrap_or(path);
30    Ok(match (req.method.unwrap_or("GET"), path_only) {
31        ("GET", "/") => Route::Index,
32        ("GET", "/status") => Route::Status,
33        ("GET", path) if path.starts_with("/files/") => Route::File(&path["/files".len()..]),
34        ("GET", "/api/status") | ("GET", "/api/system") => Route::ApiStatus,
35        ("GET", "/api/cells") => Route::ApiCells,
36        ("GET", "/api/files") => Route::Files(extract_query_param(path, "path").unwrap_or("/")),
37        #[cfg(target_os = "none")]
38        ("POST", "/api/infer") => Route::Infer {
39            body: &raw[header_end..len],
40            max_tokens: extract_query_param(path, "max_tokens")
41                .and_then(|value| value.parse::<u16>().ok()).unwrap_or(24).clamp(1, 64),
42        },
43        ("POST", path) if path.starts_with("/api/cells/") && path.ends_with("/restart") => Route::Restart,
44        _ => Route::NotFound,
45    })
46}
47
48pub fn extract_query_param<'a>(url: &'a str, key: &str) -> Option<&'a str> {
49    let query = url.split('?').nth(1)?;
50    for pair in query.split('&') {
51        if let Some((k, v)) = pair.split_once('=') {
52            if k == key { return Some(v); }
53        }
54    }
55    None
56}
