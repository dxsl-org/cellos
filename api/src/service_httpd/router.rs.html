1//! Route classification is pure: all network and backend work belongs to the reactor.
2use crate::net_ipc::{request_complete_len, RequestReadError};
3
4pub(crate) enum Route<'a> {
5    Index,
6    Status,
7    ApiStatus,
8    ApiCells,
9    ApiCluster,
10    ApiGatewayStats,
11    ApiC2cDispatch { body: &'a [u8] },
12    File(&'a str),
13    Files(&'a str),
14    #[cfg(target_os = "none")]
15    Infer { body: &'a [u8], max_tokens: u16 },
16    Restart,
17    NotFound,
18}
19
20pub(crate) fn classify<'a>(raw: &'a [u8], file_to_serve: Option<&'a str>) -> Result<Route<'a>, RequestReadError> {
21    let len = request_complete_len(raw)?.ok_or(RequestReadError::Incomplete)?;
22    if raw.len() < len { return Err(RequestReadError::Incomplete); }
23    if let Some(file) = file_to_serve { return Ok(Route::File(file)); }
24    let mut headers = [httparse::EMPTY_HEADER; 16];
25    let mut req = httparse::Request::new(&mut headers);
26    #[cfg_attr(not(target_os = "none"), allow(unused_variables))]
27    let header_end = match req.parse(&raw[..len]) {
28        Ok(httparse::Status::Complete(n)) => n,
29        _ => return Err(RequestReadError::BadRequest),
30    };
31    let path = req.path.ok_or(RequestReadError::BadRequest)?;
32    let path_only = path.split('?').next().unwrap_or(path);
33    Ok(match (req.method.unwrap_or("GET"), path_only) {
34        ("GET", "/") => Route::Index,
35        ("GET", "/status") => Route::Status,
36        ("GET", path) if path.starts_with("/files/") => Route::File(&path["/files".len()..]),
37        ("GET", "/api/status") | ("GET", "/api/system") => Route::ApiStatus,
38        ("GET", "/api/cells") => Route::ApiCells,
39        ("GET", "/api/cluster") | ("GET", "/api/cluster/status") => Route::ApiCluster,
40        ("GET", "/api/gateway") | ("GET", "/api/gateway/stats") => Route::ApiGatewayStats,
41        ("POST", "/api/c2c/dispatch") => Route::ApiC2cDispatch {
42            body: &raw[header_end..len],
43        },
44        ("GET", "/api/files") => Route::Files(extract_query_param(path, "path").unwrap_or("/")),
45        #[cfg(target_os = "none")]
46        ("POST", "/api/infer") => Route::Infer {
47            body: &raw[header_end..len],
48            max_tokens: extract_query_param(path, "max_tokens")
49                .and_then(|value| value.parse::<u16>().ok()).unwrap_or(24).clamp(1, 64),
50        },
51        ("POST", path) if path.starts_with("/api/cells/") && path.ends_with("/restart") => Route::Restart,
52        _ => Route::NotFound,
53    })
54}

56pub fn extract_query_param<'a>(url: &'a str, key: &str) -> Option<&'a str> {
57    let query = url.split('?').nth(1)?;
58    for pair in query.split('&') {
59        if let Some((k, v)) = pair.split_once('=') {
60            if k == key { return Some(v); }
61        }
62    }
63    None
64}

67mod tests {

70    #[test]
71    fn test_classify_cluster_and_gateway_routes() {
72        let req = b"GET /api/cluster HTTP/1.1\r\nHost: 127.0.0.1\r\n\r\n";
73        assert!(matches!(classify(req, None), Ok(Route::ApiCluster)));
74
75        let req_stats = b"GET /api/gateway/stats HTTP/1.1\r\nHost: 127.0.0.1\r\n\r\n";
76        assert!(matches!(classify(req_stats, None), Ok(Route::ApiGatewayStats)));
77    }
78
79    #[test]
80    fn test_classify_c2c_dispatch_payload() {
81        let req = b"POST /api/c2c/dispatch HTTP/1.1\r\nContent-Length: 9\r\n\r\nhello-c2c";
82        match classify(req, None) {
83            Ok(Route::ApiC2cDispatch { body }) => {
84                assert_eq!(body, b"hello-c2c");
85            }
86            other => panic!("expected ApiC2cDispatch, got {:?}", other.err()),
87        }
