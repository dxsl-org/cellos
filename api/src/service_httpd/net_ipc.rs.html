1//! Owned, nonblocking IPC submissions for the HTTP reactor.
2//! Every request is copied into the kernel's bounded operation queue before returning.
3
4use api::ipc::{NetRequest, NetResponse, VfsRequest, VfsResponse, IPC_BUF_SIZE};
5use ostd::ipc::{self, IpcOpId, IpcSubmitError, IpcTakeResult, IpcTerminal};
6use serde::Serialize;
7
8pub(crate) type Op = IpcOpId;
9
10pub(crate) fn submit<T: Serialize>(peer: usize, req: &T) -> Result<Op, IpcSubmitError> {
11    let mut bytes = [0u8; IPC_BUF_SIZE];
12    let encoded = api::ipc::encode(req, &mut bytes).map_err(|_| IpcSubmitError::InvalidRequest)?;
13    ipc::submit(peer, encoded)
14}
15
16/// A terminal operation is removed from kernel storage on successful take.
17/// The borrowed response must not outlive `reply`.
18pub(crate) fn take<'a>(op: Op, reply: &'a mut [u8; IPC_BUF_SIZE]) -> Option<Result<&'a [u8], IpcTerminal>> {
19    match ipc::take(op, reply) {
20        Ok(IpcTakeResult::Pending) => None,
21        Ok(IpcTakeResult::Terminal { status: IpcTerminal::Reply, len }) if len <= reply.len() => Some(Ok(&reply[..len])),
22        Ok(IpcTakeResult::Terminal { status, .. }) => Some(Err(status)),
23        Err(_) => Some(Err(IpcTerminal::Indeterminate)),
24    }
25}

43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub enum RequestReadError {
45    Incomplete,
46    BadRequest,
47    TooLarge,
48}
49
50pub(crate) const MAX_HTTP_REQUEST_BYTES: usize = 4096;
51
52/// Exactly one HTTP/1.1 request, never dispatch a partially received body.
53pub(crate) fn request_complete_len(buf: &[u8]) -> Result<Option<usize>, RequestReadError> {
54    let Some(header_end) = buf.windows(4).position(|part| part == b"\r\n\r\n") else {
55        return Ok(None);
56    };
57    let header_end = header_end + 4;
58    let mut headers = [httparse::EMPTY_HEADER; 16];
59    let mut request = httparse::Request::new(&mut headers);
60    if !matches!(request.parse(&buf[..header_end]), Ok(httparse::Status::Complete(_))) {
61        return Err(RequestReadError::BadRequest);
62    }
63    let mut body_len = None;
64    for header in request.headers.iter() {
65        if header.name.eq_ignore_ascii_case("Transfer-Encoding") {
66            return Err(RequestReadError::BadRequest);
67        }
68        if header.name.eq_ignore_ascii_case("Content-Length") {
69            if body_len.is_some() {
70                return Err(RequestReadError::BadRequest);
71            }
72            let value = core::str::from_utf8(header.value).map_err(|_| RequestReadError::BadRequest)?;
73            body_len = Some(value.trim().parse::<usize>().map_err(|_| RequestReadError::BadRequest)?);
74        }
75    }
76    let expected = header_end.checked_add(body_len.unwrap_or(0)).ok_or(RequestReadError::TooLarge)?;
77    if expected > MAX_HTTP_REQUEST_BYTES {
78        return Err(RequestReadError::TooLarge);
79    }
80    Ok(Some(expected))
81}

83#[cfg(test)]
84mod framing_tests {
85    use super::{request_complete_len, RequestReadError};
86
87    #[test]
88    fn fragmented_header_and_body_remain_incomplete() {
89        let header = b"POST /api/infer HTTP/1.1\r\nContent-Length: 5\r\n\r\n";
90        assert_eq!(request_complete_len(&header[..header.len() - 1]), Ok(None));
91        assert_eq!(request_complete_len(header), Ok(Some(header.len() + 5)));
92        let mut partial = header.to_vec();
93        partial.extend_from_slice(b"abc");
94        assert_eq!(request_complete_len(&partial), Ok(Some(header.len() + 5)));
95    }
96
97    #[test]
98    fn conflicting_framing_never_dispatches() {
99        for header in [
100            b"POST / HTTP/1.1\r\nContent-Length: 1\r\nContent-Length: 2\r\n\r\nx".as_slice(),
101            b"POST / HTTP/1.1\r\nTransfer-Encoding: chunked\r\n\r\n".as_slice(),
102            b"POST / HTTP/1.1\r\nContent-Length: 1\r\nTransfer-Encoding: chunked\r\n\r\nx".as_slice(),
103            b"POST / HTTP/1.1\r\nContent-Length: -2\r\n\r\n".as_slice(),
104        ] {
105            assert_eq!(request_complete_len(header), Err(RequestReadError::BadRequest));
106        }
107    }
108
109    #[test]
110    fn oversized_or_overflowing_length_is_rejected_before_dispatch() {
111        let huge = b"POST / HTTP/1.1\r\nContent-Length: 18446744073709551615\r\n\r\n";
112        assert_eq!(request_complete_len(huge), Err(RequestReadError::TooLarge));
113        let big = b"POST / HTTP/1.1\r\nContent-Length: 4096\r\n\r\n";
114        assert_eq!(request_complete_len(big), Err(RequestReadError::TooLarge));
115    }
