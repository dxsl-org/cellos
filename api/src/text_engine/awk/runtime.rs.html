8pub(crate) fn run(
9    program: &Program,
10    separator: Separator<'_>,
11    input: &str,
12) -> Result<Vec<String>, AwkError> {
13    let regex = match &program.filter {
14        Some(Filter::Regex(pattern)) => Some(
15            CompiledPattern::compile(PatternKind::ERELite, pattern, false)
16                .map_err(|_| AwkError::Runtime("invalid regex filter"))?,
17        ),
18        _ => None,
19    };
20    let records = RecordReader::new(input)
21        .map_err(AwkError::Record)?
22        .collect()
23        .map_err(AwkError::Record)?;
24    let mut output = Vec::new();
25    for (index, record) in records.iter().enumerate() {
26        let fields = split_fields(record, separator);
27        let env = Env {
28            line: record,
29            fields: &fields,
30            nr: index + 1,
31        };
32        if !selected(program.filter.as_ref(), regex.as_ref(), &env)? {
33            continue;
34        }
35        let line = render(program, &env)?;
36        if line.len() > MAX_RECORD_BYTES {
37            return Err(AwkError::Runtime("output record exceeds configured limit"));
38        }
39        output.push(line);
40    }
41    Ok(output)
42}

44fn split_fields<'a>(line: &'a str, separator: Separator<'_>) -> Vec<&'a str> {
45    match separator {
46        Separator::Whitespace => line.split_whitespace().collect(),
47        Separator::Literal(sep) => line.split(sep).collect(),
48    }
49}

51fn selected(
52    filter: Option<&Filter>,
53    regex: Option<&CompiledPattern>,
54    env: &Env<'_>,
55) -> Result<bool, AwkError> {
56    match filter {
57        None => Ok(true),
58        Some(Filter::Regex(_)) => Ok(regex
59            .map(|compiled| compiled.is_match(env.line))
60            .unwrap_or(false)),
61        Some(Filter::Expr(expr)) => Ok(eval(expr, env)?.truthy()),
62    }
63}

65fn render(program: &Program, env: &Env<'_>) -> Result<String, AwkError> {
66    if program.print.is_empty() {
67        return Ok(String::from(env.line));
68    }
69    let mut out = String::new();
70    for (index, expr) in program.print.iter().enumerate() {
71        if index > 0 {
72            out.push(' ');
73        }
74        out.push_str(&eval(expr, env)?.text());
75    }
76    Ok(out)
77}

79struct Env<'a> {
80    line: &'a str,
81    fields: &'a [&'a str],
82    nr: usize,
83}

85#[derive(Clone)]
86enum Value {
87    Num(i64),
88    Text(String),
89    Bool(bool),
90}
91
92impl Value {
93    fn truthy(&self) -> bool {
94        match self {
95            Self::Num(value) => *value != 0,
96            Self::Text(value) => !value.is_empty(),
97            Self::Bool(value) => *value,
98        }
99    }
100
101    fn text(&self) -> String {
102        match self {
103            Self::Num(value) => alloc::format!("{value}"),
104            Self::Text(value) => value.clone(),
105            Self::Bool(value) => alloc::format!("{}", if *value { 1 } else { 0 }),
106        }
107    }
108
109    fn num(&self) -> Result<i64, AwkError> {
110        match self {
111            Self::Num(value) => Ok(*value),
112            Self::Bool(value) => Ok(if *value { 1 } else { 0 }),
113            Self::Text(value) => value
114                .parse::<i64>()
115                .map_err(|_| AwkError::Runtime("numeric operation requires integers")),
116        }
117    }
118}

120fn eval(expr: &Expr, env: &Env<'_>) -> Result<Value, AwkError> {
121    match expr {
122        Expr::Number(value) => Ok(Value::Num(*value)),
123        Expr::Text(value) => Ok(Value::Text(value.clone())),
124        Expr::Field(0) => Ok(Value::Text(String::from(env.line))),
125        Expr::Field(field) => Ok(Value::Text(String::from(
126            env.fields.get(*field as usize - 1).copied().unwrap_or(""),
127        ))),
128        Expr::Builtin(Builtin::Nr) => Ok(Value::Num(env.nr as i64)),
129        Expr::Builtin(Builtin::Nf) => Ok(Value::Num(env.fields.len() as i64)),
130        Expr::Unary { op, expr } => {
131            let value = eval(expr, env)?;
132            match op {
133                UnaryOp::Neg => Ok(Value::Num(-value.num()?)),
134                UnaryOp::Not => Ok(Value::Bool(!value.truthy())),
135            }
136        }
137        Expr::Binary { op, lhs, rhs } => eval_binary(*op, &eval(lhs, env)?, rhs, env),
138    }
139}

141fn eval_binary(op: BinaryOp, lhs: &Value, rhs: &Expr, env: &Env<'_>) -> Result<Value, AwkError> {
142    match op {
143        BinaryOp::And => Ok(Value::Bool(lhs.truthy() && eval(rhs, env)?.truthy())),
144        BinaryOp::Or => Ok(Value::Bool(lhs.truthy() || eval(rhs, env)?.truthy())),
145        BinaryOp::Add => Ok(Value::Num(lhs.num()? + eval(rhs, env)?.num()?)),
146        BinaryOp::Sub => Ok(Value::Num(lhs.num()? - eval(rhs, env)?.num()?)),
147        BinaryOp::Mul => Ok(Value::Num(lhs.num()? * eval(rhs, env)?.num()?)),
148        BinaryOp::Div => {
149            let rhs = eval(rhs, env)?.num()?;
150            if rhs == 0 {
151                return Err(AwkError::Runtime("division by zero"));
152            }
153            Ok(Value::Num(lhs.num()? / rhs))
154        }
155        BinaryOp::Rem => {
156            let rhs = eval(rhs, env)?.num()?;
157            if rhs == 0 {
158                return Err(AwkError::Runtime("division by zero"));
159            }
160            Ok(Value::Num(lhs.num()? % rhs))
161        }
162        op => {
163            let rhs = eval(rhs, env)?;
164            Ok(Value::Bool(compare(op, lhs, &rhs)?))
165        }
166    }
167}

169fn compare(op: BinaryOp, lhs: &Value, rhs: &Value) -> Result<bool, AwkError> {
170    let numeric = lhs.num().ok().zip(rhs.num().ok());
171    if let Some((left, right)) = numeric {
172        return Ok(match op {
173            BinaryOp::Eq => left == right,
174            BinaryOp::Ne => left != right,
175            BinaryOp::Lt => left < right,
176            BinaryOp::Le => left <= right,
177            BinaryOp::Gt => left > right,
178            BinaryOp::Ge => left >= right,
179            _ => return Err(AwkError::Runtime("invalid comparison")),
180        });
181    }
182    let left = lhs.text();
183    let right = rhs.text();
184    Ok(match op {
185        BinaryOp::Eq => left == right,
186        BinaryOp::Ne => left != right,
187        BinaryOp::Lt => left < right,
188        BinaryOp::Le => left <= right,
189        BinaryOp::Gt => left > right,
190        BinaryOp::Ge => left >= right,
191        _ => return Err(AwkError::Runtime("invalid comparison")),
192    })
