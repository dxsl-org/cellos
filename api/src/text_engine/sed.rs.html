11#[derive(Debug)]
12pub enum SedError {
13    Parse(&'static str),
14    Record(RecordError),
15    OutputTooLong,
16}
17
18impl SedError {
19    pub fn message(&self) -> &'static str {
20        match self {
21            Self::Parse(message) => message,
22            Self::Record(RecordError::InputTooLarge) => "input exceeds configured limit",
23            Self::Record(RecordError::RecordTooLong) => "record exceeds configured limit",
24            Self::Record(RecordError::TooManyRecords) => "record count exceeds configured limit",
25            Self::OutputTooLong => "output record exceeds configured limit",
26        }
27    }
28}
29
30struct LineOutcome {
31    line: String,
32    deleted: bool,
33    explicit_print: bool,
34}
35
36pub fn execute(script: &str, suppress_default: bool, input: &str) -> Result<Vec<String>, SedError> {
37    let script = parser::parse(script)?;
38    let records = RecordReader::new(input)
39        .map_err(SedError::Record)?
40        .collect()
41        .map_err(SedError::Record)?;
42    let mut output = Vec::new();
43    for (index, record) in records.iter().enumerate() {
44        let outcome = apply_record(&script, record, index + 1)?;
45        if outcome.deleted {
46            continue;
47        }
48        if outcome.explicit_print {
49            output.push(outcome.line.clone());
50        }
51        if !suppress_default {
52            output.push(outcome.line);
53        }
54    }
55    Ok(output)
56}
57
58fn apply_record(
59    script: &SedScript,
60    line: &str,
61    line_number: usize,
62) -> Result<LineOutcome, SedError> {
63    match &script.command {
64        Command::Substitute {
65            pattern,
66            replacement,
67            global,
68            print,
69        } => {
70            let (line, changed) = pattern.replace(line, replacement, *global)?;
71            Ok(LineOutcome {
72                line,
73                deleted: false,
74                explicit_print: *print && changed,
75            })
76        }
77        Command::Delete(address) => Ok(LineOutcome {
78            line: String::from(line),
79            deleted: address.matches(line, line_number),
80            explicit_print: false,
81        }),
82        Command::Print(address) => Ok(LineOutcome {
83            line: String::from(line),
84            deleted: false,
85            explicit_print: address.matches(line, line_number),
86        }),
87    }
88}

90pub(super) fn validate_output_len(line: &str) -> Result<(), SedError> {
91    if line.len() > MAX_RECORD_BYTES {
92        Err(SedError::OutputTooLong)
93    } else {
94        Ok(())
95    }
96}

98#[cfg(test)]
99mod tests {
100    use super::execute;
101    use alloc::vec::Vec;
102
103    fn run(script: &str, suppress: bool, input: &str) -> Vec<String> {
104        execute(script, suppress, input).expect("script should succeed")
105    }
106
107    #[test]
108    fn substitutes_and_expands_match() {
109        let output = run(r"s|foo([0-9]+)|<&>|gp", false, "foo12\nbar");
110        assert_eq!(output, ["<foo12>", "<foo12>", "bar"]);
111    }
112
113    #[test]
114    fn supports_address_print_and_delete() {
115        let printed = run(r"/^ERR/p", true, "OK\nERR one\nERR two");
116        assert_eq!(printed, ["ERR one", "ERR two"]);
117        let deleted = run(r"/^ERR/d", false, "OK\nERR one\nWARN");
118        assert_eq!(deleted, ["OK", "WARN"]);
119    }
120
121    #[test]
122    fn supports_numeric_print() {
123        let output = run("2p", true, "one\ntwo\nthree");
124        assert_eq!(output, ["two"]);
125    }
