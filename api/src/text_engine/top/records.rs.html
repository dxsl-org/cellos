29const CPU_UNIT: &str = "core_tenths";

31fn uint(value: Option<u64>) -> Value {
32    value.map_or(Value::Null, Value::UInt)
33}
34
35fn text(value: &str) -> Value {
36    Value::String(value.into())
37}
38
39fn cpu_value(cpu: &Cpu, field: Field) -> Value {
40    match field {
41        Field::CpuNs => uint(cpu.ns),
42        Field::CpuTenths => uint(cpu.tenths.map(u64::from)),
43        _ => text(cpu.quality.label()),
44    }
45}
46
47/// The record value of `field`; moves the name out of `row`.
48fn cell_value(row: &mut CellSample, field: Field) -> Value {
49    match field {
50        Field::CellId => Value::UInt(row.cell_id),
51        Field::Generation => Value::UInt(row.generation),
52        Field::RootTid => Value::UInt(row.root_tid),
53        Field::Name => Value::String(core::mem::take(&mut row.name)),
54        Field::NameTruncated => Value::Bool(row.name_truncated),
55        Field::State => text(row.state.label()),
56        Field::Tasks => Value::UInt(u64::from(row.tasks)),
57        Field::CpuNs | Field::CpuTenths | Field::CpuQuality => cpu_value(&row.cpu, field),
58        Field::QuotaBytes => uint(row.quota_bytes),
59        Field::LimitBytes => uint(row.limit_bytes),
60        Field::Tid | Field::StackBytes | Field::SegmentBytes | Field::StackInQuota => Value::Null,
61    }
62}
63
64/// The record value of `field`; moves the name out of `row`.
65fn task_value(row: &mut TaskSample, field: Field) -> Value {
66    match field {
67        Field::Tid => Value::UInt(row.tid),
68        Field::CellId => uint(row.cell_id),
69        Field::Generation => uint(row.generation),
70        Field::Name => Value::String(core::mem::take(&mut row.name)),
71        Field::NameTruncated => Value::Bool(row.name_truncated),
72        Field::State => text(row.state.label()),
73        Field::CpuNs | Field::CpuTenths | Field::CpuQuality => cpu_value(&row.cpu, field),
74        Field::StackBytes => Value::UInt(row.stack_bytes),
75        Field::SegmentBytes => Value::UInt(row.segment_bytes),
76        Field::StackInQuota => Value::Bool(row.stack_in_quota),
77        Field::RootTid | Field::Tasks | Field::QuotaBytes | Field::LimitBytes => Value::Null,
78    }
79}

81/// One `top.cell.v1` record.
82pub fn cell_record(mut row: CellSample) -> Value {
83    Value::Record(RowKind::Cells.fields().iter().map(|&field| (field.name().into(), cell_value(&mut row, field))).collect())
84}
85
86/// One `top.task.v1` record.
87pub fn task_record(mut row: TaskSample) -> Value {
88    Value::Record(RowKind::Tasks.fields().iter().map(|&field| (field.name().into(), task_value(&mut row, field))).collect())
89}

91/// The Begin `meta` record of a frame of `kind` rows.
92pub fn meta_record(meta: &FrameMeta, kind: RowKind) -> Value {
93    let (mem_state, total, used, free) = match meta.mem {
94        SysMem::Present { total_bytes, used_bytes, free_bytes } => {
95            ("present", Some(total_bytes), Some(used_bytes), Some(free_bytes))
96        }
97        SysMem::Denied => ("denied", None, None, None),
98        SysMem::Absent => ("absent", None, None, None),
99    };
100    let values = [
101        text(kind.schema()),
102        // DV-9: always above 2^53, so a String keeps untyped JSON exact.
103        Value::String(format!("{:016x}", meta.provider_epoch)),
104        uint(meta.interval_ns),
105        Value::UInt(u64::from(meta.online_cpus)),
106        text(CPU_UNIT),
107        Value::Bool(meta.cpu_available),
108        Value::UInt(u64::from(meta.total_cells)),
109        Value::UInt(u64::from(meta.total_tasks)),
110        uint(meta.gone_cells),
111        uint(meta.gone_tasks),
112        text(mem_state),
113        uint(total),
114        uint(used),
115        uint(free),
116        Value::Bool(meta.memory_interval),
117    ];
118    Value::Record(META_FIELDS.iter().zip(values).map(|(&name, value)| (String::from(name), value)).collect())
119}

121enum RowIter {
122    Cells(vec::IntoIter<CellSample>),
123    Tasks(vec::IntoIter<TaskSample>),
124}
125
126impl RowIter {
127    fn new(rows: Rows) -> Self {
128        match rows {
129            Rows::Cells(rows) => RowIter::Cells(rows.into_iter()),
130            Rows::Tasks(rows) => RowIter::Tasks(rows.into_iter()),
131        }
132    }
133
134    fn next_value(&mut self) -> Option<Value> {
135        match self {
136            RowIter::Cells(rows) => rows.next().map(cell_record),
137            RowIter::Tasks(rows) => rows.next().map(task_record),
138        }
139    }
140}

142/// Begin, one record per pull, End.
143pub struct FrameItems {
144    begin: Option<FrameBegin>,
145    rows: RowIter,
146    end: Option<FrameEnd>,
147}
148
149impl Iterator for FrameItems {
150    type Item = Item;
151
152    fn next(&mut self) -> Option<Item> {
153        if let Some(begin) = self.begin.take() {
154            return Some(Item::Begin(begin));
155        }
156        if let Some(row) = self.rows.next_value() {
157            return Some(Item::Row(row));
158        }
159        self.end.take().map(Item::End)
160    }
161}
162
163/// The Frames items of `frame`: rows are converted lazily, one per pull.
164pub fn frame_items(frame: Frame) -> FrameItems {
165    let kind = frame.row_kind();
166    let begin = FrameBegin {
167        sample_id: frame.meta.sample_id,
168        monotonic_ns: frame.meta.capture_start_ns,
169        meta: Some(meta_record(&frame.meta, kind)),
170    };
171    let end = FrameEnd {
172        complete: frame.complete,
173        unavailable_fields: frame.unavailable_fields,
174        dropped: frame.dropped,
175    };
176    FrameItems { begin: Some(begin), rows: RowIter::new(frame.rows), end: Some(end) }
177}

179/// `top.task.v1` records of a task frame, one per pull (`ps --records`); a Cell
180/// frame yields nothing.
181pub struct TaskRows {
182    rows: vec::IntoIter<TaskSample>,
183}
184
185impl Iterator for TaskRows {
186    type Item = Value;
187
188    fn next(&mut self) -> Option<Value> {
189        self.rows.next().map(task_record)
190    }
191}
192
193pub fn task_rows(frame: Frame) -> TaskRows {
194    let rows = match frame.rows {
195        Rows::Tasks(rows) => rows,
196        Rows::Cells(_) => Vec::new(),
197    };
198    TaskRows { rows: rows.into_iter() }
