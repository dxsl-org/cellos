1//! Pure `top`/`ps` core over the Cell-aware telemetry provider (Phase 06 S6-2).
2//!
3//! The shell captures [`api::telemetry::TelemetrySnapshot`]s; this module turns
4//! them into bounded, typed frames:
5//!
6//! - [`parse_options`]: the D64-2 command line ([`USAGE`]);
7//! - [`sample`]: the [`sample::Sampler`] baseline, CPU quality and per-frame rows;
8//! - [`records`]: `top.cell.v1`/`top.task.v1` records and Begin/Row/End items;
9//! - [`text`]: the one text renderer every text mode shares.
10//!
11//! CPU is core-equivalent tenths of a percent (one busy core is 1000) and is
12//! never clamped. A value the provider could not measure is Null, never 0.
13

25/// Rows per frame of one kind (DV-11); rows beyond it make the frame incomplete.
26pub const MAX_FRAME_ROWS: usize = pipeline::MAX_SORT_RECORDS;
27/// Client cap on one snapshot buffer: covers 64 Cells + 2,048 tasks (186,464 B).
28pub const MAX_SNAPSHOT_BYTES: usize = 192 * 1024;
29/// Delay between frames when `-d` is absent.
30pub const DEFAULT_DELAY_NS: u64 = 1_000_000_000;

32pub const USAGE: &str =
33    "Usage: top [-a] [-b] [-n COUNT] [-d SECS] [-o FIELD] [--fields LIST] [--tasks | --cell ID:GEN]";
34/// `-o mem`/`-o heap` name a Cell's quota; task rows carry per-task bytes instead.
35pub const USAGE_TASK_MEMORY: &str =
36    "top: mem/heap sort Cells by quota; task rows sort by stack_bytes or segment_bytes";

43/// Which rows a frame holds (D64-1 R1: one row kind per frame).
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub enum View {
46    /// One row per live or retiring Cell incarnation (the default).
47    Cells,
48    /// Every task, ownerless kernel tasks included.
49    Tasks,
50    /// The tasks of exactly one Cell incarnation.
51    Cell { cell_id: u64, generation: u64 },
52}
53
54/// The row schema of a [`View`].
55#[derive(Clone, Copy, Debug, PartialEq, Eq)]
56pub enum RowKind {
57    Cells,
58    Tasks,
59}
60
61impl View {
62    pub fn row_kind(self) -> RowKind {
63        match self {
64            View::Cells => RowKind::Cells,
65            View::Tasks | View::Cell { .. } => RowKind::Tasks,
66        }
67    }
68}

141impl Field {
142    /// The record field name.
143    pub fn name(self) -> &'static str {
144        match self {
145            Field::CellId => "cell_id",
146            Field::Generation => "generation",
147            Field::RootTid => "root_tid",
148            Field::Tid => "tid",
149            Field::Name => "name",
150            Field::NameTruncated => "name_truncated",
151            Field::State => "state",
152            Field::Tasks => "tasks",
153            Field::CpuNs => "cpu_ns",
154            Field::CpuTenths => "cpu_tenths",
155            Field::CpuQuality => "cpu_quality",
156            Field::QuotaBytes => "quota_bytes",
157            Field::LimitBytes => "limit_bytes",
158            Field::StackBytes => "stack_bytes",
159            Field::SegmentBytes => "segment_bytes",
160            Field::StackInQuota => "stack_in_quota",
161        }
162    }
163
164    /// True when `kind` records carry this field.
165    pub fn applies_to(self, kind: RowKind) -> bool {
166        kind.fields().contains(&self)
167    }
168
169    /// Numeric resource fields sort descending; identity and text fields ascending.
170    pub fn sorts_descending(self) -> bool {
171        matches!(
172            self,
173            Field::Tasks
174                | Field::CpuNs
175                | Field::CpuTenths
176                | Field::QuotaBytes
177                | Field::LimitBytes
178                | Field::StackBytes
179                | Field::SegmentBytes
180        )
181    }
182
183    /// Resolve a field name or alias for `kind` rows.
184    ///
185    /// Aliases: `cpu` → `cpu_tenths`; `pid` → `cell_id` (Cells) or `tid` (tasks);
186    /// `mem`/`heap` → `quota_bytes` (Cells only).
187    pub fn resolve(token: &str, kind: RowKind) -> Result<Field, &'static str> {
188        let field = match (token, kind) {
189            ("cpu", _) => Field::CpuTenths,
190            ("pid", RowKind::Cells) => Field::CellId,
191            ("pid", RowKind::Tasks) => Field::Tid,
192            ("mem" | "heap", RowKind::Cells) => Field::QuotaBytes,
193            ("mem" | "heap", RowKind::Tasks) => return Err(USAGE_TASK_MEMORY),
194            _ => *kind.fields().iter().find(|field| field.name() == token).ok_or(USAGE)?,
195        };
196        Ok(field)
197    }
198}

200/// Row order for one frame.
201#[derive(Clone, Copy, Debug, PartialEq, Eq)]
202pub struct SortSpec {
203    pub field: Field,
204    /// Larger values first. Null values sort last in either direction, and ties
205    /// break on identity ascending (`cell_id, generation` or `tid`).
206    pub descending: bool,
207}
208
209impl SortSpec {
210    /// Sort by `field` in its natural direction ([`Field::sorts_descending`]).
211    pub fn new(field: Field) -> SortSpec {
212        SortSpec { field, descending: field.sorts_descending() }
213    }
214
215    /// Busiest rows first; both row kinds share the default.
216    pub fn default_for(_view: View) -> SortSpec {
217        SortSpec::new(Field::CpuTenths)
218    }
219}

221#[derive(Clone, Debug, PartialEq, Eq)]
222pub struct TopOptions {
223    pub batch: bool,
224    pub count: Option<usize>,
225    pub delay_ns: u64,
226    pub show_all: bool,
227    pub view: View,
228    pub sort: SortSpec,
229    /// Columns named by `--fields`, in the given order (fields of the view's row kind).
230    pub fields: Option<Vec<Field>>,
231}

233/// Parse `top` flags (D64-2).
234///
235/// # Errors
236/// Returns usage text (exit [`USAGE_STATUS`]) for an unknown, repeated or
237/// conflicting flag, or a missing or invalid flag value.
238pub fn parse_options(args: &[&str]) -> Result<TopOptions, &'static str> {
239    let mut batch = None;
240    let mut show_all = None;
241    let mut count = None;
242    let mut delay_ns = None;
243    let mut view = None;
244    let mut sort = None;
245    let mut fields = None;
246    let mut words = args.iter().copied();
247    while let Some(arg) = words.next() {
248        match arg {
249            "-a" => set_once(&mut show_all, Some(()))?,
250            "-b" => set_once(&mut batch, Some(()))?,
251            "-n" => set_once(
252                &mut count,
253                words.next().and_then(parse_decimal).and_then(|n| usize::try_from(n).ok()),
254            )?,
255            "-d" => set_once(&mut delay_ns, words.next().and_then(parse_delay_ns))?,
256            "-o" => set_once(&mut sort, words.next())?,
257            "--fields" => set_once(&mut fields, words.next())?,
258            "--tasks" => set_once(&mut view, Some(View::Tasks))?,
259            "--cell" => set_once(&mut view, words.next().and_then(parse_cell))?,
260            _ => return Err(USAGE),
261        }
262    }
263    let view = view.unwrap_or(View::Cells);
264    let kind = view.row_kind();
265    let sort = match sort {
266        Some(token) => SortSpec::new(Field::resolve(token, kind)?),
267        None => SortSpec::default_for(view),
268    };
269    let fields = fields.map(|list| parse_field_list(list, kind)).transpose()?;
270    Ok(TopOptions {
271        batch: batch.is_some(),
272        count,
273        delay_ns: delay_ns.unwrap_or(DEFAULT_DELAY_NS),
274        show_all: show_all.is_some(),
275        view,
276        sort,
277        fields,
278    })
279}

281/// `TelemetrySnapshot` request flags for `view`: the Cells view requests no task rows.
282pub fn request_flags(view: View, sysmem: bool) -> usize {
283    let tasks = match view.row_kind() {
284        RowKind::Cells => 0,
285        RowKind::Tasks => request::TASKS,
286    };
287    tasks | if sysmem { request::SYSTEM_MEMORY } else { 0 }
288}

290fn set_once<T>(slot: &mut Option<T>, value: Option<T>) -> Result<(), &'static str> {
291    if slot.is_some() {
292        return Err(USAGE);
293    }
294    *slot = Some(value.ok_or(USAGE)?);
295    Ok(())
296}

298fn parse_decimal(text: &str) -> Option<u64> {
299    if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
300        return None;
301    }
302    text.parse().ok()
303}
304
305/// Decimal seconds with up to nine fraction digits, in ns; `None` on overflow.
306fn parse_delay_ns(text: &str) -> Option<u64> {
307    let (whole, fraction) = match text.split_once('.') {
308        Some((whole, fraction)) => (whole, Some(fraction)),
309        None => (text, None),
310    };
311    let mut ns = parse_decimal(whole)?.checked_mul(NS_PER_SEC)?;
312    if let Some(fraction) = fraction {
313        if fraction.len() > MAX_DELAY_FRACTION_DIGITS {
314            return None;
315        }
316        let mut scaled = parse_decimal(fraction)?;
317        for _ in fraction.len()..MAX_DELAY_FRACTION_DIGITS {
318            scaled *= 10;
319        }
320        ns = ns.checked_add(scaled)?;
321    }
322    Some(ns)
323}

325/// `ID:GEN`; generation 0 is never a Cell (it marks ownerless kernel tasks).
326fn parse_cell(text: &str) -> Option<View> {
327    let (cell_id, generation) = text.split_once(':')?;
328    let generation = parse_decimal(generation).filter(|&generation| generation != 0)?;
329    Some(View::Cell { cell_id: parse_decimal(cell_id)?, generation })
330}

332fn parse_field_list(list: &str, kind: RowKind) -> Result<Vec<Field>, &'static str> {
333    let mut fields = Vec::new();
334    for token in list.split(',') {
335        let field = Field::resolve(token, kind)?;
336        if fields.contains(&field) {
337            return Err(USAGE);
338        }
339        fields.push(field);
340    }
341    Ok(fields)
