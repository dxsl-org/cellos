1//! 82574 PHY access through `MDIC` and copper autonegotiation.
2//!
3//! `MDIC` layout (82574 datasheet §10.2.2.9): data 15:0, PHY register 20:16,
4//! PHY address 25:21, opcode 27:26 (`01` write, `10` read), ready 28,
5//! interrupt-enable 29, error 30. The integrated PHY answers at address 1.
6//! MII register numbers and bits are IEEE 802.3 clause 22.
7//!
8//! After a global reset the PHY restarts from its power-on defaults and
9//! `STATUS.LU` follows the PHY's autonegotiation, so the cell restarts autoneg
10//! explicitly and then waits for link (bounded, see `controller.rs`).
11
12use crate::identity::MacKind;
13use crate::regs::{RegisterFile, MDIC};
14use types::{ViError, ViResult};
15
16pub(crate) const MDIC_DATA_MASK: u32 = 0x0000_FFFF;
17pub(crate) const MDIC_REG_SHIFT: u32 = 16;
18pub(crate) const MDIC_PHY_SHIFT: u32 = 21;
19pub(crate) const MDIC_OP_WRITE: u32 = 0b01 << 26;
20pub(crate) const MDIC_OP_READ: u32 = 0b10 << 26;
21pub(crate) const MDIC_READY: u32 = 1 << 28;
22pub(crate) const MDIC_ERROR: u32 = 1 << 30;
23/// The 82574's integrated PHY address.
24pub(crate) const PHY_ADDR: u32 = 1;
25
26/// MII Basic Mode Control Register.
27pub(crate) const MII_BMCR: u32 = 0;
28pub(crate) const BMCR_SPEED1000: u16 = 1 << 6;
29pub(crate) const BMCR_FULL_DUPLEX: u16 = 1 << 8;
30pub(crate) const BMCR_ANRESTART: u16 = 1 << 9;
31pub(crate) const BMCR_ANENABLE: u16 = 1 << 12;
32
33/// `MDIC` ready poll budget.
34const MDIC_POLLS: u32 = 100_000;
35
36/// The `MDIC` command word for one clause-22 access to the integrated PHY.
37pub(crate) const fn mdic_command(write: bool, reg: u32, data: u16) -> u32 {
38    let op = if write { MDIC_OP_WRITE } else { MDIC_OP_READ };
39    (data as u32 & MDIC_DATA_MASK) | (reg << MDIC_REG_SHIFT) | (PHY_ADDR << MDIC_PHY_SHIFT) | op
40}
41
42/// Issue one `MDIC` access and wait (bounded) for `READY`; `ERROR` fails.
43fn mdic_access(regs: &impl RegisterFile, command: u32) -> ViResult<u16> {
44    regs.wr32(MDIC, command)?;
45    for _ in 0..MDIC_POLLS {
46        let value = regs.rd32(MDIC)?;
47        if value & MDIC_READY != 0 {
48            if value & MDIC_ERROR != 0 {
49                return Err(ViError::IO);
50            }
51            return Ok((value & MDIC_DATA_MASK) as u16);
52        }
53        core::hint::spin_loop();
54    }
55    Err(ViError::IO)
56}
57
58pub(crate) fn phy_read(regs: &impl RegisterFile, reg: u32) -> ViResult<u16> {
59    mdic_access(regs, mdic_command(false, reg, 0))
60}
61
62pub(crate) fn phy_write(regs: &impl RegisterFile, reg: u32, data: u16) -> ViResult<()> {
63    mdic_access(regs, mdic_command(true, reg, data)).map(|_| ())
64}
65
66/// Advertise autonegotiation and restart it. The link then comes up when the
67/// PHY completes autoneg; the caller waits on `STATUS.LU`.
68pub(crate) fn restart_autoneg(regs: &impl RegisterFile, kind: MacKind) -> ViResult<()> {
69    match kind {
70        MacKind::I82574 => {
71            let bmcr = phy_read(regs, MII_BMCR)?;
72            phy_write(
73                regs,
74                MII_BMCR,
75                bmcr | BMCR_ANENABLE | BMCR_ANRESTART | BMCR_SPEED1000 | BMCR_FULL_DUPLEX,
76            )
77        }
78        MacKind::PchLan(_) => crate::pch::with_swflag(regs, || {
79            let bmcr = phy_read(regs, MII_BMCR)?;
80            phy_write(
81                regs,
82                MII_BMCR,
83                bmcr | BMCR_ANENABLE | BMCR_ANRESTART | BMCR_SPEED1000 | BMCR_FULL_DUPLEX,
84            )
85        }),
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92    use crate::fake::{FakeRegs, BMSR_ANEG_COMPLETE, BMSR_LINK, MII_BMSR};
93
94    /// Link and completed autoneg as the PHY reports them (`BMSR` link status
95    /// is latched-low, so the second read is the current state).
96    fn phy_link_up(regs: &FakeRegs) -> ViResult<bool> {
97        let _ = phy_read(regs, MII_BMSR)?;
98        let bmsr = phy_read(regs, MII_BMSR)?;
99        Ok(bmsr & (BMSR_LINK | BMSR_ANEG_COMPLETE) == (BMSR_LINK | BMSR_ANEG_COMPLETE))
100    }
101
102    #[test]
103    fn mdic_command_encodes_the_82574_fields() {
104        let write = mdic_command(true, MII_BMCR, 0x1340);
105        assert_eq!(write & MDIC_DATA_MASK, 0x1340);
106        assert_eq!((write >> MDIC_REG_SHIFT) & 0x1F, 0);
107        assert_eq!((write >> MDIC_PHY_SHIFT) & 0x1F, 1);
108        assert_eq!((write >> 26) & 0b11, 0b01);
109        let read = mdic_command(false, MII_BMSR, 0xFFFF);
110        assert_eq!(read & MDIC_DATA_MASK, 0xFFFF);
111        assert_eq!((read >> MDIC_REG_SHIFT) & 0x1F, 1);
112        assert_eq!((read >> 26) & 0b11, 0b10);
113        assert_eq!(read & (MDIC_READY | MDIC_ERROR), 0);
114    }
115
116    #[test]
117    fn autoneg_restart_sets_enable_and_restart_and_link_follows() {
118        let regs = FakeRegs::new();
119        regs.phy.borrow_mut()[MII_BMCR as usize] = BMCR_FULL_DUPLEX;
120        assert_eq!(phy_link_up(&regs), Ok(false));
121        restart_autoneg(&regs, MacKind::I82574).expect("MDIC write completes");
122        // The written BMCR kept the old bits and asked for autoneg.
123        let written = regs
124            .writes
125            .borrow()
126            .iter()
127            .rev()
128            .find(|(offset, value)| *offset == MDIC && value & MDIC_OP_WRITE != 0)
129            .map(|&(_, value)| (value & MDIC_DATA_MASK) as u16)
130            .expect("a BMCR write");
131        assert_eq!(written & BMCR_ANENABLE, BMCR_ANENABLE);
132        assert_eq!(written & BMCR_ANRESTART, BMCR_ANRESTART);
133        assert_eq!(written & BMCR_FULL_DUPLEX, BMCR_FULL_DUPLEX);
134        // The fake PHY completes autoneg on restart, as a real link partner would.
135        assert_eq!(phy_link_up(&regs), Ok(true));
136    }

138    #[test]
139    fn mdic_error_and_wedged_phy_fail_closed() {
140        let mut regs = FakeRegs::new();
141        regs.phy_address = 2;
142        assert_eq!(phy_read(&regs, MII_BMSR), Err(ViError::IO));
143        regs.phy_address = PHY_ADDR;
144        regs.responsive = false;
145        assert_eq!(phy_write(&regs, MII_BMCR, 0), Err(ViError::IO));
146    }

148    #[test]
149    fn pch_lan_autoneg_restart_acquires_and_releases_semaphore() {
150        let regs = FakeRegs::new();
151        regs.phy.borrow_mut()[MII_BMCR as usize] = BMCR_FULL_DUPLEX;
152        restart_autoneg(
153            &regs,
154            MacKind::PchLan(crate::identity::PchGen::SunrisePoint),
155        )
156        .expect("PCH LAN autoneg succeeds");
157        let writes = regs.writes.borrow();
158        assert!(
159            writes
160                .iter()
161                .any(|&(o, v)| o == crate::pch::EXTCNF_CTRL
162                    && v & crate::pch::EXTCNF_CTRL_SWFLAG != 0)
163        );
164        assert_eq!(
165            regs.get(crate::pch::EXTCNF_CTRL) & crate::pch::EXTCNF_CTRL_SWFLAG,
166            0
167        );
