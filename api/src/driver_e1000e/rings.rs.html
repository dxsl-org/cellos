1//! Legacy TX/RX descriptor rings in `GrantDma`-authorized memory.
2//!
3//! The 82574 keeps the 8254x **legacy** descriptor formats (datasheet §7.1.4
4//! receive, §7.2.10 transmit), selected by `RFCTL.EXSTEN = 0` and
5//! `RCTL.DTYP = 00` (both reset defaults). Each descriptor is 16 bytes; one
6//! 4 KiB ring page holds every slot this cell uses. Volatile access to the
7//! rings goes through `nic_driver::ring::DescRing`, which bounds every slot.
8
9use core::mem::size_of;
10use core::sync::atomic::{fence, Ordering};
11use nic_driver::ring::DescRing;
12
13/// TX descriptors (and TX buffers) in use.
14pub(crate) const TX_SLOTS: usize = 16;
15/// RX descriptors (and RX buffers) in use.
16pub(crate) const RX_SLOTS: usize = 16;
17
18pub(crate) type TxRing = DescRing<TxDesc, TX_SLOTS>;
19pub(crate) type RxRing = DescRing<RxDesc, RX_SLOTS>;
20
21/// Legacy transmit descriptor.
22#[repr(C)]
23#[derive(Clone, Copy)]
24pub(crate) struct TxDesc {
25    pub buf_addr: u64,
26    pub length: u16,
27    pub cso: u8,
28    pub cmd: u8,
29    pub status: u8,
30    pub css: u8,
31    pub special: u16,
32}
33
34/// Legacy receive descriptor; the controller writes everything after `buf_addr`.
35#[repr(C)]
36#[derive(Clone, Copy)]
37pub(crate) struct RxDesc {
38    pub buf_addr: u64,
39    pub length: u16,
40    pub checksum: u16,
41    pub status: u8,
42    pub errors: u8,
43    pub special: u16,
44}
45
46const _: () = assert!(size_of::<TxDesc>() == 16);
47const _: () = assert!(size_of::<RxDesc>() == 16);
48
49pub(crate) const TXD_CMD_EOP: u8 = 1 << 0;
50pub(crate) const TXD_CMD_IFCS: u8 = 1 << 1;
51pub(crate) const TXD_CMD_RS: u8 = 1 << 3;
52/// Command for a single-buffer frame: end of packet, insert FCS, report status.
53pub(crate) const TXD_CMD_FRAME: u8 = TXD_CMD_EOP | TXD_CMD_IFCS | TXD_CMD_RS;
54pub(crate) const DESC_STATUS_DD: u8 = 1 << 0;
55/// RX status: last descriptor of a frame.
56pub(crate) const RXD_STAT_EOP: u8 = 1 << 1;
57
58/// Largest frame moved in one descriptor; `RCTL.BSIZE = 00` is 2048 bytes.
59pub const BUF_SIZE: usize = 2048;
60const _: () = assert!(BUF_SIZE <= 4096, "one frame fits one DMA page");
61
62/// Ring length in bytes for `slots` legacy descriptors (`TDLEN`/`RDLEN`).
63pub(crate) const fn ring_bytes(slots: usize) -> u32 {
64    (slots * size_of::<TxDesc>()) as u32
65}
66
67/// Hand RX slot `slot` to the controller with buffer `iova`.
68pub(crate) fn post_rx(ring: &RxRing, slot: usize, iova: u64) {
69    ring.write(
70        slot,
71        RxDesc {
72            buf_addr: iova,
73            length: 0,
74            checksum: 0,
75            status: 0,
76            errors: 0,
77            special: 0,
78        },
79    );
80}
81
82/// A written-back RX descriptor, as [`rx_completed`] classifies it.
83#[derive(Clone, Copy, Debug, Eq, PartialEq)]
84pub(crate) enum RxCompletion {
85    /// DD clear: the controller still owns the slot.
86    Pending,
87    /// A whole, error-free frame of this many bytes.
88    Frame(usize),
89    /// Hand the slot back without delivering it: an error was reported, or it
90    /// is part of a frame spread over several descriptors.
91    Drop,
92}
93
94/// RX slot `slot`'s completion. `in_fragment` is true while the descriptors
95/// of a multi-descriptor frame are being dropped; it is updated here.
96pub(crate) fn rx_completed(ring: &RxRing, slot: usize, in_fragment: &mut bool) -> RxCompletion {
97    let status = ring.read(slot).status;
98    if status & DESC_STATUS_DD == 0 {
99        return RxCompletion::Pending;
100    }
101    // The rest of the write-back and the frame bytes are valid only once DD
102    // is seen: no read of them may move above the DD read, so the descriptor
103    // is read again after the fence.
104    fence(Ordering::Acquire);
105    let desc = ring.read(slot);
106    rx_written_back(status, desc.errors, desc.length, in_fragment)
107}
108
109/// Classify a descriptor whose DD bit is set. A frame is delivered only when
110/// it fits one descriptor (EOP set, not the tail of a dropped frame) and the
111/// controller reported no error.
112pub(crate) fn rx_written_back(
113    status: u8,
114    errors: u8,
115    length: u16,
116    in_fragment: &mut bool,
117) -> RxCompletion {
118    let eop = status & RXD_STAT_EOP != 0;
119    let tail_of_dropped = *in_fragment;
120    *in_fragment = !eop;
121    if !eop || tail_of_dropped || errors != 0 {
122        RxCompletion::Drop
123    } else {
124        RxCompletion::Frame(length as usize)
125    }
126}

128/// Fill TX slot `slot` for one frame of `len` bytes at `iova`.
129pub(crate) fn post_tx(ring: &TxRing, slot: usize, iova: u64, len: u16) {
130    ring.write(
131        slot,
132        TxDesc {
133            buf_addr: iova,
134            length: len,
135            cso: 0,
136            cmd: TXD_CMD_FRAME,
137            status: 0,
138            css: 0,
139            special: 0,
140        },
141    );
142}
143
144/// Whether the controller wrote back DD for TX slot `slot`.
145pub(crate) fn tx_done(ring: &TxRing, slot: usize) -> bool {
146    tx_status(ring, slot).1 & DESC_STATUS_DD != 0
147}
148
149/// Whether TX slot `slot` of a `slots`-slot ring may be filled now: it and the
150/// slot after it are both free. One slot always stays empty, so the tail
151/// written after this post (`TDT = slot + 1`) never lands on a descriptor the
152/// controller still owns — `TDT == TDH` would read as an empty ring.
153pub(crate) fn tx_has_room(ring: &TxRing, slot: usize, slots: usize) -> bool {
154    tx_room(slot, slots, |s| tx_status(ring, s))
155}
156
157fn tx_room(slot: usize, slots: usize, status: impl Fn(usize) -> (u8, u8)) -> bool {
158    [slot, (slot + 1) % slots].into_iter().all(|s| {
159        let (cmd, st) = status(s);
160        tx_slot_reusable(cmd, st)
161    })
162}
163
164/// A slot is free when it was never posted since the ring was cleared, or the
165/// controller wrote it back (DD). A posted slot without DD still belongs to
166/// the controller.
167fn tx_slot_reusable(cmd: u8, status: u8) -> bool {
168    cmd == 0 || status & DESC_STATUS_DD != 0
169}
170
171/// `(cmd, status)` of TX slot `slot`.
172fn tx_status(ring: &TxRing, slot: usize) -> (u8, u8) {
173    let desc = ring.read(slot);
174    (desc.cmd, desc.status)
175}

184        assert_eq!(offset_of!(TxDesc, buf_addr), 0);
185        assert_eq!(offset_of!(TxDesc, length), 8);

205    #[test]
206    fn ring_fits_one_page_and_lengths_are_128_byte_multiples() {
207        let bytes = ring_bytes(TX_SLOTS);
208        assert!(bytes as usize <= 4096);
209        // TDLEN/RDLEN must be 128-byte aligned (datasheet §10.2.6/§10.2.7).
210        assert_eq!(bytes % 128, 0);
211        assert_eq!(ring_bytes(RX_SLOTS) % 128, 0);
212    }

214    #[test]
215    fn rx_delivers_only_whole_error_free_frames() {
216        let mut in_fragment = false;
217        let eop = DESC_STATUS_DD | RXD_STAT_EOP;
218        assert_eq!(
219            rx_written_back(eop, 0, 60, &mut in_fragment),
220            RxCompletion::Frame(60)
221        );
222        // Any reported error (CRC, symbol, sequence, RX data) drops the frame.
223        assert_eq!(
224            rx_written_back(eop, 0x01, 60, &mut in_fragment),
225            RxCompletion::Drop
226        );
227        assert_eq!(
228            rx_written_back(eop, 0x80, 60, &mut in_fragment),
229            RxCompletion::Drop
230        );
231        assert!(!in_fragment);
232        // A frame over several descriptors is dropped up to and including EOP,
233        // so its tail is never delivered as a frame of its own.
234        assert_eq!(
235            rx_written_back(DESC_STATUS_DD, 0, 2048, &mut in_fragment),
236            RxCompletion::Drop
237        );
238        assert!(in_fragment);
239        assert_eq!(
240            rx_written_back(DESC_STATUS_DD, 0, 2048, &mut in_fragment),
241            RxCompletion::Drop
242        );
243        assert_eq!(
244            rx_written_back(eop, 0, 100, &mut in_fragment),
245            RxCompletion::Drop
246        );
247        assert!(!in_fragment);
248        assert_eq!(
249            rx_written_back(eop, 0, 64, &mut in_fragment),
250            RxCompletion::Frame(64)
251        );
252    }

254    #[test]
255    fn tx_slot_is_reusable_only_when_never_posted_or_written_back() {
256        assert!(tx_slot_reusable(0, 0), "cleared ring");
257        assert!(
258            tx_slot_reusable(TXD_CMD_FRAME, DESC_STATUS_DD),
259            "written back"

261        assert!(
262            !tx_slot_reusable(TXD_CMD_FRAME, 0),

267    /// With nothing written back, a ring of N slots takes N - 1 frames: the
268    /// tail never wraps onto a descriptor the controller still owns. One
269    /// write-back at the slot after the tail frees room again.
270    #[test]
271    fn a_full_tx_ring_keeps_one_slot_empty() {
272        use core::cell::RefCell;
273        const SLOTS: usize = TX_SLOTS;
274        let ring = RefCell::new([(0u8, 0u8); SLOTS]);
275        let room = |slot| tx_room(slot, SLOTS, |s| ring.borrow()[s]);
276        let mut next = 0;
277        let mut posted = 0;
278        while room(next) {
279            ring.borrow_mut()[next] = (TXD_CMD_FRAME, 0);
280            next = (next + 1) % SLOTS;
281            posted += 1;
282            assert!(posted < SLOTS, "the ring filled every slot");
283        }
284        assert_eq!(posted, SLOTS - 1);
285        assert_eq!(next, SLOTS - 1, "the empty slot is the tail");
286        // The head (slot 0) completes: the tail may advance onto it.
287        ring.borrow_mut()[0].1 = DESC_STATUS_DD;
288        assert!(room(next));
289        // A tail slot itself still owned by the controller is refused too.
290        ring.borrow_mut()[next] = (TXD_CMD_FRAME, 0);
291        assert!(!room(next));
292    }
