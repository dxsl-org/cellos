1//! Legacy TX/RX descriptor rings in `GrantDma`-authorized memory.
2//!
3//! The 82574 keeps the 8254x **legacy** descriptor formats (datasheet §7.1.4
4//! receive, §7.2.10 transmit), selected by `RFCTL.EXSTEN = 0` and
5//! `RCTL.DTYP = 00` (both reset defaults). Each descriptor is 16 bytes; one
6//! 4 KiB ring page holds every slot this cell uses.
7//!
8//! Law 4 exception: the volatile accesses below touch device-visible DMA memory.
9//! Every ring and buffer is a whole `DmaBuf` page that the cell owns for its
10//! lifetime, and every index is bounded by the slot count before the access.
11
12use core::mem::size_of;
13use core::sync::atomic::{fence, Ordering};
14use ostd::dma::DmaBuf;
15
16/// Legacy transmit descriptor.
17#[repr(C)]
18pub(crate) struct TxDesc {
19    pub buf_addr: u64,
20    pub length: u16,
21    pub cso: u8,
22    pub cmd: u8,
23    pub status: u8,
24    pub css: u8,
25    pub special: u16,
26}
27
28/// Legacy receive descriptor; the controller writes everything after `buf_addr`.
29#[repr(C)]
30pub(crate) struct RxDesc {
31    pub buf_addr: u64,
32    pub length: u16,
33    pub checksum: u16,
34    pub status: u8,
35    pub errors: u8,
36    pub special: u16,
37}
38
39const _: () = assert!(size_of::<TxDesc>() == 16);
40const _: () = assert!(size_of::<RxDesc>() == 16);
41
42pub(crate) const TXD_CMD_EOP: u8 = 1 << 0;
43pub(crate) const TXD_CMD_IFCS: u8 = 1 << 1;
44pub(crate) const TXD_CMD_RS: u8 = 1 << 3;
45/// Command for a single-buffer frame: end of packet, insert FCS, report status.
46pub(crate) const TXD_CMD_FRAME: u8 = TXD_CMD_EOP | TXD_CMD_IFCS | TXD_CMD_RS;
47pub(crate) const DESC_STATUS_DD: u8 = 1 << 0;
48/// RX status: last descriptor of a frame.
49pub(crate) const RXD_STAT_EOP: u8 = 1 << 1;
50
51/// Largest frame moved in one descriptor; `RCTL.BSIZE = 00` is 2048 bytes.
52pub const BUF_SIZE: usize = 2048;
53const _: () = assert!(BUF_SIZE <= 4096, "one frame fits one DMA page");
54
55/// Ring length in bytes for `slots` legacy descriptors (`TDLEN`/`RDLEN`).
56pub(crate) const fn ring_bytes(slots: usize) -> u32 {
57    (slots * size_of::<TxDesc>()) as u32
58}
59
60/// Zero a ring page before its base is programmed.
61pub(crate) fn clear(ring: &DmaBuf) {
62    // SAFETY: `ring` is one owned, writable DMA page of `ring.size()` bytes.
63    unsafe { core::ptr::write_bytes(ring.virt(), 0, ring.size()) };
64}
65
66/// Hand RX slot `slot` (< ring capacity) to the controller with buffer `iova`.
67pub(crate) fn post_rx(ring: &DmaBuf, slot: usize, iova: u64) {
68    assert!((slot + 1) * size_of::<RxDesc>() <= ring.size());
69    // SAFETY: the slot lies inside the owned ring page (asserted above).
70    unsafe {
71        let desc = (ring.virt() as *mut RxDesc).add(slot);
72        core::ptr::write_volatile(&mut (*desc).buf_addr, iova);
73        core::ptr::write_volatile(&mut (*desc).length, 0);
74        core::ptr::write_volatile(&mut (*desc).checksum, 0);
75        core::ptr::write_volatile(&mut (*desc).status, 0);
76        core::ptr::write_volatile(&mut (*desc).errors, 0);
77        core::ptr::write_volatile(&mut (*desc).special, 0);
78    }
79}

81/// A written-back RX descriptor, as [`rx_completed`] classifies it.
82#[derive(Clone, Copy, Debug, Eq, PartialEq)]
83pub(crate) enum RxCompletion {
84    /// DD clear: the controller still owns the slot.
85    Pending,
86    /// A whole, error-free frame of this many bytes.
87    Frame(usize),
88    /// Hand the slot back without delivering it: an error was reported, or it
89    /// is part of a frame spread over several descriptors.
90    Drop,
91}
92
93/// RX slot `slot`'s completion. `in_fragment` is true while the descriptors
94/// of a multi-descriptor frame are being dropped; it is updated here.
95pub(crate) fn rx_completed(ring: &DmaBuf, slot: usize, in_fragment: &mut bool) -> RxCompletion {
96    assert!((slot + 1) * size_of::<RxDesc>() <= ring.size());
97    // SAFETY: the slot lies inside the owned ring page (asserted above).
98    let desc = unsafe { (ring.virt() as *const RxDesc).add(slot) };
99    // SAFETY: as above.
100    let status = unsafe { core::ptr::read_volatile(&(*desc).status) };
101    if status & DESC_STATUS_DD == 0 {
102        return RxCompletion::Pending;
103    }
104    // The rest of the write-back and the frame bytes are valid only once DD
105    // is seen: no read of them may move above the DD read.
106    fence(Ordering::Acquire);
107    // SAFETY: as above.
108    let (errors, length) = unsafe {
109        (
110            core::ptr::read_volatile(&(*desc).errors),
111            core::ptr::read_volatile(&(*desc).length),
112        )
113    };
114    rx_written_back(status, errors, length, in_fragment)
115}

117/// Classify a descriptor whose DD bit is set. A frame is delivered only when
118/// it fits one descriptor (EOP set, not the tail of a dropped frame) and the
119/// controller reported no error.
120pub(crate) fn rx_written_back(
121    status: u8,
122    errors: u8,
123    length: u16,
124    in_fragment: &mut bool,
125) -> RxCompletion {
126    let eop = status & RXD_STAT_EOP != 0;
127    let tail_of_dropped = *in_fragment;
128    *in_fragment = !eop;
129    if !eop || tail_of_dropped || errors != 0 {
130        RxCompletion::Drop
131    } else {
132        RxCompletion::Frame(length as usize)
133    }
134}

136/// Fill TX slot `slot` for one frame of `len` bytes at `iova`.
137pub(crate) fn post_tx(ring: &DmaBuf, slot: usize, iova: u64, len: u16) {
138    assert!((slot + 1) * size_of::<TxDesc>() <= ring.size());
139    // SAFETY: the slot lies inside the owned ring page (asserted above).
140    unsafe {
141        let desc = (ring.virt() as *mut TxDesc).add(slot);
142        core::ptr::write_volatile(&mut (*desc).buf_addr, iova);
143        core::ptr::write_volatile(&mut (*desc).length, len);
144        core::ptr::write_volatile(&mut (*desc).cso, 0);
145        core::ptr::write_volatile(&mut (*desc).cmd, TXD_CMD_FRAME);
146        core::ptr::write_volatile(&mut (*desc).status, 0);
147        core::ptr::write_volatile(&mut (*desc).css, 0);
148        core::ptr::write_volatile(&mut (*desc).special, 0);
149    }
150}

152/// Whether the controller wrote back DD for TX slot `slot`.
153pub(crate) fn tx_done(ring: &DmaBuf, slot: usize) -> bool {
154    tx_status(ring, slot).1 & DESC_STATUS_DD != 0
155}
156
157/// Whether TX slot `slot` of a `slots`-slot ring may be filled now: it and the
158/// slot after it are both free. One slot always stays empty, so the tail
159/// written after this post (`TDT = slot + 1`) never lands on a descriptor the
160/// controller still owns — `TDT == TDH` would read as an empty ring.
161pub(crate) fn tx_has_room(ring: &DmaBuf, slot: usize, slots: usize) -> bool {
162    tx_room(slot, slots, |s| tx_status(ring, s))
163}
164
165fn tx_room(slot: usize, slots: usize, status: impl Fn(usize) -> (u8, u8)) -> bool {
166    [slot, (slot + 1) % slots].into_iter().all(|s| {
167        let (cmd, st) = status(s);
168        tx_slot_reusable(cmd, st)
169    })
170}
171
172/// A slot is free when it was never posted since the ring was cleared, or the
173/// controller wrote it back (DD). A posted slot without DD still belongs to
174/// the controller.
175fn tx_slot_reusable(cmd: u8, status: u8) -> bool {
176    cmd == 0 || status & DESC_STATUS_DD != 0
177}
178
179/// `(cmd, status)` of TX slot `slot`.
180fn tx_status(ring: &DmaBuf, slot: usize) -> (u8, u8) {
181    assert!((slot + 1) * size_of::<TxDesc>() <= ring.size());
182    // SAFETY: the slot lies inside the owned ring page (asserted above).
183    unsafe {
184        let desc = (ring.virt() as *const TxDesc).add(slot);
185        (
186            core::ptr::read_volatile(&(*desc).cmd),
187            core::ptr::read_volatile(&(*desc).status),
188        )
189    }
190}

192/// Copy `frame` (≤ the buffer page) into the TX buffer `buf`.
193pub(crate) fn copy_in(buf: &DmaBuf, frame: &[u8]) {
194    assert!(frame.len() <= buf.size());
195    // SAFETY: `buf` is an owned DMA page at least `frame.len()` bytes long.
196    unsafe { core::ptr::copy_nonoverlapping(frame.as_ptr(), buf.virt(), frame.len()) };
197}
198
199/// Copy `len` received bytes (≤ the buffer page) from `buf` into `out`.
200pub(crate) fn copy_out(buf: &DmaBuf, out: &mut [u8], len: usize) {
201    assert!(len <= buf.size() && len <= out.len());
202    // SAFETY: both ranges are `len` bytes inside their owners (asserted above).
203    unsafe { core::ptr::copy_nonoverlapping(buf.virt(), out.as_mut_ptr(), len) };
204}

243    #[test]
244    fn rx_delivers_only_whole_error_free_frames() {
245        let mut in_fragment = false;
246        let eop = DESC_STATUS_DD | RXD_STAT_EOP;
247        assert_eq!(
248            rx_written_back(eop, 0, 60, &mut in_fragment),
249            RxCompletion::Frame(60)
250        );
251        // Any reported error (CRC, symbol, sequence, RX data) drops the frame.
252        assert_eq!(
253            rx_written_back(eop, 0x01, 60, &mut in_fragment),
254            RxCompletion::Drop
255        );
256        assert_eq!(
257            rx_written_back(eop, 0x80, 60, &mut in_fragment),
258            RxCompletion::Drop
259        );
260        assert!(!in_fragment);
261        // A frame over several descriptors is dropped up to and including EOP,
262        // so its tail is never delivered as a frame of its own.
263        assert_eq!(
264            rx_written_back(DESC_STATUS_DD, 0, 2048, &mut in_fragment),
265            RxCompletion::Drop
266        );
267        assert!(in_fragment);
268        assert_eq!(
269            rx_written_back(DESC_STATUS_DD, 0, 2048, &mut in_fragment),
270            RxCompletion::Drop
271        );
272        assert_eq!(
273            rx_written_back(eop, 0, 100, &mut in_fragment),
274            RxCompletion::Drop
275        );
276        assert!(!in_fragment);
277        assert_eq!(
278            rx_written_back(eop, 0, 64, &mut in_fragment),
279            RxCompletion::Frame(64)
280        );
281    }

296    /// With nothing written back, a ring of N slots takes N - 1 frames: the
297    /// tail never wraps onto a descriptor the controller still owns. One
298    /// write-back at the slot after the tail frees room again.
299    #[test]
300    fn a_full_tx_ring_keeps_one_slot_empty() {
301        use core::cell::RefCell;
302        const SLOTS: usize = crate::dma_layout::TX_SLOTS;
303        let ring = RefCell::new([(0u8, 0u8); SLOTS]);
304        let room = |slot| tx_room(slot, SLOTS, |s| ring.borrow()[s]);
305        let mut next = 0;
306        let mut posted = 0;
307        while room(next) {
308            ring.borrow_mut()[next] = (TXD_CMD_FRAME, 0);
309            next = (next + 1) % SLOTS;
310            posted += 1;
311            assert!(posted < SLOTS, "the ring filled every slot");
312        }
313        assert_eq!(posted, SLOTS - 1);
314        assert_eq!(next, SLOTS - 1, "the empty slot is the tail");
315        // The head (slot 0) completes: the tail may advance onto it.
316        ring.borrow_mut()[0].1 = DESC_STATUS_DD;
317        assert!(room(next));
318        // A tail slot itself still owned by the controller is refused too.
319        ring.borrow_mut()[next] = (TXD_CMD_FRAME, 0);
320        assert!(!room(next));
321    }
