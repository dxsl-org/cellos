1//! `/bin/e1000e` PCI identity table (`reference-nic-cell-contract.md`).
2//!
3//! Each entry is `Driveable` (an implemented, tested path) or `Target` (a known
4//! SKU this cell's register model is meant to grow into, not admitted). Only
5//! `Driveable` IDs are queried; every other ID is refused by name.
6//!
7//! * `8086:10D3` — 82574L (Intel 82574 GbE Controller Family Datasheet, PCI
8//!   device ID table). QEMU's `e1000e` and VMware's "e1000e" emulate it, and it
9//!   is q35's default NIC. The only `Driveable` entry.
10//! * `8086:10F6` — 82574LA, same datasheet. `Target`: no model validates it.
11//! * I219 (PCH-integrated LAN) — `Target` for phase 13: those parts need the
12//!   PCH-LAN flash NVM path, ULP exit and the PHY workarounds before any of them
13//!   may be admitted. IDs are the PCI ID Repository (`pci.ids`) entries for the
14//!   Sunrise Point, Cannon Point, Comet Lake, Tiger Lake and Alder Lake PCHs;
15//!   phase 13 re-checks them against the I219 datasheet before promotion.
16
17/// Intel's PCI vendor ID.
18pub const VENDOR_INTEL: u16 = 0x8086;
19
20/// The MAC/PHY family a device belongs to; phase 13 adds behaviour per variant
21/// without touching the 82574 paths.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23pub enum MacKind {
24    /// 82574L/82574LA: discrete PCIe MAC + PHY.
25    I82574,
26    /// I219 behind a PCH, by PCH generation.
27    PchLan(PchGen),
28}
29
30/// PCH generation of an I219 connection.
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32pub enum PchGen {
33    SunrisePoint,
34    CannonPoint,
35    CometLake,
36    TigerLake,
37    AlderLake,
38}
39
40/// Whether the cell may bind an ID.
41#[derive(Clone, Copy, Debug, Eq, PartialEq)]
42pub enum Admission {
43    /// Implemented and validated end to end; queried at discovery.
44    Driveable,
45    /// Known SKU, not admitted; refused by name like an unknown ID.
46    Target,
47}
48
49#[derive(Clone, Copy, Debug, Eq, PartialEq)]
50pub struct Identity {
51    pub device_id: u16,
52    pub kind: MacKind,
53    pub admission: Admission,
54    pub name: &'static str,
55}
56
57const fn id(device_id: u16, kind: MacKind, admission: Admission, name: &'static str) -> Identity {
58    Identity {
59        device_id,
60        kind,
61        admission,
62        name,
63    }
64}

66use Admission::{Driveable, Target};
67use MacKind::{PchLan, I82574};
68use PchGen::{AlderLake, CannonPoint, CometLake, SunrisePoint, TigerLake};

70/// Every ID this cell knows, all under [`VENDOR_INTEL`].
71pub const IDENTITIES: [Identity; 16] = [
72    id(0x10D3, I82574, Driveable, "82574L"),
73    id(0x10F6, I82574, Target, "82574LA"),
74    id(0x156F, PchLan(SunrisePoint), Target, "I219-LM"),
75    id(0x1570, PchLan(SunrisePoint), Target, "I219-V"),
76    id(0x15B7, PchLan(SunrisePoint), Target, "I219-LM"),
77    id(0x15B8, PchLan(SunrisePoint), Target, "I219-V"),
78    id(0x15BB, PchLan(CannonPoint), Target, "I219-LM"),
79    id(0x15BC, PchLan(CannonPoint), Target, "I219-V"),
80    id(0x15BD, PchLan(CannonPoint), Target, "I219-LM"),
81    id(0x15BE, PchLan(CannonPoint), Target, "I219-V"),
82    id(0x0D4E, PchLan(CometLake), Target, "I219-LM"),
83    id(0x0D4F, PchLan(CometLake), Target, "I219-V"),
84    id(0x15FB, PchLan(TigerLake), Target, "I219-LM"),
85    id(0x15FC, PchLan(TigerLake), Target, "I219-V"),
86    id(0x1A1C, PchLan(AlderLake), Target, "I219-LM"),
87    id(0x1A1D, PchLan(AlderLake), Target, "I219-V"),
88];

90/// The device IDs handed to `sys_find_pcie_device_by_vendor`, in order: exactly
91/// the `Driveable` entries. The kernel's Ethernet gate lists the same set with
92/// `/bin/e1000e` as owner.
93pub const QUERY_DEVICE_IDS: [u16; 1] = [0x10D3];
94
95const _: () = {
96    // The query list is exactly the Driveable set.
97    let mut driveable = 0;
98    let mut i = 0;
99    while i < IDENTITIES.len() {
100        if matches!(IDENTITIES[i].admission, Admission::Driveable) {
101            assert!(IDENTITIES[i].device_id == QUERY_DEVICE_IDS[driveable]);
102            driveable += 1;
103        }
104        i += 1;
105    }
106    assert!(driveable == QUERY_DEVICE_IDS.len());
107};
108
109/// The table entry for `(vendor, device)`, whatever its admission.
110pub const fn lookup(vendor_id: u16, device_id: u16) -> Option<Identity> {
111    if vendor_id != VENDOR_INTEL {
112        return None;
113    }
114    let mut i = 0;
115    while i < IDENTITIES.len() {
116        if IDENTITIES[i].device_id == device_id {
117            return Some(IDENTITIES[i]);
118        }
119        i += 1;
120    }
121    None
122}
123
124/// The identity this cell may drive, or `None`: unknown and `Target` IDs alike
125/// fail closed, and the caller names vendor:device.
126pub const fn classify(vendor_id: u16, device_id: u16) -> Option<Identity> {
127    match lookup(vendor_id, device_id) {
128        Some(identity) if matches!(identity.admission, Admission::Driveable) => Some(identity),
129        _ => None,
130    }
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136
137    #[test]
138    fn only_the_82574l_is_driveable() {
139        let identity = classify(VENDOR_INTEL, 0x10D3).expect("82574L is driveable");
140        assert_eq!(identity.kind, MacKind::I82574);
141        assert_eq!(identity.name, "82574L");
142        assert_eq!(QUERY_DEVICE_IDS, [0x10D3]);
143    }
144
145    #[test]
146    fn targets_are_known_but_refused() {
147        let i219 = lookup(VENDOR_INTEL, 0x15B7).expect("I219 is listed");
148        assert_eq!(i219.kind, MacKind::PchLan(PchGen::SunrisePoint));
149        assert_eq!(i219.admission, Admission::Target);
150        assert_eq!(classify(VENDOR_INTEL, 0x15B7), None);
151        assert_eq!(classify(VENDOR_INTEL, 0x10F6), None);
152        for identity in IDENTITIES
153            .iter()
154            .filter(|i| i.admission == Admission::Target)
155        {
156            assert!(!QUERY_DEVICE_IDS.contains(&identity.device_id));
157        }
158    }
159
160    #[test]
161    fn sibling_families_and_foreign_vendors_fail_closed() {
162        for device in [0x100E, 0x100F, 0x10C9, 0x1533, 0x0000, 0xFFFF] {
163            assert_eq!(classify(VENDOR_INTEL, device), None, "{device:04x}");
164        }
165        assert_eq!(classify(0x15AD, 0x10D3), None);
166        assert_eq!(classify(0x1AF4, 0x10D3), None);
167    }
168
169    #[test]
170    fn table_has_no_duplicate_ids() {
171        for (index, identity) in IDENTITIES.iter().enumerate() {
172            assert!(IDENTITIES[index + 1..]
173                .iter()
174                .all(|other| other.device_id != identity.device_id));
175        }
176    }
177}
