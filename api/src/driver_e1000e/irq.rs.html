1//! MSI-X routing for the bounded bring-up, link and TX-completion waits
2//! (H-S3-4 b); the waits themselves are `nic_driver::irq::BoundedWait`.
3//!
4//! RX is **not** interrupt-driven: the net service asks for frames (`OP_RX`),
5//! so the RX cause stays masked (`rx=request-driven`).
6//!
7//! Vectors (82574 datasheet §10.2.4.9 `IVAR`): RX queue 0 → entry 0, TX queue 0
8//! → entry 1, "other" (link status change) → entry 2, folded onto fewer entries
9//! when 435 grants fewer. `CTRL_EXT.PBA_SUPPORT` enables the MSI-X PBA path.
10//! `EIAC` stays 0: auto-clear would also auto-mask the cause in `IMS` (without
11//! `CTRL_EXT.IAME`), so instead every park first clears `ICR` and re-arms `IMS`
12//! ([`rearm`]), which makes the next cause a fresh edge, then re-reads the
13//! device.
14
15use crate::regs::{
16    ivar, RegisterFile, CTRL_EXT, CTRL_EXT_PBA_SUPPORT, EIAC, ICR, ICR_LSC, ICR_OTHER, ICR_TXQ0,
17    IMC, IMS, IVAR,
18};
19use types::ViResult;
20
21/// MSI-X entries asked for: RX queue 0, TX queue 0, other causes.
22pub(crate) const WANT_VECTORS: usize = 3;
23
24/// Causes unmasked in MSI-X mode: TX write-back and link changes (via OTHER).
25pub(crate) const IMS_MSIX: u32 = ICR_LSC | ICR_TXQ0 | ICR_OTHER;
26
27/// Program the interrupt routing for `vectors` granted MSI-X entries
28/// (`None`: polled, every cause masked).
29pub(crate) fn program(regs: &impl RegisterFile, vectors: Option<u8>) -> ViResult<()> {
30    regs.wr32(IMC, u32::MAX)?;
31    let Some(granted) = vectors else {
32        return Ok(());
33    };
34    regs.wr32(CTRL_EXT, regs.rd32(CTRL_EXT)? | CTRL_EXT_PBA_SUPPORT)?;
35    regs.wr32(IVAR, ivar(granted))?;
36    regs.wr32(EIAC, 0)?;
37    regs.wr32(ICR, u32::MAX)?;
38    regs.wr32(IMS, IMS_MSIX)
39}
40
41/// Clear latched causes and re-arm before a park: a cause that lands after
42/// this point raises a fresh MSI-X message.
43pub(crate) fn rearm(regs: &impl RegisterFile) -> ViResult<()> {
44    regs.wr32(ICR, u32::MAX)?;
45    regs.wr32(IMS, IMS_MSIX)
46}

48#[cfg(test)]
49mod tests {
50    use super::*;
51    use crate::fake::FakeRegs;
52
53    #[test]
54    fn program_routes_msix_through_ivar_or_masks_everything_when_polled() {
55        let regs = FakeRegs::new();
56        program(&regs, Some(3)).unwrap();
57        assert_eq!(regs.get(IVAR), crate::regs::ivar(3));
58        assert_ne!(regs.get(CTRL_EXT) & CTRL_EXT_PBA_SUPPORT, 0);
59        assert_eq!(regs.get(EIAC), 0, "no auto-clear/auto-mask");
60        assert_eq!(regs.get(IMS), IMS_MSIX);
61        // LSC, TXQ0 and OTHER only: RX (RXQ0, bit 20) is request-driven.
62        assert_eq!(IMS_MSIX, (1 << 2) | (1 << 22) | (1 << 24));
63
64        let regs = FakeRegs::new();
65        program(&regs, None).unwrap();
66        assert_eq!(regs.get(IMC), u32::MAX);
67        let writes = regs.writes.borrow();
68        assert!(!writes.iter().any(|&(o, _)| o == IVAR || o == IMS));
69    }
70
71    #[test]
72    fn rearm_clears_causes_then_unmasks() {
73        let regs = FakeRegs::new();
74        rearm(&regs).unwrap();
75        let writes = regs.writes.borrow();
76        assert_eq!(writes.as_slice(), &[(ICR, u32::MAX), (IMS, IMS_MSIX)]);
77    }
78}
