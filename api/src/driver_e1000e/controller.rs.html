1//! Intel 82574L bring-up and the polled-RX / interrupt-assisted-TX data path.
2//!
3//! Order (82574 datasheet §4.6 initialization): quiesce bus mastering, global
4//! reset, mask interrupts, station address (`RAL0`/`RAH0` auto-loaded from NVM,
5//! `EERD` words 0..3 as the fallback), interrupt routing, PHY autoneg restart,
6//! DMA authorization, rings, TX enable, bounded link wait, RX enable. No
7//! device-visible address is programmed before every DMA object is authorized.
8

25/// Link wait: up to 50 parks (~5 s) with MSI-X, or this many spins when polled
26/// (the igb lane's budget, which covers QEMU's ~500 ms autoneg timer).
27const LINK_PARKS: u32 = 50;
28const LINK_POLLS: u32 = 6_000_000;
29/// TX completion: up to 20 parks (~2 s) with MSI-X, or this many spins.
30const TX_PARKS: u32 = 20;
31const TX_POLLS: u32 = 1_000_000;
32
33pub struct E1000eController {
34    mmio: MmioRegion,
35    irq: NicIrq,
36    tx_ring: DmaBuf,
37    rx_ring: DmaBuf,
38    tx_bufs: [DmaBuf; TX_SLOTS],
39    rx_bufs: [DmaBuf; RX_SLOTS],
40    dma_iovas: DmaIovas,
41    tx_next: usize,
42    rx_head: usize,
43    /// Dropping the descriptors of a frame spread over several of them.
44    rx_in_fragment: bool,
45    pub mac: [u8; 6],
46    /// Whether `STATUS.LU` was up before RX was enabled.
47    pub link_up: bool,
48    #[allow(dead_code)]
49    pub kind: MacKind,
50}
51
52impl E1000eController {
53    /// Bring the controller up on `mmio` (BAR0) for the function at `bdf`.
54    pub fn new(mmio: MmioRegion, bdf: u32, kind: MacKind) -> ViResult<Self> {
55        let mac = reset_and_read_mac(&mmio, kind)?;
56
57        let irq = NicIrq::setup(bdf);
58        irq.program(&mmio)?;
59        mmio.wr32(
60            CTRL,
61            (mmio.rd32(CTRL)? | CTRL_SLU | CTRL_ASDE) & !CTRL_GIO_MASTER_DISABLE,
62        )?;
63        phy::restart_autoneg(&mmio, kind)?;
64
65        let tx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
66        let tx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
67        let rx_ring = DmaBuf::alloc(1).ok_or(ViError::OutOfMemory)?;
68        let rx_bufs = try_init_array(|_| DmaBuf::alloc(1).ok_or(ViError::OutOfMemory))?;
69
70        with_authorized_dma_layout(
71            (mmio, tx_ring, rx_ring, tx_bufs, rx_bufs),
72            |resources, slot| {
73                let result = match slot {
74                    DmaSlot::TxRing => resources.1.authorize(bdf),
75                    DmaSlot::TxBuffer(index) => resources.3[index].authorize(bdf),
76                    DmaSlot::RxRing => resources.2.authorize(bdf),
77                    DmaSlot::RxBuffer(index) => resources.4[index].authorize(bdf),
78                };
79                result.map_err(|_| ViError::PermissionDenied)
80            },
81            |(mmio, tx_ring, rx_ring, tx_bufs, rx_bufs), dma_iovas| {
82                rings::clear(&tx_ring);
83                rings::clear(&rx_ring);
84                let mut ctrl = E1000eController {
85                    mmio,
86                    irq,
87                    tx_ring,
88                    rx_ring,
89                    tx_bufs,
90                    rx_bufs,
91                    dma_iovas,
92                    tx_next: 0,
93                    rx_head: 0,
94                    rx_in_fragment: false,
95                    mac,
96                    link_up: false,
97                    kind,
98                };
99                let mut result = Ok(());
100                for_each_initial_dma_program(&dma_iovas, |program| {
101                    if result.is_ok() {
102                        result = ctrl.apply(program);
103                    }
104                });
105                result.map(|()| ctrl)
106            },
107        )?
108    }
109
110    fn apply(&mut self, program: InitialDmaProgram) -> ViResult<()> {
111        let m = &self.mmio;
112        match program {
113            InitialDmaProgram::TxRingBase(iova) => {
114                m.wr32(TDBAL, iova as u32)?;
115                m.wr32(TDBAH, (iova >> 32) as u32)?;
116                m.wr32(TDLEN, rings::ring_bytes(TX_SLOTS))?;
117                m.wr32(TDH, 0)?;
118                m.wr32(TDT, 0)
119            }
120            InitialDmaProgram::RxDescriptor { slot, iova } => {
121                rings::post_rx(&self.rx_ring, slot, iova);
122                Ok(())
123            }
124            InitialDmaProgram::RxRingBase(iova) => {
125                m.wr32(RDBAL, iova as u32)?;
126                m.wr32(RDBAH, (iova >> 32) as u32)?;
127                m.wr32(RDLEN, rings::ring_bytes(RX_SLOTS))?;
128                m.wr32(RDH, 0)?;
129                m.wr32(RDT, (RX_SLOTS - 1) as u32)
130            }
131            InitialDmaProgram::Enable => {
132                m.wr32(TIPG, TIPG_COPPER)?;
133                m.wr32(TCTL, TCTL_EN | TCTL_PSP | TCTL_CT | TCTL_COLD)?;
134                // The receiver accepts frames only with link up, so wait for
135                // autoneg (bounded) before enabling it; a timeout is reported
136                // in the bind line, not fatal.
137                let mmio = &self.mmio;
138                self.link_up = self.irq.wait_for(mmio, LINK_PARKS, LINK_POLLS, || {
139                    Ok(mmio.rd32(STATUS)? & STATUS_LU != 0)
140                })?;
141                m.wr32(RCTL, RCTL_EN | RCTL_UPE | RCTL_MPE | RCTL_BAM | RCTL_SECRC)
142            }
143        }
144    }

146    /// MSI-X entries held (0 when polled), why the cell is polled, and how many
147    /// parks so far ended by an interrupt / by their deadline.
148    pub fn irq_state(&self) -> (u8, &'static str, u32, u32) {
149        let irq = &self.irq;
150        (irq.vectors(), irq.reason, irq.wakes, irq.expiries)
151    }

153    /// Transmit `frame` (no FCS) and wait, bounded, for its write-back.
154    /// `IO` when the slot to fill or the one after it was posted and never
155    /// written back: the controller may still read the first, and posting
156    /// would move the tail onto the second (one slot always stays empty).
157    pub fn send_frame(&mut self, frame: &[u8]) -> ViResult<()> {
158        if frame.is_empty() || frame.len() > BUF_SIZE {
159            return Err(ViError::InvalidInput);
160        }
161        let slot = self.tx_next;
162        if !rings::tx_has_room(&self.tx_ring, slot, TX_SLOTS) {
163            return Err(ViError::IO);
164        }
165        self.tx_next = (slot + 1) % TX_SLOTS;
166        rings::copy_in(&self.tx_bufs[slot], frame);
167        let iova = self.dma_iovas.tx_descriptor_iova(slot);
168        rings::post_tx(&self.tx_ring, slot, iova, frame.len() as u16);
169        compiler_fence(Ordering::Release);
170        self.mmio.wr32(TDT, self.tx_next as u32)?;
171
172        let ring = &self.tx_ring;
173        if self.irq.wait_for(&self.mmio, TX_PARKS, TX_POLLS, || {
174            Ok(rings::tx_done(ring, slot))
175        })? {
176            Ok(())
177        } else {
178            Err(ViError::IO)
179        }
180    }
181
182    /// Take the next received frame into `out`; 0 when none is ready.
183    /// Errored and multi-descriptor frames are dropped.
184    pub fn recv_frame(&mut self, out: &mut [u8]) -> usize {
185        for _ in 0..RX_SLOTS {
186            let head = self.rx_head;
187            let length = match rings::rx_completed(&self.rx_ring, head, &mut self.rx_in_fragment) {
188                rings::RxCompletion::Pending => return 0,
189                rings::RxCompletion::Frame(length) => Some(length.min(out.len()).min(BUF_SIZE)),
190                rings::RxCompletion::Drop => None,
191            };
192            if let Some(copy_len) = length {
193                rings::copy_out(&self.rx_bufs[head], out, copy_len);
194            }
195            rings::post_rx(&self.rx_ring, head, self.dma_iovas.rx_descriptor_iova(head));
196            self.rx_head = (head + 1) % RX_SLOTS;
197            compiler_fence(Ordering::Release);
198            let _ = self.mmio.wr32(RDT, head as u32);
199            if let Some(copy_len) = length {
200                return copy_len;
201            }
202        }
203        0
204    }
