2//! Files state and operations. Every listing and every change goes to the VFS through
3//! [`FileSystem`]; after a change the folder is listed again, so the view shows what the
4//! service holds, and a refusal is shown as the VFS reported it.
5

11use crate::fs::{self, FileSystem, FsResult, PREVIEW_READ_MAX};

15/// Longest text the path/name prompt accepts (one FAT long name).
16pub const MAX_INPUT_LEN: usize = 255;
17/// Name of the synthetic parent entry at the top of every folder except `/`.
18pub const PARENT_ENTRY: &str = "..";
19
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct Clip {
22    pub path: String,
23    pub name: String,
24    pub is_dir: bool,
25    pub cut: bool,
26}
27
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub enum PromptKind {
30    GoTo,
31    NewFolder,
32    NewFile,
33}
34
35impl PromptKind {
36    const fn label(self) -> &'static str {
37        match self {
38            Self::GoTo => "Go to",
39            Self::NewFolder => "New folder name",
40            Self::NewFile => "New file name",
41        }
42    }
43}
44
45#[derive(Debug, Clone, PartialEq, Eq)]
46pub struct Prompt {
47    pub kind: PromptKind,
48    pub text: String,
49}

51pub struct FilesApp<F: FileSystem> {
52    pub fs: F,
53    pub current_path: String,
54    pub view_mode: ViewMode,
55    pub items: Vec<FileItem>,
56    pub selected_index: usize,
57    pub places: Vec<PlacesItem>,
58    pub clipboard: Option<Clip>,
59    pub prompt: Option<Prompt>,
60    pub status_message: String,
61    pub width: u32,
62    pub height: u32,
63    pub quick_look: QuickLookState,
64}
65
66impl<F: FileSystem> FilesApp<F> {
67    pub fn new(fs: F, initial_path: &str, width: u32, height: u32) -> Self {
68        let mut app = Self {
69            fs,
70            current_path: String::from(initial_path),
71            view_mode: ViewMode::List,
72            items: Vec::new(),
73            selected_index: 0,
74            places: DEFAULT_PLACES.to_vec(),
75            clipboard: None,
76            prompt: None,
77            status_message: String::new(),
78            width,
79            height,
80            quick_look: QuickLookState::new(),
81        };
82        app.load_directory(initial_path);
83        app
84    }
85
86    fn read_listing(&mut self, path: &str) -> FsResult<Vec<FileItem>> {
87        let mut entries = self.fs.list_dir(path)?;
88        // Folders first; the VFS order is kept within each group.
89        entries.sort_by_key(|entry| !entry.is_dir);
90        let mut items = Vec::with_capacity(entries.len() + 1);
91        if path != "/" {
92            items.push(FileItem::new(PARENT_ENTRY, fs::parent(path), true, None));
93        }
94        for entry in entries {
95            let entry_path = fs::join(path, &entry.name);
96            let size = if entry.is_dir {
97                None
98            } else {
99                self.fs.stat(&entry_path).ok().map(|(size, _)| size)
100            };
101            items.push(FileItem::new(entry.name, entry_path, entry.is_dir, size));
102        }
103        Ok(items)
104    }
105
106    fn summary(&self) -> String {
107        let count = self.items.iter().filter(|item| item.name != PARENT_ENTRY).count();
108        format!("{}: {} item(s)", self.current_path, count)
109    }
110
111    /// Show `path` as the VFS lists it. A refused or failed listing keeps the current view
112    /// and puts the reason in the status bar; it never shows an empty stand-in.
113    pub fn load_directory(&mut self, path: &str) -> bool {
114        match self.read_listing(path) {
115            Ok(items) => {
116                self.current_path = String::from(path);
117                self.items = items;
118                self.selected_index = 0;
119                self.status_message = self.summary();
120                true
121            }
122            Err(e) => {
123                self.status_message = format!("Cannot open {}: {}", path, e.describe());
124                false
125            }
126        }
127    }
128
129    /// List the current folder again after a change, select `select` (or keep the old row
130    /// index) and report `status`.
131    fn refresh(&mut self, select: Option<&str>, status: String) {
132        let previous = self.selected_index;
133        let path = self.current_path.clone();
134        if !self.load_directory(&path) {
135            self.status_message = format!("{}. {}", status, self.status_message);
136            return;
137        }
138        self.selected_index = select
139            .and_then(|name| self.items.iter().position(|item| item.name == name))
140            .unwrap_or_else(|| previous.min(self.items.len().saturating_sub(1)));
141        self.status_message = status;
142    }
143
144    fn selected_entry(&self) -> Option<FileItem> {
145        self.items
146            .get(self.selected_index)
147            .filter(|item| item.name != PARENT_ENTRY)
148            .cloned()
149    }
150
151    pub fn navigate_up(&mut self) {
152        if self.current_path != "/" {
153            let parent = String::from(fs::parent(&self.current_path));
154            self.load_directory(&parent);
155        }
156    }
157
158    /// Enter a folder, or Quick Look a file.
159    pub fn open_selected(&mut self) {
160        let Some(item) = self.items.get(self.selected_index).cloned() else {
161            return;
162        };
163        if item.is_dir {
164            self.load_directory(&item.path);
165        } else {
166            self.preview_selected();
167        }
168    }
169
170    /// Open Quick Look on the selection; text kinds are read through the VFS (bounded).
171    pub fn preview_selected(&mut self) {
172        let Some(item) = self.items.get(self.selected_index).cloned() else {
173            return;
174        };
175        let text = PreviewKind::of(&item)
176            .is_text()
177            .then(|| fs::read_file(&mut self.fs, &item.path, PREVIEW_READ_MAX));
178        self.quick_look.show(&item, text);
179    }
180
181    pub fn copy_selected(&mut self, cut: bool) {
182        let Some(item) = self.selected_entry() else {
183            return;
184        };
185        self.status_message = format!("{} {}", if cut { "Cut" } else { "Copied" }, item.name);
186        self.clipboard = Some(Clip { path: item.path, name: item.name, is_dir: item.is_dir, cut });
187    }
188
189    /// Copy (or move, after Cut) the clipboard entry into the current folder under a free name.
190    pub fn paste_into_current(&mut self) {
191        let Some(clip) = self.clipboard.take() else {
192            return;
193        };
194        let existing: Vec<String> = self
195            .items
196            .iter()
197            .filter(|item| item.name != PARENT_ENTRY)
198            .map(|item| item.name.clone())
199            .collect();
200        let final_name = resolve_conflict_name(&clip.name, &existing);
201        let dst = fs::join(&self.current_path, &final_name);
202        let result = if clip.cut {
203            fs::move_entry(&mut self.fs, &clip.path, &dst, clip.is_dir)
204        } else {
205            fs::copy_entry(&mut self.fs, &clip.path, &dst, clip.is_dir)
206        };
207        match result {
208            Ok(()) => {
209                let verb = if clip.cut { "Moved" } else { "Pasted" };
210                let status = format!("{} {} -> {}", verb, clip.name, final_name);
211                if !clip.cut {
212                    self.clipboard = Some(clip);
213                }
214                self.refresh(Some(&final_name), status);
215            }
216            Err(e) => {
217                let status = format!("Paste of {} failed: {}", clip.name, e.describe());
218                self.clipboard = Some(clip);
219                // A partial copy may exist; show what the VFS now holds.
220                self.refresh(None, status);
221            }
222        }
223    }
224
225    pub fn delete_selected(&mut self) {
226        let Some(item) = self.selected_entry() else {
227            return;
228        };
229        match fs::remove_entry(&mut self.fs, &item.path, item.is_dir) {
230            Ok(()) => self.refresh(None, format!("Permanently deleted {} (irreversible)", item.name)),
231            Err(e) => self.status_message = format!("Cannot delete {}: {}", item.name, e.describe()),
232        }
233    }
234
235    pub fn switch_view_mode(&mut self, mode: ViewMode) {
236        self.view_mode = mode;
237        self.status_message = format!("View mode: {}", mode.name());
238    }
239
240    fn show_prompt(&mut self) {
241        if let Some(prompt) = &self.prompt {
242            self.status_message = format!("{}: {}_", prompt.kind.label(), prompt.text);
243        }
244    }
245
246    pub fn begin_prompt(&mut self, kind: PromptKind) {
247        self.prompt = Some(Prompt { kind, text: String::new() });
248        self.show_prompt();
249    }
250
251    pub fn prompt_input(&mut self, ch: char) {
252        if let Some(prompt) = &mut self.prompt {
253            if !ch.is_control() && prompt.text.len() + ch.len_utf8() <= MAX_INPUT_LEN {
254                prompt.text.push(ch);
255            }
256        }
257        self.show_prompt();
258    }
259
260    pub fn prompt_backspace(&mut self) {
261        if let Some(prompt) = &mut self.prompt {
262            prompt.text.pop();
263        }
264        self.show_prompt();
265    }
266
267    pub fn prompt_cancel(&mut self) {
268        if self.prompt.take().is_some() {
269            self.status_message = self.summary();
270        }
271    }
272
273    pub fn prompt_commit(&mut self) {
274        let Some(Prompt { kind, text }) = self.prompt.take() else {
275            return;
276        };
277        if kind == PromptKind::GoTo {
278            let target = fs::resolve(&self.current_path, &text);
279            self.load_directory(&target);
280            return;
281        }
282        let what = if kind == PromptKind::NewFolder { "folder" } else { "file" };
283        if let Err(why) = fs::validate_name(&text) {
284            self.status_message = format!("Cannot create {}: {}", what, why);
285            return;
286        }
287        let path = fs::join(&self.current_path, &text);
288        let result = if kind == PromptKind::NewFolder {
289            self.fs.mkdir(&path).map_err(Into::into)
290        } else {
291            fs::create_file(&mut self.fs, &path)
292        };
293        match result {
294            Ok(()) => self.refresh(Some(&text), format!("Created {} {}", what, text)),
295            Err(e) => self.status_message = format!("Cannot create {} {}: {}", what, text, e.describe()),
296        }
297    }
298}

313    fn type_prompt(app: &mut FilesApp<FakeFs>, kind: PromptKind, text: &str) {
314        app.begin_prompt(kind);

316            app.prompt_input(ch);
317        }

352    #[test]
353    fn c5_flow_create_copy_delete_round_trips_through_the_vfs() {
354        let mut app = app_at(FakeFs::new().dir("/data"), "/data");
355        type_prompt(&mut app, PromptKind::GoTo, "/tmp/c5dir");
356        assert!(app.status_message.starts_with("Cannot open /tmp/c5dir"));
357        app.fs = c5_tree();
358        type_prompt(&mut app, PromptKind::GoTo, "/tmp/c5dir");
359        assert_eq!(app.status_message, "/tmp/c5dir: 1 item(s)");
360
361        type_prompt(&mut app, PromptKind::NewFolder, "c5new");
362        assert_eq!(app.status_message, "Created folder c5new");
363        assert_eq!(names(&app), ["..", "c5new", "a.txt"]);
364        assert_eq!(app.selected_index, 1);
365        assert_eq!(app.fs.nodes.get("/tmp/c5dir/c5new"), Some(&None));
366
367        app.selected_index = 2;
368        app.copy_selected(false);
369        assert_eq!(app.status_message, "Copied a.txt");
370        app.selected_index = 1;
371        app.open_selected();
372        assert_eq!(app.current_path, "/tmp/c5dir/c5new");
373        assert_eq!(names(&app), [".."]);
374        app.paste_into_current();
375        assert_eq!(app.status_message, "Pasted a.txt -> a.txt");
376        assert_eq!(names(&app), ["..", "a.txt"]);
377        assert_eq!(app.selected_index, 1);
378        assert_eq!(app.fs.nodes["/tmp/c5dir/c5new/a.txt"].as_deref(), Some(&b"c5\n"[..]));
379
380        app.delete_selected();
381        assert_eq!(app.status_message, "Permanently deleted a.txt (irreversible)");
382        assert_eq!(names(&app), [".."]);
383        assert!(!app.fs.nodes.contains_key("/tmp/c5dir/c5new/a.txt"));
384        app.navigate_up();
385        app.selected_index = 1;
386        app.delete_selected();
387        assert_eq!(names(&app), ["..", "a.txt"]);
388        assert!(!app.fs.nodes.contains_key("/tmp/c5dir/c5new"));
389    }

411    #[test]
412    fn refused_mutations_are_reported_and_change_nothing() {
413        let mut fs = c5_tree().dir("/bin").file("/bin/tool", b"elf");
414        fs.read_only.push("/bin");
415        let mut app = app_at(fs, "/bin");
416        type_prompt(&mut app, PromptKind::NewFolder, "x");
417        assert_eq!(app.status_message, "Cannot create folder x: permission denied (VFS ERR_DENIED)");
418        type_prompt(&mut app, PromptKind::NewFile, "y");
419        assert_eq!(app.status_message, "Cannot create file y: permission denied (VFS ERR_DENIED)");
420        app.selected_index = 1;
421        app.delete_selected();
422        assert_eq!(app.status_message, "Cannot delete tool: permission denied (VFS ERR_DENIED)");
423        assert!(app.fs.nodes.contains_key("/bin/tool"));
424
425        app.copy_selected(true);
426        type_prompt(&mut app, PromptKind::GoTo, "/tmp");
427        app.paste_into_current();
428        // The copy landed, the source could not be removed: both are visible and reported.
429        assert_eq!(app.status_message, "Paste of tool failed: permission denied (VFS ERR_DENIED)");
430        assert!(app.fs.nodes.contains_key("/bin/tool"));
431        assert_eq!(names(&app), ["..", "c5dir", "tool"]);
432        assert!(app.clipboard.as_ref().is_some_and(|clip| clip.cut));
433
434        // A non-empty folder outside /data is not removed recursively (`rmdir` scope).
435        type_prompt(&mut app, PromptKind::GoTo, "/tmp");
436        app.selected_index = 1;
437        app.delete_selected();
438        assert_eq!(app.status_message, "Cannot delete c5dir: I/O error (VFS ERR_IO)");
439        assert!(app.fs.nodes.contains_key("/tmp/c5dir/a.txt"));
440    }

442    #[test]
443    fn prompt_validates_names_and_bounds_input() {
444        let mut app = app_at(c5_tree(), "/tmp/c5dir");
445        type_prompt(&mut app, PromptKind::NewFolder, "a/b");
446        assert_eq!(app.status_message, "Cannot create folder: a name cannot contain '/'");
447        type_prompt(&mut app, PromptKind::NewFile, "a.txt");
448        assert_eq!(app.status_message, "Cannot create file a.txt: already exists");
449        type_prompt(&mut app, PromptKind::NewFile, "b.txt");
450        assert_eq!(app.status_message, "Created file b.txt");
451        assert_eq!(app.fs.nodes["/tmp/c5dir/b.txt"].as_deref(), Some(&b""[..]));
452
453        app.begin_prompt(PromptKind::NewFolder);
454        for _ in 0..MAX_INPUT_LEN + 10 {
455            app.prompt_input('x');
456        }
457        assert_eq!(app.prompt.as_ref().unwrap().text.len(), MAX_INPUT_LEN);
458        app.prompt_backspace();
459        assert_eq!(app.status_message.len(), "New folder name: ".len() + MAX_INPUT_LEN - 1 + 1);
460        app.prompt_cancel();
461        assert!(app.prompt.is_none());
462        assert_eq!(app.status_message, "/tmp/c5dir: 2 item(s)");
463    }
