18/// Longest text the path/name prompt accepts (one FAT long name).

106    pub(crate) fn read_listing(&mut self, path: &str) -> FsResult<Vec<FileItem>> {
107        let mut entries = self.fs.list_dir(path)?;
108        // Folders first; the VFS order is kept within each group.
109        entries.sort_by_key(|entry| !entry.is_dir);
110        let mut items = Vec::with_capacity(entries.len() + 1);
111        if path != "/" {
112            items.push(FileItem::new(PARENT_ENTRY, fs::parent(path), true, None));
113        }
114        for entry in entries {
115            let entry_path = fs::join(path, &entry.name);
116            let size = if entry.is_dir {
117                None
118            } else {
119                self.fs.stat(&entry_path).ok().map(|(size, _)| size)
120            };
121            items.push(FileItem::new(entry.name, entry_path, entry.is_dir, size));
122        }
123        Ok(items)
124    }

126    fn summary(&self) -> String {
127        let count = self
128            .items
129            .iter()
130            .filter(|item| item.name != PARENT_ENTRY)
131            .count();
132        format!("{}: {} item(s)", self.current_path, count)
133    }

135    /// Show `path` as the VFS lists it. A refused or failed listing keeps the current view
136    /// and puts the reason in the status bar; it never shows an empty stand-in.
137    pub fn load_directory(&mut self, path: &str) -> bool {
138        match self.read_listing(path) {
139            Ok(items) => {
140                self.current_path = String::from(path);
141                self.items = items;
142                self.selected_index = 0;
143                self.status_message = self.summary();
144                true
145            }
146            Err(e) => {
147                self.status_message = format!("Cannot open {}: {}", path, e.describe());
148                false
149            }
150        }
151    }

153    /// List the current folder again after a change, select `select` (or keep the old row
154    /// index) and report `status`. The other split pane is listed again too: the change
155    /// may have touched the folder it shows.
156    pub(crate) fn refresh(&mut self, select: Option<&str>, status: String) {
157        self.reload_secondary_pane();
158        let previous = self.selected_index;
159        let path = self.current_path.clone();
160        if !self.load_directory(&path) {
161            self.status_message = format!("{}. {}", status, self.status_message);
162            return;
163        }
164        self.selected_index = select
165            .and_then(|name| self.items.iter().position(|item| item.name == name))
166            .unwrap_or_else(|| previous.min(self.items.len().saturating_sub(1)));
167        self.status_message = status;
168    }

171        self.items
172            .get(self.selected_index)
173            .filter(|item| item.name != PARENT_ENTRY)
174            .cloned()
175    }

184    /// Enter a folder, or stage a LaunchIntent for a file.
185    pub fn open_selected(&mut self) {
186        let Some(item) = self.items.get(self.selected_index).cloned() else {
187            return;
188        };
189        if item.is_dir {
190            self.load_directory(&item.path);
191        } else {
192            let intent = wire::mime::AssociationRegistry::resolve_intent(&item.path, None);
193            self.status_message = format!(
194                "{} {} with {}",
195                intent.action.as_str(),
196                item.name,
197                intent.app_id
198            );
199            self.pending_intent = Some(intent);
200        }
201    }

251    /// Copy (or move, after Cut) the clipboard entry into the current folder under a free name.
252    pub fn paste_into_current(&mut self) {
253        let Some(clip) = self.clipboard.take() else {
254            return;
255        };
256        let existing: Vec<String> = self
257            .items
258            .iter()
259            .filter(|item| item.name != PARENT_ENTRY)
260            .map(|item| item.name.clone())
261            .collect();
262        let final_name = resolve_conflict_name(&clip.name, &existing);
263        let dst = fs::join(&self.current_path, &final_name);
264        let result = if clip.cut {
265            fs::move_entry(&mut self.fs, &clip.path, &dst, clip.is_dir)
266        } else {
267            fs::copy_entry(&mut self.fs, &clip.path, &dst, clip.is_dir)
268        };
269        match result {
270            Ok(()) => {
271                let verb = if clip.cut { "Moved" } else { "Pasted" };
272                let status = format!("{} {} -> {}", verb, clip.name, final_name);
273                if !clip.cut {
274                    self.clipboard = Some(clip);
275                }
276                self.refresh(Some(&final_name), status);
277            }
278            Err(e) => {
279                let status = format!("Paste of {} failed: {}", clip.name, e.describe());
280                self.clipboard = Some(clip);
281                // A partial copy may exist; show what the VFS now holds.
282                self.refresh(None, status);
283            }
284        }
285    }

287    pub fn delete_selected(&mut self) {
288        let Some(item) = self.selected_entry() else {
289            return;
290        };
291        match fs::remove_entry(&mut self.fs, &item.path, item.is_dir) {
292            Ok(()) => self.refresh(
293                None,
294                format!("Permanently deleted {} (irreversible)", item.name),
295            ),
296            Err(e) => {
297                self.status_message = format!("Cannot delete {}: {}", item.name, e.describe())
298            }
299        }
300    }

302    /// Names in `dir`, straight from the VFS. `write` replaces an existing file, so a

304    pub(crate) fn names_in(&mut self, dir: &str) -> FsResult<Vec<String>> {
305        Ok(self
306            .fs
307            .list_dir(dir)?
308            .into_iter()
309            .map(|entry| entry.name)
310            .collect())
311    }

313    /// Move the selected entry to Trash (`/data/.trash`), preserving it for Undo.
314    pub fn trash_selected(&mut self) {
315        let Some(item) = self.selected_entry() else {
316            return;
317        };
318        // Inside the trash (not merely a sibling such as `.trash-old`) Delete is permanent.
319        if self.current_path == TRASH_DIR
320            || self
321                .current_path
322                .strip_prefix(TRASH_DIR)
323                .is_some_and(|rest| rest.starts_with('/'))
324        {
325            self.delete_selected();
326            return;
327        }
328        let _ = self.fs.mkdir(TRASH_DIR);
329        // No Trash (e.g. `/data` not mounted): say so and name the permanent path
330        // instead of reporting a bare VFS error for the item.
331        let trash_existing = match self.names_in(TRASH_DIR) {
332            Ok(names) => names,
333            Err(e) => {
334                self.status_message = format!(
335                    "Trash unavailable ({}): {} kept; Shift+Delete deletes permanently",
336                    e.describe(),
337                    item.name
338                );
339                return;
340            }
341        };
342        let target_name = resolve_conflict_name(&item.name, &trash_existing);
343        let trashed_path = fs::join(TRASH_DIR, &target_name);
344        match fs::move_entry(&mut self.fs, &item.path, &trashed_path, item.is_dir) {
345            Ok(()) => {
346                let status = format!("Moved {} to Trash (Ctrl+Z to Undo)", item.name);
347                self.trash_stack.push(TrashRecord {
348                    name: item.name,
349                    original_path: item.path,
350                    trashed_path,
351                    is_dir: item.is_dir,
352                });
353                self.refresh(None, status);
354            }
355            Err(e) => {
356                self.status_message =
357                    format!("Cannot move {} to Trash: {}", item.name, e.describe());
358            }
359        }
360    }

362    /// Restore the most recently trashed item to its original location.
363    pub fn undo_last_trash(&mut self) {
364        let Some(record) = self.trash_stack.pop() else {
365            self.status_message = String::from("Nothing to undo in Trash");
366            return;
367        };
368        // Purged or renamed in Trash since: this record can never be restored and must
369        // not block every older one.
370        if self.fs.stat(&record.trashed_path).is_err() {
371            self.status_message = format!("{} is no longer in Trash", record.name);
372            return;
373        }
374        let parent = fs::parent(&record.original_path);
375        let dest_existing = match self.names_in(parent) {
376            Ok(names) => names,
377            Err(e) => {
378                self.status_message = format!("Cannot restore {}: {}", record.name, e.describe());
379                self.trash_stack.push(record);
380                return;
381            }
382        };
383        let restored_name = resolve_conflict_name(&record.name, &dest_existing);
384        let restored_path = fs::join(parent, &restored_name);
385        match fs::move_entry(
386            &mut self.fs,
387            &record.trashed_path,
388            &restored_path,
389            record.is_dir,
390        ) {
391            Ok(()) => {
392                self.refresh(
393                    Some(&restored_name),
394                    format!("Restored {} from Trash", record.name),
395                );
396            }
397            Err(e) => {
398                let status = format!("Cannot restore {}: {}", record.name, e.describe());
399                self.trash_stack.push(record);
400                self.status_message = status;
401            }
402        }
403    }

429    pub fn prompt_input(&mut self, ch: char) {
430        if let Some(prompt) = &mut self.prompt {
431            if !ch.is_control() && prompt.text.len() + ch.len_utf8() <= MAX_INPUT_LEN {
432                prompt.text.push(ch);
433            }
434        }
435        self.show_prompt();
436    }

450
451    pub fn prompt_commit(&mut self) {
452        let Some(Prompt { kind, text }) = self.prompt.take() else {
453            return;
454        };
455        if kind == PromptKind::GoTo {
456            let target = fs::resolve(&self.current_path, &text);
457            self.load_directory(&target);
458            return;
459        }
460        let what = if kind == PromptKind::NewFolder {
461            "folder"
462        } else {
463            "file"
464        };
465        if let Err(why) = fs::validate_name(&text) {
466            self.status_message = format!("Cannot create {}: {}", what, why);
467            return;
468        }
469        let path = fs::join(&self.current_path, &text);
470        let result = if kind == PromptKind::NewFolder {
471            self.fs.mkdir(&path).map_err(Into::into)
472        } else {
473            fs::create_file(&mut self.fs, &path)
474        };
475        match result {
476            Ok(()) => self.refresh(Some(&text), format!("Created {} {}", what, text)),
477            Err(e) => {
478                self.status_message = format!("Cannot create {} {}: {}", what, text, e.describe())
479            }
480        }
481    }

728    #[test]
729    fn trash_and_undo_restores_file() {
730        let fs = FakeFs::new()
731            .dir("/data")
732            .file("/data/memo.txt", b"important\n");
733        let mut app = app_at(fs, "/data");
734        assert_eq!(names(&app), ["..", "memo.txt"]);
735        app.selected_index = 1;
736        assert!(!app.can_undo_trash());
737
738        // Trash the file
739        app.trash_selected();
740        assert!(app.can_undo_trash());
741        assert_eq!(names(&app), ["..", ".trash"]);
742        assert!(app.status_message.contains("Moved memo.txt to Trash"));
743
744        // Verify it was moved to /data/.trash/memo.txt
745        assert!(app.fs.stat("/data/.trash/memo.txt").is_ok());
746        assert!(app.fs.stat("/data/memo.txt").is_err());
747
748        // Undo trash
749        app.undo_last_trash();
750        assert!(!app.can_undo_trash());
751        assert_eq!(names(&app), ["..", ".trash", "memo.txt"]);
752        assert!(app.status_message.contains("Restored memo.txt from Trash"));
753        assert!(app.fs.stat("/data/memo.txt").is_ok());
754        assert!(app.fs.stat("/data/.trash/memo.txt").is_err());
755    }
