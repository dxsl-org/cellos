28#[derive(Debug, Clone, PartialEq, Eq)]
29pub struct FileItem {
30    pub name: String,
31    pub path: String,
32    pub is_dir: bool,
33    pub mime: api::services::mime::MimeType,
34    /// `Stat` size; `None` for folders and when the VFS refused the `Stat`.
35    pub size: Option<u64>,
36    pub icon: LineIcon,
37}
38
39impl FileItem {
40    pub fn new(
41        name: impl Into<String>,
42        path: impl Into<String>,
43        is_dir: bool,
44        size: Option<u64>,
45    ) -> Self {
46        let name = name.into();
47        let path = path.into();
48        let mime = if is_dir {
49            api::services::mime::MimeType::InodeDirectory
50        } else {
51            api::services::mime::MimeType::from_path(&path)
52        };
53        let icon = LineIcon::for_mime(mime);
54
55        Self {
56            name,
57            path,
58            is_dir,
59            size,
60            mime,
61            icon,
62        }
63    }
64
65    pub fn formatted_size(&self) -> String {
66        match self.size {
67            _ if self.is_dir => String::from("--"),
68            None => String::from("?"),
69            Some(size) if size < 1024 => format!("{} B", size),
70            Some(size) if size < 1024 * 1024 => format!("{} KB", size / 1024),
71            Some(size) => format!("{} MB", size / (1024 * 1024)),
72        }
73    }
74}

138/// Calculate a non-conflicting filename by appending ` (1)`, ` (2)`, etc.
139pub fn resolve_conflict_name(name: &str, existing_names: &[String]) -> String {
140    if !existing_names.iter().any(|n| n == name) {
141        return String::from(name);
142    }
143
144    let (stem, ext) = match name.rfind('.') {
145        Some(idx) if idx > 0 => (&name[..idx], &name[idx..]),
146        _ => (name, ""),
147    };
148
149    for i in 1..100 {
150        let candidate = format!("{} ({}){}", stem, i, ext);
151        if !existing_names.iter().any(|n| n == &candidate) {
152            return candidate;
153        }
154    }
155    format!("{} (copy){}", stem, ext)
156}

158#[cfg(test)]
159mod tests {
160    use super::*;
161
162    #[test]
163    fn test_conflict_name_resolution() {
164        let existing = alloc::vec![
165            String::from("notes.md"),
166            String::from("notes (1).md"),
167            String::from("document.txt"),
168        ];
169
170        // Unique name remains unchanged
171        assert_eq!(resolve_conflict_name("report.pdf", &existing), "report.pdf");
172
173        // Existing name gets (2) since (1) already exists
174        assert_eq!(resolve_conflict_name("notes.md", &existing), "notes (2).md");
175
176        // Existing name without (1) gets (1)
177        assert_eq!(
178            resolve_conflict_name("document.txt", &existing),
179            "document (1).txt"
180        );
181    }

183    #[test]
184    fn test_file_item_icons_and_sizes() {
185        let dir = FileItem::new("docs", "/docs", true, None);
186        assert_eq!(dir.icon, LineIcon::Folder);
187        assert_eq!(dir.formatted_size(), "--");
188
189        let doc = FileItem::new("readme.md", "/readme.md", false, Some(2048));
190        assert_eq!(doc.icon, LineIcon::Document);
191        assert_eq!(doc.formatted_size(), "2 KB");
192
193        let pdf = FileItem::new("manual.pdf", "/manual.pdf", false, Some(5 * 1024 * 1024));
194        assert_eq!(pdf.icon, LineIcon::Document);
195        assert_eq!(pdf.formatted_size(), "5 MB");
196
197        let refused = FileItem::new("secret", "/secret", false, None);
198        assert_eq!(refused.formatted_size(), "?");
199    }
