2//! Quick Look HUD for Files. Text documents show the first lines of the file as read from
3//! the VFS (bounded read); every other kind shows its real size and no fabricated content.

18pub const QUICK_LOOK_W: u32 = 540;
19pub const QUICK_LOOK_H: u32 = 400;
20/// Text lines and columns the HUD body can show.
21const PREVIEW_LINES: usize = 12;
22const PREVIEW_COLS: usize = 72;
23
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum PreviewKind {
26    Folder,
27    Image,
28    Markdown,
29    Code,
30    Pdf,
31    Unsupported,
32}
33
34impl PreviewKind {
35    pub fn of(item: &FileItem) -> Self {
36        let name = item.name.as_str();
37        if item.is_dir {
38            Self::Folder
39        } else if name.ends_with(".png") || name.ends_with(".jpg") || name.ends_with(".ppm") {
40            Self::Image
41        } else if name.ends_with(".md") {
42            Self::Markdown
43        } else if name.ends_with(".rs") || name.ends_with(".c") || name.ends_with(".toml") || name.ends_with(".txt") {
44            Self::Code
45        } else if name.ends_with(".pdf") {
46            Self::Pdf
47        } else {
48            Self::Unsupported
49        }
50    }
51
52    /// Whether Quick Look reads and shows the file's text.
53    pub const fn is_text(self) -> bool {
54        matches!(self, Self::Markdown | Self::Code)
55    }
56
57    pub const fn name(self) -> &'static str {
58        match self {
59            Self::Folder => "Folder",
60            Self::Image => "Image",
61            Self::Markdown => "Markdown Document",
62            Self::Code => "Source Code / Text",
63            Self::Pdf => "PDF Document",
64            Self::Unsupported => "Binary / Unknown",
65        }
66    }
67}
68
69pub struct QuickLookState {
70    pub is_open: bool,
71    pub file_name: String,
72    pub kind: PreviewKind,
73    pub lines: Vec<String>,
74}
75
76impl QuickLookState {
77    pub fn new() -> Self {
78        Self {
79            is_open: false,
80            file_name: String::new(),
81            kind: PreviewKind::Unsupported,
82            lines: Vec::new(),
83        }
84    }
85
86    /// Open (or retarget) Quick Look on `item`. `text` is the VFS read of a text kind and
87    /// `None` for kinds that are not read.
88    pub fn show(&mut self, item: &FileItem, text: Option<FsResult<Vec<u8>>>) {
89        self.is_open = true;
90        self.file_name = item.name.clone();
91        self.kind = PreviewKind::of(item);
92        self.lines.clear();
93        match text {
94            Some(Ok(bytes)) if bytes.is_empty() => self.lines.push(String::from("(empty file)")),
95            Some(Ok(bytes)) => {
96                let text = String::from_utf8_lossy(&bytes);
97                self.lines.extend(
98                    text.lines()
99                        .take(PREVIEW_LINES)
100                        .map(|line| line.chars().take(PREVIEW_COLS).collect::<String>()),
101                );
102            }
103            Some(Err(e)) => self.lines.push(format!("Cannot read {}: {}", item.name, e.describe())),
104            None if item.is_dir => self.lines.push(String::from("Folder. Press Enter to open it.")),
105            None => {
106                self.lines.push(format!("No inline preview for {}.", self.kind.name()));
107                self.lines.push(format!("Size: {}", item.formatted_size()));
108            }
109        }
110    }
111
112    pub fn close(&mut self) {
113        self.is_open = false;
114        self.lines.clear();
115    }
116}

118fn hud_rect(win_w: u32, win_h: u32) -> (i32, i32, u32, u32) {
119    let ql_w = QUICK_LOOK_W.min(win_w.saturating_sub(40));
120    let ql_h = QUICK_LOOK_H.min(win_h.saturating_sub(60));
121    ((win_w as i32 - ql_w as i32) / 2, (win_h as i32 - ql_h as i32) / 2, ql_w, ql_h)
122}
123
124/// A click while Quick Look is open: outside the HUD dismisses it. Returns whether the
125/// HUD consumed the click.
126pub fn handle_click(state: &mut QuickLookState, x: i32, y: i32, win_w: u32, win_h: u32) -> bool {
127    if !state.is_open {
128        return false;
129    }
130    let (ql_x, ql_y, ql_w, ql_h) = hud_rect(win_w, win_h);
131    if x < ql_x || x >= ql_x + ql_w as i32 || y < ql_y || y >= ql_y + ql_h as i32 {
132        state.close();
133    }
134    true
135}

137pub fn render(
138    surf: &mut ViSurface,
139    fonts: &mut TextFonts,
140    state: &QuickLookState,
141    win_w: u32,
142    win_h: u32,
143) {
144    if !state.is_open {
145        return;
146    }
147    let (ql_x, ql_y, ql_w, ql_h) = hud_rect(win_w, win_h);
148
149    // Dim background overlay
150    fill_rect(surf, 0, 0, win_w, win_h, Color::rgb(0, 0, 0).with_alpha(80));
151
152    // Modal background & border
153    fill_rect(surf, ql_x, ql_y, ql_w, ql_h, Color::rgb(255, 255, 255));
154    stroke_rect(surf, ql_x, ql_y, ql_w, ql_h, Color::rgb(35, 90, 215));
155
156    // Header bar (36 DIP)
157    let header_h = 36;
158    fill_rect(surf, ql_x, ql_y, ql_w, header_h, Color::rgb(240, 242, 248));
159    fill_rect(surf, ql_x, ql_y + header_h as i32 - 1, ql_w, 1, Color::rgb(215, 218, 224));
160
161    let title_text = format!("{}  ·  {}", state.file_name, state.kind.name());
162    draw_label(surf, fonts, ql_x + 14, ql_y + 8, ql_w.saturating_sub(28), 20, &title_text, FontFace::UiSemibold, 14.0, Color::rgb(20, 20, 28));
163
164    let body_x = ql_x + 20;
165    let body_w = ql_w.saturating_sub(40);
166    let (face, size) = if state.kind.is_text() {
167        (FontFace::MonoRegular, 13.0)
168    } else {
169        (FontFace::UiRegular, 14.0)
170    };
171    let mut line_y = ql_y + header_h as i32 + 16;
172    for line in &state.lines {
173        draw_label(surf, fonts, body_x, line_y, body_w, 20, line, face, size, Color::rgb(40, 45, 55));
174        line_y += 24;
175    }
176
177    // Footer with dismissal instructions
178    let footer_y = ql_y + ql_h as i32 - 28;
179    fill_rect(surf, ql_x, footer_y, ql_w, 28, Color::rgb(245, 245, 248));
180    fill_rect(surf, ql_x, footer_y, ql_w, 1, Color::rgb(215, 218, 224));
181    draw_label(
182        surf,
183        fonts,
184        ql_x + 16,
185        footer_y + 4,
186        ql_w - 32,
187        20,
188        "Space / Esc: Dismiss    ↑ / ↓: Navigate Items",
189        FontFace::UiRegular,
190        12.0,
191        Color::rgb(120, 125, 140),
192    );
193}

195#[cfg(test)]
196mod tests {
197    use super::*;
198    use crate::fs::FsError;
199    use ostd::ViError;
200
201    fn file(name: &str) -> FileItem {
202        FileItem::new(name, alloc::format!("/data/{name}"), false, Some(5))
203    }
204
205    #[test]
206    fn preview_kind_from_entry() {
207        assert_eq!(PreviewKind::of(&file("photo.png")), PreviewKind::Image);
208        assert_eq!(PreviewKind::of(&file("readme.md")), PreviewKind::Markdown);
209        assert_eq!(PreviewKind::of(&file("a.txt")), PreviewKind::Code);
210        assert_eq!(PreviewKind::of(&file("manual.pdf")), PreviewKind::Pdf);
211        assert_eq!(PreviewKind::of(&file("archive.tar")), PreviewKind::Unsupported);
212        assert_eq!(PreviewKind::of(&FileItem::new("d", "/d", true, None)), PreviewKind::Folder);
213    }
214
215    #[test]
216    fn shows_read_text_bounded_and_reports_failures() {
217        let mut ql = QuickLookState::new();
218        let long: String = core::iter::repeat('x').take(PREVIEW_COLS + 10).collect();
219        let mut text = alloc::format!("first\n{long}\n");
220        for i in 0..PREVIEW_LINES {
221            text.push_str(&alloc::format!("l{i}\n"));
222        }
223        ql.show(&file("a.txt"), Some(Ok(text.into_bytes())));
224        assert!(ql.is_open);
225        assert_eq!(ql.lines.len(), PREVIEW_LINES);
226        assert_eq!(ql.lines[0], "first");
227        assert_eq!(ql.lines[1].chars().count(), PREVIEW_COLS);
228
229        ql.show(&file("a.txt"), Some(Err(FsError::Vfs(ViError::PermissionDenied))));
230        assert_eq!(ql.lines, ["Cannot read a.txt: permission denied (VFS ERR_DENIED)"]);
231
232        ql.show(&file("photo.png"), None);
233        assert_eq!(ql.lines, ["No inline preview for Image.", "Size: 5 B"]);
234
235        assert!(handle_click(&mut ql, 10, 10, 800, 600));
236        assert!(!ql.is_open);
237        assert!(!handle_click(&mut ql, 10, 10, 800, 600));
238    }
