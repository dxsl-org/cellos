19pub const QUICK_LOOK_W: u32 = 540;
20pub const QUICK_LOOK_H: u32 = 400;
21/// Text lines and columns the HUD body can show.
22const PREVIEW_LINES: usize = 12;
23const PREVIEW_COLS: usize = 72;
24
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26pub enum PreviewKind {
27    Folder,
28    Image,
29    Markdown,
30    Code,
31    Pdf,
32    Unsupported,
33}
34
35impl PreviewKind {
36    pub fn of(item: &FileItem) -> Self {
37        if item.is_dir {
38            Self::Folder
39        } else {
40            match item.mime {
41                wire::mime::MimeType::ImagePng
42                | wire::mime::MimeType::ImageJpeg
43                | wire::mime::MimeType::ImagePpm => Self::Image,
44                wire::mime::MimeType::TextMarkdown => Self::Markdown,
45                wire::mime::MimeType::TextPlain | wire::mime::MimeType::TextSource => Self::Code,
46                wire::mime::MimeType::ApplicationPdf => Self::Pdf,
47                _ => Self::Unsupported,
48            }
49        }
50    }
51
52    /// Whether Quick Look reads and shows the file's text.
53    pub const fn is_text(self) -> bool {
54        matches!(self, Self::Markdown | Self::Code)
55    }
56
57    pub const fn name(self) -> &'static str {
58        match self {
59            Self::Folder => "Folder",
60            Self::Image => "Image",
61            Self::Markdown => "Markdown Document",
62            Self::Code => "Source Code / Text",
63            Self::Pdf => "PDF Document",
64            Self::Unsupported => "Binary / Unknown",
65        }
66    }
67}

69pub struct QuickLookState {
70    pub is_open: bool,
71    pub file_name: String,
72    pub file_path: String,
73    pub kind: PreviewKind,
74    pub mime: wire::mime::MimeType,
75    pub default_app: Option<&'static str>,
76    pub candidates: &'static [wire::mime::AppCandidate],
77    pub lines: Vec<String>,
78}
79
80impl QuickLookState {
81    pub fn new() -> Self {
82        Self {
83            is_open: false,
84            file_name: String::new(),
85            file_path: String::new(),
86            kind: PreviewKind::Unsupported,
87            mime: wire::mime::MimeType::Unknown,
88            default_app: None,
89            candidates: &[],
90            lines: Vec::new(),
91        }
92    }
93
94    /// Open (or retarget) Quick Look on `item`. `text` is the VFS read of a text kind and
95    /// `None` for kinds that are not read.
96    pub fn show(&mut self, item: &FileItem, text: Option<FsResult<Vec<u8>>>) {
97        self.is_open = true;
98        self.file_name = item.name.clone();
99        self.file_path = item.path.clone();
100        self.kind = PreviewKind::of(item);
101        self.mime = item.mime;
102        self.default_app =
103            wire::mime::AssociationRegistry::default_app(item.mime).map(|a| a.app_name);
104        self.candidates = wire::mime::AssociationRegistry::candidates_for_mime(item.mime);
105        self.lines.clear();
106        match text {
107            Some(Ok(bytes)) if bytes.is_empty() => self.lines.push(String::from("(empty file)")),
108            Some(Ok(bytes)) => {
109                let text = String::from_utf8_lossy(&bytes);
110                self.lines.extend(
111                    text.lines()
112                        .take(PREVIEW_LINES)
113                        .map(|line| line.chars().take(PREVIEW_COLS).collect::<String>()),
114                );
115            }
116            Some(Err(e)) => self
117                .lines
118                .push(format!("Cannot read {}: {}", item.name, e.describe())),
119            None if item.is_dir => self
120                .lines
121                .push(String::from("Folder. Press Enter to open it.")),
122            None => {
123                self.lines
124                    .push(format!("No inline preview for {}.", self.kind.name()));
125                self.lines.push(format!("Size: {}", item.formatted_size()));
126            }
127        }
128    }
129
130    pub fn close(&mut self) {
131        self.is_open = false;
132        self.lines.clear();
133    }
134}

136fn hud_rect(win_w: u32, win_h: u32) -> (i32, i32, u32, u32) {
137    let ql_w = QUICK_LOOK_W.min(win_w.saturating_sub(40));
138    let ql_h = QUICK_LOOK_H.min(win_h.saturating_sub(60));
139    (
140        (win_w as i32 - ql_w as i32) / 2,
141        (win_h as i32 - ql_h as i32) / 2,
142        ql_w,
143        ql_h,
144    )
145}

147/// A click while Quick Look is open: outside the HUD dismisses it. Returns whether the
148/// HUD consumed the click.
149pub fn handle_click(state: &mut QuickLookState, x: i32, y: i32, win_w: u32, win_h: u32) -> bool {
150    if !state.is_open {
151        return false;
152    }
153    let (ql_x, ql_y, ql_w, ql_h) = hud_rect(win_w, win_h);
154    if x < ql_x || x >= ql_x + ql_w as i32 || y < ql_y || y >= ql_y + ql_h as i32 {
155        state.close();
156    }
157    true
158}

160pub fn render(
161    surf: &mut ViSurface,
162    fonts: &mut TextFonts,
163    theme: &dyn ViTheme,
164    state: &QuickLookState,
165    win_w: u32,
166    win_h: u32,
167) {
168    if !state.is_open {
169        return;
170    }
171    let (ql_x, ql_y, ql_w, ql_h) = hud_rect(win_w, win_h);
172
173    // Dim scrim over the window: black at low alpha darkens every palette alike.
174    fill_rect(surf, 0, 0, win_w, win_h, Color::BLACK.with_alpha(80));
175
176    // Modal background & border
177    fill_rect(surf, ql_x, ql_y, ql_w, ql_h, theme.surface());
178    stroke_rect(surf, ql_x, ql_y, ql_w, ql_h, theme.accent());
179
180    // Header bar (36 DIP)
181    let header_h = 36;
182    fill_rect(surf, ql_x, ql_y, ql_w, header_h, theme.bg());
183    fill_rect(
184        surf,
185        ql_x,
186        ql_y + header_h as i32 - 1,
187        ql_w,
188        1,
189        theme.border(),
190    );
191
192    let title_text = format!("{}  ·  {}", state.file_name, state.kind.name());
193    draw_label(
194        surf,
195        fonts,
196        ql_x + 14,
197        ql_y + 8,
198        ql_w.saturating_sub(28),
199        20,
200        &title_text,
201        FontFace::UiSemibold,
202        14.0,
203        theme.text_primary(),
204    );
205
206    let body_x = ql_x + 20;
207    let body_w = ql_w.saturating_sub(40);
208    let (face, size) = if state.kind.is_text() {
209        (FontFace::MonoRegular, 13.0)
210    } else {
211        (FontFace::UiRegular, 14.0)
212    };
213    let mut line_y = ql_y + header_h as i32 + 16;
214    for line in &state.lines {
215        draw_label(
216            surf,
217            fonts,
218            body_x,
219            line_y,
220            body_w,
221            20,
222            line,
223            face,
224            size,
225            theme.text_primary(),
226        );
227        line_y += 24;
228    }
229
230    // Footer with dismissal instructions
231    let footer_y = ql_y + ql_h as i32 - 28;
232    fill_rect(surf, ql_x, footer_y, ql_w, 28, theme.bg());
233    fill_rect(surf, ql_x, footer_y, ql_w, 1, theme.border());
234    let footer_text = if let Some(app) = state.default_app {
235        if state.mime.is_executable() {
236            format!("Enter: Run in {}    Space/Esc: Dismiss", app)
237        } else if state.candidates.len() > 1 {
238            format!(
239                "Enter: Open ({})  ·  1..{}: Open With  ·  Esc: Dismiss",
240                app,
241                state.candidates.len()
242            )
243        } else {
244            format!("Enter: Open with {}    Space/Esc: Dismiss", app)
245        }
246    } else {
247        String::from("Space / Esc: Dismiss    ↑ / ↓: Navigate Items")
248    };
249    draw_label(
250        surf,
251        fonts,
252        ql_x + 16,
253        footer_y + 4,
254        ql_w - 32,
255        20,
256        &footer_text,
257        FontFace::UiRegular,
258        12.0,
259        theme.text_secondary(),
260    );
261}

263#[cfg(test)]
264mod tests {
265    use super::*;
266    use crate::fs::FsError;
267    use ostd::ViError;

269    fn file(name: &str) -> FileItem {
270        FileItem::new(name, alloc::format!("/data/{name}"), false, Some(5))
271    }
272
273    #[test]
274    fn preview_kind_from_entry() {
275        assert_eq!(PreviewKind::of(&file("photo.png")), PreviewKind::Image);
276        assert_eq!(PreviewKind::of(&file("readme.md")), PreviewKind::Markdown);
277        assert_eq!(PreviewKind::of(&file("a.txt")), PreviewKind::Code);
278        assert_eq!(PreviewKind::of(&file("manual.pdf")), PreviewKind::Pdf);
279        assert_eq!(
280            PreviewKind::of(&file("archive.tar")),
281            PreviewKind::Unsupported
282        );
283        assert_eq!(
284            PreviewKind::of(&FileItem::new("d", "/d", true, None)),
285            PreviewKind::Folder
286        );
287    }
288
289    #[test]
290    fn shows_read_text_bounded_and_reports_failures() {
291        let mut ql = QuickLookState::new();
292        let long = "x".repeat(PREVIEW_COLS + 10);
293        let mut text = alloc::format!("first\n{long}\n");
294        for i in 0..PREVIEW_LINES {
295            text.push_str(&alloc::format!("l{i}\n"));
296        }
297        ql.show(&file("a.txt"), Some(Ok(text.into_bytes())));
298        assert!(ql.is_open);
299        assert_eq!(ql.lines.len(), PREVIEW_LINES);
300        assert_eq!(ql.lines[0], "first");
301        assert_eq!(ql.lines[1].chars().count(), PREVIEW_COLS);
302
303        ql.show(
304            &file("a.txt"),
305            Some(Err(FsError::Vfs(ViError::PermissionDenied))),
306        );
307        assert_eq!(
308            ql.lines,
309            ["Cannot read a.txt: permission denied (VFS ERR_DENIED)"]
310        );
311
312        ql.show(&file("photo.png"), None);
313        assert_eq!(ql.lines, ["No inline preview for Image.", "Size: 5 B"]);
314
315        assert!(handle_click(&mut ql, 10, 10, 800, 600));
316        assert!(!ql.is_open);
317        assert!(!handle_click(&mut ql, 10, 10, 800, 600));
318    }
