5use ostd::display::ViSurface;
6use ostd::typography::{FontFace, TextFonts};
7use viui::canvas::Color;
8
9use crate::draw::{draw_label, fill_rect};
10use crate::model::PlacesItem;
11
12pub const PLACES_WIDTH: u32 = 160;
13
14pub fn render(
15    surf: &mut ViSurface,
16    fonts: &mut TextFonts,
17    places: &[PlacesItem],
18    active_path: &str,
19    view_h: u32,
20    theme_bg: Color,
21    theme_fg: Color,
22    theme_accent: Color,
23    theme_border: Color,
24) {
25    let sw = PLACES_WIDTH;
26    let sh = view_h;
27
28    // Sidebar background and right border
29    fill_rect(surf, 0, 44, sw, sh, theme_bg);
30    fill_rect(surf, sw as i32 - 1, 44, 1, sh, theme_border);
31
32    // Section header: PLACES
33    draw_label(surf, fonts, 14, 52, sw - 28, 20, "PLACES", FontFace::UiSemibold, 12.0, Color::rgb(130, 135, 150));
34
35    let row_h = 28;
36    let start_y = 76;
37
38    for (i, item) in places.iter().enumerate() {
39        let row_y = start_y + i as i32 * row_h;
40        let is_active = active_path == item.path || (item.path != "/" && active_path.starts_with(item.path));
41
42        if is_active {
43            fill_rect(surf, 6, row_y, sw - 12, row_h as u32, theme_accent.with_alpha(60));
44            fill_rect(surf, 6, row_y, 3, row_h as u32, theme_accent);
45        }
46
47        draw_label(surf, fonts, 14, row_y + 4, 24, 20, item.icon, FontFace::UiSemibold, 13.0, if is_active { theme_accent } else { theme_fg });
48
49        // Label
50        draw_label(surf, fonts, 42, row_y + 4, sw - 54, 20, item.name, FontFace::UiRegular, 14.0, theme_fg);
51    }
52}
53
54pub fn handle_click(places: &[PlacesItem], x: i32, y: i32) -> Option<&'static str> {
55    if x < 6 || x >= PLACES_WIDTH as i32 - 6 {
56        return None;
57    }
58    let start_y = 76;
59    let row_h = 28;
60    if y < start_y {
61        return None;
62    }
63    let idx = ((y - start_y) / row_h) as usize;
64    places.get(idx).map(|item| item.path)
65}
