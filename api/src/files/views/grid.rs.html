3//! Grid View renderer for Files application (visual icon/thumbnail card browsing).

5use ostd::display::ViSurface;
6use ostd::typography::{FontFace, TextFonts};
7use viui::canvas::Color;
8
9use crate::draw::{draw_label, fill_rect, stroke_rect};
10use crate::model::FileItem;
11
12pub const CARD_W: u32 = 100;
13pub const CARD_H: u32 = 110;
14pub const CARD_GAP: u32 = 12;
15
16pub fn render(
17    surf: &mut ViSurface,
18    fonts: &mut TextFonts,
19    items: &[FileItem],
20    selected_idx: usize,
21    content_x: i32,
22    content_y: i32,
23    content_w: u32,
24    content_h: u32,
25    theme_bg: Color,
26    theme_fg: Color,
27    theme_accent: Color,
28    theme_border: Color,
29) {
30    fill_rect(surf, content_x, content_y, content_w, content_h, theme_bg);
31
32    let cols = ((content_w.saturating_sub(CARD_GAP)) / (CARD_W + CARD_GAP)).max(1) as usize;
33    let pad_x = content_x + 12;
34    let pad_y = content_y + 12;
35
36    for (i, item) in items.iter().enumerate() {
37        let col = i % cols;
38        let row = i / cols;
39        let cx = pad_x + col as i32 * (CARD_W + CARD_GAP) as i32;
40        let cy = pad_y + row as i32 * (CARD_H + CARD_GAP) as i32;
41
42        if cy + CARD_H as i32 > content_y + content_h as i32 {
43            break;
44        }
45
46        let is_selected = i == selected_idx;
47
48        if is_selected {
49            fill_rect(surf, cx, cy, CARD_W, CARD_H, theme_accent.with_alpha(40));
50            stroke_rect(surf, cx, cy, CARD_W, CARD_H, theme_accent);
51        } else {
52            fill_rect(surf, cx, cy, CARD_W, CARD_H, Color::rgb(250, 250, 252));
53            stroke_rect(surf, cx, cy, CARD_W, CARD_H, theme_border);
54        }
55
56        // Preview box / icon
57        let icon_box_y = cy + 10;
58        let icon_box_h = 50;
59        fill_rect(surf, cx + 15, icon_box_y, CARD_W - 30, icon_box_h, if is_selected { theme_accent.with_alpha(30) } else { Color::rgb(240, 242, 248) });
60        draw_label(surf, fonts, cx + 20, icon_box_y + 12, CARD_W - 40, 24, item.icon, FontFace::UiSemibold, 18.0, if item.is_dir { theme_accent } else { theme_fg });
61
62        // Label
63        let label_y = cy + 68;
64        draw_label(surf, fonts, cx + 6, label_y, CARD_W - 12, 18, &item.name, FontFace::UiRegular, 13.0, theme_fg);
65
66        // Size subtext
67        let size_str = item.formatted_size();
68        draw_label(surf, fonts, cx + 6, label_y + 18, CARD_W - 12, 16, &size_str, FontFace::UiRegular, 11.0, Color::rgb(130, 135, 150));
69    }
70}

72pub fn handle_click(
73    items_len: usize,
74    content_x: i32,
75    content_y: i32,
76    content_w: u32,
77    x: i32,
78    y: i32,
79) -> Option<usize> {
80    if x < content_x || x >= content_x + content_w as i32 || y < content_y {
81        return None;
82    }
83    let cols = ((content_w.saturating_sub(CARD_GAP)) / (CARD_W + CARD_GAP)).max(1) as usize;
84    let pad_x = content_x + 12;
85    let pad_y = content_y + 12;
86
87    let rel_x = x - pad_x;
88    let rel_y = y - pad_y;
89    if rel_x < 0 || rel_y < 0 {
90        return None;
91    }
92
93    let col = (rel_x as u32 / (CARD_W + CARD_GAP)) as usize;
94    let row = (rel_y as u32 / (CARD_H + CARD_GAP)) as usize;
95    if col >= cols {
96        return None;
97    }
98
99    let idx = row * cols + col;
100    if idx < items_len {
101        Some(idx)
102    } else {
103        None
104    }
