30impl<F: FileSystem> FilesApp<F> {
31    /// Draw the whole window and mark it damaged.
32    pub fn render(&self, surf: &mut ViSurface, fonts: &mut TextFonts) {
33        let sw = self.width;
34        let sh = self.height;
35        let theme = theme::system();
36
37        let path_title = format!("Files: {}", self.current_path);
38        Toolbar::render(
39            surf,
40            fonts,
41            theme,
42            sw,
43            TOOLBAR_HEIGHT,
44            Some(LineIcon::Folder),
45            &path_title,
46        );
47        let mode_idx = match self.view_mode {
48            ViewMode::List => 0,
49            ViewMode::Grid => 1,
50            ViewMode::Columns => 2,
51        };
52        segments(
53            surf,
54            fonts,
55            theme,
56            layout::view_switch(sw),
57            &layout::VIEW_MODE_LABELS,
58            mode_idx,
59        );
60        // Lit while split, like the selected view mode; no index lights none.
61        let split_idx = if self.is_split() { 0 } else { usize::MAX };
62        segments(
63            surf,
64            fonts,
65            theme,
66            layout::split_button(sw),
67            &[layout::SPLIT_LABEL],
68            split_idx,
69        );
70
71        let content = layout::content_rect(sw, sh);
72        let places_defs: Vec<PlacesItemDef> = self
73            .places
74            .iter()
75            .map(|p| PlacesItemDef {
76                name: String::from(p.name),
77                path: String::from(p.path),
78                icon: p.icon,
79            })
80            .collect();
81        PlacesSidebar::render(
82            surf,
83            fonts,
84            theme,
85            0,
86            content.y,
87            PLACES_WIDTH,
88            content.h,
89            "PLACES",
90            &places_defs,
91            &self.current_path,
92        );
93
94        let (left, right) = self.pane_views();
95        match (right, layout::pane_frames(sw, sh, right.is_some())) {
96            (Some(right), (left_frame, Some(right_frame))) => {
97                render_pane(surf, fonts, theme, self.view_mode, &left, left_frame);
98                let divider_x = left_frame.x + left_frame.w as i32;
99                let divider_w = (right_frame.x - divider_x) as u32;
100                fill_rect(
101                    surf,
102                    divider_x,
103                    content.y,
104                    divider_w,
105                    content.h,
106                    theme.border(),
107                );
108                render_pane(surf, fonts, theme, self.view_mode, &right, right_frame);
109            }
110            _ => render_listing(
111                surf,
112                fonts,
113                theme,
114                self.view_mode,
115                left.items,
116                left.selected_index,
117                content,
118            ),
119        }
120
121        let status_y = sh as i32 - STATUS_HEIGHT as i32;
122        StatusBar::render(
123            surf,
124            fonts,
125            theme,
126            status_y,
127            sw,
128            STATUS_HEIGHT,
129            &self.status_message,
130        );
131        preview::render(surf, fonts, theme, &self.quick_look, sw, sh);
132        surf.damage_all();
133    }
134}

149/// One split pane: a header strip naming its folder (accent-filled when focused), its
150/// listing below, and an accent outline around the focused pane.
151fn render_pane(
152    surf: &mut ViSurface,
153    fonts: &mut TextFonts,
154    theme: &dyn ViTheme,
155    mode: ViewMode,
156    pane: &PaneView<'_>,
157    frame: ContentRect,
158) {
159    let (header_bg, header_fg) = if pane.focused {
160        (theme.accent(), Color::WHITE)
161    } else {
162        (theme.surface(), theme.text_secondary())
163    };
164    fill_rect(
165        surf,
166        frame.x,
167        frame.y,
168        frame.w,
169        PANE_HEADER_HEIGHT,
170        header_bg,
171    );
172    draw_label(
173        surf,
174        fonts,
175        frame.x + 10,
176        frame.y + 3,
177        frame.w.saturating_sub(20),
178        18,
179        pane.path,
180        FontFace::UiSemibold,
181        13.0,
182        header_fg,
183    );
184    let listing = layout::listing_rect(frame, true);
185    render_listing(
186        surf,
187        fonts,
188        theme,
189        mode,
190        pane.items,
191        pane.selected_index,
192        listing,
193    );
194    if pane.focused {
195        stroke_rect(surf, frame.x, frame.y, frame.w, frame.h, theme.accent());
196    }
197}

199/// The entries of one folder in the current view mode.
200fn render_listing(
201    surf: &mut ViSurface,
202    fonts: &mut TextFonts,
203    theme: &dyn ViTheme,
204    mode: ViewMode,
205    items: &[FileItem],
206    selected: usize,
207    rect: ContentRect,
208) {
209    match mode {
210        ViewMode::List => {
211            let defs: Vec<ListViewItemDef> = items
212                .iter()
213                .map(|item| ListViewItemDef {
214                    icon: item.icon,
215                    name: item.name.clone(),
216                    size_str: item.formatted_size(),
217                    kind_str: String::from(if item.is_dir { "Folder" } else { "File" }),
218                    is_dir: item.is_dir,
219                })
220                .collect();
221            ListView::render(
222                surf,
223                fonts,
224                theme,
225                rect.x,
226                rect.y,
227                rect.w,
228                rect.h,
229                &defs,
230                Some(selected),
231            );
232        }
233        ViewMode::Grid => {
234            let defs: Vec<GridItemDef> = items
235                .iter()
236                .map(|item| GridItemDef {
237                    icon: item.icon,
238                    name: item.name.clone(),
239                    is_dir: item.is_dir,
240                })
241                .collect();
242            GridView::render(
243                surf,
244                fonts,
245                theme,
246                rect.x,
247                rect.y,
248                rect.w,
249                rect.h,
250                &defs,
251                Some(selected),
252            );
253        }
254        ViewMode::Columns => views::columns::render(surf, fonts, items, selected, rect, theme),
255    }
