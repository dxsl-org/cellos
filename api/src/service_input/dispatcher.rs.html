1//! Focus-based input event dispatcher.
2//!
3//! Maintains a "focused cell" ID and routes translated `InputEvent`s to it
4//! via IPC Send.  Focus changes clear transient modifiers to avoid stuck keys.
5//!
6//! ## Focus default
7//!
8//! `focused` starts at 0 (no focus).  The first cell to call `SetFocus` owns
9//! the keyboard.  Events before any focus claim are silently dropped — this is
10//! preferable to the previous TID-3 shell fallback which consumed events without
11//! the shell ever reading them (shell uses sys_read(0), not the input service).
12//!
13//! ## Send-failure handling
14//!
15//! Keyboard dispatch never blocks a slow focused consumer. Pointer dispatch
16//! stashes a full compositor queue's events, then the main loop retries before
17//! consuming more raw input. No synchronous Send/SetFocus IPC cycle is created.
18
19use api::input::{encode_event, InputEvent, INPUT_EVENT_IPC_SIZE};
20use api::syscall::service;
21use ostd::syscall::{sys_lookup_service, sys_try_send};
22use ostd::session::{zeroize, Scrub};
23
24/// Report every Nth dropped keyboard event, so a stalled consumer is visible
25/// without one log line per keystroke.
26const DROP_REPORT_EVERY: u64 = 64;
27
28/// Opcode prefix byte sent to the focused cell's IPC endpoint.
29pub const INPUT_EVENT_OPCODE: u8 = 0x10;
30
31/// Routes translated events to the currently focused cell.
32pub struct Dispatcher {
33    /// Task ID of the currently focused cell (0 = no focus, events dropped).
34    focused: usize,
35    /// Keyboard events dropped because the focused cell was not draining. The
36    /// first drop and every `DROP_REPORT_EVERY`-th after it are logged.
37    dropped: u64,
38    /// Reserved for a future focus-fallback policy; `dispatch()` does not read it.
39    // reason: keyboard send failure currently leaves `focused` unchanged until a
40    // later SetFocus. Keep the field only as a placeholder for a separately
41    // specified policy, not as behaviour implemented by this dispatcher.
42    #[allow(dead_code)]
43    fallback_tid: usize,
44    /// Cached compositor TID for mouse routing (0 = not yet resolved).
45    /// Re-resolved lazily; reset to 0 when a send fails (compositor respawned).
46    compositor_tid: usize,
47    // One raw input event emits one pointer event, except HID disconnect,
48    // which can release all three recognised mouse buttons in one batch.
49    pending_mouse: [Option<InputEvent>; 3],
50    mouse_head: usize,
51    mouse_len: usize,
52    pub(crate) repeat: crate::key_repeat::KeyRepeat,
53    pub(crate) sessions: crate::session::Sessions,
54}
55
56impl Dispatcher {
57    /// Create a dispatcher with no initial focus (events dropped until SetFocus).
58    pub fn new() -> Self {
59        Self {
60            focused: 0,
61            dropped: 0,
62            fallback_tid: 0,
63            compositor_tid: 0,
64            pending_mouse: [None; 3],
65            mouse_head: 0,
66            mouse_len: 0,
67            repeat: crate::key_repeat::KeyRepeat::default(),
68            sessions: crate::session::Sessions::new(),
69        }
70    }
71
72    /// Change which cell receives input events.
73    ///
74    /// Also resets transient modifiers on the modifier-state tracker so that
75    /// Shift/Ctrl/Alt do not appear "stuck" after a focus change.
76    pub fn set_focus(&mut self, cell_endpoint: usize) {
77        if self.focused != cell_endpoint {
78            self.repeat.clear();
79        }
80        self.focused = cell_endpoint;
81    }
82
83    /// Return the currently focused endpoint.
84    #[allow(dead_code)]
85    pub fn focus(&self) -> usize {
86        self.focused
87    }
88
89    /// SessionV1 owns physical keyboard input exclusively while attached, but never while a
90    /// compositor is registered: `Sessions::physical_active` is false then, so neither this
91    /// relay nor `dispatch` can route keys into a session and legacy focus stays authoritative.
92    pub fn serial_byte(&mut self, byte: u8) -> bool {
93        self.sessions.serial_byte(byte)
94    }
95
96    /// Send a translated `InputEvent` to the focused cell.
97    ///
98    /// **Non-blocking, and that is the fix for a board failure, not a
99    /// preference.** Delivery used to block until the focused cell received the
100    /// event, which applies backpressure from a slow consumer to *everything
101    /// behind it*: while a Tier-3 guest held focus and was busy, the guest's VMM
102    /// polled input only on its WFI/preemption exits, so the input service
103    /// stayed blocked in this send, the USB cell stayed blocked feeding it, and
104    /// the network cell stayed blocked sending through the USB cell — long
105    /// enough that the kernel's liveness sweep killed net as hung
106    /// (`task 7 … missed liveness deadline`, 2026-10-04). A keyboard is a
107    /// lossy source at the consumer's pace; the events that do not fit are
108    /// dropped and counted (`dropped`), never silently and never by stalling
109    /// the cells behind them.
110    ///
111    /// The IPC message format is:
112    /// ```text
113    /// byte[0]   = INPUT_EVENT_OPCODE (0x10)
114    /// byte[1..] = encode_event() output (see libs/api/src/input.rs)
115    /// ```
116    pub fn dispatch(&mut self, event: &InputEvent) -> bool {
117        if let InputEvent::Key(key) = event {
118            if self.sessions.keyboard(key) {
119                return true;
120            }
121        }
122        if self.focused == 0 {
123            return true; // no focus — drop silently
124        }
125        if Self::try_send_event(self.focused, event).is_ok() {
126            return true;
127        }
128        // The focused cell's queue is full: it is not draining. Drop this event
129        // (and the ones that follow until it does) and say so periodically.
130        self.dropped = self.dropped.wrapping_add(1);
131        if self.dropped % DROP_REPORT_EVERY == 1 {
132            ostd::io::print("[input] dropped ");
133            ostd::io::print_usize(self.dropped as usize);
134            ostd::io::println(" keyboard event(s): focused cell is not draining");
135        }
136        false
137    }
138
139    /// Send a mouse event (`MouseMove`/`MouseButton`/`MouseScroll`) to the
140    /// compositor rather than the keyboard-focused cell.
141    ///
142    /// The compositor owns the cursor and the surface Z-order, so it is the
143    /// only correct recipient for pointer events: it repaints/moves the cursor
144    /// and hit-tests button clicks to the surface under the pointer. Routing
145    /// mouse through the keyboard focus (the historical behaviour) breaks as
146    /// soon as a non-GUI cell like the shell holds focus — the events land in
147    /// a cell that ignores them and the cursor never moves.
148    ///
149    /// A full compositor queue must not discard movement/button/wheel events.
150    /// Stash the current raw-event batch; the main loop flushes it before reading
151    /// another batch. Retry is non-blocking, so endpoint death/restart is checked
152    /// on every scheduler turn and the outer watchdog heartbeat stays live.
153    pub fn dispatch_mouse(&mut self, event: &InputEvent) {
154        if self.mouse_len == 0 {
155            self.compositor_tid = match sys_lookup_service(service::COMPOSITOR) {
156                Some(tid) => tid,
157                None => return, // no graphical consumer yet
158            };
159            if Self::try_send_event(self.compositor_tid, event).is_ok() {
160                return;
161            }
162        }
163        // Main stops consuming raw events as soon as this batch is stashed.
164        // The only multi-event batch is a HID disconnect (three mouse buttons).
165        assert!(self.mouse_len < self.pending_mouse.len());
166        let tail = (self.mouse_head + self.mouse_len) % self.pending_mouse.len();
167        self.pending_mouse[tail] = Some(*event);
168        self.mouse_len += 1;
169    }
170
171    pub fn has_pending_mouse(&self) -> bool {
172        self.mouse_len != 0
173    }
174
175    /// Retry in FIFO order without waiting inside the dispatcher. Returning
176    /// false applies backpressure to raw sources until the next scheduler turn.
177    pub fn flush_mouse(&mut self) -> bool {
178        if self.mouse_len == 0 {
179            return true;
180        }
181        self.compositor_tid = match sys_lookup_service(service::COMPOSITOR) {
182            Some(tid) => tid,
183            None => {
184                // A dead compositor has no windows to deliver these events to.
185                // Do not prevent keyboard/console input during graphical restart.
186                self.compositor_tid = 0;
187                self.pending_mouse = [None; 3];
188                self.mouse_head = 0;
189                self.mouse_len = 0;
190                return true;
191            }
192        };
193        while self.mouse_len != 0 {
194            let event = self.pending_mouse[self.mouse_head].as_ref().unwrap();
195            if Self::try_send_event(self.compositor_tid, event).is_err() {
196                return false;
197            }
198            self.pending_mouse[self.mouse_head] = None;
199            self.mouse_head = (self.mouse_head + 1) % self.pending_mouse.len();
200            self.mouse_len -= 1;
201        }
202        true
203    }
204
205    /// Encode and try-send one mouse event to `target`.
206    ///
207    /// The IPC message format is:
208    /// ```text
209    /// byte[0]   = INPUT_EVENT_OPCODE (0x10)
210    /// byte[1..] = encode_event() output (see libs/api/src/input.rs)
211    /// ```
212    fn try_send_event(target: usize, event: &InputEvent) -> Result<(), ()> {
213        // A focused-cell key event can be a typed secret: both staging buffers are wiped after the send.
214        let mut payload = [0u8; INPUT_EVENT_IPC_SIZE];
215        encode_event(event, &mut payload);
216        let mut buf = Scrub::<{ INPUT_EVENT_IPC_SIZE + 1 }>::new();
217        buf[0] = INPUT_EVENT_OPCODE;
218        buf[1..].copy_from_slice(&payload);
219        zeroize(&mut payload);
220        match sys_try_send(target, &buf) {
221            ostd::syscall::SyscallResult::Ok(0) => Ok(()),
222            _ => Err(()),
223        }
224    }
225}

227impl Default for Dispatcher {
228    fn default() -> Self {
229        Self::new()
230    }
