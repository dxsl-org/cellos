9//! Input Service Cell.
10//!
11//! Receives raw EV_KEY events from the kernel VirtIO input driver via IPC,
12//! translates scancodes to `InputEvent`s using the US QWERTY layout, and
13//! dispatches them to the currently focused cell.
14//!
15//! ## IPC protocol (inbound from kernel, sender == 0)
16//! ```text
17//! byte[0]   = event type: 0=EV_KEY, 1=EV_REL, 2=EV_ABS
18//! byte[1..5]= code  (u32 LE: scancode, REL_*, ABS_* axis)
19//! byte[5..9]= value (u32 LE: key state, signed rel delta, abs coord)
20//! ```
21//! Sender 0 is the kernel; these raw frames bypass postcard decoding entirely.
22//!
23//! ## Focus IPC (inbound from compositor/shell, sender > 0)
24//! Typed `InputRequest` encoded with postcard — see `api::ipc::InputRequest`.
25//! Sender > 0 always routes to postcard decode; opcode collisions with kernel
26//! frames are impossible by construction.
27//!
28//! ## IPC protocol (outbound to focused cell)
29//! See `dispatcher::Dispatcher::dispatch` and `api::input::encode_event`.
30

68/// Raw event type discriminant for keyboard events (kernel VirtIO push).
69const EV_KEY: u8 = 0;
70/// Raw event type for UART ASCII relay from the kernel console driver.
71/// The code field carries the raw ASCII byte; no scancode translation needed.
72const EV_ASCII: u8 = 0x04;
73/// A per-interface HID worker event carrying a logical-device ID.
74///
75/// Layout: `[EV_DEVICE][device:u32][event_type][code:u32][value:i32]`.
76const EV_DEVICE: u8 = 0x05;
77
78/// Linux evdev event types used by VirtIO input device.
79const EVDEV_KEY: u16 = 1;
80const EVDEV_REL: u16 = 2;
81const EVDEV_ABS: u16 = 3;

83/// Poll the VirtIO virtqueue once per scheduler tick (≈10 ms).
84/// Using scheduler-tick units (not mtime); see net service UNIT TRAP note.
85const POLL_SCHED_TICKS: u64 = 1;
86
87/// Watchdog interval: 5000 scheduler ticks × 10ms = 50 seconds.
88/// Must match DISPATCH_HEARTBEAT in dispatcher.rs — they share the same timeline.
89const HEARTBEAT_TICKS: u64 = 5_000;
90
91/// Input Cell entry point.
92///
93/// On startup, attempts to probe and claim a VirtIO input device. Once claimed,
94/// the kernel's `virtio_input::poll_events` / `dispatch_pending` migrate guard
95/// detects the MMIO owner and stops pushing kernel-side events — this service
96/// then owns the virtqueue exclusively.
97///
98/// Until the device is claimed (or if no VirtIO input is present), the kernel
99/// continues to push raw IPC events (sender=0) as before.
100#[cfg(not(test))]
101#[no_mangle]
102pub fn main() {
103    println("[input] Input Service v0.3: US QWERTY + VirtIO + typed focus routing");
104
105    let mut modifiers = ModifierState::new();
106    let mut mouse = MouseState::new();
107    let mut dispatcher = Dispatcher::new();
108    let mut buf = [0u8; IPC_BUF_SIZE];
109    // Last LED bitmap handed to the producers, so a lock key that is pressed and
110    // released without changing a lock costs nothing.
111    let mut leds_sent = modifiers.led_bitmap();
112
113    // Renew the watchdog before the potentially slow VirtIO MMIO probe — the
114    // cell is spawned with a default deadline and find_and_init_input can take
115    // several scheduling cycles before we reach the loop's sys_heartbeat call.
116    sys_heartbeat(HEARTBEAT_TICKS);
117
118    // Probe and claim ALL VirtIO input devices (QEMU exposes keyboard, tablet,
119    // mouse as separate virtio-input MMIO slots).  After sys_request_mmio
120    // succeeds inside find_and_init_inputs, the kernel migration guard in
121    // virtio_input.rs sees the MMIO owner and stops pushing events via
122    // dispatch_pending — so every unclaimed device would be polled by nobody.
123    let mut sources = EventSources::new();
124    let mut hid_devices = HidDeviceStates::new();
125    let mut devices: Vec<InputDevice> = find_and_init_inputs();
126    if !devices.is_empty() {
127        println("[input] VirtIO input device claimed; polling virtqueue directly");
128    } else {
129        println("[input] No VirtIO input device; relying on kernel push");
130    }
131
132    loop {
133        sys_heartbeat(HEARTBEAT_TICKS);
134
135        // Drain every VirtIO virtqueue before blocking.  This catches events
136        // that arrived since the last iteration without waiting for the timeout.
137        for dev in devices.iter_mut() {
138            drain_virtio(
139                dev,
140                &mut buf,
141                &mut modifiers,
142                &mut mouse,
143                &mut dispatcher,
144                &mut hid_devices,
145            );
146        }
147
148        // Block for at most one scheduler tick (≈10ms), or until a kernel/IPC
149        // message wakes us.
150        //
151        // Return value convention:
152        //   Ok(0)                — real timeout; buffer not modified
153        //   Ok(isize::MAX as _)  — kernel UART relay (EV_ASCII); buffer filled
154        //   Ok(n)                — typed IPC from cell n; buffer filled
155        //
156        // The sentinel is isize::MAX (not usize::MAX) because syscall() returns
157        // isize: usize::MAX == -1 as isize, which makes sys_recv_timeout return
158        // Err and the match arm never fires. isize::MAX is positive → Ok branch.
159        match sys_recv_timeout(0, &mut buf, POLL_SCHED_TICKS) {
160            SyscallResult::Ok(0) => {
161                // Real timeout — nothing from IPC; VirtIO drain already done above.
162            }
163            SyscallResult::Ok(n) if n == isize::MAX as usize => {
164                // Kernel UART relay (sentinel sender_id = isize::MAX as usize).
165                handle_kernel_event(
166                    &buf,
167                    &mut modifiers,
168                    &mut mouse,
169                    &mut dispatcher,
170                    &mut hid_devices,
171                );
172            }
173            SyscallResult::Ok(sender) => {
174                handle_message(
175                    &buf,
176                    sender,
177                    &mut modifiers,
178                    &mut mouse,
179                    &mut dispatcher,
180                    &mut sources,
181                    &mut hid_devices,
182                );
183            }
184            _ => {}
185        }
186
187        // Lock keys change what the keyboard's own LEDs show, and the device is
188        // the only place a user can see that state. Producers hear about it once
189        // per change rather than once per key, and a producer that comes up
190        // later is told at registration.
191        let leds = modifiers.led_bitmap();
192        if leds != leds_sent {
193            sources.broadcast_leds(leds);
194            leds_sent = leds;
195        }
196    }
197}

199/// Registered raw-event producers. The DWC2 host is the sole USB producer:
200/// isolated HID workers return decoded frames to that host instead of acquiring
201/// a second input-service identity.
202const MAX_EVENT_SOURCES: usize = 16;
203
204/// TIDs allowed to push raw `[opcode][code][value]` events.
205///
206/// Sender 0 is the kernel and is implicit. For DWC2, Input accepts only the
207/// kernel's current NIC-driver endpoint. The recorded identity always comes
208/// from receive metadata, never from a payload.
209struct EventSources {
210    tids: [usize; MAX_EVENT_SOURCES],
211    kinds: [u8; MAX_EVENT_SOURCES],
212    len: usize,
213}
214
215impl EventSources {
216    const fn new() -> Self {
217        Self {
218            tids: [0; MAX_EVENT_SOURCES],
219            kinds: [0; MAX_EVENT_SOURCES],
220            len: 0,
221        }
222    }
223
224    fn is_source(&self, tid: usize) -> bool {
225        tid != 0 && self.tids[..self.len].contains(&tid)
226    }
227
228    fn is_hid_host(&self, tid: usize) -> bool {
229        self.tids[..self.len]
230            .iter()
231            .zip(self.kinds[..self.len].iter())
232            .any(|(&source, &kind)| source == tid && kind == api::ipc::input_source::USB_HID_HOST)
233    }
234
235    /// Record the kernel-verified DWC2 host as the USB producer.
236    fn register(&mut self, tid: usize, kind: u8) -> bool {
237        if let Some(index) = self.tids[..self.len]
238            .iter()
239            .position(|&source| source == tid)
240        {
241            return self.kinds[index] == kind;
242        }
243        if tid == 0
244            || kind != api::ipc::input_source::USB_HID_HOST
245            || sys_lookup_service(service::NIC_DRIVER) != Some(tid)
246            || self.len == MAX_EVENT_SOURCES
247        {
248            return false;
249        }
250        self.tids[self.len] = tid;
251        self.kinds[self.len] = kind;
252        self.len += 1;
253        true
254    }
255
256    /// Tell one producer what the lock LEDs should show.
257    fn send_leds(&self, tid: usize, leds: u8) {
258        let frame = [api::ipc::OP_SET_LEDS, leds];
259        if !matches!(sys_try_send(tid, &frame), SyscallResult::Ok(0)) {
260            print("[input] WARN: lock LED state not delivered to source tid=");
261            print_usize(tid);
262            println("");
263        }
264    }
265
266    /// Tell every registered producer what the lock LEDs should show.
267    fn broadcast_leds(&self, leds: u8) {
268        for &tid in &self.tids[..self.len] {
269            self.send_leds(tid, leds);
270        }
271    }
272}

274/// Pressed-key ownership for isolated HID interfaces.
275///
276/// This is deliberately bounded: an untrusted descriptor/report cannot grow
277/// input-service state without limit, and disconnect cleanup only touches the
278/// keys that one device actually contributed.
279const MAX_HID_DEVICES: usize = 16;
280const MAX_KEYS_PER_HID_DEVICE: usize = 128;
281
282struct HidDeviceState {
283    id: u32,
284    pressed: Vec<u32>,
285}
286
287struct HidDeviceStates {
288    devices: Vec<HidDeviceState>,
289}
290
291impl HidDeviceStates {
292    const fn new() -> Self {
293        Self {
294            devices: Vec::new(),
295        }
296    }
297
298    fn record_key(&mut self, id: u32, code: u32, state: KeyState) {
299        let index = match self.devices.iter().position(|device| device.id == id) {
300            Some(index) => index,
301            None if self.devices.len() < MAX_HID_DEVICES => {
302                self.devices.push(HidDeviceState {
303                    id,
304                    pressed: Vec::new(),
305                });
306                self.devices.len() - 1
307            }
308            None => return,
309        };
310        let pressed = &mut self.devices[index].pressed;
311        match state {
312            KeyState::Pressed | KeyState::Repeated => {
313                if !pressed.contains(&code) && pressed.len() < MAX_KEYS_PER_HID_DEVICE {
314                    pressed.push(code);
315                }
316            }
317            KeyState::Released => pressed.retain(|&held| held != code),
318        }
319    }
320
321    fn take_pressed(&mut self, id: u32) -> Vec<u32> {
322        let Some(index) = self.devices.iter().position(|device| device.id == id) else {
323            return Vec::new();
324        };
325        self.devices.swap_remove(index).pressed
326    }
327
328    fn has_transient_modifier(&self, code: u32) -> bool {
329        let Some(modifier) = modifier_for_scancode(code) else {
330            return false;
331        };
332        self.devices.iter().any(|device| {
333            device
334                .pressed
335                .iter()
336                .copied()
337                .any(|held| modifier_for_scancode(held) == Some(modifier))
338        })
339    }
340}

342fn release_hid_device(
343    device: u32,
344    modifiers: &mut ModifierState,
345    mouse: &mut MouseState,
346    dispatcher: &mut Dispatcher,
347    hid_devices: &mut HidDeviceStates,
348) {
349    for code in hid_devices.take_pressed(device) {
350        // A second keyboard can keep the same logical modifier held (including
351        // left/right variants). Do not clear its aggregate state on this
352        // worker's teardown.
353        if hid_devices.has_transient_modifier(code) {
354            continue;
355        }
356        let mut release = [0u8; IPC_BUF_SIZE];
357        release[0] = EV_KEY;
358        release[1..5].copy_from_slice(&code.to_le_bytes());
359        // value remains zero: EV_KEY release.
360        handle_kernel_event(&release, modifiers, mouse, dispatcher, hid_devices);
361    }
362}

364/// Drain all pending events from the VirtIO virtqueue and dispatch them.
365fn drain_virtio(
366    dev: &mut InputDevice,
367    buf: &mut [u8; IPC_BUF_SIZE],
368    modifiers: &mut ModifierState,
369    mouse: &mut MouseState,
370    dispatcher: &mut Dispatcher,
371    hid_devices: &mut HidDeviceStates,
372) {
373    while let Some(ev) = dev.try_get_event() {
374        // Map Linux evdev event types → the same opcode encoding the kernel uses
375        // in dispatch_pending, so handle_kernel_event processes them identically.
376        let opcode: u8 = match ev.event_type {
377            EVDEV_KEY => 0, // EV_KEY
378            EVDEV_REL => 1, // EV_REL
379            EVDEV_ABS => 2, // EV_ABS
380            _ => continue,  // unknown type (EV_SYN, EV_MSC, etc.) — drop
381        };
382        buf[0] = opcode;
383        buf[1..5].copy_from_slice(&(ev.code as u32).to_le_bytes());
384        buf[5..9].copy_from_slice(&ev.value.to_le_bytes());
385        handle_kernel_event(buf, modifiers, mouse, dispatcher, hid_devices);
386    }
387}

389/// Process one incoming IPC message.
390///
391/// Discrimination is by `sender`, not opcode, to avoid collisions with postcard
392/// discriminants: kernel pushes arrive with sender=0; typed requests sender>0.
393fn handle_message(
394    buf: &[u8; IPC_BUF_SIZE],
395    sender: usize,
396    modifiers: &mut ModifierState,
397    mouse: &mut MouseState,
398    dispatcher: &mut Dispatcher,
399    sources: &mut EventSources,
400    hid_devices: &mut HidDeviceStates,
401) {
402    // Registered producers speak raw frames; all other senders are typed IPC.
403    if sender == 0 {
404        handle_kernel_event(buf, modifiers, mouse, dispatcher, hid_devices);
405    } else if sources.is_source(sender) {
406        if buf[0] == OP_HID_DEVICE_REMOVED && sources.is_hid_host(sender) {
407            let device = u32::from_le_bytes([buf[1], buf[2], buf[3], buf[4]]);
408            release_hid_device(device, modifiers, mouse, dispatcher, hid_devices);
409            return;
410        }
411        handle_kernel_event(buf, modifiers, mouse, dispatcher, hid_devices);
412    } else {
413        handle_typed_request(buf, sender, modifiers, dispatcher, sources);
414    }
415}

417/// Handle a raw VirtIO event pushed by the kernel (sender == 0).
418///
419/// Wire format: `[opcode:1][code:4 LE][value:4 LE]`
420/// opcode 0 = EV_KEY (keyboard key or mouse button via BTN_* scancode ≥ 0x110)
421/// opcode 1 = EV_REL (relative mouse: REL_X/Y/WHEEL)
422/// opcode 2 = EV_ABS (absolute mouse: ABS_X/Y)
423fn handle_kernel_event(
424    buf: &[u8; IPC_BUF_SIZE],
425    modifiers: &mut ModifierState,
426    mouse: &mut MouseState,
427    dispatcher: &mut Dispatcher,
428    hid_devices: &mut HidDeviceStates,
429) {
430    let (device, opcode, code, value) = if buf[0] == EV_DEVICE {
431        if buf.len() < 14 {
432            return;
433        }
434        (
435            Some(u32::from_le_bytes([buf[1], buf[2], buf[3], buf[4]])),
436            buf[5],
437            u32::from_le_bytes([buf[6], buf[7], buf[8], buf[9]]),
438            u32::from_le_bytes([buf[10], buf[11], buf[12], buf[13]]),
439        )
440    } else {
441        if buf.len() < 9 {
442            return;
443        }
444        (
445            None,
446            buf[0],
447            u32::from_le_bytes([buf[1], buf[2], buf[3], buf[4]]),
448            u32::from_le_bytes([buf[5], buf[6], buf[7], buf[8]]),
449        )
450    };
451
452    match opcode {
453        EV_KEY => {
454            let state = key_state_from_evdev(value);
455            if let Some(device) = device {
456                hid_devices.record_key(device, code, state);
457            }
458            // BTN_* codes (≥ 0x110) are mouse buttons, not keyboard keys.
459            // Pointer events route to the compositor (cursor + Z-order owner),
460            // not the keyboard-focused cell — see Dispatcher::dispatch_mouse.
461            if code >= BTN_LEFT {
462                if let Some(button) = btn_to_mouse_button(code) {
463                    dispatcher.dispatch_mouse(&InputEvent::MouseButton { button, state });
464                }
465                return;
466            }
467            if modifiers.update(code, state) {
468                return;
469            }
470            let (keysym, character) = translate(code, modifiers.snapshot());
471            if !dispatcher.dispatch(&InputEvent::Key(KeyEvent {
472                timestamp_ticks: sys_get_time(),
473                scancode: code,
474                keysym,
475                character,
476                modifiers: modifiers.snapshot(),
477                state,
478                _pad: [0; 2],
479            })) {
480                println("[input] WARNING: clearing failed keyboard focus");
481            }
482        }
483        1 => {
484            if let Some(ev) = mouse.apply_rel(code, value) {
485                dispatcher.dispatch_mouse(&ev);
486            }
487        }
488        2 => {
489            if let Some(ev) = mouse.apply_abs(code, value) {
490                dispatcher.dispatch_mouse(&ev);
491            }
492        }
493        EV_ASCII => {
494            // UART byte relayed by the kernel console driver.
495            // `code` carries the raw ASCII code point; skip scancode translation.
496            // Map C0 control chars to semantic KeySyms so GUI apps get proper events
497            // regardless of whether input originates from VirtIO or UART terminal.
498            let state = if value > 0 {
499                KeyState::Pressed
500            } else {
501                KeyState::Released
502            };
503            let (keysym, character) = match code {
504                0x1B => (KeySym::Escape, 0),
505                0x0D | 0x0A => (KeySym::Return, code),
506                0x08 | 0x7F => (KeySym::Backspace, code),
507                0x09 => (KeySym::Tab, code),
508                _ => (KeySym::Printable, code),
509            };
510            if !dispatcher.dispatch(&InputEvent::Key(KeyEvent {
511                timestamp_ticks: sys_get_time(),
512                scancode: 0,
513                keysym,
514                character,
515                modifiers: modifiers.snapshot(),
516                state,
517                _pad: [0; 2],
518            })) {
519                println("[input] WARNING: clearing failed keyboard focus");
520            }
521        }
522        _ => {} // unknown opcode — drop silently
523    }
524}

526/// Handle a typed `InputRequest` from a compositor or shell cell (sender > 0).
527fn handle_typed_request(
528    buf: &[u8; IPC_BUF_SIZE],
529    sender: usize,
530    modifiers: &mut ModifierState,
531    dispatcher: &mut Dispatcher,
532    sources: &mut EventSources,
533) {
534    let mut resp_buf = [0u8; 64];
535    match api::ipc::decode::<InputRequest>(buf) {
536        Ok(InputRequest::SetFocus { cell_tid: _ }) => {
537            modifiers.reset_transient();
538            // Use kernel-verified sender TID instead of the cell_tid field to
539            // prevent a cell from redirecting focus to an arbitrary TID.
540            dispatcher.set_focus(sender);
541            // Fire-and-forget: no reply. Focus is set atomically on receipt.
542            // A blocking reply would deadlock when the focused cell is not yet
543            // in sys_recv (startup race — G18 deadlock fix).
544        }
545        Ok(InputRequest::GetFocus) => {
546            let focused = dispatcher.focus() as u32;
547            if let Ok(encoded) = api::ipc::encode(&InputResponse::Focus(focused), &mut resp_buf) {
548                // GetFocus is only called by compositor (never during startup race).
549                // Use sys_try_send to be safe — compositor is in recv waiting for this.
550                let _ = sys_try_send(sender, encoded);
551            }
552        }
553        Ok(InputRequest::ClearFocus { cell_tid: _ }) => {
554            // Use kernel-verified sender TID (same as SetFocus) — prevents a cell
555            // from clearing another cell's focus. When sender == focused, drop focus.
556            if dispatcher.focus() == sender {
557                dispatcher.set_focus(0);
558            }
559            // Fire-and-forget: no reply. Same rationale as SetFocus.
560        }
561        Ok(InputRequest::RegisterEventSource { kind }) => {
562            if sources.register(sender, kind) {
563                print("[input] registered raw event source kind=");
564                print_usize(kind as usize);
565                println("");
566                // A producer that just came up does not know what the locks are
567                // set to, and its keyboard would otherwise show stale state.
568                sources.send_leds(sender, modifiers.led_bitmap());
569            } else {
570                println("[input] WARN: event-source registration refused");
571            }
572        }
573        Err(_) => {} // unknown message — drop silently
574    }
575}
