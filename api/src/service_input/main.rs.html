9//! Input Service Cell.
10//!
11//! Receives raw EV_KEY events from the kernel VirtIO input driver via IPC,
12//! translates scancodes to `InputEvent`s using the US QWERTY layout, and
13//! dispatches them to the currently focused cell.
14//!
15//! ## IPC protocol (inbound from kernel, sender == 0)
16//! Producer frames, wire in `wire::proto::input`:
17//! ```text
18//! byte[0]   = event kind: EV_KEY, EV_REL, EV_ABS, EV_ASCII (or an EV_DEVICE frame)
19//! byte[1..5]= code  (u32 LE: scancode, REL_*, ABS_* axis)
20//! byte[5..9]= value (u32 LE: key state, signed rel delta, abs coord)
21//! ```
22//! Sender 0 is the kernel; these raw frames bypass postcard decoding entirely.
23//!
24//! ## Focus IPC (inbound from compositor/shell, sender > 0)
25//! Typed `InputRequest` encoded with postcard — see `wire::ipc::InputRequest`.
26//! Sender > 0 always routes to postcard decode; opcode collisions with kernel
27//! frames are impossible by construction.
28//!
29//! ## IPC protocol (outbound to focused cell)
30//! See `dispatcher::Dispatcher::dispatch` and `wire::input::encode_event`.
31

86/// Watchdog interval: 5000 scheduler ticks × 10ms = 50 seconds.
87/// Must match DISPATCH_HEARTBEAT in dispatcher.rs — they share the same timeline.
88const HEARTBEAT_TICKS: u64 = 5_000;
89
90/// Input Cell entry point.
91///
92/// On startup, attempts to probe and claim a VirtIO input device. Once claimed,
93/// the kernel's `virtio_input::poll_events` / `dispatch_pending` migrate guard
94/// detects the MMIO owner and stops pushing kernel-side events — this service
95/// then owns the virtqueue exclusively.
96///
97/// Until the device is claimed (or if no VirtIO input is present), the kernel
98/// continues to push raw IPC events (sender=0) as before.
99#[cfg(not(test))]
100#[no_mangle]
101pub fn main() {
102    println("[input] Input Service v0.3: US QWERTY + VirtIO + typed focus routing");
103
104    let mut modifiers = ModifierState::new();
105    let (width, height) = sys_get_resolution();
106    let mut mouse = MouseState::new(width, height);
107    let mut dispatcher = Dispatcher::new();
108    let mut buf = [0u8; IPC_BUF_SIZE];
109    // Last LED bitmap handed to the producers, so a lock key that is pressed and
110    // released without changing a lock costs nothing.
111    let mut leds_sent = modifiers.led_bitmap();
112
113    // Renew the watchdog before the potentially slow VirtIO MMIO probe — the
114    // cell is spawned with a default deadline and find_and_init_input can take
115    // several scheduling cycles before we reach the loop's sys_heartbeat call.
116    sys_heartbeat(HEARTBEAT_TICKS);
117
118    // Probe and claim ALL VirtIO input devices (QEMU exposes keyboard, tablet,
119    // mouse as separate virtio-input MMIO slots).  After sys_request_mmio
120    // succeeds inside find_and_init_inputs, the kernel migration guard in
121    // virtio_input.rs sees the MMIO owner and stops pushing events via
122    // dispatch_pending — so every unclaimed device would be polled by nobody.
123    let mut sources = EventSources::new();
124    let mut hid_devices = HidDeviceStates::new();
125    let mut devices: Vec<InputDevice> = find_and_init_inputs();
126    if !devices.is_empty() {
127        println("[input] VirtIO input device claimed; polling virtqueue directly");
128    } else {
129        println("[input] No VirtIO input device; relying on kernel push");
130    }
131
132    loop {
133        sys_heartbeat(HEARTBEAT_TICKS);
134        dispatcher.sessions.maintain();
135        if !dispatcher.flush_mouse() {
136            // Compositor queue is full: retry on the next tick instead of spinning.
137            idle_wait();
138            continue;
139        }
140        // Set whenever this turn did any work; only a fully idle turn sleeps.
141        let mut busy = false;
142
143        // Nonblocking attested receive keeps ring backpressure, deferred polls,
144        // UART ESC deadlines and physical-device polling on the same owner loop.
145        // The kernel overwrites the caller trailer; clear it before each attempt
146        // so an empty receive can never reuse a previous principal.
147        buf[IPC_BUF_SIZE - abi::caller_identity::CALLER_IDENTITY_LEN..].fill(0);
148        match sys_try_recv_attested(0, &mut buf) {
149            SyscallResult::Ok(0) => {
150                // Real timeout — no IPC input.
151            }
152            SyscallResult::Ok(n) if n == isize::MAX as usize => {
153                busy = true;
154                // Kernel UART relay (sentinel sender_id = isize::MAX as usize).
155                handle_kernel_event(
156                    &buf,
157                    &mut modifiers,
158                    &mut mouse,
159                    &mut dispatcher,
160                    &mut hid_devices,
161                );
162                // The buffer carried a relayed keystroke (possibly a typed secret): wipe it before reuse.
163                ostd::session::zeroize(
164                    &mut buf[..IPC_BUF_SIZE - abi::caller_identity::CALLER_IDENTITY_LEN],
165                );
166            }
167            SyscallResult::Ok(sender) => {
168                busy = true;
169                handle_message(
170                    &buf,
171                    sender,
172                    &mut modifiers,
173                    &mut mouse,
174                    &mut dispatcher,
175                    &mut sources,
176                    &mut hid_devices,
177                );
178                // SessionV1 records (typed text, raw bytes, paste chunks) must not linger in the receive buffer.
179                ostd::session::zeroize(
180                    &mut buf[..IPC_BUF_SIZE - abi::caller_identity::CALLER_IDENTITY_LEN],
181                );
182            }
183            _ => {}
184        }
185        // Drain polled releases before checking repeat deadlines, including
186        // events that arrived while IPC receive was asleep.
187        for dev in devices.iter_mut() {
188            busy |= drain_virtio(
189                dev,
190                &mut buf,
191                &mut modifiers,
192                &mut mouse,
193                &mut dispatcher,
194                &mut hid_devices,
195            );
196            if dispatcher.has_pending_mouse() {
197                break;
198            }
199        }
200        // Polled device events are staged in the head of the same buffer.
201        if !devices.is_empty() {
202            ostd::session::zeroize(&mut buf[..16]);
203        }
204        if dispatcher.has_pending_mouse() {
205            continue;
206        }
207
208        if let Some(code) = sys_get_time_ms().and_then(|now| dispatcher.repeat.poll(now)) {
209            busy = true;
210            let (keysym, character) = translate(code, modifiers.snapshot());
211            dispatcher.dispatch(&InputEvent::Key(KeyEvent {
212                timestamp_ticks: sys_get_time(),
213                scancode: code,
214                keysym,
215                character,
216                modifiers: modifiers.snapshot(),
217                state: KeyState::Repeated,
218                _pad: [0; 2],
219            }));
220        }
221
222        // Lock keys change what the keyboard's own LEDs show, and the device is
223        // the only place a user can see that state. Producers hear about it once
224        // per change rather than once per key, and a producer that comes up
225        // later is told at registration.
226        let leds = modifiers.led_bitmap();
227        if leds != leds_sent {
228            sources.broadcast_leds(leds);
229            leds_sent = leds;
230        }
231        if busy {
232            sys_yield();
233        } else {
234            // Nothing arrived: park for one scheduler tick (10 ms) rather than burn the CPU.
235            // Receives stay nonblocking and attested, so deferred polls, UART ESC deadlines,
236            // key repeat and device polling all keep their one-tick resolution.
237            idle_wait();
238        }
239    }
240}

242/// Registered raw-event producers. The USB host is the sole USB producer:
243/// isolated HID workers return decoded frames to that host instead of acquiring
244/// a second input-service identity.
245const MAX_EVENT_SOURCES: usize = 16;
246
247/// TIDs allowed to push raw `[opcode][code][value]` events.
248///
249/// Sender 0 is the kernel and is implicit. Input accepts only the tid the kernel
250/// has published as `service::USB_HID_PRODUCER` (via `sys_register_usb_hid_producer`,
251/// syscall 423) — the singleton `service::NIC_DRIVER` role is deliberately not
252/// consulted, so a USB host cell on a PC with e1000/virtio-net cannot steal the
253/// network route. The recorded identity always comes from receive metadata,
254/// never from a payload.
255struct EventSources {
256    tids: [usize; MAX_EVENT_SOURCES],
257    kinds: [u8; MAX_EVENT_SOURCES],
258    len: usize,
259}
260
261impl EventSources {
262    const fn new() -> Self {
263        Self {
264            tids: [0; MAX_EVENT_SOURCES],
265            kinds: [0; MAX_EVENT_SOURCES],
266            len: 0,
267        }
268    }
269
270    fn is_source(&self, tid: usize) -> bool {
271        tid != 0 && self.tids[..self.len].contains(&tid)
272    }
273
274    fn is_hid_host(&self, tid: usize) -> bool {
275        self.tids[..self.len]
276            .iter()
277            .zip(self.kinds[..self.len].iter())
278            .any(|(&source, &kind)| source == tid && kind == wire::ipc::input_source::USB_HID_HOST)
279    }
280
281    /// Record the kernel-verified USB HID producer.
282    fn register(&mut self, tid: usize, kind: u8) -> bool {
283        if let Some(index) = self.tids[..self.len]
284            .iter()
285            .position(|&source| source == tid)
286        {
287            return self.kinds[index] == kind;
288        }
289        if tid == 0
290            || kind != wire::ipc::input_source::USB_HID_HOST
291            || !ostd::service::sender_is_live_provider(service::USB_HID_PRODUCER, tid)
292            || self.len == MAX_EVENT_SOURCES
293        {
294            return false;
295        }
296        self.tids[self.len] = tid;
297        self.kinds[self.len] = kind;
298        self.len += 1;
299        true
300    }
301
302    /// Tell one producer what the lock LEDs should show.
303    fn send_leds(&self, tid: usize, leds: u8) {
304        let frame = [OP_SET_LEDS, leds];
305        if !matches!(sys_try_send(tid, &frame), SyscallResult::Ok(0)) {
306            print("[input] WARN: lock LED state not delivered to source tid=");
307            print_usize(tid);
308            println("");
309        }
310    }
311
312    /// Tell every registered producer what the lock LEDs should show.
313    fn broadcast_leds(&self, leds: u8) {
314        for &tid in &self.tids[..self.len] {
315            self.send_leds(tid, leds);
316        }
317    }
318}

320/// Pressed-key ownership for isolated HID interfaces.
321///
322/// This is deliberately bounded: an untrusted descriptor/report cannot grow
323/// input-service state without limit, and disconnect cleanup only touches the
324/// keys that one device actually contributed.
325const MAX_HID_DEVICES: usize = 16;
326const MAX_KEYS_PER_HID_DEVICE: usize = 128;
327
328struct HidDeviceState {
329    id: u32,
330    pressed: Vec<u32>,
331}
332
333struct HidDeviceStates {
334    devices: Vec<HidDeviceState>,
335}
336
337impl HidDeviceStates {
338    const fn new() -> Self {
339        Self {
340            devices: Vec::new(),
341        }
342    }
343
344    fn record_key(&mut self, id: u32, code: u32, state: KeyState) {
345        let index = match self.devices.iter().position(|device| device.id == id) {
346            Some(index) => index,
347            None if state == KeyState::Released => return,
348            None if self.devices.len() < MAX_HID_DEVICES => {
349                self.devices.push(HidDeviceState {
350                    id,
351                    pressed: Vec::new(),
352                });
353                self.devices.len() - 1
354            }
355            None => return,
356        };
357        let pressed = &mut self.devices[index].pressed;
358        match state {
359            KeyState::Pressed | KeyState::Repeated => {
360                if !pressed.contains(&code) && pressed.len() < MAX_KEYS_PER_HID_DEVICE {
361                    pressed.push(code);
362                }
363            }
364            KeyState::Released => pressed.retain(|&held| held != code),
365        }
366    }
367
368    fn take_pressed(&mut self, id: u32) -> Vec<u32> {
369        let Some(index) = self.devices.iter().position(|device| device.id == id) else {
370            return Vec::new();
371        };
372        self.devices.swap_remove(index).pressed
373    }
374
375    fn has_transient_modifier(&self, code: u32) -> bool {
376        let Some(modifier) = modifier_for_scancode(code) else {
377            return false;
378        };
379        self.devices.iter().any(|device| {
380            device
381                .pressed
382                .iter()
383                .copied()
384                .any(|held| modifier_for_scancode(held) == Some(modifier))
385        })
386    }
387}

389fn release_hid_device(
390    device: u32,
391    modifiers: &mut ModifierState,
392    mouse: &mut MouseState,
393    dispatcher: &mut Dispatcher,
394    hid_devices: &mut HidDeviceStates,
395) {
396    dispatcher.repeat.remove_device(device);
397    for code in hid_devices.take_pressed(device) {
398        // A second keyboard can keep the same logical modifier held (including
399        // left/right variants). Do not clear its aggregate state on this
400        // worker's teardown.
401        if hid_devices.has_transient_modifier(code) {
402            continue;
403        }
404        let mut release = [0u8; IPC_BUF_SIZE];
405        release[0] = EV_DEVICE;
406        release[1..5].copy_from_slice(&device.to_le_bytes());
407        release[5] = EV_KEY;
408        release[6..10].copy_from_slice(&code.to_le_bytes());
409        // value remains zero: EV_KEY release from this device only.
410        handle_kernel_event(&release, modifiers, mouse, dispatcher, hid_devices);
411    }
412}

414/// Park the idle owner loop for one scheduler tick. If the cell's allowlist refuses the
415/// timer, fall back to a plain yield rather than stall.
416fn idle_wait() {
417    if !matches!(sys_set_timer(1), SyscallResult::Ok(_)) {
418        sys_yield();
419    }

422/// Drain all pending events from the VirtIO virtqueue and dispatch them.
423/// Returns whether any event was dispatched.
424fn drain_virtio(
425    dev: &mut InputDevice,
426    buf: &mut [u8; IPC_BUF_SIZE],
427    modifiers: &mut ModifierState,
428    mouse: &mut MouseState,
429    dispatcher: &mut Dispatcher,
430    hid_devices: &mut HidDeviceStates,
431) -> bool {
432    let mut active = false;
433    while !dispatcher.has_pending_mouse() {
434        let Some(ev) = dev.try_get_event() else {
435            break;
436        };
437        active = true;
438        // Map Linux evdev event types → the same opcode encoding the kernel uses
439        // in dispatch_pending, so handle_kernel_event processes them identically.
440        let opcode: u8 = match ev.event_type {
441            evdev::EV_KEY => EV_KEY,
442            evdev::EV_REL => EV_REL,
443            evdev::EV_ABS => EV_ABS,
444            _ => continue, // unknown type (EV_SYN, EV_MSC, etc.) — drop
445        };
446        buf[0] = opcode;
447        buf[1..5].copy_from_slice(&(ev.code as u32).to_le_bytes());
448        buf[5..9].copy_from_slice(&ev.value.to_le_bytes());
449        handle_kernel_event(buf, modifiers, mouse, dispatcher, hid_devices);
450    }
451    active
452}

454/// Process one incoming IPC message.
455///
456/// Discrimination is by `sender`, not opcode, to avoid collisions with postcard
457/// discriminants: kernel pushes arrive with sender=0; typed requests sender>0.
458fn handle_message(
459    buf: &[u8; IPC_BUF_SIZE],
460    sender: usize,
461    modifiers: &mut ModifierState,
462    mouse: &mut MouseState,
463    dispatcher: &mut Dispatcher,
464    sources: &mut EventSources,
465    hid_devices: &mut HidDeviceStates,
466) {
467    // Registered producers speak raw frames; all other senders are typed IPC.
468    if sender == 0 {
469        handle_kernel_event(buf, modifiers, mouse, dispatcher, hid_devices);
470    } else if sources.is_source(sender) {
471        if buf[0] == OP_HID_DEVICE_REMOVED && sources.is_hid_host(sender) {
472            let device = u32::from_le_bytes([buf[1], buf[2], buf[3], buf[4]]);
473            release_hid_device(device, modifiers, mouse, dispatcher, hid_devices);
474            return;
475        }
476        handle_kernel_event(buf, modifiers, mouse, dispatcher, hid_devices);
477    } else {
478        handle_typed_request(buf, sender, modifiers, dispatcher, sources);
479    }
480}

482/// Handle a raw VirtIO event pushed by the kernel (sender == 0).
483///
484/// Wire format (`wire::proto::input`): `[opcode:1][code:4 LE][value:4 LE]`
485/// EV_KEY = keyboard key or mouse button via BTN_* scancode ≥ 0x110
486/// EV_REL = relative mouse: REL_X/Y/WHEEL
487/// EV_ABS = absolute mouse: ABS_X/Y in guest framebuffer pixels
488/// EV_ASCII = UART byte relayed by the kernel console driver
489/// EV_DEVICE = `[EV_DEVICE][device:u32][kind][code:u32][value:i32]` from a HID worker
490fn handle_kernel_event(
491    buf: &[u8; IPC_BUF_SIZE],
492    modifiers: &mut ModifierState,
493    mouse: &mut MouseState,
494    dispatcher: &mut Dispatcher,
495    hid_devices: &mut HidDeviceStates,
496) {
497    let (device, opcode, code, value) = if buf[0] == EV_DEVICE {
498        if buf.len() < DEVICE_FRAME_LEN {
499            return;
500        }
501        (
502            Some(u32::from_le_bytes([buf[1], buf[2], buf[3], buf[4]])),
503            buf[5],
504            u32::from_le_bytes([buf[6], buf[7], buf[8], buf[9]]),
505            u32::from_le_bytes([buf[10], buf[11], buf[12], buf[13]]),
506        )
507    } else {
508        if buf.len() < PRODUCER_FRAME_LEN {
509            return;
510        }
511        (
512            None,
513            buf[0],
514            u32::from_le_bytes([buf[1], buf[2], buf[3], buf[4]]),
515            u32::from_le_bytes([buf[5], buf[6], buf[7], buf[8]]),
516        )
517    };
518
519    match opcode {
520        EV_KEY => {
521            let state = key_state_from_evdev(value);
522            if let Some(device) = device {
523                hid_devices.record_key(device, code, state);
524            }
525            // BTN_* codes (≥ 0x110) are mouse buttons, not keyboard keys.
526            // Pointer events route to the compositor (cursor + Z-order owner),
527            // not the keyboard-focused cell — see Dispatcher::dispatch_mouse.
528            if code >= BTN_LEFT {
529                if let Some(button) = btn_to_mouse_button(code) {
530                    dispatcher.dispatch_mouse(&InputEvent::MouseButton { button, state });
531                }
532                return;
533            }
534            if modifiers.update(code, state) {
535                // The compositor reads held modifiers for its own pointer
536                // gestures (Win/Shift + drag of a Cell Tab). The pointer
537                // channel reaches only it; apps still see modifiers only as
538                // flags on other keys.
539                if modifier_for_scancode(code).is_some() && state != KeyState::Repeated {
540                    dispatcher.dispatch_mouse(&InputEvent::Key(KeyEvent {
541                        timestamp_ticks: sys_get_time(),
542                        scancode: code,
543                        keysym: KeySym::Unknown,
544                        character: 0,
545                        modifiers: modifiers.snapshot(),
546                        state,
547                        _pad: [0; 2],
548                    }));
549                }
550                return;
551            }
552            let (keysym, character) = translate(code, modifiers.snapshot());
553            if let Some(now) = sys_get_time_ms() {
554                if !dispatcher.repeat.observe(device, code, state, keysym, now) {
555                    return;
556                }
557            }
558            if !dispatcher.dispatch(&InputEvent::Key(KeyEvent {
559                timestamp_ticks: sys_get_time(),
560                scancode: code,
561                keysym,
562                character,
563                modifiers: modifiers.snapshot(),
564                state,
565                _pad: [0; 2],
566            })) {
567                println("[input] WARNING: clearing failed keyboard focus");
568            }
569        }
570        EV_REL => {
571            if let Some(ev) = mouse.apply_rel(code, value) {
572                dispatcher.dispatch_mouse(&ev);
573            }
574        }
575        EV_ABS => {
576            if let Some(ev) = mouse.apply_abs(code, value) {
577                dispatcher.dispatch_mouse(&ev);
578            }
579        }
580        EV_ASCII => {
581            if dispatcher.sessions.physical_active() {
582                if value > 0 && code <= 255 {
583                    dispatcher.serial_byte(code as u8);
584                }
585                return;
586            }
587            // UART byte relayed by the kernel console driver.
588            // `code` carries the raw ASCII code point; skip scancode translation.
589            // Map C0 control chars to semantic KeySyms so GUI apps get proper events
590            // regardless of whether input originates from VirtIO or UART terminal.
591            let state = if value > 0 {
592                KeyState::Pressed
593            } else {
594                KeyState::Released
595            };
596            let (keysym, character) = match code {
597                0x1B => (KeySym::Escape, 0),
598                0x0D | 0x0A => (KeySym::Return, code),
599                0x08 | 0x7F => (KeySym::Backspace, code),
600                0x09 => (KeySym::Tab, code),
601                _ => (KeySym::Printable, code),
602            };
603            if !dispatcher.dispatch_console(&InputEvent::Key(KeyEvent {
604                timestamp_ticks: sys_get_time(),
605                scancode: 0,
606                keysym,
607                character,
608                modifiers: modifiers.snapshot(),
609                state,
610                _pad: [0; 2],
611            })) {
612                println("[input] WARNING: clearing failed keyboard focus");
613            }
614        }
615        _ => {} // unknown opcode — drop silently
616    }
617}

619/// Handle a typed `InputRequest` from a compositor or shell cell (sender > 0).
620fn handle_typed_request(
621    buf: &[u8; IPC_BUF_SIZE],
622    sender: usize,
623    modifiers: &mut ModifierState,
624    dispatcher: &mut Dispatcher,
625    sources: &mut EventSources,
626) {
627    let mut resp_buf = [0u8; 64];
628    let payload_end = IPC_BUF_SIZE - abi::caller_identity::CALLER_IDENTITY_LEN;
629    if let Ok(InputRequest::SessionV1(record)) =
630        wire::ipc::decode::<InputRequest<'_>>(&buf[..payload_end])
631    {
632        let caller = abi::caller_identity::CallerIdentity::from_recv_buf(buf);
633        dispatcher.sessions.request(record, caller, sender);
634        return;
635    }
636    match wire::ipc::decode::<InputRequest<'_>>(&buf[..payload_end]) {
637        Ok(InputRequest::SessionV1(_)) => {}
638        Ok(InputRequest::SetFocus { cell_tid: _ }) => {
639            if dispatcher.sessions.physical_active() {
640                return;
641            }
642            modifiers.reset_transient();
643            // Use kernel-verified sender TID instead of the cell_tid field to
644            // prevent a cell from redirecting focus to an arbitrary TID. Only the
645            // live compositor moves the physical keyboard; see `focus_routes`.
646            dispatcher.set_focus(sender, focus_requester(buf, sender));
647            // Fire-and-forget: no reply. Focus is set atomically on receipt.
648            // A blocking reply would deadlock when the focused cell is not yet
649            // in sys_recv (startup race — G18 deadlock fix).
650        }
651        Ok(InputRequest::GetFocus) => {
652            let focused = dispatcher.focus() as u32;
653            if let Ok(encoded) = wire::ipc::encode(&InputResponse::Focus(focused), &mut resp_buf) {
654                if let Some(operation) = ostd::ipc::current() {
655                    let _ = ostd::ipc::reply(operation, encoded);
656                } else {
657                    let _ = sys_try_send(sender, encoded);
658                }
659            }
660        }
661        Ok(InputRequest::ClearFocus { cell_tid: _ }) => {
662            // Use kernel-verified sender TID (same as SetFocus) — prevents a cell
663            // from clearing another cell's focus.
664            dispatcher.clear_focus(sender);
665            // Fire-and-forget: no reply. Same rationale as SetFocus.
666        }
667        Ok(InputRequest::RegisterEventSource { kind }) => {
668            if sources.register(sender, kind) {
669                print("[input] registered raw event source kind=");
670                print_usize(kind as usize);
671                println("");
672                // A producer that just came up does not know what the locks are
673                // set to, and its keyboard would otherwise show stale state.
674                sources.send_leds(sender, modifiers.led_bitmap());
675            } else {
676                println("[input] WARN: event-source registration refused");
677            }
678        }
679        Err(_) => {} // unknown message — drop silently
680    }
681}
682
683/// Classify a focus request against the kernel's live `COMPOSITOR` binding.
684///
685/// The tid alone is not proof: the attested caller cell and generation must
686/// equal the incarnation the kernel binds to the well-known id, which only
687/// trusted init or the supervisor can register.
688fn focus_requester(buf: &[u8; IPC_BUF_SIZE], sender: usize) -> Requester {
689    let mut record = [0u8; abi::service_binding::SERVICE_BINDING_LEN];
690    let Some(binding) = ostd::syscall::sys_lookup_service_bound(service::COMPOSITOR, &mut record)
691    else {
692        return Requester::ClientHeadless;
693    };
694    let caller = abi::caller_identity::CallerIdentity::from_recv_buf(buf);
695    let is_compositor = binding.tid == sender as u64
696        && caller.is_some_and(|caller| {
697            caller.cell_id == binding.cell_id && caller.generation == binding.generation
698        });
699    if is_compositor {
700        Requester::Compositor
701    } else {
702        Requester::ClientWithCompositor
703    }
704}
