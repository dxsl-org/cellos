3// EV_REL axis codes
4const REL_X: u32 = 0;
5const REL_Y: u32 = 1;
6const REL_WHEEL: u32 = 8;
7
8// EV_ABS axis codes
9const ABS_X: u32 = 0;
10const ABS_Y: u32 = 1;
11
12// Mouse button EV_KEY codes (BTN_*)
13pub const BTN_LEFT: u32 = 0x110;
14pub const BTN_RIGHT: u32 = 0x111;
15pub const BTN_MIDDLE: u32 = 0x112;

17/// Tracks screen-pixel position. EV_ABS producers must scale device units once,
18/// before forwarding them here; service-input never rescales pixel coordinates.
19pub struct MouseState {
20    x: i32,
21    y: i32,
22    max_x: i32,
23    max_y: i32,
24}
25
26impl MouseState {
27    pub fn new(width: u32, height: u32) -> Self {
28        Self {
29            x: 0,
30            y: 0,
31            max_x: width.saturating_sub(1).min(i32::MAX as u32) as i32,
32            max_y: height.saturating_sub(1).min(i32::MAX as u32) as i32,
33        }
34    }
35
36    /// Handle an EV_REL event. Returns the `InputEvent` to dispatch, if any.
37    pub fn apply_rel(&mut self, code: u32, raw_value: u32) -> Option<InputEvent> {
38        let value = raw_value as i32;
39        match code {
40            REL_X => {
41                let old = self.x;
42                self.x = self.x.saturating_add(value).clamp(0, self.max_x);
43                Some(InputEvent::MouseMove {
44                    x: self.x,
45                    y: self.y,
46                    dx: self.x - old,
47                    dy: 0,
48                })
49            }
50            REL_Y => {
51                let old = self.y;
52                self.y = self.y.saturating_add(value).clamp(0, self.max_y);
53                Some(InputEvent::MouseMove {
54                    x: self.x,
55                    y: self.y,
56                    dx: 0,
57                    dy: self.y - old,
58                })
59            }
60            REL_WHEEL => Some(InputEvent::MouseScroll { dx: 0, dy: value }),
61            _ => None,
62        }
63    }
64
65    /// Handle EV_ABS guest-pixel coordinates (not normalized device units).
66    pub fn apply_abs(&mut self, code: u32, raw_value: u32) -> Option<InputEvent> {
67        let value = raw_value.min(i32::MAX as u32) as i32;
68        match code {
69            ABS_X => {
70                let value = value.clamp(0, self.max_x);
71                let dx = value - self.x;
72                self.x = value;
73                Some(InputEvent::MouseMove {
74                    x: self.x,
75                    y: self.y,
76                    dx,
77                    dy: 0,
78                })
79            }
80            ABS_Y => {
81                let value = value.clamp(0, self.max_y);
82                let dy = value - self.y;
83                self.y = value;
84                Some(InputEvent::MouseMove {
85                    x: self.x,
86                    y: self.y,
87                    dx: 0,
88                    dy,
89                })
90            }
91            _ => None,
92        }
93    }
94}

96/// Map a BTN_* scancode to a `MouseButton`, or `None` if not a recognised button.
97pub fn btn_to_mouse_button(code: u32) -> Option<MouseButton> {
98    match code {
99        BTN_LEFT => Some(MouseButton::Left),
100        BTN_RIGHT => Some(MouseButton::Right),
101        BTN_MIDDLE => Some(MouseButton::Middle),
102        _ => None,
103    }
104}

106#[cfg(test)]
107mod tests {
108    use super::*;
109
110    fn position(event: Option<InputEvent>) -> (i32, i32, i32, i32) {
111        match event {
112            Some(InputEvent::MouseMove { x, y, dx, dy }) => (x, y, dx, dy),
113            _ => panic!("expected pointer movement"),
114        }
115    }
116
117    #[test]
118    fn relative_edges_do_not_accumulate_offscreen_debt() {
119        let mut mouse = MouseState::new(1920, 1080);
120        assert_eq!(position(mouse.apply_rel(REL_X, (-1200i32) as u32)), (0, 0, 0, 0));
121        assert_eq!(position(mouse.apply_rel(REL_X, 10)), (10, 0, 10, 0));
122        assert_eq!(position(mouse.apply_rel(REL_X, i32::MAX as u32)), (1919, 0, 1909, 0));
123        assert_eq!(position(mouse.apply_rel(REL_X, (-1i32) as u32)), (1918, 0, -1, 0));
124        assert_eq!(position(mouse.apply_rel(REL_Y, 2000)), (1918, 1079, 0, 1079));
125        assert_eq!(position(mouse.apply_rel(REL_Y, (-1i32) as u32)), (1918, 1078, 0, -1));
126    }
127
128    #[test]
129    fn absolute_coordinates_are_pixels_not_scaled_again() {
130        let mut mouse = MouseState::new(1280, 800);
131        assert_eq!(position(mouse.apply_abs(ABS_X, 295)), (295, 0, 295, 0));
132        assert_eq!(position(mouse.apply_abs(ABS_Y, 119)), (295, 119, 0, 119));
133        assert_eq!(position(mouse.apply_abs(ABS_X, 1279)), (1279, 119, 984, 0));
134        assert_eq!(position(mouse.apply_abs(ABS_Y, 799)), (1279, 799, 0, 680));
135        assert_eq!(position(mouse.apply_abs(ABS_X, u32::MAX)), (1279, 799, 0, 0));
136    }
137
138    #[test]
139    fn wheel_remains_signed_and_zero_extent_is_safe() {
140        let mut mouse = MouseState::new(0, 0);
141        assert_eq!(position(mouse.apply_rel(REL_X, 500)), (0, 0, 0, 0));
142        assert!(matches!(
143            mouse.apply_rel(REL_WHEEL, (-2i32) as u32),
144            Some(InputEvent::MouseScroll { dx: 0, dy: -2 })
145        ));
146    }
