115/// Selectors whose request body can carry typed or raw secrets (SubmitEvents, WritePaste, SubmitRaw).
116/// Their results are only a sequence/count, so nothing beyond the status is ever replayable.
117fn secret_bearing(selector: u16) -> bool { matches!(selector, 13 | 15 | 20) }
118fn wipe(bytes: &mut Vec<u8>) { zeroize_vec(bytes); *bytes = Vec::new(); }
119/// The record an outcome retains for `request`. A secret-bearing request is sealed at the
120/// moment it is consumed: only the SHA-256 of the exact record is kept, never its plaintext,
121/// not even while the request executes (the body is read from the receive buffer, which the
122/// caller wipes). The digest still lets an identical re-send resolve to the retained (or
123/// pending) outcome and a changed body under the same ID be refused (annex: Malformed).
124fn retained(selector: u16, request: &[u8]) -> Result<(Vec<u8>, bool), Status> {
125    let sealed = secret_bearing(selector);
126    let mut copy = Vec::new();
127    if sealed {
128        copy.try_reserve_exact(32).map_err(|_| Status::QuotaExceeded)?;
129        copy.extend_from_slice(&ostd::session::sha256(request));
130    } else {
131        reserve(&mut copy, request.len())?; copy.extend_from_slice(request);
132    }
133    Ok((copy, sealed))
134}
135impl Outcome {
136    /// Wipe and release the retained request and result bodies (the fence stays elsewhere).
137    fn scrub(&mut self) { wipe(&mut self.request); wipe(&mut self.result); self.sealed = false; }
138    /// Does `bytes` exactly equal the record this outcome retained? `digest` caches the hash of
139    /// `bytes` across candidates. Selector and request ID are header fields (not secret).
140    fn matches(&self, bytes: &[u8], digest: &mut Option<[u8; 32]>) -> bool {
141        if self.request.is_empty() { return false; }
142        if !self.sealed { return self.request == bytes; }
143        if bytes.len() < HEADER_LEN || bytes[4..6] != self.selector.to_le_bytes() || bytes[16..24] != self.id.to_le_bytes() { return false; }
144        let digest = digest.get_or_insert_with(|| ostd::session::sha256(bytes));
145        self.request[..] == digest[..]
146    }
147}

152    pub fn new() -> Self { Self { sessions: Vec::new(), frontends: Vec::new(), outcomes: Vec::new(), fences: Vec::new(), serial: None, issuer: None, next_ref: 1, kernel_operation: ZERO, challenges: Vec::new() } }
153    fn reference(&mut self) -> Result<Ref, Status> { let id = self.next_ref; self.next_ref = id.checked_add(1).ok_or(Status::QuotaExceeded)?; Ok(Ref { id, generation: 1 }) }
154    fn binding(&mut self) -> Result<ServiceBinding, Status> {
155        let mut bytes = [0; 24];
156        let binding = sys_lookup_service_bound(api::syscall::service::INPUT as u16, &mut bytes).ok_or(Status::Revoked)?;
157        if let Some(old) = self.issuer { if old != binding { return Err(Status::Revoked); } } else { self.issuer = Some(binding); }
158        Ok(binding)
159    }
160    fn kernel(&mut self, op: KernelOp, body: &[u8], reply: &mut [u8]) -> Result<usize, Status> {
161        self.binding()?;
162        let query = matches!(op, KernelOp::InspectPreparedGrant | KernelOp::InspectEndpoint | KernelOp::QueryReceiptChild | KernelOp::QueryOperation)
163            || (matches!(op, KernelOp::ResolveBindReceipt | KernelOp::ResolveBootstrap)
164                && body.get(16..20).is_some_and(|v| v == [0; 4]));
165        let header = if query {
166            // Nonmutating selectors retain no outcome and consume no request ID; they must
167            // not mint SDK correlation lanes (bounded, never retired for foreign operations).
168            sys::kernel_request(op, 1, body, reply)?
169        } else {
170            sys::kernel_call(op, self.kernel_operation, true, body, reply)?
171        };
172        if header.status != Status::Ok { return Err(header.status); }
173        Ok(header.total_len as usize)
174    }
175    fn resolve_bootstrap(&mut self, bootstrap: Ref, action: u32, session: Ref) -> Result<(u32, Principal, Ref, [Ref; 4], [Ref; 2], [Ref; 2]), Status> {
176        let mut body = [0; 40]; let mut w = Writer::new(&mut body); w.reference(bootstrap)?; w.u32(action)?; w.u32(0)?; w.reference(session)?;

178        let mut r = Reader::new(&reply[HEADER_LEN..len]); let state = r.u32()?; reserved(r.u32()?)?; let controller = r.principal()?; let operation = r.reference()?;
179        let endpoints = [r.reference()?, r.reference()?, r.reference()?, r.reference()?]; let outputs = [r.reference()?, r.reference()?]; let writers = [r.reference()?, r.reference()?]; r.finish()?;
180        Ok((state, controller, operation, endpoints, outputs, writers))
181    }

184    fn reader(&self, index: usize, p: Principal, lease: Ref) -> Result<(), Status> { let s = &self.sessions[index]; if s.frozen { return Err(Status::WouldBlock); } if (s.attachment.is_none() && !s.detached) || s.reader != p || s.lease != lease { Err(Status::Denied) } else { Ok(()) } }
185    fn frontend(&self, index: usize, p: Principal, attachment: Ref, epoch: u64) -> Result<(), Status> {
186        let s = &self.sessions[index]; if s.frozen { return Err(Status::WouldBlock); } if s.epoch != epoch { return Err(Status::WrongEpoch); }
187        match s.attachment.as_ref() {
188            Some(a) if a.reference == attachment && a.frontend == p => Ok(()),
189            // A sel-25 holder presenting a detached or superseded attachment holds a stale selector.
190            _ if s.holder == Some(p) => Err(Status::Stale),
191            _ => Err(Status::Denied),
192        }

208    fn owned_frontend(&self, reference: Ref, p: Principal) -> Result<usize, Status> { let i = self.frontend_index(reference)?; if self.frontends[i].holder == p { Ok(i) } else { Err(Status::Denied) } }
209    fn epoch(&self, index: usize, epoch: u64) -> Result<(), Status> { if self.sessions[index].epoch == epoch { Ok(()) } else { Err(Status::WrongEpoch) } }
210    fn operation_for(&self, selector: u16, body: &[u8], caller: Principal) -> Result<Ref, Status> {
211        let mut r = Reader::new(body); let reference = r.reference()?;
212        // Trust lanes: Register is fenced on its challenge Ref, Collect on its confirmation Ref.
213        if matches!(selector, 1 | 18 | 19 | 23 | 25 | 29 | 31) { return Ok(reference); }
214        if matches!(selector, 24 | 26..=28) { return self.frontends.iter().find(|f| f.reference == reference).map(|f| f.operation).ok_or(Status::Stale); }
215        if (15..=17).contains(&selector) {
216            let s = self.sessions.iter().find(|s| s.paste.as_ref().is_some_and(|p| p.reference == reference)).ok_or(Status::Stale)?;
217            return Ok(Self::holder_lane(s, caller).unwrap_or(s.operation));
218        }
219        let s = &self.sessions[self.index(reference)?];
220        if let Some(lane) = Self::holder_lane(s, caller) { return Ok(lane); }
221        Ok(s.foreground.as_ref().filter(|f| f.child.principal == caller && !f.restored).map(|f| f.operation).unwrap_or(s.operation))
222    }
223    fn participant(&self, index: usize, p: Principal) -> bool {
224        let s = &self.sessions[index];
225        s.controller == p || s.reader == p || s.holder == Some(p)
226            || s.attachment.as_ref().is_some_and(|a| a.frontend == p)
227            || s.foreground.as_ref().is_some_and(|f| f.child.principal == p)
228            || s.modes.iter().chain(s.baseline_modes.iter()).any(|m| m.holder == p)
229    }
230    /// Only principals already tied to the target session may allocate retained
231    /// outcome/fence capacity; a guessed session reference confers no bookkeeping budget.
232    fn admit(&mut self, selector: u16, body: &[u8], p: Principal, sender: usize) -> Result<(), Status> {
233        let mut r = Reader::new(body);
234        match selector {
235            29 | 31 => self.admit_trust(selector, body, p, sender),
236            1 => {
237                let operation = r.reference()?; let bootstrap = r.reference()?;
238                self.kernel_operation = operation;
239                let (state, controller, recorded, ..) = self.resolve_bootstrap(bootstrap, 0, ZERO)?;
240                if state > 1 || controller != p || recorded != operation { return Err(Status::Denied); }
241                Ok(())
242            }
243            15..=17 => {
244                let reference = r.reference()?;
245                let i = self.sessions.iter().position(|s| !s.closed && s.paste.as_ref().is_some_and(|x| x.reference == reference)).ok_or(Status::Stale)?;
246                if self.participant(i, p) { Ok(()) } else { Err(Status::Denied) }
247            }
248            // Any attested Cell may register a frontend on a live operation of its own lane.
249            23 => { let operation = r.reference()?; self.kernel_operation = operation; if self.query_operation(operation)? != 0 { return Err(Status::Cancelled); } Ok(()) }
250            24 | 26..=28 => { self.owned_frontend(r.reference()?, p)?; Ok(()) }
251            // Only the exact child bound by sel 24 may create the frontend's session.
252            25 => {
253                r.reference()?; r.reference()?; let f = &self.frontends[self.frontend_index(r.reference()?)?];
254                if f.child.is_some_and(|c| c.principal == p) { Ok(()) } else { Err(Status::Denied) }
255            }
256            18 | 19 => Ok(()),
257            _ => { let i = self.index(r.reference()?)?; if self.participant(i, p) { Ok(()) } else { Err(Status::Denied) } }
258        }
259    }
260    /// Drop request fences whose operation no live session or live foreground child uses
261    /// and that retain no outcome. A dead principal/operation cannot present an old ID.
262    fn prune(&mut self) {
263        let sessions = &self.sessions; let outcomes = &self.outcomes; let frontends = &self.frontends; let challenges = &self.challenges;
264        self.fences.retain(|f| sessions.iter().any(|s| !s.closed && (s.operation == f.operation
265                || (s.holder.is_some() && !s.orphaned && s.holder_operation == f.operation)
266                || s.foreground.as_ref().is_some_and(|g| !g.restored && g.operation == f.operation)))
267            || frontends.iter().any(|x| x.operation == f.operation)
268            || trust::challenge_lane(challenges, f.caller, f.operation)
269            || outcomes.iter().any(|o| !o.request.is_empty() && o.caller == f.caller && o.operation == f.operation));
270    }
271    fn drop_outcomes(&mut self, operation: Ref) {
272        for o in self.outcomes.iter_mut().filter(|o| o.operation == operation) { o.scrub(); o.pending = false; }
273    }
274    /// An explicitly closed session's retained outcomes die with it, except the outcome of the
275    /// close request itself (kept so its replay and acknowledgement still resolve). Dropped
276    /// outcomes are marked observed so they can never block that acknowledgement.
277    fn release_closed(&mut self, operations: &[Option<Ref>], keep: usize) {
278        for operation in operations.iter().flatten().copied() {
279            if self.frontend_lane(operation) || self.sessions.iter().any(|s| !s.closed && (s.operation == operation || s.foreground.as_ref().is_some_and(|f| !(f.restored && f.acknowledged) && f.operation == operation))) { continue; }
280            for (n, o) in self.outcomes.iter_mut().enumerate() {
281                if n == keep || o.operation != operation { continue; }
282                o.scrub(); o.pending = false; o.observed = true;
283            }
284        }
285        self.prune();
286    }
287    /// Retire the SDK correlation lane of an operation this service no longer calls on, unless
288    /// another live session or unfinished foreground still uses it.
289    fn retire_lane(&self, operation: Ref) {
290        let busy = self.frontend_lane(operation) || self.sessions.iter().any(|s| !s.closed && (s.operation == operation
291            || s.foreground.as_ref().is_some_and(|f| !(f.restored && f.acknowledged) && f.operation == operation)));
292        if !busy { ostd::session::retire_operation(operation); }
293    }
294    /// The foreground child is terminal and restored: its retained outcomes/fences die with it.
295    fn retire_foreground(&mut self, i: usize) {
296        let Some(f) = self.sessions[i].foreground.as_ref() else { return; };
297        let (principal, operation) = (f.child.principal, f.operation);
298        for o in self.outcomes.iter_mut().filter(|o| o.caller == principal && o.operation == operation) {
299            o.scrub(); o.pending = false;
300        }
301        self.prune();
302        self.retire_lane(operation);
303    }
304    fn record(&mut self, header: RequestHeaderV1, request: &[u8], p: Principal, operation: Ref) -> Result<(usize, bool), Status> {
305        if let Some(i) = self.outcomes.iter().position(|o| !o.request.is_empty() && o.caller == p && o.operation == operation && o.id == header.request_id) {
306            if !self.outcomes[i].matches(request, &mut None) { return Err(Status::Malformed); }
307            return Ok((i, true));
308        }
309        let fence = if let Some(i) = self.fences.iter().position(|f| f.caller == p && f.operation == operation) { i } else {
310            if self.fences.len() >= 512 { self.prune(); } if self.fences.len() >= 512 { return Err(Status::QuotaExceeded); } reserve(&mut self.fences, 1)?;
311            self.fences.push(Fence { caller: p, operation, next: 1, acknowledged: 0 }); self.fences.len() - 1
312        };
313        if header.request_id != self.fences[fence].next { return Err(Status::UnknownRequest); }
314        if self.outcomes.iter().filter(|o| o.caller == p && o.operation == operation && !o.request.is_empty()).count() >= 64
315            || self.outcomes.iter().filter(|o| o.caller == p && !o.request.is_empty()).count() >= 128 { return Err(Status::QuotaExceeded); }
316        let next = header.request_id.checked_add(1).ok_or(Status::QuotaExceeded)?;
317        let (copy, sealed) = retained(header.selector, request)?;
318        let mut result = Vec::new(); reserve(&mut result, MAX_SERVICE_RECORD_LEN)?;
319        let slot = self.outcomes.iter().position(|o| o.request.is_empty());
320        if slot.is_none() && self.outcomes.len() >= 1024 { return Err(Status::QuotaExceeded); }
321        if slot.is_none() { reserve(&mut self.outcomes, 1)?; }
322        let record = Outcome { caller: p, operation, id: header.request_id, selector: header.selector, request: copy, sealed, result, observed: false, pending: true, poll_session: ZERO, poll_epoch: 0, cursor: 0 };
323        let index = if let Some(slot) = slot { self.outcomes[slot] = record; slot } else { self.outcomes.push(record); self.outcomes.len() - 1 };
324        self.fences[fence].next = next;
325        Ok((index, false))
326    }

354    pub fn request(&mut self, bytes: &[u8], caller: Option<CallerIdentity>, sender: usize) {
355        let token = sys_ipc_current(); let token = if token > 0 { token as usize } else { 0 };
356        let header = match RequestHeaderV1::decode(bytes) {
357            Ok(header) => header,
358            Err(status) => {
359                if bytes.len() >= HEADER_LEN {
360                    let header = RequestHeaderV1 {
361                        selector: u16::from_le_bytes([bytes[4], bytes[5]]),
362                        total_len: bytes.len() as u32,
363                        request_id: u64::from_le_bytes(bytes[16..24].try_into().unwrap_or([0; 8])),
364                    };
365                    Self::error(header, status, sender, token);
366                }
367                return;
368            }
369        };
370        if !(1..=31).contains(&header.selector) { Self::error(header, Status::UnknownOperation, sender, token); return; }
371        let Some(caller) = caller.filter(|c| c.sender_tid == sender as u64 && c.generation != 0) else { Self::error(header, Status::Denied, sender, token); return; };
372        let p = principal(caller);
373        if bytes.len() > MAX_SERVICE_RECORD_LEN { Self::error(header, Status::Malformed, sender, token); return; }
374        if let Err(status) = self.binding() { Self::error(header, status, sender, token); return; }
375        self.maintain();
376        // Resolve exact retained bytes before resource lookup: CommitPaste,
377        // AbortPaste and CloseSession remove the live selector they mutate.
378        let mut digest = None;
379        if let Some(index) = self.outcomes.iter().position(|o| o.caller == p && o.matches(bytes, &mut digest)) {
380            self.reply_outcome(index, header, p, sender, token);
381            return;
382        }
383        let body = &bytes[HEADER_LEN..];
384        let operation = match self.operation_for(header.selector, body, p) { Ok(v) => v, Err(status) => { Self::error(header, status, sender, token); return; } };
385        if let Err(status) = self.admit(header.selector, body, p, sender) { Self::error(header, status, sender, token); return; }
386        if header.selector == 18 || header.selector == 19 { self.query_ack(header, body, p, sender, token); return; }
387        // Nonmutating reads (QueryCapabilities, QueryFrontend) retain no outcome and consume no ID.
388        if header.selector == 12 || header.selector == 26 {
389            let mut reply = [0; MAX_SERVICE_RECORD_LEN - HEADER_LEN];
390            let result = self.execute(header.selector, body, p, usize::MAX, token, &mut reply);
391            let mut record = [0; MAX_SERVICE_RECORD_LEN];
392            let (status, length) = match result { Ok(Some(n)) => (Status::Ok, n), Ok(None) => (Status::Malformed, 0), Err(s) => (s, 0) };
393            let _ = ResultHeaderV1 { selector: header.selector, status, total_len: (HEADER_LEN + length) as u32, request_id: header.request_id }.encode(&mut record[..HEADER_LEN]);
394            record[HEADER_LEN..HEADER_LEN + length].copy_from_slice(&reply[..length]);
395            Self::send(&record[..HEADER_LEN + length], sender, token);
396            return;
397        }
398        let (outcome, replay) = match self.record(header, bytes, p, operation) { Ok(v) => v, Err(status) => { Self::error(header, status, sender, token); return; } };
399        if replay {
400            self.reply_outcome(outcome, header, p, sender, token);
401            return;
402        }
403        let mut reply = Scrub::<{ MAX_SERVICE_RECORD_LEN - HEADER_LEN }>::new();
404        match self.execute(header.selector, body, p, outcome, token, &mut reply) {
405            Ok(Some(len)) => self.finish(outcome, Status::Ok, &reply[..len]),
406            Ok(None) => return,
407            Err(status) => self.finish(outcome, status, &[]),
408        }
409        let o = &mut self.outcomes[outcome]; o.observed |= Self::send(&o.result, sender, token);
410    }
411    fn query_ack(&mut self, h: RequestHeaderV1, body: &[u8], p: Principal, sender: usize, token: usize) {
412        let mut r = Reader::new(body); let parsed = (|| { let op = r.reference()?; let through = r.u64()?; r.finish()?; Ok::<_, Status>((op, through)) })();
413        let (op, id) = match parsed { Ok(v) => v, Err(s) => { Self::error(h, s, sender, token); return; } };
414        if h.selector == 19 {
415            if let Some(index) = self.outcomes.iter().position(|o| !o.request.is_empty() && o.caller == p && o.operation == op && o.id == id) {
416                self.reply_outcome(index, h, p, sender, token);
417            } else { Self::error(h, Status::UnknownRequest, sender, token); }
418            return;
419        }
420        let Some(fence) = self.fences.iter().position(|f| f.caller == p && f.operation == op) else { Self::error(h, Status::UnknownRequest, sender, token); return; };
421        if id <= self.fences[fence].acknowledged { Self::error(h, Status::Ok, sender, token); return; }
422        if id >= self.fences[fence].next || self.outcomes.iter().any(|o| o.caller == p && o.operation == op && o.id <= id && (!o.observed || o.pending)) { Self::error(h, Status::UnknownRequest, sender, token); return; }

425            o.scrub();
426        }
427        // Stable outcome indices are used by saved poll tokens. Tombstones retain
428        // their ID fence until the operation/principal is retired.
429        self.fences[fence].acknowledged = self.fences[fence].acknowledged.max(id);
430        Self::error(h, Status::Ok, sender, token);
431    }

644    /// Frontend selectors 23–28 (annex S5-3c §3, §3a).
645    fn frontend_operation(&mut self, selector: u16, body: &[u8], p: Principal, w: &mut Writer<'_>) -> Result<(), Status> {
646        match selector {
647            23 => {
648                let request = fe::RegisterFrontendV1::decode(body)?;

671                }

674            27 => {

678            }
679            28 => self.reattach(fe::ReattachFrontendV1::decode(body)?, p, w),

681        }

733    /// Sel 28: the resume holder binds a new Pending frontend to its Detached session.

757    /// The frontend is gone; the controller keeps the session (annex §3a). Its pending

774    /// Forget a frontend record. Its kind-6 receipt is revoked if still unconsumed and

782        }

784    }

796        }

929    fn admit_events(&mut self, i: usize, bytes: &[u8], count: usize) -> Result<(), Status> {
930        if self.sessions[i].capture.is_some() { return self.capture_events(i, bytes, count); }
931        if let Some(status) = self.sessions[i].input_failure { return Err(status); }
932        if self.sessions[i].draining { return Err(Status::BrokenPipe); }
933        if self.sessions[i].paste.is_some() || self.sessions[i].physical_paste.is_some() || self.sessions[i].physical_paste_discard { return Err(Status::WouldBlock); }
934        let raw_mode = self.sessions[i].mode & 1 != 0;
935        let mut prepared = Vec::new(); reserve(&mut prepared, count)?;
936        let mut raw = Scrub::<MAX_SERVICE_RECORD_LEN>::new();
937        let mut raw_len = 0; let mut offset = 0; let mut interrupts = 0; let mut interrupt_frames = 0;
938        let child_op = self.sessions[i].foreground.as_ref().filter(|f| f.committed && !f.restored).map(|f| (f.operation, f.cancelled));
939        for _ in 0..count {
940            let frame = bytes.get(offset..offset + 24).ok_or(Status::Malformed)?;
941            let kind = u16::from_le_bytes([frame[16], frame[17]]);
942            let length = u16::from_le_bytes([frame[18], frame[19]]) as usize;
943            if frame[..16] != [0; 16] || frame[20..24] != [0; 4] || !(1..=2).contains(&kind) { return Err(Status::Malformed); }
944            let payload = bytes.get(offset + 24..offset + 24 + length).ok_or(Status::Malformed)?;
945            api::services::session::validate_event(kind, payload)?;
946            if kind == 2 { api::services::session::validate_text(payload)?; }
947            let ctrl_c = kind == 1 && payload[4..8] == 2u32.to_le_bytes()
948                && matches!(u32::from_le_bytes(payload[..4].try_into().map_err(|_| Status::Malformed)?), 67 | 99) && !raw_mode;
949            // Canonical Ctrl-C is one cancellation per press; release/repeat are consumed without bytes.
950            if ctrl_c && payload[8] != 1 { offset += 24 + length; continue; }
951            let interrupt = ctrl_c;
952            let mut encoded = Scrub::<512>::new();
953            let encoded_len = if kind == 1 {
954                api::services::session::encode_key(u32::from_le_bytes(payload[..4].try_into().map_err(|_| Status::Malformed)?),
955                    u32::from_le_bytes(payload[4..8].try_into().map_err(|_| Status::Malformed)?), payload[8], &mut encoded)?
956            } else { encoded[..payload.len()].copy_from_slice(payload); payload.len() };
957            if interrupt { interrupts += 1; if child_op.is_none() { interrupt_frames += 1; prepared.push(self.event(i, 6, &[])?); } }
958            else if raw_mode {
959                if raw_len + encoded_len > raw.len() { return Err(Status::QuotaExceeded); }
960                raw[raw_len..raw_len + encoded_len].copy_from_slice(&encoded[..encoded_len]); raw_len += encoded_len;
961            } else { prepared.push(self.event(i, kind, payload)?); }
962            offset += 24 + length;
963        }
964        if offset != bytes.len() { return Err(Status::Malformed); }
965        let s = &self.sessions[i];
966        s.sequence.checked_add(count as u64).ok_or(Status::QuotaExceeded)?;
967        if s.controls.len() + interrupt_frames > CONTROL_LIMIT - 1 || s.typed.len() + prepared.len() - interrupt_frames > INPUT_LIMIT
968            || s.raw.len() + raw_len > RAW_LIMIT { return Err(Status::QuotaExceeded); }
969        // Every bounded frame and queue slot is reserved before the first effect.
970        // One cancellation of the committed child's operation per accepted batch,
971        // never a controller-root cancel; its revoked control endpoint carries no frame.
972        if let (true, Some((operation, false))) = (interrupts != 0, child_op) {
973            self.kernel_operation = operation;
974            let mut body = [0; 24]; let mut b = Writer::new(&mut body); b.reference(operation)?; b.u32(2)?; b.u32(0)?;
975            let mut result = [0; 24];
976            match self.kernel(KernelOp::CancelOperation, &body, &mut result) {
977                Ok(_) | Err(Status::Cancelled) => { if let Some(f) = self.sessions[i].foreground.as_mut() { f.cancelled = true; } }
978                // The child is already gone (or the operation retired): nothing to cancel, nothing to latch.
979                Err(Status::WrongOwner | Status::Stale | Status::InvalidChild | Status::Revoked) => {}
980                Err(status) => return Err(status),
981            }
982        }
983        let s = &mut self.sessions[i];
984        for mut event in prepared {
985            s.sequence += 1; event.sequence = s.sequence; event.bytes[..8].copy_from_slice(&s.sequence.to_le_bytes());
986            if event.bytes[16..18] == 6u16.to_le_bytes() { s.controls.push_back(event); } else { s.typed.push_back(event); }
987        }
988        s.raw.extend(&raw[..raw_len]);
989        if raw_mode { s.sequence += count as u64; }
990        Ok(())
991    }

1007    fn admit_paste(&mut self, i: usize, bytes: &[u8]) -> Result<u64, Status> {

1038    }

1049    fn drain(&mut self, i: usize) {

1051        if self.sessions[i].closed || self.sessions[i].frozen || (self.sessions[i].attachment.is_none() && !self.sessions[i].detached) || self.sessions[i].input_failure.is_some() { return; }
1052        // One bounded write per ring per turn; no full ring can stall ingress.
1053        if let Some(n) = self.sessions[i].controls.iter().position(|e| e.sent < e.bytes.len()) {
1054            let mut bytes = Scrub::<512>::new(); let event = &self.sessions[i].controls[n]; let count = (event.bytes.len() - event.sent).min(512);
1055            bytes[..count].copy_from_slice(&event.bytes[event.sent..event.sent + count]);
1056            match self.write_stream(self.sessions[i].writers[1], &bytes[..count]) {
1057                Ok(written) => self.sessions[i].controls[n].sent += written,
1058                Err(Status::WouldBlock) => {}
1059                Err(status) => { self.sessions[i].input_failure = Some(status); self.invalidate_poll(i, status); return; }
1060            }
1061        }
1062        if self.sessions[i].controls.iter().any(|e| e.sent != e.bytes.len()) { return; }
1063        if !self.sessions[i].raw.is_empty() {
1064            let mut bytes = Scrub::<512>::new(); let count = self.sessions[i].raw.len().min(512);
1065            for (destination, byte) in bytes.iter_mut().zip(self.sessions[i].raw.iter()).take(count) { *destination = *byte; }
1066            match self.write_stream(self.sessions[i].writers[0], &bytes[..count]) {
1067                Ok(written) => zeroize_deque_front(&mut self.sessions[i].raw, written),
1068                Err(Status::WouldBlock) => {}
1069                Err(status) => { self.sessions[i].input_failure = Some(status); self.invalidate_poll(i, status); return; }
1070            }
1071        }
1072        if self.sessions[i].draining && !self.sessions[i].eof && self.sessions[i].raw.is_empty() && self.sessions[i].typed.is_empty() {
1073            if self.sessions[i].controls.len() >= CONTROL_LIMIT - 1 { return; }
1074            let Ok(event) = self.event(i, 7, &[]) else { return; };
1075            let endpoint = self.sessions[i].writers[0];
1076            self.kernel_operation = self.sessions[i].foreground.as_ref().filter(|f| f.committed && !f.restored).map(|f| f.operation).unwrap_or(self.sessions[i].operation);
1077            let mut result = [0; 24];
1078            match self.kernel(KernelOp::CloseEndpoint, &endpoint.to_bytes(), &mut result) {
1079                Ok(_) => { let s = &mut self.sessions[i]; s.sequence = event.sequence; s.controls.push_back(event); s.eof = true; if s.writers[0] == s.baseline_writers[0] { s.baseline_eof = true; } }
1080                Err(status) => { self.sessions[i].input_failure = Some(status); self.invalidate_poll(i, status); }
1081            }
1082        }
1083    }
1084    pub fn maintain(&mut self) {
1085        let timestamp=now();
1086        self.maintain_frontends();
1087        self.maintain_trust();
1088        for i in 0..self.sessions.len() {
1089            if self.sessions[i].closed {continue;}
1090            self.kernel_operation = self.sessions[i].operation;
1091            let bootstrap = self.sessions[i].bootstrap;
1092            match self.resolve_bootstrap(bootstrap, 0, ZERO) {
1093                Ok((2, controller, operation, _, _, _)) if controller == self.sessions[i].controller && operation == self.sessions[i].operation => {}
1094                Ok(_) | Err(Status::Stale | Status::Revoked | Status::Cancelled | Status::NotFound) => {
1095                    let sop = self.sessions[i].operation; let fop = self.sessions[i].foreground.as_ref().map(|f| f.operation);
1096                    if self.close(i).is_ok() { self.drop_outcomes(sop); if let Some(fop) = fop { self.drop_outcomes(fop); } self.prune(); }
1097                    continue;
1098                }
1099                // An unverifiable owner query is not proof of death: retry next turn without altering routing.
1100                Err(_) => { continue; }
1101            }
1102            if self.sessions[i].holder.is_some() && !self.sessions[i].orphaned {
1103                match self.query_operation(self.sessions[i].holder_operation) {
1104                    Ok(0) => {}
1105                    Ok(_) | Err(Status::Stale | Status::Revoked | Status::Cancelled | Status::NotFound) => self.orphan(i),
1106                    // Unverifiable is not proof of holder death: retry next turn.
1107                    Err(_) => {}
1108                }
1109            }
1110            // An orphaned session ends its input once Lost(4) is out: EOF to each current reader.
1111            if self.sessions[i].orphaned && !self.sessions[i].draining { self.sessions[i].draining = true; }
1112            if self.sessions[i].prepare_deadline != 0 && timestamp >= self.sessions[i].prepare_deadline
1113                && !self.sessions[i].foreground.as_ref().is_some_and(|f| !f.restored || !f.acknowledged) {
1114                // The prepared child's five-second reservation lapsed before any receipt was registered.
1115                self.sessions[i].prepare_deadline = 0; self.sessions[i].frozen = false;
1116            }
1117            if let Err(status) = self.reconcile(i) {
1118                // Only a proven identity mismatch fails closed; transient kernel errors retry next turn.
1119                if matches!(status, Status::InvalidChild | Status::Malformed) { self.invalidate_poll(i, Status::Revoked); self.sessions[i].frozen = true; }
1120                continue;
1121            }
1122            if self.sessions[i].paste.as_ref().is_some_and(|p| p.expires<=timestamp) {self.sessions[i].paste=None;}
1123            if self.sessions[i].physical_paste.is_some() && self.sessions[i].physical_paste_expires <= timestamp {
1124                self.sessions[i].physical_paste = None;
1125                self.sessions[i].physical_paste_discard = !self.sessions[i].physical_paste_complete;
1126                self.sessions[i].physical_paste_complete = false;
1127                self.physical_loss(i, 3);
1128            }
1129            if self.sessions[i].physical_paste_complete {
1130                if let Some(staging) = self.sessions[i].physical_paste.take() {
1131                    match self.admit_paste(i, &staging) {
1132                        Ok(_) => self.sessions[i].physical_paste_complete = false,
1133                        Err(Status::WouldBlock) => self.sessions[i].physical_paste = Some(staging),
1134                        Err(_) => { self.sessions[i].physical_paste_complete = false; self.physical_loss(i, 3); }
1135                    }
1136                }
1137            }
1138            if !self.sessions[i].frozen && self.sessions[i].mode & 1 == 0 {
1139                let mut events = Scrub::<4000>::new();
1140                match sys::monotonic_ns().and_then(|ns| self.sessions[i].decoder.expire(ns, &mut events)) {
1141                    Ok(n) if n != 0 => { if self.serial_events(i, &events[..n]).is_err() { self.abort_physical_paste(i); self.physical_loss(i, 3); } }
1142                    Err(_) => { self.abort_physical_paste(i); self.physical_loss(i, 3); }
1143                    _ => {}
1144                }
1145            }
1146            self.drain(i);
1147            if let Some(poll)=self.sessions[i].poll.take() {
1148                let mut bytes=Scrub::<{ MAX_SERVICE_RECORD_LEN - HEADER_LEN }>::new();
1149                let status=if self.sessions[i].epoch!=poll.epoch || self.sessions[i].lease!=poll.lease {Some(Err(Status::WrongEpoch))} else {
1150                    match self.poll_batch(i,poll.outcome,poll.cursor,poll.maximum,&mut bytes) {Ok(Some(n))=>Some(Ok(n)),Err(s)=>Some(Err(s)),Ok(None) if timestamp>=poll.expires || (poll.deadline!=u64::MAX && now_ns()>=poll.deadline)=>Some(Err(Status::TimedOut)),Ok(None)=>None}
1151                };
1152                if let Some(status)=status {match status {Ok(n)=>self.finish(poll.outcome,Status::Ok,&bytes[..n]),Err(s)=>self.finish(poll.outcome,s,&[])} let o=&mut self.outcomes[poll.outcome]; o.observed|=Self::send(&o.result,0,poll.token);}
1153                else {self.sessions[i].poll=Some(poll);}
1154            }
1155        }
1156    }

1186    fn serial_events(&mut self, i: usize, bytes: &[u8]) -> Result<(), Status> {
1187        let mut offset = 0;
1188        while offset < bytes.len() {
