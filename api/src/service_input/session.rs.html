289/// Selectors whose request body can carry typed or raw secrets (SubmitEvents, WritePaste, SubmitRaw).
290/// Their results are only a sequence/count, so nothing beyond the status is ever replayable.
291fn secret_bearing(selector: u16) -> bool {
292    matches!(selector, 13 | 15 | 20)
293}
294fn wipe(bytes: &mut Vec<u8>) {
295    zeroize_vec(bytes);
296    *bytes = Vec::new();
297}
298/// The record an outcome retains for `request`. A secret-bearing request is sealed at the
299/// moment it is consumed: only the SHA-256 of the exact record is kept, never its plaintext,
300/// not even while the request executes (the body is read from the receive buffer, which the
301/// caller wipes). The digest still lets an identical re-send resolve to the retained (or
302/// pending) outcome and a changed body under the same ID be refused (annex: Malformed).
303fn retained(selector: u16, request: &[u8]) -> Result<(Vec<u8>, bool), Status> {
304    let sealed = secret_bearing(selector);
305    let mut copy = Vec::new();
306    if sealed {
307        copy.try_reserve_exact(32)
308            .map_err(|_| Status::QuotaExceeded)?;
309        copy.extend_from_slice(&ostd::session::sha256(request));
310    } else {
311        reserve(&mut copy, request.len())?;
312        copy.extend_from_slice(request);
313    }
314    Ok((copy, sealed))
315}
316impl Outcome {
317    /// Wipe and release the retained request and result bodies (the fence stays elsewhere).
318    fn scrub(&mut self) {
319        wipe(&mut self.request);
320        wipe(&mut self.result);
321        self.sealed = false;
322    }
323    /// Does `bytes` exactly equal the record this outcome retained? `digest` caches the hash of
324    /// `bytes` across candidates. Selector and request ID are header fields (not secret).
325    fn matches(&self, bytes: &[u8], digest: &mut Option<[u8; 32]>) -> bool {
326        if self.request.is_empty() {
327            return false;
328        }
329        if !self.sealed {
330            return self.request == bytes;
331        }
332        if bytes.len() < HEADER_LEN
333            || bytes[4..6] != self.selector.to_le_bytes()
334            || bytes[16..24] != self.id.to_le_bytes()
335        {
336            return false;
337        }
338        let digest = digest.get_or_insert_with(|| ostd::session::sha256(bytes));
339        self.request[..] == digest[..]
340    }
341}
342impl Drop for Outcome {
343    fn drop(&mut self) {
344        self.scrub();
345    }
346}

360    pub fn new() -> Self {
361        Self {

363            frontends: Vec::new(),
364            outcomes: Vec::new(),
365            fences: Vec::new(),

367            issuer: None,

373    fn reference(&mut self) -> Result<Ref, Status> {
374        let id = self.next_ref;
375        self.next_ref = id.checked_add(1).ok_or(Status::QuotaExceeded)?;
376        Ok(Ref { id, generation: 1 })
377    }
378    fn binding(&mut self) -> Result<ServiceBinding, Status> {
379        let mut bytes = [0; 24];
380        let binding = sys_lookup_service_bound(api::syscall::service::INPUT, &mut bytes)
381            .ok_or(Status::Revoked)?;
382        if let Some(old) = self.issuer {
383            if old != binding {
384                return Err(Status::Revoked);
385            }
386        } else {
387            self.issuer = Some(binding);
388        }
389        Ok(binding)
390    }
391    fn kernel(&mut self, op: KernelOp, body: &[u8], reply: &mut [u8]) -> Result<usize, Status> {
392        self.binding()?;
393        let query = matches!(
394            op,
395            KernelOp::InspectPreparedGrant
396                | KernelOp::InspectEndpoint
397                | KernelOp::QueryReceiptChild
398                | KernelOp::QueryOperation
399        ) || (matches!(
400            op,
401            KernelOp::ResolveBindReceipt | KernelOp::ResolveBootstrap
402        ) && body.get(16..20).is_some_and(|v| v == [0; 4]));
403        let header = if query {
404            // Nonmutating selectors retain no outcome and consume no request ID; they must
405            // not mint SDK correlation lanes (bounded, never retired for foreign operations).
406            sys::kernel_request(op, 1, body, reply)?
407        } else {
408            sys::kernel_call(op, self.kernel_operation, true, body, reply)?
409        };
410        if header.status != Status::Ok {
411            return Err(header.status);
412        }
413        Ok(header.total_len as usize)
414    }
415    fn resolve_bootstrap(
416        &mut self,
417        bootstrap: Ref,
418        action: u32,
419        session: Ref,
420    ) -> Result<BootstrapResolution, Status> {
421        let mut body = [0; 40];
422        let mut w = Writer::new(&mut body);
423        w.reference(bootstrap)?;
424        w.u32(action)?;
425        w.u32(0)?;
426        w.reference(session)?;
427        let mut reply = [0; 224];
428        let len = self.kernel(KernelOp::ResolveBootstrap, &body, &mut reply)?;
429        let mut r = Reader::new(&reply[HEADER_LEN..len]);
430        let state = r.u32()?;
431        reserved(r.u32()?)?;
432        let controller = r.principal()?;
433        let operation = r.reference()?;
434        let endpoints = [
435            r.reference()?,
436            r.reference()?,
437            r.reference()?,
438            r.reference()?,
439        ];
440        let outputs = [r.reference()?, r.reference()?];
441        let writers = [r.reference()?, r.reference()?];
442        r.finish()?;
443        Ok((state, controller, operation, endpoints, outputs, writers))
444    }
445    fn index(&self, reference: Ref) -> Result<usize, Status> {

450    }
451    fn controller(&self, index: usize, p: Principal) -> Result<(), Status> {

454        } else {

458    fn reader(&self, index: usize, p: Principal, lease: Ref) -> Result<(), Status> {
459        let s = &self.sessions[index];
460        if s.frozen {
461            return Err(Status::WouldBlock);
462        }
463        if (s.attachment.is_none() && !s.detached) || s.reader != p || s.lease != lease {
464            Err(Status::Denied)
465        } else {
466            Ok(())
467        }
468    }
469    fn frontend(
470        &self,
471        index: usize,
472        p: Principal,
473        attachment: Ref,
474        epoch: u64,
475    ) -> Result<(), Status> {
476        let s = &self.sessions[index];
477        if s.frozen {
478            return Err(Status::WouldBlock);
479        }
480        if s.epoch != epoch {
481            return Err(Status::WrongEpoch);
482        }
483        match s.attachment.as_ref() {
484            Some(a) if a.reference == attachment && a.frontend == p => Ok(()),
485            // A sel-25 holder presenting a detached or superseded attachment holds a stale selector.
486            _ if s.holder == Some(p) => Err(Status::Stale),
487            _ => Err(Status::Denied),
488        }
489    }
490    /// Sel-25 frontends cannot observe the controller's foreground churn: they send
491    /// epoch 0 and act on the current epoch. Kind-0 self-attachments keep the exact rule.
492    fn wire_epoch(&self, index: usize, epoch: u64) -> Result<u64, Status> {
493        let s = &self.sessions[index];
494        match (s.holder.is_some(), epoch) {
495            (false, epoch) => Ok(epoch),
496            (true, 0) => Ok(s.epoch),
497            (true, _) => Err(Status::WrongEpoch),
498        }
499    }
500    /// The request lane of a sel-25 frontend holder is its frontend's registered operation.
501    fn holder_lane(s: &Session, caller: Principal) -> Option<Ref> {

504    }
505    /// Does a live sel-25 holder or frontend record still use `operation`?
506    fn frontend_lane(&self, operation: Ref) -> bool {
507        self.sessions.iter().any(|s| {
508            !s.closed && s.holder.is_some() && !s.orphaned && s.holder_operation == operation
509        }) || self.frontends.iter().any(|f| f.operation == operation)
510    }

513            .iter()
514            .position(|f| f.reference == reference)
515            .ok_or(Status::Stale)
516    }

520            Ok(i)
521        } else {
522            Err(Status::Denied)
523        }
524    }
525    fn epoch(&self, index: usize, epoch: u64) -> Result<(), Status> {
526        if self.sessions[index].epoch == epoch {
527            Ok(())
528        } else {
529            Err(Status::WrongEpoch)
530        }

532    fn operation_for(&self, selector: u16, body: &[u8], caller: Principal) -> Result<Ref, Status> {
533        let mut r = Reader::new(body);
534        let reference = r.reference()?;
535        // Trust lanes: Register is fenced on its challenge Ref, Collect on its confirmation Ref.
536        if matches!(selector, 1 | 18 | 19 | 23 | 25 | 29 | 31) {
537            return Ok(reference);
538        }
539        if matches!(selector, 24 | 26..=28) {
540            return self
541                .frontends
542                .iter()
543                .find(|f| f.reference == reference)
544                .map(|f| f.operation)
545                .ok_or(Status::Stale);
546        }
547        if (15..=17).contains(&selector) {
548            let s = self
549                .sessions
550                .iter()
551                .find(|s| s.paste.as_ref().is_some_and(|p| p.reference == reference))
552                .ok_or(Status::Stale)?;
553            return Ok(Self::holder_lane(s, caller).unwrap_or(s.operation));
554        }
555        let s = &self.sessions[self.index(reference)?];
556        if let Some(lane) = Self::holder_lane(s, caller) {
557            return Ok(lane);
558        }
559        Ok(s.foreground
560            .as_ref()
561            .filter(|f| f.child.principal == caller && !f.restored)
562            .map(|f| f.operation)
563            .unwrap_or(s.operation))
564    }
565    fn participant(&self, index: usize, p: Principal) -> bool {
566        let s = &self.sessions[index];
567        s.controller == p
568            || s.reader == p
569            || s.holder == Some(p)
570            || s.attachment.as_ref().is_some_and(|a| a.frontend == p)
571            || s.foreground
572                .as_ref()
573                .is_some_and(|f| f.child.principal == p)
574            || s.modes
575                .iter()
576                .chain(s.baseline_modes.iter())
577                .any(|m| m.holder == p)
578    }
579    /// Only principals already tied to the target session may allocate retained
580    /// outcome/fence capacity; a guessed session reference confers no bookkeeping budget.
581    fn admit(
582        &mut self,
583        selector: u16,
584        body: &[u8],
585        p: Principal,
586        sender: usize,
587    ) -> Result<(), Status> {
588        let mut r = Reader::new(body);
589        match selector {
590            29 | 31 => self.admit_trust(selector, body, p, sender),
591            1 => {
592                let operation = r.reference()?;
593                let bootstrap = r.reference()?;
594                self.kernel_operation = operation;
595                let (state, controller, recorded, ..) =
596                    self.resolve_bootstrap(bootstrap, 0, ZERO)?;
597                if state > 1 || controller != p || recorded != operation {
598                    return Err(Status::Denied);
599                }
600                Ok(())
601            }
602            15..=17 => {
603                let reference = r.reference()?;
604                let i = self
605                    .sessions
606                    .iter()
607                    .position(|s| {
608                        !s.closed && s.paste.as_ref().is_some_and(|x| x.reference == reference)
609                    })
610                    .ok_or(Status::Stale)?;
611                if self.participant(i, p) {
612                    Ok(())
613                } else {
614                    Err(Status::Denied)
615                }
616            }
617            // Any attested Cell may register a frontend on a live operation of its own lane.
618            23 => {
619                let operation = r.reference()?;
620                self.kernel_operation = operation;
621                if self.query_operation(operation)? != 0 {
622                    return Err(Status::Cancelled);
623                }
624                Ok(())
625            }
626            24 | 26..=28 => {
627                self.owned_frontend(r.reference()?, p)?;
628                Ok(())
629            }
630            // Only the exact child bound by sel 24 may create the frontend's session.
631            25 => {
632                r.reference()?;
633                r.reference()?;
634                let f = &self.frontends[self.frontend_index(r.reference()?)?];
635                if f.child.is_some_and(|c| c.principal == p) {
636                    Ok(())
637                } else {
638                    Err(Status::Denied)
639                }
640            }
641            18 | 19 => Ok(()),
642            _ => {
643                let i = self.index(r.reference()?)?;
644                if self.participant(i, p) {
645                    Ok(())
646                } else {
647                    Err(Status::Denied)
648                }
649            }
650        }
651    }
652    /// Drop request fences whose operation no live session or live foreground child uses
653    /// and that retain no outcome. A dead principal/operation cannot present an old ID.
654    fn prune(&mut self) {
655        let sessions = &self.sessions;
656        let outcomes = &self.outcomes;
657        let frontends = &self.frontends;
658        let challenges = &self.challenges;
659        self.fences.retain(|f| {
660            sessions.iter().any(|s| {
661                !s.closed
662                    && (s.operation == f.operation
663                        || (s.holder.is_some() && !s.orphaned && s.holder_operation == f.operation)
664                        || s.foreground
665                            .as_ref()
666                            .is_some_and(|g| !g.restored && g.operation == f.operation))
667            }) || frontends.iter().any(|x| x.operation == f.operation)
668                || trust::challenge_lane(challenges, f.caller, f.operation)
669                || outcomes.iter().any(|o| {
670                    !o.request.is_empty() && o.caller == f.caller && o.operation == f.operation
671                })
672        });
673    }
674    fn drop_outcomes(&mut self, operation: Ref) {
675        for o in self
676            .outcomes
677            .iter_mut()
678            .filter(|o| o.operation == operation)
679        {
680            o.scrub();
681            o.pending = false;
682        }
683    }
684    /// An explicitly closed session's retained outcomes die with it, except the outcome of the
685    /// close request itself (kept so its replay and acknowledgement still resolve). Dropped
686    /// outcomes are marked observed so they can never block that acknowledgement.
687    fn release_closed(&mut self, operations: &[Option<Ref>], keep: usize) {
688        for operation in operations.iter().flatten().copied() {
689            if self.frontend_lane(operation)
690                || self.sessions.iter().any(|s| {
691                    !s.closed
692                        && (s.operation == operation
693                            || s.foreground.as_ref().is_some_and(|f| {
694                                !(f.restored && f.acknowledged) && f.operation == operation
695                            }))
696                })
697            {
698                continue;
699            }
700            for (n, o) in self.outcomes.iter_mut().enumerate() {
701                if n == keep || o.operation != operation {
702                    continue;
703                }
704                o.scrub();
705                o.pending = false;
706                o.observed = true;
707            }
708        }
709        self.prune();
710    }
711    /// Retire the SDK correlation lane of an operation this service no longer calls on, unless
712    /// another live session or unfinished foreground still uses it.
713    fn retire_lane(&self, operation: Ref) {
714        let busy = self.frontend_lane(operation)
715            || self.sessions.iter().any(|s| {
716                !s.closed
717                    && (s.operation == operation
718                        || s.foreground.as_ref().is_some_and(|f| {
719                            !(f.restored && f.acknowledged) && f.operation == operation
720                        }))
721            });
722        if !busy {
723            ostd::session::retire_operation(operation);
724        }
725    }
726    /// The foreground child is terminal and restored: its retained outcomes/fences die with it.
727    fn retire_foreground(&mut self, i: usize) {
728        let Some(f) = self.sessions[i].foreground.as_ref() else {
729            return;
730        };
731        let (principal, operation) = (f.child.principal, f.operation);
732        for o in self
733            .outcomes
734            .iter_mut()
735            .filter(|o| o.caller == principal && o.operation == operation)
736        {
737            o.scrub();
738            o.pending = false;
739        }
740        self.prune();
741        self.retire_lane(operation);
742    }
743    fn record(
744        &mut self,
745        header: RequestHeaderV1,
746        request: &[u8],
747        p: Principal,
748        operation: Ref,
749    ) -> Result<(usize, bool), Status> {
750        if let Some(i) = self.outcomes.iter().position(|o| {
751            !o.request.is_empty()
752                && o.caller == p
753                && o.operation == operation
754                && o.id == header.request_id
755        }) {
756            if !self.outcomes[i].matches(request, &mut None) {
757                return Err(Status::Malformed);
758            }
759            return Ok((i, true));
760        }
761        let fence = if let Some(i) = self
762            .fences
763            .iter()
764            .position(|f| f.caller == p && f.operation == operation)
765        {
766            i
767        } else {
768            if self.fences.len() >= 512 {
769                self.prune();
770            }
771            if self.fences.len() >= 512 {
772                return Err(Status::QuotaExceeded);
773            }
774            reserve(&mut self.fences, 1)?;
775            self.fences.push(Fence {
776                caller: p,
777                operation,
778                next: 1,
779                acknowledged: 0,
780            });
781            self.fences.len() - 1
782        };
783        if header.request_id != self.fences[fence].next {
784            return Err(Status::UnknownRequest);
785        }
786        if self
787            .outcomes
788            .iter()
789            .filter(|o| o.caller == p && o.operation == operation && !o.request.is_empty())
790            .count()
791            >= 64
792            || self
793                .outcomes
794                .iter()
795                .filter(|o| o.caller == p && !o.request.is_empty())
796                .count()
797                >= 128
798        {
799            return Err(Status::QuotaExceeded);
800        }
801        let next = header
802            .request_id
803            .checked_add(1)
804            .ok_or(Status::QuotaExceeded)?;
805        let (copy, sealed) = retained(header.selector, request)?;
806        let mut result = Vec::new();
807        reserve(&mut result, MAX_SERVICE_RECORD_LEN)?;
808        let slot = self.outcomes.iter().position(|o| o.request.is_empty());
809        if slot.is_none() && self.outcomes.len() >= 1024 {
810            return Err(Status::QuotaExceeded);
811        }
812        if slot.is_none() {
813            reserve(&mut self.outcomes, 1)?;
814        }
815        let record = Outcome {
816            caller: p,
817            operation,
818            id: header.request_id,
819            selector: header.selector,
820            request: copy,
821            sealed,
822            result,
823            observed: false,
824            pending: true,
825            poll_session: ZERO,
826            poll_epoch: 0,
827            cursor: 0,
828        };
829        let index = if let Some(slot) = slot {
830            self.outcomes[slot] = record;
831            slot
832        } else {
833            self.outcomes.push(record);
834            self.outcomes.len() - 1
835        };
836        self.fences[fence].next = next;
837        Ok((index, false))
838    }
839    fn finish(&mut self, index: usize, status: Status, payload: &[u8]) {
840        let o = &mut self.outcomes[index];
841        o.result.resize(HEADER_LEN + payload.len(), 0);
842        let h = ResultHeaderV1 {
843            selector: o.selector,
844            status,
845            total_len: o.result.len() as u32,
846            request_id: o.id,
847        };
848        let _ = h.encode(&mut o.result);
849        o.result[HEADER_LEN..].copy_from_slice(payload);
850        o.pending = false;
851    }
852    fn send(bytes: &[u8], sender: usize, token: usize) -> bool {
853        let mut envelope = Scrub::<{ api::ipc::IPC_BUF_SIZE }>::new();
854        let Ok(encoded) =
855            api::ipc::encode(&api::ipc::InputResponse::SessionV1(bytes), &mut envelope)
856        else {
857            return false;
858        };
859        if token != 0 {
860            sys_ipc_reply(token, encoded) == 0
861        } else {
862            matches!(
863                sys_try_send(sender, encoded),
864                ostd::syscall::SyscallResult::Ok(0)
865            )
866        }
867    }
868    fn error(header: RequestHeaderV1, status: Status, sender: usize, token: usize) {
869        let mut bytes = [0; HEADER_LEN];
870        let _ = ResultHeaderV1 {
871            selector: header.selector,
872            status,
873            total_len: HEADER_LEN as u32,
874            request_id: header.request_id,
875        }
876        .encode(&mut bytes);
877        Self::send(&bytes, sender, token);
878    }
879    fn reply_outcome(
880        &mut self,
881        index: usize,
882        header: RequestHeaderV1,
883        caller: Principal,
884        sender: usize,
885        token: usize,
886    ) {
887        let outcome = &self.outcomes[index];
888        let denial = if outcome.pending {
889            Some(Status::WouldBlock)
890        } else if outcome.poll_session != ZERO {
891            match self.sessions.iter().find(|s| {
892                !s.closed
893                    && s.reference == outcome.poll_session
894                    && s.epoch == outcome.poll_epoch
895                    && s.reader == caller
896            }) {
897                Some(session) if session.frozen => Some(Status::WouldBlock),
898                Some(_) => None,
899                None => Some(Status::WrongEpoch),
900            }
901        } else {
902            None
903        };
904        if let Some(status) = denial {
905            Self::error(header, status, sender, token);
906            return;
907        }
908        let outcome = &mut self.outcomes[index];
909        outcome.observed |= Self::send(&outcome.result, sender, token);
910    }
911    pub fn request(&mut self, bytes: &[u8], caller: Option<CallerIdentity>, sender: usize) {
912        let token = sys_ipc_current();
913        let token = if token > 0 { token as usize } else { 0 };
914        let header = match RequestHeaderV1::decode(bytes) {
915            Ok(header) => header,
916            Err(status) => {
917                if bytes.len() >= HEADER_LEN {
918                    let header = RequestHeaderV1 {
919                        selector: u16::from_le_bytes([bytes[4], bytes[5]]),
920                        total_len: bytes.len() as u32,
921                        request_id: u64::from_le_bytes(bytes[16..24].try_into().unwrap_or([0; 8])),
922                    };
923                    Self::error(header, status, sender, token);
924                }
925                return;
926            }
927        };
928        if !(1..=31).contains(&header.selector) {
929            Self::error(header, Status::UnknownOperation, sender, token);
930            return;
931        }
932        let Some(caller) = caller.filter(|c| c.sender_tid == sender as u64 && c.generation != 0)
933        else {
934            Self::error(header, Status::Denied, sender, token);
935            return;
936        };
937        let p = principal(caller);
938        if bytes.len() > MAX_SERVICE_RECORD_LEN {
939            Self::error(header, Status::Malformed, sender, token);
940            return;
941        }
942        if let Err(status) = self.binding() {
943            Self::error(header, status, sender, token);
944            return;
945        }
946        self.maintain();
947        // Resolve exact retained bytes before resource lookup: CommitPaste,
948        // AbortPaste and CloseSession remove the live selector they mutate.
949        let mut digest = None;
950        if let Some(index) = self
951            .outcomes
952            .iter()
953            .position(|o| o.caller == p && o.matches(bytes, &mut digest))
954        {
955            self.reply_outcome(index, header, p, sender, token);
956            return;
957        }
958        let body = &bytes[HEADER_LEN..];
959        let operation = match self.operation_for(header.selector, body, p) {
960            Ok(v) => v,
961            Err(status) => {
962                Self::error(header, status, sender, token);
963                return;
964            }
965        };
966        if let Err(status) = self.admit(header.selector, body, p, sender) {
967            Self::error(header, status, sender, token);
968            return;
969        }
970        if header.selector == 18 || header.selector == 19 {
971            self.query_ack(header, body, p, sender, token);
972            return;
973        }
974        // Nonmutating reads (QueryCapabilities, QueryFrontend) retain no outcome and consume no ID.
975        if header.selector == 12 || header.selector == 26 {
976            let mut reply = [0; MAX_SERVICE_RECORD_LEN - HEADER_LEN];
977            let result = self.execute(header.selector, body, p, usize::MAX, token, &mut reply);
978            let mut record = [0; MAX_SERVICE_RECORD_LEN];
979            let (status, length) = match result {
980                Ok(Some(n)) => (Status::Ok, n),
981                Ok(None) => (Status::Malformed, 0),
982                Err(s) => (s, 0),
983            };
984            let _ = ResultHeaderV1 {
985                selector: header.selector,
986                status,
987                total_len: (HEADER_LEN + length) as u32,
988                request_id: header.request_id,
989            }
990            .encode(&mut record[..HEADER_LEN]);
991            record[HEADER_LEN..HEADER_LEN + length].copy_from_slice(&reply[..length]);
992            Self::send(&record[..HEADER_LEN + length], sender, token);
993            return;
994        }
995        let (outcome, replay) = match self.record(header, bytes, p, operation) {
996            Ok(v) => v,
997            Err(status) => {
998                Self::error(header, status, sender, token);
999                return;
1000            }
1001        };
1002        if replay {
1003            self.reply_outcome(outcome, header, p, sender, token);
1004            return;
1005        }
1006        let mut reply = Scrub::<{ MAX_SERVICE_RECORD_LEN - HEADER_LEN }>::new();
1007        match self.execute(header.selector, body, p, outcome, token, &mut reply) {
1008            Ok(Some(len)) => self.finish(outcome, Status::Ok, &reply[..len]),
1009            Ok(None) => return,
1010            Err(status) => self.finish(outcome, status, &[]),
1011        }
1012        let o = &mut self.outcomes[outcome];
1013        o.observed |= Self::send(&o.result, sender, token);
1014    }
1015    fn query_ack(
1016        &mut self,
1017        h: RequestHeaderV1,
1018        body: &[u8],
1019        p: Principal,
1020        sender: usize,
1021        token: usize,
1022    ) {
1023        let mut r = Reader::new(body);
1024        let parsed = (|| {
1025            let op = r.reference()?;
1026            let through = r.u64()?;
1027            r.finish()?;
1028            Ok::<_, Status>((op, through))
1029        })();
1030        let (op, id) = match parsed {
1031            Ok(v) => v,
1032            Err(s) => {
1033                Self::error(h, s, sender, token);
1034                return;
1035            }
1036        };
1037        if h.selector == 19 {
1038            if let Some(index) = self.outcomes.iter().position(|o| {
1039                !o.request.is_empty() && o.caller == p && o.operation == op && o.id == id
1040            }) {
1041                self.reply_outcome(index, h, p, sender, token);
1042            } else {
1043                Self::error(h, Status::UnknownRequest, sender, token);
1044            }
1045            return;
1046        }
1047        let Some(fence) = self
1048            .fences
1049            .iter()
1050            .position(|f| f.caller == p && f.operation == op)
1051        else {
1052            Self::error(h, Status::UnknownRequest, sender, token);
1053            return;
1054        };
1055        if id <= self.fences[fence].acknowledged {
1056            Self::error(h, Status::Ok, sender, token);
1057            return;
1058        }
1059        if id >= self.fences[fence].next
1060            || self.outcomes.iter().any(|o| {
1061                o.caller == p && o.operation == op && o.id <= id && (!o.observed || o.pending)
1062            })
1063        {
1064            Self::error(h, Status::UnknownRequest, sender, token);
1065            return;
1066        }
1067        for o in self
1068            .outcomes
1069            .iter_mut()
1070            .filter(|o| o.caller == p && o.operation == op && o.id <= id)
1071        {
1072            if o.poll_session != ZERO {
1073                if let Some(s) = self
1074                    .sessions
1075                    .iter_mut()
1076                    .find(|s| s.reference == o.poll_session && s.epoch == o.poll_epoch)
1077                {
1078                    while s.typed.front().is_some_and(|e| e.sequence <= o.cursor) {
1079                        s.typed.pop_front();
1080                    }
1081                    s.delivered_cursor = s.delivered_cursor.max(o.cursor);
1082                }
1083            }
1084            o.scrub();
1085        }
1086        // Stable outcome indices are used by saved poll tokens. Tombstones retain
1087        // their ID fence until the operation/principal is retired.
1088        self.fences[fence].acknowledged = self.fences[fence].acknowledged.max(id);
1089        Self::error(h, Status::Ok, sender, token);
1090    }
1091    fn execute(
1092        &mut self,
1093        selector: u16,
1094        body: &[u8],
1095        p: Principal,
1096        outcome: usize,
1097        token: usize,
1098        out: &mut [u8],
1099    ) -> Result<Option<usize>, Status> {
1100        let mut r = Reader::new(body);
1101        let mut w = Writer::new(out);
1102        if selector == 1 {
1103            let operation = r.reference()?;
1104            let bootstrap = r.reference()?;
1105            let kind = r.u8()?;
1106            let flags = r.u8()?;
1107            let rows = r.u16()?;
1108            let cols = r.u16()?;
1109            reserved(r.u16()? as u32)?;
1110            r.finish()?;
1111            self.kernel_operation = operation;
1112            // GUI sessions are admitted only through RegisterFrontend/CreateForFrontend.
1113            if flags != 0 {
1114                return Err(Status::InvalidFlags);
1115            }
1116            if kind > 2 {
1117                return Err(Status::Malformed);
1118            }
1119            if kind != 0 {
1120                return Err(Status::Unsupported);
1121            }
1122            dimensions(rows, cols)?;
1123            let (i, endpoints) = self.create_session(
1124                operation,
1125                bootstrap,
1126                p,
1127                SessionShape {
1128                    kind: fe::FRONTEND_KIND_SERIAL,
1129                    rows,
1130                    cols,
1131                },
1132                None,
1133            )?;
1134            let s = &self.sessions[i];
1135            w.reference(s.reference)?;
1136            w.principal(p)?;
1137            w.reference(s.lease)?;
1138            w.u64(1)?;
1139            w.u32(2)?;
1140            w.u32(0)?;
1141            for endpoint in endpoints {
1142                w.reference(endpoint)?;
1143            }
1144            return Ok(Some(w.position()));
1145        }
1146        if (23..=28).contains(&selector) {
1147            return self
1148                .frontend_operation(selector, body, p, &mut w)
1149                .map(|_| Some(w.position()));
1150        }
1151        if (15..=17).contains(&selector) {
1152            return self
1153                .paste_operation(selector, r, p, &mut w)
1154                .map(|_| Some(w.position()));
1155        }
1156        if (29..=31).contains(&selector) {
1157            return self
1158                .trust_operation(selector, body, p, &mut w)
1159                .map(|_| Some(w.position()));
1160        }
1161        let reference = r.reference()?;
1162        let i = self.index(reference)?;
1163        self.kernel_operation = self.sessions[i].operation;
1164        match selector {
1165            2 => {
1166                let frontend = r.principal()?;
1167                let operation = r.reference()?;
1168                let input = r.reference()?;
1169                let output = r.reference()?;
1170                r.finish()?;
1171                self.controller(i, p)?;
1172                if self.sessions[i].attachment.is_some() {
1173                    return Err(Status::WouldBlock);
1174                }
1175                // A sel-25 session's frontend returns only through ReattachFrontend.
1176                if self.sessions[i].holder.is_some() {
1177                    return Err(Status::Denied);
1178                }
1179                if operation != self.sessions[i].operation || frontend != p {
1180                    return Err(Status::Denied);
1181                }
1182                let (_, _, _, endpoints, outputs, _) =
1183                    self.resolve_bootstrap(self.sessions[i].bootstrap, 0, ZERO)?;
1184                if input != endpoints[0] || output != outputs[0] {
1185                    return Err(Status::WrongSession);
1186                }
1187                // The admitted headless Serial frontend is the attested
1188                // controller, not an arbitrary payload-named process. GUI/SSH
1189                // and foreign-process forwarding require their own live bridge.
1190                let attachment = self.reference()?;
1191                self.sessions[i].attachment = Some(Attachment {
1192                    reference: attachment,
1193                    frontend,
1194                });
1195                if frontend == p && self.serial.is_none() && !compositor_registered() {
1196                    self.serial = Some(reference);
1197                }
1198                w.reference(attachment)?;
1199            }
1200            3 => {
1201                let expected = r.u64()?;
1202                let child = r.child()?;
1203                let endpoints = [
1204                    r.reference()?,
1205                    r.reference()?,
1206                    r.reference()?,
1207                    r.reference()?,
1208                ];
1209                let writers = [r.reference()?, r.reference()?];
1210                r.finish()?;
1211                self.controller(i, p)?;
1212                self.epoch(i, expected)?;
1213                self.prepare_foreground(i, child, expected, endpoints, writers, &mut w)?;
1214            }
1215            4 => {
1216                let lease = r.reference()?;
1217                let child = r.child()?;
1218                let receipt = r.reference()?;
1219                r.finish()?;
1220                self.controller(i, p)?;
1221                let s = &self.sessions[i];
1222                let f = s.foreground.as_ref().ok_or(Status::Stale)?;
1223                if f.lease != lease || f.child != child || f.receipt != receipt {
1224                    return Err(Status::Denied);
1225                }
1226                if !f.committed {
1227                    return Err(if f.restored {
1228                        Status::Cancelled
1229                    } else {
1230                        Status::WouldBlock
1231                    });
1232                }
1233                w.u64(s.epoch)?;
1234                w.principal(s.reader)?;
1235            }
1236            5 => {
1237                let lease = r.reference()?;
1238                let expected = r.u64()?;
1239                let child = r.child()?;
1240                r.finish()?;
1241                self.controller(i, p)?;
1242                let f = self.sessions[i].foreground.as_ref().ok_or(Status::Stale)?;
1243                if f.lease != lease || f.child != child || f.next != expected {
1244                    return Err(Status::WrongEpoch);
1245                }
1246                if !f.restored {
1247                    self.restore(i)?;
1248                }
1249                w.u64(self.sessions[i].epoch)?;
1250                w.principal(self.sessions[i].reader)?;
1251            }
1252            6 => {
1253                let attachment = r.reference()?;
1254                let rows = r.u16()?;
1255                let cols = r.u16()?;
1256                reserved(r.u32()?)?;
1257                r.finish()?;
1258                dimensions(rows, cols)?;
1259                self.frontend(i, p, attachment, self.sessions[i].epoch)?;
1260                let mut payload = [0; 4];
1261                payload[..2].copy_from_slice(&rows.to_le_bytes());
1262                payload[2..].copy_from_slice(&cols.to_le_bytes());
1263                let sequence = self.control(i, 5, &payload)?;
1264                self.sessions[i].rows = rows;
1265                self.sessions[i].cols = cols;
1266                w.u64(sequence)?;
1267            }
1268            7 => {
1269                let lease = r.reference()?;
1270                let epoch = r.u64()?;
1271                let mask = r.u32()?;
1272                let value = r.u32()?;
1273                r.finish()?;
1274                self.reader(i, p, lease)?;
1275                self.epoch(i, epoch)?;
1276                if mask & !31 != 0 || value & !mask != 0 {
1277                    return Err(Status::InvalidFlags);
1278                }
1279                if mask & !MODE_MASK != 0 {
1280                    return Err(Status::Unsupported);
1281                }
1282                // A physical UART's local echo is outside this owner's control: echo-off cannot be honored there.
1283                if mask & 2 != 0 && value & 2 == 0 && self.serial == Some(reference) {
1284                    return Err(Status::Unsupported);
1285                }
1286                if self.sessions[i].modes.len() + self.sessions[i].baseline_modes.len() >= 64 {
1287                    return Err(Status::QuotaExceeded);
1288                }
1289                if !self.sessions[i].raw.is_empty()
1290                    || !self.sessions[i].typed.is_empty()
1291                    || self.sessions[i].paste.is_some()
1292                    || self.sessions[i].physical_paste.is_some()
1293                    || self.sessions[i].poll.is_some()
1294                {
1295                    return Err(Status::WouldBlock);
1296                }
1297                let mode_lease = self.reference()?;
1298                let s = &mut self.sessions[i];
1299                s.modes.push(Mode {
1300                    reference: mode_lease,
1301                    holder: p,
1302                    epoch,
1303                    mask,
1304                    value,
1305                });
1306                s.mode = (s.mode & !mask) | value;
1307                w.reference(mode_lease)?;
1308                w.u32(s.mode)?;
1309                w.u32(0)?;

1316                }

1318                {

1333                            return Err(Status::Denied);
1334                        }

1336                    } else {
1337                        return Err(Status::Stale);
1338                    }
1339                }

1355                reserved(r.u16()? as u32)?;
1356                reserved(r.u32()?)?;

1361                    return Err(Status::Malformed);
1362                }

1364                    return Err(Status::WrongDirection);
1365                }

1367                    || self.outcomes.iter().enumerate().any(|(n, o)| {
1368                        n != outcome
1369                            && o.poll_session == reference
1370                            && o.poll_epoch == self.sessions[i].epoch

1372                    })

1374                    return Err(Status::WouldBlock);
1375                }
1376                if cursor != self.sessions[i].delivered_cursor {
1377                    return Err(Status::Malformed);
1378                }

1380                self.outcomes[outcome].poll_epoch = self.sessions[i].epoch;

1382                    return Ok(Some(len));
1383                }

1385                    return Err(Status::WouldBlock);
1386                }

1388                    return Err(Status::Unsupported);
1389                }

1393                }

1397                    token,
1398                    epoch: self.sessions[i].epoch,
1399                    lease,
1400                    cursor,
1401                    maximum,
1402                    expires: timestamp.saturating_add(900),

1416                    {
1417                        return Err(Status::Stale);
1418                    }

1424                        return Err(Status::Denied);
1425                    }

1427                        .attachment
1428                        .as_ref()
1429                        .is_some_and(|a| a.reference == attachment)
1430                    {
1431                        return Err(Status::Stale);
1432                    }

1437                    }
1438                    self.detach(i);
1439                }
1440            }
1441            11 => {
1442                r.finish()?;
1443                self.controller(i, p)?;

1445            }
1446            12 => {

1450                }

1454                w.u8(0)?;
1455                w.u16(s.rows)?;
1456                w.u16(s.cols)?;
1457                w.u16(0)?;

1459                w.u32(if physical { MODE_MASK & !2 } else { MODE_MASK })?;
1460                w.u32(0xff)?;
1461                w.u32(if physical { 0x5f } else { 0x7f })?;
1462                w.u32(0)?;
1463            }

1466                let epoch = r.u64()?;

1475                    return Err(Status::Malformed);
1476                }

1485                r.finish()?;

1489                    return Err(Status::Malformed);
1490                }

1492                if self.refuse_paste(i) {
1493                    return Err(Status::Cancelled);
1494                }

1496                    || self.sessions[i].physical_paste.is_some()

1498                    || self.sessions[i].draining
1499                {
1500                    return Err(Status::WouldBlock);
1501                }

1504                let paste = self.reference()?;
1505                self.sessions[i].paste = Some(Paste {
1506                    reference: paste,
1507                    frontend: p,
1508                    attachment,
1509                    epoch,
1510                    expires: now().saturating_add(5000),
1511                    total,
1512                    bytes,
1513                });
1514                w.reference(paste)?;
1515            }
1516            20 => {
1517                let attachment = r.reference()?;
1518                let epoch = r.u64()?;
1519                let count = r.u32()? as usize;
1520                reserved(r.u32()?)?;
1521                let bytes = r.take(count)?;
1522                r.finish()?;
1523                let epoch = self.wire_epoch(i, epoch)?;
1524                self.frontend(i, p, attachment, epoch)?;
1525                if count == 0 || count > 512 {
1526                    return Err(Status::Malformed);
1527                }
1528                if self.sessions[i].mode & 1 == 0 {
1529                    return Err(Status::WrongDirection);
1530                }
1531                // Raw bytes carry no key/paste provenance: they never answer a trust prompt.
1532                if self.sessions[i].capture.is_some() {
1533                    self.capture_raw(i);
1534                    w.u32(count as u32)?;
1535                    w.u32(0)?;
1536                    return Ok(Some(w.position()));
1537                }
1538                if let Some(status) = self.sessions[i].input_failure {
1539                    return Err(status);
1540                }
1541                if self.sessions[i].draining {
1542                    return Err(Status::BrokenPipe);
1543                }
1544                if self.sessions[i].paste.is_some()
1545                    || self.sessions[i].physical_paste.is_some()
1546                    || !self.sessions[i].raw.is_empty()
1547                    || self.sessions[i]
1548                        .controls
1549                        .iter()
1550                        .any(|e| e.sent != e.bytes.len())
1551                {
1552                    return Err(Status::WouldBlock);
1553                }
1554                let count = self.write_stream(self.sessions[i].writers[0], bytes)?;
1555                w.u32(count as u32)?;
1556                w.u32(0)?;
1557            }
1558            21 => {
1559                let attachment = r.reference()?;
1560                let epoch = r.u64()?;
1561                r.finish()?;
1562                let epoch = self.wire_epoch(i, epoch)?;
1563                self.frontend(i, p, attachment, epoch)?;
1564                if let Some(status) = self.sessions[i].input_failure {
1565                    return Err(status);
1566                }
1567                if self.sessions[i].paste.is_some() || self.sessions[i].physical_paste.is_some() {
1568                    return Err(Status::WouldBlock);
1569                }
1570                if !self.sessions[i].eof && self.sessions[i].controls.len() >= CONTROL_LIMIT - 1 {
1571                    return Err(Status::QuotaExceeded);
1572                }
1573                // Input ends: no answer to a trust prompt can arrive any more.
1574                self.end_capture(
1575                    i,
1576                    api::cell_session::trust::DECISION_CANCELLED,
1577                    trust::End::Release,
1578                );
1579                self.sessions[i].draining = true;
1580                self.drain(i);
1581                w.u32(if self.sessions[i].eof { 1 } else { 0 })?;
1582                w.u32(0)?;
1583                if let Some(status) = self.sessions[i].input_failure {
1584                    return Err(status);
1585                }
1586            }
1587            22 => {
1588                let lease = r.reference()?;
1589                let epoch = r.u64()?;
1590                let through = r.u64()?;
1591                r.finish()?;
1592                // The controller, while a foreground child is the reader, acknowledges the owner
1593                // notices (trust prompt / prompt end) it has drawn on its own control ring.
1594                if self.sessions[i].controller == p && self.sessions[i].reader != p {
1595                    self.acknowledge_notices(i, lease, epoch, through)?;
1596                    return Ok(Some(w.position()));
1597                }
1598                self.reader(i, p, lease)?;
1599                self.epoch(i, epoch)?;
1600                let s = &mut self.sessions[i];
1601                if s.controls
1602                    .iter()
1603                    .any(|e| e.sequence <= through && e.sent != e.bytes.len())
1604                    || through > s.sequence
1605                {
1606                    return Err(Status::Malformed);
1607                }
1608                while s.controls.front().is_some_and(|e| e.sequence <= through) {
1609                    s.controls.pop_front();
1610                }
1611            }
1612            _ => return Err(Status::UnknownOperation),
1613        }
1614        Ok(Some(w.position()))
1615    }

1682            bootstrap,
1683            controller,
1684            operation,
1685            baseline: endpoints,
1686            baseline_writers: writers,

1689            baseline_lease: lease,
1690            lease,

1692            epoch: 1,

1694            rows,
1695            cols,
1696            attachment,
1697            foreground: None,

1699            baseline_modes,

1705            poll: None,

1710            closed: false,

1713            physical_paste: None,
1714            physical_paste_expires: 0,
1715            physical_paste_complete: false,
1716            physical_paste_discard: false,
1717            delivered_cursor: 0,
1718            prepare_deadline: 0,
1719            kind,
1720            holder: frontend.map(|f| f.0),
1721            holder_operation: frontend.map_or(ZERO, |f| f.1),
1722            frontend: frontend.map(|f| f.2),

1736        } else {
1737            self.sessions.push(session);

1739        };

1754    /// Frontend selectors 23–28 (annex S5-3c §3, §3a).

1803            25 => self.create_for_frontend(fe::CreateForFrontendV1::decode(body)?, p, w),
1804            26 => {

1808                    state: f.state,
1809                    session: ZERO,
1810                    attachment: ZERO,
1811                    resume: ZERO,
1812                };

1814                    let s = &self.sessions[self.index(f.session)?];
1815                    status.session = s.reference;
1816                    status.attachment = s.attachment.as_ref().map_or(ZERO, |a| a.reference);
1817                    status.resume = s.resume;
1818                }

1821                w.bytes(&bytes)
1822            }

1826                    let i = self.index(self.frontends[fi].session)?;
1827                    self.detach(i);
1828                }
1829                self.remove_frontend(fi, false);
1830                Ok(())
1831            }
1832            28 => self.reattach(fe::ReattachFrontendV1::decode(body)?, p, w),
1833            _ => Err(Status::UnknownOperation),
1834        }
1835    }
1836    /// Sel 24: prove the child was prepared by the holder under its registered operation,
1837    /// carrying exactly this frontend in slot 8, then register the kernel kind-6 receipt.
1838    fn bind_frontend(
1839        &mut self,
1840        fi: usize,
1841        child: ChildRefV1,
1842        w: &mut Writer<'_>,
1843    ) -> Result<(), Status> {
1844        let (frontend, holder, operation) = {
1845            let f = &self.frontends[fi];
1846            if f.state != fe::FRONTEND_STATE_PENDING || f.child.is_some() {
1847                return Err(Status::Denied);
1848            }
1849            (f.reference, f.holder, f.operation)
1850        };
1851        self.kernel_operation = operation;
1852        let prepared = child.prepared_ref();
1853        let mut body = [0; 32];
1854        let mut b = Writer::new(&mut body);
1855        b.reference(prepared)?;
1856        b.u16(GRANT_KIND_INPUT_FRONTEND)?;
1857        b.u16(SLOT_FRONTEND)?;
1858        b.u32(0)?;
1859        b.u32(512)?;
1860        b.u32(0)?;
1861        let mut reply = [0; 704];
1862        let len = self.kernel(KernelOp::InspectPreparedGrant, &body, &mut reply)?;
1863        let mut r = Reader::new(&reply[HEADER_LEN..len]);
1864        let actual = r.child()?;
1865        let child_operation = r.reference()?;
1866        let digest: [u8; 32] = r.take(32)?.try_into().map_err(|_| Status::Malformed)?;
1867        let mut grant = Reader::new(r.take(40)?);
1868        let kind = grant.u16()?;
1869        let rights = grant.u16()?;
1870        let slot = grant.u16()?;
1871        reserved(grant.u16()? as u32)?;
1872        let owner_service = grant.u64()?;
1873        let object = grant.reference()?;
1874        let tail = grant.u64()?;
1875        grant.finish()?;
1876        let projection_len = r.u32()? as usize;
1877        reserved(r.u32()?)?;
1878        let count = r.u32()? as usize;
1879        reserved(r.u32()?)?;
1880        let projection = r.take(count)?;
1881        r.finish()?;
1882        if actual != child || count != projection_len || projection.len() != 32 {
1883            return Err(Status::InvalidChild);
1884        }
1885        if kind != GRANT_KIND_INPUT_FRONTEND
1886            || slot != SLOT_FRONTEND
1887            || rights != 0
1888            || tail != 0
1889            || owner_service != api::syscall::service::INPUT as u64
1890            || object != frontend
1891        {
1892            return Err(Status::InvalidChild);
1893        }
1894        let mut pr = Reader::new(projection);
1895        let preparer = pr.principal()?;
1896        let owner = pr.principal()?;
1897        pr.finish()?;
1898        if child_operation != operation || preparer != holder || owner != holder {
1899            return Err(Status::Denied);
1900        }
1901        let mut seal = [0; FRONTEND_SEAL_LEN];
1902        let mut b = Writer::new(&mut seal);
1903        b.reference(frontend)?;
1904        b.child(child)?;
1905        let mut registration = [0; 96 + FRONTEND_SEAL_LEN];
1906        let mut b = Writer::new(&mut registration);
