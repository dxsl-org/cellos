1//! One repeat owner for all keyboard producers; no allocation or catch-up bursts.
2use api::input::{KeyState, KeySym};
3
4const DELAY_MS: u64 = 400;
5const INTERVAL_MS: u64 = 33;
6
7#[derive(Default)]
8pub(crate) struct KeyRepeat {
9    held: Option<(Option<u32>, u32)>,
10    deadline_ms: u64,
11}
12
13impl KeyRepeat {
14    pub(crate) fn clear(&mut self) {
15        self.held = None;
16    }
17
18    pub(crate) fn remove_device(&mut self, device: u32) {
19        if self.held.is_some_and(|(owner, _)| owner == Some(device)) {
20            self.clear();
21        }
22    }
23
24    /// Hardware repeat is suppressed: the service owns cadence on every backend.
25    pub(crate) fn observe(
26        &mut self,
27        device: Option<u32>,
28        code: u32,
29        state: KeyState,
30        keysym: KeySym,
31        now: u64,
32    ) -> bool {
33        let key = (device, code);
34        match state {
35            KeyState::Released => {
36                if self.held == Some(key) {
37                    self.clear();
38                }
39            }
40            KeyState::Repeated => return false,
41            KeyState::Pressed => {
42                if self.held == Some(key) {
43                    return false;
44                }
45                self.clear();
46                if matches!(
47                    keysym,
48                    KeySym::Printable
49                        | KeySym::Backspace
50                        | KeySym::Delete
51                        | KeySym::Return
52                        | KeySym::Tab
53                        | KeySym::Up
54                        | KeySym::Down
55                        | KeySym::Left
56                        | KeySym::Right
57                        | KeySym::Home
58                        | KeySym::End
59                        | KeySym::PageUp
60                        | KeySym::PageDown
61                        | KeySym::F3
62                ) {
63                    self.held = Some(key);
64                    self.deadline_ms = now.saturating_add(DELAY_MS);
65                }
66            }
67        }
68        true
69    }
70
71    pub(crate) fn poll(&mut self, now: u64) -> Option<u32> {
72        let (_, code) = self.held?;
73        if now < self.deadline_ms {
74            return None;
75        }
76        self.deadline_ms = now.saturating_add(INTERVAL_MS);
77        Some(code)
78    }
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    #[test]
86    fn backspace_repeats_until_release_without_catch_up_bursts() {
87        let mut r = KeyRepeat::default();
88        assert!(r.observe(None, 14, KeyState::Pressed, KeySym::Backspace, 10));
89        assert_eq!(r.poll(409), None);
90        assert_eq!(r.poll(410), Some(14));
91        assert_eq!(r.poll(442), None);
92        assert_eq!(r.poll(443), Some(14));
93        assert_eq!(r.poll(900), Some(14));
94        assert_eq!(r.poll(900), None);
95        r.observe(None, 14, KeyState::Released, KeySym::Backspace, 901);
96        assert_eq!(r.poll(2000), None);
97    }
98
99    #[test]
100    fn last_key_wins_and_backend_repeat_does_not_double_or_delay_it() {
101        let mut r = KeyRepeat::default();
102        r.observe(None, 14, KeyState::Pressed, KeySym::Backspace, 0);
103        assert!(!r.observe(None, 14, KeyState::Repeated, KeySym::Backspace, 100));
104        assert!(!r.observe(None, 14, KeyState::Pressed, KeySym::Backspace, 200));
105        assert_eq!(r.poll(400), Some(14));
106        r.observe(None, 30, KeyState::Pressed, KeySym::Printable, 410);
107        r.observe(None, 14, KeyState::Released, KeySym::Backspace, 420);
108        assert_eq!(r.poll(810), Some(30));
109        r.clear();
110        assert_eq!(r.poll(2000), None);
111    }
112
113    #[test]
114    fn disconnect_cancels_only_owner_and_toggle_keys_do_not_repeat() {
115        let mut r = KeyRepeat::default();
116        r.observe(Some(1), 14, KeyState::Pressed, KeySym::Backspace, 0);
117        r.remove_device(2);
118        assert_eq!(r.poll(400), Some(14));
119        r.remove_device(1);
120        assert_eq!(r.poll(1000), None);
121        r.observe(None, 59, KeyState::Pressed, KeySym::F1, 1000);
122        assert_eq!(r.poll(2000), None);
123        r.observe(None, 14, KeyState::Pressed, KeySym::Backspace, 2100);
124        r.observe(Some(1), 14, KeyState::Released, KeySym::Backspace, 2200);
125        assert_eq!(r.poll(2500), Some(14));
126        r.observe(None, 61, KeyState::Pressed, KeySym::F3, 2600);
127        assert_eq!(r.poll(3000), Some(61));
128    }
