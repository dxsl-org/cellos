2//! Owner-run first-trust confirmation: SessionV1 selectors 29–31 (D79-11, D79-13).
3//!
4//! VFS registers an immutable descriptor (29). The descriptor's target, as the live
5//! committed foreground child with its exact lease and epoch, begins the confirmation
6//! (30) on a session whose frontend submits framed events (a sel-25 frontend; Serial
7//! sessions, including the physical UART, are byte streams and answer `Unsupported`).
8//! Begin discards the input still queued for the child and answers its waiting poll;
9//! from then on the session is *quarantined*: none of its typed, raw or paste input
10//! reaches the child until the confirmation is over. Only a fresh typed `yes` followed by
11//! Enter accepts; any other line, Escape, Ctrl-C, any paste or any byte without key/text
12//! provenance (SubmitRaw) rejects; the deadline expires it; frontend loss, end of input,
13//! foreground restoration and session close cancel it. Every release wipes the capture
14//! line, paste staging and the serial decoder. Only the registering VFS principal
15//! collects the decision (31), final exactly once.
16//!
17//! Input has no output path of its own, so the prompt is drawn by the session
18//! controller: `TrustPrompt` (the descriptor's host/port/algorithm/fingerprint) and
19//! `TrustPromptEnd` are owner notices written to the controller's own control ring
20//! (`baseline_writers[1]`), never to the child's. The controller acknowledges each one
21//! after drawing it (AcknowledgeControl, controller-only while a foreground child is the
22//! reader). The phases of a capture are therefore:
23//! - `Arming`: the prompt may not be on screen yet. Input is dropped and never counts.
24//! - `Armed`: the controller drew the prompt. Typed key/text events build the answer.
25//! - `Ending`: the decision is recorded; the prompt may still be on screen. Input is
26//!   dropped until the controller acknowledges `TrustPromptEnd`, then routing returns to
27//!   the child untouched (modes, lease and epoch never change).
28//!
29//! Restoration, close, frontend loss, end of input and a failed write to the controller's
30//! ring end the quarantine at once (and cancel an undecided capture; a recorded decision
31//! is kept, whether or not VFS already collected it). The kernel flushes the controller's
32//! ring when it restores the controller (RestoreSessionInput), so restore and close drop
33//! unsent notices; the controller ends its prompt on restoration itself.
34//!
35//! Accepted limitation: the attached frontend holder is the session's input authority
36//! (D02), so whatever it submits as typed key/text events is, as for every other
37//! keystroke, the user's typing. A holder that declares itself a GUI frontend and
38//! synthesizes `yes` and Enter therefore answers the prompt; the confirmation is only as
39//! trustworthy as the frontend that the session controller attached. Input cannot tell
40//! synthesized from physical keystrokes on that path and does not try to.
41//!
42//! Known gap: bytes input wrote to the child's kernel stdin ring before Begin (raw-mode
43//! routing or SubmitRaw) stay readable by the child during the prompt. Input cannot see
44//! that ring's occupancy: kernel InspectEndpoint reports only holder, operation, rights,
45//! state and direction (`kernel/src/task/cell_session.rs:873-881`), and the pipe layer
46//! exposes reader/writer counts, not buffered bytes (`kernel/src/task/pipe.rs:462-464`).

51/// An uncollected final record is released this long after its deadline.
52const RETAIN_NS: u64 = tr::EXPIRY_MAX_NS;
53/// Owner notices queued per session for the controller's control ring.

56pub(super) struct Challenge {

58    port: u16, host: Vec<u8>, algorithm: Vec<u8>, fingerprint: Vec<u8>, digest: [u8; 32],
59    session: Ref, epoch: u64, begun: bool, decision: u32,

62    fn prompt(&self) -> Result<TrustPromptV1<'_>, Status> {
63        fn text(bytes: &[u8]) -> Result<&str, Status> { core::str::from_utf8(bytes).map_err(|_| Status::Malformed) }
64        Ok(TrustPromptV1 { port: self.port, host: text(&self.host)?, algorithm: text(&self.algorithm)?, fingerprint: text(&self.fingerprint)? })
65    }

68/// Capture phase; the `u64` is the sequence of the notice the controller must acknowledge.
69#[derive(Clone, Copy, PartialEq, Eq)]
70enum Phase { Arming(u64), Armed, Ending(u64) }
71
72/// The captured line. At most eight bytes are kept: anything longer can never be `yes`.
73pub(super) struct Capture { confirmation: Ref, phase: Phase, line: [u8; 8], len: usize, overflow: bool }
74impl Capture { fn wipe(&mut self) { ostd::session::zeroize(&mut self.line); self.len = 0; } }
75impl Drop for Capture { fn drop(&mut self) { self.wipe(); } }

77/// How a capture ends.
78#[derive(Clone, Copy, PartialEq, Eq)]
79pub(super) enum End {
80    /// Tell the controller and keep the session quarantined until it acknowledges.
81    Await,
82    /// Tell the controller and release at once (frontend loss, end of input).
83    Release,
84    /// Release at once without a notice (restore, close: the kernel flushes the ring).
85    Silent,
86}

88/// One captured input item.
89enum Typed<'a> { Text(&'a [u8]), Enter, Erase, Refuse, Ignore }

91/// Never part of `yes`: stands in for a key that is not plain text.
92const OTHER: &[u8] = &[0];

94/// Does a live challenge keep the request lane `(caller, operation)` fenced? The lane of
95/// RegisterTrustChallenge is the challenge Ref, the lane of CollectTrustDecision the
96/// confirmation Ref.
97pub(super) fn challenge_lane(challenges: &[Challenge], caller: Principal, operation: Ref) -> bool {
98    challenges.iter().any(|c| c.registrar == caller && (c.challenge == operation || c.confirmation == operation))
99}

101/// A validated Key frame. Plain characters arrive as Text, so every other key (Alt/Ctrl
102/// chords, navigation, function keys) is never part of `yes`.
103fn key_item(payload: &[u8]) -> Typed<'static> {
104    let word = |at: usize| u32::from_le_bytes([payload[at], payload[at + 1], payload[at + 2], payload[at + 3]]);
105    let (code, mods, pressed) = (word(0), word(4), payload[8]);
106    if pressed == 0 { return Typed::Ignore; }
107    match code {
108        67 | 99 if mods & modifiers::CONTROL != 0 => Typed::Refuse,
109        key_code::ENTER => Typed::Enter,
110        key_code::BACKSPACE => Typed::Erase,
111        key_code::ESCAPE => Typed::Refuse,
112        _ => Typed::Text(OTHER),
113    }
114}

117    /// Sel 29/31 admission. 29: the attested sender is the live registered VFS task
118    /// (as VFS checks live KMS). 31: the principal that registered the challenge.
119    pub(super) fn admit_trust(&self, selector: u16, body: &[u8], p: Principal, sender: usize) -> Result<(), Status> {
120        if selector == 29 {
121            return if sys_lookup_service(api::syscall::service::VFS).is_some_and(|tid| tid == sender) { Ok(()) } else { Err(Status::Denied) };
122        }
123        let confirmation = Ref::from_bytes(body)?;
124        let c = self.challenges.iter().find(|c| c.confirmation == confirmation).ok_or(Status::Stale)?;
125        if c.registrar == p { Ok(()) } else { Err(Status::Denied) }
126    }

128    pub(super) fn trust_operation(&mut self, selector: u16, body: &[u8], p: Principal, w: &mut Writer<'_>) -> Result<(), Status> {
129        match selector {
130            29 => self.register_challenge(body, p, w),
131            30 => self.begin_confirmation(BeginTrustConfirmationV1::decode(body)?, p),
132            31 => self.collect_decision(Ref::from_bytes(body)?, p, w),
133            _ => Err(Status::UnknownOperation),
134        }
135    }
136
137    fn register_challenge(&mut self, body: &[u8], p: Principal, w: &mut Writer<'_>) -> Result<(), Status> {
138        let request = TrustChallengeV1::decode(body)?;
139        let now = now_ns();
140        if request.expiry_ns <= now { return Err(Status::TimedOut); }
141        if request.expiry_ns - now > tr::EXPIRY_MAX_NS { return Err(Status::Malformed); }
142        // Challenge refs are single-use: a second registration never rebinds one.
143        if self.challenges.iter().any(|c| c.registrar == p && c.challenge == request.challenge) { return Err(Status::Denied); }
144        // Uncollected records hold memory until collected or released after their deadline
145        // (also those of a dead VFS incarnation, which nobody can collect any more); a target
146        // has at most one confirmation that can still be decided.
147        if self.challenges.len() >= tr::MAX_TOTAL
148            || self.challenges.iter().filter(|c| c.registrar == p).count() >= tr::MAX_PER_REGISTRAR
149            || self.challenges.iter().filter(|c| c.target == request.target && c.decision == tr::DECISION_PENDING).count() >= tr::MAX_PER_TARGET {
150            return Err(Status::QuotaExceeded);
151        }
152        let copy = |text: &str| -> Result<Vec<u8>, Status> { let mut v = Vec::new(); reserve(&mut v, text.len())?; v.extend_from_slice(text.as_bytes()); Ok(v) };
153        let (host, algorithm, fingerprint) = (copy(request.prompt.host)?, copy(request.prompt.algorithm)?, copy(request.prompt.fingerprint)?);
154        reserve(&mut self.challenges, 1)?;
155        let confirmation = self.reference()?;
156        self.challenges.push(Challenge {
157            confirmation, challenge: request.challenge, registrar: p, target: request.target, expiry: request.expiry_ns,
158            port: request.prompt.port, host, algorithm, fingerprint, digest: ostd::session::sha256(body),
159            session: ZERO, epoch: 0, begun: false, decision: tr::DECISION_PENDING,
160        });
161        w.reference(confirmation)
162    }

164    fn begin_confirmation(&mut self, request: BeginTrustConfirmationV1, p: Principal) -> Result<(), Status> {
165        let i = self.index(request.session)?;
166        let n = self.challenges.iter().position(|c| c.confirmation == request.confirmation).ok_or(Status::Stale)?;
167        if self.challenges[n].target != p { return Err(Status::Denied); }
168        let s = &self.sessions[i];
169        let foreground = s.foreground.as_ref().is_some_and(|f| f.committed && !f.restored && f.child.principal == p);
170        if !foreground || s.reader != p || s.lease != request.lease { return Err(Status::Denied); }
171        if s.frozen { return Err(Status::WouldBlock); }
172        if s.epoch != request.epoch { return Err(Status::WrongEpoch); }
173        if now_ns() >= self.challenges[n].expiry {
174            if self.challenges[n].decision == tr::DECISION_PENDING { self.challenges[n].decision = tr::DECISION_EXPIRED; }
175            return Err(Status::TimedOut);
176        }
177        let c = &self.challenges[n];
178        if c.begun || c.decision != tr::DECISION_PENDING { return Err(Status::Stale); }
179        // A trusted answer needs a frontend whose input carries key/text/paste provenance
180        // (SubmitEvents). Serial sessions (the physical UART or the controller's own
181        // self-attachment) are byte streams where an unframed paste is indistinguishable from
182        // typing; headless or detached input has no frontend at all.
183        if s.attachment.is_none() || s.kind == fe::FRONTEND_KIND_SERIAL || self.serial == Some(request.session) { return Err(Status::Unsupported); }
184        if s.capture.is_some() || s.paste.is_some() || s.physical_paste.is_some() || s.physical_paste_discard || s.draining { return Err(Status::WouldBlock); }
185        if s.notices.len() + 2 > NOTICE_LIMIT { return Err(Status::QuotaExceeded); }
186        let mut payload = [0u8; TrustPromptV1::MAX_LEN];
187        let len = c.prompt()?.encode(&mut payload)?;
188        let confirmation = c.confirmation;
189        self.sessions[i].notices.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
190        let event = self.event(i, event_kind::TRUST_PROMPT, &payload[..len])?;
191        // Nothing queued before Begin reaches the child while the prompt is up: its unread
192        // input is discarded and an outstanding poll is answered empty, in this same step.
193        self.invalidate_poll(i, Status::WouldBlock);
194        let s = &mut self.sessions[i];
195        s.typed.clear(); zeroize_deque(&mut s.raw);
196        let prompt = event.sequence; s.sequence = prompt;
197        s.notices.push_back(event);
198        let c = &mut self.challenges[n]; c.begun = true; c.session = request.session; c.epoch = request.epoch;
199        self.sessions[i].capture = Some(Capture { confirmation, phase: Phase::Arming(prompt), line: [0; 8], len: 0, overflow: false });
200        Ok(())
201    }

203    fn collect_decision(&mut self, confirmation: Ref, p: Principal, w: &mut Writer<'_>) -> Result<(), Status> {
204        let n = self.challenges.iter().position(|c| c.confirmation == confirmation).ok_or(Status::Stale)?;
205        if self.challenges[n].registrar != p { return Err(Status::Denied); }
206        if self.challenges[n].decision == tr::DECISION_PENDING && now_ns() >= self.challenges[n].expiry {
207            if let Some(i) = self.capturing(confirmation) { self.end_capture(i, tr::DECISION_EXPIRED, End::Await); }
208            self.challenges[n].decision = tr::DECISION_EXPIRED;
209        }
210        let c = &self.challenges[n];
211        let record = TrustDecisionV1 { decision: c.decision, challenge: c.challenge, target: c.target, session: c.session, epoch: c.epoch, descriptor_sha256: c.digest };
212        let mut bytes = [0u8; TrustDecisionV1::LEN]; record.encode(&mut bytes)?; w.bytes(&bytes)?;
213        // A final decision is handed out once; the retained outcome still answers an exact replay.
214        if record.is_final() { self.challenges.swap_remove(n); }
215        Ok(())
216    }

218    fn capturing(&self, confirmation: Ref) -> Option<usize> {
219        self.sessions.iter().position(|s| !s.closed && s.capture.as_ref().is_some_and(|c| c.confirmation == confirmation))
220    }

222    /// Drop matching records; a capture of a dropped record ends cancelled.
223    fn release_challenges(&mut self, dead: impl Fn(&Challenge) -> bool) {
224        let mut n = 0;
225        while n < self.challenges.len() {
226            if !dead(&self.challenges[n]) { n += 1; continue; }
227            if let Some(i) = self.capturing(self.challenges[n].confirmation) { self.end_capture(i, tr::DECISION_CANCELLED, End::Release); }
228            self.challenges.swap_remove(n);
229        }
230    }

232    /// Queue one owner notice for the controller's control ring; returns its sequence.
233    fn notice(&mut self, i: usize, kind: u16, payload: &[u8]) -> Result<u64, Status> {
234        if self.sessions[i].notices.len() >= NOTICE_LIMIT { return Err(Status::QuotaExceeded); }
235        self.sessions[i].notices.try_reserve(1).map_err(|_| Status::QuotaExceeded)?;
236        let event = self.event(i, kind, payload)?; let sequence = event.sequence;
237        let s = &mut self.sessions[i]; s.sequence = sequence; s.notices.push_back(event);
238        Ok(sequence)
239    }

241    /// End the session's capture with `decision` unless the record already has a final one.
242    /// A capture already `Ending` only leaves quarantine on an escape path (`Release`/`Silent`).
243    pub(super) fn end_capture(&mut self, i: usize, decision: u32, end: End) {
244        let Some(capture) = self.sessions[i].capture.as_mut() else { return; };
245        if let Phase::Ending(_) = capture.phase {
246            if end != End::Await { self.release_quarantine(i); }
247            return;
248        }
249        capture.wipe();
250        let confirmation = capture.confirmation;
251        let mut accepted = false;
252        if let Some(c) = self.challenges.iter_mut().find(|c| c.confirmation == confirmation) {
253            if c.decision == tr::DECISION_PENDING { c.decision = decision; }
254            accepted = c.decision == tr::DECISION_ACCEPT;
255        }
256        let ended = if end == End::Silent { None } else { self.notice(i, event_kind::TRUST_PROMPT_END, &u32::from(accepted).to_le_bytes()).ok() };
257        match (end, ended) {
258            // Bytes typed while the prompt may still be on screen are dropped until the
259            // controller confirms it ended the prompt. Begin reserved the notice slot.
260            (End::Await, Some(sequence)) => { if let Some(c) = self.sessions[i].capture.as_mut() { c.phase = Phase::Ending(sequence); } }
261            _ => self.release_quarantine(i),
262        }
263    }

265    /// Routing returns to the child. Every incremental input state that could still carry
266    /// bytes typed during the quarantine is wiped first, so no partial sequence (escape,
267    /// UTF-8, paste staging) completes afterwards: the capture line, any paste staging and
268    /// the serial decoder.
269    fn release_quarantine(&mut self, i: usize) {
270        let s = &mut self.sessions[i];
271        s.capture = None;
272        s.paste = None; s.physical_paste = None; s.physical_paste_complete = false; s.physical_paste_discard = false;
273        s.decoder = api::services::session::SerialDecoder::new();
274    }

276    /// AcknowledgeControl from the controller while a foreground child is the reader: the
277    /// controller drew (or ended) the prompt of every delivered owner notice up to `through`.
278    pub(super) fn acknowledge_notices(&mut self, i: usize, lease: Ref, epoch: u64, through: u64) -> Result<(), Status> {
279        let s = &self.sessions[i];
280        if s.frozen { return Err(Status::WouldBlock); }
281        if lease != s.baseline_lease { return Err(Status::Denied); }
282        if epoch != s.epoch { return Err(Status::WrongEpoch); }
283        // Only notices fully written to the controller's ring can have been drawn.
284        if through > s.sequence || s.notices.iter().any(|e| e.sequence <= through) { return Err(Status::Malformed); }
285        let s = &mut self.sessions[i];
286        let Some(capture) = s.capture.as_mut() else { return Ok(()); };
287        match capture.phase {
288            Phase::Arming(sequence) if through >= sequence => capture.phase = Phase::Armed,
289            Phase::Ending(sequence) if through >= sequence => self.release_quarantine(i),
290            _ => {}
291        }
292        Ok(())
293    }

295    /// Is the capture accepting an answer now? A passed deadline first ends it expired.
296    /// While arming or ending (or with no capture) the caller drops the input.
297    fn armed(&mut self, i: usize) -> bool {
298        let Some(capture) = self.sessions[i].capture.as_ref() else { return false; };
299        if let Phase::Ending(_) = capture.phase { return false; }
300        let (confirmation, phase) = (capture.confirmation, capture.phase);
301        let now = now_ns();
302        if self.challenges.iter().find(|c| c.confirmation == confirmation).is_none_or(|c| now >= c.expiry) {
303            self.end_capture(i, tr::DECISION_EXPIRED, End::Await);
304            return false;
305        }
306        phase == Phase::Armed
307    }

309    /// Feed one item to an armed capture; true once it ended.
310    fn capture_item(&mut self, i: usize, item: Typed<'_>) -> bool {
311        let Some(c) = self.sessions[i].capture.as_mut() else { return true; };
312        let decision = match item {
313            Typed::Ignore => return false,
314            Typed::Text(bytes) => {
315                for &byte in bytes { if c.len < c.line.len() { c.line[c.len] = byte; c.len += 1; } else { c.overflow = true; } }
316                return false;
317            }
318            Typed::Erase => { if !c.overflow && c.len > 0 { c.len -= 1; c.line[c.len] = 0; } return false; }
319            Typed::Enter if !c.overflow && &c.line[..c.len] == b"yes" => tr::DECISION_ACCEPT,
320            Typed::Enter | Typed::Refuse => tr::DECISION_REJECT,
321        };
322        self.end_capture(i, decision, End::Await);
323        true
324    }

326    /// SubmitEvents / physical keys / decoded UART during a capture: the batch is validated
327    /// exactly as routed input, then consumed here. It answers only while armed; whatever
328    /// follows the deciding item, and everything while arming or ending, is dropped.
329    pub(super) fn capture_events(&mut self, i: usize, bytes: &[u8], count: usize) -> Result<(), Status> {
330        let mut offset = 0;
331        for _ in 0..count {
332            let frame = bytes.get(offset..offset + 24).ok_or(Status::Malformed)?;
333            let kind = u16::from_le_bytes([frame[16], frame[17]]);
334            let length = u16::from_le_bytes([frame[18], frame[19]]) as usize;
335            if frame[..16] != [0; 16] || frame[20..24] != [0; 4] || !(1..=2).contains(&kind) { return Err(Status::Malformed); }
336            let payload = bytes.get(offset + 24..offset + 24 + length).ok_or(Status::Malformed)?;
337            api::services::session::validate_event(kind, payload)?;
338            if kind == 2 { api::services::session::validate_text(payload)?; }
339            offset += 24 + length;
340        }
341        if offset != bytes.len() { return Err(Status::Malformed); }
342        if !self.armed(i) { return Ok(()); }
343        let mut offset = 0;
344        for _ in 0..count {
345            let kind = u16::from_le_bytes([bytes[offset + 16], bytes[offset + 17]]);
346            let length = u16::from_le_bytes([bytes[offset + 18], bytes[offset + 19]]) as usize;
347            let payload = &bytes[offset + 24..offset + 24 + length];
348            offset += 24 + length;
349            let item = if kind == 2 { Typed::Text(payload) } else { key_item(payload) };
350            if self.capture_item(i, item) { break; }
351        }
352        Ok(())
353    }

355    /// SubmitRaw during a capture. Raw bytes carry no key/paste provenance, so they never
356    /// answer: while armed they reject, otherwise they are dropped. Never routed.
357    pub(super) fn capture_raw(&mut self, i: usize) {
358        if self.armed(i) { self.end_capture(i, tr::DECISION_REJECT, End::Await); }
359    }

361    /// Any paste during a capture is refused; while armed it also rejects the capture.
362    pub(super) fn refuse_paste(&mut self, i: usize) -> bool {
363        if self.sessions[i].capture.is_none() { return false; }
364        if self.armed(i) { self.end_capture(i, tr::DECISION_REJECT, End::Await); }
365        true
366    }

368    /// Deadlines: expire captures and pending records; release uncollected final records.
369    pub(super) fn maintain_trust(&mut self) {
370        if self.challenges.is_empty() { return; }
371        for i in 0..self.sessions.len() {
372            if !self.sessions[i].closed && self.sessions[i].capture.is_some() { self.armed(i); }
373        }
374        let now = now_ns();
375        for c in self.challenges.iter_mut().filter(|c| c.decision == tr::DECISION_PENDING && now >= c.expiry) { c.decision = tr::DECISION_EXPIRED; }
376        self.release_challenges(|c| now >= c.expiry.saturating_add(RETAIN_NS));
377    }

379    /// Write queued owner notices to the controller's own control ring.
380    pub(super) fn drain_notices(&mut self, i: usize) {
381        for _ in 0..NOTICE_LIMIT {
382            let s = &self.sessions[i];
383            if s.closed { return; }
384            let Some(event) = s.notices.front() else { return; };
385            let mut bytes = [0u8; 512]; let count = (event.bytes.len() - event.sent).min(512);
386            bytes[..count].copy_from_slice(&event.bytes[event.sent..event.sent + count]);
387            let endpoint = s.baseline_writers[1];
388            self.kernel_operation = s.operation;
389            match self.write_stream(endpoint, &bytes[..count]) {
390                Ok(written) => {
391                    let s = &mut self.sessions[i];
392                    if let Some(e) = s.notices.front_mut() { e.sent += written; if e.sent == e.bytes.len() { s.notices.pop_front(); } }
393                }
394                Err(Status::WouldBlock) => return,
395                // The controller's ring is gone: no prompt can be drawn or acknowledged, so a
396                // capture can neither arm nor end. Cancel it (an undecided one) and release now.
397                Err(_) => {
398                    self.sessions[i].notices.clear();
399                    self.end_capture(i, tr::DECISION_CANCELLED, End::Silent);
400                    return;
401                }
402            }
403        }
404    }
