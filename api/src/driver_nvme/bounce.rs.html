1//! The one-page DMA buffer every sector read/write goes through (audit 08 S-11).
2//!
3//! A timed-out command is not aborted: the device may still DMA a late READ into the
4//! page, or fetch a late WRITE's payload from it. Reusing that page for the next
5//! request would let the stale transfer overwrite the next result or write the next
6//! payload to the old LBA, so a page with a command outstanding is retired - kept
7//! allocated and authorized for the driver's lifetime, never freed - and a fresh
8//! page takes its place.
9
10use crate::dma::AuthorizedDma;
11use crate::layout;
12use alloc::vec::Vec;
13use ostd::dma::DmaBuf;
14use ostd::io::print_fmt;
15use types::block_wire::MAX_RUN_SECTORS;
16
17/// Retired pages kept before the driver stops serving I/O. Each holds one page, and
18/// a device that keeps abandoning commands is not one to keep feeding memory.
19const QUARANTINE_LIMIT: usize = 16;
20
21pub struct BounceBuffer {
22    bdf: u32,
23    /// `None` once the driver refuses further I/O.
24    live: Option<AuthorizedDma<DmaBuf>>,
25    /// Pages a timed-out command may still access; held so they are never freed.
26    retired: Vec<AuthorizedDma<DmaBuf>>,
27}
28
29impl BounceBuffer {
30    /// Allocate and authorize the first page for the device at `bdf`.
31    pub fn new(bdf: u32) -> Result<Self, &'static str> {
32        Ok(Self {
33            bdf,
34            live: Some(Self::fresh(bdf)?),
35            // Reserved up front so retiring a page never allocates mid-request.
36            retired: Vec::with_capacity(QUARANTINE_LIMIT),
37        })
38    }
39
40    fn fresh(bdf: u32) -> Result<AuthorizedDma<DmaBuf>, &'static str> {
41        let buf = DmaBuf::alloc(1).ok_or("allocation failed")?;
42        let buf = match AuthorizedDma::authorize(buf, |buf| buf.authorize(bdf)) {
43            Ok(buf) => buf,
44            Err(_) => return Err("authorization failed"),
45        };
46        // A full `OP_READ_RUN` must fit one PRP1 transfer, or every run would be
47        // refused while single-sector reads still worked.
48        if layout::blocks_in_one_prp(buf.iova()) < usize::from(MAX_RUN_SECTORS) {
49            // Dropped without `free` (DmaBuf has no Drop): a page authorized for the
50            // device is never handed back to the allocator.
51            return Err("iova would need a second PRP");
52        }
53        Ok(buf)
54    }
55
56    /// The page for the next request; `None` once I/O is refused.
57    pub fn page(&self) -> Option<&AuthorizedDma<DmaBuf>> {
58        self.live.as_ref()
59    }
60
61    /// The device may still access the current page: retire it and switch to a fresh
62    /// one, or refuse further I/O when the limit is reached or no page is available.
63    pub fn quarantine(&mut self) {
64        let Some(old) = self.live.take() else {
65            return;
66        };
67        self.retired.push(old);
68        if self.retired.len() >= QUARANTINE_LIMIT {
69            let _ = print_fmt(format_args!(
70                "[nvme] {} abandoned commands: refusing further I/O\n",
71                self.retired.len()
72            ));
73            return;
74        }
75        match Self::fresh(self.bdf) {
76            Ok(buf) => {
77                let _ = print_fmt(format_args!(
78                    "[nvme] io buffer quarantined after timeout; new iova=0x{:x}\n",
79                    buf.iova()
80                ));
81                self.live = Some(buf);
82            }
83            Err(why) => {
84                let _ = print_fmt(format_args!(
85                    "[nvme] replacement io buffer {}: refusing further I/O\n",
86                    why
87                ));
88            }
89        }
90    }
91}
