1//! Time budgets for the driver's waits (Phase 10 S10-3c, D3c-1 A).
2//!
3//! Every wait used to end after a fixed number of polls. A poll is one volatile read and the
4//! loop yields every [`POLL_YIELD_EVERY`] polls, but a yield returns at once when nothing else
5//! is runnable, so the same count lasted a few seconds in an idle guest and much longer during
6//! a busy boot: a Flush (a host `fsync`) could outlast it. The budgets are now milliseconds on
7//! the kernel's monotonic clock (`GetTime`), read only at the yield cadence. Without a clock the
8//! old poll count stays the bound.
9
10use core::fmt;
11
12/// Yield every this many polls so a long wait does not starve the rest of the system (the
13/// driver Cell shares the machine with the VFS and the guest). The clock is read here too.
14pub(crate) const POLL_YIELD_EVERY: u64 = 8192;
15/// The bound when no monotonic clock is available (the former budget).
16pub(crate) const FALLBACK_POLLS: u64 = 50_000_000;
17/// Admin commands (Identify, Create CQ/SQ): Linux's `nvme_core.admin_timeout` default.
18pub(crate) const ADMIN_BUDGET_MS: u64 = 60_000;
19/// I/O commands, including Flush: Linux's `nvme_core.io_timeout` default.
20pub(crate) const IO_BUDGET_MS: u64 = 30_000;
21/// Floor of the `CSTS.RDY` budget when the controller reports `CAP.TO = 0`.
22pub(crate) const RDY_FLOOR_MS: u64 = 1_000;
23/// A completion this slow is logged, so evidence shows how close a run came to a budget.
24pub(crate) const SLOW_COMPLETION_MS: u64 = 1_000;
25
26/// The `CSTS.RDY` transition budget: `CAP.TO` (bits 31:24) in 500 ms units, the
27/// specification's worst case, with a floor of [`RDY_FLOOR_MS`].
28pub(crate) fn rdy_budget_ms(cap: u64) -> u64 {
29    (((cap >> 24) & 0xFF) * 500).max(RDY_FLOOR_MS)
30}
31
32/// How much of a budget a wait used when it gave up.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub(crate) enum Spent {
35    Millis(u64),
36    /// No monotonic clock: the poll-count bound applied.
37    Polls(u64),
38}
39
40impl fmt::Display for Spent {
41    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
42        match self {
43            Spent::Millis(ms) => write!(f, "{} ms", ms),
44            Spent::Polls(polls) => write!(f, "{} polls (no clock)", polls),
45        }
46    }
47}
48
49/// One wait: its start on the monotonic clock (if any), its budget and its poll count.
50pub(crate) struct Deadline {
51    start_ms: Option<u64>,
52    budget_ms: u64,
53    polls: u64,
54}
55
56impl Deadline {
57    /// A wait of `budget_ms` starting at `now` (`None`: no clock).
58    pub(crate) fn start(budget_ms: u64, now: Option<u64>) -> Self {
59        Self { start_ms: now, budget_ms, polls: 0 }
60    }
61
62    /// Count one poll; `true` when this poll is a yield point, where the caller yields and
63    /// then calls [`check`](Self::check).
64    pub(crate) fn tick(&mut self) -> bool {
65        self.polls += 1;
66        self.polls % POLL_YIELD_EVERY == 0
67    }
68
69    /// `Some` once the budget is spent at `now`: elapsed milliseconds with a clock, the poll
70    /// count without one.
71    pub(crate) fn check(&self, now: Option<u64>) -> Option<Spent> {
72        match (self.start_ms, now) {
73            (Some(start), Some(now)) => {
74                let elapsed = now.saturating_sub(start);
75                (elapsed >= self.budget_ms).then_some(Spent::Millis(elapsed))
76            }
77            _ => (self.polls >= FALLBACK_POLLS).then_some(Spent::Polls(self.polls)),
78        }
79    }
80
81    /// Elapsed milliseconds at `now` when at least [`SLOW_COMPLETION_MS`]; `None` otherwise
82    /// or without a clock.
83    pub(crate) fn slow(&self, now: Option<u64>) -> Option<u64> {
84        let elapsed = now?.saturating_sub(self.start_ms?);
85        (elapsed >= SLOW_COMPLETION_MS).then_some(elapsed)
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    #[test]
94    fn a_wait_expires_at_its_budget_and_not_before() {
95        let deadline = Deadline::start(IO_BUDGET_MS, Some(1_000));
96        assert_eq!(deadline.check(Some(1_000)), None);
97        assert_eq!(deadline.check(Some(1_000 + IO_BUDGET_MS - 1)), None);
98        assert_eq!(deadline.check(Some(1_000 + IO_BUDGET_MS)), Some(Spent::Millis(IO_BUDGET_MS)));
99        // A clock that went backwards is no time at all, never an early expiry.
100        assert_eq!(deadline.check(Some(10)), None);
101    }
102
103    #[test]
104    fn many_fast_polls_do_not_expire_a_clocked_wait() {
105        let mut deadline = Deadline::start(ADMIN_BUDGET_MS, Some(0));
106        deadline.polls = FALLBACK_POLLS;
107        deadline.tick();
108        assert_eq!(deadline.check(Some(ADMIN_BUDGET_MS - 1)), None);
109    }
110
111    #[test]
112    fn without_a_clock_the_poll_count_is_the_bound() {
113        for (start, now) in [(None, None), (Some(5), None), (None, Some(5))] {
114            let mut deadline = Deadline::start(IO_BUDGET_MS, start);
115            deadline.polls = FALLBACK_POLLS - 2;
116            deadline.tick();
117            assert_eq!(deadline.check(now), None);
118            deadline.tick();
119            assert_eq!(deadline.check(now), Some(Spent::Polls(FALLBACK_POLLS)));
120        }
121    }
122
123    #[test]
124    fn a_yield_point_every_8192_polls() {
125        let mut deadline = Deadline::start(IO_BUDGET_MS, Some(0));
126        let yields = (0..3 * POLL_YIELD_EVERY).filter(|_| deadline.tick()).count();
127        assert_eq!(yields, 3);
128        assert_eq!(deadline.polls, 3 * POLL_YIELD_EVERY);
129    }
130
131    #[test]
132    fn rdy_budget_follows_cap_to_with_a_floor() {
133        assert_eq!(rdy_budget_ms(0), RDY_FLOOR_MS);
134        assert_eq!(rdy_budget_ms(1 << 24), RDY_FLOOR_MS);
135        assert_eq!(rdy_budget_ms(0xF << 24), 7_500);
136        assert_eq!(rdy_budget_ms((0xFF << 24) | 0xFFFF), 127_500);
137        // Bits outside CAP.TO do not count.
138        assert_eq!(rdy_budget_ms(0xFFFF_FFFF_00FF_FFFF), RDY_FLOOR_MS);
139    }
140
141    #[test]
142    fn slow_completions_are_reported_from_one_second() {
143        let deadline = Deadline::start(IO_BUDGET_MS, Some(100));
144        assert_eq!(deadline.slow(Some(100 + SLOW_COMPLETION_MS - 1)), None);
145        assert_eq!(deadline.slow(Some(100 + SLOW_COMPLETION_MS)), Some(SLOW_COMPLETION_MS));
146        assert_eq!(deadline.slow(None), None);
147        assert_eq!(Deadline::start(IO_BUDGET_MS, None).slow(Some(5_000)), None);
148        assert_eq!(alloc::format!("{}", Spent::Millis(30_000)), "30000 ms");
149        assert_eq!(alloc::format!("{}", Spent::Polls(7)), "7 polls (no clock)");
150    }
