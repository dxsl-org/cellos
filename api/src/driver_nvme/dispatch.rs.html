1//! IPC request dispatcher — translates `DrvRequest` messages into NVMe I/O.
2//!
3//! Wire format: `types::block_wire`, shared with the AHCI/virtio-blk cells.
4
5use crate::bounce::BounceBuffer;
6use crate::controller::{IoFailure, NvmeController};
7use types::block_wire::{self, BlockOutcome, ReadRunRequest, SECTOR_SIZE};
8
9pub const REPLY_SIZE: usize = block_wire::MAX_REPLY_SIZE;
10
11#[derive(Debug, PartialEq, Eq)]
12pub enum DrvOp {
13    Read,
14    Write,
15    Flush,
16}
17
18fn parse_op(data: &[u8]) -> Option<(DrvOp, u64)> {
19    if data.len() < 2 {
20        return None;
21    }
22    let op = match u16::from_le_bytes([data[0], data[1]]) {
23        block_wire::OP_READ => DrvOp::Read,
24        block_wire::OP_WRITE => DrvOp::Write,
25        block_wire::OP_FLUSH => return Some((DrvOp::Flush, 0)),
26        _ => return None,
27    };
28    if data.len() < 10 {
29        return None;
30    }
31    let sector = u64::from_le_bytes(data[2..10].try_into().ok()?);
32    Some((op, sector))
33}
34
35fn status(out: &mut [u8; REPLY_SIZE], outcome: BlockOutcome) -> usize {
36    out[0] = outcome.to_byte();
37    1
38}
39
40/// An issued command that failed: its outcome is unknown, and a timed-out one
41/// still owns its buffer page, so that page is quarantined before any reuse.
42fn failed(bounce: &mut BounceBuffer, failure: IoFailure, out: &mut [u8; REPLY_SIZE]) -> usize {
43    if failure == IoFailure::TimedOut {
44        bounce.quarantine();
45    }
46    status(out, BlockOutcome::Indeterminate)
47}
48
49/// `OP_READ_RUN`: one NVMe read of `count` blocks into the bounce page, copied
50/// into the reply after the outcome byte. `count <= MAX_RUN_SECTORS` (7) keeps
51/// the transfer inside the page, so PRP1 alone describes it.
52fn read_run(
53    ctrl: &mut NvmeController,
54    bounce: &mut BounceBuffer,
55    request: &ReadRunRequest,
56    out: &mut [u8; REPLY_SIZE],
57) -> usize {
58    let in_range = request
59        .sector
60        .checked_add(u64::from(request.count))
61        .is_some_and(|end| end <= ctrl.n_sectors);
62    if !in_range {
63        let _ = ostd::io::print_fmt(format_args!(
64            "[nvme] out-of-range run: sector={} count={} namespace_sectors={}\n",
65            request.sector, request.count, ctrl.n_sectors
66        ));
67        return status(out, BlockOutcome::NotIssued);
68    }
69    let Some((page, iova)) = bounce.page().map(|buf| (buf.inner().virt(), buf.iova())) else {
70        return status(out, BlockOutcome::NotIssued);
71    };
72    if usize::from(request.count) > crate::layout::blocks_in_one_prp(iova) {
73        return status(out, BlockOutcome::NotIssued);
74    }
75    if let Err(failure) = ctrl.read_sectors(request.sector, request.count, iova) {
76        return failed(bounce, failure, out);
77    }
78    let len = request.reply_len();
79    out[0] = BlockOutcome::Completed.to_byte();
80    // SAFETY: the read completed, so the bounce page holds `len - 1` stable
81    // bytes (`count` blocks, checked to fit the page above).
82    unsafe {
83        core::ptr::copy_nonoverlapping(page as *const u8, out[1..len].as_mut_ptr(), len - 1);
84    }
85    len
86}

88/// Handle one incoming IPC message. Writes the reply into `out` and returns
89/// the number of bytes to send back: the outcome byte plus the sectors of a
90/// completed read, else 1.
91pub fn handle(
92    ctrl: &mut NvmeController,
93    bounce: &mut BounceBuffer,
94    data: &[u8],
95    out: &mut [u8; REPLY_SIZE],
96) -> usize {
97    if let Some(request) = ReadRunRequest::decode(data) {
98        return read_run(ctrl, bounce, &request, out);
99    }
100    let Some((op, sector)) = parse_op(data) else {
101        return status(out, BlockOutcome::NotIssued);
102    };
103
104    // A sector past the end of the namespace cannot be served, and the device
105    // only answers it with an opaque status - which is how the Tier-3 lane's
106    // persistence failures surface. Name the caller's sector and the namespace
107    // size instead, so a failing boot says whether the request or the size was
108    // wrong.
109    if matches!(op, DrvOp::Read | DrvOp::Write) && sector >= ctrl.n_sectors {
110        let _ = ostd::io::print_fmt(format_args!(
111            "[nvme] out-of-range request: op={:?} sector={} namespace_sectors={}\n",
112            op, sector, ctrl.n_sectors
113        ));
114    }
115
116    if op == DrvOp::Flush {
117        return match ctrl.flush() {
118            Ok(()) => status(out, BlockOutcome::Completed),
119            Err(failure) => failed(bounce, failure, out),
120        };
121    }
122    // `None` after too many abandoned commands, or when no clean page could
123    // replace the last one.
124    let Some((page, iova)) = bounce.page().map(|buf| (buf.inner().virt(), buf.iova())) else {
125        return status(out, BlockOutcome::NotIssued);
126    };
127
128    if op == DrvOp::Read {
129        return match ctrl.read_sector(sector, iova) {
130            Ok(()) => {
131                out[0] = BlockOutcome::Completed.to_byte();
132                // SAFETY: read_sector completed; DMA data is now stable in
133                // the CPU mapping of the live bounce page.
134                unsafe {
135                    core::ptr::copy_nonoverlapping(page, out[1..].as_mut_ptr(), SECTOR_SIZE);
136                }
137                block_wire::REPLY_SIZE
138            }
139            Err(failure) => failed(bounce, failure, out),
140        };
141    }
142
143    if data.len() < 10 + SECTOR_SIZE {
144        return status(out, BlockOutcome::NotIssued);
145    }
146    // SAFETY: write the caller payload into the CPU mapping before
147    // submitting the authorized device-visible IOVA.
148    unsafe {
149        core::ptr::copy_nonoverlapping(data[10..].as_ptr(), page, SECTOR_SIZE);
150    }
151    // Fault-injection witnesses only. `F08B-DIE` kills this cell before
152    // the write is issued, with its controller still enabled;
153    // `F09C-DIE-ISSUED` kills it right after the write's doorbell.
154    #[cfg(feature = "fault-inject")]
155    {
156        let payload = &data[10..10 + SECTOR_SIZE];
157        if payload.windows(8).any(|window| window == b"F08B-DIE") {
158            let _ = ostd::io::print_fmt(format_args!(
159                "[nvme] fault-inject: driver exiting at lba={}\n",
160                sector
161            ));
162            ostd::syscall::sys_exit(70);
163        }
164        if payload
165            .windows(15)
166            .any(|window| window == b"F09C-DIE-ISSUED")
167        {
168            ctrl.exit_after_next_doorbell();
169        }
170    }
171    match ctrl.write_sector(sector, iova) {
172        Ok(()) => status(out, BlockOutcome::Completed),
173        Err(failure) => failed(bounce, failure, out),
174    }
175}
