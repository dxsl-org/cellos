6/// Immutable SoC-profile facts used by the RV64 platform path.
7///
8/// Compatible lists remain static slices so early boot can borrow them without
9/// allocation. Policies must fail closed when hardware access is unsupported.
10#[derive(Clone, Copy, Debug, Eq, PartialEq)]
11pub struct RiscvSocProfile {
12    pub slug: &'static str,
13    pub uart_compatibles: &'static [&'static str],
14    pub plic_compatibles: &'static [&'static str],
15    pub clint_compatibles: &'static [&'static str],
16    pub rtc_compatibles: &'static [&'static str],
17    pub plic_context: PlicContextPolicy,
18    pub uart_access: UartAccessPolicy,
19    pub rtc_access: RtcAccessPolicy,
20    pub virtio_mmio: VirtioMmioPolicy,
21    pub fallback_mmio: RiscvFallbackMmio,
22    pub sdhci: Option<RiscvSdhciProfile>,
23}
24
25impl RiscvSocProfile {
26    /// Returns true when the profile allows UART MMIO probing.
27    pub const fn allows_uart_mmio(self) -> bool {
28        matches!(self.uart_access, UartAccessPolicy::Mmio)
29    }
30
31    /// Returns true when the profile allows RTC MMIO probing.
32    pub const fn allows_rtc_mmio(self) -> bool {
33        matches!(self.rtc_access, RtcAccessPolicy::Mmio)
34    }
35
36    /// Returns true when the profile should scan DTB VirtIO MMIO nodes.
37    pub const fn discovers_virtio_mmio(self) -> bool {
38        matches!(self.virtio_mmio, VirtioMmioPolicy::Discover)
39    }
40
41    /// Returns the S-mode PLIC context for a physical hart, or `None` when absent.
42    pub const fn plic_context_for_physical_hart(self, physical_hart: usize) -> Option<usize> {
43        self.plic_context
44            .s_mode_context_for_physical_hart(physical_hart)
45    }
46}
